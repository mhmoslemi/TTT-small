        for k in range(min(n_pairs_per_pass, n // 2)):
            if time.time() > deadline:
                break
            i = int(order[k])
            j = int(order[n - 1 - k])
            if i == j:
                continue
            max_d = min(1.0 - h[i], h[j])
            if max_d < 1e-9:
                continue
            for frac in (0.5, 0.3, 0.15, 0.075, 0.04, 0.02):
                delta = frac * max_d
                h[i] += delta
                h[j] -= delta
                f = _eval(h)
                if f < bf - 1e-14:
                    bf = f
                    bh = h.copy()
                    improved += 1
                    break
                else:
                    h[i] -= delta
                    h[j] += delta
        if improved == 0:
            for _ in range(80):
                if time.time() > deadline:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = rng.choice([-1, 1])
                if d > 0:
                    max_d = min(1.0 - h[i], h[j])
                else:
                    max_d = min(h[i], 1.0 - h[j])
                if max_d < 1e-9:
                    continue
                for frac in (0.4, 0.2, 0.1):
                    delta = frac * max_d
                    if d > 0:
                        h[i] += delta
                        h[j] -= delta
                    else:
                        h[i] -= delta
                        h[j] += delta
                    f = _eval(h)
                    if f < bf - 1e-14:
                        bf = f
                        bh = h.copy()
                        improved += 1
                        break
                    else:
                        if d > 0:
                            h[i] -= delta
                            h[j] += delta
                        else:
                            h[i] += delta
                            h[j] -= delta
        if improved == 0:
            stalled += 1
            if stalled > max_stall:
                break
        else:
            stalled = 0
    return bh, bf

def _try_candidate(h_start, deadline, rng, label=""):
    n = len(h_start)
    target = n * 0.5
    h = _proj_sum(h_start, target)
    c5 = _eval(h)
    # Quick Nesterov
    dl1 = min(deadline, time.time() + max(5, (deadline - time.time()) * 0.25))
    h, c5 = _nesterov(h, 800, 0.003, dl1, alpha=30000, topk=5, beta=0.88)
    # Coordinate descent
    dl2 = min(deadline, time.time() + max(5, (deadline - time.time()) * 0.6))
    h, c5 = _coord_descent(h, dl2, rng, n_pairs_per_pass=40, max_stall=15)
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    n0 = len(initial_h_values)
    n = n0
    h0 = _proj_sum(np.array(initial_h_values, dtype=float), n * 0.5)

    # Phase 1: Nesterov from initial (80s)
    dl1 = min(deadline, t0 + 80)
    h1, c5_1 = _nesterov(h0, 5000, 0.004, dl1, alpha=30000, topk=5, beta=0.87)
    best_h, best_c5 = h1.copy(), c5_1

    # Phase 2: Top-1 Nesterov from phase 1 result (50s)
    dl2 = min(deadline, t0 + 130)
    h2, c5_2 = _nesterov(best_h.copy(), 3000, 0.002, dl2, alpha=100000, topk=3, beta=0.9, use_top1=True)
    if c5_2 < best_c5:
        best_h, best_c5 = h2.copy(), c5_2

    # Phase 3: Main coordinate descent (400s)
    dl3 = min(deadline, t0 + 530)
    h3, c5_3 = _coord_descent(best_h.copy(), dl3, rng, n_pairs_per_pass=50, max_stall=30)
    if c5_3 < best_c5:
        best_h, best_c5 = h3.copy(), c5_3

    # Phase 4: Jitter restarts (250s)
    dl4 = min(deadline, t0 + 780)
    n_restarts = 5
    rem4 = max(0, dl4 - time.time())
    for r in range(n_restarts):
        if time.time() > dl4:
            break
        alloc = rem4 / (n_restarts - r)
        jit_scale = 0.003 * (1.0 + 0.5 * r)
        d = jit_scale * rng.randn(n)
        d -= d.mean()
        h_jit = _proj_sum(best_h + d, n * 0.5)
        dl_r = time.time() + alloc * 0.85
        h_r, c5_r = _try_candidate(h_jit, dl_r, rng)
        if c5_r < best_c5:
            best_h, best_c5 = h_r.copy(), c5_r

    # Phase 5: Final fine coordinate descent (remaining)
    dl5 = deadline
    h5, c5_5 = _coord_descent(best_h.copy(), dl5, rng, n_pairs_per_pass=30, max_stall=50)
    if c5_5 < best_c5:
        best_h, best_c5 = h5.copy(), c5_5

    best_h = _proj_sum(best_h, n * 0.5)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
