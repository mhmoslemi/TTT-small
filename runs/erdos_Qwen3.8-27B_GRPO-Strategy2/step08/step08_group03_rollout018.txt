    x += (S - x.sum()) / n

        d = S - x.sum()
        if abs(d) < 1e-11:

            hp = h
            y = h
            t = 1.0
            fy = fh
            Lc = max(Lc * 0.5, 1.0)
            stale += 1
    return h

def _greedy_swap(h0, rng, tlim):
    n = len(h0)
    S = int(n // 2)
    h = (h0 >= 0.5).astype(float)
    cnt = int(h.sum())
    if cnt != S:
        if cnt > S:
            idx1 = np.where(h == 1.0)[0]
            for i in np.argsort(h0[idx1])[::-1][:cnt - S]:
                h[idx1[i]] = 0.0
        else:
            idx0 = np.where(h == 0.0)[0]
            for i in np.argsort(-h0[idx0])[:S - cnt]:
                h[idx0[i]] = 1.0
    h = _repair(h, n)
    c = _c5(h)
    best_h = h.copy()
    best_c = c
    t0 = time.time()
    it = 0
    while time.time() - t0 < tlim:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        cn = _c5(h)
        if cn <= c + 1e-15:
            c = min(cn, c)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        it += 1
    return best_h, best_c

def _coord_h(h0, tlim, rng):
    n = len(h0)
    S = n * 0.5
    h = h0.copy()
    c = _c5(h)
    best_h = h.copy()
    best_c = c
    t0 = time.time()
    iters = 0
    while time.time() - t0 < tlim:
        frac = min(1.0, (time.time() - t0) / max(tlim, 1e-9))
        step = 0.08 * (0.005) ** frac
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for sgn in (1.0, -1.0):
            if sgn > 0:
                d = min(step, 1.0 - h[i], h[j])
            else:
                d = min(step, h[i], 1.0 - h[j])
            if d < 1e-12:
                continue
            h[i] += sgn * d
            h[j] -= sgn * d
            np.clip(h, 0.0, 1.0, out=h)
            cn = _c5(h)
            if cn < c - 1e-14:
                c = cn
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = h.copy()
            else:
                h[i] -= sgn * d
                h[j] += sgn * d
        iters += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)
    best = h0.copy()
    best_c = _c5(best)

    for fn in (lambda x: 1.0 - x, lambda x: x[::-1]):
        hv = _repair(fn(h0.copy()), n0)
        c = _c5(hv)
        if c < best_c:
            best, best_c = hv, c

    # Phase 1: FISTA ascending alphas
    t1 = t0 + min(150.0, 0.14 * budget_s)
    if time.time() < t1 - 1.0:
        ht = best.copy()
        for a in [30, 100, 400, 1500, 6000]:
            if time.time() >= t1:
                break
            ht = _fista(ht, a, t1 - time.time() - 1.0, L0=40.0)
            c = _c5(ht)
            if c < best_c:
                best, best_c = ht.copy(), c

    # Phase 2: Discrete greedy search
    t2 = t1 + min(180.0, 0.17 * budget_s)
    if time.time() < t2 - 1.0:
        dh, dc = _greedy_swap(best, rng, t2 - time.time() - 1.0)
        if dc < best_c:
            best, best_c = dh, dc

    # Phase 3: High-alpha FISTA polishing
    t3 = t2 + min(220.0, 0.22 * budget_s)
    if time.time() < t3 - 1.0:
        ht = best.copy()
        for a in [8000, 40000, 200000]:
            if time.time() >= t3:
                break
            ht = _fista(ht, a, t3 - time.time() - 1.0, L0=60.0)
            c = _c5(ht)
            if c < best_c:
                best, best_c = ht.copy(), c

    # Phase 4: Coordinate descent
    t4 = t3 + min(250.0, 0.25 * budget_s)
    if time.time() < t4 - 1.0:
        bh, bc = _coord_h(best, t4 - time.time() - 1.0, rng)
        if bc < best_c:
            best, best_c = bh, bc

    # Phase 5: Final discrete polish
    t5 = t3 + min(200.0, 0.20 * budget_s)
    if time.time() < t5 and time.time() < t0 + budget_s - 5.0:
        rem = min(t5 - time.time(), t0 + budget_s - 5.0 - time.time())
        if rem > 5.0:
            dh, dc = _greedy_swap(best, rng, rem)
            if dc < best_c:
                best, best_c = dh, dc

    # Symmetry variants
    for fn in (lambda x: 1.0 - x, lambda x: x[::-1],
               lambda x: 0.5 + 0.5 * (x - x[::-1]),
               lambda x: 0.5 + 0.5 * (x + x[::-1] - 1.0)):
        hv = _proj(fn(best.copy()), len(best) * 0.5)
        c = _c5(hv)
        if c < best_c:
            best, best_c = hv, c

    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
