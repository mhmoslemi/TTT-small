import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-13:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-13:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cvec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_k(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _smax_eval_grad(h, alpha, topk):
    n = len(h)
    c = _cvec(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    val = m + (s.max() + np.log(np.sum(np.exp(s)))) / alpha
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _grad_k(h, idx[ii], dx)
    return val, g

def _lfgs_b(h, alpha, topk, maxit, deadline):
    n = len(h)
    def obj(x):
        return _smax_eval_grad(x, alpha, topk)[0]
    def jac(x):
        return _smax_eval_grad(x, alpha, topk)[1]
    def pen(x):
        return obj(x) + 0.01 * (x.sum() - n*0.5)**2
    def jpen(x):
        v, g = _smax_eval_grad(x, alpha, topk)
        return v + 0.01*(x.sum()-n*0.5)**2, g + 0.02*(x.sum()-n*0.5)*np.ones(n)
    try:
        r = minimize(jpen, h, jac=None, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)]*n,
                     options={'maxiter': maxit, 'gtol': 1e-14, 'ftol': 1e-18})
        return _repair(r.x, n)
    except:
        return _repair(h, n)

def _nesterov(h, alpha0, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_eval_grad(h, alpha0, topk)[0]
    v = np.zeros(n)
    beta = 0.80
    step = step0
    alpha = alpha0
    for it in range(steps):
        if time.time() > deadline:
            break
        _, g = _smax_eval_grad(h, alpha, topk)
        hp = h.copy()
        v = beta*v + (1.0-beta)*g
        hn = _proj(h - step*v, n*0.5)
        fn = _smax_eval_grad(hn, alpha, topk)[0]
        if fn < f:
            h = hn; f = fn
            step = min(step*1.03, step0*20)
        else:
            step *= 0.4
            hn = _proj(hp - step*v, n*0.5)
            fn = _smax_eval_grad(hn, alpha, topk)[0]
            if fn < f:
                h = hn; f = fn
            else:
                step *= 0.4
                hn = _proj(hp - step*v, n*0.5)
                fn = _smax_eval_grad(hn, alpha, topk)[0]
                if fn < f:
                    h = hn; f = fn
                else:
                    break
    return _repair(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-13:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-13:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _cvec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_k(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _smax_vg(h, alpha, topk):
    n = len(h)
    c = _cvec(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    val = m + (s.max() + np.log(np.sum(np.exp(s)))) / alpha
    w = np.exp(s); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _grad_k(h, idx[ii], dx)
    return val, g

def _lfgs_opt(h, alpha, topk, maxit, dl):
    n = len(h)
    pen_c = 50.0
    def obj(x):
        v, g = _smax_vg(x, alpha, topk)
        return v + pen_c * (x.sum() - n * 0.5) ** 2
    def jac(x):
        v, g = _smax_vg(x, alpha, topk)
        return g + 2 * pen_c * (x.sum() - n * 0.5) * np.ones(n)
    try:
        r = minimize(obj, h, jac=jac, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxit, 'gtol': 1e-12, 'ftol': 1e-18, 'maxls': 30})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _nest(h, alpha, topk, steps, st0, dl):
    n = len(h)
    f = _smax_vg(h, alpha, topk)[0]
    v = np.zeros(n)
    beta = 0.82
    step = st0
    for it in range(steps):
        if time.time() > dl:
            break
        _, g = _smax_vg(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_vg(hn, alpha, topk)[0]
        if fn < f:
            h = hn; f = fn
            step = min(step * 1.04, st0 * 30)
        else:
            step *= 0.4
            for _ in range(3):
                hn = _proj(hp - step * v, n * 0.5)
                fn = _smax_vg(hn, alpha, topk)[0]
                if fn < f:
                    h = hn; f = fn; break
                step *= 0.4
            else:
                break
    return _repair(h, n)

def _symm_proj(h, k):
    n = len(h)
    if k < n:
        for p in range(max(0, n - 2 * k)):
            a, b = h[p], h[p + 2 * k]
            t = (a - b) / 2.0
            t = max(-0.5, min(0.5, t))
            h[p] = 0.5 + t
            h[p + 2 * k] = 0.5 - t
    return h

def _cycle_worst(h, n_iter, dl):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _eval(h)
    for it in range(n_iter):
        if time.time() > dl:
            break
        c = _cvec(h)
        top3 = np.argpartition(c, -3)[-3:]
        for kk in sorted(top3, key=lambda x: -c[x]):
            g = _grad_k(h, kk, dx)
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-15:
                continue
            gn = g / gnorm
            for step in [0.003, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001, 5e-5]:
                hn = _proj(h - step * gn, n * 0.5)
                cn = _eval(hn)
                if cn < best_c5 - 1e-16:
                    h = hn
                    best_c5 = cn
                    best_h = h.copy()
                    break
                elif cn < _eval(h) - 1e-16:
                    h = hn
                    break
    return best_h, best_c5

def _pair_search(h, dl, rng, max_rounds=5000):
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(h)
    window = 30
    step_sizes = [5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]
    accepted = 0
    for rr in range(max_rounds):
        if time.time() > dl:
            break
        if accepted > 0 and rr % 500 == 0:
            window = max(5, window // 2)
        i = rng.randint(n)
        j = (i + 1 + rng.randint(window)) % n
        if i == j:
            continue
        d = step_sizes[rng.randint(len(step_sizes))] * rng.choice([1, -1])
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < best_c5 - 1e-17:
                best_c5 = cn
                best_h = h.copy()
                accepted += 1
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def _block_anneal(h, niter, rng, dl, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy(); bc = c5c
    for it in range(niter):
        if time.time() > dl:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.003 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy(); hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn; c5c = c5n
            if c5n < bc:
                bc = c5n; bh = h.copy()
    return bh, bc

def _multires_refine(h, dl, rng):
    n = len(h)
    n2 = n // 2
    hc = np.zeros(n2)
    for j in range(n2):
        hc[j] = (h[2 * j] + h[2 * j + 1]) / 2.0
    hc = _repair(hc, n2)
    hc = _lfgs_opt(hc, 3000, 8, 300, dl)
    hf = np.zeros(n)
    for j in range(n2):
        hf[2 * j] = hc[j]
        hf[2 * j + 1] = hc[j] + rng.uniform(-0.02, 0.02)
    hf = np.clip(hf, 0, 1)
    hf = _repair(hf, n)
    hf = _lfgs_opt(hf, 5000, 6, 200, dl)
    return _repair(hf, n)
