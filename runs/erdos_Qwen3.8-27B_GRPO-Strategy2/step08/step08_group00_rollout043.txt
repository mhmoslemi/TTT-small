    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= x[:n - M] * wk
    return f, g

def _obj_only(x, alpha, m_lags):
    return _obj_grad(x, alpha, m_lags)[0]

def _jac_only(x, alpha, m_lags):
    return _obj_grad(x, alpha, m_lags)[1]

def _slsqp_run(h, alpha, m_lags, maxiter, ftol, bnds, half_n):
    n = len(h)
    try:
        res = minimize(_obj_only, h, jac=_jac_only,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _sum_fn,
                                     'jac': _sum_jac, 'args': (half_n,)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj_grad(h, alpha, m_lags, nsteps, s0, mu, t_limit):
    n = len(h)
    v = np.zeros(n)
    step = s0
    cur_f = _eval(h)
    for _ in range(nsteps):
        if time.time() > t_limit:
            break
        _, g = _obj_grad(h, alpha, m_lags)
        v = mu * v + (1.0 - mu) * g
        taken = False
        for _ in range(12):
            if time.time() > t_limit:
                break
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_f:
                h = h_new
                cur_f = c_new
                step = min(step * 1.06, s0 * 12.0)
                taken = True
                break
            step *= 0.45
        if not taken:
            step = min(step * 1.3, s0 * 12.0)
            if step < s0 * 0.01:
                step = s0
    return _repair(h, n), cur_f

def _local_search(h, niter, rng, base_step, npert, t_limit):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    for _ in range(niter):
        if time.time() > t_limit:
            break
        step = base_step * (0.3 + 0.7 * rng.rand())
        idx = rng.randint(0, n, size=npert)
        d = step * rng.randn(npert)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < 0 or h_new.max() > 1:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5cur:
            c5cur = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def _twopoint_polish(h, niter, rng, start_step, t_limit):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    step = start_step
    for _ in range(niter):
        if time.time() > t_limit:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 9)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                step *= 1.02
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                step *= 0.85
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=float).copy()
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Multi-start perturbations
    if _el() < 40:
        for trial in range(6):
            if _el() > 35:
                break
            scale = 0.03 / (trial + 1)
            noise = scale * rng.randn(n)
            h_try = _repair(h0 + noise, n)
            c_try = _eval(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try.copy()

    # Phase 2: Projected gradient with increasing alpha
    pg_configs = [
        (3000, 25, 500, 0.008, 0.70),
        (15000, 35, 450, 0.005, 0.75),
        (80000, 50, 400, 0.003, 0.80),
        (400000, 70, 350, 0.002, 0.85),
        (2000000, 100, 300, 0.001, 0.85),
        (10000000, 130, 250, 0.0005, 0.90),
        (50000000, 160, 200, 0.0003, 0.90),
    ]
    for alpha, ml, steps, s0, mu in pg_configs:
        if _el() > budget_s - 300:
            break
        h_new, c_new = _proj_grad(best_h.copy(), alpha, ml, steps, s0, mu,
                                  t0 + budget_s - 280)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 3: SLSQP refinement
    if _el() < budget_s - 180:
        bnds = [(0.0, 1.0)] * n
        half_n = n * 0.5
        slsqp_configs = [(50000, 40, 400), (500000, 70, 350), (5000000, 100, 300)]
        for alpha, ml, mit in slsqp_configs:
            if _el() > budget_s - 60:
                break
            h_new = _slsqp_run(best_h, alpha, ml, mit, 1e-16, bnds, half_n)
            c_new = _eval(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new

    # Phase 4: Another PG pass with highest alpha
    if _el() < budget_s - 100:
        h_new, c_new = _proj_grad(best_h.copy(), 200000000, 200, 200, 0.0002, 0.92,
                                  t0 + budget_s - 80)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 5: Local search
    if _el() < budget_s - 15:
        for bs, npert, budget_frac in [(8e-5, 3, 0.35), (2e-5, 2, 0.30), (5e-6, 2, 0.25)]:
            if _el() > budget_s - 12:
                break
            remaining = (budget_s - 12) - _el()
            niter = min(int(remaining * budget_frac / 0.0025), 200000)
            if niter < 200:
                break
            best_h, best_c5 = _local_search(best_h, niter, rng, bs, npert,
                                            t0 + budget_s - 12)

    # Phase 6: Ultra-fine 2-point polish
    if _el() < budget_s - 2:
        best_h, best_c5 = _twopoint_polish(best_h, 500000, rng, 4e-7,
                                           t0 + budget_s - 2)

    # Final: one more tiny polish
    if _el() < budget_s - 1:
        best_h, best_c5 = _twopoint_polish(best_h, 200000, rng, 2e-7,
                                           t0 + budget_s - 1)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
