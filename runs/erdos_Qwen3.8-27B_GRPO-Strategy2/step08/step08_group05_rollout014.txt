import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _sg(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    mask1 = (i1 >= 0) & (i1 < n)
    g[mask1] += (1.0 - h[i1[mask1]])
    i2 = j + off
    mask2 = (i2 >= 0) & (i2 < n)
    g[mask2] -= h[i2[mask2]]
    return g

def _lp_step(c, n, m_top, eps, rng):
    cor = _correl(c)
    idx_top = np.argpartition(cor, -m_top)[-m_top:]
    idx_top = idx_top[np.argsort(-cor[idx_top])]
    grads = np.array([_sg(c, k, n) for k in idx_top])
    c_obj = -grads[0]
    if m_top > 1:
        A_ub = grads[1:] - grads[0]
        b_ub = np.zeros(m_top - 1)
    else:
        A_ub, b_ub = None, None
    bnds = [(-eps, eps)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=np.ones((1, n)),
                      b_eq=np.zeros(1), bounds=bnds,
                      method='highs', options={'presolve': True})
    except Exception:
        return None
    if not res.success or res.x is None:
        return None
    return res.x

def _lp_descent(h, deadline, eps_start, m_top, rng):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    eps = eps_start
    stall = 0
    while time.time() < deadline:
        d = _lp_step(c, n, m_top, eps, rng)
        if d is None or np.max(np.abs(d)) < 1e-15:
            stall += 1
            if stall > 10:
                eps *= 0.4
                if eps < 5e-6:
                    break
                stall = 0
                continue
            d = rng.randn(n); d -= d.mean(); d *= eps * 0.05
            h_new = _repair(c + d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur:
                c, c5_cur = h_new, c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
            continue
        step = 1.0
        improved = False
        for _ in range(18):
            h_new = _repair(c + step * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-18:
                c, c5_cur = h_new, c5_new
                improved = True
                stall = 0
                if c5_cur < best_c5:

                break
            step *= 0.5
        if not improved:
            stall += 1
            if stall > 15:
                eps *= 0.6
                stall = 0
                if eps < 5e-6:
                    break
    return best_h, best_c5

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h); m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _sg(h, idx[ii], n)
    return g

def _nesterov(h, alpha, topk, deadline, step0):

    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta, step = 0.85, step0

            break
        g = _grad_smax(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h, f = hn, fn
            step = min(step * 1.02, step0 * 5.0)
        else:
            step *= 0.5
            if step < 1e-13:
                break
            h = _proj(hp - step * v, n * 0.5)
            f = _smax_obj(h, alpha, topk)
    return _repair(h, n)

def _coord_polish(h, deadline, rng):
