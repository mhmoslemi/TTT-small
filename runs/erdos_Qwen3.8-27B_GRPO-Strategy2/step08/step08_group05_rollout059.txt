def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_box_eq(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _smax(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    s = alpha * (ct - m)
    s -= s.max()
    return m + (s.max() + np.log(np.sum(np.exp(s)))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:].copy()
    ct = c[idx]
    s = alpha * (ct - m)
    s -= s.max()
    w = np.exp(s)

    g = np.zeros(n)

            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])

    return g

def _lp_step(h, eps, m_top, deadline):
    n = len(h)
    c = _correl(h)
    idx_top = np.argpartition(c, -m_top)[-m_top:]
    idx_top = idx_top[np.argsort(-c[idx_top])]
    # Linearize: for each top k, gradient direction
    c_obj = np.zeros(n)
    A_ub_list = []
    b_ub_list = []
    dx = 2.0 / n
    for ii, k in enumerate(idx_top):
        sh = k - (n - 1)
        g = np.zeros(n)
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:n - sp] -= h[sp:]
        g *= dx
        if ii == 0:
            c_obj = -g  # minimize top correlation
        else:
            A_ub_list.append(g - (1.0 / eps) * np.ones(n) * 0)  # g.delta <= 0
            # Actually constraint: c_k + g_k . d <= c_k, so g_k . d <= 0
            A_ub_list.append(g)
            b_ub_list.append(0.0)
    A_ub = np.array(A_ub_list) if A_ub_list else None
    b_ub = np.array(b_ub_list) if b_ub_list else None
    A_eq = np.ones((1, n))
    b_eq = np.zeros(1)
    bnds = [(-eps, eps)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bnds, method='highs', options={'presolve': True,
                      'time_limit': 5})
        if res.success:
            return np.clip(h + res.x, 0, 1)
    except Exception:
        pass
    return h

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):

        if ii == 0:
            c_obj = -g
        else:
            A_rows.append(g)
    A_ub = np.array(A_rows) if A_rows else None
    b_ub = np.zeros(len(A_rows)) if A_rows else None
    A_eq = np.ones((1, n))
    b_eq = np.zeros(1)
    bnds = [(-eps, eps)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bnds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success and res.x is not None:
            return c + res.x
    except Exception:
        pass
    return c

def _lp_descent(h, n_steps, eps, m_top, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    rng = np.random.RandomState(777)
    stall = 0
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        hn = _lp_step(c, eps, m_top)
        if np.allclose(hn, c, atol=1e-15):
            stall += 1
            if stall > 200:
                d = rng.randn(n); d -= d.mean(); d *= eps * 0.4
                hn = _repair(c + d, n)
                c5n = _eval(hn)
                if c5n < c5_cur:
                    c, c5_cur = hn, c5n; stall = 0
                    if c5_cur < best_c5:
                        best_h, best_c5 = c.copy(), c5_cur
            continue
        stagn2 = 0
        while True:
            hn = _repair(hn, n)
            c5n = _eval(hn)
            if c5n < c5_cur - 1e-15:
                c, c5_cur = hn, c5n; stall = 0
                if c5_cur < best_c5:
                    best_h, best_c5 = c.copy(), c5_cur
                break
            stagn2 += 1
            if stagn2 >= 3:
                break
            hn = _lp_step(c, eps * 0.5, m_top)
    return best_h, best_c5

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.85
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj_box_eq(h - step * v, n * 0.5)
        fn = _smax(hn, alpha, topk)
        if fn < f:
            h, f = hn, fn
            step = min(step * 1.03, step0 * 12.0)
        else:
            step *= 0.5
            if step < 1e-12:
                break
            h = _proj_box_eq(h - step * v, n * 0.5)
            f = _smax(h, alpha, topk)
    return _repair(h, n)

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy(); bc = c5c
    nb = len(blocks)
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = int(blocks[rng.randint(nb)])
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.006 * max(T, 1e-10) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _coord_refine(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy(); bc = c5c
    steps = [5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]
    per = max(1, niter // len(steps))
    for ss in steps:
        for _ in range(per):
            if time.time() > deadline:
                return bh, bc
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 6)) % n
            d = ss * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < bc:
                        bc, bh = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    dl1 = min(deadline, t0 + 200)
    for alpha, topk, mi in [(1500, 50, 400), (8000, 25, 350), (30000, 12, 300)]:
        if time.time() > dl1:
            break
        hv = _slsqp_run(best_h.copy(), alpha, topk, mi, dl1)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    dl2 = min(deadline, t0 + 420)
    hv, cv = _lp_descent(best_h.copy(), 20000, 0.005, 8, dl2)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    dl3 = min(deadline, t0 + 600)
    for alpha, topk, step0, nit in [(15000, 15, 0.003, 3000), (80000, 8, 0.001, 3000)]:
        if time.time() > dl3:
            break
        hv = _nesterov(best_h.copy(), alpha, topk, nit, step0, dl3)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    dl4 = min(deadline, t0 + 760)
    hv, cv = _lp_descent(best_h.copy(), 20000, 0.002, 5, dl4)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    dl5 = min(deadline, t0 + 900)
    blocks = np.array([1, 2, 3, 4, 5, 6, 8, 10, 14])
    hv, cv = _block_anneal(best_h.copy(), 25000, rng, dl5, 0.005, 1e-10, blocks)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()
        if time.time() < dl5:
            hv2, cv2 = _lp_descent(best_h.copy(), 5000, 0.001, 4, dl5)
            if cv2 < best_c5:
                best_c5, best_h = cv2, hv2.copy()

    if time.time() < deadline - 25:
        dl6 = min(deadline - 5, t0 + 975)
        hv = _slsqp_run(best_h.copy(), 60000, 8, 250, dl6)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    rem = max(1.0, deadline - time.time())
    npol = min(int(rem / 0.0005), 100000)
    best_h, best_c5 = _coord_refine(best_h, npol, rng, deadline)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
