    mx = max(float(P[0]), float(np.max(P[1:])), float(np.max(Nv[1:]))) if n > 1 else float(P[0])
    wP, wN = np.zeros(n), np.zeros(n)
    cnt = 0
    if P[0] >= mx - tol:
        wP[0] = 1; cnt += 1
    if n > 1:
        mP = P[1:] >= mx - tol
        if np.any(mP):
            wP[np.nonzero(mP)[0] + 1] = 1; cnt += mP.sum()
        mN = Nv[1:] >= mx - tol
        if np.any(mN):
            wN[np.nonzero(mN)[0] + 1] = 1; cnt += mN.sum()
    if cnt == 0:
        return np.zeros(n), mx
    wP /= cnt
    wN /= cnt
    return _grad_from_weights(h, wP, wN), mx

import numpy as np
import time

def _fft_size(n):
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-14:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(12):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-12:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    if abs(x.sum() - S) > 1e-8:
        lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
        for _ in range(50):
            mid = 0.5 * (lo + hi)
            if np.clip(v - mid, 0.0, 1.0).sum() > S:
                lo = mid
            else:
                hi = mid
        x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    return x

def _pn_arrays(h):
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n)
    P[0] = total - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
        mx = float(max(np.max(P), np.max(Nv)))
    else:
        mx = float(P[0])
    return P, Nv, mx

def _c5_direct(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _smooth_value(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    dx = 2.0 / n
    e = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    if n > 1:
        e += np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(e) / alpha)

def _grad_from_weights(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        Urev = np.fft.rfft(u[::-1], Nf)
        Ur = np.fft.rfft(u, Nf)
        q = -(np.fft.irfft(H * Urev, Nf)[n-1:2*n-1].real
              + np.fft.irfft(H * Ur, Nf)[:n].real)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _smooth_grad(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    wP, wN = np.zeros(n), np.zeros(n)
    e0 = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0:
            tot, e0 = 1.0, 1.0
        wP[0], wP[1:] = e0 / tot, ep / tot
        wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _grad_from_weights(h, wP, wN)

def _fista(h0, alpha, max_time, L=8.0):

        return h, h.copy(), _c5_direct(h)

            t = tn
            if y.min() < -1e-5 or y.max() > 1.0 + 1e-5 or abs(y.sum() - S) > 1e-8:
                y = _proj(y, S)
            fy = _smooth_value(y, alpha)
        else:
            y, fy = h, fh
        g = _smooth_grad(y, alpha)
        step = 1.0 / max(Lcur, 1e-12)
        accepted = False
        for _ in range(16):
            xn = _proj(y - step * g, S)
            fx = _smooth_value(xn, alpha)
            d = xn - y
            rhs = fy + float(np.dot(g, d)) + 0.5 * (1.0 / step) * float(np.dot(d, d))
            if fx <= rhs + 1e-11:
                accepted = True
                break
            step *= 0.5
        Lcur = 1.0 / max(step, 1e-12)

        else:
            stale += 1
            hp, y, t, fy = h, h, 1.0, fh
            Lcur = max(Lcur * 0.5, 1.0)
            if stale > 12:
                break
    c = _c5_direct(h)
    if c < best_c:
        best_c, best_h = c, h.copy()
    return h, best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    return _proj(np.interp(x1, x0, h), m * 0.5)

def _local_search(h0, rng, deadline, start_step=5e-5):
    tl0 = time.time()
    n = len(h0)
    h = h0.copy()
    c = _c5_direct(h)
    best_c, best_h = c, h.copy()

    tol = max(1e-8, 10.0 * 2.0 / n)

    while time.time() < deadline:

            c = _c5_direct(h)

            P, Nv, mx = _pn_arrays(h)
            tol = max(1e-8, 10.0 * 2.0 / n)
            _recompute_grad(h, P, Nv, mx, tol, g)
            order = np.argsort(g)
            low, high = order[:q], order[-q:]
            recompute = it + 1024
        frac = min(1.0, (time.time() - tl0) / max(1e-9, deadline - tl0))
        base = max(1e-10, start_step * (1e-4) ** frac)
        if rng.random() < 0.75:

            j = int(high[rng.randint(len(high))])
        else:

        if i != j and g[i] <= g[j]:
            s = 1
        elif i != j:
            s = -1
        else:
            continue
        if s > 0:
            avail = min(1.0 - h[i], h[j])
        else:
            avail = min(h[i], 1.0 - h[j])
        if avail > 1e-14:
            mult = 0.3 + 2.0 * rng.random()
            if rng.random() < 0.05:

            h[j] -= d
            cn = _c5_direct(h)
            if cn < c - 1e-15:

                if c < best_c - 1e-15:
                    best_c, best_h = cn, h.copy()

def _recompute_grad(h, P, Nv, mx, tol, g):
    n = len(h)
    wP, wN = np.zeros(n), np.zeros(n)
    cnt = 0
    if P[0] >= mx - tol:
        wP[0] = 1.0; cnt += 1
    if n > 1:
        mP = P[1:] >= mx - tol
        if np.any(mP):
            pos = np.nonzero(mP)[0] + 1
            wP[pos] = 1.0; cnt += pos.size
        mN = Nv[1:] >= mx - tol
        if np.any(mN):
            pos = np.nonzero(mN)[0] + 1
            wN[pos] = 1.0; cnt += pos.size
    if cnt == 0:
        return
    wP /= cnt
    wN /= cnt
    np.copyto(g, _grad_from_weights(h, wP, wN))

def _run_alphas(h0, alphas, deadline, L=8.0):

    best_h, best_c = h0.copy(), _c5_direct(h0)

        if deadline - time.time() < 2.0 or per <= 0.0:

        if chf < best_c:
            best_c, best_h = chf, hf
        hcur = hf if chf <= bc else bh
    return best_h, best_c, hcur
