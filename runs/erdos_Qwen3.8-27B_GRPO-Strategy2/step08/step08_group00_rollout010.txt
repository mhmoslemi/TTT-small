One more thought: the IRLS approach. The idea is:
- We want to minimize max_k c_k
- Approximate by: minimize Σ w_k * c_k where w_k = softmax(β * c_k)
- Iterate: solve the weighted problem, update weights, increase β
- This is different from a single smooth max because the weights are updated based on the *current solution's* lag values

In practice, each IRLS iteration is:
1. c = corr(h)
2. top_m = argsort(c)[-M:]
3. w = softmax(β * c[top_m])
4. Solve: min Σ_{k in top_m} w_k * c_k(h) s.t. constraints
   (This is a QP with box + equality constraints, solvable by SLSQP)
5. h = solution
6. β *= α_mult

The objective in step 4 is: Σ w_k * Σ h[i](1-h[i+k]) which is a quadratic in h.
Its gradient: Σ_k w_k * ∂c_k/∂h
Which is just a weighted combination of the individual lag gradients.

This should work well because:
- At low β, it's like minimizing the average (easier, more convex)
- At high β, it focuses on the worst lags
- The reweighting adapts to the actual solution

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -300, 300)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -300, 300)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fn(x, half_n):
    return x.sum() - half_n

def _sum_jac(x, half_n):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _weighted_obj(x, w, lags, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    return float((c[lags] * w).sum())

def _weighted_grad(x, w, lags, n):
    dx = 2.0 / n
    g = np.zeros(n)
    for ii, k in enumerate(lags):
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _irls_step(h, beta, m_lags, n):
    c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
    idx = np.argsort(c)[-m_lags:]
    s = beta * c[idx]
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_weighted_obj, h, jac=_weighted_grad,
                       args=(w, idx, n), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': 800, 'ftol': 1e-15})

    except Exception:
        return h

def _proj_grad(h, alpha, m_lags, steps, step0, mu):

    cur_c = _eval(h)

        v = mu * v + (1.0 - mu) * g
        taken = False

            c_new = _eval(h_new)

                cur_c = c_new

                taken = True
                break
            step *= 0.35

    return _repair(h, n)

        d = (0.002 + 0.008 * rng.rand()) * (1.0 if rng.rand() < 0.5 else -1.0)

                    zero_idx = np.where(h < 0.95)[0]
            else:

def _coord_polish(h, n_rounds, rng):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    for r in range(n_rounds):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, max(1, n // 10))) % n

            continue
        lo_b = max(0.0, 1.0 - h[j])
        hi_b = min(1.0, h[i] + h[j] - lo_b)

            h[i], h[j] = m1, h[i] + h[j] - m1

        h[j] = h[i] + h[j] - h[i]
        h[j] = (h[i] + h[j]) - h[i]
        c5n = _eval(h)
        if c5n < c5cur - 1e-12:
            c5cur = c5n
            if c5cur < best_c:
