    hp=h[p]; hq=h[q]
    best_d=None; best_nm=None
    for frac in fracs:

def project_h(y, target):
    y = np.asarray(y, dtype=float).copy()
    y = np.clip(y, 0.0, 1.0)
    if y.size == 0:
        return y
    s = float(y.sum())
    if abs(s - target) <= 1e-13:
        return y
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5*(lo+hi)
        t = float(np.clip(y - mid, 0.0, 1.0).sum())
        if t > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - 0.5*(lo+hi), 0.0, 1.0)
    diff = target - float(z.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - z)
            for i in idx:
                if diff <= 1e-14: break
                a = 1.0 - z[i]
                if a <= 0: continue
                if a > diff:
                    z[i] += diff; diff=0.0
                else:
                    z[i] = 1.0; diff -= a
        else:
            idx = np.argsort(z)
            nd = -diff
            for i in idx:
                if nd <= 1e-14: break
                a = z[i]
                if a <= 0: continue
                if a > nd:
                    z[i] -= nd; nd=0.0
                else:
                    z[i] = 0.0; nd -= a
    return z

def try_pair_moves(h,c,p,q,sign,maxa,fracs,best_raw,work):
    if p == q or maxa <= 1e-12: return None
    hp=h[p]; hq=h[q]
    bd=None; bn=None
    for frac in fracs:
       mag=frac*maxa
       if mag <= 1e-11: continue
       d=sign*mag
       if d > 0:
          lim=min(1.0-hp, hq)
       else:
          lim=min(hp, 1.0-hq)
       if d > 0 and d > lim + 1e-12: continue
       if d < 0 and -d > lim + 1e-12: continue
       add_pair_delta(h,c,p,q,d,1.0,work)

    if bd is None:
       return None

import numpy as np
import time


def project_h(y, target):
    y = np.asarray(y, dtype=float).copy()
    y = np.clip(y, 0.0, 1.0)
    if y.size == 0:
        return y
    s = float(y.sum())
    if abs(s - target) <= 1e-13:
        return y
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        t = float(np.clip(y - mid, 0.0, 1.0).sum())
        if t > target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(z.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            for i in np.argsort(1.0 - z):
                if diff <= 1e-14:
                    break
                a = 1.0 - z[i]
                if a <= 0:
                    continue
                if a > diff:
                    z[i] += diff
                    diff = 0.0
                else:
                    z[i] = 1.0
                    diff -= a
        else:
            nd = -diff
            for i in np.argsort(z):
                if nd <= 1e-14:
                    break
                a = z[i]
                if a <= 0:
                    continue
                if a > nd:
                    z[i] -= nd
                    nd = 0.0
                else:
                    z[i] = 0.0
                    nd -= a
    return z


def evaluate_h(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, 'full')
    return c, float(c.max())


def smooth_initial(n, rng):
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 * np.ones(n)
    h += 0.18 * np.sin(np.pi * x + rng.rand() * 6.28)
    h += 0.12 * np.sin(2.0 * np.pi * x + rng.rand() * 6.28)
    h += 0.08 * np.sin(3.0 * np.pi * x + rng.rand() * 6.28)
    return project_h(h, float(n) / 2.0)


def active_gradient(h, c, k, alpha):
    n = len(h)
    m = 2 * n - 1
    k = max(1, min(m, int(k)))
    if k >= m:
        idx = np.arange(m)
    else:
        idx = np.argpartition(c, -k)[-k:]
    cv = c[idx]
    mv = float(cv.max())
    e = np.exp(alpha * (cv - mv))
    w = e / float(e.sum())
    g = np.zeros(n)
    k0 = n - 1
    for ik, wk in zip(idx, w):
        if wk <= 1e-15:
            continue
        sh = k0 - int(ik)
        if sh >= 0:
            r = n - sh
            if r > 0:
                g[:r] += wk * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= wk * h[:r]
        else:
            sp = -sh
            r = n - sp
            if r > 0:
                g[sp:] += wk * (1.0 - h[:r])
                g[:r] -= wk * h[sp:]
    return g


def add_pair_delta(h, c, p, q, d, sgn, work):
    n = len(h)
    sd = sgn * d
    qd = sgn * d * d
    hr = h[::-1]
    np.multiply(hr, -sd, out=work)
    work += sd
    c[p:p + n] += work
    st = n - 1 - p
    np.multiply(h, -sd, out=work)
    c[st:st + n] += work
    np.multiply(hr, sd, out=work)
    work -= sd
    c[q:q + n] += work
    st = n - 1 - q
    np.multiply(h, sd, out=work)
    c[st:st + n] += work
    k0 = n - 1
    c[k0] -= 2.0 * qd
    c[k0 + p - q] += qd
    c[k0 + q - p] += qd


def try_pair_moves(h, c, p, q, sign, maxa, fracs, best_raw, work):
    if p == q or maxa <= 1e-12:
        return None
    hp = h[p]
    hq = h[q]
    bd = None
    bn = None
    for frac in fracs:
        mag = frac * maxa
        if mag <= 1e-11:
            continue
        d = sign * mag
        if d > 0:
            lim = min(1.0 - hp, hq)
            if d > lim + 1e-12:
                continue
        else:
            lim = min(hp, 1.0 - hq)
            if -d > lim + 1e-12:
                continue
        add_pair_delta(h, c, p, q, d, 1.0, work)
        h[p] = hp + d
        h[q] = hq - d
        nm = float(c.max())
        if nm < best_raw - 1e-12:
            if bn is None or nm < bn:
                bn = nm
                bd = d
        h[p] = hp
        h[q] = hq
        add_pair_delta(h, c, p, q, -d, -1.0, work)
    if bd is None:
        return None
    add_pair_delta(h, c, p, q, bd, 1.0, work)
    h[p] = hp + bd
    h[q] = hq - bd
    return bn


def sweep_gradient_pairs(h, c, best_raw, rng, work, t_end, alpha=80, k=100):
    n = len(h)
    k0 = n - 1
    half = max(1, n // 2)
    eps = 1e-10
    stall = 0
    while time.time() < t_end:
        c_true = np.correlate(h, 1.0 - h, 'full')
        rt = float(c_true.max())
        if rt > best_raw + 1e-5:
            break
        if rt < best_raw:
            best_raw = rt
        c[:] = c_true
        g = active_gradient(h, c, k, alpha)
        gm = float(g.mean())
        gorder = np.argsort(g)
        low = gorder[:half]
        high = gorder[half:]
        order = np.argsort(-np.abs(g - gm))
        acc = 0
        fracs = (0.35, 0.18)
        for p in order:
            if time.time() >= t_end:
                break
            hp = h[p]
            if g[p] >= gm:
                if hp <= eps:
                    continue
                sign = -1
                cand = low
            else:
                if hp >= 1.0 - eps:
                    continue
                sign = 1
                cand = high
            q = -1
            for _ in range(8):
                q = int(cand[rng.randint(len(cand))])
                if q == p:
                    continue
                if sign < 0:
                    if h[q] < 1.0 - eps:
                        break
                else:
                    if h[q] > eps:
                        break
            if q < 0:
                continue
            if sign < 0:
                maxa = min(h[p], 1.0 - h[q])
            else:
                maxa = min(1.0 - h[p], h[q])
            val = try_pair_moves(h, c, p, q, sign, maxa, fracs, best_raw, work)
            if val is not None:
                best_raw = val
                acc += 1
        if acc == 0:
            top = np.argsort(c)[-30:]
            acc2 = 0
            fr2 = (0.40, 0.20)
            for _ in range(500):
                if time.time() >= t_end:
                    break
                idx = int(top[rng.randint(len(top))])

                lim = n - kk if kk >= 0 else n + kk

                if val is not None:
                    best_raw = val
                    acc2 += 1
            if acc2 == 0:

            else:

    return h, c, best_raw


def fw_loop(h, c, best_raw, rng, t_end, alpha=100, k=100):

            if time.time() >= t_end:
                break
            h2 = np.clip(h + gamma * (s - h), 0.0, 1.0)
            c2 = np.correlate(h2, 1.0 - h2, 'full')
            r2 = float(c2.max())
            if r2 < best_r - 1e-14:

            if r2 < best_raw - 1e-14:
                h, c, best_raw, stall = h2, c2, r2, 0
            else:
                stall += 1
    return h, c, best_raw


def binary_from_h(h, rng, target_sum, jitter=1.0):
    n = len(h)
    base = int(target_sum)
    rem = target_sum - base
    score = h + jitter * rng.rand(n)
    b = np.zeros(n)
    if base > 0:
        idx = np.argpartition(-score, base - 1)[:base]

    if rem > 1e-12:
        zeros = np.where(b == 0.0)[0]
        if len(zeros):
            b[zeros[rng.randint(len(zeros))]] = rem
    return b
