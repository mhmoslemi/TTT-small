def _repair(h, n):
    h = np.clip(h, 0, 1).copy()
    target = 0.5*n
    if abs(h.sum() - target) > 1e-10:
        h = _proj(h, target)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1-h) if diff > 0 else np.argsort(h)
        for i in order:
            if diff > 0:
                a = min(diff, 1-h[i]); h[i] += a; diff -= a
            else:
                a = min(-diff, h[i]); h[i] -= a; diff += a
            if abs(diff) < 1e-14: break
    return h

def _grad_sub(h, tol, topk=None):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk and topk < len(c):
        active = np.argpartition(c, -topk)[-topk:]
    else:
        active = np.where(c >= m - tol)[0]
    dx = 2.0/n
    g = np.zeros(n)
    for k in active:
        ms = k - (n-1)
        if ms >= 0:
            v = n - ms
            g[:v] += 1.0 - h[ms:ms+v]
            g[ms:] -= h[:v]
        else:
            s = -ms; v = n - s
            g[s:] += 1.0 - h[:v]
            g[:v] -= h[s:]
    g *= dx / len(active)
    return g, m

def _nesterov(h, steps, step0, deadline, tol, beta=0.85, topk=None, step_max=30.0):
    n = len(h)
    tgt = 0.5*n
    f = _eval(h)
    best_h, best_f = h.copy(), f
    v = np.zeros(n)
    step = step0
    for _ in range(steps):
        if time.time() > deadline: break
        g, _ = _grad_sub(h, tol, topk)
        v = beta*v + (1-beta)*g
        hn = _proj(h - step*v, tgt)
        fn = _eval(hn)
        if fn < f - 1e-16:
            h, f = hn, fn
            step = min(step*1.02, step0*step_max)
            if f < best_f: best_h, best_f = h.copy(), f
        else:
            step *= 0.5
            if step < 1e-10: break
            hn = _proj(h - step*v, tgt)
            f = _eval(hn)
            h = hn
            if f < best_f: best_h, best_f = h.copy(), f
    return _repair(best_h, n), best_f

            # But also respect box constraints

                if ei <= 0: continue

                        best_i, best_j, e = i, j, e

                else:
                    h[best_i] -= e
                    h[best_j] += e
        if not moved: break

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):

    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        c = _correl(x)
        m = c.max()
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
        s = np.clip(alpha * (ct - m), -400, 400)
        return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    def jac(x):
        return _grad_sub(x, 0.01, topk)[0]
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - 0.5 * n,
             'jac': lambda x: np.ones(n)}]
    try:

                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18})
