```python
def _lag_repair(h, n_trials, rng, deadline):
    """Targeted repair of worst lags"""
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        cor = _correl(h)
        m = cor.max()
        # Get top 3 worst lags
        top3 = np.argsort(-cor)[:3]
        kk = top3[rng.randint(3)]
        off = kk - (n - 1)
        # Build gradient for this specific lag
        g = _sg(h, kk, n)
        # Find where the gradient is most negative (moving h in -g direction reduces c_k the most)
        # Actually, to reduce c_k, we want to move in the direction that decreases c_k
        # dc_k/dh[j] = g[j], so we want to decrease h where g[j] > 0 and increase where g[j] < 0
        # But we need to keep sum(h) constant, so we do a paired move
        pos = g > 0.3  # positions where increasing h increases c_k
        neg = g < -0.3  # positions where increasing h decreases c_k
        if pos.sum() < 1 or neg.sum() < 1:
            continue
        ip = rng.choice(np.where(pos)[0])
        in_ = rng.choice(np.where(neg)[0])
        amt = 0.01 * (0.5 + rng.random())
        h_new = h.copy()
        h_new[ip] -= amt
        h_new[in_] += amt
        if 0 <= h_new[ip] <= 1 and 0 <= h_new[in_] <= 1:
            c5n = _eval(h_new)
            if c5n < c5c:
                c5c = c5n
                h = h_new
                if c5n < bc:
                    bc, bh = c5n, h_new.copy()
    return bh, bc
```

```python
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _sg(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    mask1 = (i1 >= 0) & (i1 < n)
    g[mask1] += (1.0 - h[i1[mask1]])
    i2 = j + off
    mask2 = (i2 >= 0) & (i2 < n)
    g[mask2] -= h[i2[mask2]]
    return g

def _slsqp_run(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        c = _correl(x)
        m = c.max()
        if topk < len(c):
            idx = np.argpartition(c, -topk)[-topk:]
            ct = c[idx]
        else:
            ct = c
        s = np.clip(alpha * (ct - m), -400, 400)
        return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    def jac(x):
        nc = len(x)
        c = _correl(x)
        m = c.max()
        if topk < len(c):
            idx = np.argpartition(c, -topk)[-topk:].copy()
        else:
            idx = np.arange(len(c))
        ct = c[idx]
        s = np.clip(alpha * (ct - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(nc)
        dx = 2.0 / nc
        for ii in range(len(idx)):
            if w[ii] < 1e-13:
                continue
            kk, wk = idx[ii], w[ii] * dx
            g += wk * _sg(x, kk, nc)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-20, 'maxls': 60})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _lp_descent(h, n_steps, eps, m_top, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    fail_count = 0
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        cor = _correl(c)
        idx_top = np.argpartition(cor, -m_top)[-m_top:]
        idx_top = idx_top[np.argsort(-cor[idx_top])]
        grads = np.array([_sg(c, k, n) for k in idx_top])
        c_obj = -grads[0]
        A_ub = grads[1:] if m_top > 1 else None
        b_ub = np.zeros(m_top - 1) if m_top > 1 else None
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bnds = [(-eps, eps)] * n
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs', options={'presolve': True})
        except Exception:
            fail_count += 1
            if fail_count > 10:
                break
            continue
        if not res.success or res.x is None or np.max(np.abs(res.x)) < 1e-14:
            fail_count += 1
            if fail_count > 15:
                # Try random perturbation
                r2 = np.random.RandomState(int(time.time() * 1e6) % 100000)
                d = r2.randn(n)
                d -= d.mean()
                d *= eps * 0.5
                h_new = _repair(c + d, n)
                c5_new = _eval(h_new)
                if c5_new < c5_cur:
                    c, c5_cur = h_new, c5_new
                    if c5_cur < best_c5:
                        best_c5, best_h = c5_cur, c.copy()
                fail_count = 0
                continue
            break
        fail_count = 0
        d = res.x
        step = 1.0
        improved = False
        for _ in range(10):
            h_new = _repair(c + step * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-16:
                c, c5_cur = h_new, c5_new
                improved = True
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
                break
            step *= 0.5
        if not improved:
            # Small random kick
            r3 = np.random.RandomState(int(time.time() * 1e6) % 100000)
            d = r3.randn(n)
            d -= d.mean()
            d *= eps * 0.3
            h_new = _repair(c + d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur:
                c, c5_cur = h_new, c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h_new.copy()
    return best_h, best_c5

def _proj_grad(h, deadline, topk=5, steps=8000, step0=0.002):
    """Projected subgradient descent on exact max"""
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    step = step0
    momentum = np.zeros(n)
    beta = 0.7
    stall = 0
    for it in range(steps):
        if time.time() > deadline:
            break
        cor = _correl(c)
        m = cor.max()
        idx = np.argpartition(cor, -topk)[-topk:]
        # Equal weights on top-k
        g = np.zeros(n)
        for kk in idx:
            g += _sg(c, kk, n)
        g /= len(idx)
        # Momentum
        momentum = beta * momentum + (1 - beta) * g
        # Projected step
        c_new = _proj(c - step * momentum, n * 0.5)
        c5_new = _eval(c_new)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, c_new.copy()
        if c5_new < _eval(c) - 1e-15:
            c = c_new
            stall = 0
            step = min(step * 1.02, step0 * 3)
        else:
            stall += 1
            step *= 0.7
            if step < 1e-7:
                # Restart with perturbation
                r = np.random.RandomState(it % 10000)
                d = 0.002 * r.randn(n)
                d -= d.mean()
                c = _proj(c + d, n * 0.5)
                momentum *= 0.5
                step = step0
                stall = 0
    return best_h, best_c5

def _lag_search(h, n_trials, rng, deadline):

    n = len(h)

        # But also need to check box constraints

        if gi.max() < 0 or gj.max() > 0:

            j = rng.randint(n)

            candidates_j = np.where(g < np.percentile(g, 25))[0]

                if i == j:
                    continue
            else:
                i = candidates_i[rng.randint(len(candidates_i))]
                j = candidates_j[rng.randint(len(candidates_j))]
        amt = 0.005 * (0.3 + 0.7 * rng.random())
        h_new = h.copy()
        if g[i] > g[j]:

            h_new[j] += amt

            if c5n < c5c:
                c5c = c5n
                h = h_new
                if c5n < bc:
                    bc, bh = c5n, h_new.copy()

def _polish(h, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    scales = [5e-6, 2e-6, 1e-6, 5e-7, 2e-7]
    per_scale = 15000
    for ss in scales:
        for _ in range(per_scale):
            if time.time() > deadline:
                return bh, bc
            i = rng.randint(nc)
            j = (i + 1 + rng.randint(0, 5)) % nc
            d = ss * (2 * rng.random() - 1)
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5n < bc:
                        bc, bh = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc
```

```python
        if not improved:
            d2 = rng_kick.randn(n)
            d2 -= d2.mean(); d2 *= eps * 0.2
            h_new = _repair(c + d2, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur:
                c, c5_cur = h_new, c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
    return best_h, best_c5

def _proj_grad(h, deadline, topk, steps, step0):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    step = step0
    momentum = np.zeros(n)
    beta = 0.6
    rng_r = np.random.RandomState(777)
    for it in range(steps):
        if time.time() > deadline:
            break
        cor = _correl(c)
        idx = np.argpartition(cor, -topk)[-topk:]
        g = np.zeros(n)
        for kk in idx:
            g += _sg(c, kk, n)
        g /= len(idx)
        momentum = beta * momentum + (1.0 - beta) * g
        c_new = _proj(c - step * momentum, n * 0.5)
        c5_new = _eval(c_new)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, c_new.copy()
        c5_old = _eval(c)
        if c5_new < c5_old - 1e-15:
            c = c_new
            step = min(step * 1.03, step0 * 4.0)
        else:
            step *= 0.6
            if step < 1e-8:
                d = 0.003 * rng_r.randn(n)
                d -= d.mean()
                c = _proj(c + d, n * 0.5)
                momentum *= 0.3
                step = step0 * 0.5
    return best_h, best_c5

def _lag_search(h, n_trials, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy(); bc = c5c
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        cor = _correl(h)
        top3 = np.argsort(-cor)[:3]
        kk = top3[rng.randint(3)]
        g = _sg(h, kk, n)
        lo_q = np.percentile(g, 30)
        hi_q = np.percentile(g, 70)
        ci = np.where(g > hi_q)[0]
        cj = np.where(g < lo_q)[0]
        if len(ci) < 1 or len(cj) < 1:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
        else:
            i = ci[rng.randint(len(ci))]
            j = cj[rng.randint(len(cj))]
        amt = 0.004 * (0.4 + 0.6 * rng.random())
        h_new = h.copy()
        if g[i] > g[j]:
            h_new[i] -= amt; h_new[j] += amt
        else:
            h_new[i] += amt; h_new[j] -= amt
        if 0 <= h_new[i] <= 1 and 0 <= h_new[j] <= 1:
            c5n = _eval(h_new)
            if c5n < c5c:
                c5c = c5n; h = h_new
                if c5n < bc:
                    bc, bh = c5n, h_new.copy()
    return bh, bc

def _polish(h, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy(); bc = c5c
    scales = [4e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]
    per_scale = 8000
    for ss in scales:
        for _ in range(per_scale):
            if time.time() > deadline:
                return bh, bc
            i = rng.randint(nc)
            j = (i + 1 + rng.randint(0, 6)) % nc
            d = ss * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5n < bc:
                        bc, bh = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    dl1 = min(deadline, t0 + 60)
    for a, tk, mi in [(4000, 30, 200), (25000, 12, 180), (100000, 6, 120)]:
        if time.time() > dl1:
            break
        hv = _slsqp_run(best_h.copy(), a, tk, mi, dl1)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    dl2 = min(deadline, t0 + 310)
    hv, cv = _lp_descent(best_h.copy(), 30000, 0.003, 5, dl2)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    dl3 = min(deadline, t0 + 440)
    hv, cv = _proj_grad(best_h.copy(), dl3, topk=4, steps=12000, step0=0.002)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    dl4 = min(deadline, t0 + 620)
    hv, cv = _lp_descent(best_h.copy(), 25000, 0.0015, 4, dl4)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    dl5 = min(deadline, t0 + 760)
    hv, cv = _lag_search(best_h.copy(), 25000, rng, dl5)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    dl6 = min(deadline, t0 + 870)
    hv, cv = _lp_descent(best_h.copy(), 15000, 0.001, 3, dl6)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    hv, cv = _polish(best_h, rng, deadline)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
```
