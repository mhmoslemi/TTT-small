        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lag(h, k, n):
    sh = (n - 1) - k
    g = np.zeros(n)
    dx2 = 2.0 / n
    if sh >= 0:
        g[:n - sh] += dx2 * (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= dx2 * h[:n - sh]
    else:
        sp = -sh
        g[sp:] += dx2 * (1.0 - h[:n - sp])
        g[:n - sp] -= dx2 * h[sp:]
    return g

def _topk_grad(h, k_top, n):
    c = _corr(h, n)
    top_k = np.argsort(c)[-k_top:]
    g = np.zeros(n)
    w = 1.0 / k_top
    for k in top_k:
        g += w * _grad_lag(h, k, n)
    return g

def _proj_descend(h, k_top, n, step0, t_end, max_iter=200000):
    h = h.copy()
    step = step0
    f = _eval(h)
    stall = 0
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        g = _topk_grad(h, k_top, n)
        h_new = _repair(h - step * g, n)
        f_new = _eval(h_new)
        if f_new < f - 1e-16:
            h, f = h_new, f_new
            stall = 0
            step = min(step * 1.04, step0 * 25.0)
        else:
            step *= 0.45
            stall += 1
            if stall > 50 or step < 1e-15:
                break
    return h, f

def _nesterov_topk(h, k_top, n, steps, step0, mu, t_end):
    h = h.copy()
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    stall = 0
    for _ in range(steps):
        if time.time() > t_end:
            break
        g = _topk_grad(h, k_top, n)
        v = mu * v + (1.0 - mu) * g
        h_new = _repair(h - step * v, n)
        f_new = _eval(h_new)
        if f_new < f - 1e-16:
            h, f = h_new, f_new
            stall = 0
            step = min(step * 1.03, step0 * 15.0)
        else:
            step *= 0.4
            stall += 1
            if stall > 40 or step < 1e-15:
                v = np.zeros(n)
                step = step0 * 0.3
                stall = 0
    return h, f

class _LSE:
    def __init__(self, alpha, n):
        self.alpha = alpha
        self.n = n
        self.dx2 = 2.0 / n
    def f(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * self.dx2
        m = c.max()
        s = np.clip(self.alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s - s.max()))) / self.alpha
    def g(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * self.dx2
        m = c.max()
        s = np.clip(self.alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(self.n)
        for k in range(2 * self.n - 1):
            if w[k] < 1e-15:
                continue
            sh = (self.n - 1) - k
            wk = w[k] * self.dx2
            if sh >= 0:
                g[:self.n - sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk * x[:self.n - sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:self.n - sp])
                g[:self.n - sp] -= wk * x[sp:]
        return g

class _Con:
    def __init__(self, target):
        self.target = target
    def fun(self, x):
        return x.sum() - self.target
    def jac(self, x):
        return np.ones(x.shape[0])

def _slsqp_solve(h, alpha, maxiter, ftol, target):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    lse = _LSE(alpha, n)
    con = _Con(target)
    cons = [{'type': 'eq', 'fun': con.fun, 'jac': con.jac}]
    try:
        res = minimize(lse.f, h, jac=lse.g, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _local_3pt(h, n, n_tries, sc, t_end, rng):
    h = h.copy()
    f = _eval(h)
    best_h, best_f = h.copy(), f
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 2 + rng.randint(0, 5)) % n
        k = (i + 8 + rng.randint(0, 12)) % n
        d1 = sc * rng.randn()
        d2 = sc * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0.0 < ni < 1.0 and 0.0 < nj < 1.0 and 0.0 < nk < 1.0:
            h[i], h[j], h[k] = ni, nj, nk
            h = _repair(h, n)
            fn = _eval(h)
            if fn < f:
                f = fn
                if f < best_f:
                    best_f, best_h = f, h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
                h = _repair(h, n)
    return best_h, best_f

def _local_2pt(h, n, n_tries, sc, t_end, rng):
    h = h.copy()
    f = _eval(h)
    best_h, best_f = h.copy(), f
    no_improve = 0
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = sc * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 < ni < 1.0 and 0.0 < nj < 1.0:
            h[i], h[j] = ni, nj
            h = _repair(h, n)
            fn = _eval(h)
            if fn < f:
                f = fn
                no_improve = 0
                if f < best_f:
                    best_f, best_h = f, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                h = _repair(h, n)
                no_improve += 1
                if no_improve > 10000:
                    sc *= 0.5
                    no_improve = 0
    return best_h, best_f

def _coarse_refine(h, n, rng, t_end):
    n_coarse = max(40, n // 8)
    idx = np.linspace(0, n - 1, n_coarse, dtype=int)
    h_c = h[idx].copy()
    f_c = _eval(h_c)
    for _ in range(5000):
        if time.time() > t_end:
            break
        i = rng.randint(n_coarse)
        j = (i + 1 + rng.randint(0, 3)) % n_coarse
        d = 0.01 * rng.randn()
        ni, nj = h_c[i] + d, h_c[j] - d
        if 0 < ni < 1 and 0 < nj < 1:
            h_c[i], h_c[j] = ni, nj
            h_c = _repair(h_c, n_coarse)
            fn = _eval(h_c)
            if fn < f_c:
                f_c = fn
            else:
                h_c[i], h_c[j] = ni - d, nj + d
                h_c = _repair(h_c, n_coarse)
    x_full = np.linspace(0, 2, n, endpoint=False)
    x_coarse = np.linspace(0, 2, n_coarse, endpoint=False)
    h_interp = np.interp(x_full, x_coarse, h_c, period=2.0)
    return _repair(h_interp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_gend = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    target = n / 2.0

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _upd(hn):
        cv = _eval(hn)
        if cv < best_c5:
            return True, cv, hn.copy()
        return False, cv, hn

    # Phase 0: Coarse-grain exploration (quick)
    t_p0 = t0 + min(30, budget_s * 0.03)
    hr = _coarse_refine(best_h, n, rng, t_p0)
    cv = _eval(hr)
    if cv < best_c5:
        best_c5, best_h = cv, hr.copy()

    # Phase 1: Projected gradient, small k
    t_p1 = t0 + budget_s * 0.15
    for k in [3, 5, 10]:
        if time.time() > t_p1 - 3:
            break
        hn, _ = _proj_descend(best_h.copy(), k, n, 0.0015, t_p1)
        flag, cv, hnew = _upd(hn)
        if flag:
            best_h, best_c5 = hnew, cv

    # Phase 2: SLSQP moderate alpha
    t_p2 = t0 + budget_s * 0.32
    for alpha in [3000, 30000]:
        if time.time() > t_p2 - 5:
            break
        hn = _slsqp_solve(best_h.copy(), alpha, 1200, 1e-15, target)
        flag, cv, hnew = _upd(hn)
        if flag:
            best_h, best_c5 = hnew, cv

    # Phase 3: Nesterov accelerated, medium k
    t_p3 = t0 + budget_s * 0.48
    for k in [15, 40]:
        if time.time() > t_p3 - 5:
            break
        hn, _ = _nesterov_topk(best_h.copy(), k, n, 5000, 0.001, 0.82, t_p3)
        flag, cv, hnew = _upd(hn)
        if flag:
            best_h, best_c5 = hnew, cv

    # Phase 4: SLSQP high alpha
    t_p4 = t0 + budget_s * 0.60
    for alpha in [300000, 3000000]:
        if time.time() > t_p4 - 10:
            break
        hn = _slsqp_solve(best_h.copy(), alpha, 800, 1e-16, target)
        flag, cv, hnew = _upd(hn)
        if flag:
            best_h, best_c5 = hnew, cv

    # Phase 5: Fine projected gradient
    t_p5 = t0 + budget_s * 0.72
    hn, _ = _proj_descend(best_h.copy(), 8, n, 0.0004, t_p5)
    flag, cv, hnew = _upd(hn)
    if flag:
        best_h, best_c5 = hnew, cv

    # Phase 6: Nesterov very fine
    if time.time() < t_gend - 60:
        hn, _ = _nesterov_topk(best_h.copy(), 5, n, 8000, 0.0002, 0.85, t_gend - 45)
        flag, cv, hnew = _upd(hn)
        if flag:
            best_h, best_c5 = hnew, cv

    # Phase 7: 3-point local search
    if time.time() < t_gend - 30:
        ntr = min(40000, int((t_gend - 30 - time.time()) / 0.003))
        best_h, best_c5 = _local_3pt(best_h, n, ntr, 3.5e-5, t_gend - 15, rng)

    # Phase 8: 2-point ultra-fine
    if time.time() < t_gend - 3:
        ntr = min(60000, int((t_gend - 3 - time.time()) / 0.002))
        best_h, best_c5 = _local_2pt(best_h, n, ntr, 1.5e-5, t_gend - 0.5, rng)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
