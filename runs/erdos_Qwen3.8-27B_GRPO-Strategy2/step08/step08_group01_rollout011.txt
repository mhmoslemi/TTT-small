import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n * 0.5
    for _ in range(80):
        s = h.sum()
        if s < 1e-14:
            h[:] = 0.5
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0 and abs(diff) < 2.0:
            adj = diff / mask.sum()
            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _sobj(x, alpha, tk):
    c = _corr(x)
    idx = np.argsort(c)[-tk:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -500, 500)
    s -= s.max()
    v = np.exp(s)
    return m + np.log(v.sum()) / alpha

def _sgrad(x, alpha, tk):
    n = len(x)
    dx = 2.0 / n
    c = _corr(x)
    idx = np.argsort(c)[-tk:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -500, 500)
    e = np.exp(s - s.max())
    w = e / e.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        k = int(idx[ii])
        wk = w[ii] * dx
        if wk < 1e-17:
            continue
        nlo = max(0, -k)
        nhi = min(n - 1, n - 1 - k)
        if nhi < nlo:
            continue
        j1lo = max(0, nlo + k)
        j1hi = min(n - 1, nhi + k)
        if j1hi >= j1lo:
            g[j1lo:j1hi+1] += wk * (1.0 - x[j1lo+k:j1hi+k+1])
        g[nlo:nhi+1] -= wk * x[nlo+k:nhi+k+1]
    return g

def _mgd_adaptive(h, alpha_start, n_phases, base_iter, max_time):
    n = len(h)
    target = n * 0.5
    t0 = time.time()
    cur_c = _eval(h)
    alpha = alpha_start
    for ph in range(n_phases):
        if time.time() - t0 > max_time:
            break
        v = np.zeros(n)
        y = h.copy()
        mu = 0.7
        step = 0.003
        for it in range(base_iter):
            if time.time() - t0 > max_time:
                break
            g = _sgrad(h, alpha, 15)
            g -= g.mean()
            y = h + step * g
            v = mu * v + (1.0 - mu) * (y - h)
            h_try = _repair(y + mu * v, n)
            c_try = _eval(h_try)
            if c_try < cur_c:
                h = h_try
                cur_c = c_try
                step = min(step * 1.03, 0.02)
            else:
                step = max(step * 0.65, 0.0003)
        alpha *= 0.6
    return _repair(h, n), cur_c

def _lbfgs_run(h, alpha, maxiter, tk):
    n = len(h)
    target = n * 0.5
    pen = 1e5
    def _f(x):
        c = _corr(x)
        idx = np.argsort(c)[-tk:]
        ct = c[idx]
        m = ct.max()
        s = np.clip(alpha * (ct - m), -500, 500)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha + pen * (x.sum() - target) ** 2
    def _g(x):
        g = _sgrad(x, alpha, tk)
        g += 2.0 * pen * (x.sum() - target)
        return g
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_f, h.copy(), jac=_g, method='L-BFGS-B',
                      bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        return _repair(res.x, n)
    except Exception:
        return h

def _block_search(h, n_iter, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        bsiz = rng.randint(3, 9)
        i0 = rng.randint(0, n - bsiz)
        idx = np.arange(i0, i0 + bsiz)
        d = 0.015 * rng.randn(bsiz)
        d -= d.mean()
        saving = h[idx].copy()
        trial = h[idx] + d
        if np.any(trial < -0.01) or np.any(trial > 1.01):
            continue
        h[idx] = np.clip(trial, 0.0, 1.0)
        c_new = _eval(h)
        if c_new < cur_c:
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[idx] = saving
    return best_h, best_c

def _lag_targeted(h, n_iter, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        c = _corr(h)
        top_idx = np.argsort(c)[-5:]
        k = int(rng.choice(top_idx))
        lag = abs(k - (n - 1))
        if lag == 0:
            lag = rng.randint(1, 20)
        i = rng.randint(0, n - lag)
        j = i + lag
        d = 0.008 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _sweep_fine(h, n_passes, step0, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
