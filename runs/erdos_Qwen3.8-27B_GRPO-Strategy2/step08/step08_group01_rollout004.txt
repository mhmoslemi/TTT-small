def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _proj_sum(h, target):
    """Project h onto {x in [0,1]^n : sum(x) = target}"""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_max(h, k):
    """Gradient of O(k) = sum_i h[i]*(1-h[i+k])*dx with respect to h"""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    # O(k) = dx * sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
    # dO/dh[i] for i < n-k: dx * (1 - h[i+k])  [from first factor]
    # dO/dh[i] for i >= k: dx * (-h[i-k])  [from second factor, (1-h[i]) becomes (1-h[i]-d)]
    if k == 0:
        # O(0) = dx * sum h[i]*(1-h[i])
        g[:] = dx * (1 - 2*h)
    else:

        for i in range(n - k):
            g[i] += dx * (1 - h[i + k])
        # Second factor contributions: i in [k, n-1]
        for i in range(k, n):
            g[i] += -dx * h[i - k]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _eval_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _proj_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_k(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    else:
        g[:n - k] += dx * (1.0 - h[k:])
        g[k:] += -dx * h[:n - k]
    return g

def _mgd(h, n_iter, top_k, step0, rng):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    mu = 0.82
    step = step0
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    stall = 0
    for it in range(n_iter):
        c = _eval_all(h)
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_max = c[idx].max()
        w = np.exp(300.0 * (c[idx] - c_max))
        w /= w.sum()
        g = np.zeros(n)
        for ii, kk in enumerate(idx):
            if w[ii] > 1e-12:
                g += w[ii] * _grad_k(h, kk)
        y = h + step * g + mu * v
        v = y - h
        h_try = _proj_sum(y, target)
        c_try = _eval(h_try)
        if c_try < cur_c - 1e-15:
            h = h_try; cur_c = c_try; stall = 0
            if c_try < best_c:
                best_c = c_try; best_h = h.copy()
            step = min(step * 1.04, step0 * 4.0)
        else:
            stall += 1
            step *= 0.55
            step = max(step, step0 * 0.05)
            if stall > 200:
                break
    return _repair(best_h, n), best_c

def _subgrad_mg(h, n_iter, rng):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    step = 0.008
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for it in range(n_iter):
        c = _eval_all(h)
        ks = int(np.argmax(c))
        g = _grad_k(h, ks)
        v = 0.8 * v - step * g
        h_try = _proj_sum(h + v, target)
        c_try = _eval(h_try)

            h = h_try; cur_c = c_try

def _fourier_path(h_init, n, K, t_limit):
    x = 2.0 * np.arange(n) / n
    cols = []
    for k in range(1, K + 1):
        cols.append(np.cos(np.pi * k * x))
        cols.append(np.sin(np.pi * k * x))
    X = np.column_stack(cols)
    coeffs0, _, _, _ = np.linalg.lstsq(X, h_init - 0.5, rcond=None)
    def _obj_f(coeffs):
        h = _repair(0.5 + X @ coeffs, n)
        return _eval(h)
    res = minimize(_obj_f, coeffs0, method='Nelder-Mead',
                   options={'maxiter': 3000, 'xatol': 1e-9, 'fatol': 1e-12})
    if time.time() > t_limit:
        return h_init, _eval(h_init)
    h_f = _repair(0.5 + X @ res.x, n)
    return h_f, _eval(h_f)

        i = rng.randint(n)

def _local3(h, n_iter, step_sz, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):

def _block_move(h, n_iter, step_sz, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
