    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:

                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_shift(h, sh, n):
    g = np.zeros(n)
    if sh >= 0:
        g[:n-sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g

class _SlsqpCtx:
    def __init__(self, alpha, n, target):
        self.alpha = alpha
        self.n = n
        self.target = target
    def obj(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / self.n)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s - s.max()))) / self.alpha
    def grad(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / self.n)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(self.n)
        for k in range(2 * self.n - 1):
            if w[k] < 1e-14:
                continue
            sh = (self.n - 1) - k
            wk = w[k] * (2.0 / self.n)
            g += wk * _grad_shift(x, sh, self.n)
        return g
    def con_fun(self, x):
        return x.sum() - self.target
    def con_jac(self, x):
        return np.ones(self.n)

def _run_slsqp(h, alpha, maxiter, target):
    n = len(h)
    ctx = _SlsqpCtx(alpha, n, target)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': ctx.con_fun, 'jac': ctx.con_jac}]
    try:
        res = minimize(ctx.obj, h, jac=ctx.grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov_adaptive(h, t_end, step0=0.004):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.75
    step = step0
    best_h, best_f = h.copy(), f
    total_lags = 2 * n - 1
    k_fraction = 0.4
    while time.time() < t_end:
        c = _corr(h)
        k_top = max(1, int(k_fraction * total_lags))
        top_k = np.argsort(c)[-k_top:]
        c_max = c.max()
        gap = c[top_k] - (c_max - 0.005)
        weights = np.exp(50.0 * (c[top_k] - c_max))
        weights /= weights.sum()
        g = np.zeros(n)
        for i, k in enumerate(top_k):
            sh = (n - 1) - k
            g += weights[i] * (2.0 / n) * _grad_shift(h, sh, n)
        h_prev = h.copy()

        h_new = _repair(h - step * v, n)
        f_new = _eval(h_new)
        if f_new < f:
            h, f = h_new, f_new
            if f < best_f:

            step = min(step * 1.04, step0 * 30.0)

def _block_search(h, rng, t_end, base_scale=0.015):

    c5c = _eval(h)
    best_h, best_c = h.copy(), c5c
    sizes = [3, 5, 7, 10, 14]
    while time.time() < t_end:
        bs = sizes[rng.randint(len(sizes))]
        i = rng.randint(max(1, n - bs))
        perturb = base_scale * rng.randn(bs)
        perturb -= perturb.mean()
        new_vals = h[i:i+bs] + perturb
        if new_vals.min() >= -1e-9 and new_vals.max() <= 1.0 + 1e-9:
            old = h[i:i+bs].copy()
