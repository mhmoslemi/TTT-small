import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):

def _grad_overlap(h, k, n):
    """Gradient of overlap at lag k w.r.t. h"""
    dx = 2.0 / n
    # overlap_k = sum_i h[i] * (1-h[i+k]) * dx (for valid i)
    # d/dh[j] = dx * sum over pairs involving j
    # For lag k: overlap = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
    # d/dh[j]: 
    #   if j < n-k: term h[j]*(1-h[j+k]) contributes: (1-h[j+k])
    #   if j >= k: term h[j-k]*(1-h[j]) contributes: -h[j-k]
    g = np.zeros(n)
    if 0 <= k < n:
        # h[i] appears as first factor for i in [0, n-k)
        i1 = np.arange(min(n-k, n))
        if len(i1) > 0:
            g[i1] += (1.0 - h[i1 + k])
        # h[j] appears as second factor (1-h[j]) for j in [k, n)
        i2 = np.arange(k, n)
        if len(i2) > 0:
            g[i2] -= h[i2 - k]
    elif -n < k < 0:
        sp = -k
        # overlap = sum_{i=sp}^{n-1} h[i]*(1-h[i-sp])
        # h[j] as first factor: j in [sp, n)
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        # h[j] as second factor: (1-h[j]) for j in [0, n-sp)
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _nesterov_proj_grad(h, alpha, n_steps, initial_step, top_lags, momentum):
    """Nesterov accelerated projected gradient"""
    n = len(h)
    dx = 2.0 / n
    v = np.zeros(n)
    y = h.copy()
    cur_c = _eval(h)
    step = initial_step
    
    for it in range(n_steps):
        # Compute correlation and identify active lags
        c_raw = np.correlate(h, 1.0 - h, mode='full') * dx
        # Get top lags
        if top_lags < len(c_raw):
            active_idx = np.argsort(c_raw)[-top_lags:]
        else:
            active_idx = np.arange(len(c_raw))
        c_a = c_raw[active_idx]
        
        # Compute gradient of smooth max over active lags
        m_a = c_a.max()
        s = np.clip(alpha * (c_a - m_a), -50, 50)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        
        # Gradient
        g = np.zeros(n)
        for ii in range(len(active_idx)):
            if w[ii] < 1e-12:
                continue
            k = active_idx[ii]
            # k in [0, 2n-2], actual shift = k - (n-1)
            shift = k - (n - 1)
            g += w[ii] * _grad_overlap(h, shift, n)
        
        # Nesterov step
        v_new = momentum * v + (1.0 - momentum) * g
        v = v_new
        y = h - step * v
        
        # Project
        h_new = _proj_l2(y, n * 0.5)
        c_new = _eval(h_new)
        
        # Backtracking
        taken = False
        s_local = step

            c_try = _eval(h_try)

                h = h_try
                cur_c = c_try
                step = min(s_local * 1.05, initial_step * 5.0)
                taken = True
                break
            s_local *= 0.5
        if not taken:
            h = h_new
            cur_c = c_new
            step = min(step * 1.2, initial_step * 5.0)
        
        # Decay momentum adaptively
        if it % 500 == 0:
            momentum = min(momentum * 1.01, 0.95)
    
    return _repair(h, n), cur_c

def _binary_swap_search(h, n_trials, rng, improve_threshold=1e-10):
    """Hill climbing on binary-valued h using swap moves"""
    n = len(h)
    h = (h >= 0.5).astype(float)
    h = _repair(h, n)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for _ in range(n_trials):
        # Random swap
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i], h[j] = 0.0, 1.0
        c5n = _eval(h)
        if c5n < c5c - improve_threshold:
            c5c = c5n
            # Update ones/zeros
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c = _eval(best_h)

    # Phase 1: Nesterov projected gradient (90s)
    for alpha, steps, s0, tk, mu in [
        (2000, 400, 0.01, 12, 0.7),
        (15000, 350, 0.008, 8, 0.75),
        (100000, 300, 0.005, 5, 0.82),
        (500000, 250, 0.003, 3, 0.88),
    ]:
        if _el() > 90:
            break
        hn, cn = _nesterov_pg(best_h.copy(), alpha, steps, s0, tk, mu)
        if cn < best_c:
            best_c = cn
            best_h = hn

    # Phase 2: Binary systematic sweep (350s)
    if _el() < budget_s - 350:
        h_bin = (best_h >= 0.5).astype(float)
        h_bin = _repair(h_bin, n)
        c_bin = _eval(h_bin)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()
        # Estimate sweep speed
        _t = _el()
        _ = _eval(h_bin)
        _ = _eval(h_bin)
        _ = _eval(h_bin)
        ev_time = max((_el() - _t) / 3.0, 2e-5)
        # Full sweep = (n/2)*(n/2) evals
        sweep_cost = (n // 2) * (n // 2) * ev_time
        max_passes = max(1, int(300 / sweep_cost))
        max_passes = min(max_passes, 50)
        h_bin, c_bin = _binary_sweep(h_bin, max_passes, rng)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()

    # Phase 3: Binary random search (100s)
    if _el() < budget_s - 200:
        h_b2 = (best_h >= 0.5).astype(float)
        h_b2 = _repair(h_b2, n)
        _t = _el()
        _ = _eval(h_b2)
        ev_t = max((_el() - _t), 1e-5)
        nt = min(int(90 / ev_t * 0.7), 2000000)
        h_b2, c_b2 = _binary_random(h_b2, nt, rng)
        if c_b2 < best_c:
            best_c = c_b2
            best_h = h_b2.copy()

    # Phase 4: Continuous gradient from current best (80s)
    if _el() < budget_s - 130:
        for alpha, steps, s0, tk, mu in [
            (50000, 300, 0.006, 5, 0.8),
            (500000, 250, 0.003, 3, 0.87),
            (5000000, 200, 0.001, 2, 0.92),
        ]:
            if _el() > budget_s - 130:
                break
            hn, cn = _nesterov_pg(best_h.copy(), alpha, steps, s0, tk, mu)
            if cn < best_c:
                best_c = cn
                best_h = hn

    # Phase 5: SLSQP polish (50s)
    if _el() < budget_s - 55:
        dx = 2.0 / n
        alpha_sl = 5e5
        def _obj(x):
            c = np.correlate(x, 1.0 - x, mode='full') * dx
            m = c.max()
            s = np.clip(alpha_sl * (c - m), -300, 300)
            s -= s.max()
            return m + np.log(np.exp(s).sum()) / alpha_sl
        def _grad(x):
            c = np.correlate(x, 1.0 - x, mode='full') * dx
            m = c.max()
            s = np.clip(alpha_sl * (c - m), -300, 300)
            s -= s.max()
            w = np.exp(s); w /= w.sum()
            g = np.zeros(n)
            for ii in range(len(c)):
                if w[ii] < 1e-12:
                    continue
                sh = ii - (n - 1)
                g += w[ii] * _grad_lag(x, sh, n)
            return g
        try:
            res = _scipy_min(_obj, best_h, jac=_grad, method='SLSQP',
                            bounds=[(0,1)]*n,
                            constraints=[{'type':'eq','fun':lambda x:x.sum()-n/2.0,
                                         'jac':lambda x:np.ones(n)}],
                            options={'maxiter':150,'ftol':1e-18})
            hp = _repair(res.x, n)
            cp = _eval(hp)
            if cp < best_c:
                best_c = cp
                best_h = hp
        except Exception:
            pass

    # Phase 6: Fine 2-point local search (remaining)
    if _el() < budget_s - 5:
        time_left = budget_s - 5 - _el()
        _t = _el()
        for _ in range(40):
            _eval(best_h)
        ev_t = max((_el() - _t) / 40.0, 1e-5)
        ni = min(int(time_left / ev_t * 0.7), 5000000)
        best_h, best_c = _local2(best_h, ni, rng, 5e-6)

    best_h = _repair(best_h, n)
    best_c = _eval(best_h)
    return (best_h, best_c, n)
