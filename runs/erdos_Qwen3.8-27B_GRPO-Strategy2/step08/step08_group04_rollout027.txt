        freq = rng.uniform(2, 12)
        phase = rng.uniform(0, 2*np.pi)
        amp = rng.uniform(0.01, 0.05)
        d = amp * np.sin(freq * x + phase)
    elif ptype == 1:  # Polynomial
        deg = rng.randint(2, 5)
        coeffs = rng.randn(deg + 1) * 0.02
        d = np.polyval(coeffs, x / 2.0 - 1.0)
    elif ptype == 2:  # Block perturbation
        d = np.zeros(n)
        nb = rng.randint(3, 8)
        sizes = rng.multinomial(n, np.ones(nb)/nb)
        pos = 0
        for sz in sizes:
            amp = rng.uniform(-0.05, 0.05)
            d[pos:pos+sz] = amp
            pos += sz
    else:  # Random smooth

        # Smooth it
        k = 3
        kernel = np.ones(2*k+1) / (2*k+1)
        d = np.convolve(d, kernel, mode='same')
    d -= d.mean()  # Preserve sum
    h_new = _proj_sum(h + d, n * 0.5)
    h_new = _repair(h_new, n)
    return h_new

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid

    g = np.zeros(n)

        g[:n - m] -= h[m:]
    else:
        p = -m
        g[:n - p] += 1.0 - h[p:]
        g[p:] -= h[:n - p]

def _basin_jump(h_best, rng, n, n_tries, deadline):
    best_h = h_best.copy()
    best_c5 = _eval(best_h)
    ll = len(best_h)
    for t in range(n_tries):
        if time.time() > deadline:
            break
        scale = 0.008 * (1.0 + 0.5 * rng.random())
        h_p = _perturb(best_h, rng, ll, scale)
        dl_sub = min(deadline, time.time() + 12)
        h_r = _nesterov(h_p, 8000, 12, 1200, 0.002, dl_sub)
        c5r = _eval(h_r)
        if c5r < best_c5:
            best_c5 = c5r
            best_h = h_r.copy()
    return best_h, best_c5

def _fine_descent(h, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    rng2 = np.random.RandomState(12345)
    step_sz = 5e-5
    no_improve = 0
    for it in range(300000):
        if time.time() > deadline:
            break
        i = rng2.randint(n)
        j = (i + 1 + rng2.randint(0, 30)) % n
        d = step_sz * rng2.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n; no_improve = 0
                if c5n < bc:
                    bc = c5n; bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                no_improve += 1
                if no_improve > 500:
                    step_sz *= 0.9
                    no_improve = 0
                    if step_sz < 1e-8:
                        break
    return bh, bc

def _mirror_start(h, n):
    """Create a mirrored version of h as an alternative start."""
    hm = np.zeros(n)
    for i in range(n):
        hm[i] = 1.0 - h[n - 1 - i]
    return _repair(_proj_sum(hm, n * 0.5), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP (0-70s)
    dl1 = min(deadline, t0 + 70)
    h1 = _slsqp(h.copy(), 1500, 25, 300, dl1)
    c51 = _eval(h1)
    if c51 < best_c5:
        best_c5 = c51; best_h = h1.copy()

    # Phase 2: Nesterov multi-scale (70-300s)
    dl2 = min(deadline, t0 + 300)
    h2 = best_h.copy()
    for alpha, tk, st, ns in [(2000, 20, 0.006, 4000),
                               (8000, 12, 0.003, 4000),
                               (30000, 6, 0.001, 4000),
                               (150000, 3, 0.0005, 3000)]:
        if time.time() > dl2:
            break
        h2 = _nesterov(h2, alpha, tk, ns, st, dl2)
        c5v = _eval(h2)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h2.copy()

    # Phase 3: Basin jumping with structured perturbations (300-620s)
    dl3 = min(deadline, t0 + 620)
    n_tries = int(max(1, (dl3 - time.time()) / 25))
    h3, c53 = _basin_jump(best_h.copy(), rng, n0, n_tries, dl3)
    if c53 < best_c5:
        best_c5 = c53; best_h = h3.copy()

    # Phase 4: Try mirror start + optimize (620-720s)
    dl4 = min(deadline, t0 + 720)
    hm = _mirror_start(np.array(initial_h_values, dtype=float).copy(), n0)
    hm = _nesterov(hm, 5000, 10, 2000, 0.003, dl4)
    c5m = _eval(hm)
    if c5m < best_c5:
        best_c5 = c5m; best_h = hm.copy()

    # Phase 5: More Nesterov from best (720-820s)
    dl5 = min(deadline, t0 + 820)
    h5 = _nesterov(best_h.copy(), 300000, 3, 4000, 0.0003, dl5)
    c5v = _eval(h5)
    if c5v < best_c5:
        best_c5 = c5v; best_h = h5.copy()

    # Phase 6: Second basin jump (820-950s)
    dl6 = min(deadline, t0 + 950)
    h6, c56 = _basin_jump(best_h.copy(), rng, n0, 8, dl6)
    if c56 < best_c5:
        best_c5 = c56; best_h = h6.copy()

    # Phase 7: Fine descent (950-1080s)
    dl7 = min(deadline, t0 + 1080)
    best_h, best_c5 = _fine_descent(best_h.copy(), dl7)

    # Phase 8: Final SLSQP polish
    if deadline - time.time() > 10:
        hf = _slsqp(best_h.copy(), 50000, 4, 150, deadline)
        c5f = _eval(hf)
        if c5f < best_c5:
            best_c5 = c5f; best_h = hf.copy()

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
