import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_sum(h, target):
    # Project onto {x: sum(x)=target, 0<=x<=1}
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _topk_grads(h, topk):
    """Compute gradients of top-k correlations."""
    n = len(h)
    c = _correl(h)
    idx = np.argpartition(c, -topk)[-topk:]
    idx = idx[np.argsort(-c[idx])]
    grads = []
    for k in idx:
        sh = k - (n - 1)  # shift
        g = np.zeros(n)
        j = np.arange(n)
        # dc_k/dh[i] = dx * ((1-h[i+sh]) - h[i-sh]) with bounds
        i1 = j + sh
        m1 = (i1 >= 0) & (i1 < n)
        i2 = j - sh
        m2 = (i2 >= 0) & (i2 < n)
        g[m1] += (1.0 - h[i1[m1]])
        g[m2] -= h[i2[m2]]
        grads.append(g)
    return np.array(grads), idx, c, c[idx]

def _lp_step(h, topk, eps, deadline):
    """One LP descent step."""
    n = len(h)
    if time.time() > deadline:
        return None
    grads, idx, c, cvals = _topk_grads(h, topk)
    # LP: minimize t s.t. grads[k] @ d <= t for all k, |d_i|<=eps, sum(d)=0
    # Equivalent: minimize -grads[0]@d s.t. (grads[k]-grads[0])@d <= 0 for k=1..topk-1
    n_var = n
    # c_obj = -grads[0]
    # A_ub rows: (grads[k] - grads[0]) for k=1..topk-1
    # A_eq: ones(1,n), b_eq: 0
    # bounds: (-eps, eps) for each
    if topk > 1:
        A_ub = grads[1:] - grads[0]
    else:
        A_ub = None
    c_obj = -grads[0]
    bnds = [(-eps, eps)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=np.zeros(topk-1) if topk>1 else None,
                     A_eq=np.ones((1,n)), b_eq=np.zeros(1),
                     bounds=bnds, method='highs', options={'presolve': True, 'time_limit': 2.0})
    except Exception:
        return None
    if not res.success or res.x is None:
        return None
    return res.x

def _lp_descent(h, n_iters, eps_start, eps_end, topk, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    for it in range(n_iters):
        if time.time() > deadline:
            break
        eps = eps_start * (eps_end/eps_start) ** (it / max(n_iters, 1))
        eps = max(eps, 1e-6)
        d = _lp_step(c, topk, eps, deadline)
        if d is None:
            break
        if np.max(np.abs(d)) < 1e-14:
            break
        # Line search
        improved = False
        for frac in [1.0, 0.5, 0.25, 0.125]:
            h_new = _repair(c + frac * d, n)
            c5_new = _eval(h_new)
            if c5_new < best_c5:
                best_c5, best_h = c5_new, h_new.copy()
            if c5_new <= _eval(c) + 1e-15:
                c = h_new
                improved = True
                break
        if not improved:
            # Try smaller
            h_new = _repair(c + 0.01 * d, n)
            c5_new = _eval(h_new)
            if c5_new <= _eval(c):
                c = h_new
            else:
                break
    return best_h, best_c5

def _nesterov_refine(h, alpha, topk, n_steps, step0, deadline):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    beta = 0.85
    step = step0
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        grads, idx, c, cvals = _topk_grads(h, topk)
        # Use weighted gradient (top-k with soft weights)
        s = np.clip(alpha * (cvals - cvals[0]), -100, 100)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(topk):
            g += w[ii] * grads[ii]
        v = beta * v + (1.0 - beta) * g
        h_new = _proj_sum(h - step * v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f:
            h, f = h_new, f_new
            step = min(step * 1.05, step0 * 5.0)
        else:
            step *= 0.5
            if step < 1e-12:
                break
    return _repair(h, n), _eval(h)

def _coarse_to_fine(h_coarse, n_fine, dx_fine):
    """Upsample from coarse to fine grid."""
    # h_coarse has n_coarse values over [0,2]
    n_coarse = len(h_coarse)
    x_coarse = np.linspace(0, 2, n_coarse, endpoint=False) + 1.0/n_coarse
    x_fine = np.linspace(0, 2, n_fine, endpoint=False) + 1.0/n_fine
    h_interp = np.interp(x_fine, x_coarse, np.concatenate([h_coarse, [h_coarse[-1]]]))
    return _repair(h_interp, n_fine)

def _random_init(n, seed):

    pattern = np.zeros(period)

def _structured_init(n):

    # Pattern: high, medium, low, medium repeating
    p = np.zeros(period)

    p[period//4:period//2] = 0.5

def _coordinate_polish(h, n_iter, deadline):
    """Coordinate descent on individual points."""
    n = len(h)
    c5_cur = _eval(h)
    best_h, best_c5 = h.copy(), c5_cur
    c = _correl(h)
    top_idx = np.argsort(-c)[-20:]  # top 20 shifts
    
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        improved = False
        # Try adjusting each point in the top-shifts' influence
        for _inner in range(5):
            i = np.random.randint(n)
            c_k = _correl(h)
            # Find which shift is worst
            k_worst = np.argmax(c_k)
            sh = k_worst - (n-1)
            # Try to decrease h[i] if it contributes to the worst shift
            # dc_k/dh[i] = (1-h[i+sh]) - h[i-sh]
            i1 = i + sh
            i2 = i - sh
            grad_i = 0.0
            if 0 <= i1 < n:
                grad_i += (1.0 - h[i1])
            if 0 <= i2 < n:
                grad_i -= h[i2]
            
            if grad_i > 0:  # increasing h[i] increases the worst correlation
                delta = -0.005
            elif grad_i < 0:
                delta = 0.005
            else:
                delta = 0.005 * np.random.choice([-1, 1])
            
            h_new = h.copy()
            h_new[i] = np.clip(h_new[i] + delta, 0, 1)
            # Compensate at another point
            j = (i + n // 2) % n
            h_new[j] = np.clip(h_new[j] - delta, 0, 1)
            h_new = _repair(h_new, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-18:
                h = h_new
                c5_cur = c5_new
                improved = True
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h.copy()
                break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _topk_grads(h, topk):
    n = len(h)
    c = _correl(h)
    idx = np.argpartition(c, -topk)[-topk:]
    idx = idx[np.argsort(-c[idx])]
    grads = np.zeros((topk, n))
    j = np.arange(n)
    for ii, k in enumerate(idx):
        sh = k - (n - 1)
        g = np.zeros(n)
        i1 = j + sh
        m1 = (i1 >= 0) & (i1 < n)
        i2 = j - sh
        m2 = (i2 >= 0) & (i2 < n)
        g[m1] += (1.0 - h[i1[m1]])
        g[m2] -= h[i2[m2]]
        grads[ii] = g
    return grads, idx, c

def _lp_step(h, topk, eps, deadline):
    n = len(h)
    if time.time() > deadline:
        return None
    grads, idx, c = _topk_grads(h, topk)
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    A_ub = np.zeros((topk, n_var))
    for k in range(topk):
        A_ub[k, :n] = grads[k]
        A_ub[k, n] = -1.0
    b_ub = np.zeros(topk)
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bnds = [(-eps, eps)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bnds, method='highs', options={'presolve': True})
    except Exception:
        return None
    if not res.success or res.x is None:
        return None
    if res.fun < -1e-12:
        return res.x[:n]
    # Fallback: single worst gradient
    d = -grads[0]
    d -= d.mean()
    m = np.max(np.abs(d))
    if m < 1e-15:
        return None
    d *= eps / m
    return d

def _lp_descent(h, n_iters, eps_start, eps_end, topk, deadline):

        eps = eps_start * (eps_end / eps_start) ** t_frac

        if d is None:

        if np.max(np.abs(d)) < 1e-15:

def _nesterov_refine(h, alpha, topk, n_steps, step0, deadline):

    f = _eval(h)
    v = np.zeros(n)
    beta = 0.85
    step = step0

        grads, idx, c = _topk_grads(h, topk)
        cvals = c[idx]
        s = np.clip(alpha * (cvals - cvals[0]), -80, 80)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(topk):
            wv = w[ii]
            if wv > 1e-14:
                g += wv * grads[ii]
        v = beta * v + (1.0 - beta) * g
        h_new = _proj_sum(h - step * v, n * 0.5)
        f_new = _eval(h_new)
        if f_new < f:

            step = min(step * 1.04, step0 * 6.0)
        else:
            step *= 0.5
            if step < 1e-13:

    n = len(h)
    c5_cur = _eval(h)
