import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_all(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(k, h, n):
    """Gradient of the k-th correlation value w.r.t. h"""
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _soft_max_grad(h, top_k, alpha):
    """Compute soft-max weighted gradient focusing on top lags"""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    idx = np.argsort(c)[-top_k:]
    c_a = c[idx]
    m = c_a.max()
    s = np.clip(alpha * (c_a - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] > 1e-15:
            g += w[ii] * _grad_lag(idx[ii], h, n)
    return g, c.max()

def _proj_gradient(h, alpha, n_steps, step0, top_k, momentum, t_check):

    n = len(h)
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)

def _swap_search(h, n_iter, rng, t_check):
    """Binary swap local search with simulated annealing"""
    n = len(h)
    # Make it more binary
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    # Repair to get exactly n/2 ones
    n_ones = int(h_bin.sum())
    target_ones = n // 2
    if n_ones > target_ones:
        zeros = np.where(h_bin == 0)[0]
        ones = np.where(h_bin == 1)[0]
        # Remove excess ones (largest values first among ones)
        excess = n_ones - target_ones
        for i in range(excess):
            h_bin[ones[i]] = 0.0
    elif n_ones < target_ones:
        zeros = np.where(h_bin == 0)[0]
        need = target_ones - n_ones
        for i in range(need):
            h_bin[zeros[i]] = 1.0

    best_h = h_bin.copy()
    best_c = c5c
    T = 0.01
    T_min = 1e-6
    cool = 0.999
    
    for it in range(n_iter):
        if time.time() > t_check:
            break
        # Pick a random 1 and a random 0 to swap
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        c5n = _eval(h_bin)
        delta = c5n - c5c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, T_min)):
            c5c = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = h_bin.copy()
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        T *= cool
    return best_h, best_c

def _downsample(h, factor):
    n = len(h)
    m = n // factor
    out = h[:m*factor].reshape(m, factor).mean(axis=1).copy()
    return _repair(out, m)

def _upsample(h, factor):
    n = len(h)
    out = np.empty(n * factor)
    for i in range(n):
        out[i*factor:(i+1)*factor] = h[i]
    return out

def _slsqp_top(h, alpha, maxiter, top_k):
    n = len(h)
    dx = 2.0 / n
    def obj_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        idx = np.argsort(c)[-top_k:]
        cs = c[idx]
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def grad_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        idx = np.argsort(c)[-top_k:]
        cs = c[idx]
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] > 1e-15:
                g += w[ii] * _grad_lag(idx[ii], x, n)
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_fn, h, jac=grad_fn,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h
