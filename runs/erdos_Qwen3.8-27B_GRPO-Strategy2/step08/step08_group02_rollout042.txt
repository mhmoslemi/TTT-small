import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_worst(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max()), int(np.argmax(c))

def _proj_simplex(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(65):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(k, h, n):
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _proj_grad(h, alpha, steps, step0, top_k, momentum, tlim):

        if time.time() - t_s > tlim:
            break

        if top_k > 0 and top_k < len(c):
            idx = np.argsort(c)[-top_k:]
            cs = c[idx]
        else:
            idx = np.arange(len(c))
            cs = c
        m = cs.max()
        s = np.clip(alpha * (cs - m), -500, 500)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] < 1e-14:
                continue
            g += w[ii] * _grad_lag(int(idx[ii]), h, n)
        v = momentum * v + (1.0 - momentum) * g
        taken = False

def _downsample(h, factor):
    n = len(h)
    m = n // factor
    return h[:m * factor].reshape(m, factor).mean(axis=1).copy()

def _upsample(h, factor):
    n = len(h)
    out = np.empty(n * factor)
    for i in range(n):
        out[i * factor:(i + 1) * factor] = h[i]
    return out
