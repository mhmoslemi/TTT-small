def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:

            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lse_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lse_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    dx2 = 2.0 / n
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx2
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n - sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _slsqp(h, alpha, maxiter, ftol, target, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

    def obj(x):
        return _lse_obj(x, alpha, n)
    def grad(x):
        return _lse_grad(x, alpha, n)
    def cfun(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol,
                                'disp': False})
        if time.time() > t_end:
            return h
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

    f = _eval(h)
    v = np.zeros(n)

    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.6
    step = step0
    for _s in range(steps):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        top_k = np.argsort(c)[-k_top:]
        g = np.zeros(n)
        wk = (2.0 / n) / k_top
        for k in top_k:
            sh = (n - 1) - k
            if sh >= 0:
                g[:n - sh] += wk * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= wk * h[:n - sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - h[:n - sp])
                g[:n - sp] -= wk * h[sp:]
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _repair(h - step * v, n)
        f_new = _eval(h_new)
        if f_new < f:
            h, f = h_new, f_new
            step = min(step * 1.01, step0 * 5.0)
        else:
            step *= 0.5
            if step < 1e-13:
                v[:] = 0
                step = step0 * 0.1
    return _repair(h, n)

def _fine_2pt(h, rng, n_tries, scale, t_end):
    n = len(h)
    c5c = _eval(h)
    best_h, best_c = h.copy(), c5c
    offsets = [1, 2, 3, 5, 7, 11, n // 2, n // 3]
    for _it in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        off = offsets[rng.randint(len(offsets))]
        j = (i + off) % n
        if j == i:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c, best_h = c5n, h.copy()
            else:
                h[i] -= d
                h[j] += d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end_global = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    target = n / 2.0

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def try_update(h_new):
        nonlocal best_h, best_c5
        c5v = _eval(h_new)
        if c5v < best_c5:
            best_c5, best_h = c5v, h_new.copy()
        return c5v

    # Phase 1: SLSQP low-alpha (broad basin)
    t_ph1 = t0 + budget_s * 0.12
    if time.time() < t_ph1 - 5:
        try_update(_slsqp(best_h.copy(), 2000, 1500, 1e-14, target, t_ph1))

    # Phase 2: Nesterov top-k=80
    t_ph2 = t0 + budget_s * 0.25
    if time.time() < t_ph2 - 5:
        try_update(_nesterov_topk(best_h.copy(), 80, 3000, 0.003, t_ph2))

    # Phase 3: SLSQP medium-alpha
    t_ph3 = t0 + budget_s * 0.35
    if time.time() < t_ph3 - 10:
        try_update(_slsqp(best_h.copy(), 50000, 1000, 1e-15, target, t_ph3))

    # Phase 4: Nesterov top-k=20
    t_ph4 = t0 + budget_s * 0.50
    if time.time() < t_ph4 - 5:
        try_update(_nesterov_topk(best_h.copy(), 20, 3000, 0.0012, t_ph4))

    # Phase 5: SLSQP high-alpha
    t_ph5 = t0 + budget_s * 0.60
    if time.time() < t_ph5 - 10:
        try_update(_slsqp(best_h.copy(), 500000, 800, 1e-16, target, t_ph5))

    # Phase 6: Micro-Nesterov top-k=5
    t_ph6 = t0 + budget_s * 0.70
    if time.time() < t_ph6 - 5:
        try_update(_micro_nesterov(best_h.copy(), 5, 4000, 0.0004, t_ph6))

    # Phase 7: Coordinate descent (main polishing)
    t_ph7 = t0 + budget_s * 0.88
    if time.time() < t_ph7 - 10:
        n_sweeps = max(10, int((t_ph7 - 10 - time.time()) / 0.3))
        best_h, best_c5 = _coord_descent(best_h, 5, n_sweeps, t_ph7)

    # Phase 8: Micro-Nesterov top-k=3
    t_ph8 = t0 + budget_s * 0.93
    if time.time() < t_ph8 - 5:
        try_update(_micro_nesterov(best_h.copy(), 3, 3000, 0.0002, t_ph8))

    # Phase 9: Fine 2-point random
    t_ph9 = t0 + budget_s * 0.97
    if time.time() < t_ph9 - 5:
        ntr = min(int((t_ph9 - 5 - time.time()) / 0.003), 50000)
        best_h, best_c5 = _fine_2pt(best_h, rng, ntr, 2e-5, t_ph9)

    # Phase 10: Ultra-fine coordinate descent
    if time.time() < t_end_global - 3:
        n_sweeps = max(5, int((t_end_global - 3 - time.time()) / 0.2))
        best_h, best_c5 = _coord_descent(best_h, 3, n_sweeps, t_end_global - 1)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
