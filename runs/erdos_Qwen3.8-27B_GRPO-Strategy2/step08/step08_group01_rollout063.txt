Now let me think about the optimization strategy more carefully.

The key insight: at the optimum, there should be multiple "active" lags where the correlation equals the maximum. The subgradient of the max at a point where multiple lags are tied is the convex hull of their individual gradients. For descent, we want to move in a direction that reduces all active correlations.

A practical approach:
1. Compute all correlation values
2. Find the top m values (those within some tolerance of max)
3. Compute a weighted average gradient (weights based on how close to max)
4. Take a step in the negative gradient direction
5. Project back to feasible set

This is essentially what the previous code does with LSE, but let me make it more precise.

For the fine-tuning phase, I'll use exact 2-coordinate search:
- Pick a pair (i, j)
- Compute the current correlations
- For a small change δ (h[i]+=δ, h[j]-=δ), compute the new max correlation
- Find the optimal δ via line search

The change in corr[k] for h[i]+=δ, h[j]-=δ is:
Δcorr[k] = δ * (∂corr[k]/∂h[i] - ∂corr[k]/∂h[j])

So the new corr[k] = old_corr[k] + δ * d_k where d_k = ∂corr[k]/∂h[i] - ∂corr[k]/∂h[j]

The new max is max_k (old_corr[k] + δ * d_k)

To minimize this, we want the largest k to decrease, or at least the max to decrease.

For the active set (k where old_corr[k] ≈ max), we want δ * d_k < 0 for all active k.

The optimal δ (for small changes) is bounded by:
- 0 ≤ h[i]+δ ≤ 1 → -h[i] ≤ δ ≤ 1-h[i]
- 0 ≤ h[j]-δ ≤ 1 → h[j]-1 ≤ δ ≤ h[j]
- For each active k: old_corr[k] + δ*d_k ≤ target

The line search is a 1D problem: minimize max_k (c_k + δ*d_k) over the feasible δ range.

This is a piecewise linear function in δ, and the minimum is at one of the breakpoints.

This gives an exact (or near-exact) 2-coordinate search step.

            lo = mid
        else:
            hi = mid

    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_arr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_k(h, k, n):
    lag = k - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[lag:] += (1.0 - h[:m])
            g[:m] -= h[lag:]
    else:
        l = -lag
        m = n - l
        if m > 0:
            g[:m] += (1.0 - h[l:])
            g[l:] -= h[:m]
    return g

def _grad_smooth(h, alpha, top_k):
    n = len(h)
    c = _corr_arr(h)
    idx = np.argsort(c)[-top_k:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _grad_k(h, k, n)
    return g

def _mgd(h, n_steps, alpha, top_k, step0, rng):
    n = len(h)
    v = np.zeros(n)
    y = h.copy()
    mu = 0.7
    step = step0
    cur_c = _eval(h)
    best_c, best_h = cur_c, h.copy()
    for _ in range(n_steps):
        g = _grad_smooth(h, alpha, top_k)
        y = h + step * g
        v = mu * v + (1.0 - mu) * (y - h)
        ht = _repair(y + mu * v, n)
        ct = _eval(ht)
        if ct < cur_c:
            h, cur_c = ht, ct
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
            step = min(step * 1.03, step0 * 5.0)
        else:
            step = max(step * 0.65, step0 * 0.05)
    return best_h, best_c

def _local2(h, n_iter, step, rng, sa_t=None):
    n = len(h)
    cur_c = _eval(h)
    best_c, best_h = cur_c, h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 7)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            acc = cn < cur_c
            if not acc and sa_t is not None and sa_t > 1e-12:
                if rng.rand() < np.exp(-(cn - cur_c) / sa_t):
                    acc = True
            if acc:
                cur_c = cn
                if cn < best_c:
                    best_c, best_h = cn, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _local3(h, n_iter, step, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c, best_h = cur_c, h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        k2 = (i + 5 + rng.randint(0, 9)) % n
        if i == j or i == k2 or j == k2:
            continue
        d1, d2 = step * rng.randn(), step * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k2] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            s0, s1, s2 = h[i], h[j], h[k2]
            h[i], h[j], h[k2] = ni, nj, nk
            cn = _eval(h)
            if cn < cur_c:
                cur_c = cn
                if cn < best_c:
                    best_c, best_h = cn, h.copy()
            else:
                h[i], h[j], h[k2] = s0, s1, s2
    return best_h, best_c

def _grad_search(h, n_iter, alpha, top_k, step, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c, best_h = cur_c, h.copy()
    for _ in range(n_iter):
        g = _grad_smooth(h, alpha, top_k)
        gp = g + step * 0.05 * rng.randn(n)
        i = np.argmin(gp)
        j = np.argmax(gp)
        if i == j:
            j = (j + 1) % n
        d = step * (0.5 + 0.5 * rng.rand())
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < cur_c:
                cur_c = cn
                if cn < best_c:
                    best_c, best_h = cn, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _sweep_all(h, step, n_passes, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c, best_h = cur_c, h.copy()
    for _ in range(n_passes):
        for i in range(n):
            for off in [1, 2, 3, n // 2 - 1]:
                j = (i + off) % n
                d = step * rng.randn()
                ni, nj = h[i] + d, h[j] - d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i], h[j] = ni, nj
                    cn = _eval(h)
                    if cn < cur_c:
                        cur_c = cn
                        if cn < best_c:
                            best_c, best_h = cn, h.copy()
                    else:
                        h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: MGD with multiple settings
    mgd_cfgs = [(1500, 2000, 8, 0.003), (5000, 1500, 6, 0.002), (15000, 1000, 4, 0.001)]
    for alpha, ns, tk, s0 in mgd_cfgs:
        if _el() > 200:
            break
        ht, c = _mgd(best_h.copy(), ns, alpha, tk, s0, rng)
        c = _eval(ht)
        if c < best_c5:
            best_c5, best_h = c, ht.copy()

    # Phase 2: Perturbation + MGD
    if _el() < budget_s - 500:
        for trial in range(3):
            if _el() > budget_s - 400:
                break
            noise = 0.012 * (trial + 1) * rng.randn(n)
            noise -= noise.mean()
            hp = _repair(best_h + noise, n)
            ht, c = _mgd(hp, 400, 3000, 6, 0.003, rng)
            c = _eval(ht)
            if c < best_c5:
                best_c5, best_h = c, ht.copy()

    # Phase 3: Fine MGD
    if _el() < budget_s - 400:
        for alpha, tk, s0, ns in [(5000, 4, 6e-4, 600), (20000, 3, 3e-4, 500), (80000, 2, 1e-4, 400)]:
            if _el() > budget_s - 350:
                break
            ht, c = _mgd(best_h.copy(), ns, alpha, tk, s0, rng)
            c = _eval(ht)
            if c < best_c5:
                best_c5, best_h = c, ht.copy()

    # Phase 4: Gradient-informed 2-point search (main phase)
    if _el() < budget_s - 350:
        n_iter = min(int((budget_s - 350 - _el()) * 1200), 300000)
        best_h, best_c5 = _grad_search(best_h, n_iter, 5000, 5, 3e-4, rng)

    # Phase 5: Random 2-point search (coarser)
    if _el() < budget_s - 250:
        n_iter = min(int((budget_s - 250 - _el()) * 1500), 200000)
        best_h, best_c5 = _local2(best_h, n_iter, 5e-5, rng)

    # Phase 6: 3-point search
    if _el() < budget_s - 200:
        n_iter = min(int((budget_s - 200 - _el()) * 800), 80000)
        best_h, best_c5 = _local3(best_h, n_iter, 3e-5, rng)

    # Phase 7: SA 2-point (escape local minima)
    if _el() < budget_s - 150:
        sa_t = max(1e-5, best_c5 * 8e-5)
        n_iter = min(int((budget_s - 150 - _el()) * 1500), 150000)
        best_h, best_c5 = _local2(best_h, n_iter, 3e-5, rng, sa_t)

    # Phase 8: Fine gradient search
    if _el() < budget_s - 100:
        n_iter = min(int((budget_s - 100 - _el()) * 2000), 200000)
        best_h, best_c5 = _grad_search(best_h, n_iter, 10000, 4, 1e-4, rng)

    # Phase 9: Ultra-fine 2-point polish
    if _el() < budget_s - 60:
        n_iter = min(int((budget_s - 60 - _el()) * 3500), 400000)
        best_h, best_c5 = _local2(best_h, n_iter, 8e-6, rng)

    # Phase 10: Final MGD + micro polish
    if _el() < budget_s - 20:
        ht, _ = _mgd(best_h.copy(), 150, 100000, 3, 5e-5, rng)
        ct = _eval(ht)
        if ct < best_c5:
            best_h, best_c5 = ht.copy(), ct
        n_fin = min(int((budget_s - 20 - _el()) * 5000), 300000)
        best_h, best_c5 = _local2(best_h, n_fin, 3e-6, rng)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
