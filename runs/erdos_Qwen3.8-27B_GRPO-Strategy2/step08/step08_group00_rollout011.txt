def _pm_grad(x, p):

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    K = len(c)
    pm = (np.mean(c**p)) ** (1.0/p)
    # d(pm)/d(x_j) = pm^(1-p) * mean(c^(p-1) * dc/dx_j)
    # = pm^(1-p) / K * sum_k c_k^(p-1) * dc_k/dx_j
    scale = pm**(1.0 - p) / K
    w = c**(p-1)  # weight for each lag
    g = np.zeros(n)
    for d in range(K):
        if w[d] < 1e-18:
            continue
        wd = w[d] * scale
        L = d - (n - 1)
        if L >= 0:
            g[:n-L] += wd * (1.0 - x[L:])
            g[L:] -= wd * x[:n-L]
        else:
            M = -L
            g[M:] += wd * (1.0 - x[:n-M])
            g[:n-M] -= wd * x[M:]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        if s < 1e-12:
            return np.full(n, 0.5)
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a

                break
