import numpy as np
from scipy.optimize import minimize

_slsqp_params = {}

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _pair_coef(h, i, j, ks):
    n = len(h)
    D = np.zeros(2 * n - 1)
    v1 = (i - ks >= 0) & (i - ks < n)
    v2 = (i + ks >= 0) & (i + ks < n)
    D[v1] += 1.0 - h[i - ks[v1]]
    D[v2] -= h[i + ks[v2]]
    v1j = (j - ks >= 0) & (j - ks < n)
    v2j = (j + ks >= 0) & (j + ks < n)
    D[v1j] -= 1.0 - h[j - ks[v1j]]
    D[v2j] += h[j + ks[v2j]]
    return D

def _line_search(c_vec, D, lo, hi, nit=35):
    for _ in range(nit):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(c_vec + m1 * D) < np.max(c_vec + m2 * D):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def _coord_descent(h, n_iter, rng):
    n = len(h)
    ks = np.arange(-(n - 1), n)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(n - 1)) % n
        c_vec = _eval_vec(h)
        D = _pair_coef(h, i, j, ks)
        lo = max(-h[i], h[j] - 1.0) + 1e-12
        hi = min(1.0 - h[i], h[j]) - 1e-12
        if lo >= hi:
            continue
        delta = _line_search(c_vec, D, lo, hi)
        h[i] += delta
        h[j] -= delta
        c_new = _eval(h)
        if c_new < cur_c:
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta
    return best_h, best_c

def _mgd_grad(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):

            continue
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:n])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:n] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:n]
    return g

def _proj_target(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _mgd(h, n_iter, alpha, top_k, step0, rng):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    mu = 0.75
    step = step0
    cur_c = _eval(h)
    for _ in range(n_iter):
        g = _mgd_grad(h, alpha, top_k)
        y = h + step * g
        v = mu * v + (1.0 - mu) * (y - h)
        h_try = _proj_target(y + mu * v, target)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try
            cur_c = c_try
            step = min(step * 1.04, step0 * 4.0)
        else:
            step = max(step * 0.6, step0 * 0.1)
    return _repair(h, n), cur_c

def _slsqp_obj(x):
    alpha, top_k = _slsqp_params['alpha'], _slsqp_params['top_k']
    c = _eval_vec(x)
    idx = np.argsort(c)[-top_k:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x):
    alpha, top_k = _slsqp_params['alpha'], _slsqp_params['top_k']
    return _mgd_grad(x, alpha, top_k)

def _slsqp_cfun(x):
    return x.sum() - _slsqp_params['target']

def _slsqp_cjac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, top_k):
    n = len(h)
    _slsqp_params['alpha'] = alpha
    _slsqp_params['top_k'] = top_k
    _slsqp_params['target'] = n * 0.5
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cfun, 'jac': _slsqp_cjac}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _local2(h, n_iter, step, rng):

    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):

        ni, nj = h[i] + d, h[j] - d

                    best_h = h.copy()

    return best_h, best_c
