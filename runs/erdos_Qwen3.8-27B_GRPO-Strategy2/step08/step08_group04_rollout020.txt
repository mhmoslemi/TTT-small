    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:

            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a

                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _pmean_obj(h, p, topk):
    c = _correl(h)
    if topk < len(c):
        ct = c[np.argpartition(c, -topk)[-topk:]]
    else:
        ct = c
    return (np.mean(np.clip(ct, 1e-30, None) ** p)) ** (1.0 / p)

def _pmean_grad(h, p, topk):
    c = _correl(h)
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    n = len(h); dx = 2.0 / n
    ct = np.clip(ct, 1e-30, None)
    m = (np.mean(ct ** p)) ** (1.0 / p)
    g = np.zeros(n)
    for ii in range(len(idx)):
        w = (ct[ii] / m) ** (p - 1) / len(ct)
        g += w * _grad_lag(h, idx[ii], dx)
    return g

def _nesterov_pm(h, p0, pmax, topk, steps, deadline):
    n = len(h)
    f = _pmean_obj(h, p0, topk)
    v = np.zeros(n); beta = 0.82; step = 0.003; p = p0
    bh = h.copy(); bf = _eval(h)
    stall = 0
    for it in range(steps):
        if time.time() > deadline:
            break
        if it % 400 == 0 and it > 0:
            p = min(p * 1.6, pmax)
        g = _pmean_grad(h, p, topk)
        v = beta * v + (1.0 - beta) * g
        accepted = False
        for _ in range(25):
            hn = _proj(h - step * v, n * 0.5)
            fn = _pmean_obj(hn, p, topk)
            if fn < f:
                h = hn; f = fn; accepted = True; break
            step *= 0.5
        if not accepted:
            step *= 0.5
        c5v = _eval(h)
        if c5v < bf:
            bf = c5v; bh = h.copy(); stall = 0
            step = min(step * 1.04, 0.04)
        else:
            stall += 1
            if stall > 500:
                v *= 0.1
                stall = 0
    return _repair(bh, n), bf

def _slsqp_phase(h, p, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h, _eval(h)
    def obj(x): return _pmean_obj(x, p, topk)
    def jac(x): return _pmean_grad(x, p, topk)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18})
        hr = _repair(r.x, n)
        return hr, _eval(hr)
    except Exception:
        return h, _eval(h)

def _anneal(h, niter, rng, deadline, T0, T1):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = int(rng.choice([2, 3, 5, 7, 10, 15]))
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.003 * max(T, 1e-9)**0.5 * rng.randn(bs)
        d -= d.mean()
        hn = h.copy()
        hn[i0:i0+bs] = np.clip(seg + d, 0, 1)
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn; c5c = c5n
            if c5n < bc:
                bc = c5n; bh = h.copy()
    return bh, bc

def _grad_polish(h, niter, deadline):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    step = 5e-7
    for it in range(niter):
        if time.time() > deadline:
            break
        c = _correl(h)
        k_max = int(np.argmax(c))
        g = _grad_lag(h, k_max, 2.0 / n)
        g -= np.mean(g)
        hn = _proj(h - step * g, n * 0.5)
        c5n = _eval(hn)
        if c5n < c5c - 1e-20:
            h = hn; c5c = c5n
            if c5n < bc:
                bc = c5n; bh = h.copy()
            step = min(step * 1.08, 5e-6)
        else:
            step *= 0.6
            if step < 1e-13:
                break
    return _repair(bh, n), bc

def _fine_polish(h, niter, rng, deadline):
    nc = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    sb = 4e-7
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 6)) % nc
        d = sb * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc = c5n; bh = h.copy()
                sb = min(sb * 1.02, 2e-6)
            else:
                h[i], h[j] = ni - d, nj + d
                sb = max(sb * 0.97, 1e-9)
    return bh, bc
