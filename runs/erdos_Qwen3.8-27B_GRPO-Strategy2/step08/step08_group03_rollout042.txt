    if n > 1:
        e += np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(e) / alpha)

def _grad_from_weights(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)[n-1:2*n-1].real
        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)[:n].real
        q = -(R + L)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _smooth_grad(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0:
            tot, e0 = 1.0, 1.0
        wP[0] = e0 / tot
        wP[1:] = ep / tot
        wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _grad_from_weights(h, wP, wN)

def _fista(h0, alpha, max_time, L=15.0):
    if max_time <= 0.0:
        return h0.copy(), h0.copy(), _c5_direct(h0)
    n = len(h0)
    S = n * 0.5
    t0 = time.time()
    h = _repair(h0, n)
    hp = h.copy()
    y = h.copy()
    t = 1.0
    fy = fh = _smooth_value(h, alpha)
    best_h = h.copy()
    best_c = _c5_direct(h)
    stale = 0
    Lcur = L
    for it in range(200000):
        if time.time() - t0 > max_time:
            break
        if it and it % 4 == 0:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp)
            t = tn
            if y.min() < -1e-5 or y.max() > 1.0 + 1e-5:
                y = _proj(y, S)
            fy = _smooth_value(y, alpha)
        elif it:
            y = h
            fy = fh
        g = _smooth_grad(y, alpha)
        step = 1.0 / max(Lcur, 1e-12)
        accepted = False
        for _ in range(12):
            xn = _proj(y - step * g, S)
            fx = _smooth_value(xn, alpha)
            d = xn - y
            if fx <= fy + np.dot(g, d) + 0.5 / step * np.dot(d, d) + 1e-10:
                accepted = True
                break
            step *= 0.5
        Lcur = max(1.0 / max(step, 1e-12) * 0.82, 1.0)
        if accepted:
            hp = h
            h = xn
            fh = fx
            stale = 0
            if it % 10 == 0:
                c = _c5_direct(h)
                if c < best_c:
                    best_c, best_h = c, h.copy()
        else:
            stale += 1
            hp, y, t, fy = h, h, 1.0, fh
            if stale > 15:
                break
    c = _c5_direct(h)
    if c < best_c:
        best_c, best_h = c, h.copy()
    return h, best_h, best_c

def _active_grad(h):
    n = len(h)
    P, Nv, mx = _pn_arrays(h)
    dx = 2.0 / n
    tol = max(1e-9, 6.0 * dx)
    wP = np.zeros(n)
    wN = np.zeros(n)
    cnt = 0
    if P[0] >= mx - tol:
        wP[0] = 1.0; cnt += 1
    if n > 1:
        mP = P[1:] >= mx - tol
        if np.any(mP):
            wP[1:][mP] = 1.0; cnt += int(mP.sum())
        mN = Nv[1:] >= mx - tol
        if np.any(mN):
            wN[1:][mN] = 1.0; cnt += int(mN.sum())
    if cnt == 0:
        return np.zeros(n)
    wP *= 1.0 / cnt
    wN *= 1.0 / cnt
    return _grad_from_weights(h, wP, wN)

def _local_search(h0, rng, deadline, start_step=4e-5, bsz=3):
    tl0 = time.time()
    n = len(h0)
    h = h0.copy()
    c = _c5_direct(h)
    best_c = c
    best_h = h.copy()
    g = _active_grad(h)
    it = 0
    recomp = 800
    while time.time() < deadline:
        if it >= recomp:
            c = _c5_direct(h)
            if c < best_c - 1e-15:
                best_c, best_h = c, h.copy()
            g = _active_grad(h)
            recomp = it + 800
        frac = min(1.0, (time.time() - tl0) / max(1e-9, deadline - tl0))
        base = max(1e-13, start_step * (8e-6) ** frac)
        if rng.random() < 0.65:
            i, j = int(rng.randint(n)), int(rng.randint(n))
            if i == j:
                continue
            s = 1 if g[i] <= g[j] else -1
            avail = min(1.0 - h[i], h[j]) if s > 0 else min(h[i], 1.0 - h[j])
            if avail > 1e-14:
                d = s * min(avail, base * (0.3 + 1.7 * rng.random()))
                h[i] += d; h[j] -= d
                cn = _c5_direct(h)
                if cn < c - 1e-15:
                    c = cn
                    if c < best_c:
                        best_c, best_h = c, h.copy()
                else:
                    h[i] -= d; h[j] += d
        else:
            b = min(bsz, max(2, n // 20))
            i = int(rng.randint(0, max(1, n - b)))
            j = int(rng.randint(0, max(1, n - b)))
            if abs(i - j) < 2 * b:
                continue
            gi, gj = g[i:i+b].mean(), g[j:j+b].mean()
            s = 1 if gi <= gj else -1
            if s > 0:
                av = min((1.0 - h[i:i+b]).min(), h[j:j+b].min())
            else:
                av = min(h[i:i+b].min(), (1.0 - h[j:j+b]).min())
            if av > 1e-14:
                d = s * min(av, base * 0.4 * (0.3 + 1.7 * rng.random()))
                h[i:i+b] += d
                h[j:j+b] -= d
                np.clip(h, 0, 1, out=h)
                cn = _c5_direct(h)
                if cn < c - 1e-15:
                    c = cn
                    if c < best_c:
                        best_c, best_h = c, h.copy()
                else:
                    h[i:i+b] -= d
                    h[j:j+b] += d
        it += 1
    c = _c5_direct(h)
    if c < best_c:
        best_c, best_h = c, h.copy()
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    return _proj(np.interp(x1, x0, h), m * 0.5)

def _run_alphas(h0, alphas, total_time, deadline):
    hcur = h0.copy()
    best_h = h0.copy()
    best_c = _c5_direct(best_h)
    total_time = min(total_time, max(0.0, deadline - time.time()))
    per = total_time / max(1, len(alphas))
    for a in alphas:
        if deadline - time.time() < 2.0 or per <= 0.0:
            break
        hf, bh, bc = _fista(hcur, a, per)
        chf = _c5_direct(hf)
        chb = _c5_direct(bh)
        if bc < best_c:
            best_c, best_h = bc, bh
        if chf < best_c:
            best_c, best_h = chf, hf
        hcur = hf if chf <= chb else bh
    return best_h, best_c, hcur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)
    best = h0.copy()
    best_c = _c5_direct(best)
    for hh in (0.5 + 0.5*(h0 - h0[::-1]), 0.5 + 0.5*(h0 + h0[::-1] - 1.0)):
        c = _c5_direct(hh)
        if c < best_c:
            best, best_c = hh.copy(), c
    end1 = t0 + min(70.0, 0.07 * budget_s)
    if end1 > time.time():
        b, c, _ = _run_alphas(best, [50, 200, 1000, 5000, 30000], end1 - time.time(), end1)
        if c < best_c:
            best, best_c = b, c
    end2 = t0 + min(350.0, 0.40 * budget_s)
    if end2 > time.time():
        for f in (2, 3, 4):
            if time.time() >= end2 - 4.0:
                break
            m = n0 // f
            if m < 32:
                continue
            hm = _resample(best, m)
            slot = min(35.0, (end2 - time.time()) * 0.35)
            if slot < 4.0:
                break
            b, c, _ = _run_alphas(hm, [50, 400, 3000, 20000], slot, time.time() + slot)
            if c < best_c:
                best, best_c = b, c
            if len(best) != n0:
                hu = _resample(best, n0)
                cu = _c5_direct(hu)
                if cu < best_c:
                    best, best_c = hu.copy(), cu
    end3 = t0 + budget_s - 350.0
    if end3 > time.time() + 10.0:
        hmain = best if len(best) == n0 else _resample(best, n0)
        b, c, _ = _run_alphas(hmain, [2000, 10000, 60000, 400000], end3 - time.time(), end3)
        if c < best_c:
            best, best_c = b, c
    end4 = t0 + budget_s - 6.0
    if end4 > time.time():
        best, best_c = _local_search(best, rng, end4, 4e-5, 3)
    end5 = t0 + budget_s - 0.5
    if end5 > time.time() + 3.0:
        h2 = _proj(best + rng.normal(0, 5e-7, len(best)), len(best) * 0.5)
        b2, c2 = _local_search(h2, rng, end5, 5e-6, 2)
        if c2 < best_c:
            best, best_c = b2, c2
    for hh in (1.0 - best, best[::-1]):
        h2 = _proj(hh, len(hh) * 0.5)
        c = _c5_direct(h2)
        if c < best_c:
            best, best_c = h2, c
    best = _repair(best, len(best))
    best_c = _c5_direct(best)
    return best, best_c, len(best)
