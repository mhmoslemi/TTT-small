from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_c(h, k_idx, n):
    g = np.zeros(n)
    sh = k_idx - (n - 1)
    if sh >= 0:
        g[:n-sh] += (1.0 - h[sh:])
        if sh > 0: g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n-sp] -= h[sp:]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax_grad(h, alpha, topk, n):
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:] if topk < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13: continue
        k, wk = idx[ii], w[ii] * dx
        g += wk * _grad_c(h, k, n)
    return g

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:] if topk < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _pgd(h, alpha, topk, n, nsteps, step0, deadline, pressure=0.0):
    """Projected gradient descent with optional binary pressure."""
    target = n * 0.5
    c5c = _eval(h)
    best_h, best_c5 = h.copy(), c5c
    step = step0
    for _ in range(nsteps):
        if time.time() > deadline: break
        g = _smax_grad(h, alpha, topk, n)
        if pressure > 0:
            g -= pressure * (h - 0.5) * h * (1 - h)
        h_new = _proj(h - step * g, target)
        c5_new = _eval(h_new)
        if c5_new < c5c:
            h, c5c = h_new, c5_new
            if c5c < best_c5:
                best_c5, best_h = c5c, h.copy()
            step = min(step * 1.05, step0 * 10)
        else:
            step *= 0.5
            if step < 1e-12: break

def _lp_step(h, m_top, eps, deadline):
    """LP-based descent on top m_top shifts."""
    n = len(h)
    c = _correl(h)
    idx_top = np.argpartition(c, -m_top)[-m_top:]
    idx_top = idx_top[np.argsort(-c[idx_top])]
    grads = np.array([_grad_c(h, k, n) for k in idx_top])
    c_obj = -grads[0]
    A_ub = grads[1:] if m_top > 1 else None
    b_ub = np.zeros(m_top - 1) if m_top > 1 else None
    A_eq = np.ones((1, n))
    b_eq = np.zeros(1)
    bnds = [(-eps, eps)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bnds, method='highs', options={'presolve': True})
        if res.success and res.x is not None and np.max(np.abs(res.x)) > 1e-14:
            return res.x
    except Exception:
        pass
    return None

    c5_cur = best_c5

import numpy as np
from scipy.optimize import linprog

_N = 0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_c(h, k_idx, n):
    g = np.zeros(n)
    sh = k_idx - (n - 1)
    if sh >= 0:
        g[:n-sh] += (1.0 - h[sh:])
        if sh > 0: g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n-sp] -= h[sp:]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax_hess_free(h, alpha, topk, n):
    """Compute soft-max smooth approximation of max of top-k correlations."""
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    logsum = sm + np.log(np.sum(np.exp(s - sm)))
    return m + logsum / alpha

def _smax_grad(h, alpha, topk, n):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        g += w[ii] * dx * _grad_c(h, idx[ii], n)
    return g

def _pgd(h, alpha, topk, n, nsteps, step0, deadline, pressure=0.0):

import numpy as np
from scipy.optimize import linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_c(h, k_idx, n):
    g = np.zeros(n)
    sh = k_idx - (n - 1)
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax_grad(h, alpha, topk, n):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        g += w[ii] * dx * _grad_c(h, idx[ii], n)
    return g

def _pgd(h, alpha, topk, n, nsteps, step0, deadline, pressure=0.0):
    target = n * 0.5
    c5c = _eval(h)
    best_h, best_c5 = h.copy(), c5c
    step = step0
    for _ in range(nsteps):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk, n)
        if pressure > 0:
            g -= pressure * (h - 0.5) * h * (1.0 - h)
        h_new = _proj(h - step * g, target)
        c5_new = _eval(h_new)
        if c5_new < c5c - 1e-16:
            h, c5c = h_new, c5_new
            if c5c < best_c5:
                best_c5, best_h = c5c, h.copy()
            step = min(step * 1.04, step0 * 20.0)
        else:
            step *= 0.5
            if step < 1e-13:
                break
    return _repair(best_h, n)

def _lp_descent(h, n_steps, eps, m_top, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c5_cur = best_c5
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        c = _correl(h)
        idx_top = np.argpartition(c, -m_top)[-m_top:]
        idx_top = idx_top[np.argsort(-c[idx_top])]
        grads = np.array([_grad_c(h, k, n) for k in idx_top])
        c_obj = -grads[0]
        A_ub = grads[1:] if m_top > 1 else None
        b_ub = np.zeros(m_top - 1) if m_top > 1 else None
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bnds = [(-eps, eps)] * n
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success or res.x is None or np.max(np.abs(res.x)) < 1e-14:
            break
        d = res.x
        stp = 1.0
        improved = False
