def _nm_fourier(f0, max_time):
    """Nelder-Mead on Fourier coefficients of f"""
    n = len(f0)
    K = 25  # frequency
    Np = 2*K  # number of parameters (K sine + K cosine)
    
    # Initialize from f0
    x = np.arange(n) * (2.0/n)  # positions in [0, 2]
    # f(x) ~= sum_{k=1}^{K} a_k*sin(k*pi*x) + b_k*cos(k*pi*x)
    # (these have zero mean automatically)
    basis = np.zeros((Np, n))
    for k in range(1, K+1):
        basis[2*(k-1)] = np.sin(k*np.pi*x)
        basis[2*(k-1)+1] = np.cos(k*np.pi*x)
    
    # Least squares fit
    coeffs, _, _, _ = np.linalg.lstsq(basis, f0, rcond=None)
    
    def _obj(c):
        f = basis @ c
        f = np.clip(f, -1.0, 1.0)
        # Fix mean
        f -= f.mean()
        # Scale to ensure max |f| <= 1 after mean fix (approximately)
        m = np.max(np.abs(f))
        if m > 1.0:
            f /= m
        return 0.5 - float(np.min(np.correlate(f, f, mode='full')*(2.0/n)))/4.0
    
    # Nelder-Mead
    x0 = coeffs.copy()
    simplex = [x0.copy()]
    for i in range(Np):
        xi = x0.copy()
        xi[i] += 0.01
        simplex.append(xi)
    simplex = np.array(simplex)
    fvals = np.array([_obj(s) for s in simplex])
    
    tl0 = time.time()
    alpha, gam, rho, sig = 1.0, 2.0, 0.5, 0.5
    
    while time.time() - tl0 < max_time:
        idx = np.argsort(fvals)
        simplex = simplex[idx]
        fvals = fvals[idx]
        
        centroid = simplex[:-1].mean(axis=0)
        
        # Reflect
        xr = centroid + alpha*(centroid - simplex[-1])
        fr = _obj(xr)
        if fr < fvals[0]:
            # Expand
            xe = centroid + gam*(centroid - simplex[-1])
            fe = _obj(xe)
            if fe < fr:
                simplex[-1], fvals[-1] = xe, fe
            else:
                simplex[-1], fvals[-1] = xr, fr
        elif fr < fvals[-2]:
            simplex[-1], fvals[-1] = xr, fr
        else:
            # Contract
            if fr < fvals[-1]:
                xc = centroid + 0.5*(simplex[-1] - centroid)
            else:
                xc = centroid - 0.5*(simplex[-1] - centroid)
            fc = _obj(xc)
            if fc < fvals[-1]:
                simplex[-1], fvals[-1] = xc, fc
            else:
                # Shrink
                for i in range(1, Np+1):
                    simplex[i] = simplex[0] + sig*(simplex[i] - simplex[0])
                    fvals[i] = _obj(simplex[i])
    
    f_best = np.clip(basis @ simplex[np.argmin(fvals)], -1, 1)
    f_best -= f_best.mean()
    m = np.max(np.abs(f_best))
    if m > 1:
        f_best /= m
    h_best = _project(0.5*(f_best+1), n*0.5)
    return h_best, _c5_from_f(_h_to_f(h_best))
