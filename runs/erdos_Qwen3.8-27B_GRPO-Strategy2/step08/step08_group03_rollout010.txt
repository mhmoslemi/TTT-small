import numpy as np
import time

def _fft_size(n):
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(80):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(8):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-11:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    if abs(x.sum() - S) > 1e-8:
        lo = float(np.min(v)) - 1.0
        hi = float(np.max(v)) + 1.0
        for _ in range(45):
            mid = 0.5 * (lo + hi)
            s = np.clip(v - mid, 0.0, 1.0).sum()
            if s > S:
                lo = mid
            else:
                hi = mid
        x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    return x

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _pn_arrays(h):
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n)
    Nv = np.zeros(n)
    P[0] = total - ac[0]
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
    mx = float(P[0])
    if n > 1:
        mx = max(mx, float(np.max(P[1:])), float(np.max(Nv[1:])))
    return P, Nv, mx

def _smooth_val(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    dx = 2.0 / n
    e = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    if n > 1:
        e += np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(e) / alpha)

def _smooth_grad(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    dx = 2.0 / n
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    tot = e0
    ep = en = None
    if n > 1:
        ep = np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
    if tot <= 0:
        wP[0] = 1.0
        return dx * (1.0 - 2.0 * h)
    inv = 1.0 / tot
    wP[0] = e0 * inv
    if n > 1:
        wP[1:] = ep * inv
        wN[1:] = en * inv
    # gradient computation
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        Hf = np.fft.rfft(h, Nf)
        Urev = np.fft.rfft(u[::-1], Nf)
        Ur = np.fft.rfft(u, Nf)
        R = np.fft.irfft(Hf * Urev, Nf)[n - 1:2 * n - 1].real
        L = np.fft.irfft(Hf * Ur, Nf)[:n].real
        q = -(R + L)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

                y = _proj(y, S)
            fy = _smooth_val(y, alpha)
        else:
            y = h
            fy = fh
        g = _smooth_grad(y, alpha)
        step = 1.0 / max(Lcur, 1e-12)
        ok = False
        for _ in range(12):

            fx = _smooth_val(xn, alpha)
            d = xn - y
            rhs = fy + float(np.dot(g, d)) + 0.5 / step * float(np.dot(d, d))
            if fx <= rhs + 1e-10:

def _grad_active(h, P, Nv):
    n = len(h)
    dx = 2.0 / n
    mx = float(P[0])
    if n > 1:
        mx = max(mx, float(np.max(P[1:])), float(np.max(Nv[1:])))

    wP = np.zeros(n)
    wN = np.zeros(n)
    if P[0] >= mx - tol:
        wP[0] = 1.0
    if n > 1:
        mP = P[1:] >= mx - tol
        if np.any(mP):

        mN = Nv[1:] >= mx - tol
        if np.any(mN):
            wN[np.nonzero(mN)[0] + 1] = 1.0
    cnt = wP.sum() + wN.sum()
    if cnt == 0:
        return np.zeros(n)
    wP /= cnt
    wN /= cnt
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]

    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):

import numpy as np
import time

def _fft_size(n):
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(80):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(8):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-11:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    if abs(x.sum() - S) > 1e-8:
        lo = float(np.min(v)) - 1.0
        hi = float(np.max(v)) + 1.0
        for _ in range(45):
            mid = 0.5 * (lo + hi)
            if np.clip(v - mid, 0.0, 1.0).sum() > S:
                lo = mid
            else:
                hi = mid
        x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    return x

def _c5(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / len(h)))

def _pn_arrays(h):
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n); Nv = np.zeros(n)
    P[0] = total - ac[0]
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
    mx = float(P[0])
    if n > 1:
        mx = max(mx, float(np.max(P[1:])), float(np.max(Nv[1:])))
    return P, Nv, mx

def _smooth_val(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h); dx = 2.0 / n
    e = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    if n > 1:
        e += np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(e) / alpha)

def _grad_w(h, wP, wN):
    n = len(h); dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n); cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        Hf = np.fft.rfft(h, Nf)
        R = np.fft.irfft(Hf * np.fft.rfft(u[::-1], Nf), Nf)[n-1:2*n-1].real
        L = np.fft.irfft(Hf * np.fft.rfft(u, Nf), Nf)[:n].real
        q = -(R + L)
    return dx * (wP[0]*(1.0 - 2.0*h) + cp[n-1::-1] + cn + q)

def _smooth_grad(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    wP = np.zeros(n); wN = np.zeros(n)
    e0 = np.exp(np.clip(alpha*(P[0]-mx), -400.0, 0.0))
    tot = e0
    ep = en = None
    if n > 1:
        ep = np.exp(np.clip(alpha*(P[1:]-mx), -400.0, 0.0))
        en = np.exp(np.clip(alpha*(Nv[1:]-mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
    if tot <= 0:
        wP[0] = 1.0; return _grad_w(h, wP, wN)
    inv = 1.0/tot
    wP[0] = e0*inv
    if n > 1:
        wP[1:] = ep*inv; wN[1:] = en*inv
    return _grad_w(h, wP, wN)

def _fista(h0, alpha, budget, dl, L=10.0):
    n = len(h0); S = n*0.5; t0 = time.time()
    h = _repair(h0, n); hp = h.copy(); y = h.copy(); t = 1.0
    fy = _smooth_val(h, alpha); fh = fy
    bh = h.copy(); bc = _c5(h); Lc = L
    for it in range(200000):
        if time.time() > dl or time.time() > t0 + budget:
            break
        if it:
            tn = 0.5*(1.0 + np.sqrt(1.0 + 4.0*t*t))
            y = h + ((t-1.0)/tn)*(h-hp); t = tn
            if y.min() < -0.1 or y.max() > 1.1 or abs(y.sum()-S) > 1e-6:
                y = _proj(y, S)
            fy = _smooth_val(y, alpha)
        else:
            y = h; fy = fh
        g = _smooth_grad(y, alpha)
        step = 1.0/max(Lc, 1e-12); ok = False
        for _ in range(12):
            xn = _proj(y - step*g, S)
            fx = _smooth_val(xn, alpha)
            d = xn - y
            if fx <= fy + float(np.dot(g,d)) + 0.5/step*float(np.dot(d,d)) + 1e-10:
                ok = True; break
            step *= 0.5
        Lc = 1.0/max(step, 1e-12)
        if ok:
            hp = h; h = xn; fh = fx
            if it % 4 == 0:
                c = _c5(h)
                if c < bc: bc = c; bh = h.copy()
        else:
            hp = h; y = h; t = 1.0; fy = fh; Lc = max(Lc*0.5, 1.0)
    c = _c5(h)
    if c < bc: bc = c; bh = h.copy()
    return bh, bc, h

def _local_search(h0, rng, dl, s0=1.2e-5):
    n = len(h0); h = h0.copy(); c = _c5(h); bc = c; bh = h.copy()
    P, Nv, _ = _pn_arrays(h)
    mx = max(float(P[0]), float(np.max(P[1:])), float(np.max(Nv[1:]))) if n>1 else float(P[0])
    tol = max(1e-8, 5.0*(2.0/n))
    wP = np.zeros(n); wN = np.zeros(n)
    wP[0] = 1.0 if P[0] >= mx-tol else 0
    if n > 1:
        mp = P[1:] >= mx-tol; mn2 = Nv[1:] >= mx-tol
        if np.any(mp): wP[np.nonzero(mp)[0]+1] = 1.0
        if np.any(mn2): wN[np.nonzero(mn2)[0]+1] = 1.0
    cnt = wP.sum()+wN.sum()
    if cnt > 0: wP /= cnt; wN /= cnt
    g = _grad_w(h, wP, wN)
    order = np.argsort(g); q = max(4, n//10)
    low = order[:q]; hi = order[-q:]
    rc = 512; it = 0; tl = time.time()
    while time.time() < dl:
        if it >= rc:
            c = _c5(h)
            if c < bc: bc = c; bh = h.copy()
            P, Nv, _ = _pn_arrays(h)
            mx2 = max(float(P[0]), float(np.max(P[1:])), float(np.max(Nv[1:]))) if n>1 else float(P[0])
            tol2 = max(1e-8, 5.0*(2.0/n))
            wP = np.zeros(n); wN = np.zeros(n)
            wP[0] = 1.0 if P[0] >= mx2-tol2 else 0
            if n > 1:
                mp = P[1:] >= mx2-tol2; mn2 = Nv[1:] >= mx2-tol2
                if np.any(mp): wP[np.nonzero(mp)[0]+1] = 1.0
                if np.any(mn2): wN[np.nonzero(mn2)[0]+1] = 1.0
            cnt = wP.sum()+wN.sum()
            if cnt > 0: wP /= cnt; wN /= cnt
            g = _grad_w(h, wP, wN)
            order = np.argsort(g); low = order[:q]; hi = order[-q:]; rc = it+512
        frac = min(1.0, (time.time()-tl)/max(1e-9, dl-tl))
        base = max(1e-13, s0*(5e-6)**frac)
        if rng.random() < 0.75:
            i = int(low[rng.randint(len(low))]); j = int(hi[rng.randint(len(hi))])
        else:
            i = int(rng.randint(n)); j = int(rng.randint(n))
        if i == j: continue
        s = 1 if g[i] <= g[j] else -1
        if g[i] == g[j]: s = 1 if rng.random()<0.5 else -1
        av = min(1.0-h[i], h[j]) if s>0 else min(h[i], 1.0-h[j])
        if av <= 1e-14: continue
        m = 0.3+1.5*rng.random()
        if rng.random() < 0.02: m *= 12.0
        d = s*min(av, base*m)
        h[i] += d; h[j] -= d
        cn = _c5(h)
        if cn < c-1e-15:
            c = cn
            if c < bc: bc = c; bh = h.copy()
        else:
            h[i] -= d; h[j] += d
        it += 1
    return bh, bc

def _resample(h, m):
    n = len(h)
    if n == m: return _repair(h, n)
    if n % m == 0:
        out = h.reshape(m, n//m).mean(axis=1)
    else:
        x0 = (np.arange(n)+0.5)*(2.0/n)
        x1 = (np.arange(m)+0.5)*(2.0/m)
        out = np.interp(x1, x0, h)
    return _proj(out, m*0.5)

    rng = np.random.RandomState(seed)
    dl = t0 + budget_s - 2.0

    best = h0.copy(); bc = _c5(best)
