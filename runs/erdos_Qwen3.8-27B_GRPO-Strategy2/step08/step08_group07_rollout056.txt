def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:

                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

This gives the true log-sum-exp gradient, which is a much smoother and more informative gradient than topk. The cost is O(n * (2n-1)) = O(n²) per gradient computation, same as the correlation itself. So it's not more expensive.

With alpha large enough (say 10000+), the log-sum-exp of all shifts is very close to the max. The gradient points in the direction that most reduces the max while also slightly reducing the near-max values. This could be more effective than the topk approach.

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - tgt) > 1e-9:
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
        diff = tgt - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14: continue
        k = int(idx[ii]); sh = (n - 1) - k; wk = w[ii] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n-sp] -= wk * h[sp:]
    return g, m

def _grad_active(h, tol):
    n = len(h)
    c = _correl(h)
    m = c.max()
    active = np.where(c >= m - tol)[0]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in active:
        sh = (n - 1) - int(k)
        if sh >= 0:
            g[:n-sh] += 1.0 - h[sh:]
            if sh > 0: g[sh:] -= h[:-sh]
        else:
            sp = -sh
            g[sp:] += 1.0 - h[:-sp]
            g[:n-sp] -= h[sp:]
    g *= dx / len(active)
    return g, m

def _smax_obj(h, alpha, topk):
    c = _correl(h); m = c.max()
    if topk < len(c):
        ct = c[np.argpartition(c, -topk)]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _nest(h, mode, alpha_list, topk, tol, steps, s0, dl, beta):
    n = len(h); tgt = n * 0.5
    f = _eval(h); v = np.zeros(n); step = s0
    bh, bf = h.copy(), f
    for it in range(steps):
        if time.time() > dl: break
        if mode == 'smax':
            t = it / max(steps - 1, 1)
            ai = min(int(t * len(alpha_list)), len(alpha_list) - 1)
            alpha = alpha_list[ai]
            g, _ = _grad_smax(h, alpha, topk)
        else:

        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, tgt)
        fn = _eval(hn)

        if f < bf: bf, bh = f, h.copy()
    return _repair(bh, n), bf

def _cd(h, dl, rng, start_scale=0.008):

    c5 = _eval(h); bh = h.copy(); bc = c5
    scale = start_scale; no_imp = 0; total = 0
    while time.time() < dl:

        j = (i + 1 + rng.randint(0, n)) % n
        if i == j: j = (i + 1) % n
        d = scale * rng.randn()
        for sf in (1.0, 0.5, 0.25, -1.0, -0.5):
            st = d * sf
            ni, nj = h[i] + st, h[j] - st
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj

                    break

                    break
        total += 1; no_imp += 1
        if no_imp > 200:
            scale *= 0.7; no_imp = 0
            if scale < 1e-8: break
    return bh, bc
