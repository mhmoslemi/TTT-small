    n = len(h)
    c5c = _eval(h)
    best_h = h.copy(); best_c = c5c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = base_step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n; best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _downsample(h, factor):
    n = len(h)
    m = n // factor
    return h[:m * factor].reshape(m, factor).mean(axis=1).copy()

def _upsample(h, factor):
    return np.tile(h, factor)

def _measure_rate(h, nt=30):
    t = time.time()
    for _ in range(nt):
        _eval(h)
    el = time.time() - t
    return nt / max(el, 1e-9)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _eval(best_h)

    # Phase 1: Coarse multi-restart (100s)
    for ratio in [4, 2]:
        nc = max(24, n // ratio)
        scale = n // nc
        for restart in range(3):
            if time.time() - t0 > 100:
                break
            if restart == 0:
                hc = _repair(_downsample(best_h, scale), nc)
            elif restart == 1:
                hc = _repair(_downsample(best_h, scale) + 0.08 * rng.randn(nc), nc)
            else:
                hc = _repair(0.5 * _downsample(best_h, scale) + 0.3 * rng.rand(nc), nc)
            for alpha in [2000, 10000, 40000]:
                if time.time() - t0 > 100:
                    break
                hc = _slsqp_focus(hc, alpha, 120, 0)
            h_up = _repair(_upsample(hc, scale), n)
            c5n = _eval(h_up)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_up.copy()

    # Phase 2: Full-grid SLSQP (170s)
    slsqp_cfg = [(5000, 0, 400), (20000, 8, 300), (80000, 5, 250),
                 (300000, 3, 200), (1000000, 2, 180), (5000000, 2, 120)]
    for alpha, tk, mit in slsqp_cfg:
        if time.time() - t0 > 270:
            break
        h_new = _slsqp_focus(best_h, alpha, mit, tk)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n; best_h = h_new.copy()

    # Phase 3: Projected gradient (250s)
    pg_cfg = [(3000, 350, 0.006, 8, 0.7), (10000, 350, 0.004, 6, 0.75),
              (40000, 350, 0.003, 5, 0.8), (100000, 350, 0.002, 4, 0.8),
              (300000, 300, 0.001, 3, 0.85), (1000000, 300, 0.0005, 2, 0.85),
              (3000000, 200, 0.0003, 2, 0.9)]
    for alpha, steps, s0, tk, mu in pg_cfg:
        if time.time() - t0 > 530:
            break
        h_new = _pg_focus(best_h.copy(), alpha, steps, s0, tk, mu)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n; best_h = h_new.copy()

    # Phase 4: Second SLSQP round (50s)
    if time.time() - t0 < budget_s - 250:
        for alpha, tk, mit in [(5000000, 2, 150), (12000000, 1, 100)]:
            if time.time() - t0 > budget_s - 200:
                break
            h_new = _slsqp_focus(best_h, alpha, mit, tk)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n; best_h = h_new.copy()

    # Phase 5: Binary swap search (150s)
    if time.time() - t0 < budget_s - 200:
        tl = budget_s - 200 - (time.time() - t0)
        er = _measure_rate(best_h)
        ntries = min(int(tl * er * 0.55), 400000)
        h_b, c_b = _binary_swap_search(best_h, ntries, rng)
        if c_b < best_c5:
            best_c5 = c_b; best_h = h_b.copy()
        for thresh in [0.4, 0.6]:
            if time.time() - t0 > budget_s - 100:
                break
            h_t = _repair((best_h > thresh).astype(float), n)
            h_b2, c_b2 = _binary_swap_search(h_t, max(50000, ntries // 3), rng)
            if c_b2 < best_c5:
                best_c5 = c_b2; best_h = h_b2.copy()

    # Phase 6: Informed local search (50s)
    if time.time() - t0 < budget_s - 55:
        tl = budget_s - 55 - (time.time() - t0)
        er = _measure_rate(best_h)
        ntries = min(int(tl * er * 0.6), 2000000)
        best_h, best_c5 = _local_informed(best_h, ntries, rng, 4e-6)

    # Phase 7: Ultra-fine 2-point polish
    if time.time() - t0 < budget_s - 3:
        tl = budget_s - 3 - (time.time() - t0)
        er = _measure_rate(best_h)
        npol = min(int(tl * er * 0.75), 8000000)
        best_h, best_c5 = _local_2pt(best_h, npol, rng, 6e-8)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
