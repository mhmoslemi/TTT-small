import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Scale and redistribute
    h = h * target / max(s, 1e-15)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            room = (1.0 - h[idx]) if diff > 0 else h[idx]
            a = min(abs(diff), room)
            if a <= 0: break
            h[idx] += a * np.sign(diff)
            diff -= a * np.sign(diff)
            if abs(diff) < 1e-14: break
    return h

def _proj(h, n):
    # Project onto {0<=h<=1, sum(h)=n/2}
    target = n / 2.0
    h = np.clip(h, 0, 1)
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        # corr[k] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        # d/dh[i] for i in [0, n-1-k]: contributes (1-h[i]) from h[i+k] term
        # d/dh[i] for i in [k, n-1]: contributes -h[i-k] from (1-h[i]) term
        g[:n-k] += (1.0 - h[:n-k])
        g[k:] -= h[:n-k]
    else:
        k = -k
        g[k:] += (1.0 - h[k:])
        g[:n-k] -= h[k:]
    return g * dx

def _smax(h, alpha, topk):
    n = len(h)
    c = np.correlate(h, 1.0-h, mode='full') * (2.0/n)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha*(ct - m), -300, 300)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm))))/alpha, idx, c

def _smax_grad(h, alpha, topk):
    n = len(h)
    _, idx, _ = _smax(h, alpha, topk)
    c = np.correlate(h, 1.0-h, mode='full') * (2.0/n)
    m = c.max()
    ct = c[idx]
    s = np.clip(alpha*(ct - m), -300, 300)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0/n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] > 1e-14:
            g += w[ii] * _grad_lag(h, idx[ii] - (n-1), dx)
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline):

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline: return h
    try:
        r = minimize(lambda x: _smax(x, alpha, topk)[0], h,
                     jac=lambda x: _smax_grad(x, alpha, topk),
                     method='SLSQP', bounds=[(0,1)]*n,
                     constraints=[{'type':'eq','fun':lambda x: x.sum()-n/2,

                     options={'maxiter':maxit,'ftol':1e-20})
        return _repair(r.x, n)
    except:
        return _repair(h, n)

def _subgrad_cycle(h, topk, steps, deadline):

    step = 0.002
    beta = 0.7

        # Weight: softmax over top

        s = (ct - ct.max()) / 0.001
        w = np.exp(s); w /= w.sum()
        # Weighted gradient

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n/2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target/s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            order = np.argsort(1.0-h) if diff > 0 else np.argsort(h)
            for idx in order:
                room = min(abs(diff), (1.0-h[idx]) if diff > 0 else h[idx])
                h[idx] += room * (1 if diff > 0 else -1)
                diff -= room * (1 if diff > 0 else -1)
                if abs(diff) < 1e-14: break
    return h

def _proj(h, n):
    target = n/2.0
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo+hi)/2
        if np.clip(h-mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h-(lo+hi)/2, 0, 1)

def _grad_corr(h, idx, dx):
    """Gradient of correlation at array index idx (0..2n-2) w.r.t. h."""
    n = len(h)
    sh = (n-1) - idx  # negative of lag
    g = np.zeros(n)
    if sh >= 0:
        g[:n-sh] += (1.0-h[sh:])

    else:
        sp = -sh
        g[sp:] += (1.0-h[:n-sp])
        if sp > 0:
            g[:n-sp] -= h[sp:]
    return g * dx

def _smax_val(h, alpha, topk):
    n = len(h)
    c = np.correlate(h, 1.0-h, mode='full') * (2.0/n)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha*(ct-m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s-sm))))/alpha, idx, c

def _smax_grad(h, alpha, topk):

    _, idx, c = _smax_val(h, alpha, topk)
    m = c.max()
    ct = c[idx]
    s = np.clip(alpha*(ct-m), -400, 400)
    s -= s.max()
    w = np.exp(s); w /= w.sum()

    bh = h.copy(); bf = f

                for d in [step, -step]:
                    ni, nj = h[i]+d, h[j]-d
                    if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                        h[i], h[j] = ni, nj
                        fn = _eval(h)
                        if fn < f - 1e-18:
                            f = fn
                            if fn < bf: bf = fn; bh = h.copy()
                        else:
                            h[i], h[j] = ni-d, nj+d
        step *= 0.95
        if step < 1e-9: break
    return bh, bf
