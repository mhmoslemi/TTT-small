    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    c_k = alpha_k * t + beta_k (linear in t when other coords fixed)"""

    # c_k = dx * sum_j h[j]*(1-h[j+k-(n-1)])

    #   b-position: h[i-(n-1-k)]*(1-h[i]) if 0<=i-(n-1-k)<n
    alphas = np.zeros(m)
    for idx, k in enumerate(active_k):
        shift = k - (n - 1)  # so c_k = sum_j h[j]*(1-h[j+shift])
        a_coeff = 0.0
        b_coeff = 0.0
        j_a = i + shift  # (1-h[j_a]) multiplies h[i]
        if 0 <= j_a < n:
            a_coeff = (1.0 - h[j_a])
        j_b = i - shift  # h[j_b] multiplies (1-h[i])
        if 0 <= j_b < n:
            b_coeff = -h[j_b]
        alphas[idx] = dx * (a_coeff + b_coeff)

    c_all = _correl(h)
    betas = np.array([c_all[k] - alphas[idx] * h[i] for idx, k in enumerate(active_k)])

    # Check all pairwise intersections + boundaries
    candidates = [0.0, 1.0]

        for b_idx in range(a_idx + 1, m):

            if 0.0 <= t_star <= 1.0:
                candidates.append(t_star)

    best_val = np.inf
    for t in candidates:

        vals = alphas * t + betas
        mv = vals.max()
        if mv < best_val:
            best_val = mv
            best_t = t
    return best_t, best_val

            bh = h.copy()

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _solve_1d(h, i, active_k, c_all):
    n = len(h)
    dx = 2.0 / n
    m = len(active_k)
    shifts = active_k - (n - 1)
    ja = i + shifts
    jb = i - shifts
    va = (ja >= 0) & (ja < n)
    vb = (jb >= 0) & (jb < n)
    a_c = np.zeros(m)
    b_c = np.zeros(m)
    a_c[va] = 1.0 - h[ja[va]]
    b_c[vb] = -h[jb[vb]]
    alphas = dx * (a_c + b_c)
    betas = c_all[active_k] - alphas * h[i]
    
    # 1D max-min: find t in [0,1] minimizing max(alphas*t + betas)
    cands = [0.0, 1.0]
    for a in range(m):
        for b in range(a + 1, m):
            d = alphas[a] - alphas[b]
            if abs(d) < 1e-14:
                continue
            t = (betas[b] - betas[a]) / d
            if 0.0 < t < 1.0:
                cands.append(t)
    best_t = min(cands, key=lambda t: (alphas * t + betas).max())
    return best_t

def _coord_desc(h, deadline, m_start=5, m_end=15, sweeps=None):
    n = len(h)
    if sweeps is None:
        sweeps = 50000
    c5 = _eval(h)
    bh = h.copy()
    c_all = _correl(h)
    m_cur = m_start
    
    for sw in range(sweeps):
        if time.time() > deadline:
            break
        # Increase m over time
        frac = sw / sweeps
        m_cur = int(m_start + (m_end - m_start) * frac)
        
        if sw % 4 == 0 or m_cur != len(active_k):
            active_k = np.argpartition(c_all, -m_cur)[-m_cur:]
        
        for i in range(n):
            h[i] = _solve_1d(h, i, active_k, c_all)
        
        h = _repair(h, n)
        c_all = _correl(h)
        c5_new = float(c_all.max())
        if c5_new < c5:
            c5 = c5_new
            bh = h.copy()
    
    return bh, c5

def _nesterov(h, steps, step0, deadline, tol=0.003, beta=0.85):
    n = len(h)
    target = n * 0.5
    c5 = _eval(h)
    v = np.zeros(n)
    step = step0
    bh = h.copy()
    
    for _ in range(steps):
        if time.time() > deadline:
            break
        c = _correl(h)
        m = c.max()
        active = np.where(c >= m - tol)[0]
        dx = 2.0 / n
        g = np.zeros(n)
        for k in active:
            sh = (n - 1) - k
            if sh >= 0:
                g[:n - sh] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:-sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:-sp])
                g[:n - sp] -= h[sp:]
        g *= dx / len(active)
        
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _project(h - step * v, target)
        cn = _eval(hn)
        if cn < c5 - 1e-15:
            h, c5 = hn, cn
            step = min(step * 1.01, step0 * 15.0)
            if c5 < _eval(bh):
                bh = h.copy()
        else:
            step *= 0.5
            h = _project(hp - step * v, target)
            c5 = _eval(h)
            if c5 < _eval(bh):
                bh = h.copy()
    return _repair(bh, n), _eval(bh)

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _polish(h, niter, rng, deadline, scale=1e-5):
    n = len(h)
    c5 = _eval(h)
    bh = h.copy()
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < c5 - 1e-19:
                c5 = cn
                bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, c5

def _solve_1d_vec(h, i, active_k, c_all):

    betas = c_all[active_k] - alphas * h[i]

    # All pairwise intersections
    A = alphas[:, None] - alphas[None, :]  # m x m
    B = betas[None, :] - betas[:, None]    # m x m
    np.fill_diagonal(A, 1.0)  # avoid div by zero
    T = B / A  # m x m candidate t values

    cands = np.concatenate([[0.0, 1.0], T])
    # Evaluate
    vals = alphas[None, :] * cands[:, None] + betas[None, :]  # |cands| x m
    obj = vals.max(axis=1)
    best_idx = obj.argmin()
    return cands[best_idx]

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:

    return h

def _project(h, target):

            lo = mid
        else:
            hi = mid
