    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

    lo, hi = -2.0, 2.0

        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    h = np.clip(h, 0.0, 1.0).copy()

    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)

                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx

        # Update q via softmax
        cs = c - m
        s = cs / max(T, 1e-12)
        s -= s.max()
        q = np.exp(s)
        q /= q.sum()
        act = np.where(q > 1e-10)[0]
        eta *= 0.999
    return _repair(best_h, n), best_f

def _nesterov(h, steps, step0, tol, beta, deadline):
    n = len(h)
    target = n * 0.5
    c = _correl(h)
    m = c.max()
    f = m
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    for _ in range(steps):
        if time.time() > deadline:
            break
        active = np.where(c >= m - tol)[0]
        g = _grad_active(h, active, n)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        cn = _correl(hn)
        fn = cn.max()
        if fn < f - 1e-16:
            h, f, c = hn, fn, cn
            step = min(step * 1.012, step0 * 12.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.4
            if step < 1e-10:
                break
            hp2 = _proj(h - step * v, target)
            fn2 = _correl(hp2).max()
            if fn2 < f:
                h, f = hp2, fn2
                c = _correl(h)
                if f < best_f:
                    best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _lbfgs_refine(h, alpha, topk, maxit, deadline):
    n = len(h)
    target = n * 0.5

    def obj(x):
        cx = _correl(x)
        m = cx.max()
        if topk < len(cx):
            idx = np.argpartition(cx, -topk)[-topk:]
            ct = cx[idx]
        else:
            ct = cx
        s = np.clip(alpha * (ct - m), -500, 500)
        sm = s.max()
        return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha + 1e6 * (x.sum() - target) ** 2

    def jac(x):
        cx = _correl(x)
        m = cx.max()
        if topk < len(cx):
            idx = np.argpartition(cx, -topk)[-topk:]
            ct = cx[idx]
        else:
            idx = np.arange(len(cx))
            ct = cx
        s = np.clip(alpha * (ct - m), -500, 500)
        sm = s.max()
        w = np.exp(s - sm)
        w /= w.sum()
        dx_ = 2.0 / n
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] < 1e-13:
                continue
            k = idx[ii]; sh = (n - 1) - k
            wk = w[ii] * (1.0 / alpha) * dx_
            if sh >= 0:
                g[:n - sh] += wk * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= wk * h[:-sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - h[:-sp])
                g[:n - sp] -= wk * h[sp:]
        g += 2e6 * (x.sum() - target)
        return g

    try:
        r = minimize(obj, h, jac=jac, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxit, 'ftol': 1e-18, 'gtol': 1e-10, 'maxls': 20})
        return _repair(r.x, n)
    except Exception:
        return h

def _polish(h, niter, rng, deadline, scale):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 7)) % nc
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _anneal(h, niter, rng, deadline, blocks, T0, T1):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.002 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Primal-dual (new approach to explore different basins)
    dl1 = min(deadline, t0 + 180)
    for eta0, T0, T1, stps in [(0.003, 0.05, 0.001, 3000), (0.002, 0.02, 0.0005, 2000)]:
        if time.time() > dl1:
            break
        h2, c5v = _primal_dual(best_h.copy(), stps, eta0, T0, T1, dl1)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 2: Nesterov with exact subgradient
    dl2 = min(deadline, t0 + 380)
    for step0, tol, mom, ns in [(0.003, 0.003, 0.85, 3000),
                                (0.001, 0.001, 0.90, 3000),
                                (0.0005, 0.0005, 0.92, 2000)]:
        if time.time() > dl2:
            break
        h2, c5v = _nesterov(best_h.copy(), ns, step0, tol, mom, dl2)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 3: L-BFGS-B with soft-max (different optimizer landscape)
    dl3 = min(deadline, t0 + 550)
    for alpha, tk, mi in [(20000, 12, 200), (80000, 6, 150)]:
        if time.time() > dl3:
            break
        h2 = _lbfgs_refine(best_h.copy(), alpha, tk, mi, dl3)
        c5v = _eval(h2)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 4: Jitter + re-optimize (escape local minima)
    dl4 = min(deadline, t0 + 700)
    for scale in [0.003, 0.001, 0.005, 0.002, 0.001]:
        if time.time() > dl4:
            break
        d = scale * rng.randn(n0)
        d -= d.mean()
        hj = _repair(np.clip(best_h + d, 0, 1), n0)
        h2, c5v = _nesterov(hj, 800, 0.002, 0.002, 0.88, dl4)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 5: Block annealing
    dl5 = min(deadline, t0 + 850)
    blocks = [2, 3, 4, 5, 7, 10, 15]
    best_h, best_c5 = _anneal(best_h.copy(), 40000, rng, dl5, blocks, 0.002, 1e-10)

    # Phase 6: Fine Nesterov
    dl6 = min(deadline, t0 + budget_s - 60)
    h2, c5v = _nesterov(best_h.copy(), 4000, 0.0004, 0.0004, 0.93, dl6)
    if c5v < best_c5:
        best_c5, best_h = c5v, h2.copy()

    # Phase 7: Coordinate polish (multi-pass with decreasing scale)
    dl7 = deadline
    for scale in [5e-6, 2e-6, 1e-6, 5e-7]:
        rem = dl7 - time.time()
        if rem < 1:
            break
        n_pol = min(int(rem / 0.002), 30000)
        best_h, best_c5 = _polish(best_h, n_pol, rng, dl7, scale)

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
