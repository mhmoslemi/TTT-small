def _eval(h):

def _corr_and_grad(h, k_top, temp):
    """Return (c, gradient) using top-k softmax weighting."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if k_top >= 2 * n - 1:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        idx = np.argsort(c)[-k_top:]
        vals = c[idx]
        w_full = np.zeros(2 * n - 1)
        w_top = np.exp(vals / max(temp, 1e-8) - vals.max())
        w_top /= w_top.sum()
        w_full[idx] = w_top
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w_full[k] < 1e-15:
            continue
        wk = w_full[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g, c

class _LSECtx:
    def __init__(self, alpha, n, target):
        self.alpha = alpha
        self.n = n
        self.dx = 2.0 / n
        self.target = target
    def obj(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * self.dx
        m = c.max()
        s = np.clip(self.alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s - s.max()))) / self.alpha
    def grad(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * self.dx
        m = c.max()
        s = np.clip(self.alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(self.n)
        dx = self.dx
        for k in range(2 * self.n - 1):
            if w[k] < 1e-15:
                continue
            wk = w[k] * dx
            sh = (self.n - 1) - k
            if sh >= 0:
                g[:self.n - sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk * x[:self.n - sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:self.n - sp])
                g[:self.n - sp] -= wk * x[sp:]
        return g
    def con(self, x):
        return x.sum() - self.target

def _slsqp(h, alpha, maxiter, ftol, target):
    n = len(h)
    ctx = _LSECtx(alpha, n, target)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': ctx.con, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(ctx.obj, h, jac=ctx.grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

    b1, b2, eps = 0.85, 0.999, 1e-8

        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)

        c5 = _eval(h_new)
        if c5 < best_c5:

            h = h_new
            stall = 0
            lr = min(lr * 1.005, lr0 * 8.0)
        else:
            stall += 1
            if stall > 100:
                lr *= 0.85
                stall = 0
                if lr < lr0 * 0.005:
                    break
    return best_h, best_c5
