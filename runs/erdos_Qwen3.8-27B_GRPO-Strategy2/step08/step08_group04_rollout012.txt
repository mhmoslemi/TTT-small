def repair(h, target=None):
    h=np.asarray(h,dtype=float).copy()
    n=h.size
    h=np.clip(h,0.0,1.0)
    if target is None: target=0.5*n
    if n==0: return h
    s=h.sum()
    if s < 1e-14:
        h.fill(0.5)
        return h
    if abs(s-target)>1e-10:
        h *= target/s
        h=np.clip(h,0.0,1.0)
    diff=target-h.sum()
    if abs(diff)>1e-12:
        if diff>0:
            order=np.argsort(1.0-h)
        else:
            order=np.argsort(h)
        for idx in order:

                if a<=0: continue
                h[idx]+=a; diff-=a
            else:
                a=min(-diff, h[idx])
                if a<=0: continue
                h[idx]-=a; diff+=a
            if abs(diff)<1e-12: break
    return h

import time
import numpy as np

def repair(h, target=None):
    h = np.asarray(h, dtype=float).copy()
    n = h.size
    h = np.clip(h, 0.0, 1.0)
    if target is None:
        target = 0.5 * n
    if n == 0:
        return h
    s = h.sum()
    if s < 1e-14:
        h.fill(0.5)
        return h
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                if a <= 0:
                    continue
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                if a <= 0:
                    continue
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

            break
        if res > 0:
            mask = y < 1.0 - 1e-12
            cnt = int(mask.sum())
            if cnt == 0:
                break
            slack = 1.0 - y[mask]
            msl = float(slack.min())
            add = res / cnt
            if add <= msl + 1e-15:
                y[mask] += add
                res = target - y.sum()
            else:
                m2 = mask & (1.0 - y <= add + 1e-14)
                res -= (1.0 - y[m2]).sum()
                y[m2] = 1.0
        else:
            mask = y > 1e-12
            cnt = int(mask.sum())
            if cnt == 0:
                break
            val = y[mask]
            mn = float(val.min())
            sub = -res / cnt
            if sub <= mn + 1e-15:
                y[mask] -= sub
                res = target - y.sum()
            else:
                m2 = mask & (y <= sub + 1e-14)
                res += y[m2].sum()
                y[m2] = 0.0
    if abs(target - y.sum()) > 1e-8:
        return repair(y, target)
    return y

def corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def eval_c5(h):
    return float(np.max(corr(h))) * 2.0 / len(h)

def add_grad(g, h, k, dx, w):
    n = len(h)
    sh = n - 1 - k
    if sh >= 0:
        lg = n - sh
        if lg > 0:
            g[:lg] += w * (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= w * h[:-sh]
    else:
        sp = -sh
        g[sp:] += w * (1.0 - h[:-sp])
        g[:n - sp] -= w * h[sp:]

def smax_obj(h, alpha, topk):
    dx = 2.0 / len(h)
    c = corr(h) * dx
    m = float(c.max())
    if topk is None or topk >= c.size:
        ct = c
    else:
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    s = np.clip(alpha * (ct - m), -400.0, 400.0)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def smax_grad(h, alpha, topk):
    n = len(h)
    dx = 2.0 / n
    c = corr(h) * dx
    m = float(c.max())
    if topk is None or topk >= c.size:
        idx = np.arange(c.size)
        ct = c
    else:
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    s = np.clip(alpha * (ct - m), -400.0, 400.0)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        ww = float(w[ii])
        if ww > 1e-14:
            add_grad(g, h, int(idx[ii]), dx, ww)
    return g

def nesterov(h, deadline, max_steps, alpha, topk, step0, beta=0.8):
    h = repair(h)
    n = len(h)
    if n < 2 or time.time() > deadline:
        return h
    target = 0.5 * n
    f = smax_obj(h, alpha, topk)
    v = np.zeros(n)
    step = float(step0)
    cap = max(0.02, step0 * 30.0)
    stuck = 0
    for _ in range(max_steps):
        if time.time() > deadline:
            break
        g = smax_grad(h, alpha, topk)
        if np.linalg.norm(g) < 1e-14:
            break
        v = beta * v + (1.0 - beta) * g
        accepted = False
        s = step
        for _ in range(8):
            yn = proj_fast(h - s * v, target)
            fn = smax_obj(yn, alpha, topk)
            if fn < f - 1e-14:
                h = yn
                f = fn
                accepted = True
                break
            s *= 0.5
        if not accepted:
            s = max(step * 1e-4, 1e-12)
            yn = proj_fast(h - s * v, target)
            fn = smax_obj(yn, alpha, topk)
            if fn < f - 1e-16:
                h = yn
                f = fn
                accepted = True
        if accepted:
            stuck = 0
            step = min(step * 1.03, cap)
        else:
            stuck += 1
            step *= 0.5
            if step < max(1e-8, step0 * 1e-5) or stuck > 150:
                break
    return repair(h)

def resample(h, n_new):
    no = len(h)
    if no == n_new:
        return h.copy()
    oldx = (np.arange(no) + 0.5) * (2.0 / no)
    newx = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(newx, oldx, h, left=h[0], right=h[-1])
    return repair(y)

def fourier_init(n, rng, mmax=None, scale=None):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    f = np.zeros(n)
    if mmax is None:
        mmax = int(4 + rng.randint(10))
    mmax = min(int(mmax), max(1, n // 4))
    for m in range(1, mmax + 1):
        w = 1.0 / np.sqrt(float(m))
        f += rng.normal(0.0, w) * np.cos(np.pi * m * x)
        f += rng.normal(0.0, w) * np.sin(np.pi * m * x)
    f -= f.mean()
    sd = float(f.std())
    if sd < 1e-12:
        return repair(np.full(n, 0.5))
    if scale is None:
        scale = 0.12 + 0.18 * rng.random()
    h = 0.5 + scale * f / sd
    return repair(np.clip(h, 0.0, 1.0))

def binarize(p):
    n = len(p)
    k = n // 2
    h = np.zeros(n)
    if n % 2:
        idx = np.argsort(-p)
        h[idx[:k]] = 1.0
        h[idx[k]] = 0.5
    elif k > 0:
        idx = np.argpartition(p, n - k)[n - k:]
        h[idx] = 1.0
    return repair(h)

def choose_weighted(rng, arr, weights):
    a = np.asarray(arr)
    w = np.asarray(weights, dtype=float)
    if a.size == 0:
        return -1
    wm = float(w.max())
    if wm <= 1e-14:
        return int(a[int(rng.randint(a.size))])
    jitter = rng.rand(a.size) * (0.25 * wm)
    return int(a[int(np.argmax(w + jitter))])

def choose_active_pair(h, D, rng, step):
    n = len(h)
    k = int(np.argmax(D))
    sh = n - 1 - k
    if sh > 0:
        L = n - sh
        if L <= 0:
            return None
        a0 = np.arange(L)
        wr = h[a0] * (1.0 - h[a0 + sh])
        a = choose_weighted(rng, a0, wr)
        if a < 0:
            return None
        return a, a + sh, -step * (0.5 + 0.5 * rng.random())
    if sh < 0:
        p = -sh
        L = n - p
        if L <= 0:
            return None
        a0 = np.arange(p, n)
        wr = h[a0] * (1.0 - h[a0 - p])
        a = choose_weighted(rng, a0, wr)
        if a < 0:
            return None
        return a, a - p, -step * (0.5 + 0.5 * rng.random())
    hi = np.where(h > 0.5 + 1e-10)[0]
    lo = np.where(h < 0.5 - 1e-10)[0]
    if hi.size and lo.size:
        a = int(hi[int(rng.randint(hi.size))])
        b = int(lo[int(rng.randint(lo.size))])
        if a == b:
            a = int(rng.randint(n))
            b = int(rng.randint(n))
            if a == b:
                b = (a + 1) % n
        mag = step * (0.5 + 0.5 * rng.random())
        if h[a] > h[b]:
            return a, b, -mag
        return a, b, mag
    a = int(rng.randint(n))
    b = int(rng.randint(n))
    if a == b:
        b = (a + 1) % n
    mag = step * (0.5 + 0.5 * rng.random())
    if h[a] > h[b]:
        return a, b, -mag
    return a, b, mag

def pair_trial(h, D, i, j, d):
    n = len(h)
    Dtmp = D.copy()
    hh = h
    om = 1.0 - hh
    hi = hh[i]
    hj = hh[j]
    Dtmp[i:i + n][::-1] += d * om
    Dtmp[n - 1] += (hi + d) * (1.0 - (hi + d)) - hi * (1.0 - hi) - d * (1.0 - hi)
    Dtmp[n - 1 + i - j] += (hi + d) * (1.0 - (hj - d)) - hi * (1.0 - hj) - d * (1.0 - hj)
    Dtmp[j:j + n][::-1] += -d * om
    Dtmp[n - 1] += (hj - d) * (1.0 - (hj - d)) - hj * (1.0 - hj) - (-d) * (1.0 - hj)
    Dtmp[n - 1 + j - i] += (hj - d) * (1.0 - (hi + d)) - hj * (1.0 - hi) - (-d) * (1.0 - hi)
    st = n - 1 - i
    Dtmp[st:2 * n - 1 - i] += -d * hh
    Dtmp[n - 1] += d * hi
    Dtmp[n - 1 + j - i] += d * hj
    st = n - 1 - j
    Dtmp[st:2 * n - 1 - j] += d * hh
    Dtmp[n - 1] += -d * hj
    Dtmp[n - 1 - i + j] += -d * hi
    c5 = float(Dtmp.max()) * (2.0 / n)
    return Dtmp, c5

def local_search(h, deadline, rng, step=0.03, anneal=0.0, target_prob=0.65):
    h = repair(h)
    n = len(h)
    if n < 2:
        return h, eval_c5(h)
    D = corr(h)
    cur = float(D.max()) * (2.0 / n)
    best_h = h.copy()
    best_c = cur
    it = 0
    fail = 0
    st = float(step)
    T = float(anneal)
    while time.time() < deadline:
        it += 1
        if T > 0:
            T = float(anneal) / (1.0 + 0.002 * it)
        if it % 3000 == 0:
            D = corr(h)
            cur = float(D.max()) * (2.0 / n)
            if cur < best_c - 1e-15:
                best_c = cur
                best_h = h.copy()
        i = -1
        j = -1
        dv = 0.0
        use_t = False
        if rng.random() < target_prob:
            res = choose_active_pair(h, D, rng, st)
            if res is not None:
                i, j, dv = res
                use_t = True
        if i < 0:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                j = (i + 1) % n
            mag = st * (0.5 + 0.5 * rng.random())
            if h[i] > h[j]:
                dv = -mag
            else:
                dv = mag
            if rng.random() < 0.3:
                dv = -dv
        cands = [dv]
        if dv != 0.0:
            cands.append(dv * 0.5)
            if not use_t:
                cands.append(-0.5 * dv)
        accepted = False
        for dd in cands:
            if dd == 0.0:
                continue
            ni = h[i] + dd
            nj = h[j] - dd
            if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                continue
            Dtmp, c5n = pair_trial(h, D, i, j, dd)
            choose = c5n < cur - 1e-15
            if not choose and T > 0.0:
                diff = c5n - cur
                if 0.0 < diff < 3.0 * T:
                    if rng.random() < np.exp(-diff / T):
                        choose = True
            if choose:
                h[i] = ni
                h[j] = nj
                D = Dtmp
                cur = c5n
                accepted = True
                break
        if accepted:
            fail = 0
            st = min(st * 1.02, 0.15)
            if cur < best_c - 1e-15:
                best_c = cur
                best_h = h.copy()
        else:
            fail += 1
            if fail > 200:
                st *= 0.7
                if st < 1e-7:
                    st = 1e-6
                fail = 0
            if st < 1e-8 and fail > 500:
                break
    return best_h, eval_c5(best_h)

def consider(h, best_h, best_c, best_n):
    if h is None:
        return best_h, best_c, best_n
    h = repair(h)
    c = eval_c5(h)
    if best_h is None or c < best_c - 1e-15 or (abs(c - best_c) <= 1e-15 and len(h) < best_n):
        return h, c, len(h)
    return best_h, best_c, best_n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    full = float(budget_s)
    deadline = t0 + max(5.0, full - 8.0)
    rng = np.random.RandomState(seed)
    best_h = None
    best_c = 1e100
    best_n = 0
    ph = None
    try:
        ph = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        ph = None
    if ph is not None and ph.size:
        ph = repair(ph)
        best_h, best_c, best_n = consider(ph, best_h, best_c, best_n)
        hs = 0.5 * (ph + ph[::-1])
        best_h, best_c, best_n = consider(hs, best_h, best_c, best_n)
        hb = binarize(ph)
        best_h, best_c, best_n = consider(hb, best_h, best_c, best_n)
    else:
        h0 = fourier_init(560, rng, 24, 0.15)
        best_h, best_c, best_n = consider(h0, best_h, best_c, best_n)
    quick_dl = min(deadline, t0 + min(60.0, 0.07 * full))
    if best_h is not None and time.time() < quick_dl:
        hq, cq = local_search(best_h, quick_dl, rng, 0.025, 1e-5)
        best_h, best_c, best_n = consider(hq, best_h, best_c, best_n)
    if best_h is None:
        best_h, best_c, best_n = consider(np.full(256, 0.5), best_h, best_c, best_n)
    coarse_end = min(deadline, t0 + 0.32 * full)
    top = []
    if time.time() < coarse_end:
        n_coarse = int(max(96, min(192, best_n // 3 if best_n > 0 else 96)))
        if n_coarse % 2:
            n_coarse += 1
        base = resample(best_h, n_coarse)
        if time.time() < coarse_end:
            base = nesterov(base, min(coarse_end, time.time() + 4.0), 800, 3000, 30, 0.03, 0.8)
        base, cb = local_search(base, min(coarse_end, time.time() + 8.0), rng, 0.06, 5e-5)
        best_h, best_c, best_n = consider(base, best_h, best_c, best_n)
        top.append((float(cb), base.copy()))
        n_cand = 0
        while time.time() < coarse_end and n_cand < 160:
            n_cand += 1
            if n_cand % 4 == 1:
                mix = 0.3 + 0.5 * rng.random()
                fc = fourier_init(n_coarse, rng, int(3 + rng.randint(12)), 0.12 + 0.18 * rng.random())
                hc = mix * base + (1.0 - mix) * fc
            elif n_cand % 2 == 0:
                hc = fourier_init(n_coarse, rng, int(3 + rng.randint(14)), 0.12 + 0.20 * rng.random())
            elif n_cand % 7 == 3:
                hc = binarize(fourier_init(n_coarse, rng, int(3 + rng.randint(12)), 0.25))
            else:
                hc = fourier_init(n_coarse, rng, int(2 + rng.randint(8)), 0.10 + 0.22 * rng.random())
            if coarse_end - time.time() > 3.5:
                hc = nesterov(hc, min(coarse_end, time.time() + 0.9), 220, 3000, 24, 0.025, 0.8)
            hc, cc = local_search(hc, min(coarse_end, time.time() + 2.5), rng, 0.07, 8e-5)
            best_h, best_c, best_n = consider(hc, best_h, best_c, best_n)
            if len(top) < 8:
                top.append((float(cc), hc.copy()))
            else:
                mi = 0
                mx = top[0][0]
                for ii in range(1, len(top)):
                    if top[ii][0] > mx:
                        mx = top[ii][0]
                        mi = ii
                if cc < mx:
                    top[mi] = (float(cc), hc.copy())
    up_end = min(deadline, coarse_end + min(120.0, 0.12 * full))
    if top and time.time() < up_end:
        for _ in range(3):
            mi = 0
            mc = 1e100
            for ii in range(len(top)):
                if top[ii][0] < mc:
                    mc = top[ii][0]
                    mi = ii
            if mc > 1e99:
                break
            cc, hc = top[mi]
            top[mi] = (1e100, None)
            best_h, best_c, best_n = consider(hc, best_h, best_c, best_n)
            n_up = best_n
            if best_n < 2 * n_coarse:
                n_up = max(2 * n_coarse, 400)
            n_up = min(900, int(n_up))
            if n_up % 2:
                n_up += 1
            if n_up > n_coarse and time.time() < up_end - 30:
                hu = resample(hc, n_up)
                if time.time() < up_end - 25:
                    hu = nesterov(hu, min(up_end, time.time() + 25.0), 1500, 5000, 30, 0.015, 0.7)
                hu, cu = local_search(hu, min(up_end, time.time() + 40.0), rng, 0.02, 1e-5)
                best_h, best_c, best_n = consider(hu, best_h, best_c, best_n)
    nest_end = min(deadline, up_end + min(250.0, 0.22 * full))
    if best_h is not None and time.time() < nest_end:
        if time.time() < nest_end - 60:
            hh = best_h.copy()
            hh = nesterov(hh, min(nest_end, time.time() + 90.0), 4000, 3000, 35, 0.012, 0.8)
            best_h, best_c, best_n = consider(hh, best_h, best_c, best_n)
        if time.time() < nest_end - 40:
            hh = best_h.copy()
            hh = nesterov(hh, min(nest_end, time.time() + 70.0), 3000, 20000, 18, 0.005, 0.7)
            best_h, best_c, best_n = consider(hh, best_h, best_c, best_n)
        if time.time() < nest_end - 20:
            hh = best_h.copy()
            hh, cc = local_search(hh, min(nest_end, time.time() + 25.0), rng, 0.012, 0.0)
            best_h, best_c, best_n = consider(hh, best_h, best_c, best_n)
    if best_h is not None and time.time() + 50 < deadline:
        hh = 0.5 * (best_h + best_h[::-1])
        hh = repair(hh)
        hh, cc = local_search(hh, min(deadline, time.time() + 35.0), rng, 0.012, 0.0)
        best_h, best_c, best_n = consider(hh, best_h, best_c, best_n)
    if best_h is not None and time.time() < deadline:
        hh = best_h.copy()
        hh, cc = local_search(hh, deadline, rng, 0.012, 0.0)
        best_h, best_c, best_n = consider(hh, best_h, best_c, best_n)
    if best_h is None:
        best_h = np.full(256, 0.5)
    best_h = repair(best_h)
    best_c = eval_c5(best_h)
    return best_h, float(best_c), len(best_h)
