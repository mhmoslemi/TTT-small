def _proj(h, target):
    lo, hi = -4.0, 4.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _adam_opt(h, n_iter, alpha, top_k, lr, n):
    target = n * 0.5
    mv = np.zeros(n)
    vv = np.zeros(n)
    b1, b2 = 0.85, 0.999
    cur_c = _eval(h)
    for t in range(1, n_iter + 1):
        g = _soft_grad(h, alpha, top_k)
        mv = b1 * mv + (1 - b1) * g
        vv = b2 * vv + (1 - b2) * g * g
        mh = mv / (1.0 - b1 ** t)
        vh = vv / (1.0 - b2 ** t)
        h_try = h + lr * mh / (np.sqrt(vh) + 1e-8)
        h_try = _proj(h_try, target)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try
            cur_c = c_try
    return _repair(h, n), cur_c

def _lbfgs_pen(h, alpha, top_k, maxiter, n, lam):
    target = n * 0.5
    bnds = [(0.0, 1.0)] * n
    def _f(x):
        return _soft_obj(x, alpha, top_k) + lam * (x.sum() - target) ** 2
    def _g(x):
        return _soft_grad(x, alpha, top_k) + 2.0 * lam * (x.sum() - target)
    try:
        res = minimize(_f, h, jac=_g, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9})
        return _repair(res.x, n)
    except Exception:
        return h

def _local2(h, n_iter, step, rng, target):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    stall = 0
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                stall = 0
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                stall += 1
        if stall > 300:
            step *= 0.7
            stall = 0
    return best_h, best_c

def _local3(h, n_iter, step, rng, target):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 7)) % n
        k = (i + 5 + rng.randint(0, 15)) % n
        d1 = step * rng.randn()
        d2 = step * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            s0, s1, s2 = h[i], h[j], h[k]
            h[i], h[j], h[k] = ni, nj, nk
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = s0, s1, s2
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m*f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i*f:(i+1)*f] = h[i]
    return out

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    elapsed = lambda: time.time() - t0

    h_parent = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h_parent)
    target = n * 0.5
    best_h = h_parent.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Adam optimizer (multi-alpha schedule)
    for alpha, steps, lr in [(500, 2000, 0.3), (2000, 3000, 0.15), (10000, 2000, 0.08), (50000, 1500, 0.04)]:
        if elapsed() > 80:
            break
        h_t, c = _adam_opt(best_h.copy(), steps, alpha, 8, lr, n)
        c = _eval(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 2: L-BFGS-B with increasing alpha
    for alpha, mxi, lam in [(1000, 300, 500), (5000, 250, 1000), (20000, 200, 2000), (100000, 150, 5000)]:
        if elapsed() > 200:
            break
        h_t = _lbfgs_pen(best_h.copy(), alpha, 8, mxi, n, lam)
        c = _eval(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 3: Multi-resolution exploration
    if elapsed() < budget_s - 400:
        for ratio in [2, 4, 8]:
            if elapsed() > budget_s - 350:
                break
            nc = n // ratio
            for trial in range(3):
                if elapsed() > budget_s - 340:
                    break
                if trial == 0:
                    hc = _repair(_downsample(best_h, ratio), nc)
                else:
                    hc = np.clip(0.5 + 0.4 * rng.randn(nc), 0, 1)
                    hc = _repair(hc, nc)
                tc = _repair(_downsample(best_h, ratio), nc)
                if trial == 0:
                    hc = tc
                hc, _ = _adam_opt(hc, 500, 5000, 6, 0.2, nc)
                hc, _ = _lbfgs_pen(hc, 20000, 6, 100, nc, 2000)
                h_up = _repair(_upsample(hc, ratio), n)
                c_up = _eval(h_up)
                if c_up < best_c5:
                    best_c5 = c_up
                    best_h = h_up.copy()

    # Phase 4: More L-BFGS-B after multi-res
    if elapsed() < budget_s - 300:
        for alpha, mxi, lam in [(5000, 200, 1000), (50000, 150, 3000), (500000, 100, 10000)]:
            if elapsed() > budget_s - 280:
                break
            h_t = _lbfgs_pen(best_h.copy(), alpha, 10, mxi, n, lam)
            c = _eval(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()

    # Phase 5: Fine local search (main polishing)
    if elapsed() < budget_s - 150:
        t_remaining = budget_s - 150 - elapsed()
        n_iter = int(t_remaining * 2500)
        n_iter = min(n_iter, 400000)
        best_h, best_c5 = _local2(best_h, n_iter, 5e-5, rng, target)

    # Phase 6: 3-point local search
    if elapsed() < budget_s - 100:
        t_remaining = budget_s - 100 - elapsed()
        n3 = int(t_remaining * 1500)
        n3 = min(n3, 100000)
        best_h, best_c5 = _local3(best_h, n3, 3e-5, rng, target)

    # Phase 7: Ultra-fine 2-point polish with decreasing steps
    if elapsed() < budget_s - 30:
        t_remaining = budget_s - 30 - elapsed()
        n_fin = int(t_remaining * 4000)
        n_fin = min(n_fin, 300000)
        # Split into chunks with decreasing step
        chunk = n_fin // 4
        step_fin = 1e-5
        for ci in range(4):
            if elapsed() > budget_s - 28:
                break
            best_h, best_c5 = _local2(best_h, chunk, step_fin, rng, target)
            step_fin *= 0.4

    # Phase 8: Final Adam micro-polish + local
    if elapsed() < budget_s - 5:
        h_t, _ = _adam_opt(best_h.copy(), 500, 200000, 4, 0.01, n)
        c_t = _eval(h_t)
        if c_t < best_c5:
            best_c5 = c_t
            best_h = h_t.copy()
        n_last = int((budget_s - 5 - elapsed()) * 5000)
        n_last = min(n_last, 100000)
        best_h, best_c5 = _local2(best_h, n_last, 3e-6, rng, target)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
