import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_shift(h, k_idx, n, dx):
    g = np.zeros(n)
    sh = (n - 1) - k_idx
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _lse_smooth(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _lse_weights(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    return w

class _SLQPWrap:
    def __init__(self, alpha, n, target, dx):
        self.alpha = alpha
        self.n = n
        self.target = target
        self.dx = dx
    def obj(self, x):
        return _lse_smooth(_corr(x), self.alpha)
    def grad(self, x):
        c = _corr(x)
        w = _lse_weights(c, self.alpha)
        g = np.zeros(self.n)
        for ki in range(2 * self.n - 1):
            if w[ki] < 1e-12:
                continue
            g += w[ki] * _grad_shift(x, ki, self.n, self.dx)
        return g
    def cf(self, x):
        return x.sum() - self.target
    def cj(self, x):
        return np.ones(self.n)

def _slsqp(h, alpha, maxiter, n, target, dx):
    wrap = _SLQPWrap(alpha, n, target, dx)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': wrap.cf, 'jac': wrap.cj}]
    try:
        res = minimize(wrap.obj, h, jac=wrap.grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov_topk(h, k_top, n_steps, step0, mu, t_end, dx, n):
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    for _ in range(n_steps):
        if time.time() > t_end:
            break
        c = _corr(h)
        top_idx = np.argsort(c)[-k_top:]
        g = np.zeros(n)
        wk = dx / k_top
        for ki in top_idx:
            g += wk * _grad_shift(h, ki, n, 1.0)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _repair(h - step * v, n)
        f_new = _eval(h_new)
        if f_new < f:
            h, f = h_new, f_new
            step = min(step * 1.03, step0 * 10.0)
        else:
            step *= 0.5
            if step < 1e-10:
                v[:] = 0
                step = step0 * 0.5
                h = _repair(h_prev - step * v, n)
                f = _eval(h)
    return _repair(h, n)

def _adaptive_topk(h, t_end, dx, n):
    f = _eval(h)
    v = np.zeros(n)
    step = 0.0015
    for it in range(20000):
        if time.time() > t_end:
            break
        k_cur = min(400, 15 + it // 40)
        c = _corr(h)
        top_idx = np.argsort(c)[-k_cur:]
        g = np.zeros(n)
        wk = dx / k_cur
        for ki in top_idx:
            g += wk * _grad_shift(h, ki, n, 1.0)
        v = 0.82 * v + 0.18 * g
        h_prev = h.copy()
        h_new = _repair(h - step * v, n)
        f_new = _eval(h_new)
        if f_new < f:
            h, f = h_new, f_new
            step = min(step * 1.02, 0.012)
        else:
            step *= 0.5
            if step < 1e-10:
                step = 0.0004
                v[:] = 0
    return _repair(h, n)

def _local2(h, n_tries, rng, t_end, n):
    f = _eval(h)
    best_h, best_f = h.copy(), f
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 2)) % n
        d = 1.2e-6 * rng.randn()
        h[i] += d; h[j] -= d
        h = _repair(h, n)
        fn = _eval(h)
        if fn < f:
            f = fn
            if fn < best_f:
                best_f, best_h = fn, h.copy()
        else:
            h[i] -= d; h[j] += d
            h = _repair(h, n)
    return best_h, best_f
