```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lse(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    c_top = c[idx]
    m = c_top.max()
    s = np.clip(alpha * (c_top - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        sh = idx[ii] - (n - 1)
        wk = w[ii] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _grad_max(h, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    g = np.zeros(n)
    for ki in idx:
        sh = ki - (n - 1)
        w = dx / top_k
        if sh >= 0:
            g[:n-sh] += w * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= w * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += w * (1.0 - h[:n-sp])
            g[:n-sp] -= w * h[sp:]
    return g

def _mgd(h, n_iter, mode, alpha, top_k, step0, mu):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    for _ in range(n_iter):
        if mode == 0:
            g = _grad_lse(h, alpha, top_k)
        else:
            g = _grad_max(h, top_k)
        v = mu * v - step * g
        h_try = _proj_sum(h + v, target)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try
            cur_c = c_try
            step = min(step * 1.02, step0 * 5.0)
        else:
            step *= 0.7
            step = max(step, step0 * 0.05)
    return _repair(h, n), cur_c

def _local2(h, n_iter, step, rng, sa_temp=0.0):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            acc = c_new < cur_c
            if not acc and sa_temp > 1e-12 and rng.rand() < np.exp(-(c_new - cur_c) / sa_temp):
                acc = True
            if acc:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

    for it in range(n_iter):

        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:

            else:

    return best_h, best_c

def _slsqp_obj(x, alpha, top_k):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
    idx = np.argsort(c)[-top_k:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, top_k):
    return _grad_lse(x, alpha, top_k)

def _cons_fun(x):
    return x.sum() - len(x) * 0.5

def _cons_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, top_k, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_slsqp_obj, h, args=(alpha, top_k),
                       jac=_slsqp_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_lse(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        sh = idx[ii] - (n - 1)
        wk = w[ii] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _grad_max(h, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    g = np.zeros(n)
    for ki in idx:
        sh = ki - (n - 1)
        wk = dx / top_k
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _mgd(h, n_iter, mode, alpha, top_k, step0, mu):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    for _ in range(n_iter):
        if mode == 0:
            g = _grad_lse(h, alpha, top_k)
        else:
            g = _grad_max(h, top_k)
        v = mu * v - step * g
        h_try = _proj_sum(h + v, target)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try
            cur_c = c_try
            step = min(step * 1.02, step0 * 5.0)
        else:
            step *= 0.7
            step = max(step, step0 * 0.05)
    return _repair(h, n), cur_c

def _local2(h, n_iter, step, rng, sa_temp=0.0):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            acc = c_new < cur_c
            if not acc and sa_temp > 1e-12:
                if rng.rand() < np.exp(-(c_new - cur_c) / sa_temp):
                    acc = True
            if acc:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _dl_local(h, n_iter, step, rng, refresh=100):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    sh = 0
    for it in range(n_iter):
        if it % refresh == 0:
            sh = int(np.argmax(_eval_vec(h))) - (n - 1)
        i = rng.randint(n)
        if rng.rand() < 0.6:
            j = (i + sh) % n
        else:
            j = (i + 1 + rng.randint(0, 8)) % n
        if j == i:
            j = (i + 1) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _slsqp_obj(x, alpha, top_k):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
    idx = np.argsort(c)[-top_k:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, top_k):
    return _grad_lse(x, alpha, top_k)

def _cons_fun(x):
    return x.sum() - len(x) * 0.5

def _cons_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, top_k, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_slsqp_obj, h, args=(alpha, top_k),
                       jac=_slsqp_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _elapsed(t0):
    return time.time() - t0
