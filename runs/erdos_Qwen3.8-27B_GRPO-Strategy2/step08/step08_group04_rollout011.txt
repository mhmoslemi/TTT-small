import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) > 1e-9:
        h = _proj(h, target)
    return h

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _grad_lag(h, idx[ii], dx)
    return g

def _nesterov_run(h, alpha0, grow, topk, steps, step0, deadline, beta=0.83):
    n = len(h)
    f = _smax_obj(h, alpha0, topk)
    v = np.zeros(n)
    step = step0
    alpha = alpha0
    for it in range(steps):
        if time.time() > deadline:
            break
        if it > 0 and it % 400 == 0:
            alpha *= grow
        g = _grad_smax(h, alpha, topk)
        if np.dot(g, g) < 1e-32:
            break
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        step_eff = step
        for _ in range(12):
            hn = _proj(hp - step_eff * v, n * 0.5)
            fn = _smax_obj(hn, alpha, topk)
            if fn < f:
                h = hn
                f = fn
                break
            step_eff *= 0.5
        else:
            break
        step = min(step * 1.04, step0 * 10.0)
    return _repair(h, n)

def _slsqp_run(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    try:
        r = minimize(lambda x: _smax_obj(x, alpha, topk), h.copy(),
                     jac=lambda x: _grad_smax(x, alpha, topk),
                     method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
                                   'jac': lambda x: np.ones(n)}],
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return h

def _guided_anneal(h, niter, rng, deadline, T0, T1):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter - 1, 1))
        bs = int(rng.choice([3, 5, 8, 13, 21, 34, 55]))
        i0 = rng.randint(0, max(1, n - bs))
        c = _correl(h)
        tk = min(6, len(c))
        top_idx = np.argpartition(c, -tk)[-tk:]
        gg = np.zeros(n)
        for ki in top_idx:
            gg += _grad_lag(h, ki, 2.0 / n)
        gg_seg = gg[i0:i0 + bs]
        gg_norm = np.max(np.abs(gg_seg))
        if gg_norm > 1e-12:

        sigma = 0.005 * max(T, 1e-8) ** 0.5

        seg = h[i0:i0 + bs].copy() + d

            if c5n < bc:

                bh = h.copy()
    return bh, bc

def _micro_descent(h, nsteps, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step = 4e-6
