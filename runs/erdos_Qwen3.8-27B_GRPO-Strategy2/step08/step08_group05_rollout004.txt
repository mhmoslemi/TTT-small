import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _sg(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += (1.0 - h[i1[m1]])
    i2 = j + off
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp_one(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    dx = 2.0 / n

    def _obj(x):
        c = _correl_full(x)
        m = c.max()
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
        s = np.clip(alpha * (ct - m), -300, 300)
        return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

    def _obj_grad(x):
        c = _correl_full(x)
        m = c.max()
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
        s = np.clip(alpha * (ct - m), -300, 300)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] < 1e-14:
                continue
            g += w[ii] * _sg(x, idx[ii], n) * dx
        return g

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(_obj, h, jac=_obj_grad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18, 'maxls': 50})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _lp_descent(h, n, deadline, max_iter, eps, tk_init, tk_min):
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    stall = 0
    for it in range(max_iter):
