        if delta < 0 or rng.random() < np.exp(-delta / max(T * 0.1, 1e-12)):
            c = cn
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d
        it += 1
    return best_h, best_c

def _active_grad(h, P, Nv, topK):
    n = len(h)
    dx = 2.0 / n
    mx = float(P[0])
    if n > 1:
        mx = max(mx, float(np.max(P[1:])), float(np.max(Nv[1:])))
    tol = max(1e-9, 5.0 * dx)
    wP = np.zeros(n)
    wN = np.zeros(n)
    cnt = 0
    if P[0] >= mx - tol:
        wP[0] = 1.0
        cnt += 1
    if n > 1:
        idxP = np.argsort(-P[1:])[:topK]
        for k in idxP:
            if P[1 + k] >= mx - tol:
                wP[1 + k] = 1.0
                cnt += 1
        idxN = np.argsort(-Nv[1:])[:topK]
        for k in idxN:
            if Nv[1 + k] >= mx - tol:
                wN[1 + k] = 1.0
                cnt += 1
    if cnt == 0:
        return np.zeros(n)
    inv = 1.0 / cnt
    wP *= inv
    wN *= inv
    return _grad_from_weights(h, wP, wN)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)
    best = h0.copy()
    best_c = _c5_direct(best)
    for var in (1.0 - best, best[::-1]):
        v2 = _repair(var, n0)
        cv = _c5_direct(v2)
        if cv < best_c:
            best = v2
            best_c = cv
    e1 = t0 + min(50.0, 0.05 * budget_s)
    if time.time() < e1:
        for al in (30, 150, 800):
            if time.time() >= e1 - 1:
                break
            hf, bh, bc = _fista(best, al, min(15.0, e1 - time.time()))
            chf = _c5_direct(hf)
            if bc < best_c:
                best_c = bc
                best = bh
            if chf < best_c:
                best_c = chf
                best = hf
            best = hf if chf <= _c5_direct(bh) else bh
    e2 = t0 + min(300.0, 0.28 * budget_s)
    if budget_s > 200 and time.time() < e2 - 5:
        for m in (n0 // 2, n0 // 4):
            if time.time() >= e2 - 10:
                break
            hm = _resample(best, m)
            slot = min(50.0, e2 - time.time() - 5.0)
            if slot < 5.0:
                break
            for al in (50, 400, 3000):
                if time.time() >= e2 - 2:
                    break
                hf, bh, bc = _fista(hm, al, min(slot / 3.0, 20.0))
                chf = _c5_direct(hf)
                if bc < best_c:
                    best_c = bc
                    best = _resample(bh, n0) if len(bh) != n0 else bh
                if chf < best_c:
                    best_c = chf
                    best = _resample(hf, n0) if len(hf) != n0 else hf
                best = _resample(hf, n0) if len(hf) != n0 else hf
            best = _repair(best, n0)
    e3 = t0 + budget_s - 280.0
    if time.time() < e3 - 10:
        alphas_main = [50, 200, 800, 4000, 20000, 100000]
        per = max(5.0, (e3 - time.time()) / len(alphas_main))
        for al in alphas_main:
            if time.time() >= e3 - 2:
                break
            hf, bh, bc = _fista(best, al, min(per, e3 - time.time()))
            chf = _c5_direct(hf)
            if bc < best_c:
                best_c = bc
                best = bh
            if chf < best_c:
                best_c = chf
                best = hf
            best = hf if chf <= _c5_direct(bh) else bh
    e4 = t0 + budget_s - 3.0
    if time.time() < e4 - 1:
        dls = e4 - 2.0
        if dls - time.time() > 30:
            best, best_c = _local_sa(best, rng, dls, 5e-5, 8)
    for var in (1.0 - best, best[::-1],
                0.5 + 0.5 * (best - best[::-1]),
                0.5 + 0.5 * (best + best[::-1] - 1.0)):
        v2 = _proj(var, len(best) * 0.5)
        cv = _c5_direct(v2)
        if cv < best_c:
            best = v2
            best_c = cv
    best = _repair(best, len(best))
    best_c = _c5_direct(best)
    return best, best_c, len(best)
