import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = _corr_full(x) * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = _corr_full(x) * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g, m

def _proj_onto_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sum_fn(x, half_n):
    return x.sum() - half_n

def _sum_jac(x, half_n):
    return np.ones(len(x))

def _adam_run(h, n, alpha_lo, alpha_hi, m_lags, n_steps, lr_init, deadline, b1=0.8, b2=0.999):
    target = n / 2.0
    m_acc = np.zeros(n)
    v_acc = np.zeros(n)
    t = 0
    eps = 1e-8
    lr = lr_init
    cur_c = _eval(h)
    patience = 0
    best_lr = lr
    for step in range(n_steps):
        if time.time() > deadline:
            break
        t += 1
        frac = step / max(n_steps - 1, 1)
        alpha = alpha_lo * (alpha_hi / alpha_lo) ** frac
        g, obj_max = _smooth_grad(h, alpha, m_lags)
        m_acc = b1 * m_acc + (1 - b1) * g
        v_acc = b2 * v_acc + (1 - b2) * g * g
        m_hat = m_acc / (1 - b1 ** t)
        v_hat = v_acc / (1 - b2 ** t)
        step_size = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = _proj_onto_box_sum(h - step_size, target)
        c_new = _eval(h_new)
        if c_new < cur_c:
            h = h_new
            cur_c = c_new
            patience = 0
            lr = min(lr * 1.005, lr_init * 4.0)
            best_lr = lr
        else:
            patience += 1
            lr *= 0.92
            if patience > 200:
                lr = best_lr * 1.3
                patience = 0
                m_acc[:] = 0.0
                v_acc[:] = 0.0
                t = 0
    return h, cur_c

def _slsqp_run(h, n, alpha, m_lags, maxiter, ftol, deadline):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=lambda x: _smooth_grad(x, alpha, m_lags)[0],
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if time.time() > deadline:
            return h
        return _repair(res.x, n)
    except Exception:
        return h

def _fine_search(h, n, n_iter, rng, base_step, deadline):
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    for it in range(n_iter):
        if time.time() > deadline:
            break
        step = base_step * (0.3 + rng.rand() * 1.4)
        np_ = rng.randint(2, 6)
        idx = rng.randint(0, n, size=np_)
        d = step * rng.randn(np_)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < -0.01 or h_new.max() > 1.01:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5cur:
            c5cur = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        if it > 0 and it % 5000 == 0 and rng.rand() < 0.05:
            h = best_h.copy()
            c5cur = best_c
    return best_h, best_c

def _swap_search(h, n, n_iter, rng, step_scale, deadline):
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    for it in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 10)) % n
        d = step_scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5cur:
                c5cur = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
        if it > 0 and it % 20000 == 0 and rng.rand() < 0.03:
            h = best_h.copy()
            c5cur = best_c
    return best_h, best_c
