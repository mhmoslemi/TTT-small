from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):

    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_corr(h, k, n):
    """Gradient of correlation at lag k w.r.t. h"""
    g = np.zeros(n)

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        g += w[ii] * _grad_corr(h, idx[ii], n)
    return g

def _slsqp_sm(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj_wrap(x):
        return _smax_obj(x, alpha, topk)
    def jac_wrap(x):
        return _grad_smax(x, alpha, topk)
    def cons_fun(x):
        return x.sum() - n * 0.5
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        r = minimize(obj_wrap, h, jac=jac_wrap, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18, 'maxls': 40})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]

            if c5n < bc:

    return bh, bc

def _smart_polish(h, niter, rng, deadline):

    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    # Use decreasing step sizes
    phases = [5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7]

                return bh, bc

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_corr(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += (1.0 - h[i1[m1]])
    i2 = j + off
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

def _lp_adapt(h, n_steps, eps0, m_top, deadline, decay=0.9994):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5c = best_c5
    eps = eps0
    stall = 0
    for si in range(n_steps):
        if time.time() > deadline:
            break
        if si % 40 == 0:
            eps *= decay
            eps = max(eps, 5e-6)
        cor = _correl(c)
        idx_t = np.argpartition(cor, -m_top)[-m_top:]
        idx_t = idx_t[np.argsort(-cor[idx_t])]
        grads = np.array([_grad_corr(c, k, n) for k in idx_t])
        c_obj = -grads[0]
        A_ub = grads[1:] if m_top > 1 else None
        b_ub = np.zeros(m_top - 1) if m_top > 1 else None
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bnds = [(-eps, eps)] * n
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success or res.x is None or np.max(np.abs(res.x)) < 1e-15:
            stall += 1
            if stall > 80:
                break
            rng_s = np.random.RandomState(si * 7 + 13)
            d2 = rng_s.randn(n)
            d2 -= d2.mean()
            d2 *= eps / max(np.max(np.abs(d2)), 1e-15)
            hn = _repair(c + d2, n)
            cn = _eval(hn)
            if cn < c5c:
                c, c5c = hn, cn
                if c5c < best_c5:
                    best_c5, best_h = c5c, c.copy()
                stall = 0
        else:
            stall = 0
            d = res.x
            improved = False
            for ls in range(15):
                hn = _repair(c + (0.5 ** ls) * d, n)
                cn = _eval(hn)
                if cn < c5c - 1e-17:
                    c, c5c = hn, cn
                    improved = True
                    if c5c < best_c5:
                        best_c5, best_h = c5c, c.copy()
                    break
    return best_h, best_c5

def _sm_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -300, 300)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _sm_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -300, 300)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] > 1e-14:
            g += w[ii] * _grad_corr(h, idx[ii], n)
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _sm_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.85
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _sm_grad(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _sm_obj(hn, alpha, topk)
        if fn < f - 1e-15:
            h, f = hn, fn
            step = min(step * 1.03, step0 * 10.0)
        else:
            step *= 0.5
            if step < 1e-13:
                break
            h = _proj(hp - step * v, n * 0.5)
            f = _sm_obj(h, alpha, topk)
    return _repair(h, n)

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def of(x):
        return _sm_obj(x, alpha, topk)
    def jf(x):
        return _sm_grad(x, alpha, topk)
    def cf(x):
        return x.sum() - n * 0.5
    def cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        r = minimize(of, h, jac=jf, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18, 'maxls': 35})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.006 * max(T, 1e-10) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _grad_polish(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    cor = _correl(h)
    cor_max = np.argmax(cor)
    g = _grad_corr(h, cor_max, n)
    steps = [1e-4, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-6]
    per = max(1, niter // len(steps))
    recompute = 0
    for ss in steps:
        for _ in range(per):
            if time.time() > deadline:
                return bh, bc
            i = rng.randint(n)
            if g[i] > 1e-8:
                di = -ss
            elif g[i] < -1e-8:
                di = ss
            else:
                di = ss * (2.0 * rng.random() - 1.0)
            j = (i + 1 + rng.randint(0, 6)) % n
            dj = -di
            ni, nj = h[i] + di, h[j] + dj
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5n < bc:
                        bc, bh = c5n, h.copy()
                    recompute = 0
                else:
                    h[i], h[j] = ni - di, nj - dj
                    recompute += 1
            if recompute > 200:
                cor = _correl(h)
                cor_max = np.argmax(cor)
                g = _grad_corr(h, cor_max, n)
                recompute = 0
    return bh, bc

def _fine_pair_polish(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    steps = [3e-5, 1.5e-5, 7e-6, 3e-6, 1.5e-6, 7e-7, 3e-7]

            i = rng.randint(n)

            d = ss * (2.0 * rng.random() - 1.0)

            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
