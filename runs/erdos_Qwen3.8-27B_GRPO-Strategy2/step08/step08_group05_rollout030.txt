def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_single_shift(h, k, n):
    """Gradient of corr[k]*dx w.r.t. h."""
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    mask1 = (i1 >= 0) & (i1 < n)
    g[mask1] += (1.0 - h[i1[mask1]])
    i2 = j + off
    mask2 = (i2 >= 0) & (i2 < n)
    g[mask2] -= h[i2[mask2]]
    return g * (2.0 / n)

def _smax(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h); m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-13: continue
        k = idx[ii]
        g += w[ii] * _grad_single_shift(h, k, n)
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax(h, alpha, topk)
    v = np.zeros(n)
    beta, step = 0.85, step0
    for _ in range(steps):
        if time.time() > deadline: break
        g = _grad_smax(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax(hn, alpha, topk)
        if fn < f:
            h, f = hn, fn
            step = min(step * 1.05, step0 * 8.0)
        else:
            step *= 0.5
            if step < 1e-11: break
            h = _proj(h - step * v, n * 0.5)
            f = _smax(h, alpha, topk)
    return _repair(h, n)

def _lp_descent(h, n_steps, eps, m_top, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        cor = _correl(c)
        idx_top = np.argpartition(cor, -m_top)[-m_top:]
        idx_top = idx_top[np.argsort(-cor[idx_top])]
        grads = np.array([_grad_single_shift(c, k, n) for k in idx_top])
        c_obj = -grads[0]
        A_ub = grads[1:] if m_top > 1 else None
        b_ub = np.zeros(m_top - 1) if m_top > 1 else None
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bnds = [(-eps, eps)] * n
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs')
        except Exception:
            break
        if not res.success or res.x is None or np.max(np.abs(res.x)) < 1e-14:
            break
        d = res.x
        step = 1.0
        improved = False
        for _ in range(15):
            h_new = _repair(c + step * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-16:
                c, c5_cur = h_new, c5_new
                improved = True
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
                break
            step *= 0.5
        if not improved:
            # Try slightly larger epsilon
            d = np.random.randn(n)
            d -= d.mean()
            d *= eps * 0.5 / max(np.max(np.abs(d)), 1e-15)
            h_new = _repair(c + d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur:
                c, c5_cur = h_new, c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h_new.copy()
    return best_h, best_c5

def _tabu_local(h, n_iters, tabu_size, deadline, rng):

    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5c = best_c5
    tabu = {}
    tabu_iter = 0
    for it in range(n_iters):
        if time.time() > deadline: break
        tabu_iter += 1
        cor = _correl(c)
        m = cor.max()
        # Find top 3 shifts
        top3 = np.argsort(-cor)[:3]
        # For the worst shift, find candidate swaps
        best_shift = top3[0]
        off = best_shift - (n - 1)
        # Candidates: positions where c[i]=1 and c[i+off]=0 (contributing to overlap)
        j = np.arange(n)
        i1 = j - off
        mask1 = (i1 >= 0) & (i1 < n)
        contributing = np.zeros(n, dtype=bool)
        contributing[mask1] = (c[j[mask1]] > 0.5) & (c[i1[mask1]] < 0.5)
        # Also positions where c[i]=0 and could be swapped
        i2 = j + off
        mask2 = (i2 >= 0) & (i2 < n)
        contributing[mask2] |= (c[j[mask2]] < 0.5) & (c[i2[mask2]] > 0.5)
        
        cands = np.where(contributing)[0]
        if len(cands) < 2:
            cands = np.where(cor > m * 0.95 > 0)[0]
            if len(cands) < 2:
                cands = np.random.choice(n, size=min(20, n), replace=False)
        
        # Try random pairs from candidates
        improved = False
        ntr = min(8, len(cands))
        for _ in range(ntr):
            i = cands[rng.randint(len(cands))]
            j2 = cands[rng.randint(len(cands))]
            if i == j2: continue
            if tabu_iter - tabu.get((min(i,j2), max(i,j2)), -1000) < tabu_size:
                continue
            tabu[(min(i,j2), max(i,j2))] = tabu_iter
            # Swap: move mass from i to j2
            amt = min(c[i], 1.0 - c[j2]) * 0.3

            c[i] -= amt; c[j2] += amt
            c5n = _eval(c)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5c < best_c5:
                    best_c5, best_h = c5c, c.copy()
                improved = True
                break
            else:
                c[i] += amt; c[j2] -= amt
        if not improved and it % 100 == 0:
            # Perturbation
            idx = rng.choice(n, size=5, replace=False)
            for ii in idx:
                c[ii] += 0.01 * rng.randn()
            c = _repair(c, n)
            c5c = _eval(c)
    return best_h, best_c5

def _sweep_polish(h, niter, rng, deadline):

    sss = [1e-5, 5e-6, 2e-6, 1e-6, 5e-7]

            ni, nj = h[i] + d, h[j] - d

                    if c5n < bc: bc, bh = c5n, h.copy()

    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += (1.0 - h[i1[m1]])
    i2 = j + off
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g * (2.0 / n)

def _lp_desc(h, n_steps, eps, m_top, deadline):
    n = len(h)
    bh, bc = h.copy(), _eval(h)
    c = h.copy()
    cc = bc
    for _ in range(n_steps):
        if time.time() > deadline: break
        cor = _correl(c)
        it = np.argpartition(cor, -m_top)[-m_top:]
        it = it[np.argsort(-cor[it])]
        gr = np.array([_grad_shift(c, k, n) for k in it])
        try:
            res = linprog(-gr[0], A_ub=gr[1:], b_ub=np.zeros(m_top-1),
                          A_eq=np.ones((1,n)), b_eq=np.zeros(1),
                          bounds=[(-eps,eps)]*n, method='highs')
        except: break
        if not res.success or res.x is None or np.max(np.abs(res.x))<1e-14: break
        d = res.x
        ok = False
        for s in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
            hn = _repair(c + s*d, n)
            cn = _eval(hn)
            if cn < cc - 1e-17:
                c, cc = hn, cn; ok = True
                if cc < bc: bc, bh = cc, c.copy()
                break
        if not ok:
            d2 = np.random.randn(n); d2 -= d2.mean()
            d2 *= eps*0.5/max(np.max(np.abs(d2)),1e-15)
            hn = _repair(c+d2, n); cn = _eval(hn)
            if cn < cc: c, cc = hn, cn
            if cc < bc: bc, bh = cc, c.copy()
    return bh, bc

def _nests(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _eval(h)  # use actual max for simplicity
    v = np.zeros(n)
    beta, st = 0.85, step0
    for _ in range(steps):
        if time.time() > deadline: break
        cor = _correl(h)
        if topk < len(cor):
            idx = np.argpartition(cor, -topk)[-topk:]
        else:
            idx = np.arange(len(cor))
        ct = cor[idx]
        s = np.clip(alpha*(ct - ct.max()), -400, 400)
        w = np.exp(s - s.max()); w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] > 1e-13:
                g += w[ii] * _grad_shift(h, idx[ii], n)
        v = beta*v + (1-beta)*g
        hn = _proj(h - st*v, n*0.5)
        fn = _eval(hn)
        if fn < f:
            h, f = hn, fn
            st = min(st*1.04, step0*6)
        else:
            st *= 0.5
            if st < 1e-12: break
            h = _proj(h - st*v, n*0.5)
            f = _eval(h)
    return _repair(h, n)

import numpy as np
from scipy.optimize import linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_shift(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += (1.0 - h[i1[m1]])
    i2 = j + off
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g * (2.0 / n)

    bh, bc = h.copy(), _eval(h)

                              A_eq=A_eq, b_eq=b_eq, bounds=bnds, method='highs')
            else:

        except:
            break
        if not res.success or res.x is None or np.max(np.abs(res.x)) < 1e-15:

        ok = False
        for s in [1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.01]:
