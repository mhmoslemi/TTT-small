import numpy as np
from scipy.optimize import linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(30):
        if abs(s - target) < 1e-13:
            break
        h = np.clip(h * (target / s), 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_k(h, k, n):
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[sh:] += (1.0 - h[:n - sh])
        g[:n - sh] -= h[sh:]
    else:
        sp = -sh
        g[:n - sp] += (1.0 - h[sp:])
        g[sp:] -= h[:n - sp]
    return g

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        ct = c[np.argpartition(c, -topk)[-topk:]]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:][:]
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _grad_k(h, idx[ii], n)
    return g

def _fista(h, alpha, topk, max_iter, step0, deadline):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    y = h.copy()
    t = 1.0
    step = step0
    f_prev = _smax_obj(y, alpha, topk)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = _smax_grad(y, alpha, topk)
        h_new = _proj_box_sum(y - step * g, target)
        f_new = _smax_obj(h_new, alpha, topk)
        if f_new <= f_prev:
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = h_new + ((t - 1.0) / t_next) * (h_new - h)
            y = _proj_box_sum(y, target)
            h = h_new; f_prev = f_new; t = t_next
            step = min(step * 1.04, step0 * 12.0)
        else:
            step *= 0.4
            if step < 1e-14:
                break
            h_new = _proj_box_sum(y - step * g, target)
            f_new = _smax_obj(h_new, alpha, topk)
            if f_new <= f_prev:
                t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
                y = h_new + ((t - 1.0) / t_next) * (h_new - h)
                y = _proj_box_sum(y, target)
                h = h_new; f_prev = f_new; t = t_next
            else:
                break
        if it % 200 == 0 and it > 0:
            h = _repair(h, n)
    return _repair(h, n)

def _lp_step(c, n, eps, m_top):
    cor = _correl(c)
    idx_top = np.argpartition(cor, -m_top)[-m_top:]
    idx_top = idx_top[np.argsort(-cor[idx_top])]
    grads = [np.zeros(n) for _ in range(m_top)]
    for ii in range(m_top):
        grads[ii] = _grad_k(c, idx_top[ii], n)
    c_obj = -grads[0]
    A_ub = np.array(grads[1:]) if m_top > 1 else None
    b_ub = np.zeros(m_top - 1) if m_top > 1 else None
    A_eq = np.ones((1, n))
    b_eq = np.zeros(1)
    bnds = [(-eps, eps)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bnds, method='highs', options={'presolve': True})
    except Exception:
        return None
    if not res.success or res.x is None:
        return None
    if np.max(np.abs(res.x)) < 1e-15:
        return None
    return res.x

def _lp_descent(h, n_steps, eps_start, eps_end, m_top, deadline, rng):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    stall = 0
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = it / max(n_steps, 1)
        eps = eps_start * (eps_end / eps_start) ** frac
        mt = m_top if frac < 0.7 else max(3, m_top - 2)
        d = _lp_step(c, n, eps, mt)
        if d is None:
            stall += 1
            if stall > 80:
                if stall > 200:
                    break
                kick = 0.002 * rng.randn(n)
                kick -= kick.mean()
                kick *= eps / max(np.max(np.abs(kick)), 1e-15)
                c = _repair(c + kick, n)
                c5_cur = _eval(c)
                continue
            stall += 1
            continue
        stall = 0
        step = 1.0
        improved = False
        for _ in range(10):
            h_new = _repair(c + step * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-18:
                c, c5_cur = h_new, c5_new
                improved = True
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
                break
            step *= 0.5
        if not improved:
            d2 = 0.001 * rng.randn(n)
            d2 -= d2.mean()
            sc = eps * 0.3 / max(np.max(np.abs(d2)), 1e-15)
            h_new = _repair(c + sc * d2, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur:
                c, c5_cur = h_new, c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h_new.copy()
    return best_h, best_c5

def _anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy(); bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.004 * max(T, 1e-10) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _polish(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy(); bc = c5c
    for ss in [5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7, 5e-8]:
        ni = max(1, niter // 7)
        for _ in range(ni):
            if time.time() > deadline:
                return bh, bc
            for attempt in range(3):
                i = rng.randint(n)
                j = (i + 1 + rng.randint(0, 6)) % n
                d = ss * (2.0 * rng.random() - 1.0)
                ni2, nj2 = h[i] + d, h[j] - d
                if 0 <= ni2 <= 1 and 0 <= nj2 <= 1:
                    h[i], h[j] = ni2, nj2
                    c5n = _eval(h)
                    if c5n < c5c - 1e-20:
                        c5c = c5n
                        if c5n < bc:
                            bc, bh = c5n, h.copy()
                        break
                    else:
                        h[i], h[j] = ni2 - d, nj2 + d
    return bh, bc
