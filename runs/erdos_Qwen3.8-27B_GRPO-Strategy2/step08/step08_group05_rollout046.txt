    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

def _minimax_lp(h, topk, eps, deadline):
    """Find direction d minimizing max_j (grad_j . d) subject to constraints"""
    n = len(h)
    c = _correl(h)
    idx_top = np.argpartition(c, -topk)[-topk:]
    idx_top = idx_top[np.argsort(-c[idx_top])]
    grads = np.array([_grad_at_lag(h, k, n) for k in idx_top])
    
    # Minimize t subject to:
    # grads[j] . d <= t for j=0..topk-1
    # sum(d) = 0
    # |d[i]| <= eps
    # Variables: [d_0, ..., d_{n-1}, t] (n+1 variables)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((topk, n + 1))
    for j in range(topk):
        A_ub[j, :n] = grads[j]
        A_ub[j, n] = -1.0  # grads[j].d - t <= 0
    b_ub = np.zeros(topk)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-eps, eps)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 2.0})
    except Exception:
        return None
    if not res.success:
        return None
    return res.x[:n]

import numpy as np
from scipy.optimize import minimize, linprog

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_box(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sg(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    mask1 = (i1 >= 0) & (i1 < n)
    g[mask1] += (1.0 - h[i1[mask1]])
    i2 = j + off
    mask2 = (i2 >= 0) & (i2 < n)
    g[mask2] -= h[i2[mask2]]
    return g

def _minimax_direction(h, topk, eps):
    n = len(h)
    c = _correl(h)
    idx_top = np.argpartition(c, -topk)[-topk:]
    idx_top = idx_top[np.argsort(-c[idx_top])]
    grads = np.array([_sg(h, k, n) for k in idx_top])
    nv = n + 1
    c_obj = np.zeros(nv); c_obj[-1] = 1.0
    A_ub = np.zeros((topk, nv))
    for j in range(topk):
        A_ub[j, :n] = grads[j]
        A_ub[j, n] = -1.0
    b_ub = np.zeros(topk)
    A_eq = np.zeros((1, nv)); A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-eps, eps)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
    except Exception:
        return None
    if not res.success:
        return None
    return res.x[:n]

def _lp_minimax(h, n_iter, topk, eps, deadline):

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _smax_grad(h, alpha, topk):
    c = _correl(h)
    n = len(h)
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        g += w[ii] * _sg(h, int(idx[ii]), n)
    return g * (2.0 / n)

    f = _smax_obj(h, alpha, topk)

def _slsqp_polish(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smax_obj(x, alpha, topk)
    def jac(x):
        return _smax_grad(x, alpha, topk)
    def eqf(x):
        return x.sum() - n * 0.5
    def eqj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': eqf, 'jac': eqj}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):

    c5c = _eval(h)
    bh, bc = h.copy(), c5c

            h, c5c = hn, c5n
            if c5n < bc:

    return bh, bc

    scales = [4e-7, 2e-7, 1e-7, 5e-8, 2.5e-8]

    return (best_h, best_c5, len(best_h))
