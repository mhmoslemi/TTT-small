    valid_first = []  # valid_first[i] = (s_min, s_max) for i+s valid

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _pair_slope(h, i, j, n):
    slopes = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        v = 0.0
        if 0 <= i + s < n:
            v += 1.0 - h[i + s]
        if 0 <= i - s < n:
            v -= h[i - s]
        if 0 <= j + s < n:
            v -= 1.0 - h[j + s]
        if 0 <= j - s < n:
            v += h[j - s]
        slopes[k] = v
    return slopes

def _bisect_d(c0, slopes, dlo, dhi):
    lo, hi = dlo, dhi
    for _ in range(35):
        d = (lo + hi) * 0.5
        vals = c0 + slopes * d
        m = vals.max()
        am = vals >= m - 1e-9
        if not am.any():
            am = vals >= m - 1e-6
        sa = slopes[am]
        if sa.min() > 1e-13:
            hi = d
        elif sa.max() < -1e-13:
            lo = d
        else:
            return d
    return (lo + hi) * 0.5
