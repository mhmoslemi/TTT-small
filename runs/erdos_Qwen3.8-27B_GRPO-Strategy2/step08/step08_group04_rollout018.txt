    for it in range(n_rounds):

        g /= topk
        # Find best pair: decrease where g is most positive, increase where g is most negative
        i_src = np.argmax(g)  # most positive gradient → decreasing h there helps
        i_dst = np.argmin(g)  # most negative gradient → increasing h there helps

                        bh = h.copy()

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s > 1e-12:
        h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):

def _grad_lag(h, k, dx):

            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx
