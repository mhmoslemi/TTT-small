def _repair(h, n):

    tg = n * 0.5
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(40):
        h *= tg / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - tg) < 1e-13:
            break
    df = tg - h.sum()
    if abs(df) > 1e-13:
        order = np.argsort(1.0 - h) if df > 0 else np.argsort(h)
        for i in order:
            if df > 0:
                a = min(df, 1.0 - h[i])
                h[i] += a
                df -= a
            else:
                a = min(-df, h[i])
                h[i] -= a
                df += a

def _score(h):
    h = np.asarray(h, dtype=float)
    r = np.correlate(h, 1.0 - h, mode="full")
    return r, float(r.max()) * (2.0 / len(h))

def _top(c, t):
    m = c.size
    if t <= 0 or t >= m:
        return np.arange(m)
    t = int(t)
    if t == 1:
        return np.array([int(np.argmax(c))])
    return np.argpartition(c, -t)[-t:]

def _obj(h, a, t):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(c.max())
    ii = _top(c, t)
    s = a * (c[ii] - m)
    s = np.clip(s, -500.0, 500.0)
    s -= s.max()
    e = np.exp(s)
    return m + float(np.log(e.sum()) / a)

def _grad(h, a, t):
    n = len(h)
    dx = 2.0 / n
    off = n - 1
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(c.max())
    ii = _top(c, t)

    s = np.clip(s, -500.0, 500.0)

        else:

def _sf(x, tg):
    return x.sum() - tg

def _sj(x, tg):
    return np.ones_like(x)

def _run_slsqp(h, a, t, it, ft):
    x0 = np.asarray(h, dtype=float).copy()

                if rv < best_val - 1e-15:
                    best_t = rt
                    best_val = rv
        if best_t > 1e-12 and best_val < cur - 1e-13:
            h[s] -= best_t
            h[d2] += best_t
            np.multiply(dl, best_t, out=tm)
            tm += c
            q2 = best_t * best_t
            tm[i0] -= 2.0 * q2
            tm[i1] += q2
            tm[i2] += q2
            c = tm.copy()
            cur = best_val
            bad = 0
            acc += 1
            if acc % 64 == 0:
                last_h = h.copy()
                last_c = c.copy()
                last_cur = cur
            if acc % 160 == 0:
                cr = np.correlate(h, 1.0 - h, mode="full")
                cv = float(cr.max()) * dx
                if cv > last_cur + 1e-8:
                    h = last_h.copy()
                    c = last_c.copy()
                    cur = last_cur
                else:
                    c = cr.copy()
                    cur = cv
        else:
            bad += 1
            if bad > 500:
                break
    cr1 = np.correlate(h, 1.0 - h, mode="full")
    cv1 = float(cr1.max()) * dx
    cr2 = np.correlate(last_h, 1.0 - last_h, mode="full")
    cv2 = float(cr2.max()) * dx
    if cv1 <= cv2:
        return h, cr1, cv1
    return last_h, cr2, cv2

def _down(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1), m)

def _up(h, f):
    n = len(h)
    o = np.empty(n * f)
    for i in range(n):
        o[i * f:(i + 1) * f] = h[i]
    return _repair(o, n * f)

def _upi(h, m):
    n = len(h)
    if m == n:
        return _repair(h, n)
    x = np.linspace(0.0, n - 1, n)
    y = np.linspace(0.0, m - 1, m)
    return _repair(np.interp(y, x, h), m)

def _pert(h, sg, rng):
    d = rng.randn(len(h)) * sg
    d -= d.mean()
    return _repair(h + d, len(h))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(5.0, float(budget_s)) - 12.0
    rng = np.random.RandomState(seed)
    h = _repair(np.asarray(initial_h_values, dtype=float).copy(), len(initial_h_values))
    raw, c5 = _score(h)
    best_h = h.copy()
    best_raw = raw.copy()
    best_c = c5
    work = h.copy()

    shift_limit = min(deadline, t0 + 25.0)
    n0 = len(h)
    for s in range(n0):
        if time.time() > shift_limit:
            break
        hs = np.roll(h, s)
        r, v = _score(hs)
        if v < best_c - 1e-15:
            best_h = hs.copy()
            best_raw = r.copy()
            best_c = v
            work = hs.copy()

    du_limit = min(deadline, time.time() + 60.0)
    for f in (2, 4, 8):
        if time.time() > du_limit or n0 % f:
            continue
        m = n0 // f
        if m < 30:
            continue
        hc = _down(work, f)
        for aa, tk, it in ((300, 48, 180), (3000, 48, 220)):
            hc = _run_slsqp(hc, aa, tk, it, 1e-12)
        hu = _up(hc, f)
        r, v = _score(hu)
        if v < best_c:
            best_h = hu.copy()
            best_raw = r.copy()
            best_c = v
            work = hu.copy()

    stop = deadline - 320.0
    for aa, tk, it, ft in ((300, 128, 180, 1e-12),
                           (1500, 112, 220, 1e-13),
                           (6000, 96, 250, 1e-13),
                           (30000, 80, 250, 1e-14)):
        if time.time() > stop:
            break
        hn = _run_slsqp(work, aa, tk, it, ft)
        r, v = _score(hn)
        if v < best_c:
            best_h = hn.copy()
            best_raw = r.copy()
            best_c = v
            work = hn.copy()
        elif v < best_c + 1e-5:
            work = hn.copy()

    stop = deadline - 200.0
    for aa, tk, secs in ((50, 64, 100), (200, 96, 120), (1000, 128, 120), (4000, 80, 100)):
        if time.time() > stop:
            break
        pd = min(stop, time.time() + secs)
        hh, rr, hv = _pair(best_h, best_raw, aa, tk, 400000, pd, rng)
        if hv < best_c:
            best_h = hh.copy()
            best_raw = rr.copy()
            best_c = hv
            work = hh.copy()

    stop = deadline - 140.0
    for aa, tk, it in ((200000, 64, 180), (1000000, 48, 180)):
        if time.time() > stop:
            break
        hn = _run_slsqp(work, aa, tk, it, 1e-15)
        r, v = _score(hn)
        if v < best_c:
            best_h = hn.copy()
            best_raw = r.copy()
            best_c = v
            work = hn.copy()
        elif v < best_c + 1e-5:
            work = hn.copy()

    stop = deadline - 90.0
    if time.time() < stop:
        hh, rr, hv = _pair(best_h, best_raw, 2000, 96, 300000, min(stop, time.time() + 90.0), rng)
        if hv < best_c:
            best_h = hh.copy()
            best_raw = rr.copy()
            best_c = hv
            work = hh.copy()

    stop = deadline - 110.0
    for sg in (0.01, 0.03, 0.08):
        if time.time() > stop:
            break
        hp = _pert(best_h, sg, rng)
        hp = _run_slsqp(hp, 3000, 96, 180, 1e-13)
        r, v = _score(hp)
        if v < best_c:
            best_h = hp.copy()
            best_raw = r.copy()
            best_c = v
            work = hp.copy()
            continue
        if v > best_c + 1e-3:
            continue
        hh, rr, hv = _pair(hp, r, 1000, 96, 80000, min(stop, time.time() + 45.0), rng)
        if hv < best_c:
            best_h = hp.copy()
            best_raw = hh.copy()
            best_c = hv
            work = hp.copy()

    if len(best_h) < 960 and time.time() < deadline - 320.0:
        b1 = _upi(best_h, 960)
        r1, v1 = _score(b1)
        if v1 < best_c:
            best_h = b1.copy()
            best_raw = r1.copy()
            best_c = v1
            work = b1.copy()
        if time.time() < deadline - 200.0:
            b1 = _run_slsqp(b1, 3000, 112, 200, 1e-13)
            r1, v1 = _score(b1)
            if v1 < best_c:
                best_h = b1.copy()
                best_raw = r1.copy()
                best_c = v1
            hh, rr, hv = _pair(b1, r1, 2000, 96, 400000, min(deadline - 50.0, time.time() + 140.0), rng)
            if hv < best_c:
                best_h = hh.copy()
                best_raw = rr.copy()
                best_c = hv

    best_h = _repair(best_h, len(best_h))
    best_h, best_c = best_h, _score(best_h)[1]
    return best_h, float(best_c), len(best_h)
