But the key is: the moves need to be in the RIGHT direction. Random 2-point moves will mostly be rejected. Gradient-directed moves should have a higher acceptance rate.

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h = np.clip(h * t / s, 0, 1)
        s = h.sum()
        if abs(s - t) < 1e-14:
            break
    d = t - h.sum()
    if abs(d) > 1e-13:
        if d > 0:
            for idx in np.argsort(1.0 - h):
                a = min(d, 1.0 - h[idx])
                h[idx] += a; d -= a
                if d < 1e-14: break
        else:
            for idx in np.argsort(h):
                a = min(-d, h[idx])
                h[idx] -= a; d += a
                if -d < 1e-14: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1. - h, 'full')) * 2. / n)

def _glag(k, h, n):
    dx = 2. / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += 1. - h[i1 + sh]
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += 1. - h[i1 - sp]
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

            lo = mid
