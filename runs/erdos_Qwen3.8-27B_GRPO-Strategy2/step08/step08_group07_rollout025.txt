import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax_obj_and_grad(h, alpha, topk=None):
    n = len(h)
    c = _correl(h)
    if topk is not None and topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = alpha * ct
    s_max = s.max()
    e = np.exp(s - s_max)
    w = e / e.sum()
    dx = 2.0 / n
    obj = (s_max + np.log(e.sum())) / alpha
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += w[ii] * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= w[ii] * h[:-sh]
        else:
            sp = -sh
            g[sp:] += w[ii] * (1.0 - h[:-sp])
            g[:n - sp] -= w[ii] * h[sp:]
    g *= dx
    return obj, g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h = (h - h.min()) / (h.ptp() + 1e-15) if h.ptp() > 1e-9 else np.full(n, 0.5)
        h = np.clip(h, 0, 1)
        # scale to hit target
        if h.sum() > 0:
            h *= target / h.sum()
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a

    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax_hg(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    if topk and topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = alpha * ct
    sm = s.max()
    e = np.exp(s - sm)
    w = e / e.sum()
    obj = (sm + np.log(np.sum(e))) / alpha
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return obj, g

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a

    return h

def _nesterov(h, steps, step0, deadline, tol=0.002, beta=0.88):
