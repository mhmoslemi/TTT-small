def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:

            a = min(diff, 1.0 - h[idx])
            h[idx] += a
            diff -= a
        else:

            h[idx] -= a
            diff += a

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for idx in order:
        if diff > 0:
            a = min(diff, 1.0 - h[idx])
            h[idx] += a; diff -= a
        else:
            a = min(-diff, h[idx])
            h[idx] -= a; diff += a
        if abs(diff) < 1e-12:
            break
    return h
