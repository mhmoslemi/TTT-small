import numpy as np
from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_lse(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    c_top = c[idx]
    m = c_top.max()
    s = np.clip(alpha * (c_top - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        sh = k - (n - 1)
        wk = w[ii] * dx
        if sh >= 0:
            m2 = n - sh
            g[sh:] += wk * (1.0 - h[:m2])
            g[:m2] -= wk * h[sh:]
        else:
            p = -sh
            m2 = n - p
            g[:m2] += wk * (1.0 - h[p:p+m2])
            g[p:] -= wk * h[:m2]
    return g

def _slsqp(h, alpha, maxiter, top_k):
    n = len(h)
    target = n * 0.5
    bnds = [(0.0, 1.0)] * n

    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
        idx = np.argsort(c)[-top_k:]
        ct = c[idx]
        m = ct.max()
        s = np.clip(alpha * (ct - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha

    def _grad(x):
        return _grad_lse(x, alpha, top_k)

    def _sum_c(x):
        return x.sum() - target

    def _sum_j(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h
