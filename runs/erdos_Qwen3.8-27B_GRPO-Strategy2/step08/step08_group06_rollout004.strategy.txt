```python
            grad_candidate[:len(h)-e] += (1 - h[s:e]) / n
        
        # Update parameters
        alpha = max(1e-4, alpha * decay)
        gradient = 0.95 * gradient + 0.05 * grad_candidate
        h_new = h - alpha * gradient
        
        # Repair and accept/reject
        h_new = repair_solution(h_new, len(h))
        if evaluate_erdos(h_new) < best_value:
            h, best_value = h_new, evaluate_erdos(h_new)
            h_best = h.copy()
        
        # Periodic reinitialization (10% of iterations)
        if i > 0 and i % (max_iter/10) == 0:
            h = repair_solution(np.random.rand(len(h)) * 2 - 1, len(h))
    
    return h_best, best_value
```
