import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_c(h, k, n):
    """Gradient of correlation at shift k w.r.t. h"""
    g = np.zeros(n)
    L = k - (n - 1)  # actual lag
    if L >= 0:
        g[:n-L] += (1.0 - h[L:])
        if L > 0:
            g[L:] -= h[:-L]
    else:
        sp = -L
        g[sp:] += (1.0 - h[:-sp])
        g[:n-sp] -= h[sp:]
    return g

def _smax(h, alpha, topk):
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / len(h)
    c *= dx
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = alpha * (ct - m)
    s = np.clip(s, -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = alpha * (ct - m)
    s = np.clip(s, -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _grad_c(h, idx[ii], n)
    return g

def _subgrad(h, alpha, topk, nsteps, step0, deadline):
    n = len(h)
    v = np.zeros(n)
    beta = 0.85
    step = step0
    cur = _smax(h, alpha, topk)
    for it in range(nsteps):
        if time.time() > deadline:
            break
        g = _grad_smax(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        h_new = _proj(h - step * v, n * 0.5)
        f_new = _smax(h_new, alpha, topk)
        if f_new < cur:
            h = h_new
            cur = f_new
            step = min(step * 1.02, step0 * 10)
        else:
            step *= 0.5
            if step < 1e-12:
                break
            h = _proj(h - step * v, n * 0.5)
            cur = _smax(h, alpha, topk)
    return _repair(h, n)

def _binary_local(h, n, niter, deadline):
    """Local search on near-binary h: try flipping pairs"""
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    dh = 2.0 / n
    
    for it in range(niter):
        if time.time() > deadline:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # Try moving mass from i to j
        d = min(h[i], 1.0 - h[j])
        if d < 1e-6:
            continue
        h[i] -= d
        h[j] += d
        c5n = _eval(h)
        if c5n < bc:
            bc, bh = c5n, h.copy()
        else:
            h[i] += d
            h[j] -= d
    return bh, bc

def _pair_descent(h, n, niter, deadline, topk=5):
    """Coordinate pair descent guided by gradient"""
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    rng = np.random.RandomState(42)
    
    for it in range(niter):
        if time.time() > deadline:
            break
        # Compute gradient of smooth max
        cor = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
        idx = np.argpartition(cor, -topk)[-topk:]
        # Gradient direction: negative weighted sum of correlation gradients
        g = np.zeros(n)
        w = np.exp(cor[idx] - cor[idx].max())
        w /= w.sum()
        for ii in range(len(idx)):
            g += w[ii] * _grad_c(h, idx[ii], n)
        
        # Find best coordinate to decrease and best to increase
        i = int(np.argmin(g))  # most beneficial to increase h[i]
        j = int(np.argmax(g))  # most beneficial to decrease h[j]
        
        if i == j:
            continue
        
        # Try small move
        d = 0.01 * (h[i] + 1 - h[j])
        d = min(d, h[j], 1.0 - h[i])
        if d < 1e-8:
            continue
        
        h[i] += d
        h[j] -= d
        c5n = _eval(h)
        if c5n < bc:
            bc, bh = c5n, h.copy()
        else:
            h[i] -= d
            h[j] += d
    return bh, bc

def _block_opt(h, n, nblocks, blen, niter, deadline):
    """Optimize blocks sequentially"""
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    
    for it in range(niter):
        if time.time() > deadline:
            break
        b = it % nblocks
        i0 = b * blen
        i1 = min(i0 + blen, n)
        seg = h[i0:i1].copy()
        target_seg = np.mean(seg) * (i1 - i0)
        
        # Random perturbation of block
        pert = 0.02 * (np.random.rand(i1-i0) - 0.5)
        pert -= pert.mean()  # preserve sum
        seg_new = np.clip(seg + pert, 0, 1)
        # Re-normalize
        if seg_new.sum() > 1e-9:
            seg_new *= target_seg / seg_new.sum()
            seg_new = np.clip(seg_new, 0, 1)
        
        h_new = h.copy()
        h_new[i0:i1] = seg_new
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < bc:
            bc = c5n
            bh = h_new.copy()
            h = h_new.copy()
    return bh, bc

def _lp_step(h, n, eps, m_top, deadline, max_iters=200):
    """Single LP-based descent step"""
    c = h.copy()
    c5_cur = _eval(c)
    best_h, best_c5 = c.copy(), c5_cur
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        cor = np.correlate(c, 1.0 - c, mode='full')
        idx_top = np.argpartition(cor, -m_top)[-m_top:]
        idx_top = idx_top[np.argsort(-cor[idx_top])]
        
        grads = np.array([_grad_c(c, k, n) for k in idx_top])
        c_obj = -grads[0]
        A_ub = grads[1:] if m_top > 1 else None
        b_ub = np.zeros(m_top - 1) if m_top > 1 else None
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bnds = [(-eps, eps)] * n
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success or res.x is None:
            break
        d = res.x
        if np.max(np.abs(d)) < 1e-14:
            break
        
        step = 1.0
        improved = False
        for _ in range(10):
            h_new = _repair(c + step * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-15:
                c = h_new
                c5_cur = c5_new
                improved = True
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
                break
            step *= 0.5
        if not improved:
            break
    return best_h, best_c5

def _sweep_polish(h, niter, deadline, seed=42):
    """Final coordinate sweep"""
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    rng = np.random.RandomState(seed)
    sizes = [1e-5, 5e-6, 2e-6, 1e-6, 5e-7]
    per_size = niter // len(sizes)
    
    for ss in sizes:
        for _ in range(per_size):
            if time.time() > deadline:
                return bh, bc
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 5)) % n
            d = ss * (2 * rng.random() - 1)
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < bc:
                    bc, bh = c5n, h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
    return bh, bc

        bnds = [(-eps, eps)] * n
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success or res.x is None or np.max(np.abs(res.x)) < 1e-14:
            break
        d = res.x
        step = 1.0
        improved = False
        for _ in range(10):
            h_new = _repair(c + step * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-15:
                c, c5_cur = h_new, c5_new
                improved = True
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
                break
            step *= 0.5
        if not improved:
            break
    return best_h, best_c5

def _intensive_local(h, deadline, seed=42):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    rng = np.random.RandomState(seed)
    scales = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    per_scale = 6000
    for scale in scales:
        for _ in range(per_scale):
            if time.time() > deadline:
                return bh, bc
            cor = np.correlate(h, 1.0 - h, mode='full')
            topk = 10
            idx = np.argpartition(cor, -topk)[-topk:]
            w = np.exp(cor[idx] - cor[idx].max())
            w /= w.sum()
            g = np.zeros(n)
            for ii in range(len(idx)):
                if w[ii] < 1e-10:
                    continue
                g += w[ii] * _grad_c(h, idx[ii], n)
            i = int(np.argmin(g))
            j = int(np.argmax(g))
            if i == j:
                continue
            d = scale * (0.5 + rng.random())
            d = min(d, h[j], 1.0 - h[i])
            if d < 1e-10:
                continue
            h[i] += d
            h[j] -= d
            c5n = _eval(h)
            if c5n < bc:
                bc, bh = c5n, h.copy()
            else:
                h[i] -= d
                h[j] += d
    return bh, bc

def _random_local(h, deadline, seed=99):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    rng = np.random.RandomState(seed)
    niter = 30000
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 0.03 * rng.random()
        d = min(d, h[j], 1.0 - h[i])
        if d < 1e-10:
            continue
        h[i] += d
        h[j] -= d
        c5n = _eval(h)
        if c5n < bc:
            bc, bh = c5n, h.copy()
        else:
            h[i] -= d
            h[j] += d
    return bh, bc

def _sweep_polish(h, deadline, seed=7):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    rng = np.random.RandomState(seed)
    sizes = [2e-6, 1e-6, 5e-7, 2e-7, 1e-7, 5e-8]
    per = 12000
    for ss in sizes:
        for _ in range(per):
            if time.time() > deadline:
                return bh, bc
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 7)) % n
            d = ss * (2 * rng.random() - 1)
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < bc:
                    bc, bh = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    dl1 = min(deadline, t0 + 100)
    for alpha, topk, steps, s0 in [(8000, 25, 3000, 0.004), (30000, 12, 2000, 0.002), (120000, 6, 1500, 0.001)]:
        if time.time() > dl1:
            break
        hv = _nesterov(best_h.copy(), alpha, topk, steps, s0, dl1)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    dl2 = min(deadline, t0 + 300)
    hv, cv = _lp_step(best_h.copy(), n, 0.005, 6, dl2, 400)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    dl3 = min(deadline, t0 + 450)
    hv, cv = _intensive_local(best_h.copy(), dl3, seed=42)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    dl4 = min(deadline, t0 + 600)
    hv, cv = _lp_step(best_h.copy(), n, 0.003, 5, dl4, 300)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    dl5 = min(deadline, t0 + 780)
    hv, cv = _intensive_local(best_h.copy(), dl5, seed=123)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    dl6 = min(deadline, t0 + 880)
    hv, cv = _random_local(best_h.copy(), dl6, seed=77)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    rem = max(1, deadline - time.time())
    best_h, best_c5 = _sweep_polish(best_h, deadline, seed=31)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
