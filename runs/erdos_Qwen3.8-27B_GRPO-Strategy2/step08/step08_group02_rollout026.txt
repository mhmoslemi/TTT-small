```python
def _swap_update(c, h, i, j, n):
    """Update c when h[i]:1->0 and h[j]:0->1. Returns new c (modifies in place)."""
    # Effect of h[i] changing by -1:
    # For lag t where 0<=i-t<n: c[n-1+t] += -(1-h[i-t])  [but h[i-t] is the OLD value]
    # For lag t where 0<=i+t<n: c[n-1+t] += +h[i+t]
    # Effect of h[j] changing by +1:
    # For lag t where 0<=j-t<n: c[n-1+t] += +(1-h[j-t])
    # For lag t where 0<=j+t<n: c[n-1+t] += -h[j+t]
    
    # Position i (delta = -1, so h[i] goes from 1 to 0):
    lo1_i = max(0, i - (n-1))  # min t such that i-t >= 0, t >= 0 for lag
    hi1_i = min(n-1, i)         # max t such that i-t <= n-1
    # t ranges from -(n-1) to n-1
    # i-t in [0,n-1] means t in [i-(n-1), i], intersect with [-(n-1), n-1]
    t_lo = max(-(n-1), i-(n-1))
    t_hi = min(n-1, i)

        h_idx = i - np.arange(t_lo, t_hi+1)
        c[idx:idx+(t_hi-t_lo+1)] -= (1 - h[h_idx])
```

```python
def swap_delta(h, i, j, n):
    delta = np.zeros(2*n-1)
    # h[i-s] for s in [-(n-1), n-1]: h[i-s] where 0<=i-s<n
    # This is h[i-(n-1) : i+1] placed at appropriate positions
    # s goes from -(n-1) to n-1, index in delta is s+(n-1)
    # h[i-s]: when s=-(n-1), h[i+n-1] (invalid if i+n-1>=n); when s=n-1, h[i+n-1]... 
    # Let me think differently.
    # For the term h[i-s]: valid when 0<=i-s<n, i.e., i-n<s<=i, so s in [i-n+1, i]
    # In delta, this goes at index s+(n-1) for s in [i-n+1, i]
    s1 = np.arange(max(0, i-n+1), min(n, i+1))  # valid s values for h[i-s]
    idx1 = s1 + (n-1)
    delta[idx1] += h[i-s1]  # h[i-s] for valid s
    
    # For the term -h[j-s]: 
    s2 = np.arange(max(0, j-n+1), min(n, j+1))
    idx2 = s2 + (n-1)
    delta[idx2] -= h[j-s2]
    
    # For the term h[i+s]: valid when 0<=i+s<n, i.e., -i<=s<n-i
    s3 = np.arange(max(0, -i), min(n, n-i))  # wait, s can be negative
    # s in [-i, n-1-i] intersect [-(n-1), n-1]
    s3 = np.arange(max(-(n-1), -i), min(n-1, n-1-i)+1)
    idx3 = s3 + (n-1)
    delta[idx3] += h[i+s3]
    
    # For the term -h[j+s]:
    s4 = np.arange(max(-(n-1), -j), min(n-1, n-1-j)+1)
    idx4 = s4 + (n-1)
    delta[idx4] -= h[j+s4]
    
    return delta
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _swap_delta(h, i, j, n):
    """Compute change in full correlation when h[i]:1->0, h[j]:0->1."""
    delta = np.zeros(2*n-1)
    # h[i-s] for valid s: 0 <= i-s < n => s in [i-n+1, i]
    s_lo = max(0, i-n+1)
    s_hi = min(n-1, i)
    if s_lo <= s_hi:
        s = np.arange(s_lo, s_hi+1)
        delta[s + (n-1)] += h[i - s]
    # -h[j-s]
    s_lo = max(0, j-n+1)
    s_hi = min(n-1, j)
    if s_lo <= s_hi:
        s = np.arange(s_lo, s_hi+1)
        delta[s + (n-1)] -= h[j - s]
    # h[i+s] for valid s: 0 <= i+s < n => s in [-i, n-1-i]
    s_lo = max(-(n-1), -i)
    s_hi = min(n-1, n-1-i)
    if s_lo <= s_hi:
        s = np.arange(s_lo, s_hi+1)
        delta[s + (n-1)] += h[i + s]
    # -h[j+s]
    s_lo = max(-(n-1), -j)
    s_hi = min(n-1, n-1-j)
    if s_lo <= s_hi:
        s = np.arange(s_lo, s_hi+1)
        delta[s + (n-1)] -= h[j + s]
    return delta

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _greedy_binary(h, c, n, max_time, rng):
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h < 0.5)[0].copy()
    c_max = c.max()
    t_end = time.time() + max_time
    no_imp = 0
    batch = 3000
    while time.time() < t_end:
        improved = False
        for iteration in range(20):
            if time.time() >= t_end:
                break
            # Random batch of swaps
            best_new_max = c_max
            best_i = best_j = -1
            idx_i = rng.randint(len(ones), size=batch)
            idx_j = rng.randint(len(zeros), size=batch)
            for k in range(batch):
                i = ones[idx_i[k]]
                j = zeros[idx_j[k]]
                if i == j:
                    continue
                d = _swap_delta(h, i, j, n)
                nm = (c + d).max()
                if nm < best_new_max - 1e-12:
                    best_new_max = nm
                    best_i, best_j = i, j
            if best_i >= 0:
                d = _swap_delta(h, best_i, best_j, n)
                c += d
                h[best_i] = 0.0
                h[best_j] = 1.0
                # Update ones/zeros
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
                c_max = best_new_max
                improved = True
                no_imp = 0
        if not improved:
            no_imp += 20
            if no_imp > 200:
                break
    return h, c, c_max

def _anneal_binary(h, c, n, max_time, rng, T0=2.0, T1=0.01):
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h < 0.5)[0].copy()
    c_max = c.max()
    best_c_max = c_max
    best_h = h.copy()
    t0 = time.time()
    t_end = t0 + max_time
    steps = 0
    while time.time() < t_end:
        steps += 1
        T = T0 * (T1/T0) ** (steps / max(1, max_time*500))
        # Try a random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        d = _swap_delta(h, i, j, n)
        new_c = c + d
        new_max = new_c.max()
        delta = new_max - c_max
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            c = new_c
            h[i] = 0.0
            h[j] = 1.0
            c_max = new_max
            if c_max < best_c_max:
                best_c_max = c_max
                best_h = h.copy()
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
        else:
            # Revert (no change to c since we didn't apply)
            pass
    return best_h, best_c_max

def _continuous_refine(h, n, max_time, rng):
    """Short L-BFGS-B refinement with Penalty on max correlation."""
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0-x, mode='full') * dx
        return c.max()
    bnds = [(0.0, 1.0)] * n
    constr = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
              'jac': lambda x: np.ones(n)}]
    
    t0 = time.time()
    best = h.copy()
    best_val = _eval(h)
    
    # Multiple short L-BFGS-B runs
    for attempt in range(5):
        if time.time() - t0 > max_time:
            break
        x0 = best.copy()
        # Add tiny noise
        if attempt > 0:
            x0 += rng.randn(n) * 1e-3
            x0 = _repair(x0, n)
        try:
            res = minimize(obj, x0, method='L-BFGS-B', bounds=bnds,
                          constraints=constr, options={'maxiter': 50, 'ftol': 1e-14})
            h_new = _repair(res.x, n)
            v = _eval(h_new)
            if v < best_val:
                best_val = v
                best = h_new
        except:
            pass
    return best, best_val

def _local_fine(h, n, max_time, rng, step=1e-4):
    """Fine continuous local search with 2-point moves."""
    dx = 2.0 / n
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    t0 = time.time()
    while time.time() - t0 < max_time:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c
```

```python
                if nm < best_nm - 1e-12:
                    best_nm = float(nm)
                    best_i, best_j = i, j
            if best_i >= 0:
                d = _swap_delta(h, best_i, best_j, n)
                c += d
                h[best_i] = 0.0
                h[best_j] = 1.0
                c_max = best_nm
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
                improved = True
                np_stale = 0
        if not improved:
            np_stale += 30
            if np_stale > 300:
                break
    # Periodic resync
    c_sync = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(c_sync.max())
    c[:] = c_sync
    return h, c, c_max

def _anneal_binary(h, c, n, max_time, rng, T0=4.0, T1=0.05):
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h < 0.5)[0].copy()
    c_max = float(c.max())
    best_c_max = c_max
    best_h = h.copy()
    t0 = time.time()
    t_end = t0 + max_time
    total_steps = max(1, int(max_time * 400))
    step_count = 0
    buf = np.empty(2 * n - 1)
    while time.time() < t_end:
        step_count += 1
        T = T0 * (T1 / T0) ** (step_count / total_steps)
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        d = _swap_delta(h, i, j, n)
        np.add(c, d, out=buf)
        new_max = float(buf.max())
        delta_e = new_max - c_max
        if delta_e < 0 or rng.rand() < np.exp(-delta_e / max(T, 1e-10)):
            c[:] = buf
            h[i] = 0.0
            h[j] = 1.0
            c_max = new_max
            if c_max < best_c_max:
                best_c_max = c_max
                best_h = h.copy()
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
    return best_h, float(np.correlate(best_h, 1.0 - best_h, mode='full').max())

def _continuous_refine(h, n, max_time, rng):
    dx = 2.0 / n
    best = h.copy()
    best_val = _eval(h)
    t0 = time.time()
    for attempt in range(6):
        if time.time() - t0 > max_time:
            break
        x0 = best.copy()
        if attempt > 0:
            x0 += rng.randn(n) * 5e-3
            x0 = _repair(x0, n)
        def obj(x):
            return float(np.max(np.correlate(x, 1.0 - x, mode='full') * dx))
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, x0, method='L-BFGS-B', bounds=bnds,
                           constraints=cons, options={'maxiter': 80, 'ftol': 1e-14})
            hn = _repair(res.x, n)
            vn = _eval(hn)
            if vn < best_val:
                best_val = vn
                best = hn
        except Exception:
            pass
    return best, best_val

def _local_fine(h, n, max_time, rng, step=1e-4):
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    t0 = time.time()
    while time.time() - t0 < max_time:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=float)
    best_h = h0.copy()
    best_c5 = _eval(best_h)

    # Binary initialization
    h = _binary_init(h0, n, rng)
    c = np.correlate(h, 1.0 - h, mode='full')
    c5_bin = float(c.max() * 2.0 / n)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h.copy()

    # Phase 1: Greedy binary search
    if time.time() - t0 < budget_s - 500:
        t_rem = min(180, budget_s - 500 - (time.time() - t0))
        if t_rem > 10:
            h, c, c5_bin = _greedy_binary(h, c, n, t_rem, rng)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h.copy()

    # Phase 2: Binary annealing
    if time.time() - t0 < budget_s - 300:
        t_rem = min(280, budget_s - 300 - (time.time() - t0))
        if t_rem > 15:
            c = np.correlate(h, 1.0 - h, mode='full')
            h_a, c5_a = _anneal_binary(h, c, n, t_rem, rng)
            c5_a = float(np.max(np.correlate(h_a, 1.0 - h_a, mode='full')) * 2.0 / n)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = h_a.copy()

    # Phase 3: Second greedy pass from best binary
    if time.time() - t0 < budget_s - 250:
        t_rem = min(100, budget_s - 250 - (time.time() - t0))
        if t_rem > 10:
            h2 = _binary_init(best_h, n, rng)
            c2 = np.correlate(h2, 1.0 - h2, mode='full')
            h2, c2, c5_2 = _greedy_binary(h2, c2, n, t_rem, rng)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()

    # Phase 4: Continuous refinement
    if time.time() - t0 < budget_s - 80:
        t_rem = min(150, budget_s - 80 - (time.time() - t0))
        if t_rem > 5:
            h4, c5_4 = _continuous_refine(best_h.copy(), n, t_rem, rng)
            if c5_4 < best_c5:
                best_c5 = c5_4
                best_h = h4

    # Phase 5: Ultra-fine local search
    t_elapsed = time.time() - t0
    t_rem = budget_s - 5 - t_elapsed
    if t_rem > 5:
        h5, c5_5 = _local_fine(best_h.copy(), n, t_rem, rng, step=8e-5)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
```
