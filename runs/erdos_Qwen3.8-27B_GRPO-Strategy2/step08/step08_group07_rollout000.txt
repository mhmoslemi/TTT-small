def _proj(x, target):
    x = np.asarray(x, dtype=float)
    y = np.clip(x, 0.0, 1.0)
    d = target - y.sum()
    if abs(d) < 1e-13:
        return y
    if abs(d) < 5.0:
        yy = y.copy()
        dd = target - yy.sum()
        for _ in range(20):
            if abs(dd) < 1e-12:
                break
            if dd > 0.0:
                m = yy < 1.0
                c = m.sum()
                if c == 0: break
                yy[m] += dd / c
                yy[m] = np.clip(yy[m], 0.0, 1.0)
            else:
                m = yy > 0.0
                c = m.sum()
                if c == 0: break
                yy[m] += dd / c
                yy[m] = np.clip(yy[m], 0.0, 1.0)
            dd = target - yy.sum()
        if abs(dd) < 1e-11:
            return yy
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(50):
        mid = 0.5*(lo+hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5*(lo+hi), 0.0, 1.0)

def _weights(c, alpha, topk):
    L=c.size
    m=c.max()
    if topk < L:
        idx=np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx=np.arange(L)

def _adam_smooth(h, alpha, topk, lr, max_steps, deadline):
    n=len(h); target=0.5*n
    f=_eval(h); besth=h.copy(); bestf=f
    m=np.zeros(n); v=np.zeros(n); t=0; base=lr
    for _ in range(max_steps):
        if time.time() > deadline: break

        idx,w=_weights(c, alpha, topk)
        if idx.size == 0: break

        m = 0.9*m + 0.1*g
        v = 0.999*v + 0.001*g*g

def _lp_direction(h, eps, ymax, tol, max_lags):
    global _lp_dead
    if _lp_dead: return None
    try:
       n=len(h); c=_corr(h); C=c.max()
       idx=np.where(c >= C - tol)[0]
       if idx.size == 0: idx=np.array([int(c.argmax())])
       if idx.size > max_lags:
           idx=np.argpartition(c, -max_lags)[-max_lags:].copy()
       A=idx.size
       G=np.empty((A,n))
       dx=2.0/n; om=1.0-h; n1=n-1
       for a,k in enumerate(idx):
          g=G[a]; g.fill(0.0); sh=n1-int(k)
          if sh>=0:
             g[:n-sh] += om[sh:]
             if sh: g[sh:] -= h[:-sh]
          else:
             sp=-sh
             g[sp:] += om[:-sp]
             g[:n-sp] -= h[sp:]
          g *= dx
       d=np.maximum(0.0, C - c[idx])
       A_ub=np.empty((A,n+1))
       A_ub[:,:n]=G*eps
       A_ub[:,n]=-ymax
       c_obj=np.zeros(n+1); c_obj[-1]=1.0
       A_eq=np.zeros(n+1); A_eq[:n]=1.0
       lo=np.maximum(-1.0, -h/eps)
       hi=np.minimum(1.0, (1.0-h)/eps)
       bounds=np.column_stack((lo,hi))
       bounds=np.vstack((bounds, np.array([[-1.0,0.0]])))
       res=linprog(c_obj, A_ub=A_ub, b_ub=d, A_eq=A_eq, b_eq=np.array([0.0]),
                   bounds=bounds, method='highs')
       if res.status != 0:
           return None
       u=res.x[:n]
       if np.max(np.abs(u)) < 1e-12:
           return None
       return eps*u
    except Exception:
       _lp_dead=True
       return None

import numpy as np, time
from scipy.optimize import minimize, linprog

_lp_dead = False

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(x, t):
    x = np.asarray(x, dtype=float)
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(x - mid, 0.0, 1.0))) > t:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def _repair(h, n):
    h = _proj(np.asarray(h, dtype=float), 0.5 * n)
    d = 0.5 * n - float(h.sum())
    if abs(d) > 1e-13:
        m = h < 1.0 if d > 0 else h > 0.0
        if int(m.sum()) > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _sym(h):
    return _proj(0.5 * (h + h[::-1]), 0.5 * len(h))

def _fallback_h(n, seed):
    r = np.random.RandomState(seed)
    x = np.arange(n, dtype=float)
    h = 0.5 * np.ones(n)
    for m in range(1, 9):
        h += (0.45 / float(m)) * np.sin(2.0 * np.pi * m * x / n + r.uniform(0.0, 2.0 * np.pi))
    return _proj(np.clip(h, 0.0, 1.0), 0.5 * n)

def _weights(c, alpha, topk):
    m = float(c.max())
    thr = m - max(1.0 / alpha, 1e-12)
    idx = np.where(c >= thr)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(c))])
    if topk is not None and topk < idx.size:
        v = c[idx]
        idx = idx[np.argpartition(v, -topk)[-topk:]]
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -700.0, 700.0)
    sm = float(s.max())
    e = np.exp(s - sm)
    p = e / float(e.sum())
    k = p > 1e-14
    return idx[k], p[k]

def _grad_w(h, idx, w, dx):
    n = len(h)
    g = np.zeros(n)
    if idx.size == 0:
        return g
    om = 1.0 - h
    n1 = n - 1
    for ii in range(idx.size):
        wk = float(w[ii]) * dx
        if abs(wk) < 1e-18:
            continue
        sh = n1 - int(idx[ii])
        if sh >= 0:
            g[:n - sh] += wk * om[sh:]
            if sh == 0:
                g -= wk * h
            else:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * om[:-sp]
            g[:n - sp] -= wk * h[sp:]
    return g

def _smooth_fg(x, alpha, topk, lam):
    n = len(x)
    x = np.clip(x, 0.0, 1.0)
    c = _corr(x)
    m = float(c.max())
    idx, w = _weights(c, alpha, topk)
    s = np.clip(alpha * (c[idx] - m), -700.0, 700.0)
    sm = float(s.max())
    F = m + (sm + float(np.log(np.sum(np.exp(s - sm))))) / alpha
    g = _grad_w(x, idx, w, 2.0 / n)
    ss = float(x.sum() - 0.5 * n)
    if lam != 0.0 and ss != 0.0:
        F += lam * ss * ss
        g += (2.0 * lam * ss) * np.ones(n)
    return F, g

def _lbfgs_smooth(h, alpha, topk, lam, maxiter):
    n = len(h)
    try:
        res = minimize(_smooth_fg, h, args=(alpha, topk, lam), jac=True, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10, 'maxls': 20})
        return _repair(res.x, n)
    except Exception:
        return h

def _adam_smooth(h, alpha, topk, lr, max_steps, deadline):
    n = len(h)
    t0 = 0.5 * n
    f = _eval(h)
    bh, bf = h.copy(), f
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    base = lr
    for _ in range(max_steps):
        if time.time() > deadline:
            break
        c = _corr(h)
        idx, w = _weights(c, alpha, topk)
        if idx.size == 0:
            break
        g = _grad_w(h, idx, w, 2.0 / n)
        t += 1
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * g * g
        upd = (m / (1.0 - 0.9 ** t)) / (np.sqrt(v / (1.0 - 0.999 ** t)) + 1e-8)
        bf_l, bh_l = f, h
        for sc in (1.0, 0.5, 0.25, 0.125, 0.0625):
            cand = _proj(h - base * sc * upd, t0)
            fc = _eval(cand)
            if fc < bf_l - 1e-16:
                bf_l, bh_l = fc, cand
        if bf_l < f - 1e-16:
            h, f = bh_l, bf_l
            if f < bf:
                bf, bh = f, h.copy()
        else:
            base *= 0.5
            if base < 1e-9:
                break
    return _repair(bh, n), bf

def _exact_pg(h, deadline, step, tol, max_iter):
    n = len(h)
    t0 = 0.5 * n
    dx = 2.0 / n
    f = _eval(h)
    bh, bf = h.copy(), f
    base = step
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        c = _corr(h)
        C = float(c.max())
        idx = np.where(c >= C - tol)[0]
        if idx.size == 0:
            idx = np.array([int(np.argmax(c))])
        if idx.size > 200:
            idx = np.argpartition(c, -200)[-200:]
        g = _grad_w(h, idx, np.full(idx.size, 1.0 / idx.size), dx)
        if float(np.linalg.norm(g)) < 1e-14:
            break
        best_l, best_c, acc = f, h, None
        sc = base
        for _ in range(12):
            cand = _proj(h - sc * g, t0)
            fc = _eval(cand)
            if fc < best_l - 1e-15:
                best_l, best_c, acc = fc, cand, sc
                break
            sc *= 0.5
        if acc is None:
            base *= 0.5
            if base < 1e-9:
                break
        else:
            h, f = best_c, best_l
            base = min(acc * 1.03, 0.05)
            if f < bf:
                bf, bh = f, h.copy()
    return _repair(bh, n), bf

def _pair_polish(h, deadline, rng, scale, max_iter):
    n = len(h)
    dx = 2.0 / n
    f = _eval(h)
    bh, bf = h.copy(), f
    g = None
    zg = -1
    stops = (1.0, 0.6, 0.35, 0.2, 0.1, 0.05)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if g is None or it - zg >= 30:
            c = _corr(h)
            C = float(c.max())
            idx = np.where(c >= C - 1e-5)[0]
            if idx.size == 0:
                idx = np.array([int(np.argmax(c))])
            if idx.size > 120:
                idx = np.argpartition(c, -120)[-120:]
            g = _grad_w(h, idx, np.full(idx.size, 1.0 / idx.size), dx)
            zg = it
        kp = min(8, n)
        if rng.random() < 0.8:
            pos = np.argpartition(g, -kp)[-kp:]
            neg = np.argpartition(g, kp)[:kp]
            j = int(pos[int(rng.randint(len(pos)))])
            i = int(neg[int(rng.randint(len(neg)))])
        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        if i == j:
            continue
        me = min(h[j], 1.0 - h[i], scale)
        if me < 1e-12:
            continue
        bl, be = f, 0.0
        for sf in stops:
            e = me * sf
            if e <= 1e-14:
                continue
            h[i] += e
            h[j] -= e
            fc = _eval(h)
            h[i] -= e
            h[j] += e
            if fc < bl - 1e-17:
                bl, be = fc, e
        if be > 0.0:
            h[i] += be
            h[j] -= be
            f = bl
            scale = min(0.05, scale * 1.05)
            if f < bf:
                bf, bh = f, h.copy()
        else:
            scale *= 0.85
            if scale < 1e-10:
                break
    return _repair(bh, n), bf

def _lp_direction(h, eps, ymax, tol, max_lags):
    global _lp_dead
    if _lp_dead:
        return None
    try:
        n = len(h)
        if eps <= 0.0:
            return None
        c = _corr(h)
        C = float(c.max())
        idx = np.where(c >= C - tol)[0]
        if idx.size == 0:
            idx = np.array([int(np.argmax(c))])
        if idx.size > max_lags:
            idx = np.argpartition(c, -max_lags)[-max_lags:].copy()
        A = int(idx.size)
        G = np.empty((A, n))
        dx = 2.0 / n
        om = 1.0 - h
        n1 = n - 1
        for a, k in enumerate(idx):
            g = G[a]
            g.fill(0.0)
            sh = n1 - int(k)
            if sh >= 0:
                g[:n - sh] += om[sh:]
                if sh == 0:
                    g -= h
                else:
                    g[sh:] -= h[:-sh]
            else:
                sp = -sh
                g[sp:] += om[:-sp]
                g[:n - sp] -= h[sp:]
            g *= dx
        rs = 1000.0
        A_ub = np.empty((A, n + 1))
        A_ub[:, :n] = G * eps * rs
        A_ub[:, n] = -ymax * rs
        b = np.maximum(0.0, C - c[idx]) * rs
        co = np.zeros(n + 1)
        co[-1] = 1.0
        Ae = np.zeros(n + 1)
        Ae[:n] = 1.0
        lo = np.maximum(-1.0, -h / eps)
        hi = np.minimum(1.0, (1.0 - h) / eps)
        bd = np.vstack((np.column_stack((lo, hi)), [[-1.0, 0.0]]))
        try:
            res = linprog(co, A_ub=A_ub, b_ub=b, A_eq=Ae, b_eq=np.zeros(1), bounds=bd, method='highs')
        except Exception:
            res = linprog(co, A_ub=A_ub, b_ub=b, A_eq=Ae, b_eq=np.zeros(1), bounds=bd)
        if res.status != 0:
            return None
        u = np.asarray(res.x[:n], dtype=float)
        if not np.all(np.isfinite(u)) or float(np.max(np.abs(u))) < 1e-12:
            return None
        return eps * u
    except Exception:
        _lp_dead = True
        return None

def _apply_dir(h, f, d):
    n = len(h)
    bf, bh = f, h
    for sc in (1.0, 0.75, 0.5, 0.3, 0.15):
        cand = _proj(h + sc * d, 0.5 * n)
        fc = _eval(cand)
        if fc < bf - 1e-16:
            bf, bh = fc, cand
    return bh, bf

def _lp_refine(h, deadline, eps, ymax, tol, max_lags):
    n = len(h)
    f = _eval(h)
    bh, bf = h.copy(), f
    ec, yc, tc = eps, ymax, tol
    na = 0
    nn = 0
    for _ in range(100000):
        if time.time() > deadline:
            break
        d = _lp_direction(h, ec, yc, tc, max_lags)
        if d is None:
            nn += 1
            ec *= 0.7
            yc *= 0.7
            tc = max(1e-6, tc * 0.8)
            if nn > 8 or ec < 1e-6:
                break
            continue
        nn = 0
        h2, f2 = _apply_dir(h, f, d)
        if f2 < f - 1e-14:
            h, f = h2, f2
            na = 0
            if f < bf:
                bf, bh = f, h.copy()
            if f - f2 < 1e-8:
                ec = max(1e-6, ec * 0.85)
        else:
            na += 1
            ec *= 0.7
            yc *= 0.7
            tc = max(1e-6, tc * 0.9)
            if na > 20 or ec < 1e-5:
                break
    return _repair(bh, n), bf
