This is a much more targeted approach than random 2-point perturbations!

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_lse(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    c_top = c[idx]
    m = c_top.max()
    s = np.clip(alpha * (c_top - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        shift = k - (n - 1)
        wk = w[ii] * dx
        if shift >= 0:
            cnt = n - shift
            g[:cnt] += wk * (1.0 - h[shift:shift+cnt])
            if shift > 0:
                g[shift:] -= wk * h[:cnt]
        else:
            sp = -shift
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _adam(h, steps, alpha, top_k, lr, rng):
    n = len(h)
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    t = 0
    cur_c = _eval(h)
    for _ in range(steps):
        t += 1
        g = _grad_lse(h, alpha, top_k)
        m1 = 0.9 * m1 + 0.1 * g
        m2 = 0.999 * m2 + 0.001 * g * g
        mhat = m1 / (1.0 - 0.9**t)
        vhat = m2 / (1.0 - 0.999**t)
        step_lr = lr * min(1.0, 10.0 / (t + 1.0))
        h_try = _repair(h - step_lr * mhat / (np.sqrt(vhat) + 1e-8), n)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try
            cur_c = c_try
            lr = min(lr * 1.03, 0.08)
        else:
            lr = max(lr * 0.88, 0.002)
    return h, cur_c

def _local2(h, n_iter, step, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 11)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _targeted_local(h, n_iter, rng):
    n = len(h)
    dx = 2.0 / n
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(c)[-4:]
        k = top_idx[rng.randint(len(top_idx))]
        shift = k - (n - 1)
        if shift >= 0:
            cnt = n - shift
            valid = np.arange(cnt)
            overlap = h[valid] * (1.0 - h[shift:shift+cnt])
        else:
            sp = -shift
            valid = np.arange(sp, n)
            overlap = h[sp:n] * (1.0 - h[:n-sp])
        top_i = np.argsort(overlap)[-max(1, cnt//50):]
        i = valid[top_i[rng.randint(len(top_i))]]
        j = i + shift
        if 0 <= j < n:
            d = 0.005 * rng.random() * (1.0 if rng.rand() < 0.6 else -1.0)
            ni, nj = h[i] - d, h[j] + d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c_new = _eval(h)
                if c_new < cur_c:
                    cur_c = c_new
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni + d, nj - d
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m*f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i*f:(i+1)*f] = h[i]
    return out

def _block_perturb(h, n_iter, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(0, n - 3)
        d = 0.02 * rng.randn()
        h[i:i+3] += d
        h[(i+3):min(i+6, n)] -= d
        h = np.clip(h, 0.0, 1.0)
        c_new = _eval(h)
        if c_new < cur_c:
            cur_c = c_new
            h = _repair(h, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[i:i+3] -= d
            h[(i+3):min(i+6, n)] += d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el():
        return time.time() - t0

    h_parent = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))

        h_up = _repair(_upsample(hc, ratio), n)

        if c_up < best_c5:

            best_h = h_up.copy()
        best_h, best_c5 = _local2(best_h, 5000, 2e-4, rng)

    return (best_h, best_c5, n)
