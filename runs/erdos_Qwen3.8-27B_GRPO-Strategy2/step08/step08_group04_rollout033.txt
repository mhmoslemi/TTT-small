**Alternative idea: Use the exact max with a cutting-plane / ellipsoid approach.**

At each iteration:
1. Compute c_k for all k
2. Find k* = argmax
3. Take a step in the direction that reduces c_{k*}
4. Project onto feasible set

The gradient of c_{k*} w.r.t. h is well-defined (it's a bilinear form, but the gradient is straightforward). This is essentially a subgradient method on the max function.

The issue is that the max can "jump" to a different k* as we move, causing oscillation. Mitigation: use a smoothing (average of top-k gradients with geometric weights).

Let me also add a **pair coordinate descent** that's more efficient:

For each pair (i, j) where j is close to i:
- The change in c_k when we move mass d from j to i is:
  Δc_k = d * [(1-h[i+k]) - h[i-k] - (1-h[j+k]) + h[j-k]] (with range checks)
- We want to choose d to minimize max_k (c_k + d * Δc_k)
- This is a 1D convex problem

For the 1D problem, the optimal d is where the two "tightest" constraints cross. We can solve it by:
1. Sort the Δc_k values
2. The optimal d is at an intersection point of two lines c_k + d*Δc_k

But with 1119 lines, this is complex. A simpler approach: ternary search on d.

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0, 1).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - t) > 1e-9:
        h *= t / s
        np.clip(h, 0, 1, out=h)
        d = t - h.sum()
        if abs(d) > 1e-12:
            ord_ = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for i in ord_:
                if d > 0:
                    a = min(d, 1.0 - h[i]); h[i] += a; d -= a
                else:
                    a = min(-d, h[i]); h[i] -= a; d += a
                if abs(d) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        m = 0.5 * (lo + hi)
        if np.clip(h - m, 0, 1).sum() > target:
            lo = m
        else:
            hi = m
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_k(h, k):
    n = len(h)
    g = np.zeros(n)
    s = k - (n - 1)
    if s >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:n]
    else:
        sp = -s
        g[:n - sp] += 1.0 - h[sp:]
        g[sp:] -= h[:n - sp]
    return g

def _smax_val(c, idx, alpha):
    m = c[idx].max()
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad_smax(h, c, idx, alpha):
    n = len(h)
    dx = 2.0 / n
    m = c[idx].max()
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] > 1e-10:
            g += w[ii] * _grad_k(h, idx[ii])
    return g * dx

def _nesterov_smax(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    c = _correl(h)
    idx = np.argpartition(c, -topk)[-topk:]
    f = _smax_val(c, idx, alpha)
    v = np.zeros(n)
    b = 0.87
    st = step0
    best_h, best_f = h.copy(), f
    for it in range(steps):
        if time.time() > deadline:
            break
        if it % 2000 == 2000:
            b = max(0.8, b - 0.01)
        c = _correl(h)
        idx = np.argpartition(c, -topk)[-topk:]
        g = _grad_smax(h, c, idx, alpha)
        v = b * v + (1.0 - b) * g
        hn = _proj(h - st * v, n * 0.5)
        cn = _correl(hn)
        idxn = np.argpartition(cn, -topk)[-topk:]
        fn = _smax_val(cn, idxn, alpha)
        if fn < f:
            h, f = hn, fn
            st = min(st * 1.04, step0 * 100.0)
            if f < best_f:

        else:
            st *= 0.5
            hn = _proj(h - st * v, n * 0.5)
            cn = _correl(hn)
            fn = _smax_val(cn, np.argpartition(cn, -topk)[-topk:], alpha)

    return best_h

def _pair_cd(h, n_sweeps, deadline):

    f = _eval(h)

    deltas = [1, 2, 3, 4, 5, 7]
