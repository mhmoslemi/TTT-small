    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(-h) if diff > 0 else np.argsort(h)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(-h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad(h, alpha, topk):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c)).copy()
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-300:
        w = np.ones(len(idx)) / len(idx)
    else:
        w /= ws
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        L = idx[ii] - (n - 1)
        wk = w[ii] * dx
        if L == 0:
            g += wk * (1.0 - 2.0 * h)
        elif L > 0:
            g[:n - L] += wk * (1.0 - h[L:])
            g[L:] -= wk * h[:-L]
        else:
            P = -L
            g[P:] += wk * (1.0 - h[:n - P])
            g[:n - P] -= wk * h[P:]
    return g

def _smax_val(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_val(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.85
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_val(hn, alpha, topk)
        if fn < f:
            h, f = hn, fn
            step = min(step * 1.04, step0 * 8.0)
        else:
            step *= 0.5
            if step < 1e-12:
                break
            h = _proj(h - step * v, n * 0.5)
            f = _smax_val(h, alpha, topk)
    return _repair(h, n)

def _lp_descent(h, n_steps, eps, m_top, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        cor = _correl(c)
        idx_top = np.argpartition(cor, -m_top)[-m_top:]
        idx_top = idx_top[np.argsort(-cor[idx_top])]
        dx = 2.0 / n
        grads = np.zeros((m_top, n))
        for ii in range(m_top):
            L = idx_top[ii] - (n - 1)
            if L == 0:
                grads[ii] = (1.0 - 2.0 * c)
            elif L > 0:
                grads[ii, :n - L] += (1.0 - c[L:])
                grads[ii, L:] -= c[:-L]
            else:
                P = -L
                grads[ii, P:] += (1.0 - c[:n - P])
                grads[ii, :n - P] -= c[P:]
        grads *= dx
        c_obj = -grads[0]
        A_ub = grads[1:] if m_top > 1 else None
        b_ub = np.zeros(m_top - 1) if m_top > 1 else None
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bnds = [(-eps, eps)] * n
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success or res.x is None or np.max(np.abs(res.x)) < 1e-15:
            break
        d = res.x
        improved = False
        stp = 1.0
        for _ in range(10):
            h_new = _repair(c + stp * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-16:
                c, c5_cur = h_new, c5_new
                improved = True
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
                break
            stp *= 0.5
        if not improved:
            d2 = np.random.RandomState(_).randn(n)
            d2 -= d2.mean()
            d2 *= eps * 0.5 / max(np.max(np.abs(d2)), 1e-15)
            h_new = _repair(c + d2, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur:
                c, c5_cur = h_new, c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h_new.copy()
    return best_h, best_c5

def _con_fun(x, n):
    return x.sum() - n * 0.5

def _con_jac(x, n):
    return np.ones(len(x))

def _slsqp_opt(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smax_val(x, alpha, topk)
    def jac(x):
        return _grad(x, alpha, topk)
    def con_f(x):
        return x.sum() - n * 0.5
    def con_j(x):
        return np.ones(n)
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': con_f, 'jac': con_j}],
                     options={'maxiter': maxit, 'ftol': 1e-18, 'maxls': 50})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _anneal(h, niter, rng, deadline, T0, T1, bszs):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = bszs[rng.randint(len(bszs))]
        i0 = rng.randint(0, max(1, n - bs + 1))
        seg = h[i0:i0 + bs].copy()
        d = 0.005 * max(T, 1e-8) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _polish(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    sizes = [4e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]
    ni_per = max(1, niter // len(sizes))
    for ss in sizes:
        for _ in range(ni_per):
            if time.time() > deadline:
                return bh, bc
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 5)) % n
            d = ss * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5n < bc:
                        bc, bh = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc

    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: Fast Nesterov multi-parameter (200s)
    dl1 = min(deadline, t0 + 200)
    for a, tk, st in [(3000, 40, 0.006), (20000, 20, 0.004), (200000, 10, 0.002), (2000000, 5, 0.0008)]:
        if time.time() > dl1:
            break
        hv = _nesterov(best_h.copy(), a, tk, 10000, st, dl1)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 2: SLSQP refinement (150s)
    dl2 = min(deadline, t0 + 350)
    for a, tk in [(10000, 15), (100000, 8)]:
        if time.time() > dl2:
            break
        hv = _slsqp_opt(best_h.copy(), a, tk, 250, dl2)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 3: Aggressive LP descent (250s)
    dl3 = min(deadline, t0 + 600)
    hv, cv = _lp_descent(best_h.copy(), 5000, 0.003, 4, dl3)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 4: Block-swap iterated local search (180s)
    dl4 = min(deadline, t0 + 780)
    for attempt in range(12):
        if time.time() > dl4:
            break
        h_pert = best_h.copy()
        blen = rng.randint(8, 40)
        i1 = rng.randint(0, n - blen)
        i2 = rng.randint(0, n - blen)
        if abs(i1 - i2) >= blen:
            tmp = h_pert[i1:i1 + blen].copy()
            h_pert[i1:i1 + blen] = h_pert[i2:i2 + blen]
            h_pert[i2:i2 + blen] = tmp
        h_pert += 0.015 * (0.8 ** attempt) * rng.randn(n)
        h_pert = _proj(h_pert, n * 0.5)
        hv = _nesterov(h_pert, 100000, 8, 3000, 0.001, dl4)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()
        if time.time() < dl4 - 15:
            hv2, cv2 = _lp_descent(hv.copy(), 1500, 0.001, 3, dl4)
            if cv2 < best_c5:
                best_c5, best_h = cv2, hv2.copy()

    # Phase 5: Fine LP + anneal (120s)
    dl5 = min(deadline, t0 + 900)
    hv, cv = _lp_descent(best_h.copy(), 3000, 0.001, 3, dl5)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()
    if time.time() < dl5 - 10:
        hv, cv = _anneal(best_h.copy(), 20000, rng, dl5, 0.003, 1e-10, [2, 3, 4, 5, 6, 8])
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 6: Polish (remaining)
    rem = max(0.1, deadline - time.time())
    npol = min(int(rem / 0.001), 50000)
    best_h, best_c5 = _polish(best_h, max(npol, 3000), rng, deadline)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
