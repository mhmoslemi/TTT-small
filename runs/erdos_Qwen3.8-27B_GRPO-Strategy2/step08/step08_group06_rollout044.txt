def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _subgrad(c, h):
    """Subgradient of max(c) w.r.t. h, where c_k = sum_i h_i(1-h_{i+k})"""
    n = len(h)
    k_max = np.argmax(c)
    dx2 = 2.0 / n
    sh = (n - 1) - k_max
    g = np.zeros(n)
    if sh >= 0:
        # c = sum_{i=0}^{n-1-sh} h[i+sh] * (1-h[i]) * dx2
        g[sh:] += dx2 * (1.0 - h[:n-sh])
        g[:n-sh] -= dx2 * h[sh:]
    else:
        sp = -sh
        g[sp:] += dx2 * (1.0 - h[:n-sp])
        g[:n-sp] -= dx2 * h[sp:]
    return g

    n = len(h)

    w = np.exp(s)

            g[:n-sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

    """Frank-Wolfe linear minimization: min g^T h' s.t. 0<=h'<=1, sum h'=target"""
    idx = np.argsort(g)
    h_lin = np.zeros(n)
    # We need to pick n/2 entries (fractional) to set to 1
    # Sort g ascending, set smallest to 1
    # target = n/2, so we set the smallest target entries to 1
    # But target might not be integer... for n=560, target=280
    k_full = int(target)
    frac = target - k_full
    h_lin[idx[:k_full]] = 1.0
    if frac > 1e-12:
        h_lin[idx[k_full]] = frac
    return h_lin

def _frank_wolfe(h, alpha, n_iter, t_end):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    f = _eval(h)
    best_h, best_f = h.copy(), f
    gamma = 2.0
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = _corr(h)
        g = _subgrad_weighted(c, h, alpha)
        h_lin = _fw_step(g, n, target)
        gamma_t = 2.0 / (it + 2)
        h_new = (1 - gamma_t) * h + gamma_t * h_lin
        h_new = _repair(h_new, n)
        f_new = _eval(h_new)
        if f_new < best_f:

            h = (1 - gamma_t) * best_h + gamma_t * h_lin
            h = _repair(h, n)
        f = _eval(h)
    return best_h, best_f

def _sa_local(h, rng, t_end, n_init=3000, T0=0.002, T1=1e-6):
    """Simulated annealing local search"""
    n = len(h)
    h = h.copy()
    f = _eval(h)
    best_h, best_f = h.copy(), f
    for it in range(n_init):
        if time.time() > t_end:
            break
        T = T0 * (T1 / T0) ** (it / n_init)
        # Random 2-4 coordinate perturbation
        ncoord = rng.randint(2, 5)
        idxs = rng.choice(n, ncoord, replace=False)
        deltas = rng.randn(ncoord) * 0.01
        # Ensure sum preservation
        deltas -= deltas.mean()
        h_new = _repair(h + np.where(np.arange(n)[:,None] == idxs[None,:], deltas[None,:], 0), n)
        f_new = _eval(h_new)
        if f_new < f or rng.rand() < np.exp(-(f_new - f) / T):
            h = h_new
            f = f_new
            if f < best_f:
                best_f, best_h = f, h.copy()
    return best_h, best_f

def _coordinate_descent(h, n_rounds, t_end):
    """Coordinate descent with exact line search"""
    n = len(h)
    h = h.copy()
    f = _eval(h)
    best_h, best_f = h.copy(), f
    for rnd in range(n_rounds):

                break
            if abs(g[i]) < 1e-10:
                continue
            # Try moving h[i] in the gradient descent direction
            direction = -np.sign(g[i])
            # Binary search for best step
            lo, hi = 0.0, 0.05
            best_val = f
            best_step = 0.0
            for _bs in range(12):
                mid = (lo + hi) * 0.5
                h[i] += direction * mid
                if h[i] < 0 or h[i] > 1:
                    hi = mid
                    continue

                h_r = _repair(h, n)
                if not np.allclose(h_r, h, atol=1e-14):

                    h[i] -= direction * mid
                    hi = mid
                    continue

                if f_mid < best_val:
                    best_val = f_mid
                    best_step = mid
                    lo = mid
                else:

            if best_val < f:
                f = best_val
                if f < best_f:
                    best_f, best_h = f, h.copy()
            h = _repair(h, n)
    return best_h, best_f

    m = c.max()
    s = alpha * (c - m)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_weighted(w, h):
    n = len(h)
    g = np.zeros(n)
    dx2 = 2.0 / n
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        wk = w[k] * dx2
        sh = (n - 1) - k
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n-sh])
            g[:n-sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _grad_lse(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    return _grad_weighted(w, h)

def _lse_obj(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)

    h = h.copy()
    f = _eval(h)
    best_h, best_f = h.copy(), f
    for it in range(n_iter):
        if time.time() > t_end:
            break
        g = _grad_lse(h, alpha)
        h_v = _fw_vertex(g, n)
        best_t, best_f_new = 0.0, f
        t = 2.0 / (it + 2)
        for _bt in range(18):
            h_mid = (1.0 - t) * h + t * h_v
            h_mid = _repair(h_mid, n)
            f_mid = _eval(h_mid)
            if f_mid < best_f_new:

        h = _repair(h, n)
        f = _eval(h)
        if f < best_f:
            best_f, best_h = f, h.copy()
    return best_h, best_f

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(len(x))

def _slsqp_lse(h, alpha, maxiter, n):

    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]

def _sa_search(h, rng, t_end, n_trials, T0, T1, pert_scale):
    n = len(h)
    h = h.copy()
    f = _eval(h)
    best_h, best_f = h.copy(), f
    for it in range(n_trials):
        if time.time() > t_end:
            break
        T = T0 * (T1 / T0) ** (it / max(n_trials, 1))
        nc = rng.randint(2, 5)
        idxs = rng.choice(n, nc, replace=False)
        deltas = rng.randn(nc) * pert_scale
        deltas -= deltas.mean()
        diff = np.zeros(n)
        diff[idxs] = deltas
        h_new = _repair(h + diff, n)
        f_new = _eval(h_new)
        if f_new < f or rng.rand() < np.exp(max(-(f_new - f) / max(T, 1e-12), -500)):
            h, f = h_new, f_new
            if f < best_f:
                best_f, best_h = f, h.copy()
    return best_h, best_f

def _coord_descent(h, n_rounds, t_end):
    n = len(h)
    h = h.copy()
    f = _eval(h)
    best_h, best_f = h.copy(), f
    for rnd in range(n_rounds):
        if time.time() > t_end:
            break
        g = _grad_lse(h, 5000)
        perm = np.arange(n)
        for i in perm:
            if time.time() > t_end:
                break
            if abs(g[i]) < 1e-12:
                continue
            d = -np.sign(g[i])
            lo, hi = 0.0, 0.03
            best_step, best_val = 0.0, f
            for _bs in range(10):
                mid = (lo + hi) * 0.5
                h[i] += d * mid
                if h[i] < 0 or h[i] > 1:
                    hi = mid
                    h[i] -= d * mid
                    continue
                h_r = _repair(h, n)
                if not np.allclose(h_r, h, atol=1e-12):
                    h[i] -= d * mid
                    hi = mid
                    continue

                if f_mid < best_val:

                    lo = mid

            h[i] -= d * best_step
            h = _repair(h, n)
            if best_val < f:
                f = best_val
                if f < best_f:
                    best_f, best_h = f, h.copy()
    return best_h, best_f

def _nesterov_topk(h, k_top, steps, step0, t_end):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.8
    step = step0
    for _s in range(steps):
        if time.time() > t_end:
            break
        c = _corr(h)
        top_k = np.argsort(c)[-k_top:]
        w = np.zeros(2 * n - 1)
        w[top_k] = 1.0 / k_top
        g = _grad_weighted(w, h)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _repair(h - step * v, n)
        f_new = _eval(h_new)
        if f_new < f:
            h, f = h_new, f_new
            step = min(step * 1.03, step0 * 8.0)
        else:
            step *= 0.4
            if step < 1e-12:
                v = np.zeros(n)
                step = step0 * 0.5
            h = _repair(h_prev - step * v, n)
            f = _eval(h)
    return _repair(h, n), f

    c5v = _eval(h_new)
    if c5v < best_c5:
        return h_new.copy(), c5v
    return best_h, best_c5
