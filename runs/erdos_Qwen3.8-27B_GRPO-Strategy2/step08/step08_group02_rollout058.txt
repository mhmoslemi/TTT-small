            cs = c

        s -= s.max()

            cs = c

        s = np.clip(alpha * (cs - m), -300, 300)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] < 1e-14:
                continue
            g += w[ii] * _grad_single(idx[ii], x, n)
        g += 1000.0 * (x.sum() - n / 2.0)
        return g

                                           'gtol': 1e-14, 'maxfun': maxiter * 3,

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)

        h *= target / s
        h = np.clip(h, 0.0, 1.0)

        ni, nj, nk = h[i]+d1, h[j]+d2, h[k]-d1-d2
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0 and 0.0 <= nk <= 1.0:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c, best_h = c5n, h.copy()
            else:
                h[i], h[j], h[k] = ni-d1, nj-d2, nk+d1+d2
    return best_h, best_c

def _downsample(h, factor):
    n = len(h)
    m = n // factor
    return h[:m*factor].reshape(m, factor).mean(axis=1).copy()

def _upsample(h, factor):
    n = len(h)
    out = np.empty(n * factor)
    for i in range(n):
        out[i*factor:(i+1)*factor] = h[i]
    return out

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el():
        return time.time() - t0

    n = len(initial_h_values)
    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _eval(best_h)

    # Phase 0: Coarse-to-fine
    for ratio in [4, 2]:
        if _el() > 30:
            break
        nc = max(24, n // ratio)
        hc = _repair(_downsample(best_h, n // nc), nc)
        hc = _lbbfgs(hc, 500, 8, 100)
        h_up = _repair(_upsample(hc, n // nc), n)
        if _eval(h_up) < best_c5:
            best_c5 = _eval(h_up)
            best_h = h_up.copy()

    # Phase 1: L-BFGS-B with increasing sharpness
    for alpha, tk, mit in [(50, 30, 60), (200, 15, 80), (1000, 8, 80),
                           (5000, 5, 70), (20000, 3, 60), (100000, 2, 50)]:
        if _el() > 55:
            break
        h_new = _lbbfgs(best_h, alpha, tk, mit)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5, best_h = c5n, h_new.copy()

    # Phase 2: Focal momentum gradient
    for ni, tk, al, s0, mu in [(300, 8, 500, 0.008, 0.7),
                                (400, 6, 2000, 0.006, 0.75),
                                (500, 5, 8000, 0.004, 0.8),
                                (600, 4, 30000, 0.003, 0.8),
                                (800, 3, 100000, 0.002, 0.85),
                                (1000, 2, 500000, 0.001, 0.85)]:
        if _el() > 320:
            break
        h_new, c_new = _focal_momentum(best_h, ni, tk, al, s0, mu)
        if c_new < best_c5:
            best_c5, best_h = c_new, h_new.copy()

    # Phase 3: Second L-BFGS-B round
    if _el() < budget_s - 300:
        for alpha, tk, mit in [(50000, 2, 50), (200000, 2, 40), (1000000, 1, 30)]:
            if _el() > budget_s - 290:
                break
            h_new = _lbbfgs(best_h, alpha, tk, mit)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5, best_h = c5n, h_new.copy()

    # Phase 4: More focal descent
    if _el() < budget_s - 220:
        for ni, tk, al, s0, mu in [(1500, 2, 500000, 0.0005, 0.9),
                                    (2000, 1, 2000000, 0.0003, 0.9)]:
            if _el() > budget_s - 210:
                break
            h_new, c_new = _focal_momentum(best_h, ni, tk, al, s0, mu)
            if c_new < best_c5:
                best_c5, best_h = c_new, h_new.copy()

    # Phase 5: Worst-lag local search
    if _el() < budget_s - 100:
        time_left = budget_s - 100 - _el()
        _t = _el()
        for _ in range(30):
            _eval_worst(best_h)
        eps = max((_el() - _t) / 30.0, 1e-6)
        ntries = min(int(time_left * (1.0/eps) * 0.55), 3000000)
        best_h, best_c5 = _local_worst(best_h, ntries, rng, 4e-6)

    # Phase 6: 3-point local search
    if _el() < budget_s - 50:
        time_left = budget_s - 50 - _el()
        _t = _el()
        for _ in range(30):
            _eval(best_h)
        eps = max((_el() - _t) / 30.0, 1e-6)
        ntries = min(int(time_left * (1.0/eps) * 0.5), 2000000)
        best_h, best_c5 = _local_3pt(best_h, ntries, rng, 2e-6)

    # Phase 7: 2-point ultra-fine
    if _el() < budget_s - 3:
        time_left = budget_s - 3 - _el()
        _t = _el()
        for _ in range(30):
            _eval(best_h)
        eps = max((_el() - _t) / 30.0, 1e-6)
        npol = min(int(time_left * (1.0/eps) * 0.7), 8000000)
        best_h, best_c5 = _local_2pt(best_h, npol, rng, 5e-8)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
