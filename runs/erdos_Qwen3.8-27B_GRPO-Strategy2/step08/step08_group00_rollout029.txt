    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:

    # Proportional adjustment
    diff = target - s
    # Distribute to non-saturated entries
    free = (h > 1e-10) & (h < 1 - 1e-10)
    if free.sum() == 0:
        # All saturated, need to pick which to adjust

            idx = np.argsort(h)[:int(abs(diff))+1]
            for i in idx:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add

            idx = np.argsort(-h)[:int(abs(diff))+1]
            for i in idx:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub

    if diff > 0:
        room = (1.0 - h[free]).sum()
        if room < diff:
            h[free] = 1.0
            diff -= room
            free2 = ~free
            room = (1.0 - h[free2]).sum()
            if room > 0:
                adj = min(1.0, diff / room)

    s = h.sum()

import numpy as np
from scipy.optimize import minimize, minimize_scalar

def _repair(h, n):
    """Clip to [0,1] and adjust sum to n/2"""
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not mask.any():
                break
            room = (1.0 - h[mask]).sum()
            frac = min(1.0, diff / room) if room > 0 else 0
            h[mask] = np.minimum(h[mask] + (1.0 - h[mask]) * frac, 1.0)
        else:
            mask = h > 1e-15
            if not mask.any():
                break
            h[mask] = np.maximum(h[mask] * (1.0 + diff / h[mask].sum()), 0.0)

    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)

        elif abs(L) < n:
            M = abs(L)
            g[:n-M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n-M]
    return g

def _sum_fn(x, half_n):
    return x.sum() - half_n

def _sum_jac(x, half_n):
    return np.ones(len(x))

def _slsqp_h(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

def _nesterov_pg(h, alpha, m_lags, n_iter, step0):
    n = len(h)
    target = n / 2.0
    x0 = h.copy()
    x1 = h.copy()
    y = h.copy()
    t = 1.0
    step = step0
    mu = 0.9
    v = np.zeros(n)
    cur_c = _eval_max(h)
    
    for it in range(n_iter):
        # Nesterov interpolation
        t_prev = t
        t = (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev)) / 2.0
        beta = (t_prev - 1.0) / t
        y = x1 + beta * (x1 - x0)

        y_clip = np.clip(y, 0, 1)
        g = _smooth_grad(y_clip, alpha, m_lags)

        v = mu * v + (1.0 - mu) * g

        taken = False

                x1 = x_new
                cur_c = c_new

                taken = True

            step *= 0.5

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            m = h < 1.0 - 1e-15
            if not m.any():
                break
            r = (1.0 - h[m]).sum()
            if r <= 0:
                break
            h[m] = np.minimum(h[m] + (1.0 - h[m]) * min(1.0, diff/r), 1.0)
        else:
            m = h > 1e-15
            if not m.any():
                break
            r = h[m].sum()
            if r <= 0:
                break
            h[m] = np.maximum(h[m] * max(0.0, 1.0 + diff/r), 0.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            a = min(abs(diff), 1.0 - h[idx]) if diff > 0 else min(-diff, h[idx])
            h[idx] += a if diff > 0 else -a
            diff += -a if diff > 0 else a
            if abs(diff) < 1e-13:
                break
    return h

def _eval_c(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval_max(h):
    return float(np.max(_eval_c(h)))

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _softmax_obj(x, alpha, m_lags):
    c = _eval_c(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400.0, 400.0)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _softmax_grad(x, alpha, m_lags):
    n = len(x)
    c = _eval_c(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400.0, 400.0)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        wk = w[ii]
        L = idx[ii] - (n - 1)
        if L == 0:
            g -= wk * 2.0 * x
        else:
            M = abs(L)
            if M < n:
                g[:n - M] += wk * (1.0 - x[M:])
                g[M:] -= wk * x[:n - M]
    return g

def _con_fun(x, half_n):
    return x.sum() - half_n

def _con_jac(x, half_n):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_softmax_obj, h, jac=_softmax_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _nesterov(h, alpha, m_lags, n_iter, step0, t_end):
    n = len(h)
    tgt = n / 2.0
    x0 = h.copy()
    x1 = h.copy()
    t = 1.0
    step = step0
    mu = 0.85
    v = np.zeros(n)
    cur_c = _eval_max(h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        tp = t
        t = (1.0 + np.sqrt(1.0 + 4.0 * tp * tp)) / 2.0
        beta = (tp - 1.0) / t
        y = x1 + beta * (x1 - x0)
        yc = np.clip(y, 0.0, 1.0)
        g = _softmax_grad(yc, alpha, m_lags)
        v = mu * v + (1.0 - mu) * g
        taken = False
        for _ in range(8):
            xn = _proj(y - step * v, tgt)
            cn = _eval_max(xn)
            if cn < cur_c - 1e-14:
                x0 = x1
                x1 = xn
                cur_c = cn
                step = min(step * 1.06, step0 * 30.0)
                taken = True
                break
            step *= 0.5
        if not taken:
            x0 = x1
            step = min(step * 1.3, step0 * 30.0)
            if step < 1e-12:
                break
    return _repair(x1, n), cur_c

def _local_block(h, n_iter, rng, bs, t_end):
    n = len(h)
    bh = h.copy()
    bc = _eval_max(bh)
    cc = bc
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        s = rng.randint(0, n - bs + 1)
        hn = bh.copy()
        hn[s:s + bs] += 0.01 * rng.randn(bs)
        hn = _repair(hn, n)
        cn = _eval_max(hn)
        if cn < cc - 1e-10:
            cc = cn
            bh = hn
            if cn < bc:
                bc = cn
    return bh, bc

def _two_point(h, n_iter, rng, t_end):
    n = len(h)
    bh = h.copy()
    bc = _eval_max(bh)
    cc = bc
    hc = h.copy()
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 10)) % n
        d = 5e-7 * rng.randn()
        ni, nj = hc[i] + d, hc[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            hc[i], hc[j] = ni, nj
            cn = _eval_max(hc)
            if cn < cc:
                cc = cn
                if cn < bc:
                    bc = cn
                    bh = hc.copy()
            else:
                hc[i], hc[j] = ni - d, nj + d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def _el():
