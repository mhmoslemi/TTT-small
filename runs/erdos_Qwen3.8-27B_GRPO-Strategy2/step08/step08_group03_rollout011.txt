def _fps(n):
    m = 2*n - 1
    N = 1
    while N < m: N <<= 1
    return N

def _repair(h, n):
    S = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(60):
        s = h.sum()
        if abs(s - S) < 1e-12: break
        h *= S / max(s, 1e-15)
        np.clip(h, 0.0, 1.0, out=h)
    d = S - h.sum()
    if abs(d) > 1e-12:
        o = np.argsort(1-h if d>0 else h)
        for i in o:
            a = min(abs(d), 1-h[i] if d>0 else h[i])
            h[i] += np.sign(d)*a
            d -= np.sign(d)*a

def _proj(v, S):

    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fps(n)
        H = np.fft.rfft(h, Nf)
        q = -(np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)[n - 1:2 * n - 1].real
              + np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)[:n].real)
    return dx * (wP[0] * (1 - 2 * h) + linP + linN + q)

def _sg(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(a * (P[0] - mx), -400, 0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(a * (P[1:] - mx), -400, 0))
        en = np.exp(np.clip(a * (Nv[1:] - mx), -400, 0))
        tot += ep.sum() + en.sum()
        if tot <= 0:
            tot, e0 = 1.0, 1.0
        wP[0] = e0 / tot
        wP[1:] = ep / tot
        wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _gw(h, wP, wN)

def _fista(h0, a, mt, L=12.0):
    n = len(h0)
    S = n * 0.5
    t0 = time.time()
    h = _repair(h0, n)
    hp = h.copy()
    y = h.copy()
    t = 1.0
    fy = _sw(h, a)
    best_h = h.copy()
    best_c = _c5(h)
    stale = 0
    LC = L
    for it in range(200000):
        if time.time() - t0 > mt:
            break
        if it:
            tn = 0.5 * (1 + np.sqrt(1 + 4 * t * t))
            y = h + ((t - 1) / tn) * (h - hp)
            t = tn
            if y.min() < -1e-5 or y.max() > 1 + 1e-5 or abs(y.sum() - S) > 1e-8:
                y = _proj(y, S)
            fy = _sw(y, a)
        else:
            y = h
        g = _sg(y, a)
        step = 1.0 / max(LC, 1e-12)
        ok = False
        for _ in range(12):
            xn = _proj(y - step * g, S)
            fx = _sw(xn, a)
            d = xn - y
            if fx <= fy + np.dot(g, d) + 0.5 / step * np.dot(d, d) + 1e-11:
                ok = True
                break
            step *= 0.5
        LC = 1.0 / max(step, 1e-12)
        if ok:
            hp = h
            h = xn
            stale = 0
            if it % 8 == 0:
                c = _c5(h)
                if c < best_c:
                    best_c, best_h = c, h.copy()
        else:
            stale += 1
            hp = h
            y = h
            t = 1.0
            fy = _sw(h, a)
            LC = max(LC * 0.5, 1.0)
            if stale > 12:
                break
    c = _c5(h)
    if c < best_c:
        best_c, best_h = c, h.copy()
    return h, best_h, best_c

def _agrad(h, P, Nv):
