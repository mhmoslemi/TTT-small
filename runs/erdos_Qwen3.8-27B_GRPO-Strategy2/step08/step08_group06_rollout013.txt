def _project(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Binary search for shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        shifted = np.clip(h - mid, 0, 1)
        if shifted.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Fine adjustment
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n

def _coord_descent(h, n_tries, rng, t_end):
    """Coordinate descent: pick 2 indices, transfer mass"""
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    current_c = best_c
    for _ in range(n_tries):

                if c < best_c:

    return best_h, best_c

def _slsqp(h, alpha, maxiter, target, t_end):

        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

        return g

import numpy as np
from scipy.optimize import minimize
import time


def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    if h.sum() < 1e-14:
        return np.full(n, 0.5)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h


def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _grad_active(h, K):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    idx = np.argsort(c)[-K:]
    g = np.zeros(n)
    w = dx / K
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += w * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= w * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += w * (1.0 - h[:n - sp])
            g[:n - sp] -= w * h[sp:]
    return g, max(c) * dx


class _LSE:
    def __init__(self, alpha, n):
        self.alpha = alpha
        self.n = n

    def objective(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / self.n)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s - s.max()))) / self.alpha

    def gradient(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / self.n)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(self.n)
        dx2 = 2.0 / self.n
        for k in range(2 * self.n - 1):
            if w[k] < 1e-14:
                continue
            sh = (self.n - 1) - k
            wk = w[k] * dx2
            if sh >= 0:
                g[:self.n - sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk * x[:self.n - sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:self.n - sp])
                g[:self.n - sp] -= wk * x[sp:]
        return g


class _SumCon:
    def __init__(self, n, target):
        self.n = n
        self.target = target
        self._ones = np.ones(n)

    def fun(self, x):
        return x.sum() - self.target

    def jac(self, x):
        return self._ones


def _fista(h, K, steps, step0, t_end):
    n = len(h)
    x = h.copy()
    y = h.copy()
    f = _eval(h)
    t_val = 1.0
    step = step0
    for s in range(steps):
        if time.time() > t_end:
            break
        g, _ = _grad_active(y, K)
        x_new = _project(y - step * g, n)
        f_new = _eval(x_new)
        if f_new < f - 1e-16:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            y = _project(x_new + ((t_val - 1.0) / t_new) * (x_new - x), n)
            x, f = x_new, f_new
            t_val = t_new
            step = min(step * 1.015, step0 * 8.0)
        else:
            step *= 0.5
            if step < 1e-15:
                step = step0 * 0.2
                y = x.copy()
                t_val = 1.0
            else:
                x_new = _project(y - step * g, n)
                f_new = _eval(x_new)
                if f_new < f:
                    x, f = x_new, f_new
                y = x.copy()
                t_val = 1.0
    return x


def _slsqp_opt(h, alpha, maxiter, n, target, t_end):
    lse = _LSE(alpha, n)
    cons = [{'type': 'eq', 'fun': _SumCon(n, target).fun, 'jac': _SumCon(n, target).jac}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_LSE(alpha, n).objective, h, jac=_LSE(alpha, n).gradient,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if time.time() < t_end:
            return _project(res.x, n)
    except Exception:
        pass
    return _project(h, n)


def _coord_descent(h, n_tries, rng, t_end, scale):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    cur_c = best_c
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 30)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c = _eval(h)
            if c < cur_c:
                cur_c = c
                if c < best_c:
                    best_c, best_h = c, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c


def _focal_restart(h, K, t_end):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    active = np.argsort(c)[-K:]
    m = c.max()
    pressure = np.zeros(n)
    w = 2.0 / (n * K)
    for k in active:
        sh = (n - 1) - k
        if sh >= 0:
            pressure[:n - sh] += w * (1.0 - h[sh:])
            if sh > 0:
                pressure[sh:] -= w * h[:n - sh]
        else:
            sp = -sh
            pressure[sp:] += w * (1.0 - h[:n - sp])
            pressure[:n - sp] -= w * h[sp:]
    h_new = _project(h - 0.01 * pressure, n)
    if _eval(h_new) < _eval(h):
        return h_new
    return h
