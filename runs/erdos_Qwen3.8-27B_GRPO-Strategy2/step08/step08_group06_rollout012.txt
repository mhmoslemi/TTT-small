import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(60):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:

    return h

    idx = np.argsort(c)[-k_top:]

    dx = 2.0 / n
    w = dx / k_top

        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += w * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= w * h[:n-sh]

            sp = -sh
            g[sp:] += w * (1.0 - h[:n-sp])
            g[:n-sp] -= w * h[sp:]
    return g

    x = np.arange(n) * 2.0 / n
    p = np.zeros(2 * K)
    for k in range(1, K + 1):
        p[2*(k-1)] = 2.0 * np.mean((h - 0.5) * np.cos(np.pi * k * x))
        p[2*(k-1)+1] = 2.0 * np.mean((h - 0.5) * np.sin(np.pi * k * x))

    p0 = _proj_fourier(h, n, K)
    c0 = _eval(h)
    best_p = p0.copy()
    best_c = c0

        return _eval(hv)

def _block_anneal(h, n, t_start, t_end, rng, T0, T1):

    best_h, best_c = h.copy(), c_cur
    n_steps = 0
    while time.time() < t_end:

    lr = 0.005

    for _ in range(steps):

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(60):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _topk_subgrad(h, k_top, n):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    idx = np.argsort(c)[-k_top:]
    g = np.zeros(n)
    dx = 2.0 / n
    w = dx / k_top
    for k in idx:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += w * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= w * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += w * (1.0 - h[:n - sp])
            g[:n - sp] -= w * h[sp:]
    return g

def _proj_fourier(h, n, K):
    x = np.arange(n) * 2.0 / n
    p = np.zeros(2 * K)
    for k in range(1, K + 1):
        p[2 * (k - 1)] = 2.0 * np.mean((h - 0.5) * np.cos(np.pi * k * x))
        p[2 * (k - 1) + 1] = 2.0 * np.mean((h - 0.5) * np.sin(np.pi * k * x))
    return p

def _fwd_fourier(p, n, K):
    x = np.arange(n) * 2.0 / n
    h = 0.5 * np.ones(n)
    for k in range(1, K + 1):
        h += p[2 * (k - 1)] * np.cos(np.pi * k * x)
        h += p[2 * (k - 1) + 1] * np.sin(np.pi * k * x)
    return h

def _fourier_search(h_start, n, K, t_end, rng):
    p0 = _proj_fourier(h_start, n, K)
    best_p = p0.copy()
    best_c = _eval(h_start)
    nd = 2 * K
    for attempt in range(6):
        if time.time() > t_end:
            break
        if attempt == 0:
            p_try = best_p.copy()
        else:
            scale = 0.003 * (1.0 + 0.7 * attempt)
            p_try = best_p + rng.randn(nd) * scale
        def obj_f(pv):
            hv = _repair(_fwd_fourier(pv, n, K), n)
            return _eval(hv)
        try:
            res = minimize(obj_f, p_try, method='Nelder-Mead',
                           options={'maxiter': 1500, 'xatol': 1e-11, 'fatol': 1e-13})
            hv = _repair(_fwd_fourier(res.x, n, K), n)
            cv = _eval(hv)
            if cv < best_c:
                best_c = cv
                best_p = res.x.copy()
        except Exception:
            pass
    return _repair(_fwd_fourier(best_p, n, K), n)

def _subgrad_polish(h, n, k_top, max_steps, lr0, t_end):
    lr = lr0
    c = _eval(h)
    for _ in range(max_steps):
        if time.time() > t_end:
            break
        g = _topk_subgrad(h, k_top, n)
        h_new = _repair(h - lr * g, n)
        c_new = _eval(h_new)
        if c_new < c:
            h = h_new
            c = c_new
            lr = min(lr * 1.03, 0.08)
        else:
            lr *= 0.45
            if lr < 1e-10:
                break
    return _repair(h, n)

def _block_anneal(h, n, t_start, t_end, rng, T0, T1):
    c_cur = _eval(h)
    best_h, best_c = h.copy(), c_cur
    while time.time() < t_end:
        elapsed = time.time() - t_start
        total = max(t_end - t_start, 1e-8)
        frac = min(1.0, elapsed / total)
        T = T0 * (T1 / T0) ** frac
        L = int(rng.choice([3, 5, 7, 10, 15, 20]))
        i = int(rng.randint(0, max(n - L, 1)))
        blk = h[i:i + L].copy()
        blk_sum = blk.sum()
        delta = rng.randn(L)
        delta -= delta.mean()
        new_blk = np.clip(blk + 0.12 * T * delta, 0.0, 1.0)
        diff = blk_sum - new_blk.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - new_blk) if diff > 0 else np.argsort(new_blk)
            for j in order:
                if diff > 0:
                    a = min(diff, 1.0 - new_blk[j]); new_blk[j] += a; diff -= a
                else:
                    a = min(-diff, new_blk[j]); new_blk[j] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
        h[i:i + L] = new_blk
        c_new = _eval(h)
        dc = c_new - c_cur
        if dc < 0 or (T > 1e-10 and rng.rand() < np.exp(-dc / T)):
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
        else:
            h[i:i + L] = blk
    return best_h, best_c

def _fine_coord(h, n, t_end, rng, scale):
    c = _eval(h)
    best_h, best_c = h.copy(), c
    while time.time() < t_end:
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        d = rng.randn() * scale
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < c:
                c = c_new
                if c < best_c:
                    best_c, best_h = c, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end_g = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Quick subgradient warmup from parent
    h1 = _subgrad_polish(best_h.copy(), n, 3, 3000, 0.01, t0 + budget_s * 0.06)
    c1 = _eval(h1)
    if c1 < best_c5:
        best_c5, best_h = c1, h1

    # Phase 2: Fourier optimization from best
    h2 = _fourier_search(best_h.copy(), n, 35, t0 + budget_s * 0.22, rng)
    c2 = _eval(h2)
    if c2 < best_c5:
        best_c5, best_h = c2, h2

    # Phase 3: Block annealing (escapes local minima)
    h3, c3 = _block_anneal(best_h.copy(), n, time.time(),
                           t0 + budget_s * 0.42, rng, 0.08, 0.002)
    if c3 < best_c5:
        best_c5, best_h = c3, h3

    # Phase 4: Subgradient refinement on best
    h4 = _subgrad_polish(best_h.copy(), n, 2, 4000, 0.005, t0 + budget_s * 0.52)
    c4 = _eval(h4)
    if c4 < best_c5:
        best_c5, best_h = c4, h4

    # Phase 5: Fourier with more modes from current best
    h5 = _fourier_search(best_h.copy(), n, 50, t0 + budget_s * 0.62, rng)
    c5 = _eval(h5)
    if c5 < best_c5:
        best_c5, best_h = c5, h5

    # Phase 6: Short block anneal for final escape
    h6, c6 = _block_anneal(best_h.copy(), n, time.time(),
                           t0 + budget_s * 0.75, rng, 0.04, 0.001)
    if c6 < best_c5:
        best_c5, best_h = c6, h6

    # Phase 7: Fine subgradient polish
    h7 = _subgrad_polish(best_h.copy(), n, 2, 3000, 0.002, t0 + budget_s * 0.82)
    c7 = _eval(h7)
    if c7 < best_c5:
        best_c5, best_h = c7, h7

    # Phase 8: Ultra-fine coordinate descent
    h8, c8 = _fine_coord(best_h.copy(), n, t_end_g - 10, rng, 2e-5)

    h9, c9 = _fine_coord(best_h.copy(), n, t_end_g - 1, rng, 5e-6)
    if c9 < best_c5:
        best_c5, best_h = c9, h9

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
