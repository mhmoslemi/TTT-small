def _adam_smooth(h, alpha0, alpha_f, n_steps, top_k, lr):
    n = len(h)
    dx = 2.0 / n
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c = _eval(h)
    for t in range(1, n_steps+1):
        alpha = alpha0 * (alpha_f/alpha0) ** (t/n_steps)
        c = np.correlate(h, 1.0-h, mode='full') * dx
        if top_k > 0:
            idx = np.argsort(c)[-top_k:]
            cs = c[idx]
        else:
            idx = np.arange(len(c))
            cs = c
        m_a = cs.max()
        s = np.clip(alpha * (cs - m_a), -400, 400)
        s -= s.max()
        obj = m_a + np.log(np.exp(s).sum()) / alpha
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for wi, ii in zip(w, idx):
            if wi > 1e-15:
                _grad_lag_into(ii, h, n, g, wi)
        # ADAM update
        b1, b2, eps = 0.9, 0.999, 1e-8
        m = b1*m + (1-b1)*g
        v = b2*v + (1-b2)*g**2
        m_hat = m / (1-b1**t)
        v_hat = v / (1-b2**t)
        h_new = _proj(h - lr * m_hat / (np.sqrt(v_hat) + eps), n*0.5)
        c5n = _eval(h_new)
        if c5n < best_c:
            best_c = c5n
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_worst(h):

    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max()), int(np.argmax(c))

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_into(k, h, n, g, scale):
    dx = scale * 2.0 / n
    sh = (n - 1) - k
    if sh >= 0:
        if sh == 0:
            g += (1.0 - 2.0 * h) * dx
        else:
            g[:n-sh] += (1.0 - h[sh:]) * dx
            g[sh:] -= h[:n-sh] * dx
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp]) * dx
        g[:n-sp] -= h[sp:] * dx

def _slsqp_focus(h, alpha, maxiter, top_k):
    n = len(h)
    dx = 2.0 / n
    def obj_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        if top_k > 0:
            idx = np.argsort(c)[-top_k:]
            cs = c[idx]
        else:
            cs = c
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400); s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def grad_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        if top_k > 0:
            idx = np.argsort(c)[-top_k:]
            cs = c[idx]
        else:
            idx = np.arange(len(c)); cs = c
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400); s -= s.max()
        w = np.exp(s); w /= w.sum()
        g = np.zeros(n)
        for wi, ii in zip(w, idx):
            if wi > 1e-15:
                _grad_into(ii, x, n, g, wi)
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)

def _adam_smooth(h, alpha0, alpha_f, n_steps, top_k, lr):
    n = len(h)
    dx = 2.0 / n
    m = np.zeros(n); v = np.zeros(n)
    best_h = h.copy(); best_c = _eval(h)
    b1, b2, eps = 0.9, 0.999, 1e-8
    g_buf = np.zeros(n)
    for t in range(1, n_steps + 1):
        alpha = alpha0 * (alpha_f / alpha0) ** (t / n_steps)
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        if top_k > 0 and top_k < len(c):
            idx = np.argsort(c)[-top_k:]
            cs = c[idx]
        else:
            idx = np.arange(len(c)); cs = c
        m_a = cs.max()
        s = np.clip(alpha * (cs - m_a), -400, 400); s -= s.max()
        w = np.exp(s); w /= w.sum()
        g_buf[:] = 0.0
        for wi, ii in zip(w, idx):
            if wi > 1e-15:
                _grad_into(ii, h, n, g_buf, wi)
        m[:] = b1*m + (1-b1)*g_buf
        v[:] = b2*v + (1-b2)*g_buf**2
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        h_new = _proj(h - lr * m_hat / (np.sqrt(v_hat) + eps), n * 0.5)
        c5n = _eval(h_new)
        if c5n < best_c:
            best_c = c5n; best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def _gcls(h, n_iter, rng, top_k=5, init_step=0.006):
    n = len(h)

    stall = 0
    for it in range(n_iter):

        idx = np.argsort(c)[-top_k:]
        wa = c[idx].copy(); wa /= wa.sum()
        g_buf[:] = 0.0
        for w, ii in zip(wa, idx):
            _grad_into(ii, h, n, g_buf, w)
        j_i = int(np.argmin(g_buf))
        j_d = int(np.argmax(g_buf))
        if j_i == j_d:
            j_d = (j_i + 1 + rng.randint(0, 5)) % n
        accepted = False
        for mult in (1.0, 0.5, 0.25, 2.0, 4.0):
            ss = step * mult
            ni = h[j_i] + ss; nj = h[j_d] - ss
            if 0.0 <= ni and 0.0 <= nj <= 1.0:
                h[j_i], h[j_d] = ni, nj
                c5n = _eval(h)

                    stall = 0
                    if c5n < best_c:
                        best_c = c5n; best_h = h.copy()
                    accepted = True
                    break
                h[j_i], h[j_d] = ni - ss, nj + ss
        if not accepted:
            step *= 0.65
            stall += 1
        if it % 5000 == 4999:
            i = rng.randint(n); j = (i + 1 + rng.randint(0, 10)) % n
