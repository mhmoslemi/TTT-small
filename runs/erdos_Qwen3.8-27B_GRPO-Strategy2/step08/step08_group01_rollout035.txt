    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el():
        return time.time() - t0

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h0)
    best_h, best_c5 = h0.copy(), _eval(h0)

    # Phase 1: MGD with increasing alpha
    for alpha, steps, s0 in [(3000, 400, 0.004), (20000, 600, 0.0015), (100000, 500, 0.0005)]:
        if _el() > 160:
            break
        ht, _ = _mgd(best_h.copy(), steps, alpha, s0, rng)
        c = _eval(ht)
        if c < best_c5:
            best_c5, best_h = c, ht.copy()

    # Phase 2: SLSQP refinement
    if _el() < budget_s - 350:
        for alpha, mxi in [(20000, 500), (80000, 400), (400000, 300)]:
            if _el() > budget_s - 300:
                break
            ht = _slsqp_run(best_h.copy(), alpha, mxi)
            c = _eval(ht)
            if c < best_c5:
                best_c5, best_h = c, ht.copy()

    # Phase 3: Multi-start with structured perturbations
    if _el() < budget_s - 450:
        for t in range(6):
            if _el() > budget_s - 400:
                break
            for scale in [0.004, 0.01, 0.02]:
                if _el() > budget_s - 390:
                    break
                noise = scale * rng.randn(n)
                noise -= noise.mean()
                hp = _repair(best_h + noise, n)
                hp, _ = _mgd(hp, 250, 30000, 0.001, rng)
                c = _eval(hp)
                if c < best_c5:
                    best_c5, best_h = c, hp.copy()

    # Phase 4: SLSQP after multi-start
    if _el() < budget_s - 250:
        ht = _slsqp_run(best_h.copy(), 100000, 350)
        c = _eval(ht)
        if c < best_c5:
            best_c5, best_h = c, ht.copy()

    # Phase 5: Gradient-guided local search
    if _el() < budget_s - 150:
        remaining = budget_s - 150 - _el()
        ni = min(int(remaining * 1200), 100000)
        best_h, best_c5 = _grad_ls(best_h, ni, 5e-5, rng)

    # Phase 6: Standard local search with annealing
    if _el() < budget_s - 80:
        remaining = budget_s - 80 - _el()
        sa = best_c5 * 4e-5
        ni = min(int(remaining * 1500), 120000)
        best_h, best_c5 = _ls(best_h, ni, 4e-5, rng, sa)

    # Phase 7: Cooled local search (no SA)
    if _el() < budget_s - 30:
        remaining = budget_s - 30 - _el()
        ni = min(int(remaining * 2500), 200000)
        best_h, best_c5 = _ls(best_h, ni, 3e-5, rng)

    # Phase 8: Ultra-fine polish
    if _el() < budget_s - 10:
        remaining = budget_s - 10 - _el()
        ni = min(int(remaining * 4000), 300000)
        best_h, best_c5 = _ls(best_h, ni, 5e-6, rng)

    # Phase 9: Final MGD + local
    if _el() < budget_s - 5:
        ht, _ = _mgd(best_h.copy(), 300, 500000, 0.0002, rng)
        c = _eval(ht)
        if c < best_c5:
            best_c5, best_h = c, ht.copy()
    if _el() < budget_s - 2:
        remaining = budget_s - 2 - _el()
        ni = min(int(remaining * 6000), 200000)
        best_h, best_c5 = _ls(best_h, ni, 3e-6, rng)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
