def _proj(h, target):
    lo = -2.0
    hi = 2.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    return np.clip(h - t, 0.0, 1.0)


def _repair(h, n):
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0.0:
            idx = np.where(h < 1.0 - 1e-10)[0]
            if idx.size:
                h[idx] += diff / float(idx.size)
                h = np.clip(h, 0.0, 1.0)
        else:
            idx = np.where(h > 1e-10)[0]
            if idx.size:
                h[idx] += diff / float(idx.size)
                h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) > 1e-12:
        h = _proj(h, target)
    return h


def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def _eval(h):
    n = len(h)
    return float(np.max(_corr(h)) * (2.0 / n))


def _top_indices(c, k):
    L = len(c)
    k = min(int(k), L)
    if k <= 0:
        return np.empty(0, dtype=int)
    if k >= L:
        return np.arange(L, dtype=int)
    return np.argpartition(c, L - k)[-k:]


def _grad_shift_into(h, k, g):
    n = len(h)
    k0 = n - 1
    if k <= k0:
        sh = k0 - k
        if sh == 0:
            g += 1.0 - 2.0 * h
        else:
            m = n - sh
            g[:m] += 1.0 - h[sh:]
            g[sh:] -= h[:m]
    else:
        sp = k - k0
        m = n - sp
        g[sp:] += 1.0 - h[:m]
        g[:m] -= h[sp:]


def _gradient(h, c, kg):
    g = np.zeros(len(h), dtype=float)
    for k in _top_indices(c, kg):
        _grad_shift_into(h, int(k), g)
    return g


def _eval_pair(c, delta, q1, q2, d, work, temp):
    np.multiply(delta, d, out=work)
    np.add(work, c, out=temp)
    dd = d * d
    if q1 >= 0:
        temp[q1] += dd
    if q2 >= 0 and q2 != q1:
        temp[q2] += dd
    return float(np.max(temp))

def _perturb(h, rng, amp, n):
    h = np.asarray(h, dtype=float).copy()
    nblocks = max(1, n // 100)
    for _ in range(nblocks):
        size = int(rng.randint(1, min(61, n + 1)))

        d = amp * rng.randn(size)
        d -= d.mean()
        seg += d
        h[i0:i0 + size] = np.clip(seg, 0.0, 1.0)
    return _repair(h, n)


def _quantiles(h, k):
    n = len(h)
    k = min(int(k), n)
    if k <= 0:
        e = np.empty(0, dtype=int)
        return e, e
    if k >= n:
        idx = np.arange(n, dtype=int)
        return idx, idx
    low = np.argpartition(h, k - 1)[:k]
    high = np.argpartition(h, n - k)[-k:]
    return low, high


def _violation_pair(h, k, n):
    k0 = n - 1
    if k <= k0:
        sh = k0 - k
        if sh <= 0 or sh >= n:
            return -1, -1
        m = n - sh
        if m <= 0:
            return -1, -1
        terms = h[:m] * (1.0 - h[sh:])
        i = int(np.argmax(terms))
        return i, i + sh
    else:
        sp = k - k0
        if sp <= 0 or sp >= n:
            return -1, -1
        m = n - sp
        if m <= 0:
            return -1, -1
        terms = h[sp:] * (1.0 - h[:m])
        i = int(np.argmax(terms))
        return i + sp, i


def _add_cand(out, seen, a, b, n):
    a = int(a)
    b = int(b)
    if a < 0 or b < 0 or a >= n or b >= n or a == b:
        return
    if a > b:
        a, b = b, a
    key = a * n + b
    if key not in seen:
        seen.add(key)
        out.append((a, b))
