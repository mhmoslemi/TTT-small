def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _lsexp_grad(h, alpha, topk):

    n = len(h)
    c = _correl(h)

    else:
        idx = np.arange(len(c)); ct = c
    s = alpha * (ct - ct.max())
    w = np.exp(s); w /= w.sum()
    # Gradient of (1/alpha)*log(sum(w_k * exp(alpha*c_k))) = sum(w_k * grad c_k)
    dx = 2.0 / n
    g = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _smooth_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = alpha * (ct - ct.max())
    w = np.exp(s); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = idx[ii]; sh = (n - 1) - k; wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _fista_smooth(h, alpha_list, maxit, deadline):
    n = len(h)
    target = n * 0.5
    best_h, best_f = h.copy(), _eval(h)
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        topk = min(40, 2 * n - 1)
        L = alpha * (2.0 / n) * topk * 2.0
        step = min(2.0 / (L + 1e-10), 0.04)
        xk = _proj(h - step * _smooth_grad(h, alpha, topk), target)
        z = xk.copy()
        tk = 1.0
        for _ in range(maxit):
            if time.time() > deadline:
                break
            g = _smooth_grad(z, alpha, topk)
            t_old = tk
            xn = _proj(z - step * g, target)
            tk = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
            z = xn + ((t_old - 1.0) / tk) * (xn - xk)
            z = np.clip(z, 0.0, 1.0)
            xk = xn
            f = _eval(xk)
            if f < best_f:
                best_f, best_h = f, xk.copy()
        h = xk.copy()
    return best_h, best_f

def _nesterov_sm(h, alpha, topk, steps, step0, deadline, beta=0.9):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _smooth_grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-16:
            h, f = hn, fn
            step = min(step * 1.02, step0 * 25.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.5
            h = _proj(h - step * v, target)
            f = _eval(h)
            if f < best_f:
                best_f, best_h = f, h.copy()
    return best_h, best_f

def _slsqp(h, deadline, maxit=250):
    n = len(h)
    if time.time() > deadline:
        return h
    alpha, topk = 15000, 25
    def obj(x):
        c = _correl(x); m = c.max()
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
        s = np.clip(alpha * (ct - m), -400, 400)
        return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    def jac(x):
        return _smooth_grad(x, alpha, topk)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return h

def _flip_search(h, n_iter, rng, deadline):
    n = len(h)
    c5c = _eval(h); bh = h.copy()
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.4, 0.4)
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < _eval(bh):
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh

def _anneal(h, n_iter, rng, deadline, blocks, T0, T1):
    n = len(h)
    c5c = _eval(h); bh = h.copy(); bc = c5c
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(n_iter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.012 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean(); seg += d; seg = np.clip(seg, 0, 1)
        hn = h.copy(); hn[i0:i0+bs] = seg; hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _jitter_reopt(h, rng, step0, deadline):
    n = len(h)
    c5c = _eval(h)
    for scale in [0.003, 0.008, 0.015]:
        d = scale * rng.randn(n); d -= d.mean()
        hn = _repair(np.clip(h + d, 0, 1), n)
        hn2, c5n = _nesterov_sm(hn, 30000, 20, 400, step0, deadline, beta=0.88)
        if c5n < c5c:
            return hn2, c5n
    return h, c5c
