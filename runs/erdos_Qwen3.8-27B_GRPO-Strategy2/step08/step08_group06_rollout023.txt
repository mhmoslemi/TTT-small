c_k = sum_j h[j] * (1-h)[j + k - n + 1]

(1-h)[i+k-n+1] (if valid index) - h[i-k+n-1] (if valid index)

For the 1D minimization: given coefficients a_k for all k (2n-1 values) and constants b_k = c_k - a_k * h[i] (current value), find t minimizing max_k (b_k + a_k * t).

This is a linear programming problem in 1D: minimize z subject to b_k + a_k * t ≤ z for all k, 0 ≤ t ≤ 1. We can solve this by:
- The optimum is where the two "tightest" constraints cross
- Or use bisection: check if at t, all constraints are satisfied with a common z

Actually, the simplest approach: the function f(t) = max_k (b_k + a_k * t) is the upper envelope of lines. Its minimum is at a "kink point" where two lines cross, or at t=0 or t=1.

For a practical implementation:
- Evaluate f at t=0, t=1
- Check if the derivative changes sign (i.e., is there a kink in (0,1)?)
- If so, binary search for it

The derivative (subgradient) of f at t is a_{k(t)} where k(t) = argmax_k (b_k + a_k * t).

def _coord_1d(h, i, c, n):
    """Find best h[i] given current c_k values."""
    # Compute coefficients a_k for h[i] in c_k
    # c_k = sum_j h[j]*(1-h)[j+k-(n-1)]
    # Coefficient of h[i]: (1-h)[i+k-(n-1)] - h[i-k+(n-1)] (if in range)
    a = np.zeros(2*n-1)
    for k in range(2*n-1):
        m = k - (n-1)
        # First term: (1-h)[i+m] if 0 <= i+m < n
        idx1 = i + m
        if 0 <= idx1 < n:
            a[k] += (1.0 - h[idx1])
        # Second term: -h[i-m] if 0 <= i-m < n
        idx2 = i - m
        if 0 <= idx2 < n:
            a[k] -= h[idx2]
    # b_k = c_k - a_k * h[i]
    b = c - a * h[i]
    # Minimize max_k (b_k + a_k * t) over t in [0,1]
    # Bisection on subgradient
    lo, hi = 0.0, 1.0
    for _ in range(30):
        mid = (lo + hi) / 2
        vals = b + a * mid
        k_max = np.argmax(vals)
        deriv = a[k_max]
        if deriv > 1e-10:
            hi = mid
        elif deriv < -1e-10:
            lo = mid
        else:
            break
    return (lo + hi) / 2

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _eval_c(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(np.max(c)), c

def _coeff_vector(h, i, n):
    """Coefficient of h[i] in each c_k of the raw correlate."""
    n2 = 2 * n - 1
    a = np.zeros(n2)
    m_range = np.arange(n2) - (n - 1)
    idx1 = i + m_range
    valid1 = (idx1 >= 0) & (idx1 < n)
    a[valid1] += (1.0 - h[idx1[valid1]])
    idx2 = i - m_range
    valid2 = (idx2 >= 0) & (idx2 < n)
    a[valid2] -= h[idx2[valid2]]
    return a

def _coord_1d(h, i, c_raw, n):
    """Find optimal h[i] given c_raw (raw correlate values)."""
    n2 = 2 * n - 1
    a = _coeff_vector(h, i, n)
    b = c_raw - a * h[i]
    lo, hi = 0.0, 1.0
    for _ in range(25):
        mid = (lo + hi) * 0.5
        vals = b + a * mid
        deriv = a[np.argmax(vals)]
        if deriv > 1e-12:
            hi = mid
        elif deriv < -1e-12:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def _coord_descent(h, n_sweeps, t_end):
    n = len(h)
    for _sw in range(n_sweeps):
        if time.time() > t_end:
            break
        c_raw = np.correlate(h, 1.0 - h, mode='full')
        changed = False
        for i in range(n):
            new_val = _coord_1d(h, i, c_raw, n)
            if abs(new_val - h[i]) > 1e-10:
                h[i] = new_val
                changed = True
        if not changed:
            break
        c_raw = np.correlate(h, 1.0 - h, mode='full')
    return h

def _exact_subgrad(h, max_iter, t_end, step0=0.3):
    n = len(h)
    step = step0
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        c5, c = _eval_c(h)
        thresh = c5 - c5 * 0.02
        active = np.where(c >= thresh)[0]
        g = np.zeros(n)
        dx2 = 2.0 / n
        for k in active:
            m = k - (n - 1)
            idx1 = np.arange(n) + m
            v1 = (idx1 >= 0) & (idx1 < n)
            g[v1] += dx2 * (1.0 - h[idx1[v1]])
            idx2 = np.arange(n) - m
            v2 = (idx2 >= 0) & (idx2 < n)
            g[v2] -= dx2 * h[idx2[v2]]
        g /= max(len(active), 1)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        h_new = _repair(h - (step / gn) * g, n)
        new_c5 = _eval(h_new)
        if new_c5 < c5 - 1e-12:
            h = h_new
            step = min(step * 1.05, 3.0)
        else:
            step *= 0.6
            if step < 1e-10:
                break
    return h

def _bimodal_push(h, n_iters, t_end):
    """Push values toward 0/1 to create bimodal structure."""
    n = len(h)
    for _ in range(n_iters):
        if time.time() > t_end:
            break
        c_raw = np.correlate(h, 1.0 - h, mode='full')
        # For each i, the zero-shift contribution is h[i]*(1-h[i])
        # Push toward 0 or 1 depending on which reduces max overlap
        for i in range(0, n, 3):
            a = _coeff_vector(h, i, n)
            b = c_raw - a * h[i]
            # Try pushing toward 0 and 1
            v0 = b + a * 0.0
            v1 = b + a * 1.0
            m0 = np.max(v0)
            m1 = np.max(v1)
            if m0 < m1:
                h[i] = max(0.0, h[i] - 0.05)
            else:
                h[i] = min(1.0, h[i] + 0.05)
        h = _repair(h, n)
    return h

def _local_2pt(h, n_tries, rng, t_end):
    n = len(h)
    best_h, best_c = h.copy(), _eval(h)
    cur_c = best_c
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = 2e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5 = _eval(h)
            if c5 < cur_c:
                cur_c = c5
                if c5 < best_c:
                    best_c, best_h = c5, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _local_3pt(h, n_tries, rng, t_end):
    n = len(h)
    best_h, best_c = h.copy(), _eval(h)
    cur_c = best_c
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 2 + rng.randint(0, 8)) % n
        k = (i + 10 + rng.randint(0, 20)) % n
        d1, d2 = 1.5e-5 * rng.randn(2)
        ni, nj, nk = h[i]+d1, h[j]+d2, h[k]-d1-d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5 = _eval(h)
            if c5 < cur_c:
                cur_c = c5
                if c5 < best_c:
                    best_c, best_h = c5, h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return best_h, best_c

def _dft_smooth(h, t_end, n_passes=30):
    """Adjust DFT to make autocorrelation more uniform."""
    n = len(h)
    h = h.copy()
    for _ in range(n_passes):
        if time.time() > t_end:
            break
        H = np.fft.rfft(h)
        H[0] = 0.0
        mag = np.abs(H)
        if mag[1:].max() < 1e-12:
            break
        # Target: flatten magnitude (geometric mean of non-DC)
        target_mag = np.geomsort(mag[1:])[len(mag) // 2 - 1] if len(mag) > 2 else 1.0
        target_mag = np.median(mag[1:])
        alpha = 0.05
        new_mag = mag.copy()
        new_mag[1:] = (1 - alpha) * mag[1:] + alpha * target_mag
        phase = np.angle(H)
        H = new_mag * np.exp(1j * phase)
        h_new = np.fft.irfft(H, n=n)
        h_new = _repair(h_new, n)
        if _eval(h_new) < _eval(h):
            h = h_new
    return h
