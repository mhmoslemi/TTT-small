def grad_lag(h, lag_idx):
    """Exact gradient of the overlap at a specific correlation index."""
    n = len(h)
    dx = 2.0 / n
    s = lag_idx - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        i1 = np.arange(n - s)
        g[i1] += dx * (1.0 - h[i1 + s])
        if s > 0:
            i2 = np.arange(s, n)
            g[i2] -= dx * h[i2 - s]
    else:
        sp = -s
        i1 = np.arange(sp, n)
        g[i1] += dx * (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= dx * h[i2 + sp]
    return g

    c = full_corr(h)
    idx = np.argsort(c)[-k:]
    g = np.zeros(len(h))
    for ii in idx:
        g += grad_lag(h, ii)
    return g

Let me think about another approach: **coordinate descent with exact 1D optimization**. For each coordinate i, fix all other coordinates and optimize h[i] (and one coupling coordinate j) to minimize the max overlap. This is a 2D subproblem that can be solved by grid search or bisection.

Actually, for a 2-point move (i, j), the objective as a function of d (where h[i] += d, h[j] -= d) is:
max_k [c[k](d)] where c[k](d) = c[k](0) + d * (∂c[k]/∂h[i] - ∂c[k]/∂h[j])

So each c[k] is affine in d, and the max of affine functions is convex in d. We can find the minimum by checking where the kink points are.

This is actually a nice property! The minimum of a convex piecewise-linear function can be found by checking the O(2n) kink points and finding where the sign of the derivative changes.

def exact_2d_step(h, i, j):
    """Find optimal d for 2-point move h[i]+=d, h[j]-=d."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    # For each lag, compute the slope of c[lag] w.r.t. d
    slopes = np.zeros(2*n-1)
    for lag in range(2*n-1):
        s = lag - (n-1)
        di = 0.0  # contribution from h[i]
        dj = 0.0  # contribution from h[j]
        if s >= 0:
            # c[lag] = sum_{k=0}^{n-1-s} h[k](1-h[k+s])
            # d/dh[i]: if 0<=i<=n-1-s and i+s<n, then (1-h[i+s])
            # if 0<=i-s<=n-1-s, i.e., s<=i<=n-1, then -h[i-s]
            if 0 <= i <= n-1-s:
                di += dx * (1-h[i+s])
            if s <= i <= n-1:
                di -= dx * h[i-s]
            # same for j
            if 0 <= j <= n-1-s:
                dj += dx * (1-h[j+s])
            if s <= j <= n-1:
                dj -= dx * h[j-s]
        else:
            sp = -s
            # c[lag] = sum_{k=sp}^{n-1} h[k](1-h[k-sp])
            if sp <= i <= n-1:
                di += dx * (1-h[i-sp])
            if 0 <= i <= n-1-sp:
                di -= dx * h[i+sp]
            if sp <= j <= n-1:
                dj += dx * (1-h[j-sp])
            if 0 <= j <= n-1-sp:
                dj -= dx * h[j+sp]
        slopes[lag] = di - dj  # slope of c[lag] w.r.t. d
    
    # c[lag](d) = c[lag](0) + d * slopes[lag]
    # We want to minimize max_lag c[lag](d)
    # This is a LP: min t s.t. c[lag](0) + d*slopes[lag] <= t for all lag
    # Equivalently: max_lag (c[lag](0) + d*slopes[lag])
    # This is convex in d, piecewise linear
    # Find the minimum by checking all kink points
    
    # Kink points: where c[a](d) = c[b](d) for some a, b
    # c[a](d) = c[a](0) + d*s[a]
    # c[a](d) = c[b](d) => d = (c[b](0) - c[a](0)) / (s[a] - s[b])
    
    # But there are O(n^2) kink points, which is too many
    # Instead, use a simpler approach: ternary search or bisection on the convex function
    
    # Actually, since it's piecewise linear, we can do:
    # - The subgradient of max_lag (c[lag](0) + d*s[lag]) at d=0 is the set of s[lag*] for lag* achieving the max
    # - We need the d where 0 is in the subgradient, i.e., where the max is achieved at lags with both positive and negative slopes (or a single lag with zero slope)
    
    # Simpler: just do a golden section search over d in [dmin, dmax]
    def f(d):
        if 0 <= h[i]+d <= 1 and 0 <= h[j]-d <= 1:
            new_c = c + d * slopes
            return new_c.max()
        return 1e10
    
    # Determine valid range of d
    dmin = max(-(h[i]), -(1-h[j]))  # h[i]+d >= 0, h[j]-d >= 0 => d >= -h[j], h[i]+d <= 1 => d <= 1-h[i]
    # Wait: h[i]+d >= 0 => d >= -h[i]; h[j]-d >= 0 => d <= h[j]
    # h[i]+d <= 1 => d <= 1-h[i]; h[j]-d <= 1 => d >= h[j]-1
    dmin = max(-h[i], h[j]-1)
    dmax = min(1-h[i], h[j])
    
    if dmin >= dmax:
        return h, 0.0
    
    # Golden section search
    gr = (5**0.5 - 1) / 2
    a, b = dmin, dmax
    c_g = b - gr*(b-a)
    d_g = a + gr*(b-a)
    fc, fd = f(c_g), f(d_g)
    for _ in range(50):
        if fc < fd:
            b = d_g; d_g = c_g; fd = fc
            c_g = b - gr*(b-a)
            fc = f(c_g)
        else:
            a = c_g; c_g = d_g; fc = fd
            d_g = a + gr*(b-a)
            fd = f(d_g)
    
    d_opt = (a+b)/2
    h[i] += d_opt
    h[j] -= d_opt
    return h, f(d_opt)
