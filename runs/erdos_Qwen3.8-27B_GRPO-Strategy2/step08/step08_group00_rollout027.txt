    return np.correlate(h, 1.0-h, mode='full') * 2.0/n

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0, 1).astype(float)
    d = t - h.sum()
    if abs(d) < 1e-14:
        return h
    if d > 0:
        order = np.argsort(1.0-h)
    else:
        order = np.argsort(h)
    rem = abs(d)
    for i in order:
        if rem < 1e-14:
            break
        if d > 0:
            a = min(rem, 1.0-h[i])
            h[i] += a
        else:
            a = min(rem, h[i])
            h[i] -= a
        rem -= a
    return h

def _proj(h, t):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5*(lo+hi)
        if np.clip(h-mid,0,1).sum() > t:
            lo = mid
        else:
            hi = mid
    return np.clip(h-0.5*(lo+hi), 0, 1)

def _slsqp_smooth(h, alpha, mlags, maxiter):
    n = len(h)
    def obj(x):
        c = np.correlate(x, 1.0-x, mode='full')*2.0/n
        m = c.max()
        idx = np.argsort(c)[-mlags:]
        s = alpha*(c[idx]-m)
        s -= s.max()
        return m + np.log(np.exp(s).sum())/alpha
    def sum_fun(x):
        return x.sum() - n*0.5
    bnds = [(0,1)]*n
    cons = [{'type':'eq','fun':sum_fun,'jac':lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter':maxiter,'ftol':1e-12})
        return _repair(res.x, n)
    except:
        return h

    c = np.correlate(h, 1.0-h, mode='full')*2.0/n

    best_h, best_c5 = _local_search(best_h, t_end, rng, 3e-4, n//15)

    # Phase 4: Top-lag targeted search
    t_end = t0 + min(budget_s - 80, _el() + 150)
    best_h, best_c5 = _top_lag_search(best_h, t_end, rng, 1.5e-4)

    # Phase 5: Medium local search
    t_end = t0 + min(budget_s - 40, _el() + 150)
    best_h, best_c5 = _local_search(best_h, t_end, rng, 5e-5, n//40)

    # Phase 6: Fine local search
    t_end = t0 + min(budget_s - 15, _el() + 120)
    best_h, best_c5 = _local_search(best_h, t_end, rng, 1e-5, 8)

    # Phase 7: Ultra-fine polish
    t_end = t0 + budget_s - 2
    best_h, best_c5 = _local_search(best_h, t_end, rng, 2e-6, 3)

    # Final repair and evaluation
    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
