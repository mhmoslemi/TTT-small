def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_all(h, n, indices):
    """Compute gradients for specified shift indices (into the full correlate array)."""
    dx = 2.0 / n
    grads = np.zeros((len(indices), n))
    for ii, k in enumerate(indices):
        off = k - (n - 1)
        g = np.zeros(n)
        j = np.arange(n)
        i1 = j - off
        m1 = (i1 >= 0) & (i1 < n)
        g[m1] += (1.0 - h[i1[m1]])
        i2 = j + off
        m2 = (i2 >= 0) & (i2 < n)
        g[m2] -= h[i2[m2]]
        grads[ii] = g * dx
    return grads

    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha, idx, ct

        c_obj = -grads[0]
        A_ub = grads[1:] if m_top > 1 else None

import numpy as np
from scipy.optimize import minimize, linprog
import time

_OPT = {}

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - t) > 1e-9:
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        d = t - h.sum()
        if abs(d) > 1e-12:
            ord_ = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
            for idx in ord_:
                if d > 0:
                    a = min(d, 1.0 - h[idx]); h[idx] += a; d -= a
                else:
                    a = min(-d, h[idx]); h[idx] -= a; d += a
                if abs(d) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_shift(h, k):
    n = len(h)
    dx = 2.0 / n
    off = k - (n - 1)
    g = np.zeros(n)
    j = np.arange(n)
    i1 = j - off
    m1 = (i1 >= 0) & (i1 < n)
    if m1.any():
        g[m1] += (1.0 - h[i1[m1]])
    i2 = j + off
    m2 = (i2 >= 0) & (i2 < n)
    if m2.any():
        g[m2] -= h[i2[m2]]
    return g * dx

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sobj(x):
    a, tk = _OPT['a'], _OPT['tk']
    c = _correl(x)
    m = c.max()
    if tk < len(c):
        idx = np.argpartition(c, -tk)[-tk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(a * (ct - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / a

def _sgrad(x):
    a, tk = _OPT['a'], _OPT['tk']
    n = len(x)
    c = _correl(x)
    m = c.max()
    if tk < len(c):
        idx = np.argpartition(c, -tk)[-tk:].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(a * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] > 1e-14:
            g += w[ii] * _grad_shift(x, idx[ii])
    return g

def _ceq(x):
    return x.sum() - _OPT['n'] * 0.5

def _ceq_j(x):
    return np.ones(len(x))

def _slsqp(h, a, tk, maxit, dl):
    _OPT.update(a=a, tk=tk, n=len(h))
    try:
        r = minimize(_sobj, h, jac=_sgrad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * len(h),
                     constraints=[{'type': 'eq', 'fun': _ceq, 'jac': _ceq_j}],
                     options={'maxiter': maxit, 'ftol': 1e-18, 'maxls': 60})
        return _repair(r.x, len(h))
    except Exception:
        return h

def _lp_desc(h, nsteps, eps0, mtop, dl):

    bh, bc5 = h.copy(), _eval(h)
    c, c5c = h.copy(), bc5

def _nesterov(h, a, tk, steps, st0, dl):

def _ganneal(h, niter, rng, dl, T0, T1):

            if c5n < bc:

                else:
