        for i in order:
            j = (i + 1 + rng.randint(0, 8)) % n
            if i == j:
                continue
            d = step * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                cn = _eval(h)
                if cn < cur:
                    cur = cn
                    if cn < best_c:
                        best_c = cn; best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    hp = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(hp)
    best_h = hp.copy()
    best_c5 = _eval(best_h)

    # Phase 1: MGD sweep (diverse alpha)
    for alpha, steps, tk, s0 in [(100,4000,12,0.004),(500,3000,8,0.003),(2000,2500,6,0.002),(10000,2000,5,0.001),(50000,1500,4,0.0005)]:
        if time.time() - t0 > 160:
            break
        ht, c = _mgd(best_h.copy(), steps, alpha, tk, s0, rng)
        c = _eval(ht)
        if c < best_c5:
            best_c5 = c; best_h = ht.copy()

    # Phase 2: SLSQP refinement
    if time.time() - t0 < budget_s - 280:
        for alpha, tk, mxi in [(2000,10,1000),(10000,8,800),(50000,6,600),(200000,5,500),(500000,4,400)]:
            if time.time() - t0 > budget_s - 230:
                break
            ht = _slsqp(best_h.copy(), alpha, mxi, tk)
            c = _eval(ht)
            if c < best_c5:
                best_c5 = c; best_h = ht.copy()

    # Phase 3: Structured perturbations + re-optimize
    if time.time() - t0 < budget_s - 350:
        perts = []
        for freq in [1, 2, 3, 5, 7, 13]:
            t = np.arange(n) / n * 2.0 * np.pi
            perts.append(0.025 * np.sin(freq * t + rng.rand() * 6.28))
            perts.append(0.025 * np.cos(freq * t + rng.rand() * 6.28))
        for _ in range(6):
            p = 0.015 * rng.randn(n); p -= p.mean(); perts.append(p)
        for pert in perts:
            if time.time() - t0 > budget_s - 200:
                break
            hp2 = _repair(best_h + pert, n)
            ht, c = _mgd(hp2, 600, 5000, 6, 0.002, rng)
            c = _eval(ht)
            if c < best_c5:
                best_c5 = c; best_h = ht.copy()
            ht = _slsqp(ht, 10000, 300, 5)
            c = _eval(ht)
            if c < best_c5:
                best_c5 = c; best_h = ht.copy()

    # Phase 4: Multi-resolution
    if time.time() - t0 < budget_s - 350:
        for ratio in [2, 4]:
            if time.time() - t0 > budget_s - 300:
                break
            nc = n // ratio
            hc = _repair(_downsample(best_h, ratio), nc)
            hc, _ = _mgd(hc, 800, 5000, 5, 0.006, rng)
            hu = _repair(_upsample(hc, ratio), n)
            cu = _eval(hu)
            if cu < best_c5:
                best_c5 = cu; best_h = hu.copy()
            best_h, best_c5 = _local2(best_h, 5000, 5e-5, rng)
            best_h, best_c5 = _mgd(best_h, 300, 10000, 5, 0.001, rng)
            best_c5 = _eval(best_h)

    # Phase 5: SLSQP round 2
    if time.time() - t0 < budget_s - 200:
        for alpha, tk, mxi in [(5000,8,800),(30000,6,600),(200000,4,400)]:
            if time.time() - t0 > budget_s - 160:
                break
            ht = _slsqp(best_h.copy(), alpha, mxi, tk)
            c = _eval(ht)
            if c < best_c5:
                best_c5 = c; best_h = ht.copy()

    # Phase 6: SA local search
    if time.time() - t0 < budget_s - 130:
        temp = max(1e-7, best_c5 * 1e-5)
        nsa = min(int((budget_s - 130 - (time.time()-t0)) * 2500), 120000)
        best_h, best_c5 = _local2(best_h, nsa, 4e-5, rng, temp)

    # Phase 7: Sweep-based coordinate descent
    if time.time() - t0 < budget_s - 70:
        npass = min(int((budget_s - 70 - (time.time()-t0)) * 200), 3000)
        best_h, best_c5 = _sweep_local(best_h, npass, 2e-5, rng)

    # Phase 8: 3-point search
    if time.time() - t0 < budget_s - 50:
        n3 = min(int((budget_s - 50 - (time.time()-t0)) * 2000), 100000)
        best_h, best_c5 = _local3(best_h, n3, 2e-5, rng)

    # Phase 9: MGD final
    if time.time() - t0 < budget_s - 25:
        ht, _ = _mgd(best_h, 600, 100000, 4, 0.0004, rng)
        ct = _eval(ht)
        if ct < best_c5:
            best_c5 = ct; best_h = ht.copy()

    # Phase 10: Ultra-fine 2-point
    if time.time() - t0 < budget_s - 3:
        nfin = min(int((budget_s - 3 - (time.time()-t0)) * 5000), 250000)
        best_h, best_c5 = _local2(best_h, nfin, 2e-6, rng)

    # Final micro MGD
    if time.time() - t0 < budget_s - 0.5:
        ht, _ = _mgd(best_h, 300, 500000, 3, 0.0002, rng)
        ct = _eval(ht)
        if ct < best_c5:
            best_c5 = ct; best_h = ht.copy()
        nfin2 = min(int((budget_s - 0.5 - (time.time()-t0)) * 6000), 100000)
        best_h, best_c5 = _local2(best_h, nfin2, 1e-6, rng)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
