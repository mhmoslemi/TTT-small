            i = rng.randint(n); j = (i + 1 + rng.randint(0, 4)) % n
            d = ss * (2 * rng.random() - 1)
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5n < bc: bc, bh = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: SLSQP with annealed alpha (0-180s)
    dl1 = min(deadline, t0 + 180)
    for a, tk, mi in [(3000, 40, 400), (15000, 25, 400), (80000, 12, 300)]:
        if time.time() > dl1: break
        hv = _slsqp_full(best_h.copy(), a, tk, mi, dl1)
        cv = _eval(hv)
        if cv < best_c5: best_c5, best_h = cv, hv.copy()

    # Phase 2: LP descent with sum-of-peaks (180-450s)
    dl2 = min(deadline, t0 + 450)
    for mt, ep in [(10, 0.003), (6, 0.002), (4, 0.001)]:
        if time.time() > dl2: break
        hv, cv = _lp_sum(best_h.copy(), 8000, ep, mt, dl2, rng)
        if cv < best_c5: best_c5, best_h = cv, hv.copy()

    # Phase 3: Nesterov accelerated smooth max (450-600s)
    dl3 = min(deadline, t0 + 600)
    for a, tk, st, ns in [(50000, 12, 0.002, 3000), (200000, 6, 0.001, 2000)]:
        if time.time() > dl3: break
        hv = _nesterov(best_h.copy(), a, tk, ns, st, dl3)
        cv = _eval(hv)
        if cv < best_c5: best_c5, best_h = cv, hv.copy()

    # Phase 4: Second LP pass with smaller eps (600-780s)
    dl4 = min(deadline, t0 + 780)
    for mt, ep in [(5, 0.0008), (3, 0.0005)]:
        if time.time() > dl4: break
        hv, cv = _lp_sum(best_h.copy(), 12000, ep, mt, dl4, rng)
        if cv < best_c5: best_c5, best_h = cv, hv.copy()

    # Phase 5: Subspace SLSQP on top-gradient coords (780-870s)
    dl5 = min(deadline, t0 + 870)
    for sc in [30, 20, 15]:
        if time.time() > dl5: break
        hv = _subspace_slsqp(best_h.copy(), sc, 50000, dl5)
        cv = _eval(hv)
        if cv < best_c5: best_c5, best_h = cv, hv.copy()

    # Phase 6: Block perturbations (870-960s)
    dl6 = min(deadline, t0 + 960)
    blocks = [1, 2, 3, 4, 6, 8]
    c5c = best_c5
    for it in range(20000):
        if time.time() > dl6: break
        T = 0.002 * (1e-10 / 0.002) ** (it / 20000.0)
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = best_h[i0:i0 + bs].copy()
        d = 0.002 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = best_h.copy(); hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            best_h, c5c = hn, c5n
            if c5n < best_c5: best_c5, best_h = c5n, hn.copy()

    # Phase 7: Fine polish (960-end)
    best_h, best_c5 = _polish(best_h, min(int(max(0, deadline - time.time()) / 0.001), 60000), deadline, rng)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
