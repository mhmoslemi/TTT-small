            else:

    h2, c5 = _coord_cd(best_h.copy(), cc, n0, rng, dl3, 60, 5)
    if c5 < best_c5:
        best_c5, best_h = c5, h2.copy()

    # Phase 4: Nesterov with fine tolerance
    dl4 = min(deadline, t0 + 580)
    h, c5 = _nesterov(best_h.copy(), 3000, 0.0015, dl4, 0.90, 0.001)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 5: Coordinate descent round 2 (more sweeps)
    dl5 = min(deadline, t0 + 750)
    cc = _correl(best_h.copy())
    h2, c5 = _coord_cd(best_h.copy(), cc, n0, rng, dl5, 100, 4)
    if c5 < best_c5:
        best_c5, best_h = c5, h2.copy()

    # Phase 6: Multi-start perturbation + coordinate CD
    dl6 = min(deadline, t0 + 880)
    for trial in range(6):
        if time.time() > dl6:
            break
        d = 0.003 * rng.randn(n0)
        d -= d.mean()
        hp = _repair(np.clip(best_h + d, 0, 1), n0)
        cc = _correl(hp.copy())
        sub_dl = min(dl6, time.time() + 120)
        h2, c5 = _coord_cd(hp, cc, n0, rng, sub_dl, 15, 3)
        if c5 < best_c5:
            best_c5, best_h = c5, h2.copy()

    # Phase 7: Final fine Nesterov + coordinate CD
    dl7 = min(deadline, t0 + budget_s - 30)
    h, c5 = _nesterov(best_h.copy(), 2000, 0.0005, dl7, 0.92, 0.0005)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    cc = _correl(best_h.copy())
    h2, c5 = _coord_cd(best_h.copy(), cc, n0, rng, deadline, 50, 3)
    if c5 < best_c5:
        best_c5, best_h = c5, h2.copy()

    # Phase 8: Polish
    best_h, best_c5 = _polish(best_h, 30000, rng, deadline, 5e-7)

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
