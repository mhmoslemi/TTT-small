from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_k(h, k, n):
    """Gradient of correlation k with respect to h. k is 0-indexed from correlate output."""
    g = np.zeros(n)
    off = k - (n - 1)  # actual shift
    j = np.arange(n)
    i1 = j - off
    mask1 = (i1 >= 0) & (i1 < n)
    g[mask1] += (1.0 - h[i1[mask1]])
    i2 = j + off
    mask2 = (i2 >= 0) & (i2 < n)
    g[mask2] -= h[i2[mask2]]
    return g

    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    for it in range(n_steps):
        if time.time() > deadline:
            break

        thresh = thresh_start * (thresh_end / thresh_start) ** t_frac

        active = np.where(cor > c_max - thresh)[0]

        # Actually find the true worst
        active_sorted = active[np.argsort(-cor[active])]
        g0 = _grad_k(c, active_sorted[0], n)
        c_obj = -g0
        
        A_ub_rows = []
        for k in active_sorted[1:]:
            gk = _grad_k(c, k, n)
            A_ub_rows.append(gk)
        A_ub = np.array(A_ub_rows) if A_ub_rows else None
        b_ub = np.zeros(len(A_ub_rows)) if A_ub_rows else None

        except Exception:

Let me also think about a different approach: **Bregman/conditional gradient (Frank-Wolfe) method**.

The problem is: min f(h) = max_k c_k(h) s.t. h in P where P = {h: 0<=h<=1, sum(h)=n/2}.

Frank-Wolfe:
1. Compute subgradient g = ∇c_k*(h) where k* is argmax
2. Solve linear minimization: s = argmin_{v in P} g^T v
3. Step: h_new = h + gamma * (s - h)
4. gamma is chosen by line search

The linear minimization over P is: min g^T v s.t. 0<=v<=1, sum(v)=n/2. This has closed form: sort g, take the n/2 smallest components and set v to 1 there, 0 elsewhere (for integer n/2) or with one fractional component.

Wait, that's for the case where v is binary. For continuous v in [0,1] with sum = n/2, the solution is:
- Sort g in increasing order: g_(1) <= g_(2) <= ... <= g_(n)
- Set v_(i) = 1 for i <= k, v_(k+1) = fractional, v_(i) = 0 for i > k+1
- Where k is chosen so that sum(v) = n/2

Actually for the continuous case: v_(i) = 1 if g_(i) < threshold, 0 if g_(i) > threshold, fractional at the threshold. This is the same projection!

So Frank-Wolfe step is: compute subgradient, find the extreme point of P in the direction -g, take a step toward it.

This is actually very efficient! Each iteration is O(n log n) for sorting. We can do 100k+ iterations.

def _frank_wolfe(h, n_iters, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    gamma = 1.0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        cor = _correl(c)
        k_worst = np.argmax(cor)
        g = _grad_k(c, k_worst, n)
        
        # Solve: min g^T v s.t. 0<=v<=1, sum(v)=n/2
        # Closed form: sort g, take bottom n/2
        sorted_g = np.sort(g)
        threshold = sorted_g[n//2]  # approximately
        # More precisely, find the threshold
        v = np.where(g < threshold, 1.0, 0.0)
        # Adjust for exact sum
        diff = n/2.0 - v.sum()
        if diff > 0:
            # Need to increase some v's
            frac = np.where((v == 0) & (g < threshold), 1, 
                           np.where(v == 0, 2, 3))
            # Actually just set the next ones to fractional
            order = np.argsort(g)
            for idx in order:
                if diff <= 0:
                    break
                if v[idx] < 1:
                    add = min(diff, 1 - v[idx])
                    v[idx] += add
                    diff -= add
        elif diff < 0:
            order = np.argsort(-g)
            for idx in order:
                if diff >= 0:
                    break
                if v[idx] > 0:
                    rem = min(-diff, v[idx])
                    v[idx] -= rem
                    diff += rem
        
        # Step
        h_new = c + gamma * (v - c)
        h_new = _repair(h_new, n)
        c5_new = _eval(h_new)
        
        if c5_new < c5_cur:
            c, c5_cur = h_new, c5_new
            if c5_cur < best_c5:
                best_c5, best_h = c5_cur, c.copy()
            gamma = min(gamma * 1.02, 1.0)
        else:
            gamma *= 0.8
            if gamma < 1e-6:
                break
    
    return best_h, best_c5

Let me also think about the "pair descent" more carefully. The key insight is: if we change h[i] by +d and h[j] by -d, the change in correlation k is:
Δc_k = d * (∂c_k/∂h[i] - ∂c_k/∂h[j])

So if we know the gradients, we can predict the change without re-evaluating. This makes pair descent much faster.

Let me code a more efficient pair descent:
- Precompute gradients for top-M correlations
- For each pair (i,j), the change in correlation k is d*(g_k[i] - g_k[j])
- We want all top correlations to decrease (or at least the max to decrease)
- This gives linear constraints on d

Actually this is exactly the LP descent with n=2 variables! We can solve it analytically:
- Minimize d*(g_0[i] - g_0[j]) subject to d*(g_k[i] - g_k[j]) <= 0 for all other active k
- And 0 <= h[i]+d <= 1, 0 <= h[j]-d <= 1

This is a 1D LP, solvable in O(M) where M is the number of active constraints.

Let me implement this as a "fast pair LP":

```python
def _fast_pair_lp(h, n_iters, rng, M, deadline):
    """Pair moves using 1D LP on precomputed gradients."""
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Recompute correlations periodically
        if it % 50 == 0:
            cor = _correl(c)
            top_M = np.argpartition(cor, -M)[-M:]
            grads = np.array([_grad_k(c, k, n) for k in top_M])
        
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 7)) % n
        
        # 1D LP: find d that maximally reduces worst while not increasing others
        # Objective: minimize d * (grads[0][i] - grads[0][j])
        # Constraints: d * (grads[k][i] - grads[k][j]) <= 0 for k=1..M-1
        # Bounds: d >= max(-c[i], -(1-c[j])) [h[i]+d>=0, h[j]-d<=1]
        #         d <= min(1-c[i], c[j])
        
        obj_coef = grads[0][i] - grads[0][j]
        if abs(obj_coef) < 1e-15:
            continue
        
        d_min = max(-c[i], -(1.0 - c[j]))
        d_max = min(1.0 - c[i], c[j])
        if d_min >= d_max:
            continue
        
        # For each constraint k: d * a_k <= 0 where a_k = grads[k][i] - grads[k][j]
        # If a_k > 0: d <= 0
        # If a_k < 0: d >= 0
        # If a_k = 0: no constraint
        
        effective_min = d_min
        effective_max = d_max
        
        for k in range(1, M):
            a_k = grads[k][i] - grads[k][j]
            if a_k > 1e-12:
                effective_max = min(effective_max, 0.0)
            elif a_k < -1e-12:
                effective_min = max(effective_min, 0.0)
        
        if effective_min >= effective_max:
            # No feasible improvement direction
            # Try random small move
            d = 1e-6 * (2 * rng.random() - 1)
            if d_min <= d <= d_max:
                c[i] += d
                c[j] -= d
                c5_new = _eval(c)
                if c5_new < c5_cur:
                    c5_cur = c5_new
                    if c5_cur < best_c5:
                        best_c5, best_h = c5_cur, c.copy()
                else:
                    c[i] -= d
                    c[j] += d
            continue
        
        # Choose d to minimize obj_coef * d
        if obj_coef > 0:
            d = effective_min  # most negative
        else:
            d = effective_max  # most positive
        
        if abs(d) < 1e-12:
            continue
        
        # Line search: try d, 0.5d, 0.25d
        improved = False
        for frac in [1.0, 0.5, 0.25, 0.1]:
            d_try = d * frac
            c[i] += d_try
            c[j] -= d_try
            c5_new = _eval(c)
            if c5_new < c5_cur - 1e-17:
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
                improved = True
                break
            else:
                c[i] -= d_try
                c[j] += d_try
        if not improved:
            c[i] -= 0  # no change
            c[j] -= 0
    
    return best_h, best_c5
```

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_k(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += (1.0 - h[i1[m1]])
    i2 = j + off
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

def _lp_adaptive(h, n_steps, eps0, eps1, th0, th1, deadline, max_active=80):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    stalled = 0
    rng_st = np.random.RandomState(12345)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        tf = it / max(n_steps - 1, 1)
        eps = eps0 * (eps1 / eps0) ** tf
        th = th0 * (th1 / th0) ** tf
        cor = _correl(c)
        c_max = cor.max()
        active = np.where(cor > c_max - th)[0]
        if len(active) > max_active:
            active = np.argpartition(cor, -max_active)[-max_active:]
        if len(active) < 2:
            active = np.argpartition(cor, -10)[-10:]
        active = active[np.argsort(-cor[active])]
        g0 = _grad_k(c, active[0], n)
        c_obj = -g0
        A_rows = []
        for k in active[1:]:
            A_rows.append(_grad_k(c, k, n))
        A_ub = np.array(A_rows) if A_rows else None
        b_ub = np.zeros(len(A_rows)) if A_rows else None
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bnds = [(-eps, eps)] * n
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success or res.x is None or np.max(np.abs(res.x)) < 1e-15:
            stalled += 1
            if stalled > 200:
                d_p = rng_st.randn(n) * eps * 0.3
                d_p -= d_p.mean()
                c = _repair(c + d_p, n)
                c5_cur = _eval(c)

def _slsqp_surr(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        c = _correl(x)
        m = c.max()
        idx = np.argpartition(c, -topk)[-topk:] if topk < len(c) else np.arange(len(c))
        ct = c[idx]
        s = np.clip(alpha * (ct - m), -400, 400)
        return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    def jac(x):
        g = np.zeros(n)
        cor = _correl(x)
        m = cor.max()
        idx = np.argpartition(cor, -topk)[-topk:].copy() if topk < len(cor) else np.arange(len(cor))
        ct = cor[idx]
        s = np.clip(alpha * (ct - m), -400, 400)
        w = np.exp(s - s.max()); w /= w.sum()
        for ii in range(len(idx)):
            if w[ii] > 1e-13:
                g += w[ii] * _grad_k(x, idx[ii], n)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-20, 'maxls': 40})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

    best_h, best_c5 = h.copy(), _eval(h)

    dl1 = min(deadline, t0 + 50)
    for a, tk, mi in [(20000, 15, 200), (80000, 8, 150)]:
        if time.time() > dl1:
            break
        hv = _slsqp_surr(best_h.copy(), a, tk, mi, dl1)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    dl2 = min(deadline, t0 + 320)
    hv, cv = _lp_adaptive(best_h.copy(), 40000, 0.004, 0.0004, 0.015, 0.0001, dl2)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    dl3 = min(deadline, t0 + 460)
    hv, cv = _pair_lp(best_h.copy(), 50000, rng, 20, dl3)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    dl4 = min(deadline, t0 + 560)
    hv, cv = _fw_subgrad(best_h.copy(), 30000, dl4, 0.0015)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    dl5 = min(deadline, t0 + 720)
    hv, cv = _lp_adaptive(best_h.copy(), 30000, 0.0015, 0.0002, 0.003, 0.00005, dl5)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    dl6 = min(deadline, t0 + 880)
    hv, cv = _pair_lp(best_h.copy(), 40000, rng, 30, dl6)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    rem = max(0, deadline - time.time())
    npol = min(int(rem / 0.0012), 50000)
    best_h, best_c5 = _fine_sweep(best_h, npol, rng, deadline)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
