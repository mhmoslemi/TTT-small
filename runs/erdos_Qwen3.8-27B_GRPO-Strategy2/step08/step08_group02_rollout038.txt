def _max_step(h,d):

def _repair(h,n):
    h=np.clip(h,0.0,1.0).astype(float).copy()
    target=n/2.0
    s=h.sum()
    if s<1e-12: return np.full(n,0.5)
    for _ in range(100):
        h*=target/s
        np.clip(h,0.0,1.0,out=h)
        s=h.sum()
        if abs(s-target)<1e-12: break
    diff=target-h.sum()
    if abs(diff)>1e-12:
        if diff>0:
            order=np.argsort(1.0-h)
        else:
            order=np.argsort(h)
        for idx in order:
            if diff>0:
                a=min(diff,1.0-h[idx]); h[idx]+=a; diff-=a
            else:
                a=min(-diff,h[idx]); h[idx]-=a; diff+=a
            if abs(diff)<1e-13: break
    return h

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = 0.5 * n
    s = float(h.sum())
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(120):
        h *= target / s
        np.clip(h, 0.0, 1.0, out=h)
        s = float(h.sum())
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h


def _project(h, target):
    lo, hi = -4.0, 4.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def _counts(h):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode="full")


def _c5_counts(c, n):
    return float(np.max(c) * (2.0 / n))


def _resize_to(arr, m):
    n = len(arr)
    if n == m:
        return _repair(arr.copy(), m)
    xo = np.linspace(0.0, 1.0, n, endpoint=False)
    xn = np.linspace(0.0, 1.0, m, endpoint=False)
    y = np.interp(xn, xo, np.asarray(arr, dtype=float))
    return _repair(y, m)


def _bit_random_ones(n, rng):
    m = n // 2
    ones = 0
    idx = rng.choice(n, size=m, replace=False)
    for x in idx:
        ones |= (1 << int(x))
    return ones


def _bit_max(ones, zeros, n):
    best = 0
    lo = -(n - 1)
    hi = n
    for d in range(lo, hi):
        if d >= 0:
            cnt = (ones & (zeros << d)).bit_count()
        else:
            cnt = (ones & (zeros >> (-d))).bit_count()
        if cnt > best:
            best = cnt
    return best


def _bit_counts_list(ones, zeros, n):
    out = []
    for d in range(-(n - 1), n):
        if d >= 0:
            cnt = (ones & (zeros << d)).bit_count()
        else:
            cnt = (ones & (zeros >> (-d))).bit_count()
        out.append(cnt)
    return out


def _bit_to_array(ones, n):
    a = np.zeros(n, dtype=float)
    for i in range(n):
        if (ones >> i) & 1:
            a[i] = 1.0
    return a


def _ons_from_array(arr):
    n = len(arr)
    target = n // 2
    a = np.clip(np.asarray(arr, dtype=float), 0.0, 1.0)
    mask = a > 0.5
    cnt = int(mask.sum())
    if cnt > target:
        idx = np.nonzero(mask)[0]
        order = np.argsort(a[idx])
        take = cnt - target
        for k in range(take):
            mask[idx[order[k]]] = 0
    elif cnt < target:
        idx = np.nonzero(~mask)[0]
        order = np.argsort(-a[idx])
        take = target - cnt
        for k in range(take):
            mask[idx[order[k]]] = 1
    cnt = int(mask.sum())
    if cnt > target:
        idx = np.nonzero(mask)[0]
        for k in range(cnt - target):
            mask[idx[k]] = 0
    elif cnt < target:
        idx = np.nonzero(~mask)[0]
        for k in range(target - cnt):
            mask[idx[k]] = 1
    ones = 0
    for pos in np.nonzero(mask)[0]:
        ones |= (1 << int(pos))
    return ones


def _bit_anneal(n, iters, t_end, rng, start_ones=None):

    if start_ones is None:
        ones = _bit_random_ones(n, rng)
    else:

    zeros = mask_all ^ ones

    h = [0] * n
    ones_l = []
    zeros_l = []
    pos_o = [-1] * n
    pos_z = [-1] * n
    for i in range(n):
        if (ones >> i) & 1:
            pos_o[i] = len(ones_l)
            ones_l.append(i)
            h[i] = 1
        else:

            zeros_l.append(i)

    cur_max = _bit_max(ones, zeros, n)
    best_max = cur_max
    best_ones = ones
    T0, T1 = 1.0, 1e-4
    den = max(1, iters - 1)

    for it in range(iters):
        if time.time() > t_end:
            break
        if not ones_l or not zeros_l:
            break

        move_i = -1
        move_j = -1
        ok = False

        if (it & 1) == 0 and cur_max > 0:
            counts = _bit_counts_list(ones, zeros, n)
            cm = max(counts)
            chosen_p = -1
            seen = 0
            for p, cv in enumerate(counts):
                if cv >= cm - 1:
                    seen += 1
                    if rng.random() < (1.0 / seen):
                        chosen_p = p
            if chosen_p >= 0:
                d = chosen_p - (n - 1)
                if d > 0:
                    seen = 0
                    chosen_k = -1
                    for k in range(n - d):
                        if h[k + d] and not h[k]:
                            seen += 1
                            if rng.random() < (1.0 / seen):
                                chosen_k = k
                    if seen > 0:
                        move_i = chosen_k + d
                        move_j = chosen_k
                        ok = True
                elif d < 0:
                    m = -d

                    chosen_k = -1
                    for k in range(n - m):

                                chosen_k = k
                    if seen > 0:
                        move_i = chosen_k
                        move_j = chosen_k + m
                        ok = True

        if ok and (move_i < 0 or move_j < 0 or pos_o[move_i] < 0 or pos_z[move_j] < 0):
            ok = False

        if not ok:
            move_i = ones_l[rng.randrange(len(ones_l))]
            move_j = zeros_l[rng.randrange(len(zeros_l))]

        new_ones = (ones & ~(1 << move_i)) | (1 << move_j)
        new_zeros = mask_all ^ new_ones
        new_max = _bit_max(new_ones, new_zeros, n)

        T = T0 * ((T1 / T0) ** (float(it) / float(den)))
        accept = False
        if new_max < cur_max:
            accept = True
        elif new_max == cur_max:
            accept = rng.random() < 0.5

            pos_o[move_j] = len(ones_l) - 1
            zeros_l.append(move_i)
            pos_z[move_i] = len(zeros_l) - 1

            cur_max = new_max
            if new_max < best_max:
                best_max = new_max

def _feasible_direction(h, d):
    d = d.copy()
    low = h <= 1e-12
    high = h >= 1.0 - 1e-12
    d[(low) & (d < 0.0)] = 0.0
    d[(high) & (d > 0.0)] = 0.0
    d -= d.mean()
    return d


def _max_step(h, d):
    ma = 1e100
    pos = d > 1e-14
    if np.any(pos):
        ma = min(ma, float(np.min((1.0 - h[pos]) / d[pos])))
    neg = d < -1e-14
    if np.any(neg):
        ma = min(ma, float(np.min(h[neg] / (-d[neg]))))
    if not np.isfinite(ma) or ma <= 1e-14:
        return 0.0
    return ma


def _direction_active(h, c, top_k, tau_frac, rng):
    n = len(h)
    cmax = float(np.max(c))
    if top_k <= 0:
        return np.zeros(n)
    top_k = min(int(top_k), len(c))
    idx = np.argsort(c)[-top_k:]
    if top_k == 1 and rng is not None and cmax > 0.0:
        near = np.nonzero(c >= cmax - 1e-9)[0]
        if len(near):
            idx = near[rng.randrange(len(near))].reshape(1)

    vals = np.asarray(c[idx], dtype=float)
    if len(idx) > 1 and tau_frac > 0.0:
        tau = max(1.0, cmax * tau_frac)
        s = (vals - cmax) / tau
        s = np.clip(s, -30.0, 0.0)
        w = np.exp(s)
        w /= w.sum()
    else:
        w = np.full(len(idx), 1.0 / len(idx))

    v = np.zeros(n, dtype=float)
    off = n - 1
    for ii in range(len(idx)):
        wp = float(w[ii])
        d = int(idx[ii]) - off
        if d == 0:
            v += wp * (2.0 * h - 1.0)
        elif d > 0:
            q = h[d:] * (1.0 - h[:n - d])
            if n - d > 0:
                v[d:] -= wp * q
                v[:n - d] += wp * q
        else:
            m = -d
            q = h[:n - m] * (1.0 - h[m:])
            if n - m > 0:
                v[:n - m] -= wp * q
                v[m:] += wp * q

    v -= v.mean()
    return v


def _active_descent(h, top_k, tau_frac, max_it, t_end, rng):
    n = len(h)
    h = h.copy()
    c = _counts(h)

        if mvc < 1e-14:
            break
        d = v / mvc
        d = _feasible_direction(h, d)
        mvd = float(np.max(np.abs(d)))
        if mvd < 1e-14:
            break
        d /= mvd
        ma = _max_step(h, d)
        if ma <= 1e-13:
            break
        ma *= 0.98

        best_val = curc
        best_ht = None
        best_ct = None
        for fr in (0.9, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005):
            a = fr * ma
            if a <= 1e-15:
                continue
            ht = h + a * d
            ct = _counts(ht)
            val = float(np.max(ct))
            if val < best_val - 1e-12:
                best_val = val
                best_ht = ht
                best_ct = ct

        if best_ht is not None:
            h = best_ht
            c = best_ct
            curc = best_val
            stall = 0

def _polish_continuous(h, t_end, rng):
