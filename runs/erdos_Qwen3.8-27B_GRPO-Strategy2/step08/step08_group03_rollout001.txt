def _local(h0, rng, deadline, s0=8e-5):
    n = len(h0)
    h = h0.copy()
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    c5 = c.max() * (2.0 / n)
    best_c = c5; best_h = h.copy()
    G = _gmat(h)
    step = s0; stale = 0
    tl = time.time()
    while time.time() < deadline:
        if stale > 3000:
            step *= 0.4; stale = 0
            if step < 1e-12: break
        el = (time.time() - tl) / max(1e-9, deadline - tl)
        bs = step * max(0.01, (1.0 - el))
        i = rng.randint(n); j = rng.randint(n)
        if i == j: continue
        d = bs * (0.3 + 1.4 * rng.random())
        if rng.random() < 0.5:
            a = min(1.0 - h[i], d); b = min(h[j], a)
            sg = 1
        else:
            a = min(h[i], d); b = min(1.0 - h[j], a)
            sg = -1
        if b < 1e-13: continue
        dd = sg * b
        dc = dd * (G[:, i] - G[:, j])
        nv = c.max() + dc
        mn = float(np.min(nv))
        cn = -mn if mn < 0 else float(np.max(nv))
        if cn < c5 - 1e-14:
            c = nv; c5 = cn
            h[i] += dd; h[j] -= dd
            _gmat_upd(G, i, dd); _gmat_upd(G, j, -dd)
            stale = 0
            if c5 < best_c: best_c = c5; best_h = h.copy()
        else:
            stale += 1
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m: return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    return _proj(np.interp(x1, x0, h), 0.5 * m)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    best = h0.copy(); best_c = _c5(best)
    for hh in (1.0 - h0, h0[::-1],
               0.5 + 0.5 * (h0 - h0[::-1]),
               0.5 + 0.5 * (h0 + h0[::-1] - 1.0)):
        h2 = _proj(hh, 0.5 * len(hh))
        c2 = _c5(h2)
        if c2 < best_c: best = h2.copy(); best_c = c2
    main_end = t0 + budget_s - 180
    alphas = [15, 60, 250, 1000, 4000, 15000, 60000, 200000, 500000]
    per = max(5.0, (main_end - time.time()) / len(alphas))
    hc = best.copy()
    for a in alphas:
        rem = main_end - time.time()
        if rem < 5: break
        slot = min(per, rem)
        bf, bc = _fista(hc, a, slot)
        if bc < best_c: best = bf; best_c = bc
        hc = bf
    if len(best) != n0:
        best = _resample(best, n0); best_c = _c5(best)
    ls_end = t0 + budget_s - 30
    if ls_end > time.time() + 10:
        best, best_c = _local(best, rng, ls_end, 8e-5)
    for hh in (1.0 - best, best[::-1],
               0.5 + 0.5 * (best - best[::-1]),
               0.5 + 0.5 * (best + best[::-1] - 1.0)):
        h2 = _proj(hh, 0.5 * len(hh))
        c2 = _c5(h2)
        if c2 < best_c: best = h2; best_c = c2
    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
