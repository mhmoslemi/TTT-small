import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_top(h, alpha, top_k):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    idx = np.argsort(c)[-top_k:]
    ct = c[idx]
    m = ct.max()
    s = alpha * (ct - m)
    s = np.clip(s, -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, ki in enumerate(idx):
        wk = w[ii]
        if wk < 1e-15:
            continue
        lag = ki - (n - 1)
        if lag >= 0:
            a = np.arange(n - lag)
            g[a + lag] += wk * (1.0 - h[a])
            if lag > 0:
                b = np.arange(lag, n)
                g[b] -= wk * h[b - lag]
        else:
            sp = -lag
            a = np.arange(sp, n)
            g[a] += wk * (1.0 - h[a - sp])
            b = np.arange(n - sp)
            g[b + sp] -= wk * h[b]
    return g

def _mgd(h, n_iter, alpha, top_k, step0, rng):
    n = len(h)
    v = np.zeros(n)

    cur_c = _eval(h)
    best_h = h.copy()
    best_c = cur_c

        g = _grad_top(h, alpha, top_k)
        v = mu * v - step * g
        h_try = _repair(h + v, n)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h, cur_c = h_try, c_try
            if c_try < best_c:
                best_c, best_h = c_try, h.copy()
            step = min(step * 1.03, step0 * 5.0)
        else:
            v *= 0.3
            step = max(step * 0.65, step0 * 0.03)
    return best_h, best_c
