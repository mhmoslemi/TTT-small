            if f < best_f: best_f, best_h = f, h.copy()
        else:
            step *= 0.5
            if step < step0 * 0.01:
                step = step0 * 0.1
                v = np.zeros(n)
            h2 = _proj(h - step * v, target)
            f2 = _eval(h2)
            if f2 < best_f: best_f, best_h = f2, h2.copy()
            if f2 <= f:
                h, f = h2, f2
    return best_h, best_f

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline: return h
    def obj(x):
        c = _correl(x); m = c.max()
        if topk < len(c):
            idx = np.argpartition(c, -topk)[-topk:]; ct = c[idx]
        else:
            ct = c
        s = np.clip(alpha * (ct - m), -400, 400)
        return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=lambda x: _grad_sm(x, alpha, topk)[0],
                     method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     constraints=cons, options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except:
        return h

def _anneal_guided(h, niter, rng, deadline, scale_explore, scale_noise):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    blocks = [2, 3, 4, 5, 7, 10, 15, 20]
    for it in range(niter):
        if time.time() > deadline: break
        T = 0.005 * (1e-10 / 0.005) ** (it / max(niter, 1))
        if it % 50 == 0:
            g, _ = _grad_exact(h, 0.002)
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        if it % 50 < 25:
            d = -0.002 * g[i0:i0+bs] + scale_noise * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        else:
            d = scale_explore * rng.randn(bs)
        d -= d.mean(); seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy(); hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc: bc, bh = c5n, h.copy()
    return bh, bc

def _polish(h, deadline, scale=1e-5):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    rng = np.random.RandomState(12345)
    while time.time() < deadline:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < bc: bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _grad_anneal(h, niter, rng, deadline, step0):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    T = 0.003
    for it in range(niter):
        if time.time() > deadline: break
        T *= 0.9997
        g, _ = _grad_exact(h, 0.003)
        noise = T * 0.5 * rng.randn(n)
        noise -= noise.mean()
        hn = _proj(h - step0 * (g + noise), n * 0.5)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-14)):
            h, c5c = hn, c5n
            if c5n < bc: bc, bh = c5n, h.copy()
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy(); best_c5 = _eval(best_h)

    # Phase 1: SLSQP quick refinement
    dl1 = min(deadline, t0 + 90)
    for alpha, tk, mi in [(5000, 40, 300), (20000, 20, 300), (80000, 10, 250)]:
        if time.time() > dl1: break
        h = _slsqp(best_h.copy(), alpha, tk, mi, dl1)
        c5v = _eval(h)
        if c5v < best_c5: best_c5, best_h = c5v, h.copy()

    # Phase 2: Smooth-to-sharp Nesterov (coarse alpha -> fine alpha)
    dl2 = min(deadline, t0 + 350)
    for alpha, tk, st in [(2000, 50, 0.02), (5000, 30, 0.01), (15000, 15, 0.005), (40000, 8, 0.003)]:
        if time.time() > dl2 - 30: break
        h2, c5v = _nesterov(best_h.copy(), 'smax', alpha, tk, 0, st, 3000, dl2, beta=0.87)
        if c5v < best_c5: best_c5, best_h = c5v, h2.copy()

    # Phase 3: Multi-restart exact Nesterov
    dl3 = min(deadline, t0 + 550)
    for scale in [0.003, 0.008, 0.015, 0.001, 0.02]:
        if time.time() > dl3 - 60: break
        hp = _repair(np.clip(best_h + scale * rng.randn(n), 0, 1), n)
        h2, c5v = _nesterov(hp, 'exact', 0, 0, 0.004, 0.003, 3000, dl3, beta=0.89)
        if c5v < best_c5: best_c5, best_h = c5v, h2.copy()

    # Phase 4: Gradient-guided annealing
    dl4 = min(deadline, t0 + 720)
    best_h, best_c5 = _anneal_guided(best_h.copy(), 40000, rng, dl4, 0.003, 0.002)

    # Phase 5: Gradient anneal (fine)
    dl5 = min(deadline, t0 + 850)
    best_h, best_c5 = _grad_anneal(best_h.copy(), 20000, rng, dl5, 0.001)

    # Phase 6: Very fine Nesterov
    dl6 = min(deadline, t0 + budget_s - 60)
    h2, c5v = _nesterov(best_h.copy(), 'exact', 0, 0, 0.0003, 0.0004, 8000, dl6, beta=0.93)
    if c5v < best_c5: best_c5, best_h = c5v, h2.copy()

    # Also try from perturbed best
    if time.time() < dl6 - 30:
        hp = _repair(np.clip(best_h + 0.001 * rng.randn(n), 0, 1), n)
        h3, c5v3 = _nesterov(hp, 'exact', 0, 0, 0.0005, 0.001, 3000, dl6, beta=0.91)
        if c5v3 < best_c5: best_c5, best_h = c5v3, h3.copy()

    # Phase 7: Final polish
    best_h, best_c5 = _polish(best_h, deadline, scale=5e-6)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
