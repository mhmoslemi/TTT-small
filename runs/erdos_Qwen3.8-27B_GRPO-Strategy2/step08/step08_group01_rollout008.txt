def _adam_lse(h, n_iter, alpha0, alpha1, top_k, lr, t0, be):
    n = len(h)
    target = n / 2.0
    b1, b2, eps = 0.85, 0.999, 1e-8
    mu = np.zeros(n)
    sv = np.zeros(n)
    cur_c = _eval(h)
    best_h = h.copy()
    best_c = cur_c
    for it in range(n_iter):
        if time.time() > t0 + be - 30:
            break
        frac = it / max(n_iter - 1, 1)
        alpha = alpha0 * (alpha1 / alpha0) ** frac
        val, g = _lse_val_grad(h, alpha, top_k)
        mu = b1 * mu + (1 - b1) * g
        sv = b2 * sv + (1 - b2) * g * g
        mu_hat = mu / (1 - b1 ** (it + 1))
        sv_hat = sv / (1 - b2 ** (it + 1))

        h_try = h - step
        h_try = _repair(h_try, n)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try
            cur_c = c_try

                best_h = h.copy()

def _slsqp_run(h, alpha, maxiter, top_k):
    n = len(h)
    target = n / 2.0
    bnds = [(0.0, 1.0)] * n
    def _obj(x):
        v, _ = _lse_val_grad(x, alpha, top_k)
        return v
    def _grad(x):
        _, g = _lse_val_grad(x, alpha, top_k)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return out

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    be = t0 + budget_s

    h_parent = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Adam with increasing alpha (focus on max)
    adam_cfgs = [
        (300, 500, 100, 6, 0.01, 200),
        (500, 3000, 60, 5, 0.008, 250),
        (3000, 50000, 30, 4, 0.004, 300),
        (8000, 200000, 15, 4, 0.002, 200),
    ]
    for ni, a0, a1, tk, lr, tlimit in adam_cfgs:
        if time.time() > be - 350:
            break
        h_t = best_h.copy()
        h_t, c = _adam_lse(h_t, ni, a0, a1, tk, lr, t0, budget_s)
        c = _eval(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 2: Multi-start perturbations + Adam
    if time.time() < be - 400:
        for trial in range(5):
            if time.time() > be - 370:
                break
            scale = 0.015 * (trial + 1)
            noise = scale * rng.randn(n)
            noise -= noise.mean()
            h_p = _repair(best_h + noise, n)
            c_p = _eval(h_p)
            if c_p < best_c5:
                best_c5 = c_p
                best_h = h_p.copy()
            h_t, c = _adam_lse(h_p, 600, 2000, 50000, 5, 0.005, t0, budget_s)
            c = _eval(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()

    # Phase 3: SLSQP with multiple alphas
    if time.time() < be - 300:
        for alpha, tk, mxi in [(2000, 10, 500), (15000, 6, 400), (100000, 4, 300), (500000, 3, 250)]:
            if time.time() > be - 260:
                break
            h_t = _slsqp_run(best_h.copy(), alpha, mxi, tk)
            c = _eval(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()

    # Phase 4: Multi-resolution
    if time.time() < be - 350:
        for ratio in [2, 4]:
            if time.time() > be - 320:
                break
            nc = n // ratio
            hc = _repair(_downsample(best_h, ratio), nc)
            hc, _ = _adam_lse(hc, 300, 1000, 50000, 5, 0.01, t0, budget_s)
            h_up = _repair(_upsample(hc, ratio), n)
            c_up = _eval(h_up)
            if c_up < best_c5:
                best_c5 = c_up
                best_h = h_up.copy()
            best_h, best_c5 = _local2(best_h.copy(), 3000, 1e-4, rng)

    # Phase 5: SLSQP after multi-res
    if time.time() < be - 200:
        for alpha, tk, mxi in [(8000, 7, 400), (50000, 5, 300), (300000, 4, 200)]:
            if time.time() > be - 170:
                break
            h_t = _slsqp_run(best_h.copy(), alpha, mxi, tk)
            c = _eval(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()

    # Phase 6: Adam fine-tune
    if time.time() < be - 150:
        h_t, c = _adam_lse(best_h.copy(), 2000, 100000, 1000000, 3, 0.001, t0, budget_s)
        c = _eval(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 7: SA local search
    if time.time() < be - 100:
        rem = be - 100 - time.time()
        nsa = min(int(rem * 1200), 60000)
        best_h, best_c5 = _local2(best_h, nsa, 4e-5, rng, max(1e-6, best_c5 * 5e-5))

    # Phase 8: Sweep local
    if time.time() < be - 60:
        rem = be - 60 - time.time()
        npass = min(int(rem * 100), 1500)
        best_h, best_c5 = _sweep_local(best_h, npass, 3e-5, rng)

    # Phase 9: 3-point local
    if time.time() < be - 30:
        rem = be - 30 - time.time()
        n3 = min(int(rem * 800), 40000)
        best_h, best_c5 = _local3(best_h, n3, 2e-5, rng)

    # Phase 10: Ultra-fine final polish
    if time.time() < be - 5:
        rem = be - 5 - time.time()
        nfin = min(int(rem * 3000), 150000)
        best_h, best_c5 = _local2(best_h, nfin, 3e-6, rng)
        # One more SLSQP with very high alpha
        if time.time() < be - 10:
            h_t = _slsqp_run(best_h.copy(), 1000000, 200, 3)
            c = _eval(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()
        rem2 = be - 3 - time.time()
        nfin2 = min(int(rem2 * 3500), 80000)
        if nfin2 > 100:
            best_h, best_c5 = _local2(best_h, nfin2, 1e-6, rng)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
