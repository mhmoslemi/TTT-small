def _do_slsqp(h0, alpha, ml, maxiter, ftol, t_end):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h0=h0, jac=_smooth_grad,
                       args=(alpha, ml), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h0

def _proj_box_sum(h, n):
    target = n * 0.5
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _proj_grad(h, alpha, ml, steps, step0, t_end):
    n = len(h); dv = 2.0 / n
    v = np.zeros(n); mu = 0.55; step = step0
    cur = _eval(h)
    for _ in range(steps):
        if time.time() > t_end: break
        g = _smooth_grad(h, alpha, ml)
        g -= g.mean()
        v = mu * v + (1.0 - mu) * g
        vn = np.linalg.norm(v)
        if vn < 1e-20: break
        v /= vn
        taken = False
        for _t in range(12):
            h_new = _proj_box_sum(h - step * v, n)
            c_new = _eval(h_new)
            if c_new < cur - 1e-16:
                h = h_new; cur = c_new
                step = min(step * 1.08, step0 * 10.0)
                taken = True; break
            step *= 0.4
        if not taken:
            step = min(step * 1.2, step0 * 10.0)
            if step < step0 * 1e-7: break
    return _repair(h, n)

def _local_search(h, n_iter, rng, base_step, t_end):
    n = len(h)
    c5 = _eval(h); best_h = h.copy(); best_c = c5
    step = base_step
    for it in range(n_iter):
        if time.time() > t_end: break
        i = rng.randint(n)
        span = min(n - 1, max(2, int(0.3 * n)))
        j = (i + 1 + rng.randint(0, span - 1)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5 - 1e-16:
                c5 = c5n
                if c5n < best_c:
                    best_c = c5n; best_h = h.copy()
                step = min(step * 1.01, base_step * 5.0)
            else:
                h[i], h[j] = ni - d, nj + d
                step *= 0.97
    return best_h, best_c

def _grad_local_search(h, n_iter, rng, t_end):
    n = len(h); dv = 2.0 / n
    c5 = _eval(h); best_h = h.copy(); best_c = c5
    step = 2e-5
    for it in range(n_iter):
        if time.time() > t_end: break
        raw = np.correlate(h, 1.0 - h, mode='full')
        c = raw * dv
        cm = c.max()
        kstar = np.argmax(c)
        L = kstar - (n - 1)
        d = abs(L)
        g = np.zeros(n)
        if L >= 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:n]
        else:
            g[:n - d] += (1.0 - h[d:])
            g[d:] -= h[:n - d]
        g -= g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-20: break
        gdir = -g / gnorm
        n_pert = rng.randint(3, 12)
        idx = rng.randint(0, n, size=n_pert)
        moves = step * gdir[idx]
        moves -= moves.mean()
        h_new = h.copy()
        h_new[idx] += moves
        if h_new.min() < 0 or h_new.max() > 1:
            h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5 - 1e-16:
            h = h_new; c5 = c5n
            if c5n < best_c:
                best_c = c5n; best_h = h.copy()
            step = min(step * 1.01, 2e-4)
        else:
            rng2 = rng.randint(n)
            j = (rng2 + 1 + rng.randint(0, 50)) % n
            dd = step * rng.randn()
            ni, nj = h[rng2] + dd, h[j] - dd
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[rng2], h[j] = ni, nj
                c5n2 = _eval(h)
                if c5n2 < c5:
                    c5 = c5n2
                    if c5n2 < best_c:
                        best_c = c5n2; best_h = h.copy()
                else:
                    h[rng2], h[j] = ni - dd, nj + dd
            step *= 0.96
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):

                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n; best_h = h2.copy()
                else:
