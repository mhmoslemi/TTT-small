```python
from scipy.optimize import linprog

def _linearized_step(h, n, dx, top_k=None, tol=0.001):
    """One step of linearized minimax."""
    c = _corr(h)
    cmax = c.max()
    # Select active constraints (within tol of max)
    if top_k is not None:
        active = np.argsort(c)[-top_k:]
    else:
        active = np.where(c >= cmax - tol)[0]
    
    # Compute gradients for active shifts
    grads = []
    for k in active:
        grads.append(_grad_shift(k, h, n, dx))  # shape (n,)
    grads = np.array(grads)  # shape (len(active), n)
    
    # LP: min t
    # s.t. grads @ delta <= t - c[active]
    #      -1 <= delta_i <= 1 - h_i
    #      sum(delta) = 0
    
    m = len(active)
    # Variables: [delta_0, ..., delta_{n-1}, t]
    # Objective: minimize t -> c = [0,...,0,1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active k: grad_k @ delta - t <= -c_k
    # i.e., [grad_k, -1] @ [delta, t] <= -c_k
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = -c[active]
    
    # Bounds on delta: -h_i <= delta_i <= 1 - h_i
    bounds = [(-h[i], 1 - h[i]) for i in range(n)] + [(None, None)]
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')

        delta = res.x[:n]
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - tgt) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > tgt:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = tgt - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corrl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_one(k, h, n, dx):
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _lse_f(x, alpha, n):
    c = _corrl(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _lse_g(x, alpha, n):
    c = _corrl(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n); dx = 2.0 / n
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        g += _grad_one(k, x, n, w[k] * dx)
    return g

class _LSE:
    def __init__(s, a, n):
        s.a = a; s.n = n
    def f(s, x):
        return _lse_f(x, s.a, s.n)
    def g(s, x):
        return _lse_g(x, s.a, s.n)
    def cf(s, x):
        return x.sum() - s.n * 0.5
    def cj(s, x):
        return np.ones(s.n)

def _slsqp(h0, alpha, n, maxit, ftol):
    o = _LSE(alpha, n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': o.cf, 'jac': o.cj}]
    try:
        res = minimize(o.f, h0, jac=o.g, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h0

def _lp_minimax(h, n, dx, top_k, max_it, t_end):
    best_h, best_c = h.copy(), _eval(h)
    for _ in range(max_it):
        if time.time() > t_end:
            break
        c = _corrl(h)
        active = np.argsort(c)[-top_k:]
        grads = np.array([_grad_one(k, h, n, dx) for k in active])
        c_act = c[active]
        m = len(active)
        c_obj = np.zeros(n + 1); c_obj[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = grads
        A_ub[:, n] = -1.0
        b_ub = -c_act.copy()
        bounds = [(-h[i], 1.0 - h[i]) for i in range(n)] + [(None, None)]
        A_eq = np.zeros((1, n + 1)); A_eq[0, :n] = 1.0
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq,
                          b_eq=np.zeros(1), bounds=bounds, method='highs')
        except Exception:
            break
        if not res.success:
            break
        delta = res.x[:n]
        if np.max(np.abs(delta)) < 1e-13:
            break
        lam = 1.0
        ok = False
        for _ in range(25):
            hn = _repair(h + lam * delta, n)
            cn = _eval(hn)
            if cn < best_c:
                best_c, best_h, h = cn, hn.copy(), hn
                ok = True; break
            lam *= 0.6
        if not ok:
            break
    return best_h, best_c

def _adam_act(h, n, dx, top_k, n_it, t_end, lr):
    best_h, best_c = h.copy(), _eval(h)
    m1 = np.zeros(n); m2 = np.zeros(n); t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    for _ in range(n_it):
        if time.time() > t_end:
            break
        t += 1
        c = _corrl(h)
        active = np.argsort(c)[-top_k:]
        g = np.zeros(n)
        for k in active:
            g += _grad_one(k, h, n, dx)
        g /= len(active)
        m1 = b1 * m1 + (1 - b1) * g
        m2 = b2 * m2 + (1 - b2) * g * g
        m1h = m1 / (1 - b1 ** t)
        m2h = m2 / (1 - b2 ** t)
        step = lr * m1h / (np.sqrt(m2h) + eps)
        hn = _repair(h - step, n)
        cn = _eval(hn)
        if cn < best_c:
            best_c, best_h, h = cn, hn.copy(), hn
    return best_h, best_c

def _local_3pt(h, n, rng, t_end):
    best_h, best_c = h.copy(), _eval(h)
    while time.time() < t_end:
        i = rng.randint(n)
        j = (i + rng.randint(2, 10)) % n
        k = (i + rng.randint(11, 25)) % n
        d1 = 1.5e-5 * rng.randn(); d2 = 1.5e-5 * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if ni < 0 or ni > 1 or nj < 0 or nj > 1 or nk < 0 or nk > 1:
            continue
        h[i], h[j], h[k] = ni, nj, nk
        cn = _eval(h)
        if cn < best_c:
            best_c, best_h = cn, h.copy()
        else:
            h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return best_h, best_c

def _local_2pt(h, n, rng, t_end):
    best_h, best_c = h.copy(), _eval(h)
    while time.time() < t_end:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = 5e-7 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _eval(h)
        if cn < best_c:
            best_c, best_h = cn, h.copy()
        else:
            h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    dx = 2.0 / n
    bh = h.copy(); bc5 = _eval(bh)

    def _upd(hn):
        nonlocal bh, bc5
        cv = _eval(hn)
        if cv < bc5:
            bc5, bh = cv, hn.copy()

    # Phase 1: SLSQP low alpha
    te = t0 + budget_s * 0.08
    if time.time() < te - 2:
        _upd(_slsqp(bh.copy(), 300, n, 3000, 1e-14))

    # Phase 2: LP minimax broad
    te = t0 + budget_s * 0.18
    if time.time() < te - 2:
        bh, bc5 = _lp_minimax(bh, n, dx, 60, 600, te)

    # Phase 3: SLSQP medium
    te = t0 + budget_s * 0.28
    if time.time() < te - 2:
        _upd(_slsqp(bh.copy(), 5000, n, 3000, 1e-15))

    # Phase 4: LP minimax medium
    te = t0 + budget_s * 0.38
    if time.time() < te - 2:
        bh, bc5 = _lp_minimax(bh, n, dx, 25, 600, te)

    # Phase 5: Adam on active set
    te = t0 + budget_s * 0.50
    if time.time() < te - 2:
        bh, bc5 = _adam_act(bh, n, dx, 30, 200000, te, 0.004)

    # Phase 6: SLSQP high
    te = t0 + budget_s * 0.60
    if time.time() < te - 2:
        _upd(_slsqp(bh.copy(), 50000, n, 2500, 1e-15))

    # Phase 7: LP minimax tight
    te = t0 + budget_s * 0.70
    if time.time() < te - 2:
        bh, bc5 = _lp_minimax(bh, n, dx, 12, 500, te)

    # Phase 8: Adam fine
    te = t0 + budget_s * 0.80
    if time.time() < te - 2:
        bh, bc5 = _adam_act(bh, n, dx, 10, 200000, te, 0.001)

    # Phase 9: SLSQP very high
    te = t0 + budget_s * 0.88
    if time.time() < te - 2:
        _upd(_slsqp(bh.copy(), 300000, n, 2000, 1e-16))

    # Phase 10: LP final
    te = t0 + budget_s * 0.92
    if time.time() < te - 2:
        bh, bc5 = _lp_minimax(bh, n, dx, 8, 300, te)

    # Phase 11: 3-point local search
    te = t0 + budget_s * 0.97
    if time.time() < te - 2:
        bh, bc5 = _local_3pt(bh, n, rng, te)

    # Phase 12: 2-point ultra-fine
    te = t0 + budget_s - 1.0
    if time.time() < te - 0.5:
        bh, bc5 = _local_2pt(bh, n, rng, te)

    bh = _repair(bh, n)
    bc5 = _eval(bh)
    return (bh, bc5, n)
```
