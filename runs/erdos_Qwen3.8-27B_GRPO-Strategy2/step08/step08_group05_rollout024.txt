def _smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(-c, topk)[:topk]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    logsum = s.max() + np.log(np.sum(np.exp(s - s.max())))
    return m + logsum / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(-c, topk)[:topk].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-12:
            continue
        g += w[ii] * _grad_k(h, idx[ii], n)
    return g

def _proj_sub(h, target):
    """Project onto {h : 0<=h<=1, sum(h)=target} using bisection"""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _mom_grad(h, alpha, topk, steps, lr, deadline):
    """Momentum gradient descent with smooth max"""
    n = len(h)
    f = _smax(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.9
    best_h, best_f = h.copy(), _eval(h)
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        v = beta * v + (1 - beta) * g
        h_new = _proj_sub(h - lr * v, n * 0.5)
        f_new = _smax(h_new, alpha, topk)
        if f_new <= f + 1e-16:
            h = h_new
            f = f_new
        else:
            # Halve step
            h_new = _proj_sub(h - lr * 0.5 * v, n * 0.5)
            f = _smax(h_new, alpha, topk)
            h = h_new
        c5 = _eval(h)
        if c5 < best_f - 1e-15:
            best_f, best_h = c5, h.copy()
    return best_h, best_f

def _slsqp_opt(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smax(x, alpha, topk)
    def jac(x):
        return _smax_grad(x, alpha, topk)
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
                                   'jac': lambda x: np.ones(n)}],
                     options={'maxiter': maxit, 'ftol': 1e-18, 'maxls': 50})
        if r.success:
            return _repair(r.x, n)
        return _repair(r.x, n)
    except Exception:
        return h

def _bit_flip(h, n, n_flips, rng, deadline):
    """Try flipping small blocks of values"""
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_c = best_c5
    for _ in range(n_flips):
        if time.time() > deadline:
            break

        h1 = c.copy()
        h1[i0:i0 + bs] = comp

        c5_1 = _eval(h1)

        d -= d.mean()

        h2 = _repair(h2, n)

        if c5_1 < c5_c:
            c, c5_c = h1, c5_1
            if c5_c < best_c5:
                best_c5, best_h = c5_c, c.copy()
        elif c5_2 < c5_c:

            if c5_c < best_c5:

def _polish(h, n_iter, rng, deadline):
    """Fine two-point swap polish"""
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c5_c = best_c5
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        ss = 1e-5 * (0.5 ** (min(_ // 2000, 6)))
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = ss * (2 * rng.random() - 1)
        ni, nj = h[i] + d, h[j] - d

                if c5_n < best_c5:
                    best_c5, best_h = c5_n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

        else:
            lr *= 0.5
            if lr < 1e-12:
                break
            h = _proj_sub(h - lr * v, n * 0.5)
            f = _smax(h, alpha, topk)
        c5 = _eval(h)
        if c5 < best_c5 - 1e-15:
            best_c5, best_h = c5, h.copy()
    return best_h, best_c5

def _perturb_reopt(h, n, rng, deadline, n_rounds):
    best_h, best_c5 = h.copy(), _eval(h)
    for _ in range(n_rounds):
        if time.time() > deadline:
            break
        bs = rng.randint(2, 10)
        i0 = rng.randint(0, max(1, n - bs))
        scale = rng.uniform(0.05, 0.2)
        d = scale * rng.randn(bs)
        d -= d.mean()
        h2 = h.copy()
        h2[i0:i0 + bs] = np.clip(h2[i0:i0 + bs] + d, 0, 1)
        h2 = _repair(h2, n)
        c5_2 = _eval(h2)
        if c5_2 < best_c5:
            best_h, best_c5 = h2.copy(), c5_2
            h = h2
            hv, cv = _lp_descent(h.copy(), min(deadline, time.time() + 30),
                                  0.003, 0.001, 5, 200)
            if cv < best_c5:
                best_c5, best_h = cv, hv.copy()
                h = hv.copy()
    return best_h, best_c5

def _polish(h, n_iter, rng, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c5_c = best_c5
    for i in range(n_iter):
        if time.time() > deadline:
            break
        ss = 5e-6 * (0.4 ** (min(i // 3000, 8)))
        if ss < 1e-8:
            break
        a = rng.randint(n)
        b = (a + 1 + rng.randint(0, 3)) % n
        d = ss * (2 * rng.random() - 1)
        na, nb = h[a] + d, h[b] - d
        if 0 <= na <= 1 and 0 <= nb <= 1:
            h[a], h[b] = na, nb
            c5_n = _eval(h)
            if c5_n < c5_c - 1e-19:
                c5_c = c5_n
                if c5_n < best_c5:
                    best_c5, best_h = c5_n, h.copy()
            else:
                h[a], h[b] = na - d, nb + d
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: Quick SLSQP (50s)
    dl1 = min(deadline, t0 + 50)
    for a, tk in [(1500, 25), (8000, 12)]:
        if time.time() > dl1:
            break
        hv = _slsqp_opt(best_h.copy(), a, tk, 150, dl1)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 2: Coarse-to-fine LP descent (350s)
    dl2 = min(deadline, t0 + 400)
    configs = [
        (0.010, 0.005, 5, 3000),
        (0.005, 0.002, 6, 3000),
        (0.003, 0.001, 5, 3000),
        (0.002, 0.0005, 6, 3000),
    ]
    for e0, e1, mt, mi in configs:
        if time.time() > dl2:
            break
        hv, cv = _lp_descent(best_h.copy(), dl2, e0, e1, mt, mi)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 3: Perturbation + re-optimization (130s)
    dl3 = min(deadline, t0 + 530)
    hv, cv = _perturb_reopt(best_h.copy(), n, rng, dl3, 40)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 4: Fine LP descent (200s)
    dl4 = min(deadline, t0 + 730)
    hv, cv = _lp_descent(best_h.copy(), dl4, 0.002, 0.0003, 5, 8000)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()
    hv, cv = _lp_descent(best_h.copy(), dl4, 0.001, 0.0002, 4, 5000)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 5: SLSQP + Momentum (100s)
    dl5 = min(deadline, t0 + 830)
    hv = _slsqp_opt(best_h.copy(), 20000, 10, 200, dl5)
    cv = _eval(hv)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()
    hv, cv = _mom_grad(best_h.copy(), 100000, 8, 2000, 0.0008, dl5)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 6: Final LP + polish (remaining)
    hv, cv = _lp_descent(best_h.copy(), deadline, 0.001, 0.0002, 4, 3000)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    rem = max(0, deadline - time.time())
    npol = min(int(rem / 0.001), 60000)
    best_h, best_c5 = _polish(best_h, max(npol, 1000), rng, deadline)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
