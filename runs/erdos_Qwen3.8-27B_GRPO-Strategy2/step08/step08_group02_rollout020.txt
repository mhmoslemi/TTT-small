        # dC/dh[i] = (1-h[i-sp])*dx for i in [sp, n-1]
        # dC/dh[i-sp] = -h[i]*dx for i in [sp, n-1], i.e., j in [0, n-1-sp]
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp]) * dx
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp] * dx
    return g

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _penalty_grad(h, alpha, top_k):
    """Penalty gradient: sum over top-k lags weighted by softplus of excess."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    if top_k > 0 and top_k < len(c):
        # Use softmax weights
        s = np.clip(alpha * (c - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        active = np.where(w > 1e-10)[0]
    else:
        s = np.clip(alpha * (c - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        active = np.where(w > 1e-10)[0]
    g = np.zeros(n)
    for idx in active:
        if w[idx] < 1e-12:
            continue
        g += w[idx] * _grad_corr(idx, h, n)
    return g

def _proj_grad_opt(h, alpha, steps, step0, top_k, momentum, t_max):
    """Projected gradient with penalty, time-limited."""
    n = len(h)
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    for it in range(steps):
        if time.time() > t_max:
            break
        g = _penalty_grad(h, alpha, top_k)
        v = momentum * v + (1.0 - momentum) * g
        taken = False
        for _t in range(15):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-16:
                h = h_new
                cur_c = c_new
                step = min(step * 1.15, step0 * 15.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.3, step0 * 15.0)
        if it % 300 == 0:
            v *= 0.3
    return _repair(h, n), cur_c

def _slsqp_focus(h, alpha, maxiter, top_k, t_max):
    n = len(h)
    dx = 2.0 / n
    def obj_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def grad_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()

        active = np.where(w > 1e-10)[0]
        for idx in active:
            g += w[idx] * _grad_corr(idx, x, n)
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_worst(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max()), int(np.argmax(c))

def _cd_sweep(h, n_sweeps=5, rng=None, damp=1.0):
    n = len(h)
    dx = 2.0 / n
    N = 2 * n - 1
    lags = np.arange(N) - (n - 1)
    one_h = 1.0 - h
    C = np.correlate(h, one_h, mode='full') * dx
    for sweep in range(n_sweeps):
        improved = False
        order = np.arange(n) if rng is None else rng.permutation(n)
        for i in order:
            omh = 1.0 - h
            f1_idx = i + lags
            f2_idx = i - lags
            B = np.zeros(N)
            v1 = (f1_idx >= 0) & (f1_idx < n)
            B[v1] = dx * omh[f1_idx[v1]]
            v2 = (f2_idx >= 0) & (f2_idx < n)
            B[v2] -= dx * h[f2_idx[v2]]
            C = np.correlate(h, omh, mode='full') * dx
            A = C - B * h[i]
            hi_v = min(C.max(), (A + B).max())
            lo_v = A.min() - np.abs(B).max()
            if hi_v - lo_v < 1e-15:
                t_opt = hi_v
            else:
                for _ in range(45):
                    mid = (lo_v + hi_v) * 0.5
                    pos = B > 1e-14
                    neg = B < -1e-14
                    zro = ~pos & ~neg
                    ok = True
                    if np.any(zro) and A[zro].max() > mid:
                        ok = False
                    if ok:
                        t_lo = 0.0; t_hi = 1.0
                        if np.any(pos):
                            t_hi = min(t_hi, float((mid - A[pos]).min() / B[pos].max() * B[pos].max()))
                        if np.any(neg):
                            bnd = (mid - A[neg]) / B[neg]
                            t_lo = max(t_lo, float(bnd.max()))
                        if np.any(pos):
                            bnd_p = (mid - A[pos]) / B[pos]
                            t_hi = min(t_hi, float(bnd_p.min()))
                        if t_lo > t_hi + 1e-14:
                            ok = False
                    if ok:
                        hi_v = mid
                    else:
                        lo_v = mid
                t_opt = hi_v
            t_new = h[i] + damp * (t_opt - h[i])
            if t_new < 0: t_new = 0.0
            elif t_new > 1: t_new = 1.0
            if abs(t_new - h[i]) > 1e-13:
                h[i] = t_new
                improved = True
        if not improved:
            break
    return h

def _grad_corr(k, h, n):
    dx = 2.0 / n
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        ii = np.arange(n - s)
        g[ii] += (1.0 - h[ii + s]) * dx
        ii2 = np.arange(s, n)
        g[ii2] -= h[ii2 - s] * dx
    else:
        sp = -s
        ii = np.arange(sp, n)
        g[ii] += (1.0 - h[ii - sp]) * dx
        ii2 = np.arange(n - sp)
        g[ii2] -= h[ii2 + sp] * dx
    return g

def _proj(h, target):
    lo, hi = -4.0, 4.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _penalty_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for idx in active:
        g += w[idx] * _grad_corr(idx, h, n)
    return g

def _proj_grad(h, alpha, steps, step0, momentum, t_max):
    n = len(h)
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    for it in range(steps):
        if time.time() > t_max:
            break
        g = _penalty_grad(h, alpha)
        v = momentum * v + (1.0 - momentum) * g
        taken = False
        for _t in range(14):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-16:
                h = h_new; cur_c = c_new
                step = min(step * 1.12, step0 * 12.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.25, step0 * 12.0)
        if it % 200 == 0:
            v *= 0.25
    return _repair(h, n), cur_c

def _slsqp_focus(h, alpha, maxiter, t_max):
    n = len(h)
    dx = 2.0 / n
    def obj_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def grad_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        s -= s.max()
        w = np.exp(s); w /= w.sum()
        g = np.zeros(n)
        for idx in np.where(w > 1e-10)[0]:
            g += w[idx] * _grad_corr(idx, x, n)
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _local_informed(h, n_iter, rng, base_step, t_max):
    n = len(h)
    c5c, k_w = _eval_worst(h)
    best_h, best_c = h.copy(), c5c
    for _ in range(n_iter):
        if time.time() > t_max:
            break
        s = k_w - (n - 1)
        if 0 < abs(s) < n:
            i = rng.randint(n - abs(s))
            j = i + abs(s)
            d = abs(base_step * rng.randn())
            ni, nj = h[i] - d, h[j] + d
            if 0.0 <= ni and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n, k_w = _eval_worst(h)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c:
                        best_c, best_h = c5n, h.copy()
                else:
                    h[i], h[j] = ni + d, nj - d
                continue
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        d = base_step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n, k_w = _eval_worst(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c, best_h = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _local_2pt(h, n_iter, rng, base_step, t_max):
