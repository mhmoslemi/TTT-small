    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += (1.0 - h[i1[m1]])
    i2 = j + off
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

def _proj_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lp_descent(h, n_steps, eps, m_top, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c5_cur = best_c5
    cur = h.copy()
    stall = 0
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        cor = _correl(cur)
        idx_top = np.argpartition(cor, -m_top)[-m_top:]
        idx_top = idx_top[np.argsort(-cor[idx_top])]
        grads = np.array([_grad_k(cur, k, n) for k in idx_top])
        c_obj = -grads[0]
        A_ub = grads[1:] if m_top > 1 else None
        b_ub = np.zeros(m_top - 1) if m_top > 1 else None
        bnds = [(-eps, eps)] * n
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=np.ones((1, n)), b_eq=np.zeros(1),
                          bounds=bnds, method='highs')
        except Exception:
            break
        if not res.success or res.x is None:
            stall += 1
            if stall > 100:
                break
            continue
        if np.max(np.abs(res.x)) < 1e-15:
            stall += 1
            if stall > 100:
                break
            d = np.random.RandomState(stall).randn(n)
            d -= d.mean(); d *= eps * 0.1
        else:
            stall = 0
            d = res.x
        step = 1.0
        improved = False
        for _ in range(8):
            h_new = _repair(cur + step * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-15:
                cur = h_new.copy()
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h_new.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            rng_p = np.random.RandomState(int(time.time() * 1e6) % 999983)
            d2 = rng_p.randn(n)
            d2 -= d2.mean()
            d2 *= eps * 0.3 / max(np.max(np.abs(d2)), 1e-15)
            h_new = _repair(cur + d2, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur:
                cur = h_new.copy()
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h_new.copy()
    return best_h, best_c5

def _smax(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:] if topk < len(c) else np.arange(len(c))
    s = np.clip(alpha * (c[idx] - m), -300, 300)
    sm = s.max()
    return m + np.log(np.sum(np.exp(s - sm)) + 1e-30) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:].astype(int) if topk < len(c) else np.arange(len(c))
    s = np.clip(alpha * (c[idx] - m), -300, 300)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii, k in enumerate(idx):
        if w[ii] < 1e-13:
            continue
        g += w[ii] * dx * _grad_k(h, int(k), n)
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.85
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _grad_smax(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        h_new = _proj_box_sum(h - step * v, n * 0.5)
        f_new = _smax(h_new, alpha, topk)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.03, step0 * 20.0)
        else:
            step *= 0.5
            if step < 1e-13:
                break
            h = _proj_box_sum(h - step * v, n * 0.5)
            f = _smax(h, alpha, topk)
    return _repair(h, n)

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smax(x, alpha, topk)
    def jac(x):
        return _grad_smax(x, alpha, topk)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5, 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18, 'maxls': 40})
        return _repair(r.x if r.x is not None else h, n)
    except Exception:
        return _repair(h, n)

def _anneal(h, n_iter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(n_iter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.002 * max(T, 1e-10) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _polish(h, n_iter, rng, deadline):
    n = len(h)
    bh = h.copy()
    bc = _eval(h)
    c5c = bc
    scales = [1e-6, 5e-7, 2e-7, 1e-7, 5e-8, 2e-8]
    per = max(1, n_iter // len(scales))
    for scale in scales:
        for _ in range(per):
            if time.time() > deadline:
                return bh, bc
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 7)) % n
            d = scale * (2 * rng.random() - 1)
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _eval(h)
                if c < c5c - 1e-19:
                    c5c = c
                    if c < bc:
                        bc, bh = c, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc

    g = np.zeros(n)
    dx = 2.0 / n
    for ii, k in enumerate(idx):
        if w[ii] < 1e-13:
            continue
        g += w[ii] * dx * _grad_k(h, int(k), n)
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.85
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _grad_smax(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        h_new = _proj_box_sum(h - step * v, n * 0.5)
        f_new = _smax(h_new, alpha, topk)
        if f_new < f:
            h = h_new
            f = f_new
            step = min(step * 1.03, step0 * 15.0)
        else:
            step *= 0.5
            if step < 1e-13:
                break
            h = _proj_box_sum(h - step * v, n * 0.5)
            f = _smax(h, alpha, topk)
    return _repair(h, n)

def _eq_sum_fun(x):
    return x.sum() - len(x) * 0.5

def _eq_sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smax(x, alpha, topk)
    def jac(x):
        return _grad_smax(x, alpha, topk)
    cons = [{'type': 'eq', 'fun': _eq_sum_fun, 'jac': _eq_sum_jac}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18, 'maxls': 40})
        return _repair(r.x if r.x is not None else h, n)
    except Exception:
        return _repair(h, n)

def _anneal(h, n_iter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(n_iter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.002 * max(T, 1e-10) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _polish(h, n_iter, rng, deadline):
    n = len(h)
    bh = h.copy()
    bc = _eval(h)
    c5c = bc
    scales = [2e-6, 1e-6, 5e-7, 2e-7, 1e-7, 5e-8, 2e-8]
    per = max(1, n_iter // len(scales))
    for scale in scales:
        for _ in range(per):
            if time.time() > deadline:
                return bh, bc
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 7)) % n
            d = scale * (2 * rng.random() - 1)
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _eval(h)
                if c < c5c - 1e-19:
                    c5c = c
                    if c < bc:
                        bc, bh = c, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 6
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float), n)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: SLSQP smooth basin finding
    dl1 = min(deadline, t0 + 90)
    for a, tk, mi in [(4000, 30, 400), (15000, 15, 300), (60000, 10, 200)]:
        if time.time() > dl1:
            break
        hv = _slsqp(best_h.copy(), a, tk, mi, dl1)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 2: LP descent - moderate constraints
    dl2 = min(deadline, t0 + 300)
    hv, cv = _lp_descent(best_h.copy(), 25000, 0.003, 10, dl2)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 3: Nesterov with high alpha (flat landscape exploitation)
    dl3 = min(deadline, t0 + 480)
    for a, tk, st in [(150000, 12, 0.001), (500000, 8, 0.0005)]:
        if time.time() > dl3:
            break
        hv = _nesterov(best_h.copy(), a, tk, 3000, st, dl3)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 4: LP descent - strong constraints (many top correlations)
    dl4 = min(deadline, t0 + 680)
    for eps, mt in [(0.001, 18), (0.0005, 25)]:
        if time.time() > dl4:
            break
        hv, cv = _lp_descent(best_h.copy(), 18000, eps, mt, dl4)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 5: Anneal to escape, then LP refine
    dl5 = min(deadline, t0 + 900)
    blocks = [1, 2, 3, 4, 5, 7, 10]
    hv, cv = _anneal(best_h.copy(), 30000, rng, dl5, 0.002, 1e-10, blocks)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()
    if time.time() < dl5 - 20:
        hv2, cv2 = _lp_descent(best_h.copy(), 8000, 0.001, 12, dl5)
        if cv2 < best_c5:
            best_c5, best_h = cv2, hv2.copy()

    # Phase 6: Ultra-fine polish
    rem = max(0, deadline - time.time())
    npol = min(int(rem / 0.001), 50000)
    best_h, best_c5 = _polish(best_h, npol, rng, deadline)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
