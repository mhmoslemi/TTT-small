import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h = np.clip(h * (target / s), 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _grad_lag(h, k, n):
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += 1.0 - h[sh:]
        if sh > 0: g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += 1.0 - h[:-sp]
        g[:n - sp] -= h[sp:]
    return g * dx

def _sm(h, a, k):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))
    m = c.max()
    idx = np.argpartition(c, -k)[-k:]
    s = np.clip(a * (c[idx] - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / a

def _smg(h, a, k):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    idx = np.argpartition(c, -k)[-k:]
    s = np.clip(a * (c[idx] - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for i in range(len(idx)):
        if w[i] > 1e-14: g += w[i] * _grad_lag(h, int(idx[i]), n)
    return g

def _nest(h, a0, tk, mx, s0, dl, ag=1.3, gp=400):
    n = len(h)
    f = _sm(h, a0, tk)
    v = np.zeros(n); b = 0.83; st = s0; a = a0
    for it in range(mx):
        if time.time() > dl: break
        if it > 0 and it % gp == 0: a = min(a * ag, 5e7)
        g = _smg(h, a, tk)
        v = b * v + (1.0 - b) * g
        hn = _proj(h - st * v, n * 0.5)
        fn = _sm(hn, a, tk)
        if fn < f:
            h, f, st = hn, fn, min(st * 1.015, s0 * 30.0)
        else:
            st *= 0.5; ok = False
            for _ in range(4):
                if time.time() > dl: break
                hn = _proj(h - st * v, n * 0.5)
                fn = _sm(hn, a, tk)
                if fn < f: h, f, ok = hn, fn, True; break
                st *= 0.5
            if not ok: break
    return _repair(h, n)

def _down(h, f):
    n = len(h); nn = n // f
    return _repair(h[:nn * f].reshape(nn, f).mean(1), nn)

def _up(h, f):
    return np.repeat(h, f)

def _ls(h, nit, dl, s0=0.004):
    n = len(h)
    c5 = _eval(h); bh, bc = h.copy(), c5
    st = s0
    for it in range(nit):
        if time.time() > dl: break
        g = _smg(h, 2000, 12)
        pos = np.argsort(-g)[:25]
        neg = np.argsort(g)[:25]
        moved = False
        for a in pos[:12]:
            if moved: break
            for b in neg[:12]:
                if a == b: continue
                if h[a] < st * 0.5 or h[b] > 1.0 - st * 0.5: continue
                for d in [st, st * 0.6, st * 1.5]:
                    if h[a] < d or h[b] > 1.0 - d: continue
                    h[a] -= d; h[b] += d
                    c5n = _eval(h)
                    if c5n < c5 - 1e-16:
                        c5 = c5n
                        if c5 < bc: bc, bh = c5, h.copy()
                        moved = True; break
                    h[a] += d; h[b] -= d
                if moved: break
        if not moved:
            st *= 0.82
            if st < 1e-10: break
    return bh, bc

def _ann(h, nit, rng, dl, T0=0.004, T1=1e-10):
    n = len(h)
    c5 = _eval(h); bh, bc = h.copy(), c5
    bl = [1, 2, 3, 4, 6, 8, 12, 16, 24, 32, 48]
    for it in range(nit):
        if time.time() > dl: break
        T = T0 * (T1 / T0) ** (it / max(nit, 1))
        bs = bl[rng.randint(len(bl))]
        i0 = rng.randint(0, max(1, n - bs))
        amp = 0.004 * max(T, 1e-9) ** 0.3
        d = amp * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(h[i0:i0 + bs] + d, 0, 1)
        hn = h.copy(); hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5 or rng.random() < np.exp(-(c5n - c5) / max(T, 1e-12)):
            h, c5 = hn, c5n
            if c5n < bc: bc, bh = c5n, h.copy()
    return bh, bc

def _slsqf(h, dl, maxit=400):
    n = len(h)
    if time.time() > dl: return h
    def _obj(x):
        return _sm(x, 50000, 10)
    def _jac(x):
        return _smg(x, 50000, 10)
    try:
        r = minimize(_obj, h, jac=_jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
                                   'jac': lambda x: np.ones(n)}],
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _diverse_restart(h, rng, dl, alpha, topk, step0):
    n = len(h)
    pert = rng.randn(n) * 0.02
    pert -= pert.mean()
    hp = _repair(h + pert, n)
    hr = _nest(hp, alpha, topk, 2000, step0, dl)
    return hr

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    dl = t0 + budget_s - 5

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    bh = h.copy(); bc = _eval(h)

    # P1: Initial Nesterov (0-80s)
    d1 = min(dl, t0 + 80)
    h1 = _nest(h, 3000, 30, 6000, 0.005, d1)
    c1 = _eval(h1)
    if c1 < bc: bc, bh = c1, h1.copy()

    # P2: Multi-resolution exploration (80-240s)
    d2 = min(dl, t0 + 240)
    for nt in [n0 // 2, n0 // 4]:
        if time.time() > d2: break
        cn = len(bh)
        if nt < cn:
            ht = _down(bh, cn // nt)
        else:
            ht = bh.copy()
        ht = _nest(ht, 5000, 20, 3000, 0.004, d2)
        ct = _eval(ht)
        if ct < bc: bc, bh = ct, ht.copy()
    # Upsample back to n0 and refine
    if len(bh) < n0 and time.time() < d2:
        bu = _up(bh, n0 // len(bh))
        bu = _nest(bu, 10000, 20, 2000, 0.003, d2)
        cu = _eval(bu)
        if cu < bc: bc, bh = cu, bu.copy()

    # P3: SLSQP refinement (240-380s)
    d3 = min(dl, t0 + 380)
    hs = _slsqf(bh.copy(), d3)
    cs = _eval(hs)
    if cs < bc: bc, bh = cs, hs.copy()

    # P4: Diverse restart + re-optimize (380-460s)
    d4 = min(dl, t0 + 460)
    hd = _diverse_restart(bh, rng, d4, 5000, 20, 0.003)
    cd = _eval(hd)
    if cd < bc:
        bc, bh = cd, hd.copy()
    elif time.time() < d4:
        h4b = _diverse_restart(bh, rng, d4, 10000, 15, 0.002)
        c4b = _eval(h4b)
        if c4b < bc: bc, bh = c4b, h4b.copy()

    # P5: Aggressive multi-param Nesterov (460-640s)
    d5 = min(dl, t0 + 640)
    h5 = bh.copy()
    for a0, tk, st in [(10000, 20, 0.004), (50000, 12, 0.002),
                        (300000, 8, 0.001), (1000000, 5, 0.0005)]:
        if time.time() > d5: break
        h5 = _nest(h5, a0, tk, 5000, st, d5)
        c5 = _eval(h5)
        if c5 < bc: bc, bh = c5, h5.copy()

    # P6: Extended annealing (640-820s)
    d6 = min(dl, t0 + 820)
    h6, c6 = _ann(bh.copy(), 80000, rng, d6, 0.005, 1e-10)
    if c6 < bc: bc, bh = c6, h6.copy()

    # P7: Gradient-guided local search (820-970s)
    d7 = min(dl, t0 + 970)
    h7, c7 = _ls(bh.copy(), 100000, d7, 0.003)
    if c7 < bc: bc, bh = c7, h7.copy()

    # P8: Final Nesterov + SLSQP polish (970-end)
    d8 = dl
    h8 = _nest(bh.copy(), 2000000, 4, 3000, 0.0004, d8)
    c8 = _eval(h8)
    if c8 < bc: bc, bh = c8, h8.copy()
    if time.time() < dl - 20:
        h8 = _slsqf(bh.copy(), dl, maxit=200)
        c8 = _eval(h8)
        if c8 < bc: bc, bh = c8, h8.copy()

    bh = _repair(bh, len(bh))
    bc = _eval(bh)
    return (bh, bc, len(bh))
