import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_topk(h, top_indices):
    n = len(h)
    dx2 = 2.0 / n
    g = np.zeros(n)
    for k in top_indices:
        sh = (n - 1) - k
        wk = dx2
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g / len(top_indices)

def _bb_step(h, g, h_prev, g_prev, step_init):
    if h_prev is None or g_prev is None:
        return step_init
    dy = h - h_prev
    dg = g - g_prev
    alpha = np.dot(dy, dy) / max(np.dot(dy, dg), 1e-12)
    return np.clip(alpha, 1e-6, 0.1)

def _proj_grad_topk(h, k_top, steps, step0, t_end):
    n = len(h)
    h_prev = h.copy()
    g_prev = None
    step = step0
    f = _eval(h)
    for _s in range(steps):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        top_k = np.argsort(c)[-k_top:]
        g = _grad_topk(h, top_k)
        step = _bb_step(h, g, h_prev, g_prev, step0)
        h_prev = h.copy()
        g_prev = g.copy()
        h_new = _repair(h - step * g, n)
        f_new = _eval(h_new)
        if f_new < f:
            h, f = h_new, f_new
        else:
            h = _repair(h_prev - step * 0.3 * g, n)
            f = _eval(h)
            step *= 0.5
    return h

def _sa_local(h, n_tries, rng, t_end, block_size=3):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    cur_h = h.copy()
    cur_c = best_c
    T0, T1 = 0.005, 0.0001
    for _it in range(n_tries):
        if time.time() > t_end:
            break
        T = T0 * (T1/T0) ** (_it / max(n_tries, 1))
        i = rng.randint(n - block_size + 1)
        d = rng.randn(block_size) * 0.002
        cur_h[i:i+block_size] += d
        cur_h = _repair(cur_h, n)
        cur_c = _eval(cur_h)
        delta = cur_c - best_c
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-8)):
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()
        else:
            cur_h = best_h.copy()
            cur_c = best_c
    return best_h, best_c

def _lse_objective(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _lse_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx2 = 2.0 / n
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx2
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

def _slsqp_run(h, alpha, maxiter, ftol, target):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    class Ctx:
        def obj(self, x):
            return _lse_objective(x, alpha, n)
        def grad(self, x):
            return _lse_grad(x, alpha, n)
        def con_fun(self, x):
            return x.sum() - target
        def con_jac(self, x):
            return np.ones(n)
    ctx = Ctx()
    cons = [{'type': 'eq', 'fun': ctx.con_fun, 'jac': ctx.con_jac}]
    try:
        res = minimize(ctx.obj, h, jac=ctx.grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _lse_obj(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.float64(np.log(np.sum(np.exp(s - s.max()))) / alpha)

def _lse_grad(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx2 = 2.0 / n
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[k] * dx2
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:n-sp])
            g[:n-sp] -= wk * x[sp:]
    return g

class _Ctx:
    def __init__(self, alpha, n, target):
        self.alpha = alpha
        self.n = n
        self.target = target
    def obj(self, x):
        return _lse_obj(x, self.alpha, self.n)
    def grad(self, x):
        return _lse_grad(x, self.alpha, self.n)
    def con_fun(self, x):
        return x.sum() - self.target
    def con_jac(self, x):
        return np.ones(self.n)

def _slsqp(h, alpha, maxiter, ftol, target):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    ctx = _Ctx(alpha, n, target)
    cons = [{'type': 'eq', 'fun': ctx.con_fun, 'jac': ctx.con_jac}]
    try:
        res = minimize(ctx.obj, h, jac=ctx.grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)
