import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _proj(v, S):
    x = np.clip(v, 0.0, 1.0).copy()
    for _ in range(60):
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _repair(h, n):
    return _proj(h, n * 0.5)

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    return _proj(np.interp(x1, x0, h), m * 0.5)

def _subgrad(h, tol_frac=0.03):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(corr.max())
    active = np.where(corr >= mx - tol_frac * np.maximum(mx, 1e-10))[0]
    g = np.zeros(n)
    for m in active:
        k = int(m) - (n - 1)
        if k > 0:
            g[:n-k] += 1.0 - h[k:]
            g[k:] -= h[:n-k]
        elif k < 0:
            kk = -k
            g[kk:] += 1.0 - h[:n-kk]
            g[:n-kk] -= h[kk:]
        else:
            g += 1.0 - 2.0 * h
    g /= len(active)
    return g, mx, len(active)

def _sgd_search(h0, rng, deadline, tol_frac=0.03):
    n = len(h0)
    h = _repair(h0, n).copy()
    c = _c5(h)
    best_c = c
    best_h = h.copy()
    t0 = time.time()
    lr = 0.5
    temp = 0.02  # simulated annealing temperature
    t_end = deadline - t0
    it = 0
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t0) / max(1e-9, t_end))
        temp_cur = temp * (0.01) ** frac
        g, mx, na = _subgrad(h, tol_frac)
        # Adaptive learning rate
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        step = lr / gnorm  # normalized step
        h_new = _proj(h - step * g, n * 0.5)
        c_new = _c5(h_new)
        # Accept if improved or with SA probability
        if c_new < c:
            h = h_new
            c = c_new
            lr = min(lr * 1.02, 2.0)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            # Simulated annealing: accept with prob exp(-delta/temp)
            delta = c_new - c
            if delta > 0:
                p = np.exp(-delta / max(temp_cur, 1e-12))
                if rng.random() < p:
                    h = h_new
                    c = c_new
            lr = max(lr * 0.95, 0.01)
        # Periodic restart
        if it % 500 == 499 and frac < 0.9:
            # Small perturbation
            h = _proj(h + rng.randn(n) * 0.01 * (1 - frac), n * 0.5)
        it += 1
    return best_h, best_c

import numpy as np
import time
from scipy.optimize import minimize

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _proj(v, S):
    x = np.clip(np.asarray(v, dtype=float), 0.0, 1.0)
    for _ in range(80):
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _repair(h, n):
    return _proj(h, n * 0.5)

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    return _proj(np.interp(x1, x0, h), m * 0.5)

def _subgrad(h, tol_frac=0.02):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(corr.max())
    thr = mx - tol_frac * max(mx, 0.01)
    active = np.where(corr >= thr)[0]
    g = np.zeros(n)
    for m in active:
        k = int(m) - (n - 1)
        if k > 0:
            g[:n-k] += 1.0 - h[k:]
            g[k:] -= h[:n-k]
        elif k < 0:
            kk = -k
            g[kk:] += 1.0 - h[:n-kk]
            g[:n-kk] -= h[kk:]
        else:
            g += 1.0 - 2.0 * h
    g /= len(active)
    return g, mx, len(active)

def _cg_search(h0, rng, deadline, tol_frac=0.02):
    """Subgradient + simulated annealing search"""
    n = len(h0)
    h = _repair(h0, n).copy()
    c = _c5(h)
    best_c = c
    best_h = h.copy()
    t0 = time.time()
    lr = 0.3
    base_temp = 0.015
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t0) / max(1e-9, deadline - t0))
        temp = base_temp * (0.005) ** frac
        g, mx, na = _subgrad(h, tol_frac)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        step = lr / gnorm
        h_new = _proj(h - step * g, n * 0.5)
        c_new = _c5(h_new)
        if c_new <= c:
            h = h_new
            c = c_new
            lr = min(lr * 1.03, 3.0)
            if c < best_c:
                best_c, best_h = c, h.copy()
        else:
            if rng.random() < np.exp(-(c_new - c) / max(temp, 1e-15)):
                h = h_new
                c = c_new
            lr = max(lr * 0.97, 0.01)
        if it := int(frac * 10000):
            pass  # just to track
    return best_h, best_c

def _binary_refine(h0, rng, deadline):
    n = len(h0)
    h = np.zeros(n)
    h[np.argsort(h0)[::-1][:n//2]] = 1.0
    c = _c5(h)
    best_c = c
    best_h = h.copy()
    t0 = time.time()
    while time.time() < deadline:
        # Random swap
        if rng.random() < 0.8:
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] != h[j] and i != j:
                h[i], h[j] = h[j], h[i]
                cn = _c5(h)
                if cn < c - 1e-12:
                    c = cn
                    if c < best_c:
                        best_c, best_h = c, h.copy()
                else:
                    h[i], h[j] = h[j], h[i]
        else:
            # Block swap
            L = rng.randint(2, n//30 + 2)
            i = rng.randint(0, n-L)
            j = rng.randint(0, n-L)
            if abs(i-j) < L:
                continue
            s1, s2 = h[i:i+L].copy(), h[j:j+L].copy()
            h[i:i+L], h[j:j+L] = s2, s1
            cn = _c5(h)
            if cn < c - 1e-12:
                c = cn
                if c < best_c:
                    best_c, best_h = c, h.copy()
            else:
                h[i:i+L], h[j:j+L] = s1, s2
    return best_h, best_c

def _fourier_init(n, K, rng):
    """Generate a random Fourier-based initial function"""
    h = np.full(n, 0.5)
    for k in range(1, K+1):
        a = rng.uniform(-0.15, 0.15)
        b = rng.uniform(-0.15, 0.15)
        h += a * np.cos(2*np.pi*k*np.arange(n)/n)
        h += b * np.sin(2*np.pi*k*np.arange(n)/n)
    return _proj(h, n * 0.5)

    return best, best_c, n0

def _alt_search(h0, rng, deadline, alpha=50, inner_iters=30):
    n = len(h0)
    h = _repair(h0, n).copy()
    c = _c5(h)
    best_c = c
    best_h = h.copy()
    t0 = time.time()
    lr = 0.1
    
    while time.time() < deadline:
        # Fix h, update p (softmax weights)
        corr = np.correlate(h, 1.0-h, mode='full')
        # Active set with softmax
        emax = corr.max()
        w = np.exp(alpha * (corr - emax))
        wsum = w.sum()
        if wsum < 1e-15:
            w = np.zeros_like(corr)
            w[np.argmax(corr)] = 1.0
        else:
            w /= wsum
        
        # Fix p, optimize h (projected gradient)
        for _ in range(inner_iters):
            if time.time() >= deadline:
                break
            g = _weighted_grad(h, w, n)
            h = _proj(h - lr * g, n * 0.5)
        
        c = _c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
    
    return best_h, best_c

def _weighted_grad(h, w, n):
    """Gradient of sum_k w_k * c(k) with respect to h"""
    g = np.zeros(n)
    for m in range(len(w)):
        if w[m] < 1e-10:
            continue
        k = m - (n-1)
        if k > 0:
            g[:n-k] += (1.0 - h[k:]) * w[m]
            g[k:] -= h[:n-k] * w[m]
        elif k < 0:
            kk = -k
            g[kk:] += (1.0 - h[:n-kk]) * w[m]
            g[:n-kk] -= h[kk:] * w[m]
        else:
            g += (1.0 - 2.0*h) * w[m]
    return g
