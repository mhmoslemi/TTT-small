        return h

def _local_search(h, n_iter, rng, step, n_pert, t0, tlimit):
    n = len(h)
    c5 = _eval(h)
    best_h = h.copy()
    best_c = c5
    for _ in range(n_iter):
        if time.time() - t0 > tlimit:
            break
        s = step * (0.3 + 0.7 * rng.rand())
        idx = rng.randint(0, n, size=n_pert)
        d = s * rng.randn(n_pert)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < -0.2 or h_new.max() > 1.2:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5:
            h = h_new
            c5 = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1), m)

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return _repair(out, n * f)

def _make_sin_init(n, freqs, amps, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for f, a in zip(freqs, amps):
        phase = rng.rand() * 2 * np.pi
        h += a * np.sin(2 * np.pi * f * x / 2.0 + phase)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    tl = budget_s
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: FISTA progressive alpha sweep
    alpha_sched = [
        (1000, 40, 0.5, 2000),
        (5000, 60, 0.3, 4000),
        (20000, 100, 0.2, 4000),
        (100000, 150, 0.1, 3000),
        (500000, 250, 0.05, 2000),
        (2000000, 400, 0.02, 2000),
    ]
    for alpha, ml, st, niter in alpha_sched:
        if _el() > tl * 0.35:
            break
        h_new, c5n = _fista(best_h, alpha, ml, niter, st, t0, tl * 0.95)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 2: Multi-resolution
    for ratio in [4, 8]:
        if _el() > tl * 0.50:
            break
        nc = max(32, n // ratio)
        f = n // nc
        hc = _downsample(best_h, f)
        for alpha, ml in [(3000, 25), (30000, 40)]:
            if _el() > tl * 0.52:
                break
            hc, _ = _fista(hc, alpha, ml, 3000, 0.4, t0, tl * 0.52)
        h_up = _upsample(hc, f)
        c5u = _eval(h_up)
        if c5u < best_c5:
            best_c5 = c5u
            best_h = h_up

    # Phase 3: SLSQP at full resolution
    for alpha, ml in [(20000, 80), (200000, 150), (2000000, 250)]:
        if _el() > tl * 0.72:
            break
        h_new = _slsqp_round(best_h, alpha, ml, 1000, 1e-15, t0, tl * 0.72)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 4: High-alpha FISTA
    for alpha, ml, st in [(5000000, 500, 0.01), (20000000, 800, 0.005), (100000000, 1200, 0.002)]:
        if _el() > tl * 0.84:
            break
        h_new, c5n = _fista(best_h, alpha, ml, 1500, st, t0, tl * 0.88)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 5: Try alternate starts (quick check)
    if _el() < tl - 200:
        alt_starts = [
            _make_sin_init(n, [3, 7, 13], [0.15, 0.08, 0.04], rng),
            _make_sin_init(n, [5, 11, 19], [0.12, 0.06, 0.03], rng),
        ]
        for hs in alt_starts:
            if _el() > tl - 80:
                break
            hs, c5s = _fista(hs, 50000, 100, 5000, 0.15, t0, tl - 80)
            if c5s < best_c5:
                best_c5 = c5s
                best_h = hs

    # Phase 6: Local search
    if _el() < tl - 40:
        ntries = min(int((tl - 40 - _el()) / 0.003), 250000)
        best_h, best_c5 = _local_search(best_h, ntries, rng, 5e-6, 4, t0, tl - 40)

    # Phase 7: Ultra-fine 2-point polish
    if _el() < tl - 2:
        h2 = best_h.copy()
        nc2 = len(h2)
        c5c = _eval(h2)
        npol = min(int((tl - 2 - _el()) / 0.002), 150000)
        for _ in range(npol):
            if _el() > tl - 2:
                break
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 5)) % nc2
            d = 1.5e-7 * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
