                h[idx] -= a
                diff += a

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)

            lo = mid
        else:

    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[:n-sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n-sp] -= h[sp:]
    return g * dx

    dx = 2.0 / n

        c = np.correlate(x, 1.0 - x, mode='full') * dx
        idx = np.argsort(c)[-top_k:] if 0 < top_k < len(c) else np.arange(len(c))
        cs = c[idx]
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha

        c = np.correlate(x, 1.0 - x, mode='full') * dx
        idx = np.argsort(c)[-top_k:] if 0 < top_k < len(c) else np.arange(len(c))
        cs = c[idx]
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)

        w = np.exp(s); w /= w.sum()
        g = np.zeros(n)

                g += w[ii] * _grad_lag(int(idx[ii]), x, n)
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_worst(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max()), int(np.argmax(c))

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(k, h, n):
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _slsqp(h, alpha, maxiter, top_k):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        idx = np.argsort(c)[-top_k:] if 0 < top_k < len(c) else np.arange(len(c))
        cs = c[idx]; m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400); s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        idx = np.argsort(c)[-top_k:] if 0 < top_k < len(c) else np.arange(len(c))
        cs = c[idx]; m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400); s -= s.max()
        w = np.exp(s); w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] > 1e-15:
                g += w[ii] * _grad_lag(int(idx[ii]), x, n)
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _pg(h, alpha, steps, step0, top_k, momentum):
    n = len(h); v = np.zeros(n); step = step0; cur_c = _eval(h)
    for it in range(steps):
        c = _corr(h)
        active = np.argsort(c)[-top_k:]
        ca = c[active]; m = ca.max()
        s = np.clip(alpha * (ca - m), -400, 400); s -= s.max()
        w = np.exp(s); w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(active)):
            if w[ii] > 1e-15:
                g += w[ii] * _grad_lag(int(active[ii]), h, n)
        v = momentum * v + (1.0 - momentum) * g
        taken = False
        for _ in range(15):
            hn = _proj(h - step * v, n * 0.5)
            cn = _eval(hn)
            if cn < cur_c - 1e-16:
                h, cur_c, taken = hn, cn, True
                step = min(step * 1.1, step0 * 10)
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.4, step0 * 10)
        if it % 200 == 0:
            v *= 0.3
    return _repair(h, n)

def _tabu_local(h, n_iter, rng, base_step, sa_temp):
    n = len(h)
    best_h, best_c, cur_c = h.copy(), _eval(h), _eval(h)
    seen, seq = {}, []
    for it in range(n_iter):
        c5c, k_w = _eval_worst(h)
        cur_c = c5c
        sh = (n - 1) - k_w
        moved = False
        if 0 < abs(sh) < n:
            if sh > 0:
                valid = np.arange(n - sh)
                pot = h[valid] * (1.0 - h[valid + sh])
            else:

            order = np.argsort(pot)[-min(40, len(valid)):][::-1]
            for ci in order:

                key = (min(i, j), max(i, j))
                if key in seen:
                    continue
                for sgn in (1, -1):

                    ni, nj = h[i] + d, h[j] - d
                    if 0.0 < ni < 1.0 and 0.0 < nj < 1.0:
                        h[i], h[j] = ni, nj
                        c5n, _ = _eval_worst(h)
                        acc = c5n < cur_c
                        if not acc and sa_temp > 1e-15:
                            acc = rng.random() < np.exp(-(c5n - cur_c) / sa_temp)
                        if acc:
                            cur_c = c5n
                            if c5n < best_c:
                                best_c, best_h = c5n, h.copy()
                            seq.append(key); seen[key] = it
                            if len(seq) > 300:
                                seen.pop(seq.pop(0), None)
                            moved = True
                        else:
                            h[i], h[j] = ni - d, nj + d
                        break
                if moved:
                    break
        if not moved or rng.random() < 0.15:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 30)) % n
            d = base_step * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0.0 < ni < 1.0 and 0.0 < nj < 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                acc = c5n < cur_c

                    cur_c = c5n
                    if c5n < best_c:
                        best_c, best_h = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
        if it % 2000 == 0 and sa_temp > 1e-15:
            sa_temp *= 0.92
    return best_h, best_c

def _fine_polish(h, n_iter, rng, base_step):
    n = len(h)
    best_h, best_c, cur_c = h.copy(), _eval(h), _eval(h)
    for _ in range(n_iter):

        j = (i + 1 + rng.randint(0, 6)) % n

def _down(h, f):
    n = len(h); m = n // f
    return h[:m * f].reshape(m, f).mean(1).copy()

def _up(h, f):
    out = np.empty(len(h) * f)
    for i in range(len(h)):
        out[i * f:(i + 1) * f] = h[i]
    return out

                       (500000, 3, 250), (2000000, 2, 200), (10000000, 1, 150)]:
        if _el() > 160: break
        hn = _slsqp(best_h, a, mi, tk)
        cn = _eval(hn)
        if cn < best_c5:
            best_c5, best_h = cn, hn.copy()

    for a, st, s0, tk, mu in [(3000, 500, 0.006, 8, 0.7), (10000, 500, 0.004, 6, 0.75),
                               (50000, 400, 0.003, 5, 0.8), (200000, 300, 0.002, 4, 0.8),
                               (1000000, 200, 0.001, 3, 0.85)]:
        if _el() > 410: break

        cn = _eval(hn)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))

For (2), the correlation c[k] = sum_i h[i] * (1-h[i+k]) can be written as:
c[k] = sum_i h[i] - sum_i h[i]h[i+k]

The second term is the autocorrelation. In the frequency domain, the autocorrelation is the inverse FFT of |FFT(h)|². So minimizing the max autocorrelation is equivalent to making the |FFT(h)|² as flat as possible (which is the "equiripple" property of optimal sequences).

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
    diff = t - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_worst(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max()), int(np.argmax(c))

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(k, h, n):
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _slsqp_obj(x, alpha, top_k):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-top_k:] if 0 < top_k < len(c) else np.arange(len(c))
    cs = c[idx]; m = cs.max()
    s = np.clip(alpha * (cs - m), -400, 400); s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, top_k):
    n = len(x); dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-top_k:] if 0 < top_k < len(c) else np.arange(len(c))
    cs = c[idx]; m = cs.max()
    s = np.clip(alpha * (cs - m), -400, 400); s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] > 1e-15:
            g += w[ii] * _grad_lag(int(idx[ii]), x, n)
    return g

def _slsqp(h, alpha, maxiter, top_k):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, args=(alpha, top_k),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h
