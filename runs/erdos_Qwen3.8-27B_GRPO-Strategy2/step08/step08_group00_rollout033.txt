import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _obj_smooth(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _grad_smooth(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _optimize(x, alpha, m_lags, maxiter):
    n = len(x)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_smooth, x, jac=_grad_smooth,
                       args=(alpha, m_lags), method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter,
                       'ftol': 1e-18, 'gtol': 1e-14, 'maxfun': maxiter * 2})
        x_out = res.x
    except Exception:
        x_out = x.copy()
    x_out = x_out - (x_out.mean() - 0.5)
    return _repair(x_out, n)

def _two_point(h, n_steps, rng, delta_scale):
    n = len(h)
    c5 = _eval(h)
    for _ in range(n_steps):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 40)) % n
        d = delta_scale * (0.3 + 0.7 * rng.rand())
        if rng.rand() > 0.5:
            d = -d
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n <= c5:
                c5 = c5n
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5

    h_p = _optimize(h_p, 20000000, 80, 800)
    h_p = _optimize(h_p, 200000000, 120, 600)
    return h_p
