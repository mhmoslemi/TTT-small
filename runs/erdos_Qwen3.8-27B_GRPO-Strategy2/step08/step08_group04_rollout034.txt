import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

        else:

        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]

            f = fn
            step = min(step * 1.01, step0 * 5.0)

            hn = _proj(hp - step * v, n * 0.5)
            fn = _eval(hn)
            if fn < f:
                h = hn
                f = fn
            else:
                break
    return _repair(h, n)

Let k go from -(n-1) to (n-1) in the "full" output (index 0 to 2n-2, lag = index - (n-1)).

For lag ℓ: c[ℓ] = sum_n h[n+ℓ] * (1-h[n]) where 0≤n<n, 0≤n+ℓ<n.

c[-ℓ] = sum_n h[n-ℓ] * (1-h[n]) where 0≤n<n, 0≤n-ℓ<n.
Let m = n-ℓ: c[-ℓ] = sum_m h[m] * (1-h[m+ℓ]) where 0≤m<n, 0≤m+ℓ<n.

And c[ℓ] = sum_n h[n+ℓ] * (1-h[n]) where 0≤n<n, 0≤n+ℓ<n.
Let m = n+ℓ: c[ℓ] = sum_m h[m] * (1-h[m-ℓ]) where 0≤m-ℓ<n, 0≤m<n.

So c[ℓ] = sum_m h[m](1-h[m-ℓ]) and c[-ℓ] = sum_m h[m](1-h[m+ℓ]).

These are not equal in general. So no symmetry to exploit.

import numpy as np
from scipy.optimize import minimize
import time


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h


def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _proj(h, target):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(44):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() >= target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def _grad_k(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += 1.0 - h[sh:]
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += 1.0 - h[:-sp]
        g[:n - sp] -= h[sp:]
    return g * dx


def _smax_obj(h, alpha, topk):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    return m + float(s.max() + np.log(np.sum(np.exp(s - sm)))) / alpha


def _smax_grad(h, alpha, topk):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] > 1e-10:
            g += w[ii] * _grad_k(h, int(idx[ii]), dx)
    return g


class _SmaxWrap:
    def __init__(self, alpha, topk):
        self.a = alpha
        self.tk = topk
    def __call__(self, x):
        return _smax_obj(x, self.a, self.tk)
    def grad(self, x):
        return _smax_grad(x, self.a, self.tk)


def _sum_con_fun(x):
    return x.sum() - len(x) * 0.5


def _sum_con_jac(x):
    return np.ones(len(x))


def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    wrap = _SmaxWrap(alpha, topk)
    cons = [{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}]
    try:
        r = minimize(wrap, h, jac=wrap.grad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-22})
        return _repair(r.x.copy(), n)
    except Exception:
        return _repair(h.copy(), n)


def _nesterov(h, alpha, topk, steps, step0, beta, deadline):
    n = len(h)
    target = n * 0.5
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    step = step0
    stall = 0
    for it in range(steps):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f - 1e-16:
            h = hn; f = fn
            step = min(step * 1.006, step0 * 12.0)
            stall = 0
        else:
            step *= 0.55
            if step < 1e-14:
                break
            hn = _proj(h - step * v, target)
            fn = _smax_obj(hn, alpha, topk)
            if fn < f:
                h = hn; f = fn
            else:
                stall += 1
                if stall > 50:
                    v *= 0.3
                    step = step0 * 0.05
                    stall = 0
                else:
                    hn = _proj(h - step * 0.5 * v, target)
                    fn = _smax_obj(hn, alpha, topk)
                    if fn < f:
                        h = hn; f = fn; step *= 0.5
                    else:
                        v *= 0.5
    return _repair(h.copy(), n)


def _targeted_cd(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(c))
        sh = (n - 1) - k_star
        if sh >= 0:
            contrib = h[:n-sh] * (1.0 - h[sh:])
            i_best = int(np.argmax(contrib))
        else:
            sp = -sh
            contrib = h[sp:] * (1.0 - h[:n-sp])
            i_best = int(np.argmax(contrib)) + sp
        partner = (i_best + (1 if sh >= 0 else -1)) % n
        for d in [1e-5, -1e-5, 5e-6, -5e-6, 2e-5, -2e-5]:
            ni = h[i_best] + d
            nj = h[partner] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i_best], h[partner] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5n < bc:
                        bc = c5n; bh = h.copy()
                    break
                h[i_best], h[partner] = ni - d, nj + d
    return bh, bc


def _multi_restart(h, rng, deadline, n_restarts=8):
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(h)
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        sub_dl = time.time() + (deadline - time.time()) / (n_restarts - r)
        hn = h.copy()
        mask = rng.rand(n) < 0.15
        pert = 0.03 * rng.randn(n)
        hn[mask] = np.clip(hn[mask] + pert[mask], 0, 1)
        hn = _repair(hn, n)
        hl = _nesterov(hn, 30000, 6, 800, 0.003, 0.87, sub_dl)
        c5v = _eval(hl)
        if c5v < best_c5:
            best_c5 = c5v; best_h = hl.copy()
    return best_h, best_c5


def _final_polish(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step = 5e-6
    for it in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc = c5n; bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                if it % 5000 == 0:
                    step = max(step * 0.9, 1e-8)
    return bh, bc
