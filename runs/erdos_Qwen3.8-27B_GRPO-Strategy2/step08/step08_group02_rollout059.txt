        idx = np.argsort(c)[::-1][:topk]
        val = np.sum(c[idx] ** p)
        if binary_weight > 0:
            val += binary_weight * np.sum(4.0 * x * (1.0 - x))
        return val

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _corr(h):
    n = len(h)

def _grad_lag(ki, h, n):
    dx = 2.0 / n
    sh = ki - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        m = n - sh
        i1 = np.arange(m)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _opt_lbgfs(h, p, topk, maxiter, binary_wt):
    n = len(h)

        c = _corr(x)
