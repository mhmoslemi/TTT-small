    s=x.sum();

          if np.clip(x-md,0,1).sum()>tg: lo=md
          else: hi=md
       x=np.clip(x-(lo+hi)*0.5,0,1)
       d=tg-x.sum()

import time
import numpy as np
from scipy.optimize import minimize, linprog

def _ex(t):
    o = np.empty_like(t)
    p = t >= 0
    o[p] = 1.0 / (1.0 + np.exp(-t[p]))
    e = np.exp(t[~p])
    o[~p] = e / (1.0 + e)
    return o

def _hp(z, tg):
    z = np.nan_to_num(np.asarray(z, float), nan=0.0, posinf=50.0, neginf=-50.0)
    z = np.clip(z, -50.0, 50.0)
    lo = float(np.min(z)) - 70.0
    hi = float(np.max(z)) + 70.0
    for _ in range(45):
        md = 0.5 * (lo + hi)
        if _ex(z - md).sum() > tg:
            lo = md
        else:
            hi = md
    lam = 0.5 * (lo + hi)
    h = _ex(z - lam)
    p = h * (1.0 - h)
    S = float(p.sum())
    if S < 1e-12:
        S = 1e-12
    return h, p, S

def _zh(z, tg):
    return _pth(_hp(z, tg)[0], tg)

def _cv(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _cm(h):
    return float(np.max(_cv(h)))

def _sw(c, a):
    m = float(np.max(c))
    if not np.isfinite(m):
        w = np.zeros(c.shape); w[int(np.argmax(c))] = 1.0; return w
    s = np.exp(a * (c - m))
    s = np.where(np.isfinite(s), s, 0.0)
    t = float(s.sum())
    if t <= 0:
        w = np.zeros(c.shape); w[int(np.argmax(c))] = 1.0; return w
    return s / t

def _gl(h, k):
    n = len(h); dx = 2.0 / n; off = n - 1
    d = int(k) - off
    g = np.zeros(n)
    if d > 0:
        m = n - d
        g[d:] += dx * (1.0 - h[:m])
        g[:m] -= dx * h[d:]
    elif d < 0:
        s = -d; m = n - s
        g[:m] += dx * (1.0 - h[s:])
        g[s:] -= dx * h[:m]
    else:
        g += dx * (1.0 - 2.0 * h)
    return g

def _gw(h, w):
    n = len(h); dx = 2.0 / n; off = n - 1
    nz = np.flatnonzero(w > 0.0)
    if nz.size == 0:
        return np.zeros(n)
    if nz.size <= 180:
        g = np.zeros(n)
        for k in nz:
            wk = float(w[k]) * dx
            if wk == 0.0:
                continue
            d = int(k) - off
            if d > 0:
                m = n - d
                g[d:] += wk * (1.0 - h[:m])
                g[:m] -= wk * h[d:]
            elif d < 0:
                s = -d; m = n - s
                g[:m] += wk * (1.0 - h[s:])
                g[s:] -= wk * h[:m]
            else:
                g += wk * (1.0 - 2.0 * h)
        return g
    A = np.convolve(w, 1.0 - h, mode='full')[off:off + n]
    B = np.convolve(h, w[::-1], mode='full')[off:off + n]
    return dx * (A - B)

def _fz(x, tg, a):
    z = np.nan_to_num(np.asarray(x, float), nan=0.0, posinf=50.0, neginf=-50.0)
    z = np.clip(z, -50.0, 50.0)
    z -= z.mean()
    h, p, S = _hp(z, tg)
    c = _cv(h)
    m = float(np.max(c))
    if not np.isfinite(m):
        return 1e100, np.zeros_like(z)
    s = np.exp(a * (c - m))
    s = np.where(np.isfinite(s), s, 0.0)
    t = float(s.sum())
    f = m + (np.log(t) / a if t > 0.0 else 0.0)
    if S <= 1e-12 or t <= 0.0:
        return f, np.zeros_like(z)
    w = s / t
    g = _gw(h, w)
    gz = p * (g - float(np.dot(p, g)) / S)
    gz -= gz.mean()
    return f, np.nan_to_num(gz, nan=0.0, posinf=1e3, neginf=-1e3)

def _lbg(z, tg, a, mf):

                     options={'maxiter': 130, 'maxfun': int(mf), 'ftol': 1e-12,
                              'gtol': 1e-7, 'maxcor': 30})

    return z

    x = np.clip(x, 0.0, 1.0)

        if abs(d) < 1e-8:
            if d > 0:
                i = int(np.argmax(1.0 - x)); x[i] = min(1.0, x[i] + d)
            else:
                i = int(np.argmax(x)); x[i] = max(0.0, x[i] + d)
    return x

def _basis(n, m):
    x = np.arange(n) * (1.0 / float(n))
    C = [np.ones(n)]
    for j in range(1, m + 1):
        C.append(np.cos(2.0 * np.pi * j * x))
        C.append(np.sin(2.0 * np.pi * j * x))
    return np.column_stack(C)

def _fc(a, B, tg, al):

    if si > 0 and time.time() < dl - 2:
        h3, c3 = _sub(best, tg, min(dl - 2, time.time() + 40.0), rg, si)
        if c3 < bc:
            best, bc = h3, c3
    if ls > 0 and time.time() < dl - 1:
        h4, c4 = _loc(best, tg, min(dl - 1, time.time() + ls), 3e-5, max(1e-7, bc * 5e-5), rg)
        if c4 < bc:
            best, bc = h4, c4
    return best, bc

def _cf(h, tg, dl, rg):
    n = len(h); best = h.copy(); bc = _cm(h)
    for M, tr, sc in ((8, 2, 0.35), (16, 2, 0.45), (28, 2, 0.55)):
        if time.time() >= dl:
            break
        B = _basis(n, M)
        P = B.shape[1]
        bnds = [(-5.0, 5.0)] * P
        for r in range(tr):
            if time.time() >= dl:
                break
            st = rg.randn(P) * sc
            for a in (10, 100, 1000):
                if time.time() >= dl:
                    break
                rem = max(0.0, dl - time.time())
                act = max(30, min(220, int(rem * 35)))
                try:
                    rr = minimize(_fc, st, jac=True, args=(B, tg, a), method='L-BFGS-B',
                                 bounds=bnds,
                                 options={'maxiter': 70, 'maxfun': act, 'ftol': 1e-10,
                                          'gtol': 1e-5, 'maxcor': 20})
                    if rr.x is not None and np.all(np.isfinite(rr.x)):
                        st = rr.x
                except Exception:
                    pass
                ht = _zh(B @ st, tg); ch = _cm(ht)
                if np.isfinite(ch) and ch < bc:
                    bc = ch; best = ht.copy()
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    end = t0 + float(budget_s)
    rg = np.random.RandomState(seed)
    try:
        hp = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        hp = np.zeros(560); hp[:280] = 1.0
    n = len(hp)
    tg = n * 0.5
    hp = _pth(hp, tg)
    best = hp.copy()
    bc = _cm(best)

    for xt in (best[::-1].copy(), _pth(0.5 * (best + best[::-1]), tg),
               _pth(1.0 - best, tg)):
        xc = _cm(xt)
        if xc < bc:
            best, bc = xt, xc
    xs, xs_c = _sh(best)
    if xs_c < bc:
        best, bc = xs, xs_c
    zb = _zfh(best)
    for vv in (-2.0, -1.0, -0.5, 0.5, 1.0, 2.0):
        xt = _zh(zb + vv, tg); xc = _cm(xt)
        if xc < bc:
            best, bc = xt, xc

    dl_main = t0 + min(300.0, 0.35 * float(budget_s))
    if time.time() < dl_main:
        ht, ct = _ref(best, tg, dl_main, (5, 20, 100, 500, 2000, 8000), 550, rg, 30, 40.0, 3)
        if ct < bc:
            best, bc = ht, ct

    dl_cf = t0 + min(700.0, 0.70 * float(budget_s))
    if time.time() < dl_cf - 180:
        ht, ct = _cf(best, tg, dl_cf - 180, rg)
        if ct < bc:
            best, bc = ht, ct
            hl, cl = _ref(best, tg, min(dl_cf, end - 180), (20, 200, 2000), 350, rg, 18, 18.0, 2)
            if cl < bc:
                best, bc = hl, cl

    dl_st = t0 + min(950.0, 0.90 * float(budget_s))
    cands = []
    if time.time() < dl_st - 60:
        for xx in (best[::-1].copy(), _pth(0.5 * (best + best[::-1]), tg),
                   _pth(1.0 - best, tg), _sh(best)[0]):
            xc = _cm(xx)
            if xc < bc + 0.008 and len(cands) < 6:
                cands.append((xx, xc))
        zc = _zfh(best)
        for vv in (-0.6, 0.6, -1.4, 1.4):
            xx = _zh(zc + vv, tg); xc = _cm(xx)
            if xc < bc + 0.015 and len(cands) < 6:
                cands.append((xx, xc))
        for sc in (0.008, 0.025):
            xx = _pth(best + sc * rg.randn(n), tg); xc = _cm(xx)
            if xc < bc + 0.035 and len(cands) < 6:
                cands.append((xx, xc))
        for rr in (2, 4):
            if n % rr == 0:
                xx = _pth(np.repeat(_down(best, rr), rr), tg); xc = _cm(xx)
                if xc < bc + 0.04 and len(cands) < 6:
                    cands.append((xx, xc))

    for idx in range(len(cands)):
        if time.time() >= dl_st:
            break
        rem = max(1.0, dl_st - time.time())
        alloc = max(20.0, min(80.0, rem / max(1, len(cands) - idx)))
        xx, xc = cands[idx]
        a_dl = min(dl_st, time.time() + (100.0 if xc < bc else alloc))
        ht, ct = _ref(xx, tg, a_dl, (20, 200, 2000), 300, rg, 12, min(12.0, alloc * 0.3), 1)
        if ct < bc:
            best, bc = ht, ct

    final_dl = end - 3.0
    steps = (1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7)
    for cyc in range(16):
        if time.time() >= final_dl - 5:
            break
        ht, ct = _lp(best, tg, min(final_dl - 5, time.time() + 18.0), 2)
        if ct < bc:
            best, bc = ht, ct
        if time.time() >= final_dl - 5:
            break
        ht, ct = _sub(best, tg, min(final_dl - 5, time.time() + 28.0), rg, 10)
        if ct < bc:
            best, bc = ht, ct
        if time.time() >= final_dl - 4:
            break
        st = steps[min(cyc, len(steps) - 1)]
        ht, ct = _loc(best, tg, min(final_dl - 2, time.time() + 45.0), st, max(1e-8, bc * 5e-5), rg)
        if ct < bc:
            best, bc = ht, ct

    best = _pth(best, tg)
    bc = _cm(best)
    if not np.isfinite(bc):
        best = _pth(hp, tg); bc = _cm(best)
    return best, bc, len(best)
