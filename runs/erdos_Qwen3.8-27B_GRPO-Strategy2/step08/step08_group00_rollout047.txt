import numpy as np
import time

def _fft_corr(h):
    n = len(h)
    N = 1
    while N < 2 * n:
        N <<= 1
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    return np.fft.irfft(H * np.conj(G), N)[:2 * n - 1]

def _eval(h):
    return float(np.max(_fft_corr(h)) * 2.0 / len(h))

def _proj(h, target):
    lo = h.min() - 0.5
    hi = h.max() + 0.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _slsqp(h, alpha, m_lags, maxiter, ftol):
    from scipy.optimize import minimize
    n = len(h)
    dx = 2.0 / n
    def _obj(x):
        c = _fft_corr(x) * dx
        m = c.max()
        idx = np.argsort(c)[-m_lags:]
        s = np.clip(alpha * (c[idx] - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def _grad(x):
        c = _fft_corr(x) * dx
        m = c.max()
        idx = np.argsort(c)[-m_lags:]
        s = np.clip(alpha * (c[idx] - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii, k in enumerate(idx):
            if w[ii] < 1e-14:
                continue
            L = k - (n - 1)
            if L >= 0:
                g[:n - L] += w[ii] * (1.0 - x[L:])
                g[L:] -= w[ii] * x[:n - L]
            else:
                M = -L
                g[M:] += w[ii] * (1.0 - x[:n - M])
                g[:n - M] -= w[ii] * x[M:]
        return g
    def _sumf(x):
        return x.sum() - n * 0.5
    def _sumj(x):
        return np.ones(n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sumf, 'jac': _sumj}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _proj(res.x, n * 0.5)
    except Exception:
        return h

def _pgd(h, alpha, m_lags, n_steps, step0, rng):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c5 = np.max(_fft_corr(h)) * dx
    v = np.zeros(n)
    step = step0
    mu = 0.75
    for _ in range(n_steps):
        c = _fft_corr(h) * dx
        m = c.max()
        top = np.argpartition(c, -m_lags)[-m_lags:]
        s = np.clip(alpha * (c[top] - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii, k in enumerate(top):
            if w[ii] < 1e-14:
                continue
            L = k - (n - 1)
            if L >= 0:
                g[:n - L] += w[ii] * (1.0 - h[L:])
                g[L:] -= w[ii] * h[:n - L]
            else:
                M = -L
                g[M:] += w[ii] * (1.0 - h[:n - M])
                g[:n - M] -= w[ii] * h[M:]
        v = mu * v + (1.0 - mu) * g
        taken = False
        for _ in range(25):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = np.max(_fft_corr(h_new)) * dx
            if c_new < c5:
                h = h_new
                c5 = c_new
                step = min(step * 1.12, step0 * 30.0)
                taken = True
                break
            step *= 0.45
        if not taken:
            step = min(step * 1.3, step0 * 30.0)
    return h, c5

def _local_search(h, n_iter, rng, base_step):
    n = len(h)
    dx = 2.0 / n
    c5 = np.max(_fft_corr(h)) * dx
    for _ in range(n_iter):
        step = base_step * (0.4 + 0.8 * rng.random())
        idx = rng.randint(0, n, size=4)
        d = step * rng.randn(4)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < -0.1 or h_new.max() > 1.1:
            continue
        h_new = _proj(h_new, n * 0.5)
        c_new = np.max(_fft_corr(h_new)) * dx
        if c_new < c5:
            c5 = c_new
            h = h_new
    return h, c5

def _two_point_polish(h, n_iter, rng, max_step):
    n = len(h)
    dx = 2.0 / n
    c5 = np.max(_fft_corr(h)) * dx
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 10)) % n
        d = max_step * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i] = ni
            h[j] = nj
            c_new = np.max(_fft_corr(h)) * dx
            if c_new < c5:
                c5 = c_new
            else:
                h[i] = ni - d
                h[j] = nj + d
    return h, c5

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _proj(h[:m * f].reshape(m, f).mean(axis=1), m * 0.5)

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return _proj(out, n * f * 0.5)
