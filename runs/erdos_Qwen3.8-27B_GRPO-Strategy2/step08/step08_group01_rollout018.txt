from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _proj(h, target):
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        sh = k - (n - 1)
        if sh == 0:
            g += wk * (1.0 - 2.0 * h)
        elif sh > 0:
            g[sh:] += wk * (1.0 - h[:n-sh])
            g[:n-sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n-sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n-sp]
    return g

def _mgd(h, n_iter, alpha, top_k, step0, deadline):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    mu = 0.75
    step = step0
    cur_c = _eval(h)
    best_h, best_c = h.copy(), cur_c
    for it in range(n_iter):
        if time.time() > deadline:
            break
        y = h + mu * v
        g = _grad(y, alpha, top_k)
        y_new = y - step * g
        v = mu * v + (1.0 - mu) * (y_new - h) / max(step, 1e-10)
        h_try = _proj(y_new + mu * v * step, target)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try
            cur_c = c_try
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
            step = min(step * 1.03, step0 * 5.0)
        else:
            step *= 0.65
            step = max(step, step0 * 0.05)
    return _repair(h, n), best_c

def _slsqp(h, alpha, maxiter, top_k):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    target = n * 0.5
    def _obj(x):
        c = _eval_vec(x)
        idx = np.argsort(c)[-top_k:]
        ct = c[idx]
        m = ct.max()
        s = np.clip(alpha * (ct - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def _grad_obj(x):
        return _grad(x, alpha, top_k)
    def _eq(x):
        return x.sum() - target
    def _eq_j(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _eq, 'jac': _eq_j}]
    try:
        res = minimize(_obj, h, jac=_grad_obj, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _binary_search(h, n_iter, rng, deadline):

    n = len(h)
    # First, threshold to binary
    h_bin = (h > 0.5).astype(float)

    n_ones = h_bin.sum()
    if n_ones != n / 2:

        else:
            for i in range(int(n/2 - n_ones)):
                h_bin[sorted_idx[n-1-i]] = 1.0
    c = _eval(h_bin)
    best_h, best_c = h_bin.copy(), c
    
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]

        if time.time() > deadline:
            break
        oi = ones[rng.randint(len(ones))]
        zi = zeros[rng.randint(len(zeros))]
        h_bin[oi] = 0.0
        h_bin[zi] = 1.0
        c_new = _eval(h_bin)
        if c_new < c:
            c = c_new
            if c < best_c:
                best_c, best_h = c, h_bin.copy()
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
        else:
            h_bin[oi] = 1.0
            h_bin[zi] = 0.0
    return best_h, best_c

def _local2(h, n_iter, step, rng, deadline, sa_temp=None):
    n = len(h)
    cur_c = _eval(h)
    best_h, best_c = h.copy(), cur_c
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 9)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            accept = c_new < cur_c
            if not accept and sa_temp is not None and sa_temp > 1e-12:
                if rng.rand() < np.exp(-(c_new - cur_c) / sa_temp):
                    accept = True
            if accept:
                cur_c = c_new
                if c_new < best_c:
                    best_c, best_h = c_new, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _sweep(h, n_passes, step, rng, deadline):

    cur_c = _eval(h)
    best_h, best_c = h.copy(), cur_c
    for p in range(n_passes):

                    if c_new < best_c:
                        best_c, best_h = c_new, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _proj(h, target):
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        sh = k - (n - 1)
        if sh == 0:
            g += wk * (1.0 - 2.0 * h)
        elif sh > 0:
            g[sh:] += wk * (1.0 - h[:n-sh])
            g[:n-sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n-sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n-sp]
    return g

def _mgd(h, n_iter, alpha, top_k, step0, deadline):
    n = len(h)
    target = n * 0.5
    vel = np.zeros(n)
    mu = 0.72
    step = step0
    cur_c = _eval(h)
    best_h, best_c = h.copy(), cur_c
    for it in range(n_iter):
        if time.time() > deadline:
            break
        y = h + mu * vel
        g = _grad(y, alpha, top_k)
        y_new = y - step * g
        h_try = _proj(y_new, target)
        c_try = _eval(h_try)
        if c_try < cur_c:
            vel = h_try - h
            h = h_try
            cur_c = c_try
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
            step = min(step * 1.04, step0 * 6.0)
        else:
            step *= 0.6
            step = max(step, step0 * 0.05)
            vel *= mu
    return _repair(h, n), best_c

def _slsqp(h, alpha, maxiter, top_k):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    target = n * 0.5
    def _obj(x):
        c = _eval_vec(x)
        idx = np.argsort(c)[-top_k:]
        ct = c[idx]
        m2 = ct.max()
        s = np.clip(alpha * (ct - m2), -400, 400)
        s -= s.max()
        return m2 + np.log(np.exp(s).sum()) / alpha
    def _go(x):
        return _grad(x, alpha, top_k)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h, jac=_go, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _bin_search(h_init, n_iter, rng, deadline):
    n = len(h_init)
    h = (h_init > 0.5).astype(float)
    no = int(h.sum())
    if no != n // 2:
        si = np.argsort(h_init)
        if no > n // 2:
            for i in range(no - n // 2):
                h[si[i]] = 0.0
        else:
            for i in range(n // 2 - no):
                h[si[n - 1 - i]] = 1.0
    c = _eval(h)
    best_h, best_c = h.copy(), c
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        oi = ones[rng.randint(len(ones))]
        zi = zeros[rng.randint(len(zeros))]
        h[oi] = 0.0
        h[zi] = 1.0
        cn = _eval(h)
        if cn < c:
            c = cn
            if c < best_c:
                best_c, best_h = c, h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[oi] = 1.0
            h[zi] = 0.0
    return best_h, best_c

def _local2(h, n_iter, step, rng, deadline, sa_t=None):
    n = len(h)
    cur_c = _eval(h)
    best_h, best_c = h.copy(), cur_c
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 9)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            acc = cn < cur_c
            if not acc and sa_t is not None and sa_t > 1e-14:
                if rng.rand() < np.exp(-(cn - cur_c) / sa_t):
                    acc = True
            if acc:
                cur_c = cn
                if cn < best_c:
                    best_c, best_h = cn, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _sweep(h, n_passes, step, rng, deadline):
    n = len(h)
    cur_c = _eval(h)
    best_h, best_c = h.copy(), cur_c
    for _ in range(n_passes):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            j = (i + 1 + rng.randint(0, 5)) % n
            d = step * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                cn = _eval(h)
                if cn < cur_c:
                    cur_c = cn
                    if cn < best_c:
                        best_c, best_h = cn, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _attack_worst(h, n_iter, rng, deadline):
    """Focus search on the worst shift's contributing positions."""
    n = len(h)
    cur_c = _eval(h)
    best_h, best_c = h.copy(), cur_c
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c_vec = _eval_vec(h)
        worst_k = np.argmax(c_vec)
        sh = worst_k - (n - 1)
        if sh >= 0:
            positions_bad = np.where((h[sh:] > 0.5) & (h[:n-sh] < 0.5))[0] + sh
        else:
            sp = -sh
            positions_bad = np.where((h[:n-sp] > 0.5) & (h[sp:] < 0.5))[0]
        if len(positions_bad) == 0:
            positions_bad = np.arange(n)
        i = positions_bad[rng.randint(len(positions_bad))]
        j = (i + rng.randint(1, 11)) % n
        d = 3e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < cur_c:

                if cn < best_c:
                    best_c, best_h = cn, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c
