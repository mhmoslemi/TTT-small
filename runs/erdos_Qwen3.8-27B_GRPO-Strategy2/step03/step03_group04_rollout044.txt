import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)


def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h


def _eq_target(x, target):
    return np.sum(x) - target


def _do_slsqp(x0, bounds, target, alpha, maxiter):
    res = minimize(_smooth_max, x0, args=(alpha,), method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': _eq_target, 'args': (target,)}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return res.x


def _pair_search(h, n, dx, deltas, max_passes, deadline):
    c5_cur = _c5(h, dx)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        imp = False
        for i in range(n):
            if time.time() > deadline:
                break
            hi = h[i]
            for j in range(i + 1, n):
                if time.time() > deadline:
                    break
                hj = h[j]
                for d in deltas:
                    ni, nj = hi + d, hj - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, dx)
                    if c < c5_cur - 1e-15:
                        c5_cur, hi, hj = c, ni, nj
                        imp = True
                    else:
                        h[i], h[j] = hi, hj
                h[i], h[j] = hi, hj
            h[i] = hi
        if not imp:
            break
    return h, c5_cur

    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    h1 = _repair(1.0 - h0, n)
    c5_1 = _c5(h1, dx)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 1: SLSQP warm-up
    dl1 = t0 + budget_s * 0.15
    for alpha in [500, 2000, 10000, 50000, 200000]:
        if time.time() > dl1:

        try:

            c5 = _c5(hc, dx)
            if c5 < best_c5:

        except Exception:
            pass

    # Phase 2: Subgradient on top-K lags
    dl2 = t0 + budget_s * 0.30

    c5_cur = _c5(h, dx)

    dl6 = t0 + budget_s * 0.95
    h = best_h.copy()
    c5_cur = _c5(h, dx)

        j = rng.randint(n)
        if i == j:

        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < c5_cur:
                c5_cur = c
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    if c5_cur < best_c5:
        best_c5, best_h = c5_cur, h.copy()

    # Phase 7: Final SLSQP
    if time.time() < t0 + budget_s - 5:
        try:
            rx = _do_slsqp(best_h.copy(), bounds, target, 500000, 1000)
            hc = _repair(rx, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:

            pass

            hf = _repair(1.0 - best_h, n)
            rx = _do_slsqp(hf, bounds, target, 200000, 500)

            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc.copy()
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
