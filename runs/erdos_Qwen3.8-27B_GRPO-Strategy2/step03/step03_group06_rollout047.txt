import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _to_binary(h, n, target, rng):
    """Convert h to a binary vector with exactly target ones"""
    b = np.zeros(n)
    idx = np.argsort(h)[::-1][:target]
    b[idx] = 1.0
    return b

def _sa_binary(h, n, dx, rng, t_end, T0=0.03, T1=0.001, cool=0.99999):

    cur = h.copy()
    cur_c5 = _c5(cur, dx)
    best = cur.copy()
    best_c5 = cur_c5
    T = T0
    n_ones = target  # 35

                    best_c5 = nc
                    best = cur.copy()
            else:
                cur[i] = 1.0
                cur[j] = 0.0
        if not improved:
            stall += 200
    return best, best_c5

def _dbl_swap(h, n, dx, rng, t_end, max_tries=50000):
    cur = h.copy()
    cur_c5 = _c5_val(cur, dx)
    best = cur.copy()
    best_c5 = cur_c5
    for _ in range(max_tries):
        if time.time() >= t_end:
            break
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        a, b = rng.choice(ones, size=2, replace=False)
        c, d = rng.choice(zeros, size=2, replace=False)
        cur[a] = 0.0; cur[b] = 0.0; cur[c] = 1.0; cur[d] = 1.0
        nc = _c5_val(cur, dx)
        if nc < cur_c5 - 1e-14:
            cur_c5 = nc
            if nc < best_c5:
                best_c5 = nc
                best = cur.copy()
        else:
            cur[a] = 1.0; cur[b] = 1.0; cur[c] = 0.0; cur[d] = 0.0
    return best, best_c5

def _targeted(h, n, dx, rng, t_end):
    cur = h.copy()
    cur_c5 = _c5_val(cur, dx)
    best = cur.copy()
    best_c5 = cur_c5
    while time.time() < t_end:
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        mx = np.max(corr)
        worst = np.where(corr >= mx - 0.01)[0]
        improved = False
        for wl in worst:
            d = int(wl) - (n - 1)
            if d == 0:
                continue
            ad = abs(d)
            if ad >= n:
                continue
            cands = []
            if d > 0:
                for ii in range(n - ad):
                    if cur[ii] == 1 and cur[ii + ad] == 0:
                        cands.append(ii)
            else:
                for ii in range(ad, n):
                    if cur[ii] == 1 and cur[ii - ad] == 0:
                        cands.append(ii)
            if not cands:
                continue
            zeros = np.where(cur == 0)[0]
            ones = np.where(cur == 1)[0]
            for _ in range(30):
                if time.time() >= t_end:
                    break
                ii = cands[rng.randint(len(cands))]
                jj = int(zeros[rng.randint(len(zeros))])
                cur[ii] = 0.0
                cur[jj] = 1.0
                nc = _c5_val(cur, dx)
                if nc < cur_c5 - 1e-14:
                    cur_c5 = nc
                    improved = True
                    if nc < best_c5:
                        best_c5 = nc
                        best = cur.copy()
                else:
                    cur[ii] = 1.0
                    cur[jj] = 0.0
        if not improved:
            for _ in range(300):
                if time.time() >= t_end:
                    break
                ones = np.where(cur == 1)[0]
                zeros = np.where(cur == 0)[0]
                a, b = rng.choice(ones, size=2, replace=False)
                c2, d2 = rng.choice(zeros, size=2, replace=False)
                cur[a] = 0; cur[b] = 0; cur[c2] = 1; cur[d2] = 1
                nc = _c5_val(cur, dx)
                if nc < cur_c5 - 1e-14:
                    cur_c5 = nc
                    improved = True
                    if nc < best_c5:
                        best_c5 = nc
                        best = cur.copy()
                else:
                    cur[a] = 1; cur[b] = 1; cur[c2] = 0; cur[d2] = 0
            if not improved:
                break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n // 2
    h0 = np.array(initial_h_values, dtype=float)
    hb = _to_binary(h0, n, target)
    best_h = hb.copy()
    best_c5 = _c5_val(best_h, dx)

    starts = [hb.copy()]
    t = np.arange(n) / n
    for f in range(1, 10):
        ph = rng.uniform(0, 2 * np.pi)
        hs = 0.5 + 0.4 * np.sin(2 * np.pi * f * t + ph) + 0.1 * np.cos(4 * np.pi * f * t + 2 * ph)
        starts.append(_to_binary(hs, n, target))
    for _ in range(25):
        starts.append(_to_binary(rng.uniform(0, 1, n), n, target))
    starts = [_to_binary(s, n, target) for s in starts]

    # Phase 1: SA on top starts (long)
    t_end = t0 + 350
    for s in starts[:6]:
        if time.time() >= t_end:
            break
        hs, cs = _sa(s, n, dx, rng, min(t_end, time.time() + 60), 0.05, 0.003, 0.99999)
        if cs < best_c5:
            best_h, best_c5 = hs, cs

    # Phase 2: SA on remaining starts
    t_end = t0 + 600
    for s in starts[6:]:
        if time.time() >= t_end:
            break
        hs, cs = _sa(s, n, dx, rng, min(t_end, time.time() + 15), 0.04, 0.002, 0.99999)
        if cs < best_c5:
            best_h, best_c5 = hs, cs

    # Phase 3: Targeted local search
    if time.time() < t0 + 850:
        t_e = min(t0 + 850, time.time() + 150)
        ht, ct = _targeted(best_h, n, dx, rng, t_e)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    # Phase 4: Hill climbing
    if time.time() < t0 + 950:
        t_e = min(t0 + 950, time.time() + 80)
        hl, cl = _hill(best_h, n, dx, rng, t_e)
        if cl < best_c5:
            best_h, best_c5 = hl, cl

    # Phase 5: Double swap
    if time.time() < t0 + 1020:
        t_e = min(t0 + 1020, time.time() + 50)
        hd, cd = _dbl_swap(best_h, n, dx, rng, t_e, max_tries=40000)
        if cd < best_c5:
            best_h, best_c5 = hd, cd

    # Phase 6: Final SA (low temp)
    if time.time() < t0 + budget_s - 20:
        t_e = min(t0 + budget_s - 20, time.time() + 70)
        hs, cs = _sa(best_h, n, dx, rng, t_e, 0.015, 0.001, 0.99999)
        if cs < best_c5:
            best_h, best_c5 = hs, cs

    # Phase 7: Final targeted + hill
    if time.time() < t0 + budget_s - 5:
        t_e = t0 + budget_s - 5
        ht, ct = _targeted(best_h, n, dx, rng, t_e)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    return (best_h, best_c5, n)
