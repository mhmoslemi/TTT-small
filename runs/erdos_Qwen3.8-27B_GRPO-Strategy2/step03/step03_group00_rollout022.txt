import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _corr_vals(h, n):
    """Return all correlation values (2n-1 values) scaled by dx."""
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5(h, n):
    return float(np.max(_corr_vals(h, n)))

def _grad_lag(h, n, L):
    """Gradient of the L-th lag correlation w.r.t. h, scaled by dx."""
    dx = 2.0 / n
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        g[:n-L] -= h[L:]
        g[L:] += (1.0 - h[:n-L])
    return g * dx

def _worst_lags(c, m=5):
    """Return indices of top-m correlation values."""
    n = len(c)  # 2n-1
    idx = np.argsort(c)[::-1][:m]
    return idx

def _working_set_descent(h, n, lr_init, budget_end, rng, M=5):
    """Subgradient descent with working set."""
    best_h = h.copy()
    best_c5 = _c5(h, n)
    c = _corr_vals(h, n)
    
    for t in range(20000):
        if time.time() > budget_end:
            break
        # Update working set every 20 iterations
        if t % 20 == 0:
            c = _corr_vals(h, n)
            wl = _worst_lags(c, M)
        
        # Compute gradient as average over working set
        g = np.zeros(n)
        for k in wl:
            L = abs(k - (n - 1))
            g += _grad_lag(h, n, L)
        g /= len(wl)
        
        # Step size: decreasing
        lr = lr_init / (1.0 + 0.01 * t)
        
        # Take step and project
        h_new = h - lr * g
        h_new = _repair(h_new, n)
        
        c5_new = _c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

- Subdiagonal (j' = j-L < j): H[j, j-L] = -1 for j=L,...,n-1
  In numpy: idx = np.arange(n-L); H[idx+L, idx] = -1

So:
```python
idx = np.arange(n-L)
H[idx, idx+L] -= 1.0
H[idx+L, idx] -= 1.0
```

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _c5(h, n):
    return float(np.max(_corr(h, n)))

def _grad_hess(h, n, Ls):
    dx = 2.0 / n
    M = len(Ls)
    g = np.zeros(n)
    H = np.zeros((n, n))
    for L in Ls:
        if L == 0:
            g += 1.0 - 2.0 * h
            H[n, n]  # no-op, use diagonal
            H[np.arange(n), np.arange(n)] += -2.0
        else:
            g[:n - L] -= h[L:]
            g[L:] += 1.0 - h[:n - L]
            idx = np.arange(n - L)
            H[idx, idx + L] -= 1.0
            H[idx + L, idx] -= 1.0
    g *= dx / M
    H *= dx / M
    return g, H

def _newton_iter(h, n, c5_cur, lags, lam=0.5):
    g, H = _grad_hess(h, n, lags)
    reg = H + lam * np.eye(n)
    try:
        dh = np.linalg.solve(reg, -g)
    except np.linalg.LinAlgError:
        dh = -g * 0.01
    best_h = h.copy()
    best_c = c5_cur
    for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03]:
        h_new = _repair(h + alpha * dh, n)
        c_new = _c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if best_c >= c5_cur:
        g_norm = np.dot(g, g)
        if g_norm > 1e-20:
            dh2 = -g / g_norm
            for alpha in [0.01, 0.005, 0.001]:
                h_new = _repair(h + alpha * dh2, n)
                c_new = _c5(h_new, n)
                if c_new < c5_cur:
                    best_h = h_new
                    best_c = c_new
                    break
    return best_h, best_c

def _slsqp_smooth(x, n, alpha):
    c = _corr(x, n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp_grad(x, n, alpha):
    c = _corr(x, n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    n2 = len(c)
    for k in range(n2):
        if w[k] < 1e-13:
            continue
        L = abs(k - (n - 1))
        if L == 0:
            g += w[k] * (1.0 - 2.0 * x)
        else:
            g[:n - L] += w[k] * (-x[L:])
            g[L:] += w[k] * (1.0 - x[:n - L])
    return g

    rng = np.random.RandomState(seed)

    raw = np.array(initial_h_values, dtype=float)

    best_c5 = _c5(h, n)

    # Phase 1: Newton with working set (45% time)
    p1_end = t0 + budget_s * 0.45
    M_bal = 5
    stuck = 0
    for it in range(20000):
        if elapsed() > p1_end:
            break
        c = _corr(h, n)
        top = np.argsort(c)[::-1]
        # Alternate between M=1 (focused) and M=M_bal (balanced)
        if it % 30 < 20:
            lags = [abs(top[0] - (n - 1))]
        else:
            lags_list = [abs(k - (n - 1)) for k in top[:M_bal]]
            # Remove duplicates

            lags = []
            for L in lags_list:

            if len(lags) < 2:

        h_new, c_new = _newton_iter(h, n, best_c5, lags, lam=0.3 + 0.5 * (it % 10) / 10.0)

    # Phase 4: Simulated annealing (12% time)
    p4_end = t0 + budget_s * 0.92
    h = best_h.copy()
    c5_cur = _c5(h, n)
    T = 0.0004
    min_T = 5e-10
    cool = 0.99985
    steps = np.array([0.003, 0.008, 0.02, 0.05, 0.10])

        hi, hj = h[i], h[j]

        h[i] = max(0.0, min(1.0, ni))

        c_new = _c5(h, n)
        if c_new < c5_cur or (T > 1e-10 and rng.rand() < np.exp(-min((c_new - c5_cur) / T, 50))):
            c5_cur = c_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        else:

        T = max(T * cool, min_T)

    # Phase 5: Ultra-fine local search (remaining)
    h = best_h.copy()
    c5_cur = _c5(h, n)
    end_t = t0 + budget_s - 5
