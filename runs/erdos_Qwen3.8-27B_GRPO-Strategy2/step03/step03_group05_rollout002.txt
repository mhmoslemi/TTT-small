    h = h.copy()

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    return _project(h, target)

    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _softgrad(x, alpha):
    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)

        pk = pw[k]
        if pk < 1e-14:

            g[M:] -= pk * x[:n - M]

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _project_sum(h, target):
    h = h.copy()
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    return _project_sum(h, target)

def _softobj(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _softgrad(x, alpha):
    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    return g / alpha

def _proj_grad_opt(h, n, dx, target, alpha, lr, maxit, tlim):
    t0 = time.time()
    h = h.copy()
    c5_cur = _c5(h, dx)
    for it in range(maxit):
        if time.time() - t0 > tlim:
            break
        g = _softgrad(h, alpha)
        g = g - g.mean()
        step = lr
        improved = False
        for _bt in range(25):
            h_new = _project_sum(h - step * g, target)
            c5_new = _c5(h_new, dx)
            if c5_new < c5_cur - 1e-16:
                h = h_new
                c5_cur = c5_new
                improved = True
                break
            step *= 0.5
        if not improved:
            break
    return h, c5_cur

def _periodic_obj(base, p, n, dx):
    h = np.tile(base, n // p)
    return _c5(h, dx)

def _base_sum(base, target_sum):
    return base.sum() - target_sum

def _try_periodic(p, n, dx, target, rng, tlim):
    t0 = time.time()
    if time.time() - t0 > tlim:
        return None
    reps = n // p
    base_sum = target / reps
    n_ones = int(base_sum)
    frac = base_sum - n_ones
    base = np.zeros(p)
    perm = rng.permutation(p)
    base[perm[:n_ones]] = 1.0
    if frac > 1e-10:
        base[perm[n_ones]] = frac
    h0 = _project_sum(np.tile(base, reps), target)
    h0_base = h0[:p].copy()
    cons = {'type': 'eq', 'fun': _base_sum, 'args': (base_sum,)}
    bounds_p = [(0.0, 1.0)] * p
    try:
        res = minimize(_periodic_obj, h0_base, args=(p, n, dx),
                       method='SLSQP', bounds=bounds_p, constraints=cons,
                       options={'maxiter': 500, 'ftol': 1e-14})
        hb = np.clip(res.x, 0.0, 1.0)
        hb = _project_sum(hb, base_sum)
        h = _project_sum(np.tile(hb, reps), target)
        return h, _c5(h, dx)
    except Exception:
        h = _project_sum(np.tile(base, reps), target)
        return h, _c5(h, dx)

def _binary_local_search(h, n, dx, target, rng, tlim):
    t0 = time.time()
    h = h.copy()
    c5_cur = _c5(h, dx)
    for _it in range(50000):
        if time.time() - t0 > tlim:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        if h[i] > 0.5 and h[j] < 0.5:
            h[i] = 0.0
            h[j] = 1.0
            c = _c5(h, dx)
            if c > c5_cur + 1e-15:
                h[i] = 1.0
                h[j] = 0.0
        elif h[i] < 0.5 and h[j] > 0.5:
            h[i] = 1.0
            h[j] = 0.0
            c = _c5(h, dx)
            if c > c5_cur + 1e-15:
                h[i] = 0.0
                h[j] = 1.0
    return h, _c5(h, dx)

def _coord_polish(h, n, dx, target, rng, tlim):
    t0 = time.time()
    h = h.copy()
    c5_cur = _c5(h, dx)
    deltas = np.array([0.001, 0.003, 0.01, 0.03, 0.05, -0.001, -0.003, -0.01, -0.03, -0.05])
    for _sweep in range(200):
        if time.time() - t0 > tlim:
            break
        g = _softgrad(h, 80000)
        order = np.argsort(np.abs(g))[::-1][:30]
        improved = False
        for ii in range(len(order)):
            if time.time() - t0 > tlim:
                break
            i = order[ii]
            for jj in range(ii + 1, min(ii + 15, len(order))):
                j = order[jj]
                best_d, best_c = 0.0, c5_cur
                for d in deltas:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < best_c:
                            best_c, best_d = c, d
                        h[i], h[j] = h[i] - d if best_d == 0 else ni, h[j] - d if best_d == 0 else nj
                        h[i], h[j] = min(max(ni - best_d, 0), 1), min(max(nj + best_d, 0), 1) if best_d != 0 else (h[i], h[j])
                # redo cleanly
                h[i], h[j] = h[i], h[j]
                # simpler: just try swaps
                for d in deltas:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < best_c:
                            best_c, best_d = c, d
                        h[i], h[j] = ni - d, nj + d
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    c5_cur = best_c
                    improved = True
                    break
            if improved:
                h = _project_sum(h, target)
                c5_cur = _c5(h, dx)
                break
        if not improved:
            break
    return h, c5_cur

def _slsqp_refine(h, n, dx, target, alpha, tlim):
    t0 = time.time()
    if time.time() - t0 > tlim:
        return h, _c5(h, dx)
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': _base_sum, 'args': (target,)}
    try:
        res = minimize(_softobj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-15})
        hc = _project_sum(res.x, target)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0

    best_h = _repair(np.array(initial_h_values, dtype=float), n)
    best_c5 = _c5(best_h, dx)

    def elapsed():
        return time.time() - t0

    # Phase 1: Periodic pattern exploration
    periods = [7, 10, 14, 20, 35]
    for p in periods:
        if elapsed() > budget_s - 600:
            break
        res = _try_periodic(p, n, dx, target, rng, 120)
        if res is not None:
            h_p, c_p = res
            if c_p < best_c5:
                best_c5, best_h = c_p, h_p.copy()

    # Phase 2: Multi-start projected gradient with varying alpha
    starts = [best_h.copy()]
    for _ in range(4):
        h_rand = rng.uniform(0, 1, n)
        starts.append(_repair(h_rand, n))
    # Binary starts
    for _ in range(2):
        h_bin = np.zeros(n)
        h_bin[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(h_bin)

    for h0 in starts:
        if elapsed() > budget_s - 300:
            break
        for alpha in [2000, 10000, 50000, 200000]:
            if elapsed() > budget_s - 300:
                break
            h_g, c_g = _proj_grad_opt(h0, n, dx, target, alpha, 0.01, 400,
                                      min(90, budget_s - 300 - elapsed()))
            if c_g < best_c5:
                best_c5, best_h = c_g, h_g.copy()
            h0 = h_g.copy()

    # Phase 3: SLSQP refinement
    if elapsed() < budget_s - 200:
        h_s, c_s = _slsqp_refine(best_h, n, dx, target, 100000,
                                  min(150, budget_s - 200 - elapsed()))
        if c_s < best_c5:
            best_c5, best_h = c_s, h_s.copy()
        h_s, c_s = _slsqp_refine(best_h, n, dx, target, 300000,
                                  min(100, budget_s - 200 - elapsed()))
        if c_s < best_c5:
            best_c5, best_h = c_s, h_s.copy()

    # Phase 4: Binary local search
    if elapsed() < budget_s - 120:
        h_b = np.where(best_h > 0.5, 1.0, 0.0)
        # Ensure sum = target
        excess = h_b.sum() - target
        if excess > 0:
            idx = np.where(h_b == 1.0)[0]
            h_b[idx[:int(excess)]] = 0.0
        elif excess < 0:
            idx = np.where(h_b == 0.0)[0]
            h_b[idx[:int(-excess)]] = 1.0
        h_b, c_b = _binary_local_search(h_b, n, dx, target, rng,
                                         min(90, budget_s - 120 - elapsed()))
        if c_b < best_c5:
            best_c5, best_h = c_b, h_b.copy()

    # Phase 5: Coordinate polish on best
    if elapsed() < budget_s - 60:
        h_pol, c_pol = _coord_polish(best_h, n, dx, target, rng,
                                      min(55, budget_s - 60 - elapsed()))
        if c_pol < best_c5:
            best_c5, best_h = c_pol, h_pol.copy()

    # Phase 6: Final high-alpha projected gradient
    if elapsed() < budget_s - 20:
        h_f, c_f = _proj_grad_opt(best_h, n, dx, target, 500000, 0.003, 300,
                                   min(18, budget_s - 20 - elapsed()))
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f.copy()

    # Phase 7: Ultra-fine random polish
    if elapsed() < budget_s - 3:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        tiny = np.array([0.0005, 0.001, 0.002, 0.005, -0.0005, -0.001, -0.002, -0.005])
        n_iters = min(20000, int((budget_s - 3 - elapsed()) * 3000))
        for _ in range(n_iters):
            if elapsed() > budget_s - 3:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue

            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5_cur:
                    c5_cur = c
                    best_c5 = c
                    best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
