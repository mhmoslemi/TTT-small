import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _c5_exact(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _smooth_obj(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -400, 400)
    val = m + np.log(np.sum(np.exp(s))) / alpha
    return val + 100.0 * (np.sum(x) - target) ** 2

def _lbfgs(h, n, dx, alpha, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, args=(alpha, target), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-8})
        hc = _repair(res.x, n)
        return hc, _c5_exact(hc, dx)
    except:
        return h, _c5_exact(h, dx)

def _slsqp(h, n, dx, alpha, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}
    try:

        else:

    return h, best_c5

        else:

def _anneal(h, n, dx, rng, deadline):
    """Simulated annealing with pair moves."""
    h_cur = h.copy()
    best_c5 = _c5_exact(h_cur, dx)
    h_best = h_cur.copy()
    c5_cur = best_c5

def _directed_step(h, n, dx, rng, deadline, n_iters):
    """Directed search: focus on reducing the maximum lag."""
    best_c5 = _c5_exact(h, dx)
    h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    
    for it in range(n_iters):
        if time.time() >= deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = np.argmax(corr)
        # Physical lag = k_max - mid
        lag = k_max - mid  # in units of dx
        # To reduce corr[k_max], we want to reduce h[i]*(1-h[i+lag]) terms
        # Find the index i that contributes most to corr[k_max]
        # corr[k_max] = sum_i h[i] * (1-h[i+lag]) for valid i
        
        # For the "first" part (lag >= 0): valid i are 0 to n-1-lag
        # For the "second" part (lag < 0): valid i are -lag to n-1
        if lag >= 0:
            valid_i = np.arange(0, n - lag)
        else:
            valid_i = np.arange(-lag, n)
        
        if len(valid_i) == 0:
            continue
        
        contrib = h[valid_i] * (1.0 - h[valid_i + lag])
        i_best = valid_i[np.argmax(contrib)]
        
        # Move mass from i_best toward i_best+lag (to increase h there)
        # Or: pick random i from valid_i and move mass
        if rng.random() < 0.3:
            i = i_best
        else:
            i = valid_i[rng.randint(len(valid_i))]
        
        j = i + lag if 0 <= i + lag < n else i
        if i == j:
            continue

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _smooth_obj(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha + 50.0 * (np.sum(x) - target) ** 2

def _lbfgs(h, n, dx, alpha, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, args=(alpha, target), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-9})
        return _repair(res.x, n), _c5(_repair(res.x, n), dx)
    except Exception:
        return h, _c5(h, dx)

def _slsqp(h, n, dx, alpha, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': lambda x, t=target: np.sum(x) - t, 'jac': lambda x, n=n: np.ones(n)}
    try:
        res = minimize(lambda x: _smooth_obj(x, alpha, target), h, method='SLSQP',
                       bounds=bounds, constraints=[con], options={'maxiter': maxiter, 'ftol': 1e-17})
        return _repair(res.x, n), _c5(_repair(res.x, n), dx)
    except Exception:
        return h, _c5(h, dx)

def _directed_search(h, n, dx, rng, deadline):
    best = _c5(h, dx)
    h = h.copy()
    mid = n - 1
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = int(np.argmax(corr))
        lag = k_max - mid
        if lag >= 0:
            vi = np.arange(0, n - lag)
        else:
            vi = np.arange(-lag, n)
        if len(vi) == 0:
            break
        contrib = h[vi] * (1.0 - h[vi + lag])
        tot = contrib.sum()
        if tot < 1e-12:
            i = rng.randint(n)
            j = rng.randint(n)
        else:
            i = int(vi[rng.choice(len(vi), p=contrib / tot)])
            j = i + lag
            if not (0 <= j < n) or i == j:
                i = rng.randint(n)
                j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(0.003, 0.02) * (1.0 if rng.random() < 0.5 else -1.0)
        ni, nj = h[i] + d, h[j] - d
        if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
            continue
        h[i], h[j] = ni, nj
        c5n = _c5(h, dx)
        if c5n < best - 1e-15:
            best = c5n
        else:
            h[i], h[j] = ni - d, nj + d
    return h, best

def _random_pairs(h, n, dx, rng, deadline):
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.02, 0.012, 0.007, 0.004, 0.002, 0.001]
    for sc in scales:
        tEnd = min(deadline, time.time() + 60)
        while time.time() < tEnd:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _c5(h, dx)
            if c5n < best - 1e-15:
                best = c5n
            else:
                h[i], h[j] = ni - d, nj + d
    return h, best

def _block_move(h, n, dx, rng, deadline, bs):
    best = _c5(h, dx)
    h = h.copy()
    while time.time() < deadline:
        i = rng.randint(0, max(1, n - bs))
        j = rng.randint(0, max(1, n - bs))
        if abs(i - j) < bs:
            continue
        d = rng.uniform(0.003, 0.012) * (1.0 if rng.random() < 0.5 else -1.0)
        seg_i = h[i:i+bs]
        seg_j = h[j:j+bs]
        if np.any(seg_i + d < -1e-10) or np.any(seg_i + d > 1.0 + 1e-10):
            continue
        if np.any(seg_j - d < -1e-10) or np.any(seg_j - d > 1.0 + 1e-10):
            continue
        h[i:i+bs] += d
        h[j:j+bs] -= d
        c5n = _c5(h, dx)
        if c5n < best - 1e-15:
            best = c5n
        else:
            h[i:i+bs] -= d
            h[j:j+bs] += d
    return h, best

def _anneal(h, n, dx, rng, deadline):
    h_cur = h.copy()
    best = _c5(h_cur, dx)
    h_best = h_cur.copy()
    c5_cur = best
    T = 0.0015
    T_min = 5e-6
    cool = 0.99985
    while time.time() < deadline and T > T_min:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 0.012 * (2.0 * rng.random() - 1.0)
        ni, nj = h_cur[i] + d, h_cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            T *= cool
            continue
        h_cur[i], h_cur[j] = ni, nj
        c5n = _c5(h_cur, dx)
        delta = c5n - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            c5_cur = c5n
            if c5n < best:
                best = c5n
                h_best = h_cur.copy()
        else:
            h_cur[i], h_cur[j] = ni - d, nj + d
        T *= cool
    return h_best, best

def _perturb(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        f = rng.randint(2, 8)
        hp += 0.1 * np.sin(2*np.pi*f*np.arange(n)/n + rng.uniform(0, 6.28))
    elif kind == 1:
        j = rng.randint(n//3, 2*n//3)
        hp = np.roll(hp, j)
    elif kind == 2:
        blk = rng.randint(5, 15)
        i = rng.randint(0, n - blk)
        hp[i:i+blk] = np.roll(hp[i:i+blk], rng.randint(1, blk))
    else:
        hp += rng.uniform(-0.05, 0.05, n)
    return _repair(hp, n)
