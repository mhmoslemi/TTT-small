        if w[idx] < 1e-14:
            continue
        d = abs(idx - center)
        wi = w[idx]
        if d == 0:
            grad += wi * (1.0 - 2.0 * h)
        else:
            grad[d:] += wi * (1.0 - h[:-d])
            grad[:n-d] -= wi * h[d:]
    return obj, grad

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _lbfgs(h0, n, dx, alpha, maxiter=500, tol=1e-14):

        return _smooth_obj_grad(x, alpha)[1]
    best_h, best_c5 = h0.copy(), _c5(h0, dx)

def _pen_lbfgs(h0, n, dx, alpha, pen, maxiter=400):

def _pair_local(h, n, dx, rng, deadline, ntrials=50000):
    best_c5 = _c5(h, dx)
    h = h.copy()
    corr = _corr(h)
    scales = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    per = max(200, ntrials // len(scales))
    for sc in scales:
        imp = False
        for _ in range(per):
            if time.time() >= deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5 = _c5(h, dx)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                imp = True
            else:
                h[i], h[j] = h[i] - d, h[j] + d
        if not imp and sc < 0.003:
            break
    return h, best_c5

def _block_local(h, n, dx, rng, deadline, niter=5000):
    best_c5 = _c5(h, dx)
    h = h.copy()
    for bs in [2, 3]:
        iters = niter // 2
        for _ in range(iters):
            if time.time() >= deadline:
                break
            i = rng.randint(0, n - bs)
            j = rng.randint(0, n - bs)
            if abs(i - j) < bs:
                continue
            eps = rng.uniform(0.003, 0.02) * (1 if rng.random() < 0.5 else -1)
            if np.any(h[i:i+bs] + eps < 0) or np.any(h[i:i+bs] + eps > 1):
                continue
            if np.any(h[j:j+bs] - eps < 0) or np.any(h[j:j+bs] - eps > 1):
                continue
            h[i:i+bs] += eps
            h[j:j+bs] -= eps
            c5 = _c5(h, dx)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
            else:
                h[i:i+bs] -= eps
                h[j:j+bs] += eps
    return h, best_c5

def _focal_opt(h, n, dx, alpha, n_focal, maxiter=300):
    """Optimize weighted sum of top lags + smooth max of rest."""
    corr = _corr(h)
    top_idx = np.argsort(corr)[-n_focal:]
    top_set = set(top_idx)
    bounds = [(0.0, 1.0)] * n
    
    def fobj(x):
        c = _corr(x)
        m = np.max(c)
        s = np.clip(alpha * (c - m), -500, 500)
        e = np.exp(s)
        se = np.sum(e)
        obj = m + np.log(se) / alpha
        # Add extra weight to top lags
        for ti in top_set:
            obj += c[ti]
        return obj
    
    def fgrad(x):
        c = _corr(x)
        m = np.max(c)
        s = np.clip(alpha * (c - m), -500, 500)
        e = np.exp(s)
        se = np.sum(e)
        w = e / se
        nc = 2 * n - 1
        center = n - 1
        grad = np.zeros(n)
        for idx in range(nc):
            if w[idx] < 1e-14:
                continue
            d = abs(idx - center)
            wi = w[idx]
            if idx in top_set:
                wi += 1.0
            if d == 0:
                grad += wi * (1.0 - 2.0 * x)
            else:
                grad[d:] += wi * (1.0 - x[:-d])
                grad[:n-d] -= wi * x[d:]
        return grad
    
    best_h, best_c5 = h.copy(), _c5(h, dx)
    try:
        res = minimize(fobj, h, jac=fgrad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-7})
        hc = _repair(res.x, n)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_h, best_c5 = hc, c5
    except:
        pass
    return best_h, best_c5

def _cos_init(n, A=0.5):
