import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break

    s = np.clip(alpha * (c - m), -500, 500)

            g[idx] -= wm * x[idx + k]
        else:
            idx = np.arange(-k, n)
            g[idx + k] += wm * (ones - x[idx])
            g[idx] -= wm * x[idx + k]
    return g

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': iters, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _sf(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _sj(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-15:
            continue
        wm = w[mi]
        k = mi - (n - 1)
        if k >= 0:
            idx = np.arange(n - k)
            g[idx + k] += wm * (1.0 - x[idx])
            g[idx] -= wm * x[idx + k]
        else:
            idx = np.arange(-k, n)
            g[idx + k] += wm * (1.0 - x[idx])
            g[idx] -= wm * x[idx + k]
    return g

def _slsqp(h, n, alpha, iters, ftol):

             'jac': lambda x: np.ones(n)}]

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': iters, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _line_search_pair(h, i, j, n):
    """Find best d for transfer from j to i via ternary search on max corr."""
    lo = max(-h[j], -1e-6)
    hi = min(1.0 - h[i], 1e-6 + h[j])
    if hi - lo < 1e-10:
        return 0.0, _eval(h)
    best_d = 0.0
    best_v = _eval(h)
    for _ in range(40):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        h[i], h[j] = h[i] + m1, h[j] - m1
        v1 = _eval(h)
        h[i], h[j] = h[i] + m2, h[j] - m2
        v2 = _eval(h)
        if v1 < v2:
            hi = m2
            if v1 < best_v:
                best_v = v1
                best_d = m1
        else:
            lo = m1
            if v2 < best_v:
                best_v = v2
                best_d = m2
        if hi - lo < 1e-9:
            break
    h[i], h[j] = h[i] + best_d, h[j] - best_d
    return best_d, best_v

def _coord_sweeps(h, n, max_sweeps, t0, deadline, tol):
