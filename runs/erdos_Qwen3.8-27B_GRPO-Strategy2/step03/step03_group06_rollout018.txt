import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _coord_slopes(h, j, n):
    n_lags = 2 * n - 1
    taus = np.arange(n_lags) - (n - 1)
    Ls = np.maximum(0, -taus)
    Us = np.minimum(n, n - taus)
    slope = np.zeros(n_lags)
    cond1 = (Ls <= j) & (j < Us) & (j + taus >= 0) & (j + taus < n)
    i1 = np.where(cond1)[0]
    if len(i1) > 0:
        slope[i1] += (1.0 - h[j + taus[i1]])
    jt = j - taus
    cond2 = (Ls <= jt) & (jt < Us) & (jt >= 0) & (jt < n)
    i2 = np.where(cond2)[0]
    if len(i2) > 0:
        slope[i2] -= h[jt[i2]]
    return slope

def _coord_descent(h, n, max_sweeps=200):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = _c5(h, dx)
    improved_total = False
    for sweep in range(max_sweeps):
        improved = False
        current_c = _corr(h)
        for j in range(n):
            slope = _coord_slopes(h, j, n)
            base = current_c - slope * h[j]
            xs = np.linspace(0, 1, 64)
            vals = np.max(base[None, :] + slope[:, None] * xs[None, :], axis=1)
            idx = np.argmin(vals)
            x_lo = xs[max(0, idx - 1)]
            x_hi = xs[min(63, idx + 1)]
            for _ in range(30):
                x_mid = 0.5 * (x_lo + x_hi)
                if np.max(base + slope * x_mid) <= np.max(base + slope * x_lo):
                    x_hi = x_mid
                else:
                    x_lo = x_mid
            x_opt = 0.5 * (x_lo + x_hi)
            if abs(x_opt - h[j]) > 1e-10:
                h[j] = x_opt
                improved = True
                current_c = _corr(h)
        h = _repair(h, n)
        c5_now = _c5(h, dx)
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            improved_total = True
        if not improved:
            break
    return h, _c5(h, dx)

def _proj_grad(h, n, deadline, n_steps=30000):
    dx = 2.0 / n
    target = n / 2.0
    best_c5 = _c5(h, dx)
    h_cur = h.copy()
    alpha = 0.01
    while time.time() < deadline:
        c = _corr(h_cur)
        m_star = np.argmax(c)
        slope = _coord_slopes(h_cur, 0, n)  # dummy to get structure
        # Compute full gradient of c(m_star)
        tau = m_star - (n - 1)
        L = max(0, -tau)
        U = min(n, n - tau)
        grad = np.zeros(n)
        js = np.arange(L, U)
        valid2 = (js + tau >= 0) & (js + tau < n)
        grad[js[valid2]] -= h_cur[js[valid2] + tau]
        jt = js - tau
        valid1 = (jt >= 0) & (jt < n) & (jt >= L) & (jt < U)
        grad[js[valid1]] += (1.0 - h_cur[jt[valid1]])
        # Also add small penalty gradient for non-max lags (active set)
        top_k = np.argsort(c)[-5:]
        for mk in top_k[1:]:
            tau_k = mk - (n - 1)
            L_k = max(0, -tau_k)
            U_k = min(n, n - tau_k)
            gk = np.zeros(n)
            js_k = np.arange(L_k, U_k)
            v2 = (js_k + tau_k >= 0) & (js_k + tau_k < n)
            gk[js_k[v2]] -= h_cur[js_k[v2] + tau_k]
            jt_k = js_k - tau_k
            v1 = (jt_k >= 0) & (jt_k < n) & (jt_k >= L_k) & (jt_k < U_k)
            gk[js_k[v1]] += (1.0 - h_cur[jt_k[v1]])
            grad += 0.3 * gk
        grad -= grad.mean()  # center for sum constraint
        h_new = h_cur - alpha * grad
        h_new = _repair(h_new, n)
        c5_new = _c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            h_cur = h_new
            alpha = min(alpha * 1.1, 0.05)
        else:
            c5_cur = _c5(h_cur, dx)
            if c5_new < c5_cur:
                h_cur = h_new
            else:
                alpha *= 0.7
            if alpha < 0.0005:
                alpha = 0.01
    return h_cur, best_c5

def _fine_pair(h, n, dx, rng, deadline, n_tries=60000):
    best_c5 = _c5(h, dx)
    h = h.copy()
    scales = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004]
    per_scale = max(300, n_tries // len(scales))
    for scale in scales:
        improved = False
        for _ in range(per_scale):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
                continue
            h[i], h[j] = ni, nj
            c5 = _c5(h, dx)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                improved = True
            else:
                h[i], h[j] = h[i] - d, h[j] + d
        if not improved and scale < 0.003:
            break
    return h, best_c5

def _slsqp_polish(h, n, alpha, maxiter=300):
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def _obj(x, a=alpha):
        c = _corr(x)
        m = np.max(c)
        s = a * (c - m)
        s = np.clip(s, -400, 400)
        return m + np.log(np.sum(np.exp(s))) / a

    def _obj_grad(x, a=alpha):
        c = _corr(x)
        m = np.max(c)
        s = a * (c - m)
        s = np.clip(s, -400, 400)
        w = np.exp(s) / np.sum(np.exp(s))
        grad = np.zeros(n)
        for mk in range(2 * n - 1):
            if w[mk] < 1e-10:
                continue
            tau = mk - (n - 1)
            L = max(0, -tau)
            U = min(n, n - tau)
            js = np.arange(L, U)
            v2 = (js + tau >= 0) & (js + tau < n)
            grad[js[v2]] -= w[mk] * x[js[v2] + tau]
            jt = js - tau
            v1 = (jt >= 0) & (jt < n) & (jt >= L) & (jt < U)
            grad[js[v1]] += w[mk] * (1.0 - x[jt[v1]])
        return grad

    eq_con = {'type': 'eq', 'fun': None, 'jac': None}
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(res.x, n)
        c5 = _c5(hc, dx)
        return hc, c5
    except Exception:
        return h, _c5(h, dx)

def _binary_seed(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def _lag_aware_perturb(h, n, rng, deadline):

    best_c5 = _c5(h, dx)
