def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    """Gradient of correlation w.r.t. h[p]."""
    lags = np.arange(-(n-1), n)
    l1 = np.arange(-p, n-p)
    t1 = lags[l1]  # not needed, just indices
    # Actually g[p] is the vector of dcorr_k/dh[p] for all k
    # dcorr_k/dh[p] = (1-h[p+k])*[0<=p+k<n] - h[p-k]*[0<=p-k<n]
    g = np.zeros(2*n-1)
    for k in range(2*n-1):
        l = k - (n-1)
        val = 0.0
        if 0 <= p + l < n:
            val += (1 - h[p+l])
        if 0 <= p - l < n:
            val -= h[p-l]
        g[k] = val
    return g

def _greedy_sweep(h, n, time_limit, max_rounds=200):
    """Greedy pair sweep: always pick the pair with the best improvement."""
    improved = False
    for _ in range(max_rounds):
        if time.time() > time_limit:
            break
        corr = _corr(h)
        cur_max = corr.max()
        best_improve = 1e-12
        best_pair = None
        best_d = 0
        best_new_corr = None
        
        for i in range(n):
            gi = _g_vec(h, i, n)
            for j in range(i+1, n):
                gj = _g_vec(h, j, n)
                gd = gi - gj
                # Find d that minimizes max(corr + d*gd)
                # subject to 0 <= h[i]+d <= 1, 0 <= h[j]-d <= 1
                d_min = max(-h[i], h[j]-1.0)
                d_max = min(1.0-h[i], h[j])
                if d_max - d_min < 1e-14:
                    continue
                # Coarse search
                ds = np.linspace(d_min, d_max, 20)
                vals = np.max(corr[None,:] + ds[:,None]*gd[None,:], axis=1)
                bi = vals.argmin()
                if vals[bi] >= cur_max - 1e-13:
                    continue
                # Bisect
                lo = max(d_min, ds[max(0,bi-1)])
                hi = min(d_max, ds[min(19,bi+1)])
                for _ in range(25):
                    mid = 0.5*(lo+hi)
                    if np.max(corr + mid*gd) <= np.max(corr + hi*gd):
                        hi = mid
                    else:
                        lo = mid
                d_best = 0.5*(lo+hi)
                new_max = np.max(corr + d_best*gd)
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_pair = (i, j)
                    best_d = d_best
                    best_new_corr = corr + d_best*gd
        
        if best_pair is None:
            break
        i, j = best_pair
        h[i] += best_d
        h[j] -= best_d
        h = np.clip(h, 0, 1)
        improved = True
    
    return h, improved

def _smooth_max_val(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return float(m + np.log(np.exp(s).sum()) / alpha)

    return _smooth_max_val(_corr(x), alpha)

def _smooth_grad(x, alpha):
    c = _corr(x)
    n = len(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    grad = np.zeros(n)
    for j in range(n):
        l1 = np.arange(-j, n - j)
        t1 = w[l1 + (n - 1)] @ (1.0 - x[j + l1])
        l2 = np.arange(j - n + 1, j + 1)
        t2 = w[l2 + (n - 1)] @ x[j - l2]

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
