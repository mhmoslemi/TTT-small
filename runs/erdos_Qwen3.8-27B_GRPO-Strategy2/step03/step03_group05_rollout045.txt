    """Create a block pattern with n_blocks alternating 1/0 regions."""
    h = np.zeros(n)
    block_size = n // n_blocks
    for i in range(n_blocks):
        start = i * block_size
        end = min((i + 1) * block_size, n)
        if i % 2 == 0:
            h[start:end] = 1.0
    return _repair(h, n)

def _optimize_active(h, n, T, dx, max_iters, t_limit):
    """Active-set projected gradient descent."""
    t0 = time.time()
    h = h.copy()
    target = n / 2.0
    c5_cur = _c5(h, dx)
    best = c5_cur
    for it in range(max_iters):
        if time.time() > t_limit:
            break
        c = _corr(h)
        top_idx = np.argsort(c)[::-1][:T]
        g = _active_grad(h, top_idx)
        # Project gradient to maintain sum constraint
        g -= g.mean()
        # Adaptive step size (backtracking)
        accepted = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            h_new = _repair(h - step * g, n)
            c5_new = _c5(h_new, dx)
            if c5_new < c5_cur - 1e-15:
                h = h_new
                c5_cur = c5_new
                accepted = True
                if c5_new < best:
                    best = c5_new
                break
        if not accepted:
            # Try smaller top set
            if T > 2:
                T = max(2, T // 2)
            else:
                # Diversify: add small random perturbation
                perturb = 0.001 * np.random.randn(n)
                h = _repair(h + perturb, n)
                c5_cur = _c5(h, dx)
                T = 5
    return h, best

def _optimize_lbfqsb(h, n, dx, alpha, lam, t_limit):
    """L-BFGS-B with soft-max objective."""
    def obj(x):
        c = _corr(x)
        m = np.max(c)
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha + lam * (np.sum(x) - n / 2.0) ** 2
    
    def grad(x):
        c = _corr(x)
        m = np.max(c)
        s = np.clip(alpha * (c - m), -500, 500)
        ew = np.exp(s)
        pw = ew / ew.sum()
        g = np.zeros(n)
        for ki in range(2 * n - 1):
            L = ki - (n - 1)
            pk = pw[ki]
            if pk < 1e-14:
                continue
            if L == 0:
                g += pk * (1.0 - 2.0 * x)
            elif L > 0:
                g[L:] += pk * (1.0 - x[:n - L])
                g[:n - L] -= pk * x[L:]
            else:
                M = -L
                g[:n - M] += pk * (1.0 - x[M:])
                g[M:] -= pk * x[:n - M]
        g *= 1.0 / alpha
        g += 2.0 * lam * (np.sum(x) - n / 2.0)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h.copy(), jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 500, 'ftol': 1e-17})
        return _repair(res.x, n)
    except:
        return h

def _polish(h, n, dx, t_limit):
    """Ultra-fine coordinate descent polish."""
    h = h.copy()
    c5_cur = _c5(h, dx)
    best = c5_cur
    rng = np.random.RandomState(123)
    steps = [0.0005, 0.001, 0.002, 0.005, -0.0005, -0.001, -0.002, -0.005]
    for sweep in range(200):
        if time.time() > t_limit:
            break
        improved = False
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                for d in steps:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c = _c5(_repair(h, n), dx)
                        if c < c5_cur - 1e-15:
                            h_i, h_j = h[i], h[j]
                            h = _repair(h, n)
                            c5_cur = c
                            improved = True
                            if c < best:
                                best = c
                        else:
                            h[i], h[j] = ni - d, nj + d
                        break
    return h, best

        g = _active_grad(h, top_idx)
        order = np.argsort(np.abs(g))[::-1][:30]
        improved = False
        for ii in range(len(order)):
            i = order[ii]
            for jj in range(ii + 1, min(ii + 25, len(order))):
                j = order[jj]
                dvals = [0.002, 0.005, 0.01, 0.02, -0.002, -0.005, -0.01, -0.02,
                         0.0005, -0.0005]
                best_d = 0.0
                best_c = c5_cur
                for d in dvals:
                    ni, nj = h[i] + d, h[j] - d
                    if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                        h[i], h[j] = ni, nj
                        hc = _repair(h, n)
                        c = _c5(hc, dx)
                        if c < best_c:
                            best_c = c
                            best_d = d
                        h[i], h[j] = ni - d, nj + d
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    h = _repair(h, n)
                    c5_cur = _c5(h, dx)
                    improved = True
                    if c5_cur < best_c5:
                        best_c5 = c5_cur
                        best_h = h.copy()
                    break
            if time.time() > t_limit:
                break
        if not improved and sweep > 20:
            break
    return best_h, best_c5

def _fine_polish(h, n, dx, t_limit):
    h = h.copy()
    c5_cur = _c5(h, dx)
    best_c5 = c5_cur
    best_h = h.copy()
    for sweep in range(500):
        if time.time() > t_limit:
            break
        c = _corr(h)
        top_idx = np.argsort(c)[::-1][:2]
        g = _active_grad(h, top_idx)
        g -= g.mean()
        order = np.argsort(np.abs(g))[::-1][:20]
        improved = False
        for i in order:
            for d in [0.0003, 0.0007, 0.0015, -0.0003, -0.0007, -0.0015,
                      0.0001, -0.0001]:
                ni = h[i] + d
                if 0.0 <= ni <= 1.0:
                    h[i] = ni
                    hc = _repair(h, n)
                    c_val = _c5(hc, dx)
                    if c_val < c5_cur - 1e-15:
                        h = hc
                        c5_cur = c_val
                        improved = True
                        if c5_cur < best_c5:
                            best_c5 = c5_cur
                            best_h = h.copy()
                    else:
                        h[i] = ni - d
                    if improved:
                        break
        if not improved:
            break
    return best_h, best_c5

def _make_block_pattern(n, n_blocks):
    h = np.zeros(n)
    block = n // n_blocks
    for i in range(n_blocks):
        s = i * block
        e = min(s + block, n)
        if i % 2 == 0:
            h[s:e] = 1.0
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = 1.0
    best_n = 0

    def _rem():
        return budget_s - (time.time() - t0)

    n_list = [70]
    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        pass
    else:
        n_list = [64, 70, 72, 80, 84, 90, 96]

    # Primary: n=70 from initial_h_values
    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        n = 70
        dx = 2.0 / n
        try:
            h0 = _repair(np.array(initial_h_values, dtype=float), n)
        except Exception:
            h0 = _make_block_pattern(n, 2)
        c5_0 = _c5(h0, dx)
        best_h = h0.copy()
        best_c5 = c5_0
        best_n = n

        # Phase 1: L-BFGS-B to polish (short)
        if _rem() > 150:
            h1 = _lbfqsb(h0, n, dx, 2000, 30.0, t0 + min(80, _rem() - 100))
            c5_1 = _c5(h1, dx)
            if c5_1 < best_c5:
                best_c5 = c5_1
                best_h = h1.copy()
            h1 = _lbfqsb(h1, n, dx, 8000, 10.0, t0 + min(140, _rem() - 100))
            c5_1 = _c5(h1, dx)
            if c5_1 < best_c5:
                best_c5 = c5_1
                best_h = h1.copy()

        # Phase 2: Active-set optimization (main work)
        if _rem() > 100:
            h2, c5_2 = _active_opt(best_h, n, dx, t0 + min(350, _rem() - 80))
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()

        # Phase 3: Coordinate descent
        if _rem() > 80:
            h3, c5_3 = _coord_desc(best_h, n, dx, t0 + min(250, _rem() - 60))
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()

        # Phase 4: Fine polish
        if _rem() > 40:
            h4, c5_4 = _fine_polish(best_h, n, dx, t0 + min(120, _rem() - 20))
            if c5_4 < best_c5:
                best_c5 = c5_4
                best_h = h4.copy()

    # Secondary: try other n values with shorter budgets
    for n in [64, 72, 80, 84, 96]:
        if _rem() < 60:
            break
        dx = 2.0 / n
        cands = [_make_block_pattern(n, 2), _make_block_pattern(n, 3),
                 _make_block_pattern(n, 4)]
        cands.append(_repair(rng.uniform(0, 1, n), n))
        x_lin = np.linspace(0, 2, n)
        cands.append(_repair(np.clip(0.5 + 0.35 * np.sin(2 * np.pi * x_lin), 0, 1), n))
        for h_init in cands:
            if _rem() < 40:
                break
            h1 = _lbfqsb(h_init, n, dx, 5000, 30.0, t0 + min(25, _rem() - 20))
            h2, c5_2 = _active_opt(h1, n, dx, t0 + min(20, _rem() - 15))
            h3, c5_3 = _coord_desc(h2, n, dx, t0 + min(10, _rem() - 5))
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
                best_n = n

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
