                    n = n2
            except Exception:
                pass
        if n == n2 and elapsed() < budget_s - 15:
            best_h, best_c5 = fine_search(best_h.copy(), n, t0 + budget_s - 5, rng)

    best_h = repair(best_h, n)
    best_c5 = c5_val(best_h, n)
    return (best_h, best_c5, n)
