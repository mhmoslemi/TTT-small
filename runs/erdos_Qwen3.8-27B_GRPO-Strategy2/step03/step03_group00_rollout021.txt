import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h))) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _so(x, n, alpha):
    c = _corr(x) * (2.0 / n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sg(x, n, alpha):
    dxv = 2.0 / n
    c = _corr(x) * dxv
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    grad = np.zeros(n)
    for i in range(n):
        lo = i
        hi = min(2 * n - 2, i + n - 1)
        grad[i] = dxv * np.dot(w[lo:hi + 1], 1.0 - 2.0 * x[lo - i:hi - i + 1])
    return grad

def _slsqp_wrap(h_init, n, alpha, maxiter, t_end):
    bounds = [(0.0, 1.0)] * n
    tgt = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - tgt, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_so, h_init, jac=_sg, args=(n, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h_init

def _interp(h_old, n_old, n_new):
    x_old = np.arange(n_old) / n_old * 2.0
    x_new = np.arange(n_new) / n_new * 2.0
    # Periodic interpolation (wrap)
    xo = np.concatenate([x_old - 2.0, x_old, x_old + 2.0])
    ho = np.concatenate([h_old, h_old, h_old])
    return np.interp(x_new, xo, ho)

def _pair_cd(h, n, t_end, rng, max_sweeps=500):
    c = _corr(h).copy()
    c5_cur = float(np.max(c)) * (2.0 / n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = 0
        idx = rng.permutation(n)
        for a in range(n):
            for b in range(a + 1, n):
                if time.time() > t_end:
                    break
                i, j = idx[a], idx[b]
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], 1.0 - h[j], h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                m_lo = min(i, j)
                m_hi = min(2 * n - 2, max(i, j) + n - 1)
                B = np.zeros(m_hi - m_lo + 1)
                lo_i = max(m_lo, i); hi_i = min(m_hi, i + n - 1)
                if hi_i >= lo_i:
                    B[lo_i - m_lo:hi_i - m_lo + 1] += 1.0 - 2.0 * h[lo_i - i:hi_i - i + 1]
                lo_j = max(m_lo, j); hi_j = min(m_hi, j + n - 1)
                if hi_j >= lo_j:
                    B[lo_j - m_lo:hi_j - m_lo + 1] -= 1.0 - 2.0 * h[lo_j - j:hi_j - j + 1]
                cs = c[m_lo:m_hi + 1]
                dgrid = np.concatenate([[0.0, d_lo, d_hi],
                                        np.linspace(d_lo, d_hi, 6)[1:-1]])
                fv = np.max(cs[None, :] + np.outer(dgrid, B), axis=1)
                bi = np.argmin(fv)
                if fv[0] - fv[bi] < 1e-13:
                    continue
                d_val = dgrid[bi]
                h[i] += d_val
                h[j] -= d_val
                c[m_lo:m_hi + 1] += d_val * B
                new_c5 = float(np.max(c)) * (2.0 / n)
                if new_c5 < c5_cur:
                    c5_cur = new_c5
                    imp += 1
                else:
                    h[i] -= d_val
                    h[j] += d_val
                    c[m_lo:m_hi + 1] -= d_val * B
        if imp == 0:
            break
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float).copy()
    n0 = len(h0)
    best_h = _repair(h0, n0)
    best_c5 = _c5(best_h, n0)

    # Phase 1: SLSQP at initial resolution
    for alpha in [5000, 20000]:
        for r in range(3):
            if time.time() > t_end - 400:
                break
            hi = best_h.copy() if r == 0 else _repair(best_h + 0.04 * rng.randn(n0), n0)
            hc = _slsqp_wrap(hi, n0, alpha, 400, t_end - 350)
            cc = _c5(hc, n0)
            if cc < best_c5:
                best_c5 = cc
                best_h = hc.copy()

    # Phase 2: Interpolate to n=140, optimize
    n1 = 140
    h1 = _repair(_interp(best_h, n0, n1), n1)
    c5_1 = _c5(h1, n1)
    best_h1 = h1.copy()
    best_c5_1 = c5_1

    for alpha in [10000, 30000, 80000]:
        for r in range(3):
            if time.time() > t_end - 250:
                break
            hi = best_h1.copy() if r == 0 else _repair(best_h1 + 0.03 * rng.randn(n1), n1)
            hc = _slsqp_wrap(hi, n1, alpha, 600, t_end - 200)
            cc = _c5(hc, n1)
            if cc < best_c5_1:
                best_c5_1 = cc
                best_h1 = hc.copy()

    # Phase 3: Pair coordinate descent at n=140
    h_cd = best_h1.copy()
    h_cd, c5_cd = _pair_cd(h_cd, n1, t_end - 80, rng, max_sweeps=800)
    if c5_cd < best_c5_1:
        best_c5_1 = c5_cd
        best_h1 = h_cd.copy()

    # Phase 4: Try n=200 if time permits
    if time.time() < t_end - 120:
        n2 = 200
        h2 = _repair(_interp(best_h1, n1, n2), n2)
        c5_2 = _c5(h2, n2)
        best_h2 = h2.copy()
        best_c5_2 = c5_2
        for alpha in [20000, 50000]:
            if time.time() > t_end - 60:
                break
            hi = best_h2.copy()
            hc = _slsqp_wrap(hi, n2, alpha, 400, t_end - 50)
            cc = _c5(hc, n2)
            if cc < best_c5_2:
                best_c5_2 = cc
                best_h2 = hc.copy()
        if time.time() < t_end - 40:
            h2c, c5_2c = _pair_cd(best_h2, n2, t_end - 20, rng, max_sweeps=400)
            if c5_2c < best_c5_2:
                best_c5_2 = c5_2c
                best_h2 = h2c.copy()
        # Compare with n1 solution
        if best_c5_2 < best_c5_1:
            best_h1 = best_h2
            best_c5_1 = best_c5_2
            n1 = n2

    # Phase 5: Final hill-climbing at n1
    if time.time() < t_end - 10:
        hf = best_h1.copy()
        c5f = _c5(hf, n1)
        c = _corr(hf).copy()
        steps = np.array([0.0005, 0.001, 0.002, 0.005, 0.01])
        while time.time() < t_end - 5:
            i = rng.randint(n1)
            j = rng.randint(n1)
            if i == j:
                continue
            d = steps[rng.randint(len(steps))] * (2.0 if rng.rand() < 0.5 else -1.0)
            d_lo = max(-hf[i], hf[j] - 1.0)
            d_hi = min(1.0 - hf[i], 1.0 - hf[j], hf[j])
            d = np.clip(d, d_lo, d_hi)
            if abs(d) < 1e-12:
                continue
            m_lo = min(i, j)
            m_hi = min(2 * n1 - 2, max(i, j) + n1 - 1)
            B = np.zeros(m_hi - m_lo + 1)
            lo_i = max(m_lo, i); hi_i = min(m_hi, i + n1 - 1)
            if hi_i >= lo_i:
                B[lo_i - m_lo:hi_i - m_lo + 1] += 1.0 - 2.0 * hf[lo_i - i:hi_i - i + 1]
            lo_j = max(m_lo, j); hi_j = min(m_hi, j + n1 - 1)
            if hi_j >= lo_j:
                B[lo_j - m_lo:hi_j - m_lo + 1] -= 1.0 - 2.0 * hf[lo_j - j:hi_j - j + 1]
            new_max = float(np.max(np.concatenate([c[:m_lo],
                                                    c[m_lo:m_hi + 1] + d * B,
                                                    c[m_hi + 1:]])))
            new_c5 = new_max * (2.0 / n1)
            if new_c5 < c5f - 1e-15:
                hf[i] += d
                hf[j] -= d
                c[m_lo:m_hi + 1] += d * B
                c5f = new_c5
                if c5f < best_c5_1:
                    best_c5_1 = c5f
                    best_h1 = hf.copy()
            else:
                # Revert not needed since c wasn't modified

    best_h1 = _repair(best_h1, n1)
    final_c5 = _c5(best_h1, n1)
    return (best_h1, final_c5, n1)
