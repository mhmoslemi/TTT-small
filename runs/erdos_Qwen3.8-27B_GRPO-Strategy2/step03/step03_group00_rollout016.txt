import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h))) * (2.0 / n)

def _single_coord_grad(h, i, n):
    """Compute ∂c[k]/∂h[i] * dx for all k (length 2n-1)."""
    dx = 2.0 / n
    grad = np.zeros(2 * n - 1)
    for k_full in range(2 * n - 1):
        lag = k_full - (n - 1)
        # First: i appears as h[p+lag], p = i-lag
        p1 = i - lag
        if 0 <= p1 < n:
            grad[k_full] += (1.0 - h[p1])
        # Second: i appears as h[p], p = i, other is h[i+lag]
        p2 = i + lag
        if 0 <= p2 < n:
            grad[k_full] -= h[p2]
    return grad * dx

def _min_1d(c, grad, old):
    """Minimize max_k (c[k] + (t-old)*grad[k]) over t in [0,1]."""
    # Ternary search
    lo, hi = 0.0, 1.0
    best_t = old
    best_f = float('inf')
    for _ in range(30):

        tm = (lo + hi) / 2.0
        fm = np.max(c + (tm - old) * grad)
        if fm < best_f:
            best_f = fm
            best_t = tm
    return best_t, best_f

    dx = 2.0 / n
    grad = np.zeros(2 * n - 1)
    for k_full in range(2 * n - 1):
        lag = k_full - (n - 1)
        p1 = i - lag
        if 0 <= p1 < n:
            grad[k_full] += (1.0 - h[p1])
        p2 = i + lag
        if 0 <= p2 < n:
            grad[k_full] -= h[p2]
    return grad * dx

def _min_1d(c, grad, old):
    lo, hi = 0.0, 1.0
    best_t = old
    best_f = float(np.max(c))
    if np.all(np.abs(grad) < 1e-15):
        return best_t, best_f
    for _ in range(35):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(c + (m1 - old) * grad)
        f2 = np.max(c + (m2 - old) * grad)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
        tm = (lo + hi) / 2.0
        fm = np.max(c + (tm - old) * grad)
        if fm < best_f:
            best_f = fm
            best_t = tm
    return best_t, best_f

def _coord_descent(h, n, max_sweeps=500, t_limit=None):
    c = _corr(h) * (2.0 / n)
    best_c5 = float(np.max(c))
    best_h = h.copy()
    for sweep in range(max_sweeps):
        if t_limit is not None and time.time() > t_limit:
            break
        improved = False
        for i in range(n):
            old = h[i]
            grad = _single_coord_grad(h, i, n)
            best_t, _ = _min_1d(c, grad, old)
            delta = best_t - old
            if abs(delta) > 1e-14:
                h[i] = best_t
                c += delta * grad
                cur = float(np.max(c))
                if cur < best_c5 - 1e-13:
                    best_c5 = cur
                    best_h = h.copy()
                    improved = True
                elif cur > best_c5 + 1e-13:
                    c = _corr(h) * (2.0 / n)
        if not improved and sweep > 10:
            break
    return best_h, best_c5

def _anneal(h, n, t_limit, rng):
    c5_cur = _c5(h, n)
    best_h = h.copy()
    best_c5 = c5_cur
    T = 0.00015
    min_T = 1e-11
    cool = 0.99985
    steps = np.array([0.0005, 0.002, 0.005, 0.015, 0.04])
    while time.time() < t_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = steps[rng.randint(len(steps))] * (1.0 if rng.rand() < 0.5 else -1.0)
        hi, hj = h[i], h[j]
        ni, nj = hi + d, hj - d
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
            continue
        h[i], h[j] = ni, nj
        c5_new = _c5(h, n)
        if c5_new < c5_cur or (T > 1e-11 and rng.rand() < np.exp(-(c5_new - c5_cur) / T)):
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            h[i], h[j] = hi, hj
        T = max(T * cool, min_T)
    return best_h, best_c5

def _fine_pair_search(h, n, t_limit, rng):
    c5_cur = _c5(h, n)
    best_h = h.copy()
    best_c5 = c5_cur
    deltas = np.array([0.0003, 0.001, 0.004, 0.012, -0.0003, -0.001, -0.004, -0.012])
    for _pass in range(200):
        if time.time() > t_limit:
            break
        improved = False
        for i in range(n):
            if time.time() > t_limit:
                break
            for j in range(i + 1, n):
                if time.time() > t_limit:
                    break
                hi, hj = h[i], h[j]
                for d in deltas:
                    ni, nj = hi + d, hj - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, n)
                    if c < c5_cur - 1e-14:
                        c5_cur = c
                        hi, hj = ni, nj
                        improved = True
                    else:
                        h[i], h[j] = hi, hj
                h[i], h[j] = hi, hj
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5(best_h, n)
    end_t = t0 + budget_s

    # Phase 1: Exact coordinate descent from initial
    limit = min(end_t - 200, t0 + 120)
    best_h, best_c5 = _coord_descent(best_h, n, max_sweeps=800, t_limit=limit)
    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)

    # Phase 2: Multiple perturbation + coordinate descent
    for attempt in range(80):
        if time.time() > end_t - 180:
            break
        scale = 0.005 + 0.02 * rng.rand()
        h2 = best_h + scale * rng.randn(n)
        h2 = _repair(h2, n)
        limit2 = min(end_t - 150, time.time() + 8)
        h2, c5_2 = _coord_descent(h2, n, max_sweeps=100, t_limit=limit2)
        h2 = _repair(h2, n)
        c5_2 = _c5(h2, n)
        if c5_2 < best_c5 - 1e-12:
            best_h = h2.copy()
            best_c5 = c5_2

    # Phase 3: Larger perturbations to explore landscape
    for attempt in range(20):
        if time.time() > end_t - 120:
            break
        h2 = best_h + 0.05 * rng.randn(n)
        h2 = _repair(h2, n)
        limit3 = min(end_t - 100, time.time() + 6)
        h2, c5_2 = _coord_descent(h2, n, max_sweeps=80, t_limit=limit3)
        h2 = _repair(h2, n)
        c5_2 = _c5(h2, n)
        if c5_2 < best_c5 - 1e-12:
            best_h = h2.copy()
            best_c5 = c5_2

    # Phase 4: Simulated annealing
    if time.time() < end_t - 80:
        limit4 = min(end_t - 50, time.time() + 120)
        best_h, best_c5 = _anneal(best_h, n, limit4, rng)
        best_h = _repair(best_h, n)
        best_c5 = _c5(best_h, n)

    # Phase 5: Final exact coordinate descent
    if time.time() < end_t - 30:
        limit5 = min(end_t - 15, time.time() + 60)
        best_h, best_c5 = _coord_descent(best_h, n, max_sweeps=500, t_limit=limit5)
        best_h = _repair(best_h, n)
        best_c5 = _c5(best_h, n)

    # Phase 6: Fine pair search for last-mile improvement
    if time.time() < end_t - 10:
        limit6 = end_t - 5
        best_h, best_c5 = _fine_pair_search(best_h, n, limit6, rng)
        best_h = _repair(best_h, n)
        best_c5 = _c5(best_h, n)

    return (best_h, best_c5, n)
