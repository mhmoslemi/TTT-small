def _grad_prefix(h, top_k):

    n = len(h)
    c = _corr(h)
    idx = np.argsort(c)[::-1][:top_k]

    ct = c[idx]

import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _softobj(x, alpha, lam):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-12:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _proj(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
    for idx in order:
        if diff > 0:
            a = min(diff, 1.0 - h[idx])
        else:
            a = max(diff, -h[idx])
        h[idx] += a
        diff -= a
        if abs(diff) < 1e-14:
            break
    return h

def _topK_grad(h, K):

    c = _corr(h)
    idx = np.argsort(c)[::-1][:K]

    g = np.zeros(n)

        else:
            M = -L
            g[:n - M] += wt * (1.0 - h[M:])
            g[M:] -= wt * h[:n - M]
    g -= g.mean()
    return g, c.max() * (2.0 / n)
