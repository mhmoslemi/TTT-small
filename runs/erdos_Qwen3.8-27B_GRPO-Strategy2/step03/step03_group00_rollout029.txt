The objective is: f(h) = log-sum-exp(alpha * (corr(h) - m)) + m where m = max(corr(h))

This is C² smooth and its gradient is computable.

The constraint is: h ∈ [0,1]^n, sum(h) = n/2

```python
def project(h, n):
    target = n / 2.0
    # Binary search for shift to satisfy sum constraint, with clipping
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0, 1)
    # Fix any tiny residual
    diff = target - h_proj.sum()
    if abs(diff) > 1e-12:
        # Add/subtract from entries with most room
        if diff > 0:
            room = 1 - h_proj
        else:
            room = h_proj
        order = np.argsort(room)  # largest room first (or smallest h for removing)
        if diff > 0:
            order = np.argsort(-room)
        else:
            order = np.argsort(room)
        for idx in order:
            if diff > 0:
                add = min(diff, 1 - h_proj[idx])
                h_proj[idx] += add
                diff -= add
            else:
                sub = min(-diff, h_proj[idx])
                h_proj[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h_proj
```

    return np.correlate(h, 1.0 - h, mode='full')

    dx = 2.0 / n
    return float(np.max(corr)) * dx

def _c5(h, n):
    return _c5_from_corr(_corr_full(h), n)

def _project(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = h.copy()
    # Binary search for the right shift
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - hp)
        else:
            order = np.argsort(hp)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - hp[idx])
                hp[idx] += a
                diff -= a
            else:
                a = min(-diff, hp[idx])
                hp[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return hp

def _smooth_obj_and_grad(h, n, alpha):
    """Compute smooth max objective and gradient."""
    corr = _corr_full(h)
    dx = 2.0 / n
    c = corr * dx
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    e = np.exp(s)
    total = e.sum()
    w = e / total
    obj = m + np.log(total) / alpha
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        L = abs(k - (n - 1))
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            # (1-h[j+L]) for j = 0..n-1-L

            # -h[j-L] for j = L..n-1
            for j in range(L, n):
                grad[j] -= wk * h[j - L]
    grad *= dx
    return obj, grad

import time

def _c5_vec(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Scale and re-clip
    if s > 0:
        h *= target / s
    h = np.clip(h, 0.0, 1.0)
    # Fix residual with targeted adjustments
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
            diff -= a if diff > 0 else -a
            if abs(diff) < 1e-14:
                break
    return h

def _grad_smooth(h, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c = corr * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    total = e.sum()

        if L == 0:
            grad += wk * (1.0 - 2.0*h)
        else:
            grad[:n-L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:n-L]
    return grad * dx, float(m + np.log(total) / alpha)

    best_c5 = _c5_vec(h, n)
    best_h = h.copy()
    for p in range(max_passes):
        if time.time() > budget_end:
            break
        improved = False
        for i in range(n):
            if time.time() > budget_end:
                break
            for j in range(i+1, n):
                # Find optimal d for(h[i]+=d, h[j]-=d)
                d_lo = max(-h[i], h[j]-1.0)
                d_hi = min(1.0-h[i], h[j])

                    best_h = h.copy()
                    improved = True
                else:
                    h[i], h[j] = h[i]-d_opt, h[j]+d_opt
        if not improved:
            break
    return best_h, best_c5
