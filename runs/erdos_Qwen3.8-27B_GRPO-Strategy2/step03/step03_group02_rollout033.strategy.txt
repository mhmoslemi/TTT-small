                              options={'maxiter': int(phase_time/15)} , tol=1e-16)
               hc = repair(res.x, target_sum)
               hc_c5 = _obj_smax(hc, 130)
               if hc_c5 < best_c5:
                   best_h, best_c5 = hc, hc_c5
           except:  # Budget tight
               repair_step = best_h * (1 + np.random.randn(n) * 0.001)
               repaired = repair(repair_step, target_sum)
               try:
                   hc = minimize(_obj_smax, repaired, args=(90,), method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': lambda x: _eq_con(x, target_sum)}, tol=1e-16)
                   hc_c5 = _obj_smax(hc.x, 120)
                   if hc_c5 < best_c5: best_c5, best_h = hc_c5, hc.x
               except: pass

           # Monte Carlo perturbation
           # Sample complete restarts for distributivity
           for mc in range(int(phase_time/60)):
               h_restart = np.random.rand(n)
               final_sum = h_restart.sum()
               h_restart = repair(h_restart*target_sum/final_sum, target_sum)
               c5_restart = _obj_smax(h_restart, 95)
               if c5_restart < best_c5:
                   best_c5, best_h = c5_restart, h_restart.copy()

       return best_h, best_c5*2.0/n, n   # Return dx-adjusted bound
