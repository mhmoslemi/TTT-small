    h = np.clip(h.astype(float), 0.0, 1.0).copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    return float(np.max(compute_corr(h)) * (2.0 / n))

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        d = target - s
        if abs(d) < 1e-13:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-13:
                break
            a = min(abs(d), (1.0 - h[idx]) if d > 0 else h[idx])
            h[idx] += a if d > 0 else -a
            d -= a if d > 0 else -a
    return h

def soft_max(corr, alpha):
    m = np.max(corr)
    c = np.clip(alpha * (corr - m), -500, 500)
    e = np.exp(c)
    return m + np.log(e.sum() / len(corr)) / alpha, e / e.sum()

def corr_grad(h, n, weights):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-10:
            continue
        mk = k - (n - 1)
        if mk == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            j_lo = max(0, -mk)
            j_hi = min(n - 1, n - 1 - mk)
            js = np.arange(j_lo, j_hi + 1)
            jms = js + mk
            grad[jms] += wk * (1.0 - h[js])
            grad[js] += wk * (-h[jms])
    return grad * dx

def _slsqp_obj(h, n, alpha):
    corr = compute_corr(h)
    val, _ = soft_max(corr, alpha)
    return val

def _slsqp_grad(h, n, alpha):
    corr = compute_corr(h)
    _, w = soft_max(corr, alpha)
    return corr_grad(h, n, w)

def _pen_obj(h, n, alpha, pen):
    corr = compute_corr(h)
    val, _ = soft_max(corr, alpha)
    s = h.sum() - n / 2.0
    return val + pen * s * s

def _pen_grad(h, n, alpha, pen):
    corr = compute_corr(h)
    _, w = soft_max(corr, alpha)
    g = corr_grad(h, n, w)
    s = h.sum() - n / 2.0
    g += 2.0 * pen * s
    return g

def _make_pen_funcs(n, alpha, pen):
    def fo(h):
        return _pen_obj(h, n, alpha, pen)
    def go(h):
        return _pen_grad(h, n, alpha, pen)
    return fo, go

def _make_plain_funcs(n, alpha):
    def fo(h):
        return _slsqp_obj(h, n, alpha)
    def go(h):
        return _slsqp_grad(h, n, alpha)
    return fo, go

def fourier_start(n, M, rng):
    coeffs = rng.randn(2 * M + 1) * 0.25
    angles = 2.0 * np.pi * np.arange(n) / n
    h = np.full(n, 0.5) + coeffs[0] * 0.5
    for k in range(1, M + 1):
        h += coeffs[2 * k - 1] * np.cos(k * angles)
        h += coeffs[2 * k] * np.sin(k * angles)
    return repair(h, n)

def block_start(n, rng, nblocks):
    h = np.zeros(n)
    bs = max(1, n // nblocks)
    for i in range(nblocks):
        lo = i * bs
        hi = min(n, (i + 1) * bs)
        h[lo:hi] = 0.75 if i % 2 == 0 else 0.25
    h += 0.04 * rng.randn(n)
    return repair(h, n)

def symm_start(n, rng):
    half = n // 2
    h_half = 0.5 + 0.3 * rng.randn(half)
    h = np.zeros(n)
    h[:half] = h_half
    h[half:] = 1.0 - h_half[::-1]
    return repair(h, n)

def lbfgs_run(h, n, alpha, pen, maxiter, t_end):
    bounds = [(0.0, 1.0)] * n
    if pen > 0:
        fo, go = _make_pen_funcs(n, alpha, pen)
    else:
        fo, go = _make_plain_funcs(n, alpha)
    try:
        res = minimize(fo, h, jac=go, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return repair(res.x, n)
    except Exception:
        return h

def active_set_refine(h, n, n_active, alpha, maxiter, t_end):
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[::-1][:n_active]
    top_val = corr[top_idx].max()
    for _ in range(maxiter):
        if time.time() > t_end:
            break
        w = np.zeros(2 * n - 1)
        w[top_idx] = 1.0 / n_active
        def _ao(hh):
            c = compute_corr(hh)
            return np.dot(w, c) + 100.0 * (hh.sum() - n / 2.0) ** 2
        def _ag(hh):
            c = compute_corr(hh)
            g = corr_grad(hh, n, w)
            g += 200.0 * (hh.sum() - n / 2.0)
            return g
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(_ao, h, jac=_ag, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 200, 'ftol': 1e-16})
            h_new = repair(res.x, n)
        except Exception:
            h_new = h
        c5_new = compute_c5(h_new, n)
        c5_old = compute_c5(h, n)
        if c5_new < c5_old - 1e-14:
            h = h_new
        else:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[::-1][:n_active]
    return h

def coord_descent(h, n, n_passes, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for p in range(n_passes):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(4 * n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for d in steps:
                for sign in (1, -1):
                    delta = sign * d
                    ni, nj = hi + delta, hj - delta
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-15:
                        best = c5
                        improved = True
                        break
                    h[i], h[j] = hi, hj
                if improved:
                    break
        if not improved and p > 40:
            break
    return h, best
