Let me also think about the pair swap optimization more carefully:

For a swap h[i] += δ, h[j] -= δ, the change in full[p] is:
Δfull[p] = δ * (∂full[p]/∂h[i] - ∂full[p]/∂h[j])

If we want to reduce the max, we want Δfull[p] < 0 for the active p's.

For a given pair (i,j) and direction δ, the max over active p of (Δfull[p]) should be negative. We can find the optimal δ by looking at the constraint that the largest active lag doesn't increase.

This is essentially a line search problem: find δ that minimizes max_p full[p](h + δ*(e_i - e_j)) subject to 0 ≤ h ≤ 1.

Since full[p] is linear in δ (for fixed i,j), this is:
minimize max_p (full[p] + δ * g_p)
where g_p = ∂full[p]/∂h[i] - ∂full[p]/∂h[j]

The optimal δ is where the top two lines cross. This can be found in O(k log k) time where k is the number of active lags.

This is a very efficient local search! Let me implement it.

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def corr_grad_all(h, n):
    """Compute gradients for all lags. Returns array of shape (2n-1, n)."""
    nfull = 2 * n - 1
    grads = np.zeros((nfull, n))
    for p in range(nfull):
        j = p - n + 1
        for m in range(n):
            if 0 <= m - j < n:
                grads[p, m] += (1.0 - h[m - j])
            if 0 <= m + j < n:
                grads[p, m] -= h[m + j]
    return grads

def active_set_lp(h, n, max_iters=100, t_end=None, verbose=False):
    """Active-set LP optimization."""
    target = n / 2.0
    h = h.copy()
    best_c5 = compute_c5(h, n)
    nfull = 2 * n - 1
    
    for it in range(max_iters):
        if t_end and time.time() > t_end:
            break
        
        corr = compute_corr(h)
        grads = corr_grad_all(h, n)
        
        # LP: min t s.t. grads[p,:] · h - t <= corr[p] for all p
        #     sum(h) = target, 0<=h<=1, t>=0
        A_ub = np.column_stack([grads, -np.ones(nfull)])
        b_ub = corr.copy()
        
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            break
        
        h_new = repair(res.x[:n], n)
        c5_new = compute_c5(h_new, n)
        
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            h = h_new
        else:
            # Try perturbation to escape
            rng_local = np.random.RandomState(it * 7 + 3)
            h = repair(h + 0.005 * rng_local.randn(n), n)
            # Re-loop
            continue
    
    return h, best_c5

def pair_swap_refine(h, n, n_passes=50, t_end=None, rng=None):
    """Pair swap local search with analytical line search."""
    h = h.copy()
    best_c5 = compute_c5(h, n)
    nfull = 2 * n - 1
    corr = compute_corr(h)
    grads = corr_grad_all(h, n)
    
    for path in range(n_passes):
        if t_end and time.time() > t_end:
            break
        improved = False
        for _ in range(3 * n):
            if t_end and time.time() > t_end:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            # Line search: find best delta
            # full[p] changes by delta * (grads[p,i] - grads[p,j])
            d = grads[:, i] - grads[:, j]
            # Corr changes: corr_new[p] = corr[p] + delta * d[p]
            # We want to minimize max(p) corr_new[p]
            # subject to 0<=h[i]+delta<=1 and 0<=h[j]-delta<=1
            lo = max(0.0, h[j] - (1.0 - h[i]) if False else -h[i], -(1.0 - h[j]))
            hi = min(1.0 - h[i], h[j])
            # Actually: h[i]+δ ∈ [0,1] → δ ∈ [-h[i], 1-h[i]]
            #            h[j]-δ ∈ [0,1] → δ ∈ [h[j]-1, h[j]]
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi - lo < 1e-12:
                continue
            
            # Minimize max_p (corr[p] + δ*d[p]) over δ ∈ [lo, hi]
            # The max is a piecewise-linear function of δ.
            # Find vertices where the argmax changes.
            # For simplicity, check a few candidate δ values:
            candidates = [lo, hi, 0.0]
            # Add the points where two top corrs cross
            top_lags = np.argsort(corr)[::-1][:5]
            for a_lag in top_lags:
                for b_lag in top_lags:
                    if a_lag == b_lag:
                        continue
                    denom = d[b_lag] - d[a_lag]
                    if abs(denom) > 1e-12:
                        delta_cross = -(corr[a_lag] - corr[b_lag]) / denom
                        if lo < delta_cross < hi:
                            candidates.append(delta_cross)
            
            best_delta = 0.0
            best_val = np.inf
            for dc in candidates:
                dc = np.clip(dc, lo, hi)
                new_corr = corr + dc * d
                val = np.max(new_corr)
                if val < best_val - 1e-15:
                    best_val = val
                    best_delta = dc
            
            if best_delta != 0.0:
                h[i] += best_delta
                h[j] -= best_delta
                h = np.clip(h, 0.0, 1.0)
                # Recompute corr and grads
                corr = compute_corr(h)
                grads = corr_grad_all(h, n)
                c5 = float(np.max(corr) * 2.0 / n)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    improved = True
        if not improved and path > 10:
            break
    
    return h, best_c5

def slsqp_pass(h, n, alpha, maxiter, t_end=None):
    """SLSQP with log-sum-exp smoothing."""
    target = n / 2.0
    
    def obj(hv):
        corr = compute_corr(hv)
        m = np.max(corr)
        s = alpha * (corr - m)
        s = np.clip(s, -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    
    def eq_con(hv):
        return hv.sum() - target
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con}]
    try:
        if t_end and time.time() > t_end:
            return h
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return repair(res.x, n)
    except Exception:
        return h

def make_random_start(n, seed_val, kind=0):
    rng = np.random.RandomState(seed_val)
    if kind == 0:
        h = 0.5 + 0.3 * rng.randn(n)
    elif kind == 1:
        h = np.where(np.arange(n) < n // 2, 1.0, 0.0) + 0.1 * rng.randn(n)
    elif kind == 2:
        f = rng.randint(2, 8)
        ph = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.35 * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 3:
        h = 0.5 + 0.3 * np.cos(np.pi * np.arange(n) / n) ** 2
    else:
        h = rng.uniform(0.2, 0.8, n)
    return repair(h, n)

    best_c5 = np.inf
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

_state = {}

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def corr_grad_all(h, n):
    nfull = 2 * n - 1
    grads = np.zeros((nfull, n))
    for p in range(nfull):
        j = p - n + 1
        for m in range(n):
            if 0 <= m - j < n:
                grads[p, m] += (1.0 - h[m - j])
            if 0 <= m + j < n:
                grads[p, m] -= h[m + j]
    return grads

def _slsqp_obj(hv):
    corr = compute_corr(hv)
    m = np.max(corr)
    a = _state['alpha']
    s = np.clip(a * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a

def _slsqp_con(hv):
    return hv.sum() - _state['target']

def slsqp_pass(h, n, alpha, maxiter, t_end=None):
    _state['alpha'] = alpha
    _state['target'] = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con}]
    if t_end and time.time() > t_end:
        return h
    try:
        res = minimize(_slsqp_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return repair(res.x, n)
    except Exception:
        return h

def active_set_lp(h, n, max_iters, t_end):
    target = n / 2.0
    h = h.copy()
    best_c5 = compute_c5(h, n)
    nfull = 2 * n - 1
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    for it in range(max_iters):
        if time.time() > t_end:
            break
        corr = compute_corr(h)
        grads = corr_grad_all(h, n)
        A_ub = np.column_stack([grads, -np.ones(nfull)])
        b_ub = corr.copy()
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            break
        h_new = repair(res.x[:n], n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            h = h_new
        else:
            rng_loc = np.random.RandomState(it * 13 + 7)
            h = repair(h + 0.004 * rng_loc.randn(n), n)
    return h, best_c5

def pair_swap_refine(h, n, n_passes, t_end, rng):
    h = h.copy()
    best_c5 = compute_c5(h, n)
    nfull = 2 * n - 1
    corr = compute_corr(h)
    grads = corr_grad_all(h, n)
    for path in range(n_passes):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(3 * n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = grads[:, i] - grads[:, j]
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi - lo < 1e-12:
                continue
            candidates = [lo, hi, 0.0]
            top_lags = np.argsort(corr)[::-1][:8]
            for a_l in top_lags:
                for b_l in top_lags:
                    if a_l == b_l:
                        continue
                    denom = d[b_l] - d[a_l]
                    if abs(denom) > 1e-12:
                        dc = -(corr[a_l] - corr[b_l]) / denom
                        if lo < dc < hi:
                            candidates.append(dc)
            best_delta = 0.0
            best_val = np.inf
            for dc in candidates:
                dc = float(np.clip(dc, lo, hi))
                val = float(np.max(corr + dc * d))
                if val < best_val - 1e-15:
                    best_val = val
                    best_delta = dc
            if abs(best_delta) > 1e-12:
                h[i] += best_delta
                h[j] -= best_delta
                h = np.clip(h, 0.0, 1.0)
                corr = compute_corr(h)
                grads = corr_grad_all(h, n)
                c5_v = float(np.max(corr) * 2.0 / n)
                if c5_v < best_c5 - 1e-14:
                    best_c5 = c5_v
                    improved = True
        if not improved and path > 15:
            break
    return h, best_c5

        h = 0.5 * h + 0.2 * rng.randn(n)
    else:
        h = rng.uniform(0.2, 0.8, n)
    return repair(h, n)
```
