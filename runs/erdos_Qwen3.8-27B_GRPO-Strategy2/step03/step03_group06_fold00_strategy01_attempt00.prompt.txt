def _corr_arr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _compute_c5(h, dx):
    return float(np.max(_corr_arr(h)) * dx)

def _smooth_obj(x, alpha):
    corr = _corr_arr(x)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_obj_pen(x, alpha, target, pen):
    return _smooth_obj(x, alpha) + pen * (np.sum(x) - target) ** 2

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)

        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()

        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub

    return h
