import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
        d = target - s
        if abs(d) < 1e-15:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-15:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def make_fourier_basis(n, M):
    idx = np.arange(n)
    cols = []
    for m in range(1, M + 1):
        cols.append(np.cos(2 * np.pi * m * idx / n))
        cols.append(np.sin(2 * np.pi * m * idx / n))
    return np.column_stack(cols)

def fourier_nm(h0, n, M, alpha, maxiter=5000):
    B = make_fourier_basis(n, M)
    h_c = h0 - 0.5
    coeffs0 = 2.0 * (B.T @ h_c) / n

    def obj(c):
        h_raw = 0.5 + B @ c
        h = np.clip(h_raw, 0.0, 1.0)
        corr = compute_corr(h)
        m = np.max(corr)
        s = np.clip(alpha * (corr - m), -500, 500)
        val = m + np.log(np.sum(np.exp(s))) / alpha
        val += 0.05 * np.sum((h - h_raw) ** 2)
        return val

    best_h = h0.copy()
    best_val = compute_c5(h0, n)
    for restart in range(3):
        c0 = coeffs0.copy()
        if restart > 0:
            rng_r = np.random.RandomState(restart * 99 + 7)
            c0 += 0.02 * rng_r.randn(len(c0))
        try:
            res = minimize(obj, c0, method='Nelder-Mead',
                           options={'maxiter': maxiter, 'xatol': 1e-12, 'fatol': 1e-14})
            h_out = repair(np.clip(0.5 + B @ res.x, 0, 1), n)
            c5_out = compute_c5(h_out, n)
            if c5_out < best_val:
                best_h = h_out
                best_val = c5_out
        except Exception:
            pass
    return best_h

def slsqp(h, n, alpha, topk, fw, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(hv):
        corr = compute_corr(hv)
        m = np.max(corr)
        s = np.clip(alpha * (corr - m), -500, 500)
        val = m + np.log(np.sum(np.exp(s))) / alpha
        if topk > 1:
            sc = np.sort(corr)[::-1][:topk]
            val += fw * np.sum((sc - sc.mean()) ** 2)
        return val

    def con(hv):
        return hv.sum() - target

    cons = [{'type': 'eq', 'fun': con}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(res.x, n)
    except Exception:
        return h

                k2 = -k
                t1 = np.zeros(n)
                t1[k2:] = 1.0 - h[:n - k2]
                t2 = np.zeros(n)
                t2[:n - k2] = h[k2:]
                grad += wp * (t1 - t2)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = repair(h_new, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best:
            best = c5_new
            h = h_new
        else:
            lr *= 0.7
            if lr < 1e-10:
                break
    return h, best

def coord_desc(h, n, t_end, rng, n_rounds=400):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004]
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(5 * n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for d in steps:
                for sign in (1, -1):
                    ni = hi + sign * d
                    nj = hj - sign * d
                    if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-14:
                        best = c5
                        improved = True
                        break
                    h[i], h[j] = hi, hj
                if improved:
                    break
        if not improved and r > 60:
            break
    return h, best

def make_random_start(n, seed_val):
    rng = np.random.RandomState(seed_val)
    kind = seed_val % 5
    if kind == 0:
        h = 0.5 + 0.3 * rng.randn(n)
    elif kind == 1:
        h = np.where(np.arange(n) < n // 2, 1.0, 0.0)
        h = 0.8 * h + 0.2 * rng.uniform(0, 1, n)
    elif kind == 2:
        f = rng.randint(1, 6)
        h = 0.5 + 0.35 * np.sin(2 * np.pi * f * np.arange(n) / n + rng.uniform(0, 6.28))
    elif kind == 3:
        block = max(2, n // 5)
        h = np.zeros(n)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
        h = 0.7 * h + 0.3 * rng.uniform(0, 1, n)
    else:
        h = 0.5 + 0.35 * rng.uniform(-1, 1, n)
    return repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed + 54321)
    best_h = None
    best_c5 = np.inf
    best_n = 70

    init_h = None
    init_n = 70
    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
        init_n = len(init_h)
    except NameError:
        pass
    if init_h is not None:
        init_h = repair(init_h, init_n)
        best_h = init_h.copy()
        best_c5 = compute_c5(init_h, init_n)
        best_n = init_n

    # Phase 1: Gradient-based local search on initial
    if best_h is not None and time.time() - t0 < budget_s * 0.12:
        h_g, c5_g = grad_smooth_step(best_h, best_n, 5000, 0.002, 2000)
        if c5_g < best_c5:
            best_h, best_c5 = h_g, c5_g
        h_g2, c5_g2 = grad_smooth_step(best_h, best_n, 20000, 0.001, 1500)
        if c5_g2 < best_c5:
            best_h, best_c5 = h_g2, c5_g2

    # Phase 2: Fourier re-optimization
    if best_h is not None and time.time() - t0 < budget_s * 0.25:
        for M in [4, 6, 8]:
            if time.time() - t0 > budget_s * 0.25:
                break
            for alpha in [5000, 20000]:
                if time.time() - t0 > budget_s * 0.25:
                    break
                h_f = fourier_nm(best_h, best_n, M, alpha, maxiter=3000)
                c5_f = compute_c5(h_f, best_n)
                if c5_f < best_c5:
                    best_h, best_c5 = h_f, c5_f

    # Phase 3: SLSQP main refinement
    h_work = best_h.copy()
    for alpha in [3000, 10000, 30000, 80000, 150000]:
        if time.time() - t0 > budget_s * 0.50:
            break
        for topk in [1, 3, 5, 7]:
            if time.time() - t0 > budget_s * 0.50:
                break
            for fw in [0.0, 0.003, 0.01, 0.04]:
                if time.time() - t0 > budget_s * 0.50:
                    break
                h_new = slsqp(h_work, best_n, alpha, topk, fw, 700)
                c5_new = compute_c5(h_new, best_n)
                if c5_new < best_c5 - 1e-14:
                    best_h, best_c5 = h_new.copy(), c5_new
                    h_work = h_new
                elif c5_new < compute_c5(h_work, best_n) - 1e-14:
                    h_work = h_new

    # Phase 4: Gradient steps on best
    if time.time() - t0 < budget_s * 0.65:
        h_g, c5_g = grad_smooth_step(best_h, best_n, 50000, 0.001, 3000)
        if c5_g < best_c5:
            best_h, best_c5 = h_g, c5_g

    # Phase 5: Coordinate descent
    if time.time() - t0 < budget_s * 0.80:
        h_cd, c5_cd = coord_desc(best_h, best_n, t0 + budget_s * 0.80, rng, 500)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd

    # Phase 6: SLSQP + grad on best
    if time.time() - t0 < budget_s * 0.88:
        h_fin = slsqp(best_h, best_n, 100000, 3, 0.01, 1000)
        c5_fin = compute_c5(h_fin, best_n)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin, c5_fin
        h_g3, c5_g3 = grad_smooth_step(best_h, best_n, 30000, 0.0005, 2000)
        if c5_g3 < best_c5:
            best_h, best_c5 = h_g3, c5_g3

    # Phase 7: Try alternative n values
    if time.time() - t0 < budget_s * 0.96:
        for n_alt in [56, 64, 72, 84, 100]:
            if time.time() - t0 > budget_s * 0.96:
                break
            if n_alt == best_n:
                continue
            for kk in range(2):
                if time.time() - t0 > budget_s * 0.96:
                    break
                h_alt = make_random_start(n_alt, seed + n_alt * 31 + kk)
                h_alt = fourier_nm(h_alt, n_alt, 5, 10000, maxiter=2000)
                h_alt = slsqp(h_alt, n_alt, 30000, 3, 0.01, 500)
                h_alt, c5_t = grad_smooth_step(h_alt, n_alt, 20000, 0.001, 1000)
                c5_alt = compute_c5(h_alt, n_alt)
                if c5_alt < best_c5 - 1e-14:
                    best_h, best_c5, best_n = h_alt, c5_alt, n_alt

    # Phase 8: Final coordinate descent polish
    if time.time() - t0 < budget_s * 0.995:
        h_p, c5_p = coord_desc(best_h, best_n, t0 + budget_s * 0.995, rng, 150)
        if c5_p < best_c5:
            best_h, best_c5 = h_p, c5_p

    if best_h is None:
        best_n = 70
        best_h = repair(np.full(best_n, 0.5), best_n)
        best_c5 = compute_c5(best_h, best_n)

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
