import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h))) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_obj(x, n, alpha):
    c = _corr(x) * (2.0 / n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = _corr(x) * dx
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    total = np.sum(e)
    w = e / total
    grad = np.zeros(n)
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        j_min = max(0, -lag)
        j_max = min(n, n - lag)
        for j in range(j_min, j_max):
            grad[j] -= wk * x[j + lag]
            grad[j + lag] += wk * (1.0 - x[j])
    return grad * dx

    c = _corr(x) * (2.0 / n)
    idx = np.argsort(c)[::-1][:k]
    return np.sum(c[idx])

def _topk_grad(x, n, k):
    dx = 2.0 / n
    c = _corr(x) * dx
    idx = np.argsort(c)[::-1][:k]

        j_min = max(0, -lag)
        j_max = min(n, n - lag)
        for j in range(j_min, j_max):
            grad[j] -= x[j + lag]
            grad[j + lag] += (1.0 - x[j])
    return grad * dx / k

    def obj(x):
        c = _corr(x) * (2.0 / n)
        m = np.max(c)
        s = np.clip(alpha * (c - m), -500, 500)
        sm = m + np.log(np.sum(np.exp(s))) / alpha
        penalty_term = penalty * (x.sum() - n / 2.0) ** 2
        return sm + penalty_term
    def grad(x):
        g = _smooth_grad(x, n, alpha)
        g += 2.0 * penalty * (x.sum() - n / 2.0) * np.ones(n)
        return g
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h_init, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return res.x, res.nit

def _slsqp_run(h_init, n, alpha, maxiter=400):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    res = minimize(
        lambda x: _smooth_obj(x, n, alpha), h_init,
        jac=lambda x: _smooth_grad(x, n, alpha),
        method='SLSQP', bounds=bounds, constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-16})
    return res.x

def _coord_descent(h, n, budget_end, rng, steps):

    for _pass in range(10000):
        if time.time() > budget_end:
            break
        improved = False

            hi = h[i]

                if time.time() > budget_end:

                hj = h[j]

                best_c = c5_cur
                for d in steps:

                    h[i], h[j] = ni, nj
                    c = _c5(h, n)

                if best_d != 0:
                    h[i], h[j] = hi + best_d, hj - best_d
                    c5_cur = best_c
                    improved = True
        if not improved:

    return h, c5_cur

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h))) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_obj(x, n, alpha):
    c = _corr(x) * (2.0 / n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = _corr(x) * dx
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / np.sum(e)
    grad = np.zeros(n)
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        j_min = max(0, -lag)
        j_max = min(n, n - lag)
        for j in range(j_min, j_max):
            grad[j] -= wk * x[j + lag]
            grad[j + lag] += wk * (1.0 - x[j])
    return grad * dx

def _hinge_obj(x, n, t, alpha):
    c = _corr(x) * (2.0 / n)
    viol = np.maximum(c - t, 0)
    return np.sum(viol ** 2) * alpha + 0.001 * np.sum((x - 0.5) ** 2)

def _hinge_grad(x, n, t, alpha):
    c = _corr(x) * (2.0 / n)
    viol = np.maximum(c - t, 0)
    active = viol > 0
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for k in range(len(viol)):
        if not active[k]:
            continue
        coeff = 2.0 * alpha * viol[k] * dx_val
        lag = k - (n - 1)
        j_min = max(0, -lag)
        j_max = min(n, n - lag)
        for j in range(j_min, j_max):
            grad[j] -= coeff * x[j + lag]
            grad[j + lag] += coeff * (1.0 - x[j])
    grad += 0.002 * (x - 0.5)
    return grad

    def obj(x):
        return _smooth_obj(x, n, alpha) + penalty * (x.sum() - n / 2.0) ** 2
    def grad(x):
        return _smooth_grad(x, n, alpha) + 2.0 * penalty * (x.sum() - n / 2.0)
    res = minimize(obj, h_init, jac=grad, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
    return res.x

def _slsqp(h_init, n, alpha, maxiter=300):
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    res = minimize(lambda x: _smooth_obj(x, n, alpha), h_init,
                   jac=lambda x: _smooth_grad(x, n, alpha),
                   method='SLSQP', bounds=[(0.0, 1.0)] * n,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
    return res.x

def _hinge_lbfgsb(h_init, n, t, alpha, maxiter=200):

        if best_d != 0:
            h[i] += best_d
            h[j] -= best_d
            c5_cur = best_c
    return h, c5_cur
