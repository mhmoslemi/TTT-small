import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _subgradient(h, corr, eps_rel=1e-4):
    """Compute subgradient of max(corr) w.r.t. h"""
    n = len(h)
    m = float(np.max(corr))
    # Active set: lags within eps_rel of max
    active = np.where(corr >= m - eps_rel * m)[0]
    g = np.zeros(n)
    for k_idx in active:
        k = k_idx - (n - 1)  # actual lag
        # corr(k) = sum_i h[i]*(1-h[i+k]) for valid i
        # d corr(k)/d h[j] = (1-h[j+k]) - h[j-k] (with OOB = 0)
        for j in range(n):
            jp = j + k
            jm = j - k
            grad = 0.0
            if 0 <= jp < n:
                grad += (1.0 - h[jp])
            if 0 <= jm < n:
                grad -= h[jm]
            g[j] += grad
    g /= len(active)
    return g

def _project_onto_feasible(v, n, target):
    """Project v onto [0,1]^n ∩ {sum = target}"""
    # Bisection on lambda: h = clip(v - lambda, 0, 1), find lambda s.t. sum(h) = target
    lo, hi = -2.0, 2.0  # wide range
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    # Fine correction
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12: break
    return h

def _project(v, n, target):
    lo, hi = v.min() - 2.0, v.max() + 2.0
    for _ in range(80):
        mid = 0.5*(lo+hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target: lo = mid
        else: hi = mid
    h = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0-h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12: break
    return h

def _g_vec(h, p, n):
    kk = np.arange(-(n-1), n)
    g = np.zeros(2*n-1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha*(c-m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s))))/alpha

def _pgd_phase(h, n, target, dx, t_end, alphas=(50, 200, 1000, 3000)):
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    for alpha in alphas:
        lr = 0.5 / (alpha * n)  # conservative initial step
        h_c = best_h.copy()
        c_prev = _c5(h_c, dx)
        stagnant = 0
        while time.time() < t_end - 30:
            c = _corr(h_c)
            m = float(np.max(c))
            s = np.clip(alpha*(c-m), -500, 500)
            w = np.exp(s); w /= w.sum()
            g = np.zeros(n)
            for p in range(n):
                g[p] = np.dot(w, _g_vec(h_c, p, n))
            h_new = _project(h_c - lr*g, n, target)
            c5_new = _c5(h_new, dx)
            if c5_new < best_c5 - 1e-10:
                best_h = h_new.copy()
                best_c5 = c5_new
                stagnant = 0
                # maybe increase lr slightly
                lr = min(lr * 1.1, 1.0/(alpha*n*0.1))
            elif c5_new < c_prev - 1e-12:
                h_c = h_new
                c_prev = c5_new
                stagnant += 1
            else:
                lr *= 0.7
                stagnant += 1
            h_c = h_new if c5_new < c_prev else h_c
            if stagnant > 200:
                break
    return best_h, best_c5

def _sweep_round(h, n, time_limit, n_coarse=25, n_bisect=35):
    corr = _corr(h)
    improved = False
    order = np.random.permutation(n)
    for ai in range(n):
        i = order[ai]
        if time.time() > time_limit:
            return h, corr, improved
        for jj in range(ai+1, n):
            j = order[jj]
            if time.time() > time_limit:
                return h, corr, improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj-1.0)
            d_max = min(1.0-hi, hj)
            if d_max - d_min < 1e-12: continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None,:] + ds[:,None]*gd[None,:], axis=1)
            bi = np.argmin(vals)
            cur_max = float(np.max(corr))
            if vals[bi] >= cur_max - 1e-13: continue
            lo = max(d_min, ds[max(0,bi-1)])
            hi_b = min(d_max, ds[min(n_coarse-1,bi+1)])
            for _ in range(n_bisect):
                mid = 0.5*(lo+hi_b)
                if np.max(corr + mid*gd) <= np.max(corr + hi_b*gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5*(lo+hi_b)
            h[i] = hi + d_best; h[j] = hj - d_best
            corr = corr + d_best*gd
            improved = True
    return h, corr, improved

def _slsqp_step(h, alpha, maxiter, bounds, target, n):
    def obj(x): return _smooth_max(_corr(x), alpha)
    def eq_con(x): return float(np.sum(x)) - target
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type':'eq','fun':eq_con}],
                       options={'maxiter':maxiter,'ftol':1e-14,'maxls':50})
        return _repair(res.x, n)
    except: return h
