def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _overlaps(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * 2.0 / len(x)

def _obj_grad(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)

        if w[mi] < 1e-14:

        if k == 0:

        g = np.zeros(n)

        g[k:] -= h[:n-k]
    else:
        a = -k
        g = np.zeros(n)
        g[a:] += (1.0 - h[:n-a])
        g[:n-a] -= h[a:]

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,

    try:

                       args=(alpha,), method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

                                    best_c5 = c5c; best_h = h.copy()
                                break
                            h[i], h[j] = h[i] - s, h[j] + s

    # Phase 8: Symmetry check
    h_sym = (best_h + 1.0 - best_h[::-1]) / 2.0
    h_sym = _repair(h_sym, n)
    c5_sym = _eval(h_sym)
    if c5_sym < best_c5:
        best_c5 = c5_sym
        best_h = h_sym.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
