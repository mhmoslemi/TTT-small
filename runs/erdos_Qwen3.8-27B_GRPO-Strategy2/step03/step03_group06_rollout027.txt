                diff -= a

def subgradient_step(h, n, alpha_top, step_size):
    """One subgradient step targeting top lags."""
    corr = compute_corr(h)
    # Find top lags with softmax weighting
    top_k = 10
    sorted_corr = np.sort(corr)[::-1]
    weights = np.exp(alpha_top * (corr - np.max(corr)))
    weights /= weights.sum()
    
    # Compute weighted gradient
    grad = np.zeros(n)
    n_corr = len(corr)
    for m in range(n_corr):
        if weights[m] < 1e-6:
            continue
        w = weights[m]
        lag = m - (n - 1)  # can be negative
        if lag >= 0:
            # q = sum_{i=0}^{n-1-lag} h[i] - sum_{i=0}^{n-1-lag} h[i]*h[i+lag]
            # dq/dh[j] = I(j <= n-1-lag) - h[j+lag]*I(j <= n-1-lag) - h[j-lag]*I(j >= lag)
            for j in range(n - lag):
                grad[j] += w * (1.0 - h[j+lag])
            for j in range(lag, n):
                grad[j] -= w * h[j-lag]
        else:
            p = -lag
            # q = sum_{j=p}^{n-1} h[j] - sum_{j=p}^{n-1} h[j]*h[j-p]
            for j in range(p, n):
                grad[j] += w * (1.0 - h[j-p])
            for j in range(0, n - p):
                grad[j] -= w * h[j+p]
    
    return grad

def compute_aug_grad(h, n, weights):
    """Compute weighted sum of gradients using vectorization."""
    grad = np.zeros(n)
    n_corr = 2*n - 1
    corr = compute_corr(h)  # already computed outside
    
    for m in range(n_corr):
        w = weights[m]
        if w < 1e-8:
            continue
        lag = m - (n-1)
        if lag >= 0:
            # First part: d/dh[j] of sum_{i=0}^{n-1-lag} h[i] for j in [0, n-1-lag]
            # This is 1 for j in [0, n-1-lag-1]
            # Second part: -d/dh[j] of sum_{i=0}^{n-1-lag} h[i]*h[i+lag]
            # For j in [0, n-1-lag-1]: contributes -h[j+lag]
            # For j in [lag, n-1]: contributes -h[j-lag]
            idx1 = np.arange(n - lag)
            grad[idx1] += w * (1.0 - h[idx1 + lag])
            idx2 = np.arange(lag, n)
            grad[idx2] -= w * h[idx2 - lag]
        else:
            p = -lag
            idx1 = np.arange(p, n)
            grad[idx1] += w * (1.0 - h[idx1 - p])
            idx2 = np.arange(0, n - p)
            grad[idx2] -= w * h[idx2 + p]
    
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, dx):
    return float(np.max(compute_corr(h)) * dx)

def smooth_max(corr, alpha):
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def obj_smooth(x, alpha, target, pen):
    corr = compute_corr(x)
    return smooth_max(corr, alpha) + pen * (np.sum(x) - target) ** 2

def obj_con(x, alpha):
    corr = compute_corr(x)
    return smooth_max(corr, alpha)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i]); h[i] += a; diff -= a
            else:
                a = min(-diff, h[i]); h[i] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def aug_gradient(h, n, top_lags):
    grad = np.zeros(n)
    for m in top_lags:
        lag = m - (n - 1)
        if lag >= 0:
            i1 = np.arange(n - lag)
            grad[i1] += (1.0 - np.roll(np.append(h, 0), -lag)[:n - lag])
            i2 = np.arange(lag, n)
            grad[i2] -= np.roll(np.append(0, h), lag)[:n - lag]
            # direct: h[j+lag] for j in [0,n-lag), h[j-lag] for j in [lag,n]
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
        else:
            p = -lag
            grad[p:] += (1.0 - h[:-p])
            grad[:n - p] -= h[p:]
    return grad

def lbfgs_opt(h, n, dx, alpha, maxiter=300):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best_h, best_c5 = h, compute_c5(h, dx)

def slsqp_opt(h, n, dx, alpha, maxiter=400):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}

        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5n = compute_c5(h, dx)
        if c5n < c5_cur or rng.random() < np.exp(-(c5n - c5_cur) / T):
            c5_cur = c5n
            if c5n < c5_best:
                c5_best, h_best = c5n, h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return h_best, c5_best

def fine_ls(h, n, dx, rng, deadline):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    scales = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    per = 25000
    for scale in scales:
        improved = False
        for _ in range(per):
            if time.time() >= deadline:
                return h, best_c5
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            a, b = h[i] + d, h[j] - d
            if a < -1e-9 or a > 1 + 1e-9 or b < -1e-9 or b > 1 + 1e-9:
                continue
            h[i], h[j] = a, b
            c5 = compute_c5(h, dx)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5; improved = True
            else:
                h[i], h[j] = a - d, b + d
        if not improved and scale < 0.001:
            break
    return h, best_c5

def block_descent(h, n, dx, rng, deadline):
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    target = n / 2.0
    bs = 5
    n_blocks = n // bs
    for epoch in range(200):
        if time.time() >= deadline:
            break
        order = rng.permutation(n_blocks)
        for bi in order:
            i0 = bi * bs
            i1 = i0 + bs
            block = h[i0:i1].copy()
            rest_sum = target - block.sum()
            # Optimize this block: minimize max_corr with block free, rest fixed
            # Use small L-BFGS on just bs variables
            def block_obj(bv):
                hx = h.copy()
                hx[i0:i1] = np.clip(bv, 0, 1)
                hx_sum = hx.sum()
                if hx_sum < 1e-9:
                    return 1.0
                corr = compute_corr(hx)
                return smooth_max(corr, 2000.0)
            try:
                res = minimize(block_obj, block, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * bs,
                               options={'maxiter': 50, 'ftol': 1e-14})
                new_block = np.clip(res.x, 0.0, 1.0)
                # Renormalize: scale block to hit target sum
                cur_rest = h.sum() - block.sum()
                desired_block_sum = target - cur_rest
                if desired_block_sum > 0 and new_block.sum() > 1e-9:
                    new_block *= desired_block_sum / new_block.sum()
                    new_block = np.clip(new_block, 0.0, 1.0)
                h[i0:i1] = new_block
                hc = repair(h, n)
                c5 = compute_c5(hc, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    h = hc
            except Exception:
                pass
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    dx0 = 2.0 / n0
    h0 = repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5, best_n = h0, compute_c5(h0, dx0), n0

    # Phase 1: L-BFGS warmup (40s)
    t_end = t0 + 40
    for alpha in [100, 500, 2000]:
        if time.time() >= t_end or rem() < 500:
            break
        ht, c5t = lbfgs_opt(best_h, best_n, 2.0/best_n, alpha, maxiter=300)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 2: SLSQP refinement (80s)
    t_end = t0 + 120
    for alpha in [500, 2000, 8000, 30000]:
        if time.time() >= t_end or rem() < 400:
            break
        ht, c5t = slsqp_opt(best_h, best_n, 2.0/best_n, alpha, maxiter=500)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 3: Subgradient descent targeting top lags (80s)
    if rem() > 350:
        t_end = time.time() + min(80, rem() - 350)
        ht, c5t = subgrad_descent(best_h, best_n, 2.0/best_n, rng, t_end)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 4: SLSQP after subgradient (40s)
    if rem() > 300:
        t_end = time.time() + min(40, rem() - 300)
        for alpha in [5000, 20000]:
            if time.time() >= t_end:
                break
            ht, c5t = slsqp_opt(best_h, best_n, 2.0/best_n, alpha, maxiter=400)
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t

    # Phase 5: Block descent (150s)
    if rem() > 300:
        t_end = time.time() + min(150, rem() - 300)
        ht, c5t = block_descent(best_h, best_n, 2.0/best_n, rng, t_end)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 6: Multi-restart with binary SA and other n (200s)
    if rem() > 250:
        t_end = time.time() + min(200, rem() - 250)
        for n_try in [60, 70, 80, 90, 100, 120]:
            if time.time() >= t_end or rem() < 150:
                break
            dx_try = 2.0 / n_try
            hb, _ = binary_sa(n_try, rng, time.time() + 30, max_iter=80000)
            ht, c5t = lbfgs_opt(hb, n_try, dx_try, 500, maxiter=200)
            ht, c5t = slsqp_opt(ht, n_try, dx_try, 8000, maxiter=300)
            if c5t < best_c5:
                best_h, best_c5, best_n = ht, c5t, n_try
            # Also try from current best reshaped
            if n_try != best_n and rem() > 180:
                ht2 = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, best_n), best_h)
                ht2 = repair(ht2, n_try)
                ht2, c5t2 = lbfgs_opt(ht2, n_try, dx_try, 1000, maxiter=200)
                ht2, c5t2 = slsqp_opt(ht2, n_try, dx_try, 10000, maxiter=300)
                if c5t2 < best_c5:
                    best_h, best_c5, best_n = ht2, c5t2, n_try

    # Phase 7: Fine local search (150s)
    if rem() > 120:
        t_end = time.time() + min(150, rem() - 100)
        ht, c5t = fine_ls(best_h, best_n, 2.0/best_n, rng, t_end)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 8: Final SLSQP + subgradient (60s)
    if rem() > 60:
        t_end = time.time() + min(30, rem() - 40)
        for alpha in [30000, 100000]:
            if time.time() >= t_end:
                break
            ht, c5t = slsqp_opt(best_h, best_n, 2.0/best_n, alpha, maxiter=300)
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t
        if rem() > 30:
            t_end2 = time.time() + min(20, rem() - 20)
            ht, c5t = subgrad_descent(best_h, best_n, 2.0/best_n, rng, t_end2)
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t

    # Phase 9: Ultra-fine local search (30s)
    if rem() > 20:
        t_end = time.time() + max(5, min(25, rem() - 5))
        ht, c5t = fine_ls(best_h, best_n, 2.0/best_n, rng, t_end)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
