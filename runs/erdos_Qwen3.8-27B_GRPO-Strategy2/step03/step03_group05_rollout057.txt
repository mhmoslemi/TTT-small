def _proj(h, target):

    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            a = min(diff, 1.0 - h[i])
            h[i] += a
            diff -= a
            if diff < 1e-14: break

                    c5_cur = _c5(h, n)
    return h, c5_cur

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h))) * (2.0 / n)

def _proj(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            a = min(diff, 1.0 - h[i])
            h[i] += a
            diff -= a
            if diff < 1e-14:
                break
    else:
        order = np.argsort(-h)
        for i in order:
            a = min(-diff, h[i])
            h[i] -= a
            diff += a
            if -diff < 1e-14:
                break
    return h

def _grad_topk(h, top_idx):
    n = len(h)
    g = np.zeros(n)
    for ki in top_idx:
        L = ki - (n - 1)
        if L == 0:
            g += 1.0 - 2.0 * h
        elif L > 0:
            g[L:] += 1.0 - h[:n - L]
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += 1.0 - h[M:]
            g[M:] -= h[:n - M]
    return g

    h += 0.004 * rng.randn(n)
