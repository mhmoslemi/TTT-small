import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
        d = target - s
        if abs(d) < 1e-15:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-15:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def smooth_max(corr, alpha):
    m = corr.max()
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def objective(h, n, alpha, topk, flat_w, pen):
    corr = compute_corr(h)
    val = smooth_max(corr, alpha)
    if topk > 1:
        sc = np.sort(corr)[::-1][:topk]
        val += flat_w * np.sum((sc - sc.mean())**2)
    val += pen * (h.sum() - n/2.0)**2
    return val

def make_start(n, seed_val, kind):
    rng = np.random.RandomState(seed_val)
    if kind == 0:
        h = 0.5 + 0.3 * rng.randn(n)
    elif kind == 1:
        f = rng.randint(2, 8)
        ph = rng.uniform(0, 6.283)
        h = 0.5 + 0.35 * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 2:
        h = np.zeros(n)
        # Thue-Morse-like: alternate blocks
        b = max(1, n // 20)
        for i in range(0, n, 2*b):
            h[i:i+b] = 1.0
        h = 0.3 + 0.4*h + 0.15*rng.randn(n)
    elif kind == 3:
        h = 0.5 + 0.25 * np.cos(4 * np.pi * np.arange(n) / n)
    elif kind == 4:
        h = 0.5 + 0.2 * rng.randn(n) + 0.15 * np.sin(2*np.pi*3*np.arange(n)/n)
    else:
        h = rng.uniform(0.1, 0.9, n)
    return repair(h, n)

def adam_opt(h, n, lr, n_steps, alpha, topk, flat_w, pen, beta1, beta2, eps_adam, t_end):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    
    for step in range(n_steps):
        if time.time() > t_end:
            break
        t += 1
        
        # Compute gradient numerically (central difference)
        corr = compute_corr(h)
        sm = smooth_max(corr, alpha)
        if topk > 1:
            sc = np.sort(corr)[::-1][:topk]
            sm += flat_w * np.sum((sc - sc.mean())**2)
        pen_term = pen * (h.sum() - n/2.0)**2
        base_val = sm + pen_term
        
        grad = np.zeros(n)
        eps_g = 1e-5
        for j in range(n):
            h[j] += eps_g
            c2 = compute_corr(h)
            sm2 = smooth_max(c2, alpha)
            if topk > 1:
                sc2 = np.sort(c2)[::-1][:topk]
                sm2 += flat_w * np.sum((sc2 - sc2.mean())**2)
            pen2 = pen * (h.sum() - n/2.0)**2
            h[j] -= 2*eps_g
            c3 = compute_corr(h)
            sm3 = smooth_max(c3, alpha)
            if topk > 1:
                sc3 = np.sort(c3)[::-1][:topk]
                sm3 += flat_w * np.sum((sc3 - sc3.mean())**2)
            pen3 = pen * (h.sum() - n/2.0)**2
            h[j] += eps_g
            grad[j] = (sm2 + pen2 - sm3 - pen3) / (2*eps_g)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        
        # Project
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = repair(h_new, n)
        
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        
        # Decay lr
        if step % 200 == 199:
            lr *= 0.9
    
    return best_h, best_c5

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def smooth_max(corr, alpha):
    m = corr.max()
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def obj_limited(h, n, alpha, topk, flat_w, pen):
    corr = compute_corr(h)
    val = smooth_max(corr, alpha)
    if topk > 1:
        sc = np.sort(corr, kind='mergesort')[::-1][:topk]
        val += flat_w * np.sum((sc - sc.mean())**2)
    val += pen * (h.sum() - n / 2.0) ** 2
    return val

def run_lbfgs(h, n, maxiter, alpha, topk, flat_w, pen, ftol=1e-15, gtol=1e-12):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_limited, h, args=(n, alpha, topk, flat_w, pen),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol})
        return repair(res.x, n)
    except Exception:
        return h

def coord_descent(h, n, n_trials, t_end, rng, base_steps=None):
    h = h.copy()
    best = compute_c5(h, n)
    if base_steps is None:
        base_steps = [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    
    for trial in range(n_trials):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        for d in base_steps:
            for sign in (1, -1):
                delta = sign * d
                ni, nj = hi + delta, hj - delta
                if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                    continue
                h[i], h[j] = ni, nj
                c5 = compute_c5(h, n)
                if c5 < best - 1e-14:
                    best = c5
                    break
                h[i], h[j] = hi, hj
            else:
                continue
            break
    
    return h, best

def make_start(n, seed_val, kind):
    rng = np.random.RandomState(seed_val)
    if kind == 0:
        h = 0.5 + 0.3 * rng.randn(n)
    elif kind == 1:
        f = rng.randint(2, 10)
        ph = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.35 * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 2:
        # Block pattern
        b = max(1, n // 10)
        h = np.zeros(n)
        for i in range(0, n, 2 * b):
            h[i:i + b] = 1.0
        h = 0.3 + 0.4 * h + 0.2 * rng.randn(n)
    elif kind == 3:
        h = 0.5 + 0.25 * np.cos(4 * np.pi * np.arange(n) / n)
    elif kind == 4:
        h = 0.5 + 0.2 * rng.randn(n) + 0.15 * np.sin(4 * np.pi * np.arange(n) / n)
    elif kind == 5:
        # Two-frequency
        h = 0.5 + 0.2 * np.sin(2 * np.pi * 2 * np.arange(n) / n) + 0.15 * np.cos(2 * np.pi * 5 * np.arange(n) / n + 1.0)
    else:
        h = rng.uniform(0.1, 0.9, n)
    return repair(h, n)
