import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(-(1.0 - h) if diff > 0 else h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return h

def _coeff_matrix(h, n):
    """Compute a[i,m] = coefficient of h[i] in corr[m]."""
    m_total = 2 * n - 1
    A = np.zeros((n, m_total))
    for i in range(n):
        for m in range(m_total):
            k = m - (n - 1)
            coeff = 0.0
            if 0 <= i + k < n:
                coeff += (1.0 - h[i + k])
            if 0 <= i - k < n:
                coeff -= h[i - k]
            A[i, m] = coeff
    return A
