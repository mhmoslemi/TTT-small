def _repair(h, n):
    """Clip and adjust sum to n/2."""
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    scale = target / s if s > 0 else 0.5
    h *= scale
    h = np.clip(h, 0.0, 1.0)

    diff = target - h.sum()
    if diff > 0:

    for i in idx:
        if diff > 0:
            a = min(diff, 1.0 - h[i])
            h[i] += a; diff -= a
        else:
            a = min(-diff, h[i])
            h[i] -= a; diff += a
        if abs(diff) < 1e-12:

    return float(c.max() * 2.0 / n)

def _grad_c(k, h):
    """Gradient of correlation at lag k (0 to 2n-2) w.r.t. h."""
    n = len(h)
    L = k - (n - 1)  # actual lag
    g = np.zeros(n)
    if L == 0:
        g = 1.0 - 2.0 * h
    elif L > 0:
        # c_L = sum_{m=0}^{n-1-L} h[m+L]*(1-h[m])
        # d/dh[j]: (1-h[j-L]) if j >= L; -h[j+L] if j <= n-1-L

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_active(h, top_k=5):
    n = len(h)
    c = _corr(h)
    idx = np.argpartition(c, -top_k)[-top_k:]
    w = c[idx] - c[idx].min() + 1e-10
    w /= w.sum()
    g = np.zeros(n)
    for i, k in enumerate(idx):
        L = k - (n - 1)
        wg = w[i]
        if L == 0:
            g += wg * (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += wg * (1.0 - h[:n - L])
            g[:n - L] -= wg * h[L:]
        else:
            M = -L
            g[:n - M] += wg * (1.0 - h[M:])
            g[M:] -= wg * h[:n - M]
    return g

def _sf(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sj(x, alpha):
    nn = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(nn)
    for k in range(2 * nn - 1):
        if w[k] < 1e-14:
            continue
        L = k - (nn - 1)
        wl = w[k]
        if L == 0:
            g += wl * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += wl * (1.0 - x[:nn - L])
            g[:nn - L] -= wl * x[L:]
        else:
            M = -L
            g[:nn - M] += wl * (1.0 - x[M:])
            g[M:] -= wl * x[:nn - M]
    return g

def _pgd(h0, n, iters, alpha, beta, top_k, t0, tlimit):
    h = h0.astype(float).copy()
    v = np.zeros(n)
    best = _eval(h)
    best_h = h.copy()
    for it in range(iters):
        if time.time() > tlimit:
            break
        y = h + beta * v
        g = _grad_active(y, top_k)
        h_new = _repair(y - alpha * g, n)
        v = beta * v + (h - h_new)
        h = h_new
        f = _eval(h)
        if f < best:
            best = f
            best_h = h.copy()
        if it > 0 and it % 500 == 0:
            if best > _eval(best_h) + 5e-6:
                alpha *= 0.8
            elif alpha < 0.05:
                alpha *= 1.1
        if it > 5000:
            beta = max(beta * 0.9995, 0.5)
    return best_h, best

def _slsqp(h, n, alpha, maxiter):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_sf, h, jac=_sj, args=(alpha,),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-19})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)
