            room = np.argsort(-(1.0 - h))
            for idx in room:

                h[idx] += add
                diff -= add
                if diff < 1e-12:

            for idx in room:
                rem = min(-diff, h[idx])
                h[idx] -= rem
                diff += rem

                h[idx] += add
                diff -= add
                if diff < 1e-12:

                rem = min(-diff, h[idx])

    k = m - (n - 1)

        g[k:] += (1.0 - h[:n-k])  # j from k to n-1, h[j-k] from h[0] to h[n-1-k]

        g[:n-mk] += (1.0 - h[mk:])
