```python
                    diff -= a
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + float(np.log(np.sum(np.exp(s))) / alpha)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eq_constraint(x, target):
    return np.sum(x) - target

def _slsqp_obj(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _slsqp_jac(x, alpha):
    # Numerical gradient (simple)
    eps = 1e-7
    g = np.zeros(len(x))
    for i in range(len(x)):
        xp = x.copy()
        xp[i] += eps
        xm = x.copy()
        xm[i] -= eps
        g[i] = (_slsqp_obj(xp, alpha) - _slsqp_obj(xm, alpha)) / (2*eps)
    return g

def _do_slsqp(x0, bounds, target, alpha, maxiter):
    res = minimize(_slsqp_obj, x0, args=(alpha,), method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': _eq_constraint, 'args': (target,)}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return res.x

def _slope_for_pair(h, i, j, n):
    # Compute slope vector for all lags when transferring δ from i to j
    slope = np.zeros(2 * n - 1)
    idx = n - 1  # center index in corr array
    for m in range(n - 1, -1, -1):  # m = 0 to n-1
        s = 0.0
        if 0 <= i + m < n: s += h[i + m]
        if 0 <= j + m < n: s -= h[j + m]
        if 0 <= i - m < n: s += h[i - m]
        if 0 <= j - m < n: s -= h[j - m]
        if m == 0:
            slope[idx] = s
        else:
            slope[idx + m] = s
            slope[idx - m] = s
    return slope

def _find_opt_delta(corr, slope, lo, hi):
    # Find delta in [lo, hi] minimizing max(corr + slope*delta)
    if hi - lo < 1e-14:
        return lo
    # Evaluate at boundaries
    f_lo = float(np.max(corr + slope * lo))
    f_hi = float(np.max(corr + slope * hi))
    best_f = min(f_lo, f_hi)
    best_d = lo if f_lo <= f_hi else hi
    # Find crossing points
    n_l = len(corr)
    for a in range(n_l):
        for b in range(a + 1, n_l):

            if abs(ds) < 1e-14:

            d = (corr[b] - corr[a]) / ds

                f = float(np.max(corr + slope * d))
                if f < best_f - 1e-15:
                    best_f = f
                    best_d = d
    return best_d
```

```python
def _slope_for_pair(h, i, j, n):
    """Compute slope vector for all lags when transferring δ from position i to position j."""
    L = 2 * n - 1
    center = n - 1
    slope = np.zeros(L)
    for k in range(L):
        m = k - center  # lag offset
        s = 0.0
        # Term 1: h[i] appears in a[n+m] position when n = i-m
        if 0 <= i - m < n:
            s -= (1.0 - h[i - m])
        # Term 2: h[i] appears in v[n] position when n = i, need i+m valid
        if 0 <= i + m < n:
            s += h[i + m]
        # Term 3: h[j] appears in a[n+m] position when n = j-m
        if 0 <= j - m < n:
            s += (1.0 - h[j - m])
        # Term 4: h[j] appears in v[n] position when n = j, need j+m valid
        if 0 <= j + m < n:
            s -= h[j + m]
        slope[k] = s
    return slope
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + float(np.log(np.sum(np.exp(s))) / alpha)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eq_c(x, target):
    return np.sum(x) - target

def _slsqp_f(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _do_slsqp(x0, bounds, target, alpha, maxiter):
    res = minimize(_slsqp_f, x0, args=(alpha,), method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': _eq_c, 'args': (target,)}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return res.x

def _pair_slope(h, i, j, n):
    L = 2 * n - 1
    center = n - 1
    sl = np.zeros(L)
    for k in range(L):
        m = k - center
        s = 0.0
        if 0 <= i - m < n:
            s -= (1.0 - h[i - m])
        if 0 <= i + m < n:
            s += h[i + m]
        if 0 <= j - m < n:
            s += (1.0 - h[j - m])
        if 0 <= j + m < n:
            s -= h[j + m]
        sl[k] = s
    return sl

def _opt_delta(corr, sl, lo, hi):
    if hi - lo < 1e-14:
        return lo, float(np.max(corr + sl * lo))
    nl = len(corr)
    sa = sl[:, None]
    sb = sl[None, :]
    ca = corr[:, None]
    cb = corr[None, :]
    ds = sa - sb
    with np.errstate(divide='ignore', invalid='ignore'):
        cross = (cb - ca) / ds
    mask = (np.abs(ds) > 1e-14) & (cross > lo + 1e-14) & (cross < hi - 1e-14)
    tri = np.arange(nl)[:, None] < np.arange(nl)[None, :]
    valid_cross = cross[mask & tri]
    cands = np.concatenate(([lo, hi], valid_cross))
    if len(cands) == 0:
        return lo, float('inf')
    vals = corr[None, :] + sl[None, :] * cands[:, None]
    fv = vals.max(axis=1)
    bi = int(np.argmin(fv))
    return cands[bi], float(fv[bi])

def _gradient_step(h, corr, n, step):
    top_k = min(8, 2 * n - 1)
    top = np.argsort(corr)[-top_k:]
    g = np.zeros(n)
    center = n - 1
    for k in top:
        m = k - center
        for i in range(n):
            if 0 <= i - m < n:
                g[i] += (1.0 - h[i - m])
            if 0 <= i + m < n:
                g[i] -= h[i + m]
    g /= top_k
    h_new = h - step * g
    h_new = _repair(h_new, n)
    return h_new

    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def elapsed():
        return time.time() - t0

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    h_alt = _repair(1.0 - h0, n)
    c5_alt = _c5(h_alt, dx)
    if c5_alt < best_c5:
        best_h, best_c5 = h_alt.copy(), c5_alt

    # Phase 1: SLSQP
    for alpha in [500, 3000, 20000, 80000]:
        if elapsed() > budget_s - 500:
            break
        for att in range(2):
            if elapsed() > budget_s - 500:
                break
            x0 = _repair(best_h + (0.003 * att) * rng.randn(n), n)
            try:
                rx = _do_slsqp(x0, bounds, target, alpha, 2000)
                hc = _repair(rx, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_h, best_c5 = hc, c5
            except Exception:
                pass

    # Phase 2: Exact pair-swap local search
    h = best_h.copy()
    corr = _corr(h).copy()
    c5_cur = _c5(h, dx)
    sweep = 0
    while elapsed() < budget_s - 200:
        improved = False
        order = list(range(n))
        rng.shuffle(order)
        for ii in range(n):
            for jj in range(ii + 1, n):
                i, j = order[ii], order[jj]
                hi, hj = h[i], h[j]
                lo = max(hi - 1.0, -hj)
                hib = min(hi, 1.0 - hj)
                if hib - lo < 1e-14:
                    continue

                d_opt, f_opt = _opt_delta(corr, sl, lo, hib)
                cur_max = float(np.max(corr))
                if f_opt < cur_max - 1e-13:
                    h[i] -= d_opt
                    h[j] += d_opt
                    h[i] = max(0.0, min(1.0, h[i]))
                    h[j] = max(0.0, min(1.0, h[j]))
                    corr = corr + sl * d_opt
                    if _c5(h, dx) < best_c5:
                        best_c5, best_h = _c5(h, dx), h.copy()
                    improved = True
        sweep += 1
        if not improved:
            break
        if sweep % 50 == 0:
            h = _repair(h, n)
            corr = _corr(h)

    # Phase 3: Gradient refinement on top lags
    if elapsed() < budget_s - 120:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        step = 0.02
        stall = 0
        for it in range(3000):
            if elapsed() > budget_s - 120:
                break
            corr = _corr(h)
            h_new = _gradient_step(h, corr, n, step)
            c5_new = _c5(h_new, dx)
            if c5_new < c5_cur - 1e-14:
                h = h_new
                c5_cur = c5_new
                stall = 0
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h.copy()
            else:
                stall += 1
                step *= 0.85
                if stall > 80:
                    break
        # Follow up with pair swap
        corr = _corr(h)
        for _sw in range(5):
            imp2 = False
            for ii in range(n):
                for jj in range(ii + 1, n):
                    i, j = ii, jj
                    hi2, hj2 = h[i], h[j]
                    lo2 = max(hi2 - 1.0, -hj2)
                    hib2 = min(hi2, 1.0 - hj2)
                    if hib2 - lo2 < 1e-14:
                        continue
                    sl2 = _pair_slope(h, i, j, n)
                    d2, f2 = _opt_delta(corr, sl2, lo2, hib2)
                    cm = float(np.max(corr))
                    if f2 < cm - 1e-13:
                        h[i] -= d2
                        h[j] += d2
                        h[i] = max(0.0, min(1.0, h[i]))
                        h[j] = max(0.0, min(1.0, h[j]))
                        corr = corr + sl2 * d2
                        imp2 = True
            c5_cur = _c5(h, dx)
            if c5_cur < best_c5:
                best_c5, best_h = c5_cur, h.copy()
            if not imp2:
                break

    # Phase 4: Fine random search
    if elapsed() < budget_s - 40:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        deltas = np.array([5e-5, 1e-4, 2e-4, 5e-4, -5e-5, -1e-4, -2e-4, -5e-4])
        while elapsed() < budget_s - 40:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = deltas[rng.randint(len(deltas))]
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5_cur - 1e-15:
                    c5_cur = c
                    if c5_cur < best_c5:
                        best_c5, best_h = c5_cur, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    # Phase 5: Final SLSQP polish
    if elapsed() < budget_s - 3:
        try:
            rx = _do_slsqp(best_h, bounds, target, 100000, 2000)
            hc = _repair(rx, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_h, best_c5 = hc, c5
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
```
