            g, c, max_c = _active_grad(h_ag, n)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            g = g / gn
            step = min(0.08, 0.8 / gn)
            improved = False
            for _ in range(18):
                h_try = _repair(h_ag - step * g, n)
                c5_try = _c5(h_try, dx)
                if c5_try < c5_ag - 1e-13:
                    h_ag = h_try
                    c5_ag = c5_try
                    improved = True
                    stall = 0
                    if c5_ag < best_c5:
                        best_c5 = c5_ag
                        best_h = h_ag.copy()
                    break
                step *= 0.4
            if not improved:
                stall += 1
                if stall > 300:
                    h_ag = _repair(h_ag + 0.02 * rng.randn(n), n)
                    c5_ag = _c5(h_ag, dx)
                    stall = 0

    # Phase 3: 0-1 Simulated Annealing
    if tleft() > 50:
        h_01 = (best_h > 0.5).astype(float)
        h_01 = _repair(h_01, n)
        h_sa = h_01.copy()
        c5_sa = _c5(h_sa, dx)
        T = 0.004
        max_it = min(300000, int(tleft() * 180))
        for it in range(max_it):
            if tleft() < 50:
                break
            T = max(3e-5, T * 0.99997)
            ones = np.where(h_sa > 0.5)[0]
            zeros = np.where(h_sa < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_sa[i], h_sa[j] = 0.0, 1.0
            c_new = _c5(h_sa, dx)
            d = c_new - c5_sa
            if d < 0 or rng.rand() < np.exp(-max(d, 0) / T):
                c5_sa = c_new
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
            else:
                h_sa[i], h_sa[j] = 1.0, 0.0

    # Phase 4: Upscale to 2n and optimize
    if tleft() > 250:
        n2 = 2 * n
        dx2 = 2.0 / n2
        target2 = n2 / 2.0
        _con_target = target2
        bounds2 = [(0.0, 1.0)] * n2
        h2 = _upscale(best_h, n, n2)
        h2 = _repair(h2, n2)
        c5_2 = _c5(h2, dx2)
        for alpha in [5000, 50000]:
            if tleft() < 120:
                break
            try:
                res = minimize(_smooth_obj, h2, args=(alpha,),
                    method='SLSQP', bounds=bounds2,
                    constraints=[{'type': 'eq', 'fun': _con_fun}],
                    options={'maxiter': 300, 'ftol': 1e-14, 'iprint': 0})
                hc = _repair(res.x, n2)
                c5 = _c5(hc, dx2)
                if c5 < c5_2:
                    c5_2, h2 = c5, hc
            except Exception:
                pass
        if tleft() > 120:
            c5_cur = c5_2
            for it in range(50000):
                if tleft() < 120:
                    break
                g, c, max_c = _active_grad(h2, n2)
                gn = np.linalg.norm(g)
                if gn < 1e-14:
                    break
                g = g / gn
                step = min(0.04, 0.5 / gn)
                for _ in range(12):
                    h_try = _repair(h2 - step * g, n2)
                    c_try = _c5(h_try, dx2)
                    if c_try < c5_cur - 1e-13:
                        h2 = h_try
                        c5_cur = c_try
                        c5_2 = c_try
                        break
                    step *= 0.5
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n2
            best_dx = dx2

    # Phase 5: Multi-start SLSQP on best resolution
    if tleft() > 35:
        nn = best_n
        _con_target = nn / 2.0
        bb = [(0.0, 1.0)] * nn
        for r in range(8):
            if tleft() < 25:
                break
            if r < 3:
                hp = best_h + 0.02 * rng.randn(nn)
            elif r < 5:
                hp = best_h.copy()
                for _ in range(4):
                    idx = rng.randint(nn)
                    hp[idx] = 1.0 - hp[idx]
            else:
                freq = 2.0 * np.pi * (r + 1) / nn
                hp = best_h + 0.03 * np.sin(np.arange(nn) * freq)
            hp = _repair(hp, nn)
            try:
                res = minimize(_smooth_obj, hp, args=(150000,),
                    method='SLSQP', bounds=bb,
                    constraints=[{'type': 'eq', 'fun': _con_fun}],
                    options={'maxiter': 400, 'ftol': 1e-15, 'iprint': 0})
                hc = _repair(res.x, nn)
                c5 = _c5(hc, best_dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc
            except Exception:
                pass

    # Phase 6: Gradient-guided fine pair search
    if tleft() > 10:
        h_f = best_h.copy()
        c5_f = _c5(h_f, best_dx)
        nn = best_n
        max_it = min(400000, int(tleft() * 6000))
        for it in range(max_it):
            if tleft() < 10:
                break
            i = rng.randint(nn)
            j = (i + 1 + rng.randint(nn - 1)) % nn
            c = _corr(h_f)
            m_star = np.argmax(c)
            g = _grad_lag(h_f, nn, m_star)
            if g[i] < g[j]:
                d = min(1.0 - h_f[i], h_f[j]) * 0.06
            else:
                d = -min(h_f[i], 1.0 - h_f[j]) * 0.06
            old_i, old_j = h_f[i], h_f[j]
            found = False
            for frac in [1.0, 0.5, 0.25]:
                dd = d * frac
                ni, nj = h_f[i] + dd, h_f[j] - dd
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h_f[i], h_f[j] = ni, nj
                    h_f = _repair(h_f, nn)
                    c_try = _c5(h_f, best_dx)
                    if c_try < c5_f - 1e-14:
                        c5_f = c_try
                        if c5_f < best_c5:
                            best_c5 = c5_f
                            best_h = h_f.copy()
                        found = True
                        break
            if not found:
                h_f = _repair(h_f, nn)
                h_f[i], h_f[j] = old_i, old_j
                h_f = _repair(h_f, nn)

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, best_dx)
    return (best_h, best_c5, best_n)
