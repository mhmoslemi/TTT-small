def _coord_descent_step(h, dx, n, max_passes=1000):
    """One full coordinate descent pass over all h[i]."""
    corr = _corr(h)
    best_c5 = _c5(h, dx)
    
    for i in range(n):
        lag_offsets = np.arange(2*n-1) - (n-1)  # lags from -(n-1) to n-1
        # Coefficient a_p for h[i] in c[p]
        a = np.zeros(2*n-1)
        for p in range(2*n-1):
            lag = lag_offsets[p]
            if 0 <= i + lag < n:
                a[p] += (1.0 - h[i+lag])
            if 0 <= i - lag < n:
                a[p] += (-h[i-lag])
        
        # f(delta) = max_p (corr[p] + a[p]*delta)
        # delta ∈ [-h[i], 1-h[i]]
        lo, hi = -h[i], 1.0 - h[i]
        
        def f(d):
            return np.max(corr + a * d)
        
        from scipy.optimize import minimize_scalar
        res = minimize_scalar(f, bounds=(lo, hi), method='bounded', options={'xatol': 1e-10})
        h[i] += res.x
        corr_new = _corr(h)  # Recompute (or update incrementally)
        new_c5 = np.max(corr_new) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            corr = corr_new
        else:
            # Revert
            h[i] -= res.x
            # corr unchanged
    return h, best_c5

    corr = _corr(h).astype(float)

    best_h = h.copy()

        for i in range(n):

            a = np.zeros(2*n-1)
            for lag in range(-(n-1), n):

                if 0 <= i - lag < n:
                    c_p += (-h[i-lag])
                a[idx] = c_p

            lo, hi = -h[i], 1.0 - h[i]
            if lo > hi:

            # f is convex piecewise linear
            # Try: is current delta=0 optimal?
            # Subgradient at 0: set of a[p] for p in argmax(corr)

        val += 100.0 * (np.sum(x) - target) ** 2

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
    for idx in order:
        if diff > 0:
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
        if abs(diff) < 1e-12:
            break
    return h

def _f_1d(d, corr, a):
    return np.max(corr + a * d)

def _smooth_obj(x, alpha, target, mu):
    corr = _corr(x)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + mu * (np.sum(x) - target) ** 2

def _topk_obj(x, alpha, k, target, mu):
    corr = _corr(x)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    sm = m + np.log(np.sum(np.exp(s))) / alpha
    sk = np.sort(corr)[::-1][:k].sum()
    return sm + 0.4 * sk + mu * (np.sum(x) - target) ** 2

def _con_eq(x, target):
    return np.sum(x) - target

def _coord_descent(h, dx, n, deadline, max_passes=50000):
    h = h.copy()
    corr = _corr(h)
    current_max = np.max(corr)
    best_c5 = current_max * dx
    best_h = h.copy()
    for _ in range(max_passes):
        if time.time() >= deadline:
            break
        any_imm = False
        for i in range(n):
            a = np.zeros(2 * n - 1)
            for lag in range(-(n - 1), n):
                p = lag + (n - 1)
                c = 0.0
                if 0 <= i + lag < n:
                    c += 1.0 - h[i + lag]
                if 0 <= i - lag < n:
                    c += -h[i - lag]
                a[p] = c
            if np.max(np.abs(a)) < 1e-15:
                continue
            lo = -h[i]
            hi = 1.0 - h[i]
            if hi - lo < 1e-14:
                continue
            res = minimize_scalar(_f_1d, args=(corr, a), bounds=(lo, hi),
                                  method='bounded', options={'xatol': 1e-11})
            d = res.x
            if abs(d) < 1e-13:
                continue
            new_max = np.max(corr + a * d)
            if new_max < current_max - 1e-15:
                corr += a * d
                h[i] = np.clip(h[i] + d, 0.0, 1.0)
                current_max = new_max
                if current_max * dx < best_c5:
                    best_c5 = current_max * dx
                    best_h = h.copy()
                any_imm = True
        if not any_imm:
            break
    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return best_h, best_c5

def _lbfgs_opt(h, n, dx, alpha, maxiter, use_topk):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    if use_topk:
        res = minimize(_topk_obj, h, method='L-BFGS-B', bounds=bounds,
                       args=(alpha, 3, target, 50.0),
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
    else:
        res = minimize(_smooth_obj, h, method='L-BFGS-B', bounds=bounds,
                       args=(alpha, target, 50.0),
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
    hc = _repair(res.x, n)
    return hc, _c5(hc, dx)

def _slsqp_opt(h, n, dx, alpha, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq, 'args': (target,)}]
    res = minimize(_smooth_obj, h, method='SLSQP', bounds=bounds,
                   args=(alpha, target, 0.0), constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-13})
    hc = _repair(res.x, n)
    return hc, _c5(hc, dx)

def _perturb(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        block = rng.randint(3, 10)
        i = rng.randint(0, n - block)
        hp[i:i + block] = np.roll(hp[i:i + block], rng.randint(1, block))
    elif kind == 1:
        f = rng.randint(1, 6)
        ph = rng.uniform(0, 6.283)
        amp = rng.uniform(0.04, 0.18)
        hp += amp * np.sin(2.0 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 2:
        j = rng.randint(n // 5, 4 * n // 5)
        hp = np.concatenate([h[j:], h[:j]])
    elif kind == 3:
        hp += rng.normal(0, 0.06, n)
    elif kind == 4:
        b1 = rng.randint(0, n // 2)
        b2 = b1 + rng.randint(n // 4, n - b1)
        ln = min(rng.randint(2, 7), n - b2)
        tmp = hp[b1:b1 + ln].copy()
        hp[b1:b1 + ln] = hp[b2:b2 + ln]
        hp[b2:b2 + ln] = tmp
    else:
        w = rng.uniform(0.5, 1.5)
        hp = 0.5 + w * (h - 0.5)
    return _repair(hp, n)

def _pairwise_local(h, n, dx, rng, deadline, n_iters=200000):
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)
    corr = _corr(best_h)
    scales = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    per_scale = max(200, n_iters // len(scales))
    for scale in scales:
        imp = False
        for _ in range(per_scale):
            if time.time() >= deadline:
                return best_h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni = best_h[i] + d
            nj = best_h[j] - d
            if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
                continue
            best_h[i], best_h[j] = ni, nj
            nc = _corr(best_h)
            nm = np.max(nc)
            if nm * dx < best_c5 - 1e-14:
                best_c5 = nm * dx
                corr = nc
                imp = True
            else:
                best_h[i], best_h[j] = best_h[i] - d, best_h[j] + d
        if not imp:
            break
    return best_h, best_c5
