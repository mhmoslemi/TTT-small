    # Phase 2: Projected gradient (fast iterations)
    if _el() < budget_s - 250:
        h2, c5 = _proj_grad(best_h, n0, 100000, 5000, 0.005, _dl(budget_s - 250))
        if c5 < best_c5:
            best_c5 = c5; best_h = h2
        h2, c5 = _proj_grad(best_h, n0, 500000, 5000, 0.002, _dl(budget_s - 250))
        if c5 < best_c5:
            best_c5 = c5; best_h = h2

    # Phase 3: Chase-the-max subgradient
    if _el() < budget_s - 200:
        h2, c5 = _chase_max(best_h, n0, 10000, _dl(budget_s - 200))
        if c5 < best_c5:
            best_c5 = c5; best_h = h2

    # Phase 4: Multi-start SLSQP at n=70
    rng = np.random.RandomState(seed)
    if _el() < budget_s - 120:
        for r in range(15):
            if _el() > budget_s - 120: break
            sc = 0.005 + 0.03 * rng.rand()
            hp = _repair(best_h + sc * rng.randn(n0), n0)
            h = _slsqp(hp, n0, 100000, 1500)
            c5 = _eval(h)
            if c5 < best_c5:
                best_c5 = c5; best_h = h.copy()

    # Phase 5: Fine local search
    if _el() < budget_s - 80:
        h2, c5 = _fine_local(best_h, n0, _dl(budget_s - 80), 42, 40000)
        if c5 < best_c5:
            best_c5 = c5; best_h = h2
    if _el() < budget_s - 40:
        h2, c5 = _fine_local(best_h, n0, _dl(budget_s - 40), 137, 30000)
        if c5 < best_c5:
            best_c5 = c5; best_h = h2

    # Phase 6: Try other n values for diversity
    if _el() < budget_s - 60:
        for en in [64, 75, 84, 90]:
            if _el() > budget_s - 45: break
            h_try, c_try = _try_n(en, seed + en, budget_s, t0)
            if h_try is not None and c_try < best_c5:
                best_c5 = c_try; best_h = h_try; best_n = en

    # Phase 7: Final polish
    if _el() < budget_s - 15:
        h = _slsqp(best_h, best_n, 1000000, 2000)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5; best_h = h.copy()
        h2, c5 = _fine_local(best_h, best_n, _dl(budget_s - 8), 999, 15000)
        if c5 < best_c5:
            best_c5 = c5; best_h = h2

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
