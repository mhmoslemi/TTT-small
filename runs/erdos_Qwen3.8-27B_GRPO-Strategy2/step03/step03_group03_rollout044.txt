import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def _grad_pnorm(h, p):
    n = len(h)
    c = _corr(h)
    weights = np.zeros(2 * n - 1)
    if p > 1:
        s = float(np.sum(c ** p))
        if s < 1e-30:
            return np.zeros(n)
        scale = s ** (1.0 / p - 1.0)
        weights = (c ** (p - 1)) * scale
    else:
        mx = float(np.max(c))
        if mx < 1e-15:
            return np.zeros(n)
        weights = np.where(c > mx * 0.99, 1.0, 0.0) / max(1, np.sum(c > mx * 0.99))
    g = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if abs(w) < 1e-18:
            continue
        k = m - (n - 1)
        if k >= 0:
            g[:n - k] += w * (1.0 - h[k:])
            g[k:] -= w * h[:n - k]
        else:
            k2 = -k
            g[k2:] += w * (1.0 - h[:n - k2])
            g[:n - k2] -= w * h[k2:]
    return g

def _pnorm_obj(h, p, target, mu):
    c = _corr(h)
    if p > 1:
        val = float(np.sum(c ** p) ** (1.0 / p))
    else:
        val = float(np.max(c))
    pen = mu * (np.sum(h) - target) ** 2
    return val + pen

def _pnorm_obj_grad(h, p, target, mu):
    c = _corr(h)
    if p > 1:
        s = float(np.sum(c ** p))
        val = s ** (1.0 / p) if s > 0 else 0.0
    else:
        val = float(np.max(c))
    pen = mu * (np.sum(h) - target) ** 2
    g = _grad_pnorm(h, p)
    g += 2.0 * mu * (np.sum(h) - target)
    return val + pen, g

def _subgrad_step(h, n, target, step):
    c = _corr(h)
    mx = c.max()
    active = c > mx - 0.05 * (c.max() - c.min() + 1e-15)
    if not active.any():
        active = c > 0.9 * mx
    g = _grad_pnorm(h, 0)
    return h - step * g

def _fista_pnorm(h0, n, target, p, mu, max_iter, tol):
    h = h0.copy()
    y = h0.copy()
    t_k = 1.0
    best_obj = _pnorm_obj(h0, p, target, mu)
    for it in range(max_iter):
        fy, gy = _pnorm_obj_grad(y, p, target, mu)
        lr = (1.0 + 1e-8) / (1.0 + 1e-6 * it) * 2.0
        y_new = _project(y - lr * gy, target)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = y_new + ((t_k - 1.0) / t_new) * (y_new - h)
        h = y_new
        t_k = t_new
        obj = _pnorm_obj(h, p, target, mu)
        if obj < best_obj:
            best_obj = obj
        if it > 10 and abs(best_obj - _pnorm_obj(h, p, target, mu)) < tol:
            break
    return h, best_obj

def _coord_descent(h, n, target, max_iter):

    best_c5 = _c5_val(h, 2.0 / n)
    no_improve = 0
    for it in range(max_iter):
        c = _corr(h)
        m_star = int(np.argmax(c))
        k = m_star - (n - 1)

        if k >= 0:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
        else:
            k2 = -k

            g[:n - k2] -= h[k2:]
        best_improve = 0.0
        best_i, best_d = -1, 0.0

            if abs(g[i]) < 1e-12:
                continue

                d = min(0.1, h[i])
            else:
                d = min(0.1, 1.0 - h[i])
            improve = abs(g[i]) * d

                best_improve = improve
                best_i = i
                best_d = d
        if best_i < 0:

            h = _project(h, target)

                if c5 < best_c5:
                    best_c5 = c5
                    break
                else:
                    h[i], h[j] = ni - d, nj + d
                    h = _project(h, target)
    return h, best_c5
