import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _smax(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smax(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _slsqp(h, n, target, bounds, alpha, maxiter):
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=[{'type': 'eq', 'fun': _eq_con, 'args': (target,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 30})
        return _repair(res.x, n)
    except Exception:
        return h

def _pair_opt(d_min, d_max, corr, gd, n_grid, n_bisect):
    """Exact 1-D convex piecewise-linear minimization using subgradient bisection."""
    if d_max - d_min < 1e-13:
        return 0.0
    ds = np.linspace(d_min, d_max, n_grid)
    vals = np.max(corr[None, :] + np.outer(ds, gd), axis=1)
    bi = int(np.argmin(vals))
    cur_max = np.max(corr)
    if vals[bi] >= cur_max - 1e-13:
        return 0.0
    lo = ds[max(0, bi - 1)]
    hi_b = ds[min(n_grid - 1, bi + 1)]
    for _ in range(n_bisect):
        mid = 0.5 * (lo + hi_b)
        active = corr + mid * gd
        max_v = np.max(active)
        mask = active >= max_v - 1e-11
        slopes = gd[mask]
        if len(slopes) == 0:
            break
        if slopes.min() > 1e-11:
            hi_b = mid
        elif slopes.max() < -1e-11:
            lo = mid
        else:
            break
    best_d = 0.5 * (lo + hi_b)
    if abs(best_d) < 1e-13:
        return 0.0
    return best_d

def _coord_descent(h, n, corr, max_time, n_grid=50, n_bisect=50):
    improved = False
    n_imp = 0
    order = np.random.permutation(n)
    for ai in range(n):
        i = order[ai]
        if time.time() > max_time:
            break
        hi = h[i]
        gi = _g_vec(h, i, n)
        for aj in range(ai + 1, n):
            j = order[aj]
            hj = h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gd = gi - _g_vec(h, j, n)
            d_best = _pair_opt(d_min, d_max, corr, gd, n_grid, n_bisect)
            if abs(d_best) > 1e-12:
                hi += d_best
                h[i] = hi
                h[j] = hj - d_best
                corr = corr + d_best * gd
                # Update gi since h changed
                gi = _g_vec(h, i, n)
                improved = True
                n_imp += 1
    return h, corr, improved, n_imp

    x = np.linspace(0.0, 2.0, n, endpoint=False)

        h += amp * np.cos(np.pi * m * x)
