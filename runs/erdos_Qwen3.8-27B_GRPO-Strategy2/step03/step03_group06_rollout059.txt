from scipy.optimize import minimize
import time

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    h = h * (target / max(s, 1e-12))
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
    return h

def _to_binary(h, n):
    h = (h >= 0.5).astype(float)
    s = h.sum()
    target = n // 2
    diff = target - int(s)
    if diff > 0:
        zeros = np.where(h == 0)[0]
        h[zeros[:diff]] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        h[ones[:(-diff)]] = 0.0
    return h

def _smooth_max(corr, alpha):
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp_polish(h, n, dx, alpha, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return _smooth_max(corr, alpha) + 100.0 * (np.sum(x) - target)**2
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except:
        return h.copy()

def _binary_sa(h0, n, dx, rng, deadline, T0=0.05, cool_rate=None):

    c5 = _compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = c5
    T = T0
    # Estimate total iterations and set cool rate
    # We want T to drop from T0 to 0.0001 over the phase
    # T * cool^n = 0.0001 => cool = (0.0001/T0)^(1/n)
    # We don't know n yet, so use a fixed rate
    cool = cool_rate if cool_rate else 0.99999
    no_improve = 0
    while time.time() < deadline and T > 0.00005:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if rng.random() < 0.7:
            # Single swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            c5_new = _compute_c5(h, dx)
            if c5_new < c5 or rng.random() < np.exp(-(c5_new - c5) / T):
                c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i], h[j] = 1.0, 0.0
                no_improve += 1
        else:
            # Double swap
            if len(ones) >= 2 and len(zeros) >= 2:
                i1 = ones[rng.randint(len(ones))]
                i2 = ones[rng.randint(len(ones))]
                if i1 == i2:
                    continue
                j1 = zeros[rng.randint(len(zeros))]
                j2 = zeros[rng.randint(len(zeros))]
                if j1 == j2:
                    continue
                h[i1], h[i2], h[j1], h[j2] = 0.0, 0.0, 1.0, 1.0
                c5_new = _compute_c5(h, dx)
                if c5_new < c5 or rng.random() < np.exp(-(c5_new - c5) / T):
                    c5 = c5_new
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    h[i1], h[i2], h[j1], h[j2] = 1.0, 1.0, 0.0, 0.0
                    no_improve += 1
        T *= cool
        # Kick restart
        if no_improve > 5000 and T < T0 * 0.1:
            T = T0 * 0.05
            no_improve = 0
    return best_h, best_c5

def _fine_local(h, n, dx, rng, deadline, n_tries=100000):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= tgt / s
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if abs(d) < 1e-12:
                break
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (tgt / s)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if abs(d) < 1e-12:
                break
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
    return h

def _bin(h, n):
    h = (h >= 0.5).astype(float)
    d = n // 2 - int(h.sum())
    if d > 0:
        z = np.where(h == 0)[0]
        h[z[:d]] = 1.0
    elif d < 0:
        o = np.where(h == 1)[0]
        h[o[:(-d)]] = 0.0
    return h

def _sa(h0, n, dx, rng, deadline, T0):
    h = h0.copy()
    c = _c5(h, dx)
    bh, bc = h.copy(), c
    T = T0
    nstag = 0
    while time.time() < deadline and T > 1e-5:
        o = np.flatnonzero(h)
        z = np.flatnonzero(1.0 - h)
        if len(o) < 1 or len(z) < 1:
            break
        if rng.random() < 0.72:
            i = o[rng.randint(len(o))]
            j = z[rng.randint(len(z))]
            h[i], h[j] = 0.0, 1.0
            cn = _c5(h, dx)
            if cn <= c or rng.random() < np.exp(-(cn - c) / max(T, 1e-8)):
                c = cn
                if cn < bc:

                    nstag += 1

def _obj_fn(x, alpha, tgt):
    cr = np.correlate(x, 1.0 - x, mode='full')
    m = cr.max()
    s = np.clip(alpha * (cr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + 50.0 * (x.sum() - tgt) ** 2
