import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h


def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _obj_alpha(x, alpha):
    return _smooth_max(_corr(x), alpha)


def _eq_target(x, target):
    return np.sum(x) - target


def _do_slsqp(x0, bounds, target, alpha, maxiter):
    res = minimize(_obj_alpha, x0, args=(alpha,), method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': _eq_target, 'args': (target,)}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return res.x


def _delta_c_vec(h, i, j, d, s_arr, n, n_corr):
    dc = np.zeros(n_corr)
    v1i = (i + s_arr >= 0) & (i + s_arr < n)
    v2i = (i - s_arr >= 0) & (i - s_arr < n)
    v1j = (j + s_arr >= 0) & (j + s_arr < n)
    v2j = (j - s_arr >= 0) & (j - s_arr < n)
    if np.any(v1i):
        dc[v1i] += d * (1.0 - h[i + s_arr[v1i]])
    if np.any(v2i):
        dc[v2i] -= d * h[i - s_arr[v2i]]
    if np.any(v1j):
        dc[v1j] -= d * (1.0 - h[j + s_arr[v1j]])
    if np.any(v2j):
        dc[v2j] += d * h[j - s_arr[v2j]]
    return dc
