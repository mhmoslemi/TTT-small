import numpy as np
from scipy.optimize import minimize, linprog
import time

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def c5_value(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def smooth_max(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -400, 400)
    e = np.exp(s)
    total = e.sum()
    weights = e / total
    g = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        i_min = max(0, -lag)
        i_max = min(n, n - lag)
        for i in range(i_min, i_max):
            jp = i + lag
            g[jp] += w * (1.0 - h[i])
            g[i] -= w * h[jp]
    return g * dx

```python
def solve_lp(h_t, n):
    dx = 2.0 / n
    m = 2*n - 1  # number of lags
    # Compute c_k and gradients at h_t
    c_t = np.zeros(m)
    G = np.zeros((m, n))  # G[k, i] = ∂c_k/∂h[i] at h_t
    for k in range(m):
        lag = k - (n - 1)
        if lag >= 0:
            # h[i] * h[i+lag] for i = 0 to n-1-lag
            i_range = range(n - lag)
        else:
            i_range = range(-lag, n)
        # c_k = sum h[i]*(1-h[i+lag]) * dx
        c_t[k] = 0.0
        for i in i_range:
            jp = i + lag
            c_t[k] += h_t[i] * (1 - h_t[jp])
        c_t[k] *= dx
        # Gradient
        for i in i_range:
            jp = i + lag
            G[k, jp] += (1 - h_t[i]) * dx  # ∂/∂h[jp] of h[i]*(1-h[jp])
            G[k, i] -= h_t[jp] * dx        # ∂/∂h[i] of h[i]*(1-h[jp])
    
    # LP: min t
    # s.t. G[k,:] @ (h - h_t) + c_t[k] <= t  =>  G[k,:] @ h - t <= G[k,:] @ h_t - c_t[k]
    #      0 <= h[i] <= 1
    #      sum(h) = n/2
    
    # Variables: [h_0, ..., h_{n-1}, t]
    # Objective: [0, ..., 0, 1] (minimize t)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Inequality constraints: G[k,:] @ h - t <= G[k,:] @ h_t - c_t[k]
    A_ub = np.hstack([G, -np.ones((m, 1))])
    b_ub = G @ h_t - c_t
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(None, None)]  # t is free
    
    from scipy.optimize import linprog
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if result.success:
        return result.x[:n], result.x[n]
    return h_t, c_t.max()
```

import numpy as np
from scipy.optimize import minimize, linprog
import time

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def c5_value(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_cg(h, n):
    dx = 2.0 / n
    m = 2 * n - 1
    c = np.zeros(m)
    G = np.zeros((m, n))
    for k in range(m):
        lag = k - (n - 1)
        lo = max(0, -lag)
        hi = min(n, n - lag)
        idx = np.arange(lo, hi)
        c[k] = np.sum(h[idx] * (1.0 - h[idx + lag])) * dx
        G[k, idx] += (1.0 - h[idx + lag]) * dx
        G[k, idx + lag] -= h[idx] * dx
    return c, G

def solve_lp(h_t, n):
    m = 2 * n - 1
    c_t, G = compute_cg(h_t, n)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.hstack([G, -np.ones((m, 1))])
    b_ub = G @ h_t - c_t
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[n]
    return h_t.copy(), c_t.max()

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    total = e.sum()
    w = e / total
    _, G = compute_cg(x, n)
    return G.T @ w

def slsqp_run(h_init, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    return minimize(smooth_obj, h_init, args=(n, alpha),
                    jac=smooth_grad, method='SLSQP',
                    bounds=bounds, constraints=cons,
                    options={'maxiter': maxiter, 'ftol': 1e-18})

def local_search(h, n, time_limit):
    dx = 2.0 / n
    c5_cur = c5_value(h, n)
    deltas = np.array([0.001, 0.003, 0.01, 0.03, -0.001, -0.003, -0.01, -0.03])
    no_improve = 0
    while time.time() < time_limit and no_improve < 3:
        improved = False
        no_improve += 1
        for i in range(n):
            for j in range(i + 1, n):
                hi, hj = h[i], h[j]
                for d in deltas:
                    ni, nj = hi + d, hj - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c = c5_value(h, n)
                    if c < c5_cur - 1e-15:
                        c5_cur = c
                        hi, hj = ni, nj
                        improved = True
                    else:
                        h[i], h[j] = hi, hj
        if improved:
            no_improve = 0
    return h, c5_cur

def coordinate_descent(h, n, time_limit, rng):
    c5_cur = c5_value(h, n)
    steps = np.array([0.0005, 0.001, 0.002, 0.005, 0.01, -0.0005, -0.001, -0.002, -0.005, -0.01])
    while time.time() < time_limit:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(n - 1)) % n
        d = steps[rng.randint(len(steps))]
        hi, hj = h[i], h[j]
        ni, nj = hi + d, hj - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = c5_value(h, n)
            if c < c5_cur:
                c5_cur = c
            else:
                h[i], h[j] = hi, hj
    return h, c5_cur
