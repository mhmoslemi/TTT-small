    # Phase B: Upsample to 2x and optimize
    if _el() < budget_s - 400:
        n1 = n0 * 2
        h_hi = _upsample(best_h, 2)
        h_hi = _repair(h_hi, n1)
        for alpha in [500, 5000, 50000, 200000]:
            if _el() > budget_s - 250:
                break
            h_hi = _slsqp(h_hi, alpha, 1500, 1e-19)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1

        # Projected gradient at high resolution
        if _el() < budget_s - 200:
            h_hi = _proj_grad(best_h, 100000, 300, 0.003)
            h_hi = _repair(h_hi, n1)
            c5 = _eval(h_hi)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hi.copy()
                best_n = n1

    # Phase C: Multi-start at best resolution
    n_cur = best_n
    if _el() < budget_s - 200:
        for _r in range(15):
            if _el() > budget_s - 200:
                break
            strat = rng.randint(4)
            if strat == 0:
                hp = _repair(best_h + 0.015 * rng.randn(n_cur), n_cur)
            elif strat == 1:
                hp = _repair(1.0 - best_h + 0.01 * rng.randn(n_cur), n_cur)
            elif strat == 2:
                hp = best_h.copy()
                for _b in range(rng.randint(2, 4)):
                    bs = rng.randint(3, max(4, n_cur // 10))
                    st = rng.randint(0, n_cur - bs)
                    hp[st:st+bs] += 0.04 * rng.randn(bs)
                hp = _repair(hp, n_cur)
            else:
                # Frequency-domain perturbation
                hp = best_h.copy()
                freqs = rng.randint(1, n_cur // 2 + 1, size=3)
                for f in freqs:
                    amp = 0.005 * rng.randn()
                    t = np.arange(n_cur)
                    hp += amp * np.sin(2 * np.pi * f * t / n_cur)
                hp = _repair(hp, n_cur)
            h_new = _slsqp(hp, 200000, 1000, 1e-19)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = n_cur

    # Phase D: Lag-targeted descent
    if _el() < budget_s - 100:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        dx = 2.0 / nc
        for _round in range(100):
            if _el() > budget_s - 100:
                break
            c = np.correlate(h, 1.0 - h, mode='full')
            k_w = c.argmax()
            sh = (nc - 1) - k_w
            g = np.zeros(nc)
            if sh >= 0:
                i1 = np.arange(nc - sh)
                g[i1] += (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, nc)
                    g[i2] -= h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, nc)
                g[i1] += (1.0 - h[i1 - sp])
                i2 = np.arange(nc - sp)
                g[i2] -= h[i2 + sp]
            g *= dx
            for step in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
                h_new = _proj_to_sum(h - step * g, nc * 0.5)
                c5n = _eval(h_new)
                if c5n < c5c - 1e-15:
                    h = h_new
                    c5c = c5n
                    if c5c < best_c5:
                        best_c5 = c5c
                        best_h = h.copy()
                        best_n = nc
                    break

    # Phase E: Fine random pairwise search
    if _el() < budget_s - 40:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        for scale in [0.005, 0.002, 0.0005, 0.0001]:
            if _el() > budget_s - 40:
                break
            for _it in range(8000):
                if _el() > budget_s - 40:
                    break
                i = rng.randint(nc)
                j = rng.randint(nc)
                if i == j:
                    continue
                d = scale * rng.randn()
                ni, nj = h[i] + d, h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-16:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                        best_n = nc
                else:
                    h[i], h[j] = ni - d, nj + d

    # Phase F: Final SLSQP polish
    if _el() < budget_s - 10:
        h = _slsqp(best_h, 500000, 3000, 1e-20)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = len(h)

    # Phase G: Ultra-fine touch
    if _el() < budget_s - 3:
        h = best_h.copy()
        nc = len(h)
        c5c = _eval(h)
        for _it in range(15000):
            if _el() > budget_s - 3:
                break
            i = rng.randint(nc)
            j = rng.randint(nc)
            if i == j:
                continue
            d = 5e-5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-17:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                        best_n = nc
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
