import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return h

def _grad_lag(h, n, m):
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        nk = n - k
        g[k:] += (1.0 - h[:nk])
        g[:nk] -= h[k:]
    else:
        nk = n + k
        g[:nk] += (1.0 - h[nk:])
        g[nk:] -= h[:nk]
    return g

def _smooth_obj(x, alpha, target):
    c = _corr(x)
    m = float(c.max())
    shifted = np.clip(alpha * (c - m), -50, 50)
    val = m + np.log(np.sum(np.exp(shifted))) / alpha
    val += 100.0 * (np.sum(x) - target) ** 2
    return val

def _block_perturb(h, rng, n, scale):

    bs = rng.randint(0, max(1, n - 6))
    bl = 3 + rng.randint(4)
    h_p[bs:bs+bl] += rng.randn(min(bl, n - bs)) * scale
    return _repair(h_p, n)
