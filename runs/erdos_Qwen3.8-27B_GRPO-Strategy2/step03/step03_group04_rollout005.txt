            if v1 < v2:
                hi = m2
            else:
                lo = m1
        x_opt = (lo + hi) / 2
        v_opt = np.max(alphas * x_opt + betas)
        if v_opt < best_val:
            best_x = x_opt
            best_val = v_opt
    else:
        best_x = xs[idx]
        best_val = vals[idx]
    
    return best_x, best_val

Wait, but this is approximate because the "alphas" calculation assumes linearity, which fails for k=n-1 (where h[p]² appears). Let me handle that case.

For k = n-1: c_{n-1} = sum_i h[i]*(1-h[i])
If we change h[p]: c_{n-1} changes by delta*(1-h[p]) - delta*h[p] = delta*(1-2h[p])
But this is only valid for an infinitesimal delta. For finite changes, the change is:
h[p+delta]*(1-h[p+delta]) - h[p]*(1-h[p]) = (h[p]+delta)(1-h[p]-delta) - h[p](1-h[p])
= h[p] - h[p]² + delta - delta*h[p] - delta*h[p] + delta² - delta²... 

Let me just compute: (x+d)(1-x-d) - x(1-x) = (x+d)(1-x) - (x+d)d - x(1-x) = (x+d-x)(1-x) - d(x+d) = d(1-x) - d(x+d) = d(1-2x-d)

So for k=n-1, the change in c is d*(1-2*h[p]-d), which is quadratic in d.

For other k, the change is d*(1-h[j]) or d*(-h[i]), which is linear in d.

So my linear approximation is exact for all k ≠ n-1, and slightly off for k = n-1 (by the d² term). For small d, this is negligible.

Let me just use the linear approximation and handle the n-1 lag specially if needed.

import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)


def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h


def _grad_weighted(h, lags, weights, n):
    g = np.zeros(n)
    center = n - 1
    for m, w in zip(lags, weights):
        for i in range(n):
            j = i + m - center
            if 0 <= j < n:
                g[i] += w * (1.0 - h[j])
                g[j] -= w * h[i]
    return g


def _obj_alpha(x, alpha):
    return _smooth_max(_corr(x), alpha)


def _eq_target(x, target):
    return np.sum(x) - target


def _do_slsqp(x0, bounds, target, alpha, maxiter, ftol):
    res = minimize(_obj_alpha, x0, args=(alpha,), method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': _eq_target, 'args': (target,)}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x


def _coord_opt(h, p, n):
    """Optimize coordinate p holding others fixed via 1D search."""
    c = _corr(h)
    center = n - 1
    alphas = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        a = 0.0
        j = p + k - center
        if 0 <= j < n:
            a += (1.0 - h[j])
        i = p - k + center
        if 0 <= i < n:
            a -= h[i]
        alphas[k] = a
    betas = c - alphas * h[p]
    xs = np.linspace(0.0, 1.0, 201)
    vals = np.array([np.max(alphas * x + betas) for x in xs])
    idx = int(np.argmin(vals))
    lo = xs[max(idx - 2, 0)]
    hi = xs[min(idx + 2, 200)]
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(alphas * m1 + betas) < np.max(alphas * m2 + betas):
            hi = m2
        else:
            lo = m1
    x_opt = (lo + hi) / 2.0
    return x_opt


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def elapsed():
        return time.time() - t0

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    h1 = _repair(1.0 - h0, n)
    c5_1 = _c5(h1, dx)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 1: SLSQP with increasing smoothness
    for alpha in [500, 2000, 8000, 30000, 100000, 400000]:
        if elapsed() > budget_s - 350:
            break
        for att in range(2):
            if elapsed() > budget_s - 350:
                break
            sc = 0.0 if att == 0 else 0.008
            x0 = _repair(best_h + sc * rng.randn(n), n)
            try:
                rx = _do_slsqp(x0, bounds, target, alpha, 2500, 1e-15)
                hc = _repair(rx, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc.copy()
            except Exception:
                pass

    # Phase 2: Weighted gradient descent with adaptive step
    if elapsed() < budget_s - 250:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        for step_base in [0.04, 0.015, 0.005, 0.002]:
            for it in range(2000):
                if elapsed() > budget_s - 250:
                    break
                c = _corr(h)
                cmx = float(np.max(c))
                top_n = 10
                top_idx = np.argsort(c)[-top_n:]
                top_vals = c[top_idx]
                w = np.exp(-80.0 * (cmx - top_vals))
                wsum = w.sum()
                if wsum < 1e-30:
                    w = np.ones(top_n) / top_n
                else:
                    w = w / wsum
                g = _grad_weighted(h, top_idx, w, n)
                gn = float(np.linalg.norm(g))
                if gn < 1e-14:
                    break
                step = step_base / gn
                h_new = _repair(h - step * g, n)
                c5_new = _c5(h_new, dx)
                if c5_new < c5_cur - 1e-15:
                    h, c5_cur = h_new, c5_new
                    if c5_cur < best_c5:
                        best_c5, best_h = c5_cur, h.copy()
                elif it > 200 and it % 50 == 0:
                    break

    # Phase 3: Coordinate descent (exact 1D optimization per coordinate)
    if elapsed() < budget_s - 150:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        for _round in range(80):
            if elapsed() > budget_s - 150:
                break
            imp = False
            for p in range(n):
                if elapsed() > budget_s - 150:
                    break
                x_opt = _coord_opt(h, p, n)
                if abs(x_opt - h[p]) > 1e-8:
                    h[p] = x_opt
                    h = _repair(h, n)
                    c5_new = _c5(h, dx)
                    if c5_new < c5_cur - 1e-14:
                        c5_cur = c5_new
                        imp = True
                        if c5_cur < best_c5:
                            best_c5, best_h = c5_cur, h.copy()
                    else:
                        h[p] = _repair(h, n)[p]  # keep it
            if c5_cur < best_c5:
                best_c5, best_h = c5_cur, h.copy()
            if not imp:
                break

    # Phase 4: Targeted pair search on critical lags
    if elapsed() < budget_s - 80:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        deltas = np.array([0.003, 0.001, 0.0004, -0.003, -0.001, -0.0004])
        for _round in range(300):
            if elapsed() > budget_s - 80:
                break
            c = _corr(h)
            top_k = np.argsort(c)[-6:]
            imp = False
            center = n - 1
            for m in top_k:
                if elapsed() > budget_s - 80:
                    break
                for i in range(n):
                    j = i + m - center
                    if 0 <= j < n and i != j:
                        hi, hj = h[i], h[j]
                        for d in deltas:
                            ni, nj = hi + d, hj - d
                            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                                continue
                            h[i], h[j] = ni, nj
                            cv = _c5(h, dx)
                            if cv < c5_cur - 1e-14:
                                c5_cur, hi, hj = cv, ni, nj
                                imp = True
                            else:
                                h[i], h[j] = hi, hj
                        h[i], h[j] = hi, hj
            if c5_cur < best_c5:
                best_c5, best_h = c5_cur, h.copy()
            if not imp:
                break

    # Phase 5: Full pairwise local search
    if elapsed() < budget_s - 40:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        deltas = np.array([0.0015, 0.0006, 0.0002, 0.0001,
                           -0.0015, -0.0006, -0.0002, -0.0001])
        for _pass in range(80):
            if elapsed() > budget_s - 40:
                break
            imp = False
            for i in range(n):
                if elapsed() > budget_s - 40:
                    break
                hi = h[i]
                for j in range(i + 1, n):
                    hj = h[j]
                    for d in deltas:
                        ni, nj = hi + d, hj - d
                        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                            continue
                        h[i], h[j] = ni, nj
                        cv = _c5(h, dx)
                        if cv < c5_cur - 1e-14:
                            c5_cur, hi, hj = cv, ni, nj
                            imp = True
                        else:
                            h[i], h[j] = hi, hj
                    h[i], h[j] = hi, hj
                h[i] = hi
            if c5_cur < best_c5:

            if not imp:
                break

    # Phase 6: Fine random search
    if elapsed() < budget_s - 10:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        d_opts = np.array([5e-5, 1e-4, 2e-4, 5e-4, 1e-3,

        while elapsed() < budget_s - 10:

            j = rng.randint(n)
            if i == j:

            d = d_opts[rng.randint(len(d_opts))]
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                cv = _c5(h, dx)
                if cv < c5_cur:

            rx = _do_slsqp(best_h, bounds, target, 500000, 5000, 1e-16)
            hc = _repair(rx, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:

        except Exception:
            pass

    # Phase 8: Try 1-best_h variant

        try:

            rx = _do_slsqp(hf, bounds, target, 100000, 2000, 1e-16)
            hc = _repair(rx, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
