def _lag_grad(h, n, L):
    """Gradient of correlation at lag L (L = k - (n-1))"""
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += (1.0 - h[:n - L])
        g[:n - L] -= h[L:]
    else:
        M = -L
        g[:n - M] += (1.0 - h[M:])
        g[M:] -= h[:n - M]
    return g

def _softobj(x, alpha, lam, n):

    m = np.max(c)
    s = np.clip(alpha * (c - m), -200, 200)
    return m + np.log(np.sum(np.exp(s))) / alpha + lam * (np.sum(x) - n/2.0)**2

def _softgrad(x, alpha, lam, n):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -200, 200)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)

        L = k - (n-1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        g += pk * _lag_grad(x, n, L)
    g /= alpha
    g += 2.0 * lam * (np.sum(x) - n/2.0)
    return g

def _minmax_lp(h, n, top_k):
    """Solve linearized min-max: min t s.t. c_k(h)+grad_k·d <= t, sum(d)=0, -h<=d<=1-h"""
    c = _corr(h)
    idx = np.argsort(c)[::-1][:top_k]
    # Variables: d (n), t (1). Total n+1 variables.
    # Constraints for each active lag k:
    #   c_k + grad_k · d <= t
    #   => grad_k · d - t <= -c_k
    # Sum: sum(d) = 0
    # Bounds: -h[i] <= d[i] <= 1-h[i]
    # t >= 0 (or some lower bound)
    
    m = len(idx)
    # c matrix: m x (n+1)
    cmat = np.zeros((m, n+1))
    bvec = np.zeros(m)
    for ii, ki in enumerate(idx):
        L = ki - (n-1)
        gk = _lag_grad(h, n, L)
        cmat[ii, :n] = gk
        cmat[ii, n] = -1.0  # -t
        bvec[ii] = -c[ki]
    
    # A_eq: sum(d) = 0
    Aeq = np.zeros((1, n+1))
    Aeq[0, :n] = 1.0
    beq = np.zeros(1)
    
    # Bounds: -h[i] <= d[i] <= 1-h[i], t >= 0 (use -1e6 to inf)
    lb = np.concatenate([-h, [-1e6]])
    ub = np.concatenate([1.0 - h, [1e6]])
    
    # Objective: minimize t => [0, 0, ..., 0, 1]
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0
    
    from scipy.optimize import linprog
    res = linprog(c_obj, A_ub=cmat, b_ub=bvec, A_eq=Aeq, b_eq=beq,
                  bounds=list(zip(lb, ub)), method='highs')
    if res.success:
        d = res.x[:n]
        h_new = _repair(h + d, n)
        return h_new
    return h

    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        return np.full(n, 0.5)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _lag_grad(h, n, L):
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += (1.0 - h[:n - L])
        g[:n - L] -= h[L:]
    else:
        M = -L
        g[:n - M] += (1.0 - h[M:])
        g[M:] -= h[:n - M]
    return g

def _softobj(x, alpha, n):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -300, 300)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _softgrad(x, alpha, n):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -300, 300)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        g += pk * _lag_grad(x, n, L)
    g /= alpha
    return g

    A_eq = np.zeros((1, n + 1))

    c_obj = np.zeros(n + 1)
