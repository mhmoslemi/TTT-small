_state = {}

def _obj(v):
    corr = np.correlate(v, 1.0 - v, mode='full')
    m = np.max(corr)
    a = _state['alpha']
    s = a * (corr - m)
    s = np.clip(s, -80, 80)
    val = m + np.log(np.sum(np.exp(s))) / a
    tk = _state['topk']
    if tk > 1:
        idx = np.argsort(corr)[::-1][:tk]
        sc = corr[idx]
        val += _state['fw'] * np.sum((sc - sc.mean())**2)
    val += _state['pen'] * (v.sum() - _state['target'])**2
    return val

def _eq(v):
    return v.sum() - _state['target']

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a; d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a; d += a
    return h

def do_slsqp(h, n, alpha, topk, fw, pen, maxiter):
    _state.update({'alpha': alpha, 'topk': topk, 'fw': fw, 'pen': pen, 'target': n/2.0})

    cons = [{'type': 'eq', 'fun': _eq}]

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})

def targeted_cd(h, n, t_end, rng):
    """Coordinate descent targeting worst correlations with SA."""
    h = h.copy()
    best = compute_c5(h, n)
    dx = 2.0 / n

    T_sa = 0.002  # simulated annealing temperature
    no_improve = 0
    
    while time.time() < t_end:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k_shift = max_idx - (n - 1)

            # Pairs (i, i+k_shift) contributing most

                c = h[i] * (1.0 - h[i + k_shift])

                        ni, nj = h[i] - d, h[i + k_shift] + d
                        if 0 <= ni <= 1 and 0 <= nj <= 1:
                            h[i], h[i + k_shift] = ni, nj
                            c5 = compute_c5(h, n)
                            if c5 < best - 1e-15:
                                best = c5; no_improve = 0; break
                            elif rng.random() < np.exp(-(c5-best)/T_sa):
                                no_improve = 0; break
                            h[i], h[i + k_shift] = h[i]+d, h[i+k_shift]-d
                    if c5 < best + 0.001:  # accepted

            j = rng.randint(n)

                break
    return h, best
