def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    for _ in range(10000):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        d = target - s
        if d > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h += d / n
            h = np.clip(h, 0.0, 1.0)
            continue
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_val_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    M = np.max(corr)
    s = alpha * (corr - M)
    s = np.clip(s, -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    val = M + np.log(e.sum()) / alpha

        if wi < 1e-14:
            continue

            lag = idx - (n - 1)
            for i in range(n - lag):
                grad[i + lag] += wi * (1.0 - h[i])
                grad[i] -= wi * h[i + lag]

            for i in range(lag, n):
                grad[i] += wi * (1.0 - h[i - lag])
