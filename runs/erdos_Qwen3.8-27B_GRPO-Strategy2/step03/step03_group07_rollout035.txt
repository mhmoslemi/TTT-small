    i = np.arange(n)

Let me also think about one more optimization trick: **correlation-aware coordinate descent**. Instead of random pairs, choose pairs that are likely to reduce the max correlation.

For the maximizing shift k*, the correlation is:
c[k*] = sum_j h[j] * (1-h[j+shift])

To reduce c[k*], we want to:
- Decrease h[j] where (1-h[j+shift]) is large (i.e., h[j+shift] is small)
- Decrease h[j+shift] where h[j] is large
- Or increase h[j+shift] where (1-h[j+shift]) is large (to make the product smaller)

A simple heuristic: for the maximizing shift k*, find the j that maximizes h[j]*(1-h[j+shift]), and try to reduce either h[j] or h[j+shift].

Let me implement this as a targeted coordinate descent:

```python
def targeted_cd(h, n, n_iters, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    corr = compute_corr(h)
    for it in range(n_iters):
        if time.time() > t_end: break
        # Find maximizing shift
        k_star = np.argmax(corr)
        shift = k_star - (n - 1)
        # Find j that maximizes contribution to c[k_star]
        # c[k_star] = sum_j h[j] * (1-h[j+shift])
        if shift >= 0:
            js = np.arange(0, n - shift)
            contrib = h[js] * (1 - h[js + shift])
        else:
            s2 = -shift
            js = np.arange(s2, n)
            contrib = h[js] * (1 - h[js - s2])
        
        j_star = js[np.argmax(contrib)]
        # Move mass: reduce h[j_star] or h[j_star + shift]
        partner = j_star + shift
        if partner < 0 or partner >= n:
            partner = rng.randint(n)
        
        # Line search
        i, j = j_star, partner
        hi, hj = h[i], h[j]
        lo, hi_l = -hj + 1e-12, 1.0 - hi - 1e-12
        if lo >= hi_l:
            i, j = j, i
            hi, hj = h[i], h[j]
            lo, hi_l = -hj + 1e-12, 1.0 - hi - 1e-12
            if lo >= hi_l: continue
        
        for _ in range(15):
            mid = 0.5 * (lo + hi_l)
            h[i], h[j] = hi + mid, hj - mid
            v = compute_c5(h, n)
            if v < best: best = v
            h[i], h[j] = hi, hj
            mid1, mid2 = 0.5*(lo+mid), 0.5*(mid+hi_l)
            h[i], h[j] = hi+mid1, hj-mid1
            v1 = compute_c5(h, n)
            h[i], h[j] = hi+mid2, hj-mid2
            v2 = compute_c5(h, n)
            h[i], h[j] = hi, hj
            if v1 <= v2: hi_l = mid
            else: lo = mid
    
    return h, best

If I increase h[m], which term changes?
- The term where j=m: h[m]*(1-h[m+sh]) → increases h[m], contribution increases by (1-h[m+sh])
- The term where j+sh=m, i.e., j=m-sh: h[m-sh]*(1-h[m]) → h[m] increases, (1-h[m]) decreases, contribution decreases by h[m-sh]

So ∂c_k/∂h[m] = (1-h[m+sh]) - h[m-sh] (when both indices valid).

If h[m+sh] is small (say 0), then (1-h[m+sh]) = 1, and the first term is positive.
If h[m-sh] is large (say 1), then h[m-sh] = 1, and the second term is -1.

So if h[m+sh]=0 and h[m-sh]=1, the gradient is 1-1=0.
If h[m+sh]=0 and h[m-sh]=0, the gradient is 1-0=1 (increase h[m] helps).
If h[m+sh]=1 and h[m-sh]=0, the gradient is 0-0=0.
If h[m+sh]=1 and h[m-sh]=1, the gradient is 0-1=-1 (increase h[m] hurts).

This makes sense. The gradient tells us how changing h[m] affects the correlation at shift sh.

def coord_ls(h, n, n_iters, t_end, rng, n_ls=15):
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(n_iters):
        if time.time() > t_end: break
        i, j = rng.randint(n), rng.randint(n)
        if i == j: continue
        hi, hj = h[i], h[j]
        lo, hi_l = -hj + 1e-12, 1.0 - hi - 1e-12
        if lo >= hi_l: continue
        for _ in range(n_ls):

            h[i], h[j] = hi + mid, hj - mid
            v = compute_c5(h, n)
            if v < best: best = v
            h[i], h[j] = hi, hj

            v2 = compute_c5(h, n)
            h[i], h[j] = hi, hj
            hi_l, lo = (mid, lo) if v1 <= v2 else (lo, mid)
    return h, best

def targeted_cd(h, n, n_iters, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(n_iters):
        if time.time() > t_end: break
        corr = compute_corr(h)
        k_star = int(np.argmax(corr))
        sh = k_star - (n - 1)
        if sh >= 0:
            js = np.arange(n - sh)
            contrib = h[js] * (1.0 - h[js + sh])
        else:
            s2 = -sh
            js = np.arange(s2, n)
            contrib = h[js] * (1.0 - h[js - s2])
        j_star = js[int(np.argmax(contrib))]
        partner = j_star + sh
        if partner < 0 or partner >= n:
            partner = rng.randint(n)
        # Pair move with line search
        i, j = j_star, partner
        hi, hj = h[i], h[j]
        lo, hi_l = -hj + 1e-12, 1.0 - hi - 1e-12
        if lo >= hi_l:
            i, j = j, i
            hi, hj = h[i], h[j]
            lo, hi_l = -hj + 1e-12, 1.0 - hi - 1e-12
            if lo >= hi_l: continue
        for _ in range(12):
            mid = 0.5 * (lo + hi_l)
            h[i], h[j] = hi + mid, hj - mid
            v = compute_c5(h, n)
            if v < best: best = v
            h[i], h[j] = hi, hj
            m1, m2 = 0.5*(lo+mid), 0.5*(mid+hi_l)
            h[i], h[j] = hi+m1, hj-m1
            v1 = compute_c5(h, n)
            h[i], h[j] = hi+m2, hj-m2
            v2 = compute_c5(h, n)
            h[i], h[j] = hi, hj
            if v1 <= v2: hi_l = mid
            else: lo = mid
    return h, best

def anneal(h, n, t_cool_start, t_end, rng):
    h = h.copy()
    cur_v = compute_c5(h, n)
    best_h, best_v = h.copy(), cur_v
    T0, T1 = 0.04, 0.00005
    total_t = t_end - t_cool_start
    for _ in range(300000):
        if time.time() > t_end: break
        elapsed = time.time() - t_cool_start
        frac = min(1.0, elapsed / max(1.0, total_t))
        T = T0 * (T1 / T0) ** frac
        np_ = rng.randint(1, 5)
        idxs = rng.choice(n, np_, replace=False)
        h_save = h[idxs].copy()
        h[idxs] = np.clip(h[idxs] + 0.2 * rng.randn(np_), 0, 1)
        h = repair(h, n)
        v_new = compute_c5(h, n)
        if v_new <= cur_v or rng.random() < np.exp(-max(0, v_new - cur_v) / max(T, 1e-12)):
            cur_v = v_new
            if v_new < best_v:
                best_v = v_new
                best_h = h.copy()
        else:
            h[idxs] = h_save
            h = repair(h, n)
    return best_h, best_v
```

def harmonic_eval(coeffs, n, K):
    i = np.arange(n)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        a, b = coeffs[2*(k-1)], coeffs[2*(k-1)+1]
        h += a * np.cos(2*np.pi*k*i/n) + b * np.sin(2*np.pi*k*i/n)
    return h

def harmonic_opt(n, K, rng, t_end, n_trials=3):
    best_h, best_v = repair(np.full(n, 0.5), n), compute_c5(np.full(n, 0.5), n)
    for _ in range(n_trials):

        def hfun(c):

            v2 = compute_c5(h2, n)
            if v2 < best_v: best_h, best_v = h2, v2

def make_start(n, seed_val, kind):
