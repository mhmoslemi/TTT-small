        h_t, c5_t = _intensive_ls(best_h, n, dx, rng, t_end, 200000)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 5: Triple moves
    if rem() > 150:
        t_end = time.time() + min(80, rem() - 120)
        h_t, c5_t = _triple_ls(best_h, n, dx, rng, t_end, 40000)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 6: SLSQP final polish
    if rem() > 80:
        t_end = time.time() + min(60, rem() - 50)
        for a in [50000, 150000]:
            if time.time() >= t_end:
                break
            h_t, c5_t = _slsqp(best_h, n, dx, a, 250)
            if c5_t < best_c5:
                best_h, best_c5 = h_t, c5_t

    # Phase 7: Final intensive pair search
    if rem() > 30:
        t_end = time.time() + max(3, rem() - 5)
        h_t, c5_t = _intensive_ls(best_h, n, dx, rng, t_end, 100000)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
