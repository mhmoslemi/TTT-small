        c = _corr(x)
        _, w = _sm_grad(c, alpha)
        return _corr_grad_weighted(x, w)
    def cefun(x):
        return np.sum(x) - target
    def cefun_jac(x):
        return np.ones(len(x))
    res = minimize(fun, h0, method='SLSQP', jac=jac, bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': cefun, 'jac': cefun_jac}],
                   options={'maxiter': maxit, 'ftol': 1e-18})
    return res.x


def _subgrad_top(h, lags, n):
    nm1 = n - 1
    g = np.zeros(n)
    for li in lags:
        lag = li - nm1
        nv = n - abs(lag)
        if lag >= 0:
            for i in range(nv):
                g[i + lag] += (1.0 - h[i])
                g[i] -= h[i + lag]
        else:
            d = -lag
            for i in range(nv):
                g[i] += (1.0 - h[i + d])
                g[i + d] -= h[i]
    return g / len(lags)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def tl(need):
        return (time.time() - t0) + need < budget_s

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)
    h1 = _repair(1.0 - h0, n)
    c5_1 = _c5(h1, dx)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 1: SLSQP with analytical gradient, alpha sweep
    for alpha in [120, 500, 2500, 10000, 40000]:
        if not tl(300):
            break
        for att in range(3):
            if not tl(240):
                break
            sc = 0.0 if att == 0 else 0.004 * (att + 1)
            x0 = _repair(best_h + sc * rng.randn(n), n)
            try:
                rx = _slsqp_solve(x0, bounds, target, alpha, 900)
                hc = _repair(rx, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc.copy()
            except Exception:
                pass

    # Phase 2: Momentum subgradient with adaptive K
    if tl(280):
        h = best_h.copy()
        c5c = _c5(h, dx)
        v = np.zeros(n)
        mom = 0.65
        step = 0.025
        tot = 2 * n - 1
        for ph in range(15):
            if not tl(80):
                break
            K = min(2 + 2 * ph, tot)
            s2 = step * (0.55 ** (ph // 5))
            for it in range(500):
                if not tl(60):
                    break
                c = _corr(h)
                top = np.argsort(c)[-K:]
                g = _subgrad_top(h, top, n)
                v = mom * v + g
                h_new = _repair(h - s2 * v, n)
                c5n = _c5(h_new, dx)
                if c5n < c5c - 1e-16:
                    h, c5c = h_new, c5n
                    step = min(step * 1.04, 0.09)
                    if c5c < best_c5:
                        best_c5, best_h = c5c, h.copy()
                else:
                    v *= 0.4
                    step *= 0.75
                    if step < 0.001:
                        break

    # Phase 3: Exhaustive pair search with multi-scale deltas
    if tl(200):
        h = best_h.copy()
        c5c = _c5(h, dx)
        deltas = np.array([0.006, 0.0025, 0.0008, 0.0002, -0.006, -0.0025, -0.0008, -0.0002])
        for _pass in range(400):
            if not tl(60):
                break
            imp = False
            for i in range(n):
                if not tl(55):
                    break
                hi = h[i]
                for j in range(i + 1, n):
                    hj = h[j]
                    for d in deltas:
                        ni, nj = hi + d, hj - d
                        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                            continue
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < c5c - 1e-14:
                            c5c = c
                            hi, hj = ni, nj
                            imp = True
                        else:
                            h[i], h[j] = hi, hj
                    h[i], h[j] = hi, hj
                h[i] = hi
            if c5c < best_c5:
                best_c5, best_h = c5c, h.copy()
            if not imp:
                # Try perturbation to escape
                h = _repair(h + 0.003 * rng.randn(n), n)
                c5c = _c5(h, dx)
                if c5c < best_c5:
                    best_c5, best_h = c5c, h.copy()

    # Phase 4: Fine random pair search
    if tl(60):
        h = best_h.copy()
        c5c = _c5(h, dx)
        opts = np.array([3e-4, 7e-4, 1.5e-3, 3e-3, -3e-4, -7e-4, -1.5e-3, -3e-3])
        while tl(15):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = opts[rng.randint(len(opts))]
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5c:
                    c5c = c
                    if c5c < best_c5:
                        best_c5, best_h = c5c, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    # Phase 5: Final SLSQP polish
    if tl(4):
        try:
            rx = _slsqp_solve(best_h, bounds, target, 50000, 600)
            hc = _repair(rx, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    # Phase 6: Symmetry + final SLSQP
    if tl(3):
        try:
            hf = _repair(1.0 - best_h, n)
            rx = _slsqp_solve(hf, bounds, target, 20000, 400)
            hc = _repair(rx, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
