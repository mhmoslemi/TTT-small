    _state['alpha'] = alpha
    _state['lam'] = lam
    _state['target'] = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def eq_c(h):
        return h.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': eq_c}]
    try:
        res = minimize(obj_sm, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(res.x, n)
    except Exception:
        return h

def coord_descent(h, n, n_rounds, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(5 * n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for d in steps:
                for sign in (1, -1):
                    delta = sign * d
                    ni, nj = hi + delta, hj - delta
                    if ni < -1e-10 or ni > 1+1e-10 or nj < -1e-10 or nj > 1+1e-10:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-15:
                        best = c5
                        improved = True
                        break
                    h[i], h[j] = hi, hj
                if improved:
                    break
        if not improved and r > 80:
            break
    return h, best

def simulated_anneal(h, n, t_end, rng, n_restarts=10):
    h = h.copy()
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    T_start = 0.1
    T_end = 0.0001
    steps = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    n_total = 50000
    t0 = time.time()
    for it in range(n_total):
        if time.time() > t_end:
            break
        T = T_start * (T_end / T_start) ** (it / n_total)
        i = rng.randint(n)
        d = steps[rng.randint(len(steps))] * (1 if rng.rand() < 0.5 else -1)
        hi = h[i]
        h[i] = hi + d
        if h[i] < 0 or h[i] > 1:
            h[i] = hi
            continue
        # Adjust another coordinate to maintain sum
        j = rng.randint(n)
        if j == i:
            continue
        diff = h[i] - hi
        h[j] -= diff
        if h[j] < 0 or h[j] > 1:
            h[i] = hi
            h[j] += diff
            continue
        c5_new = compute_c5(h, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        else:
            delta = c5_new - best_c5
            if rng.rand() < np.exp(-delta / max(T, 1e-10)):
                pass  # accept
            else:
                h[i] = hi
                h[j] += diff
    return best_h, best_c5
