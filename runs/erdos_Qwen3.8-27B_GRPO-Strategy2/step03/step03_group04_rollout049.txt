```python
def _guided_move(h, active, n, dx):
    """For each active lag, try to reduce its correlation by pair moves."""
    c5_cur = _c5(h, dx)
    c = _corr(h)
    improved = False
    for m in active:
        s = m - (n - 1)
        if s == 0:
            continue  # Can't do pair move for self-correlation
        for i in range(n):
            j = i + s
            if j >= n or j < 0:
                continue
            # Try to reduce C[m]: increase h[i] and decrease h[j] (or vice versa)
            # C[m] contains h[i]*(1-h[j]) term
            # Derivative w.r.t. h[i]: (1-h[j]) + contribution from other terms
            # For a small change: ΔC ≈ (1-h[j])*d_i - h[i]*d_j - (1-h[i])*d_j + h[j]*d_i
            # = (1-h[j]+h[j])*d_i + (-h[i]-(1-h[i]))*d_j  -- wait this isn't right for lag m
            # Actually for lag m, the term h[i]*(1-h[j]) where j=i+s:
            # ∂/∂h[i] = (1-h[j]), ∂/∂h[j] = -h[i]
            # So to decrease C[m], increase h[i] (if h[j]>0) or decrease h[j] (if h[i]<1)
            for d in [0.001, 0.002, 0.005, -0.001, -0.002, -0.005]:
                ni = h[i] + d
                nj = h[j] - d  # keep sum constant
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c5_new = _c5(h, dx)
                    if c5_new < c5_cur - 1e-15:
                        c5_cur = c5_new
                        improved = True
                    else:
                        h[i], h[j] = ni - d, nj + d
    return h, c5_cur, improved
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h


def _active_grad(h, active, n):
    g = np.zeros(n)
    for m in active:
        s = m - (n - 1)
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                g[i] += (1.0 - h[j])
                g[j] -= h[i]
    return g / len(active)


def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _obj_smooth(x, alpha):
    return _smooth_max(_corr(x), alpha)


def _eq_c(x, target):
    return np.sum(x) - target


def _find_active(c, frac=0.003, max_lags=25):
    mx = np.max(c)
    tol = mx * frac + 1e-5
    idx = np.where(c >= mx - tol)[0]
    if len(idx) > max_lags:
        idx = np.argsort(c)[-max_lags:]
    return idx

    c5_1 = _c5(h1, dx)

    # Phase 1: SLSQP with smooth max (multi-alpha)

        if elapsed() > budget_s - 400:
            break
        x0 = _repair(best_h + 0.003 * rng.randn(n), n)
        try:
            res = minimize(_obj_smooth, x0, args=(alpha,), method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _eq_c, 'args': (target,)}],
                           options={'maxiter': 300, 'ftol': 1e-15})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:

        except Exception:
            pass

    # Phase 2: Nesterov momentum on active-set gradient
    h = best_h.copy()
```
