import time
from scipy.optimize import minimize

def repair(h, n):

    return np.correlate(h, 1.0 - h, mode='full')

def c5_val(corr, dx):
    return float(np.max(corr)) * dx

def g_vec(h, p, n):
    """Gradient of all correlations w.r.t. h[p] - h[q] swap"""
    kk = np.arange(-(n-1), n)
    g = np.zeros(2*n - 1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def sweep_round(h, n, corr, time_limit, n_coarse=20, n_bisect=30):
    improved = False
    order = np.arange(n)
    for ai in range(n-1):
        i = order[ai]
        for j in order[ai+1:]:
            if time.time() > time_limit:
                return h, corr, improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-14:
                continue
            gd = g_vec(h, i, n) - g_vec(h, j, n)
            # Coarse search
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None,:] + ds[:,None]*gd[None,:], axis=1)
            bi = np.argmin(vals)
            cur_max = np.max(corr)
            if vals[bi] >= cur_max - 1e-13:
                continue
            # Bisection
            lo = ds[max(0, bi-1)]
            hi_b = ds[min(n_coarse-1, bi+1)]
            lo = max(lo, d_min)
            hi_b = min(hi_b, d_max)
            for _ in range(n_bisect):
                mid = 0.5*(lo + hi_b)
                if np.max(corr + mid*gd) <= np.max(corr + hi_b*gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5*(lo + hi_b)
            if abs(d_best) < 1e-12:
                continue
            h[i] += d_best
            h[j] -= d_best
            corr += d_best * gd
            improved = True
    return h, corr, improved

    """Solve QP: minimize mean of active correlations subject to constraints"""
    from scipy.optimize import minimize
    
    def obj(h):
        c = np.correlate(h, 1-h, mode='full')
        return np.mean(c[active_indices])
    
    def obj_grad(h):
        # Gradient of mean active correlation
        grad = np.zeros(n)
        for m in active_indices:
            s = m - (n-1)
            if s == 0:
                grad += (1 - 2*h) / len(active_indices)
            else:
                e = np.ones(n) / len(active_indices)
                ip = np.arange(n) + s
                mp = (ip >= 0) & (ip < n)
                e[mp] -= h[ip[mp]] / len(active_indices)
                im = np.arange(n) - s
                mm = (im >= 0) & (im < n)
                e[mm] -= h[im[mm]] / len(active_indices)
                grad += e
        return grad

             'jac': lambda h: np.ones(n)}]

    res = minimize(obj, h, jac=obj_grad, method='SLSQP', 
                   bounds=bounds, constraints=cons,

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add; diff -= add
            else:
                sub = min(-diff, h[idx]); h[idx] -= sub; diff += sub
            if abs(diff) < 1e-12: break
    return h

def pair_grad(h, i, j, n):

    corr_len = 2*n - 1
    a = np.zeros(corr_len)
    b = np.zeros(corr_len)
    for m in range(corr_len):
        s = m - (n-1)
        # Change in corr[m] from h[i]+=d, h[j]-=d
        # corr[m] = sum_k h[k]*(1-h[k+s])
        # Terms involving h[i]:
        #   k=i: h[i]*(1-h[i+s]) → change: d*(1-h[i+s])
        #   k=i-s: h[i-s]*(1-h[i]) → change: -d*h[i-s]  (i.e., h[i-s]*(1-(h[i]+d)))
        # Terms involving h[j]:
        #   k=j: h[j]*(1-h[j+s]) → change: -d*(1-h[j+s])
        #   k=j-s: h[j-s]*(1-h[j]) → change: +d*h[j-s]
        # Quadratic terms (when i+s=j or j+s=i):
        #   If i+s=j: term k=i is h[i]*h[j] → (h[i]+d)*(h[j]-d) → extra -d²
        #   If j+s=i: term k=j is h[j]*h[i] → (h[j]-d)*(h[i]+d) → extra +d²
        
        ai = 0.0
        bi = 0.0
        # Contribution from h[i] increase
        if 0 <= i+s < n:
            ai += (1.0 - h[i+s])
        if 0 <= i-s < n:
            ai -= h[i-s]
        # Contribution from h[j] decrease
        if 0 <= j+s < n:
            ai -= (1.0 - h[j+s])
        if 0 <= j-s < n:
            ai += h[j-s]
        # Quadratic correction
        if i + s == j and 0 <= i < n:  # h[i]*h[i+s] = h[i]*h[j]
            bi += -1.0
        if j + s == i and 0 <= j < n:  # h[j]*h[j+s] = h[j]*h[i]
            bi += 1.0
        a[m] = ai
        b[m] = bi
    return a, b

    corr_len = 2*n - 1
    # s ranges from -(n-1) to n-1, indexed as m = s + (n-1)
    s = np.arange(corr_len) - (n-1)
    
    # h[i+s]: valid when 0 <= i+s < n
    idx_is = i + s
    mask_is = (idx_is >= 0) & (idx_is < n)
    h_is = np.where(mask_is, h[np.clip(idx_is, 0, n-1)], 0.0)
    
    # h[i-s]: valid when 0 <= i-s < n
    idx_im = i - s
    mask_im = (idx_im >= 0) & (idx_im < n)
    h_im = np.where(mask_im, h[np.clip(idx_im, 0, n-1)], 0.0)
    
    # h[j+s]
    idx_js = j + s
    mask_js = (idx_js >= 0) & (idx_js < n)
    h_js = np.where(mask_js, h[np.clip(idx_js, 0, n-1)], 0.0)

    m_js_i = j + s == i
    b[m_is_j] -= 1.0
    b[m_js_i] += 1.0
    
    return a, b

def find_best_d(corr, a, b, d_min, d_max, n_iter=40):
    """Minimize max(corr + a*d + b*d²) over d in [d_min, d_max]"""
    gr = (np.sqrt(5) - 1) / 2
    lo, hi = d_min, d_max
    c = hi - gr * (hi - lo)
    dd = lo + gr * (hi - lo)
    fc = np.max(corr + a*c + b*c*c)
    fd = np.max(corr + a*dd + b*dd*dd)
    for _ in range(n_iter):
        if fc < fd:
            hi = dd; dd = c; fd = fc
            c = hi - gr * (hi - lo)
            fc = np.max(corr + a*c + b*c*c)
        else:
            lo = c; c = dd; fc = fd
            dd = lo + gr * (hi - lo)
            fd = np.max(corr + a*dd + b*dd*dd)
    d_best = 0.5 * (lo + hi)
    return d_best, float(np.max(corr + a*d_best + b*d_best*d_best))

def repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add; diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub; diff += sub
            if abs(diff) < 1e-12: break
    return h

def compute_corr(h):

def pair_grad_fast(h, i, j, n):
    corr_len = 2 * n - 1
    s = np.arange(corr_len) - (n - 1)
    idx_is = i + s
    mask_is = (idx_is >= 0) & (idx_is < n)
    h_is = np.where(mask_is, h[np.clip(idx_is, 0, n-1)], 0.0)
    idx_im = i - s
    mask_im = (idx_im >= 0) & (idx_im < n)
    h_im = np.where(mask_im, h[np.clip(idx_im, 0, n-1)], 0.0)
    idx_js = j + s
    mask_js = (idx_js >= 0) & (idx_js < n)
    h_js = np.where(mask_js, h[np.clip(idx_js, 0, n-1)], 0.0)
    a = (1.0 - h_is) * mask_is.astype(float) - h_im * mask_im.astype(float) \

def find_best_d(corr, a, b, d_min, d_max, n_iter=45):
    if d_max - d_min < 1e-14:
        return 0.0, float(np.max(corr))

    lo, hi = d_min, d_max
    c = hi - gr * (hi - lo)
    dd = lo + gr * (hi - lo)
    fc = float(np.max(corr + a * c + b * c * c))
    fd = float(np.max(corr + a * dd + b * dd * dd))

    f_best = float(np.max(corr + a * d_best + b * d_best * d_best))
