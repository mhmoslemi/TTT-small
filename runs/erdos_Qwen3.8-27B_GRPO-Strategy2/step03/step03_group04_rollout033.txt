                c = _c5(h, dx)
                if c < c5_cur:
                    c5_cur = c
                    if c5_cur < best_c5:
                        best_c5, best_h = c5_cur, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    # Phase 5: Final SLSQP polish
    if elapsed() < budget_s - 8:
        try:
            rx = _slsqp(best_h, bounds, target, 200000, 2500)
            hc = _repair(rx, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    # Phase 6: Ultra-fine pair polish
    if elapsed() < budget_s - 2:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        h, c5_cur = _pair_search(h, c5_cur, np.array([4e-5, -4e-5]), 40, budget_s - 2)
        if c5_cur < best_c5:
            best_c5, best_h = c5_cur, h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
