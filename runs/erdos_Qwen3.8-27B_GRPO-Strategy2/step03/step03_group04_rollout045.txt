def _full_h(x):
    return np.concatenate([x, 1.0 - x])

def _c5_half(x):
    h = _full_h(x)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * (2.0 / len(h))

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _corr(h):

def _grad_x(x, h, top_lags, n):
    n_half = n // 2
    g = np.zeros(n_half)
    for ki in top_lags:

            a1 = m - L
            if 0 <= a1 < n:
                v1 += (1.0 - h[a1])

            b2 = jm + L
            if 0 <= b2 < n:
                v2 -= h[b2]
            g[m] += v1 - v2
    return g / len(top_lags)

                    diff -= a
