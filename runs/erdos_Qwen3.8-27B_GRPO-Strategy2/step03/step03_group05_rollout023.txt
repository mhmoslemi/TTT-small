    return np.correlate(h, 1.0 - h, mode='full')

    c_safe = np.maximum(c, 1e-30)
    return (np.sum(c_safe ** p)) ** (1.0 / p)

def _lsexp(c, alpha):
    m = np.max(c)
    return m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha

def _grad_corr_weighted(h, weights):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-15:
            continue
        L = k - (n - 1)
        if L == 0:
            g += wk * (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += wk * (1.0 - h[:n - L])
            g[:n - L] -= wk * h[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n - M]
    return g

    return _pnorm(c, p) + lam * (np.sum(x) - len(x) / 2.0) ** 2

                diff -= a

                diff += a
