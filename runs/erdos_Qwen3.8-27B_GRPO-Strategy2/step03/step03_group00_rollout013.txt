def project_constraint(h, n):
    """Project h onto {x : 0<=x[i]<=1, sum(x)=n/2}"""
    target = n / 2.0
    # Bisection on lambda: find lambda such that sum(clip(h[i] - lambda, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fix small numerical errors
    diff = target - h_proj.sum()
    if abs(diff) > 1e-14:
        # Distribute diff among entries not at boundary
        interior = (h_proj > 1e-8) & (h_proj < 1 - 1e-8)
        if interior.any():
            hProj[interior] += diff / interior.sum()

    for _pass in range(300):
        if time.time() > budget_end:
            break
        improved = False
        idx = np.arange(n)
        rng.shuffle(idx)
        for i in idx:
            if time.time() > budget_end:
                break
            hi = h[i]
            for j in idx:
                if j == i:
                    continue
                hj = h[j]
                for d in deltas:
                    ni, nj = hi + d, hj - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, n)
                    if c < c5_cur - 1e-15:
                        c5_cur = c
                        improved = True
                        hi, hj = ni, nj
                    else:
                        h[i], h[j] = hi, hj
                h[i], h[j] = hi, hj
        if not improved:
            break
    return h, c5_cur

def _slsqp_wrap(x, n, alpha):
    return _smooth_obj(x, n, alpha)

def _slsqp_граду(x, n, alpha):
    return _smooth_grad(x, n, alpha)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    n_init = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n_init)
    best_h = h0.copy()
    best_n = n_init
    best_c5 = _c5(best_h, best_n)

    # Phase 1: PGD with increasing alpha
    for alpha in [3000, 10000, 30000, 100000, 300000]:
        if elapsed() > budget_s * 0.35:
            break
        h_work, c5_w = _pgd_optimize(best_h, best_n, alpha,
                                     t0 + budget_s * 0.35, 0.05, 5000)
        c5_w = _c5(h_work, best_n)
        if c5_w < best_c5:
            best_c5 = c5_w
            best_h = h_work.copy()

    # Phase 2: SLSQP polishing
    for alpha in [50000, 200000]:
        if elapsed() > budget_s * 0.55:
            break
        for r in range(3):
            if elapsed() > budget_s * 0.55:
                break
            if r == 0:
                h_init = best_h.copy()
            else:
                h_init = _repair(best_h + 0.02 * rng.randn(best_n), best_n)
            try:
                res = _slsqp_run(h_init, best_n, alpha, 600)
                hc = _repair(res.x, best_n)
                c5 = _c5(hc, best_n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc
            except Exception:
                pass

    # Phase 3: Multi-resolution search
    for n_try in [50, 80, 100, 120, 140]:
        if elapsed() > budget_s * 0.78:
            break
        if n_try == best_n:
            continue
        x_old = np.linspace(0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0, 2.0, n_try, endpoint=False)
        h_new = np.interp(x_new, x_old, best_h, period=2.0)
        h_new = _repair(h_new, n_try)
        h_work, c5_w = _pgd_optimize(h_new, n_try, 50000,
                                     t0 + budget_s * 0.78, 0.03, 2000)
        c5_w = _c5(h_work, n_try)
        if c5_w < best_c5 - 1e-6:
            best_c5 = c5_w
            best_h = h_work
            best_n = n_try
        # Quick SLSQP on new resolution
        if c5_w < best_c5 + 0.002 and elapsed() < budget_s * 0.82:
            try:
                res = _slsqp_run(h_work, n_try, 100000, 400)
                hc = _repair(res.x, n_try)
                c5 = _c5(hc, n_try)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc
                    best_n = n_try
            except Exception:
                pass

    # Phase 4: Fine pair search
    if elapsed() < budget_s - 40:
        best_h, best_c5 = _fine_pair_search(best_h, best_n,
                                            t0 + budget_s - 20, rng)

    # Phase 5: Ultra-fine single moves with projection
    if elapsed() < budget_s - 10:
        h = best_h.copy()
        c5_cur = _c5(h, best_n)
        tiny = np.array([0.0001, 0.0003, 0.001, -0.0001, -0.0003, -0.001])
        end_t = t0 + budget_s - 10
        while time.time() < end_t:
            i = rng.randint(best_n)
            j = rng.randint(best_n)
            if i == j:
                continue
            d = tiny[rng.randint(len(tiny))]
            hi, hj = h[i], h[j]
            ni, nj = hi + d, hj - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5(h, best_n)
                if c < c5_cur:
                    c5_cur = c
                    if c < best_c5:
                        best_c5 = c
                        best_h = h.copy()
                else:
                    h[i], h[j] = hi, hj

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, best_n)
    return (best_h, best_c5, best_n)
