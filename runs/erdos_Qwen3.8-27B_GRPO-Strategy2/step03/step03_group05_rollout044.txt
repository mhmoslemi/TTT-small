import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _lag_grad(h, L, n):
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += (1.0 - h[:n - L])
        g[:n - L] -= h[L:]
    else:
        M = -L
        g[:n - M] += (1.0 - h[M:])
        g[M:] -= h[:n - M]
    return g

def _softobj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _topsum_obj(x, top_lags, lam):
    c2 = _corr(x)
    s = c2[top_lags].sum()
    pen = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return s + pen

def _topsum_grad(x, top_lags, lam):
    n = len(x)
    g = np.zeros(n)
    for ki in top_lags:
        L = ki - (n - 1)
        g += _lag_grad(x, L, n)
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _triple_grad_obj(x, top_lags, lam):
    c2 = _corr(x)
    m = c2.max()
    s = np.clip(5000.0 * (c2 - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    objective = c2[top_lags].sum() + 0.5 * np.log(np.sum(ew)) / 5000.0
    pen = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return objective + pen

def _triple_grad(x, top_lags, lam):
    n = len(x)
    c = _corr(x)
    m = c.max()
    s = np.clip(5000.0 * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for ki in top_lags:
        L = ki - (n - 1)
        g += _lag_grad(x, L, n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k] * 0.5
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

                for jj in range(ii + 1, min(ii + 12, len(order))):
                    j = order[jj]
                    best_d = 0.0
                    best_c = c5_cur
                    for d in steps:

                    if best_d != 0.0:

                        h[j] -= best_d
                        h = _repair(h, n)

    if _elapsed() < budget_s - 40:

        c5_cur = _c5(h, dx)
        for _ in range(60000):
            if _elapsed() > budget_s - 40:

            i, j = rng.randint(n), rng.randint(n)
            if i == j:

                    h[i], h[j] = ni - d, nj + d
