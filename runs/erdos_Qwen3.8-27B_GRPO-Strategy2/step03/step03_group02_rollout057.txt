                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_pair(h, i, j, N):
    lags = np.arange(-(N-1), N)
    g = np.zeros(2*N-1)
    ip = i + lags
    im = i - lags
    jp = j + lags
    jm = j - lags
    g -= np.where((im>=0)&(im<N), h[im], 0.0)
    g -= np.where((ip>=0)&(ip<N), h[ip], 0.0)
    g += np.where((jm>=0)&(jm<N), h[jm], 0.0)
    g += np.where((jp>=0)&(jp<N), h[jp], 0.0)
    return g

def _sweep(h, corr, N, time_limit, n_coarse=20, n_bisect=25):
    improved = False
    order = np.random.permutation(N)
    for ai in range(N-1):
        i = order[ai]
        for aj in range(ai+1, N):
            j = order[aj]
            if time.time() > time_limit:
                return h, corr, improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj-1.0)
            d_max = min(1.0-hi, hj)
            if d_max - d_min < 1e-14:
                continue
            gd = _g_pair(h, i, j, N)
            ds = np.linspace(d_min, d_max, n_coarse)
            best_val = np.max(corr)
            best_d = 0.0
            for d in ds:
                v = np.max(corr + d*gd)
                if v < best_val:
                    best_val = v
                    best_d = d
            if abs(best_d) < 1e-14:
                continue
            step = (d_max - d_min) / n_coarse
            lo = max(d_min, best_d - step)
            hival = min(d_max, best_d + step)
            for _ in range(n_bisect):
                mid = 0.5*(lo + hival)
                if np.max(corr + mid*gd) <= np.max(corr + hival*gd):
                    hival = mid
                else:
                    lo = mid
            d_best = 0.5*(lo + hival)
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best*gd
            improved = True
    return h, corr, improved

def _subgrad_round(h, N, n_iter=500):
    target = N/2.0
    h = h.copy()
    lr = 0.1
    best = _c5(h, 2.0/N)
    for it in range(n_iter):
        corr = _corr(h)
        k_star = np.argmax(corr)
        lag = k_star - (N-1)
        sg = np.ones(N)
        # d(corr[k])/d(h[i]) = 1 - h[i-lag] - h[i+lag]
        il = np.arange(N) - lag
        ip = np.arange(N) + lag
        mask_l = (il>=0)&(il<N)
        mask_p = (ip>=0)&(ip<N)
        sg[mask_l] -= h[il[mask_l]]
        sg[mask_p] -= h[ip[mask_p]]
        h_new = _repair(h - lr*sg, N)
        c_new = _c5(h_new, 2.0/N)
        if c_new < best:
            h = h_new
            best = c_new
        else:
            lr *= 0.95
    return h

def _slsqp(h, N, alpha, maxiter=200):
    dx = 2.0/N
    target = N/2.0
    bounds = [(0.0, 1.0)]*N
    
    def obj(x):
        c = _corr(x)
        m = np.max(c)
        s = np.exp(alpha*(c-m))
        return m + np.log(np.sum(s))/alpha
    
    def obj_grad(x):
        c = _corr(x)
        m = np.max(c)
        s = np.exp(alpha*(c-m))
        w = s/np.sum(s)
        # grad of smooth max = weighted average of subgradients
        grad = np.zeros(N)
        for k in range(2*N-1):
            lag_k = k - (N-1)
            sg = np.ones(N)
            il = np.arange(N) - lag_k
            ip = np.arange(N) + lag_k
            ml = (il>=0)&(il<N)
            mp = (ip>=0)&(ip<N)
            sg[ml] -= x[il[ml]]
            sg[mp] -= x[ip[mp]]
            grad += w[k]*sg
        return grad
    
    cons = [{'type':'eq', 'fun': lambda x: x.sum()-target, 'jac': lambda x: np.ones(N)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter':maxiter, 'ftol':1e-14, 'maxls':50})
        return _repair(res.x, N)
    except:
        return h

def _sweep(h, corr, N, time_limit, n_coarse=20, n_bisect=25):
    improved = False
    lags = np.arange(-(N-1), N)
    order = np.random.permutation(N)
    for ai in range(N-1):
        i = order[ai]
        for aj in range(ai+1, N):
            j = order[aj]
            if time.time() > time_limit:
                return h, corr, improved
            hi = h[i]; hj = h[j]
            d_min = max(-hi, hj-1.0)
            d_max = min(1.0-hi, hj)
            if d_max - d_min < 1e-14:
                continue
            # Compute subgradients
            gd = np.zeros(2*N-1)
            for k_idx in range(2*N-1):
                lag = lags[k_idx]
                val = 0.0
                if 0 <= i-lag < N: val -= h[i-lag]
                if 0 <= i+lag < N: val -= h[i+lag]
                if 0 <= j-lag < N: val += h[j-lag]
                if 0 <= j+lag < N: val += h[j+lag]
                gd[k_idx] = val
            # Cross term: -d² if |i-j| == |lag_k|
            diff_ij = abs(i - j)
            cross = (np.abs(lags) == diff_ij).astype(float)
            
            # Coarse search
            ds = np.linspace(d_min, d_max, n_coarse)
            best_val = np.max(corr)
            best_d = 0.0
            for d in ds:
                c_new = corr + d*gd - d*d*cross
                v = np.max(c_new)
                if v < best_val:
                    best_val = v
                    best_d = d
            if abs(best_d) < 1e-14:
                continue
            # Bisection
            step = (d_max-d_min)/n_coarse
            lo = max(d_min, best_d-step)
            hival = min(d_max, best_d+step)
            for _ in range(n_bisect):
                mid = 0.5*(lo+hival)
                if np.max(corr + mid*gd - mid*mid*cross) <= np.max(corr + hival*gd - hival*hival*cross):
                    hival = mid
                else:
                    lo = mid
            d_best = 0.5*(lo+hival)
            h[i] = hi+d_best
            h[j] = hj-d_best
            corr = corr + d_best*gd - d_best*d_best*cross
            improved = True
    return h, corr, improved

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, N):
    h = np.clip(h, 0.0, 1.0).copy()
    target = N / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(N, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _sweep(h, corr, N, time_limit, n_coarse=20, n_bisect=25):
    improved = False
    lags = np.arange(-(N-1), N)
    cross_mask = None
    order = np.random.permutation(N)
    for ai in range(N - 1):
        i = order[ai]
        for aj in range(ai + 1, N):
            j = order[aj]
            if time.time() > time_limit:
                return h, corr, improved
            hi = h[i]; hj = h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-14:
                continue
            ip = i + lags; im = i - lags
            jp = j + lags; jm = j - lags
            gd = (np.where((im>=0)&(im<N), -h[im], 0.0) +
                  np.where((ip>=0)&(ip<N), -h[ip], 0.0) +
                  np.where((jm>=0)&(jm<N), h[jm], 0.0) +
                  np.where((jp>=0)&(jp<N), h[jp], 0.0))
            diff_ij = abs(i - j)
            cross = (np.abs(lags) == diff_ij).astype(float)
            ds = np.linspace(d_min, d_max, n_coarse)
            best_val = np.max(corr)
            best_d = 0.0
            for d in ds:
                v = np.max(corr + d * gd - d * d * cross)
                if v < best_val:
                    best_val = v
                    best_d = d
            if abs(best_d) < 1e-14:
                continue
            step = (d_max - d_min) / n_coarse
            lo = max(d_min, best_d - step)
            hival = min(d_max, best_d + step)
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hival)
                if np.max(corr + mid*gd - mid*mid*cross) <= np.max(corr + hival*gd - hival*hival*cross):
                    hival = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hival)
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd - d_best * d_best * cross
            improved = True
    return h, corr, improved

def _slsqp(h, N, alpha, maxiter=200):
    target = N / 2.0
    bounds = [(0.0, 1.0)] * N
    lags = np.arange(-(N-1), N).astype(float)

    def obj(x):
        c = _corr(x)
        m = np.max(c)
        s = np.exp(np.clip(alpha * (c - m), -500, 500))
        return m + np.log(np.sum(s)) / alpha

    def obj_grad(x):
        c = _corr(x)
        m = np.max(c)
        w = np.exp(np.clip(alpha * (c - m), -500, 500))
        w /= w.sum()
        grad = np.ones(N)
        for k in range(2 * N - 1):
            lag = int(lags[k])
            if lag == 0:
                grad -= 2.0 * w[k] * x
            else:
                il = np.arange(N) - lag
                ip2 = np.arange(N) + lag
                ml = (il >= 0) & (il < N)
                mp = (ip2 >= 0) & (ip2 < N)
                grad[ml] -= w[k] * x[il[ml]]
                grad[mp] -= w[k] * x[ip2[mp]]
        return grad

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(N)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 40})
        return _repair(res.x, N)
    except Exception:
        return h

def _subgrad_round(h, N, n_iter=300):
    dx = 2.0 / N
    h = h.copy()
    lr = 0.08
    best = _c5(h, dx)
    lags = np.arange(-(N-1), N)
    for it in range(n_iter):
        corr = _corr(h)
        k_star = np.argmax(corr)
        lag = k_star - (N - 1)
        sg = np.ones(N)
        idx_l = np.arange(N) - lag
        idx_p = np.arange(N) + lag
        ml = (idx_l >= 0) & (idx_l < N)
        mp = (idx_p >= 0) & (idx_p < N)
        sg[ml] -= h[idx_l[ml]]
        sg[mp] -= h[idx_p[mp]]
        h_new = _repair(h - lr * sg, N)
        c_new = _c5(h_new, dx)
        if c_new < best:
            h = h_new
            best = c_new
        elif c_new < best + 1e-8:
            h = h_new
            best = c_new
            lr *= 0.98
        else:
            lr *= 0.9
    return h

def _init_fourier(N, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, N, endpoint=False)
    h = np.full(N, 0.5)
    for m in range(1, 12):
        amp = rng.uniform(0.1, 0.5) / (1 + 0.3 * m)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * m * x / 4.0 + phase)
    return _repair(h, N)

def _init_resample(N, Np):
    if N == Np:
        return initial_h_values.copy()
    xp = np.linspace(0, 2, Np, endpoint=False)
    xt = np.linspace(0, 2, N, endpoint=False)
    h = np.interp(xt, xp, initial_h_values, period=2.0)
    return _repair(h, N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    h_parent = np.array(initial_h_values, dtype=float)
    N_parent = len(h_parent)

    best_h = _repair(h_parent, N_parent)
    best_c5 = _c5(best_h, 2.0 / N_parent)
    best_N = N_parent

    n_candidates = [30, 40, 50, 60, 70, 80, 90, 100, 120]
    if N_parent not in n_candidates:
        n_candidates.append(N_parent)
    n_candidates = sorted(set(n_candidates))

    phase1_end = t0 + budget_s * 0.65
    phase2_end = t0 + budget_s * 0.85

    # Phase 1: multi-n search
    for N in n_candidates:
        if time.time() > phase1_end:
            break
        t_start_n = time.time()
        time_alloc = 90.0
        t_limit_n = min(t_start_n + time_alloc, phase1_end)

        starts = []
        starts.append(_init_resample(N, N_parent))
        starts.append(_init_fourier(N, seed + N))
        starts.append(_init_fourier(N, seed + N + 1000))
        starts.append(_repair(np.random.RandomState(seed + N + 2000).uniform(0, 1, N), N))

        for si, h0 in enumerate(starts):
            if time.time() > t_limit_n - 5:
                break
            h = _repair(h0, N)
            c5 = _c5(h, 2.0 / N)
            corr = _corr(h)

            # Coordinate sweeps
            stall = 0
            t_sweep_end = t_limit_n - 20
            for it in range(200):
                if time.time() > t_sweep_end:
                    break
                h, corr, imp = _sweep(h, corr, N, time.time() + 12, 20, 25)
                if imp:
                    c5_new = _c5(h, 2.0 / N)
                    if c5_new < c5 - 1e-14:
                        c5 = c5_new
                        stall = 0
                    else:
                        stall += 1
                else:
                    stall += 1
                if stall >= 4 and si == 0:
                    h2 = _slsqp(h, N, alpha=300, maxiter=100)
                    c5_2 = _c5(h2, 2.0 / N)
                    if c5_2 < c5:
                        h, c5, corr = h2, c5_2, _corr(h2)
                        stall = 0
                elif stall >= 6:
                    h = _repair(h + 0.04 * np.random.randn(N), N)
                    corr = _corr(h)
                    c5 = _c5(h, 2.0 / N)
                    stall = 0

            # Subgradient
            if time.time() < t_limit_n - 8:
                h2 = _subgrad_round(h, N, n_iter=150)
                c5_2 = _c5(h2, 2.0 / N)
                if c5_2 < c5:
                    h, c5 = h2, c5_2

            # SLSQP
            if time.time() < t_limit_n - 4:
                h2 = _slsqp(h, N, alpha=600, maxiter=200)
                c5_2 = _c5(h2, 2.0 / N)
                if c5_2 < c5:
                    h, c5 = h2, c5_2

            # Final sweep
            if time.time() < t_limit_n - 1:
                h, corr, _ = _sweep(h, _corr(h), N, t_limit_n - 1, 25, 30)
                c5 = _c5(h, 2.0 / N)

            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()
                best_N = N

    # Phase 2: intensive on best N
    if time.time() < phase2_end:
        N = best_N
        h = best_h.copy()
        corr = _corr(h)
        c5 = best_c5
        stall = 0
        for it in range(500):
            if time.time() > phase2_end - 20:
                break
            h, corr, imp = _sweep(h, corr, N, time.time() + 20, 25, 30)
            if imp:
                c5_new = _c5(h, 2.0 / N)
                if c5_new < c5 - 1e-14:
                    c5 = c5_new
                    best_c5 = c5
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
            if stall >= 5:
                h2 = _slsqp(h, N, alpha=400 + 100 * (stall % 10), maxiter=150)
                c5_2 = _c5(h2, 2.0 / N)
                if c5_2 < c5:
                    h, c5, corr = h2, c5_2, _corr(h2)
                    best_c5 = c5
                    best_h = h.copy()
                else:
                    h = _repair(h + 0.02 * np.random.randn(N), N)
                    corr = _corr(h)
                    c5 = _c5(h, 2.0 / N)
                stall = 0

    # Phase 3: final polish
    if time.time() < budget_s - 5:
        h = best_h.copy()
        h, corr, _ = _sweep(h, _corr(h), best_N, time.time() + 10, 30, 35)
        c5 = _c5(h, 2.0 / best_N)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    if time.time() < budget_s - 3:
        h2 = _slsqp(best_h, best_N, alpha=2000, maxiter=400)
        c5_2 = _c5(h2, 2.0 / best_N)
        if c5_2 < best_c5:
            best_h, best_c5 = h2.copy(), c5_2

    best_h = _repair(best_h, best_N)
    best_c5 = _c5(best_h, 2.0 / best_N)
    return (best_h, best_c5, best_N)
