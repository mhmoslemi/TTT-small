                else:

    return h

    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -50, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -50, 500)
    w = np.exp(s)
    w /= w.sum()
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-7: continue
        mlag = n - 1 - k
        for i in range(n):
            j = i + mlag
            if 0 <= j < n:
                g[i] += w[k] * (1.0 - h[j])
                g[j] -= w[k] * h[i]
    return g

def _slsqp(x0, bounds, target, alpha, maxiter=1000):
    def f(x): return _smobj(x, alpha)
    def df(x): return _smgrad(x, alpha)
    def eq(x): return x.sum() - target
    res = minimize(f, x0, jac=df, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': eq}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})

    """Momentum projected gradient for smooth max"""
    h = h.copy()
    v = np.zeros(n)
    best = _c5(h, dx)
    for it in range(nsteps):
        g = _smgrad(h, alpha)
        v = mu * v - lr * g
        h_new = _repair(h + v, n)
        c = _c5(h_new, dx)
        if c < best:
            h = h_new
            best = c
        lr *= 0.998
    return h, best

    c = _corr(h)
    idx = np.argsort(c)[-k:]

            if 0 <= j < n:
                g[i] += (1.0 - h[j])
                g[j] -= h[i]

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-13:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-13: break
    return h

def _smobj(h, alpha):
    c = _corr(h)
    m = float(c.max())
    s = np.clip(alpha * (c - m), -50.0, 500.0)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _smgrad(h, alpha):
    c = _corr(h)
    m = float(c.max())
    s = np.clip(alpha * (c - m), -50.0, 500.0)
    w = np.exp(s)
    w /= w.sum()
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-7: continue
        mlag = n - 1 - k
        lo = max(0, -mlag)
        hi = min(n, n - mlag)
        g[lo:hi] += wk * (1.0 - h[lo + mlag:hi + mlag])
        g[lo + mlag:hi + mlag] -= wk * h[lo:hi]
    return g

def _topk_grad(h, n, k):
    c = _corr(h)
    idx = np.argsort(c)[-k:]
    g = np.zeros(n)
    for m in idx:
        mlag = n - 1 - m
        lo = max(0, -mlag)
        hi = min(n, n - mlag)
        g[lo:hi] += (1.0 - h[lo + mlag:hi + mlag])
        g[lo + mlag:hi + mlag] -= h[lo:hi]
    return g / k

def _slsqp(x0, bounds, target, alpha, maxiter):
    def f(x): return _smobj(x, alpha)
    def df(x): return _smgrad(x, alpha)
    def eq(x): return x.sum() - target
    res = minimize(f, x0, jac=df, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': eq}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return res.x

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def T(): return time.time() - t0

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)
    h1 = _repair(1.0 - h0, n)
    c1 = _c5(h1, dx)
    if c1 < best_c5:
        best_h, best_c5 = h1.copy(), c1

    # Phase A: Smooth max momentum gradient descent
    if T() < budget_s - 350:
        for alpha in [200, 800, 3000, 10000, 40000]:
            if T() > budget_s - 350: break
            for att in range(2):
                if T() > budget_s - 350: break
                x0 = best_h if att == 0 else _repair(best_h + 0.003 * rng.randn(n), n)
                h_c = x0.copy()
                v = np.zeros(n)
                lr = 0.012
                for it in range(600):
                    g = _smgrad(h_c, alpha)
                    v = 0.85 * v - lr * g
                    h_new = _repair(h_c + v, n)
                    c_val = _c5(h_new, dx)
                    if c_val < _c5(h_c, dx):
                        h_c = h_new
                    lr *= 0.999
                c_final = _c5(h_c, dx)
                if c_final < best_c5:
                    best_h, best_c5 = h_c.copy(), c_final

    # Phase B: SLSQP at multiple alphas
    if T() < budget_s - 200:
        for alpha in [1000, 5000, 20000, 80000]:

            for trial in range(2):

                try:

                except Exception:

    # Phase C: Subgradient on top-K lags (increasing K)
    if T() < budget_s - 120:

            if c5_c < best_c5:

    # Phase D: Pair local search

            if T() > budget_s - 50: break

                    for j in range(i + 1, n):

                            if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                                continue
                            h[i], h[j] = ni, nj
                            c = _c5(h, dx)
                            if c < c5_cur - 1e-14:
                                c5_cur = c; imp = True
                            else:
                                h[i], h[j] = ni - d, nj + d
                if not imp: break
        if c5_cur < best_c5:
            best_h, best_c5 = h.copy(), c5_cur

    # Phase E: Fine random pair search
    if T() < budget_s - 15:
        h = best_h.copy()
        c5_cur = best_c5
        opts = np.array([2e-4, 8e-5, 4e-5, -2e-4, -8e-5, -4e-5])
        t_end = time.time() + max(0, budget_s - 15 - T())
        while time.time() < t_end:
            i, j = rng.randint(n), rng.randint(n)
            if i == j: continue
            d = opts[rng.randint(6)]
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5_cur:
                    c5_cur = c
                    best_h, best_c5 = h.copy(), c5_cur
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    # Phase F: Final SLSQP polish
    if T() < budget_s - 3:
        try:
            rx = _slsqp(best_h, bounds, target, 150000, 2000)
            hc = _repair(rx, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_h, best_c5 = hc.copy(), c5
        except Exception:
            pass

    # Phase G: Try from 1-best_h
    if T() < budget_s - 2:
        try:
            hf = _repair(1.0 - best_h, n)
            rx = _slsqp(hf, bounds, target, 50000, 1000)
            hc = _repair(rx, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_h, best_c5 = hc.copy(), c5
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
