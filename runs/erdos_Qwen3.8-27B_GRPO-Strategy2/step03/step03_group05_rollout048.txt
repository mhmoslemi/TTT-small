    """Compute gradient for top-k lags only."""
    n = len(h)
    c = _corr(h)
    top = np.argsort(c)[::-1][:k]
    g = np.zeros(n)
    for ki in top:
        L = int(ki) - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n-L])
            g[:n-L] -= h[L:]
        else:
            M = -L
            g[:n-M] += (1.0 - h[M:])
            g[M:] -= h[:n-M]
    return g

def _project(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    h = h.copy()
    for idx in order:
        if diff > 0:
            a = min(diff, 1.0 - h[idx])
            h[idx] += a
            diff -= a
        else:
            a = min(-diff, h[idx])
            h[idx] -= a
            diff += a
        if abs(diff) < 1e-13:
            break
    return h

For the QP approach:
At point h0, with active set A = {top m lags}:
minimize sum_{L∈A} [c_L(h0) + ∇c_L(h0)·(h-h0)] + λ/2 * ||h-h0||²
s.t. 0≤h≤1, sum(h) = n/2

The linearized objective:
sum_{L∈A} c_L(h0) + sum_{L∈A} ∇c_L(h0)·(h-h0) + λ/2 * ||h-h0||²

Let G = sum_{L∈A} ∇c_L(h0) = gradient of the sum of active lags.

Then: minimize G·h + λ/2 * ||h-h0||² (ignoring constants)
s.t. 0≤h≤1, sum(h) = n/2

Without box constraints: h* = h0 - G/λ + μ/λ where μ gives sum h* = n/2.
μ = (sum(h0) - n/2 + sum(G)/λ) ... 

Actually: h* = h0 - G/(2λ) + μ/(2λ) * 1
sum(h*) = sum(h0) - sum(G)/(2λ) + n*μ/(2λ) = n/2
Since sum(h0) = n/2: -sum(G)/(2λ) + n*μ/(2λ) = 0 → μ = sum(G)/n

So h*_unconstrained = h0 - (G - mean(G)*1)/(2λ)

Then project to [0,1] with sum constraint. This is a "river" or "tropical" projection.

For the projection:
1. clip to [0,1]
2. Fix the sum by adding/subtracting from the least "bound-constrained" coordinates

This is essentially a proximal gradient step with step size 1/(2λ).

The key advantage: by using the EXACT sum of active lags (not a soft-max), we get a sharper descent direction.

Let me implement this. The algorithm:
```
for iteration:
    c = _corr(h)
    A = top m lags of c
    G = sum of gradients for lags in A
    # Proximal step
    h_new = h - (G - G.mean()) / (2*lambda)
    h_new = project(h_new)  # clip to [0,1], fix sum
    # Line search or accept
    if c5(h_new) < c5(h):
        h = h_new
    else:
        lambda *= 0.5 (or accept with smaller step)
```

The choice of λ controls the step size: small λ → large steps (faster but less stable), large λ → small steps (safer but slower).

I'll use adaptive λ: start moderate, increase when steps are rejected.

from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h))) * 2.0 / n

def _grad_lag(h, L, n):
    g = np.zeros(n)
    if L == 0:
        g = 1.0 - 2.0 * h
    elif 0 < L < n:
        g[L:] += (1.0 - h[:n-L])
        g[:n-L] -= h[L:]
    elif -n < L < 0:
        M = -L
        g[:n-M] += (1.0 - h[M:])
        g[M:] -= h[:n-M]
    return g

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    h = h.copy()
    for idx in order:
        if diff > 0:
            a = min(diff, 1.0 - h[idx])
            h[idx] += a
            diff -= a
        else:
            a = min(-diff, h[idx])
            h[idx] -= a
            diff += a
        if abs(diff) < 1e-13:
            break
    return h

    g = np.zeros(n)
    for ki in top:
        L = int(ki) - (n - 1)
        if -n < L < n:
            g += _grad_lag(h, L, n)
    return g

    h_new = h - (G - G.mean()) / (2.0 * lam)
