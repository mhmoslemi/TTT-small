```python
def coord_descent(h, n, dx, max_sweeps=1000):
    c5 = compute_c5(h, dx)
    c = np.correlate(h, 1-h, mode='full')  # length 2n-1
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            # Compute coefficient vector a for position i
            a = np.zeros(2*n-1)
            # First term: (1-h[i+n-1-k]) for k in [i, i+n-1]
            k_start, k_end = i, i+n-1
            for k in range(k_start, k_end+1):
                j = i + n - 1 - k
                a[k] += (1 - h[j])
            # Second term: -h[i-n+1+k] for k in [n-1-i, 2n-2-i]
            k_start2, k_end2 = n-1-i, 2*n-2-i
            for k in range(k_start2, k_end2+1):
                j = i - n + 1 + k
                a[k] -= h[j]
            
            # Now find δ that minimizes max_k (c[k] + δ*a[k]) subject to 0≤h[i]+δ≤1
            # Feasible δ range: [-h[i], 1-h[i]]
            delta_min, delta_max = -h[i], 1.0 - h[i]
            
            # The objective is max of linears - find minimum
            # Kink points where c[k1] + δ*a[k1] = c[k2] + δ*a[k2]
            # δ = (c[k2]-c[k1]) / (a[k1]-a[k2])
            
            # Collect candidate δ values
            candidates = [delta_min, delta_max]
            # Also the current point
            candidates.append(0.0)
            
            # Find dominant lags and their intersections
            # For efficiency, only consider top few c[k] values
            top_k = np.argsort(c)[-20:]  # top 20 correlations
            for ik in range(len(top_k)):
                for jk in range(len(top_k)):
                    if ik == jk: continue
                    k1, k2 = top_k[ik], top_k[jk]
                    if a[k1] != a[k2]:
                        d = (c[k2] - c[k1]) / (a[k1] - a[k2])
                        if delta_min - 1e-10 <= d <= delta_max + 1e-10:
                            candidates.append(d)
            
            # Evaluate at all candidates
            best_delta = 0.0
            best_val = np.max(c)
            for d in candidates:
                d = np.clip(d, delta_min, delta_max)
                val = np.max(c + d * a)
                if val < best_val - 1e-14:
                    best_val = val
                    best_delta = d
            
            if abs(best_delta) > 1e-12:
                h[i] += best_delta
                c = c + best_delta * a  # update correlation
                improved = True
                new_c5 = np.max(c) * dx
                if new_c5 < c5:
                    c5 = new_c5
        
        if not improved:
            break
    return h, c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _coeff_a(h, i, n):
    """Coefficient vector: d(corr)/dh[i] for all lags."""
    a = np.zeros(2 * n - 1)
    # First term: (1-h[j]) where j = i+n-1-k, valid 0<=j<=n-1 => k in [i, i+n-1]
    ks, ke = i, i + n - 1
    js = np.arange(ke, ks - 1, -1)  # j values from i+n-1-k
    # j = i + n - 1 - k, so for k from ks to ke: j from i to i+n-1-k... let me just loop
    for k in range(ks, ke + 1):
        j = i + n - 1 - k
        if 0 <= j <= n - 1:
            a[k] += (1.0 - h[j])
    # Second term: -h[j] where j = i-n+1+k, valid 0<=j<=n-1 => k in [n-1-i, 2n-2-i]
    ks2, ke2 = n - 1 - i, 2 * n - 2 - i
    for k in range(ks2, ke2 + 1):
        j = i - n + 1 + k
        if 0 <= j <= n - 1:
            a[k] -= h[j]
    return a

def _coord_step(h, i, n, c, a):
    """Find optimal delta for coordinate i given current corr c and coeff a."""
    dmin, dmax = -h[i], 1.0 - h[i]
    if dmax - dmin < 1e-15:
        return 0.0
    # Find kink points among top lags
    top_k = np.argsort(c)[-40:]
    candidates = [dmin, dmax, 0.0]
    for ip in range(len(top_k)):
        for jq in range(ip + 1, len(top_k)):
            k1, k2 = top_k[ip], top_k[jq]
            da = a[k1] - a[k2]
            if abs(da) < 1e-14:
                continue
            d = (c[k2] - c[k1]) / da
            if dmin - 1e-10 <= d <= dmax + 1e-10:
                candidates.append(np.clip(d, dmin, dmax))
    best_d, best_v = 0.0, float(np.max(c))
    for d in candidates:
        d = float(np.clip(d, dmin, dmax))
        v = float(np.max(c + d * a))
        if v < best_v - 1e-15:
            best_v, best_d = v, d
    return best_d

def _coord_descent(h, n, dx, max_sweeps=5000, deadline=9e9):
    c = _corr(h)
    c5 = float(np.max(c) * dx)
    prev_c5 = c5 + 1e-6
    for sw in range(max_sweeps):
        if time.time() >= deadline:
            break
        improved = False
        order = np.arange(n)
        for i in order:
            a = _coeff_a(h, i, n)
            d = _coord_step(h, i, n, c, a)
            if abs(d) > 1e-13:
                h[i] += d
                c += d * a
                improved = True
        new_c5 = float(np.max(c) * dx)
        if new_c5 < c5:
            c5 = new_c5
        if not improved or abs(c5 - prev_c5) < 1e-16:
            break
        prev_c5 = c5
    return h, c5

def _pair_descent(h, n, dx, max_sweeps=500, deadline=9e9):
    c = _corr(h)
    c5 = float(np.max(c) * dx)
    for sw in range(max_sweeps):

        aj = _coeff_a(h, j, n)

        # Objective: max_k(c[k] + d*(ai[k]-aj[k]))
        # Feasible: 0<=h[i]+d<=1, 0<=h[j]-d<=1

def _binary_sa(h, n, dx, rng, deadline, T0=0.02, Tmin=0.0001, cool=0.9995):

            for idx in idx_0[:target - ones]:
                h_b[idx] = 1.0
    c5_best = _binary_c5(h_b, n, dx)
    best = h_b.copy()
    T = T0
    while time.time() < deadline and T > Tmin:
        i = rng.randint(n)
        # Swap with a position of opposite value
        if h_b[i] > 0.5:
            zeros = np.where(h_b < 0.5)[0]
        else:
            zeros = np.where(h_b > 0.5)[0]
        if len(zeros) == 0:
            continue
        j = zeros[rng.randint(len(zeros))]
        h_b[i], h_b[j] = 1.0 - h_b[i], 1.0 - h_b[j]
        c5_new = _binary_c5(h_b, n, dx)

                best = h_b.copy()
        else:
            h_b[i], h_b[j] = 1.0 - h_b[i], 1.0 - h_b[j]
        T *= cool
    return best, c5_best
```
