So: ∂result[p]/∂h[j] = (1-h[j-d])*I(0≤j-d<n) - h[j+d]*I(0≤j+d<n)

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _full_h(h_half):
    return np.concatenate([h_half, 1.0 - h_half])

def _corr_lag_grad(h, d, n):
    g = np.zeros(n)
    if d >= 0:
        if d < n:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] += -h[d:]
    else:
        dd = -d
        if dd < n:
            g[:n - dd] += (1.0 - h[dd:])
            g[dd:] += -h[:n - dd]
    return g

def _smooth_obj(h, dx, alpha):
    c = _corr(h) * dx
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _smooth_grad_full(h, dx, alpha):
    n = len(h)
    c = _corr(h) * dx
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / np.sum(e)
    g = np.zeros(n)
    for p in range(len(c)):
        if w[p] < 1e-8:
            continue
        d = p - (n - 1)
        g += w[p] * _corr_lag_grad(h, d, n)
    return g * dx

def _smooth_grad_sym(h_half, dx, alpha):
    n2 = len(h_half)
    h = _full_h(h_half)
    g_full = _smooth_grad_full(h, dx, alpha)
    return g_full[:n2] - g_full[n2:]

def _obj_sym(h_half, dx, alpha):
    h = _full_h(h_half)
    return _smooth_obj(h, dx, alpha)

def _repair(h, n, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    n = len(initial_h_values)
    n2 = n // 2
    dx = 2.0 / n
    target = n / 2.0
    bounds_full = [(0.0, 1.0)] * n
    bounds_sym = [(0.0, 1.0)] * n2

    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n, target)
    best_c5 = _c5(best_h, dx)

    h1 = _repair(1.0 - best_h, n, target)
    c5_1 = _c5(h1, dx)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 1: Symmetric L-BFGS-B (35 free vars, constraint auto-satisfied)
    h_half0 = (best_h[:n2] + (1.0 - best_h[n2:])) / 2.0
    h_half0 = np.clip(h_half0, 0.0, 1.0)

    for alpha in [50, 100, 300, 1000, 3000]:
        if elapsed() > budget_s - 350:
            break
        for att in range(3):
            if elapsed() > budget_s - 350:
                break
            if att == 0:
                x0 = h_half0.copy()
            else:
                x0 = np.clip(h_half0 + 0.01 * att * rng.randn(n2), 0.0, 1.0)

            def obj_s(x):
                return _obj_sym(x, dx, alpha)

            def grad_s(x):
                return _smooth_grad_sym(x, dx, alpha)

            try:
                res = minimize(obj_s, x0, jac=grad_s, method='L-BFGS-B',
                               bounds=bounds_sym,
                               options={'maxiter': 800, 'ftol': 1e-20, 'gtol': 1e-14})
                h_sym = _full_h(np.clip(res.x, 0, 1))
                c5 = _c5(h_sym, dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, h_sym.copy()
                    h_half0 = np.clip(res.x, 0, 1).copy()
            except Exception:
                pass

    # Phase 2: Full 70-var L-BFGS-B with projection
    if elapsed() < budget_s - 250:
        for alpha in [200, 1000, 5000]:
            if elapsed() > budget_s - 250:
                break

            def obj_f(x):
                return _smooth_obj(x, dx, alpha)

            def grad_f(x):
                return _smooth_grad_full(x, dx, alpha)

            try:
                res = minimize(obj_f, best_h, jac=grad_f, method='L-BFGS-B',
                               bounds=bounds_full,
                               options={'maxiter': 500, 'ftol': 1e-20, 'gtol': 1e-14})
                hc = _repair(res.x, n, target)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc.copy()
            except Exception:
                pass

    # Phase 3: SLSQP with smooth max for constraint handling
    if elapsed() < budget_s - 150:
        for alpha in [500, 2000]:
            if elapsed() > budget_s - 150:
                break

            def obj_sl(x):
                return _smooth_obj(x, dx, alpha)

            def grad_sl(x):
                return _smooth_grad_full(x, dx, alpha)

            def eq_c(x):
                return np.sum(x) - target

            try:
                res = minimize(obj_sl, best_h, jac=grad_sl, method='SLSQP',
                               bounds=bounds_full,
                               constraints=[{'type': 'eq', 'fun': eq_c}],
                               options={'maxiter': 600, 'ftol': 1e-16})
                hc = _repair(res.x, n, target)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc.copy()
            except Exception:
                pass

    # Phase 4: Projected subgradient with momentum on active lags
    if elapsed() < budget_s - 100:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        momentum = np.zeros(n)
        beta = 0.85

        for phase in range(6):
            lr_p = 0.008 * (0.55 ** phase)
            for it in range(200):
                if elapsed() > budget_s - 100:
                    break
                c = _corr(h)
                max_c = np.max(c)
                thresh = max_c - 0.005 * max_c
                active = np.where(c >= thresh)[0]
                g = np.zeros(n)
                for p in active:
                    d = p - (n - 1)
                    g += _corr_lag_grad(h, d, n)
                g /= len(active)
                momentum = beta * momentum + (1 - beta) * g
                h_new = np.clip(h - lr_p * momentum * dx, 0.0, 1.0)
                h_new = _repair(h_new, n, target)
                c5_new = _c5(h_new, dx)
                if c5_new < c5_cur:
                    h, c5_cur = h_new, c5_new
                    if c5_cur < best_c5:
                        best_c5, best_h = c5_cur, h.copy()

    # Phase 5: Pair local search
    if elapsed() < budget_s - 40:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        deltas = [0.002, 0.0008, 0.0002, -0.002, -0.0008, -0.0002]
        for _pass in range(40):
            if elapsed() > budget_s - 40:
                break
            imp = False
            for i in range(n):
                if elapsed() > budget_s - 40:
                    break
                hi = h[i]
                for j in range(i + 1, n):
                    hj = h[j]
                    for d in deltas:
                        ni, nj = hi + d, hj - d
                        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                            continue
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < c5_cur - 1e-15:
                            c5_cur, hi, hj = c, ni, nj
                            imp = True
                        else:
                            h[i], h[j] = hi, hj
                    h[i], h[j] = hi, hj
            if c5_cur < best_c5:
                best_c5, best_h = c5_cur, h.copy()
            if not imp:
                break

    # Phase 6: Ultra-fine random pair search
    if elapsed() < budget_s - 5:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        opts = [3e-5, 7e-5, 1.5e-4, -3e-5, -7e-5, -1.5e-4]
        while elapsed() < budget_s - 5:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = opts[rng.randint(len(opts))]
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5_cur:
                    c5_cur = c
                    best_c5 = min(best_c5, c5_cur)
                    best_h = h.copy()
                else:
                    h[i] = ni - d
                    h[j] = nj + d

    best_h = _repair(best_h, n, target)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
