def _repair(h, n):

    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

        else:

    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h))) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _project(v, n):
    target = n / 2.0

    c = _corr(h) * dx
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(float(np.sum(np.exp(s)))) / alpha

def _smooth_grad(h, n, alpha):
    dx = 2.0 / n
    c = _corr(h) * dx
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    e = np.exp(s)
    w = e / float(np.sum(e))
    g = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-14:

        lag = k - (n - 1)
        i0 = max(0, -lag)
        i1 = min(n, n - lag)
        for i in range(i0, i1):

            g[j] += wk * (1.0 - h[i])
            g[i] -= wk * h[j]

    h = h.copy()
    h_n = h.copy()  # Nesterov lookahead
    v = np.zeros(n)
    beta = 0.9
    for alpha in alphas:
        for it in range(max_iter_per_alpha):
            g = _smooth_grad(h_n, n, alpha)
            v = beta * v - (1.0 - beta) * g
            h_new = _project(h - v, n)
            # Check convergence
            if np.max(np.abs(h_new - h)) < 1e-15:
                break
            h = h_new
            t = (it + 1) / (it + 2)
            h_n = h + (t - 1) / t * (h - h_n)  # Nesterov update
    return h
