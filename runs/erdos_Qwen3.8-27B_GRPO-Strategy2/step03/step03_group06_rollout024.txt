def _repair(h, n):

    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:

                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

    corr = _corr(x)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _constrained_slsqp(h, n, dx, aidx, target_val, maxiter=300):
    """Minimize sum(h) deviation with constraints corr_k <= target_val/dx"""
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons_list = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    t_target = target_val / dx
    for k_idx in aidx:
        # corr at full-correlation index k_idx
        # full correlation: index 0 = lag -(n-1), index n-1 = lag 0, index 2n-2 = lag n-1
        # lag = k_idx - (n-1)
        # corr[lag] = sum_i h[i] * (1-h[i+lag]) for valid i
        # In numpy correlate: result[k_idx] corresponds to specific overlap
        def make_con(ki):
            def con_fun(x):
                c = _corr(x)
                return c[ki] - t_target
            return con_fun
        cons_list.append({'type': 'ineq', 'fun': lambda x, ki=k_idx: (t_target - _corr(x)[ki])})

                      method='SLSQP', bounds=bounds, constraints=cons_list,
                      options={'maxiter': maxiter, 'ftol': 1e-14})

    except:
        return h

    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-9:
                break
    return h
