```python
    c = np.correlate(h, 1.0 - h, mode='full')

    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _repair(h, n):

    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)

            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
```

```python
    """Returns (2n-1, n) matrix where row m is gradient of c[m] w.r.t. h"""
    G = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        k = m - (n-1)
        if k >= 0:
            G[m, :n-k] += (1.0 - h[k:])
            G[m, k:] -= h[:n-k]
        else:
            G[m, :n+k] += (1.0 - h[-k:])
            G[m, -k:] -= h[:n+k]
    return G
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5_raw(h):
    return np.max(np.correlate(h, 1.0 - h, mode='full'))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_obj(x, alpha, target):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + 1000.0 * (x.sum() - target) ** 2

def _grad_matrix(h, n):
    G = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        if k >= 0:
            G[m, :n-k] += (1.0 - h[k:])
            G[m, k:] -= h[:n-k]
        else:
            G[m, :n+k] += (1.0 - h[-k:])
            G[m, -k:] -= h[:n+k]
    return G

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5_raw(best_h) * dx

    # Phase 1: SLSQP from parent

                hc = _repair(res.x, n)

            except Exception:
                pass

    # Phase 2: Soft-gradient descent with momentum

                    f = f_try
                    if f < best_c5:

    return (best_h, best_c5, n)
```
