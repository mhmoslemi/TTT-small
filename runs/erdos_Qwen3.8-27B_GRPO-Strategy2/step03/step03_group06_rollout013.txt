import numpy as np
from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _compute_c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _soft_max_val(corr, alpha):
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _soft_max_weights(corr, alpha):
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    e = np.exp(s)
    return e / np.sum(e)

def _gradient(h, alpha, dx):
    """Exact gradient of soft_max(corr)*dx w.r.t. h."""
    n = len(h)
    corr = _corr_full(h)
    w = _soft_max_weights(corr, alpha)
    # grad[j] = sum_{k=j}^{j+n-1} w[k] * (1-2*h[k-j]) * dx
    # Let m = k-j, so m goes from 0 to n-1, k = j+m
    grad = np.zeros(n)
    for j in range(n):
        valid = np.arange(j, j+n) < len(corr)
        m_idx = np.arange(n)[valid]
        k_idx = np.arange(j, j+n)[valid]
        grad[j] = np.sum(w[k_idx] * (1.0 - 2.0 * h[m_idx])) * dx
    return grad

def _objective(h, alpha, target, pen):
    corr = _corr_full(h)
    val = _soft_max_val(corr, alpha) * (2.0 / len(h))  # dx
    return val + pen * (np.sum(h) - target) ** 2

def _objective_grad(h, alpha, target, pen):
    dx = 2.0 / len(h)
    corr = _corr_full(h)
    w = _soft_max_weights(corr, alpha)
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        k_idx = np.arange(j, j+n)
        valid = k_idx < len(corr)
        m_idx = np.arange(n)[valid]
        k_valid = k_idx[valid]
        grad[j] = np.sum(w[k_valid] * (1.0 - 2.0 * h[m_idx])) * dx
    diff = np.sum(h) - target
    grad += 2.0 * pen * diff * np.ones(n)
    return grad

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:
                break
    return h

def _lbfgs(h0, n, dx, alpha, target, maxiter=300, pen_schedule=None):
    """L-BFGS-B with penalty for sum constraint."""
    bounds = [(0.0, 1.0)] * n
    if pen_schedule is None:
        pen_schedule = [10, 100, 1000, 10000]
    best_h, best_c5 = h0.copy(), _compute_c5(h0, dx)
    for pen in pen_schedule:
        try:
            res = minimize(
                _objective, h0, args=(alpha, target, pen),
                jac=_objective_grad, method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-8}
            )
            hc = _repair(res.x, n)
            c5 = _compute_c5(hc, dx)
            if c5 < best_c5:

                h0 = hc
        except Exception:

    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _soft_max(corr, alpha):
    m = corr.max()
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _soft_max_w(corr, alpha):
    m = corr.max()
    s = np.clip(alpha * (corr - m), -500, 500)
    e = np.exp(s)
    return e / e.sum()

def _grad(h, alpha, dx, target, pen):
    """Exact gradient of softmax(corr)*dx + pen*(sum(h)-target)^2."""
    n = len(h)
    corr = _corr_full(h)
    w = _soft_max_w(corr, alpha)
    # grad[j] = dx * sum_{m=0}^{n-1} w[j+m] * (1-2*h[m])
    v = 1.0 - 2.0 * h
    g = np.correlate(w, v, mode='valid') * dx
    g += 2.0 * pen * (h.sum() - target)
    return g

def _obj(h, alpha, target, pen):
    corr = _corr_full(h)
    return _soft_max(corr, alpha) * (2.0/len(h)) + pen*(h.sum()-target)**2

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i]); h[i] += a; diff -= a
            else:
                a = min(-diff, h[i]); h[i] -= a; diff += a
            if abs(diff) < 1e-10:
                break
    return h

def _binary_proj(h, n):
    """Project to 0-1 vector with exactly n/2 ones."""
    target = n // 2
    # Round: take top n/2 values as 1
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def _lbfgs(h0, n, dx, alpha, target, maxiter=200, pens=(50, 500, 5000, 50000)):
    bounds = [(0.0, 1.0)] * n
    best_h, best_c5 = h0.copy(), _c5(h0, dx)
    for pen in pens:
        try:
            res = minimize(_obj, h0, args=(alpha, target, pen),
                          jac=_grad, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-7})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_h, best_c5 = hc, c5
                h0 = hc
        except Exception:
            pass
    return best_h, best_c5

def _discrete_ls(h, n, dx, rng, deadline, n_iters=50000):
    """Local search on 0-1 vectors."""
    b = _binary_proj(h, n)
    c5 = _c5(b, dx)
    best_c5 = c5
    best_b = b.copy()
    target = n // 2
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    for _ in range(n_iters):
        if time.time() >= deadline:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i], b[j] = 0, 1
        new_c5 = _c5(b, dx)
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_b = b.copy()
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
        else:
            b[i], b[j] = 1, 0
    return best_b, best_c5

def _continuous_ls(h, n, dx, rng, deadline, n_iters=30000):
    """Pair-swap local search in continuous space."""
    h = h.copy()
    best_c5 = _c5(h, dx)
    scales = [0.02, 0.01, 0.005, 0.002, 0.001]
    for scale in scales:
        ni = max(1000, n_iters // len(scales))
        for _ in range(ni):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            a, b2 = h[i] + d, h[j] - d
            if a < 0 or a > 1 or b2 < 0 or b2 > 1:
                continue
            h[i], h[j] = a, b2
            c5 = _c5(h, dx)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, best_c5

def _anneal(h, n, dx, rng, deadline):
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    c5_cur = best_c5
    T = 0.005
    while time.time() < deadline and T > 1e-6:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 0.01 * (2.0 * rng.random() - 1.0)
        a, b2 = h_cur[i] + d, h_cur[j] - d
        if a < 0 or a > 1 or b2 < 0 or b2 > 1:
            continue
        h_cur[i], h_cur[j] = a, b2
        c5_new = _c5(h_cur, dx)
        if c5_new < c5_cur or rng.random() < np.exp(-(c5_new - c5_cur) / T):
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_cur.copy()
        else:
            h_cur[i], h_cur[j] = h_cur[i] - d, h_cur[j] + d
        T *= 0.9998
    return best_h, best_c5

def _perturb(h, n, rng, kind):

    if kind == 0:
        b = rng.randint(3, 15)
        i = rng.randint(0, n - b)

    elif kind == 3:

import numpy as np
from scipy.optimize import minimize
import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr_full(h)) * dx)

def _soft_max(corr, alpha):
    m = corr.max()
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _soft_max_w(corr, alpha):
    m = corr.max()
    s = np.clip(alpha * (corr - m), -500, 500)
    e = np.exp(s)
    return e / e.sum()

def _grad_fn(h, alpha, dx, target, pen):
    n = len(h)
    corr = _corr_full(h)
    w = _soft_max_w(corr, alpha)
    v = 1.0 - 2.0 * h
    g = np.correlate(w, v, mode='valid') * dx
    g += 2.0 * pen * (h.sum() - target)
    return g

def _obj_fn(h, alpha, target, pen):
    corr = _corr_full(h)
    return _soft_max(corr, alpha) * (2.0 / len(h)) + pen * (h.sum() - target) ** 2

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i]); h[i] += a; diff -= a
            else:
                a = min(-diff, h[i]); h[i] -= a; diff += a
            if abs(diff) < 1e-10:
                break
    return h

def _binary_proj(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def _lbfgs(h0, n, dx, alpha, target, maxiter=250, pens=None):
    if pens is None:
        pens = (50, 500, 5000, 50000)
    bounds = [(0.0, 1.0)] * n
    best_h, best_c5 = h0.copy(), _c5(h0, dx)
    for pen in pens:
        try:
            res = minimize(_obj_fn, best_h, args=(alpha, target, pen),
                          jac=_grad_fn, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-7})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc
        except Exception:
            pass
    return best_h, best_c5

def _discrete_ls(h, n, dx, rng, deadline, n_iters=80000):
    b = _binary_proj(h, n)
    best_b = b.copy()
    best_c5 = _c5(b, dx)
    for _ in range(n_iters):
        if time.time() >= deadline:
            break
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i], b[j] = 0, 1
        nc = _c5(b, dx)
        if nc < best_c5 - 1e-12:
            best_c5 = nc
            best_b = b.copy()
        else:
            b[i], b[j] = 1, 0
    return best_b, best_c5

def _cont_ls(h, n, dx, rng, deadline, n_iters=40000):
    h = h.copy()
    best_c5 = _c5(h, dx)
    scales = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    per = max(500, n_iters // len(scales))
    for scale in scales:
        for _ in range(per):
            if time.time() >= deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            a, b2 = h[i] + d, h[j] - d
            if a < 0 or a > 1 or b2 < 0 or b2 > 1:
                continue
            h[i], h[j] = a, b2
            c5 = _c5(h, dx)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, best_c5

def _anneal(h, n, dx, rng, deadline):
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    c5_cur = best_c5
    T = 0.004
    while time.time() < deadline and T > 5e-7:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 0.008 * (2.0 * rng.random() - 1.0)
        a, b2 = h_cur[i] + d, h_cur[j] - d
        if a < 0 or a > 1 or b2 < 0 or b2 > 1:
            continue
        h_cur[i], h_cur[j] = a, b2
        cn = _c5(h_cur, dx)
        if cn < c5_cur or (c5_cur - cn < 0.1 and rng.random() < np.exp(-(cn - c5_cur) / T)):
            c5_cur = cn
            if cn < best_c5:
                best_c5 = cn
                best_h = h_cur.copy()
        else:
            h_cur[i], h_cur[j] = h_cur[i] - d, h_cur[j] + d
        T *= 0.9997
    return best_h, best_c5

def _perturb(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        bsz = rng.randint(3, 15)
        i = rng.randint(0, n - bsz)
        hp[i:i+bsz] = np.roll(hp[i:i+bsz], rng.randint(1, bsz))
    elif kind == 1:
        f = rng.randint(1, 6)
        p = rng.uniform(0, 2 * np.pi)
        hp += rng.uniform(0.05, 0.2) * np.sin(2*np.pi*f*np.arange(n)/n + p)
    elif kind == 2:
        f1, f2 = rng.randint(1, 5), rng.randint(1, 5)
        p1, p2 = rng.uniform(0, 6.28)
        hp += rng.uniform(0.04, 0.14) * np.sin(2*np.pi*f1*np.arange(n)/n+p1) * np.sin(2*np.pi*f2*np.arange(n)/n+p2)
    elif kind == 3:
        j = rng.randint(n//4, 3*n//4)
        hp = np.concatenate([h[j:], h[:j]])
    elif kind == 4:
        hp = hp + rng.normal(0, 0.06, n)
    elif kind == 5:
        b = _binary_proj(h, n)
        hp = b + rng.normal(0, 0.03, n)
    else:
        hp = h[rng.permutation(n)]
    return _repair(hp, n)

def _grad_step(h, n, dx, alpha, target, deadline, lr=0.005):
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    for _ in range(5000):
        if time.time() >= deadline:
            break
        corr = _corr_full(h)
        w = _soft_max_w(corr, alpha)
        v = 1.0 - 2.0 * h
        g = np.correlate(w, v, mode='valid') * dx
        g -= g.mean()
        h_new = h - lr * g
        h_new = np.clip(h_new, 0, 1)
        h_new = _repair(h_new, n)
        c5n = _c5(h_new, dx)
        if c5n < best_c5 - 1e-15:
            best_c5 = c5n
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.02, 0.02)
        else:
            lr *= 0.85
        if lr < 1e-9:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
