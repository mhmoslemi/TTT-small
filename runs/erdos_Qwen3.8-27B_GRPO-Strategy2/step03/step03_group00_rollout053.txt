    """Gradient of c_k (k-th 'full' correlation value) w.r.t. h."""
    L = k - (n - 1)  # actual lag
    grad = np.zeros(n)
    if L == 0:
        # c_0 = sum h[i](1-h[i]), grad = 1-2h
        return 1.0 - 2.0 * h
    elif L > 0:
        # c_L = sum_{i=0}^{n-1-L} h[i+L]*(1-h[i])
        # grad[i] = (1-h[i+L]) for i=0..n-1-L (h[i] as second factor in (1-h[i]))
        #         - h[i-L] for i=L..n-1 (h[i] as first factor h[i])
        # Wait: c_L = sum_{i=0}^{n-1-L} h[i+L] - h[i+L]*h[i]
        # d/dh[j]: 
        #   h[j] appears in h[i+L] when i+L=j, i=j-L: valid if 0<=j-L<=n-1-L, i.e., L<=j<=n-1
        #     contribution: +1
        #   h[j] appears in -h[i+L]*h[i] when i+L=j (i=j-L): contribution -h[i] = -h[j-L]
        #   h[j] appears in -h[i+L]*h[i] when i=j: contribution -h[i+L] = -h[j+L]

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h
