                h[idx] -= sub
                diff += sub

    return h

    dx_val = 2.0 / n
    c = _corr(x) * dx_val
    m = np.max(c)

    def el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h0.copy()
    best_c5 = _c5(best_h, n0)
    best_n = n0

    # Phase 1: SLSQP at native resolution
    for alpha in [3000, 10000, 30000]:
        if el() > budget_s - 350:
            break
        for r in range(5):
            if el() > budget_s - 350:
                break
            h_i = best_h.copy() if r == 0 else _repair(best_h + 0.04 * rng.randn(best_n), best_n)
            try:
                res = _slsqp(h_i, best_n, lambda x: _sm_obj(x, best_n, alpha),
                             lambda x: _sm_grad(x, best_n, alpha), 400)
                hc = _repair(res.x, best_n)
                c5 = _c5(hc, best_n)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc.copy()
            except Exception:
                pass

    # Phase 2: Interpolate to n=100 and optimize
    n1 = 100
    h1 = _interp(best_h, best_n, n1)
    if _c5(h1, n1) < best_c5:
        best_c5, best_h, best_n = _c5(h1, n1), h1, n1
    else:
        best_h = h1
        best_n = n1
        best_c5 = _c5(best_h, best_n)

    for alpha in [8000, 25000, 80000]:
        if el() > budget_s - 280:
            break
        for r in range(4):
            if el() > budget_s - 280:
                break
            h_i = best_h.copy() if r == 0 else _repair(best_h + 0.03 * rng.randn(best_n), best_n)
            try:
                res = _slsqp(h_i, best_n, lambda x: _sm_obj(x, best_n, alpha),
                             lambda x: _sm_grad(x, best_n, alpha), 600)
                hc = _repair(res.x, best_n)
                c5 = _c5(hc, best_n)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc.copy()
            except Exception:
                pass

    # Phase 3: Top-k SLSQP
    for K in [3, 5, 8]:
        if el() > budget_s - 180:
            break
        for r in range(3):
            if el() > budget_s - 180:
                break
            h_i = best_h.copy() if r == 0 else _repair(best_h + 0.02 * rng.randn(best_n), best_n)
            try:
                res = _slsqp(h_i, best_n, lambda x: _topk_obj(x, best_n, K),
                             lambda x: _topk_grad(x, best_n, K), 400)
                hc = _repair(res.x, best_n)
                c5 = _c5(hc, best_n)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc.copy()
            except Exception:
                pass

    # Phase 4: Projected gradient with top-M focus
    if el() < budget_s - 120:
        h, c5 = _proj_grad(best_h, best_n, M=5, steps=2000,
                           budget_end=t0 + budget_s - 120, rng=rng)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Phase 5: Simulated annealing
    if el() < budget_s - 80:
        h = best_h.copy()
        c5_cur = _c5(h, best_n)
        T = 5e-5; cool = 0.9997; min_T = 1e-10
        steps = np.array([0.002, 0.005, 0.01, 0.03, 0.08])
        while el() < budget_s - 80:
            i = rng.randint(best_n)
            j = rng.randint(best_n)
            if i == j:
                continue
            d = steps[rng.randint(len(steps))] * (1 if rng.rand() < 0.5 else -1)
            hi, hj = h[i], h[j]
            ni, nj = hi + d, hj - d
            ni, nj = max(0.0, min(1.0, ni)), max(0.0, min(1.0, nj))
            h[i], h[j] = ni, nj
            c5_new = _c5(h, best_n)
            if c5_new < c5_cur or (T > min_T and rng.rand() < np.exp(-(c5_new - c5_cur) / T)):
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h.copy()
            else:
                h[i], h[j] = hi, hj
            T = max(T * cool, min_T)

    # Phase 6: Deterministic local search
    if el() < budget_s - 25:
        best_h, best_c5 = _local_search(best_h, best_n, t0 + budget_s - 25)

    # Phase 7: Ultra-fine random descent
    if el() < budget_s - 5:
        h = best_h.copy()
        c5_cur = _c5(h, best_n)
        tiny = np.array([0.0002, 0.0004, 0.0008, -0.0002, -0.0004, -0.0008])
        while time.time() < t0 + budget_s - 5:
            i = rng.randint(best_n)
            j = rng.randint(best_n)
            if i == j:
                continue
            d = tiny[rng.randint(len(tiny))]
            hi, hj = h[i], h[j]
            ni, nj = hi + d, hj - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5(h, best_n)
                if c < c5_cur:
                    c5_cur = c
                    if c < best_c5:
                        best_c5, best_h = c, h.copy()
                else:
                    h[i], h[j] = hi, hj

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, best_n)
    return (best_h, best_c5, best_n)
