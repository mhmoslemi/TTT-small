                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_matrix(h0, n):
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        lo1 = max(0, lag)
        hi1 = min(n, lag + n)

        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if lo2 < hi2:
            G[k, lo2:hi2] -= h0[lo2 + lag:hi2 + lag]
    return G

                    imp = True
                else:
                    h[i], h[j] = ni - d, nj + d
        if not imp:
            break
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Try 1-h symmetry
    h1 = _repair(1.0 - h0, n)
    c5_1 = _c5(h1, dx)
    if c5_1 < best_c5:
        best_h, best_c5 = h1.copy(), c5_1

    # Phase 1: SLP with full steps
    if time.time() - t0 < budget_s - 400:
        best_h, best_c5 = _slp_phase(best_h, n, dx, 80,
                                     [1.0, 1.0, 1.0], t0, budget_s, 400)

    # Phase 2: SLP with damped steps for stability
    if time.time() - t0 < budget_s - 300:
        best_h, best_c5 = _slp_phase(best_h, n, dx, 60,
                                     [0.7, 0.5, 0.3], t0, budget_s, 300)

    # Phase 3: SLP from 1-best
    if time.time() - t0 < budget_s - 250:
        h_alt = _repair(1.0 - best_h, n)
        h_alt, c5_alt = _slp_phase(h_alt, n, dx, 60,
                                   [1.0, 0.5], t0, budget_s, 250)
        if c5_alt < best_c5:
            best_h, best_c5 = h_alt, c5_alt

    # Phase 4: Random restarts with SLP
    if time.time() - t0 < budget_s - 150:
        for restart in range(30):
            if time.time() - t0 > budget_s - 150:
                break
            sc = 0.03 * (1.0 + 0.3 * (restart % 5))
            h_rr = _repair(best_h + sc * rng.randn(n), n)
            h_rr, c5_rr = _slp_phase(h_rr, n, dx, 40,
                                     [1.0, 0.6], t0, budget_s, 150)
            if c5_rr < best_c5:
                best_h, best_c5 = h_rr, c5_rr

    # Phase 5: Subgradient refinement
    if time.time() - t0 < budget_s - 80:
        h_sg, c5_sg = _subgrad_refine(best_h.copy(), n, dx, t0, budget_s, 80)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    # Phase 6: SLP polish after subgradient
    if time.time() - t0 < budget_s - 60:
        best_h, best_c5 = _slp_phase(best_h, n, dx, 40,
                                     [0.5, 0.3, 0.1], t0, budget_s, 60)

    # Phase 7: Pair search (uses remaining time)
    if time.time() - t0 < budget_s - 10:
        best_h, best_c5 = _pair_search(best_h.copy(), n, dx, rng,
                                       t0, budget_s, 10)

    # Phase 8: Final SLP micro-polish
    if time.time() - t0 < budget_s - 3:
        for _ in range(10):
            if time.time() - t0 > budget_s - 3:
                break
            h_lp, _ = _try_solvers(best_h, n)
            if h_lp is not None:
                h_new = _repair(0.3 * best_h + 0.7 * h_lp, n)
                c5_new = _c5(h_new, dx)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new, c5_new

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
