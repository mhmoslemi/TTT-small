    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corr(h))) * dx

def _repair(h, n, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:

        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _grad_for_lags(h, lags, n):
    """Compute gradient of sum of selected lags' correlations w.r.t. h."""

            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2
    return np.clip(h - mid, 0, 1)

    best_vj = h[j]

        c = _c5(h, dx)
        if c < best_c:
            best_c = c
            best_vi = v
            best_vj = s - v
    h[i], h[j] = best_vi, best_vj
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _project(h, n, target):
    """Project onto {x: 0<=x<=1, sum=target} via bisection on shift."""
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def _grad_lags(h, lags, n):
    g = np.zeros(n)
    for L in lags:
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    return g

def _solve_2d(h, i, j, n, dx, n_scan=120):
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-12:
        return h, _c5(h, dx)
    best_c = _c5(h, dx)
    best_v = h[i]
    vs = np.linspace(lo, hi, n_scan)
    for v in vs:
        h[i] = v; h[j] = s - v
        c = _c5(h, dx)
        if c < best_c:
            best_c = c; best_v = v
    h[i] = best_v; h[j] = s - best_v
    return h, best_c

def _quick_grad_polish(h, n, target, dx, max_it=500, t_limit=None, t0=None):
    step = 0.008
    for it in range(max_it):
        if t_limit is not None and time.time() - t0 > t_limit:
            break
        c = _corr(h)
        top = np.argsort(c)[::-1][:6] - (n - 1)
        g = _grad_lags(h, list(top), n)
        h_new = _project(h - step * g, n, target)
        c_new = _c5(h_new, dx)
        c_old = _c5(h, dx)
        if c_new < c_old:
            h = h_new
            step = min(step * 1.08, 0.04)
        else:
            step *= 0.65
            if step < 1e-8:
                break
    return h

    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    def elapsed():

    # Phase 1: Subgradient descent for min-max (different from soft-max)
    if elapsed() < budget_s - 350:
        h = best_h.copy()
        C0 = _c5(h, dx)
        for it in range(8000):
            if elapsed() > budget_s - 350:
                break
            c = _corr(h)
            cmax = c.max()
            # Tolerance band around max
            tol = max(0.05 * cmax, 0.005)
            top_mask = c >= (cmax - tol)
            top_lags = np.where(top_mask)[0] - (n - 1)
            # Also ensure absolute top-5 are included
            abs_top = np.argsort(c)[::-1][:5] - (n - 1)
            top_lags = np.unique(np.concatenate([top_lags, abs_top]))
            g = _grad_lags(h, list(top_lags), n)

        ht = _project(base + res.x @ P, n, target)

        c5_h = _c5(h, dx)

            i, j = rng.randint(n, 2)
            if i == j:
                continue
            d = deltas[rng.randint(len(deltas))]
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)

                        best_h = h.copy()
                else:
                    h[i] -= d; h[j] += d

    if elapsed() < budget_s - 20:
        patterns = []
        # Block patterns
        for nb in [2, 3, 4, 5]:
            ht = np.zeros(n)
            bs = n // nb
            for b in range(nb):
