```python
    # (1-h[j+k]) valid for j: this is (1-h) circularly shifted, masked
    # h[j-k] valid for j: this is h circularly shifted, masked
    
    # Use the fact that for shift k:
    # If k >= 0: (1-h[j+k]) is valid for j = 0,...,n-1-k, and h[j-k] is valid for j = k,...,n-1
    # If k < 0: (1-h[j+k]) is valid for j = -k,...,n-1, and h[j-k] is valid for j = 0,...,n-1+k
    
    grad = np.zeros(n)
    for m in range(len(w)):
        if w[m] < 1e-15:
            continue
        k = m - (n - 1)
        if k >= 0:
            # (1-h[j+k]) valid for j in [0, n-1-k]
            grad[:n-k] += w[m] * (1.0 - h[k:])
            # h[j-k] valid for j in [k, n-1]
            grad[k:] -= w[m] * h[:n-k]
        else:
            # (1-h[j+k]) valid for j in [-k, n-1]
            grad[-k:] += w[m] * (1.0 - h[:n+k])
            # h[j-k] valid for j in [0, n-1+k]
            grad[:n+k] -= w[m] * h[-k:]
    return grad
```

This is O(n * n) = O(n²) with vectorized inner operations, so it's fast.

- Take step: h_new = h - step * grad
- Project: find lambda such that sum(clip(h_new - lambda, 0, 1)) = target

```python
        hc2 = _slsqp_full(hc, n, target, bounds, max(1000, alpha), 400)
        c5_2 = _c5(hc2, dx)
        if c5_2 < best_c5 - 1e-15:
            best_c5 = c5_2
            best_h = hc2.copy()

    # Phase 3: Binary rounding + reoptimize (try different thresholds)
    for thresh in [0.03, 0.08, 0.15, 0.25]:
        if tl() < 60:
            break
        hc = best_h.copy()
        close0 = hc < thresh
        close1 = hc > 1.0 - thresh
        n_free = n - close0.sum() - close1.sum()
        if n_free < 3 or n_free > n - 3:
            continue
        hc[close0] = 0.0
        hc[close1] = 1.0
        def obj_bin(x):
            return _smooth_max(_corr(x), 5000)
        cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
        try:
            res = minimize(obj_bin, hc, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-15, 'maxls': 100})
            hc = _repair(res.x, n)
            # Enforce the binary constraints
            hc[close0] = 0.0
            hc[close1] = 1.0
            hc = _repair(hc, n)
        except Exception:
            pass
        c5 = _c5(hc, dx)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = hc.copy()

    # Phase 4: Intensive pair sweeps with diverse orderings
    stall = 0
    round_num = 0
    while tl() > 100:
        round_num += 1
        if round_num % 4 == 1:
            order = np.random.permutation(n)
        elif round_num % 4 == 2:
            order = np.argsort(best_h)
        elif round_num % 4 == 3:
            order = np.argsort(-best_h)
        else:
            order = np.argsort(np.random.uniform(0, 1, n) * (1.0 - np.abs(2.0 * best_h - 1.0)))

        h = best_h.copy()
        corr = _corr(h)
        chunk = min(35.0, tl() - 100)
        h, corr, imp, ni_ = _pair_sweep(h, n, corr, time.time() + chunk, order, 25, 50)
        if imp:
            c5 = _c5(h, dx)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        # SLSQP polish after sweep improvements
        if imp and tl() > 130:
            hc = _slsqp_full(best_h, n, target, bounds, 10000, 500)
            c5_s = _c5(hc, dx)
            if c5_s < best_c5 - 1e-15:
                best_c5 = c5_s
                best_h = hc.copy()

        # Perturbation when stalled
        if stall >= 3 and tl() > 130:
            scale = min(0.04, 0.008 * stall)
            h_p = best_h + scale * np.random.randn(n)
            h_p = _repair(h_p, n)
            hc = _slsqp_full(h_p, n, target, bounds, 3000, 300)
            hc_corr = _corr(hc)
            ord_p = np.random.permutation(n)
            hc, hc_corr, _, _ = _pair_sweep(hc, n, hc_corr, time.time() + 15, ord_p, 20, 40)
            c5_t = _c5(hc, dx)
            if c5_t < best_c5 - 1e-15:
                best_c5 = c5_t
                best_h = hc.copy()
            stall = 0

    # Phase 5: Final high-precision sweeps
    while tl() > 20:
        order = np.random.permutation(n)
        h = best_h.copy()
        corr = _corr(h)
        chunk = min(12.0, tl() - 15)
        h, corr, imp, ni_ = _pair_sweep(h, n, corr, time.time() + chunk, order, 35, 55)
        if imp:
            c5 = _c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 6: Final SLSQP with very high alpha
    for alpha in [10000, 30000, 80000]:
        if tl() < 6:
            break
        hc = _slsqp_full(best_h, n, target, bounds, alpha, 800)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
```
