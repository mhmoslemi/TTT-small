```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return h

def _coord_terms(h, i, n):
    """Compute A[k] and C[k] such that c[k] = A[k] + C[k]*t + q*(t - t^2) for t=h[i].
    q=1 for k=center (k_shift=0), q=0 otherwise."""
    center = n - 1
    n_terms = 2 * n - 1
    A = np.zeros(n_terms)
    C = np.zeros(n_terms)
    q = np.zeros(n_terms)
    
    for ks in range(-(n-1), n):
        k = center + ks
        # c[k] = sum_{m: valid} h[m] * (1 - h[m+ks])
        m_lo = max(0, -ks)
        m_hi = min(n, n - ks)
        
        base = 0.0
        c_i = 0.0
        for m in range(m_lo, m_hi):
            m2 = m + ks
            if m == i and m2 == i:
                # k=0, m=i: term is h[i]*(1-h[i]) = t - t^2
                q[k] = 1.0
            elif m == i:
                # h[i] * (1 - h[m2]) = t * (1 - h[m2])
                c_i += (1.0 - h[m2])
            elif m2 == i:
                # h[m] * (1 - h[i]) = h[m] - t*h[m]
                base += h[m]
                c_i -= h[m]
            else:
                base += h[m] * (1.0 - h[m2])
        
        A[k] = base
        C[k] = c_i
    
    return A, C, q

def _min_1d(A, C, q, center):
    """Minimize max_k (A[k] + C[k]*t + q[k]*(t-t^2)) over t in [0,1]."""
    def f(t):
        vals = A + C * t
        if q[center] > 0:
            vals[center] += t - t * t
        return np.max(vals)
    
    # Use minimize_scalar
    res = minimize_scalar(f, bounds=(0, 1), method='bounded', 
                         options={'xatol': 1e-10})
    return res.x

def _coord_descent(h, n, max_passes=200):
    """Perform coordinate descent on h."""
    h = h.copy()
    center = n - 1
    best_c5 = _c5(h, 2.0/n)
    
    for p in range(max_passes):
        improved = False
        for i in range(n):
            A, C, q = _coord_terms(h, i, n)
            t_new = _min_1d(A, C, q, center)
            if abs(t_new - h[i]) > 1e-10:
                h[i] = t_new
                improved = True
                c5_new = _c5(h, 2.0/n)
                if c5_new < best_c5:
                    best_c5 = c5_new
        if not improved:
            break
    return h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return h

def _coord_opt(h, i, n):
    center = n - 1
    c = _corr(h)
    hi = h[i]
    C = np.zeros(2 * n - 1)
    for ks in range(-(n - 1), n):
        if ks == 0:
            continue
        k = center + ks
        i2 = i + ks
        if 0 <= i2 < n:
            C[k] += (1.0 - h[i2])
        m = i - ks
        if 0 <= m < n:
            C[k] -= h[m]
    B = c - C * hi
    B[center] -= (hi - hi * hi)
    
    def f(t):
        vals = B + C * t
        vals[center] += t - t * t
        return np.max(vals)
    
    best_t = hi
    best_f = f(hi)
    grid = np.linspace(0, 1, 200)
    for t in grid:
        ft = f(t)
        if ft < best_f - 1e-14:
            best_f = ft
            best_t = t
    lo = max(0.0, best_t - 0.005)
    hi_b = min(1.0, best_t + 0.005)
    if hi_b - lo > 1e-8:
        res = minimize_scalar(f, bounds=(lo, hi_b), method='bounded',
                             options={'xatol': 1e-10})
        if res.fun < best_f:
            best_t = float(res.x)
    return best_t

def _coord_descent(h, n, max_passes=100, tol=1e-12):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = _c5(h, dx)
    for p in range(max_passes):
        max_change = 0.0
        for i in range(n):
            t_new = _coord_opt(h, i, n)
            change = abs(t_new - h[i])
            if change > tol:
                h[i] = t_new
                max_change = max(max_change, change)
        c5 = _c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
        if max_change < tol:
            break
    return h, best_c5

def _interpolate(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def _subgradient(h, n, n_iter=5000):
    dx = 2.0 / n
    h = h.copy()
    c5_cur = _c5(h, dx)
    for it in range(n_iter):
        c = _corr(h)
        max_c = c.max()
        active = np.where(c > 0.95 * max_c)[0]
        # Compute gradient of average of active correlations
        g = np.zeros(n)
        for m in active:
            k = m - (n - 1)
            if k >= 0:
                g[k:] += (1.0 - h[:n - k]) / len(active)
                g[:n - k] -= h[k:] / len(active)
            else:
                g[:n + k] += (1.0 - h[-k:]) / len(active)
                g[-k:] -= h[:n + k] / len(active)
        step = 0.05 / (1.0 + 0.01 * it)
        h_new = h - step * g
        h_new = _repair(h_new, n)
        c5_new = _c5(h_new, dx)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return budget_s - (time.time() - t0)
    
    # Start from initial_h_values
    n = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, 2.0 / n)
    best_n = n
    
    # Phase 1: Coordinate descent from initial
    if tleft() > 200:
        h_cd, c5_cd = _coord_descent(h0, n, max_passes=200)
        if c5_cd < best_c5:
            best_c5, best_h, best_n = c5_cd, h_cd, n
    
    # Phase 2: Subgradient refinement
    if tleft() > 150:
        h_sg, c5_sg = _subgradient(best_h, best_n, n_iter=10000)
        if c5_sg < best_c5:
            best_c5, best_h, best_n = c5_sg, h_sg, best_n
        # Follow up with coordinate descent
        h_cd2, c5_cd2 = _coord_descent(best_h, best_n, max_passes=100)
        if c5_cd2 < best_c5:
            best_c5, best_h, best_n = c5_cd2, h_cd2, best_n
    
    # Phase 3: Multi-start with perturbations
    for trial in range(8):
        if tleft() < 100:
            break
        noise_scale = 0.05 * (1 - trial / 10.0)
        h_pert = best_h + noise_scale * rng.randn(*best_h.shape)
        h_pert = _repair(h_pert, best_n)
        h_cd3, c5_cd3 = _coord_descent(h_pert, best_n, max_passes=150)
        if c5_cd3 < best_c5:
            best_c5, best_h, best_n = c5_cd3, h_cd3, best_n
    
    # Phase 4: Try different n values
    for n_try in [50, 60, 80, 100]:
        if tleft() < 80:
            break
        h_interp = _interpolate(best_h, n_try)
        h_interp = _repair(h_interp, n_try)
        h_cd4, c5_cd4 = _coord_descent(h_interp, n_try, max_passes=100)
        if c5_cd4 < best_c5:
            best_c5, best_h, best_n = c5_cd4, h_cd4, n_try
    
    # Phase 5: SLSQP polish
    if tleft() > 60:
        target = best_n / 2.0
        bounds = [(0.0, 1.0)] * best_n
        
        def obj_slsqp(x):
            c = _corr(x)
            m = np.max(c)
            alpha = 10000
            s = np.clip(alpha * (c - m), -500, 500)
            return m + np.log(np.sum(np.exp(s))) / alpha
        
        def con_sum(x):
            return np.sum(x) - target
        
        try:
            res = minimize(obj_slsqp, best_h, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': con_sum}],
                          options={'maxiter': 500, 'ftol': 1e-16, 'iprint': 0})
            hc = _repair(res.x, best_n)
            c5 = _c5(hc, 2.0 / best_n)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass
    
    # Phase 6: Final coordinate descent
    if tleft() > 30:
        h_final, c5_final = _coord_descent(best_h, best_n, max_passes=200)
        if c5_final < best_c5:
            best_c5, best_h, best_n = c5_final, h_final, best_n
    
    # Final repair and evaluation
    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, 2.0 / best_n)
    
    return (best_h, best_c5, best_n)
```

```python
def _update_corr(c, h, i, old_val, new_val):
    """Update correlation c after changing h[i] from old_val to new_val."""
    n = len(h)
    center = n - 1
    delta = new_val - old_val
    if abs(delta) < 1e-15:
        return
    # c[k] = sum_m h[m]*(1-h[m+ks]) where k = center + ks
    # Terms involving h[i]:
    # m = i: h[i]*(1-h[i+ks]) → change by delta*(1-h[i+ks])
    # m+ks = i (m = i-ks): h[i-ks]*(1-h[i]) → change by -delta*h[i-ks]
    for ks in range(-(n-1), n):
        k = center + ks
        if ks == 0:
            # h[i]*(1-h[i]) changes by delta*(1-2*new_val) approximately
            # Actually: new term = new_val*(1-new_val), old term = old_val*(1-old_val)
            # change = new_val - new_val² - old_val + old_val² = delta - delta*(new+old)
            change = delta - delta * (new_val + old_val)
            c[k] += change
        else:
            change = 0.0
            i2 = i + ks
            if 0 <= i2 < n:
                change += delta * (1 - h[i2])
            m = i - ks
            if 0 != m and 0 <= m < n:  # m != i already guaranteed by ks != 0
                change -= delta * h[m]
            c[k] += change
```

```python
from scipy.optimize import minimize, minimize_scalar
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return h

def _coord_opt(h, i, n):
    center = n - 1
    c = _corr(h)
    hi = h[i]
    C = np.zeros(2 * n - 1)
    for ks in range(-(n - 1), n):
        if ks == 0:
            continue
        k = center + ks
        i2 = i + ks
        if 0 <= i2 < n:
            C[k] += (1.0 - h[i2])
        m = i - ks
        if 0 <= m < n:
            C[k] -= h[m]

    B[center] -= (hi - hi * hi)

    def f(t):
        vals = B + C * t
        vals[center] += t - t * t
        return np.max(vals)

    best_f = f(hi)
    grid = np.linspace(0, 1, 150)
    for t in grid:
        ft = f(t)
        if ft < best_f - 1e-14:
            best_f = ft
            best_t = t
    lo = max(0.0, best_t - 0.008)
    hi_b = min(1.0, best_t + 0.008)
    if hi_b - lo > 1e-9:
        res = minimize_scalar(f, bounds=(lo, hi_b), method='bounded',
                             options={'xatol': 1e-11})
        if res.fun < best_f:

def _coord_descent(h, n, max_passes=300, tol=1e-13):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = _c5(h, dx)
    for p in range(max_passes):
        max_change = 0.0
        for i in range(n):
            t_new = _coord_opt(h, i, n)
            ch = abs(t_new - h[i])
            if ch > tol:
                h[i] = t_new
                if ch > max_change:
                    max_change = ch
        c5 = _c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
        if max_change < tol:
            break
    return h, best_c5

def _subgradient(h, n, n_iter=8000):
    dx = 2.0 / n
    h = h.copy()
    c5_cur = _c5(h, dx)
    for it in range(n_iter):
        c = _corr(h)
        max_c = c.max()
        active = np.where(c > 0.97 * max_c)[0]
        g = np.zeros(n)
        for m in active:
            k = m - (n - 1)
            if k >= 0:
                g[k:] += (1.0 - h[:n - k])
                g[:n - k] -= h[k:]
            else:
                g[:n + k] += (1.0 - h[-k:])
                g[-k:] -= h[:n + k]
        g /= len(active)
        step = 0.04 / (1.0 + 0.008 * it)
        h_new = _repair(h - step * g, n)
        c5_new = _c5(h_new, dx)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
    return h, c5_cur

def _interpolate(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
```
