from scipy.optimize import minimize

# Module-level parameter storage to avoid lambdas/closures
_params = {}

def _smooth_max(corr, alpha):
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _obj_lbfgs(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return _smooth_max(corr, _params['alpha']) + _params['pen'] * (np.sum(x) - _params['target'])**2

def _obj_slsqp(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return _smooth_max(corr, _params['alpha'])

def _eq_con(x):
    return np.sum(x) - _params['target']

def _compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

    c5_best = c5_cur

import numpy as np
from scipy.optimize import minimize
import time

_params = {}

def _smooth_max(corr, alpha):
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _obj_lbfgs(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return _smooth_max(corr, _params['alpha']) + _params['pen'] * (np.sum(x) - _params['target'])**2

def _obj_slsqp(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return _smooth_max(corr, _params['alpha'])

def _eq_con(x):
    return np.sum(x) - _params['target']

def _compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _lbfgs_opt(h, n, dx, alpha, pen, maxiter):
    _params['alpha'] = alpha
    _params['pen'] = pen
    _params['target'] = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_lbfgs, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-7})
        hc = _repair(res.x, n)
        return hc, _compute_c5(hc, dx)
    except Exception:
        return h, _compute_c5(h, dx)

    _params['target'] = n / 2.0

        h_t, c5_t = _pairwise_local(best_h, n, dx, rng, tEnd + 1, 5000)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    best_h = _repair(best_h, n)
    best_c5 = _compute_c5(best_h, dx)
    return (best_h, best_c5, n)
