    return float(np.max(corr) * dx)

def _to_binary(h, n):
    """Convert continuous h to binary with exactly n/2 ones."""
    b = (h >= 0.5).astype(float)
    n_ones = int(b.sum())
    target = n // 2
    if n_ones == target:
        return b
    # Repair
    if n_ones > target:

        idx = np.where(b == 1)[0]

        order = np.argsort(-h[idx])

        need = target - n_ones
        idx = np.where(b == 0)[0]

When we swap h[i] from 1→0 and h[j] from 0→1:
- The correlation c[k] changes only for lags k where i or j is involved.
- Specifically, c[k] = sum_n h[n+k]*(1-h[n])
- Changing h[i]: affects terms where n+k=i (i.e., k=i-n) or n=i
- Changing h[j]: similar

For the "full" correlation of length 2n-1 with indices from -(n-1) to (n-1):
c[k] = sum_{n} h[n+k] * (1-h[n]) where 0≤n≤n-1 and 0≤n+k≤n-1

When h[i] goes from 1 to 0:
- Terms where n+k=i: these contribute h[i]*(1-h[n]) = 1*(1-h[n]). They become 0. So c[k=i-n] decreases by (1-h[n]) for each valid n.

import numpy as np
import time
from scipy.optimize import minimize

def _c5_val(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _to_binary(h, n):
    target = n // 2
    b = (h >= 0.5).astype(float)
    n_ones = int(b.sum())
    if n_ones == target:
        return b
    if n_ones > target:
        idx = np.where(b == 1)[0]
        order = np.argsort(-h[idx])
        for i in range(n_ones - target):
            b[idx[order[i]]] = 0.0
    else:
        need = target - n_ones
        idx = np.where(b == 0)[0]
        order = np.argsort(-h[idx])
        for i in range(need):
            b[idx[order[i]]] = 1.0
    return b

def _greedy_binary(h, n, max_passes=20):
    h = h.copy()
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_cur = float(np.max(corr)) * (2.0 / n)
    for _ in range(max_passes):
        best_c5 = c5_cur
        best_swap = None
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                c = np.correlate(h, 1.0 - h, mode='full')
                v = float(np.max(c)) * (2.0 / n)
                h[i], h[j] = 1.0, 0.0
                if v < best_c5 - 1e-15:
                    best_c5 = v
                    best_swap = (i, j)
        if best_swap is None:
            break
        i, j = best_swap
        h[i], h[j] = 0.0, 1.0
        ones.remove(i)
        ones.append(j)
        zeros.remove(j)
        zeros.append(i)
        c5_cur = best_c5
    return h, c5_cur

def _sa_binary(h, n, rng, deadline, T_init=0.4, cool=0.99999, two_swap_prob=0.15):
    h = h.copy()
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_cur = float(np.max(corr)) * (2.0 / n)
    c5_best = c5_cur
    h_best = h.copy()
    T = T_init
    while time.time() < deadline:
        do_2swap = rng.random() < two_swap_prob
        if do_2swap and len(ones) >= 2 and len(zeros) >= 2:
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones) - 1)]
            if i2 == i1:
                i2 = ones[(ones.index(i1) + 1) % len(ones)]
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros) - 1)]
            if j2 == j1:
                j2 = zeros[(zeros.index(j1) + 1) % len(zeros)]
            h[i1], h[i2], h[j1], h[j2] = 0.0, 0.0, 1.0, 1.0
            c = np.correlate(h, 1.0 - h, mode='full')
            c5_new = float(np.max(c)) * (2.0 / n)
            d = c5_new - c5_cur
            if d <= 0 or rng.random() < np.exp(-d / T):
                c5_cur = c5_new
                ones.remove(i1)
                ones.remove(i2)
                ones.extend([j1, j2])
                zeros.remove(j1)
                zeros.remove(j2)
                zeros.extend([i1, i2])
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h.copy()
            else:
                h[i1], h[i2], h[j1], h[j2] = 1.0, 1.0, 0.0, 0.0
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            c = np.correlate(h, 1.0 - h, mode='full')
            c5_new = float(np.max(c)) * (2.0 / n)
            d = c5_new - c5_cur
            if d <= 0 or rng.random() < np.exp(-d / T):
                c5_cur = c5_new
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h.copy()
            else:
                h[i], h[j] = 1.0, 0.0
        T = max(T * cool, 0.0003)
    return h_best, c5_best

def _cont_refine(h, n, deadline):
    dx = 2.0 / n
    target = n / 2.0
    a = 3000.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = a * np.clip(corr - m, -200, 200)
        return (m + np.log(np.sum(np.exp(s))) / a) * dx
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = a * np.clip(corr - m, -200, 200)
        ex = np.exp(s)
        z = np.sum(ex)
        weights = ex / z
        dc_dcorr = weights
        dm_dcorr = np.zeros_like(corr)
        dm_dcorr[np.argmax(corr)] = 1.0
        dc_dcorr += (1.0 / a) * (1.0 - m * 0)  # simplified
        g = np.zeros(n)
        for k in range(len(corr)):
            w = dc_dcorr[k] * dx
            shift = k - (n - 1)
            if shift >= 0:
                lo = max(0, shift)
                hi = min(n, n - shift)
                for idx in range(lo, hi):
                    g[idx] += w * (1.0 - (x[idx - shift] if idx >= shift else 0))
                    if idx - shift < n:
                        g[idx - shift] += w * x[idx]
            else:
                lo = max(0, -shift)
                hi = min(n + shift, n)
                for idx in range(lo, hi):
                    g[idx + (-shift)] += w * x[idx]
                    g[idx] += w * (1.0 - x[idx + (-shift)])
        return g * dx
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': 800, 'ftol': 1e-16})
        hc = np.clip(res.x, 0.0, 1.0)
        diff = target - hc.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - hc) if diff > 0 else np.argsort(hc)
            for idx in order:
                if diff > 0:
                    add = min(diff, 1.0 - hc[idx])
                    hc[idx] += add
                    diff -= add
                else:
                    sub = min(-diff, hc[idx])
                    hc[idx] -= sub
                    diff += sub
                if abs(diff) < 1e-12:
                    break
        c5 = _c5_val(hc, n)
        return hc, c5
    except Exception:
        return h, _c5_val(h, n)
