def _coord_decent_step(h, j, c_arr, n):
    # Compute p_k and q_k for the j-th coordinate
    # c_k = p_k * h[j] + q_k for all k
    # Current: c_k = p_k * h[j] + q_k, so q_k = c_k - p_k * h[j]
    
    p = np.zeros(2*n-1)
    for m in range(2*n-1):
        k = m - (n-1)
        if k >= 0:
            # h[j] appears as first factor if k <= j <= n-1
            if k <= j <= n-1:
                p[m] += (1.0 - h[j-k])
            # h[j] appears as second factor if j <= n-1-k
            if j <= n-1-k:
                p[m] -= h[j+k]
        else:
            kp = -k
            # h[j] appears as first factor if j <= n-1-kp
            if j <= n-1-kp:
                p[m] += (1.0 - h[j+kp])
            # h[j] appears as second factor if kp <= j <= n-1
            if kp <= j <= n-1:
                p[m] -= h[j-kp]
    
    q = c_arr - p * h[j]

    # Upper bound: max at t=1 is max(p_k + q_k)
    
    lo = max(min(c_arr), 0)  # actually lower bound
    hi = max(c_arr)  # current value
    
    # Actually, let me just do: the minimum is between min of (max at 0, max at 1) and ... 
    # Let me just binary search
    max_at_0 = q.max()
    max_at_1 = (p + q).max()
    
    # The answer is at most min(max_at_0, max_at_1)
    # And at least max_k(min(over t in [0,1]) of p_k*t + q_k)
    
    # Binary search on v
    for _ in range(60):

        # Check if exists t in [0,1] with p_k*t + q_k <= mid for all k
        # Constraints: p_k*t <= mid - q_k for all k
        # If p_k > 0: t <= (mid - q_k)/p_k

        # If p_k == 0: need q_k <= mid

        t_lo = 0.0
        t_hi = 1.0
        for pk, qk in zip(p, q):
            if pk > 1e-12:

    # Lower bound: 0 (all c_k >= 0 since h in [0,1])
    # Upper bound: max(max(q), max(p+q))
    lo = 0.0
    hi = max(q.max(), (p + q).max())
    
    if hi <= lo:
        return 0.0
    
    best_t = 0.0
    for _ in range(40):

        t_hi = 1.0
        feasible = True
        for k in range(n_lags):

        else:

    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
            else:
                a = max(diff, -h[idx])
            h[idx] += a
            diff -= a
            if abs(diff) < 1e-10:
                break
    return h

def _grad_one(h, n, m):
    """Gradient of c_m with respect to h."""
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    else:
        kp = -k
        g[:n-kp] += (1.0 - h[kp:])
        g[kp:] -= h[:n-kp]
    return g

def _smooth_obj(x, beta):
    c = _corr(x)
    m = c.max()
    return m + np.log(np.sum(np.exp(np.clip(beta*(c-m), -500, 500))))/beta

def _sum_con(x, target):
    return np.sum(x) - target

def _compute_p(h, j, n):

    p = np.zeros(2*n-1)

        if k >= 0:
            if k <= j:
                p[m] += (1.0 - h[j-k])
            if j+k < n:
                p[m] -= h[j+k]
        else:
            kp = -k
            if j+kp < n:
                p[m] += (1.0 - h[j+kp])
            if j-kp >= 0:
                p[m] -= h[j-kp]
    return p

def _solve_1d_max(p, q):

            elif p[k] < -1e-15:
                tl = max(tl, (mid - q[k]) / p[k])
            elif q[k] > mid + 1e-15:
                ok = False
                break
        if not ok or tl > th + 1e-15:
            lo = mid
        else:
            hi = mid
            best_t = max(0.0, min(1.0, (tl+th)*0.5))
    return best_t

def _coord_sweep(h, c_arr, n):

    n_changed = 0

        if abs(delta) > 1e-10:

            c_arr += p * delta

For lag k ≥ 0 (m = k + n - 1):
$c_k = \sum_{i=0}^{n-1-k} h[i+k](1-h[i])$

$\partial c_k / \partial h[j]$:
- h[j] appears as $h[i+k]$ when $i = j-k$, valid if $0 \le j-k \le n-1-k$, i.e., $k \le j \le n-1$
  Contribution: $(1-h[j-k])$
- h[j] appears as $(1-h[i])$ when $i = j$, valid if $0 \le j \le n-1-k$, i.e., $j \le n-1-k$
  Contribution: $-h[j+k]$

So for m corresponding to k ≥ 0:

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            a = min(diff, 1.0 - h[idx]) if diff > 0 else max(diff, -h[idx])
            h[idx] += a
            diff -= a
            if abs(diff) < 1e-10:
                break
    return h

def _grad_one(h, n, m):
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        kp = -k
        g[:n - kp] += (1.0 - h[kp:])
        g[kp:] -= h[:n - kp]
    return g

def _smooth_obj(x, beta):
    c = _corr(x)
    m = c.max()
    return m + np.log(np.sum(np.exp(np.clip(beta * (c - m), -500, 500)))) / beta

def _sum_con(x, target):
    return np.sum(x) - target

def _compute_p(h, j, n):
    p = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        if k >= 0:
            if k <= j:
                p[m] += (1.0 - h[j - k])
            if j + k < n:
                p[m] -= h[j + k]
        else:
            kp = -k
            if j + kp < n:
                p[m] += (1.0 - h[j + kp])
            if j >= kp:
                p[m] -= h[j - kp]
    return p

def _solve_1d(p, q):
    n_lags = len(p)
    pos = p > 1e-15
    neg = p < -1e-15
    zero = ~(pos | neg)
    hi = max(float(q.max()), float((p + q).max()))
    if hi <= 1e-14:
        return 0.0
    lo = 0.0
    best_t = 0.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        tl, th = 0.0, 1.0
        if np.any(pos):
            th = min(th, float(np.min((mid - q[pos]) / p[pos])))
        if np.any(neg):
            tl = max(tl, float(np.max((mid - q[neg]) / p[neg])))
        if (np.any(zero) and np.any(q[zero] > mid + 1e-14)) or tl > th + 1e-14:
            lo = mid
        else:
            hi = mid
            best_t = max(0.0, min(1.0, (tl + th) * 0.5))
    return best_t

def _coord_sweep(h, c_arr, n):
    n_changed = 0
    for j in range(n):
        p = _compute_p(h, j, n)
        q = c_arr - p * h[j]
        t_new = _solve_1d(p, q)
        delta = t_new - h[j]
        if abs(delta) > 1e-9:
            h[j] = t_new
            c_arr += p * delta
            n_changed += 1
    return h, c_arr, n_changed

def _cd_loop(h, c_arr, n, max_sweeps, min_left, best_ref, t0, budget_s, dx):
    best_c5 = best_ref[0]
    best_h = best_ref[1]
    for sweep in range(max_sweeps):
        if (budget_s - (time.time() - t0)) < min_left:
            break
        h, c_arr, nc = _coord_sweep(h, c_arr, n)
        h = _repair(h, n)
        c_arr = _corr(h)
        mv = c_arr.max() * dx
        if mv < best_c5:
            best_c5 = mv
            best_h = h.copy()
        if nc == 0 and sweep > 3:
            break
    best_ref[0] = best_c5
    best_ref[1] = best_h
    return h, c_arr

    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_c5 = _c5(best_h, dx)
    best_ref = [best_c5, best_h]

    def tleft():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP smooth max
    for beta in [80, 500, 3000]:
        if tleft() < 160:
            break
        try:
            res = minimize(_smooth_obj, best_ref[1], args=(beta,),
                method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': _sum_con, 'args': (target,)}],
                options={'maxiter': 3000, 'ftol': 1e-18, 'iprint': 0})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
                best_ref[0], best_ref[1] = c5, hc
        except Exception:
            pass

    # Phase 2: Coordinate descent (exact 1D, fast)
    if tleft() > 280:
        h = best_ref[1].copy()
        c_arr = _corr(h)
        _cd_loop(h, c_arr, n, 3000, 280, best_ref, t0, budget_s, dx)

    # Phase 3: Nesterov subgradient on true max
    if tleft() > 120:
        h = best_ref[1].copy()
        y = h.copy()
        for it in range(40000):
            if tleft() < 120:
                break
            c = _corr(y)
            m_idx = int(np.argmax(c))
            g = _grad_one(y, n, m_idx)
            gf = np.linalg.norm(g)
            if gf < 1e-15:
                break
            step = 0.15 / (1.0 + 0.3 * np.sqrt(float(it)))
            if it > 0:
                z = h + 0.85 * (h - y)
            else:
                z = h.copy()
            z = z - step * g
            y = _repair(z, n)
            h = y.copy()
            c5n = _c5(h, dx)
            if c5n < best_c5:
                best_c5, best_h = c5n, h.copy()
                best_ref[0], best_ref[1] = c5n, best_h

    # Phase 4: Coordinate descent polish
    if tleft() > 100:
        h = best_ref[1].copy()
        c_arr = _corr(h)
        _cd_loop(h, c_arr, n, 1000, 100, best_ref, t0, budget_s, dx)

    # Phase 5: Multi-scale random restarts + CD
    if tleft() > 50:
        for scale in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001]:
            if tleft() < 50:
                break
            for _r in range(3):
                if tleft() < 50:
                    break
                h = best_ref[1] + scale * rng.randn(n)
                h = _repair(h, n)
                c_arr = _corr(h)
                h, c_arr = _cd_loop(h, c_arr, n, 200, 50, best_ref, t0, budget_s, dx)

    # Phase 6: Final SLSQP polish
    if tleft() > 30:
        try:
            res = minimize(_smooth_obj, best_ref[1], args=(10000,),
                method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': _sum_con, 'args': (target,)}],
                options={'maxiter': 2000, 'ftol': 1e-18, 'iprint': 0})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
                best_ref[0], best_ref[1] = c5, hc
        except Exception:
            pass

    # Phase 7: Final coordinate descent
    if tleft() > 15:
        h = best_ref[1].copy()
        c_arr = _corr(h)
        _cd_loop(h, c_arr, n, 500, 15, best_ref, t0, budget_s, dx)

    best_h = _repair(best_ref[1], n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
