import numpy as np
import time

def _corr_dx(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_val(h, n):
    return float(np.max(np.correlate(h, 1.0-h, mode='full'))) * (2.0/n)

def _project(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5*(lo+hi)
        s = np.clip(h+mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5*(lo+hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0-h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0-h[idx])
                h[idx] += a; diff -= a
            else:
                a = max(diff, h[idx]-1.0)
                h[idx] += a; diff -= a
            if abs(diff) < 1e-14: break
    return h

def _grad_smooth(h, n, alpha, topk=None):
    c = _corr_dx(h, n)
    m = c.max()
    s = np.clip(alpha*(c-m), -500, 500)
    e = np.exp(s)
    w = e/e.sum()
    if topk and topk < len(c):
        idx = np.argsort(w)[::-1][:topk]
        w = w.copy()
        w[~np.isin(np.arange(len(w)), idx)] = 0
        w /= w.sum()
    dxv = 2.0/n
    g = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-14: continue
        lag = k-(n-1)
        if lag >= 0:
            L = n-lag
            g[lag:lag+L] += wk*(1.0-h[:L])
            g[:L] -= wk*h[lag:lag+L]
        else:
            L = n+lag
            g[:L] += wk*(1.0-h[-lag:])
            g[-lag:] -= wk*h[:L]
    return g*dxv
