import numpy as np
from scipy.optimize import minimize
import time

_st = {}

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    return float(np.max(compute_corr(h)) * 2.0 / n)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def _smmax(corr, a):
    m = np.max(corr)
    s = np.clip(a * (corr - m), -500, 0)
    return m + np.log(np.sum(np.exp(s))) / a

def _obj(h):
    corr = compute_corr(h)
    v = _smmax(corr, _st['a'])
    k = _st['k']
    if k > 1:
        sc = np.sort(corr)[::-1][:k]
        v += _st['fw'] * (sc[0] - sc[-1])
    return v

def _eq(h):
    return h.sum() - _st['tgt']

def slsqp(h, n, alpha, topk, fw, iters):
    _st.update(a=alpha, k=topk, fw=fw, tgt=n / 2.0)
    try:
        r = minimize(_obj, h, method='SLSQP',
                     bounds=[(0, 1)] * n,
                     constraints=[{'type': 'eq', 'fun': _eq}],
                     options={'maxiter': iters, 'ftol': 1e-14})
        return repair(r.x, n)
    except Exception:
        return h

def subgrad(h, n, topk, lr):
    corr = compute_corr(h)
    idx = np.argsort(corr)[::-1][:topk]
    g = np.zeros(n)
    for kk in idx:
        lag = int(kk) - (n - 1)
        if lag >= 0:
            g[:n - lag] += (1.0 - h[lag:])
            if lag > 0:
                g[lag:] -= h[:n - lag]
        else:
            L = -lag
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
    g /= len(idx)
    return repair(h - lr * g, n)

def cd2(h, n, t_end, rng, niters):
    h = h.copy()
    best = compute_c5(h, n)
    steps = (0.04, 0.02, 0.01, 0.005, 0.002, 0.001)
    for it in range(niters):
        if time.time() > t_end:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        found = False
        for d in steps:
            for sgn in (1, -1):
                ni, nj = h[i] + sgn * d, h[j] - sgn * d
                if ni < -1e-10 or ni > 1 + 1e-10 or nj < -1e-10 or nj > 1 + 1e-10:
                    continue
                h[i], h[j] = ni, nj
                c = compute_c5(h, n)
                if c < best - 1e-14:
                    best = c
                    found = True
                    break
                h[i], h[j] = ni - sgn * d, nj + sgn * d
            if found:
                break
        if not found and it > 600:
            break
    return h, best

def cos2_start(n, rng, amp=0.0):
    i = np.arange(n)
    h = np.cos(2.0 * np.pi * i / n) ** 2
    if amp > 0:
        h += amp * rng.randn(n)
    return repair(h, n)
