            h2, c2 = _fcd(h2, n, 3000, t0 + budget_s * 0.78, rng)
            if c2 < best_c5 - 1e-14:
                best_h, best_c5 = h2.copy(), c2
    if time.time() - t0 < budget_s * 0.92:
        h, c5 = _fcd(best_h, n, 100000, t0 + budget_s * 0.92, rng)
        if c5 < best_c5 - 1e-14:
            best_h, best_c5 = h.copy(), c5
    if time.time() - t0 < budget_s * 0.96:
        for na in [64, 66, 72, 80]:
            if time.time() > t0 + budget_s * 0.96:
                break
            if na == n:
                continue
            for kk in range(2):
                if time.time() > t0 + budget_s * 0.96:
                    break
                ha = _mk(na, seed + na * 13 + kk, kk % 5)
                ha = _slsqp(ha, na, 20000, 3, 0.01, 300)
                ha, ca = _fcd(ha, na, 5000, t0 + budget_s * 0.96, rng)
                if ca < best_c5 - 1e-14:
                    best_h, best_c5, best_n = ha.copy(), ca, na
    if time.time() - t0 < budget_s * 0.99:
        h2 = _slsqp(best_h, best_n, 100000, 7, 0.05, 500)
        c2 = _c5(h2, best_n)
        if c2 < best_c5 - 1e-14:
            best_h, best_c5 = h2.copy(), c2
        h, c5 = _fcd(best_h, best_n, 30000, t0 + budget_s * 0.995, rng)
        if c5 < best_c5 - 1e-14:
            best_h, best_c5 = h.copy(), c5
    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, best_n)
    return (best_h, best_c5, best_n)
