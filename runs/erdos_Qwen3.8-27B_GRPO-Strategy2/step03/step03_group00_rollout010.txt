            e = np.exp(s)
            return _lag_grad(e / e.sum(), x, n)
    else:  # 'topk'

        def obj(x):
            c = _corr_vals(x, n)
            return c[np.argsort(c)[::-1][:k]].sum()

            idx = np.argsort(c)[::-1][:k]
            w = np.zeros(len(c))
            w[idx] = 1.0 / k
            return _lag_grad(w, x, n)
    
    res = minimize(obj, h_init, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
    return res.x

def _hinge_slsqp(h_init, n, C_target, maxiter):
    """Minimize sum of (c_k - C_target)_+^2 + lambda * (sum - n/2)^2"""
    bounds = [(0.0, 1.0)] * n
    tgt = n / 2.0
    
    def obj(x):
        c = _corr_vals(x, n)
        excess = np.maximum(c - C_target, 0)
        return np.sum(excess ** 2)
    
    def grad(x):
        c = _corr_vals(x, n)
        mask = c > C_target
        if not mask.any():
            return np.zeros(n)
        w = np.zeros(len(c))
        w[mask] = 2.0 * (c[mask] - C_target)
        return _lag_grad(w, x, n)
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt, 'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h_init, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
    return res.x

def _pair_search(h, n, end_t, rng, steps):

    while time.time() < end_t:

        else:

    return h, c5_cur

def _proj_grad(h, n, end_t, alpha, lr0):
    """Projected gradient on smooth max."""
    lr = lr0
    c5_cur = _c5(h, n)
    while time.time() < end_t:

        m = c.max()
        s = np.clip(alpha * (c - m), -250, 250)
        e = np.exp(s)
        w = e / e.sum()
        g = _lag_grad(w, h, n)
        h_new = np.clip(h - lr * g, 0, 1)
        h_new *= (n / 2.0) / max(h_new.sum(), 1e-12)
        h_new = np.clip(h_new, 0, 1)
        c_new = _c5(h_new, n)
        if c_new < c5_cur:
            h, c5_cur = h_new, c_new
            lr = min(lr * 1.05, 0.01)
        else:
            lr *= 0.7
    return h, c5_cur

def _corr_vals(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5(h, n):
    return float(np.max(_corr_vals(h, n)))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-15:
                break
    return h

def _upsample(h, nf, nt):
    xo = np.linspace(0, 2.0, nf, endpoint=False)
    xn = np.linspace(0, 2.0, nt, endpoint=False)
    return np.interp(xn, xo, h)

def _lag_grad(weights, h, n):
    g = np.zeros(n)
    dx = 2.0 / n
    for k in range(len(weights)):
        w = weights[k]
        if w < 1e-13:
            continue
        lag = k - (n - 1)
        lo = max(0, -lag)
        hi = min(n, n - lag)
        for i in range(lo, hi):
            g[i + lag] += w * (1.0 - h[i])
            g[i] -= w * h[i + lag]
    return g * dx

            best_c5, best_h = cc, hc

    # Phase 8: Medium pair search
    if rem() > 35:
        steps = np.array([0.0005, 0.001, 0.003, 0.008, 0.02,
                          -0.0005, -0.001, -0.003, -0.008, -0.02])
        hc = best_h.copy()
        hc, cc = _pair_search(hc, best_n, t0 + budget_s - 20, rng, steps)
        hc = _repair(hc, best_n)
        cc = _c5(hc, best_n)
        if cc < best_c5:
            best_c5, best_h = cc, hc

    # Phase 9: Ultra-fine pair search
    if rem() > 10:
        steps = np.array([0.0001, 0.0002, 0.0005, 0.001,
                          -0.0001, -0.0002, -0.0005, -0.001])
        hc = best_h.copy()
        hc, cc = _pair_search(hc, best_n, t0 + budget_s - 3, rng, steps)
        hc = _repair(hc, best_n)
        cc = _c5(hc, best_n)
        if cc < best_c5:
            best_c5, best_h = cc, hc

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, best_n)
    return (best_h, best_c5, best_n)
