   - Candidate d values: boundaries (-h[i], 1-h[j]) and all pairwise intersections
   - For the top K lags (by current value), compute intersections: d = (c_m - c_m')/(a_m' - a_m)
   - Evaluate f(d) at all candidates, take the best

Let me code this up:

```python
def _optimal_shift(h, i, j, cval, K=30, n=None):
    """Find optimal d for h[i]→h[i]+d, h[j]→h[j]-d"""
    if n is None:
        n = len(h)
    nL = 2 * n - 1
    cmax_pos = n - 1  # zero-lag position
    
    # Compute a_m = (1-h[i+L]-h[i-L]) - (1-h[j+L]-h[j-L])
    a = np.zeros(nL)
    L_range = np.arange(-(n-1), n)
    for idx, L in enumerate(L_range):
        # h[i+L] if valid
        hi_L = h[i+L] if 0 <= i+L < n else 0.0
        hi_mL = h[i-L] if 0 <= i-L < n else 0.0
        hj_L = h[j+L] if 0 <= j+L < n else 0.0
        hj_mL = h[j-L] if 0 <= j-L < n else 0.0
        a[idx] = (1 - hi_L - hi_mL) - (1 - hj_L - hj_mL)
    
    # f(d) = max_m (cval[m] + d * a[m])
    # Minimize over d ∈ [-h[i], 1-h[j]]
    d_lo = -h[i]
    d_hi = 1 - h[j]
    
    if d_lo >= d_hi - 1e-15:
        return 0.0
    
    # Get top K lags by current value
    top_idx = np.argsort(cval)[::-1][:K]
    
    # Collect candidate d values
    candidates = [d_lo, d_hi, 0.0]
    for ii in range(len(top_idx)):
        for jj in range(ii+1, len(top_idx)):
            m1, m2 = top_idx[ii], top_idx[jj]
            da = a[m1] - a[m2]
            if abs(da) > 1e-15:
                d = (cval[m2] - cval[m1]) / da
                if d_lo <= d <= d_hi:
                    candidates.append(d)
    
    # Also: for each top lag, find where it equals the current max
    cmax = np.max(cval)
    for m in top_idx:
        if abs(a[m]) > 1e-15:
            d = (cmax - cval[m]) / a[m]
            if d_lo <= d <= d_hi:
                candidates.append(d)
    
    best_d = 0.0
    best_val = np.inf
    for d in candidates:
        d = max(d_lo, min(d_hi, d))
        val = np.max(cval + d * a)
        if val < best_val:
            best_val = val
            best_d = d
    
    return best_d
```

This is O(K² + K + n) per pair, which is very fast.

```python
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)

        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:

            if d_lo - 1e-12 <= d <= d_hi + 1e-12:
                candidates.append(max(d_lo, min(d_hi, d)))

        if val < best_val - 1e-15:

    cval = _corr(h)
    c5_cur = float(np.max(cval)) * dx_val
    no_improve = 0

        order = rng.permutation(n)
        for i in order:

                new_c5 = float(np.max(cval)) * dx_val

                    improved = True
                else:

                    h[j] += d
                    cval = _corr(h)
        if not improved:
            break
    return h, c5_cur
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_obj(x, n, alpha):
    c = _corr(x) * (2.0 / n)
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dx_val = 2.0 / n
    c = _corr(x) * dx_val
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -400, 400)
    e = np.exp(s)
    total = float(np.sum(e))
    weights = e / total
    g = np.zeros(n)
    for k in range(len(c)):
        w = weights[k]
        if w < 1e-13:
            continue
        lag = k - (n - 1)
        i_min = max(0, -lag)
        i_max = min(n, n - lag)
        for i in range(i_min, i_max):
            jp = i + lag
            g[jp] += w * (1.0 - x[i])
            g[i] -= w * x[jp]
    return g * dx_val

def _slsqp(h_init, n, alpha, maxiter, bounds, cons):
    return minimize(
        lambda x: _smooth_obj(x, n, alpha), h_init,
        jac=lambda x: _smooth_grad(x, n, alpha),
        method='SLSQP', bounds=bounds, constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-16})

def _optimal_shift(h, i, j, cval, n, K=30):
    nL = 2 * n - 1
    a = np.zeros(nL)
    for m in range(nL):
        L = m - (n - 1)
        s1 = 0.0
        ip = i + L
        if 0 <= ip < n:
            s1 += h[ip]
        im = i - L
        if 0 <= im < n:
            s1 += h[im]
        s2 = 0.0
        jp = j + L
        if 0 <= jp < n:
            s2 += h[jp]
        jm = j - L
        if 0 <= jm < n:
            s2 += h[jm]
        a[m] = s1 - s2
    
    d_lo = -h[i]
    d_hi = 1.0 - h[j]
    if d_lo >= d_hi - 1e-15:
        return 0.0
    
    K = min(K, nL)
    top_idx = np.argsort(cval)[::-1][:K]
    
    cmax = float(np.max(cval))
    best_d = 0.0
    best_val = cmax
    
    # Check boundaries and 0
    for d in [d_lo, d_hi, 0.0]:
        val = float(np.max(cval + d * a))
        if val < best_val:
            best_val = val
            best_d = d
    
    # Pairwise intersections among top K
    for ii in range(len(top_idx)):
        m1 = top_idx[ii]
        for jj in range(ii + 1, len(top_idx)):
            m2 = top_idx[jj]
            da = a[m1] - a[m2]
            if abs(da) > 1e-15:
```

```python
def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h))) * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= tgt / s
    h = np.clip(h, 0.0, 1.0)
    diff = tgt - h.sum()
    if abs(diff) > 1e-14:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[k]); h[k] += a; diff -= a
            else:
                a = min(-diff, h[k]); h[k] -= a; diff += a
            if abs(diff) < 1e-14: break
    return h

def _lsm_obj(x, n, alpha):
    dx = 2.0 / n
    c = _corr(x) * dx
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lsm_grad(x, n, alpha):
    dx = 2.0 / n
    c = _corr(x) * dx
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    tot = float(np.sum(e))
    w = e / tot
    g = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-12: continue
        lag = k - (n - 1)
        lo = max(0, -lag); hi = min(n, n - lag)
        for i in range(lo, hi):
            jp = i + lag
            g[jp] += wk * (1.0 - x[i])
            g[i] -= wk * x[jp]
    return g * dx

                    method='SLSQP', bounds=bnd, constraints=cns,
                    options={'maxiter': maxit, 'ftol': 1e-16})
```
