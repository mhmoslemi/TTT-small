    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    stall = 0
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        gain = False
        for i in range(n):
            if time.time() > t_end:
                break
            for j in range(i + 1, n):
                hi, hj = h[i], h[j]
                for d in steps:
                    ni, nj = hi - d, hj + d
                    if ni >= 0.0 and nj <= 1.0:
                        h[i], h[j] = ni, nj
                        c = compute_c5(h, n)
                        if c < best - 1e-14:
                            best = c
                            gain = True
                            break
                        h[i], h[j] = hi, hj
                    ni, nj = hi + d, hj - d
                    if ni <= 1.0 and nj >= 0.0:
                        h[i], h[j] = ni, nj
                        c = compute_c5(h, n)
                        if c < best - 1e-14:
                            best = c
                            gain = True
                            break
                        h[i], h[j] = hi, hj
                    if gain:
                        break
        if not gain:
            stall += 1
            if stall > 8:
                break
        else:
            stall = 0
    return h, best

def rand_coord(h, n, t_end, nit, rng):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.025, 0.012, 0.006, 0.003, 0.001]
    for _ in range(nit):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        for d in steps:
            improved = False
            for sgn in (1, -1):
                ni, nj = hi + sgn * d, hj - sgn * d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i], h[j] = ni, nj
                    c = compute_c5(h, n)
                    if c < best - 1e-14:
                        best = c
                        improved = True
                        break
                    h[i], h[j] = hi, hj
            if improved:
                break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s
    rng = np.random.RandomState(seed + 9999)
    best_h = None
    best_c5 = np.inf
    best_n = 70
    try:
        init = np.array(initial_h_values, dtype=float).copy()
        best_n = len(init)
        best_h = project_h(init, best_n)
        best_c5 = compute_c5(best_h, best_n)
    except NameError:
        best_h = project_h(0.5 + 0.2 * rng.randn(70), 70)
        best_c5 = compute_c5(best_h, 70)
    n = best_n
    h_w = best_h.copy()
    # Phase 1: subgradient (new technique)
    t1 = t0 + budget_s * 0.12
    h_w, c5s = subgrad_descent(h_w, n, t1)
    if c5s < best_c5 - 1e-14:
        best_h, best_c5 = h_w.copy(), c5s
    # Phase 2: SLSQP escalating
    t2 = t0 + budget_s * 0.42
    for alpha in [30000, 100000, 300000, 800000]:
        if time.time() > t2:
            break
        for topk, fw in [(3, 0.0), (5, 0.01), (5, 0.04), (7, 0.01)]:
            if time.time() > t2:
                break
            hn = do_slsqp(h_w.copy(), n, alpha, topk, fw, 500)
            cn = compute_c5(hn, n)
            if cn < best_c5 - 1e-14:
                best_h, best_c5 = hn.copy(), cn
                h_w = hn
            elif cn < compute_c5(h_w, n) - 1e-14:
                h_w = hn
    # Phase 3: systematic coordinate descent
    t3 = t0 + budget_s * 0.72
    hcd, c5cd = sys_coord(h_w, n, t3, 12000)
    if c5cd < best_c5 - 1e-14:
        best_h, best_c5 = hcd.copy(), c5cd
        h_w = hcd
    else:
        h_w = best_h.copy()
    # Phase 4: perturb + reoptimize
    t4 = t0 + budget_s * 0.87
    for trial in range(4):
        if time.time() > t4:
            break
        h_pert = project_h(h_w + 0.06 * rng.randn(n), n)
        h_pert = do_slsqp(h_pert, n, 150000, 5, 0.02, 400)
        h_pert, cp = sys_coord(h_pert, n, min(t4, time.time() + 25), 2000)
        if cp < best_c5 - 1e-14:
            best_h, best_c5 = h_pert.copy(), cp
            h_w = h_pert
    # Phase 5: nearby n
    t5 = t0 + budget_s * 0.95
    for na in [n - 2, n + 2, n + 4, n - 4]:
        if time.time() > t5 or na < 20:
            continue
        for tr in range(2):
            if time.time() > t5:
                break
            ha = project_h(0.5 + 0.25 * rng.randn(na), na)
            ha = do_slsqp(ha, na, 80000, 3, 0.01, 350)
            ha, ca = sys_coord(ha, na, min(t5, time.time() + 20), 1500)
            if ca < best_c5 - 1e-14:
                best_h, best_c5, best_n = ha.copy(), ca, na
    # Phase 6: final polish
    if best_h is not None and time.time() < t0 + budget_s * 0.995:
        hf = do_slsqp(best_h.copy(), best_n, 500000, 5, 0.02, 400)
        cf = compute_c5(hf, best_n)
        if cf < best_c5 - 1e-14:
            best_h, best_c5 = hf.copy(), cf
        hf, cf2 = sys_coord(best_h, best_n, t0 + budget_s * 0.998, 8000)
        if cf2 < best_c5 - 1e-14:
            best_h, best_c5 = hf.copy(), cf2
    best_h = project_h(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
