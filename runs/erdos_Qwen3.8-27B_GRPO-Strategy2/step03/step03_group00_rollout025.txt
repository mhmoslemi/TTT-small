        for i in range(n):
            if time.time() > t_end:
                break
            for j in range(i+1, n):
                if time.time() > t_end:
                    break
                # Try small transfers
                hi, hj = h[i], h[j]

                        continue
                    h[i], h[j] = hi - amt, hj + amt
                    if _c5_value(h, n) < c5_cur:
                        c5_cur = _c5_value(h, n)
                        hi, hj = h[i], h[j]
                        improved = True
                    else:
                        h[i], h[j] = hi, hj
                amt = frac * min(hj, 1.0 - hi)
                # reversed
                amt = 0.5 * min(hj, 1.0 - hi)
                if amt > 1e-10:
                    h[i], h[j] = hi + amt, hj - amt
                    if _c5_value(h, n) < c5_cur:
                        c5_cur = _c5_value(h, n)
                        hi, hj = h[i], h[j]
                        improved = True
                    else:
                        h[i], h[j] = hi, hj
                h[i] = hi
        if not improved:
            break
    return h, c5_cur

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, n):
    return float(np.max(_corr(h))) * (2.0 / n)

def _proj(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    if s > target:
        lo, hi = 0.0, 2.0
        for _ in range(80):
            mid = 0.5 * (lo + hi)
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        return np.clip(h - 0.5 * (lo + hi), 0, 1)
    else:
        lo, hi = 0.0, 2.0
        for _ in range(80):
            mid = 0.5 * (lo + hi)
            if np.clip(h + mid, 0, 1).sum() < target:
                lo = mid
            else:
                hi = mid
        return np.clip(h + 0.5 * (lo + hi), 0, 1)

def _sobj(x, n, alpha):
    c = _corr(x) * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(x, n, alpha):
    dx_v = 2.0 / n
    c = _corr(x) * dx_v
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for mi in range(2 * n - 1):
        wi = w[mi] * dx_v
        if wi < 1e-15:
            continue
        lag = mi - (n - 1)
        if lag >= 0:
            i1 = np.arange(n - lag)
            i2 = i1 + lag
            g[i1] += wi * (1.0 - x[i2])
            g[i2] -= wi * x[i1]
        else:
            l = -lag
            i1 = np.arange(l, n)
            i2 = i1 - l
            g[i1] += wi * (1.0 - x[i2])
            g[i2] -= wi * x[i1]
    return g

def _con_fun(x):
    return x.sum() - x.shape[0] * 0.5

def _con_jac(x):
    return np.ones(x.shape[0])

def _do_slsqp(h0, n, alpha, maxit):
    bnd = [(0.0, 1.0)] * n
    cn = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    res = minimize(_sobj, h0, args=(n, alpha),
                   jac=_sgrad, method='SLSQP',
                   bounds=bnd, constraints=cn,
                   options={'maxiter': maxit, 'ftol': 1e-18})
    return res.x

def _topk_obj(x, n, k):
    c = _corr(x) * (2.0 / n)
    idx = np.argsort(c)[::-1][:k]
    return c[idx].sum()

def _topk_grad(x, n, k):
    dx_v = 2.0 / n
    c = _corr(x) * dx_v
    idx = np.argsort(c)[::-1][:k]
    g = np.zeros(n)
    for mi in idx:
        lag = mi - (n - 1)
        if lag >= 0:
            i1 = np.arange(n - lag)
            i2 = i1 + lag
            g[i1] += (1.0 - x[i2])
            g[i2] -= x[i1]
        else:
            l = -lag
            i1 = np.arange(l, n)
            i2 = i1 - l
            g[i1] += (1.0 - x[i2])
            g[i2] -= x[i1]
    return g * dx_v / k

def _do_topk_slsqp(h0, n, k, maxit):
    bnd = [(0.0, 1.0)] * n
    cn = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    res = minimize(_topk_obj, h0, args=(n, k),
                   jac=_topk_grad, method='SLSQP',
                   bounds=bnd, constraints=cn,
                   options={'maxiter': maxit, 'ftol': 1e-16})
    return res.x

def _local_search(h, n, t_end, rng):
    c5 = _c5_val(h, n)
    deltas = [0.002, 0.006, 0.015, 0.04, -0.002, -0.006, -0.015, -0.04]

                if time.time() > t_end:
                    break
                hj = h[j]

                    ni, nj = hi + d, hj - d

                            hi, hj = ni, nj
                            imp = True

                h[i] = hi
        if not imp:

    return h, c5

def _fine_random(h, n, t_end, rng):
    c5c = _c5_val(h, n)
    tiny = [0.0004, 0.001, 0.0025, -0.0004, -0.001, -0.0025]

        i = rng.randint(n)
        j = rng.randint(n)

            continue

            c = _c5_val(h, n)
            if c < c5c:
                c5c = c
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5c
