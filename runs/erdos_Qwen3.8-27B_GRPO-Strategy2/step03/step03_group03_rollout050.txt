    return m + np.log(np.sum(np.exp(np.clip(alpha*(c-m), -500, 500))))/alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n/2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target/s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0-h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0-h[idx])
            else:
                add = max(diff, h[idx]-1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-10:
                break
    return h

def _grad_c(h, n):
    """Gradient matrix: shape (2n-1, n), rows are gradients of each correlation."""
    n2 = 2*n - 1
    G = np.zeros((n2, n))
    for m in range(n2):
        k = m - (n-1)
        if k >= 0:
            G[m, k:] += (1.0 - h[:n-k])
            G[m, :n-k] -= h[k:]
        else:
            G[m, :n+k] += (1.0 - h[-k:])
            G[m, -k:] -= h[:n+k]
    return G

            n2 = 2*n - 1
            # Build LP: min t s.t. G@(h_new - h) + c <= t, 0<=h_new<=1, sum h_new = target
            # Variables: [h_new (n), t (1)]
            # Objective: minimize t → c_obj = [0,...,0, 1]
            # Constraints:
            #   G@(h_new - h) + c <= t  →  G@h_new - t <= t - c  →  G@h_new - t <= (0,...,0,...) - c + t... 
            #   Actually: G@h_new - t <= -c (since G@h_new - G@h + c <= t → G@h_new - t <= G@h - c)
            #   Wait: c + G@(h_new - h) <= t → G@h_new - G@h + c <= t → G@h_new - t <= G@h - c
            A_ub = np.zeros((n2, n+1))
            A_ub[:, :n] = G
            A_ub[:, n] = -1.0
            b_ub = (G @ h - c)  # G@h_new - t <= G@h - c
            # 0 <= h_new <= 1: bounds
            # sum h_new = target: equality
            A_eq = np.zeros((1, n+1))
            A_eq[0, :n] = 1.0
            b_eq = np.array([target])
            c_obj = np.zeros(n+1)
            c_obj[n] = 1.0
