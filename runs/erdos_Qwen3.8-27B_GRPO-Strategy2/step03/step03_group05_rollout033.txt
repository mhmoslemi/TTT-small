def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _shift_grad(h, L, n):
    """Gradient of correlation at shift L w.r.t. h"""
    g = np.zeros(n)
    if L == 0:
        g = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += (1.0 - h[:n - L])
        g[:n - L] -= h[L:]
    else:
        M = -L
        g[:n - M] += (1.0 - h[M:])
        g[M:] -= h[:n - M]
    return g

def _softobj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _active_slsqp(h, n, dx, K, max_iter=500):
    """Minimize max of top K correlations using SLSQP"""
    c = _corr(h)
    top_idx = np.argsort(c)[::-1][:K]
    
    def obj(v):
        return v[-1]  # minimize t
    
    def obj_grad(v):
        g = np.zeros_like(v)
        g[-1] = 1.0
        return g
    
    cons = []
    for ki in top_idx:
        L = ki - (n - 1)
        def make_con(L):
            def con_fn(v):
                h_v = v[:-1]
                t = v[-1]
                return t - _shift_val(h_v, L) * dx
            def con_jac(v):
                h_v = v[:-1]
                g = _shift_grad(h_v, L, n)
                jj = np.zeros(n+1)
                jj[:n] = g
                jj[n] = -1.0
                return jj
            return {'type': 'ineq', 'fun': con_fn, 'jac': con_jac}
        cons.append(make_con(L))
    
    # Equality constraint: sum(h) = n/2
    cons.append({
        'type': 'eq',
        'fun': lambda v: np.sum(v[:-1]) - n/2.0,
        'jac': lambda v: np.concatenate([np.ones(n), [0.0]])
    })
    
    x0 = np.concatenate([h, [np.max(c) * dx]])
    bounds = [(0,1)]*n + [(0, 1)]
    
    res = minimize(obj, x0, jac=obj_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return res.x[:-1]

        return np.sum(h[L:] * (1-h[:n-L]))

import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _shift_val(h, L):
    n = len(h)
    if L == 0:
        return np.sum(h * (1.0 - h))
    elif L > 0:
        return np.sum(h[L:] * (1.0 - h[:n - L]))
    else:
        M = -L
        return np.sum(h[:n - M] * (1.0 - h[M:]))

def _shift_grad(h, L):
    n = len(h)
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += (1.0 - h[:n - L])
        g[:n - L] -= h[L:]
    else:
        M = -L
        g[:n - M] += (1.0 - h[M:])
        g[M:] -= h[:n - M]
    return g

def _softobj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-12:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

**New idea: Direct subgradient method on the max**

The function f(h) = max_k c_k * dx is piecewise linear (each c_k is bilinear in h, but for fixed h it's a scalar). Its subgradient at h is the gradient of the active constraints.

1. Compute c for current h
2. Find all "near-active" shifts: those with c_k ≥ max(c) - eps
3. Compute the weighted average gradient of these shifts (weights proportional to how close they are to max)
4. Take a step: h_new = h - lr * grad_avg
5. Project to feasible set (0≤h≤1, sum=n/2)
6. If c5 decreased, accept; else halve lr
7. Repeat

This is a subgradient method. It's slower than gradient methods but doesn't need smoothing.

**Even better: Use the exact max gradient (not smoothed)**

At each step:
1. c = _corr(h)
2. max_c = max(c)
3. active = {k : c[k] ≥ max_c - eps} (eps small, like 1e-6)
4. grad = average of _shift_grad(h, L_k) for k in active
5. Project step: h_new = proj(h - lr * grad)
6. Line search: try a few lr values

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _shift_val(h, L):
    n = len(h)
    if L == 0:
        return float(np.sum(h * (1.0 - h)))
    elif L > 0:
        return float(np.sum(h[L:] * (1.0 - h[:n - L])))
    else:
        M = -L
        return float(np.sum(h[:n - M] * (1.0 - h[M:])))

def _shift_grad(h, L):
    n = len(h)
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += (1.0 - h[:n - L])
        g[:n - L] -= h[L:]
    else:
        M = -L
        g[:n - M] += (1.0 - h[M:])
        g[M:] -= h[:n - M]
    return g

def _softobj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-12:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _aslsqp_obj(v):
    return v[-1]

def _aslsqp_obj_grad(v):
    g = np.zeros(len(v))
    g[-1] = 1.0
    return g

def _aslsqp_ineq_fun(v, dx, L):
    return v[-1] - _shift_val(v[:-1], L) * dx

def _aslsqp_ineq_jac(v, dx, L):
    g = -_shift_grad(v[:-1], L)
    jj = np.zeros(len(v))
    jj[:-1] = g
    jj[-1] = 1.0
    return jj

def _aslsqp_eq_fun(v, n):
    return np.sum(v[:-1]) - n / 2.0

def _aslsqp_eq_jac(v, n):
    jj = np.zeros(len(v))
    jj[:-1] = 1.0
    return jj

def _subgrad_step(h, n, dx, lr, tol):
    c = _corr(h)
    max_c = np.max(c)
    active = c >= max_c - tol
    if not np.any(active):
        active = c >= max_c - 0.01
    n_act = np.sum(active)
    g = np.zeros(n)
    for k in np.where(active)[0]:
        L = int(k - (n - 1))
        g += _shift_grad(h, L)
    g /= max(n_act, 1)
    g -= g.mean()
    return h - lr * g

    x_old = np.linspace(0, 2, n_old + 1)
    y_old = np.concatenate([h, [h[0]]])
    x_new = np.linspace(0, 2, n_new + 1)
    y_new = np.interp(x_new[:-1], x_old, y_old)
    return y_new
