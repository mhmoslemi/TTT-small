    rng = np.random.RandomState(12345)
    best = h.copy()
    best_c = _eval(best)
    cur = h.copy()
    cur_c = best_c
    T = t0_start
    while time.time() - t0 < tlimit:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = 0.01 * rng.randn()
        ni, nj = cur[i] + d, cur[j] - d
        if not (0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0):
            continue
        cur[i], cur[j] = ni, nj
        c_new = _eval(cur)
        delta = c_new - cur_c
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            cur_c = c_new
            if c_new < best_c - 1e-15:
                best_c = c_new
                best = cur.copy()
        else:
            cur[i], cur[j] = ni - d, nj + d
        T *= t_factor
        if T < 1e-8:
            T = t0_start  # reheat
    return best, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _sf(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a

def _sj(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        mmin = max(0, k - n + 1)
        mmax = min(n - 1, k)
        idx = np.arange(mmin, mmax + 1)
        g[idx] += w[k] * (1.0 - 2.0 * x[k - idx])
    return g

def _do_slsqp(h, n, alpha, iters):
    bnds = [(0.0, 1.0)] * n
    nc = {'n': n, 'alpha': alpha}
    def _cf(x):
        return np.sum(x) - nc['n'] / 2.0
    def _cj(x):
        return np.ones(nc['n'])
    def _fo(x):
        return _sf(x, nc['alpha'])
    def _jo(x):
        return _sj(x, nc['alpha'])
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        res = minimize(_fo, h, jac=_jo, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': iters, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _binary_search(h, n, tlimit):
    t0 = time.time()
    best = np.where(h > 0.5, 1.0, 0.0).copy()
    n_ones = int(np.sum(best))
    if n_ones > n // 2:
        idxs = np.where(best == 1.0)[0]
        for idx in idxs[:n_ones - n // 2]:
            best[idx] = 0.0
    elif n_ones < n // 2:
        idxs = np.where(best == 0.0)[0]
        for idx in idxs[:n // 2 - n_ones]:
            best[idx] = 1.0
    best_c = _eval(best)
    ones = list(np.where(best > 0.5)[0])
    zeros = list(np.where(best <= 0.5)[0])
    while time.time() - t0 < tlimit:
        improved = False
        for pi in range(len(ones)):
            for pj in range(len(zeros)):
                i, j = ones[pi], zeros[pj]
                best[i], best[j] = 0.0, 1.0
                c = _eval(best)
                if c < best_c - 1e-15:
                    best_c = c
                    ones[pi] = j
                    zeros[pj] = i
                    improved = True
                else:
                    best[i], best[j] = 1.0, 0.0
        if not improved:
            break
    return best, best_c

def _pairwise_sweep(h, n, steps, tlimit):
    t0 = time.time()
    best_c = _eval(h)
    for d in steps:
        if time.time() - t0 > tlimit:
            break
        improved = True
        while improved and time.time() - t0 < tlimit:
            improved = False
            for i in range(n):
                for j in range(i + 1, n):
                    bc = best_c
                    bni, bnj = h[i], h[j]
                    for dd in (d, -d):
                        ni, nj = h[i] + dd, h[j] - dd
                        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                            h[i], h[j] = ni, nj
                            c = _eval(h)
                            if c < bc - 1e-15:
                                bc = c
                                bni, bnj = ni, nj
                    h[i], h[j] = bni, bnj
                    if bc < best_c - 1e-15:
                        best_c = bc
                        improved = True
    return h, best_c

def _triplet_sweep(h, n, tlimit):
    t0 = time.time()
    best_c = _eval(h)
    steps = [0.05, 0.02, 0.01, 0.005, 0.002]
    improved = True
    while improved and time.time() - t0 < tlimit:
        improved = False
        for d in steps:
            if time.time() - t0 > tlimit:
                break
            for i in range(n):
                for j in range(i + 1, n):
                    for k in range(j + 1, n):
                        for ii, jj, ti, tj in [(i, j, d, -d), (i, k, d, -d), (j, k, -d, d)]:
                            ni, nj = h[ii] + ti, h[jj] + tj
                            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                                h[ii], h[jj] = ni, nj
                                c = _eval(h)
                                if c < best_c - 1e-15:
                                    best_c = c
                                    improved = True
                                else:
                                    h[ii], h[jj] = ni - ti, nj - tj
    return h, best_c

def _sa_optimize(h, n, tlimit):
    t0 = time.time()
    rng = np.random.RandomState(777)
    best = h.copy()
    best_c = _eval(best)
    cur = h.copy()
    cur_c = best_c
    T = 0.003
    while time.time() - t0 < tlimit:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = 0.015 * rng.randn()
        ni, nj = cur[i] + d, cur[j] - d
        if not (0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0):
            continue
        cur[i], cur[j] = ni, nj
        c_new = _eval(cur)
        delta = c_new - cur_c
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            cur_c = c_new
            if c_new < best_c - 1e-15:
                best_c = c_new
                best = cur.copy()
        else:
            cur[i], cur[j] = ni - d, nj + d
        T *= 0.9997
        if T < 1e-8:
            T = 0.003
    return best, best_c
