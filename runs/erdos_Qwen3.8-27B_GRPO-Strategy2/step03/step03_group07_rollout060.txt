def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a; d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a; d += a
    return h

_state = {}

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx]); h[idx] += a; d -= a
            else:
                a = min(-d, h[idx]); h[idx] -= a; d += a
    return h

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def opt_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = _state['alpha'] * (corr - m)
    s = np.clip(s, -400, 400)
    val = m + np.log(np.sum(np.exp(s))) / _state['alpha']
    tk = _state['topk']
    if tk > 1:
        sc = np.sort(corr)[::-1][:tk]
        val += _state['fw'] * np.var(sc)
    val += _state['pen'] * (x.sum() - _state['target']) ** 2
    return val

def do_slsqp(h, n, alpha, topk, fw, pen, maxiter):

    bounds = [(0.0, 1.0)] * n

                       options={'maxiter': maxiter, 'ftol': 1e-16})
