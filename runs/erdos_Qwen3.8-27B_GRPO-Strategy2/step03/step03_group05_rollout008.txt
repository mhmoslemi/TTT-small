from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h))) * (2.0 / n)

    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
            else:
                a = -min(-diff, h[idx])
            h[idx] += a
            diff -= a
            if abs(diff) < 1e-14:
                break
    return h

def _grad_shifts(h, indices):
    n = len(h)
    g = np.zeros(n)
    for ki in indices:
        L = ki - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    return g

def _nm_handler(x, n):
    hx = _repair(x, n)
    return _c5(hx, n)

def _make_start(n, rng, style):
    h = np.full(n, 0.5)
    if style == 0:
        h += 0.3 * np.sin(2.0 * np.pi * np.arange(n) / (n / 4.0))
    elif style == 1:
        h += 0.4 * np.cos(2.0 * np.pi * np.arange(n) / 7.0)
    elif style == 2:
        h += 0.25 * np.sin(2.0 * np.pi * np.arange(n) / 3.0) + 0.15 * np.sin(2.0 * np.pi * np.arange(n) / 11.0)
    else:
        h += 0.2 * rng.randn(n)
    return _repair(np.clip(h, 0, 1), n)

def _pgd_loop(h, n, t0, deadline, m_start, m_max, lr_init):

    lr = lr_init
    h = h.copy()

            top = np.argsort(c)[::-1][:m]
            g = _grad_shifts(h, top)
            g -= g.mean()
            best_h = h.copy()
            best_c = c5_cur
            for shr in [1.0, 0.5, 0.25, 0.1, 0.05]:
                h_try = _repair(h - lr * shr * g, n)
                c_try = _c5(h_try, n)
                if c_try < best_c:
                    best_c = c_try
                    best_h = h_try
            if best_c < c5_cur - 1e-14:
                h = best_h
                c5_cur = best_c
                stall = 0
                lr = min(lr * 1.02, 0.3)
            else:
                stall += 1
                lr *= 0.7

                    break
    return h, c5_cur

def _nm_polish(h, n, t0, deadline, maxit):
    if time.time() > deadline:
        return h
    try:
        res = minimize(_nm_handler, h, args=(n,), method='Nelder-Mead',
                       options={'maxiter': maxit, 'xatol': 1e-9, 'fatol': 1e-11,
                                'adaptive': True})

def _pair_cd(h, n, t0, deadline, n_sweeps):
    dx = 2.0 / n
    c5_cur = _c5(h, n)
    rng_local = np.random.RandomState(int(h.sum() * 1000) % 10000)

    return h, c5_cur

    rng_local = np.random.RandomState(12345)
    c5_cur = _c5(h, n)

        else:

            h = _repair(h, n)
        T *= 0.9998
    return h, c5_cur
