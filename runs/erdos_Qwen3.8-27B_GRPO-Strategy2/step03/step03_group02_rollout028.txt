import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(h, target):
    n = len(h)
    lo, hi = -h.max() - 1.0, 1.0 - h.min() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    # Fix any tiny residual
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
        else:
            order = np.argsort(x)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - x[idx])
                x[idx] += add
                diff -= add
            else:
                sub = min(-diff, x[idx])
                x[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return x

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _grad_smooth_max(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s)
    w /= w.sum()
    
    Lags = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for k in range(len(Lags)):
        wk = w[k]
        if wk < 1e-15:
            continue
        L = Lags[k]
        f = np.zeros(n)
        b = np.zeros(n)
        if L >= 0:
            f[:n - L] = 1.0 - h[L:]
            b[L:] = h[:n - L]
        else:
            f[-L:] = 1.0 - h[:n + L]
            b[:n + L] = h[-L:]
        grad += wk * (f - b)
    return grad, c, w

def _pgrad_step(h, target, alpha, step, n_inner=30):
    """Projected gradient descent for smooth max."""
    h = h.copy()
    for _ in range(n_inner):
        grad, c, w = _grad_smooth_max(h, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        # Backtracking line search
        s = step
        f_cur = _smooth_max(c, alpha)
        for _ in range(30):
            h_new = _project(h - s * grad, target)
            c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
            f_new = _smooth_max(c_new, alpha)
            if f_new <= f_cur - 1e-4 * s * gnorm * gnorm:
                h = h_new
                c = c_new
                break
            s *= 0.5
        else:
            break
        step = s * 1.5  # increase step for next iteration
    return h

def _coordinate_refine(h, n, dx, target, time_limit, n_coarse=20):
    """Exact coordinate-pair descent on the max correlation."""
    improved = False
    c = _corr(h)
    cur_max = float(np.max(c))
    order = np.random.permutation(n)
    
    for ai, i in enumerate(order):
        for j in order[ai+1:]:
            if time.time() > time_limit:
                return h, cur_max, improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            # Compute how c changes: c_new = c + d * (g_i - g_j)
            # g[p] = d(c[p])/d(h[i]) - d(c[p])/d(h[j]) ... 
            # Actually let's use the _g_vec approach
            # c[k] changes by d * (dc[k]/dh[i] - dc[k]/dh[j])
            # For efficiency, precompute the difference vector
            
            # dc/dh[i]: for lag L, contribution is (1-h[i+L])*I[valid] - h[i-L]*I[valid]
            gi = np.zeros(2*n-1)
            gj = np.zeros(2*n-1)
            Lags = np.arange(-(n-1), n)
            for k, L in enumerate(Lags):
                # d(c[k])/d(h[i]):
                # (1-h[i+L]) if 0<=i+L<n else 0, minus h[i-L] if 0<=i-L<n else 0

                    gi[k] += (1.0 - h[i + L])
                if 0 <= i - L < n:
                    gi[k] -= h[i - L]
                # d(c[k])/d(h[j]):
                if 0 <= j + L < n:
                    gj[k] += (1.0 - h[j + L])
                if 0 <= j - L < n:
                    gj[k] -= h[j - L]
            
            gd = gi - gj
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(c[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = np.argmin(vals)
            if vals[bi] >= cur_max - 1e-14:
                continue
            lo = max(d_min, ds[max(0, bi-1)])
            hi_b = min(d_max, ds[min(n_coarse-1, bi+1)])
            for _ in range(30):

                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            h[i] = hi + d_best
            h[j] = hj - d_best
            c = c + d_best * gd
            cur_max = float(np.max(c))
            improved = True
    return h, cur_max, improved

def _fourier_init(n, seed, n_harm=15):
    """Generate a random Fourier-based initialization."""
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2.0, n, endpoint=False) + 1.0/n  # cell centers
    h = np.full(n, 0.5)
    for m in range(1, n_harm + 1):
        amp = rng.uniform(0.05, 0.25) * (0.8 ** (m // 5))
        phase = rng.uniform(0, 2*np.pi)
        h += amp * np.cos(np.pi * m * x / 2.0 + phase)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    h = _project(h, target)
    return h

def _slsqp_step(h, n, dx, target, alpha, maxiter=200):
    def obj(x):
        return _smooth_max(np.correlate(x, 1.0-x, mode='full'), alpha)
    def obj_grad(x):
        g, _, _ = _grad_smooth_max(x, alpha)
        return g
    def eq_con(x):
        return float(np.sum(x)) - target
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 40})
        return _project(res.x, target)
    except Exception:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(h, target):
    n = len(h)
    lo, hi = -h.max() - 1.0, 1.0 - h.min() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h + mid, 0.0, 1.0).sum() < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - x) if diff > 0 else np.argsort(x)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - x[idx]); x[idx] += a; diff -= a
            else:
                a = min(-diff, x[idx]); x[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return x

def _smooth_max_val(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _g_vec(h, p, n):
    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _grad_smooth(h, alpha):
    n = len(h)
    c = _corr(h)
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        L = k - (n - 1)
        f = np.zeros(n)
        b = np.zeros(n)
        if L >= 0:
            f[:n - L] = 1.0 - h[L:]
            b[L:] = h[:n - L]
        else:
            f[-L:] = 1.0 - h[:n + L]
            b[:n + L] = h[-L:]
        grad += w[k] * (f - b)
    return grad, c

def _pgrad(h, target, alpha, step, n_iter=40):
    h = h.copy()
    for _ in range(n_iter):
        grad, c = _grad_smooth(h, alpha)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        s = step
        fc = _smooth_max_val(c, alpha)
        for _ in range(25):
            hx = _project(h - s * grad, target)
            cn = _corr(hx)
            fn = _smooth_max_val(cn, alpha)
            if fn <= fc - 1e-5 * s * gn * gn:
                h = hx
                break
            s *= 0.5
        step = max(s * 1.3, 1e-6 / alpha)
    return h

def _coord_sweep(h, n, target, t_limit, ncoarse=22, nbisect=32):
    c = _corr(h)
    cur_max = float(np.max(c))
    improved = False
    order = np.random.permutation(n)
    for ai in range(n):
        i = order[ai]
        for j in order[ai + 1:]:
            if time.time() > t_limit:
                return h, cur_max, improved
            hi, hj = h[i], h[j]
            dlo = max(-hi, hj - 1.0)
            dhi = min(1.0 - hi, hj)
            if dhi - dlo < 1e-13:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(dlo, dhi, ncoarse)
            vals = np.max(c[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = np.argmin(vals)
            if vals[bi] >= cur_max - 1e-14:
                continue
            lo = max(dlo, ds[max(0, bi - 1)])
            hh = min(dhi, ds[min(ncoarse - 1, bi + 1)])
            for _ in range(nbisect):
                mid = 0.5 * (lo + hh)
                if np.max(c + mid * gd) <= np.max(c + hh * gd):
                    hh = mid
                else:
                    lo = mid
            d = 0.5 * (lo + hh)
            h[i] = hi + d
            h[j] = hj - d
            c = c + d * gd
            nm = float(np.max(c))
            if nm < cur_max:
                cur_max = nm
            improved = True
    return h, cur_max, improved

def _slsqp(h, n, target, alpha, mxit=200):
    def obj(x):
        return _smooth_max_val(_corr(x), alpha)
    def objg(x):
        g, _ = _grad_smooth(x, alpha)
        return g
    def eqc(x):
        return float(np.sum(x)) - target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eqc}]
    try:
        r = minimize(obj, h, jac=objg, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': mxit, 'ftol': 1e-14, 'maxls': 40})
        return _project(r.x, target)
    except Exception:
        return h
