    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()

    if abs(diff) < 1e-14:
        return h

        order = np.argsort(-h)
        rem = -diff
        for idx in order:
            take = min(rem, h[idx])
            h[idx] -= take
            rem -= take
            if rem < 1e-14:
                break
    else:

        rem = diff

            take = min(rem, 1.0 - h[idx])
            h[idx] += take
            rem -= take
            if rem < 1e-14:

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    h = np.clip(h, 0.0, 1.0).copy()

                break

            h[idx] -= take

    n_iter = 0
    while time.time() < t_end:
        if n_iter % 500 == 0:
            c = _corr(h)
            m_star = int(np.argmax(c))
            m = m_star - (n - 1)
            if m >= 0:
                g_star = np.zeros(n)
                g_star[:n - m] = 1.0 - h[m:]
                g_star[m:] = -h[:n - m]
            else:
                p = -m
                g_star = np.zeros(n)
                g_star[p:] = 1.0 - h[:n - p]
                g_star[:n - p] = -h[p:]
        # Gradient-guided pair selection
        free_i = np.where(h < 0.999)[0]
        free_j = np.where(h > 0.001)[0]
        if len(free_i) == 0 or len(free_j) == 0:
