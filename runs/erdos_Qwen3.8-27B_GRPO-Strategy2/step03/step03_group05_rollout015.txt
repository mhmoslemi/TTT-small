```python
import cvxpy as cp

def solve_lp(h0, n, dx, t_current):
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2]
    
    corr0 = np.correlate(h0, 1-h0, mode='full')
    for k in range(2*n-1):
        L = k - (n-1)
        # Linearize f_k(h) around h0
        # f_k(h) = sum h[i]*(1-h[j]) for valid (i,j) at lag k
        # ∂f_k/∂h[i] = (1-h0[j]) for forward, -h0[j'] for backward
        grad = np.zeros(n)
        if L == 0:
            grad = 1 - 2*h0
        elif L > 0:
            grad[L:] += (1 - h0[:n-L])
            grad[:n-L] -= h0[L:]
        else:
            M = -L
            grad[:n-M] += (1 - h0[M:])
            grad[M:] -= h0[:n-M]
        
        f_k_linear = float(np.dot(grad, h)) + (corr0[k] - float(np.dot(grad, h0)))
        constraints.append(f_k_linear <= t)
    
    problem = cp.Problem(cp.Minimize(t), constraints)
    problem.solve(solver=cp.ECOS)  # or GPIL, CLARABEL
    
    if problem.status == 'optimal':
        return h.value, t.value
    return None, None
```

```python
    """Compute gradients of all correlate components w.r.t. h"""
    grads = np.zeros((2*n-1, n))

        L = k - (n-1)
        g = np.zeros(n)
        if L == 0:
            g = 1.0 - 2.0 * h
        elif L > 0:

        else:
            M = -L
            g[:n-M] += (1.0 - h[M:])
            g[M:] -= h[:n-M]

                diff -= a
```

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _grad_one(h, L, n):
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += (1.0 - h[:n-L])
        g[:n-L] -= h[L:]
    else:
        M = -L
        g[:n-M] += (1.0 - h[M:])
        g[M:] -= h[:n-M]
    return g

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _softobj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    w = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        pk = w[k]
        if pk < 1e-14:
            continue
        L = k - (n - 1)
        g += pk * _grad_one(x, L, n)
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _obj_wrap(x, alpha, lam):
    return _softobj(x, alpha, lam)

def _jac_wrap(x, alpha, lam):
    return _softgrad(x, alpha, lam)

def _solve_slp(h0, n, dx, max_iter=60, time_limit=90):
    h = h0.copy()
    t0 = time.time()
    for _ in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        corr0 = _corr(h)
        c5_cur = float(np.max(corr0)) * dx
        var_h = cp.Variable(n)
        var_t = cp.Variable()
        constraints = [var_h >= 0, var_h <= 1, cp.sum(var_h) == n / 2.0]
        for k in range(2 * n - 1):
            L = k - (n - 1)
            g = _grad_one(h, L, n)
            f_lin = corr0[k] + float(np.dot(g, var_h - h))
            constraints.append(f_lin <= var_t)
        prob = cp.Problem(cp.Minimize(var_t), constraints)
        solved = False
        for solver in [cp.ECOS, cp.PDLP]:
            try:
                prob.solve(solver=solver, max_iters=20000)
                solved = True
                break
            except Exception:
                continue
        if not solved or prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        h_lp = _repair(np.array(var_h.value).flatten(), n)
        best_step_c5 = c5_cur
        best_alpha = 0.0
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            h_tr = (1 - alpha) * h + alpha * h_lp
            h_tr = _repair(h_tr, n)
            c5_tr = _c5(h_tr, dx)
            if c5_tr < best_step_c5:
                best_step_c5 = c5_tr
                best_alpha = alpha
        if best_alpha == 0.0:
            break
        h = (1 - best_alpha) * h + best_alpha * h_lp
        h = _repair(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h0 = _repair(np.array(initial_h_values, dtype=float), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    def elapsed():
        return time.time() - t_start

    # Phase 1: SLP
    if elapsed() < budget_s - 450:
        h_slp = _solve_slp(h0, n, dx, max_iter=80, time_limit=100)
        c5_s = _c5(h_slp, dx)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_slp.copy()

    # Phase 2: Progressive L-BFGS-B soft-max
    alphas = [50, 200, 1000, 5000, 30000, 120000]
    for a in alphas:
        if elapsed() > budget_s - 320:
            break
        try:
            res = minimize(_obj_wrap, best_h.copy(), jac=_jac_wrap,
                           args=(a, 50.0), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-14})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc
        except Exception:
            pass

    # Phase 3: SLP again from improved point
    if elapsed() < budget_s - 250:
        h_slp = _solve_slp(best_h, n, dx, max_iter=60, time_limit=80)
        c5_s = _c5(h_slp, dx)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_slp.copy()

    # Phase 4: Active-set projected gradient with top-K

        for m_top in [2, 3, 4, 6, 8]:

            for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:

                    break

                while improved:

                    g = np.zeros(n)

                    else:

    # Phase 5: Pair coordinate descent
    if elapsed() < budget_s - 80:

        for _sweep in range(80):
            if elapsed() > budget_s - 80:
                break
            g = _softgrad(h, 50000, 0.0)

            improved = False
            for ii in range(len(order)):
                i = order[ii]
                for jj in range(ii + 1, min(ii + 25, len(order))):
                    j = order[jj]
                    best_d = 0.0
                    best_c = c5_cur
                    for d in dlist:
                        ni, nj = h[i] + d, h[j] - d
                        if 0 <= ni <= 1 and 0 <= nj <= 1:
                            h[i], h[j] = ni, nj
                            c = _c5(h, dx)
                            if c < best_c:
                                best_c = c
                                best_d = d
                            h[i], h[j] = ni - d, nj + d
                    if best_d != 0.0:
                        h[i] += best_d
                        h[j] -= best_d
                        h = _repair(h, n)
                        c5_cur = _c5(h, dx)
                        improved = True
                        if c5_cur < best_c5:
                            best_c5 = c5_cur
                            best_h = h.copy()
                        break
            if not improved:
                break

    # Phase 6: Short simulated annealing
    if elapsed() < budget_s - 35:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        T = 3e-5
        for _it in range(30000):
            if elapsed() > budget_s - 35:
                break
            i, j = rng.randint(n), rng.randint(n)

                continue
            d = rng.uniform(-0.03, 0.03)
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:

                    best_h = h.copy()
```
