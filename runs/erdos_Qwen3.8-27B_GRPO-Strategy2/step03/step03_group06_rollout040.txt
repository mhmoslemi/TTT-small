import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair_cont(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-10:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _repair_bin(h, n, rng):
    h = (np.array(h, dtype=float) >= 0.5).astype(float)
    target = int(n / 2)
    cnt = int(h.sum())
    if cnt > target:
        ones = np.where(h == 1.0)[0]
        h[rng.choice(ones, cnt - target, replace=False)] = 0.0
    elif cnt < target:
        zeros = np.where(h == 0.0)[0]
        h[rng.choice(zeros, target - cnt, replace=False)] = 1.0
    return h

def _smoobj(x, alpha, target, pen):
    c = _corr(x)
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -600, 600)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (x.sum() - target) ** 2

def _smoobj_eq(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -600, 600)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sumeq(x, target):
    return x.sum() - target

def _lbfgs(h, n, alpha, maxiter, bounds):
    target = n / 2.0
    try:
        res = minimize(_smoobj, h, args=(alpha, target, 500.0),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-9})
        return _repair_cont(res.x, n)
    except Exception:
        return h

def _slsqp(h, n, alpha, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sumeq, 'args': (target,)}]
    try:
        res = minimize(_smoobj_eq, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair_cont(res.x, n)
    except Exception:
        return h

def _bin_sa(h, n, dx, rng, deadline):
    h = _repair_bin(h, n, rng)
    best = h.copy()
    best_c5 = _c5(h, dx)
    cur = h.copy()
    cur_c5 = best_c5
    T = 0.008
    T_min = 0.00005
    cool = 0.99997
    ones_idx = np.where(h == 1.0)[0]
    zeros_idx = np.where(h == 0.0)[0]
    while time.time() < deadline and T > T_min:
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        cur[i], cur[j] = 0.0, 1.0
        nc = _c5(cur, dx)
        d = nc - cur_c5
        if d < 0 or rng.random() < np.exp(-d / T):
            cur_c5 = nc
            ones_idx = np.where(cur == 1.0)[0]
            zeros_idx = np.where(cur == 0.0)[0]
            if nc < best_c5:
                best_c5 = nc
                best = cur.copy()
        else:
            cur[i], cur[j] = 1.0, 0.0
        T *= cool
    return best, best_c5

def _lag_target(h, n, dx, rng, deadline, top_k=5):
    best_c5 = _c5(h, dx)
    h = h.copy()
    while time.time() < deadline:
        c = _corr(h)
        wl = np.argmax(c)
        d = wl - (n - 1)
        if d > 0:
            v = np.arange(0, n - d)
            contrib = h[v] * (1.0 - h[v + d])
        elif d < 0:
            v = np.arange(-d, n)
            contrib = h[v] * (1.0 - h[v + d])
        else:
            v = np.arange(n)
            contrib = h[v] * (1.0 - h[v])
        if len(v) == 0:
            continue
        pi = int(np.argmax(contrib))
        i = v[pi]
        j = rng.randint(n)
        if j == i:
            continue
        amt = 0.001 * rng.random() + 0.0003
        if h[i] > amt:
            ni, nj = h[i] - amt, h[j] + amt
            if nj <= 1.0:
                h[i], h[j] = ni, nj
                nc = _c5(h, dx)
                if nc < best_c5:
                    best_c5 = nc
                else:
                    h[i], h[j] = ni + amt, nj - amt
    return _repair_cont(h, n), best_c5

def _fine(h, n, dx, rng, deadline, n_tries=200000):
    best_c5 = _c5(h, dx)
    h = h.copy()
    scales = [0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    per = max(400, n_tries // len(scales))
    for sc in scales:
        imp = False
        for _ in range(per):
            if time.time() >= deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            dv = sc * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + dv, h[j] - dv
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            nc = _c5(h, dx)
            if nc < best_c5 - 1e-15:
                best_c5 = nc
                imp = True
            else:
                h[i], h[j] = h[i] - dv, h[j] + dv
        if not imp and sc < 0.0005:
            break
    return _repair_cont(h, n), best_c5

def _phase_a_restart(h, n, dx, rng, bounds, deadline):
    hp = h + rng.normal(0, 0.015, n)
    hp = _repair_cont(hp, n)
    for a in [1000, 8000, 40000]:
        if time.time() >= deadline:
            break
        ht = _lbfgs(hp, n, a, 150, bounds)
        c5 = _c5(ht, dx)
        if c5 < _c5(h, dx):
            h = ht
    ht = _slsqp(h, n, 20000, 400)
    c5 = _c5(ht, dx)
    if c5 < _c5(h, dx):
        h = ht
    return h
