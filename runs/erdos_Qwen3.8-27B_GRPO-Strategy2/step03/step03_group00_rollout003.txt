def cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def c5(h):
    return float(np.max(cvals(h)))

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def sg_subgrad(h, n, topk=5, sharp=100.0):
    c = cvals(h)
    m = np.max(c)
    idx = np.argsort(c)[::-1][:topk]
    w = np.exp(sharp * (c[idx] - m))
    w /= w.sum()
    g = np.zeros(n)
    for ii, wi in zip(idx, w):
        lag = ii - (n - 1)
        if lag >= 0:
            i = np.arange(0, n - lag)
            g[i + lag] += wi * (1.0 - h[i])
            g[i] -= wi * h[i + lag]
        else:
            i = np.arange(-lag, n)
            g[i + lag] += wi * (1.0 - h[i])
            g[i] -= wi * h[i + lag]
    return g
