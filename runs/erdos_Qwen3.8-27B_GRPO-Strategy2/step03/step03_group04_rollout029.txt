```python
    """Solve min_{x in [0,1]} max(D0 + (1+lambda)*x - x^2, max_k(A[k] + (B[k]+lambda)*x))"""

    # Linear k: l(x) = A[k] + (B[k]+lambda)*x

        lk_c = A[k]
```

```python
        hi = min(n, n - k)
        # B_k
        b = 0.0
        if lo <= i - k < hi:
            b += 1.0 - h[i - k]
        if lo <= i < hi and 0 <= i + k < n:
            b -= h[i + k]
        # const_k = corr[k_idx] - h[i] * B_k
        A[idx] = corr[k_idx] - h[i] * b
        B[idx] = b
        idx += 1
    
    return _solve_1d(D0, A, B, lam)
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-12:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-14:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-14:
                    break
    return h


def _solve_1d(D0, A, B, lam):
    m = len(A)
    cands = [0.0, 1.0]
    xq = (1.0 + lam) / 2.0
    if 0.0 < xq < 1.0:
        cands.append(xq)
    for k in range(m):
        b2 = -(1.0 - B[k])
        c2 = -(D0 - A[k])
        disc = b2 * b2 - 4.0 * c2
        if disc >= -1e-15:
            disc = max(0.0, disc)
            sq = np.sqrt(disc)
            r1 = (-b2 + sq) / 2.0
            r2 = (-b2 - sq) / 2.0
            if -1e-12 <= r1 <= 1.0 + 1e-12:
                cands.append(np.clip(r1, 0.0, 1.0))
            if -1e-12 <= r2 <= 1.0 + 1e-12:
                cands.append(np.clip(r2, 0.0, 1.0))
    grid = np.linspace(0.0, 1.0, 31)[1:-1]
    cands.extend(grid.tolist())
    Bm = B + lam
    best_x = 0.0
    best_val = np.inf
    for x in cands:
        x = np.clip(x, 0.0, 1.0)
        q = D0 + (1.0 + lam) * x - x * x
        f = max(q, np.max(A + Bm * x)) if m > 0 else q
        if f < best_val:
            best_val = f
            best_x = x
    return best_x


def _coord_sweep(h, n, lam, order):
    for i in order:
        ci = h[i]
        corr = _corr(h)
        D0 = corr[n - 1] - ci + ci * ci
        m = 2 * n - 1
        A = np.empty(m - 1)
        B = np.empty(m - 1)
        idx = 0
        for k_idx in range(m):
            if k_idx == n - 1:
                continue
            k = k_idx - (n - 1)
            lo = max(0, -k)
            hi = min(n, n - k)
            b = 0.0
            if lo <= i - k < hi:
                b += 1.0 - h[i - k]
            if lo <= i < hi and 0 <= i + k < n:
                b -= h[i + k]
            A[idx] = corr[k_idx] - ci * b
            B[idx] = b
            idx += 1
        x_opt = _solve_1d(D0, A, B, lam)
        h[i] = x_opt
    return h


def _c5_and_corr(h):
    c = _corr(h)
    return float(np.max(c)), c


def _eval_c5(h, dx):
    return float(np.max(_corr(h)) * dx)


def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -400.0, 400.0)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _obj_smooth(x, alpha):
    return _smooth_max(_corr(x), alpha)


def _eq_con(x, target):
    return np.sum(x) - target


def _do_slsqp(x0, bounds, target, alpha, maxiter):
    res = minimize(_obj_smooth, x0, args=(alpha,), method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': _eq_con, 'args': (target,)}],
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return res.x
```
