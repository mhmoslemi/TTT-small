import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smo(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smo_pen(h, alpha, tgt, pen):
    return _smo(h, alpha) + pen * (h.sum() - tgt) ** 2

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= tgt / s
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-14:
                break
    return h

def _lbfgs(h, n, dx, alpha, mi=300):
    tgt = n / 2.0
    bnds = [(0.0, 1.0)] * n
    best_h, best_c = h.copy(), _c5(h, dx)
    for p in [50.0, 500.0, 5000.0]:
        res = minimize(lambda x: _smo_pen(x, alpha, tgt, p), best_h,
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': mi, 'ftol': 1e-18, 'gtol': 1e-8})
        hc = _repair(res.x, n)
        c = _c5(hc, dx)
        if c < best_c:
            best_h, best_c = hc, c
    return best_h, best_c

def _slsqp(h, n, dx, alpha, mi=500):
    tgt = n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt}]
    res = minimize(lambda x: _smo(x, alpha), h, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': mi, 'ftol': 1e-16})
    hc = _repair(res.x, n)
    return hc, _c5(hc, dx)

        jmax = int(np.argmax(c_vals))

        k = jmax - (n - 1)
        # Find positions that contribute most to this lag
        # c[jmax] = sum_i h[i] * (1-h[i+k]) for appropriate i

        i = rng.randint(n)
        j = rng.randint(n)

    with np.errstate(divide='ignore', invalid='ignore'):

    # Only consider j < k, 0 < X < 1
    tri = np.triu(np.ones((m, m), dtype=bool), k=1)
    X = X[tri]
    X = X[np.isfinite(X)]
    X = X[(X > 0) & (X < 1)]
    
    candidates = np.concatenate([[0.0, 1.0], X])
    
    # Evaluate max at each candidate
    vals = a[:, None] * candidates[None, :] + b[:, None]  # (m, len(cand))
    maxvals = vals.max(axis=0)
    idx = np.argmin(maxvals)
    
    h[i] = candidates[idx]
    return maxvals[idx] * (2.0 / n)  # C5 value
```

This is fully vectorized and should be very fast! For n=70, m=139:
- A, B are 139x139 = 19,321 elements: fast
- X computation: fast
- vals is 139 * (num_candidates) where num_candidates could be up to ~9600: 1.33M elements - fast with NumPy

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smo(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smo_pen(h, alpha, tgt, pen):
    return _smo(h, alpha) + pen * (h.sum() - tgt) ** 2

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= tgt / s
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-14:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a; d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a; d += a
            if abs(d) < 1e-14:
                break
    return h

def _lbfgs(h, n, dx, alpha, mi=300):
    tgt = n / 2.0
    bnds = [(0.0, 1.0)] * n
    best_h, best_c = h.copy(), _c5(h, dx)
    for p in [50.0, 500.0, 5000.0]:
        try:
            res = minimize(lambda x: _smo_pen(x, alpha, tgt, p), best_h,
                           method='L-BFGS-B', bounds=bnds,
                           options={'maxiter': mi, 'ftol': 1e-18, 'gtol': 1e-8})
            hc = _repair(res.x, n)
            c = _c5(hc, dx)
            if c < best_c:
                best_h, best_c = hc, c
        except:
            pass
    return best_h, best_c

def _slsqp(h, n, dx, alpha, mi=500):
    tgt = n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt}]
    try:
        res = minimize(lambda x: _smo(x, alpha), h, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': mi, 'ftol': 1e-16})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except:
        return h, _c5(h, dx)

def _coord_opt(h, n, i):
    """Optimize single coordinate i to minimize C5, others fixed."""
    m = 2 * n - 1
    old = h[i]
    h[i] = 0.0
    c0 = _corr(h).copy()
    h[i] = 1.0
    c1 = _corr(h).copy()
    a = c1 - c0
    b = c0
    
    A = a[:, None] - a[None, :]
    B = b[None, :] - b[:, None]
    with np.errstate(divide='ignore', invalid='ignore'):
        X = B / A
    tri = np.triu(np.ones((m, m), dtype=bool), k=1)
    vals = X[tri]
    vals = vals[np.isfinite(vals)]
    vals = vals[(vals > 0) & (vals < 1)]
    cand = np.concatenate([[0.0, 1.0], vals])
    
    # Evaluate
    Y = a[:, None] * cand[None, :] + b[:, None]
    mx = Y.max(axis=0)
    idx = np.argmin(mx)
    
    h[i] = cand[idx]
    return mx[idx]

def _coord_descent(h, n, dx, rng, deadline, max_sweeps=1000):
    """Coordinate descent: cycle through all coordinates, optimize each."""
    best_c = _c5(h, dx)
    h = h.copy()
    n_coords = len(h)
    imp_count = 0
    for sweep in range(max_sweeps):
        if time.time() >= deadline:
            break
        # Random order for diversity
        order = rng.permutation(n_coords) if sweep % 3 == 0 else np.arange(n_coords)
        improved = False
        for i in order:
            _coord_opt(h, n, i)
        # Check constraint
        h = _repair(h, n)
        c = _c5(h, dx)
        if c < best_c - 1e-15:
            best_c = c
            improved = True
            imp_count = 0
        else:
            imp_count += 1
        if imp_count > 5:
            break
    return h, best_c

Let me also consider: what if I do "paired" coordinate descent? I.e., optimize (h[i], h[j]) jointly as a 2D problem, with the constraint h[i] + h[j] = h[i]_old + h[j]_old (to maintain the sum).

For a 2D min-max: min_{x,y} max_k (a_k*x + b_k*y + c_k) s.t. x+y=s, 0≤x≤1, 0≤y≤1
This reduces to: min_x max_k (a_k*x + b_k*(s-x) + c_k) = min_x max_k ((a_k-b_k)*x + b_k*s + c_k) s.t. max(0, s-1) ≤ x ≤ min(1, s)

So it's still a 1D problem! The slope for each lag k is (a_k - b_k), and the intercept is b_k*s + c_k.

This is elegant. Let me implement paired coordinate descent:

```python
def _paired_coord_opt(h, n, i, j):
    """Optimize h[i] and h[j] jointly, maintaining h[i]+h[j]=const."""
    m = 2 * n - 1
    s = h[i] + h[j]  # must be preserved
    
    # Compute needed quantities
    # For each lag k, corr[k] = a_k*h[i] + b_k*h[j] + c_k
    # where a_k, b_k, c_k are determined by the other coordinates
    
    # Set h[i]=0, h[j]=0: get c_k
    h[i] = 0.0; h[j] = 0.0
    c_arr = _corr(h).copy()
    
    # Set h[i]=1, h[j]=0: get a_k
    h[i] = 1.0
    a_arr = _corr(h).copy()
    
    # Set h[i]=0, h[j]=1: get b_k
    h[i] = 0.0; h[j] = 1.0
    b_arr = _corr(h).copy()
    
    # Now corr[k] = a_k*x + b_k*y + c_k where x=h[i], y=h[j], x+y=s
    # = (a_k - b_k)*x + b_k*s + c_k
    slope = a_arr - b_arr
    intercept = b_arr * s + c_arr
    
    # min_x max_k (slope_k * x + intercept_k) s.t. max(0,s-1) <= x <= min(1,s)
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if lo >= hi:
        h[i] = lo; h[j] = s - lo
        return
    
    # Find optimal x
    A = slope[:, None] - slope[None, :]
    B = intercept[None, :] - intercept[:, None]
    with np.errstate(divide='ignore', invalid='ignore'):
        X = B / A
    tri = np.triu(np.ones((m, m), dtype=bool), k=1)
    vals = X[tri]
    vals = vals[np.isfinite(vals)]
    vals = vals[(vals > lo) & (vals < hi)]
    cand = np.concatenate([[lo, hi], vals])
    
    Y = slope[:, None] * cand[None, :] + intercept[:, None]
    mx = Y.max(axis=0)
    idx = np.argmin(mx)
    
    h[i] = cand[idx]
    h[j] = s - h[i]
```

This means my linear-coordinate optimization is only valid for coordinates that don't appear in the "self-product" of the correlation. But in np.correlate(h, 1-h, 'full'), the "self" terms (where the same index appears in both h and 1-h) only occur at the central lag (k=0, i.e., j=n-1 in the output).

Wait, let me re-examine. np.correlate(h, 1-h, 'full')[j] = Σ_n h[n + j - (N-1)] * (1-h[n])

For j = N-1 (k=0): Σ_n h[n] * (1-h[n]) = Σ_n (h[n] - h[n]²)

For j ≠ N-1 (k ≠ 0): the indices h[n+k] and h[n] are different (for k≠0), so there's no h[n]² term. It's linear in each h[n] separately.

- When optimizing a single coordinate h[i]:

So the 1D problem is: min_x max(C_0 + x - x², a_k1*x + b_k1, a_k2*x + b_k2, ...) s.t. 0≤x≤1

This is min_x max(quadratic, lines). The maximum of a concave quadratic and lines is... well, the quadratic x - x² is concave (opens downward), so its maximum is at x=0.5. The lines are just lines.

The "upper envelope" is the max of all these curves. I need to find where this envelope is minimized.

The critical points are:
1. x = 0, x = 1 (boundaries)
2. x = 0.5 (vertex of quadratic)
3. Points where the quadratic crosses a line: C_0 + x - x² = a_k*x + b_k → x² + (a_k - 1)*x + (b_k - C_0) = 0
4. Points where two lines cross: a_k*x + b_k = a_m*x + b_m → x = (b_m - b_k)/(a_k - a_m)

For each of these candidate points, evaluate the max and pick the smallest.

This is O(m²) for line-line crossings + O(m) for quadratic-line crossings + O(1) for special points. Total O(m²) per coordinate.

Let me implement this:

```python
def _coord_opt_analytic(h, n, i):
    m = 2 * n - 1
    mid = n - 1  # index of k=0 lag
    
    # Get slopes and intercepts for linear lags (k≠0)
    h[i] = 0.0
    c0 = _corr(h).copy()
    h[i] = 1.0
    c1 = _corr(h).copy()
    h[i] = 0.5
    c05 = _corr(h).copy()
    
    # For linear lags (j ≠ mid): corr[j] = a_j * x + b_j
    a = np.empty(m)
    b = np.empty(m)
    for j in range(m):
        if j == mid:
            continue
        # a_j = (c1[j] - c0[j]) / 1.0
        a[j] = c1[j] - c0[j]
        b[j] = c0[j]
    
    # For quadratic lag (j == mid): corr[mid] = -x² + x + C0
    # c0[mid] = C0, c05[mid] = -0.25 + 0.5 + C0 = C0 + 0.25
    # So: -x² + x + C0 where C0 = c0[mid]
    C0 = c0[mid]
    
    # Build candidate list
    candidates = [0.0, 1.0, 0.5]
    
    # Line-line crossings
    lin_idx = [j for j in range(m) if j != mid]
    for p in range(len(lin_idx)):
        for q in range(p+1, len(lin_idx)):
            j, k = lin_idx[p], lin_idx[q]
            da = a[j] - a[k]
            if abs(da) > 1e-15:
                x = (b[k] - b[j]) / da
                if 0.0 < x < 1.0:
                    candidates.append(x)
    
    # Quadratic-line crossings: -x² + x + C0 = a_j*x + b_j
    # x² + (a_j - 1)*x + (b_j - C0) = 0
    for j in lin_idx:
        A_coef = 1.0
        B_coef = a[j] - 1.0
        C_coef = b[j] - C0
        disc = B_coef**2 - 4*A_coef*C_coef
        if disc >= 0:
            sqrt_disc = np.sqrt(disc)
            x1 = (-B_coef + sqrt_disc) / (2*A_coef)
            x2 = (-B_coef - sqrt_disc) / (2*A_coef)
            for x in [x1, x2]:
                if 0.0 < x < 1.0:
