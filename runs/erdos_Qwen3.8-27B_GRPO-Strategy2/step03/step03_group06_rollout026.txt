def _compute_c5(h, dx):

def _repair_cont(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)

            order = np.argsort(h)

                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-10:

    return h

def _bin_sa(h_init, n, dx, rng, deadline, n_iters=2000000):
    # Binary SA: state is which n/2 positions are 1
    target = n // 2
    # Initialize from h_init: pick top n/2 values
    h = h_init.copy()

    zeros_idx = np.where(h_bin == 0)[0]

        for i in range(extra):
            h_bin[ones_idx[i]] = 0
            zeros_idx = np.where(h_bin == 0)[0]
    elif len(ones_idx) < target:

        for i in range(need):
            h_bin[zeros_idx[i]] = 1

    c5_cur = _compute_c5(h_bin, dx)
    best_c5 = c5_cur
    best_state = h_bin.copy()
    
    T = 0.5
    T_min = 0.001
    cool = (T_min / T) ** (1.0 / n_iters)
    
    for it in range(n_iters):

            break
        # Pick a random one and a random zero, swap
        ones_idx = np.where(h_bin == 1)[0]
