    if s < 1e-12:

                h[idx] += add
                diff -= add
            else:

                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

    dx_val = 2.0 / n
    g = np.zeros(n)

            if 0 <= jp < n:
                g[jp] += (1.0 - h[i])
                g[i] -= h[jp]

    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _topk_grad(h, n, k_indices):
    dx_val = 2.0 / n
    g = np.zeros(n)
    for k_idx in k_indices:
        lag = k_idx - (n - 1)
        if lag == 0:
            for i in range(n):
                g[i] += (1.0 - h[i]) - h[i]
        else:
            i_min = max(0, -lag)
            i_max = min(n, n - lag)
            for i in range(i_min, i_max):
                jp = i + lag
                g[jp] += (1.0 - h[i])
                g[i] -= h[jp]
    return g * dx_val / len(k_indices)

def _proj_step(h, g, step, n):
    z = h - step * g
    target = n / 2.0
    lo, hi = z.min() - 1.0, z.max() + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(z + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)

def _fine_search(h, n, budget_end, rng):

def _anneal(h_init, n, budget_end, rng):
    h = h_init.copy()
    c5_cur = _c5(h, n)
    best_h = h.copy()
    best_c5 = c5_cur
    T = 0.001
    min_T = 1e-10
    cool = 0.9998
    steps = np.array([0.002, 0.005, 0.01, 0.03])
    while time.time() < budget_end:
        i = rng.randint(n)
        j = (i + rng.randint(1, n)) % n
        d = steps[rng.randint(len(steps))] * (1 if rng.rand() < 0.5 else -1)
        hi, hj = h[i], h[j]
        ni, nj = hi + d, hj - d
        if ni < -0.01 or ni > 1.01 or nj < -0.01 or nj > 1.01:
            continue
        h[i] = max(0.0, min(1.0, ni))
        h[j] = max(0.0, min(1.0, nj))
        c5_new = _c5(h, n)
        if c5_new < c5_cur or (T > 1e-11 and rng.rand() < np.exp(-(c5_new - c5_cur) / T)):
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            h[i], h[j] = hi, hj
        T = max(T * cool, min_T)
    return best_h, best_c5

def _slsqp_smooth(h_init, n, alpha, maxiter):
    def obj(x):
        c = _corr(x)
        m = np.max(c)
        s = alpha * (c - m)
        s = np.clip(s, -400, 400)
        return m + np.log(np.sum(np.exp(s))) / alpha
    
    def grad(x):
        c = _corr(x)
        m = np.max(c)
        s = alpha * (c - m)
        s = np.clip(s, -400, 400)
        e = np.exp(s)
        total = e.sum()
        if total < 1e-300:
            return np.zeros(n)
        w = e / total
        g = np.zeros(n)
        for k_idx in range(len(c)):
            wk = w[k_idx]
            if wk < 1e-14:
                continue
            lag = k_idx - (n - 1)
            i_min = max(0, -lag)
            i_max = min(n, n - lag)
            for i in range(i_min, i_max):
                jp = i + lag
                g[jp] += wk * (1.0 - x[i])
                g[i] -= wk * x[jp]
        return g * (2.0 / n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h_init, jac=grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x
