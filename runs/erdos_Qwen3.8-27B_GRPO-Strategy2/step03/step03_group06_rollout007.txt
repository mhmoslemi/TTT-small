def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    e = alpha * (c - m)
    e = np.clip(e, -500, 500)
    return m + np.log(np.exp(e).sum()) / alpha

def _smooth_grad(x, alpha):
    c = _corr(x)
    m = c.max()
    e = alpha * (c - m)
    e = np.clip(e, -500, 500)
    s = np.exp(e)
    w = s / s.sum()
    v = 1.0 - 2.0 * x
    full = np.correlate(w, v, mode='full')
    n = len(x)
    return alpha * full[n - 1::-1]

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _slsqp_analytic(h, n, dx, alpha, maxiter=500):

        return _smooth_obj(x, alpha)
    def obj_grad(x):
        return _smooth_grad(x, alpha)
    con = {'type': 'eq', 'fun': lambda x: x.sum() - target,
           'jac': lambda x: np.ones(n)}
    try:

                       bounds=bounds, constraints=[con],

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = c.max()
    e = alpha * (c - m)
    e = np.clip(e, -500, 500)
    return m + np.log(np.exp(e).sum()) / alpha

def _smooth_grad(x, alpha):
    c = _corr(x)
    m = c.max()
    e = alpha * (c - m)
    e = np.clip(e, -500, 500)
    s = np.exp(e)
    w = s / s.sum()
    v = 1.0 - 2.0 * x
    n = len(x)
    full = np.correlate(w, v, mode='full')
    return full[n - 1:2 * n - 1]

def _p_norm_obj(x, p):
    c = _corr(x)
    return np.power(np.sum(np.power(c, p)), 1.0 / p)

def _p_norm_grad(x, p):
    c = _corr(x)
    n = len(x)
    v = 1.0 - 2.0 * x
    full = np.correlate(np.power(c, p - 1), v, mode='full')
    return full[n - 1:2 * n - 1] * np.power(np.sum(np.power(c, p)), 1.0 / p - 1.0)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _slsqp_logsumexp(h, n, dx, alpha, maxiter=600):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': lambda x: x.sum() - target,
           'jac': lambda x: np.ones(n)}
    try:
        res = minimize(lambda x: _smooth_obj(x, alpha), h,
                       jac=lambda x: _smooth_grad(x, alpha),
                       method='SLSQP', bounds=bounds, constraints=[con],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _slsqp_pnorm(h, n, dx, p, maxiter=400):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': lambda x: x.sum() - target,
           'jac': lambda x: np.ones(n)}
    try:
        res = minimize(lambda x: _p_norm_obj(x, p), h,
                       jac=lambda x: _p_norm_grad(x, p),
                       method='SLSQP', bounds=bounds, constraints=[con],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _local_search(h, n, dx, rng, deadline, max_iters=500000):
    best_c5 = _c5(h, dx)
    h = h.copy()
    scale = 0.02
    patience = 0
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * (2.0 * rng.random() - 1.0)
        ni, nj = h[i] + d, h[j] - d
        if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
            continue
        h[i], h[j] = ni, nj
        c5_new = _c5(h, dx)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            patience = 0
            scale = max(0.0005, scale * 0.95)
        else:
            h[i], h[j] = ni - d, nj + d
            patience += 1
            if patience > 20000:
                scale = max(0.0003, scale * 0.7)
                patience = 0
    return h, best_c5

def _block_search(h, n, dx, rng, deadline, max_iters=30000):
    best_c5 = _c5(h, dx)
    h = h.copy()
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        bs = rng.choice([2, 3, 4])
        i = rng.randint(0, n - bs + 1)
        j = rng.randint(0, n - bs + 1)
        if abs(i - j) < bs:
            continue
        eps = rng.uniform(0.003, 0.015) * (1.0 if rng.random() < 0.5 else -1.0)
        seg_i = h[i:i + bs] + eps
        seg_j = h[j:j + bs] - eps
        if np.any(seg_i < -1e-9) or np.any(seg_i > 1 + 1e-9) or \
           np.any(seg_j < -1e-9) or np.any(seg_j > 1 + 1e-9):
            continue
        h[i:i + bs] = seg_i
        h[j:j + bs] = seg_j
        c5_new = _c5(h, dx)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
        else:
            h[i:i + bs] -= eps
            h[j:j + bs] += eps
    return h, best_c5

def _perturb(h, n, rng, kind=0):
    hp = h.copy()
    if kind == 0:
        f = rng.randint(1, 6)
        ph = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.04, 0.15)
        hp += amp * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 1:
        block = rng.randint(3, min(8, n // 4))
        i = rng.randint(0, n - block)
        hp[i:i + block] = np.roll(hp[i:i + block], rng.randint(1, block))
    elif kind == 2:
        j = rng.randint(n // 4, 3 * n // 4)
        hp = np.concatenate([hp[j:], hp[:j]])
    else:
        hp += rng.normal(0, 0.04, n)
    return _repair(hp, n)

def _targeted_local(h, n, dx, rng, deadline, max_iters=100000):
    best_c5 = _c5(h, dx)
    h = h.copy()
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        c = _corr(h)
        top_k = np.argsort(c)[-5:]
        i = top_k[rng.randint(len(top_k))]
        lo = max(0, i - n + 1)
        hi = min(n - 1, i)
        idx = rng.randint(lo, hi + 1)
        j2 = rng.randint(n)
        if j2 == idx:
            continue
        scale = max(0.001, 0.015 * (1.0 - it / max_iters))
        d = scale * (2.0 * rng.random() - 1.0)
        ni, nj = h[idx] + d, h[j2] - d
        if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
            continue
        h[idx], h[j2] = ni, nj
        c5_new = _c5(h, dx)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
        else:
            h[idx], h[j2] = ni - d, nj + d
    return h, best_c5
