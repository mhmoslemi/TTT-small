def _grad_all_lags(h, n):

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _grad_all_lags(h, n):
    """Gradient of each corr[m] w.r.t. h[j]. Shape: (2n-1, n)"""
    grad = np.zeros((2*n-1, n))
    idx = np.arange(n)
    for k in range(-(n-1), n):
        m = k + (n-1)
        mask1 = (idx + k >= 0) & (idx + k < n)
        grad[m, mask1] += (1.0 - h[idx[mask1] + k])
        mask2 = (idx - k >= 0) & (idx - k < n)
        grad[m, mask2] -= h[idx[mask2] - k]
    return grad

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _subgradient_step(h, n, lr, top_k, time_limit):
    """One subgradient step minimizing max correlation."""
    corr = _corr(h)
    max_c = np.max(corr)

        active = np.arange(len(corr))
    else:

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _grad_all_lags(h, n):
    grad = np.zeros((2*n-1, n))
    idx = np.arange(n)
    for k in range(-(n-1), n):
        m = k + (n-1)
        mask1 = (idx + k >= 0) & (idx + k < n)
        grad[m, mask1] += (1.0 - h[idx[mask1] + k])
        mask2 = (idx - k >= 0) & (idx - k < n)
        grad[m, mask2] -= h[idx[mask2] - k]
    return grad

def _smooth_max_and_grad(h, n, alpha):
    corr = _corr(h)
    c_max = corr.max()
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -100, 100)
    exps = np.exp(shifted)
    s = exps / exps.sum()
    val = c_max + np.log(exps.sum()) / alpha
    G = _grad_all_lags(h, n)
    grad = s @ G
    return val, grad, corr

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _sd_optimize(h, n, t0, time_budget, alpha_start=50, alpha_end=200):
    """Subgradient descent optimization."""
    dx = 2.0 / n
    best = h.copy()
    best_c5 = _c5(h, dx)
    n_iter = 0
    lr = 0.02
    stall = 0
    
    while time.time() - t0 < time_budget:
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, n_iter / 2000)
        val, grad, corr = _smooth_max_and_grad(h, n, alpha)
        
        # Adaptive learning rate
        h_new = h - lr * grad
        h_new = _repair(h_new, n)
        new_c5 = _c5(h_new, dx)
        
        if new_c5 < best_c5 - 1e-10:
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = h.copy()
            stall = 0
            lr = min(lr * 1.05, 0.05)
        else:
            stall += 1
            lr *= 0.95
            if stall > 200:
                # Restart with perturbation
                h = _repair(h + 0.03 * np.random.randn(n), n)
                lr = 0.02
                stall = 0
        
        n_iter += 1
        if n_iter % 500 == 0:
            lr = max(lr * 0.9, 0.001)
    
    return best, best_c5

def _sweep_round(h, n, corr, time_limit, n_coarse=20, n_bisect=22):
    improved = False
    order = np.random.permutation(n)
    n_improved = 0
    for ai in range(n):
        i = order[ai]
        for j in order[ai+1:]:
            if time.time() > time_limit:
                return h, corr, improved, n_improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            # Gradient of corr w.r.t. d = (h[i]-d) and (h[j]+d)
            # d(corr)/d(d) = -dc/dh[i] + dc/dh[j]
            idx = np.arange(n)
            gd = np.zeros(2*n-1)
            for k in range(-(n-1), n):
                m = k + (n-1)
                # contribution from h[i]: (1-h[i+k]) * I(valid) - h[i-k] * I(valid)
                g_i = 0.0
                if 0 <= i+k < n:
                    g_i += 1.0 - h[i+k]
                if 0 <= i-k < n:
                    g_i -= h[i-k]
                g_j = 0.0
                if 0 <= j+k < n:
                    g_j += 1.0 - h[j+k]
                if 0 <= j-k < n:
                    g_j -= h[j-k]
                gd[m] = -g_i + g_j
            # Find best d via bisection
            cur_max = corr.max()
            lo, hi_b = d_min, d_max
            if hi_b - lo < 1e-10:
                continue
            # Coarse scan
            ds = np.linspace(lo, hi_b, n_coarse)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = np.argmin(vals)
            if vals[bi] >= cur_max - 1e-12:
                continue
            lo = max(lo, ds[max(0, bi-1)])
            hi_b = min(hi_b, ds[min(n_coarse-1, bi+1)])
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            h[i] += d_best
            h[j] -= d_best
            corr = corr + d_best * gd
            improved = True
            n_improved += 1
    return h, corr, improved, n_improved

from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _grad_row(h, p, n):
    lags = np.arange(2*n - 1) - (n - 1)
    g = np.zeros(2*n - 1)
    m1 = (p + lags >= 0) & (p + lags < n)
    m2 = (p - lags >= 0) & (p - lags < n)
    g[m1] += (1.0 - h[p + lags[m1]])
    g[m2] -= h[p - lags[m2]]
    return g

def _grad_all(h, n):
    return np.array([_grad_row(h, p, n) for p in range(n)])

def _smg(h, n, alpha):
    corr = _corr(h)
    cm = corr.max()
    sh = np.clip(alpha * (corr - cm), -300, 300)
    ex = np.exp(sh)
    w = ex / ex.sum()
    G = _grad_all(h, n)
    return G @ w, corr

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-14:
        ord_ = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in ord_:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-14:

    return h

    target = n/2

        corr = _corr(x)
        cm = corr.max()

        return cm + np.log(np.sum(np.exp(sh)))/alpha
