from scipy.optimize import minimize, fmin
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _grad_c_k(h, k):
    """Gradient of c(k) = sum h[i]*(1-h[i+k]) w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    L = k  # shift
    if L == 0:
        g = 1.0 - 2.0 * h
    elif L > 0:
        # c(k) = sum_{i=0}^{n-1-L} h[i]*(1-h[i+L])
        for i in range(n - L):
            g[i] += (1.0 - h[i + L])
            g[i + L] += (-h[i])
    else:
        M = -L
        for i in range(M, n):
            g[i] += (1.0 - h[i - M])
            g[i - M] += (-h[i])
    return g

def _grad_max_c(h, top_k=5):
    """Subgradient of max_k c(k)"""
    c = _corr(h)
    idx = np.argsort(c)[::-1][:top_k]
    g = np.zeros(len(h))
    for ki in idx:
        L = ki - (len(h) - 1)
        g += _grad_c_k(h, L)
    g /= top_k
    return g

def _softobj(x, alpha, lam):
    n = len(x)
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    penalty = lam * (np.sum(x) - n / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam):
    n = len(x)
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        g += pk * _grad_c_k(x, L)
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

                else:
                    step *= 0.7
                    if step < 1e-5: break

    # Phase 5: Block Nelder-Mead on groups of 5
    if elapsed() < budget_s - 100:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        for sweep in range(15):
            if elapsed() > budget_s - 100: break
            improved = False
            for start in range(0, n - 4, 4):
                if elapsed() > budget_s - 100: break
                idx = np.arange(start, min(start + 5, n))
                h_save = h.copy()
                h = _block_nm(h, idx, n, dx, maxfev=60)
                c5_new = _c5(h, dx)
                if c5_new < c5_cur - 1e-14:
                    c5_cur = c5_new
                    improved = True
                    if c5_new < best_c5:
                        best_c5, best_h = c5_new, h.copy()
                else:
                    h = h_save
            if not improved: break

    # Phase 6: Gradient-directed pair search
    if elapsed() < budget_s - 50:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        for sweep in range(300):
            if elapsed() > budget_s - 50: break
            c = _corr(h)
            top = np.argsort(c)[::-1][:3]
            g = np.zeros(n)
            for ki in top:
                g += _grad_ck(h, ki - (n - 1))
            g /= len(top)
            order = np.argsort(np.abs(g))[::-1][:25]
            found = False
            for ii in range(len(order)):
                i = order[ii]
                for jj in range(ii + 1, min(ii + 18, len(order))):
                    j = order[jj]
                    best_d, best_c = 0.0, c5_cur
                    for d in [0.003, 0.007, 0.015, 0.03, -0.003, -0.007, -0.015, -0.03,
                              0.001, -0.001, 0.002, -0.002]:
                        ni, nj = h[i] + d, h[j] - d
                        if 0 <= ni <= 1 and 0 <= nj <= 1:
                            h[i], h[j] = ni, nj
                            c = _c5(h, dx)
                            if c < best_c:
                                best_c, best_d = c, d
                            h[i], h[j] = ni - d, nj + d
                    if best_d != 0:
                        h[i] += best_d; h[j] -= best_d
                        h = _repair(h, n)
                        c5_cur = _c5(h, dx)
                        found = True
                        if c5_cur < best_c5:
                            best_c5, best_h = c5_cur, h.copy()
                        break
                if found: break
            if not found: break

    # Phase 7: Ultra-fine random coordinate polish
    if elapsed() < budget_s - 5:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        for _ in range(100000):
            if elapsed() > budget_s - 5: break
            i, j = rng.randint(n), rng.randint(n)
            if i == j: continue
            d = rng.choice([0.0004, 0.001, 0.0025, 0.006, -0.0004, -0.001, -0.0025, -0.006])
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5_cur - 1e-16:
                    c5_cur = c
                    if c < best_c5:
                        best_c5, best_h = c, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
