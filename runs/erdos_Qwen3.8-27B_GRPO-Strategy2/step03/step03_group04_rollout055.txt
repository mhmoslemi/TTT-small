                pass  # handled above

    # Final check
    h_best = _unpack35(best_a, n)
    final_c5 = _c5_h(h_best, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Verify integral
    assert abs(h_best.sum() - n / 2.0) < 1e-6, f"Integral violated: {h_best.sum()}"
    assert np.all(h_best >= -1e-10) and np.all(h_best <= 1.0 + 1e-10)

    return (h_best, best_c5, n)
