from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _proj_box_hyperplane(h, target):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=target}."""
    n = len(h)
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = 0.5 * (lo + hi)
        hp = np.clip(h - lam, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)

def _topk_grad(h, top_k):
    """Compute gradient of sum of top-k overlaps."""
    n = len(h)
    c = _corr(h)
    # Index in c corresponds to lag: c[j] is lag j-(n-1)
    order = np.argsort(c)[::-1][:top_k]
    g = np.zeros(n)

                g[L:] += (1.0 - h[:n-L])
                g[:n-L] -= h[L:]
        else:
            M = -L
            if M < n:
                g[:n-M] += (1.0 - h[M:])
                g[M:] -= h[:n-M]
    return g / top_k

    """Nesterov accelerated projected gradient on active set."""
    target = float(n) / 2.0
    h = _proj_box_hyperplane(h.copy(), target)
    c5_cur = _c5(h, dx)
    best_c5 = c5_cur
    best_h = h.copy()
    
    beta = 0.9  # momentum
    step = 0.1
    top_k = 5
    v = np.zeros(n)
    
    for it in range(max_iter):
        if time.time() - t_start > t_budget:
            break
        
        # Nesterov lookahead
        y = h + beta * v
        g = _topk_grad(y, top_k)
        
        # Projected gradient step
        h_new = _proj_box_hyperplane(y - step * g, target)
        v = h_new - y  # new momentum (simplified)
        # Actually for Nesterov: v_new = beta * v + (h_new - y)
        # But simpler: v = h_new - h (basic momentum)
        
        c5_new = _c5(h_new, dx)
        if c5_new < c5_cur:
            v = 0.9 * v + 0.1 * (h_new - h)
            h = h_new
            c5_cur = c5_new
            step = min(step * 1.05, 0.5)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            step *= 0.7
            if step < 1e-7:
                break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _proj(h, target):
    n = len(h)
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _topk_grad(h, topk):
    n = len(h)
    c = _corr(h)
    order = np.argsort(c)[::-1][:topk]
    g = np.zeros(n)
    for ki in order:
        L = ki - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif 0 < L < n:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        elif -n < L < 0:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    return g

def _active_grad_set(h, topk):
    n = len(h)
    c = _corr(h)
    order = np.argsort(c)[::-1][:topk]
    g = np.zeros(n)
    for ki in order:
        L = ki - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif 0 < L < n:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        elif -n < L < 0:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    return g

def _projected_grad_descent(h, n, dx, target, t_end, topk_schedule=None):
    h = _proj(h.copy(), target)
    c5_cur = _c5(h, dx)
    best_c5 = c5_cur
    best_h = h.copy()
    step = 0.3
    n_it = 0
    while time.time() < t_end:
        n_it += 1
        if topk_schedule:
            frac = min(n_it / float(len(topk_schedule) * 50), 1.0)
            idx = int(frac * len(topk_schedule))
            topk = topk_schedule[min(idx, len(topk_schedule) - 1)]
        else:
            topk = 5
        g = _active_grad_set(h, topk)
        g -= g.mean()
        h_try = _proj(h - step * g, target)
        c5_try = _c5(h_try, dx)
        if c5_try < c5_cur - 1e-16:
            h = h_try
            c5_cur = c5_try
            step = min(step * 1.03, 1.0)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h.copy()
        else:
            step *= 0.75
            if step < 1e-8:
                break
    return best_h, best_c5

def _pair_cd(h, n, dx, target, t_end, max_sweeps=200):
    h = h.copy()
    c5_cur = _c5(h, dx)
    best_c5 = c5_cur
    best_h = h.copy()
    c_full = _corr(h)
    c_max = float(np.max(c_full))
    deltas = np.array([0.001, 0.003, 0.008, 0.02, 0.05,
                       -0.001, -0.003, -0.008, -0.02, -0.05])
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        g = _active_grad_set(h, 5)
        order = np.argsort(np.abs(g))[::-1][:30]
        improved = False
        for ii in range(len(order)):
            i = order[ii]
            for jj in range(ii + 1, min(ii + 15, len(order))):
                j = order[jj]
                best_d = 0.0
                best_c = c5_cur
                for d in deltas:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < best_c:
                            best_c = c
                            best_d = d
                        h[i], h[j] = ni - d, nj + d
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    h = _proj(h, target)
                    c5_cur = _c5(h, dx)
                    improved = True
                    if c5_cur < best_c5:
                        best_c5 = c5_cur
                        best_h = h.copy()
                    break
        if not improved:
            break
    return best_h, best_c5

def _anneal(h, n, dx, target, t_end, rng, n_iter=30000):
    h = h.copy()
    c5_cur = _c5(h, dx)
    best_c5 = c5_cur
    best_h = h.copy()
    T = 3e-5
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.03, 0.03)
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        h = _proj(h, target)
        c_new = _c5(h, dx)
        if c_new < c5_cur or (T > 1e-9 and rng.rand() < np.exp(-(c_new - c5_cur) / T)):
            c5_cur = c_new
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d
            h = _proj(h, target)
        T *= 0.9998
    return best_h, best_c5

def _softobj(x, alpha, lam, n):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    penalty = lam * (np.sum(x) - n / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam, n):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif 0 < L < n:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        elif -n < L < 0:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= x[:n - M]
    g /= alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _lbfgs_round(h, n, dx, target, alpha, max_iter, t_end):
    bounds = [(0.0, 1.0)] * n
    try:
        def obj(x):
            return _softobj(x, alpha, 50.0, n)
        def jac(x):
            return _softgrad(x, alpha, 50.0, n)
        res = minimize(obj, h.copy(), jac=jac, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-16})
        hc = _proj(res.x, target)
        c5 = _c5(hc, dx)
        return hc, c5
    except Exception:
        return h.copy(), _c5(h, dx)

def run(seed=42, budget_s=1100, **kwargs):
