    """Compute Jacobian: J[i,k] = dc[k]/dh[i] for all i,k."""
    n = len(h)

    J = np.zeros((n, m))

        if L >= 0:
            for i in range(n - L):
                J[i, k] += (1.0 - h[i+L])
                J[i+L, k] -= h[i]
        else:
            M = -L
            for i in range(n - M):
                J[i+M, k] += (1.0 - h[i])
                J[i, k] -= h[i+M]
    return J

import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:

                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])

                    diff += a

                            if lo < db < hi:

                # Find best d

                best_val = np.inf

                    val = np.max(c + d * delta)
                    if val < best_val:

                    h[j] -= best_d
                    h[i] = np.clip(h[i], 0, 1)
                    h[j] = np.clip(h[j], 0, 1)
                    cmax = best_val
                    improved = True
        if not improved:
            break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _sf(h, alpha, topk):
    c = _corr(h)
    m = len(c)
    if topk < m:
        idx = np.argpartition(c, -topk)[-topk:]
        vals = c[idx]
    else:
        vals = c
    mm = vals.max()
    s = np.clip(alpha * (vals - mm), -400, 400)
    s -= s.max()
    return mm + np.log(np.exp(s).sum()) / alpha

def _sj(h, alpha, topk):
    n = len(h)
    c = _corr(h)
    m = len(c)
    g = np.zeros(n)
    if topk < m:
        idx = np.argpartition(c, -topk)[-topk:]
        vals = c[idx]
    else:
        idx = np.arange(m)
        vals = c
    mm = vals.max()
    s = np.clip(alpha * (vals - mm), -400, 400)
    s -= s.max()
    ex = np.exp(s)
    w = ex / ex.sum()
    for ki in range(len(idx)):
        wk = w[ki]
        if wk < 1e-15:
            continue
        k = idx[ki]
        L = k - (n - 1)
        if L >= 0:
            for i in range(n - L):
                g[i] += wk * (1.0 - h[i + L])
                g[i + L] -= wk * h[i]
        else:
            M = -L
            for i in range(n - M):
                g[i + M] += wk * (1.0 - h[i])
                g[i] -= wk * h[i + M]
    return g

def _con_fun(x):
    return np.sum(x) - len(x) / 2.0

def _con_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, topk, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_sf, h.copy(), jac=_sj, args=(alpha, topk),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _interp(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return _repair(np.interp(xd, xs, h_src), n_dst)

def _refine(h, rng, n_iter, t_stop):
    n = len(h)
    h = h.copy()
    c5 = _eval(h)
    scale = 0.008
    no_imp = 0
    for _ in range(n_iter):
        if time.time() > t_stop:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5 - 1e-15:
            c5 = c5n
            no_imp = 0
            scale = min(scale * 1.04, 0.04)
        else:
            h[i], h[j] = ni - d, nj + d
            no_imp += 1
            if no_imp > 300:
                scale *= 0.75
                no_imp = 0
                if scale < 1e-7:
                    break
    return h, c5

def _optimize_h(h, t_stop, max_alphas=12):
    n = len(h)
    c5 = _eval(h)
    alphas = [30, 60, 120, 250, 500, 1000, 2000, 5000, 10000, 20000, 40000, 80000]
    for ai in range(min(max_alphas, len(alphas))):
        if time.time() > t_stop:
            break
        alpha = alphas[ai]
        topk = min(2 * n - 1, max(3, int(2 + 1.5 * ai)))
        h_try = _slsqp(h, alpha, topk, 1200, 1e-18)
        c5_try = _eval(h_try)
        if c5_try < c5:
            c5 = c5_try
            h = h_try
    return h, c5

def _optimize_at_n(h_init, n_init, n_target, t_stop, rng):
    h = _interp(h_init, n_init, n_target)
    h, c5 = _optimize_h(h, t_stop, max_alphas=10)
    if time.time() < t_stop - 20:
        h, c5 = _refine(h, rng, 30000, t_stop - 10)
    return h, c5
