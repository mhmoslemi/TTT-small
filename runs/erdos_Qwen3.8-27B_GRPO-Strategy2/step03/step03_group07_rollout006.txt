def compute_weighted_corr_grad(h, n, w):

    w: weights for each correlation (length 2n-1, sums to 1)
    Returns: gradient of sum_k w[k]*corr[k] w.r.t. h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):

    return float(np.max(compute_corr(h)) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    for _ in range(10000):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        # Water-filling projection
        if s > target:
            # Need to reduce sum by (s - target)
            d = s - target
            order = np.argsort(h)[::-1]  # largest first
            for idx in order:
                if d < 1e-14:
                    break
                a = min(d, h[idx])
                h[idx] -= a
                d -= a
        else:

            order = np.argsort(h)  # smallest first
