    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += 1.0 - h[i1[m1]]
    i2 = p + kk
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

def _sweep(h, n, corr, tlim, nc=20, nb=30):
    nimp = 0
    order = np.random.permutation(n)
    for ai in range(n):
        i = order[ai]
        for j in order[ai+1:]:

            hi, hj = h[i], h[j]
            dlo = max(-hi, hj-1.0)

            gd = _gvec(h, i, n) - _gvec(h, j, n)
            ds = np.linspace(dlo, dhi, nc)
            vals = np.max(corr[None,:] + ds[:,None]*gd[None,:], axis=1)
            bi = np.argmin(vals)
            if vals[bi] >= np.max(corr) - 1e-14: continue
            lo = max(dlo, ds[max(0,bi-1)])
            hi_b = min(dhi, ds[min(nc-1,bi+1)])
            for _ in range(nb):
                mid = 0.5*(lo+hi_b)
                if np.max(corr+mid*gd) <= np.max(corr+hi_b*gd): hi_b = mid
                else: lo = mid
            db = 0.5*(lo+hi_b)
            if abs(db) < 1e-14: continue
            h[i] += db; h[j] -= db
            corr += db*gd
            nimp += 1
    return h, corr, nimp

                     bounds=bnds, constraints=cons,
                     options={'maxiter':mi,'ftol':1e-15,'maxls':40})

    except: return h

import numpy as np
from scipy.optimize import minimize

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5val(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _smax(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha
