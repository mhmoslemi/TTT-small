def _pair_optimal_t(h, i, j, corr, n):
    """Find optimal t for h[i]+=t, h[j]-=t to minimize max corr."""
    # Linear coefficients for each corr element
    # corr_k -> a_k * t + c_k
    # a_k = (1-h[i+k_idx]) - h[i_k_idx] - (1-h[j+k_idx]) + h[j_k_idx]
    # where indices are boundary-checked
    
    n_full = 2*n - 1
    a = np.zeros(n_full)
    
    for m in range(n_full):
        lag = m - (n - 1)
        # Contribution from h[i] changing by +t:
        # h[i] appears in term m=i as h[m](1-h[m+lag]) if valid: coeff = (1-h[i+lag]) if 0<=i+lag<n
        # h[i] appears in term m=i-lag as h[m](1-h[m+lag]) if valid: coeff = -h[i-lag] if 0<=i-lag<n
        ai = 0.0
        if 0 <= i + lag < n:
            ai += (1 - h[i+lag])
        if 0 <= i - lag < n:
            ai -= h[i-lag]
        
        # Contribution from h[j] changing by -t:
        aj = 0.0
        if 0 <= j + lag < n:
            aj += (1 - h[j+lag])
        if 0 <= j - lag < n:
            aj -= h[j-lag]
        
        a[m] = ai - aj  # coefficient of t
    
    # Now find t minimizing max_m (corr[m] + a[m]*t) subject to bounds on t
    # Bounds: h[i]+t in [0,1] -> t in [-h[i], 1-h[i]]
    #         h[j]-t in [0,1] -> t in [h[j]-1, h[j]]
    t_lo = max(-h[i], h[j]-1)
    t_hi = min(1-h[i], h[j])
    
    if t_lo >= t_hi:
        return 0.0
    
    # Find t that minimizes max of linear functions on [t_lo, t_hi]
    # The minimum of the upper envelope of lines occurs at either:
    # - t=0 (current point), or
    # - An intersection of two lines where the upper envelope has a local minimum
    
    # Simple approach: check t=0, t_lo, t_hi, and intersections
    best_t = 0.0
    best_val = np.max(corr)
    
    # Check boundaries
    for t_check in [t_lo, t_hi, 0.0]:
        val = np.max(corr + a * t_check)
        if val < best_val:
            best_val = val
            best_t = t_check
    
    # Check intersections of pairs of lines (but this is O(n²) which is 139² ≈ 19321)
    # For n=70, this is fine
    # Line m: y = corr[m] + a[m]*t
    # Intersection of lines p and q: t = (corr[q]-corr[p]) / (a[p]-a[q]) if a[p]!=a[q]
    
    # Actually, for efficiency, let's just sample the upper envelope
    # The min of max of lines is at a vertex of the upper envelope
    # We can find this by: for each pair, compute intersection, check if it's on upper envelope
    
    # Simpler: just do a ternary search or check a few key points
    # Actually for n_full=139 lines, checking all pairs is ~19K which is fine
    
    best_t = 0.0
    best_val = np.max(corr)
    
    # Collect all candidate t values
    candidates = set()
    candidates.add(0.0)
    candidates.add(t_lo)
    candidates.add(t_hi)
    
    # Add intersections within [t_lo, t_hi]
    # To limit computation, only consider lines with |a| > threshold
    active = np.where(np.abs(a) > 1e-12)[0]
    for pi in range(len(active)):

            if abs(a[p] - a[q]) < 1e-12:
                continue
            t_int = (corr[q] - corr[p]) / (a[p] - a[q])
            if t_lo <= t_int <= t_hi:

        if val < best_val - 1e-15:
            best_val = val
            best_t = t_c
    
    return best_t

def _pair_coeff(h, i, j, n):
    """Compute coefficient a[m] such that corr_new[m] = corr[m] + a[m]*t for h[i]+=t, h[j]-=t"""
    n_full = 2 * n - 1
    a = np.zeros(n_full)
    for m in range(n_full):

            ci -= h[i - lag]

            cj += (1.0 - h[j + lag])
        if 0 <= j - lag < n:
            cj -= h[j - lag]
        a[m] = ci - cj
    return a

import numpy as np
import time
from scipy.optimize import minimize

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _pair_coeff(h, i, j, n):
    lags = np.arange(-(n-1), n)
    n_full = 2 * n - 1
    ci = np.zeros(n_full)
    im = i + lags
    mask = (im >= 0) & (im < n)
    ci[mask] += (1.0 - h[im[mask]])
    im2 = i - lags
    mask2 = (im2 >= 0) & (im2 < n)
    ci[mask2] -= h[im2[mask2]]
    cj = np.zeros(n_full)
    jm = j + lags
    mask = (jm >= 0) & (jm < n)
    cj[mask] += (1.0 - h[jm[mask]])
    jm2 = j - lags
    mask2 = (jm2 >= 0) & (jm2 < n)
    cj[mask2] -= h[jm2[mask2]]
    return ci - cj

def _min_max_line(corr, a, t_lo, t_hi, n_iter=25):
    for _ in range(n_iter):
        t1 = t_lo + (t_hi - t_lo) / 3.0
        t2 = t_hi - (t_hi - t_lo) / 3.0
        if np.max(corr + a * t1) < np.max(corr + a * t2):
            t_hi = t2
        else:
            t_lo = t1
    return (t_lo + t_hi) / 2.0

def _pair_descent(h, n, dx, rng, deadline, n_sweeps=500):
    h = h.copy()
    corr = _corr(h)
    best_c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    
    for sweep in range(n_sweeps):
        if time.time() >= deadline:
            break
        order = rng.permutation(n)
        improved = 0
        for ii in range(n):
            i = int(order[ii])
            j = rng.randint(n)
            if i == j:
                continue
            a = _pair_coeff(h, i, j, n)
            t_lo = max(-h[i], h[j] - 1.0)
            t_hi = min(1.0 - h[i], h[j])
            if t_hi - t_lo < 1e-12:
                continue
            t_opt = _min_max_line(corr, a, t_lo, t_hi)
            if abs(t_opt) < 1e-14:
                continue
            h[i] += t_opt
            h[j] -= t_opt
            corr += a * t_opt
            c5 = float(np.max(corr) * dx)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h.copy()
                improved += 1
        if improved == 0 and sweep > 20:
            break
    
    # Final repair
    h = _repair(best_h, n)
    c5 = _c5(h, dx)
    if c5 < best_c5:
        best_c5 = c5
    return best_h, best_c5

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    h = h * (target / max(s, 1e-15))
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
    for idx in order:
        if diff > 0:
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
        if abs(diff) < 1e-12:
            break
    return h
