from scipy.optimize import minimize, minimize_scalar
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h))) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _grad_matrix(h, n):
    m = 2 * n - 1
    G = np.zeros((m, n))
    for k in range(m):
        lag = k - (n - 1)
        i_min = max(0, -lag)
        i_max = min(n, n - lag)
        for i in range(i_min, i_max):
            jp = i + lag
            G[k, jp] += (1.0 - h[i])
            G[k, i] -= h[jp]
    return G

def _pair_descend(h, n, max_passes=50, tol=1e-12):
    """Cyclic pair coordinate descent with exact 1D line search"""
    dx = 2.0 / n
    best = _c5(h, n)
    for _pass in range(max_passes):
        improved = False
        G = _grad_matrix(h, n)  # G[k, i] = dc_k/dh[i]
        c = _corr(h) * dx
        for i in range(n):
            for j in range(i + 1, n):
                if time.time() > ...:  # time check
                    return h, best
                # f(d) = max_k (c_k + d * (G[k,i] - G[k,j])) * dx
                A = G[:, i] - G[:, j]
                d_min = max(-h[i], h[j] - 1.0)
                d_max = min(1.0 - h[i], h[j])
                if d_min >= d_max:
                    continue
                # Minimize max_k (c_k + d*A_k) over d ∈ [d_min, d_max]
                # This is: dx * min_d max_k (c_k + d*A_k)
                # Scale: minimize max_k (c_k + d*A_k) (factor dx is constant)
                
                def fobj(d):
                    return np.max(c + d * A)
                
                res = minimize_scalar(fobj, bounds=(d_min, d_max), method='bounded',
                                     options={'xatol': 1e-10})
                d_opt = res.x
                if abs(d_opt) < 1e-14:
                    continue
                h[i] += d_opt
                h[j] -= d_opt
                h[i] = np.clip(h[i], 0.0, 1.0)
                h[j] = np.clip(h[j], 0.0, 1.0)
                new_c5 = _c5(h, n)
                if new_c5 < best - 1e-15:
                    best = new_c5
                    improved = True
                else:
                    h[i] -= d_opt
                    h[j] += d_opt
        if not improved:
            break
    return h, best

    c1 = np.max(c)
    idx1 = np.argmax(c)
    
    # Candidates
    candidates = [d_min, d_max, 0.0]
    
    # Where top line = 0
    if abs(A[idx1]) > 1e-15:
        d_zero = -c[idx1] / A[idx1]
        if d_min <= d_zero <= d_max:
            candidates.append(d_zero)
    
    # Where top line meets second
    c_sorted = np.sort(c)[::-1]
    if c_sorted[1] < c_sorted[0] and abs(A[idx1]) > 1e-15:
        # Find the second highest
        mask = np.ones(len(c), dtype=bool)
        mask[idx1] = False
        idx2 = np.argmax(np.where(mask, c, -np.inf))
        if abs(A[idx1] - A[idx2]) > 1e-15:
            d_cross = (c[idx2] - c[idx1]) / (A[idx1] - A[idx2])
            if d_min <= d_cross <= d_max:
                candidates.append(d_cross)
    
    # Evaluate at candidates
    best_d = 0
    best_val = np.max(c)
    for d in candidates:
        d = np.clip(d, d_min, d_max)
        val = np.max(c + d * A)
        if val < best_val:
            best_val = val
            best_d = d

def _grad_matrix(h, n):
    m = 2 * n - 1
    G = np.zeros((m, n))
    # For each lag d from -(n-1) to n-1:
    for d in range(-(n-1), n):
        k = d + n - 1
        # Term 1: (1-h[p-d]) for valid p
        if d >= 0:
            p1 = np.arange(d, n)  # p from d to n-1
            G[k, p1] += (1.0 - h[p1 - d])
            p2 = np.arange(0, n - d)  # p from 0 to n-1-d
            G[k, p2] -= h[p2 + d]
        elif d < 0:
            e = -d
            p1 = np.arange(0, n - e)  # p from 0 to n-1-e
            G[k, p1] += (1.0 - h[p1 + e])
            p2 = np.arange(e, n)  # p from e to n-1
            G[k, p2] -= h[p2 - e]
        else:
            p = np.arange(n)
            G[k, p] += (1.0 - h[p])
            G[k, p] -= h[p]
    return G

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h))) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _grad_matrix(h, n):

            p2 = np.arange(e, n)
