import numpy as np
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(_corr(h))) * dx

def _project(h, target_sum):
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - hp.sum()
    if abs(diff) > 1e-12:
        idx = np.argsort(1.0 - hp) if diff > 0 else np.argsort(hp)
        for i in idx:
            if diff > 0:
                t = min(diff, 1.0 - hp[i])
                hp[i] += t
                diff -= t
            else:
                t = min(-diff, hp[i])
                hp[i] -= t
                diff += t
            if abs(diff) < 1e-12:
                break
    return hp

def _sgrad(h, n, alpha):
    dx = 2.0 / n
    c = _corr(h) * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        i0 = max(0, -lag)
        i1 = min(n, n - lag)
        for i in range(i0, i1):
            j = i + lag
            g[j] += w[k] * (1.0 - h[i])
            g[i] -= w[k] * h[j]
    return g * dx

def _pgd(h0, n, alphas, lr, iters_per_alpha, momentum, rng, t_end):
    h = h0.copy()
    target = n / 2.0
    best_c5 = _c5(h, n)
    best_h = h.copy()
    for alpha in alphas:
        v = np.zeros(n)
        for it in range(iters_per_alpha):
            if time.time() > t_end:
                break
            g = _sgrad(h, n, alpha)
            v = momentum * v - lr * g
            h_new = _project(h + v, target)
            c5_new = _c5(h_new, n)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            if c5_new <= best_c5 + 0.5e-6:
                h = h_new
            else:
                v *= 0.5
                h = _project(h + v, target)
    return best_h, best_c5

def _fine_search(h, n, budget_end, rng):
    c5_cur = _c5(h, n)
    steps = [0.001, 0.003, 0.01, 0.03, -0.001, -0.003, -0.01, -0.03]
    for _ in range(200):
        if time.time() > budget_end:
            break
        improved = False
        for i in range(n):
            if time.time() > budget_end:
                break
            for j in range(i + 1, n):
                if time.time() > budget_end:
                    break
                for d in steps:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c = _c5(h, n)
                        if c < c5_cur - 1e-15:
                            c5_cur = c
                            improved = True
                        else:
                            h[i], h[j] = ni - d, nj + d  # undo
        if not improved:
            break
    return h, c5_cur

def _anneal(h, n, T, t_end, rng):
    c5_cur = _c5(h, n)
    best_c5 = c5_cur
    best_h = h.copy()

import numpy as np
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h))) * (2.0 / n)

def _project(h, target_sum):
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - hp.sum()
    if abs(diff) > 1e-12:
        idx = np.argsort(1.0 - hp) if diff > 0 else np.argsort(hp)
        for i in idx:
            t = min(abs(diff), abs(1.0 - hp[i] if diff > 0 else hp[i]))
            hp[i] += t if diff > 0 else -t
            diff -= abs(t) if diff > 0 else -abs(t)
            if abs(diff) < 1e-12:
                break
    return hp

def _sgrad(h, n, alpha):
    dx = 2.0 / n
    c = _corr(h) * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        i0 = max(0, -lag)
        i1 = min(n, n - lag)
        for i in range(i0, i1):
            j = i + lag
            g[j] += wk * (1.0 - h[i])
            g[i] -= wk * h[j]
    return g * dx

def _topk_subgrad(h, n, k):
    dx = 2.0 / n
    c = _corr(h) * dx
    idx = np.argsort(c)[::-1][:k]
    g = np.zeros(n)
    for mk in idx:
        lag = mk - (n - 1)
        i0 = max(0, -lag)
        i1 = min(n, n - lag)
        for i in range(i0, i1):
            j = i + lag
            g[j] += (1.0 - h[i])
            g[i] -= h[j]
    return g * dx / k

def _pgd(h0, n, alpha_schedule, lr0, decay, iters, mom, rng, t_end):
    h = h0.copy()
    target = n / 2.0
    best_c5 = _c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    N = len(alpha_schedule) * iters
    for it in range(N):
        if time.time() > t_end:
            break
        alpha = alpha_schedule[min(it // iters, len(alpha_schedule) - 1)]
        phase = it % iters
        if phase % 3 == 0 and alpha < 5000:
            g = _topk_subgrad(h, n, max(3, int(alpha / 500)))
        else:
            g = _sgrad(h, n, alpha)
        lr = lr0 * (decay ** it)
        v = mom * v - lr * g
        h_new = _project(h + v, target)
        c5_new = _c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new <= _c5(h, n) + 1e-14:
            h = h_new
        else:
            v *= 0.3
            h = _project(h + v, target)
    return best_h, best_c5

def _fine_search(h, n, t_end, rng):
    c5_cur = _c5(h, n)
    steps = np.array([0.002, 0.005, 0.012, 0.03, -0.002, -0.005, -0.012, -0.03])
    for _pass in range(300):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            hi = h[i]
            for j in range(i + 1, n):
                if time.time() > t_end:
                    break
                hj = h[j]
                for d in steps:
                    ni, nj = hi + d, hj - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c = _c5(h, n)
                        if c < c5_cur - 1e-15:
                            c5_cur = c
                            hi, hj = ni, nj
                            improved = True
                        else:
                            h[i], h[j] = hi, hj
                h[i] = hi
            h[j] = hj
        if not improved:
            break
    return h, c5_cur
