def _optimal_delta(h, i, j, top_lags, n, current_c):
    """Find optimal d for changing h[i]+=d, h[j]-=d to minimize max of top correlations."""
    # Coefficients A_p for each top lag
    A = np.zeros(len(top_lags))
    for idx, m in enumerate(top_lags):
        p = m - (n - 1)
        a1 = (1.0 - h[i + p]) if 0 <= i + p < n else 0.0
        a2 = -h[i - p] if 0 <= i - p < n else 0.0
        a3 = -(1.0 - h[j + p]) if 0 <= j + p < n else 0.0
        a4 = h[j - p] if 0 <= j - p < n else 0.0
        A[idx] = a1 + a2 + a3 + a4
    
    # Current top correlations
    c_top = np.array([current_c[m] for m in top_lags])
    
    # Find d to minimize max(c_top + d*A)
    # Also need 0 <= h[i]+d <= 1 and 0 <= h[j]-d <= 1
    d_min = max(-h[i], h[j] - 1.0)  # h[i]+d >= 0, h[j]-d <= 1
    d_max = min(1.0 - h[i], h[j])     # h[i]+d <= 1, h[j]-d >= 0
    
    if d_min >= d_max:
        return 0.0
    
    # Minimize max(c_top + d*A) over d in [d_min, d_max]
    # The function is max of linear functions of d → convex in d
    # Use ternary search
    lo, hi = d_min, d_max
    for _ in range(50):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = np.max(c_top + m1 * A)
        f2 = np.max(c_top + m2 * A)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2

import time

    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):

    s = h.sum()

                a = min(-diff, h[k])
                h[k] -= a
                diff += a

def _smoothmax(c, alpha):
    m = c.max()
    return m + np.log(np.exp(np.clip(alpha * (c - m), -500, 500)).sum()) / alpha


def _slsqp(x0, n, target, alpha, iters):

    res = minimize(lambda x: _smoothmax(_corr(x), alpha), x0, method='SLSQP',
                   bounds=bnds,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target}],
                   options={'maxiter': iters, 'ftol': 1e-18})
    return res.x


def _optimal_d(h, i, j, c_full, n, K=8):
    top_m = np.argsort(c_full)[-K:]
    A = np.zeros(K)
    c_top = np.zeros(K)
    for idx, m in enumerate(top_m):
        p = m - (n - 1)
        a = 0.0
        if 0 <= i + p < n:
            a += (1.0 - h[i + p])
        if 0 <= i - p < n:
            a -= h[i - p]
        if 0 <= j + p < n:
            a -= (1.0 - h[j + p])
        if 0 <= j - p < n:
            a += h[j - p]
        A[idx] = a
        c_top[idx] = c_full[m]
    d_lo = max(-h[i], h[j] - 1.0)
    d_hi = min(1.0 - h[i], h[j])
    if d_lo >= d_hi - 1e-15:
        return 0.0
    lo, hi = d_lo, d_hi
    best_f = np.max(c_top)
    for _ in range(50):
        m1 = lo + (hi - lo) * 0.333
        m2 = hi - (hi - lo) * 0.333
        f1 = np.max(c_top + m1 * A)
        f2 = np.max(c_top + m2 * A)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
        best_f = min(best_f, f1, f2)
    d = (lo + hi) * 0.5
    if np.max(c_top + d * A) >= best_f - 1e-12:
        return 0.0
    return d


def _try_pair(h, i, j, c_full, n, dx):
    d = _optimal_d(h, i, j, c_full, n)
    if abs(d) < 1e-12:
        return h, _c5(h, dx), False
    h[i] += d
    h[j] -= d
    h = np.clip(h, 0.0, 1.0)
    c5_new = _c5(h, dx)
    return h, c5_new, True
