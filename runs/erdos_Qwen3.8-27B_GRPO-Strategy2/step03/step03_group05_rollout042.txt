def _softobj(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = np.minimum(alpha * (c - m), 50.0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _softgrad(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = np.minimum(alpha * (c - m), 50.0)
    ew = np.exp(s - s.max())
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])

        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]

def _repair(h, n):

    if s < 1e-12:

    h = np.clip(h, 0.0, 1.0)

    if abs(diff) > 1e-13:

        if diff > 0:

                diff -= a

            g += 1.0 - 2.0 * h
            H += -2.0 * np.eye(n)
        elif L > 0:
            g[:n-L] += 1.0 - h[L:]
            g[L:] -= h[:n-L]
            for i in range(n - L):

                H[i+L, i] -= 1.0
        else:

            g[:n-M] += 1.0 - h[M:]
            g[M:] -= h[:n-M]
            for i in range(n - M):
                H[i, i+M] -= 1.0
                H[i+M, i] -= 1.0

    # Minimize f = c_active subject to sum(h) = target
    # KKT: H*delta + g + lambda*1 = 0, 1^T*delta = 0
    # [H  1; 1^T  0] [delta; lambda] = [-g; 0]
    A = np.zeros((n+1, n+1))
    A[:n, :n] = H
    A[:n, n] = 1.0
    A[n, :n] = 1.0
    b = -np.concatenate([g, [0.0]])
    try:
        sol = np.linalg.solve(A, b)
    except:
        sol = np.linalg.lstsq(A, b, rcond=None)[0]
    delta = sol[:n]
    
    # Line search
    best_c5 = _c5(h, dx)
    for alpha in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:

        c5_new = _c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            h = h_new
            break
    return h, best_c5
