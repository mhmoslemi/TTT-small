def _repair(h, n):
    # Clip and adjust sum to n/2
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    # Proportional scaling with iterative projection
    for _ in range(50):
        h *= target / h.sum()
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target) < 1e-10:
            break
    # Final adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            room = min(abs(diff), 1.0 - h[idx]) if diff > 0 else min(abs(diff), h[idx])
            if room <= 0:
                continue
            h[idx] += room if diff > 0 else -room
            diff += -room if diff > 0 else room
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):

    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

    m = c.max()

        k = I - (n - 1)
        if k >= 0:
            g[k:] += wi * (1. - h[:n - k])
            g[:n - k] -= wi * h[k:]
        else:
            s2 = -k
            g[:n - s2] += wi * (1. - h[s2:])
            g[s2:] -= wi * h[:n - s2]
    return g

def _slq(h, n, a, mi):
    bd = [(0., 1.)] * n
    cn = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2., 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(_sm, h, jac=_smg, args=(a,), method='SLSQP',
                     bounds=bd, constraints=cn,
                     options={'maxiter': mi, 'ftol': 1e-19})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _proj(y, n):
    c = n / 2.
    lo, hi = y.min() - 2., y.max() + 2.
    for _ in range(90):
        md = (lo + hi) / 2.
        if np.clip(y - md, 0., 1.).sum() > c:
            lo = md
        else:
            hi = md
    return np.clip(y - (lo + hi) / 2., 0., 1.)

def _grad_desc(h, n, a, niter, lr0):
    h = _repair(h.copy(), n)
    lr = lr0
    fcur = _sm(h, a)
    for _ in range(niter):
        g = _smg(h, a)
        ls = lr
        for _ in range(25):
            h2 = _proj(h - ls * g, n)
            f2 = _sm(h2, a)
            if f2 < fcur - 1e-8:
                h, fcur = h2, f2
                lr = min(ls * 1.1, 0.1)
                break
            ls *= 0.5
        else:
            break
    return _repair(h, n)

def _nm_block(h, n, rng, blk, mi):
    idx = np.random.RandomState(rng.randint(10**9)).choice(n, size=min(blk, n), replace=False)
    xi = h[idx].copy()
    base = h.copy()
    def fsub(z):
        h2 = base.copy(); h2[idx] = np.clip(z, 0., 1.)
        return _eval(_repair(h2, n))
    try:
        r = minimize(fsub, xi, method='Nelder-Mead',
                     options={'maxiter': mi, 'xatol': 1e-12, 'fatol': 1e-15, 'adaptive': True})
        h2 = base.copy(); h2[idx] = np.clip(r.x, 0., 1.)
        return _repair(h2, n)
    except Exception:
        return h

def _resample(ho, no, nn):
    xo = (np.arange(no) + 0.5) * 2.0 / no
    xn = (np.arange(nn) + 0.5) * 2.0 / nn
    return _repair(np.interp(xn, xo, ho), nn)

def _struct_start(n, rng):
    m = rng.randint(2, 6)
    vals = np.sort(rng.uniform(0.1, 0.9, m))
    h = np.zeros(n)
    for i in range(n):
        h[i] = np.interp(i, np.linspace(0, n - 1, m), vals)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el(): return time.time() - t0

    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    best_n = n
    # Phase 1: SLSQP warm-up with increasing alpha
    for a in [300, 3000, 30000, 300000, 3000000]:
        if _el() > budget_s - 300: break
        h = _slq(best_h, best_n, a, 2500)
        c5 = _eval(h)
        if c5 < best_c5: best_c5, best_h = c5, h.copy()

    # Phase 2: Projected gradient refinement
    if _el() < budget_s - 220:
        for a in [5000, 50000, 500000]:
            if _el() > budget_s - 220: break
            h = _grad_desc(best_h, best_n, a, 400, 0.02)
            c5 = _eval(h)
            if c5 < best_c5: best_c5, best_h = c5, h.copy()

    # Phase 3: Block Nelder-Mead
    if _el() < budget_s - 160:
        h = best_h.copy(); c5c = _eval(h)
        for _ in range(min(80, n)):
            if _el() > budget_s - 160: break
            h = _nm_block(h, best_n, rng, max(8, best_n // 5), 800)
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c5: best_c5, best_h = c5n, h.copy()

    # Phase 4: Multi-start SLSQP with structured perturbations
    if _el() < budget_s - 120:
        for _r in range(50):
            if _el() > budget_s - 120: break
            rtype = _r % 4
            if rtype == 0:
                sc = 0.003 + 0.02 * rng.rand()
                hp = _repair(best_h + sc * rng.randn(best_n), best_n)
            elif rtype == 1:
                hp = _struct_start(best_n, rng)
            elif rtype == 2:
                idx = rng.choice(best_n, size=best_n // 3, replace=False)
                hp = best_h.copy(); hp[idx] = np.clip(rng.uniform(0.1, 0.9, len(idx)), 0., 1.)
                hp = _repair(hp, best_n)
            else:
                sc = 0.05 + 0.1 * rng.rand()
                hp = _repair(best_h + sc * np.sin(np.linspace(0, 4 * np.pi, best_n) + rng.rand() * 10), best_n)
            h = _slq(hp, best_n, 200000, 1200)
            c5 = _eval(h)
            if c5 < best_c5: best_c5, best_h = c5, h.copy()

    # Phase 5: Try different resolutions
    if _el() < budget_s - 80:
        for nn in [50, 60, 80, 100, 120]:
            if _el() > budget_s - 80: break
            h2 = _resample(best_h, best_n, nn)
            h2 = _slq(h2, nn, 200000, 2000)
            c5 = _eval(h2)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h2.copy(), nn

    # Phase 6: Fine random pairwise moves
    if _el() < budget_s - 30:
        h = best_h.copy(); c5c = _eval(h)
        sc = 0.004; ni = 0
        for _ in range(100000):
            if _el() > budget_s - 30: break
            i, j = rng.randint(best_n), rng.randint(best_n)
            if i == j: continue
            d = sc * rng.randn()
            a2, b2 = h[i] + d, h[j] - d
            if a2 < 0 or a2 > 1 or b2 < 0 or b2 > 1: continue
            h[i], h[j] = a2, b2
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n; ni = 0
                sc = min(sc * 1.12, 0.015)
                if c5n < best_c5: best_c5, best_h = c5n, h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
                ni += 1
                if ni > 400: sc *= 0.65; ni = 0
                if sc < 1e-7: break

    # Phase 7: Ultra-high alpha SLSQP polish
    if _el() < budget_s - 15:
        h = _slq(best_h, best_n, 5000000, 4000)
        c5 = _eval(h)
        if c5 < best_c5: best_c5, best_h = c5, h.copy()

    # Phase 8: Tiny coordinate adjustments
    if _el() < budget_s - 5:
        h = best_h.copy(); c5c = _eval(h)
        for _ in range(50000):
            if _el() > budget_s - 5: break
            i, j = rng.randint(best_n), rng.randint(best_n)
            if i == j: continue
            d = rng.choice([1, -1]) * rng.choice([1, 2, 5]) * 1e-5
            a2, b2 = h[i] + d, h[j] - d
            if 0 < a2 < 1 and 0 < b2 < 1:
                h[i], h[j] = a2, b2
                c5n = _eval(h)
                if c5n < c5c - 1e-17:
                    c5c = c5n
                    if c5n < best_c5: best_c5, best_h = c5n, h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
