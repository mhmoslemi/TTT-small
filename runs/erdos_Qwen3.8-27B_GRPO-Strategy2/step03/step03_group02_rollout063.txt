from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _waterfill(x, target):
    """Project x onto {sum(·)=target, 0≤·≤1}"""
    n = len(x)
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mu = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mu, 0.0, 1.0))
        if s > target:
            lo = mu
        else:
            hi = mu
    x = np.clip(x - 0.5*(lo+hi), 0.0, 1.0)
    # Fix residual
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - x)
        else:
            order = np.argsort(x)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - x[idx])
                x[idx] += a
                diff -= a
            else:
                a = min(-diff, x[idx])
                x[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return x

def _grad_smax(h, alpha):
    n = len(h)
    corr = _corr(h)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    w = np.exp(s) / np.sum(np.exp(s))
    lags = np.arange(-(n-1), n)
    A = np.zeros(n)
    B = np.zeros(n)
    for j in range(n):
        va = (j + lags >= 0) & (j + lags < n)
        A[j] = np.dot(w[va], h[j + lags[va]])
        vb = (j - lags >= 0) & (j - lags < n)
        B[j] = np.dot(w[vb], h[j - lags[vb]])
    grad = 1.0 - A - B
    smax_val = m + np.log(np.sum(np.exp(s))) / alpha
    return grad, smax_val, corr

    # Armijo line search
    best_h = h.copy()
    best_val = smax_val

        if v_new < best_val:

        lr *= 0.5

def _g_vec(h, p, n):
    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

    n_pairs = 0
    n_improved = 0
    cur_max = np.max(corr)
    for ai in range(n):
        i = order[ai]
        for j in order[ai + 1:]:
            n_pairs += 1

            hi, hj = h[i], h[j]

            range_d = d_max - d_min

            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(d_min, d_max, n_coarse)

            bi = np.argmin(vals)
            if vals[bi] >= cur_max - 1e-13:

            lo = max(d_min, ds[max(0, bi - 1)])
            hi_b = min(d_max, ds[min(n_coarse - 1, bi + 1)])

                mid = 0.5 * (lo + hi_b)

            cur_max = max(cur_max, np.max(corr))
            improved = True

import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _waterfill(x, target):
    n = len(x)
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mu = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mu, 0.0, 1.0))
        if s > target:
            lo = mu
        else:
            hi = mu
    x = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - x) if diff > 0 else np.argsort(x)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - x[idx])
            else:
                a = max(diff, -x[idx])
            x[idx] += a
            diff -= a
            if abs(diff) < 1e-14:
                break
    return x

def _grad_smax(h, alpha):
    n = len(h)
    corr = _corr(h)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    w = np.exp(s) / np.sum(np.exp(s))
    lags = np.arange(-(n - 1), n)
    A = np.zeros(n)
    B = np.zeros(n)
    for j in range(n):
        va = (j + lags >= 0) & (j + lags < n)
        A[j] = np.dot(w[va], h[j + lags[va]])
        vb = (j - lags >= 0) & (j - lags < n)
        B[j] = np.dot(w[vb], h[j - lags[vb]])
    grad = 1.0 - A - B
    val = m + np.log(np.sum(np.exp(s))) / alpha
    return grad, val

def _proj_grad(h, alpha, lr, target):

    best_h, best_v = h.copy(), val

        _, vn = _grad_smax(hn, alpha)

        else:
            lr *= 0.5
            if lr < 1e-10:
                break

def _g_vec(h, p, n):
    kk = np.arange(-(n - 1), n)

    i1 = p - kk
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += (1.0 - h[i1[m1]])
    i2 = p + kk
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

def _sweep(h, n, corr, tlim, nc=20, nb=25, skip_thresh=0.0):
    improved = False
    order = np.random.permutation(n)
    cur_max = float(np.max(corr))
    ni = 0
    for ai in range(n):
        i = order[ai]
        for j in order[ai + 1:]:
            if time.time() > tlim:
                break
            hi, hj = h[i], h[j]
            dmin = max(-hi, hj - 1.0)
            dmax = min(1.0 - hi, hj)
            if dmax - dmin < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(dmin, dmax, nc)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = np.argmin(vals)
            if vals[bi] >= cur_max - skip_thresh:

            hibi = min(dmax, ds[min(nc - 1, bi + 1)])
            for _ in range(nb):
                mid = 0.5 * (lo + hibi)
                if np.max(corr + mid * gd) <= np.max(corr + hibi * gd):
                    hibi = mid
                else:
                    lo = mid
            d = 0.5 * (lo + hibi)
            h[i], h[j] = hi + d, hj - d
            corr = corr + d * gd
            nm = float(np.max(corr))
            if nm > cur_max:
                cur_max = nm
            elif nm < cur_max:
                cur_max = nm
            improved = True
            ni += 1
    return h, corr, improved, ni

    lo, hi = 0.0, 1.0
    for _ in range(80):

        s = np.sum(np.clip(x - mu, 0.0, 1.0))
        if s > target:

        else:
            hi = mu

        order = np.argsort(1.0 - x) if diff > 0 else np.argsort(x)
        for idx in order:
            a = min(diff, 1.0 - x[idx]) if diff > 0 else max(diff, -x[idx])
            x[idx] += a
            diff -= a
            if abs(diff) < 1e-14:

    m = float(np.max(corr))

    w = np.exp(s) / np.sum(np.exp(s))

    B = np.zeros(n)
    for j in range(n):

        A[j] = np.dot(w[va], h[j + lags[va]])
        vb = (j - lags >= 0) & (j - lags < n)
        B[j] = np.dot(w[vb], h[j - lags[vb]])
    return 1.0 - A - B, m + float(np.log(np.sum(np.exp(s)))) / alpha

    i2 = p + kk
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

                return h, corr, improved, ni

            dmin = max(-hi, hj - 1.0)

            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(dmin, dmax, nc)

            bi = int(np.argmin(vals))

            corr = corr + d * gd

    order = np.random.permutation(n)
    cur_max = float(np.max(corr))
    ni = 0
    for ai in range(n):
        i = order[ai]
        for j in order[ai + 1:]:
            if time.time() > tlim:
                return h, corr, improved, ni
            hi, hj = h[i], h[j]
            dmin = max(-hi, hj - 1.0)
            dmax = min(1.0 - hi, hj)
            if dmax - dmin < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(dmin, dmax, nc)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = int(np.argmin(vals))
            if vals[bi] >= cur_max - 1e-14:
                continue
            lo = max(dmin, ds[max(0, bi - 1)])
            hibi = min(dmax, ds[min(nc - 1, bi + 1)])
            for _ in range(nb):
                mid = 0.5 * (lo + hibi)
                if np.max(corr + mid * gd) <= np.max(corr + hibi * gd):
                    hibi = mid
                else:
                    lo = mid
            d = 0.5 * (lo + hibi)
            h[i], h[j] = hi + d, hj - d
            corr = corr + d * gd
            cur_max = float(np.max(corr))
            improved = True
            ni += 1
    return h, corr, improved, ni

def _smax_obj(x, alpha):
    c = _corr(x)
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _slsqp(h, alpha, target, bounds, maxit):
    try:
        r = minimize(_smax_obj, h, args=(alpha,), method='SLSQP', bounds=bounds,
                     constraints=[{'type': 'eq', 'fun': _eq_con, 'args': (target,)}],
                     options={'maxiter': maxit, 'ftol': 1e-14, 'maxls': 40})
        return _waterfill(r.x, target)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h = _waterfill(np.array(initial_h_values, dtype=float).copy(), target)
    best_h, best_c5 = h.copy(), _c5(h, dx)

    # Phase 1: Projected gradient with ramping alpha
    for alpha in [50, 150, 400, 1000]:
        if time.time() - t0 > budget_s * 0.08:
            break
        for _ in range(30):
            if time.time() - t0 > budget_s * 0.08:
                break
            h = _proj_grad(h, alpha, 0.03, target, 12)
        c = _c5(h, dx)
        if c < best_c5 - 1e-15:
            best_c5, best_h = c, h.copy()

    # Phase 2: Coordinate sweeps with periodic SLSQP and perturbation
    h, corr = best_h.copy(), _corr(best_h)
    stall, it = 0, 0
    while time.time() - t0 < budget_s - 200:
        it += 1
        rem = budget_s - 200 - (time.time() - t0)
        if rem < 10:
            break
        ch = min(55.0, rem)
        h, corr, imp, _ = _sweep(h, n, corr, time.time() + ch, 20, 25)
        if imp:
            c5 = float(np.max(corr) * dx)
            if c5 < best_c5 - 1e-14:
                best_c5, best_h = c5, h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if it % 5 == 0 and time.time() - t0 < budget_s - 220:
            hs = _slsqp(best_h.copy(), 200 + 40 * it, target, bounds, 180)
            cs = _c5(hs, dx)
            if cs < best_c5 - 1e-14:
                best_c5, best_h = cs, hs.copy()
                h, corr = hs.copy(), _corr(hs)
                stall = 0
        if stall >= 3 and time.time() - t0 < budget_s - 240:
            scale = min(0.07, 0.012 + 0.01 * stall)
            hp = _waterfill(best_h + scale * np.random.randn(n), target)
            hp, hpc, _, _ = _sweep(hp, n, _corr(hp), time.time() + 20, 18, 20)
            cp = float(np.max(hpc) * dx)
            if cp < best_c5 - 1e-14:
                best_c5, best_h = cp, hp.copy()
                h, corr = hp.copy(), _corr(hp)
            stall = 0

    # Phase 3: SLSQP polish with high alpha
    if time.time() - t0 < budget_s - 50:
        for alpha in [1000, 2500, 5000]:
            if time.time() - t0 > budget_s - 50:
                break
            hs = _slsqp(best_h.copy(), alpha, target, bounds, 300)
            cs = _c5(hs, dx)
            if cs < best_c5 - 1e-14:
                best_c5, best_h = cs, hs.copy()

    # Phase 4: Final fine sweeps
    h, corr = best_h.copy(), _corr(best_h)
    while time.time() - t0 < budget_s - 5:
        ch = min(18.0, budget_s - 5 - (time.time() - t0))
        if ch < 3:
            break
        h, corr, imp, _ = _sweep(h, n, corr, time.time() + ch, 25, 30)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5 - 1e-15:
            best_c5, best_h = c5, h.copy()
        if not imp:
            break

    best_h = _waterfill(best_h, target)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
