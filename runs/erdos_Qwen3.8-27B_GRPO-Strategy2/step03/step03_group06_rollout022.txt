    s = np.clip(alpha * (c - m), -50, 50)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (x.sum() - target) ** 2

def _lbfgs(h, n, dx, alpha, target, maxiter):
    b = [(0.0, 1.0)] * n

                        method='L-BFGS-B', bounds=b,

            hc = _repair(r.x, n); cc = _c5(hc, dx)
            if cc < bc: bh, bc = hc, cc
        except: pass

        return hc, _c5(hc, dx)
    except:
        return h, _c5(h, dx)

def _coord_descent(h, n, dx, deadline):
    h = h.copy()

    bc = np.max(corr) * dx
    for _ in range(500):

            h[i] -= 1.0

                m2 = hi - (hi-lo)/3
                if np.max(corr + m1*g) < np.max(corr + m2*g): hi = m2
                else: lo = m1
            d = (lo+hi)/2
            if abs(d) > 1e-10:
                h[i] += d; corr += d*g
                c = np.max(corr)*dx
                if c < bc: bc = c; imp = True

def _pair_descent(h, n, dx, deadline):

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-12:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h

def _obj(x, alpha, target, pen):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -50, 50)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (x.sum() - target) ** 2

def _lbfgs(h, n, dx, alpha, target, maxiter):
    b = [(0.0, 1.0)] * n
    bh, bc = h.copy(), _c5(h, dx)
    for p in [10, 100, 1000, 10000]:
        try:
            r = minimize(lambda x, pp=p: _obj(x, alpha, target, pp), bh,
                        method='L-BFGS-B', bounds=b,
                        options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
            hc = _repair(r.x, n); cc = _c5(hc, dx)
            if cc < bc:
                bh, bc = hc, cc
        except Exception:
            pass
    return bh, bc

def _slsqp(h, n, dx, alpha, target, maxiter):
    b = [(0.0, 1.0)] * n
    try:
        r = minimize(lambda x: _obj(x, alpha, target, 5000), h,
                    method='SLSQP', bounds=b,
                    constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target}],
                    options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(r.x, n)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _compute_grads(h, corr, n):
    gs = np.zeros((n, 2 * n - 1))
    for i in range(n):
        h[i] += 1.0
        gs[i] = _corr(h) - corr
        h[i] -= 1.0
    return gs

def _ternary_max(corr, g, lo, hi, iters=60):
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr + m1 * g) < np.max(corr + m2 * g):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def _coord_descent(h, n, dx, deadline, passes=300):
    h = h.copy()
    corr = _corr(h)
    bc = np.max(corr) * dx
    for _ in range(passes):
        if time.time() >= deadline:
            break
        imp = False
        gs = _compute_grads(h, corr, n)
        for i in range(n):
            g = gs[i]
            lo, hi = -h[i], 1.0 - h[i]
            if hi - lo < 1e-12:
                continue
            d = _ternary_max(corr, g, lo, hi)
            if abs(d) > 1e-10:
                h[i] += d
                corr = corr + d * g
                c = np.max(corr) * dx
                if c < bc:
                    bc = c
                    imp = True
        if not imp:
            break
    _corr_check = _corr(h)
    return h, max(bc, np.max(_corr_check) * dx)

def _pair_descent(h, n, dx, deadline, passes=100):
    h = h.copy()
    corr = _corr(h)
    bc = np.max(corr) * dx
    for _ in range(passes):
        if time.time() >= deadline:
            break
        imp = False
        gs = _compute_grads(h, corr, n)
        for i in range(n):
            for j in range(i + 1, n):
                g = gs[i] - gs[j]
                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0 - h[i], h[j])
                if hi - lo < 1e-12:
                    continue
                d = _ternary_max(corr, g, lo, hi)
                if abs(d) > 1e-10:
                    h[i] += d
                    h[j] -= d
                    corr = corr + d * g
                    c = np.max(corr) * dx
                    if c < bc:
                        bc = c
                        imp = True
        if not imp:
            break
    _corr_check = _corr(h)
    return h, max(bc, np.max(_corr_check) * dx)

def _rand_perturb(h, n, rng, amp):
    hp = h + rng.uniform(-amp, amp, n)
    return _repair(hp, n)

def _anti_periodic(n):
    h = np.zeros(n)
    half = n // 2
    for i in range(half):
        h[i] = 1.0
        if i + half < n:
            h[i + half] = 0.0
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0

    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 0: Try structured starts, pick best
    if rem() > 30:

        t = np.linspace(0, 2, n)

        if ct < best_c5:

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
