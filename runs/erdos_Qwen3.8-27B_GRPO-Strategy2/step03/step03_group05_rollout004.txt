import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _softobj(x, alpha, lam, n):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    penalty = lam * (np.sum(x) - n / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam, n):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _active_grad(h, top_idx, n):
    g = np.zeros(n)
    for ki in top_idx:
        L = ki - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    return g

def _lbfbs(h, n, alpha, lam, maxiter, bounds):
    def obj(x):
        return _softobj(x, alpha, lam, n)
    def jac(x):
        return _softgrad(x, alpha, lam, n)
    try:
        res = minimize(obj, h.copy(), jac=jac, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-17})
        return _repair(res.x, n)
    except Exception:
        return h.copy()

def _active_descent(h, n, dx, top_k, n_iter, steps):
    h = h.copy()
    best_c5 = _c5(h, dx)
    for _ in range(n_iter):
        c = _corr(h)
        top = np.argsort(c)[::-1][:top_k]
        g = _active_grad(h, top, n)
        g -= g.mean()
        improved = False
        for step in steps:
            h_new = _repair(h - step * g, n)
            c5_new = _c5(h_new, dx)
            if c5_new < best_c5 - 1e-15:
                h = h_new
                best_c5 = c5_new
                improved = True
                break
        if not improved:
            break
    return h, best_c5

def _pair_cd(h, n, dx, n_sweeps, dlist, max_pairs=15):
    h = h.copy()
    c5_cur = _c5(h, dx)
    for _ in range(n_sweeps):
        improved = False
        c = _corr(h)
        top = np.argsort(c)[::-1][:5]
        g = _active_grad(h, top, n)
        order = np.argsort(np.abs(g))[::-1][:min(30, n)]
        for ii in range(len(order)):
            i = order[ii]
            for jj in range(ii + 1, min(ii + max_pairs, len(order))):
                j = order[jj]
                best_d = 0.0
                best_c = c5_cur
                for d in dlist:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c2 = _c5(h, dx)
                        if c2 < best_c:
                            best_c = c2
                            best_d = d
                        h[i], h[j] = ni - d, nj + d
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    c5_cur = best_c
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, c5_cur

def _sa_polish(h, n, dx, rng, T0, n_iter, cool):
    h = h.copy()
    c5_cur = _c5(h, dx)
    best_c5 = c5_cur
    T = T0
    for _ in range(n_iter):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.05, 0.05)
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        h = _repair(h, n)
        c_new = _c5(h, dx)
        if c_new < c5_cur or (T > 1e-10 and rng.rand() < np.exp(-(c_new - c5_cur) / T)):
            c5_cur = c_new
            if c_new < best_c5:
                best_c5 = c_new
        else:
            h[i] -= d
            h[j] += d
            h = _repair(h, n)
        T *= cool
    return h, c5_cur, best_c5

def _structured_init(n, seed):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    best_h = None
    best_c5 = float('inf')
    cands = []
    x = np.arange(n)
    for p in [1, 2, 3, 5, 7]:
        cands.append(0.5 + 0.3 * np.sin(2 * np.pi * p * x / n))
    for nb in [2, 3, 4, 5, 7]:
        h = np.zeros(n)
        block = n // nb
        for b in range(nb):
            if b % 2 == 0:
                h[b*block:(b+1)*block] = 1.0
        cands.append(h)
    cands.append(rng.rand(n))
    cands.append((rng.rand(n) > 0.5).astype(float))
    for c in cands:
        hc = _repair(c, n)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc
    return best_h, best_c5
