def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx), corr

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def project_h(h, n):
    target = n / 2.0
    # Bisection on shift lambda
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

        order = np.argsort(1.0 - hp) if diff > 0 else np.argsort(hp)

            if diff > 0:
                a = min(diff, 1.0 - hp[idx])
                hp[idx] += a
                diff -= a
            else:

                diff += a

        if j2lo <= j2hi:

**New approach for fine-tuning**: Instead of random pairs, use the gradient to determine WHICH pairs to adjust. The gradient tells us the steepest descent direction. We can decompose this into individual coordinate moves.

Specifically:
1. Compute gradient g
2. The best single-coordinate move is to increase h[i] where g[i] is most negative (or decrease where g[i] is most positive), while adjusting another coordinate to maintain the sum constraint
3. The best pair (i,j) to move is the one where g[i] - g[j] is most negative (move mass from j to i)

This is essentially "greedy gradient" or "coordinate descent along the gradient direction."

def directed_cd(h, n, t_end, rng):

    best_c5, _ = compute_c5(h, n)
    
    for it in range(100000):
        if time.time() > t_end:
            break
        corr = compute_corr(h)
        c5 = float(np.max(corr) * 2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
        
        g = grad_active(h, n, corr, 0.1)
        # Best pair: maximize g[j] - g[i] (move mass from i to j)
        # But we also need to respect bounds
        # Try top few candidates
        sorted_g = np.argsort(g)  # ascending: most negative first
        # We want to increase h where g is negative, decrease where g is positive
        # So move from high-g to low-g
        candidates = []
        for ii in sorted_g[:5]:  # most negative g → want to increase
            for jj in sorted_g[-5:]:  # most positive g → want to decrease
                if ii != jj:
                    candidates.append((ii, jj))
        
        improved = False
        for (i, j) in candidates:
            for step in [0.01, 0.005, 0.002, 0.001, 0.0005]:
                hi, hj = h[i], h[j]
                ni, nj = hi + step, hj - step
                if ni > 1 or nj < 0:
                    continue
                h[i], h[j] = ni, nj
                c5_try = float(np.max(compute_corr(h)) * 2.0 / n)
                if c5_try < best_c5 - 1e-16:
                    best_c5 = c5_try
                    improved = True
                    break
                h[i], h[j] = hi, hj
            if improved:
                break

            # Try random perturbation
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:

                        h[i], h[j] = h[i] - delta, h[j] + delta

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    corr = compute_corr(h)
    return float(np.max(corr) * 2.0 / n), corr

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def project_h(h, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - hp) if diff > 0 else np.argsort(hp)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - hp[idx])
                hp[idx] += a
                diff -= a
            else:
                a = min(-diff, hp[idx])
                hp[idx] -= a
                diff += a
    return hp

def grad_active(h, n, corr, frac):
    m = corr.max()
    span = m - corr.min() + 1e-15
    thresh = m - frac * span
    active = np.where(corr >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    g = np.zeros(n)
    half = n - 1
    for p in active:
        lag = p - half
        j1lo = max(0, -lag)
        j1hi = min(n - 1, n - 1 - lag)
        if j1lo <= j1hi:
            idx = np.arange(j1lo, j1hi + 1)
            g[idx] += 1.0 - h[idx + lag]
        j2lo = max(0, lag)
        j2hi = min(n - 1, lag + n - 1)
        if j2lo <= j2hi:
            idx = np.arange(j2lo, j2hi + 1)
            g[idx] -= h[idx - lag]
    g /= len(active)
    return g

def subgrad_step(h_init, n, t_end, rng, lr0, mu, frac):
    h = h_init.copy()
    best_h = h.copy()
    best_c5, _ = compute_c5(h, n)
    v = np.zeros(n)
    lr = lr0
    pat = 0
    for it in range(800000):
        if time.time() > t_end:
            break
        corr = compute_corr(h)
        c5 = float(np.max(corr) * 2.0 / n)
        if c5 < best_c5 - 1e-16:
            best_c5 = c5
            best_h = h.copy()
            pat = 0
        else:
            pat += 1
        if pat > 15000:
            h = project_h(h + 0.015 * rng.randn(n), n)
            v *= 0.3
            lr = min(lr * 1.3, 0.4)
            pat = 0
            continue
        g = grad_active(h, n, corr, frac)
        v = mu * v - lr * g
        h = project_h(h + v, n)
        if it % 3000 == 0:
            lr *= 0.98
    return best_h, best_c5

def directed_cd(h_init, n, t_end, rng):
    h = h_init.copy()
    best_c5, _ = compute_c5(h, n)
    step_sizes = [0.02, 0.01, 0.005, 0.0025, 0.001, 0.0005, 0.0002]
    stall = 0
    for it in range(200000):
        if time.time() > t_end:
            break
        corr = compute_corr(h)
        c5 = float(np.max(corr) * 2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            stall = 0
        else:
            stall += 1
        g = grad_active(h, n, corr, 0.12)
        sg = np.argsort(g)
        improved = False
        top = min(6, n // 2)
        for ii in sg[:top]:
            for jj in sg[-top:]:
                if ii == jj:
                    continue
                for step in step_sizes:
                    hi, hj = h[ii], h[jj]
                    ni, nj = hi + step, hj - step
                    if ni > 1.0 or nj < 0.0:
                        continue
                    h[ii], h[jj] = ni, nj
                    c5t = float(np.max(compute_corr(h)) * 2.0 / n)
                    if c5t < best_c5 - 1e-16:
                        best_c5 = c5t
                        improved = True
                        break
                    h[ii], h[jj] = hi, hj
                if improved:
                    break
            if improved:
                break
        if not improved:
            if stall > 500:
                for _ in range(100):
                    if time.time() > t_end:
                        break
                    i = rng.randint(n)
                    j = rng.randint(n)
                    if i == j:
                        continue
                    d = rng.uniform(0.02, 0.08) * (1 if rng.random() < 0.5 else -1)
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c5t = float(np.max(compute_corr(h)) * 2.0 / n)
                        if c5t < best_c5 - 1e-16:
                            best_c5 = c5t
                            stall = 0
                            break
                        h[i], h[j] = h[i] - d, h[j] + d
    return h, best_c5

def sim_anneal(h_init, n, t_end, rng):
    h = h_init.copy()
    best_h = h.copy()
    best_c5, _ = compute_c5(h, n)
    T = 0.004
    T_min = 0.000005
    cool = 0.99985
    for it in range(400000):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-0.06, 0.06)
        ni, nj = h[i] + delta, h[j] - delta
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = float(np.max(compute_corr(h)) * 2.0 / n)
        dc = c5n - best_c5
        if dc < 0 or (T > T_min and rng.random() < np.exp(-dc / T)):
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h.copy()
        else:
            h[i], h[j] = h[i] - delta, h[j] + delta
        T = max(T * cool, T_min)
    return best_h, best_c5

def slsqp_refine(h_init, n, t_end, alpha, topk, maxiter):
    target = n / 2.0
    def obj(h):
        corr = compute_corr(h)
        m = np.max(corr)
        s = alpha * (corr - m)
        s = np.clip(s, -400, 400)
        val = m + np.log(np.sum(np.exp(s))) / alpha
        if topk > 1:
            sc = np.sort(corr)[::-1][:topk]
            val += 0.01 * np.sum((sc - sc.mean())**2)
        return val
    def con_fn(h):
        return h.sum() - target
    def con_jac(h):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fn, 'jac': con_jac}]
    try:
        res = minimize(obj, h_init.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(res.x, n)
    except Exception:
        return h_init
