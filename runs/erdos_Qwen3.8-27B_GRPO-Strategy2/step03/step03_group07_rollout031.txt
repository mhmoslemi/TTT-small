def compute_grad_weighted(h, n, w):
    """w has shape (2n-1,), returns gradient of sum_s w[s]*corr_s w.r.t. h"""
    grad = np.zeros(n)
    n_full = 2*n - 1
    for s in range(n_full):
        if w[s] < 1e-12:
            continue
        shift = s - (n-1)
        # First term: (1-h[j-shift]) for valid j
        # j-shift in [0, n-1] => j in [shift, shift+n-1] intersect [0, n-1]
        j_start = max(0, shift)
        j_end = min(n-1, shift + n - 1)
        idx = np.arange(j_start, j_end+1)
        src = idx - shift
        grad[j_start:j_end+1] += w[s] * (1 - h[src])
        
        # Second term: -h[j+shift] for valid j
        # j+shift in [0, n-1] => j in [-shift, n-1-shift] intersect [0, n-1]
        j_start2 = max(0, -shift)
        j_end2 = min(n-1, n-1-shift)
        idx2 = np.arange(j_start2, j_end2+1)
        src2 = idx2 + shift
        grad[j_start2:j_end2+1] -= w[s] * h[src2]
    
    return grad

def compute_c5(h, n):

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        if abs(s2 - target) < 1e-14:
            break
        d = target - s2
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx]); h[idx] += a
            else:
                a = min(-d, h[idx]); h[idx] -= a
            d -= np.sign(d) * a
    return h

def proj_sum(h, n):

        if jlo <= jhi:

            grad[jlo:jhi + 1] += ws * (1.0 - h[src])

        jhi2 = min(n - 1, n - 1 - shift)
        if jlo2 <= jhi2:
            src2 = np.arange(jlo2, jhi2 + 1) + shift

    return grad

    h_fin, c5_fin = coord_descent(best_h, best_n, 40, t_end, rng)
    if c5_fin < best_c5 - 1e-14:
        best_h = h_fin.copy()
        best_c5 = c5_fin
    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
