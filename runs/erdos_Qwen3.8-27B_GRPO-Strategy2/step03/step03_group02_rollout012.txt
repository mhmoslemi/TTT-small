    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _sweep_pair(h, n, corr, time_limit, n_coarse=35, n_bisect=45):
    improved = False
    n_improved = 0
    order = np.random.permutation(n)
    for ai in range(n - 1):
        i = order[ai]
        for aj in range(ai + 1, n):
            j = order[aj]
            if time.time() > time_limit:
                return h, corr, improved, n_improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = int(np.argmin(vals))
            if vals[bi] >= np.max(corr) - 1e-13:
                continue
            lo = ds[max(0, bi - 1)]
            hi_b = ds[min(n_coarse - 1, bi + 1)]
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            improved = True
            n_improved += 1
    return h, corr, improved, n_improved

def _sweep_single(h, n, corr, time_limit, n_coarse=50, n_bisect=50):
    improved = False
    order = np.random.permutation(n)
    for i in order:
        if time.time() > time_limit:
            return h, corr, improved
        hi = h[i]
        gd = _g_vec(h, i, n)
        ds = np.linspace(0, 1, n_coarse)
        vals = np.max(corr[None, :] + (ds[:, None] - hi) * gd[None, :], axis=1)
        bi = int(np.argmin(vals))
        if vals[bi] >= np.max(corr) - 1e-13:
            continue
        lo = ds[max(0, bi - 1)]
        hi_b = ds[min(n_coarse - 1, bi + 1)]
        for _ in range(n_bisect):
            mid = 0.5 * (lo + hi_b)
            if np.max(corr + (mid - hi) * gd) <= np.max(corr + (hi_b - hi) * gd):
                hi_b = mid
            else:
                lo = mid
        new_val = 0.5 * (lo + hi_b)
        h[i] = new_val
        corr = corr + (new_val - hi) * gd
        if abs(new_val - hi) > 1e-7:
            improved = True
    return h, corr, improved

def _focused_sweep(h, n, corr, time_limit, top_k=5, n_bisect=45):
    """Focus on reducing top-k correlations specifically."""
    improved = False
    cur_corr = corr.copy()
    n_corr = len(cur_corr)
    for it in range(200):
        if time.time() > time_limit:
            break
        top_idx = np.argsort(cur_corr)[-top_k:]
        for ki in top_idx:
            # Find which coordinates most affect this correlation
            # corr[ki] depends on h pairs at specific offsets
            offset = ki - (n - 1)
            if offset >= 0:
                pairs = [(i, i + offset) for i in range(n - offset)]
            else:
                pairs = [(i - offset, i) for i in range(-offset, n)]
            if not pairs:
                continue
            # Perturb the pair with max h[i]*(1-h[j])
            best_i, best_j = pairs[0]
            best_val = h[best_i] * (1 - h[best_j])
            for (pi, pj) in pairs:
                v = h[pi] * (1 - h[pj])
                if v > best_val:
                    best_val = v
                    best_i, best_j = pi, pj
            # Try swapping: increase h at the 0-sided, decrease at 1-sided
            if h[best_i] > 0.5:
                gd = _g_vec(h, best_i, n)
                delta = -0.5 * h[best_i]
            else:
                gd = _g_vec(h, best_i, n)
                delta = 0.5 * (1.0 - h[best_i])
            new_h_i = h[best_i] + delta
            new_h_i = np.clip(new_h_i, 0, 1)
            actual_delta = new_h_i - h[best_i]
            trial_corr = cur_corr + actual_delta * gd
            # Maintain sum constraint by adjusting a "donor" coordinate
            if abs(actual_delta) > 1e-12:
                # Find best donor to maintain sum
                donor_candidates = np.array([p for p in range(n) if p != best_i])
                if len(donor_candidates) > 0:
                    best_donor = donor_candidates[0]
                    best_trial = np.max(trial_corr)
                    for dc in donor_candidates[:20]:
                        gd_d = _g_vec(h, dc, n)
                        d_delta = -actual_delta
                        d_new = np.clip(h[dc] + d_delta, 0, 1)
                        d_actual = d_new - h[dc]
                        tc = cur_corr + actual_delta * gd + d_actual * gd_d
                        if d_actual * actual_delta < 0 and abs(d_actual) > 1e-12:
                            if np.max(tc) < best_trial:
                                best_trial = np.max(tc)
                                best_donor = dc
                    if best_donor != donor_candidates[0]:
                        gd_d = _g_vec(h, best_donor, n)
                        d_delta = -actual_delta
                        d_new = np.clip(h[best_donor] + d_delta, 0, 1)
                        d_actual = d_new - h[best_donor]
                        h[best_i] = new_h_i
                        h[best_donor] = d_new
                        cur_corr = cur_corr + actual_delta * gd + d_actual * gd_d
                        if np.max(cur_corr) < np.max(corr) - 1e-14:
                            improved = True
                    else:
                        h[best_i] = new_h_i
                        cur_corr = trial_corr
                else:
                    h[best_i] = new_h_i
                    cur_corr = trial_corr
    return h, cur_corr, improved

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)
    best_corr = _corr(best_h)

    # Phase 1: SLSQP warm-up
    for alpha in [50, 150, 400, 1000]:
        if time.time() - t0 > budget_s * 0.12:
            break
        cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
        try:
            res = minimize(_obj_smax, best_h, args=(alpha,), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 350, 'ftol': 1e-14, 'maxls': 50})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = hc.copy()
                best_corr = _corr(best_h)
        except Exception:
            pass

    # Phase 2: Intensive pair sweeps with binary push
    stall = 0
    round_count = 0
    while time.time() - t0 < budget_s - 80:
        round_count += 1
        h = best_h.copy()
        corr = best_corr.copy()
        remaining = budget_s - 80 - (time.time() - t0)
        chunk = min(45.0, remaining)
        tl = time.time() + chunk

        # Pair sweep with high precision
        h, corr, imp, n_imp = _sweep_pair(h, n, corr, tl, 35, 45)
        if imp:
            corr = _corr(h)
            c5 = float(np.max(corr) * dx)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h.copy()
                best_corr = corr.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        # Every 5 rounds, try binary push
        if round_count % 5 == 0 and time.time() - t0 < budget_s - 100:
            h_bp = _binary_push(best_h, n, 0.12)
            c5_bp = _c5(h_bp, dx)
            if c5_bp < best_c5 - 1e-6:
                bc = _corr(h_bp)
                h_bp, bc, _, _ = _sweep_pair(h_bp, n, bc, time.time() + 15, 30, 40)
                c5_bp2 = _c5(h_bp, dx)
                if c5_bp2 < best_c5 - 1e-15:
                    best_c5 = c5_bp2
                    best_h = h_bp.copy()
                    best_corr = bc.copy()
                    stall = 0

        # Periodic SLSQP to escape local minima
        if stall >= 3 and time.time() - t0 < budget_s - 100:
            alpha = 300 + 100 * stall
            cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
            try:
                res = minimize(_obj_smax, best_h, args=(alpha,), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 250, 'ftol': 1e-14, 'maxls': 40})
                hc = _repair(res.x, n)
                hc_corr = _corr(hc)
                hc, hc_corr, _, _ = _sweep_pair(hc, n, hc_corr, time.time() + 18, 30, 40)
                c5_t = _c5(hc, dx)
                if c5_t < best_c5 - 1e-15:
                    best_c5 = c5_t
                    best_h = hc.copy()
                    best_corr = hc_corr.copy()
                    stall = 0
            except Exception:
                stall = 0
            stall = max(0, stall - 1)

        # Strong perturbation when very stalled
        if stall >= 8 and time.time() - t0 < budget_s - 100:
            scale = min(0.04, 0.01 + 0.005 * (stall - 4))
            h_p = best_h + scale * np.random.randn(n)
            h_p = _repair(h_p, n)
            hp_corr = _corr(h_p)
            h_p, hp_corr, _, _ = _sweep_pair(h_p, n, hp_corr, time.time() + 25, 30, 40)
            c5_p = _c5(h_p, dx)
            if c5_p < best_c5 - 1e-15:
                best_c5 = c5_p
                best_h = h_p.copy()
                best_corr = hp_corr.copy()
            stall = 0

    # Phase 3: Final precision single-coordinate sweeps
    if time.time() - t0 < budget_s - 25:
        for _ in range(8):
            if time.time() - t0 > budget_s - 12:
                break
            h = best_h.copy()
            corr = best_corr.copy()
            h, corr, imp = _sweep_single(h, n, corr, time.time() + 18, 50, 55)
            if imp:
                c5 = _c5(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    best_corr = _corr(h)

    # Phase 4: Final SLSQP with very high alpha
    if time.time() - t0 < budget_s - 5:
        cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
        try:
            res = minimize(_obj_smax, best_h, args=(5000,), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 500, 'ftol': 1e-15, 'maxls': 50})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()
        except Exception:
            pass

    # Phase 5: One more precision pass
    if time.time() - t0 < budget_s - 2:
        h = best_h.copy()
        corr = _corr(h)
        h, corr, _ = _sweep_pair(h, n, corr, time.time() + (budget_s - 2 - (time.time() - t0)), 40, 50)
        c5 = _c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
