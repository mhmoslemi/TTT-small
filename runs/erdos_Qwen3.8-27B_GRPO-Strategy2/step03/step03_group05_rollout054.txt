import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _grad_topk(h, k):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[::-1][:k]
    g = np.zeros(n)
    for ki in idx:
        L = ki - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    return g

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _lp_target(g, n):
    target = n / 2.0
    h_lp = np.zeros(n)
    idx_sorted = np.argsort(g)
    remaining = target
    for i in idx_sorted:
        if remaining >= 1.0 - 1e-12:
            h_lp[i] = 1.0
            remaining -= 1.0
        elif remaining > 1e-12:
            h_lp[i] = remaining
            remaining = 0.0
            break
    return h_lp

def _softobj(x, alpha, lam, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + lam * (x.sum() - n / 2.0) ** 2

def _softgrad(x, alpha, lam, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for ki in range(2 * n - 1):
        L = ki - (n - 1)
        pk = pw[ki]
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= x[:n - M]
    g /= alpha
    g += 2.0 * lam * (x.sum() - n / 2.0)
    return g

def _optimize(h, n, dx, rng, time_limit, t0):
    def elapsed():
        return time.time() - t0

    best_h = h.copy()
    best_c5 = _c5(h, dx)
    h_work = h.copy()
    c5_work = best_c5

    # Phase 1: LP-guided top-k with momentum
    config = [
        (3, 500, 0.06),
        (5, 400, 0.05),
        (3, 200, 0.03),
        (8, 350, 0.04),
        (5, 250, 0.02),
        (12, 250, 0.03),
        (8, 200, 0.015),
        (15, 200, 0.02),
        (3, 200, 0.01),
    ]
    momentum = np.zeros(n)
    mom_coeff = 0.3

    for k_top, n_iter, base_step in config:
        if elapsed() > t0 + time_limit - 80:
            break
        stuck = 0
        for it in range(n_iter):
            if elapsed() > t0 + time_limit - 80:
                break
            g = _grad_topk(h_work, k_top)
            momentum = mom_coeff * momentum + (1 - mom_coeff) * g
            g_c = momentum - momentum.mean()
            h_lp = _lp_target(g, n)
            direction_lp = h_lp - h_work

            improved = False
            for alpha in [base_step, base_step * 0.5, base_step * 0.25, base_step * 0.1, base_step * 0.04]:
                h_new = _repair(h_work + alpha * direction_lp, n)
                c_new = _c5(h_new, dx)
                if c_new < c5_work - 1e-15:
                    h_work = h_new
                    c5_work = c_new
                    improved = True
                    if c_new < best_c5:
                        best_c5 = c_new
                        best_h = h_work.copy()
                    break

            if not improved:
                for alpha in [base_step * 0.02, base_step * 0.008, base_step * 0.003, base_step * 0.001]:
                    h_new = _repair(h_work - alpha * g_c, n)
                    c_new = _c5(h_new, dx)
                    if c_new < c5_work - 1e-15:
                        h_work = h_new
                        c5_work = c_new
                        improved = True
                        if c_new < best_c5:
                            best_c5 = c_new
                            best_h = h_work.copy()
                        break

            if not improved:
                stuck += 1
                if stuck > 40:
                    break
            else:
                stuck = 0

    # Phase 2: L-BFGS-B soft-max polish
    if elapsed() < t0 + time_limit - 60:
        for alpha_sm in [5000, 30000, 100000]:
            if elapsed() > t0 + time_limit - 40:
                break
            lam = 40.0
            try:
                res = minimize(
                    lambda x: _softobj(x, alpha_sm, lam, n),
                    best_h.copy(),
                    jac=lambda x: _softgrad(x, alpha_sm, lam, n),
                    method='L-BFGS-B', bounds=[(0, 1)] * n,
                    options={'maxiter': 300, 'ftol': 1e-18})
                hc = _repair(res.x, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc
            except Exception:
                pass

    # Phase 3: Very fine top-k gradient
    if elapsed() < t0 + time_limit - 35:
        h_w = best_h.copy()
        c5_w = best_c5
        for k_top in [3, 5, 8, 5]:
            if elapsed() > t0 + time_limit - 35:
                break
            for _ in range(150):
                if elapsed() > t0 + time_limit - 35:
                    break
                g = _grad_topk(h_w, k_top)
                g_c = g - g.mean()
                imp = False
                for alpha in [0.003, 0.001, 0.0004, 0.0001]:
                    h_new = _repair(h_w - alpha * g_c, n)
                    c_new = _c5(h_new, dx)
                    if c_new < c5_w - 1e-15:
                        h_w = h_new
                        c5_w = c_new
                        imp = True
                        if c_new < best_c5:
                            best_c5 = c_new
                            best_h = h_w.copy()
                        break
                if not imp:
                    break

    # Phase 4: Coordinate descent
    if elapsed() < t0 + time_limit - 20:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        for _sweep in range(15):
            if elapsed() > t0 + time_limit - 20:
                break
            g = _grad_topk(h, 8)
            order = np.argsort(np.abs(g))[::-1][:12]
            imp = False
            for ii in range(len(order)):
                if imp:
                    break
                i = order[ii]
                for jj in range(ii + 1, min(ii + 8, len(order))):
                    if imp:
                        break
                    j = order[jj]
                    for d in [0.003, 0.007, 0.015, 0.03, -0.003, -0.007, -0.015, -0.03]:
                        ni, nj = h[i] + d, h[j] - d
                        if 0 <= ni <= 1 and 0 <= nj <= 1:
                            h[i], h[j] = ni, nj
                            c = _c5(h, dx)
                            if c < c5_cur - 1e-15:
                                c5_cur = c
                                imp = True
                                if c < best_c5:
                                    best_c5 = c
                                    best_h = h.copy()
                                break
                            else:
                                h[i], h[j] = ni - d, nj + d
            h = _repair(h, n)

    # Phase 5: Tiny random search
    if elapsed() < t0 + time_limit - 2:
        h = best_h.copy()
        c5_cur = best_c5
        for _ in range(4000):
            if elapsed() > t0 + time_limit - 2:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.choice([0.001, 0.002, 0.005, 0.01, -0.001, -0.002, -0.005, -0.01])
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5_cur:
                    c5_cur = c
                    best_c5 = c
                    best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    return best_h, best_c5
