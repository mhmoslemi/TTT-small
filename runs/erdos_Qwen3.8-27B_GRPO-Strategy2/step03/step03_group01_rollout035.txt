import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _project(h, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0.0, 1.0).sum() < target:
            lo = mid
        else:
            hi = mid
    hc = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hc.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = hc < 1.0 - 1e-12
        else:
            mask = hc > 1e-12
        cnt = mask.sum()
        if cnt > 0:
            hc[mask] += diff / cnt
            hc = np.clip(hc, 0.0, 1.0)
    return hc

def _coord_descent(h, n, max_passes=40):
    m = 2 * n - 1
    zlag = n - 1
    for _p in range(max_passes):
        improved = False
        for i in range(n):
            t0 = h[i]
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = t0
            a = c1 - c0
            b = c0.copy()
            idx = np.arange(m) != zlag
            a_sub = a[idx]
            b_sub = b[idx]
            da = a_sub[:, None] - a_sub[None, :]
            db = b_sub[None, :] - b_sub[:, None]
            mask = np.abs(da) > 1e-14
            with np.errstate(divide='ignore', invalid='ignore'):
                tb = np.where(mask, db / np.where(mask, da, 1.0), 999.0)
            valid = (tb >= 0.0) & (tb <= 1.0)
            cands = np.concatenate(([0.0, 1.0, 0.5, t0], tb[valid]))
            disc = (a_sub - 1.0)**2 - 4.0*(b_sub - b[zlag])
            vd = disc >= 0
            tr1 = (-(a_sub[vd] - 1.0) + np.sqrt(disc[vd])) * 0.5
            tr2 = (-(a_sub[vd] - 1.0) - np.sqrt(disc[vd])) * 0.5
            tr1 = tr1[(tr1 >= 0) & (tr1 <= 1)]
            tr2 = tr2[(tr2 >= 0) & (tr2 <= 1)]
            cands = np.concatenate((cands, tr1, tr2))
            cands = np.unique(np.clip(cands, 0.0, 1.0))
            aff = a_sub[:, None] * cands[None, :] + b_sub[:, None]
            conc = b[zlag] + cands - cands**2
            obj = np.maximum(aff.max(axis=0), conc)
            best_t = cands[np.argmin(obj)]
            if abs(best_t - t0) > 1e-12:
                h[i] = best_t
                improved = True
        if not improved:
            break
    return h

def _subgrad(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    mx = c.max()
    active = np.where(c >= mx - 1e-10)[0]
    grad = np.zeros(n)
    for k_idx in active:

        elif lag > 0:

            grad[lag:] -= h[:n - lag]
        else:
            L = -lag
            grad[L:] += (1.0 - h[:n - L])
            grad[:n - L] -= h[L:]
    grad /= len(active)
    return grad

def _subgrad_opt(h, n, steps=500, alpha0=0.05):
    h = h.copy()
    for s in range(steps):
        g = _subgrad(h, n)
        alpha = alpha0 / (1.0 + 0.01 * s)
        h_new = _project(h - alpha * g, n)
        if _eval(h_new) < _eval(h):
            h = h_new
    return h

def _sf(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sj(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-14:
            continue
        lag = k_idx - (n - 1)
        wg = w[k_idx]
        if lag == 0:
            g += wg * (1.0 - 2.0 * x)
        elif lag > 0:
            g[:n - lag] += wg * (1.0 - x[lag:])
            g[lag:] -= wg * x[:n - lag]
        else:
            L = -lag
            g[L:] += wg * (1.0 - x[:n - L])
            g[:n - L] -= wg * x[L:]
    return g

def _slsqp_opt(h, n, alpha, iters=2000):
    bnds = [(0.0, 1.0)] * n
    cons_sum = n / 2.0
    def _con_fun(x):
        return np.sum(x) - cons_sum
    def _con_jac(x):
        return np.ones(n)
    try:
        res = minimize(_sf, h, jac=_sj, args=(alpha,), method='SLSQP',
                       bounds=bnds, constraints=[{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}],
                       options={'maxiter': iters, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _con_fun(x):
    return np.sum(x) - x.size / 2.0

def _con_jac(x):
    return np.ones(x.size)
