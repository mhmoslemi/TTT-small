    m_new = beta1 * m + (1 - beta1) * grad
    v_new = beta2 * v + (1 - beta2) * grad**2

import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + float(np.log(np.sum(np.exp(s))) / alpha)

def _obj_alpha(h, alpha, n):
    return _smooth_max(_corr(h), alpha)

def _obj_grad(h, alpha, n):
    c = _corr(h)
    m = float(np.max(c))
    e = np.exp(np.clip(alpha * (c - m), -500, 500))
    w = e / e.sum()
    g = np.zeros(n)
    lags = np.arange(-(n-1), n)
    for i in range(n):
        idx1 = i - lags
        idx2 = i + lags
        mask1 = (idx1 >= 0) & (idx1 < n)
        mask2 = (idx2 >= 0) & (idx2 < n)
        term1 = np.where(mask1, 1.0 - h[np.clip(idx1, 0, n-1)], 0.0)
        term2 = np.where(mask2, h[np.clip(idx2, 0, n-1)], 0.0)
        g[i] = np.sum(w * (term1 - term2))
    return _smooth_max(c, alpha), g

def _eq_target(h, target):
    return np.sum(h) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    def obj(x):
        return _smooth_max(_corr(x), alpha)
    def obj_jac(x):
        c, g = _obj_grad(x, alpha, len(x))
        return c, g  # SLSQP wants (value, gradient)
    # SLSQP doesn't accept jac as (f, g) tuple, let me use separate
    def f(x):
        return _smooth_max(_corr(x), alpha)
    def g(x):
        _, g = _obj_grad(x, alpha, len(x))
        return g
    res = minimize(f, h0, jac=g, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target,
                                 'jac': lambda x: np.ones(len(x))}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return res.x

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + float(np.log(np.sum(np.exp(s))) / alpha)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_func(h, alpha, n):
    c = _corr(h)
    m = float(np.max(c))
    e = np.exp(np.clip(alpha * (c - m), -500, 500))
    w = e / e.sum()
    idx = np.arange(n)[:, None] - np.arange(n)[None, :] + (n - 1)
    A = w[idx]
    return A @ (1.0 - h) - A.T @ h

def _slsqp_obj(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _slsqp_grad(x, alpha, n):
    return _grad_func(x, alpha, n)

def _slsqp_con_fun(x, target):
    return np.sum(x) - target

def _slsqp_con_jac(x, target):
    return np.ones(len(x))

def _slsqp_run(x0, bounds, target, alpha, maxiter):
    res = minimize(_slsqp_obj, x0, jac=_slsqp_grad, args=(alpha, len(x0)),
                   method='SLSQP', bounds=bounds,

                                 'jac': _slsqp_con_jac, 'args': (target,)}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return res.x

def _pair_eval(t, h, i, j, s, dx):
    h[i] = t
    h[j] = s - t
    return float(np.max(_corr(h)) * dx)
