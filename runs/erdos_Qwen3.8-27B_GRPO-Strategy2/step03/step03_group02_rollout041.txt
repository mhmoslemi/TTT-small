def _block_slsqp(h, n, target, bounds, alpha, block_size=10, maxiter=200):
    idx = np.random.choice(n, block_size, replace=False)
    idx.sort()
    x0 = h[idx].copy()
    fixed_sum = h.sum() - x0.sum()
    target_block = target - fixed_sum
    
    def obj_block(x):
        h_mod = h.copy()
        h_mod[idx] = x
        return _smooth_max(_corr(h_mod), alpha)
    
    def con_block(x):
        return float(np.sum(x)) - target_block
    
    cons = [{'type': 'eq', 'fun': con_block, 'args': ()}]
    bnds = [(0.0, 1.0)] * block_size
    try:
        res = minimize(obj_block, x0, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = h.copy()
        h_new[idx] = res.x
        return h_new
    except:
        return h

    idx = np.sort(np.random.choice(n, block_size, replace=False))
    x0 = h[idx].copy()
    target_block = target - (h.sum() - x0.sum())
    cons = [{'type': 'eq', 'fun': _block_eq, 'args': (target_block,)}]
    bnds = [(0.0, 1.0)] * block_size
    try:
        res = minimize(_block_obj, x0, args=(h, idx, alpha), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-13, 'maxls': 30})
        if res.success or res.fun < _smooth_max(_corr(h), alpha):
            h_new = h.copy()
            h_new[idx] = np.clip(res.x, 0.0, 1.0)
            return _repair(h_new, n)
    except Exception:
        pass
    return h

def _full_slsqp(h, n, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: SLSQP warm-up with alpha ramping
    for alpha in [40, 120, 350, 900]:
        if time.time() - t0 > budget_s * 0.10:
            break
        hc = _full_slsqp(best_h, n, target, bounds, alpha, 350)
        c5 = _c5(hc, dx)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = hc.copy()

    # Phase 2: Alternating pair and triple sweeps
    stall = 0
    iteration = 0
    while time.time() - t0 < budget_s - 100:
        iteration += 1
        h = best_h.copy()
        corr = _corr(h)
        chunk = min(30.0, budget_s - 100 - (time.time() - t0))
        if iteration % 3 == 0:
            h, corr, imp = _triple_sweep(h, n, corr, time.time() + chunk * 0.5, 100, 16)
            h, corr, imp2 = _pair_sweep(h, n, corr, time.time() + chunk * 0.3, 25, 30)
            imp = imp or imp2
        else:
            h, corr, imp = _pair_sweep(h, n, corr, time.time() + chunk, 27, 38)
        if imp:
            c5 = _c5(h, dx)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        # Block SLSQP every few iterations
        if iteration % 3 == 0 and time.time() - t0 < budget_s - 120:
            for _ in range(2):
                if time.time() - t0 > budget_s - 120:
                    break
                alpha_b = 200 + 80 * iteration
                hc = _block_slsqp(best_h, n, target, alpha_b, block_size=10, maxiter=120)
                hc_corr = _corr(hc)
                hc, hc_corr, _ = _pair_sweep(hc, n, hc_corr, time.time() + 8, 20, 25)
                c5_t = _c5(hc, dx)
                if c5_t < best_c5 - 1e-14:
                    best_c5 = c5_t
                    best_h = hc.copy()
                    stall = 0

        # Perturbation when stalled
        if stall >= 5 and time.time() - t0 < budget_s - 130:
            pert_scale = min(0.06, 0.012 + 0.008 * stall)
            h_pert = best_h + pert_scale * np.random.randn(n)
            h_pert = _repair(h_pert, n)
            hc = _full_slsqp(h_pert, n, target, bounds, 500, 200)
            hc_corr = _corr(hc)
            hc, hc_corr, _ = _triple_sweep(hc, n, hc_corr, time.time() + 12, 60, 14)
            hc, hc_corr, _ = _pair_sweep(hc, n, hc_corr, time.time() + 10, 25, 30)
            c5_t = _c5(hc, dx)
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
                best_h = hc.copy()
            # Also try subgradient-inspired perturbation
            if time.time() - t0 < budget_s - 130:
                corr_b = _corr(best_h)
                ks = np.argsort(corr_b)[-5:]
                grad = np.zeros(n)
                for ki in ks:
                    lag = int(ki - (n - 1))
                    for ii in range(n):
                        j = ii + lag
                        if 0 <= ii < n and 0 <= j < n:
                            grad[ii] += (1.0 - best_h[j])
                            grad[j] -= best_h[ii]
                h_sg = best_h - 0.01 * grad / (np.max(np.abs(grad)) + 1e-12)
                h_sg = _repair(h_sg, n)
                hc2 = _full_slsqp(h_sg, n, target, bounds, 400, 200)
                hc2_corr = _corr(hc2)
                hc2, hc2_corr, _ = _pair_sweep(hc2, n, hc2_corr, time.time() + 10, 25, 30)
                c5_t2 = _c5(hc2, dx)
                if c5_t2 < best_c5 - 1e-14:
                    best_c5 = c5_t2
                    best_h = hc2.copy()
            stall = 0

    # Phase 3: Final high-precision sweeps
    if time.time() - t0 < budget_s - 20:
        for _ in range(4):
            if time.time() - t0 > budget_s - 15:
                break
            h = best_h.copy()
            corr = _corr(h)
            h, corr, _ = _pair_sweep(h, n, corr, time.time() + 15, 32, 42)
            h, corr, _ = _triple_sweep(h, n, corr, time.time() + 10, 80, 20)
            c5 = _c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 4: Final SLSQP polish
    if time.time() - t0 < budget_s - 4:
        try:
            hc = _full_slsqp(best_h, n, target, bounds, 3000, 500)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
