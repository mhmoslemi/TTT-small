            break
        _G['k'] = k
        _G['target'] = n_p / 2.0
        _G['objfn'] = obj_topk
        h_new = fdslsq(h_work.copy(), n_p, bnd, 400)
        c5_new = compute_c5(h_new, n_p)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new.copy()
            best_c5 = c5_new
            h_work = h_new

    # Phase 2: Softmax with escalating sharpness
    for alpha in [500, 2000, 8000, 30000, 100000, 500000]:
        if time.time() - t0 > budget_s * 0.40:
            break
        _G['alpha'] = alpha
        _G['target'] = n_p / 2.0
        _G['objfn'] = obj_softmax
        h_new = fdslsq(h_work.copy(), n_p, bnd, 350)
        c5_new = compute_c5(h_new, n_p)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new.copy()
            best_c5 = c5_new
            h_work = h_new

    # Phase 3: L-BFGS-B penalty refinement
    if time.time() - t0 < budget_s * 0.55:
        for alpha in [10000, 100000]:
            if time.time() - t0 > budget_s * 0.55:
                break
            for pen in [1000, 10000, 100000]:
                if time.time() - t0 > budget_s * 0.55:
                    break
                _G['alpha'] = alpha
                _G['pen'] = pen
                _G['target'] = n_p / 2.0
                try:
                    res = minimize(obj_pen, h_work.copy(), method='L-BFGS-B',
                                  bounds=bnd,
                                  options={'maxiter': 3000, 'ftol': 1e-22,
                                           'gtol': 1e-14})
                    h_new = repair(res.x, n_p)
                    c5_new = compute_c5(h_new, n_p)
                    if c5_new < best_c5 - 1e-14:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        h_work = h_new
                except Exception:
                    pass

    # Phase 4: Coordinate descent (main)
    if time.time() - t0 < budget_s * 0.78:
        t_cd = t0 + budget_s * 0.78
        h_cd, c5_cd = coord_cd(h_work.copy(), n_p, t_cd, rng,
                               n_rounds=1200, n_trials_mult=8)
        if c5_cd < best_c5 - 1e-14:
            best_h = h_cd.copy()
            best_c5 = c5_cd
            h_work = h_cd

    # Phase 5: Perturbation + re-optimize (escape local minima)
    if time.time() - t0 < budget_s * 0.88:
        for perturb_try in range(3):
            if time.time() - t0 > budget_s * 0.88:
                break
            h_pert = h_work.copy()
            noise_scale = 0.03 * (perturb_try + 1)
            h_pert += noise_scale * rng.randn(n_p)
            h_pert = repair(h_pert, n_p)
            _G['alpha'] = 50000
            _G['target'] = n_p / 2.0
            _G['objfn'] = obj_softmax
            h_pert = fdslsq(h_pert, n_p, bnd, 300)
            t_pert_cd = min(t0 + budget_s * 0.88, time.time() + 20)
            h_pert, c5_pert = coord_cd(h_pert, n_p, t_pert_cd, rng,
                                       n_rounds=100, n_trials_mult=4)
            if c5_pert < best_c5 - 1e-14:
                best_h = h_pert.copy()
                best_c5 = c5_pert
                h_work = h_pert

    # Phase 6: Try alternative n values
    if time.time() - t0 < budget_s * 0.94:
        for n_alt in [64, 80, 96]:
            if time.time() - t0 > budget_s * 0.94:
                break
            if n_alt == best_n:
                continue
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_alt)
            h_alt = np.interp(x_new, x_old, best_h)
            h_alt = repair(h_alt, n_alt)
            bnd_alt = [(0.0, 1.0)] * n_alt
            _G['alpha'] = 50000
            _G['target'] = n_alt / 2.0
            _G['objfn'] = obj_softmax
            h_alt = fdslsq(h_alt, n_alt, bnd_alt, 300)
            t_alt_cd = min(t0 + budget_s * 0.94, time.time() + 15)
            h_alt, c5_alt = coord_cd(h_alt, n_alt, t_alt_cd, rng,
                                     n_rounds=80, n_trials_mult=4)
            if c5_alt < best_c5 - 1e-14:
                best_h = h_alt.copy()
                best_c5 = c5_alt
                best_n = n_alt
                h_work = h_alt

    # Phase 7: Final SLSQP polish
    if best_h is not None and time.time() - t0 < budget_s * 0.975:
        bnd_f = [(0.0, 1.0)] * best_n
        _G['alpha'] = 1000000
        _G['target'] = best_n / 2.0
        _G['objfn'] = obj_softmax
        h_fin = fdslsq(best_h.copy(), best_n, bnd_f, 600)
        c5_fin = compute_c5(h_fin, best_n)
        if c5_fin < best_c5 - 1e-14:
            best_h = h_fin
            best_c5 = c5_fin

    # Phase 8: Final fine coordinate descent
    if time.time() - t0 < budget_s * 0.998:
        t_end_final = t0 + budget_s * 0.998
        h_fcd, c5_fcd = coord_cd(best_h.copy(), best_n, t_end_final, rng,
                                 n_rounds=400, n_trials_mult=5,
                                 steps=[0.004, 0.002, 0.001, 0.0005, 0.0002,
                                        0.0001])
        if c5_fcd < best_c5 - 1e-14:
            best_h = h_fcd
            best_c5 = c5_fcd

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
