import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    for _ in range(5000):
        h *= target / max(s, 1e-30)
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def soft_max_obj(h, alpha, topk=1, flat_w=0.0):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    val = m + np.log(np.sum(np.exp(s))) / alpha
    if topk > 1:
        sc = np.sort(corr)[::-1][:topk]
        val += flat_w * np.sum((sc - sc.mean())**2)
    return val

def analytic_grad(h, alpha, topk=1, flat_w=0.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    K = 2 * n - 1
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    e = np.exp(s)
    S = e.sum()
    w = e / S
    val = m + np.log(S) / alpha
    grad = np.zeros(n)
    for idx in range(K):
        d = idx - (n - 1)
        wk = w[idx]
        if wk < 1e-18:
            continue
        jm1 = max(0, d)
        jM1 = min(n - 1, n + d - 1)
        if jm1 <= jM1:
            idx1 = np.arange(jm1, jM1 + 1) - d
            grad[jm1:jM1 + 1] += wk * (1.0 - h[idx1])
        jm2 = max(0, -d)
        jM2 = min(n - 1, n - d - 1)
        if jm2 <= jM2:
            idx2 = np.arange(jm2, jM2 + 1) + d
            grad[jm2:jM2 + 1] -= wk * h[idx2]
    if topk > 1:
        sc = np.sort(corr)[::-1][:topk]
        val += flat_w * np.sum((sc - sc.mean())**2)
    return val, grad

Actually, one more important idea: **the objective landscape near the optimum**. If we're at a local minimum, the coordinate descent can help by making small adjustments to individual h[i] values. But the parent's coordinate descent is quite slow (random pairs, multiple step sizes). 

A faster approach: **cyclic coordinate descent** with **line search**. For each coordinate i, given the current h, find the optimal value of h[i] (keeping all others fixed). This is a 1D optimization problem.

Fixing h[j] for j≠i, the corr_d as a function of h[i] is:
corr_d = (terms not involving h[i]) + h[i] * (coefficient) - (h[i]^2 * 1 if i appears in both h[i+d] and h[i])

Actually, for a fixed shift d:
- h[i] appears in h[i+d] when the "i" index in the formula is i-d: contributes h[i]*(1-h[i-d])
- h[i] appears in h[i] when the "i" index is i: contributes h[i+d]*(1-h[i])
- Combined for shift d: h[i]*(1-h[i-d]) + h[i+d]*(1-h[i]) (if both are valid)
  = h[i] - h[i]*h[i-d] + h[i+d] - h[i]*h[i+d]
  = (h[i] + h[i+d]) - h[i]*(h[i-d] + h[i+d])
  
This is a quadratic in h[i]: -A_i * h[i]^2 + B_i * h[i] + C_i
where A_i depends on whether h[i] is the "self" term (it isn't in cross-correlation... wait)

Hmm, actually in the cross-correlation h[i+d]*(1-h[i]), h[i] appears linearly in (1-h[i]) and h[i+d] appears linearly. So for fixed other variables:
corr_d = [terms not involving h[i]] + (contribution from h[i] as first factor) + (contribution from h[i] as second factor)

h[i] as first factor (h[i+d] where i+d = j, i.e., j=i and d=0, or more generally h[i] = h[(i-d)+d]):
contribution = h[i] * (1-h[i-d]) for valid i-d

h[i] as second factor (the 1-h[i] part):
contribution = (1-h[i]) * h[i+d] = h[i+d] - h[i]*h[i+d] for valid i+d

So total contribution of h[i] to corr_d:
= h[i]*(1-h[i-d]) + h[i+d] - h[i]*h[i+d]  (with validity conditions)
= h[i] + h[i+d] - h[i]*(h[i-d] + h[i+d])  (assuming both valid)

This is linear in h[i] (not quadratic!) since the d=0 case would give h[i]*(1-h[i]) which IS quadratic, but for d≠0 it's linear.

For d=0: h[i]*(1-h[i]) = h[i] - h[i]^2. This IS quadratic in h[i].

So the total corr_d as a function of h[i] (with everything else fixed):
- For d=0: linear term h[i] - h[i]^2 (plus constants from other i's)
- For d≠0: linear in h[i] (plus constants)

The max over d of all these (which includes the d=0 quadratic) is a piecewise function. Minimizing it over h[i] ∈ [0,1] is a 1D problem that can be solved by checking critical points.

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    for _ in range(3000):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        h *= target / max(s, 1e-30)
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            a = min(abs(d), h[idx]) if d < 0 else min(d, 1.0 - h[idx])
            h[idx] += a if d > 0 else -a
            d -= a if d > 0 else -a
    return h

def soft_max_val_grad(h, alpha, topk=1, flat_w=0.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    K = 2 * n - 1
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    e = np.exp(s)
    S = e.sum()
    w = e / S
    val = m + np.log(S) / alpha
    grad = np.zeros(n)
    for idx in range(K):
        d = idx - (n - 1)
        wk = w[idx]
        if wk < 1e-18:
            continue
        jm1 = max(0, d)
        jM1 = min(n - 1, n + d - 1)
        if jm1 <= jM1:
            grad[jm1:jM1+1] += wk * (1.0 - h[np.arange(jm1, jM1+1) - d])
        jm2 = max(0, -d)
        jM2 = min(n - 1, n - d - 1)
        if jm2 <= jM2:
            grad[jm2:jM2+1] -= wk * h[np.arange(jm2, jM2+1) + d]
    if topk > 1:
        sc = np.sort(corr)[::-1][:topk]
        val += flat_w * np.sum((sc - sc.mean())**2)
    return val, grad
