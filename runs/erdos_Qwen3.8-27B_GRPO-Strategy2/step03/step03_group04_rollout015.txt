import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)


def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h


def _grad_one_lag(h, idx, n):
    """Gradient of correlation at index idx (shift = idx-(n-1)) w.r.t. h."""
    d = idx - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        for i in range(n):
            v = 0.0
            if i <= n - 1 - d:
                v += (1.0 - h[i + d])
            if i >= d:
                v -= h[i - d]
            g[i] = v
    else:
        e = -d
        for i in range(n):
            v = 0.0
            if i >= e:
                v += (1.0 - h[i - e])
            if i <= n - 1 - e:
                v -= h[i + e]
            g[i] = v
    return g


def _grad_active(h, active_idx, n):
    n_a = len(active_idx)
    if n_a == 0:
        return np.zeros(n)
    g = np.zeros(n)
    for idx in active_idx:
        g += _grad_one_lag(h, idx, n)
    return g / n_a


def _obj_alpha(x, alpha):
    return _smooth_max(_corr(x), alpha)


def _eq_target(x, target):
    return x.sum() - target


def _do_slsqp(x0, bounds, target, alpha, maxiter):
    res = minimize(_obj_alpha, x0, args=(alpha,), method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': _eq_target, 'args': (target,)}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return res.x
