    # = (1/p) * (sum |c|^p)^(1/p - 1) * p * |c|^(p-1) * sign(c) * dc/dh

def _project(h, target):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=target}"""
    n = len(h)
    # Water-filling: find lambda such that sum(clip(h_i - lambda, 0, 1)) = target
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any tiny numerical issues
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        # Distribute among non-boundary points
        free = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        if free.sum() > 0:
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

            p[f] += d / nf
        else:
            p += d / len(p)
    return np.clip(p, 0.0, 1.0)

def _sum_eq(x, target):
    return x.sum() - target

def _grad_lag(h, n, m):
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        g[:n + k] += (1.0 - h[-k:])
        g[-k:] -= h[:n + k]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'args': (target,)}]

    def tleft():
        return budget_s - (time.time() - t0)

    h = _project(np.array(initial_h_values, dtype=float).copy(), target)
    best_h = h.copy()
    best_c5 = _c5(h, dx)

    # Phase 1: SLSQP with increasing alpha
    for alpha in [2000, 8000, 25000]:
        if tleft() < 320:
            break
        try:
            res = minimize(_lse_obj, h, args=(alpha,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-16, 'iprint': 0})
            hc = _project(np.clip(res.x, 0, 1), target)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h, h = c5, hc, hc
        except Exception:
            pass

    # Phase 2: Coordinate descent with fine grid
    grid_fine = np.concatenate(([0.0, 1.0], np.linspace(0.02, 0.98, 30)))
    for _sweep in range(500):
        if tleft() < 320:
            break
        any_change = False
        for i in range(n):
            if tleft() < 320:
                break
            old = h[i]
            bv = _c5(h, dx)
            bt = old
            for t in grid_fine:
                if abs(t - old) < 1e-8:
                    continue
                h[i] = t
                h = _project(h, target)
                v = _c5(h, dx)
                if v < bv - 1e-13:
                    bv, bt = v, t
            h[i] = bt
            h = _project(h, target)
            if bt != old:
                any_change = True
            cv = _c5(h, dx)
            if cv < best_c5:
                best_c5, best_h = cv, h.copy()
        if not any_change:
            break

    # Phase 3: Gradient-guided paired moves
    if tleft() > 200:
        h = best_h.copy()
        for _it in range(50000):
            if tleft() < 200:
                break
            c = _corr(h)
            m_star = int(np.argmax(c))
            g = _grad_lag(h, n, m_star)
            i_best = int(np.argmax(g))
            j_best = int(np.argmin(g))
            if g[i_best] - g[j_best] < 1e-10:
                i_best, j_best = rng.randint(n), (rng.randint(n) + 1) % n
            dm = min(1.0 - h[i_best], h[j_best])
            if dm < 1e-7:
                dm2 = min(1.0 - h[j_best], h[i_best])
                if dm2 < 1e-7:
                    continue
                i_best, j_best = j_best, i_best
                dm = dm2
            bv = _c5(h, dx)
            bd = 0.0
            for d in np.linspace(0.05 * dm, dm, 10):
                h[i_best], h[j_best] = h[i_best] + d, h[j_best] - d
                h = _project(h, target)
                v = _c5(h, dx)
                if v < bv:
                    bv, bd = v, d
            if bd > 0:
                h[i_best], h[j_best] = h[i_best] + bd, h[j_best] - bd
                h = _project(h, target)
                cv = _c5(h, dx)
                if cv < best_c5:
                    best_c5, best_h = cv, h.copy()
            else:
                # random pair fallback
                ri, rj = rng.randint(n), (rng.randint(n) + 1) % n
                rd = min(1.0 - h[ri], h[rj])
                if rd > 1e-5:
                    hd = rd * rng.uniform(0.2, 0.8)
                    h[ri], h[rj] = h[ri] + hd, h[rj] - hd
                    h = _project(h, target)
                    cv = _c5(h, dx)
                    if cv < best_c5:
                        best_c5, best_h = cv, h.copy()

    # Phase 4: SLSQP multi-restart
    for _r in range(12):
        if tleft() < 80:
            break
        hp = best_h + 0.012 * rng.randn(n)
        hp = _project(hp, target)
        try:
            res = minimize(_lse_obj, hp, args=(15000,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 350, 'ftol': 1e-15, 'iprint': 0})
            hc = _project(np.clip(res.x, 0, 1), target)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    # Phase 5: Binary push then re-optimize
    if tleft() > 100:
        hb = best_h.copy()
        mask_lo = hb < 0.08
        mask_hi = hb > 0.92
        hb[mask_lo] = 0.0
        hb[mask_hi] = 1.0
        hb = _project(hb, target)
        c5b = _c5(hb, dx)
        if c5b < best_c5:
            best_c5, best_h = c5b, hb
        else:
            # partial push: only most extreme
            thr = np.percentile(best_h[best_h > 0.05 & (best_h < 0.95)], 10)
            hb2 = best_h.copy()
            hb2[hb2 < 0.05] = 0.0
            hb2[hb2 > 0.95] = 1.0
            hb2 = _project(hb2, target)
            c5b2 = _c5(hb2, dx)
            if c5b2 < best_c5:
                best_c5, best_h = c5b2, hb2

    # Phase 6: Fine random perturbation
    if tleft() > 15:
        h = best_h.copy()
        c5_cur = best_c5
        for _ in range(800000):
            if tleft() < 15:
                break
            mag = 10.0 ** (rng.randint(3) - 5)
            d = rng.choice([-1.0, 1.0]) * mag
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                h = _project(h, target)
                c5 = _c5(h, dx)
                if c5 < c5_cur:
                    c5_cur = c5
                    if c5 < best_c5:
                        best_c5, best_h = c5, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _project(best_h, target)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
