import numpy as np
from scipy.optimize import minimize
import time

def corr_func(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_func(h, dx):
    return float(np.max(corr_func(h))) * dx

def repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def lag_gradient(h, L, n):
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[:n-L] += 1.0 - h[L:]
        g[L:] -= h[:n-L]
    else:
        M = -L
        g[M:] += 1.0 - h[:n-M]
        g[:n-M] -= h[M:]
    return g

def soft_obj(x, alpha, lam, target):
    c = corr_func(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + lam * (np.sum(x) - target) ** 2

def soft_grad(x, alpha, lam, target, n):
    c = corr_func(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        g += pk * lag_gradient(x, L, n)
    g /= alpha
    g += 2.0 * lam * (np.sum(x) - target)
    return g

def slsq_sum(x):
    return np.sum(x)

def slsq_sum_grad(x):
    return np.ones(len(x))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def el():
        return time.time() - t0

    h = repair(np.array(initial_h_values, dtype=float), n)
    best_h = h.copy()
    best_c5 = c5_func(best_h, dx)

    # Phase 1: L-BFGS-B multi-alpha
    for alpha in [20000, 80000, 300000, 1000000]:
        if el() > budget_s - 320:
            break
        for lam in [40.0, 120.0]:
            if el() > budget_s - 320:
                break
            try:
                res = minimize(soft_obj, h, jac=soft_grad,
                               args=(alpha, lam, target, n),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 800, 'ftol': 1e-18, 'gtol': 1e-14})
                hc = repair(res.x, n)
                c5 = c5_func(hc, dx)
                if c5 < best_c5:
                    best_h, best_c5, h = hc, c5, hc
            except Exception:
                pass

    # Phase 2: SLSQP exact constraint
    cons = [{'type': 'eq', 'fun': slsq_sum, 'jac': slsq_sum_grad}]
    for alpha in [100000, 500000]:
        if el() > budget_s - 220:
            break
        try:
            res = minimize(soft_obj, best_h, jac=soft_grad,
                           args=(alpha, 0.0, target, n),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 1000, 'ftol': 1e-16})
            hc = repair(res.x, n)
            c5 = c5_func(hc, dx)
            if c5 < best_c5:
                best_h, best_c5 = hc, c5
                h = hc
        except Exception:
            pass

    # Phase 3: Active-set smooth descent on top lags
    h = best_h.copy()
    for n_top in [3, 5, 7, 10]:
        if el() > budget_s - 200:
            break
        for step in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]:

                if not moved:

    # Phase 4: Targeted pair coordinate descent

                important[L:] += np.abs(1 - h[L:]) + np.abs(h[:n-L])
            else:
                M = -L

                important[:n-M] += np.abs(1 - h[:n-M]) + np.abs(h[M:])
        order = np.argsort(important)[::-1][:35]
        found = False
        for ii in range(len(order)):
            for jj in range(ii + 1, min(ii + 12, len(order))):
                i, j = order[ii], order[jj]
                bd, bc = 0.0, c5_cur
                for d in psteps:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        cn = c5_func(h, dx)
                        if cn < bc:
                            bc, bd = cn, d
                        h[i], h[j] = ni - d, nj + d
                if bd != 0:
                    h[i] += bd
                    h[j] -= bd
                    h = repair(h, n)
                    c5_cur = c5_func(h, dx)
                    if c5_cur < best_c5:
                        best_h, best_c5 = h.copy(), c5_cur
                    found = True
                    break
            if found:
                break
        if not found:
            break

    # Phase 5: Simulated annealing
    h = best_h.copy()
    c5_cur = c5_func(h, dx)
    T = 1.2e-5
    for _it in range(100000):
        if el() > budget_s - 45:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.02, 0.02)
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        h = repair(h, n)
        cn = c5_func(h, dx)
        if cn < c5_cur or (T > 1e-12 and rng.rand() < np.exp(-(cn - c5_cur) / T)):
            c5_cur = cn
            if cn < best_c5:
                best_h, best_c5 = h.copy(), cn
        else:
            h[i] -= d
            h[j] += d
            h = repair(h, n)
        T *= 0.99987
        if T < 1e-12:
            T = 1e-12

    # Phase 6: Ultra-fine micro-polish
    h = best_h.copy()
    c5_cur = c5_func(h, dx)
    tiny = np.array([8e-5, 2e-4, 5e-4, 1e-3, 2e-3,
                     -8e-5, -2e-4, -5e-4, -1e-3, -2e-3])
    for _ in range(50000):
        if el() > budget_s - 2:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = tiny[rng.randint(len(tiny))]
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            cn = c5_func(h, dx)
            if cn < c5_cur:
                c5_cur = cn
                best_h = h.copy()
                best_c5 = cn
            else:
                h[i], h[j] = ni - d, nj + d

    best_h = repair(best_h, n)
    best_c5 = c5_func(best_h, dx)
    return (best_h, best_c5, n)
