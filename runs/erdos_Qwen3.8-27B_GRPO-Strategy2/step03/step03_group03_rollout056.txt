**Let me try a fundamentally different approach: use the KKT conditions.**

At the optimum, for the active lags A:
- c_k = t for all k ∈ A
- The gradient of t = max_{k∈A} c_k is g_A = (1/|A|) Σ_{k∈A} g_k
- g_A should be orthogonal to the feasible set (i.e., for free variables, g_A[j] = 0; for h[j]=0, g_A[j] ≥ 0; for h[j]=1, g_A[j] ≤ 0)

So the KKT conditions are:
1. c_k = t for k ∈ A
2. c_k ≤ t for k ∉ A
3. For j with 0 < h[j] < 1: (Σ_{k∈A} g_k[j]) / |A| = 0
4. For j with h[j] = 0: (Σ_{k∈A} g_k[j]) / |A| ≥ 0
5. For j with h[j] = 1: (Σ_{k∈A} g_k[j]) / |A| ≤ 0

This gives us a system to solve! We can use it as a verification and also as a guide for optimization.

**Practical algorithm based on KKT:**

1. Start with h from initial_h_values
2. Compute all c_k, find active set A = {k : c_k > t - ε}
3. Compute g_A = mean of gradients over A
4. For free variables (0 < h[j] < 1), we want g_A[j] = 0
5. This means: adjust h to make g_A as close to 0 as possible on free variables
6. This is a linear problem! g_A depends linearly on h (since c_k is bilinear, its gradient is linear in h).

g_k[j] = (1-h[j+k])·[j<n-k] - h[j-k]·[j≥k]

So the KKT condition "g_A[j] = 0 for free j" is a LINEAR system in h!

This is a huge insight. We can solve:

The linear part gives us directions to move. The bilinear part (c_k = t) is the hard part.

def _de_obj(x):
    h = np.clip(x, 0, 1)
    pen = 500.0 * (h.sum() - _g_target) ** 2
    return _c5(h, 2.0 / len(h)) + pen

def _repair(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(40):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        m = (h < 0.999999) if diff > 0 else (h > 1e-6)
        if not m.any():
            m = (h < 1.0) if diff > 0 else (h > 0.0)
        if not m.any():
            break
        h[m] += diff / m.sum()
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmin(h))
        h[idx] = float(np.clip(h[idx] + diff, 0, 1))
    return h

def _grad_k(h, n, k):
    g = np.zeros(n)
    if k >= 0:
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    else:
        kk = -k
        g[kk:] += (1.0 - h[:n - kk])
        g[:n - kk] -= h[kk:]
    return g

def _active_grad(h, n, thresh):
    c = _corr(h)
    mc = c.max()
    active = np.where(c > thresh * mc)[0]
    g = np.zeros(n)
    for m in active:
        g += _grad_k(h, n, m - (n - 1))
    return g / max(len(active), 1)

def run(seed=42, budget_s=1100, **kwargs):
    global _g_target
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - (time.time() - t0)

    n = len(initial_h_values)
    dx = 2.0 / n
    _g_target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h0 = _repair(np.array(initial_h_values, dtype=float), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: Differential evolution (global exploration)
    if tleft() > 200:
        try:
            bounds_de = np.array([(0.0, 1.0)] * n)
            res = differential_evolution(_de_obj, bounds_de, maxiter=300,
                                         popsize=80, tol=1e-10, seed=seed,
                                         polish=False, init='sobol',
                                         mutation=(0.3, 1.0), recombination=0.8)
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    # Phase 2: SLSQP with multi-alpha smoothing
    for alpha in [10000, 100000, 1000000, 5000000]:
        if tleft() < 120:
            break
        try:
            res = minimize(_smooth_obj, best_h, args=(alpha,), method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _sum_con}],
                           options={'maxiter': 3000, 'ftol': 1e-18, 'iprint': 0})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    # Phase 3: Subgradient with adaptive step and Nesterov momentum
    if tleft() > 250:
        h = best_h.copy()
        v = np.zeros(n)
        c5_cur = best_c5
        step = 0.03
        mu = 0.85
        stagnation = 0
        for it in range(300000):
            if tleft() < 250:
                break
            g = _active_grad(h, n, 0.96)
            v = mu * v - step * g
            h_new = _repair(h + v, n)
            c5_new = _c5(h_new, dx)
            if c5_new < c5_cur:
                h, c5_cur, v = h_new, c5_new, v
                step = min(step * 1.01, 0.15)
                stagnation = 0
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h.copy()
            else:
                v *= 0.5
                step *= 0.8
                stagnation += 1
                if stagnation > 30:
                    # Broader active set
                    g2 = _active_grad(h, n, 0.85)
                    h_try = _repair(h - 0.005 * g2, n)
                    c5_try = _c5(h_try, dx)
                    if c5_try < c5_cur:
                        h, c5_cur = h_try, c5_try
                        v = np.zeros(n)
                        step = 0.02
                        stagnation = 0
                        if c5_cur < best_c5:
                            best_c5, best_h = c5_cur, h.copy()
                    else:
                        step = 0.005
                        mu = 0.7
                        v = np.zeros(n)
                        stagnation = 0

    # Phase 4: O(n) coordinate descent on worst lag
    if tleft() > 120:
        h = best_h.copy()
        c5_cur = best_c5
        for _round in range(60000):
            if tleft() < 120:
                break
            c = _corr(h)
            mi = int(np.argmax(c))
            k = mi - (n - 1)
            g = _grad_k(h, n, k)
            neg = np.where(g < -1e-7)[0]
            pos = np.where(g > 1e-7)[0]
            if len(neg) == 0 or len(pos) == 0:
                i1, i2 = rng.randint(n), rng.randint(n)
                if i1 != i2:
                    d = 0.003 * rng.randn()
                    if 0 <= h[i1] + d <= 1 and 0 <= h[i2] - d <= 1:
                        h[i1] += d
                        h[i2] -= d
                        h = _repair(h, n)
                        if _c5(h, dx) < c5_cur:
                            c5_cur = _c5(h, dx)
                            if c5_cur < best_c5:
                                best_c5, best_h = c5_cur, h.copy()
                continue
            bi = neg[int(np.argmin(g[neg]))]
            bj = pos[int(np.argmax(g[pos]))]
            dmax = min(1.0 - h[bi], h[bj]) * 0.9
            lo, hi = 1e-8, dmax
            for _ in range(10):
                mid = (lo + hi) * 0.5
                h[bi] += mid
                h[bj] -= mid
                h = _repair(h, n)
                c5_try = _c5(h, dx)
                h[bi] -= mid
                h[bj] += mid
                h = _repair(h, n)
                if c5_try < c5_cur:
                    lo, c5_cur = mid, c5_try
                else:
                    hi = mid
            if lo > 1e-7:
                h[bi] += lo
                h[bj] -= lo
                h = _repair(h, n)
                c5_cur = _c5(h, dx)
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h.copy()

    # Phase 5: Multi-restart SLSQP from perturbed points
    if tleft() > 60:
        for _r in range(10):
            if tleft() < 35:
                break
            scale = 0.005 + 0.008 * _r
            h_pert = _repair(best_h + scale * rng.randn(n), n)
            try:
                res = minimize(_smooth_obj, h_pert, args=(2000000,),
                               method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': _sum_con}],
                               options={'maxiter': 1500, 'ftol': 1e-17, 'iprint': 0})
                hc = _repair(res.x, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc
            except Exception:
                pass

    # Phase 6: Ultra-fine random pair search
    if tleft() > 20:
        h = best_h.copy()
        c5_cur = best_c5
        for _ in range(200000):
            if tleft() < 20:
                break
            i1 = rng.randint(n)
            i2 = rng.randint(n)
            if i1 == i2:
                continue
            for d in (0.0005, 0.001, 0.002, 0.005, -0.0005, -0.001, -0.002, -0.005):
                ni, nj = h[i1] + d, h[i2] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i1], h[i2] = ni, nj
                    h = _repair(h, n)
                    c5_try = _c5(h, dx)
                    if c5_try < c5_cur:
                        c5_cur = c5_try
                        if c5_cur < best_c5:
                            best_c5, best_h = c5_cur, h.copy()
                        break
                    else:
                        h[i1], h[i2] = ni - d, nj + d
                        h = _repair(h, n)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
