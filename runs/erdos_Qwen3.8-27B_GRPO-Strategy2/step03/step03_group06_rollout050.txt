from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair_binary(h, n, target):
    """Repair to binary with exact sum."""
    h = np.round(h)
    h = np.clip(h, 0, 1)
    s = h.sum()
    diff = int(round(target - s))
    if diff > 0:
        # Need to add diff ones
        zeros = np.where(h == 0)[0]
        idx = np.random.choice(zeros, diff, replace=False)
        h[idx] = 1
    elif diff < 0:
        ones = np.where(h == 1)[0]
        idx = np.random.choice(ones, -diff, replace=False)
        h[idx] = 0
    return h

def _sa_binary(h, n, dx, target, rng, deadline, T0=0.5, T_min=0.01):
    """Simulated annealing on binary vectors."""
    h = h.copy()
    c5_cur = _c5(h, dx)
    best_c5 = c5_cur
    h_best = h.copy()
    T = T0
    cool = 0.99995
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    while time.time() < deadline and T > T_min:
        # Swap: move a 1 to a 0 position
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i] = 0
        h[j] = 1
        c5_new = _c5(h, dx)
        delta = c5_new - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            c5_cur = c5_new
            # Update ones/zeros sets
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if c5_new < best_c5:
                best_c5 = c5_new
                h_best = h.copy()
        else:
            h[i] = 1
            h[j] = 0
        T *= cool
    return h_best, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(c.max())
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _repair(h, n, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _slsqp(h, n, dx, alpha, maxiter=400):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    def obj(x):
        return _smooth_max(_corr(x), alpha)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[con], options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(res.x, n, target)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _lbfgs(h, n, dx, alpha, maxiter=300):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best_h, best_c5 = h, _c5(h, dx)
    for pen in [50.0, 500.0, 5000.0]:
        def obj(x):
            return _smooth_max(_corr(x), alpha) + pen * (np.sum(x) - target) ** 2
        try:
            res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'gtol': 1e-6, 'ftol': 1e-15})
            hc = _repair(res.x, n, target)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_h, best_c5 = hc, c5
        except Exception:
            pass
    return best_h, best_c5

def _binary_sa(h0, n, dx, target, rng, deadline, T0=0.3):
    h = np.round(np.clip(h0, 0, 1))
    diff = target - h.sum()
    if diff > 0:
        z = np.where(h == 0)[0]
        h[np.random.RandomState(rng.randint(1e9)).choice(z, int(diff), replace=False)] = 1
    elif diff < 0:
        o = np.where(h == 1)[0]
        h[np.random.RandomState(rng.randint(1e9)).choice(o, int(-diff), replace=False)] = 0
    c5_cur = _c5(h, dx)
    best_c5 = c5_cur
    h_best = h.copy()
    T = T0
    cool = 0.99997
    while time.time() < deadline and T > 0.005:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0, 1
        c5_new = _c5(h, dx)
        d = c5_new - c5_cur
        if d < 0 or rng.random() < np.exp(-d / T):
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                h_best = h.copy()
        else:
            h[i], h[j] = 1, 0
        T *= cool
    return h_best, best_c5

def _fine_local(h, n, dx, rng, deadline, n_tries=300000):
    best_c5 = _c5(h, dx)
    h = h.copy()
    scales = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    per_scale = max(1000, n_tries // len(scales))
    for sc in scales:
        improved = False
        for _ in range(per_scale):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:

            h[i], h[j] = ni, nj

            if c5n < best_c5 - 1e-15:
                best_c5 = c5n
                improved = True
            else:
                h[i], h[j] = ni - d, nj + d
        if not improved and sc < 0.002:
            break
    return h, best_c5

def _targeted_local(h, n, dx, rng, deadline, n_tries=200000):
    best_c5 = _c5(h, dx)
    h = h.copy()
    corr = _corr(h)
    for _ in range(n_tries):
        if time.time() >= deadline:
            break
        top = np.argsort(corr)[-3:]
        kw = top[rng.randint(3)]
        offset = kw - (n - 1)
        if offset >= 0:
            idx = np.arange(n - offset)
            mask = (h[idx + offset] > 0.1) & (h[idx] < 0.9)
        else:
            idx = np.arange(-offset, n)
            mask = (h[idx] > 0.1) & (h[idx + offset] < 0.9)
        if mask.sum() == 0:
            i, j = rng.randint(n), rng.randint(n)
        else:
            cand = idx[mask]
            i = cand[rng.randint(len(cand))]
            j = i + offset if offset > 0 else i
            if offset > 0:
                j = i
                i = i + offset
            else:
                j = i - offset
                i = i
            if i == j or i < 0 or i >= n or j < 0 or j >= n:
                i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = 0.008 * rng.random()
        if offset >= 0:
            hi, hj = h[i] - d, h[j] + d
        else:
            hi, hj = h[j] - d, h[i] + d
        hi = np.clip(hi, 0, 1)
        hj = np.clip(hj, 0, 1)
        h[i], h[j] = hi, hj
        c5n = _c5(h, dx)
        if c5n < best_c5 - 1e-15:
