import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _g_vec(h, i, n):
    m = np.arange(-(n-1), n)
    g = np.zeros(2*n - 1)
    idx1 = i + m
    mask1 = (idx1 >= 0) & (idx1 < n)
    g[mask1] += (1.0 - h[idx1[mask1]])
    idx2 = i - m
    mask2 = (idx2 >= 0) & (idx2 < n)
    g[mask2] -= h[idx2[mask2]]
    return g

def _minimax_1d(c, g, d_min, d_max):
    """Find d in [d_min, d_max] minimizing max(c + d*g). Convex PL."""
    if d_max - d_min < 1e-14:
        return 0.0
    lo, hi = d_min, d_max
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        t = c + mid * g
        am = np.max(t)
        grads = g[t >= am - 1e-12]
        if np.min(grads) > 1e-10:
            hi = mid
        elif np.max(grads) < -1e-10:
            lo = mid
        else:
            return mid
    return 0.5 * (lo + hi)

def _sweep_single(h, c, n, order=None, time_limit=None):
    """Single-coordinate descent sweep."""
    improved = False
    if order is None:
        order = np.arange(n)
    for i in order:
        if time_limit is not None and time.time() > time_limit:
            break
        hi = h[i]
        d_min = -hi
        d_max = 1.0 - hi
        if d_max - d_min < 1e-14:
            continue
        g = _g_vec(h, i, n)
        d_best = _minimax_1d(c, g, d_min, d_max)
        if abs(d_best) > 1e-14:
            c_new_max = np.max(c + d_best * g)
            if c_new_max < np.max(c) - 1e-13:
                h[i] += d_best
                c += d_best * g
                improved = True
    return h, c, improved

def _sweep_pair(h, c, n, order=None, time_limit=None, n_coarse=20, n_bisect=30):
    """Pairwise coordinate descent sweep."""
    improved = False
    if order is None:
        order = np.random.permutation(n)
    for ai, i in enumerate(order):
        for j in order[ai+1:]:
            if time_limit is not None and time.time() > time_limit:
                return h, c, improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-14:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            # Minimize max(c + d*gd) for d in [d_min, d_max]
            d_best = _minimax_1d(c, gd, d_min, d_max)
            if abs(d_best) > 1e-14:
                if np.max(c + d_best * gd) < np.max(c) - 1e-13:
                    h[i] += d_best
                    h[j] -= d_best
                    c += d_best * gd
                    improved = True
    return h, c, improved

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _slsqp(h, n, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 30})
        return _repair(res.x, n)
    except Exception:
        return h

        h = rng.random(n)

    c = _corr(h)

        if c5 < best_c5:
            best_c5 = c5

        c5 = float(np.max(c) * dx)
        if c5 < best_c5:

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _g_vec(h, i, n):
    m = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = i + m
    mk1 = (idx1 >= 0) & (idx1 < n)
    g[mk1] += (1.0 - h[idx1[mk1]])
    idx2 = i - m
    mk2 = (idx2 >= 0) & (idx2 < n)
    g[mk2] -= h[idx2[mk2]]
    return g

def _minimax_1d(c, g, dlo, dhi):
    if dhi - dlo < 1e-14:
        return 0.0
    lo, hi = dlo, dhi
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        t = c + mid * g
        am = np.max(t)
        grads = g[t >= am - 1e-12]
        if np.min(grads) > 1e-10:
            hi = mid
        elif np.max(grads) < -1e-10:
            lo = mid
        else:
            return mid
    return 0.5 * (lo + hi)

def _sweep1(h, c, n, order, tlim):
    improved = False
    for i in order:
        if time.time() > tlim:
            break
        dlo, dhi = -h[i], 1.0 - h[i]
        if dhi - dlo < 1e-14:
            continue
        g = _g_vec(h, i, n)
        d = _minimax_1d(c, g, dlo, dhi)
        if abs(d) > 1e-14 and np.max(c + d * g) < np.max(c) - 1e-13:
            h[i] += d
            c += d * g
            improved = True
    return h, c, improved

def _sweep2(h, c, n, order, tlim):
    improved = False
    for ai in range(len(order)):
        for aj in range(ai + 1, len(order)):
            if time.time() > tlim:
                return h, c, improved
            i, j = order[ai], order[aj]
            dlo = max(-h[i], h[j] - 1.0)
            dhi = min(1.0 - h[i], h[j])
            if dhi - dlo < 1e-14:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            d = _minimax_1d(c, gd, dlo, dhi)
            if abs(d) > 1e-14 and np.max(c + d * gd) < np.max(c) - 1e-13:
                h[i] += d
                h[j] -= d
                c += d * gd
                improved = True
    return h, c, improved

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= tgt / s
    h = np.clip(h, 0.0, 1.0)
    diff = tgt - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for k in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[k])
                h[k] += a
                diff -= a
            else:
                a = min(-diff, h[k])
                h[k] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _smax(c, a):
    m = float(np.max(c))
    s = a * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / a

def _obj_sm(x, a):
    return _smax(_corr(x), a)

def _eqc(x, t):
    return float(np.sum(x)) - t

def _slsqp(h, n, t, bnd, a, mi):
    cons = [{'type': 'eq', 'fun': _eqc, 'args': (t,)}]
    try:
        r = minimize(_obj_sm, h, args=(a,), method='SLSQP',
                     bounds=bnd, constraints=cons,
                     options={'maxiter': mi, 'ftol': 1e-14, 'maxls': 30})
        return _repair(r.x, n)
    except Exception:
        return h
