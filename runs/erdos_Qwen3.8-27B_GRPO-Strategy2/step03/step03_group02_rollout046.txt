        chunk = min(35.0, t_b - time.time())
        h, corr_s, imp = _pair_sweep(h, n, corr_s, time.time() + chunk)
        if imp:
            c5_s = _c5(h, dx)
            if c5_s < best_c5 - 1e-14:
                best_c5 = c5_s
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        # Periodic SLSQP
        if sw_iter % 3 == 0 and time.time() < t_b - 30:
            alpha_s = 200 + 80 * sw_iter
            hc = _slsqp_step(best_h, n, target, alpha_s, 300)
            hc_corr = _corr(hc)
            hc, hc_corr, _ = _pair_sweep(hc, n, hc_corr, time.time() + 15)
            c5_t = _c5(hc, dx)
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
                best_h = hc.copy()
                stall = 0

        # Perturbation when stalled
        if stall >= 3 and time.time() < t_b - 40:
            scale = min(0.12, 0.04 + 0.02 * stall)
            h_p = best_h + scale * rng.randn(n)
            h_p = _project(h_p, target)
            hc = _gradient_descent(h_p, target, n, 100, 400, time.time() + 10)
            hc_corr = _corr(hc)
            hc, hc_corr, _ = _pair_sweep(hc, n, hc_corr, time.time() + 15)
            c5_t = _c5(hc, dx)
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
                best_h = hc.copy()
                stall = 0
            # Also try fourier perturbation
            if stall > 4:
                hf = _fourier_perturb(best_h, n, rng)
                hf_corr = _corr(hf)
                hf, hf_corr, _ = _pair_sweep(hf, n, hf_corr, time.time() + 12)
                c5_f = _c5(hf, dx)
                if c5_f < best_c5 - 1e-14:
                    best_c5 = c5_f
                    best_h = hf.copy()
                    stall = 0
            stall = 0

    # Phase C: High-alpha SLSQP polish (15% budget)
    t_c = t0 + budget_s * 0.88
    for alpha_p in [500, 1000, 2000, 3000]:
        if time.time() > t_c:
            break
        hc = _slsqp_step(best_h, n, target, alpha_p, 400)
        c5_p = _c5(hc, dx)
        if c5_p < best_c5 - 1e-14:
            best_c5 = c5_p
            best_h = hc.copy()

    # Phase D: Final fine sweeps (10% budget)
    t_d = t0 + budget_s * 0.985
    for _ in range(4):
        if time.time() > t_d:
            break
        h = best_h.copy()
        corr_d = _corr(h)
        h, corr_d, _ = _pair_sweep(h, n, corr_d, time.time() + 18, 31, 45)
        c5_d = _c5(h, dx)
        if c5_d < best_c5:
            best_c5 = c5_d
            best_h = h.copy()

    # Final gradient polish
    if time.time() < t0 + budget_s * 0.99:
        hg = _gradient_descent(best_h, target, n, 500, 1000, t0 + budget_s * 0.99)
        c5_g = _c5(hg, dx)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = hg.copy()

    best_h = _project(best_h, target)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
