from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _g_vec(h, p, n):
    kk = np.arange(-(n-1), n)
    g = np.zeros(2*n-1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _slsqp_step(h, n, dx, target, bounds, alpha, maxiter):

        c = _corr(x)
        m = np.max(c)
        s = alpha * (c - m)
        s = np.clip(s, -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha

from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _compute_G(h, n):
    G = np.zeros((n, 2*n-1))
    for p in range(n):
        G[p, n-1-p : 2*n-1-p] += (1.0 - h)
        G[p, p : p+n] -= h[::-1]
    return G

def _update_G(G, n, a, delta):
    # h[a] changed by delta
    k_lo, k_hi = a-(n-1), a
    for k in range(k_lo, k_hi+1):
        G[a-k, k+n-1] -= delta
    k_lo, k_hi = -a, n-1-a
    for k in range(k_lo, k_hi+1):
        G[a+k, k+n-1] -= delta

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _compute_G(h, n):
    G = np.zeros((n, 2*n-1))
    for p in range(n):
        G[p, n-1-p:2*n-1-p] += (1.0 - h)
        G[p, p:p+n] -= h[::-1]
    return G

def _update_G(G, n, a, delta):
    p1 = np.arange(n-1, -1, -1)
    k1 = np.arange(a, a+n)
    G[p1, k1] -= delta
    p2 = np.arange(n)
    k2 = np.arange(n-1-a, 2*n-1-a)
    G[p2, k2] -= delta

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _pair_sweep(h, n, time_limit, max_rounds=200):
