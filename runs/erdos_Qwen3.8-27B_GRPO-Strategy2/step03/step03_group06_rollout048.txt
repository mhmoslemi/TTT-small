import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0, 1)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0:
                d = min(diff, 1.0 - h[i])
                h[i] += d
                diff -= d
            else:
                d = min(-diff, h[i])
                h[i] -= d
                diff += d
            if abs(diff) < 1e-14:
                break
    return h

def _soft_max(corr, alpha):
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -100, 100)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _obj_smooth(x, alpha, pen, target):
    c = _corr(x)
    return _soft_max(c, alpha) + pen * (x.sum() - target) ** 2

def _lbfgs(h, n, dx, alpha, pen, maxiter):
    target = n / 2.0
    bounds = [(0, 1)] * n
    try:
        res = minimize(_obj_smooth, h, args=(alpha, pen, target),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-8})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except:
        return h, _c5(h, dx)

def _slsqp(h, n, dx, alpha, maxiter):
    target = n / 2.0
    bounds = [(0, 1)] * n
    con = {'type': 'eq', 'fun': lambda x: x.sum() - target}
    try:
        res = minimize(lambda x: _soft_max(_corr(x), alpha), h,

                       options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except:
        return h, _c5(h, dx)
