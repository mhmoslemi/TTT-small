        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad_loop(h, n, max_iter, t_budget_fn, remaining_fn):
    c5_curr = _eval(h)
    lr = 0.005
    stall = 0
    for it in range(max_iter):
        if remaining_fn() is None or remaining_fn() < 30:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        g = _subgrad_true(c, h, n)
        step = lr
        improved = False
        for _bs in range(25):
            h_try = _repair(h - step * g, n)
            c5_try = _eval(h_try)
            if c5_try < c5_curr - 1e-14:
                h = h_try
                c5_curr = c5_try
                improved = True
                break
            step *= 0.5
        if not improved:
            stall += 1
            lr *= 0.8
        else:
            stall = 0
        if it % 200 == 199:
            lr *= 0.97
        if lr < 1e-9:
            break
        if stall > 500:
            break
    return h, c5_curr

def _coord_search(h, n, max_iter, remaining_fn, scale0=0.02):
    c5_curr = _eval(h)
    scale = scale0
    stall = 0
    rng = np.random.RandomState(int(time.time() * 1000) % 100000)
    for it in range(max_iter):
        if remaining_fn() is None or remaining_fn() < 15:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5_curr - 1e-15:
            c5_curr = c5n
            stall = 0
            scale = min(scale * 1.05, 0.03)
        else:
            h[i], h[j] = ni - d, nj + d
            stall += 1
            if stall > 300:
                scale *= 0.6
                stall = 0
                if scale < 5e-7:
                    break
    return h, c5_curr

def _active_refine(h, n, max_iter, remaining_fn):
    c5_curr = _eval(h)
    for outer in range(5):
        if remaining_fn() is None or remaining_fn() < 20:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        m = c.max()
        active = np.where(c >= m - 2e-5)[0]
        if len(active) == 0:
            break
        # Try to reduce active lags via targeted moves
        for it in range(max_iter):
            if remaining_fn() is None or remaining_fn() < 20:
                break
            i = int(remaining_fn() * 1000) % n
            j = (i + 1 + (it * 7) % n) % n
            for d in [0.01, -0.01, 0.005, -0.005, 0.002, -0.002]:
                ni, nj = h[i] + d, h[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5_curr - 1e-15:
                    c5_curr = c5n
                    break
                h[i], h[j] = ni - d, nj + d
    return h, c5_curr

def _interpolate_h(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0, 2, n_src, endpoint=False) + 1.0 / n_src
    x_dst = np.linspace(0, 2, n_dst, endpoint=False) + 1.0 / n_dst
    return np.interp(x_dst, x_src, h_src)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)

    def _rem():
        el = time.time() - t0
        rem = budget_s - el
        return rem if rem > 0 else None

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _eval(best_h)

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]

    # Phase 1: SLSQP warm-up with moderate alpha
    for alpha in [500, 5000, 50000]:
        if _rem() is None or _rem() < 400:
            break
        h_try = _slsqp(best_h, n, alpha, 1500, bnds, cons)
        c5_try = _eval(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 2: True-objective subgradient descent (key new component)
    if _rem() is not None and _rem() > 250:
        h_try, c5_try = _subgrad_loop(best_h.copy(), n, 100000, None, _rem)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 3: SLSQP polish with high alpha
    for alpha in [200000, 2000000]:
        if _rem() is None or _rem() < 120:
            break
        h_try = _slsqp(best_h, n, alpha, 2000, bnds, cons)
        c5_try = _eval(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 4: Active-set refinement
    if _rem() is not None and _rem() > 100:
        h_try, c5_try = _active_refine(best_h.copy(), n, 50000, _rem)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 5: Coordinate search with adaptive scale
    if _rem() is not None and _rem() > 60:
        h_try, c5_try = _coord_search(best_h.copy(), n, 80000, _rem, 0.015)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 6: Try another n for potentially better resolution
    for n_try in [n + 10, n - 10, n + 20]:
        if _rem() is None or _rem() < 100:
            break
        if n_try < 30:
            continue
        h_interp = _interpolate_h(best_h, n, n_try)
        h_interp = _repair(h_interp, n_try)
        bnds_t = [(0.0, 1.0)] * n_try
        cons_t = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_try / 2.0,
                   'jac': lambda x: np.ones(n_try)}]
        h_s = _slsqp(h_interp, n_try, 50000, 1000, bnds_t, cons_t)
        h_s, c5_s = _subgrad_loop(h_s, n_try, 30000, None, _rem)
        if _rem() is not None and _rem() > 40:
            h_s, c5_s = _coord_search(h_s, n_try, 20000, _rem, 0.01)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s
            n = n_try

    # Phase 7: Final subgradient polish
    if _rem() is not None and _rem() > 40:
        h_try, c5_try = _subgrad_loop(best_h.copy(), n, 50000, None, _rem)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 8: Ultra-fine coordinate moves
    if _rem() is not None and _rem() > 15:
        h_try, c5_try = _coord_search(best_h.copy(), n, 50000, _rem, 0.003)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
