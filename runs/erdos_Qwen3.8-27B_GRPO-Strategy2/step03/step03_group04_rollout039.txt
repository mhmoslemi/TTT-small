Let me also make the local search more efficient. Instead of computing the full C5 for each trial, I can compute only the change.

If I change h[i] → h[i]+d and h[j] → h[j]-d, the change in C[m] (for each m) is:
ΔC[m] = d * ∂C[m]/∂h[i] - d * ∂C[m]/∂h[j]

And I only need to check if the maximum decreases. The new maximum is:
max_m (C[m] + ΔC[m])

I can precompute ∂C[m]/∂h[i] for all m for each i, but that's O(n²) storage. For n=70, that's 4900 values per i, or 343000 total. That's fine.

Actually, let me precompute the "gradient snapshot": for each position m in h, the vector of partial derivatives ∂C[k]/∂h[m] for all k. Then a pair change (i, j, d) gives:
ΔC = d * grad_i - d * grad_j

And the new max is max(C + d*(grad_i - grad_j)).

This is O(n) per evaluation instead of O(n²). But the precomputation is O(n³). For n=70, that's 343000, which is trivial.

Let me implement this optimization for the local search phase.

Actually, for the local search, the gradient of C w.r.t. h[i] is:
For each lag m (0 to 2n-2 in the full correlate):
∂C[m]/∂h[i] = contribution from the term where h[i] appears

From my earlier analysis:
For m = n-1+k (k≥0): ∂C[n-1+k]/∂h[i] = (1-h[i-k])*[i≥k] - h[i+k]*[i≤n-1-k]
For m = n-1-k (k≥1): ∂C[n-1-k]/∂h[i] = (1-h[i+k])*[i≤n-1-k] - h[i-k]*[i≥k]

So I can precompute a matrix G of shape (n, 2n-1) where G[i, m] = ∂C[m]/∂h[i].

Then for a pair change (i, j) with delta d:
new_C = C + d * (G[i] - G[j])
new_max = max(new_C) * dx

This is O(n) per trial after precomputation.

    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corr(h)) * dx)

def _sm_max(c, alpha):
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sm_weights(c, alpha):
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    e = np.exp(s)
    return e / e.sum()

def _grad_sm(h, w, n):
    g = np.zeros(n)
    for k in range(n):
        wkp = w[n - 1 + k]
        if wkp > 1e-14:
            g[k:] += wkp * (1.0 - h[:n - k])
            g[:n - k] -= wkp * h[k:]
        if k > 0:
            wkn = w[n - 1 - k]
            if wkn > 1e-14:
                g[:n - k] += wkn * (1.0 - h[k:])
                g[k:] -= wkn * h[:n - k]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    diff = target - h.sum()

        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)

            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
        if abs(diff) < 1e-12:
            break
    return h

    G = np.zeros((n, 2 * n - 1))

        for k in range(n):

            # dC[n-1+k]/dh[i] = (1-h[i-k])*[i>=k] - h[i+k]*[i<=n-1-k]
            if i >= k:
                G[i, n - 1 + k] += (1.0 - h[i - k])
            if i <= n - 1 - k:
                G[i, n - 1 + k] -= h[i + k]
            if k > 0:
                # C[n-1-k] = sum h[j]*(1-h[j+k])
                # dC[n-1-k]/dh[i] = (1-h[i+k])*[i<=n-1-k] - h[i-k]*[i>=k]
                if i <= n - 1 - k:
                    G[i, n - 1 - k] += (1.0 - h[i + k])
                if i >= k:
                    G[i, n - 1 - k] -= h[i - k]
    return G, C

import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)


def _sm_max(c, alpha):
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _sm_weights(c, alpha):
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    e = np.exp(s)
    return e / e.sum()


def _grad_sm(h, w, n):
    g = np.zeros(n)
    for k in range(n):
        wkp = w[n - 1 + k]
        if wkp > 1e-14:
            g[k:] += wkp * (1.0 - h[:n - k])
            g[:n - k] -= wkp * h[k:]
        if k > 0:
            wkn = w[n - 1 - k]
            if wkn > 1e-14:
                g[:n - k] += wkn * (1.0 - h[k:])
                g[k:] -= wkn * h[:n - k]
    return g


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for idx in order:
        if diff > 0:
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
        if abs(diff) < 1e-12:
            break
    return h


def _pb_obj(x, target, alpha, mu):
    c = _corr(x)
    return _sm_max(c, alpha) + mu * (x.sum() - target) ** 2


def _pb_jac(x, target, alpha, mu):
    c = _corr(x)
    w = _sm_weights(c, alpha)
    g = _grad_sm(x, w, len(x))
    g += 2.0 * mu * (x.sum() - target)
    return g


def _slq_eq(x, target):
    return x.sum() - target


def _slq_eq_jac(x, target):
    return np.ones(len(x))


def _slq_obj(x, alpha):
    return _sm_max(_corr(x), alpha)


def _slq_jac(x, alpha):
    c = _corr(x)
    w = _sm_weights(c, alpha)
    return _grad_sm(x, w, len(x))

    def elapsed():
        return time.time() - t0

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    cands = [h0, _repair(1.0 - h0, n)]
    xs = np.arange(n) / (n * 0.5)
    cands.append(_repair(0.5 + 0.25 * np.sin(np.pi * xs), n))
    best_h = None
    best_c5 = np.inf
    for hc in cands:
        v = _c5(hc, dx)
        if v < best_c5:
            best_c5 = v
            best_h = hc.copy()

    if elapsed() < budget_s - 35:

        c5_cur = _c5(h, dx)

        try:
            res = minimize(_slq_obj, best_h.copy(), args=(200000,),
                          jac=_slq_jac, method='SLSQP',

    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
