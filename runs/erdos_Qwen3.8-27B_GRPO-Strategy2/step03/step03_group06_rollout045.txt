import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _eq_fun(x, target):
    return np.sum(x) - target

def _to_binary(h, n, rng):
    b = (h >= 0.5).astype(float)
    diff = n // 2 - int(b.sum())
    if diff > 0:
        zeros = np.where(b == 0)[0]
        perm = zeros[np.argsort(-h[zeros])]
        b[perm[:diff]] = 1.0
    elif diff < 0:
        ones = np.where(b == 1)[0]
        perm = ones[np.argsort(h[ones])]
        b[perm[:(-diff)]] = 0.0
    return b

def _rand_binary(n, rng):
    pos = rng.choice(n, n // 2, replace=False)
    b = np.zeros(n)
    b[pos] = 1.0
    return b

def _sa_binary(h, n, dx, rng, deadline, T0=0.035, Tmin=0.0002, cool=0.99998):
    c5_cur = _c5(h, dx)
    best_c5 = c5_cur
    h_best = h.copy()
    T = T0
    ones_idx = np.where(h > 0.5)[0].copy()
    zeros_idx = np.where(h <= 0.5)[0].copy()
    stagnant = 0
    while time.time() < deadline and T > Tmin:
        ri = rng.randint(len(ones_idx))
        rj = rng.randint(len(zeros_idx))
        i = int(ones_idx[ri])
        j = int(zeros_idx[rj])
        h[i] = 0.0
        h[j] = 1.0
        c5_new = _c5(h, dx)
        if c5_new <= c5_cur:
            accept = True
        else:
            delta = c5_new - c5_cur
            if delta > 5.0 * T:
                accept = False
            else:
                accept = rng.random() < np.exp(-delta / T)
        if accept:
            c5_cur = c5_new
            ones_idx[ri] = j
            zeros_idx[rj] = i
            stagnation_good = c5_new < best_c5 - 1e-15
            if stagnation_good:
                best_c5 = c5_new
                h_best = h.copy()
                stagnant = 0
            else:
                stagnant += 1
            if stagnant > 40000:
                T = max(T, T0 * 0.04)
                stagnant = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
    return h_best, best_c5

def _binary_ls(h, n, dx, deadline):
    h = h.copy()
    c5 = _c5(h, dx)
    while time.time() < deadline:
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        best_val = c5
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                v = _c5(h, dx)
                if v < best_val:
                    best_val = v
                    best_pair = (int(i), int(j))
                h[i] = 1.0
                h[j] = 0.0
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            c5 = best_val
            improved = True
        if not improved:
            break
    return h, c5

def _smooth_obj(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _cont_refine(h, n, dx, deadline):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    for alpha in [2000, 8000, 20000]:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                _smooth_obj, best_h, method='L-BFGS-B', bounds=bounds,
                args=(alpha,),
                options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-8}
            )
            hc = _repair(res.x, n)
            c5v = _c5(hc, dx)
            if c5v < best_c5 - 1e-12:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass
    return best_h, best_c5

def _pair_perturb_cont(h, n, dx, rng, deadline):
    best_c5 = _c5(h, dx)
    h = h.copy()
    scales = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0004]
    per_scale = 3000
    for scale in scales:
        improved = False
        for _ in range(per_scale):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                continue
            h[i], h[j] = ni, nj
            c5v = _c5(h, dx)
            if c5v < best_c5 - 1e-14:
                best_c5 = c5v
                improved = True
            else:
                h[i], h[j] = ni - d, nj + d
        if not improved and scale < 0.003:
            break
    return h, best_c5
