    c = _corr(x)
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def _smooth_g(x, alpha, target, pen):
    n = len(x)
    c = _corr(x)
    K = len(c)
    m = np.max(c)
    s = alpha * (c - m)

        wm = w[m_idx]
        if wm < 1e-14:

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_f(x, alpha, target):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_g(x, alpha, target):
    n = len(x)
    c = _corr(x)
    K = len(c)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / np.sum(e)
    grad = np.zeros(n)
    for mi in range(K):
        wm = w[mi]
        if wm < 1e-15:
            continue
        shift = mi - (n - 1)
        i_lo = max(0, -shift)
        i_hi = min(n - 1, n - 1 - shift)
        idx = np.arange(i_lo, i_hi + 1)
        jdx = idx + shift
        grad[idx] += wm * (1.0 - x[jdx])
        grad[jdx] -= wm * x[idx]
    return grad

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx]); h[idx] += add; diff -= add
            else:
                sub = min(-diff, h[idx]); h[idx] -= sub; diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _slsqp_opt(h, n, dx, alpha, maxiter=500):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return _smooth_f(x, alpha, target)
    def jac(x):
        return _smooth_g(x, alpha, target)
    def con_f(x):
        return np.sum(x) - target
    def con_g(x):
        return np.ones(n)
    cons = {'type': 'eq', 'fun': con_f, 'jac': con_g}
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac, bounds=bounds,
                       constraints=[cons], options={'maxiter': maxiter, 'ftol': 1e-16})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _lbfgs_opt(h, n, dx, alpha, maxiter=300):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best_h, best_c5 = h.copy(), _c5(h, dx)
    for pen in [50.0, 500.0, 2000.0, 10000.0]:
        def pf(x):
            return _smooth_f(x, alpha, target) + pen * (np.sum(x) - target) ** 2
        def pg(x):
            return _smooth_g(x, alpha, target) + 2.0 * pen * (np.sum(x) - target)
        try:
            res = minimize(pf, best_h, method='L-BFGS-B', jac=pg, bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
            hc = _repair(res.x, n)
            c5v = _c5(hc, dx)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass
    return best_h, best_c5

def _pair_local(h, n, dx, rng, deadline, n_tries=80000):

    h = h.copy()
    scales = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.00008]
