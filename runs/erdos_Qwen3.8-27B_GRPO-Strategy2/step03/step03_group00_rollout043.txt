def _c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr_vec(h))) * (2.0 / n)

def _c5_vec(h, n):
    return _corr_vec(h) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_obj(x, n, alpha):
    c = _c5_vec(x, n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, n, alpha):
    dx_val = 2.0 / n
    c = _c5_vec(x, n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    total = np.sum(e)
    weights = e / total
    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-13:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += w * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += w * (1.0 - x[:n - lag])
            g[:n - lag] -= w * x[lag:]
        else:
            L = -lag
            g[:n - L] += w * (1.0 - x[L:])
            g[L:] -= w * x[:n - L]
    return g * dx_val
