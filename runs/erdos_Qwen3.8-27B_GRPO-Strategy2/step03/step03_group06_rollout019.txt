    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)

    h = np.clip(h, 0.0, 1.0)

    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corr(h)) * dx)

def _lsm_max(corr, alpha):

    s = np.clip(alpha * (corr - m), -500, 500)

def _repair(h, n):

    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)

            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub

    try:

        hc = _repair(res.x, n)

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _lsm(corr, alpha):
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _project(v, S):
    """Project v onto box [0,1] with sum = S via bisection."""
    lo_t, hi_t = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo_t + hi_t)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > S:
            lo_t = mid
        else:
            hi_t = mid
    return np.clip(v - 0.5 * (lo_t + hi_t), 0.0, 1.0)

def _grad_corr_k(h, k):
    """Gradient of correlation at shift index k (in full output)."""
    n = len(h)
    s = k - (n - 1)  # actual shift
    g = np.zeros(n)
    for t in range(n):
        if 0 <= t - s < n:
            g[t] += (1.0 - h[t - s])
        if 0 <= t + s < n:
            g[t] -= h[t + s]
    return g

    best_c5 = _c5(best_h, dx)
    h = h.copy()

    target = n / 2.0

            k_star = int(np.argmax(corr))
            g = _grad_corr_k(h, k_star)
            g -= np.mean(g)

    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return _lsm(_corr(x), alpha) + pen * (np.sum(x) - target) ** 2
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _slsqp(h, n, dx, alpha, maxiter=400):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return _lsm(_corr(x), alpha)
    con = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=[con],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _pair_search(h, n, dx, rng, deadline, n_iters, s0, s1):
    best_c5 = _c5(h, dx)
    h = h.copy()
    ratio = s1 / s0
    for it in range(n_iters):
        if time.time() >= deadline:
            break
        sc = s0 * ratio ** (it / max(1, n_iters))
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = sc * (2.0 * rng.random() - 1.0)
        ni, nj = h[i] + d, h[j] - d
        if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
            continue
        h[i], h[j] = ni, nj
        c5n = _c5(h, dx)
        if c5n < best_c5 - 1e-15:
            best_c5 = c5n
        else:
            h[i], h[j] = h[i] - d, h[j] + d
    return h, best_c5

def _block_search(h, n, dx, rng, deadline, n_iters=15000):
    best_c5 = _c5(h, dx)
    h = h.copy()
    for bs in [2, 3, 4]:
        for _ in range(n_iters // 3):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(0, max(1, n - bs + 1))
            j = rng.randint(0, max(1, n - bs + 1))
            if abs(i - j) < bs:
                continue
            eps = rng.uniform(0.002, 0.015) * (1.0 if rng.random() < 0.5 else -1.0)
            ok = True
            for di in range(bs):
                if not (0.0 <= h[i+di]+eps <= 1.0 and 0.0 <= h[j+di]-eps <= 1.0):
                    ok = False
                    break
            if not ok:
                continue
            h[i:i+bs] += eps
            h[j:j+bs] -= eps
            c5n = _c5(h, dx)
            if c5n < best_c5 - 1e-15:
                best_c5 = c5n
            else:
                h[i:i+bs] -= eps
                h[j:j+bs] += eps
    return h, best_c5

def _anneal(h, n, dx, rng, deadline):
    best_c5 = _c5(h, dx)
    hc = h.copy()
    cc = best_c5
    hb = hc.copy()
    T, Tmin = 0.003, 0.00003
    while time.time() < deadline and T > Tmin:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = 0.006 * (2.0 * rng.random() - 1.0)
        ni, nj = hc[i]+d, hc[j]-d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        hc[i], hc[j] = ni, nj
        cn = _c5(hc, dx)
        if cn < cc or rng.random() < np.exp(-max(0.0, cn-cc)/T):
            cc = cn
            if cn < best_c5:
                best_c5 = cn
                hb = hc.copy()
        else:
            hc[i], hc[j] = ni-d, nj+d
        T *= 0.9996
    return hb, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Smooth L-BFGS (35s)
    tE = t0 + 35
    for a in [300, 2000, 10000]:
        if time.time() >= tE or rem() < 700:
            break
        ht, ct = _smooth_lbfgs(best_h, n, dx, a, maxiter=250)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    # Phase 2: SLSQP (50s)
    tE = t0 + 85
    for a in [5000, 30000, 150000]:
        if time.time() >= tE or rem() < 600:
            break
        ht, ct = _slsqp(best_h, n, dx, a, maxiter=350)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    # Phase 3: Subgradient descent (250s) - core novel method
    if rem() > 400:
        tE = time.time() + min(250, rem() - 350)
        ht, ct = _subgradient_opt(best_h, n, dx, tE, lr0=0.008, n_phases=5)
        if ct < best_c5:
            best_h, best_c5 = ht, ct
        # Second subgradient pass with smaller lr
        if rem() > 250:
            tE2 = time.time() + min(100, rem() - 200)
            ht2, ct2 = _subgradient_opt(best_h, n, dx, tE2, lr0=0.002, n_phases=3)
            if ct2 < best_c5:
                best_h, best_c5 = ht2, ct2

    # Phase 4: Pair search (150s)
    if rem() > 200:
        tE = time.time() + min(150, rem() - 180)
        ht, ct = _pair_search(best_h, n, dx, rng, tE, 800000, 0.015, 0.0003)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    # Phase 5: Block search (70s)
    if rem() > 140:
        tE = time.time() + min(70, rem() - 130)
        ht, ct = _block_search(best_h, n, dx, rng, tE, 12000)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    # Phase 6: Fine annealing (80s)
    if rem() > 110:
        tE = time.time() + min(80, rem() - 100)
        ht, ct = _anneal(best_h, n, dx, rng, tE)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    # Phase 7: SLSQP polish (40s)
    if rem() > 70:
        tE = time.time() + min(40, rem() - 60)
        for a in [50000, 300000, 1000000]:
            if time.time() >= tE:
                break
            ht, ct = _slsqp(best_h, n, dx, a, maxiter=250)
            if ct < best_c5:
                best_h, best_c5 = ht, ct

    # Phase 8: Ultra-fine pair search (50s)
    if rem() > 50:
        tE = time.time() + min(50, rem() - 40)
        ht, ct = _pair_search(best_h, n, dx, rng, tE, 500000, 0.003, 0.00003)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    # Phase 9: Final subgradient (20s)
    if rem() > 25:
        tE = time.time() + min(20, rem() - 10)
        ht, ct = _subgradient_opt(best_h, n, dx, tE, lr0=0.0005, n_phases=2)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    # Phase 10: Final ultra-fine
    if rem() > 10:
        tE = time.time() + max(2, min(8, rem() - 2))
        ht, ct = _pair_search(best_h, n, dx, rng, tE, 150000, 0.0005, 0.000005)
        if ct < best_c5:
            best_h, best_c5 = ht, ct

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
