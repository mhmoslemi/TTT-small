import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max_vals(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    p = e / e.sum()
    return m + float(np.log(np.sum(e))) / alpha, p

def _h_grad(c, h, n, alpha):
    _, p = _smooth_max_vals(c, alpha)
    lags = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for ci in range(2 * n - 1):
        w = p[ci]
        if w < 1e-12:
            continue
        lag = lags[ci]
        j = np.arange(n)
        j1 = j + lag
        m1 = (j1 >= 0) & (j1 < n)
        j2 = j - lag
        m2 = (j2 >= 0) & (j2 < n)
        grad[m1] += w * (1.0 - h[j1[m1]])
        grad[m2] -= w * h[j2[m2]]
    return grad

def _obj_grad(x, alpha):
    c = _corr(x)
    val, _ = _smooth_max_vals(c, alpha)
    g = _h_grad(c, x, len(x), alpha)
    return val, g

def _project(y, n):
    target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = -min(-diff, h[idx])
            h[idx] += add
            diff -= add if diff > 0 else -add
            if abs(diff) < 1e-12:
                break
    return h

def _slsqp(h, n, alpha, maxiter=300):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'args': (target,)}]
    try:
        res = minimize(_obj_grad, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 25})
        if res.fun < _smooth_max_vals(_corr(h), alpha)[0] + 1e-10:
            return _repair(res.x, n)
    except Exception:
        pass
    return h

def _eq_fun(x, target):
    return np.sum(x) - target

def _pgd(h, n, alpha, n_iter, t_limit, lr_init=0.005):
    """Projected gradient descent with backtracking."""
    c = _corr(h)
    f_cur, _ = _smooth_max_vals(c, alpha)
    lr = lr_init
    for _ in range(n_iter):
        if time.time() > t_limit:
            break
        grad = _h_grad(c, h, n, alpha)
        y = h - lr * grad
        h_new = _project(y, n)
        c_new = _corr(h_new)
        f_new, _ = _smooth_max_vals(c_new, alpha)
        if f_new < f_cur - 1e-15:
            h, c, f_cur = h_new, c_new, f_new
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.5
            if lr < 1e-7:
                break
    return h

def _coord_sweep(h, n, t_limit, n_coarse=20, n_bisect=30):
    c = _corr(h)
    cur_max = float(np.max(c))
    lags = np.arange(-(n - 1), n)
    order = np.random.permutation(n)
    improved = 0
    for ai, i in enumerate(order):
        if time.time() > t_limit:
            break
        hi = h[i]
        for j in order[ai + 1:]:
            if time.time() > t_limit:
                break
            hj = h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            eff = np.zeros(2 * n - 1)
            for ci in range(2 * n - 1):
                lag = lags[ci]
                di = 0.0
                dj = 0.0
                if 0 <= i + lag < n:
                    di += 1.0 - h[i + lag]
                if 0 <= i - lag < n:
                    di -= h[i - lag]
                if 0 <= j + lag < n:
                    dj += 1.0 - h[j + lag]
                if 0 <= j - lag < n:
                    dj -= h[j - lag]
                eff[ci] = di - dj
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(c[None, :] + ds[:, None] * eff[None, :], axis=1)
            bi = int(np.argmin(vals))
            if vals[bi] >= cur_max - 1e-13:
                continue
            lo = max(d_min, float(ds[max(0, bi - 1)]))
            hi_b = min(d_max, float(ds[min(n_coarse - 1, bi + 1)]))
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if np.max(c + mid * eff) <= np.max(c + hi_b * eff):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            if abs(d_best) < 1e-13:
                continue
            h[i] += d_best
            h[j] -= d_best
            c += d_best * eff
            improved += 1
    return h, improved

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: SLSQP warm-start with varying alpha
    for alpha in [40, 120, 400]:
        if time.time() - t0 > 50:
            break
        hn = _slsqp(best_h, n, alpha, 250)
        c5 = _c5(hn, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hn

    # Phase 2: Main loop - PGD + coord sweeps + perturbation
    anneal_temp = 0.08
    stall = 0
    iter_count = 0
    while time.time() - t0 < budget_s - 120:
        iter_count += 1
        # PGD phase
        alpha_pgd = min(80 + 20 * iter_count, 600)
        t_pgd = min(time.time() + 35, budget_s - 100)
        h = best_h.copy()
        h = _pgd(h, n, alpha_pgd, 5000, t_pgd, lr_init=0.003)
        c5 = _c5(h, dx)

        # Coord sweep
        if time.time() < budget_s - 100:
            t_sweep = min(time.time() + 25, budget_s - 90)
            h, _ = _coord_sweep(h, n, t_sweep, 18, 28)
            c5 = _c5(h, dx)

        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
            anneal_temp = max(0.01, anneal_temp * 0.95)
        else:
            stall += 1

        # Perturbation + re-optimize when stalled
        if stall >= 2 and time.time() - t0 < budget_s - 130:
            scale = anneal_temp * (1 + 0.2 * min(stall, 5))
            h_p = best_h + scale * np.random.randn(n)
            h_p = _project(h_p, n)
            h_p = _pgd(h_p, n, 100, 3000, min(time.time() + 30, budget_s - 100))
            if time.time() < budget_s - 95:
                h_p, _ = _coord_sweep(h_p, n, min(time.time() + 15, budget_s - 85), 15, 25)
            c5_p = _c5(h_p, dx)
            if c5_p < best_c5 - 1e-14:
                best_c5 = c5_p
                best_h = h_p.copy()
                stall = 0
                anneal_temp = max(0.01, anneal_temp * 0.7)
            anneal_temp *= 0.9

        # Periodic SLSQP to escape
        if iter_count % 5 == 0 and time.time() - t0 < budget_s - 130:
            alpha_sl = min(200 + 50 * iter_count, 1500)
            hn = _slsqp(best_h, n, alpha_sl, 200)
            c5_s = _c5(hn, dx)
            if c5_s < best_c5 - 1e-14:
                best_c5 = c5_s
                best_h = hn
                stall = 0

    # Phase 3: Final high-precision polish
    if time.time() - t0 < budget_s - 20:
        for alpha_final in [800, 2000, 5000]:
            if time.time() - t0 > budget_s - 8:
                break
            hn = _slsqp(best_h, n, alpha_final, 400)
            c5 = _c5(hn, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hn

    # Phase 4: Final coord sweep
    if time.time() - t0 < budget_s - 5:
        h, _ = _coord_sweep(best_h, n, time.time() + 3, 25, 35)
        c5 = _c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
