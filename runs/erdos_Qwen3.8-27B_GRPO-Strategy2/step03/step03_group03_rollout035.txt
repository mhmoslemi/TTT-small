from scipy.optimize import minimize
import time

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_val(h, dx):
    return np.max(corr_full(h)) * dx

def smooth_max_obj(h, alpha, dx):
    c = corr_full(h)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * dx

def smooth_max_grad(h, alpha, dx):
    c = corr_full(h)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s) / np.sum(np.exp(s))
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            g[lag:n] += w[k] * (1.0 - h[:n - lag])
            g[:n - lag] -= w[k] * h[lag:]
        else:
            l2 = -lag
            g[l2:n] += w[k] * (1.0 - h[:n - l2])
            g[:n - l2] -= w[k] * h[l2:n]
    return g * dx

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return h

    """Gradient of corr[k] w.r.t. h"""
    lag = k - (n - 1)

        g[:n - lag] -= h[lag:]
    else:
        l2 = -lag
        g[l2:n] += (1.0 - h[:n - l2])
        g[:n - l2] -= h[l2:n]
    return g
