from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_scaled(h):
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / len(h)

def _sf(h, alpha):
    c = _corr_scaled(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sf_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wl = w[idx]
        if wl < 1e-15:
            continue
        lag = idx - (n - 1)
        js = max(0, lag)
        je = min(n - 1, n - 1 + lag)
        if js <= je:
            g[js:je+1] += wl * dx * (1.0 - h[js-lag:je+1-lag])
        js2 = max(0, -lag)
        je2 = min(n - 1, n - 1 - lag)
        if js2 <= je2:
            g[js2:je2+1] -= wl * dx * h[js2+lag:je2+1+lag]
    return g

def _project(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2} using bisection on lambda."""
    target = n / 2.0
    lo, hi = h.min() - target, h.max()
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def _slsqp(h0, n, alpha, maxiter):
    bnds = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
            'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_sf, h0, jac=_sf_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=con,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except:
        return _repair(h0.copy(), n)

def _grad_descent(h0, n, alpha, budget_fn, max_iter=5000):
    """Projected gradient with Armijo line search."""
    h = h0.copy()
    f_prev = _sf(h, alpha)
    step = 0.01
    for it in range(max_iter):
        if budget_fn():
            break
        g = _sf_grad(h, alpha)
        g_norm = np.sqrt(np.sum(g**2))
        if g_norm < 1e-15:
            break
        # Armijo line search
        for _ in range(30):
            h_new = _project(h - step * g, n)
            f_new = _sf(h_new, alpha)
            if f_new <= f_prev - 0.01 * step * g_norm:
                h = h_new
                f_prev = f_new
                break
            step *= 0.5
        else:
            step *= 0.5
            if step < 1e-12:
                break
    return _repair(h, n)

    h = h0.copy()
    c_current = _eval(h)
    T = 0.005  # initial temperature

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                    h[i] += a; diff -= a
                else:
                    a = min(-diff, h[i])
                    h[i] -= a; diff += a

1. Evaluate g_k(h) for all k
2. Identify the top m lags
3. Linearize g_k around current h for these top lags
4. Solve the resulting QP: min t s.t. linearized g_k(h) ≤ t, 0≤h≤1, sum(h)=n/2

The QP step can be solved with cvxpy. But since it's a QP (linear constraints), it should be fast.

Actually wait - the linearization of g_k around h is:
g_k(h + Δ) ≈ g_k(h) + ∇g_k(h)^T Δ

So the QP is:
min t
s.t. g_k(h) + ∇g_k(h)^T Δ ≤ t for k in top_m
     0 ≤ h_i + Δ_i ≤ 1
     sum(h) + sum(Δ) = n/2 (which is sum(Δ) = 0 since sum(h) = n/2)
     t free

This is a linear program (LP)! We can solve it with scipy or cvxpy. The number of variables is n+1=71, and the number of constraints is m (top lags) + 2n (bounds) + 1 (sum) ≈ m + 141.

For m=10 (top 10 lags), this is a small LP that can be solved in milliseconds.

This is essentially a **cutting plane** or **Benders decomposition** approach!

Algorithm:
1. Start with h₀
2. For iteration t=1,2,...:
   a. Evaluate g_k(h_t) for all k
   b. Find top m lags S_t
   c. Solve LP: min t s.t. g_k(h_t) + ∇g_k(h_t)^T Δ ≤ t, k∈S_t, 0≤h_t+Δ≤1, sum(Δ)=0
   d. h_{t+1} = h_t + Δ
   e. Add new constraints for newly active lags
   f. Stop when improvement < ε

    from scipy.optimize import linprog
    h = h0.copy()
    c5 = _eval(h)
    constraints = []  # list of (grad, gk_val)

        if time_check():
            break

        c = np.correlate(h, 1.0-h, mode='full') * 2.0/n
        m = c.max()
        
        # Find top lags
        top_frac = 0.1  # top 10%
        threshold = m - (m - c.min()) * top_frac
        active = np.where(c >= threshold)[0]
        
        if len(active) < 2:
            active = np.argsort(c)[-5:]
        
        # Build LP
        # Variables: Δ (n) + t (1) = n+1 variables
        # Minimize: t
        # Constraints:
        #   For each active k: ∇g_k(h)^T Δ + g_k(h) ≤ t  →  ∇g_k(h)^T Δ - t ≤ -g_k(h)
        #   0 ≤ h[i] + Δ[i] ≤ 1  →  -h[i] ≤ Δ[i] ≤ 1-h[i]
        #   sum(Δ) = 0
        
        c_lp = np.zeros(n + 1)
        c_lp[-1] = 1.0  # minimize t

        b_ub = []

            grad = np.zeros(n)
            for j in range(n):

                if 0 <= j-lag < n:
                    grad[j] += dx * (1.0 - h[j-lag])
                if 0 <= j+lag < n:
                    grad[j] -= dx * h[j+lag]

        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = [0.0]
        
        A_ub = np.array(A_ub)

        res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in range(n):
                i = order[idx]
                if diff > 0:
                    a = min(diff, 1.0 - h[i])
                    h[i] += a; diff -= a
                else:
                    a = min(-diff, h[i])
                    h[i] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sf(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / len(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sf_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wl = w[idx]
        if wl < 1e-15:
            continue
        lag = idx - (n - 1)
        js, je = max(0, lag), min(n - 1, n - 1 + lag)
        if js <= je:
            g[js:je + 1] += wl * dx * (1.0 - h[js - lag:je + 1 - lag])
        js2, je2 = max(0, -lag), min(n - 1, n - 1 - lag)
        if js2 <= je2:
            g[js2:je2 + 1] -= wl * dx * h[js2 + lag:je2 + 1 + lag]
    return g

def _slsqp(h0, n, alpha, maxiter):
    bnds = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
            'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_sf, h0, jac=_sf_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=con,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return _repair(h0.copy(), n)

def _proj_grad(h0, n, alpha, max_iter, tc):
    h = h0.copy()
    f_prev = _sf(h, alpha)
    step = 0.1
    for _ in range(max_iter):
        if tc():
            break
        g = _sf_grad(h, alpha)
        gn = np.sqrt(np.sum(g * g))
        if gn < 1e-14:
            break
        for _ in range(40):
            y = h - step * g
            lo, hi = y.min() - n, y.max()
            for _2 in range(60):
                mid = (lo + hi) / 2
                p = np.clip(y - mid, 0.0, 1.0)
                if p.sum() > n / 2.0:
                    lo = mid
                else:
                    hi = mid
            h_new = np.clip(y - (lo + hi) / 2, 0.0, 1.0)
            f_new = _sf(h_new, alpha)
            if f_new <= f_prev - 1e-4 * step * gn:
                if f_prev - f_new < 1e-16:
                    return h
                h = h_new
                f_prev = f_new
                step = min(step * 1.4, 5.0)
                break
            step *= 0.5
    return _repair(h, n)

def _benders(h0, n, tc, max_iter=300):
    h = h0.copy()
    c5 = _eval(h)
    dx = 2.0 / n
    for it in range(max_iter):
        if tc():
            break
        c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
        m = c.max()
        spread = m - c.min()
        thresh = m - spread * 0.08
        active = np.where(c >= thresh)[0]
        if len(active) < 3:
            active = np.argsort(c)[-8:]
        A_ub_rows = []
        b_ub_vals = []
        for idx in active:
            lag = idx - (n - 1)
            grad = np.zeros(n)
            for j in range(n):
                if 0 <= j - lag < n:
                    grad[j] += dx * (1.0 - h[j - lag])
                if 0 <= j + lag < n:
                    grad[j] -= dx * h[j + lag]
            row = np.zeros(n + 1)
            row[:n] = grad
            row[n] = -1.0
            A_ub_rows.append(row)
            b_ub_vals.append(-c[idx])
        A_ub = np.array(A_ub_rows)
        b_ub = np.array(b_ub_vals)
        bounds = [(-h[i], 1.0 - h[i]) for i in range(n)]
        bounds.append((None, None))
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                          bounds=bounds, method='highs')
        except Exception:
            break
        if not res.success:
            break
        delta = res.x[:n]
        improved = False
        for frac in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            h_try = _repair(h + frac * delta, n)
            c5_try = _eval(h_try)
            if c5_try < c5 - 1e-12:
                h = h_try
                c5 = c5_try
                improved = True
                break
        if not improved:
            break
    return h, c5

def _sim_anneal(h0, n, rng, tc, max_iter=80000):
    h = h0.copy()
    c_cur = _eval(h)
    T = 0.004
    best_c = c_cur
    best_h = h.copy()
    for it in range(max_iter):
        if tc():
            break
        r = rng.rand()
        if r < 0.25:
            idx = rng.choice(n, 3, replace=False)
            d1 = rng.uniform(-0.06, 0.06)
            d2 = rng.uniform(-0.06, 0.06)
            moves = [d1, d2, -(d1 + d2)]
        else:
            idx = rng.choice(n, 2, replace=False)
            d = rng.uniform(-0.08, 0.08)
            moves = [d, -d]
        orig = [h[i] for i in idx]
        ok = True
        for k, i in enumerate(idx):
            v = orig[k] + moves[k]
            if v < -0.01 or v > 1.01:
                ok = False
                break
        if not ok:
            continue
        for k, i in enumerate(idx):
            h[i] = orig[k] + moves[k]
        c_new = _eval(h)
        delta = c_new - c_cur
        if delta < 0 or rng.rand() < np.exp(-max(delta, 0) / max(T, 1e-12)):
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            for k, i in enumerate(idx):
                h[i] = orig[k]
        T = max(T * 0.99985, 2e-5)
    return best_h, best_c
