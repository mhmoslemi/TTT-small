def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Proportional scaling then fix
    if s > 1e-12:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _slsqp_opt(h, n, dx, alpha, maxiter=500):

    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s > 1e-12:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _eq_fun(x, target):
    return np.sum(x) - target

def _slsqp_opt(h, n, dx, alpha, maxiter=500):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': _eq_fun, 'args': (target,)}
    try:
        res = minimize(_smooth_obj, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=[con],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(res.x, n)
        c5 = _compute_c5(hc, dx)
        return hc, c5
    except Exception:
        return h, _compute_c5(h, dx)

def _lbfgs_opt(h, n, dx, alpha, pen, maxiter=300):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj_pen(x):
        return _smooth_obj(x, alpha) + pen * (np.sum(x) - target) ** 2
    try:
        res = minimize(obj_pen, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-7})
        hc = _repair(res.x, n)
        c5 = _compute_c5(hc, dx)
        return hc, c5
    except Exception:
        return h, _compute_c5(h, dx)

def _binary_sa(h_init, n, dx, rng, deadline, T_start=0.006, T_end=0.000005):
    target_ones = n // 2
    h_cur = np.where(h_init > 0.5, 1.0, 0.0).astype(float)
    ones_idx = np.where(h_cur == 1.0)[0]
    zeros_idx = np.where(h_cur == 0.0)[0]
    if len(ones_idx) > target_ones:
        h_cur[ones_idx[target_ones:]] = 0.0
        zeros_idx = np.where(h_cur == 0.0)[0]
        ones_idx = np.where(h_cur == 1.0)[0]
    elif len(ones_idx) < target_ones:
        need = target_ones - len(ones_idx)
        h_cur[zeros_idx[:need]] = 1.0
        zeros_idx = np.where(h_cur == 0.0)[0]
        ones_idx = np.where(h_cur == 1.0)[0]
    c5_cur = _compute_c5(h_cur, dx)
    c5_best = c5_cur
    h_best = h_cur.copy()
    T = T_start
    n_steps = int(0.35 / 0.0000005) if (T_start - T_end) > 0 else 1000000
    n_steps = int(np.log(T_start / T_end) / np.log(T_start / T_end * (1 - 1.0 / n_steps))) if T_start > T_end else 100000
    n_steps = 500000
    cool = (T_end / T_start) ** (1.0 / n_steps)
    for step in range(n_steps):
        if time.time() >= deadline:
            break
        if step % 2 == 0:
            i1 = rng.randint(len(ones_idx))
            i2 = rng.randint(len(zeros_idx))
            p1, p2 = ones_idx[i1], zeros_idx[i2]
            h_cur[p1] = 0.0
            h_cur[p2] = 1.0
            c5_new = _compute_c5(h_cur, dx)
            d = c5_new - c5_cur
            if d < 0 or rng.random() < np.exp(-d / T):
                c5_cur = c5_new
                ones_idx[i1] = p2
                zeros_idx[i2] = p1
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_cur.copy()
            else:
                h_cur[p1] = 1.0
                h_cur[p2] = 0.0
        else:
            if len(ones_idx) >= 2 and len(zeros_idx) >= 2:
                i1, i2 = rng.randint(len(ones_idx)), rng.randint(len(ones_idx))
                j1, j2 = rng.randint(len(zeros_idx)), rng.randint(len(zeros_idx))
                if i1 == i2 or j1 == j2:
                    T *= cool
                    continue
                a, b = ones_idx[i1], ones_idx[i2]
                c, d2 = zeros_idx[j1], zeros_idx[j2]
                h_cur[a] = 0.0
                h_cur[b] = 0.0
                h_cur[c] = 1.0
                h_cur[d2] = 1.0
                c5_new = _compute_c5(h_cur, dx)
                delta = c5_new - c5_cur
                if delta < 0 or rng.random() < np.exp(-delta / T):
                    c5_cur = c5_new
                    ones_idx[i1] = c
                    ones_idx[i2] = d2
                    zeros_idx[j1] = a
                    zeros_idx[j2] = b
                    if c5_new < c5_best:
                        c5_best = c5_new
                        h_best = h_cur.copy()
                else:
                    h_cur[a] = 1.0
                    h_cur[b] = 1.0
                    h_cur[c] = 0.0
                    h_cur[d2] = 0.0
        T *= cool
    return h_best, c5_best

def _fine_local(h, n, dx, rng, deadline, scales=None):
    if scales is None:
        scales = [0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]
    best_c5 = _compute_c5(h, dx)
    h = h.copy()
    for scale in scales:
        improved = False
        count = 0
        max_count = 30000
        while count < max_count and time.time() < deadline:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                count += 1
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                count += 1
                continue
            h[i], h[j] = ni, nj
            c5 = _compute_c5(h, dx)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                improved = True
            else:
                h[i], h[j] = ni - d, nj + d
            count += 1
        if not improved and scale < 0.001:
            break
    return h, best_c5

def _triple_local(h, n, dx, rng, deadline):
    best_c5 = _compute_c5(h, dx)
    h = h.copy()
    for scale in [0.008, 0.004, 0.002, 0.001]:
        for _ in range(15000):
            if time.time() >= deadline:
                return h, best_c5
            i, j, k = rng.randint(n), rng.randint(n), rng.randint(n)
            if i == j or j == k or i == k:
                continue
            d1 = scale * (2.0 * rng.random() - 1.0)
            d2 = scale * (2.0 * rng.random() - 1.0)
            d3 = -(d1 + d2)
            ni, nj, nk = h[i] + d1, h[j] + d2, h[k] + d3
            if ni < 0 or ni > 1 or nj < 0 or nj > 1 or nk < 0 or nk > 1:
                continue
            h[i], h[j], h[k] = ni, nj, nk
            c5 = _compute_c5(h, dx)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk - d3
    return h, best_c5

def _block_swap_local(h, n, dx, rng, deadline, n_iters=5000):
    best_c5 = _compute_c5(h, dx)
    h = h.copy()
    for bs in [3, 2]:
        for _ in range(n_iters // 2):
            if time.time() >= deadline:
                return h, best_c5
            start1 = rng.randint(0, n - bs + 1)
            start2 = rng.randint(0, n - bs + 1)
            if abs(start1 - start2) < bs:
                continue
            eps = rng.uniform(0.003, 0.02) * (1.0 if rng.random() < 0.5 else -1.0)
            ok = True
            for di in range(bs):
                if h[start1 + di] + eps < 0 or h[start1 + di] + eps > 1:
                    ok = False
                if h[start2 + di] - eps < 0 or h[start2 + di] - eps > 1:
                    ok = False
            if not ok:
                continue
            h[start1:start1 + bs] += eps
            h[start2:start2 + bs] -= eps
            c5 = _compute_c5(h, dx)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
            else:
                h[start1:start1 + bs] -= eps
                h[start2:start2 + bs] += eps
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h0.copy(), _compute_c5(h0, dx)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP polish (40s)
    if rem() > 120:
        for alpha in [2000, 8000]:
            h_t, c5_t = _slsqp_opt(best_h, n, dx, alpha, maxiter=400)
            if c5_t < best_c5:
                best_h, best_c5 = h_t, c5_t

    # Phase 2: Binary SA - main search (500s)
    if rem() > 600:
        sa_deadline = time.time() + min(480, rem() - 550)
        # First: SA from current solution's binary projection
        h_b, c5_b = _binary_sa(best_h, n, dx, rng, sa_deadline)
        if c5_b < best_c5:
            # Refine the binary solution
            h_t, c5_t = _slsqp_opt(h_b, n, dx, 5000, maxiter=400)
            if c5_t < c5_b:
                h_b, c5_b = h_t, c5_t
            best_h, best_c5 = h_b, c5_b

        # Second: SA from random binary
        if rem() > 300:
            h_rand = np.zeros(n)
            h_rand[rng.choice(n, n // 2, replace=False)] = 1.0
            sa_deadline2 = time.time() + min(200, rem() - 280)
            h_b2, c5_b2 = _binary_sa(h_rand, n, dx, rng, sa_deadline2, T_start=0.01)
            if c5_b2 < best_c5:
                h_t, c5_t = _slsqp_opt(h_b2, n, dx, 5000, maxiter=400)
                if c5_t < c5_b2:
                    h_b2, c5_b2 = h_t, c5_t
                best_h, best_c5 = h_b2, c5_b2

        # Third: SA from periodic pattern
        if rem() > 250:
            h_per = np.zeros(n)
            period = int(np.ceil(n / 2.3))
            for i in range(n):
                if (i % period) < period // 2:
                    h_per[i] = 1.0
            h_per = _repair(h_per, n)
            sa_deadline3 = time.time() + min(150, rem() - 230)
            h_b3, c5_b3 = _binary_sa(h_per, n, dx, rng, sa_deadline3, T_start=0.008)
            if c5_b3 < best_c5:
                h_t, c5_t = _slsqp_opt(h_b3, n, dx, 8000, maxiter=300)
                if c5_t < c5_b3:
                    h_b3, c5_b3 = h_t, c5_t
                best_h, best_c5 = h_b3, c5_b3

    # Phase 3: L-BFGS + SLSQP refinement cycle (100s)
    if rem() > 180:
        for alpha in [3000, 10000, 30000]:
            if rem() < 100:
                break
            h_t, c5_t = _slsqp_opt(best_h, n, dx, alpha, maxiter=500)
            if c5_t < best_c5:
                best_h, best_c5 = h_t, c5_t

    # Phase 4: Block swaps (60s)
    if rem() > 130:
        t_d = time.time() + min(50, rem() - 110)
        h_t, c5_t = _block_swap_local(best_h, n, dx, rng, t_d, n_iters=8000)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 5: Fine local search (80s)
    if rem() > 100:
        t_d = time.time() + min(70, rem() - 80)
        h_t, c5_t = _fine_local(best_h, n, dx, rng, t_d)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 6: Triple-point local (50s)
    if rem() > 80:
        t_d = time.time() + min(40, rem() - 60)
        h_t, c5_t = _triple_local(best_h, n, dx, rng, t_d)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 7: Final SLSQP with very high alpha (40s)
    if rem() > 50:
        h_t, c5_t = _slsqp_opt(best_h, n, dx, 50000, maxiter=400)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 8: Ultra-fine local (30s)
    if rem() > 25:
        t_d = time.time() + max(3, min(20, rem() - 5))
        h_t, c5_t = _fine_local(best_h, n, dx, rng, t_d,
                                scales=[0.0008, 0.0004, 0.0002, 0.0001, 0.00005])
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 9: Final mini-SLSQP
    if rem() > 15:
        h_t, c5_t = _slsqp_opt(best_h, n, dx, 100000, maxiter=300)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    best_h = _repair(best_h, n)
    best_c5 = _compute_c5(best_h, dx)
    return (best_h, best_c5, n)
