import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _smo(h, a, tgt):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a + 1e4 * (h.sum() - tgt) ** 2

def _fix(h, n):
    h = np.clip(h, 0., 1.)
    t = n / 2.
    s = h.sum()
    if s < 1e-14: return np.full(n, .5)
    h *= t / s
    h = np.clip(h, 0., 1.)
    d = t - h.sum()
    if abs(d) > 1e-14:
        o = np.argsort(1. - h) if d > 0 else np.argsort(h)
        for i in o:
            if d > 0:
                a = min(d, 1. - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-14: break
    return h

def _bfgs(h, n, dx, a, tgt, mi=200):
    bh, bc = h.copy(), _c5(h, dx)
    try:
        r = minimize(lambda x: _smo(x, a, tgt), h, method='L-BFGS-B',
                    bounds=[(0., 1.)]*n, options={'maxiter': mi, 'ftol': 1e-18, 'gtol': 1e-9})
        hc = _fix(r.x, n); cv = _c5(hc, dx)
        if cv < bc: bh, bc = hc, cv
    except: pass
    return bh, bc

def _ls(h, n, dx, rng, dl, ns=40000):
    bh, bc = h.copy(), _c5(h, dx)
    st = 0
    for sc in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5]:
        imp = False
        for _ in range(ns // 9):
            if time.time() >= dl: return bh, bc
            st += 1
            i, j = rng.randint(n), rng.randint(n)
            if i == j: continue
            d = sc * (2 * rng.random() - 1)
            ni, nj = bh[i] + d, bh[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue
            bh[i], bh[j] = ni, nj
            cv = _c5(bh, dx)
            if cv < bc - 1e-15: bc, imp = cv, True
            else: bh[i], bh[j] = ni - d, nj + d

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _smo_obj(x, alpha, tgt):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + 1e4 * (x.sum() - tgt)**2

def _fix(h, n):
    h = np.clip(h, 0., 1.)
    t = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0., 1.)
    d = t - h.sum()
    if abs(d) > 1e-14:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-14:
                break
    return h

def _bfgs(h, n, dx, alpha, tgt, mi=250):
    bh, bc = h.copy(), _c5(h, dx)
    try:
        r = minimize(lambda x: _smo_obj(x, alpha, tgt), h, method='L-BFGS-B',
                    bounds=[(0., 1.)] * n,
                    options={'maxiter': mi, 'ftol': 1e-18, 'gtol': 1e-9})
        hc = _fix(r.x, n)
        cv = _c5(hc, dx)
        if cv < bc:
            bh, bc = hc, cv
    except Exception:
        pass
    return bh, bc

def _local(h, n, dx, rng, dl, total=60000):
    bh, bc = h.copy(), _c5(h, dx)
    scales = [0.008, 0.004, 0.002, 0.001, 0.0005, 0.00025, 0.0001, 5e-5, 2e-5, 1e-5]
    per = max(50, total // len(scales))
    for sc in scales:
        improved = False
        for _ in range(per):
            if time.time() >= dl:
                return bh, bc
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            ni, nj = bh[i] + d, bh[j] - d
            if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
                continue
            bh[i], bh[j] = ni, nj
            cv = _c5(bh, dx)
            if cv < bc - 1e-15:
                bc = cv
                improved = True
            else:
                bh[i], bh[j] = ni - d, nj + d
        if not improved and sc < 3e-4:
            break
    return bh, bc

def _block(h, n, dx, rng, dl, iters=4000):
    bh, bc = h.copy(), _c5(h, dx)
    for bs in [2, 3]:
        for _ in range(iters // 2):
            if time.time() >= dl:
                return bh, bc
            i = rng.randint(0, n - bs + 1)
            j = rng.randint(0, n - bs + 1)
            if abs(i - j) < bs:
                continue
            eps = rng.uniform(0.001, 0.008) * (1.0 if rng.random() < 0.5 else -1.0)
            ok = True
            for di in range(bs):
                if not (0 <= bh[i+di]+eps <= 1 and 0 <= bh[j+di]-eps <= 1):
                    ok = False
                    break
            if not ok:
                continue
            bh[i:i+bs] += eps
            bh[j:j+bs] -= eps
            cv = _c5(bh, dx)
            if cv < bc - 1e-15:
                bc = cv
            else:
                bh[i:i+bs] -= eps
                bh[j:j+bs] += eps
    return bh, bc

def _lag_targeted(h, n, dx, rng, dl, iters=20000):
    bh, bc = h.copy(), _c5(h, dx)
    for _ in range(iters):
        if time.time() >= dl:
            return bh, bc
        corr = np.correlate(bh, 1.0 - bh, mode='full')
        top = np.argsort(corr)[-3:]
        k_idx = top[rng.randint(len(top))]
        k = abs(k_idx - (n - 1))
        if k == 0 or k >= n:
            i, j = rng.randint(n), rng.randint(n)
        else:
            i = rng.randint(n - k)
            j = i + k
            if rng.random() < 0.5:
                i, j = j, i
        d = rng.uniform(0.0005, 0.005) * (1 if rng.random() < 0.5 else -1)
        ni, nj = bh[i] + d, bh[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        bh[i], bh[j] = ni, nj
        cv = _c5(bh, dx)
        if cv < bc - 1e-15:
            bc = cv
        else:
            bh[i], bh[j] = ni - d, nj + d
    return bh, bc

def _perturb(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        hp += rng.uniform(-0.06, 0.06, n)
    elif kind == 1:
        f = rng.randint(1, 7)
        ph = rng.uniform(0, 2 * np.pi)
        hp += 0.06 * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 2:
        j = rng.randint(n // 5, 4 * n // 5)
        hp = np.concatenate([hp[j:], hp[:j]])
    elif kind == 3:
        f1, f2 = rng.randint(1, 5), rng.randint(1, 5)
        p1, p2 = rng.uniform(0, 6.28), rng.uniform(0, 6.28)
        t = np.arange(n) / n
        hp += 0.05 * np.sin(2*np.pi*f1*t + p1) * np.sin(2*np.pi*f2*t + p2)
    else:
        hp[:] = 0.5 + 0.2 * rng.randn(n)
    return _fix(hp, n)
