                diff -= add

                diff += sub

import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(-h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    kk = np.arange(-(n-1), n)
    g = np.zeros(2*n-1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _bisect(corr, gd, dlo, dhi, n_it=42):
    lo, hi = dlo, dhi
    for _ in range(n_it):
        mid = 0.5*(lo+hi)
        if np.max(corr + mid*gd) <= np.max(corr + hi*gd):
            hi = mid
        else:
            lo = mid
    return 0.5*(lo+hi)

    order = np.random.permutation(n)
    for ai in range(n-1):

            dhi = min(1.0-hi, hj)

            gd = _g_vec(h, i, n) - _g_vec(h, j, n)

            h[j] -= d_best

**New idea: Use the sub-gradient of the max to guide a projected sub-gradient method.**

The sub-gradient of f(h) = max_k c_k(h) at a point h (where c_k is differentiable) is:
∇f(h) = (1/|A|) * sum_{k∈A} ∇c_k(h)

where A = {k : c_k(h) = max}.

The projected sub-gradient step would be:
h_new = h - alpha * ∇f(h)
then project onto {0≤h≤1, sum(h)=target}

The projection onto {0≤h≤1} ∩ {sum=h_target} is:
- First clip to [0,1]
- Then adjust to meet the sum constraint (similar to _repair)

But actually, since c_k is bilinear (not linear), the "gradient" depends on h:
∂c_k/∂h[i] = sum of (1-h[j]) for all j where h[i] appears as h[i] in c_k
             - sum of h[j] for all j where h[i] appears as 1-h[i] in c_k

This is exactly what _g_vec computes (with sign).

def _subgrad_step(h, n, alpha, target):
    corr = _corr(h)
    mx = np.max(corr)
    # Active set: correlations within 1e-6 of max
    active = np.where(corr >= mx - 1e-6 * n)[0]
    # Compute sub-gradient
    # c_k = sum_i h[i]*(1-h[i+k])
    # ∂c_k/∂h[i] involves all j where i appears
    # Use _g_vec: for changing h[i] by +1, the change in c_k is g_vec[i]_k
    # Actually _g_vec(h, i, n) gives the vector of ∂c_k/∂h[i] for all k
    # So the sub-gradient at h[i] is sum over active k of the k-th component
    # But we need the full gradient vector: for each i, sum over active k of ∂c_k/∂h[i]
    # This is: for each i, sum_{k active} g_vec(h, i, n)[k]
    # = sum_{k active} g_vec(h, i, n)[k]
    
    # More efficiently: the sub-gradient is sum_{k in active} of the k-th row of the "Jacobian"
    # Let G[i,k] = ∂c_k/∂h[i]
    # We need sg[i] = sum_{k in active} G[i,k]
    
    # G[i,k] for c_k = sum_j h[j]*(1-h[j+k]):
    # h[i] appears as h[j] when j=i: contribution to c_k is (1-h[i+k]) if 0≤i+k<n
    # h[i] appears as h[j+k] when j = i-k: contribution to c_k is -h[i-k] if 0≤i-k<n
    # So G[i,k] = (1-h[i+k])*[0≤i+k<n] - h[i-k]*[0≤i-k<n]
    
    # This is O(n^2) to compute for all i,k but we only need the sum over active k.
    sg = np.zeros(n)
    for i in range(n):
        for k_idx in active:
            k = int(k_idx) - (n-1)  # convert to lag
            # G[i, k_idx] = (1-h[i+k])*[0≤i+k<n] - h[i-k]*[0≤i-k<n]
            val = 0.0
            if 0 <= i+k < n:
                val += (1.0 - h[i+k])
            if 0 <= i-k < n:
                val -= h[i-k]
            sg[i] += val
    # Projected gradient step
    h_new = h - alpha * sg
    h_new = _repair(h_new, n)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _bisect(corr, gd, dlo, dhi, n_it=48):
    lo, hi = dlo, dhi
    for _ in range(n_it):
        mid = 0.5 * (lo + hi)
        if np.max(corr + mid * gd) <= np.max(corr + hi * gd):
            hi = mid
        else:
            lo = mid
    return 0.5 * (lo + hi)

def _sweep(h, n, corr, t_lim, n_coarse=15, n_bis=45):
    improved = False
    order = np.random.permutation(n)
    for ai in range(n - 1):
        for ai2 in range(ai + 1, n):
            if time.time() > t_lim:
                return h, _corr(h), improved
            i, j = int(order[ai]), int(order[ai2])
            hi, hj = h[i], h[j]
            dlo = max(-hi, hj - 1.0)
            dhi = min(1.0 - hi, hj)
            if dhi - dlo < 1e-13:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            cur_mx = float(np.max(corr))
            ds = np.linspace(dlo, dhi, n_coarse)
            vals = np.array([np.max(corr + d * gd) for d in ds])
            bi = int(np.argmin(vals))
            if vals[bi] >= cur_mx - 1e-13:
                continue
            lo_b = max(dlo, ds[max(0, bi - 1)])
            hi_b = min(dhi, ds[min(n_coarse - 1, bi + 1)])
            d_best = _bisect(corr, gd, lo_b, hi_b, n_bis)
            h[i] += d_best
            h[j] -= d_best
            corr = corr + d_best * gd
            improved = True
    return h, _corr(h), improved

def _slsqp(h, n, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 60})
        return _repair(res.x, n)
    except Exception:
        return h

def _fourier_perturb(h, n, eps):
    h_p = h.copy()
    m = np.random.randint(2, n // 2)
    phase = np.random.uniform(0, 2 * np.pi)
    h_p += eps * np.cos(2 * np.pi * m * np.arange(n) / n + phase)
    return _repair(h_p, n)

def _block_swap(h, n, block_size):
    h_p = h.copy()
    i = np.random.randint(0, n - block_size)
    j = np.random.randint(0, n - block_size)
    if abs(i - j) < block_size:
        j = (j + block_size) % (n - block_size)
        if abs(i - j) < block_size:
            return h_p
    h_p[i:i+block_size], h_p[j:j+block_size] = h[j:j+block_size].copy(), h[i:i+block_size].copy()
    return _repair(h_p, n)
