def _project(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2} using Bregman (Euclidean) projection."""
    target = n / 2.0
    s = h.sum()
    if s == 0:
        return np.full(n, target/n)
    # Bisection for lambda
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        ps = proj.sum()
        if ps > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)

        mask = (proj > 1e-14) & (proj < 1-1e-14)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

A tighter upper bound: h[i](1-h[j]) ≤ h[i](1-0) = h[i] or ≤ (1-h[j])*1 = 1-h[j]. The minimum of these gives: h[i](1-h[j]) ≤ min(h[i], 1-h[j]).

This doesn't help directly with optimization since min is non-smooth.
