def _corr_and_grad(h, n, alpha):
    """Compute corr values and gradient of smooth max."""
    corr = np.correlate(h, 1.0 - h, mode='full')  # 2n-1 values
    # Smooth max and its gradient
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    shifted = np.clip(shifted, -500, 500)
    exps = np.exp(shifted)
    weights = exps / np.sum(exps)
    f_val = max_c + np.log(np.sum(exps)) / alpha
    
    # Gradient
    grad = np.zeros(2*n - 1)  # temp, then reshape to n
    # Actually grad should be n-dimensional
    grad_h = np.zeros(n)
    for lag in range(-(n-1), n):
        w = weights[lag + n - 1]
        if w < 1e-15:
            continue
        if lag >= 0:
            # ∂corr_lag/∂h[i] = (1-h[i-lag]) for lag≤i≤n-1
            #                 - h[i+lag] for 0≤i≤n-1-lag
            for i in range(lag, n):
                grad_h[i] += w * (1.0 - h[i-lag])
            for i in range(0, n-lag):
                grad_h[i] += w * (-h[i+lag])
        else:
            l = -lag  # l > 0
            # corr_{-l} = sum_{j=l}^{n-1} h[j-l](1-h[j])
            # ∂/∂h[i]: h[i] appears as h[j-l] when j=i+l, valid i+l≥l, i≤n-1-l → 0≤i≤n-1-l

    return f_val, corr, grad_h

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max_val(corr, alpha):
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_max_grad(h, n, alpha):
    corr = _corr(h)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    exps = np.exp(s)
    w = exps / np.sum(exps)
    val = m + np.log(np.sum(exps)) / alpha
    
    grad = np.zeros(n)
    for lag in range(-(n-1), n):
        wi = w[lag + n - 1]
        if wi < 1e-14:
            continue
        if lag >= 0:
            if lag < n:
                grad[lag:] += wi * (1.0 - h[:n-lag])
                grad[:n-lag] -= wi * h[lag:]
        else:
            l = -lag
            if l < n:
                grad[:n-l] += wi * (1.0 - h[l:])
                grad[l:] -= wi * h[:n-l]
    return val, corr, grad

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # Water-filling projection
    if diff > 0:
        # Need to increase sum: increase smallest values
        order = np.argsort(h)
        for idx in order:
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
            if diff < 1e-12:
                break
    else:
        # Need to decrease sum: decrease largest values
        order = np.argsort(h)
        for idx in order[::-1]:
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
            if diff > -1e-12:
                break
    return h

def _g_vec(h, p, n):
    """Gradient vector for moving mass from j to i (h[i]+=d, h[j]-=d)."""
    kk = np.arange(-(n-1), n)
    g = np.zeros(2*n-1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _pair_sweep(h, n, time_limit, n_coarse=21, n_bisect=30, max_rounds=5):
    corr = _corr(h)
    best_c5 = np.max(corr)
    for rnd in range(max_rounds):
        if time.time() > time_limit:
            break
        improved = False
        order = np.random.permutation(n)
        for ai in range(n):
            i = order[ai]
            for j in order[ai+1:]:
                if time.time() > time_limit:
                    return h, _corr(h)
                d_min = max(-h[i], h[j] - 1.0)
                d_max = min(1.0 - h[i], h[j])
                if d_max - d_min < 1e-12:
                    continue
                gd = _g_vec(h, i, n) - _g_vec(h, j, n)
                # Coarse scan
                ds = np.linspace(d_min, d_max, n_coarse)
                newmaxs = np.array([np.max(corr + d * gd) for d in ds])
                bi = np.argmin(newmaxs)
                if newmaxs[bi] >= best_c5 - 1e-13:
                    continue
                # Bisection
                lo = ds[max(0, bi-1)]
                hi = ds[min(n_coarse-1, bi+1)]
                for _ in range(n_bisect):
                    mid = 0.5*(lo + hi)
                    if np.max(corr + mid*gd) <= np.max(corr + hi*gd):
                        hi = mid
                    else:
                        lo = mid
                d_best = 0.5*(lo + hi)
                h[i] += d_best
                h[j] -= d_best
                corr = corr + d_best * gd
                best_c5 = np.max(corr)
                improved = True
        if not improved:
            break
    return h, _corr(h)

def _grad_descent(h, n, budget_time, alpha_init=200, lr_init=0.002):
    """Smoothed gradient descent with projection."""
    best_h = h.copy()
    best_c5 = np.max(_corr(h))
    alpha = alpha_init
    lr = lr_init
    t_end = time.time() + budget_time
    stall = 0
    while time.time() < t_end:
        val, corr, grad = _smooth_max_grad(h, n, alpha)
        if np.linalg.norm(grad) < 1e-10:
            break
        # Backtracking
        for _ in range(30):
            h_trial = _repair(h - lr * grad, n)
            c5_trial = np.max(_corr(h_trial))
            if c5_trial < best_c5:
                h = h_trial
                if c5_trial < best_c5 - 1e-10:
                    best_h = h.copy()
                    best_c5 = c5_trial
                    stall = 0
                break
            lr *= 0.5
        else:
            stall += 1
            if stall > 20:
                alpha *= 2
                lr = lr_init
                break
        # Slowly increase alpha for sharper max
        alpha *= 1.01
    return best_h, best_c5

def _fourier_init(n, seed):

    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2.0, n, endpoint=False)

        b = rng.uniform(-0.15, 0.15)
        h += a * np.cos(m * np.pi * x) + b * np.sin(m * np.pi * x)
    h = _repair(h, n)
    return h

    h[half:] = h_half[::-1] if n % 2 == 0 else np.append(h_half[::-1], 0.5)

def _qp_iteration(h0, n, t_bound, solver='ECOS'):
    """One SLQP iteration: linearize and solve LP."""
    import cvxpy as cp
    h = cp.Variable(n)
    t = cp.Variable()
    
    corr0 = _corr(h0)  # 2n-1 values, indexed by lag = -(n-1) to (n-1)
    
    constraints = [sum(h) == n/2, h >= 0, h <= 1, t >= 0]
    
    for lag in range(-(n-1), n):
        # Linearize corr_lag at h0
        # corr_lag = sum valid h[j+lag](1-h[j])
        # Gradient at h0:
        grad_lag = np.zeros(n)
        if lag >= 0:
            grad_lag[lag:] += (1.0 - h0[:n-lag])
            grad_lag[:n-lag] -= h0[lag:]
        else:
            l = -lag
            grad_lag[:n-l] += (1.0 - h0[l:])
            grad_lag[l:] -= h0[:n-l]
        
        linear_approx = corr0[lag+n-1] + np.dot(grad_lag, h - h0)
        constraints.append(linear_approx <= t / dx)  # need to pass dx
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=getattr(cp, solver))
    
    if h.value is not None:
        return _repair(np.array(h.value), n), t.value
    return h0, np.inf

        order = np.argsort(h)
        for idx in order[::-1]:
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
            if diff > -1e-12:
                break
    return h

def _grad_single_lag(h0, n, lag):
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            g[lag:] += (1.0 - h0[:n - lag])
            g[:n - lag] -= h0[lag:]
    else:
        l = -lag
        if l < n:
            g[:n - l] += (1.0 - h0[l:])
            g[l:] -= h0[:n - l]
    return g

def _all_gradients(h0, n):
    n_lags = 2 * n - 1
    A = np.zeros((n_lags, n))
    for k in range(n_lags):
        lag = k - (n - 1)
        A[k] = _grad_single_lag(h0, n, lag)
    return A

def _sol_lp(h0, n, dx):
    import cvxpy as cp
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    corr0 = _corr(h0)
    A = _all_gradients(h0, n)
    b_lin = corr0 - A @ h0
    constraints = [
        h_var >= 0, h_var <= 1,
        cp.sum(h_var) == n / 2,
        A @ h_var + b_lin <= t_var / dx,
        t_var >= 0
    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=5000)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK)
        except Exception:
            return h0.copy(), np.inf
    if h_var.value is not None and np.isfinite(t_var.value):
        h_sol = _repair(np.array(h_var.value, dtype=float), n)
        return h_sol, float(t_var.value)
    return h0.copy(), np.inf

def _smooth_grad_descent(h, n, time_limit, alpha=400, lr=0.002):
    best_h = h.copy()
    best_c5 = np.max(_corr(h))
    t_end = time.time() + time_limit
    step = 0
    while time.time() < t_end:
        corr = _corr(h)
        m = np.max(corr)
        s = np.clip(alpha * (corr - m), -500, 500)
        exps = np.exp(s)
        w = exps / np.sum(exps)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            lag = k - (n - 1)
            grad += w[k] * _grad_single_lag(h, n, lag)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        lr_cur = lr
        accepted = False
        for _ in range(40):
            h_new = _repair(h - lr_cur * grad / gn, n)
            c5_new = np.max(_corr(h_new))
            if c5_new < best_c5:
                h = h_new
                accepted = True
                if c5_new < best_c5 - 1e-13:
                    best_h = h.copy()
                    best_c5 = c5_new
                break
            lr_cur *= 0.5
        if not accepted:
            break
        alpha *= 1.015
        step += 1
    return best_h, best_c5

def _pair_sweep(h, n, time_limit, n_coarse=21, n_bisect=32):
    corr = _corr(h)
    order = np.random.permutation(n)
    kk = np.arange(-(n - 1), n)
    for ai in range(n):
        i = order[ai]
        if time.time() > time_limit:
            break
        for j in order[ai + 1:]:
            if time.time() > time_limit:
                break
            d_min = max(-h[i], h[j] - 1.0)
            d_max = min(1.0 - h[i], h[j])
            if d_max - d_min < 1e-12:
                continue
            g = np.zeros(2 * n - 1)
            idx1 = i - kk
            m1 = (idx1 >= 0) & (idx1 < n)
            g[m1] += (1.0 - h[idx1[m1]])
            idx2 = i + kk
            m2 = (idx2 >= 0) & (idx2 < n)
            g[m2] -= h[idx2[m2]]
            idx1j = j - kk
            m1j = (idx1j >= 0) & (idx1j < n)
            g[m1j] -= (1.0 - h[idx1j[m1j]])
            idx2j = j + kk
            m2j = (idx2j >= 0) & (idx2j < n)
            g[m2j] += h[idx2j[m2j]]
            ds = np.linspace(d_min, d_max, n_coarse)
            newmaxs = np.array([np.max(corr + d * g) for d in ds])
            bi = np.argmin(newmaxs)
            if newmaxs[bi] >= np.max(corr) - 1e-13:
                continue
            lo = ds[max(0, bi - 1)]
            hi = ds[min(n_coarse - 1, bi + 1)]
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi)
                if np.max(corr + mid * g) <= np.max(corr + hi * g):
                    hi = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi)
            h[i] += d_best
            h[j] -= d_best
            corr = corr + d_best * g
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: SLQP via LP (main optimization)
    h_cur = h.copy()
    t_lp_end = t0 + budget_s * 0.45
    last_improve = t0
    for it in range(5000):
        if time.time() > t_lp_end:
            break
        h_new, t_lp = _sol_lp(h_cur, n, dx)
        c5_new = _c5(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            last_improve = time.time()
        # Check if we've stagnated
        if time.time() - last_improve > 60:
            # Nudge: perturb and try again
            h_cur = best_h + 0.02 * np.random.randn(n)
            h_cur = _repair(h_cur, n)
        else:
            h_cur = h_new
        # If LP gives same solution, try smaller step
        if np.allclose(h_new, h_cur, atol=1e-10):
            h_cur = _repair(h_cur + 0.01 * np.random.randn(n), n)

    # Phase 2: Gradient descent polish
    if time.time() - t0 < budget_s * 0.6:
        h_gd, c5_gd = _smooth_grad_descent(best_h, n, time_limit=budget_s * 0.1, alpha=500, lr=0.003)
        if c5_gd < best_c5:
            best_c5 = c5_gd
            best_h = h_gd.copy()

    # Phase 3: More LP iterations from improved point
    if time.time() - t0 < budget_s * 0.8:
        h_cur = best_h.copy()
        t_lp2_end = t0 + budget_s * 0.8
        last_imp2 = time.time()
        for it in range(3000):
            if time.time() > t_lp2_end:
                break
            h_new, t_lp = _sol_lp(h_cur, n, dx)
            c5_new = _c5(h_new, dx)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                last_imp2 = time.time()
            if time.time() - last_imp2 > 40:
                h_cur = _repair(best_h + 0.015 * np.random.randn(n), n)
            else:
                h_cur = h_new

    # Phase 4: Final coordinate sweeps
    t_sweep_end = t0 + budget_s - 5
    while time.time() < t_sweep_end:
        chunk = min(40.0, t_sweep_end - time.time())
        h_sw = _pair_sweep(best_h.copy(), n, time.time() + chunk, 25, 35)
        c5_sw = _c5(h_sw, dx)
        if c5_sw < best_c5:
            best_c5 = c5_sw
            best_h = h_sw.copy()
        else:
            # Tryperturbation + sweep
            h_pert = _repair(best_h + 0.01 * np.random.randn(n), n)
            h_sw2 = _pair_sweep(h_pert, n, time.time() + min(25, t_sweep_end - time.time()), 21, 30)
            c5_sw2 = _c5(h_sw2, dx)
            if c5_sw2 < best_c5:
                best_c5 = c5_sw2
                best_h = h_sw2.copy()
            if t_sweep_end - time.time() < 15:
                break

    # Phase 5: Final gradient polish
    if time.time() - t0 < budget_s - 3:
        h_final, c5_final = _smooth_grad_descent(best_h, n, time_limit=min(10, budget_s - 3 - (time.time() - t0)), alpha=800, lr=0.001)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    final_h = _repair(best_h, n)
    final_c5 = _c5(final_h, dx)
    return (final_h, final_c5, n)
