    best_c5 = np.inf
    best_n = 70

    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
        init_n = len(init_h)
        init_h = repair(init_h, init_n)
        best_h = init_h.copy()
        best_c5 = compute_c5(init_h, init_n)
        best_n = init_n
    except NameError:
        n0 = 70
        h0 = 0.5 + 0.3 * np.sin(2 * np.pi * np.arange(n0) / n0) + 0.05 * rng.randn(n0)
        best_h = repair(h0, n0)
        best_c5 = compute_c5(best_h, n0)
        best_n = n0

    n_p = best_n
    h_work = best_h.copy()

    # Phase 1: SLSQP with escalating alpha (30%)
    t1 = t0 + budget_s * 0.30
    for alpha in [1000, 5000, 20000, 80000]:
        if time.time() > t1:
            break
        for topk, fw in [(1, 0.0), (3, 0.005), (5, 0.01)]:
            if time.time() > t1:
                break
            h_new = do_slsqp(h_work.copy(), n_p, alpha, topk, fw, 500)
            c5_new = compute_c5(h_new, n_p)
            if c5_new < best_c5 - 1e-15:
                best_h = h_new.copy()
                best_c5 = c5_new
                h_work = h_new
            elif c5_new < compute_c5(h_work, n_p) - 1e-15:
                h_work = h_new

    # Phase 2: L-BFGS refinement (12%)
    t2 = t0 + budget_s * 0.42
    for alpha, pen in [(20000, 500), (80000, 5000), (200000, 20000)]:
        if time.time() > t2:
            break
        h_new = do_lbfgs(h_work.copy(), n_p, alpha, pen, 3000)
        c5_new = compute_c5(h_new, n_p)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new.copy()
            best_c5 = c5_new
            h_work = h_new

    # Phase 3: Aggressive fine coordinate descent (36%)
    t3 = t0 + budget_s * 0.78
    h_cd, c5_cd = fine_cd(best_h.copy(), best_n, t3, rng)
    if c5_cd < best_c5 - 1e-15:
        best_h = h_cd.copy()
        best_c5 = c5_cd

    # Phase 4: SLSQP polish + more CD (17%)
    t4 = t0 + budget_s * 0.95
    if time.time() < t4:
        h_new = do_slsqp(best_h.copy(), best_n, 100000, 3, 0.01, 600)
        c5_new = compute_c5(h_new, best_n)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new.copy()
            best_c5 = c5_new
        h_cd2, c5_cd2 = fine_cd(best_h.copy(), best_n, t4, rng)
        if c5_cd2 < best_c5 - 1e-15:
            best_h = h_cd2.copy()
            best_c5 = c5_cd2

    # Phase 5: Try alternative n (5%)
    t5 = t0 + budget_s * 0.995
    if time.time() < t5:
        for n_alt in [64, 72, 80]:
            if time.time() > t5:
                break
            if n_alt == best_n:
                continue
            f = rng.randint(2, 5)
            h_alt = 0.5 + 0.25 * np.sin(2 * np.pi * f * np.arange(n_alt) / n_alt + rng.uniform(0, 6.28))
            h_alt = repair(h_alt, n_alt)
            h_alt = do_slsqp(h_alt, n_alt, 20000, 3, 0.005, 300)
            h_alt, c5_alt = fine_cd(h_alt, n_alt, min(time.time() + 20, t5), rng)
            if c5_alt < best_c5 - 1e-15:
                best_h = h_alt.copy()
                best_c5 = c5_alt
                best_n = n_alt

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
