```python
def _grad_violation(h, n, thresh):
    """Gradient of sum_k max(0, corr_k - thresh)^2 w.r.t. h"""
    corr = _corr(h)
    viol = np.maximum(corr - thresh, 0.0)
    active = viol > 0
    grad = np.zeros(n)
    for k in range(2*n-1):
        if not active[k]:
            continue
        lag = n - 1 - k
        w = 2.0 * viol[k]
        # d(corr_k)/d(h[i]) = (1-h[i+lag]) - h[i-lag] for valid indices
        for i in range(n):
            g = 0.0
            if 0 <= i + lag < n:
                g += 1.0 - h[i + lag]
            if 0 <= i - lag < n:
                g -= h[i - lag]
            grad[i] += w * g
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _slsqp_step(h, n, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 50})
        return _repair(res.x, n)
    except Exception:

    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _sweep_round(h, n, corr, time_limit, n_coarse=30, n_bisect=40):
    improved = False
    order = np.random.permutation(n)
    n_pairs = 0
    n_improved = 0
    for ai, i in enumerate(order):
        for j in order[ai + 1:]:
            n_pairs += 1
            if time.time() > time_limit:
                return h, corr, improved, n_pairs, n_improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            range_d = d_max - d_min
            if range_d < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = np.argmin(vals)
            cur_max = np.max(corr)
            if vals[bi] >= cur_max - 1e-13:
                continue
            lo = max(d_min, ds[max(0, bi - 1)])
            hi_b = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            if abs(d_best) < 1e-12:
                continue
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            improved = True
            n_improved += 1
    return h, corr, improved, n_pairs, n_improved

def _gradient_descent(h, n, dx, thresh, lr_init, time_limit, max_steps=2000):

    h = h.copy()
    lr = lr_init
    for step in range(max_steps):
        if time.time() > time_limit:
            break
        corr = _corr(h)
        viol = np.maximum(corr - thresh, 0.0)
        if viol.max() < 1e-12:
            break
        # Compute gradient of sum(viol^2)
        w = 2.0 * viol
        grad = np.zeros(n)
        for m in range(-(n-1), n):
            wv = w[n-1+m]
            if wv < 1e-15:
                continue
            # d(corr_{n-1+m})/d(h[i]) = (1-h[i+m]) - h[i-m] (valid indices)
            if m >= 0:
                # (1-h[i+m]) valid for i=0..n-1-m
                grad[:n-m] += wv * (1.0 - h[m:])
                # -h[i-m] valid for i=m..n-1
                grad[m:] -= wv * h[:n-m]
            else:
                mm = -m
                # (1-h[i+m]) = (1-h[i-mm]) valid for i=mm..n-1
                grad[mm:] += wv * (1.0 - h[:n-mm])
                # -h[i-m] = -h[i+mm] valid for i=0..n-1-mm
                grad[:n-mm] -= wv * h[mm:]
        h_new = h - lr * grad
        h_new = _repair(h_new, n)
        # Line search: reduce lr if it didn't help
        corr_new = _corr(h_new)
        if np.sum(np.maximum(corr_new - thresh, 0)**2) > np.sum(viol**2) * 1.01:
            lr *= 0.7
            h = h  # don't update
        else:
            h = h_new
            if lr < 0.5 * lr_init:
                lr *= 1.1  # slowly increase back
    return h
```

This is a 1D problem: minimize max of (2n-1) linear functions of δ. The optimal δ is at a breakpoint or at the boundary.

This is exactly what the pair sweep does but with δ constrained to pairs. A single-coordinate version would be:
- For each i, find the optimal h[i] given all others
- This is a 1D minimax of linear functions, solvable in O(n log n)

This is a standard "minimax of L1-convex function" problem. The function f(δ) = max_k (a_k + b_k * δ) is convex piecewise-linear, and its minimum is at a point where two or more lines cross (or at a boundary).

For n=70, we have 139 lines. We can find the minimum by:
1. Evaluate f at δ=0
2. Find the active set (lines achieving max at δ=0)
3. Find the δ where the max is minimized (could be where two active lines cross, or where an active line becomes 0)
4. Binary search or direct computation

Actually, the simplest approach: the minimum of max of linear functions occurs at a "vertex" of the lower envelope, which is where two lines cross at their maximum. We can find this by:
- For each pair of lines (k1, k2), find the crossing point δ* = (a2-a1)/(b1-b2)
- Check if δ* is in [0,1] and if at δ*, both lines have the same value and no other line is higher
- The minimum is the lowest such value

But checking all pairs is O(n²) per coordinate, which is O(n³) total. For n=70, that's 343K per full sweep... manageable.

Actually, a simpler approach: binary search on the value V. For a given V, is there a δ such that for all k: a_k + b_k*δ ≤ V? This gives:
- For b_k > 0: δ ≤ (V - a_k)/b_k
- For b_k < 0: δ ≥ (V - a_k)/b_k  
- For b_k = 0: require a_k ≤ V

So the feasible δ interval is [max_{b_k<0}(V-a_k)/b_k, min_{b_k>0}(V-a_k)/b_k] intersected with [0, 1]. If this interval is non-empty, V is feasible.

Binary search on V gives the minimum in O(n log n * log(precision)) per coordinate. For 70 coordinates and ~40 bisection steps, that's 70*40*139 ≈ 390K operations per full sweep. Very fast!

```python
def _minimize_plines(corr, g, delta_lo, delta_hi, n_iter=50):
    """Minimize max_k (corr[k] + delta * g[k]) over delta in [delta_lo, delta_hi]"""
    lo, hi = delta_lo, delta_hi
    for _ in range(n_iter):
        mid1 = lo + (hi - lo) / 3
        mid2 = hi - (hi - lo) / 3
        f1 = np.max(corr + mid1 * g)
        f2 = np.max(corr + mid2 * g)
        if f1 < f2:
            hi = mid2
        else:
            lo = mid1
    return 0.5 * (lo + hi)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _slsqp_step(h, n, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return h

def _g_vec(h, p, n):
    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _precompute_g(h, n):
    gs = np.zeros((n, 2 * n - 1))
    kk = np.arange(-(n - 1), n)
    for i in range(n):
        idx1 = i - kk
        m1 = (idx1 >= 0) & (idx1 < n)
        gs[i, m1] += (1.0 - h[idx1[m1]])
        idx2 = i + kk
        m2 = (idx2 >= 0) & (idx2 < n)
        gs[i, m2] -= h[idx2[m2]]
    return gs

def _pair_sweep(h, n, g_all, corr, time_limit, n_coarse=30, n_bisect=40):
    improved = False
    order = np.random.permutation(n)
    n_pairs = 0
    for ai, i in enumerate(order):
        for j in order[ai + 1:]:
            n_pairs += 1
            if time.time() > time_limit:
                return h, g_all, corr, improved
            d_min = max(-h[i], h[j] - 1.0)
            d_max = min(1.0 - h[i], h[j])
            if d_max - d_min < 1e-12:
                continue
            gd = g_all[i] - g_all[j]
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = np.argmin(vals)
            if vals[bi] >= np.max(corr) - 1e-13:
                continue
            lo = max(d_min, ds[max(0, bi - 1)])
            hi_b = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            if abs(d_best) < 1e-13:
                continue
            h[i] += d_best
            h[j] -= d_best
            corr += d_best * gd
            g_all[i] = _g_vec(h, i, n)
            g_all[j] = _g_vec(h, j, n)
            improved = True
    return h, g_all, corr, improved

    order = np.random.permutation(n)

        if time.time() > time_limit:
            break
        if h[i] < 1e-12 or h[i] > 1.0 - 1e-12:

        d_min, d_max = -h[i], 1.0 - h[i]

        gd[i] = 0.0
        lo, hi = d_min, d_max
        for _ in range(n_bisect):
            m1 = lo + (hi - lo) / 3.0

        h[i] += d_opt
        corr = _corr(h)
        g_all[i] = _g_vec(h, i, n)
        improved = True
    return h, g_all, corr, improved

def _grad_desc(h, n, thresh, lr, time_limit, max_steps=3000):
    h = h.copy()
    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        corr = _corr(h)
        viol = np.maximum(corr - thresh, 0.0)
        if viol.max() < 1e-11:
            break
        w = 2.0 * viol
        grad = np.zeros(n)
        for m in range(-(n - 1), n):
            wv = w[n - 1 + m]
            if wv < 1e-14:
                continue
            if m >= 0:
                grad[m:] += wv * (1.0 - h[:n - m])
                grad[:n - m] -= wv * h[m:]
            else:
                mm = -m
                grad[:n - mm] += wv * (1.0 - h[mm:])
                grad[mm:] -= wv * h[:n - mm]
        old_vio = viol.sum() ** 2
        h_new = _repair(h - lr * grad, n)
        new_vio = np.sum(np.maximum(_corr(h_new) - thresh, 0.0) ** 2)
        if new_vio < old_vio:
            h = h_new
            lr = min(lr * 1.03, 0.05)
        else:
            lr *= 0.6
        if lr < 1e-6:
            lr = 0.01
    return h

            hc = _slsqp_step(best_h, best_n, best_n / 2.0, bf, 8000, 800)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc.copy()
        except Exception:
            pass

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, best_n)
```
