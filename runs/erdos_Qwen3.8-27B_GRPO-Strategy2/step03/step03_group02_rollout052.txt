    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    # softmax weights
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    w = np.exp(s) / np.sum(np.exp(s))
    
    # For each lag, compute g_lag
    # lag = k - (n-1) where k indexes into corr (full mode)
    # corr index k corresponds to lag = k - (n-1)
    # g_lag[p] = (1-h[p+lag])*[valid] - h[p-lag]*[valid]
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-10:
            continue
        lag = k - (n-1)
        # First term: (1-h[p+lag]) for valid p+lag
        if lag >= 0:
            # p+lag < n => p < n-lag
            valid_p = range(n - lag)
            grad[valid_p] += w[k] * (1.0 - h[lag:n])
        else:

            la = -lag  # positive
            valid_p = range(la, n)

import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _grad_corr_weighted(h, w):
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[:n - lag] += wk * (1.0 - h[lag:])
            if lag < n:
                grad[lag:] -= wk * h[:n - lag]
        else:
            la = -lag
            if la < n:
                grad[la:] += wk * (1.0 - h[:n - la])
            if n - la > 0:
                grad[:n - la] -= wk * h[la:]
    return grad

def _grad_smooth_max(h, alpha):
    n = len(h)
    corr = _corr(h)
    m = float(np.max(corr))
    s = np.clip(alpha * (corr - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    grad = _grad_corr_weighted(h, w)
    return grad, corr, m

def _pgd_step(h, n, dx, alpha, max_linesearch=30):
    """One projected gradient descent step."""
    grad, corr, m = _grad_smooth_max(h, alpha)
    f_curr = m * dx
    step_size = 1.0
    for _ in range(max_linesearch):
        h_trial = h - step_size * grad
        h_trial = _repair(h_trial, n)
        f_trial = _c5(h_trial, dx)
        if f_trial < f_curr - 1e-15:
            return h_trial
        step_size *= 0.5
    return h

def _sweep_round(h, n, time_limit, n_coarse=20, n_bisect=30):
    corr = _corr(h)
    improved = False
    n_improved = 0
    order = np.random.permutation(n)
    for ai in range(n):
        i = order[ai]
        for j in order[ai + 1:]:
            if time.time() > time_limit:
                return h, corr, improved, n_improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-14:
                continue
            # Gradient of corr w.r.t. (h[i]+d, h[j]-d)
            # corr_new = corr + d * (g_i - g_j)
            # g_i[p] = ∂corr/∂h[i] vector... 
            # Actually, changing h[i] by +d and h[j] by -d:
            # Δcorr[k] = d * (∂corr[k]/∂h[i] - ∂corr[k]/∂h[j])
            
            # Compute g_i and g_j
            gi = np.zeros(2*n-1)  # ∂corr[k]/∂h[i] for all k
            gj = np.zeros(2*n-1)
            for k in range(2*n-1):
                lag = k - (n-1)
                # ∂corr[k]/∂h[i]:
                gi[k] = 0.0
                if 0 <= i + lag < n:
                    gi[k] += 1.0 - h[i + lag]
                if 0 <= i - lag < n:
                    gi[k] -= h[i - lag]
                # ∂corr[k]/∂h[j]:
                gj[k] = 0.0
                if 0 <= j + lag < n:
                    gj[k] += 1.0 - h[j + lag]
                if 0 <= j - lag < n:
                    gj[k] -= h[j - lag]
            
            gd = gi - gj
            cur_max = np.max(corr)
            
            # Coarse search
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.empty(n_coarse)
            for ii in range(n_coarse):
                vals[ii] = np.max(corr + ds[ii] * gd)
            bi = np.argmin(vals)
            if vals[bi] >= cur_max - 1e-13:
                continue
            
            # Bisection
            lo = max(d_min, ds[max(0, bi - 1)])
            hi_b = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            improved = True
            n_improved += 1
    return h, corr, improved, n_improved
```

Actually, for the sweep round, the key operation for each pair (i,j) is:
1. Compute gd[k] = ∂corr[k]/∂h[i] - ∂corr[k]/∂h[j] for all k
2. Find d that minimizes max(corr + d*gd)

For step 1, I need ∂corr[k]/∂h[i] for all k. This depends on h. Specifically:
∂corr[k]/∂h[i] = (1-h[i+lag_k])*[valid] - h[i-lag_k]*[valid]

where lag_k = k - (n-1).

So for fixed i, as k varies, lag_k varies from -(n-1) to (n-1).
(1-h[i+lag]) is: for lag = -(n-1), ..., n-1
- Valid when 0 ≤ i+lag < n, i.e., -i ≤ lag < n-i
- h[i+lag] = h[i+lag] (just an array lookup)

Similarly for h[i-lag]:
- Valid when 0 ≤ i-lag < n, i.e., -n+i < lag ≤ i
- h[i-lag] = h[i-lag]

So for fixed i:
g_i[k] = (1-h[i+lag_k]) * [0≤i+lag_k<n] - h[i-lag_k] * [0≤i-lag_k<n]

def _g_row(h, i, n):
    lags = np.arange(-(n-1), n)  # length 2n-1
    gi = np.zeros(2*n-1)
    # First term: (1-h[i+lag]) where 0≤i+lag<n
    idx = i + lags
    m1 = (idx >= 0) & (idx < n)
    gi[m1] += 1.0 - h[idx[m1]]
    # Second term: -h[i-lag] where 0≤i-lag<n
    idx2 = i - lags
    m2 = (idx2 >= 0) & (idx2 < n)
    gi[m2] -= h[idx2[m2]]
    return gi
```

import numpy as np
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _g_row(h, i, n):
    lags = np.arange(-(n-1), n)
    gi = np.zeros(2*n-1)
    idx = i + lags
    m1 = (idx >= 0) & (idx < n)
    gi[m1] += 1.0 - h[idx[m1]]
    idx2 = i - lags
    m2 = (idx2 >= 0) & (idx2 < n)
    gi[m2] -= h[idx2[m2]]
    return gi

def _grad_c5(h, alpha):
    n = len(h)
    corr = _corr(h)
    m = float(np.max(corr))
    s = np.clip(alpha * (corr - m), -500, 500)
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(2*n-1)
        w[np.argmax(corr)] = 1.0
    else:
        w /= ws
    # grad = sum_k w[k] * g_row(h, ..., k)
    # But g_row gives d(corr[k])/d(h[i]) for fixed i
    # We need: grad[i] = sum_k w[k] * d(corr[k])/d(h[i])
    # = sum_k w[k] * g_i[k]
    # where g_i[k] = d(corr[k])/d(h[i])
    # This is: for each i, dot(w, _g_row(h, i, n))
    # But _g_row(h, i, n) gives d(corr)/d(h[i]) as a function of k
    # So grad[i] = dot(w, _g_row(h, i, n))
    # Wait no. _g_row(h, i, n) gives for each k, d(corr[k])/d(h[i]).
    # The gradient of f(h) = sum_k w[k] * corr[k] w.r.t. h[i] is:
    # grad[i] = sum_k w[k] * d(corr[k])/d(h[i]) = dot(w, g_i)
    # where g_i = _g_row(h, i, n) (length 2n-1)
    # So grad = [dot(w, _g_row(h, i, n)) for i in range(n)]
    
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)
    for i in range(n):
        gi = np.zeros(2*n-1)
        idx = i + lags
        m1 = (idx >= 0) & (idx < n)
        gi[m1] += 1.0 - h[idx[m1]]
        idx2 = i - lags
        m2 = (idx2 >= 0) & (idx2 < n)
        gi[m2] -= h[idx2[m2]]
        grad[i] = np.dot(w, gi)
    return grad, corr, m

def _optimize_multi(h, n, dx, deadline):
    """Alternate between PGD and sweep rounds."""
    alpha = 50.0
    stall = 0
    while time.time() < deadline:
        # PGD phase: several gradient steps
        for _ in range(20):
            if time.time() > deadline:
                break
            grad, corr, m = _grad_c5_vec(h, alpha)
            # Projected gradient step
            step = 1.0
            for _ in range(20):
                h_new = h - step * grad
                h_new = _repair(h_new, n)
                if _c5(h_new, dx) < m * dx:
                    break
                step *= 0.5
            else:
                break  # no improvement
            h = h_new
            alpha *= 1.1  # slowly increase concentration
        
        # Sweep phase: one full round
        h, corr, improved, ni = _sweep_round(h, n, time.time() + min(30, max(1, deadline - time.time())))
        if not improved:
            stall += 1
        else:

        if stall >= 3:

            h = _repair(h, n)
            stall = 0
            alpha = 50.0

Also, I should be more careful with the step size. The gradient of the smooth max has a specific scale. Let me think about what step size makes sense.

The smooth max objective is f(h) = smooth_max(corr) * dx. The gradient has components of order O(1) (since each correlation is O(n) and the softmax weights sum to 1, the gradient components are O(1)). The step size should be O(1/L) where L is the Lipschitz constant of the gradient.

import numpy as np
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _grad_c5(h, alpha):
    n = len(h)
    corr = _corr(h)
    m = float(np.max(corr))
    s = np.clip(alpha * (corr - m), -500, 500)
    w = np.exp(s)
    ws = w.sum()
    if ws > 1e-30:
        w /= ws
    else:
        w = np.zeros(2 * n - 1)
        w[np.argmax(corr)] = 1.0
    lags = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for i in range(n):
        ip = i + lags
        m1 = (ip >= 0) & (ip < n)
        im = i - lags
        m2 = (im >= 0) & (im < n)
        grad[i] = np.sum(w[m1] * (1.0 - h[ip[m1]])) - np.sum(w[m2] * h[im[m2]])
    return grad, corr, m

def _subgrad_dir(h):
    n = len(h)
    corr = _corr(h)
    m = float(np.max(corr))
    tol = 1e-9 * m + 1e-12
    active = np.where(corr >= m - tol)[0]
    lags_full = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for k in active:
        lag = lags_full[k]
        for i in range(n):
            val = 0.0
            if 0 <= i + lag < n:
                val += 1.0 - h[i + lag]
            if 0 <= i - lag < n:
                val -= h[i - lag]
            grad[i] += val
    grad /= max(1, len(active))
    return grad, corr, m

def _g_row(h, i, n):
    lags = np.arange(-(n - 1), n)
    gi = np.zeros(2 * n - 1)
    ip = i + lags
    m1 = (ip >= 0) & (ip < n)
    gi[m1] += 1.0 - h[ip[m1]]
    im = i - lags
    m2 = (im >= 0) & (im < n)
    gi[m2] -= h[im[m2]]
    return gi

def _sweep_round(h, n, deadline, n_coarse=20, n_bisect=30):
    corr = _corr(h)
    improved = False
    n_improved = 0
    order = np.random.permutation(n)
    for ai in range(n):
        i = order[ai]
        for j in order[ai + 1:]:
            if time.time() > deadline:
                return h, corr, improved, n_improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-14:
                continue
            gd = _g_row(h, i, n) - _g_row(h, j, n)
            cur_max = np.max(corr)
            ds = np.linspace(d_min, d_max, n_coarse)
            best_v = cur_max
            best_d = 0.0
            for ii in range(n_coarse):
                v = np.max(corr + ds[ii] * gd)
                if v < best_v - 1e-14:
                    best_v = v
                    best_d = ds[ii]
            if best_v >= cur_max - 1e-13:
                continue
            lo = max(d_min, ds[max(0, np.argmin([np.max(corr + ds[ii]*gd) for ii in range(n_coarse)]) - 1)] if n_coarse > 1 else d_min)
            hi_b = min(d_max, ds[min(n_coarse - 1, np.argmin([np.max(corr + ds[ii]*gd) for ii in range(n_coarse)]) + 1)])
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            improved = True
            n_improved += 1
    return h, corr, improved, n_improved

def _fourier_init(n):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.ones(n) * 0.5
    for m in range(1, 30):
        coef = 0.15 / np.sqrt(m) * (1 if m % 2 else -0.5)
        h += coef * np.cos(2 * np.pi * m * x / 2.0)
        h += coef * 0.5 * np.sin(2 * np.pi * m * x / 2.0)
    return _repair(h, n)

def _block_init(n):
    h = np.zeros(n)
    block = 7
    for i in range(n):
        h[i] = 1.0 if (i // block) % 2 == 0 else 0.0
    return _repair(h, n)

def _optimize_inner(h, n, dx, deadline, seed):
    np.random.seed(seed)
    best_local = h.copy()
    best_local_c5 = _c5(h, dx)
    alpha = 40.0
    stall = 0
    it = 0
    while time.time() < deadline:
        it += 1
        if it % 10 == 1:
            h, corr, m = _grad_c5(h, alpha)
        else:
            h_grad, corr, m = _subgrad_dir(h)
            h = h_grad
        step = 2.0 / n
        improved_pg = False
        for _ in range(25):
            h_new = h - step * h
            h_new = _repair(h_new, n)
            c5_new = _c5(h_new, dx)
            if c5_new < m * dx - 1e-15:
                h = h_new
                improved_pg = True
                break
            step *= 0.6
        if improved_pg:
            alpha = min(alpha * 1.08, 500.0)
            stall = 0
        else:
            sweep_deadline = min(deadline, time.time() + 15.0)
            h, corr, sw_imp, ni = _sweep_round(h, n, sweep_deadline, 20, 30)
            if sw_imp:
                c5_new = _c5(h, dx)
                stall = 0
                alpha = 40.0
            else:
                stall += 1
        c5_cur = _c5(h, dx)
        if c5_cur < best_local_c5 - 1e-14:
            best_local_c5 = c5_cur
            best_local = h.copy()
        if stall >= 5 and time.time() < deadline - 20:
            scale = min(0.08, 0.02 + 0.01 * stall)
            h = _repair(best_local + scale * np.random.randn(n), n)
            alpha = 40.0
            stall = 0
    return best_local, best_local_c5
