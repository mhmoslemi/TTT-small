    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmin(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def _grad_k(h, n, k):
    lag = k - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        v = n - lag
        g[:v] += 1.0 - h[lag:]
        g[lag:] -= h[:v]
    else:
        v = n + lag
        g[-lag:] += 1.0 - h[:v]
        g[:v] -= h[-lag:]
    return g

def _grad_mat(h, n):
    G = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        G[m] = _grad_k(h, n, m)
    return G

def _nesterov(h, n, target, dx, t_end):
    x = h.copy()
    y = x.copy()
    best_x, best_f = x.copy(), _c5(x, dx)
    alpha = 0.1
    t = 0
    while time.time() < t_end:
        t += 1
        c = _corr(y)
        n_act = max(3, c.size // 8)
        idx = np.argpartition(-c, n_act)[:n_act]
        G = _grad_mat(y, n)
        g = G[idx].mean(axis=0)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        for _ in range(15):
            xn = _project(y - alpha * g, target)
            fn = _c5(xn, dx)
            if fn < best_f + 0.5 * alpha * gn or alpha < 1e-10:
                break
            alpha *= 0.5
        beta = (1.0 + 1.0 / np.sqrt(float(t))) / (2.0 + 1.0 / np.sqrt(float(t)))
        y = _project(xn + beta * (xn - x), target)
        x = xn
        if fn < best_f:
            best_f, best_x = fn, x.copy()
        if t % 300 == 0:
            alpha = min(alpha * 1.03, 0.2)
    return best_x, best_f

def _pair_cd(h, n, target, dx, t_end, rng):
    h = h.copy()
    c5v = _c5(h, dx)
    c_arr = _corr(h)
    G = _grad_mat(h, n)
    stale = 0
    while time.time() < t_end:
        improved = False
        for _ in range(300):
            if time.time() > t_end:
                return h, c5v
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j or h[i] >= 0.9999 or h[j] <= 0.0001:
                continue
            slope = G[:, i] - G[:, j]
            lo_d, hi_d = -h[j], 1.0 - h[i]
            if hi_d - lo_d < 1e-10:
                continue
            a, b = lo_d, hi_d
            for _ in range(30):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                if (c_arr + m1 * slope).max() < (c_arr + m2 * slope).max():
                    b = m2
                else:
                    a = m1
            d = 0.5 * (a + b)
            if abs(d) < 1e-10:
                continue
            new_max = (c_arr + d * slope).max()
            if new_max < c_arr.max() - 1e-11:
                h[i] += d
                h[j] -= d
                h = _project(h, target)
                c_arr = _corr(h)
                c5v = c_arr.max() * dx
                G = _grad_mat(h, n)
                improved = True
                stale = 0
            else:
                stale += 1
                if stale > 5000:
                    stale = 0
        if not improved:
            break
    return h, c5v

def _smooth_fn(x, alpha, target, pen):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (x.sum() - target) ** 2

def _sum_con(x, target):
    return x.sum() - target

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    h = _project(np.array(initial_h_values, dtype=float), target)
    best_h, best_c5 = h.copy(), _c5(h, dx)
    bounds = [(0.0, 1.0)] * n

    def _tl():
        return budget_s - (time.time() - t0)

    # Phase 1: Nesterov (40%)
    if _tl() > 80:
        h1, c1 = _nesterov(best_h, n, target, dx, t0 + budget_s * 0.40)
        if c1 < best_c5:
            best_c5, best_h = c1, h1

    # Phase 2: Pairwise CD (25%)
    if _tl() > 80:
        h2, c2 = _pair_cd(best_h, n, target, dx, t0 + budget_s * 0.65, rng)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 3: SLSQP (15%)
    if _tl() > 40:
        for alpha in [50000, 200000]:
            if _tl() < 30:
                break

                res = minimize(_smooth_fn, best_h, args=(alpha, target, 100.0),
                              method='SLSQP', bounds=bounds,
                              constraints=[{'type': 'eq', 'fun': _sum_con, 'args': (target,)}],
                              options={'maxiter': 500, 'ftol': 1e-18, 'iprint': 0})
                hc = _project(res.x, target)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
