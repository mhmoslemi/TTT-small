import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    # Proportional scaling then fix
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _softobj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _init_patterns(n, rng):
    """Generate diverse initial patterns."""
    patterns = []
    # Uniform
    patterns.append(np.full(n, 0.5))
    # Sinusoidal
    x = np.linspace(0, 2, n, endpoint=False)
    patterns.append(np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x), 0, 1))
    patterns.append(np.clip(0.5 + 0.4 * np.sin(4 * np.pi * x), 0, 1))
    patterns.append(np.clip(0.5 + 0.5 * np.cos(2 * np.pi * 1.5 * x), 0, 1))
    # Block patterns
    for p in [3, 4, 5, 7]:
        if p <= n:
            block = np.zeros(n)
            for i in range(n):
                phase = (i // (n // p)) % p
                if phase < p // 2:
                    block[i] = 1.0
            patterns.append(block)
    # Random binary-ish

    patterns.append(rb)
    rn = np.clip(rng.randn(n) * 0.3 + 0.5, 0, 1)
    patterns.append(rn)
    return patterns

                g = _active_grad(h, top_idx) / m_top
                g_c = g - g.mean()
                h_new, c5_new = _golden_line(h, g_c, dx, n, scale=0.02)
                if c5_new < c5_cur - 1e-14:
                    h = h_new
                    c5_cur = c5_new
                    if c5_cur < best_c5:
                        best_c5 = c5_cur
                        best_h = h.copy()
                    break
                else:
                    h, c5_cur = h_new, c5_new
                    h = _repair(h, n)

    # Phase 4: Pair coordinate with line search on top-gradient pairs
    if elapsed() < budget_s * 0.82:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        for _sweep in range(60):
            if elapsed() > budget_s * 0.82:
                break
            g = _softgrad(h, 100000, 0.0)
            order = np.argsort(np.abs(g))[::-1][:30]
            improved = False
            for ii in range(len(order)):
                if elapsed() > budget_s * 0.82:
                    break
                i = order[ii]
                for jj in range(ii + 1, min(ii + 12, len(order))):
                    j = order[jj]
                    d = np.zeros(n)
                    d[i] = 1.0
                    d[j] = -1.0
                    h_new, c5_new = _golden_line(h, d, dx, n, scale=0.005, max_iter=25)
                    if c5_new < c5_cur - 1e-14:
                        h = h_new
                        c5_cur = c5_new
                        improved = True
                        if c5_cur < best_c5:
                            best_c5 = c5_cur
                            best_h = h.copy()
                        break
            if not improved:
                break

    # Phase 5: Block perturbation + re-optimization (escape local optima)
    if elapsed() < budget_s * 0.9:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        for _blk in range(20):
            if elapsed() > budget_s * 0.9:
                break
            # Perturb a random block
            blen = rng.randint(3, max(4, n // 8))
            bstart = rng.randint(0, n - blen)
            h_p = h.copy()
            h_p[bstart:bstart + blen] = np.clip(
                h_p[bstart:bstart + blen] + 0.02 * rng.randn(blen), 0, 1)
            h_p = _repair(h_p, n)
            # Short L-BFGS from perturbed point
            try:
                res = minimize(
                    lambda x: _softobj(x, 100000, 200.0), h_p,
                    jac=lambda x: _softgrad(x, 100000, 200.0),
                    method='L-BFGS-B', bounds=bounds,
                    options={'maxiter': 200, 'ftol': 1e-17})
                hc = _repair(res.x, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc.copy()
            except Exception:
                pass

    # Phase 6: Ultra-fine random pair polish
    if elapsed() < budget_s - 3:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        steps = np.array([0.0003, 0.0007, 0.0015, 0.003,
                         -0.0003, -0.0007, -0.0015, -0.003])
        for _ in range(60000):
            if elapsed() > budget_s - 3:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = steps[rng.randint(len(steps))]
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5_cur - 1e-15:
                    c5_cur = c
                    if c < best_c5:
                        best_c5 = c
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
