def _fast_sweep(h, n, time_limit, n_coarse=30, n_bisect=30):
    corr = _corr(h)
    cur_max = float(np.max(corr))
    improved = False
    n_improved = 0
    
    # Pre-compute all g_vecs
    all_g = np.array([_g_vec(h, i, n) for i in range(n)])  # shape (n, 2n-1)
    
    while time.time() < time_limit:
        any_change = False
        for i in range(n):
            for j in range(i+1, n):
                if time.time() > time_limit:
                    return h, corr, improved, n_improved
                hi, hj = h[i], h[j]
                d_min = max(-hi, hj - 1.0)
                d_max = min(1.0 - hi, hj)
                if d_max - d_min < 1e-12:
                    continue
                gd = all_g[i] - all_g[j]
                # Find best d
                ds = np.linspace(d_min, d_max, n_coarse)
                # max(corr + d*gd) for each d
                vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
                bi = int(np.argmin(vals))
                if vals[bi] >= cur_max - 1e-13:
                    continue
                # Bisection
                lo = max(d_min, ds[max(0, bi-1)])
                hi_b = min(d_max, ds[min(n_coarse-1, bi+1)])
                for _ in range(n_bisect):
                    mid = 0.5*(lo + hi_b)
                    if np.max(corr + mid*gd) <= np.max(corr + hi_b*gd):
                        hi_b = mid
                    else:
                        lo = mid
                d_best = 0.5*(lo + hi_b)
                if abs(d_best) < 1e-14:
                    continue
                h[i] = hi + d_best
                h[j] = hj - d_best
                corr = corr + d_best * gd
                cur_max = float(np.max(corr))
                # Update all_g
                all_g = np.array([_g_vec(h, p, n) for p in range(n)])
                improved = True
                n_improved += 1
                any_change = True
        if not any_change:
            break
    return h, corr, improved, n_improved

            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(dlo, dhi, nc)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi2 = int(np.argmin(vals))
            cm = float(np.max(corr))
            if vals[bi2] > cm - 1e-12:
                continue
            lo = dlo if bi2 == 0 else ds[bi2 - 1]
            hh = dhi if bi2 == nc - 1 else ds[bi2 + 1]
            for _ in range(nb):
                mid = 0.5 * (lo + hh)
                if np.max(corr + mid * gd) <= np.max(corr + hh * gd):
                    hh = mid
                else:
                    lo = mid
            d = 0.5 * (lo + hh)
            if abs(d) < 1e-14:
                continue
            h[i] += d
            h[j] -= d
            corr += d * gd
            n_imp += 1
    return h, n_imp
