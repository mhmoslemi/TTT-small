def solve_1d_minimax(a, b):
    """Find min_v such that exists x in [0,1] with a[k] + x*b[k] <= v for all k."""
    def feasible(v):
        lo = 0.0
        hi = 1.0
        for k in range(len(a)):
            if b[k] > 1e-15:
                hi = min(hi, (v - a[k]) / b[k])
            elif b[k] < -1e-15:
                lo = max(lo, (v - a[k]) / b[k])
            else:
                if a[k] > v:
                    return False
        return lo <= hi
    
    # Binary search on v
    v_lo = np.max(np.minimum(a, a + b))  # lower bound
    v_hi = np.max(np.maximum(a, a + b))  # upper bound
    for _ in range(60):
        v_mid = (v_lo + v_hi) / 2
        if feasible(v_mid):
            v_hi = v_mid
        else:
            v_lo = v_mid
    
    # Find optimal x
    v = v_hi
    lo = 0.0
    hi = 1.0
    for k in range(len(a)):
        if b[k] > 1e-15:
            hi = min(hi, (v - a[k]) / b[k])
        elif b[k] < -1e-15:
            lo = max(lo, (v - a[k]) / b[k])

    x = np.clip(x, 0.0, 1.0)
    return x, v

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _solve_1d_minimax(a, b):
    n_c = len(a)
    def feasible(v):
        lo = 0.0
        hi = 1.0
        bp = b > 1e-14
        bn = b < -1e-14
        bz = ~bp & ~bn
        if np.any(bz) and np.max(a[bz]) > v + 1e-15:
            return False
        if np.any(bp):
            hi = min(hi, np.min((v - a[bp]) / b[bp]))
        if np.any(bn):
            lo = max(lo, np.max((v - a[bn]) / b[bn]))
        return lo <= hi + 1e-14
    v_lo = float(np.max(np.minimum(a, a + b)))

            co = np.correlate(h, 1.0 - h, mode='full') * dx

        nc5 = _c5(h, n)

    return h, best_c5

def _subgrad_step(h, n, eta, k_top):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[::-1][:k_top]
    g = np.zeros(n)
    for ki in idx:
        lag = ki - (n - 1)
        i_min = max(0, lag) if lag > 0 else 0
        i_max = min(n, n + lag) if lag < 0 else n
        for i in range(i_min, i_max):
            jp = i - lag
            if 0 <= jp < n:
                g[i] += (1.0 - h[jp]) / k_top
                g[jp] -= h[i] / k_top
    h_new = h - eta * g
    return h_new

def _slsqp(h0, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = np.max(c)
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def grad(x):
        dx = 2.0 / n
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(c)
        s = np.clip(alpha * (c - m), -400, 400)
        e = np.exp(s)
        w = e / np.sum(e)
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wi = w[k]
            if wi < 1e-14:
                continue
            lag = k - (n - 1)
            lo = max(0, lag) if lag > 0 else 0
            hi = min(n, n + lag) if lag < 0 else n
            for i in range(lo, hi):
                jp = i - lag
                if 0 <= jp < n:
                    g[i] += wi * (1.0 - x[jp])
                    g[jp] -= wi * x[i]
        return g * dx
    def con_fun(x):
        return np.sum(x) - n / 2.0
    res = minimize(obj, h0, jac=grad, method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return res.x

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _solve_1d_minimax(a, b):
    def feasible(v):
        lo, hi = 0.0, 1.0
        bp = b > 1e-14
        bn = b < -1e-14
        bz = ~bp & ~bn
        if np.any(bz) and np.max(a[bz]) > v + 1e-15:
            return False
        if np.any(bp):
            hi = min(hi, float(np.min((v - a[bp]) / b[bp])))
        if np.any(bn):
            lo = max(lo, float(np.max((v - a[bn]) / b[bn])))
        return lo <= hi + 1e-14
    v_lo = float(np.max(np.minimum(a, a + b)))
    v_hi = float(np.max(np.maximum(a, a + b)))
    for _ in range(80):
        vm = (v_lo + v_hi) * 0.5
        if feasible(vm):
            v_hi = vm
        else:
            v_lo = vm
    v = v_hi
    lo, hi = 0.0, 1.0
    bp = b > 1e-14
    bn = b < -1e-14
    if np.any(bp):
        hi = min(hi, float(np.min((v - a[bp]) / b[bp])))
    if np.any(bn):
        lo = max(lo, float(np.max((v - a[bn]) / b[bn])))
    x = (max(lo, 0.0) + min(hi, 1.0)) * 0.5
    return float(np.clip(x, 0.0, 1.0))

def _coord_descent(h, n, max_sweeps, deadline, rng=None):
    best_c5 = _c5(h, n)
    dx = 2.0 / n
    order = np.arange(n)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if rng is not None and sweep % 3 == 2:
            order = rng.permutation(n)
        improved = False
        for m in order:
            orig = h[m]
            eps = 1e-7
            h[m] = orig + eps
            cp = np.correlate(h, 1.0 - h, mode='full') * dx
            h[m] = orig - eps
            cm = np.correlate(h, 1.0 - h, mode='full') * dx
            h[m] = orig
            co = np.correlate(h, 1.0 - h, mode='full') * dx
            b = (cp - cm) / (2.0 * eps)
            a = co - orig * b
            xn = _solve_1d_minimax(a, b)
            if abs(xn - orig) > 1e-12:
                h[m] = xn
                improved = True
        nc5 = _c5(h, n)
        if nc5 < best_c5:
            best_c5 = nc5
        if not improved:
            break
    return h, best_c5

def _slsqp(h0, n, alpha, maxiter):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(c)
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(c)
        s = np.clip(alpha * (c - m), -400, 400)
        e = np.exp(s)
        w = e / np.sum(e)
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wi = w[k]
            if wi < 1e-14:
                continue
            lag = k - (n - 1)
            lo = max(0, lag) if lag > 0 else 0
            hi = min(n, n + lag) if lag < 0 else n
            for i in range(lo, hi):
                jp = i - lag
                if 0 <= jp < n:
                    g[i] += wi * (1.0 - x[jp])
                    g[jp] -= wi * x[i]
        return g * dx
    def cfun(x):
        return np.sum(x) - n / 2.0
    res = minimize(obj, h0, jac=grad, method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': cfun, 'jac': lambda x: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return res.x

def _pair_search(h, n, deltas, max_passes, deadline):
    c5c = _c5(h, n)
    best_c = c5c
    best = h.copy()
    for ps in range(max_passes):
        if time.time() > deadline:
            break
        imp = False
        for i in range(n):
            if time.time() > deadline:
                break
            hi = h[i]
            for j in range(i + 1, n):
                if time.time() > deadline:
                    break
                hj = h[j]
                for d in deltas:
                    ni, nj = hi + d, hj - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c = _c5(h, n)
                        if c < c5c - 1e-14:
                            c5c = c
                            hi, hj = ni, nj
                            imp = True
                        else:
                            h[i], h[j] = hi, hj
            h[i] = hi
        if c5c < best_c - 1e-15:
            best_c = c5c
            best = h.copy()
        if not imp:
            break
    return best, best_c
