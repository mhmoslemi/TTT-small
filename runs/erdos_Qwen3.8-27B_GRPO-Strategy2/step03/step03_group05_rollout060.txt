def _project(h, target):
    """Project h onto {x in [0,1]^n : sum(x) = target}"""
    n = len(h)
    # Binary search for lambda
    lo, hi = h.min() - target, h.max() + target
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo+hi)/2, 0, 1)
    # Fix any remaining error
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        # Adjust one element
        if diff > 0:
            idx = np.argmax(1 - x)
            x[idx] = min(1, x[idx] + diff)
        else:
            idx = np.argmax(x)
            x[idx] = max(0, x[idx] + diff)
    return x

def _periodic_score(p, period):
    """Compute max circular autocorrelation deficit for a period pattern"""
    n_per = period
    ones_per = p.sum()
    # c_r = ones_per - circular_autocorr(r)
    max_c = 0
    for r in range(1, n_per):
        shifted = np.roll(p, -r)
        overlap = np.sum(p * shifted)  # circular autocorrelation at shift r
        c_r = ones_per - overlap
        if c_r > max_c:
            max_c = c_r
    return max_c

def _all_grads(h):
    """Return G[k, L] = dcorr(L)/dh[k] for all k, L"""
    n = len(h)
    G = np.zeros((n, 2*n-1))
    G[:, n-1] = 1.0 - 2.0*h  # L=0
    for L in range(1, n):
        # Positive shift
        G[L:, n-1+L] += 1.0 - h[:n-L]  # positions L to n-1: (1-h[pos-L])
        G[:n-L, n-1+L] -= h[L:]  # positions 0 to n-1-L: -h[pos+L]
        # Negative shift (M=L)
        G[:n-L, n-1-L] += 1.0 - h[L:]  # positions 0 to n-1-L: (1-h[pos+L])
        G[L:, n-1-L] -= h[:n-L]  # positions L to n-1: -h[pos-L]
    return G

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo+hi)*0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-13:
        idx = int(np.argmax(1.0-x)) if diff > 0 else int(np.argmax(x))
        x[idx] = np.clip(x[idx]+diff, 0.0, 1.0)
    return x

def _corr(h):
    return np.correlate(h, 1.0-h, mode='full')
