    return np.max(_corr(h)) * dx

def _repair(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    # Iterative water-filling
    h = h.copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        # Adjust: add uniform amount to non-saturated
        if s > target:
            excess = s - target
            free = h < 1.0 - 1e-15
            if free.sum() == 0:
                break
            adj = excess / free.sum()
            h[free] = np.maximum(h[free] - adj, 0.0)

    g = np.zeros(n)

    return g

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            free = h < 1.0
            if not free.any():
                break
            h[free] -= (s - target) / free.sum()
            np.clip(h, 0, 1, out=h)
        else:
            free = h > 0.0
            if not free.any():
                break
            h[free] += (target - s) / free.sum()
            np.clip(h, 0, 1, out=h)
    return h

def _grad_lag(h, n, k):
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
    else:
        kk = -k
        if kk < n:
            g[:n-kk] += (1.0 - h[kk:])
            g[kk:] -= h[:n-kk]
    return g

def _smooth_obj(x, alpha, n):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _constr_fun(x, target):
    return float(np.sum(x) - target)

def _greedy_pair_search(h, n, dx, max_rounds, t_left_fn, slack_fn):

    max_c = float(c_arr.max())
    for _ in range(max_rounds):
        if t_left_fn() < slack_fn:
            break
        m_star = int(np.argmax(c_arr))
        k_star = m_star - (n - 1)
        g_star = _grad_lag(h, n, k_star)
        best_improve = 1e-14
        best_i, best_j, best_d = -1, -1, 0.0

                d = min(1.0 - h[i], h[j])

                    best_improve = d * dg
                    best_i, best_j, best_d = i, j, d
        if best_i < 0:

        accepted = False

            if c_new < max_c - 1e-14:
                max_c = c_new
                c_arr = _corr(h)
                accepted = True
                break
            h[best_i] -= d
            h[best_j] += d
        if not accepted:
            break
    return h, max_c * dx

def _sa_phase(h, n, dx, T_init, cooling, max_iters, t_left_fn, slack, rng):
    c_cur = _c5(h, dx)

        npert = int(rng.randint(2, 5))

        h_t = h.copy()
        h_t[idx] += rng.randn(npert) * T * 8.0

        if c_t < c_cur or rng.rand() < np.exp(-(c_t - c_cur) / max(T, 1e-10)):

    return h, c_cur
