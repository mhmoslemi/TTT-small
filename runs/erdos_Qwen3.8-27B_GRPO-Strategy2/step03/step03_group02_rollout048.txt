import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, i, n):
    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = i - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = i + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _grad_smooth_max(h, alpha):
    c = _corr(h)
    n = len(h)
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    grad = np.zeros(n)
    for k_idx in range(len(c)):
        if w[k_idx] < 1e-14:
            continue
        k = k_idx - (n - 1)
        i_lo = max(0, k)
        i_hi = min(n, n + k)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += w[k_idx] * (1.0 - h[i_lo - k:i_hi - k])
        i_lo2 = max(0, -k)
        i_hi2 = min(n, n - k)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] -= w[k_idx] * h[i_lo2 + k:i_hi2 + k]
    return grad

def _pa_sweep(h, n, corr, time_limit, n_coarse, n_bisect):
    improved = False
    order = np.random.permutation(n)
    for ai in range(n - 1):
        i = order[ai]
        for j in order[ai + 1:]:
            if time.time() > time_limit:
                return h, corr, improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = np.argmin(vals)
            if vals[bi] >= np.max(corr) - 1e-13:
                continue
            lo = max(d_min, ds[max(0, bi - 1)])
            hi_b = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            improved = True
    return h, corr, improved

def _slsqp_step(h, n, alpha, maxiter, target, bounds):
    def obj(x):
        return _smooth_max(_corr(x), alpha)
    def eqcon(x):
        return float(np.sum(x)) - target
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eqcon}],
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return h

def _lifb_step(h, n, alpha, maxiter, bounds):
    def obj(x):
        return _smooth_max(_corr(x), alpha)
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj_grad(h, n, alpha, lr_init, n_steps, time_limit):
    h_cur = h.copy()
    dx = 2.0 / n
    lr = lr_init
    base_c5 = _c5(h_cur, dx)
    for _ in range(n_steps):
        if time.time() > time_limit:
            break
        g = _grad_smooth_max(h_cur, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        lr = min(lr, 0.5 / g_norm)
        accepted = False
        for bt in range(20):
            h_trial = _repair(h_cur - lr * g, n)
            c5_t = _c5(h_trial, dx)
            if c5_t < base_c5 - 1e-15:
                h_cur = h_trial
                base_c5 = c5_t
                accepted = True
                break
            lr *= 0.5
        if not accepted:
            lr = lr_init * 0.1
    return h_cur

def _fourier_opt(h, n, alpha, maxit, time_limit):
    H = np.fft.rfft(h.astype(float))
    M = min(12, n // 5)
    H0 = H[1:M + 1].copy()

    def h_from_p(p):
        new_H = H.copy()
        new_H[1:M + 1] = p
        return _repair(np.fft.irfft(new_H, n=n), n)

    def obj(p):
        return _smooth_max(_corr(h_from_p(p)), alpha)

    best_p = H0.copy()
    best_val = obj(H0)
    try:
        res = minimize(obj, H0, method='Nelder-Mead',
                      options={'maxiter': maxit, 'xatol': 1e-10,
                               'fatol': 1e-12, 'maxfev': 400,
                               'adaptive': True})
        if time.time() < time_limit and obj(res.x) < best_val:
            best_p = res.x.copy()
            best_val = obj(res.x)
    except Exception:
        pass
    return h_from_p(best_p)
