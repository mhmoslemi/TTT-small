def _sj(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    w0 = w[n-1]
    if w0 > 1e-14:
        g += w0 * (1.0 - 2.0 * x)
    for s in range(1, n):
        wp = w[n-1+s]
        wm = w[n-1-s]
        if wp > 1e-14:
            g[:n-s] += wp * (1.0 - x[s:])
            g[s:] -= wp * x[:n-s]
        if wm > 1e-14:
            g[s:] += wm * (1.0 - x[:n-s])
            g[:n-s] -= wm * x[s:]
    return g * (2.0 / n)

    return (np.sum(c**p))**(1.0/p)

def _subgrad_step(h, n, step, alpha):
    """Take one subgradient step with smoothing."""
    c = np.correlate(h, 1-h, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    w0 = w[n-1]
    if w0 > 1e-14:
        g += w0 * (1.0 - 2.0 * h)
    for s in range(1, n):
        wp = w[n-1+s]
        wm = w[n-1-s]
        if wp > 1e-14:
            g[:n-s] += wp * (1.0 - h[s:])
            g[s:] -= wp * h[:n-s]
        if wm > 1e-14:
            g[s:] += wm * (1.0 - h[:n-s])
            g[:n-s] -= wm * h[s:]
    g *= 2.0 / n
    # Project: subtract mean to maintain sum constraint
    g -= g.mean()
    # Backtracking line search
    c0 = _eval(h)
    lr = step
    for _ in range(20):
        h_new = _repair(h - lr * g, n)
        c_new = _eval(h_new)
        if c_new < c0 - 1e-15 * lr * np.abs(g).max():
            return h_new, c_new
        lr *= 0.5
    return h, c0

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _sf(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a

def _sj(x, a):

    g = np.zeros(n)

        L = k - (n - 1)

def _slsqp(h, alpha, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

             'jac': lambda x: np.ones(n)}]

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h.copy(), n)

def _pair_opt(h, i, j):
    """Optimal transfer from j to i via 1D scan."""
    n = len(h)
    hi, hj = h[i], h[j]

    d_hi = min(1.0 - hi, hj) - 1e-12
    if d_hi - d_lo < 1e-10:
        return
    c0 = _eval(h)
    best_d = 0.0
    best_c = c0
    # Scan
    for k in range(21):
        d = d_lo + (d_hi - d_lo) * k / 20.0
        h[i], h[j] = hi + d, hj - d
        c = _eval(h)

            best_d = d

        c = _eval(h)
        if c < best_c:
            best_c = c
            best_d = d
    h[i], h[j] = hi + best_d, hj - best_d
