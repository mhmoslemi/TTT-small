def _proj_grad(h, n, alpha, lr, iters):
    """Projected subgradient with Nesterov"""
    y = h.copy()
    h_old = h.copy()
    for t in range(iters):
        g = _sj(h_old.copy(), alpha)
        # Nesterov: y = h - lr * grad(h), then project
        # Actually standard: h_new = y - lr*g, y_new = h_new + (t/(t+2))*(h_new - h_old)
        h_new = y - lr * g
        h_new = _repair(h_new, n)
        y = h_new + (t / (t + 2.0)) * (h_new - h_old)
        y = _repair(y, n)
        h_old = h_new
    return h_old

    n_candidates = [n0 - 6, n0 - 3, n0 + 3, n0 + 6, n0 + 10, n0 - 10, n0 + 14]
    n_candidates = sorted(set(n for n in n_candidates if 30 <= n <= 120 and n != n0))
    for nc in n_candidates:
        if _el() > budget_s - 300:
            break
        hi = _interp_h(best_h, nc)
        hi = _slsqp(hi, nc, 50000, 1500)
        hi = _slsqp(hi, nc, 300000, 1000)
        c5 = _eval(hi)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, hi.copy(), nc

    # Phase 3: Subgradient + periodic SLSQP on best
    if _el() < budget_s - 200:
        h = _subgrad_opt(best_h, best_n, 80000, 4000, 0.008)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 4: Diverse multi-start at best_n
    if _el() < budget_s - 120:
        n_use = best_n
        inits = [
            _repair(best_h + 0.04 * rng.randn(n_use), n_use),
            _repair(best_h + 0.08 * rng.randn(n_use), n_use),
            _binary_init(n_use, rng),
            _struct_init(n_use, 3, 0.3, rng),
            _struct_init(n_use, 5, 0.1, rng),
            _struct_init(n_use, 2, 0.7, rng),
        ]
        for h_init in inits:
            if _el() > budget_s - 120:
                break
            h = _slsqp(h_init, n_use, 50000, 1500)
            h = _slsqp(h, n_use, 200000, 800)
            c5 = _eval(h)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h.copy(), n_use

    # Phase 5: Greedy gradient-guided pair search
    if _el() < budget_s - 60:
        h5, c5_5 = _greedy_pairs(best_h, best_n, 15000, rng,
                                  lambda: _el() > budget_s - 60)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Final high-precision SLSQP
    if _el() < budget_s - 10:
        h = _slsqp(best_h, best_n, 500000, 3000)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 7: Ultra-fine random perturbations
    if _el() < budget_s - 3:
        h = best_h.copy()
        c5c = best_c5
        for _ in range(20000):
            if _el() > budget_s - 3:
                break
            i = rng.randint(best_n)
            d = rng.choice([-1, 1]) * rng.choice([0.5, 1, 3]) * 1e-5
            ni = h[i] + d
            if 0 <= ni <= 1:
                h[i] = ni
                # Compensate another point
                j = (i + 1 + rng.randint(best_n - 1)) % best_n
                h[j] -= d
                h = _repair(h, best_n)
                c5n = _eval(h)
                if c5n < c5c - 1e-16:
                    c5c = c5n
                    best_c5, best_h = c5c, h.copy()
                else:
                    h = best_h.copy()

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
