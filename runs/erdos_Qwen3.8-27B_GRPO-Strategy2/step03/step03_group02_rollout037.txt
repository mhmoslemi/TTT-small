```python
def _coord_opt_h(h, p, corr, n):
    """Optimize h[p] given all other h fixed. corr is the current correlation vector."""
    # Which correlation values depend on h[p]?
    # c[lag+n-1] involves h[p] if 0<=p+lag<n or 0<=p-lag<n
    # lag ranges from -(n-1) to (n-1)
    # p+lag valid: -(p) <= lag <= n-1-p
    # p-lag valid: p-(n-1) <= lag <= p
    
    # For each lag involving p:
    # dc/dh[p] = (1-h[p+lag])*I(p+lag valid) - h[p-lag]*I(p-lag valid)
    # This is the slope of c[lag] as a function of h[p]
    
    lags = np.arange(-(n-1), n)
    jp = p + lags
    jm = p - lags
    mp = (jp >= 0) & (jp < n)
    mm = (jm >= 0) & (jm < n)
    active = mp | mm  # lags where c depends on h[p]
    
    # For active lags, c_k = corr[k] + (new_h_p - old_h_p) * slope_k
    slopes = np.zeros(2*n-1)
    slopes[mp] += (1.0 - h[jp[mp]])
    slopes[mm] -= h[jm[mm]]
    
    old_hp = h[p]
    # We want: minimize max_k corr[k] + d * slopes[k] where d = new_hp - old_hp
    # subject to 0 <= old_hp + d <= 1 and integral constraint (handled separately)
    
    d_lo = max(0, -old_hp)
    d_hi = min(1, 1 - old_hp)
    
    # Find the d that minimizes max(corr + d * slopes)
    # This is a 1D convex minimization
    # Sample and bisect
    n_grid = 50
    ds = np.linspace(d_lo, d_hi, n_grid)
    vals = np.array([np.max(corr + d * slopes) for d in ds])
    bi = np.argmin(vals)
    if vals[bi] >= corr.max() - 1e-13:
        return 0.0  # no improvement
    
    lo = ds[max(0, bi-1)]
    hi_b = ds[min(n_grid-1, bi+1)]
    for _ in range(35):
        mid = 0.5*(lo + hi_b)
        if np.max(corr + mid*slopes) <= np.max(corr + hi_b*slopes):
            hi_b = mid
        else:
            lo = mid
    return 0.5*(lo + hi_b)
```

        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _coord_sweep(h, n, corr, time_limit):
    """Coordinate descent: optimize each h[p] in turn."""
    n_imp = 0
    order = np.random.permutation(n)
    for p in order:
        if time.time() > time_limit:
            break
        old_hp = h[p]
        lags = np.arange(-(n-1), n)
        jp = p + lags
        jm = p - lags
        mp = (jp >= 0) & (jp < n)
        mm = (jm >= 0) & (jm < n)
        active = mp | mm
        if not active.any():
            continue
        slopes = np.zeros(2*n-1)
        if mp.any():
            slopes[mp] += (1.0 - h[jp[mp]])
        if mm.any():
            slopes[mm] -= h[jm[mm]]
        d_lo = max(0.0, -old_hp)
        d_hi = min(1.0, 1.0 - old_hp)
        if d_hi - d_lo < 1e-14:
            continue
        n_grid = 41
        ds = np.linspace(d_lo, d_hi, n_grid)
        cur_mx = corr.max()
        vals = np.array([np.max(corr + d * slopes) for d in ds])
        bi = np.argmin(vals)
        if vals[bi] >= cur_mx - 1e-13:
            continue
        lo = ds[max(0, bi-1)]
        hi_b = ds[min(n_grid-1, bi+1)]
        for _ in range(30):
            mid = 0.5*(lo + hi_b)
            if np.max(corr + mid*slopes) <= np.max(corr + hi_b*slopes):
                hi_b = mid
            else:
                lo = mid
        d_best = 0.5*(lo + hi_b)
        if abs(d_best) > 1e-14:
            h[p] = old_hp + d_best
            corr += d_best * slopes
            n_imp += 1
    return h, corr, n_imp

def _pair_sweep(h, n, corr, time_limit, n_grid=31, n_bisect=28):
    """Pair coordinate descent."""
    order = np.random.permutation(n)
    n_imp = 0
    for ai in range(n):
        for bi in range(ai+1, n):
            if time.time() > time_limit:
                return h, corr, n_imp
            i, j = order[ai], order[bi]
            hi, hj = h[i], h[j]
            d_lo = max(0.0, hj-1.0, -hi)
            d_hi = min(1.0-hi, hj)
            if d_hi - d_lo < 1e-14:
                continue
            lags = np.arange(-(n-1), n)
            jp_i, jm_i = i + lags, i - lags
            jp_j, jm_j = j + lags, j - lags
            mp_i = (jp_i >= 0) & (jp_i < n)
            mm_i = (jm_i >= 0) & (jm_i < n)
            mp_j = (jp_j >= 0) & (jp_j < n)
            mm_j = (jm_j >= 0) & (jm_j < n)
            gd = np.zeros(2*n-1)
            if mp_i.any(): gd[mp_i] += (1.0 - h[jp_i[mp_i]])
            if mm_i.any(): gd[mm_i] -= h[jm_i[mm_i]]
            if mp_j.any(): gd[mp_j] -= (1.0 - h[jp_j[mp_j]])
            if mm_j.any(): gd[mm_j] += h[jm_j[mm_j]]
            ds = np.linspace(d_lo, d_hi, n_grid)
            cur_mx = corr.max()
            vals = np.array([np.max(corr + d*gd) for d in ds])
            bi2 = np.argmin(vals)
            if vals[bi2] >= cur_mx - 1e-12:
                continue
            lo = ds[max(0, bi2-1)]
            hi_b = ds[min(n_grid-1, bi2+1)]
            for _ in range(n_bisect):
                mid = 0.5*(lo+hi_b)

            h[i] += d_best
            h[j] -= d_best
            corr += d_best * gd
            n_imp += 1

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0

    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _c5(best_h, dx)
    best_corr = _corr(best_h)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP warm-start
    for alpha in [50, 150, 400, 1000, 2000]:
        if rem() < 180: break
        hc = _slsqp(best_h, n, alpha, 500)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5 = c5; best_h = hc.copy(); best_corr = _corr(best_h)

    # Phase 2: Intensive coord + pair sweeps
    for sw in range(25):
        if rem() < 120: break
        h = best_h.copy(); corr = best_corr.copy()
        h, corr, _ = _coord_sweep(h, n, corr, time.time() + min(12, rem() - 30), 41, 32)
        c5 = _c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5; best_h = h.copy(); best_corr = corr.copy()
        if rem() < 120: break
        h = best_h.copy(); corr = best_corr.copy()
        h, corr, _ = _pair_sweep(h, n, corr, time.time() + min(20, rem() - 30), 31, 28)
        c5 = _c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5; best_h = h.copy(); best_corr = corr.copy()

    # Phase 3: SLSQP + refine + perturbation
    stall = 0
    for it in range(40):
        if rem() < 120: break
        alpha = 300 + 200 * (it % 4)
        hc = _slsqp(best_h, n, alpha, 400)
        hc_corr = _corr(hc)
        tL = time.time() + min(15, rem() - 40)
        hc, hc_corr, _ = _coord_sweep(hc, n, hc_corr, tL, 41, 32)
        tL2 = time.time() + min(18, rem() - 40)
        hc, hc_corr, _ = _pair_sweep(hc, n, hc_corr, tL2, 29, 26)
        c5 = _c5(hc, dx)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5; best_h = hc.copy(); best_corr = hc_corr.copy(); stall = 0
        else:
            stall += 1
        if stall >= 3 and rem() > 160:
            scale = min(0.12, 0.03 + 0.02 * stall)
            hp = best_h + scale * np.random.randn(n)
            hp = _repair(hp, n)
            hc2 = _slsqp(hp, n, 600, 400)
            hc2c = _corr(hc2)
            tL3 = time.time() + min(20, rem() - 50)
            hc2, hc2c, _ = _pair_sweep(hc2, n, hc2c, tL3, 31, 30)
            c5_2 = _c5(hc2, dx)
            if c5_2 < best_c5:
                best_c5 = c5_2; best_h = hc2.copy(); best_corr = hc2c.copy()
            stall = 0

    # Phase 4: Final high-precision sweeps
    for _ in range(6):
        if rem() < 18: break
        h = best_h.copy(); corr = best_corr.copy()
        h, corr, _ = _coord_sweep(h, n, corr, time.time() + min(10, rem() - 10), 51, 38)
        h, corr, _ = _pair_sweep(h, n, corr, time.time() + min(10, rem() - 10), 41, 35)
        c5 = _c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5; best_h = h.copy(); best_corr = corr.copy()

    # Phase 5: Final SLSQP polish
    if rem() > 8:
        hc = _slsqp(best_h, n, 5000, 1000)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5 = c5; best_h = hc.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
