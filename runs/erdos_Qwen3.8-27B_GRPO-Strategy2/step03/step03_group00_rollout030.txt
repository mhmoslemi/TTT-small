def _pair_optimal_d(h, i, j, c, n):
    """Find optimal d for h[i]+=d, h[j]-=d to minimize max overlap."""
    dx = 2.0 / n
    # d range
    d_lo = max(-h[i], h[j] - 1.0)
    d_hi = min(1.0 - h[i], h[j])
    if d_hi <= d_lo:
        return 0.0
    
    # Compute a_k = b_k^i - b_k^j for all lags k
    # b_k^i = (1-h[i-lag])*1{valid} - h[i+lag]*1{valid} for lag = k-(n-1)
    a = np.zeros(2*n - 1)
    for k in range(2*n - 1):
        lag = k - (n - 1)
        # b for index i
        bi = 0.0
        if 0 <= i - lag < n:
            bi += (1.0 - h[i - lag])
        if 0 <= i + lag < n:
            bi -= h[i + lag]
        # b for index j
        bj = 0.0
        if 0 <= j - lag < n:
            bj += (1.0 - h[j - lag])
        if 0 <= j + lag < n:
            bj -= h[j + lag]
        a[k] = bi - bj
    
    # f(d) = max_k (c[k] + a[k]*d) * dx
    # Find min over d in [d_lo, d_hi]
    # Evaluate at boundaries and at breakpoints of top lines
    candidates = [0.0, d_lo, d_hi]
    
    # Top indices by c[k]
    top = np.argsort(c)[::-1][:8]
    for p in range(len(top)):
        for q in range(p+1, len(top)):
            kp, kq = top[p], top[q]
            denom = a[kp] - a[kq]
            if abs(denom) < 1e-15:
                continue
            d_bp = (c[kq] - c[kp]) / denom
            if d_lo <= d_bp <= d_hi:
                candidates.append(d_bp)
    
    best_d = 0.0
    best_val = float('inf')
    for d_cand in candidates:
        f_val = np.max(c + a * d_cand)
        if f_val < best_val:
            best_val = f_val
            best_d = d_cand
    
    return best_d

For i+lag to be in [0,n): 0 ≤ i+k-n+1 < n → n-1-i ≤ k ≤ 2n-2-i. With k∈[0,2n-2], valid range is [max(0, n-1-i), min(2n-2, 2n-2-i)].

So:
b[k] = (1-h[i-k+n-1]) * 1{max(0,1-n+i)≤k≤min(2n-2,n-1+i)} - h[i+k-n+1] * 1{max(0,n-1-i)≤k≤min(2n-2,2n-2-i)}

I can compute these as slicings:

For the first term: indices into h are i-k+n-1 for k in the valid range.
When k = max(0, 1-n+i): index = i - (1-n+i) + n-1 = n-1
When k = min(2n-2, n-1+i): index = i - (n-1+i) + n-1 = 0
So the indices go from n-1 down to 0 (or a subset). It's reversed h.

For the second term: indices into h are i+k-n+1 for k in the valid range.
When k = max(0, n-1-i): index = i + (n-1-i) - n + 1 = 0
When k = min(2n-2, 2n-2-i): index = i + (2n-2-i) - n + 1 = n-1
So indices go from 0 up to n-1. It's h itself.

So: 
term1 = np.zeros(2n-1); term1[valid_range_1] = 1-h[n-1:n-1-(len_range_1):-1]  (reversed h, sliced)
term2 = np.zeros(2n-1); term2[valid_range_2] = h[0:len_range_2]  (h, sliced)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                h[idx] += min(diff, 1.0 - h[idx])
                diff -= min(diff, 1.0 - h[idx])
            else:
                h[idx] -= min(-diff, h[idx])
                diff += min(-diff, h[idx])
            if abs(diff) < 1e-14:
                break
    return h

def _c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _c5_vec(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smooth_f(x, n, alpha):
    c = _c5_vec(x, n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_df(x, n, alpha):
    dx = 2.0 / n
    c = _c5_vec(x, n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    tot = np.sum(e)
    w = e / tot
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        i_lo = max(0, -lag)
        i_hi = min(n, n - lag)
        for i in range(i_lo, i_hi):
            jp = i + lag
            g[jp] += wk * (1.0 - x[i])
            g[i] -= wk * x[jp]
    return g * dx

def _slsqp(h_init, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    return minimize(_smooth_f, h_init, args=(n, alpha),
                    jac=lambda x: _smooth_df(x, n, alpha),
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': maxiter, 'ftol': 1e-18})

def _pair_sweep(h, n, max_sweeps, budget_end):
    """Exact pair-move local search using linearity in d."""
    c5_cur = _c5(h, n)
    c = _c5_vec(h, n)
    for sweep in range(max_sweeps):
        if time.time() > budget_end:
            break
        improved = False
        for i in range(n):
            if time.time() > budget_end:
                return h, c5_cur
            for j in range(i + 1, n):
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi <= d_lo + 1e-15:
                    continue
                # Compute a[k] = b_k^i - b_k^j
                a = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    lag = k - (n - 1)
                    bi = 0.0
                    if 0 <= i - lag < n:
                        bi += 1.0 - h[i - lag]
                    if 0 <= i + lag < n:
                        bi -= h[i + lag]
                    bj = 0.0
                    if 0 <= j - lag < n:
                        bj += 1.0 - h[j - lag]
                    if 0 <= j + lag < n:
                        bj -= h[j + lag]
                    a[k] = bi - bj
                # Find min of max(c + a*d) over d in [d_lo, d_hi]
                candidates = np.array([0.0, d_lo, d_hi])
                # Add breakpoints from top lines
                top = np.argsort(c)[::-1][:6]
                for p in range(len(top)):
                    for q in range(p + 1, len(top)):
                        kp, kq = top[p], top[q]
                        denom = a[kp] - a[kq]
                        if abs(denom) > 1e-14:
                            d_bp = (c[kq] - c[kp]) / denom
                            if d_lo < d_bp < d_hi:
                                candidates = np.append(candidates, d_bp)
                best_d = 0.0
                best_val = np.max(c)  # f(0)
                for dc in candidates:
                    val = np.max(c + a * dc)
                    if val < best_val - 1e-15:
                        best_val = val
                        best_d = dc
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    h = _repair(h, n)
                    c = _c5_vec(h, n)
                    c5_new = float(np.max(c))
                    if c5_new < c5_cur - 1e-15:
                        c5_cur = c5_new
                        improved = True
                    else:
                        h[i] -= best_d
                        h[j] += best_d
                        h = _repair(h, n)
                        c = _c5_vec(h, n)
        if not improved:
            break
    return h, c5_cur
