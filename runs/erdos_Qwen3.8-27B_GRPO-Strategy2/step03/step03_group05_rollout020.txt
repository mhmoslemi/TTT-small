import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-10:
                break
    return h

                    if c < bc: bc, bt = c, t
            h[i], h[j] = bt, s - bt

        if not improved: break

    c = _corr(x)
    m = np.max(c)
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a + lam * (np.sum(x) - len(x) / 2.0) ** 2

def _softgrad(x, a, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(a * (c - m), -500, 500)
    ew = np.exp(s); pw = ew / ew.sum()
    n = len(x); g = np.zeros(n)
