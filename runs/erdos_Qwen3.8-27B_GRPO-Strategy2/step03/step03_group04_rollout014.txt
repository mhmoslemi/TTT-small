import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h


def _coord_step(h, p, n):
    K = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.arange(K) - (n - 1)
    j1 = p + m
    j2 = p - m
    C1 = np.where((j1 >= 0) & (j1 < n), 1.0 - h[np.clip(j1, 0, n - 1)], 0.0)
    C2 = np.where((j2 >= 0) & (j2 < n), h[np.clip(j2, 0, n - 1)], 0.0)
    A = C1 - C2
    B = corr - h[p] * A
    t_cands = np.array([0.0, 1.0])
    da = A[:, np.newaxis] - A[np.newaxis, :]
    db = B[np.newaxis, :] - B[:, np.newaxis]
    mask = np.abs(da) > 1e-15
    with np.errstate(divide='ignore', invalid='ignore'):
        t_mat = np.where(mask, db / np.where(mask, da, 1.0), np.nan)
    valid = mask & (t_mat > 0.0) & (t_mat < 1.0) & ~np.isnan(t_mat)
    if valid.any():
        t_extra = t_mat[valid]
        t_cands = np.concatenate([t_cands, t_extra])
    vals = A[np.newaxis, :] * t_cands[:, np.newaxis] + B[np.newaxis, :]
    c_vals = np.max(vals, axis=1)
    idx = np.argmin(c_vals)
    return float(t_cands[idx]), float(c_vals[idx])


def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _do_slsqp(x0, bounds, target, alpha, maxiter):
    def obj(x):
        return _smooth_max(_corr(x), alpha)
    def eq_fun(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eq_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return res.x
    except Exception:
        return x0
