import numpy as np
from scipy.optimize import minimize
import time

_state = {}

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def obj_direct(h):
    n = _state['n']
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    alpha = _state['alpha']
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    val = m + np.log(np.sum(np.exp(s))) / alpha
    val += _state['pen'] * (h.sum() - target) ** 2
    # top-k flattening
    k = _state['topk']
    if k > 1:
        sc = np.sort(corr)[::-1][:k]
        val += _state['flat_w'] * (np.max(sc) - np.min(sc))
    return val

def do_lbfgs_sigmoid(h0, n, alpha, topk, flat_w, pen, maxiter):
    """L-BFGS-B in sigmoid parameterization."""
    _state['n'] = n
    _state['alpha'] = alpha
    _state['topk'] = topk
    _state['flat_w'] = flat_w
    _state['pen'] = pen
    target = n / 2.0
    # Convert h0 to z: h = 1/(1+exp(-z)) => z = log(h/(1-h))
    h_clipped = np.clip(h0, 1e-6, 1-1e-6)
    z0 = np.log(h_clipped / (1 - h_clipped))
    
    def z_obj(z):
        h = 1.0 / (1.0 + np.exp(-np.clip(z, -30, 30)))
        return obj_direct(h)
    
    try:
        res = minimize(z_obj, z0, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-25, 'gtol': 1e-14})
        h_new = 1.0 / (1.0 + np.exp(-np.clip(res.x, -30, 30)))
        return repair(h_new, n)
    except Exception:
        return h0

def do_lbfgs_direct(h0, n, alpha, topk, flat_w, pen, maxiter):
    """L-BFGS-B directly on h with box constraints."""
    _state['n'] = n
    _state['alpha'] = alpha
    _state['topk'] = topk
    _state['flat_w'] = flat_w
    _state['pen'] = pen

    except Exception:

def fine_coord_descent(h, n, t_end, rng):
    """Very fine coordinate descent."""
    h = h.copy()
    best_c5 = compute_c5(h, n)
    # Multiple scales of steps
    for scale in [1.0, 0.5, 0.25, 0.1, 0.04, 0.01, 0.004, 0.001, 0.0004, 0.0001, 0.00004, 0.00001]:
        if time.time() > t_end:
            break
        n_no_improve = 0
        max_no_improve = max(200, n * 10)
        while n_no_improve < max_no_improve and time.time() < t_end:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            improved = False
            for d in [scale, scale * 0.5, scale * 0.25]:
                for sign in [1, -1]:
                    delta = sign * d
                    ni = hi + delta
                    nj = hj - delta
                    if ni < -1e-12 or ni > 1.0 + 1e-12:
                        continue
                    if nj < -1e-12 or nj > 1.0 + 1e-12:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best_c5 - 1e-16:
                        best_c5 = c5
                        improved = True
                        break
                    h[i], h[j] = hi, hj
                if improved:
                    break
            if improved:
                n_no_improve = 0
            else:
                n_no_improve += 1
    return h, best_c5

    h = h.copy()
    target = n / 2.0
    best_c5 = compute_c5(h, n)
    scale = 0.01
    for it in range(1000000):

import numpy as np
from scipy.optimize import minimize
import time

_state = {}

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(8000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def obj_softmax(h):
    n = _state['n']
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    alpha = _state['alpha']
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    val = m + np.log(np.sum(np.exp(s))) / alpha
    val += _state['pen'] * (h.sum() - target) ** 2
    k = _state['topk']
    if k > 1:
        sc = np.sort(corr)[::-1][:k]
        val += _state['flat_w'] * (np.max(sc) - np.min(sc))
    return val

    _state['n'] = n
    _state['alpha'] = alpha
    _state['topk'] = topk
    _state['flat_w'] = flat_w
    _state['pen'] = pen
    hc = np.clip(h0, 1e-6, 1.0 - 1e-6)
    z0 = np.log(hc / (1.0 - hc))

    def z_obj(z):
        h = 1.0 / (1.0 + np.exp(-np.clip(z, -40, 40)))
        return obj_softmax(h)

    try:
        res = minimize(z_obj, z0, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-25, 'gtol': 1e-15})
        h_new = 1.0 / (1.0 + np.exp(-np.clip(res.x, -40, 40)))
        return repair(h_new, n)
    except Exception:
        return h0

    _state['flat_w'] = flat_w
    _state['pen'] = pen
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_softmax, h0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-25, 'gtol': 1e-15})
        return repair(res.x, n)
    except Exception:
        return h0

def do_slsqp(h0, n, alpha, topk, flat_w, maxiter):
    _state['n'] = n
    _state['alpha'] = alpha
    _state['topk'] = topk
    _state['flat_w'] = flat_w
    _state['pen'] = 0.0
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

        return h.sum() - target
