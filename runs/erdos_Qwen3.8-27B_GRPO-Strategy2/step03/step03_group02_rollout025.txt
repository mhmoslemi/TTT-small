def _corr(h):

def _c5(h, dx):

def _smooth_max_val(c, alpha):
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad_smooth_max(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c - np.max(c)
    w = np.exp(alpha * cm)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:

        else:

        for idx in order:

                h[idx] += add
                diff -= add
            else:

                h[idx] -= sub

def _nesterov_step(h, n, dx, target, alpha, lr, momentum, max_iter, time_limit):
    """Nesterov accelerated projected gradient on smooth max."""
    best_h = h.copy()
    best_val = _smooth_max_val(_corr(h), alpha)
    y = h.copy()
    v = np.zeros(n)
    t_mom = 1.0
    
    for it in range(max_iter):
        if time.time() > time_limit:
            break

        # Nesterov: y = x + (t-1)/(t+2) * (x - x_prev)

        # Project gradient step

        h_new = _project_onto_simplex_cut(h_new, target)
        v = h.copy()
        h = h_new
        t_mom = t_new
        
        val = _smooth_max_val(_corr(h), alpha)
        if val < best_val:
            best_val = val
            best_h = h.copy()

def _g_vec(h, p, n):
    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _sweep_round(h, n, corr, time_limit, n_coarse=25, n_bisect=35):
    improved = False
    order = np.random.permutation(n)
    n_pairs = 0
    for ai, i in enumerate(order):
        for j in order[ai + 1:]:
            n_pairs += 1
            if time.time() > time_limit:
                return h, corr, improved, n_pairs
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = np.argmin(vals)
            if vals[bi] >= np.max(corr) - 1e-13:
                continue
            lo = max(d_min, ds[max(0, bi - 1)])
            hi_b = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            improved = True
    return h, corr, improved, n_pairs

    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corr(h)) * dx)

def _smooth_max_val(c, alpha):
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _grad_corr_k(h, k):
    """Gradient of c[k] w.r.t. h, where c = correlate(h, 1-h, 'full')."""
    n = len(h)
    g = np.zeros(n)
    # c[k] = sum_i h[i+k]*(1-h[i]) for valid i
    # dc[k]/dh[j] = (1-h[j-k]) if 0<=j-k<n - h[j+k] if 0<=j+k<n
    for j in range(n):
        val = 0.0
        ik = j - k
        if 0 <= ik < n:
            val += (1.0 - h[ik])
        jk = j + k
        if 0 <= jk < n:
            val -= h[jk]
        g[j] = val
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _sweep_round(h, n, corr, time_limit, n_coarse, n_bisect):
    improved = False
    order = np.random.permutation(n)
    for ai, i in enumerate(order):
        for j in order[ai + 1:]:
            if time.time() > time_limit:
                return h, corr, improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = int(np.argmin(vals))
            if vals[bi] >= np.max(corr) - 1e-13:
                continue
            lo = max(d_min, ds[max(0, bi - 1)])
            hi_b = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            improved = True
    return h, corr, improved

def _subgrad_step(h, n, target, top_m, lr, max_iter, time_limit):
