import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def precompute_G(h, n):
    G = np.zeros((n, 2 * n - 1))
    for m in range(2 * n - 1):
        s = m - (n - 1)
        lo1 = max(0, s)
        hi1 = min(n, n + s)
        if hi1 > lo1:
            G[lo1:hi1, m] += (1.0 - h[lo1 - s:hi1 - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if hi2 > lo2:
            G[lo2:hi2, m] -= h[lo2 + s:hi2 + s]
    return G

def verify_G(h, n):

    corr = compute_corr(h)
    eps = 1e-7
    for _ in range(5):
        i = np.random.randint(n)
        h2 = h.copy()
        h2[i] += eps
        h2 = np.clip(h2, 0, 1)
        corr2 = compute_corr(h2)
        actual = (corr2 - corr) / eps
        diff = np.abs(actual - G[i, :])
        valid_mask = (h[i] + eps <= 1)
        if np.max(diff[valid_mask] if valid_mask.any() else diff) > 0.01:
            return False
    return True

def coord_descent_fast(h, n, n_passes, n_pairs, rng, t_end):
    corr = compute_corr(h).astype(float)
    G = precompute_G(h, n)
    best = np.max(corr)
    dx = 2.0 / n
    n_deltas = 12
    for p in range(n_passes):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(n_pairs):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = G[i, :] - G[j, :]
            d_lo = max(-h[i] + 1e-12, h[j] - 1.0 + 1e-12)
            d_hi = min(1.0 - h[i] - 1e-12, h[j] - 1e-12)
            if d_hi <= d_lo:
                continue
            deltas = np.linspace(d_lo, d_hi, n_deltas)
            new_maxes = np.max(corr[None, :] + deltas[:, None] * d[None, :], axis=1)
            bi = new_maxes.argmin()
            if new_maxes[bi] < best - 1e-13:
                delta = deltas[bi]
                h[i] += delta
                h[j] -= delta
                corr = corr + delta * d
                best = new_maxes[bi]
                G = precompute_G(h, n)
                improved = True
        if not improved and p > 30:
            break
    return h, best * dx, corr

def subgradient_step(h, n, corr, temperature, step_size, rng):
    max_c = np.max(corr)
    w = np.exp((corr - max_c) / temperature)
    w /= w.sum()
    G = precompute_G(h, n)
    grad = G.T @ w
    direction = -grad
    new_h = h + step_size * direction
    new_h = repair(new_h, n)
    return new_h

    return repair(h, n)
