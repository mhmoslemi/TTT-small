import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)


def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _repair(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h


def _grad_lags(h, lags, n):
    g = np.zeros(n)
    for m in lags:
        for i in range(n):
            j = i + m - (n - 1)
            if 0 <= j < n:
                g[i] += (1.0 - h[j])
                g[j] -= h[i]
    return g / len(lags)


def _eq_target(x, target):
    return np.sum(x) - target


def _opt_coord(h, i, n):

            h[i] = new

def _slsqp_step(x0, bounds, target, alpha, maxiter, ftol):

        res = minimize(_smooth_max, x0, args=(alpha,), method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _eq_target, 'args': (target,)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return x0

                    h_new = _repair(h - step * g, n)
                    c5_new = _c5(h_new, dx)
                    if c5_new < c5_cur - 1e-15:
                        h, c5_cur = h_new, c5_new
                        if c5_cur < best_c5:
                            best_c5, best_h = c5_cur, h.copy()

    # Phase 5: Pair local search
    if elapsed() < budget_s - 60:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        deltas = np.array([0.01, 0.005, 0.002, 0.001, -0.01, -0.005, -0.002, -0.001])
        for _pass in range(60):
            if elapsed() > budget_s - 60:
                break
            imp = False
            for i in range(n):
                hi = h[i]
                for j in range(i + 1, n):
                    hj = h[j]
                    for d in deltas:
                        ni, nj = hi + d, hj - d
                        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                            h[i], h[j] = ni, nj
                            c = _c5(h, dx)
                            if c < c5_cur - 1e-14:
                                c5_cur, hi, hj = c, ni, nj
                                imp = True
                            else:
                                h[i], h[j] = hi, hj
                    h[i], h[j] = hi, hj
                h[i] = hi
            if c5_cur < best_c5:
                best_c5, best_h = c5_cur, h.copy()
            if not imp and _pass > 3:
                break

    # Phase 6: Coordinate descent again
    if elapsed() < budget_s - 40:
        h = _coord_descent(best_h.copy(), n, max_sweeps=30)
        h = _repair(h, n)
        c5 = _c5(h, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 7: Fine random pair search
    if elapsed() < budget_s - 15:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        opts = np.array([1e-4, 2e-4, 5e-4, 1e-3, -1e-4, -2e-4, -5e-4, -1e-3])
        while elapsed() < budget_s - 15:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = opts[rng.randint(len(opts))]
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5_cur:
                    c5_cur = c
                    if c5_cur < best_c5:
                        best_c5, best_h = c5_cur, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    # Phase 8: Final SLSQP + coordinate
    if elapsed() < budget_s - 5:
        rx = _slsqp_step(best_h, bounds, target, 200000, 3000, 1e-18)
        hc = _repair(rx, n)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc
        h = _coord_descent(best_h.copy(), n, max_sweeps=20)
        h = _repair(h, n)
        c5 = _c5(h, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Phase 9: 1-h attempt
    if elapsed() < budget_s - 3:
        hf = _repair(1.0 - best_h, n)
        rx = _slsqp_step(hf, bounds, target, 100000, 1500, 1e-18)
        hc = _repair(rx, n)
        c5 = _c5(hc, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, hc
            h = _coord_descent(best_h.copy(), n, max_sweeps=15)
            h = _repair(h, n)
            c5 = _c5(h, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, h

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
