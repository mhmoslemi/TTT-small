from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_c_k(h, k, n):
    # Gradient of c_k = sum_i h[i]*(1-h[i+k]) with respect to h

    # derivative w.r.t. h[j] gets contribution from:
    #   i=j: h[j+k] (if j+k < n)
    #   i+k=j: h[j-k] (if j-k >= 0)
    for i in range(n):
        j = i + k
        if 0 <= j < n:
            g[i] -= h[j]
            g[j] -= h[i]
    return g

    m = np.max(c)

        k = idx - offset
        g += widx * _grad_c_k(h, k, n)
