    g = np.zeros(n)
    for ki in idx:
        lag = ki - (n - 1)
        if lag >= 0:
            g[lag:] += (1.0 - x[:n - lag])
            g[:n - lag] -= x[lag:]
        else:
            m2 = -lag
            g[:n - m2] += (1.0 - x[m2:])
            g[m2:] -= x[:n - m2]
    return g * (2.0 / n) / k

def _slsqp(h0, n, alpha, mi):
    bnds = [(0.0, 1.0)] * n
    def cf(x):
        return np.sum(x) - n / 2.0
    def cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        return minimize(_smax_obj, h0, args=(n, alpha),
                        jac=lambda x: _smax_grad(x, n, alpha),
                        method='SLSQP', bounds=bnds, constraints=cons,
                        options={'maxiter': mi, 'ftol': 1e-18})
    except Exception:
        return None

def _slsqp_topk(h0, n, k, mi):
    bnds = [(0.0, 1.0)] * n
    def cf(x):
        return np.sum(x) - n / 2.0
    def cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        return minimize(lambda x: np.sum(_vals(x, n)[np.argsort(_vals(x, n))[::-1][:k]]),
                        h0,
                        jac=lambda x: _topk_grad(x, n, k),
                        method='SLSQP', bounds=bnds, constraints=cons,
                        options={'maxiter': mi, 'ftol': 1e-16})
    except Exception:
        return None

def _proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = n / 2.0
    for _ in range(30):
        d = t - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = 1.0 - h > 1e-8
        else:
            mask = h > 1e-8
        if not mask.any():
            break
        h[mask] += d * 0.6 / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return _repair(h, n)

def _subgrad_loop(h, n, budget_end, rng, k_start=8, k_end=2):
    c5_cur = _c5(h, n)
    best = c5_cur
    best_h = h.copy()
    t = 0.003
    min_t = 1e-7
    cool = 0.9998
    for step in range(200000):
        if time.time() > budget_end:
            break
        c = _vals(h, n)
        k_now = max(k_end, k_start - step // 5000)
        idx = np.argsort(c)[::-1][:k_now]
        g = np.zeros(n)
        for ki in idx:
            lag = ki - (n - 1)
            if lag >= 0:
                g[lag:] += (1.0 - h[:n - lag])
                g[:n - lag] -= h[lag:]
            else:
                m2 = -lag
                g[:n - m2] += (1.0 - h[m2:])
                g[m2:] -= h[:n - m2]
        g *= (2.0 / n) / k_now
        h_new = _proj(h - t * g, n)
        c5_new = _c5(h_new, n)
        if c5_new < c5_cur or (t > 1e-8 and rng.rand() < np.exp(-(c5_new - c5_cur) / (t * 50))):
            h = h_new
            c5_cur = c5_new
            if c5_cur < best:
                best = c5_cur
                best_h = h.copy()
        else:
            h = h  # revert
        t = max(t * cool, min_t)
    return best_h, best

def _grad_coord_descent(h, n, budget_end, rng):
    c5_cur = _c5(h, n)
    best = c5_cur
    best_h = h.copy()
    n_imp = 0
    for step in range(300000):
        if time.time() > budget_end:
            break
        c = _vals(h, n)
        top_idx = np.argsort(c)[::-1][:3]
        g = np.zeros(n)
        for ki in top_idx:
            lag = ki - (n - 1)
            if lag >= 0:
                g[lag:] += (1.0 - h[:n - lag])
                g[:n - lag] -= h[lag:]
            else:
                m2 = -lag
                g[:n - m2] += (1.0 - h[m2:])
                g[m2:] -= h[:n - m2]
        g *= (2.0 / n) / 3
        # Find best ascending and descending points
        asc = np.argmin(g)
        desc = np.argmax(g)
        if asc == desc:
            asc = (desc + 1) % n
        # Also consider random pair
        if rng.rand() < 0.3:
            asc = rng.randint(n)
            desc = rng.randint(n)
            if asc == desc:
                desc = (desc + 1) % n
        # Try several step sizes
        found = False
        for sgn in [-1, 1]:
            for d in [0.005, 0.01, 0.02, 0.002, 0.001]:
                ds = sgn * d
                ni = h[asc] + ds
                nj = h[desc] - ds
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[asc], h[desc] = ni, nj
                c5_new = _c5(h, n)
                if c5_new < c5_cur - 1e-16:
                    c5_cur = c5_new
                    if c5_cur < best:
                        best = c5_cur
                        best_h = h.copy()
                    n_imp += 1
                    found = True
                    break
                else:
                    h[asc], h[desc] = nj, ni
                    h[asc], h[desc] = h[desc], h[asc]
                    # restore
                    h[asc] = ni
                    h[desc] = nj
                    h[desc] = h[asc] - ds
                    h[asc] = h[desc] + ds
                    break
            if found:
                break
        # Periodic random fine adjustment
        if step % 100 == 99:
            for _ in range(5):
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = rng.choice([0.0005, 0.001, -0.0005, -0.001])
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c5_new = _c5(h, n)
                    if c5_new < c5_cur - 1e-16:
                        c5_cur = c5_new
                        if c5_cur < best:
                            best = c5_cur
                            best_h = h.copy()
                    else:
                        h[i], h[j] = h[j] + d, h[i] - d
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    n = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, n)

    # Phase 1: SLSQP with increasing alpha
    for alpha in [2000, 5000, 15000, 40000, 100000]:
        if elapsed() > budget_s - 600:
            break
        for r in range(3):
            if elapsed() > budget_s - 600:
                break
            if r == 0:
                h_init = best_h.copy()
            else:
                h_init = _repair(best_h + 0.02 * rng.randn(n), n)
            res = _slsqp(h_init, n, alpha, 400)
            if res is not None:
                hc = _repair(res.x, n)
                c5 = _c5(hc, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc.copy()

    # Phase 2: Top-k SLSQP focused on active lags
    for k_val in [4, 3, 2]:
        if elapsed() > budget_s - 500:
            break
        for r in range(3):
            if elapsed() > budget_s - 500:
                break
            h_init = best_h.copy() if r == 0 else _repair(best_h + 0.01 * rng.randn(n), n)
            res = _slsqp_topk(h_init, n, k_val, 300)
            if res is not None:
                hc = _repair(res.x, n)
                c5 = _c5(hc, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc.copy()

    # Phase 3: Subgradient with simulated annealing
    if elapsed() < budget_s - 400:
        best_h, best_c5 = _subgrad_loop(best_h, n, t0 + budget_s - 400, rng, k_start=8, k_end=2)

    # Phase 4: Gradient-guided coordinate descent
    if elapsed() < budget_s - 200:
        best_h, best_c5 = _grad_coord_descent(best_h, n, t0 + budget_s - 200, rng)

    # Phase 5: Another round of SLSQP polish with high alpha
    if elapsed() < budget_s - 100:
        for alpha in [50000, 100000, 200000]:
            if elapsed() > budget_s - 80:
                break
            res = _slsqp(best_h, n, alpha, 500)
            if res is not None:
                hc = _repair(res.x, n)
                c5 = _c5(hc, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc.copy()

    # Phase 6: Ultra-fine coordinate descent
    if elapsed() < budget_s - 40:
        best_h, best_c5 = _grad_coord_descent(best_h, n, t0 + budget_s - 40, rng)

    # Phase 7: Try higher resolution
    if elapsed() < budget_s - 80:
        n2 = 140
        x_old = np.linspace(0, 2.0, n, endpoint=False)
        x_new = np.linspace(0, 2.0, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h, period=2.0)
        h2 = _repair(h2, n2)
        c5_2 = _c5(h2, n2)
        if c5_2 < best_c5 + 0.001:
            for alpha in [20000, 80000]:
                if elapsed() > budget_s - 50:
                    break
                res = _slsqp(h2, n2, alpha, 300)
                if res is not None:
                    hc = _repair(res.x, n2)
                    c5 = _c5(hc, n2)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = hc
                        n = n2
            if n == n2 and elapsed() < budget_s - 30:
                best_h, best_c5 = _grad_coord_descent(best_h, n, t0 + budget_s - 30, rng)

    # Final repair
    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)
    return (best_h, best_c5, n)
