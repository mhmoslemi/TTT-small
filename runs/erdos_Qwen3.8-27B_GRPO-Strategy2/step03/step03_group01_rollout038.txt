```python
def _project(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    # Bisection on tau: find tau such that sum(clip(h-tau, 0, 1)) = target
    lo, hi = -h.max() - 1, h.max() + 1  # wide bounds
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.clip(h - tau, 0, 1)
```

```python
def _optimal_pair_move(h, i, j, n):
    """Find optimal d for h[i]+=d, h[j]-=d that minimizes max lag."""
    # d range: h[i]+d in [0,1] and h[j]-d in [0,1]
    d_lo = max(-h[i], h[j] - 1.0)
    d_hi = min(1.0 - h[i], h[j])
    if d_lo >= d_hi:
        return h, 0.0
    
    # The objective f(d) = max_k c[m(d)] is convex in d
    # Find minimum by ternary search
    for _ in range(40):
        d1 = d_lo + (d_hi - d_lo) / 3.0
        d2 = d_hi - (d_hi - d_lo) / 3.0

    d_opt = 0.5 * (d_lo + d_hi)
    h[i], h[j] = h[i] + d_opt, h[j] - d_opt
    return h, d_opt
```

```python
def _project(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo, hi = float(h.min()) - 1.0, float(h.max()) + 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_lag(h, m):
    n = len(h)
    g = np.zeros(n)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    elif 0 < m < n:
        g[m:] += 1.0 - h[:n - m]
        g[:n - m] -= h[m:]
    elif -n < m < 0:
        mp = -m
        g[:n - mp] += 1.0 - h[mp:]
        g[mp:] -= h[:n - mp]
    return g

def _subgrad(h, tk):
    n = len(h)
    lags = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
    mx = float(lags.max())
    thr = mx - max(0.04 * abs(mx) + 1e-6, 1e-6)
    act = np.where(lags >= thr)[0]
    if len(act) > tk:
        act = np.argsort(lags)[::-1][:tk]
    elif len(act) == 0:
        act = np.array([int(np.argmax(lags))])
    g = np.zeros(n)
    for k in act:
        g += _grad_lag(h, int(k) - n + 1)
    return g / len(act), mx
```
