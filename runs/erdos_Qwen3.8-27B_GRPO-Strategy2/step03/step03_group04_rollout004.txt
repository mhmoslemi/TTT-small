            try:
                rx = _slsqp(x0, bounds, target, alpha, 2500)
                hc = _repair(rx, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc.copy()
            except Exception:
                pass

    # Phase 2: Gradient on top-5 lags
    if _time() < budget_s - 300:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        for ph in range(12):
            step = 0.1 * (0.45 ** ph)
            for it in range(400):
                if _time() > budget_s - 300:
                    break
                c = _corr(h)
                top = np.argsort(c)[-5:]
                g = _grad_lags(h, top, n)
                h_new = _repair(h - step * g, n)
                c5_new = _c5(h_new, dx)
                if c5_new < c5_cur - 1e-15:
                    h, c5_cur = h_new, c5_new
                    if c5_cur < best_c5:
                        best_c5, best_h = c5_cur, h.copy()

    # Phase 3: Coarse pair search
    if _time() < budget_s - 200:
        deltas3 = np.array([0.01, 0.005, 0.002, 0.001, -0.01, -0.005, -0.002, -0.001])
        h, c5_cur, _ = _pair_search(best_h.copy(), _c5(best_h, dx), deltas3, n, dx, t0, budget_s - 120, best_c5, 300)
        if c5_cur < best_c5:
            best_c5, best_h = c5_cur, h.copy()

    # Phase 4: Triple search
    if _time() < budget_s - 150:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        h, c5_cur = _triple_search(h, c5_cur, n, dx, t0, budget_s - 80, best_c5, 30)
        if c5_cur < best_c5:
            best_c5, best_h = c5_cur, h.copy()

    # Phase 5: Fine pair search
    if _time() < budget_s - 60:
        deltas5 = np.array([0.002, 0.001, 0.0005, 0.0002, -0.002, -0.001, -0.0005, -0.0002])
        h, c5_cur, _ = _pair_search(best_h.copy(), _c5(best_h, dx), deltas5, n, dx, t0, budget_s - 40, best_c5, 500)
        if c5_cur < best_c5:
            best_c5, best_h = c5_cur, h.copy()

    # Phase 6: Ultra-fine pair search
    if _time() < budget_s - 30:
        deltas6 = np.array([5e-4, 2e-4, 1e-4, -5e-4, -2e-4, -1e-4])
        h, c5_cur, _ = _pair_search(best_h.copy(), _c5(best_h, dx), deltas6, n, dx, t0, budget_s - 15, best_c5, 800)
        if c5_cur < best_c5:
            best_c5, best_h = c5_cur, h.copy()

    # Phase 7: Random fine polish
    if _time() < budget_s - 10:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        opts = np.array([5e-4, 2e-4, 1e-4, 5e-5, -5e-4, -2e-4, -1e-4, -5e-5])
        while _time() < budget_s - 8:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = opts[rng.randint(len(opts))]
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5_cur:
                    c5_cur = c
                    if c5_cur < best_c5:
                        best_c5, best_h = c5_cur, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    # Phase 8: Final SLSQP polish
    if _time() < budget_s - 3:
        for a in [80000, 200000]:
            if _time() > budget_s - 3:
                break
            try:
                rx = _slsqp(best_h, bounds, target, a, 3000)
                hc = _repair(rx, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc
            except Exception:
                pass

    # Try 1-h final
    if _time() < budget_s - 2:
        try:
            hf = _repair(1.0 - best_h, n)
            c5f = _c5(hf, dx)
            if c5f < best_c5:
                best_c5, best_h = c5f, hf
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
