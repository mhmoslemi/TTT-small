            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c_cur:
                    c_cur = c
                    if c_cur < best_c5:
                        best_c5, best_h = c_cur, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    # Phase 6: Final SLSQP polish
    if bl() > 5:
        try:
            rx = _do_slsqp(best_h, bounds, target, 200000, 3000)
            hc = _repair(rx, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    # Phase 7: One more local search pass after polish
    if bl() > 10:
        h = best_h.copy()
        c_cur = _c5(h, dx)
        deltas = [0.001, 0.0005, 0.0002, -0.001, -0.0005, -0.0002]
        for _pass in range(500):
            if bl() < 5:
                break
            imp = False
            for i in range(n):
                if bl() < 5:
                    break
                hi = h[i]
                for j in range(i + 1, n):
                    hj = h[j]
                    for d in deltas:
                        ni, nj = hi + d, hj - d
                        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                            continue
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < c_cur - 1e-15:
                            c_cur, hi, hj = c, ni, nj
                            imp = True
                        else:
                            h[i], h[j] = hi, hj
                    h[i], h[j] = hi, hj
                h[i] = hi
            if c_cur < best_c5:
                best_c5, best_h = c_cur, h.copy()
            if not imp:
                break

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
