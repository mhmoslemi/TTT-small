                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5_cur:
                    c5_cur = c
                    best_c5 = c
                    best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
