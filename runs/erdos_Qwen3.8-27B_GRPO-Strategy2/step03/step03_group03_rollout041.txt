    g = np.zeros(n)
    if k >= 0:
        g[k:] += 1.0 - h[:n - k]
        g[:n - k] -= h[k:]
    else:
        g[:n + k] += 1.0 - h[-k:]
        g[-k:] -= h[:n + k]
    return g

def _thue_morse_h(n):
    bits = np.array([bin(i).count('1') & 1 for i in range(n)], dtype=float)
    return bits

def _run_lbfgs(h, n, dx, alpha, maxiter, bounds):
    try:
        res = minimize(_lse_obj, h, args=(dx, alpha), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n), _c5(_repair(res.x, n), dx)
    except Exception:
        return h, _c5(h, dx)

def _run_hinge(h, n, dx, T, lam, maxiter, bounds):
    try:
        res = minimize(_hinge_obj, h, args=(dx, T, n, lam), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _subgradient_phase(h, n, dx, t_left_fn, min_t):
    c5_cur = _c5(h, dx)
    momentum = np.zeros(n)
    beta = 0.5
    for it in range(200000):
        if t_left_fn() < min_t:
            break
        c = _corr(h)
        mx = c.max()
        active = np.where(c > 0.95 * mx)[0]
        g = np.zeros(n)
        for m in active:
            g += _grad_lag(h, n, m)
        g /= max(1, len(active))
        step = 0.06 / (1.0 + 0.0008 * it)
        momentum = beta * momentum + (1.0 - beta) * g
        h_new = h - step * momentum
        h_new = _repair(h_new, n)
        c5_new = _c5(h_new, dx)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
    return h, c5_cur

def _coord_descent(h, n, dx, rng, t_left_fn, min_t, n_samples):
    c5_cur = _c5(h, dx)
    c = _corr(h)
    for rnd in range(50000):
        if t_left_fn() < min_t:
            break
        m_star = int(np.argmax(c))
        g = _grad_lag(h, n, m_star)
        best_imp, best_i, best_j, best_d = 0.0, -1, -1, 0.0
        for _ in range(n_samples):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            dg = g[i] - g[j]
            if dg <= 0:
                continue
            dmax = min(1.0 - h[i], h[j])
            if dmax < 1e-10:
                continue
            d = dmax * (0.2 + 0.5 * rng.rand())
            imp = dg * d
            if imp > best_imp:
                best_imp, best_i, best_j, best_d = imp, i, j, d
        if best_i < 0:
            break
        h[best_i] += best_d
        h[best_j] -= best_d
        h = _repair(h, n)
        new_c5 = _c5(h, dx)
        if new_c5 < c5_cur:
            c5_cur = new_c5
            c = _corr(h)
        else:
            h[best_i] -= best_d
            h[best_j] += best_d
            h = _repair(h, n)
    return h, c5_cur

def _fine_search(h, n, dx, rng, t_left_fn, min_t, max_trials):
    c5_cur = _c5(h, dx)
    for _ in range(max_trials):
        if t_left_fn() < min_t:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(n - 1)) % n
        d = rng.choice([-1, 1]) * 10.0 ** rng.uniform(-6, -3)
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < c5_cur:
                c5_cur = c
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    n = len(initial_h_values)
    dx = 2.0 / n
    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n

    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h_init.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: L-BFGS-B with log-sum-exp, increasing alpha
    h_work = h_init.copy()
    for alpha in [500, 2000, 10000, 50000]:
        if tl() < 300:
            break
        h_work, c5 = _run_lbfgs(h_work, n, dx, alpha, 600, bounds)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work

    # Phase 2: Hinge-based optimization (binary search on target)
    if tl() > 250:
        h_work = best_h.copy()
        lo, hi = 0.30, best_c5 + 0.05
        for _bisect in range(10):
            if tl() < 250:
                break
            mid = 0.5 * (lo + hi)
            h_work, c5 = _run_hinge(h_work, n, dx, mid, 50.0, 800, bounds)
            actual = _c5(h_work, dx)
            if actual <= mid + 1e-4:
                lo = mid
            else:
                hi = mid
            if c5 < best_c5:
                best_c5, best_h = c5, h_work

    # Phase 3: Subgradient with momentum on active lags
    if tl() > 180:
        h_work, c5 = _subgradient_phase(best_h.copy(), n, dx, tl, 180)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work

    # Phase 4: Coordinate descent on worst lag
    if tl() > 120:
        h_work, c5 = _coord_descent(best_h.copy(), n, dx, rng, tl, 120, 500)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work

    # Phase 5: SLSQP polish
    if tl() > 60:
        h_work = best_h.copy()
        try:
            def f_sl(x):
                c = _corr(x)
                m = c.max()
                s = np.clip(80000.0 * (c - m), -500.0, 500.0)
                return (m + np.log(np.sum(np.exp(s))) / 80000.0) * dx
            res = minimize(f_sl, h_work, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target}],
                           options={'maxiter': 3000, 'ftol': 1e-18, 'iprint': 0})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    # Phase 6: Fine random pair search
    if tl() > 20:
        h_work, c5 = _fine_search(best_h.copy(), n, dx, rng, tl, 20, 300000)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work

    # Phase 7: Second round of hinge + coord descent on refined solution
    if tl() > 40:
        h_work = best_h.copy()
        T_try = best_c5 - 0.001
        h_work, c5 = _run_hinge(h_work, n, dx, max(0.25, T_try), 100.0, 500, bounds)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work
        if tl() > 25:
            h_work, c5 = _coord_descent(best_h.copy(), n, dx, rng, tl, 25, 300)
            if c5 < best_c5:
                best_c5, best_h = c5, h_work

    # Phase 8: Final ultra-fine search
    if tl() > 10:
        h_work, c5 = _fine_search(best_h.copy(), n, dx, rng, tl, 10, 200000)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
