import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _build_from_t(t, half):
    h = np.zeros(2 * half)
    h[:half] = t
    h[half:] = 1.0 - t
    return h

def _ap_obj(t, alpha, half):
    h = _build_from_t(t, half)
    c = _corr(h)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _ap_grad(t, alpha, half):
    h = _build_from_t(t, half)
    c = _corr(h)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = 2 * half
    g_full = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-15:
            continue
        if L == 0:
            g_full += pk * (1.0 - 2.0 * h)
        elif L > 0:
            g_full[L:] += pk * (1.0 - h[:n-L])
            g_full[:n-L] -= pk * h[L:]
        else:
            M = -L
            g_full[:n-M] += pk * (1.0 - h[M:])
            g_full[M:] -= pk * h[:n-M]
    # Chain rule: dh/dt

    # h[half:] = 1-t, so dh[half:]/dt = -I
