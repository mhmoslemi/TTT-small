def coord_descent_fast(h, n, n_rounds, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    corr = compute_corr(h)
    steps = [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(6 * n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for d in steps:
                for sign in (1, -1):
                    delta = sign * d
                    ni, nj = hi + delta, hj - delta
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-14:
                        best = c5
                        improved = True
                        break
                    h[i], h[j] = hi, hj
                if improved:
                    break
        if not improved and r > 30:
            break
    return h, best

        best_h = repair(np.full(best_n, 0.5), best_n)
        best_c5 = compute_c5(best_h, best_n)

    h_work = best_h.copy()
    n_p = best_n

    # Phase 1: Projected gradient descent with multiple alpha/lr combos
    t1 = t0 + budget_s * 0.28
    for alpha in [800, 3000, 10000, 30000, 80000]:
        if time.time() > t1:
            break
        for lr in [0.003, 0.008, 0.015]:
            if time.time() > t1:
                break
            h_new, c5_new = pgd_loop(h_work.copy(), n_p, alpha, 3, 0.003, lr, 8000, t1)
            if c5_new < best_c5 - 1e-14:
                best_h = h_new.copy()
                best_c5 = c5_new
                h_work = h_new

    # Phase 2: SLSQP refinement with varying params
    t2 = t0 + budget_s * 0.52
    for alpha in [10000, 30000, 80000]:
        if time.time() > t2:
            break
        for topk in [3, 5]:
            if time.time() > t2:
                break
            for fw in [0.0, 0.005, 0.02, 0.08]:
                if time.time() > t2:
                    break
                h_new = do_slsqp(h_work.copy(), n_p, alpha, topk, fw, 600)
                c5_new = compute_c5(h_new, n_p)
                if c5_new < best_c5 - 1e-14:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h_work = h_new
                elif compute_c5(h_new, n_p) < compute_c5(h_work, n_p) - 1e-12:
                    h_work = h_new

    # Phase 3: Extended coordinate descent
    t3 = t0 + budget_s * 0.80
    if time.time() < t3:
        h_cd, c5_cd = coord_descent(h_work, n_p, 600, t3, rng)
        if c5_cd < best_c5 - 1e-14:
            best_h = h_cd.copy()
            best_c5 = c5_cd
            h_work = h_cd

    # Phase 4: Try nearby n values
    t4 = t0 + budget_s * 0.92
    if time.time() < t4:
        for n_alt in [64, 72, 80, 56, 96]:
            if time.time() > t4:
                break
            if n_alt == best_n:
                continue
            h_alt = make_start(n_alt, seed + n_alt * 17, (n_alt + seed) % 5)
            h_alt, c5_alt = pgd_loop(h_alt, n_alt, 15000, 3, 0.01, 0.01, 4000, t4)
            if c5_alt < best_c5 - 1e-14:
                h_alt = do_slsqp(h_alt, n_alt, 30000, 5, 0.02, 400)
                c5_alt = compute_c5(h_alt, n_alt)
                if c5_alt < best_c5 - 1e-14:
                    best_h = h_alt.copy()
                    best_c5 = c5_alt
                    best_n = n_alt

    # Phase 5: Final SLSQP + coord polish on best
    if best_h is not None:
        t5 = t0 + budget_s * 0.97
        if time.time() < t5:
            h_fin = do_slsqp(best_h.copy(), best_n, 80000, 5, 0.03, 600)
            c5_fin = compute_c5(h_fin, best_n)
            if c5_fin < best_c5 - 1e-14:
                best_h = h_fin.copy()
                best_c5 = c5_fin
        t6 = t0 + budget_s * 0.998
        if time.time() < t6:
            h_p, c5_p = coord_descent(best_h, best_n, 300, t6, rng)
            if c5_p < best_c5 - 1e-14:
                best_h = h_p.copy()
                best_c5 = c5_p

    if best_h is None:
        best_h = repair(np.full(best_n, 0.5), best_n)
        best_c5 = compute_c5(best_h, best_n)

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
