def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (t / s)
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h

def _lsq(x, alpha, target, pen):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.log(s.exp().sum()) / alpha + pen * (x.sum() - target) ** 2

def _lbfgs(h, n, alpha, maxiter=400, pen=100.0):

                     bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-9})
        return _repair(r.x, n)
    except Exception:
        return h

def _binary_anneal(n, dx, rng, deadline, T0=2.0, Tmin=0.01, cool=0.9995):
    """Anneal in binary space, then interpolate"""
    h = (rng.rand(n) < 0.5).astype(float)
    # Fix sum
    ones = np.sum(h > 0.5)
    if ones > n//2:
        idx = np.where(h > 0.5)[0]
        for i in rng.choice(idx, ones - n//2, replace=False):
            h[i] = 0
    elif ones < n//2:
        idx = np.where(h < 0.5)[0]
        for i in rng.choice(idx, n//2 - ones, replace=False):
            h[i] = 1
    bc = _c5(h, dx)
    T = T0
    while time.time() < deadline and T > Tmin:
        # Flip two bits (to maintain sum)
        ones_idx = np.where(h > 0.5)[0]
        zeros_idx = np.where(h < 0.5)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        h[i], h[j] = 0.0, 1.0
        c = _c5(h, dx)
        if c < bc or rng.random() < np.exp(-(c - bc) / T):
            bc = c
        else:
            h[i], h[j] = 1.0, 0.0
        T *= cool
    # Smooth: add small random perturbation
    h = h + 0.05 * rng.randn(n)
    h = _repair(h, n)
    return h, _c5(h, dx)

That's NOT non-decreasing (it decreases from 2 to 0). So f is NOT convex!

So the 1D subproblem is NOT convex in general. Ternary search might not work perfectly. However, in practice, the function is "nice enough" that a few local minima are the worst case, and we can handle that with multiple starting points for the 1D search.

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (t / s)
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-14:
                break
    return h

def _lsq(x, alpha, target, pen):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.log(s.exp().sum()) / alpha + pen * (x.sum() - target) ** 2

def _lbfgs(h, n, alpha, maxiter=400, pen=100.0):
    t = n / 2.0
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(_lsq, h, args=(alpha, t, pen), method='L-BFGS-B',
                     bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return _repair(r.x, n)
    except Exception:
        return h

def _slsqp(h, n, alpha, maxiter=400):
    t = n / 2.0
    bnds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': lambda x: float(x.sum() - t)}
    try:
        r = minimize(lambda x: _lsq(x, alpha, t, 0.0), h, method='SLSQP',
                     bounds=bnds, constraints=[con],
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return h

def _coord_descent(h, n, dx, grid, n_passes, deadline):
    """Coordinate descent with 1D grid search per variable."""
    h = h.copy()
    for p in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            old = h[i]
            best_val = _c5(h, dx)
            best_x = old
            for x in np.linspace(0.0, 1.0, grid):
                if x == old:
                    continue
                h[i] = x
                c = _c5(h, dx)
                if c < best_val:
                    best_val = c
                    best_x = x
            h[i] = best_x
            if abs(best_x - old) > 1e-7:
                improved = True
        h = _repair(h, n)
        if not improved:
            break
    return h, _c5(h, dx)

def _pair_local(h, n, dx, rng, deadline, ntries):
    bc = _c5(h, dx)
    h = h.copy()
    scales = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    per = max(500, ntries // len(scales))
    for sc in scales:
        imp = 0
        for _ in range(per):
            if time.time() > deadline:
                return h, bc
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            ni = h[i] + d
            nj = h[j] - d
            if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
                continue
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < bc - 1e-16:
                bc = c
                imp += 1
            else:
                h[i], h[j] = ni - d, nj + d
        if imp == 0 and sc <= 0.0005:
            break
    return h, bc

def _triple_local(h, n, dx, rng, deadline, ntries):
    """Modify 3 points: +d, +d, -2d to maintain sum."""
    bc = _c5(h, dx)
    h = h.copy()
    for _ in range(ntries):
        if time.time() > deadline:
            break
        i, j, k = rng.randint(n), rng.randint(n), rng.randint(n)
        if i == j or j == k or i == k:
            continue
        d = 0.004 * (2.0 * rng.random() - 1.0)
        ni = h[i] + d
        nj = h[j] + d
        nk = h[k] - 2.0 * d
        if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10 or nk < -1e-10 or nk > 1.0 + 1e-10:
            continue
        h[i], h[j], h[k] = ni, nj, nk
        c = _c5(h, dx)
        if c < bc - 1e-16:
            bc = c
        else:
            h[i], h[j], h[k] = ni - d, nj - d, nk + 2.0 * d
    return h, bc

def _gen_init(n, rng):
    kind = rng.randint(5)
    if kind == 0:
        h = np.zeros(n)
        blk = max(2, n // rng.randint(4, 10))
        for i in range(0, n, blk):
            h[i:i+blk] = 1.0
    elif kind == 1:
        t = np.arange(n) / n
        f = rng.randint(2, 6)
        ph = rng.uniform(0, 6.28)
        h = (0.5 + 0.5 * np.sin(2.0 * np.pi * f * t + ph) > 0.5).astype(float)
    elif kind == 2:
        h = (rng.rand(n) < 0.5).astype(float)
    elif kind == 3:
        gr = 0.6180339887
        h = (((np.arange(n) * gr) % 1.0) > 0.5).astype(float)
    else:
        h = rng.rand(n) * 0.3
    return _repair(h, n)

    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
