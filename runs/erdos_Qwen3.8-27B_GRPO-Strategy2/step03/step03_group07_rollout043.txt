            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def make_start(n, seed_val, kind):
    rng = np.random.RandomState(seed_val)
    if kind == 0:
        h = 0.5 + 0.3 * rng.randn(n)
    elif kind == 1:
        h = np.where(np.arange(n) < n // 2, 1.0, 0.0) + 0.1 * rng.randn(n)
    elif kind == 2:
        f = rng.randint(2, 8)
        ph = rng.uniform(0, 6.283)
        h = 0.5 + 0.35 * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 3:
        block = max(2, n // 5)
        h = np.zeros(n)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
        h = 0.6 * h + 0.3 * rng.randn(n)
    elif kind == 4:
        h = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * np.arange(n) / n)
    else:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * 5 * np.arange(n) / n) + 0.15 * np.cos(2 * np.pi * 2 * np.arange(n) / n)
    return repair(h, n)

def obj_func(h):
    corr = compute_corr(h)
    m = np.max(corr)
    a = _obj_state['alpha']
    s = a * (corr - m)
    s = np.clip(s, -500, 500)
    val = m + np.log(np.sum(np.exp(s))) / a
    k = _obj_state['topk']
    if k > 1:
        sc = np.sort(corr)[::-1][:k]
        val += _obj_state['flat_w'] * np.sum((sc - sc.mean())**2)
    return val

def eq_con(h):
    return h.sum() - _obj_state['target']

def slsqp_opt(h, n, alpha, topk, flat_w, maxiter):
    _obj_state['alpha'] = alpha
    _obj_state['topk'] = topk
    _obj_state['flat_w'] = flat_w
    _obj_state['target'] = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con}]
    try:
        res = minimize(obj_func, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return repair(res.x, n)
    except Exception:
        return h

def targeted_cd(h, n, t_end, rng, n_passes=200):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.04, 0.02, 0.01, 0.005, 0.003, 0.001, 0.0005]
    for p in range(n_passes):
        if time.time() > t_end:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[::-1][:6]
        improved = False
        for _ in range(6 * n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for d in steps:
                for sign in (1, -1):
                    ni = hi + sign * d
                    nj = hj - sign * d
                    if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                        continue
                    h[i], h[j] = ni, nj
                    c5_new = compute_c5(h, n)
                    if c5_new < best - 1e-15:
                        best = c5_new
                        improved = True
                        break
                    h[i], h[j] = hi, hj
                if improved:
                    break
        if not improved and p > 40:
            mask = rng.rand(n) < 0.08
            h[mask] += 0.008 * rng.randn(int(mask.sum()))
            h = repair(h, n)
    return h, best

def anneal_cd(h, n, t_end, rng, n_rounds=200):
    h = h.copy()
    best = compute_c5(h, n)
    best_h = h.copy()
    T = 0.015
    T_min = 0.0005
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        T = max(T_min, T * 0.97)
        for _ in range(5 * n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = T * rng.randn()
            hi, hj = h[i], h[j]
            ni, nj = hi + d, hj - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5_new = compute_c5(h, n)
            if c5_new < best:
                best = c5_new
                best_h = h.copy()
            elif rng.rand() > np.exp(-(c5_new - best) / max(T, 1e-8)):
                h[i], h[j] = hi, hj
    return best_h, best

def fine_cd(h, n, t_end, rng, n_passes=100):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for p in range(n_passes):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(10 * n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for d in steps:
                for sign in (1, -1):
                    ni = hi + sign * d
                    nj = hj - sign * d
                    if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                        continue
                    h[i], h[j] = ni, nj
                    c5_new = compute_c5(h, n)
                    if c5_new < best - 1e-15:
                        best = c5_new
                        improved = True
                        break
                    h[i], h[j] = hi, hj
                if improved:
                    break
        if not improved:
            break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed + 9999)
    best_h = None
    best_c5 = np.inf
    best_n = 70

    init_h = None
    init_n = 70
    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
        init_n = len(init_h)
    except NameError:
        pass

    if init_h is not None:
        init_h = repair(init_h, init_n)
        best_h = init_h.copy()
        best_c5 = compute_c5(init_h, init_n)
        best_n = init_n

    n_p = best_n
    h_work = best_h.copy() if best_h is not None else repair(np.full(n_p, 0.5), n_p)

    # Phase 1: SLSQP escalation
    for alpha, tk, fw in [(2000,3,0.01),(5000,3,0.01),(10000,5,0.02),(30000,5,0.03),(80000,7,0.04)]:
        if time.time() - t0 > budget_s * 0.28:
            break
        h_new = slsqp_opt(h_work.copy(), n_p, alpha, tk, fw, 500)
        c5_new = compute_c5(h_new, n_p)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new.copy()
            best_c5 = c5_new
        h_work = h_new if c5_new < compute_c5(h_work, n_p) else h_work

    # Phase 2: Targeted CD
    t_end2 = t0 + budget_s * 0.50
    h_cd, c5_cd = targeted_cd(h_work.copy(), n_p, t_end2, rng, 250)
    if c5_cd < best_c5 - 1e-14:
        best_h = h_cd.copy()
        best_c5 = c5_cd
    h_work = h_cd if c5_cd < compute_c5(h_work, n_p) else h_work

    # Phase 3: Annealing to escape
    t_end3 = t0 + budget_s * 0.68
    h_ann, c5_ann = anneal_cd(h_work.copy(), n_p, t_end3, rng, 180)
    if c5_ann < best_c5 - 1e-14:
        best_h = h_ann.copy()
        best_c5 = c5_ann
    h_work = h_ann if c5_ann < compute_c5(h_work, n_p) else h_work

    # Phase 4: SLSQP polish
    if time.time() - t0 < budget_s * 0.82:
        h_fin = slsqp_opt(h_work.copy(), n_p, 60000, 5, 0.03, 700)
        c5_fin = compute_c5(h_fin, n_p)
        if c5_fin < best_c5 - 1e-14:
            best_h = h_fin.copy()
            best_c5 = c5_fin
        h_work = h_fin if c5_fin < compute_c5(h_work, n_p) else h_work

    # Phase 5: Fine CD
    t_end5 = t0 + budget_s * 0.93
    h_fin2, c5_fin2 = fine_cd(h_work.copy(), n_p, t_end5, rng, 150)
    if c5_fin2 < best_c5 - 1e-14:
        best_h = h_fin2.copy()
        best_c5 = c5_fin2

    # Phase 6: Try alternate n
    if time.time() - t0 < budget_s * 0.96:
        for n_alt in [64, 80]:
            if time.time() - t0 > budget_s * 0.96:
                break
            if n_alt == best_n:
                continue
            h_alt = make_start(n_alt, seed + n_alt * 11, 4)
            h_alt = slsqp_opt(h_alt, n_alt, 15000, 5, 0.02, 500)
            c5_alt = compute_c5(h_alt, n_alt)
            if c5_alt < best_c5 - 1e-14:
                best_h = h_alt.copy()
                best_c5 = c5_alt
                best_n = n_alt

    # Phase 7: Final polish
    if best_h is not None and time.time() - t0 < budget_s * 0.99:
        h_p = slsqp_opt(best_h.copy(), best_n, 100000, 7, 0.04, 400)
        c5_p = compute_c5(h_p, best_n)
        if c5_p < best_c5 - 1e-14:
            best_h = h_p
            best_c5 = c5_p
        h_p2, c5_p2 = fine_cd(best_h, best_n, t0 + budget_s * 0.995, rng, 80)
        if c5_p2 < best_c5 - 1e-14:
            best_h = h_p2
            best_c5 = c5_p2

    if best_h is None:
        best_n = 70
        best_h = repair(np.full(best_n, 0.5), best_n)
        best_c5 = compute_c5(best_h, best_n)

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
