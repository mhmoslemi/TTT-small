        else:

    d = 0.5 * (lo + hi_b)
    h = h.copy()
    h[i] = hi + d
    h[j] = hj - d
    corr = corr + d * g
    return h, corr, True

def _sweep_round(h, n, corr, tlim, nc, nb, order=None):
    if order is None:
        order = np.random.permutation(n)
    imp = False
    for ai in range(n):
        for aj in range(ai + 1, n):
            if time.time() > tlim:
                return h, corr, imp
            h, corr, it = _sweep_pair(h, n, corr, int(order[ai]), int(order[aj]), nc, nb)
            imp = imp or it
    return h, corr, imp

def _subspace_opt(h, n, target, dx, npick, tlim):
    """Optimize a subset of coordinates most affecting the max correlation."""
    corr = _corr(h)
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    lag = max_idx - (n - 1)
    h_cur = h.copy()
    for _ in range(50):
        if time.time() > tlim:
            break
        grad = _sm_grad(h_cur, 500)
        idx = np.argsort(-np.abs(grad))[:npick]
        sub = h_cur[idx].copy()
        def _sub_obj(x, idx=idx, h_cur=h_cur, n=n):
            x_full = h_cur.copy()
            x_full[idx] = x
            return _sm_obj(x_full, 500)
        def _sub_eq(x, idx=idx, h_cur=h_cur, n=n, target=target):
            return float(np.sum(h_cur) - sum(h_cur[idx]) + float(np.sum(x))) - target
        def _sub_jac(x, idx=idx):
            return np.ones(len(idx))
        cons = [{'type': 'eq', 'fun': _sub_eq, 'jac': _sub_jac, 'args': ()}]
        bnds = [(0.0, 1.0)] * len(idx)
        try:
            res = minimize(_sub_obj, sub, method='SLSQP', bounds=bnds,
                           constraints=cons, options={'maxiter': 100, 'ftol': 1e-14})
            h_cur[idx] = np.clip(res.x, 0, 1)
            c5_new = _c5(h_cur, dx)
            if c5_new >= _c5(h, dx) + 1e-15:
                h_cur[idx] = sub
        except Exception:
            pass
        h_cur = _repair(h_cur, n)
    return h_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _c5(best_h, dx)

    # Phase 1: PGD with increasing alpha
    for alpha in [50, 100, 300, 800, 2000]:
        if time.time() - t0 > budget_s * 0.18:
            break
        hp = _pgd(best_h, alpha, n, max_iter=3000)
        c5 = _c5(hp, dx)
        if c5 < best_c5 - 1e-15:
            best_c5, best_h = c5, hp.copy()

    # Phase 2: SLSQP refinement
    for alpha in [300, 800, 2000, 5000]:
        if time.time() - t0 > budget_s * 0.32:
            break
        hc = _slsqp(best_h, n, target, bounds, alpha, 600)
        c5 = _c5(hc, dx)
        if c5 < best_c5 - 1e-15:
            best_c5, best_h = c5, hc.copy()

    # Phase 3: Pair sweeps + periodic SLSQP + perturbation
    stall = 0
    it = 0
    while time.time() - t0 < budget_s - 130:
        it += 1
        h = best_h.copy()
        corr = _corr(h)
        tlim = min(time.time() + 35.0, t0 + budget_s - 130)
        order = np.random.permutation(n)
        h, corr, imp = _sweep_round(h, n, corr, tlim, 25, 35, order)
        if imp:
            c5 = _c5(h, dx)
            if c5 < best_c5 - 1e-15:
                best_c5, best_h = c5, h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if it % 3 == 0 and time.time() - t0 < budget_s - 160:
            alpha = 500 + 300 * it
            hc = _slsqp(best_h, n, target, bounds, alpha, 500)
            hc_corr = _corr(hc)
            hc, hc_corr, _ = _sweep_round(hc, n, hc_corr, time.time() + 15, 20, 30)
            c5t = _c5(hc, dx)
            if c5t < best_c5 - 1e-15:
                best_c5, best_h = c5t, hc.copy()
                stall = 0

        if stall >= 3 and time.time() - t0 < budget_s - 160:
            scale = min(0.06, 0.008 * stall)
            hp = best_h + scale * np.random.randn(n)
            hp = _repair(hp, n)
            hp = _pgd(hp, 800, n, max_iter=2000)
            hp_corr = _corr(hp)
            hp, hp_corr, _ = _sweep_round(hp, n, hp_corr, time.time() + 18, 25, 35)
            c5t = _c5(hp, dx)
            if c5t < best_c5 - 1e-15:
                best_c5, best_h = c5t, hp.copy()
            stall = 0

    # Phase 4: Final precise sweeps
    if time.time() - t0 < budget_s - 20:
        for _ in range(4):
            if time.time() - t0 > budget_s - 15:
                break
            h = best_h.copy()
            corr = _corr(h)
            h, corr, _ = _sweep_round(h, n, corr, time.time() + 14, 31, 42)
            c5 = _c5(h, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()

    # Phase 5: Final SLSQP
    if time.time() - t0 < budget_s - 4:
        try:
            hc = _slsqp(best_h, n, target, bounds, 8000, 1000)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc.copy()
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
