```python
from scipy.optimize import minimize

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:

                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

    """Compute c_k for all lags k when changing h[i]."""
    # c_k = [0<=i+k<n ? (1-h[i+k]) : 0] - [0<=i-k<n ? h[i-k] : 0]

    if k2_min <= k2_max:

        c[ks + n - 1] -= h[i - ks]
    return c

        return 0.0
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(corr + m1 * c)
        f2 = np.max(corr + m2 * c)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return 0.5 * (lo + hi)

def _coord_sweep(h, n, corr, time_limit, n_iter=50):
    """Single-coordinate sweeps until time limit."""
    improved = False
    n_sweeps = 0
    while time.time() < time_limit:
        any_improve = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > time_limit:
                break
            hi = h[i]
            d_min = -hi
            d_max = 1.0 - hi
            if d_max - d_min < 1e-12:
                continue
            c = _coord_coef(h, i, n)
            delta = _optimal_delta(corr, c, d_min, d_max, n_iter)
            new_max = np.max(corr + delta * c)
            cur_max = np.max(corr)
            if new_max < cur_max - 1e-13:
                h[i] += delta
                corr = corr + delta * c
                any_improve = True
                improved = True
        n_sweeps += 1
        if not any_improve:
            break
    return h, corr, improved, n_sweeps

    h = np.clip(h, 0.0, 1.0)

    try:

            if 0 <= i - k < n:
                g -= h[i - k]
            grad[i] += w_k * g
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _coord_coef(h, i, n):
    nk = 2 * n - 1
    c = np.zeros(nk)
    k1_min = max(0, -i)
    k1_max = min(n - 1, n - 1 - i)
    if k1_min <= k1_max:
        ks = np.arange(k1_min, k1_max + 1)
        c[ks + n - 1] += (1.0 - h[i + ks])
    k2_min = max(0, i - (n - 1))
    k2_max = min(n - 1, i)
    if k2_min <= k2_max:
        ks = np.arange(k2_min, k2_max + 1)
        c[ks + n - 1] -= h[i - ks]
    return c

def _optimal_delta(corr, c, d_min, d_max, n_iter=45):
    lo, hi = d_min, d_max
    if hi - lo < 1e-14:
        return 0.0
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr + m1 * c) < np.max(corr + m2 * c):
            hi = m2
        else:
            lo = m1
    return 0.5 * (lo + hi)

def _coord_sweep(h, n, corr, time_limit, n_iter=45):
    improved = False
    total_sweeps = 0
    while time.time() < time_limit:
        any_imp = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > time_limit:
                break
            hi = h[i]
            d_min = -hi
            d_max = 1.0 - hi
            if d_max - d_min < 1e-12:
                continue
            c = _coord_coef(h, i, n)
            delta = _optimal_delta(corr, c, d_min, d_max, n_iter)
            if abs(delta) < 1e-14:
                continue
            new_max = np.max(corr + delta * c)
            if new_max < np.max(corr) - 1e-13:
                h[i] += delta
                corr = corr + delta * c
                any_imp = True
                improved = True
        total_sweeps += 1
        if not any_imp:
            break
    return h, corr, improved, total_sweeps

def _pair_refine(h, n, corr, time_limit, n_coarse=20):
    improved = False
    for _ in range(5):
        if time.time() > time_limit:
            break
        order = np.random.permutation(n)
        any_imp = False
        for ai in range(n):
            i = order[ai]
            if time.time() > time_limit:
                break
            for bi in range(ai + 1, min(ai + 8, n)):
                j = order[bi]
                hi, hj = h[i], h[j]
                d_min = max(-hi, hj - 1.0)
                d_max = min(1.0 - hi, hj)
                if d_max - d_min < 1e-12:
                    continue
                ci = _coord_coef(h, i, n)
                cj = _coord_coef(h, j, n)
                gd = ci - cj
                ds = np.linspace(d_min, d_max, n_coarse)
                vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
                bi2 = np.argmin(vals)
                if vals[bi2] >= np.max(corr) - 1e-13:
                    continue
                lo = max(d_min, ds[max(0, bi2 - 1)])
                hi_b = min(d_max, ds[min(n_coarse - 1, bi2 + 1)])
                for _ in range(30):
                    mid = 0.5 * (lo + hi_b)
                    if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
                        hi_b = mid
                    else:
                        lo = mid
                d_best = 0.5 * (lo + hi_b)
                h[i] = hi + d_best
                h[j] = hj - d_best
                corr = corr + d_best * gd
                any_imp = True
                improved = True
        if not any_imp:
            break
    return h, corr, improved

def _fourier_init(n, scale=0.35):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, 25):
        amp = scale / (m * np.log(m + 1))
        h += amp * np.cos(np.pi * m * x)
        h += amp * 0.5 * np.sin(np.pi * m * x + 0.3 * m)
    return _repair(np.clip(h, 0.0, 1.0), n)

def _slsqp_step(h, n, dx, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return h

def _optimize_from(h, n, dx, target, bounds, t0, budget_s, min_remain=60):
    best_h = _repair(h.copy(), n)
    best_c5 = _c5(best_h, dx)
    # SLSQP
    if time.time() - t0 < budget_s - min_remain - 40:
        for alpha in [100, 300, 800]:
            if time.time() - t0 > budget_s - min_remain - 30:
                break
            hc = _slsqp_step(best_h, n, dx, target, bounds, alpha, 300)
            c5 = _c5(hc, dx)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = hc.copy()
    # Coord sweeps
    if time.time() - t0 < budget_s - min_remain:
        hc = best_h.copy()
        corr = _corr(hc)
        hc, corr, _, _ = _coord_sweep(hc, n, corr, min(time.time() + 25, budget_s - min_remain))
        c5 = _c5(hc, dx)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = hc.copy()
    # Pair refine
    if time.time() - t0 < budget_s - min_remain:
        hc = best_h.copy()
        corr = _corr(hc)
        hc, corr, _, _ = _pair_refine(hc, n, corr, min(time.time() + 10, budget_s - min_remain))
        c5 = _c5(hc, dx)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = hc.copy()
    return best_h, best_c5
```
