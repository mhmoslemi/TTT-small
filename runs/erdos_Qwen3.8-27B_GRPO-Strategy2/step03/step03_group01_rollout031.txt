import time

    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for idx in order:

                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
        else:
            order = np.argsort(h)
            for idx in order:
                if diff >= -1e-12: break
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def _eval_corr(h):
    return np.correlate(h, 1.0-h, mode='full')

def _sf(x, a):
    c = _eval_corr(x)
    m = c.max()
    s = np.clip(a*(c-m), -500, 500)
    return m + np.log(np.sum(np.exp(s)))/a

    n = len(x)
    c = _eval_corr(x)
    m = c.max()
    s = np.clip(a*(c-m), -500, 500)
    w = np.exp(s - s.max())

    g = np.zeros(n)

        if w[k] < 1e-14: continue

            g[idx1] += wl * (1.0 - x[idx1-M])
            idx2 = np.arange(0, n-M)
            g[idx2] -= wl * x[idx2+M]

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                if diff <= 1e-12:
                    break
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
        else:
            order = np.argsort(h)
            for idx in order:
                if diff >= -1e-12:
                    break
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _sf(x, a):
    c = _eval_corr(x)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a

def _sf_jac(x, a):
    n = len(x)
    c = _eval_corr(x)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        L = k - (n - 1)
        wl = w[k]
        if L >= 0:
            i1 = np.arange(0, n - L)
            g[i1] += wl * (1.0 - x[i1 + L])
            i2 = np.arange(L, n)
            g[i2] -= wl * x[i2 - L]
        else:
            M = -L
            i1 = np.arange(M, n)
            g[i1] += wl * (1.0 - x[i1 - M])
            i2 = np.arange(0, n - M)
            g[i2] -= wl * x[i2 + M]
    return g

def _slsqp(h, n, alpha, iters):
    bnds = [(0.0, 1.0)] * n
    v = cp.Variable(n)
    s = cp.sum(v) - n / 2.0
    def _c(x):
        return np.sum(x) - n / 2.0
    def _cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _c, 'jac': _cj}]
    try:
        res = minimize(_sf, h, _sf_jac, method='SLSQP', bounds=bnds,
                       constraints=cons, args=(alpha,),
                       options={'maxiter': iters, 'ftol': 1e-19})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _grad_lag(h, k):
    """Gradient of correlation at lag k (index k in full correlation)"""
    n = len(h)
    L = k - (n - 1)
    g = np.zeros(n)
    if L >= 0:
        for i in range(n - L):
            g[i] += (1.0 - h[i + L])
            g[i + L] -= h[i]
    else:
        M = -L
        for i in range(n - M):
            g[i + M] += (1.0 - h[i])
            g[i] -= h[i + M]
    return g

def _oa_refine(h, n, top_k, max_iters, time_limit):
    """Outer approximation: LP at each step"""
    t0 = time.time()
    cur = h.copy()
    cur_f = _eval(cur)
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        c = _eval_corr(cur)
        m = c.max()
        # Get top_k lags (and their neighbors for robustness)
        idx = np.argsort(c)[::-1][:top_k]
        # Build LP
        vh = cp.Variable(n)
        vt = cp.Variable()
        cons = [vh >= 0, vh <= 1, cp.sum(vh) == n / 2]
        for k in idx:
            gk = _grad_lag(cur, k)
            ck = c[k]
            # c_k ≈ ck + gk·(h - cur) = (ck - gk·cur) + gk·h
            rhs = ck - gk @ cur
            cons.append(cp.dot(gk, vh) - vt <= rhs)
        # Also add the current max as a lower bound on t for validity
        # (ensures t >= max actual c_k at the new point)
        prob = cp.Problem(cp.Minimize(vt), cons)
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                break
        if vh.value is None:
            break
        new_h = np.round(vh.value, 10)
        new_h = _repair(new_h, n)
        new_f = _eval(new_h)
        if new_f < cur_f - 1e-12:
            cur = new_h
            cur_f = new_f
        elif np.linalg.norm(new_h - cur) < 1e-8:
            break
        else:
            # Accept anyway if close
            if abs(new_f - cur_f) < 1e-10:
                cur = new_h
    return cur, cur_f

def _sa_refine(h, n, rng, t_start, t_end, iters, alpha):
    """Simulated annealing with soft-max energy"""
    cur = h.copy()
    cur_e = _sf(cur, alpha)
    cur_f = _eval(cur)
    best = cur.copy()
    best_f = cur_f
    for it in range(iters):
        T = t_start * (t_end / t_start) ** (it / max(iters, 1))
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.1, 0.1)
        ni, nj = cur[i] + d, cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        cur[i], cur[j] = ni, nj
        new_e = _sf(cur, alpha)
        new_f = _eval(cur)
        if new_e < cur_e:
            cur_e = new_e
            if new_f < best_f:
                best_f = new_f
                best = cur.copy()
        else:
            if rng.random() < np.exp(-(new_e - cur_e) / max(T, 1e-15)):
                cur_e = new_e
                if new_f < best_f:
                    best_f = new_f
                    best = cur.copy()
            else:
                cur[i], cur[j] = ni - d, nj + d
    return best, best_f

def _pair_descent(h, n, rng, time_limit):
    """Adaptive pair coordinate descent"""
    t0 = time.time()
    cur = h.copy()
    cur_f = _eval(cur)
    step = 0.03
    no_imp = 0
    while time.time() - t0 < time_limit:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = step * rng.randn()
        ni, nj = cur[i] + d, cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        cur[i], cur[j] = ni, nj
        nf = _eval(cur)
        if nf < cur_f - 1e-15:
            cur_f = nf
            no_imp = 0
            step = min(step * 1.05, 0.01)
        else:
            cur[i], cur[j] = cur[i] - d, cur[j] + d
            no_imp += 1
            if no_imp > 300:
                step *= 0.7
                no_imp = 0
                if step < 1e-6:
                    break
    return cur, cur_f

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                if diff <= 1e-12:
                    break
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
        else:
            order = np.argsort(h)
            for idx in order:
                if diff >= -1e-12:
                    break
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _sf(x, a):
    c = _eval_corr(x)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a

def _sf_jac(x, a):
    n = len(x)
    c = _eval_corr(x)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        L = k - (n - 1)
        wl = w[k]
        if L >= 0:
            i1 = np.arange(0, n - L)
            g[i1] += wl * (1.0 - x[i1 + L])
            i2 = np.arange(L, n)
            g[i2] -= wl * x[i2 - L]
        else:
            M = -L
            i1 = np.arange(M, n)
            g[i1] += wl * (1.0 - x[i1 - M])
            i2 = np.arange(0, n - M)
            g[i2] -= wl * x[i2 + M]
    return g

def _csum(x):
    return np.sum(x) - len(x) / 2.0

def _csum_jac(x):
    return np.ones(len(x))

def _slsqp(h, n, alpha, iters):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _csum, 'jac': _csum_jac}]
    try:
        res = minimize(_sf, h, _sf_jac, method='SLSQP', bounds=bnds,
                       constraints=cons, args=(alpha,),
                       options={'maxiter': iters, 'ftol': 1e-19})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _grad_lag(h, k):
    n = len(h)
    L = k - (n - 1)
    g = np.zeros(n)
    if L >= 0:
        for i in range(n - L):
            g[i] += (1.0 - h[i + L])
            g[i + L] -= h[i]
    else:
        M = -L
        for i in range(n - M):
            g[i + M] += (1.0 - h[i])
            g[i] -= h[i + M]
    return g

def _oa_refine(h, n, top_k, max_iters, time_limit):
    t0 = time.time()
    cur = h.copy()
    cur_f = _eval(cur)
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        c = _eval_corr(cur)
        idx = np.argsort(c)[::-1][:top_k]
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_rows = []
        b_vals = []
        for k in idx:
            gk = _grad_lag(cur, k)
            rhs = c[k] - np.dot(gk, cur)
            row = np.zeros(n + 1)
            row[:n] = gk
            row[n] = -1.0
            A_rows.append(row)
            b_vals.append(rhs)
        A_ub = np.array(A_rows)
        b_ub = np.array(b_vals)
        A_eq = np.array([[np.ones(n), 0.0]])
        b_eq = np.array([n / 2.0])
        bounds = [(0, 1)] * n + [(None, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            break
        if not res.success or res.x is None:
            break
        new_h = _repair(res.x[:n], n)
        new_f = _eval(new_h)
        if new_f < cur_f - 1e-12:
            cur_f = new_f
            cur = new_h
        elif abs(new_f - cur_f) < 1e-10 and np.linalg.norm(new_h - cur) < 1e-6:
            break
    return cur, cur_f

def _sa_refine(h, n, rng, t_start, t_end, iters, alpha):
    cur = h.copy()
    cur_e = _sf(cur, alpha)
    cur_f = _eval(cur)
    best = cur.copy()
    best_f = cur_f
    for it in range(iters):
        T = t_start * (t_end / t_start) ** (it / max(iters - 1, 1))
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        scale = 0.1 if it < iters // 3 else 0.03
        d = scale * rng.randn()
        ni, nj = cur[i] + d, cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        cur[i], cur[j] = ni, nj
        new_e = _sf(cur, alpha)
        new_f = _eval(cur)
        if new_e < cur_e:
            cur_e = new_e
            if new_f < best_f:
                best_f = new_f
                best = cur.copy()
        else:
            if rng.random() < np.exp(-(new_e - cur_e) / max(T, 1e-15)):
                cur_e = new_e
                if new_f < best_f:
                    best_f = new_f
                    best = cur.copy()
            else:
                cur[i], cur[j] = ni - d, nj + d
    return best, best_f

def _pair_descent(h, n, rng, time_limit):
    t0 = time.time()
    cur = h.copy()
    cur_f = _eval(cur)
    step = 0.03
    no_imp = 0
    while time.time() - t0 < time_limit:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = step * rng.randn()
        ni, nj = cur[i] + d, cur[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        cur[i], cur[j] = ni, nj
        nf = _eval(cur)
        if nf < cur_f - 1e-15:
            cur_f = nf
            no_imp = 0
            step = min(step * 1.05, 0.008)
        else:
            cur[i], cur[j] = cur[i] - d, cur[j] + d
            no_imp += 1
            if no_imp > 400:
                step *= 0.65
                no_imp = 0
                if step < 5e-7:
                    break
    return cur, cur_f

def _binary_ls(h, n, rng, time_limit):
    t0 = time.time()
    b = (h > 0.5).astype(float)
    target = n // 2
    ones = b.sum()
    if ones > target:
        idx1 = np.where(b == 1)[0]
        order = rng.permutation(len(idx1))
        for ii in range(int(ones - target)):
            b[idx1[order[ii]]] = 0
    elif ones < target:
        idx0 = np.where(b == 0)[0]
        order = rng.permutation(len(idx0))
        for ii in range(int(target - ones)):
            b[idx0[order[ii]]] = 1
    cur_f = _eval(b)
    best_b = b.copy()
    best_f = cur_f
    ones_idx = np.where(b == 1)[0]
    zeros_idx = np.where(b == 0)[0]
    while time.time() - t0 < time_limit:
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        oi = ones_idx[rng.randint(len(ones_idx))]
        zi = zeros_idx[rng.randint(len(zeros_idx))]
        b[oi], b[zi] = 0, 1
        nf = _eval(b)
        if nf < cur_f - 1e-15:
            cur_f = nf
            ones_idx = np.where(b == 1)[0]
            zeros_idx = np.where(b == 0)[0]
            if nf < best_f:
                best_f = nf
                best_b = b.copy()
        else:
            b[oi], b[zi] = 1, 0
    return best_b, best_f
