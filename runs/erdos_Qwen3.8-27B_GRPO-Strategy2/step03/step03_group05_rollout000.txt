                if lr < 1e-6:
                    break

    # Phase 3: SLSQP with high-alpha soft-max
    if elapsed() < budget_s - 150:
        for a in [50000, 200000, 800000]:
            if elapsed() > budget_s - 150:
                break
            try:
                cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
                res = minimize(
                    lambda x, a_=a: x_norm_soft(x, a_), best_h.copy(),
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': 600, 'ftol': 1e-17})
                hc = _repair(res.x, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc
            except Exception:
                pass

    # Phase 4: Gradient-guided pair coordinate descent (long)
    if elapsed() < budget_s - 100:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        dlist_pos = np.array([0.001, 0.002, 0.005, 0.01, 0.02, 0.04])
        dlist = np.concatenate([dlist_pos, -dlist_pos])
        for _sweep in range(120):
            if elapsed() > budget_s - 100:
                break
            c = _corr(h)
            top_idx = np.argsort(c)[::-1][:6]
            g = _active_grad(h, top_idx)
            order = np.argsort(np.abs(g))[::-1][:30]
            improved = False
            for ii in range(len(order)):
                i = order[ii]
                for jj in range(ii + 1, min(ii + 15, len(order))):
                    j = order[jj]
                    best_d = 0.0
                    best_c = c5_cur
                    for d in dlist:
                        ni, nj = h[i] + d, h[j] - d
                        if 0 <= ni <= 1 and 0 <= nj <= 1:
                            h[i], h[j] = ni, nj
                            c = _c5(h, dx)
                            if c < best_c:
                                best_c = c
                                best_d = d
                            h[i], h[j] = ni - d, nj + d
                    if best_d != 0.0:
                        h[i] += best_d
                        h[j] -= best_d
                        h = _repair(h, n)
                        c5_cur = _c5(h, dx)
                        improved = True
                        if c5_cur < best_c5:
                            best_c5 = c5_cur
                            best_h = h.copy()
                        break
            if not improved:
                break

    # Phase 5: Short simulated annealing
    if elapsed() < budget_s - 50:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        T = 2e-5
        for _it in range(50000):
            if elapsed() > budget_s - 50:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-0.03, 0.03)
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            h = _repair(h, n)
            c_new = _c5(h, dx)
            if c_new < c5_cur or (T > 1e-10 and rng.rand() < np.exp(-(c_new - c5_cur) / T)):
                c5_cur = c_new
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h.copy()
            else:
                h[i] -= d
                h[j] += d
                h = _repair(h, n)
            T *= 0.9998
            if T < 1e-10:
                T = 1e-10

    # Phase 6: Multi-restart short L-BFGS-B
    if elapsed() < budget_s - 25:
        for _r in range(8):
            if elapsed() > budget_s - 25:
                break
            scale = 0.005 + 0.005 * _r
            hp = best_h + scale * rng.randn(n)
            hp = _repair(hp, n)
            try:
                res = minimize(
                    lambda x: _softobj(x, 500000), hp,
                    jac=lambda x: _softgrad(x, 500000),
                    method='L-BFGS-B', bounds=bounds,
                    options={'maxiter': 300, 'ftol': 1e-17})
                hc = _repair(res.x, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc
            except Exception:
                pass

    # Phase 7: Ultra-fine polish
    if elapsed() < budget_s - 3:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        tiny = np.array([0.0002, 0.0005, 0.001, 0.002, -0.0002, -0.0005, -0.001, -0.002])
        for _ in range(25000):
            if elapsed() > budget_s - 3:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = tiny[rng.randint(len(tiny))]
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5_cur:
                    c5_cur = c
                    best_c5 = c
                    best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)

def x_norm_soft(x, a_):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(a_ * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / a_
