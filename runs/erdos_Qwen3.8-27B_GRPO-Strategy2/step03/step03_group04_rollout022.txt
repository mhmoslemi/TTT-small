def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _build_sym(h_half):
    return np.concatenate([h_half, 1.0 - h_half])

    mx = float(np.max(c))
    s = np.clip(alpha * (c - mx), -500.0, 500.0)
    return mx + float(np.log(np.sum(np.exp(s)))) / alpha

    mx = float(np.max(c))

    e = np.exp(s)
