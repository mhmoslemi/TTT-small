import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _grad_one_lag(x, L, n):
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * x
    elif L > 0:
        g[L:] += (1.0 - x[:n - L])
        g[:n - L] -= x[L:]
    else:
        M = -L
        g[:n - M] += (1.0 - x[M:])
        g[M:] -= x[:n - M]
    return g

def _softobj(x, alpha, topk_frac, var_w, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    val = m + np.log(np.sum(np.exp(s))) / alpha
    if var_w > 0:
        srt = np.argsort(c)[::-1]
        tk = max(2, int(topk_frac * len(c)))
        ck = c[srt[:tk]]
        val += var_w * np.var(ck)
    val += lam * (np.sum(x) - len(x) / 2.0) ** 2
    return val

def _softgrad(x, alpha, topk_frac, var_w, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-15:
            continue
        g += pk * _grad_one_lag(x, L, n)
    g *= 1.0 / alpha
    if var_w > 0:
        srt = np.argsort(c)[::-1]
        tk = max(2, int(topk_frac * len(c)))
        idx_k = srt[:tk]
        ck = c[idx_k]
        mu = ck.mean()
        for ik in idx_k:
            L = ik - (n - 1)
            dck = 2.0 * (c[ik] - mu) / tk
            g += dck * _grad_one_lag(x, L, n)
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _active_grad(h, top_idx, n):
    g = np.zeros(n)
    for ki in top_idx:
        L = ki - (n - 1)
        g += _grad_one_lag(h, L, n)
    g /= len(top_idx)
    return g

def _make_structured(n, seed):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.zeros(n)
    choice = seed % 4
    if choice == 0:
        q = max(1, n // 5)
        h[:q] = 1.0
        h[n//2:n//2+q] = 1.0
    elif choice == 1:
        q = max(1, n // 7)
        for s in [0, n//3, 2*n//3]:
            h[s:s+q] = 1.0
    elif choice == 2:
        h[:] = 0.5 + 0.3 * np.cos(3 * np.pi * x)
    else:
        h[:] = 0.5 * (1 + np.cos(2 * np.pi * x * 1.5))
    return _repair(h, n)

def _purify(h, n, thresh=0.02):
    hp = h.copy()
    hp[hp < thresh] = 0.0
    hp[hp > 1.0 - thresh] = 1.0
    hp = _repair(hp, n)
    return hp
