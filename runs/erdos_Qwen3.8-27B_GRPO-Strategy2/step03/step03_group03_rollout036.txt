def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, dx):
    return np.max(_corr(h)) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    add = min(diff, 1.0 - h[idx])
                else:
                    add = max(diff, h[idx] - 1.0)
                h[idx] += add
                diff -= add
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_lag(h, n, m):
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        g[:n + k] += (1.0 - h[-k:])
        g[-k:] -= h[:n + k]
    return g

    c = _corr(x)

    s = np.clip(alpha * (c - m), -500, 500)
