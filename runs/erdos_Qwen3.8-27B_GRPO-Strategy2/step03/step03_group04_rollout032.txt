import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(x, alpha):
    m = np.max(x)
    s = np.clip(alpha * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _all_grads(h, n):
    one_minus_h = 1.0 - h
    G = np.zeros((n, 2 * n - 1))
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        if lag >= 0:
            G[lag:] += one_minus_h[:n - lag]
            G[:n - lag] -= h[lag:]
        else:
            L = -lag
            G[:n - L] += one_minus_h[L:]
            G[L:] -= h[:n - L]
    return G

def _smooth_obj(x, dx, alpha):
    c = _corr(x)
    return _smooth_max(c, alpha) * dx

def _smooth_grad(x, dx, alpha):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    w = np.exp(s)
    w /= w.sum()
    n = len(x)
    g = np.zeros(n)
    omh = 1.0 - x
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            g[lag:] += w[k] * omh[:n - lag]
            g[:n - lag] -= w[k] * x[lag:]
        else:
            L = -lag
            g[:n - L] += w[k] * omh[L:]
            g[L:] -= w[k] * x[:n - L]
    return g * dx

def _do_slsqp(x0, bounds, target, dx, alpha, maxiter):
    try:
        res = minimize(_smooth_obj, x0, args=(dx, alpha), jac=_smooth_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return res.x
    except Exception:
        return x0

def _pair_optimal_d(c, gi, gj, hi, hj):
    d_min = max(-hi, hj - 1.0)
    d_max = min(1.0 - hi, hj)
    if d_max - d_min < 1e-15:
        return 0.0
    lo, hi_b = d_min, d_max
    for _ in range(40):
        m1 = lo + (hi_b - lo) / 3.0
        m2 = hi_b - (hi_b - lo) / 3.0
        f1 = np.max(c + m1 * (gi - gj))
        f2 = np.max(c + m2 * (gi - gj))
        if f1 < f2:
            hi_b = m2
        else:
            lo = m1
    return (lo + hi_b) / 2.0

        for i in range(n):
            for j in range(i + 1, n):
                gi, gj = G[i], G[j]
                d = _pair_optimal_d(c, gi, gj, h[i], h[j])
                if abs(d) < 1e-14:
                    continue
                h[i] += d
                h[j] -= d
                c += d * (gi - gj)
                nc5 = float(np.max(c) * dx)
                if nc5 < c5_cur - 1e-16:
                    c5_cur = nc5
                    imp = True
                else:
                    h[i] -= d
                    h[j] += d
                    c -= d * (gi - gj)
        if c5_cur < 380:
            G = _all_grads(h, n)
        if not imp:
            break
    return h, c5_cur

def _sliding_window_search(h, dx, rng, time_limit, t0, window=6, n_tries=200000):
    n = len(h)
    c5_cur = _c5_val(h, dx)
    best = h.copy()
    for _ in range(n_tries):
        if time.time() > time_limit:
            break
        i = rng.randint(0, n - window + 1)
        h_old = h[i:i+window].copy()
        h[i:i+window] = sorted(h_old.tolist(), key=rng.random)
        c5_new = _c5_val(h, dx)
        if c5_new < c5_cur - 1e-15:
            c5_cur = c5_new
            best = h.copy()
        else:
            h[i:i+window] = h_old
    return best, c5_cur

def _perturb_slsqp(hbest, dx, bounds, target, alpha, time_limit, t0, rng):
    n = len(hbest)
    h = _repair(hbest + 0.01 * rng.randn(n), n)
    c5_cur = _c5_val(h, dx)
    for _ in range(50):
        if time.time() > time_limit:
            break
        c = _corr(h)
        m = np.max(c)
        top = np.argsort(c)[-5:]
        grad = np.zeros(n)
        G = _all_grads(h, n)
        for k in top:
            grad += G[:, k]
        grad /= len(top)
        for lr in [0.02, 0.01, 0.005, 0.002]:
            h_try = _repair(h - lr * grad, n)
            c5_try = _c5_val(h_try, dx)
            if c5_try < c5_cur - 1e-16:
                h, c5_cur = h_try, c5_try
                break
    if c5_cur < _c5_val(hbest, dx):
        return _do_slsqp(h, bounds, target, dx, alpha, 500)
    return hbest

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h0.copy(), _c5_val(h0, dx)

    h1 = _repair(1.0 - h0, n)
    c5_1 = _c5_val(h1, dx)
    if c5_1 < best_c5:
        best_h, best_c5 = h1.copy(), c5_1

    for alpha in [500, 2000, 10000, 50000]:
        if time.time() > t_start + 60:
            break
        rx = _do_slsqp(best_h, bounds, target, dx, alpha, 800)
        hc = _repair(rx, n)
        c5 = _c5_val(hc, dx)
        if c5 < best_c5 - 1e-15:
            best_h, best_c5 = hc.copy(), c5

    if time.time() < t_start + budget_s - 150:
        h, c5 = best_h.copy(), best_c5
        c = _corr(h)
        G = _all_grads(h, n)
        h, c5 = _local_search(h, c, G, dx, 300, t_start + budget_s - 150, t_start)
        if c5 < best_c5:
            best_h, best_c5 = _repair(h, n), c5

    if time.time() < t_start + budget_s - 100:
        h, c5 = best_h.copy(), best_c5
        c = _corr(h)
        G = _all_grads(h, n)
        h, c5 = _local_search(h, c, G, dx, 500, t_start + budget_s - 100, t_start)
        if c5 < best_c5:
            best_h, best_c5 = _repair(h, n), c5

    if time.time() < t_start + budget_s - 60:
        h, c5 = _sliding_window_search(best_h, dx, rng, t_start + budget_s - 60, t_start,
                                        window=5, n_tries=300000)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    if time.time() < t_start + budget_s - 50:
        h, c5 = _sliding_window_search(best_h, dx, rng, t_start + budget_s - 50, t_start,
                                        window=8, n_tries=200000)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    if time.time() < t_start + budget_s - 30:
        h, c5 = best_h.copy(), best_c5
        c = _corr(h)
        G = _all_grads(h, n)
        h, c5 = _local_search(h, c, G, dx, 500, t_start + budget_s - 30, t_start)
        if c5 < best_c5:
            best_h, best_c5 = _repair(h, n), c5

    if time.time() < t_start + budget_s - 10:
        hp = _perturb_slsqp(best_h, dx, bounds, target, 20000,
                            t_start + budget_s - 10, t_start, rng)
        hc = _repair(hp, n)
        c5 = _c5_val(hc, dx)
        if c5 < best_c5:
            best_h, best_c5 = hc, c5

    if time.time() < t_start + budget_s - 5:
        h, c5 = best_h.copy(), best_c5
        c = _corr(h)
        G = _all_grads(h, n)
        h, c5 = _local_search(h, c, G, dx, 200, t_start + budget_s - 5, t_start)
        if c5 < best_c5:
            best_h, best_c5 = _repair(h, n), c5

    best_h = _repair(best_h, n)
    best_c5 = _c5_val(best_h, dx)
    return (best_h, best_c5, n)
