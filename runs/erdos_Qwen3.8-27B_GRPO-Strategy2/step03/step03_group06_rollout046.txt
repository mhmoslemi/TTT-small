import numpy as np
import time
from scipy.optimize import minimize

def _eval(h, dx):
    corr = np.correlate(h.astype(np.float64), 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _eval_max(h):
    corr = np.correlate(h.astype(np.float64), 1.0 - h, mode='full')
    return float(np.max(corr))

def _smooth_max(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    return m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha

def _smooth_max_pen(x, alpha, target, pen):
    return _smooth_max(x, alpha) + pen * (np.sum(x) - target) ** 2

def _continuous_refine(h, n, dx, deadline):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best_h = h.copy()
    best_c5 = _eval(best_h, dx)
    for alpha in [100, 500, 2000, 10000]:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                lambda x, a=alpha: _smooth_max_pen(x, a, target, 100.0),
                best_h, method='L-BFGS-B', bounds=bounds,
                options={'maxiter': 200, 'ftol': 1e-15}
            )
            hc = np.clip(res.x, 0, 1)
            s = hc.sum()
            if s > 0:
                hc *= target / s
                hc = np.clip(hc, 0, 1)
            c5 = _eval(hc, dx)
            if c5 < best_c5:
                best_h, best_c5 = hc, c5
        except:
            pass
    return best_h, best_c5

def _bin_sa(h_bin, n, target, T_start, T_min, cool, deadline, rng):
    """Binary simulated annealing."""
    cur = h_bin.copy()
    cur_max = _eval_max(cur)
    best_local = h_bin.copy()
    best_local_max = cur_max
    T = T_start
    
    while time.time() < deadline and T > T_min:
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        
        cur[i] = 0.0
        cur[j] = 1.0
        new_max = _eval_max(cur)
        
        if new_max <= cur_max or rng.random() < np.exp(-(new_max - cur_max) / T):
            cur_max = new_max
            if new_max < best_local_max:
                best_local_max = new_max
                best_local = cur.copy()
        else:
            cur[i] = 1.0
            cur[j] = 0.0
        T *= cool
    
    return best_local, best_local_max

import numpy as np
import time
from scipy.optimize import minimize

def _eval_c5(h, dx):
    corr = np.correlate(h.astype(np.float64), 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _eval_max(h):
    corr = np.correlate(h.astype(np.float64), 1.0 - h, mode='full')
    return float(np.max(corr))

def _smooth_obj(x, alpha, target, pen):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def _bin_sa(h_init, n, target, T0, T_min, cool, deadline, rng):
    cur = h_init.astype(np.float64).copy()
    cur_max = _eval_max(cur)
    best = cur.copy()
    best_max = cur_max
    T = T0
    n_ones = target
    n_zeros = n - target
    while time.time() < deadline and T > T_min:
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        cur[i] = 0.0
        cur[j] = 1.0
        nm = _eval_max(cur)
        if nm <= cur_max or (nm - cur_max) / T < np.log(1.0 - 1e-15 + rng.random()):
            cur_max = nm
            if nm < best_max:
                best_max = nm
                best = cur.copy()
        else:
            cur[i] = 1.0
            cur[j] = 0.0
        T *= cool
    return best, best_max

def _steepest_descent(h_bin, n, deadline, rng):
    h = h_bin.astype(np.float64).copy()
    cur_max = _eval_max(h)
    improved = True
    while improved and time.time() < deadline:
        improved = False

            h[i] = 0.0
            h[j] = 1.0
            v = _eval_max(h)
            if v < best_val:
                best_val = v
                best_move = (i, j)
            h[i] = 1.0
            h[j] = 0.0
        if best_move is not None:
            i, j = best_move
            h[i] = 0.0
            h[j] = 1.0
            cur_max = best_val
            improved = True
    return h, cur_max

def _cont_refine(h, n, dx, deadline):

    bounds = [(0.0, 1.0)] * n

    for alpha in [80, 300, 1000, 4000, 12000]:
        if time.time() >= deadline:

                lambda x, a=alpha: _smooth_obj(x, a, target, 500.0),
                best_h, method='L-BFGS-B', bounds=bounds,
                options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-8}
            )
            hc = np.clip(res.x, 0.0, 1.0)
            s = hc.sum()
            if s > 1e-12:

            c5 = _eval_c5(hc, dx)
            if c5 < best_c5:
                best_h, best_c5 = hc, c5
        except Exception:
            pass
    return best_h, best_c5
