    _params['target'] = target
    bounds = [(0.0, 1.0)] * n

    h0 = _proj(np.array(initial_h_values, dtype=float), target)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: L-BFGS-B with soft-max, increasing alpha
    alphas = [20, 80, 300, 1500, 8000, 40000, 150000]
    for a in alphas:
        if time.time() - t0 > budget_s - 350:
            break
        _params['alpha'] = a
        _params['lam'] = 30.0
        try:
            res = minimize(_softobj, best_h.copy(), jac=_softgrad,
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 600, 'ftol': 1e-18, 'gtol': 1e-14})
            hc = _proj(res.x, target)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc
        except Exception:
            pass

    # Phase 2: SLSQP with exact constraint
    for a in [3000, 20000, 100000, 400000]:
        if time.time() - t0 > budget_s - 250:
            break
        _params['alpha'] = a
        _params['lam'] = 0.0
        try:
            res = minimize(_softobj, best_h, jac=_softgrad,
                          method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}],
                          options={'maxiter': 1200, 'ftol': 1e-18})
            hc = _proj(res.x, target)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc
        except Exception:
            pass

    # Phase 3: Active-set projected gradient with adaptive step
    if time.time() - t0 < budget_s - 120:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        for m_top in [3, 5, 8, 15]:
            if time.time() - t0 > budget_s - 120:
                break
            step = 0.008
            stuck = 0
            for _it in range(3000):
                if time.time() - t0 > budget_s - 120:
                    break
                c = _corr(h)
                idx = np.argsort(c)[::-1][:m_top]
                g = _active_grad(h, idx, n)
                g_c = g - g.mean()
                ok = False
                for _bt in range(25):
                    h_new = _proj(h - step * g_c, target)
                    c5_new = _c5(h_new, dx)
                    if c5_new < c5_cur - 1e-16:
                        h = h_new
                        c5_cur = c5_new
                        if c5_cur < best_c5:
                            best_c5 = c5_cur
                            best_h = h.copy()
                        stuck = 0
                        ok = True
                        break
                    step *= 0.6
                if not ok:
                    stuck += 1
                    step *= 1.5
                    if step > 0.04 or stuck > 30:
                        break

    # Phase 4: Systematic pair coordinate descent
    if time.time() - t0 < budget_s - 60:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        _params['alpha'] = 20000
        _params['lam'] = 0.0
        for _sweep in range(300):
            if time.time() - t0 > budget_s - 60:
                break
            g = _softgrad(h)
            order = np.argsort(np.abs(g))[::-1][:35]
            improved = False
            for ii in range(len(order)):
                i = order[ii]
                for jj in range(ii + 1, min(ii + 25, len(order))):
                    j = order[jj]
                    best_d = 0.0
                    best_c = c5_cur
                    # Coarse then fine 1D search
                    for d in np.concatenate([np.linspace(-0.05, 0.05, 21),
                                             np.linspace(-0.01, 0.01, 11)]):
                        ni, nj = h[i] + d, h[j] - d
                        if 0 <= ni <= 1 and 0 <= nj <= 1:
                            h[i], h[j] = ni, nj
                            c = _c5(h, dx)
                            h[i], h[j] = h[i] - d, h[j] + d
                            if c < best_c:
                                best_c = c
                                best_d = d
                    if best_d != 0.0:
                        h[i] += best_d
                        h[j] -= best_d
                        c5_cur = best_c
                        improved = True
                        if c5_cur < best_c5:
                            best_c5 = c5_cur
                            best_h = h.copy()
                        break
            if not improved:
                break

    # Phase 5: Long fine random polish
    if time.time() - t0 < budget_s - 5:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        extreme = 0
        for _it in range(500000):
            if time.time() - t0 > budget_s - 5:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-0.004, 0.004)
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < c5_cur:
                c5_cur = c
                best_c5 = c
                best_h = h.copy()
                extreme = 0
            else:
                h[i], h[j] = ni - d, nj + d
                extreme += 1
            if extreme > 200000 and _it > 50000:
                break

    # Phase 6: Final micro-polish with tiny steps
    if time.time() - t0 < budget_s - 2:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        tiny_steps = np.array([0.0002, 0.0005, 0.001, -0.0002, -0.0005, -0.001])
        for _it in range(100000):
            if time.time() - t0 > budget_s - 2:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = tiny_steps[rng.randint(6)]
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5_cur:
                    c5_cur = c
                    best_c5 = c
                    best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _proj(best_h, target)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
