    h = h.copy()

            break

        for i in range(n):
            if time_limit_fn():
                break

            for target_val in [0.0, 1.0]:
                delta = target_val - h0
                if abs(delta) < 1e-10:
                    continue
                # Need to compensate: find j such that h[j] - delta stays in [0,1]
                # delta > 0 means we increased h[i], need to decrease some h[j] by delta
                # delta < 0 means we decreased h[i], need to increase some h[j] by -delta
                if delta > 0:

                    # Wait: h[j] + (-delta) <= 1 => h[j] <= 1 + delta (delta is negative, so 1+delta < 1)
                    # Actually: new_h[j] = h[j] - delta = h[j] + |delta|
                    # Need h[j] + |delta| <= 1, i.e., h[j] <= 1 - |delta| = 1 + delta
                if len(candidates) == 0:
                    continue
                # Try the candidate that gives best improvement
                # For speed, just try a few
                trial_best = c5
                trial_j = -1
                trial_h = None
                sample_candidates = candidates
                if len(sample_candidates) > 10:
                    sample_candidates = candidates[rng_0.randint(0, len(candidates), 10)]

                    h = trial_h
                    c5 = trial_best
                    improved = True

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sf(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sj(x, alpha):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        wk = w[k]
        if lag >= 0:
            L = n - lag
            g[:L] += wk * (1.0 - x[lag:])
            g[lag:] -= wk * x[:L]
        else:
            L = n + lag
            g[-lag:] += wk * (1.0 - x[:L])
            g[:L] -= wk * x[-lag:]
    return g

def _slsqp(h, n, alpha, maxiter, ftol):
    bnds = [(0.0, 1.0)] * n
    tgt = n / 2.0
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_sf, h, jac=_sj, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _sum_fun(x):
    return np.sum(x) - len(x) / 2.0

def _sum_jac(x):
    return np.ones(len(x))

def _binary_relax(h, n):
    best = h.copy()
    best_c5 = _eval(h)
    n_ones = n // 2
    sorted_idx = np.argsort(h)
    for pos in range(max(0, n_ones - 6), min(n, n_ones + 6)):
        h_bin = np.zeros(n)
        h_bin[sorted_idx[pos:]] = 1.0
        h_bin = _repair(h_bin, n)
        c5 = _eval(h_bin)
        if c5 < best_c5:
            best_c5 = c5
            best = h_bin.copy()
    h_pushed = np.where(h > 0.4, 1.0, 0.0)
    h_pushed = _repair(h_pushed, n)
    c5 = _eval(h_pushed)
    if c5 < best_c5:
        best = h_pushed.copy()
        best_c5 = c5
    return best, best_c5

def _pair_descent(h, n, max_rounds, time_check, dirs=None):
    h = h.copy()
    c5c = _eval(h)
    if dirs is None:
        dirs = [0.05, 0.1, 0.2, -0.05, -0.1, -0.2]
    for rnd in range(max_rounds):
        if time_check():
            break
        improved = False
        for i in range(n):
            if time_check():
                break
            for j in range(i + 1, n):
                if time_check():
                    break
                for d in dirs:
                    ni = h[i] + d
                    nj = h[j] - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c5n = _eval(h)
                    if c5n < c5c - 1e-15:
                        c5c = c5n
                        improved = True
                    else:
                        h[i], h[j] = ni - d, nj + d
        if not improved:
            break
    return h, c5c
