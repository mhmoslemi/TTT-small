import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _softobj(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _softgrad(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-15:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g *= 1.0 / alpha
    return g

def _active_grad(h, m_top=5):
    n = len(h)
    c = _corr(h)
    cmax = np.max(c)
    idx = np.argsort(c)[::-1][:m_top]
    wts = np.exp(-80.0 * (cmax - c[idx]))
    wts /= wts.sum()
    g = np.zeros(n)
    for i, ki in enumerate(idx):
        L = ki - (n - 1)
        w = wts[i]
        if L == 0:
            g += w * (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += w * (1.0 - h[:n - L])
            g[:n - L] -= w * h[L:]
        else:
            M = -L
            g[:n - M] += w * (1.0 - h[M:])
            g[M:] -= w * h[:n - M]
    return g

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _interp_h(src, n_new):
    n_src = len(src)
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, n_src), src)

def _lbfgs(h, n, alpha, bounds, maxiter=500):
    dx = 2.0 / n
    try:
        res = minimize(_softobj, h.copy(), (alpha,),
                       jac=_softgrad, args=(alpha,),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _subgrad_loop(h, n, t_start, t_end):
    dx = 2.0 / n
    c5_cur = _c5(h, dx)
    eta = 0.004
    for _ in range(5000):
        if time.time() > t_end:
            break
        g = _active_grad(h, 4)
        g -= g.mean()
        h_new = _repair(h - eta * g, n)
        c5_new = _c5(h_new, dx)
        if c5_new < c5_cur - 1e-15:
            h = h_new
            c5_cur = c5_new
            eta = min(eta * 1.03, 0.015)
        else:
            eta *= 0.6
            if eta < 1e-7:
                break
    return h, c5_cur

def _coord_desc(h, n, rng, t_end, sweeps=40):
    dx = 2.0 / n
    c5_cur = _c5(h, dx)
    for _ in range(sweeps):
        if time.time() > t_end:
            break
        g = _active_grad(h, 6)
        order = np.argsort(np.abs(g))[::-1][:25]
        improved = False
        for ii in range(len(order)):
            i = order[ii]
            for jj in range(ii + 1, min(ii + 12, len(order))):
                j = order[jj]
                bd, bc = 0.0, c5_cur
                for d in [0.001, 0.003, 0.007, 0.015, 0.03, 0.06,
                          -0.001, -0.003, -0.007, -0.015, -0.03, -0.06]:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < bc:
                            bc, bd = c, d
                        h[i], h[j] = ni - d, nj + d
                if bd != 0:
                    h[i] += bd
                    h[j] -= bd
                    c5_cur = _c5(h, dx)
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur

def _anneal(h, n, rng, t_end, iters=15000):
    dx = 2.0 / n
    c5_cur = _c5(h, dx)
    T = 5e-6
    for _ in range(iters):
        if time.time() > t_end:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.03, 0.03)
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        h = _repair(h, n)
        c_new = _c5(h, dx)
        if c_new < c5_cur or (T > 1e-11 and rng.rand() < np.exp(-(c_new - c5_cur) / T)):
            c5_cur = c_new
        else:
            h[i] -= d
            h[j] += d
            h = _repair(h, n)
        T *= 0.99992
    return h, c5_cur

def _fine_polish(h, n, rng, t_end, iters=12000):
    dx = 2.0 / n
    c5_cur = _c5(h, dx)
    steps = np.array([0.0004, 0.0009, 0.002, -0.0004, -0.0009, -0.002,
                      0.0001, -0.0001])
    for _ in range(iters):
        if time.time() > t_end:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = steps[rng.randint(len(steps))]
        h[i] += d
        h[j] -= d
        h = _repair(h, n)
        c = _c5(h, dx)
        if c > c5_cur:
            h[i] -= d
            h[j] += d
            h = _repair(h, n)
        else:
            c5_cur = c
    return h, c5_cur
