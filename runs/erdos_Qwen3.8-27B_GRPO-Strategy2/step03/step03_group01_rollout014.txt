This is a linear program! Because f(k) is linear in h (it's bilinear: h[i]*(1-h[j]) = h[i] - h[i]*h[j], which is NOT linear).

Wait, h[i]*(1-h[j]) = h[i] - h[i]*h[j]. This is NOT linear in h when i≠j. So it's a quadratic program (or non-convex if we don't have special structure).

Actually, for the case i=j (L=0): h[i]*(1-h[i]) = h[i] - h[i]², which is concave (we're minimizing, so this makes the problem non-convex).

For i≠j: h[i]*(1-h[j]) = h[i] - h[i]*h[j]. Cross terms make this non-convex in general.

So it's a non-convex optimization problem. The smooth max (log-sum-exp) makes it twice differentiable but still non-convex.

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)

                diff -= a

    k_worst, cmax = _worst_lag(h)
    L = k_worst - (n-1)
    g = np.zeros(n)
    omh = 1.0 - h
    if L >= 0:
        g[:n-L] += omh[L:]
        if n-2*L > 0:
            g[L:n-L] -= h[:n-2*L]
    else:
        M = -L
        g[M:] += omh[:n-M]
        g[:n-M] -= h[M:]
    # Move from high-g to low-g positions
    hi = int(np.argmax(g))
    lo = int(np.argmin(g))
    if hi == lo:
        return h, False
    d = min(gamma, h[hi]-h[lo], h[hi], h[lo]) * 2.0
    if d < 1e-8:
        return h, False
    h[hi] -= d
    h[lo] += d
    return h, True

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)

    def _el():
        return time.time() - t0

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP polish with increasing alpha
    for alpha in [300, 3000, 30000, 150000]:
        if _el() > budget_s - 300:
            break
        h_new = _slsqp(best_h, n, alpha, 2000, 1e-18)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 2: L-BFGS-B with high penalty
    if _el() < budget_s - 250:
        h_new = _lbfgs(best_h, n, 100000, 8000, 5e4)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 3: Multi-start with diverse initializations
    if _el() < budget_s - 180:
        starts = []
        for _r in range(10):
            sc = 0.003 + 0.025 * rng.rand()
            starts.append(_repair(best_h + sc * rng.randn(n), n))
        starts.append(_binary_proj(best_h, n))
        for sh in [1, 2, -1, -2, 3]:
            starts.append(_repair(np.roll(best_h, sh), n))
        starts.append(_repair(best_h[::-1].copy(), n))
        for t in [0.35, 0.45, 0.55, 0.65]:
            starts.append(_repair(np.where(best_h > t, 1.0, 0.0), n))
        # Sawtooth pattern
        sw = np.zeros(n)
        for i in range(n):
            sw[i] = 1.0 if (i % 7) < 4 else 0.0
        starts.append(_repair(sw, n))
        # Block pattern
        bp = np.zeros(n)
        bp[:n//3] = 1.0
        bp[2*n//3:] = 1.0
        starts.append(_repair(bp, n))

        for hs in starts:
            if _el() > budget_s - 180:
                break
            h_new = _slsqp(hs, n, 50000, 1500, 1e-17)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    # Phase 4: Targeted gradient moves on worst lag
    if _el() < budget_s - 120:
        h = best_h.copy()
        c5c = _eval(h)
        gamma = 0.02
        for _it in range(5000):
            if _el() > budget_s - 120:
                break
            h, moved = _targeted_move(h, n, gamma)
            if not moved:
                gamma *= 0.8
                if gamma < 1e-6:
                    break
                continue
            c5n = _eval(h)
            if c5n < c5c - 1e-15:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h = best_h.copy()
                gamma *= 0.7

    # Phase 5: Pair coordinate descent with adaptive steps
    if _el() < budget_s - 80:
        h = best_h.copy()
        c5c = _eval(h)
        step_sets = [
            np.array([0.01, 0.03, 0.05, 0.1, -0.01, -0.03, -0.05, -0.1]),
            np.array([0.003, 0.01, 0.02, 0.05, -0.003, -0.01, -0.02, -0.05]),
            np.array([0.001, 0.003, 0.008, 0.02, -0.001, -0.003, -0.008, -0.02]),
            np.array([0.0003, 0.001, 0.003, 0.01, -0.0003, -0.001, -0.003, -0.01]),
        ]
        for steps in step_sets:
            if _el() > budget_s - 80:
                break
            for _pass in range(200):
                if _el() > budget_s - 80:
                    break
                imp = False
                for i in range(0, n, 1):
                    if _el() > budget_s - 80:
                        break
                    for j in range(i+1, n, 1):
                        if _el() > budget_s - 80:
                            break
                        hi_val, hj_val = h[i], h[j]
                        for d in steps:
                            ni, nj = hi_val+d, hj_val-d
                            if ni < -1e-12 or ni > 1+1e-12 or nj < -1e-12 or nj > 1+1e-12:
                                continue
                            h[i], h[j] = ni, nj
                            c5n = _eval(h)
                            if c5n < c5c - 1e-15:
                                c5c = c5n
                                imp = True
                                hi_val, hj_val = ni, nj
                                if c5n < best_c5:
                                    best_c5 = c5n
                                    best_h = h.copy()
                            else:
                                h[i], h[j] = hi_val, hj_val
                        h[i], h[j] = hi_val, hj_val
                if not imp:
                    break

    # Phase 6: Random fine-tuning with gradient-guided direction
    if _el() < budget_s - 30:
        h = best_h.copy()
        c5c = _eval(h)
        scale = 0.003
        no_imp = 0
        for _ in range(200000):
            if _el() > budget_s - 30:
                break
            # Gradient-guided: pick i where gradient is high, j where low
            g = _grad_obj(h, 50000)
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if g[i] - g[j] < 1e-8:
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    continue
            d = scale * rng.randn()
            if g[i] > g[j]:
                d = abs(d)
            else:
                d = -abs(d)
            ni, nj = h[i]+d, h[j]-d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-15:
                c5c = c5n
                no_imp = 0
                scale = min(scale * 1.03, 0.015)
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i]-d, h[j]+d
                no_imp += 1
                if no_imp > 300:
                    scale *= 0.75
                    no_imp = 0
                    if scale < 1e-6:
                        break

    # Phase 7: Final SLSQP polish
    if _el() < budget_s - 10:
        h_new = _slsqp(best_h, n, 500000, 2000, 1e-19)
        c5 = _eval(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 8: Ultra-fine random moves
    if _el() < budget_s - 3:
        h = best_h.copy()
        c5c = _eval(h)
        for _ in range(50000):
            if _el() > budget_s - 3:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.choice([1, -1]) * 1e-5 * rng.choice([1, 2, 5])
            ni, nj = h[i]+d, h[j]-d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-16:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = h[i]-d, h[j]+d

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
