            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            mc = float(_corr(h).max())
            if mc < c5_raw - 1e-17:
                c5_raw = mc
                h = _repair(h, n)
                c5_chk = _c5(h, dx)
                if c5_chk < best_c5:
                    best_c5 = c5_chk
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
