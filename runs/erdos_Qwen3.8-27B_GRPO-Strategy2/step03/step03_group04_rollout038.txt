import numpy as np
from scipy.optimize import fmin_l_bfgs_b
import time


def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5_val(h, dx):
    return float(np.max(_corr_full(h)) * dx)


def _soft_max(c, alpha):
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)


def _soft_max_w(c, alpha):
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    e = np.exp(s)
    ssum = e.sum()
    if ssum < 1e-300:
        return np.full(len(c), 1.0 / len(c))
    return e / ssum


def _corr_grad(x, w, n):
    g = np.zeros(n)
    for m in range(2 * n - 1):
        lag = m - (n - 1)
        wk = w[m]
        if wk < 1e-14:
            continue
        lo1 = max(0, -lag)
        hi1 = min(n, n - lag)
        if lo1 < hi1:
            g[lo1:hi1] += wk * (1.0 - x[lo1 + lag:hi1 + lag])
        lo2 = max(0, lag)
        hi2 = min(n, n + lag)
        if lo2 < hi2:
            g[lo2:hi2] -= wk * x[lo2 - lag:hi2 - lag]
    return g


def _build_sym(a, n):
    n2 = n // 2
    h = np.empty(n)
    h[:n2] = a
    h[n2:] = 1.0 - a
    return h


def _obj_sym(a, alpha, n):
    h = _build_sym(a, n)
    c = _corr_full(h)
    return float(_soft_max(c, alpha))


def _grad_sym(a, alpha, n):
    h = _build_sym(a, n)
    c = _corr_full(h)
    w = _soft_max_w(c, alpha)
    g_h = _corr_grad(h, w, n)
    n2 = n // 2
    return g_h[:n2] - g_h[n2:]


def _obj_pen(x, alpha, n, mu):
    c = _corr_full(x)
    v = _soft_max(c, alpha)
    s = np.sum(x) - n / 2.0
    return v + mu * s * s


def _grad_pen(x, alpha, n, mu):
    c = _corr_full(x)
    w = _soft_max_w(c, alpha)
    g = _corr_grad(x, w, n)
    s = np.sum(x) - n / 2.0
    g += 2.0 * mu * s
    return g


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h


def _improve(state, h, dx, n):
    h = _repair(h, n)
    c = _c5_val(h, dx)
    if c < state[0]:
        state[0] = c
        state[1] = h.copy()
    return h

    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    n2 = n // 2

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    state = [float(_c5_val(h0, dx)), h0.copy()]

    # Try 1-h0 symmetry
    h1 = _repair(1.0 - h0, n)
    _improve(state, h1, dx, n)

    # Symmetric projection for initialization
    a_sym = np.clip((h0[:n2] + (1.0 - h0[n2:])) / 2.0, 0.0, 1.0)

    # Phase 1: L-BFGS-B in symmetric subspace (n2 vars, auto integral)
    bounds_sym = [(0.0, 1.0)] * n2
    for alpha in [100, 500, 2000, 10000, 50000, 200000]:
        if time.time() - t0 > budget_s - 320:
            break
        for att in range(2):
            if time.time() - t0 > budget_s - 320:

            x0a = a_sym.copy()
            if att > 0:
                x0a = np.clip(x0a + 0.004 * rng.randn(n2), 0.0, 1.0)
            try:
                res = fmin_l_bfgs_b(_obj_sym, x0a, fprime=_grad_sym,
                                    args=(alpha, n),

                                    bounds=bounds_sym, factr=1e8)
                h = _build_sym(res, n)
                _improve(state, h, dx, n)
                a_sym = np.clip(res, 0.0, 1.0).copy()
            except Exception:
                pass

    # Phase 2: L-BFGS-B in full space with integral penalty
    bounds_full = [(0.0, 1.0)] * n
    mu = 2000.0
    for alpha in [500, 2000, 10000, 50000, 200000]:
        if time.time() - t0 > budget_s - 250:

        for att in range(2):
            if time.time() - t0 > budget_s - 250:

                pass

    # Phase 3: Deliberate coordinate descent (exact c5)
    if time.time() - t0 < budget_s - 60:

        c_cur = state[0]

                            j = (i + 1) % n
                            h[j] += diff
                            if h[j] < 0.0 or h[j] > 1.0:
                                h[i] = hi_old
                                h[j] -= diff
                                continue
                        c = _c5_val(h, dx)
                        if c < c_cur - 1e-15:
                            hi_old = nd
                            c_cur = c
                            improved = True
                        else:
                            h[i] = hi_old
                h[i] = hi_old
            if c_cur < state[0]:
                state[0] = c_cur
                state[1] = h.copy()
            if not improved:
                break

    # Phase 4: Random pair transfer search (preserves integral)
    if time.time() - t0 < budget_s - 10:
        h = state[1].copy()
        c_cur = state[0]
        while time.time() - t0 < budget_s - 10:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.choice([5e-5, 1e-4, 2e-4, 5e-4, 1e-3, 2e-3,
                            -5e-5, -1e-4, -2e-4, -5e-4, -1e-3, -2e-3])
            ni, nj = h[i] + d, h[j] - d
            if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                continue
            h[i], h[j] = ni, nj
            c = _c5_val(h, dx)
            if c < c_cur:
                c_cur = c
                if c_cur < state[0]:
                    state[0] = c_cur
                    state[1] = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d

    # Phase 5: Final L-BFGS-B polish at very high alpha
    if time.time() - t0 < budget_s - 5:
        try:
            res = fmin_l_bfgs_b(_obj_pen, state[1], fprime=_grad_pen,
                                args=(500000, n, 5000.0),
                                pgtol=1e-16, maxiter=3000, iprint=0,
                                bounds=bounds_full, factr=1e8)
            _improve(state, res, dx, n)
        except Exception:
            pass

    # Also try from 1-best
    if time.time() - t0 < budget_s - 3:
        hf = _repair(1.0 - state[1], n)
        try:
            res = fmin_l_bfgs_b(_obj_pen, hf, fprime=_grad_pen,
                                args=(200000, n, 5000.0),
                                pgtol=1e-16, maxiter=2000, iprint=0,
                                bounds=bounds_full, factr=1e8)
            _improve(state, res, dx, n)
        except Exception:
            pass

    best_c5 = _c5_val(best_h, dx)
    return (best_h, best_c5, n)
