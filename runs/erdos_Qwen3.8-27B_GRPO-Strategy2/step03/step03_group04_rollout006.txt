    """Return gradient of all correlations w.r.t. h. Shape: (2n-1, n)"""
    n = len(h)
    shifts = np.arange(-(n-1), n)
    grad = np.zeros((2*n-1, n))
    for idx, m in enumerate(shifts):
        if m >= 0:
            # c[n-1+m] = sum_{i=m}^{n-1} h[i](1-h[i-m])
            for i in range(m, n):
                grad[idx, i] += (1.0 - h[i-m])
                grad[idx, i-m] += -h[i]
        else:
            s = -m
            # c[n-1-s] = sum_{i=0}^{n-1-s} h[i](1-h[i+s])
            for i in range(n - s):
                grad[idx, i] += (1.0 - h[i+s])
                grad[idx, i+s] += -h[i]
    return grad
