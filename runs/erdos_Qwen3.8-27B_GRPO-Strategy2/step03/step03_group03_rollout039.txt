def _pair_optimal_delta(h, n, i, j):
    """Find optimal δ to transfer from h[j] to h[i]."""
    corr = _corr(h)  # length 2n-1
    # Compute B_m = coefficient of δ in c_m after transfer h[i]+=δ, h[j]-=δ
    B = np.zeros(2*n-1)
    for m in range(2*n-1):
        d = m - (n-1)
        # c(m) = sum over valid (p, p+d) of h[p]*(1-h[p+d])
        # h[i] appears as h[p] when p=i (need i+d valid), or as h[p+d] when p=i-d (need i-d valid)
        # h[j] similarly
        bi = 0.0  # coefficient for h[i]
        bj = 0.0  # coefficient for h[j]
        # For h[i] (positive contribution to c when h[i] increases):
        # As h[p]: term h[i]*(1-h[i+d]) → coefficient (1-h[i+d]) if i+d valid
        # As (1-h[p+d]): term h[i-d]*(1-h[i]) → coefficient -h[i-d] if i-d valid
        if 0 <= i + d < n:
            bi += (1.0 - h[i + d]) if (i + d) != j else (1.0 - (h[j] - 0))  # at δ=0
        if 0 <= i - d < n:
            bi -= h[i - d] if (i - d) != j else h[j]
        # For h[j] (negative contribution since h[j] decreases):
        if 0 <= j + d < n:
            bj += (1.0 - h[j + d]) if (j + d) != i else h[i]
        if 0 <= j - d < n:
            bj -= h[j - d] if (j - d) != i else h[i]
        B[m] = bi - bj  # change in c_m per unit δ (h[i] up, h[j] down)
    
    # c_m + δ * B_m, find δ that minimizes max_m(c_m + δ*B_m)
    # Valid range: -h[i] ≤ δ ≤ 1-h[j]
    d_lo = -h[i]
    d_hi = 1.0 - h[j]
    
    # Convex minimization: binary search on subgradient
    for _ in range(60):
        mid = (d_lo + d_hi) / 2
        vals = corr + mid * B
        act = np.where(vals == vals.max())[0]
        grad = np.mean(B[act])
        if grad > 1e-10:
            d_hi = mid
        elif grad < -1e-10:
            d_lo = mid
        else:
            break
    return (d_lo + d_hi) / 2

    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return h

def _B_matrix(h, n):
    B = np.zeros((n, 2 * n - 1))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        # For each i: B[i,m] = (1-h[i+k])*[0<=i+k<n] - h[i-k]*[0<=i-k<n]
        for i in range(n):
            v = 0.0
            ik = i + k
            if 0 <= ik < n:
                v += 1.0 - h[ik]
            ik = i - k
            if 0 <= ik < n:
                v -= h[ik]
            B[i, m] = v
    return B

def _optimal_delta(corr, Bpair, Qpair, h_i, h_j):
    """Minimize max(corr + delta*Bpair + delta^2*Qpair) over delta in [-h_i, 1-h_j]."""
    d_lo = -h_i
    d_hi = 1.0 - h_j
    if d_hi - d_lo < 1e-15:
        return 0.0
    for _ in range(60):
        mid = 0.5 * (d_lo + d_hi)
        vals = corr + mid * Bpair + mid * mid * Qpair
        mx = vals.max()
        act = vals >= mx - 1e-13
        grad = (Bpair[act] + 2.0 * mid * Qpair[act]).mean()
        if grad > 1e-11:
            d_hi = mid
        elif grad < -1e-11:
            d_lo = mid
        else:
            break
    delta = 0.5 * (d_lo + d_hi)
    return float(np.clip(delta, d_lo, d_hi))

    best_c5 = _c5(best_h, dx)
    
    def tleft():
        return budget_s - (time.time() - t0)
    
    # Generate all pairs
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
    
    # Precompute Q matrix (Q[i,j,m] = 1 if cross-term exists)
    # Q for pair (i,j): m = (n-1)+(j-i) and m = (n-1)+(i-j)
    def _Q_for_pair(i, j):
        Q = np.zeros(n_lags)
        m1 = (n - 1) + (j - i)
        m2 = (n - 1) + (i - j)
        if 0 <= m1 < n_lags:
            Q[m1] = 1.0
        if 0 <= m2 < n_lags:
            Q[m2] = 1.0
        return Q
    
    # Phase 1: Cyclic pair descent with exact delta
    h = best_h.copy()
    corr = _corr(h)
    Bmat = _B_matrix(h, n)
    c5_cur = _c5(h, dx)
    
    max_sweeps = 100
    for sweep in range(max_sweeps):
        if tleft() < 180:

        any_improve = False
        for idx in order:

            if dE < 0 or rng.rand() < np.exp(-dE / max(T, 1e-10)):
                c5_cur = new_c5
                Bmat = _B_matrix(h, n)
                if c5_cur < best_c5:

            else:

                else:
