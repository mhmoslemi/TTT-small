                break
            imp = False
            order = np.arange(n)
            rng.shuffle(order)
            for i in order:
                if elapsed() > budget_s - 200:
                    break
                hi = h[i]
                for d in base_deltas:
                    ni = hi + d
                    if ni < 0.0 or ni > 1.0:
                        continue
                    # Need to maintain sum: adjust a random other element
                    j = rng.randint(n)
                    if j == i:
                        j = (i + 1) % n
                    h[i] = ni
                    h[j] = h[j] - d
                    h = _repair(h, n)
                    c_new = _c5(h, dx)
                    if c_new < c5_cur - 1e-15:
                        c5_cur = c_new
                        imp = True
                    else:
                        h = _repair(h, n)
                h = _repair(h, n)
            if c5_cur < best_c5:
                best_c5, best_h = c5_cur, h.copy()
            if not imp and _pass > 8:
                break

    # Phase 3: Pair local search (fine)
    if elapsed() < budget_s - 80:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        deltas = np.array([0.003, 0.001, 0.0004, -0.003, -0.001, -0.0004])
        for _pass in range(50):
            if elapsed() > budget_s - 80:
                break
            imp = False
            for i in range(n):
                if elapsed() > budget_s - 80:
                    break
                hi = h[i]
                for j in range(i + 1, n):
                    hj = h[j]
                    for d in deltas:
                        ni, nj = hi + d, hj - d
                        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                            continue
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < c5_cur - 1e-14:
                            c5_cur, hi, hj = c, ni, nj
                            imp = True
                        else:
                            h[i], h[j] = hi, hj
                    h[i], h[j] = hi, hj
                h[i] = hi
            if c5_cur < best_c5:
                best_c5, best_h = c5_cur, h.copy()
            if not imp and _pass > 5:
                break

    # Phase 4: SLSQP polish (multiple alpha levels)
    if elapsed() < budget_s - 20:
        for alpha in [50000, 200000, 500000]:
            if elapsed() > budget_s - 10:
                break
            try:
                rx = _slsqp(best_h.copy(), n, target, alpha, 400)
                hc = _repair(rx, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc
            except Exception:
                pass

    # Phase 5: Ultra-fine random pair search
    if elapsed() < budget_s - 5:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        opts = np.array([3e-4, 1e-4, 5e-5, 1e-3, -3e-4, -1e-4, -5e-5, -1e-3])
        while elapsed() < budget_s - 3:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = opts[rng.randint(len(opts))]
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5_cur:
                    c5_cur = c
                    if c5_cur < best_c5:
                        best_c5, best_h = c5_cur, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    # Phase 6: Try 1-h and re-optimize
    if elapsed() < budget_s - 3:
        try:
            hf = _repair(1.0 - best_h, n)
            rx = _slsqp(hf, n, target, 100000, 300)
            hc = _repair(rx, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
