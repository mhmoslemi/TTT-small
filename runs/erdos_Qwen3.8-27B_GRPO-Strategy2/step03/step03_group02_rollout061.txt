```python
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(h, target):
    lo, hi = float(h.min()) - 1.0, float(h.max()) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _grad_matrix(h, n):
    G = np.zeros((2 * n - 1, n))
    for s in range(-(n - 1), n):
        a = abs(s)
        if a < n:
            if s >= 0:
                G[s + n - 1, :n - a] += 1.0 - h[a:]
                G[s + n - 1, a:] -= h[:n - a]
            else:
                G[s + n - 1, a:] += 1.0 - h[:n - a]
                G[s + n - 1, :n - a] -= h[a:]
    return G

        p2 = i + s
        if 0 <= p1 < n:
            G[s + n - 1, p1] -= delta

        i = int(order[ai])
        for j in int(order[ai + 1]):
            if time.time() > t_end:
                return h, G, corr, improved_total
            hi, hj = h[i], h[j]
            dmin, dmax = max(-hi, hj - 1.0), min(1.0 - hi, hj)
            if dmax - dmin < 1e-12:
                continue
            gd = G[:, i] - G[:, j]
            nc = 16
            ds = np.linspace(dmin, dmax, nc)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = int(np.argmin(vals))
            if vals[bi] >= float(np.max(corr)) - 1e-12:

            lo = ds[max(0, bi - 1)]
            hi_b = ds[min(nc - 1, bi + 1)]

            _update_G(G, n, j, -d)
```

```python
        p1 = i - s
        p2 = i + s
        if 0 <= p1 < n:
            G[s + n - 1, p1] -= delta
        if 0 <= p2 < n:
            G[s + n - 1, p2] -= delta
```
