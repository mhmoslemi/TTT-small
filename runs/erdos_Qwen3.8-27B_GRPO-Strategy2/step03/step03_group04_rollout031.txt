    for m_idx in range(2*n-1):

        wm = v[m_idx]

        lo = max(0, -m)
        hi = min(n, n - m)
        if lo < hi:
            g[lo:hi] += wm * (1.0 - h[lo+m:hi+m])
        # Add contribution from -h[i-m] for valid i
        lo2 = max(0, m)
        hi2 = min(n, n + m)
        if lo2 < hi2:
            g[lo2:hi2] -= wm * h[lo2-m:hi2-m]

def _min_1d_convex(a, b, lo=0.0, hi=1.0):
    """Minimize max(a*x + b) for x in [lo, hi]. a, b are arrays."""
    def _eval(x):
        vals = a * x + b
        return float(np.max(vals))
    
    def _active(x):
        vals = a * x + b
        mx = np.max(vals)
        return vals >= mx - 1e-15
    
    # Bisection on subgradient
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        act = _active(mid)
        if np.all(a[act] <= 1e-15):
            # Subgradient ≤ 0, minimum is to the right
            lo = mid
        elif np.all(a[act] >= -1e-15):
            # Subgradient ≥ 0, minimum is to the left
            hi = mid
        else:
            # 0 is in subgradient at mid
            return mid
    return 0.5 * (lo + hi)

                    for d in deltas:
                        ni, nj = hi_v + d, hj_v - d
                        if ni < -1e-15 or ni > 1 + 1e-15 or nj < -1e-15 or nj > 1 + 1e-15:
                            continue
                        h[i], h[j] = min(max(ni, 0.0), 1.0), min(max(nj, 0.0), 1.0)
                        hv = _c5(h, dx)
                        if hv < c5_cur - 1e-14:
                            c5_cur = hv
                            hi_v, hj_v = h[i], h[j]
                            imp = True
                        else:
                            h[i], h[j] = hi_v, hj_v
                    h[i], h[j] = hi_v, hj_v
        if not imp:
            break
    return h, c5_cur

def _fine_random(h, n, dx, t0, remaining_s):
    c5_cur = _c5(h, dx)
    rng = np.random.RandomState(12345)
    opts = np.array([1e-4, 3e-4, 1e-3, -1e-4, -3e-4, -1e-3])
    while time.time() - t0 < budget_s - remaining_s:
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = opts[rng.randint(len(opts))]
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < c5_cur:
                c5_cur = c
            else:
                h[i], h[j] = ni - d, nj + d
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    global budget_s
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    h1 = _repair(1.0 - h0, n)
    c5_1 = _c5(h1, dx)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 1: SLSQP initialization
    for alpha in [500, 5000]:
        if time.time() - t0 > budget_s - 500:
            break
        try:
            rx = _do_slsqp(best_h.copy(), bounds, target, alpha, 800)
            hc = _repair(rx, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc.copy()
        except Exception:
            pass

    # Phase 2: Coordinate descent (exact 1D)
    if time.time() - t0 < budget_s - 300:
        hc, c5c = _coord_sweep(best_h.copy(), n, dx, t0, 300)
        if c5c < best_c5:
            best_c5, best_h = c5c, hc.copy()
        hc2, c5c2 = _coord_sweep(1.0 - best_h, n, dx, t0, 300)
        if c5c2 < best_c5:
            best_c5, best_h = c5c2, hc2.copy()

    # Phase 3: Soft gradient descent
    if time.time() - t0 < budget_s - 200:
        hg, c5g = _grad_step(best_h.copy(), n, dx, t0, 200)
        if c5g < best_c5:
            best_c5, best_h = c5g, hg.copy()
        hc3, c5c3 = _coord_sweep(best_h.copy(), n, dx, t0, 200)
        if c5c3 < best_c5:
            best_c5, best_h = c5c3, hc3.copy()

    # Phase 4: Targeted pair search
    if time.time() - t0 < budget_s - 100:
        hp, c5p = _pair_search(best_h.copy(), n, dx, t0, 100, 50)
        if c5p < best_c5:
            best_c5, best_h = c5p, hp.copy()

    # Phase 5: Another coordinate pass
    if time.time() - t0 < budget_s - 50:
        hc4, c5c4 = _coord_sweep(best_h.copy(), n, dx, t0, 60)
        if c5c4 < best_c5:
            best_c5, best_h = c5c4, hc4.copy()

    # Phase 6: Fine random search
    if time.time() - t0 < budget_s - 20:
        hr, c5r = _fine_random(best_h.copy(), n, dx, t0, 20)
        if c5r < best_c5:
            best_c5, best_h = c5r, hr.copy()

    # Phase 7: Final SLSQP polish
    if time.time() - t0 < budget_s - 5:
        try:
            rx = _do_slsqp(best_h.copy(), bounds, target, 50000, 2000)
            hc = _repair(rx, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc.copy()
        except Exception:
            pass

    # Final coordinate polish
    if time.time() - t0 < budget_s - 2:
        hf, c5f = _coord_sweep(best_h.copy(), n, dx, t0, 2)
        if c5f < best_c5:
            best_c5, best_h = c5f, hf.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
