import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
            else:
                a = min(-diff, h[idx])
            h[idx] += a
            diff -= a
            if abs(diff) < 1e-13:
                break
    return h

def _softobj(x, alpha, lam, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    penalty = lam * (np.sum(x) - n / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _active_grad(h, top_idx, n):
    g = np.zeros(n)
    for ki in top_idx:
        L = ki - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    return g

def _lbfgs_run(h, n, alpha, lam, maxiter, t0, budget, reserve):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            _softobj, h, args=(alpha, lam, n),
            jac=_softgrad, method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-17})
        if time.time() > t0 + budget - reserve:
            return None
        return _repair(res.x, n)
    except Exception:
        return None

def _slsqp_run(h, n, alpha, reserve, t0, budget):
    bounds = [(0.0, 1.0)] * n
    one_vec = np.ones(n)
    def cons_fun(x):
        return np.sum(x) - n / 2.0
    def cons_jac(x):
        return one_vec
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(
            _softobj, h, args=(alpha, 0.0, n),
            method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': 600, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return None

    best_n = n_init

    # Try other n values with algorithmic starts
    for n_try in [50, 60, 80]:
        if time.time() > t0 + budget_s - 300:
            break
        dx = 2.0 / n_try
        for init_type in range(2):
            if time.time() > t0 + budget_s - 250:
                break
            if init_type == 0:
                x = np.linspace(0.0, 2.0, n_try)
                ht = 1.0 - np.abs(2.0 * x - 1.0)
            else:
                ht = 0.5 + 0.3 * np.sin(np.pi * 2.0 * np.arange(n_try) / n_try * 3.0 + 0.5)
            ht = np.clip(ht, 0.0, 1.0)
            ht = _repair(ht, n_try)
            hc, c5_new = _optimize_single(ht, n_try, dx, t0, budget_s - 100, rng)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = hc.copy()
                best_n = n_try

    # Main optimization on best candidate
    n = best_n
    dx = 2.0 / n
    h, c5_main = _optimize_single(best_h, n, dx, t0, budget_s, rng)
    if c5_main < best_c5:
        best_c5 = c5_main
        best_h = h.copy()

    # Multi-restart with perturbations
    for _r in range(4):
        if time.time() > t0 + budget_s - 30:
            break
        hp = best_h + 0.006 * rng.randn(n)
        hp = _repair(hp, n)
        hc, c5_new = _optimize_single(hp, n, dx, t0, budget_s, rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = hc.copy()

    # Ultra-fine final polish
    if time.time() < t0 + budget_s - 3:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        tiny = np.array([0.0002, 0.0005, 0.001, -0.0002, -0.0005, -0.001])
        for _ in range(10000):
            if time.time() > t0 + budget_s - 2:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = tiny[rng.randint(len(tiny))]
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5_cur:
                    c5_cur = c
                    if c < best_c5:
                        best_c5 = c
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
