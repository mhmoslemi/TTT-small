    bounds = [(0.0, 1.0)] * n

    best_c5 = _c5(h, dx)
    for alpha in [80, 300, 1000, 4000, 12000, 30000]:
        if time.time() >= deadline:
            break
        def obj(x, a=alpha):
            c = _corr(x)
            m = np.max(c)
            return m + np.log(np.sum(np.exp(a * (c - m)))) / a + 20.0 * (np.sum(x) - target) ** 2
        try:
            res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 350, 'ftol': 1e-16, 'gtol': 1e-8})
            hc = _repair(res.x, n, target)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_h, best_c5 = hc, c5
        except Exception:
            pass
    return best_h, best_c5

def _slsqpolish(h, n, dx, deadline):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    for alpha in [3000, 15000, 60000]:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                lambda x, a=alpha: _smooth_max(_corr(x), a) + 50.0 * (np.sum(x) - target) ** 2,
                best_h, method='SLSQP', bounds=bounds,
                options={'maxiter': 400, 'ftol': 1e-15})
            hc = _repair(res.x, n, target)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_h, best_c5 = hc, c5
        except Exception:
            pass
    return best_h, best_c5

def _fine_local(h, n, dx, rng, deadline, n_tries=50000):
    best_c5 = _c5(h, dx)
    h = h.copy()
    scales = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    per_scale = max(300, n_tries // len(scales))
    for scale in scales:
        improved = False
        for _ in range(per_scale):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
                continue
            h[i], h[j] = ni, nj
            c5 = _c5(h, dx)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                improved = True
            else:
                h[i], h[j] = h[i] - d, h[j] + d
        if not improved and scale < 0.001:
            break
    return h, best_c5

def _block_local(h, n, dx, rng, deadline, n_iters=5000):
    best_c5 = _c5(h, dx)
    h = h.copy()
    for bs in [2, 3, 4]:
        for _ in range(n_iters // 3):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(0, n - bs + 1)
            j = rng.randint(0, n - bs + 1)
            if abs(i - j) < bs:
                continue
            eps = rng.uniform(0.003, 0.02) * (1.0 if rng.random() < 0.5 else -1.0)
            ok = True
            for di in range(bs):
                if h[i + di] + eps < -1e-10 or h[i + di] + eps > 1.0 + 1e-10:
                    ok = False
                if h[j + di] - eps < -1e-10 or h[j + di] - eps > 1.0 + 1e-10:
                    ok = False
            if not ok:
                continue
            h[i:i + bs] += eps
            h[j:j + bs] -= eps
            c5 = _c5(h, dx)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
            else:
                h[i:i + bs] -= eps
                h[j:j + bs] += eps
    return h, best_c5

def _perturb_start(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        blk = rng.randint(4, 14)
        i = rng.randint(0, n - blk)
        seg = hp[i:i + blk]
        hp[i:i + blk] = np.roll(seg, rng.randint(1, blk))
    elif kind == 1:
        f = rng.randint(1, 6)
        ph = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.08, 0.2)
        hp = hp + amp * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 2:
        f1, f2 = rng.randint(1, 5), rng.randint(1, 5)
        p1, p2 = rng.uniform(0, 6.28), rng.uniform(0, 6.28)
        amp = rng.uniform(0.05, 0.15)
        t = np.arange(n) / n
        hp = hp + amp * np.sin(2 * np.pi * f1 * t + p1) * np.sin(2 * np.pi * f2 * t + p2)
    elif kind == 3:
        j = rng.randint(n // 5, 4 * n // 5)
        hp = np.concatenate([h[j:], h[:j]])
    else:
        hp = h + rng.uniform(-0.08, 0.08, n)
    return _repair(hp, n, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n, target)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: Binary SA from diverse starts (~450s)
    t_end1 = t_start + 450
    starts = []
    starts.append(_make_binary(best_h, n, rng))
    for r in range(6):
        h_r = (rng.random(n) < 0.5).astype(float)
        starts.append(_make_binary(h_r, n, rng))
    for blk in [4, 5, 7, 10]:
        pat = np.tile([1.0] * blk + [0.0] * blk, n // blk + 1)[:n]
        starts.append(_make_binary(pat, n, rng))
    # Shifted versions of initial
    for sh in [7, 17, 23]:
        starts.append(_make_binary(np.roll(best_h, sh), n, rng))

    for s in starts:
        if time.time() >= t_end1:
            break
        t_end_sa = min(t_end1, time.time() + 35)
        h_b, c5_b = _binary_sa(s, n, dx, rng, t_end_sa, t0=4.0, tmin=0.03, cool=0.99997)
        if time.time() < t_end1 - 25:
            t_end_cr = min(t_end1 - 15, time.time() + 25)
            h_c, c5_c = _cont_refine(h_b, n, dx, t_end_cr)
            if c5_c < best_c5:
                best_h, best_c5 = h_c, c5_c
            if c5_b < best_c5:
                best_h, best_c5 = _repair(h_b, n, target), c5_b
        elif c5_b < best_c5:
            best_h, best_c5 = _repair(h_b, n, target), c5_b

    # Phase 2: Multi-restart continuous from perturbed best (~200s)
    t_end2 = min(t_start + 700, time.time() + 200)
    for r in range(6):
        if time.time() >= t_end2:
            break
        start = _perturb_start(best_h, n, rng, r % 5)
        t_end_cr = min(t_end2, time.time() + 25)
        h_t, c5_t = _cont_refine(start, n, dx, t_end_cr)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 3: SLSQP polish (~100s)
    t_end3 = min(t_start + 850, time.time() + 100)
    h_t, c5_t = _slsqpolish(best_h, n, dx, t_end3)
    if c5_t < best_c5:
        best_h, best_c5 = h_t, c5_t

    # Phase 4: Block + fine local (~120s)
    if time.time() < t_end2 + 120:
        t_end4 = min(t_start + 1000, time.time() + 60)
        h_t, c5_t = _block_local(best_h, n, dx, rng, t_end4, n_iters=4000)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t
    t_end5 = min(t_start + 1060, time.time() + 60)
    h_t, c5_t = _fine_local(best_h, n, dx, rng, t_end5, n_tries=40000)
    if c5_t < best_c5:
        best_h, best_c5 = h_t, c5_t

    # Phase 5: Final polish
    if time.time() < t_start + budget_s - 15:
        t_end6 = time.time() + max(2, min(30, budget_s - (time.time() - t_start) - 10))
        h_t, c5_t = _slsqpolish(best_h, n, dx, t_end6)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t
    if time.time() < t_start + budget_s - 5:
        t_end7 = time.time() + max(1, min(12, budget_s - (time.time() - t_start) - 3))
        h_t, c5_t = _fine_local(best_h, n, dx, rng, t_end7, n_tries=15000)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    best_h = _repair(best_h, n, target)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
