```python
import numpy as np
import time

FRACS = (0.001, 0.003, 0.008, 0.02, 0.05, 0.12, 0.25, 0.5, 1.0)
SMALL = (2e-6, 1e-5, 5e-5, 2e-4)

def _fs(n):
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _pr(v, S):
    x = np.array(v, dtype=float)
    np.clip(x, 0.0, 1.0, out=x)
    d = S - x.sum()
    for _ in range(30):
        if abs(d) < 1e-12:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
        d = S - x.sum()
    if abs(d) > 1e-10:
        o = np.argsort(1.0 - x) if d > 0 else np.argsort(x)
        for i in o:
            if abs(d) < 1e-10:
                break
            if d > 0:
                a = min(d, 1.0 - x[i])
                if a > 0:
                    x[i] += a
                    d -= a
            else:
                a = min(-d, x[i])
                if a > 0:
                    x[i] -= a
                    d += a
    return x

def _pn(h):
    n = h.size
    N = _fs(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pf = np.empty(n + 1)
    pf[0] = 0.0
    np.cumsum(h, out=pf[1:])
    tot = pf[n]
    P = np.empty(n)
    Nv = np.zeros(n)
    P[0] = tot - ac[0]
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pf[n - idx] - ac[idx]
        Nv[idx] = tot - pf[idx] - ac[idx]
    mx = float(max(P.max(), Nv.max() if n > 1 else 0.0))
    return P, Nv, mx

def _st(h):
    P, Nv, mx = _pn(h)
    n = h.size
    return P[1:].copy(), Nv[1:].copy(), float(P[0]), 2.0 * mx / n, mx

def _c5(h):
    P, Nv, mx = _pn(h)
    return float(2.0 * mx / h.size)

def _gs(h, B, R, O0, mx):
    n = h.size
    g = np.zeros(n)
    tol = max(1e-9, 6.0 * (2.0 / n), 0.008 * mx)
    cut = mx - tol
    ib = np.nonzero(B >= cut)[0]
    for k in ib:
        p = k + 1
        L = n - p
        g[:L] += 1.0 - h[p:]
        g[p:] += -h[:L]
    ir = np.nonzero(R >= cut)[0]
    for k in ir:
        p = k + 1
        L = n - p
        g[:L] += -h[p:]
        g[p:] += 1.0 - h[:L]
    c = ib.size + ir.size
    if O0 >= cut:
        g += 1.0 - 2.0 * h
        c += 1
    if c:
        g /= c
    return g

def _mc(h, i, j):
    n = h.size
    cb = np.zeros(n - 1)
    cr = np.zeros(n - 1)
    for a, s in ((j, 1.0), (i, -1.0)):
        m1 = n - a - 1
        if m1:
            cb[:m1] += s * (1.0 - h[a + 1:a + m1 + 1])
            cr[:m1] += -s * h[a + 1:a + m1 + 1]
        m2 = a
        if m2:
            rv = h[a - 1::-1]
            cb[:m2] += -s * rv
            cr[:m2] += s * (1.0 - rv)
    return cb, cr

def _ap(h, B, R, O0, i, j, d, cb, cr, p0):
    xi = h[i]
    xj = h[j]
    B += d * cb
    R += d * cr
    if p0 >= 0:
        B[p0] += d * d
        R[p0] += d * d
    O0 += 2.0 * d * (xi - xj) - 2.0 * d * d
    h[i] -= d
    h[j] += d
    return O0

def _bd(B, R, O0, cb, cr, p0, xi, xj, D):
    if B.size == 0 or D <= 1e-12:
        return 0.0, 1e100
    vals = [D * f for f in FRACS if D * f > 1e-12]
    for a in SMALL:
        if D > a:
            vals.append(a)
    if not vals:
        return 0.0, 1e100
    d = np.unique(np.asarray(vals, dtype=float))
    dd = d.reshape(-1, 1)
    t = np.maximum(B[None, :] + dd * cb[None, :], R[None, :] + dd * cr[None, :])
    if 0 <= p0 < t.shape[1]:
        t[:, p0] += d * d
    row = t.max(axis=1)
    v0 = O0 + 2.0 * d * (xi - xj) - 2.0 * d * d
    cand = np.maximum(row, v0)
    k = int(np.argmin(cand))
    return float(d[k]), float(cand[k])

def _lo(h0, budget, rng):
    t0 = time.time()
    dl = t0 + max(0.0, budget)
    n = h0.size
    if n < 2:
        return h0, _c5(h0)
    S = 0.5 * n
    h = _pr(h0, S)
    B, R, O0, C, mx = _st(h)
    bh = h.copy()
    bc = C
    g = None
    since = 0
    no = 0
    temp = 0.4
    q = max(12, n // 10)
    K = 40 if n <= 700 else 24

    while time.time() < dl - 1.0:
        if since > 40:
            B, R, O0, C, mx = _st(h)
            since = 0
            g = None
            if C > bc + 0.02:
                h = _pr(bh, S)
                B, R, O0, C, mx = _st(h)
        if g is None:
            g = _gs(h, B, R, O0, mx)
            o = np.argsort(g)
            low = o[:q]
            high = o[::-1][:q]

        bp = None
        for _ in range(K):
            i = int(high[rng.randint(len(high))])
            j = int(low[rng.randint(len(low))])
            if i == j:
                continue
            xi = h[i]
            xj = h[j]
            if g[i] < g[j]:
                i, j = j, i
                xi, xj = xj, xi
            D = min(xi, 1.0 - xj)
            if D <= 1e-7:
                if min(xj, 1.0 - xi) <= 1e-7:
                    continue
                i, j = j, i
                xi, xj = xj, xi
                D = min(xi, 1.0 - xj)
            if D <= 1e-7:
                continue
            cb, cr = _mc(h, i, j)
            p0 = abs(i - j) - 1
            d, raw = _bd(B, R, O0, cb, cr, p0, xi, xj, D)
            if d > 1e-12 and raw < mx - 1e-12:
                if bp is None or raw < bp[5]:
                    bp = (i, j, cb, cr, d, raw)

        if bp is not None:
            i, j, cb, cr, d, raw = bp
            p0 = abs(i - j) - 1
            O0 = _ap(h, B, R, O0, i, j, d, cb, cr, p0)
            mx = raw
            C = 2.0 * mx / n
            since += 1
            no = 0
            temp = max(0.02, temp * 0.995)
            if C < bc:
                bc = C
                bh = h.copy()
            if since % 6 == 0:
                g = None
            if since % 30 == 0:
                B, R, O0, C, mx = _st(h)
                since = 0
                g = None
                if C < bc:
                    bc = C
                    bh = h.copy()
        else:
            no += 1
            temp = max(0.02, temp * 0.995)
            if no % 3 == 0:
                acc = False
                for _ in range(6):
                    i = int(rng.randint(n))
                    j = int(rng.randint(n))
                    if i == j:
                        continue
                    xi = h[i]
                    xj = h[j]
                    if g is not None and g[i] < g[j]:
                        i, j = j, i
                        xi, xj = xj, xi
                    D = min(xi, 1.0 - xj)
                    if D <= 1e-7:
                        if min(xj, 1.0 - xi) <= 1e-7:
                            continue
                        i, j = j, i
                        xi, xj = xj, xi
                        D = min(xi, 1.0 - xj)
                    if D <= 1e-7:
                        continue
                    cb, cr = _mc(h, i, j)
                    p0 = abs(i - j) - 1
                    d, raw = _bd(B, R, O0, cb, cr, p0, xi, xj, D)
                    if d <= 1e-12:
                        continue
                    want = raw < mx - 1e-12
                    if not want and raw - mx < temp:
                        want = np.exp(-(raw - mx) / temp) > rng.random()
                    if want:
                        O0 = _ap(h, B, R, O0, i, j, d, cb, cr, p0)
                        mx = raw
                        C = 2.0 * mx / n
                        since += 1
                        acc = True
                        break
                if acc:
                    no = 0
                    if C < bc:
                        bc = C
                        bh = h.copy()
                    if since > 30:
                        B, R, O0, C, mx = _st(h)
                        since = 0
                        g = None
                    continue

            if no >= 30:
                sh = 0.02
                ok = False
                for _ in range(4):
                    h2 = _pr(h - sh * g, S)
                    B2, R2, O02, C2, mx2 = _st(h2)
                    if C2 < C - 1e-10:
                        h = h2
                        B, R, O0, C, mx = B2, R2, O02, C2, mx2
                        ok = True
                        break
                    sh *= 0.4
                if ok:
                    since = 0
                    no = 0
                    g = None
                    if C < bc:
                        bc = C
                        bh = h.copy()
                    continue
                if C > bc + 0.01:
                    h = _pr(bh, S)
                    st = _st(h)
                else:
                    amp = 0.05 + 0.05 * (no / 200.0)
                    hp2 = _pr(bh + (rng.random(n) - 0.5) * amp, S)
                    st = _st(hp2)
                    if st[3] > bc + 0.02:
                        h = _pr(bh, S)
                        st = _st(h)
                    else:
                        h = hp2
                B, R, O0, C, mx = st
                no = 0
                since = 0
                g = None
                if C < bc:
                    bc = C
                    bh = h.copy()

    B, R, O0, C, mx = _st(h)
    if C < bc:
        bc = C
        bh = h.copy()
    bh = _pr(bh, S)
    bc = _c5(bh)
    return bh, bc

def _rs(h, m):
    n = h.size
    m = int(m)
    if m % 2:
        m += 1
    if n == m:
        return _pr(h, 0.5 * m)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    return _pr(np.interp(x1, x0, h), 0.5 * m)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        ip = np.asarray(initial_h_values, dtype=float)
    else:
        ip = np.full(560, 0.5)
    if ip.size == 0:
        ip = np.full(560, 0.5)

    n0 = ip.size
    S = 0.5 * n0
    candidates = []

    h0 = _pr(ip, S)
    candidates.append((_c5(h0), h0))
    candidates.append((_c5(_pr(1.0 - h0, S)), _pr(1.0 - h0, S)))
    candidates.append((_c5(_pr(h0[::-1], S)), _pr(h0[::-1], S)))

    x = (np.arange(n0) + 0.5) * (2.0 / n0)
    for fq in (1, 2, 3, 4):
        for ph in (0.0, np.pi * 0.5):
            hv = _pr(0.5 + 0.5 * np.cos(np.pi * fq * x + ph), S)
            candidates.append((_c5(hv), hv))
    ht = np.minimum(x, 2.0 - x)
    if ht.sum() > 0:
        ht = ht * (S / ht.sum())
    candidates.append((_c5(_pr(np.clip(ht, 0.0, 1.0), S)), _pr(np.clip(ht, 0.0, 1.0), S)))

    bh = None
    bc = 10.0
    for c, hv in candidates:
        if c < bc:
            bc = c
            bh = hv.copy()
    if bh is None:
        bh = _pr(ip, S)
        bc = _c5(bh)

    rem = budget_s - (time.time() - t0)
    reserve = 0.0
    if rem > 600 and bh.size < 1120:
        reserve = min(200.0, rem - 150.0)
    t1 = max(10.0, rem - reserve - 20.0)
    h1, c1 = _lo(bh.copy(), t1, rng)
    if c1 < bc:
        bh, bc = h1, c1

    rem2 = budget_s - (time.time() - t0)
    if reserve > 0 and rem2 > 150 and bh.size < 1120:
        m = 2 * bh.size
        if m > 1120:
            m = 1120
        if m > bh.size:
            h2, c2 = _lo(_rs(bh, m), max(10.0, rem2 - 10.0), rng)
            if c2 < bc:
                bh, bc = h2, c2

    bh = _pr(bh, 0.5 * bh.size)
    bc = _c5(bh)
    return bh, float(bc), int(bh.size)
```
