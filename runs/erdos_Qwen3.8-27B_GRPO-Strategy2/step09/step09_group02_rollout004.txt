    s = h.sum()
    for _ in range(200):
        if abs(s - target) < 1e-12:

        h *= target / s
        h = np.clip(h, 0.0, 1.0)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0

        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:

            a = min(abs(diff), 1.0 - h[idx]) if diff > 0 else min(-diff, h[idx])
            h[idx] += a if diff > 0 else -a
            diff += -a if diff > 0 else a

def _smooth_val(c, alpha, m_lags):
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha, idx

    c = _corr(x)
    val, _ = _smooth_val(c, alpha, m_lags)
    return val + lam * (x.sum() - target) ** 2

    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:

            g[M:] -= w[ii] * x[:n - M]
