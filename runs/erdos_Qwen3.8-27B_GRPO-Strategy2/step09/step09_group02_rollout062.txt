                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_topk(h, topk):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    idx = np.argsort(c)[-topk:]
    g = np.zeros(n)
    for k in idx:
        L = k - (n - 1)
        if L >= 0:
            g[L:] += dx * (1.0 - h[:n - L])
            g[:n - L] -= dx * h[L:]
        else:
            M = -L
            g[:n - M] += dx * (1.0 - h[M:])
            g[M:] -= dx * h[:n - M]
    return g - g.mean()

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _pg(h, tlimit, mu=0.85, tk0=40, tk1=3, s0=0.004):
    n = len(h)
    v = np.zeros(n)
    st = s0
    cur = _eval(h)
    best, bc = h.copy(), cur
    stall, i = 0, 0
    while time.time() < tlimit:
        fr = min(1.0, i / 8000.0)
        tk = max(tk1, int(tk0 - (tk0 - tk1) * fr))
        g = _grad_topk(h, tk)
        v = mu * v + (1.0 - mu) * g
        ng = np.linalg.norm(v)
        if ng < 1e-16:
            break
        v /= ng
        ok = False
        for _ in range(15):
            hn = _proj(h - st * v, n // 2)
            cn = _eval(hn)
            if cn < cur - 1e-18:
                h, cur = hn, cn
                st = min(st * 1.08, 0.12)
                if cur < bc:
                    bc, best = cur, h.copy()
                ok, stall = True, 0
                break
            st *= 0.4
        if not ok:
            st = max(st * 1.5, s0 * 0.5)
            stall += 1
            if stall > 300:
                v *= 0.1
                stall = 0
        i += 1
    return best, bc

def _ds(h, f):
    m = len(h) // f
    return h[:m * f].reshape(m, f).mean(axis=1)

def _us(h):
    m = len(h)
    o = np.empty(2 * m)
    o[0::2] = h
    o[1::2] = h
    return o

def _ctf(h, nt, tlim):
    lv = [h.copy()]
    while len(lv[-1]) > max(28, nt // 8):
        lv.append(_ds(lv[-1], 2))
    hw = lv[-1].copy()
    for i in range(len(lv) - 1, -1, -1):
        if time.time() > tlim:
            break
        nl = len(lv[i])
        tk = max(5, int(20 * nl / nt))
        s0 = 0.02 * (nt / nl)
        hw, _ = _pg(hw, tlim, mu=0.82, tk0=tk, tk1=max(3, tk // 4), s0=s0)
        if i > 0:
            while len(hw) < len(lv[i - 1]):
                hw = _us(hw)
                hw = _proj(hw, len(hw) // 2)
            best_b, best_c = None, _eval(lv[i - 1])
            for alpha in [0.3, 0.5, 0.7, 1.0]:
                bl = alpha * hw + (1.0 - alpha) * lv[i - 1]
                bl = _proj(bl, len(bl) // 2)
                c = _eval(bl)
                if c < best_c:
                    best_c, best_b = c, bl
            if best_b is not None:
                hw = best_b
    return hw

def _lsearch(h, niter, rng, tlim):
    n = len(h)
    cur = _eval(h)
    best, bc = h.copy(), cur
    for _ in range(niter):
        if time.time() > tlim:
            break
        sc = 1 << rng.randint(0, 5)
        np_ = max(1, 8 // sc)
        idx = rng.randint(0, n, size=np_)
        amp = 0.015 / np.sqrt(sc)
        d = amp * rng.randn(np_)
        d -= d.mean()
        hn = h.copy()
        hn[idx] += d
        if hn.min() < -0.1 or hn.max() > 1.1:
            continue
        hn = _repair(hn, n)
        cn = _eval(hn)
        if cn < cur or (cur - bc < 1e-6 and rng.rand() < 0.03):
            h, cur = hn, cn
            if cur < bc:
                bc, best = cur, h.copy()
    return best, bc

def _polish(h, niter, rng, tlim):
    n = len(h)
    cur = _eval(h)
    best, bc = h.copy(), cur
    for it in range(niter):
        if time.time() > tlim:
            break
        fr = it / max(1, niter)
        amp = 5e-7 * (1.0 - fr) + 1e-9
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 25)) % n
        d = amp * rng.rand() * (1 if rng.rand() < 0.5 else -1)
        if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
            h[i] += d; h[j] -= d
            cn = _eval(h)
            if cn < cur - 1e-18:
                cur = cn
                if cn < bc:
                    bc, best = cn, h.copy()
            else:
                h[i] -= d; h[j] += d
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = _repair(h, n)
    bh, bc5 = h.copy(), _eval(h)

    # Phase 1: Coarse-to-fine
    t1 = t0 + 100
    h_cf = _ctf(h, n, t1)
    h_cf = _proj(h_cf, n // 2)
    c = _eval(h_cf)
    if c < bc5:
        bc5, bh = c, h_cf

    # Phase 2: Long projected gradient (broad focus)
    t2 = t0 + 400
    bh, bc5 = _pg(bh, t2, mu=0.85, tk0=35, tk1=3, s0=0.004)

    # Phase 3: Second CTF pass from improved point
    if time.time() < t0 + budget_s - 400:
        t3 = t0 + 550
        h_cf2 = _ctf(bh, n, t3)
        h_cf2 = _proj(h_cf2, n // 2)
        c = _eval(h_cf2)
        if c < bc5:
            bc5, bh = c, h_cf2

    # Phase 4: Focused gradient (narrow topk)
    t4 = t0 + budget_s - 250
    if time.time() < t4:
        bh, bc5 = _pg(bh, t4, mu=0.88, tk0=15, tk1=2, s0=0.002)

    # Phase 5: Multi-scale local search
    t5 = t0 + budget_s - 100
    if time.time() < t5:
        ni = int((t5 - time.time()) * 80) + 500
        bh, bc5 = _lsearch(bh, ni, rng, t5)

    # Phase 6: Fine 2-point polish (adaptive amplitude)
    t6 = t0 + budget_s - 15
    if time.time() < t6:
        ni = int((t6 - time.time()) * 400) + 2000
        bh, bc5 = _polish(bh, ni, rng, t6)

    # Phase 7: Ultra-fine polish
    t7 = t0 + budget_s - 1
    if time.time() < t7:
        ni = int((t7 - time.time()) * 600) + 500
        bh, bc5 = _polish(bh, ni, rng, t7)

    bh = _repair(bh, n)
    bc5 = _eval(bh)
    return (bh, bc5, n)
