import numpy as np
import time
import math

def _repair(h):
    n = len(h)
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if s < 1e-14:
        h[:] = 0.5
        return h
    for _ in range(40):
        s = float(h.sum())
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)[::-1]
            rem = diff
            for i in order:
                add = min(rem, 1.0 - h[i])
                if add <= 0:
                    continue
                h[i] += add
                rem -= add
                if abs(rem) < 1e-13:
                    break
        else:
            order = np.argsort(h)
            rem = -diff
            for i in order:
                sub = min(rem, h[i])
                if sub <= 0:
                    continue
                h[i] -= sub
                rem -= sub
                if abs(rem) < 1e-13:
                    break
    return h

def _proj(x, target):
    x = np.asarray(x, dtype=float)
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(32):
        mid = 0.5 * (lo + hi)
        if float(np.clip(x - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(y.sum())
    if abs(diff) > 1e-9:
        order = np.argsort(1.0 - y)[::-1] if diff > 0 else np.argsort(y)
        rem = float(diff)
        for i in order:
            if abs(rem) < 1e-12:
                break
            if rem > 0:
                add = min(rem, 1.0 - y[i])
                y[i] += add
                rem -= add
            else:
                sub = min(-rem, y[i])
                y[i] -= sub
                rem += sub
    return y

def _c_from_h(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _apply_point(c, h, p, d, dx, tmp):
    if d == 0.0:
        return
    n = len(h)
    s = dx * d
    c[p:p + n] += s
    np.multiply(h[::-1], s, out=tmp)
    c[p:p + n] -= tmp
    np.multiply(h, s, out=tmp)
    c[n - 1 - p:n - 1 - p + n] -= tmp
    c[n - 1] -= dx * d * d
    h[p] += d

def _top_weights(c, topm, alpha):
    m = len(c)
    if topm <= 0:
        topm = 1
    if topm >= m:
        idx = np.arange(m)
    else:
        idx = np.argpartition(c, m - topm)[-topm:]
    vals = c[idx]
    cmax = float(vals.max())
    if alpha <= 1e-12:
        vals = np.full(topm, 1.0 / topm)
    else:
        s = alpha * (vals - cmax)
        s = s - s.max()
        s = np.clip(s, -60.0, 0.0)
        vals = np.exp(s)
        sm = float(vals.sum())
        if sm <= 0.0:
            vals[0] = 1.0
        else:
            vals /= sm
    return idx, vals

def _gradient_top(h, idx, vals, buf):
    n = len(h)
    g = np.zeros(n)
    off = n - 1
    for ii in range(len(idx)):
        w = float(vals[ii])
        if w <= 1e-15:
            continue
        L = int(idx[ii]) - off
        if L >= 0:
            l = n - L
            if l <= 0:
                continue
            g[L:n] += w
            np.multiply(h[:l], w, out=buf[:l])
            g[L:n] -= buf[:l]
            np.multiply(h[L:], w, out=buf[:l])
            g[:l] -= buf[:l]
        else:
            M = -L
            l = n - M
            if l <= 0:
                continue
            g[:l] += w
            np.multiply(h[M:], w, out=buf[:l])
            g[:l] -= buf[:l]
            np.multiply(h[:l], w, out=buf[:l])
            g[M:] -= buf[:l]
    return g

def _pair_delta_cm(h, c, p, q, a, dx, delta, cnew, tmp):
    n = len(h)
    L = 2 * n - 1
    off = n - 1
    delta.fill(0.0)
    s = dx * a
    delta[p:p + n] += s
    np.multiply(h[::-1], s, out=tmp)
    delta[p:p + n] -= tmp
    np.multiply(h, s, out=tmp)
    delta[n - 1 - p:n - 1 - p + n] -= tmp
    delta[off] -= dx * a * a

    s2 = -s
    delta[q:q + n] += s2
    np.multiply(h[::-1], s2, out=tmp)
    delta[q:q + n] -= tmp
    np.multiply(h, s2, out=tmp)
    delta[n - 1 - q:n - 1 - q + n] -= tmp
    delta[off] -= dx * a * a

    a2 = dx * a * a
    if p != q:
        i1 = off + (p - q)
        if 0 <= i1 < L:
            delta[i1] += a2
        i2 = off + (q - p)
        if 0 <= i2 < L:
            delta[i2] += a2
    np.add(c, delta, out=cnew)
    return float(cnew.max())

            if noimp > 120:
                if allow_worse:
                    h_try = _proj(h + 0.05 * rng.randn(n), 0.5 * n)
                    c_try = _c_from_h(h_try)
                    if float(c_try.max()) < cur + 5e-4:
                        h = h_try
                        c = c_try
                        cur = float(c.max())
                h = _repair(h)
                c = _c_from_h(h)
                cur = float(c.max())
                step = 0.8
                noimp = 0
                idx, vals = _top_weights(c, 20, 50.0)
                g = _gradient_top(h, idx, vals, gbuf)
                d = -g
                d -= d.mean()
                cosort = np.argsort(d)[::-1]
                nsort = np.argsort(d)
    h = _repair(best_h)
    cb = _c_from_h(h)
    return h, cb

def _pg_search(h, c, deadline, rng, topm=20, alpha=30.0, step0=0.10, momentum=0.5, refresh=25):
    n = len(h)
    target = 0.5 * n
    h = h.copy()
    c = c.copy()
    cur = float(c.max())
    best_h = h.copy()
    best_c = cur
    v = np.zeros(n)
    step = step0
    gbuf = np.empty(n)
    idx = None
    vals = None
    g = None
    it = 0
    noimp = 0
    while time.time() < deadline:
        if idx is None or it % refresh == 0:
            tm = topm
            al = alpha
            if noimp > 25:
                tm = min(3 * topm, len(c) - 1)
                al = max(5.0, 0.5 * alpha)
            idx, vals = _top_weights(c, tm, al)
            g = _gradient_top(h, idx, vals, gbuf)
        v = momentum * v - g
        v -= v.mean()
        norm = float(np.linalg.norm(v))
        if norm < 1e-12:
            v = rng.randn(n)
            v -= v.mean()
            norm = float(np.linalg.norm(v))
        if norm <= 0.0:
            v = np.ones(n)
        else:
            v /= norm
        if step < 1e-6:
            step = 0.25 * step0
        taken = False
        for _ in range(7):
            if time.time() >= deadline:
                break
            h_try = _proj(h + step * v, target)
            c_try = _c_from_h(h_try)
            cm = float(c_try.max())
            if cm < cur - 1e-14:
                h = h_try
                c = c_try
                cur = cm
                taken = True
                noimp = 0
                step = min(step * 1.12, step0 * 25.0)
                break
            step *= 0.5
            if step < 1e-9:
                break
        if not taken:
            step = min(step * 1.7, step0 * 25.0)
            noimp += 1
            if noimp > 30:
                v += 0.5 * rng.randn(n)
                v -= v.mean()
                noimp = 0
            if it % 5000 == 0:
                h = _repair(h)
                c = _c_from_h(h)
                cur = float(c.max())
        it += 1
        if cur < best_c:
            best_c = cur
            best_h = h.copy()
    best_h = _repair(best_h)
    cb = _c_from_h(best_h)
    return best_h, cb

def _run_one(h, deadline, rng):
    h = _repair(h)
    c = _c_from_h(h)
    now = time.time()
    total = max(0.0, deadline - now)
    if total <= 1e-6:
        return h, c
    if total > 60.0:
        d1 = now + 0.30 * total
        h, c = _pg_search(h, c, d1, rng, 10, 80.0, 0.12)
        d2 = now + 0.55 * total
        h, c = _pg_search(h, c, d2, rng, 40, 20.0, 0.08)
    d3 = now + 0.85 * total
    if d3 > now:
        h, c = _local_delta_search(h, c, d3, rng, allow_worse=True)
    if total > 40.0:
        d4 = now + 0.93 * total
        if d4 < deadline - 2.0:
            h, c = _pg_search(h, c, d4, rng, 20, 50.0, 0.06)
    h, c = _local_delta_search(h, c, deadline, rng, allow_worse=False)
    return h, c

def _rank_binary(h):
    n = len(h)
    order = np.argsort(h)
    hb = np.zeros(n, dtype=float)
    k = n // 2
    hb[order[-k:]] = 1.0
    if n % 2 == 1:
        hb[order[k]] = 0.5
    return hb

def _symmetrize(h):
    ys = 0.5 * (h + 1.0 - h[::-1])
    n = len(h)
    if n % 2 == 1:
        ys[n // 2] = 0.5
    return _repair(ys)

def _downsample_to(h, m):
    n = len(h)
    if m <= 0:
        m = 1
    if m >= n:
        return _repair(h)
    f = max(1, n // m)
    m = n // f
    if m <= 0:
        m = 1
    arr = np.reshape(h[:m * f], (m, f)).mean(axis=1)
    return _repair(arr)

def _upsample_to(h, m):
    n = len(h)
    f = max(1, m // n)
    arr = np.repeat(h, f)
    if len(arr) > m:
        arr = arr[:m]
    return _repair(arr)

def _consider(h, best_h, best_c, best_n):
    hh = _repair(h)
    cc = float(_c_from_h(hh).max())
    if cc < best_c:
        return hh, cc, len(hh)
    return best_h, best_c, best_n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    rng = np.random.RandomState(seed)
    ihv = globals().get('initial_h_values')
    if ihv is not None and len(ihv) > 0:
        h0 = np.asarray(ihv, dtype=float).copy()
    else:
        n = 560
        x = np.linspace(0.0, 2.0, n)
        h0 = 0.5 + 0.45 * np.sin(2.0 * np.pi * x) + 0.12 * np.sin(7.0 * np.pi * x)
        h0 = np.clip(h0, 0.0, 1.0)
    h0 = _repair(h0)
    best_h = h0.copy()
    best_c = float(_c_from_h(h0).max())
    best_n = len(h0)

    d1 = t0 + 0.38 * budget
    if time.time() < d1 - 1.0:
        h1, c1 = _run_one(h0, d1, rng)
        best_h, best_c, best_n = _consider(h1, best_h, best_c, best_n)

    d2 = d1 + 0.09 * budget
    if time.time() < d2 - 1.0:
        h2, c2 = _run_one(_symmetrize(best_h), d2, rng)
        best_h, best_c, best_n = _consider(h2, best_h, best_c, best_n)

    d3 = d2 + 0.09 * budget
    if time.time() < d3 - 1.0:
        h3, c3 = _run_one(_rank_binary(best_h), d3, rng)
        best_h, best_c, best_n = _consider(h3, best_h, best_c, best_n)

    d4 = d3 + 0.15 * budget
    if time.time() < d4 - 1.0:
        if best_n > 400:
            mc = 280
        elif best_n > 200:
            mc = 140
        else:
            mc = max(80, best_n // 2)
        hc, chc = _run_one(_downsample_to(best_h, mc), d4, rng)
        best_h, best_c, best_n = _consider(hc, best_h, best_c, best_n)
        if len(h0) > len(hc):
            hup = _upsample_to(hc, len(h0))
            best_h, best_c, best_n = _consider(hup, best_h, best_c, best_n)

    d5 = min(d4 + 0.27 * budget, t0 + budget - 2.0)
    if time.time() < d5 - 0.5:
        hf = _repair(best_h)
        cf = _c_from_h(hf)
        hf, cf = _local_delta_search(hf, cf, d5, rng, allow_worse=False)
        best_h, best_c, best_n = _consider(hf, best_h, best_c, best_n)

    best_h = _repair(best_h)
    best_c = float(_c_from_h(best_h).max())
    return best_h, best_c, len(best_h)
