import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smooth_obj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    ml = min(ml, len(c) - 1)
    idx = np.argpartition(-c, ml)[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    ml = min(ml, len(c) - 1)
    idx = np.argpartition(-c, ml)[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fn(x, half):
    return x.sum() - half

def _sum_jac(x, half):
    return np.ones(len(x))

def _proj_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp(h, alpha, ml, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, ml), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 100})
        return _repair(res.x, n)
    except Exception:
        return h

def _nesterov(h, alpha, ml, steps, lr0, mom, t_fn, rem_fn):
    n = len(h)
    v = np.zeros(n)
    lr = lr0
    for it in range(steps):
        if rem_fn() < 30:
            break
        g = _smooth_grad(h, alpha, ml)
        v = mom * v - lr * g
        h_new = _proj_sum(h + v, n * 0.5)
        step = lr
        taken = False
        for _t in range(12):
            h_try = _proj_sum(h - step * g, n * 0.5)
            c_try = _eval(h_try)
            if c_try < _eval(h):
                h = h_try
                v = mom * v
                lr = min(step * 1.2, lr0 * 30.0)
                taken = True
                break
            step *= 0.5
        if not taken:
            h = h_new
            lr = min(lr * 1.1, lr0 * 30.0)
    return h

def _fine_search(h, n_iter, rng, alpha, ml, t_fn, rem_fn):

    c5_cur = _eval(h)
    best_h = h.copy()
    best_c = c5_cur
    g = _smooth_grad(h, alpha, ml)
    step_decay = 0.9995
    scale = 1e-4
    stall = 0
    for _ in range(n_iter):
        if rem_fn() < 10:
            break
        if stall > 5000:
            scale = max(1e-6, scale * 0.7)
            stall = 0
        top3 = np.argpartition(-np.abs(g), 3)[:3]
        i = top3[rng.randint(3)]
        j = (i + 1 + rng.randint(0, 6)) % n
        d = scale * rng.randn()
        h_new = h.copy()
        h_new[i] += d
        h_new[j] -= d
        if h_new.min() < 0 or h_new.max() > 1:
            d *= -0.5
            h_new[i] += d
            h_new[j] -= d
            if h_new.min() < 0 or h_new.max() > 1:
                continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5_cur:
            h = h_new
            c5_cur = c5n
            stall = 0
            g = _smooth_grad(h, alpha, ml)
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
                scale = max(scale * 1.05, 2e-4)
        else:
            stall += 1
            scale *= step_decay
    return best_h, best_c

def _polish(h, n_iter, rng, t_fn, rem_fn):
    n = len(h)
    c5_cur = _eval(h)
    h2 = h.copy()
    for _ in range(n_iter):
        if rem_fn() < 1:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 10)) % n
        d = 5e-7 * rng.randn()
        ni, nj = h2[i] + d, h2[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h2[i], h2[j] = ni, nj
            c5n = _eval(h2)
            if c5n < c5_cur:
                c5_cur = c5n
            else:
                h2[i], h2[j] = ni - d, nj + d
    return h2, c5_cur
