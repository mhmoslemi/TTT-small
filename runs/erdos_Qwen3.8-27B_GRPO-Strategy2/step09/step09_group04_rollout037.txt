    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:

                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:

                h[idx] -= a; diff += a

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _subgrad(h, m_act):
    """Subgradient of max of top m_act correlations."""
    n = len(h)
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(c_raw)[-m_act:]
    g = np.zeros(n)
    for k in top:
        L = k - (n - 1)
        if L >= 0:
            g[:n-L] += (1.0 - h[L:])
            g[L:] -= h[:n-L]
        else:
            M = -L
            g[M:] += (1.0 - h[:n-M])
            g[:n-M] -= h[M:]
    g /= len(top)
    return g * dx

def _smooth_obj(x, alpha, m_act):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    top = np.argsort(c)[-m_act:]
    s = alpha * (c[top] - c[top].max())
    s -= s.max()
    return c[top].max() + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_act):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    top = np.argsort(c)[-m_act:]
    s = alpha * (c[top] - c[top].max())
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(top):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[:n-L] += wk * (1.0 - x[L:])
            g[L:] -= wk * x[:n-L]
        else:
            M = -L
            g[M:] += wk * (1.0 - x[:n-M])
            g[:n-M] -= wk * x[M:]
    return g

def _cons_fun(x, hn):
    return x.sum() - hn

def _cons_jac(x, hn):
    return np.ones(len(x))

def _slsqp_pass(h, alpha, m_act, maxiter, n):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (n*0.5,)}]
    try:
        r = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha, m_act),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(r.x, n)
    except:
        return h

def _nesterov_subgrad(h, n, m_act, niter, base_step, mu, seed):
    """Nesterov-accelerated subgradient method."""
    rng = np.random.RandomState(seed)
    target = n / 2.0
    h_curr = h.copy()
    v = np.zeros(n)
    best_local = _eval(h_curr)
    best_h = h_curr.copy()
    
    for it in range(niter):
        g = _subgrad(h_curr, m_act)
        step = base_step / (1.0 + 0.002 * it)
        # Nesterov: evaluate gradient at reflected point
        h_reflect = _proj(h_curr - step * v, target)
        g_ref = _subgrad(h_reflect, m_act)
        v = mu * v + (1.0 - mu) * g_ref
        h_new = _proj(h_reflect - step * v, target)
        cn = _eval(h_new)
        if cn <= best_local + 1e-12:
            h_curr = h_new
            if cn < best_local:
                best_local = cn
                best_h = h_curr.copy()
        else:
            # Backtrack
            step2 = step * 0.5
            h_new = _proj(h_curr - step2 * v, target)
            cn = _eval(h_new)
            h_curr = h_new
            if cn < best_local:
                best_local = cn
                best_h = h_curr.copy()
    return best_h, best_local

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_all(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_shift(h, L):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if L >= 0:
        g[:n - L] += (1.0 - h[L:])
        g[L:] -= h[:n - L]
    else:
        M = -L
        g[M:] += (1.0 - h[:n - M])
        g[:n - M] -= h[M:]
    return g * dx

def _subgrad_top(h, m_act):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(c)[-m_act:]
    g = np.zeros(n)
    for k in top:
        g += _grad_shift(h, k - (n - 1))
    return g / len(top)

def _smooth_obj(x, alpha, m_act):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    top = np.argsort(c)[-m_act:]
    s = alpha * (c[top] - c[top].max())
    s -= s.max()
    return c[top].max() + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_act):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    top = np.argsort(c)[-m_act:]
    s = alpha * (c[top] - c[top].max())
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(top):
        if w[ii] < 1e-15:
            continue
        g += w[ii] * _grad_shift(x, k - (n - 1))
    return g

def _cf(x, hn):
    return x.sum() - hn

def _cj(x, hn):
    return np.ones(len(x))

def _slsqp(h, alpha, m_act, iters, n):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n * 0.5,)}]
    try:
        r = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha, m_act),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': iters, 'ftol': 1e-18})
        return _repair(r.x, n)
    except:
        return h

def _nesterov_sg(h, n, m_act, niter, bstep, mu, t_end):

    hc = h.copy()
    v = np.zeros(n)
    bc = _eval(hc)

        if time.time() > t_end:
            break
        step = bstep / (1.0 + 0.003 * it)

        g = _subgrad_top(hr, m_act)
        v = mu * v + (1.0 - mu) * g
        hn = _proj(hr - step * v, target)
        cn = _eval(hn)
        if cn <= bc + 1e-11:
            hc = hn
            if cn < bc:
                bc, bh = cn, hc.copy()
        else:
            hn = _proj(hc - 0.3 * step * v, target)
            cn = _eval(hn)
            hc = hn
            if cn < bc:
                bc, bh = cn, hc.copy()
        m_act_eff = min(80, m_act + it // 2000)
    return bh, bc

def _equalize(h, n, m_act, niter, t_end):
    target = n / 2.0
    hc = h.copy()
    bc = _eval(hc)
    bh = hc.copy()
    step = 0.01
    for it in range(niter):
        if time.time() > t_end:
            break
        c = _corr_all(hc)
        top = np.argsort(c)[-m_act:]
        ts = top[np.argsort(c[top])]
        hi_k = ts[-1]
        lo_k = ts[0]
        g = _grad_shift(hc, hi_k - (n - 1)) - _grad_shift(hc, lo_k - (n - 1))
        hn = _proj(hc - step * g, target)
        cn = _eval(hn)
        if cn <= bc + 1e-12:
            hc = hn
            if cn < bc:
                bc, bh = cn, hc.copy()
        else:
            step *= 0.6
        step = min(step * 1.02, 0.01)
    return bh, bc
