def _optimal_t(g_active, beta_active, h_j):
    # Find t in [0,1] minimizing max_k (g_k + beta_k * t)
    # The functions are linear in t, so the max is piecewise linear
    # The minimum occurs at t=0, t=1, or at an intersection of two lines
    K = len(beta_active)
    candidates = [0.0, 1.0]
    for i in range(K):
        for j in range(i+1, K):
            db = beta_active[j] - beta_active[i]
            if abs(db) > 1e-15:
                t = (g_active[i] - g_active[j]) / db
                if 0.0 <= t <= 1.0:
                    candidates.append(t)
    best_t = 0.0
    best_val = np.inf
    for t in candidates:
        val = np.max(g_active + beta_active * t)
        if val < best_val:
            best_val = val
            best_t = t
    return best_t

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-15:
                break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _cd_pass(h, rng, n_bisect=25):
    n = len(h); c = _corr(h); cmax_sz = 2*n - 1; nm1 = n - 1
    jlist = np.arange(n); rng.shuffle(jlist)
    for j in jlist:
        j = int(j)
        beta = np.zeros(cmax_sz)
        lo1 = j; hi1 = min(cmax_sz - 1, j + nm1)
        lags1 = np.arange(lo1, hi1 + 1) - nm1
        beta[lo1:hi1+1] += (1.0 - h[j - lags1])
        lo2 = max(0, nm1 - j); hi2 = min(cmax_sz - 1, cmax_sz - 1 - j)
        lags2 = np.arange(lo2, hi2 + 1) - nm1
        beta[lo2:hi2+1] -= h[j + lags2]
        g = c - beta * h[j]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(n_bisect):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0

            else:
                lo_t = t1

            c += beta * (t_new - h[j]); h[j] = t_new
    return h

def _sqp(h, K, max_iter):
    n = len(h); c = _corr(h)
    active = np.argsort(c)[-K:]
    for _ in range(max_iter):
        G = np.column_stack([_grad_lag(h, k - (n - 1), n) for k in active])
        nv = n + 1
        c_obj = np.zeros(nv); c_obj[-1] = 1.0
        A_ub = np.hstack([G.T, -np.ones((K, 1))])
        A_eq = np.zeros((1, nv)); A_eq[0, :n] = 1.0
        bounds = [(-h[i], 1.0 - h[i]) for i in range(n)] + [(0.0, None)]
        try:

        if res.status != 0:

        delta = res.x[:n]

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)

        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-15:

        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:

                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a

        scale = scales[trial % len(scales)]
        mode = trial % 4
        n_cp = 3 + (trial % 4)
        hp = _perturb(best_h, rng, n, mode, scale)
        for _ in range(n_cp):
            hp = _cd_pass(hp, rng)
            hp = _repair(hp, n)
        c5p = _eval(hp)
        if c5p < best_c5:
            best_c5, best_h = c5p, hp.copy()
            if time.time() < end3 - 15:
                hs = _sqp(best_h.copy(), 15, 10)
                c5s = _eval(hs)
                if c5s < best_c5:
                    best_c5, best_h = c5s, hs.copy()
        trial += 1

    # Phase 4: Final CD + SQP polish
    end4 = t0 + budget_s * 0.93
    while time.time() < end4:
        best_h = _cd_pass(best_h, rng, n_bisect=30)
        best_h = _repair(best_h, n)
        c5 = _eval(best_h)
        if c5 < best_c5:
            best_c5 = c5
        if time.time() < end4 - 20:
            hs = _sqp(best_h.copy(), 20, 10)
            c5s = _eval(hs)
            if c5s < best_c5:
                best_h, best_c5 = hs.copy(), c5s

    # Phase 5: Final cleanup
    if time.time() < t0 + budget_s - 4:
        best_h = _cd_pass(best_h, rng, n_bisect=30)
        best_h = _repair(best_h, n)
        c5 = _eval(best_h)
        if c5 < best_c5:
            best_c5 = c5

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
