def _slp(h, max_iter=20, n_top=None):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best_c5 = _eval(h_cur)
    best_h = h_cur.copy()
    
    for it in range(max_iter):
        c = np.correlate(h_cur, 1-h_cur, mode='full') * dx
        # Select top shifts if n_top is specified
        if n_top and n_top < len(c):
            idx = np.argpartition(-c, n_top-1)[:n_top]
        else:
            idx = np.argsort(-c)[:len(c)]
        
        # Compute gradient of each c_k with respect to h
        # c_k = sum_i h[i] * (1-h[i+L]) * dx where L = k-(n-1)
        # dc_k/dh[j] for the shift L:
        #   From h[i] term: if i=j, contribution is (1-h[j+L])*dx for j+L < n
        #   From (1-h[i+L]) term: if i+L=j, i=j-L, contribution is -h[j-L]*dx for j-L >= 0
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
        
        for k in idx:
            L = int(k) - (n-1)
            if L >= 0:
                # c = sum_{i=0}^{n-1-L} h[i](1-h[i+L]) dx
                # dc/dh[j] = (1-h[j+L])*dx for j in [0, n-1-L]
                #          + (-h[j-L])*dx for j in [L, n-1]
                grad = np.zeros(n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - t) < 1e-10:
            break
    d = t - h.sum()
    if abs(d) > 1e-10:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * 2.0 / n)

    lo, hi = -5.0, 5.0

        if np.clip(h - mid, 0, 1).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    n = len(x); dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    ku = min(k, len(c) - 1)
    idx = np.argpartition(-c, ku - 1)[:ku]
    return float((np.mean(c[idx] ** p)) ** (1.0 / p))

def _lk_grad(x, k, p):
    n = len(x); dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    ku = min(k, len(c) - 1)
    idx = np.argpartition(-c, ku - 1)[:ku]
    ck = c[idx]
    mp = (np.mean(ck ** p)) ** ((p - 1) / p) / ku

        else:

    return x.sum() - len(x) * 0.5

    return np.ones(len(x))

                     options={'maxiter': mi, 'ftol': ft, 'maxls': 30})
        return _repair(r.x, n)
    except Exception:
        return h

    g = np.zeros(n)
    if L >= 0:
        g[:n - L] += (1 - h[L:])
        g[L:] += (-h[:n - L])
    else:

        g[:n - M] += (-h[M:])

def _slp_iter(h, n_top, t_start):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    cut = min(n_top, len(c) - 1)
    idx = np.argpartition(-c, cut - 1)[:cut]

        else:

    return bh, best

    g = _lk_grad(h, k, p)

    n = len(h); dx = 2.0 / n
    L = int(k) - (n - 1)
    g = np.zeros(n)
    if L >= 0:
        g[:n - L] += (1 - h[L:]) * dx
        g[L:] += (-h[:n - L]) * dx
    else:
        M = -L
        g[M:] += (1 - h[:n - M]) * dx
        g[:n - M] += (-h[M:]) * dx
    return g

def _slp_solve(h, n_top):
    n = len(h); dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    cut = min(n_top, len(c) - 1)
    idx = np.argpartition(-c, cut - 1)[:cut]
    hv = cp.Variable(n)
    tv = cp.Variable()
    cons = [hv >= 0, hv <= 1, cp.sum(hv) == n * 0.5]
    for k in idx:
        g = _gc(h, int(k))
        cons.append(g @ hv <= tv - c[k] + g @ h)
    prob = cp.Problem(cp.Minimize(tv), cons)
    for sv in [cp.GLOP, cp.CBC, cp.CVXOPT, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=sv, verbose=False)
            if hv.value is not None:
                return np.clip(hv.value, 0.0, 1.0)
        except Exception:
            continue
    return None

def _slp_phase(h, max_it, ntop, t_end):
    n = len(h)
    bc = _eval(h); bh = h.copy()
    for _ in range(max_it):
        if time.time() > t_end:
            break
        hn = _slp_solve(h, ntop)
        if hn is None:
            break
        hn = _repair(hn, n)
        c5 = _eval(hn)
        if c5 < bc:
            bc = c5; bh = hn.copy(); h = hn
        elif c5 < _eval(h) + 1e-9:
            h = hn
        else:
            h = _repair(0.5 * h + 0.5 * hn, n)
    return bh, bc

def _nest(h, gf, args, steps, lr0, mom):
    n = len(h); v = np.zeros(n)
    bc = _eval(h); bh = h.copy(); lr = lr0
    for _ in range(steps):
        g = gf(h, *args)
        v = mom * v - lr * g
        hn = _proj(h + v, n * 0.5)
        cn = _eval(hn)
        if cn < bc:
            bc = cn; bh = hn.copy()
        c0 = _eval(h); ok = False; st = lr
        for _t in range(12):
            ht = _proj(h - st * g, n * 0.5)
            if _eval(ht) < c0:
                h = ht; v = mom * v; lr = min(st * 1.05, lr0 * 30)
                ok = True; break
            st *= 0.5
        if not ok:
            h = hn; lr = min(lr * 1.03, lr0 * 30)
    return bh

def _ds(h, f):
    n = len(h); m = n // f
    return _repair(h[:m*f].reshape(m, f).mean(1).copy(), m)

def _us(h, f):
    n = len(h); out = np.empty(n * f)
    out.reshape(n, f)[:] = h[:, None]
    return _repair(out, n * f)

def _ls(h, ni, rng, k, p):
    n = len(h); cc = _eval(h); bc = cc; bh = h.copy()
    g = _lk_g(h, k, p)
    for _ in range(ni):
        az = np.flatnonzero(np.abs(g) > 1e-15)
        if len(az) < 2:
            break
        i = az[rng.randint(len(az))]; j = az[rng.randint(len(az))]
        d = 3e-5 * rng.randn()
        hn = h.copy(); hn[i] += d; hn[j] -= d
        if hn.min() < 0 or hn.max() > 1:
            continue
        hn = _repair(hn, n)
        cn = _eval(hn)
        if cn < cc:
            h = hn; cc = cn; g = _lk_g(h, k, p)
            if cn < bc:
                bc = cn; bh = h.copy()
    return bh, bc
