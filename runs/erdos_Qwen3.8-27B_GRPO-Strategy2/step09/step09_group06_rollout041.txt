        s = np.clip(s - s.max(), -500, 500)
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        dx_val = 2.0 / n
        for ii, kk in enumerate(idx):
            if w[ii] < 1e-15:
                continue
            wk = w[ii] * dx_val
            L = kk - (n - 1)
            if L >= 0:
                g[L:] += wk * (1.0 - h[:n - L])
                g[:n - L] -= wk * h[L:]
            else:
                M = -L
                g[:n - M] += wk * (1.0 - h[M:])
                g[M:] -= wk * h[:n - M]
        s_sum = h.sum() - n / 2.0
        g += 50.0 * s_sum
        v = mu * v + (1.0 - mu) * g
        taken = False
        for _t in range(15):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-15:
                h = h_new
                cur_c = c_new
                step = min(step * 1.1, step0 * 20.0)
                taken = True
                break
            step *= 0.35
        if not taken:
            step = min(step * 1.3, step0 * 20.0)
    return _repair(h, n)

def _local_search(h, n_iter, rng, base_step):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    for it in range(n_iter):
        frac = it / max(n_iter, 1)
        step = base_step * (1.5 - 0.5 * frac) * (0.5 + rng.rand())
        k = rng.choice([2, 2, 3, 3, 4, 5, 6])
        idx = rng.randint(0, n, size=k)
        d = step * rng.randn(k)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < -0.02 or h_new.max() > 1.02:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5cur:
            c5cur = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def _two_point_polish(h, n_polish, rng):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_polish):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 9)) % n
        d = 1e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _multi_res(h, r, alpha_list):
    n = len(h)
    nc = max(40, n // r)
    f = n // nc
    hc = _repair(h[:nc * f].reshape(nc, f).mean(axis=1).copy(), nc)
    for alpha, m in alpha_list:
        hc = _slb(hc, alpha, m, nc, 50.0, 300, 1e-15)
    h_up = _repair(np.repeat(hc, f), n)
    return h_up

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _el():
        return time.time() - t0

    # Phase 1: Multi-resolution warm-up (quick)
    for ratio in [4, 8]:
        if _el() > 20:
            break
        hc = _multi_res(best_h, ratio, [(1000, 15), (5000, 25)])
        c5u = _eval(hc)
        if c5u < best_c5:
            best_c5 = c5u
            best_h = hc.copy()

    # Phase 2: L-BFGS-B alpha sweep (main optimization - much faster than SLSQP)
    lgbm_configs = [
        (500, 20, 50.0, 500, 1e-14),
        (2000, 25, 50.0, 500, 1e-14),
        (5000, 30, 50.0, 400, 1e-14),
        (10000, 35, 50.0, 400, 1e-14),
        (30000, 40, 100.0, 400, 1e-14),
        (100000, 45, 100.0, 350, 1e-14),
        (300000, 50, 200.0, 350, 1e-14),
        (1e6, 55, 200.0, 300, 1e-14),
        (3e6, 60, 500.0, 300, 1e-14),
        (1e7, 65, 500.0, 250, 1e-14),
        (3e7, 70, 1000.0, 250, 1e-14),
        (1e8, 75, 1000.0, 200, 1e-14),
        (3e8, 80, 2000.0, 200, 1e-14),
        (1e9, 85, 5000.0, 150, 1e-14),
        (3e9, 90, 5000.0, 150, 1e-14),
    ]
    for alpha, m, pen, mi, ftol in lgbm_configs:
        if _el() > 450:
            break
        h_new = _slb(best_h, alpha, m, n, pen, mi, ftol)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 3: Projected gradient with momentum
    if _el() < budget_s - 300:
        pg_configs = [
            (5000, 50, 800, 0.005),
            (20000, 60, 600, 0.003),
            (100000, 70, 500, 0.002),
            (500000, 80, 400, 0.001),
            (2e6, 90, 300, 0.0005),
            (1e7, 100, 300, 0.0003),
        ]
        for alpha, ml, steps, s0 in pg_configs:
            if _el() > budget_s - 250:
                break
            h_new = _proj_grad(best_h.copy(), alpha, ml, steps, s0)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 4: Second L-BFGS-B pass at very high alpha
    if _el() < budget_s - 150:
        for alpha, m, pen in [(3e9, 90, 5000.0), (1e10, 95, 10000.0), (3e10, 100, 20000.0)]:
            if _el() > budget_s - 120:
                break
            h_new = _slb(best_h, alpha, m, n, pen, 200, 1e-14)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 5: Intensive local search
    if _el() < budget_s - 80:
        rem = budget_s - 80 - _el()
        ntries = int(rem / 0.0012)
        best_h, best_c5 = _local_search(best_h, ntries, rng, 4e-5)

    # Phase 6: Ultra-fine 2-point polish
    if _el() < budget_s - 2:
        rem = budget_s - 2 - _el()
        npol = int(rem / 0.0006)
        best_h, best_c5 = _two_point_polish(best_h, npol, rng)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
