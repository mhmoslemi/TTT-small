def _slsqp_pass(h, alpha, ml, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n * 0.5,)}]
    try:
        r = minimize(_smobj, h, jac=_smgrad, args=(alpha, ml),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return h


def _lbfgs_pass(h, alpha, ml, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n * 0.5,)}]
    try:
        r = minimize(_smobj, h, jac=_smgrad, args=(alpha, ml),
                     method='L-BFGS-B', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16,
                              'maxfun': maxiter * 4})
        return _repair(r.x, n)
    except Exception:
        return h


def _adam_pass(h, alpha, ml, nsteps, lr_init, t_end):
    n = len(h)
    p = h.copy()
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    lr = lr_init
    best = p.copy()
    best_c = _eval(p)
    b1, b2 = 0.9, 0.999
    for t in range(1, nsteps + 1):
        if time.time() > t_end:
            break
        g = _smgrad(p, alpha, ml)
        g -= g.mean()
        m_v = b1 * m_v + (1.0 - b1) * g
        v_v = b2 * v_v + (1.0 - b2) * (g * g)
        m_hat = m_v / (1.0 - b1 ** t)
        v_hat = v_v / (1.0 - b2 ** t)
        step = -lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        step -= step.mean()
        p_new = _repair(p + step, n)
        cn = _eval(p_new)
        cp = _eval(p)
        if cn <= cp + 1e-16:
            p = p_new
            if cn < best_c:
                best_c = cn
                best = p.copy()
            lr = min(lr * 1.03, lr_init * 5.0)
        else:
            lr *= 0.85
            if lr < 1e-12:
                p = best.copy()
                lr = lr_init * 0.4
    return best, best_c


def _active_grad_step(h, nsteps, t_end):
    n = len(h)
    p = h.copy()
    c_cur = _eval(p)
    best = p.copy()
    best_c = c_cur
    for _ in range(nsteps):
        if time.time() > t_end:
            break
        c = _corr_scaled(p)
        kmax = c.argmax()
        L = kmax - (n - 1)
        g = np.zeros(n)
        if L >= 0:
            g[:n - L] = (1.0 - p[L:])
            g[L:] -= p[:n - L]
        else:
            M = -L
            g[M:] = (1.0 - p[:n - M])
            g[:n - M] -= p[n - M:]
        g *= (2.0 / n)
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        for step in [1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]:
            hn = _repair(p - step * g, n)
            cn = _eval(hn)
            if cn < c_cur:
                p = hn
                c_cur = cn
                if cn < best_c:
                    best_c = cn
                    best = p.copy()
                break
    return best, best_c


def _coord_descent(h, n_iter, rng, t_end):
    n = len(h)
    p = h.copy()
    c_cur = _eval(p)
    best = p.copy()
    best_c = c_cur
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 7)) % n
        d = 3e-7 * rng.randn()
        ni, nj = p[i] + d, p[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            p[i], p[j] = ni, nj
            cn = _eval(p)
            if cn < c_cur:
                c_cur = cn
                if cn < best_c:
                    best_c = cn
                    best = p.copy()
            else:
                p[i], p[j] = ni - d, nj + d
    return best, best_c


def _multi_scale(h, n, t_end):
    coarse_n = n // 8
    if coarse_n < 20:
        return h
    hc = np.zeros(coarse_n)
    for i in range(coarse_n):
        lo = i * (n // coarse_n)
        hi = (i + 1) * (n // coarse_n)
        hc[i] = h[lo:hi].mean()
    hc = _repair(hc, coarse_n)
    for alpha in [10000, 100000, 1000000]:
        if time.time() > t_end:
            break
        ml = min(coarse_n - 1, 30)
        hn = _slsqp_pass(hc, alpha, ml, 800)
        hc = hn
    hi = np.interp(np.arange(n) * 0.5, np.arange(coarse_n) * 0.5, hc)
    hi = hi * 2.0
    hi = _repair(hi, n)
    return hi


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    best_h = _repair(h0, n)
    best_c = _eval(best_h)

    # Phase 1: SLSQP with increasing alpha (correct gradient)
    for alpha in [10000, 50000, 200000, 1000000, 5000000, 20000000]:
        if time.time() - t0 > 250:
            break
        ml = min(100, max(20, int(20 * np.log10(max(alpha, 1e4) / 1e4) + 25)))
        hn = _slsqp_pass(best_h, alpha, ml, 2000)
        cn = _eval(hn)
        if cn < best_c:
            best_c, best_h = cn, hn

    # Phase 2: L-BFGS-B refinement
    if time.time() - t0 < budget_s - 350:
        for alpha in [100000, 1000000, 10000000]:
            if time.time() - t0 > budget_s - 300:
                break
            ml = min(100, 25 + int(15 * np.log10(max(alpha, 1e5) / 1e5)))
            hn = _lbfgs_pass(best_h, alpha, ml, 1500)
            cn = _eval(hn)
            if cn < best_c:
                best_c, best_h = cn, hn

    # Phase 3: Adam with correct gradient
    if time.time() - t0 < budget_s - 300:
        for alpha, lr, ns in [(200000, 0.008, 4000), (2000000, 0.004, 4000),
                              (20000000, 0.002, 3000)]:
            if time.time() - t0 > budget_s - 200:
                break
            ml = min(100, 25 + int(12 * np.log10(max(alpha, 2e5) / 2e5)))
            hn, cn = _adam_pass(best_h, alpha, ml, ns, lr, budget_s - 150)
            if cn < best_c:
                best_c, best_h = cn, hn

    # Phase 4: High-alpha SLSQP (near-exact max)
    if time.time() - t0 < budget_s - 200:
        for alpha in [100000000, 500000000, 2000000000]:
            if time.time() - t0 > budget_s - 120:
                break
            ml = min(120, 50)
            hn = _slsqp_pass(best_h, alpha, ml, 3000)
            cn = _eval(hn)
            if cn < best_c:
                best_c, best_h = cn, hn

    # Phase 5: Active-set gradient (exact max gradient)
    if time.time() - t0 < budget_s - 100:
        nsteps = min(5000, int((budget_s - 100 - (time.time() - t0)) / 0.05))
        best_h, best_c = _active_grad_step(best_h, nsteps, budget_s - 80)

    # Phase 6: Multi-scale restart
    if time.time() - t0 < budget_s - 150:
        h_ms = _multi_scale(best_h, n, budget_s - 150)
        hn = _slsqp_pass(h_ms, 5000000, 50, 1000)
        cn = _eval(hn)
        if cn < best_c:
            best_c, best_h = cn, hn
        if time.time() - t0 < budget_s - 50:
            hn = _slsqp_pass(best_h, 50000000, 60, 2000)
            cn = _eval(hn)
            if cn < best_c:
                best_c, best_h = cn, hn

    # Phase 7: Fine coordinate descent
    if time.time() - t0 < budget_s - 3:
        n_iter = min(1000000, int((budget_s - 3 - (time.time() - t0)) / 0.0018))
        best_h, best_c = _coord_descent(best_h, n_iter, rng, budget_s - 2)

    best_h = _repair(best_h, n)
    best_c = _eval(best_h)
    return (best_h, best_c, n)
