    """Power-sum smooth max of top m_lags correlations."""
    c = _corr_vals(h)
    idx = np.argsort(c)[-m_lags:]
    top = c[idx]
    top = np.maximum(top, 1e-15)
    return (np.sum(top ** p)) ** (1.0 / p)

def _ps_grad(h, p, m_lags):
    """Gradient of power-sum smooth max."""
    n = len(h)
    c = _corr_vals(h)
    idx = np.argsort(c)[-m_lags:]
    top = c[idx]
    top_safe = np.maximum(top, 1e-15)
    # d/dh of (sum top^p)^(1/p) = (1/p) * (sum top^p)^(1/p - 1) * p * top^(p-1) * dtop/dh
    # = (sum top^p)^(1/p - 1) * top^(p-1) * dcorr/dh
    sum_top_p = np.sum(top_safe ** p)
    scale = sum_top_p ** (1.0/p - 1.0)
    g = np.zeros(n)
    for ii, k in enumerate(idx):

        L = k - (n - 1)

            g[:n-L] -= w * h[L:]
        else:
            M = -L
            g[:n-M] += w * (1.0 - h[M:])
            g[M:] -= w * h[:n-M]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-11:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-11:
        order = np.argsort(h) if diff > 0 else np.argsort(-h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0, 1)

def _lse_obj(x, alpha, ml):
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = np.clip(alpha*(c[idx]-m), -350, 350)
    s -= s.max()
    return m + np.log(np.exp(s).sum())/alpha

def _lse_jac(x, alpha, ml):
    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = np.clip(alpha*(c[idx]-m), -350, 350)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15: continue
        wk = w[ii]*(2.0/n)
        L = k-(n-1)
        if L >= 0:
            g[L:] += wk*(1.0-x[:n-L])
            g[:n-L] -= wk*x[L:]
        else:
            M = -L
            g[:n-M] += wk*(1.0-x[M:])
            g[M:] -= wk*x[:n-M]
    return g

def _sf_jac(x, target):
    return x.sum() - target

def _sf_jac_arr(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, ml, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)]*n
    cons = [{'type':'eq','fun':_sf_jac,'jac':_sf_jac_arr,'args':(n*0.5,)}]
    try:
        res = minimize(_lse_obj, h, jac=_lse_jac, args=(alpha,ml),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter':maxiter,'ftol':ftol})
        return _repair(res.x, n)
    except:
        return h

def _ps_grad(h, p, ml):
    n = len(h)
    c = _corr(h)
    idx = np.argsort(c)[-ml:]
    top = np.maximum(c[idx], 1e-15)
    stp = np.sum(top**p)
    scale = stp**(1.0/p - 1.0)
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        w = scale*(top[ii]**(p-1))*(2.0/n)
        L = k-(n-1)
        if L >= 0:
            g[L:] += w*(1.0-h[:n-L])
            g[:n-L] -= w*h[L:]
        else:
            M = -L
            g[:n-M] += w*(1.0-h[M:])
            g[M:] -= w*h[:n-M]
    return g

def _rmsprop(h, p, ml, steps, step0):
    n = len(h)
    v = np.zeros(n); sq = np.zeros(n)
    step = step0; cur = _eval(h); mom = 0.8
    for _ in range(steps):
        g = _ps_grad(h, p, ml)
        v = mom*v+(1-mom)*g
        sq = 0.95*sq+0.05*g*g
        d = v/(np.sqrt(sq)+1e-8)
        ok = False
        for _ in range(14):
            hn = _proj(h-step*d, n*0.5)
            cn = _eval(hn)
            if cn < cur:
                h, cur, ok = hn, cn, True
                step = min(step*1.12, step0*30)
                break
            step *= 0.38
        if not ok:
            step = min(step*1.35, step0*30)
    return _repair(h, n)

def _two_pt(h, niter, p, ml):
    n = len(h)
    cur = _eval(h); best = h.copy(); bestc = cur
    for _ in range(niter):
        g = _ps_grad(h, p, ml)
        gp = g - g.mean()
        i = int(np.argmax(gp)); j = int(np.argmin(gp))
        if i == j: continue
        found = False
        for frac in [0.4, 0.2, 0.6, 0.1, 0.8, 0.3, 0.5]:
            dv = frac*0.004
            ni, nj = h[i]+dv, h[j]-dv
            if 0<=ni<=1 and 0<=nj<=1:
                h[i], h[j] = ni, nj
                c = _eval(h)
                if c < cur:
                    cur = c
                    if c < bestc: best, bestc = h.copy(), c
                    found = True
                    break
                h[i], h[j] = ni-dv, nj+dv
        if not found:
            h[i] -= 0; h[j] += 0
    return best, bestc

def _fine(h, niter, rng, step):
    n = len(h); cur = _eval(h); best = h.copy(); bestc = cur
    for _ in range(niter):
        i = rng.randint(n)
        j = (i+1+rng.randint(0,5))%n
        d = step*rng.randn()
        ni, nj = h[i]+d, h[j]-d
        if 0<=ni<=1 and 0<=nj<=1:
            h[i], h[j] = ni, nj
            c = _eval(h)
            if c < cur:
                cur = c
                if c < bestc: best, bestc = h.copy(), c
            else:
                h[i], h[j] = ni-d, nj+d
    return best, bestc

def _down(h, f):
    n = len(h); m = n//f
    return _repair(h[:m*f].reshape(m,f).mean(1).copy(), m)

def _up(h, f):
    n = len(h); out = np.empty(n*f)
    for i in range(n):
        out[i*f:(i+1)*f] = h[i]
    return _repair(out, n*f)

def _basin_hop(h, rng, n_pert, alpha, ml, maxiter):
    n = len(h)
    h2 = h + n_pert*rng.randn(n)
    h2 = _proj(h2, n*0.5)
    h2 = _slsqp(h2, alpha, ml, 300, 1e-15)
    return h2
