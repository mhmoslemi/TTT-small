import numpy as np
import time

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-14:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(50):
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            np.clip(x, 0.0, 1.0, out=x)
            break
        x[m] += d / c
        np.clip(x, 0.0, 1.0, out=x)
    return x

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _corr_max_and_active(h, tol_frac=0.02, top_k=30):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(corr.max())
    # Active: within tol_frac of max OR top top_k
    tol = mx * tol_frac
    active_idx = np.nonzero(corr >= mx - tol)[0]
    # Also ensure we include top_k
    top_idx = np.argsort(corr)[-top_k:]
    active_idx = np.unique(np.concatenate([active_idx, top_idx]))
    lags = active_idx - (n - 1)
    return mx, lags, corr

def _grad_active(h, active_lags):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in active_lags:
        if k == 0:
            g[:] += (1.0 - 2.0 * h)
        else:
            g[:n-k] += (1.0 - h[k:])
            g[k:] += -h[:n-k]
    g *= dx
    # Normalize by count
    g /= len(active_lags)
    return g

def _active_grad_descent(h0, deadline, lr_init=0.1, lr_decay=0.999, tol_frac=0.02, top_k=20):
    n = len(h0)
    h = h0.copy()
    S = n * 0.5
    best_c = _c5(h)
    best_h = h.copy()
    lr = lr_init
    t0 = time.time()
    while time.time() < deadline:
        mx, lags, corr = _corr_max_and_active(h, tol_frac, top_k)
        g = _grad_active(h, lags)
        # Projected gradient step
        h_new = h - lr * g
        h_new = _proj(h_new, S)
        c_new = _c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if c_new <= mx * (2.0/n) * (1 + 1e-10):
            h = h_new
        else:
            lr *= lr_decay
            if lr < 1e-10:
                # Try smaller step
                h_new = h - lr * 10 * g
                h_new = _proj(h_new, S)
                c_new = _c5(h_new)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                h = h_new
        if time.time() - t0 > 5 and lr < 1e-8:
            # Restart with slightly perturbed h
            rng = np.random.RandomState(int(time.time()*1000) % 10000)
            h = h + rng.randn(n) * 0.01
            h = _repair(h, n)
            lr = lr_init * 0.5
    return best_h, best_c

def _pair_local_search(h0, deadline, n_steps_target=50000):

    h = h0.copy()
    S = n * 0.5
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(corr.max())
    best_c = mx * (2.0/n)
    best_h = h.copy()
    t0 = time.time()
    total_t = max(0.1, deadline - t0)
    rng = np.random.RandomState(12345)
    it = 0
    while time.time() < deadline:

        T = 0.3 * (0.01) ** frac
        # Find active lag

        # Targeted move at active lag
        if abs(k_act) < n and abs(k_act) > 0 and rng.random() < 0.8:

            if k_act > 0:

                j = i + k
                # Move: push h[i] toward h[j]

                d = diff * (0.1 + 0.4 * rng.random())
            else:

                j = rng.randint(n - k)
                i = j + k
                diff = h[j] - h[i]
                d = diff * (0.1 + 0.4 * rng.random())
            # clamp

            if h[i] + d > 1: d = 1 - h[i]

import numpy as np

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-14:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    for _ in range(80):
        d = S - x.sum()
        if abs(d) < 1e-14:
            break
        m = (x < 1.0 - 1e-12) if d > 0 else (x > 1e-12)
        c = m.sum()
        if c == 0:
            np.clip(x, 0.0, 1.0, out=x)
            break
        x[m] += d / c
    np.clip(x, 0.0, 1.0, out=x)
    return x

def _c5(h):
    n = len(h)

def _get_active(h, frac=0.03, topk=15):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(corr.max())
    tol = mx * frac
    mask = corr >= mx - tol
    top_idx = np.argsort(corr)[-topk:]
    idx = np.unique(np.concatenate([np.nonzero(mask)[0], top_idx]))
    lags = (idx - (n - 1)).astype(int)
    return mx, lags, corr

def _grad_topk(h, lags, topk=15):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    kl = list(lags)
    for k in kl:
        if k == 0:
            g[:] += (1.0 - 2.0 * h)
        else:
            g[:n-k] += (1.0 - h[k:])
            g[k:] += -h[:n-k]
    g *= dx / max(len(kl), 1)
    return g
