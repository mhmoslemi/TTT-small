def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break

def _eval(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / n)

    c = _corr_full(x)
    m = c.max()
    idx = np.argsort(c)[::-1][:top_k]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _repair(h, n):

    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:

    diff = target - h.sum()
    if abs(diff) > 1e-10:

            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:

    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(x, alpha, top_k):
    n = len(x)
    c = _corr_raw(x) * (2.0 / n)
    m = c.max()
    idx = np.argsort(c)[::-1][:top_k]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sls_smooth(h, alpha, top_k, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]

                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

    n = len(h)
    h = h.astype(float).copy()
    best_c5 = _eval(h)
    best_h = h.copy()
    
    for it in range(max_iter):
        c = _corr_raw(h)
        c5_raw = c.max()
        c5 = c5_raw * dx

        idx = np.argsort(c)[::-1][:n_active]

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]

            # gradient w.r.t h:
            grad = np.zeros(n)
            if shift >= 0:

                grad[:m] += (1.0 - h[shift:shift+m])  # (1-h[i+shift]) for i=0..m-1
                grad[shift:] -= h[:n-shift]            # -h[i] for i+shift = j, i=j-shift

            cons.append(grad @ h_var <= t_var + rhs)

        try:

            try:
                prob.solve(solver=cp.GLPK)
            except Exception:
                break

            break

def _grad_shift(h, s, n):
    """
    Compute gradient of r[shift=s] w.r.t. h.
    r at shift s: for s>=0, sum_{n_idx=0}^{n-1-s} h[n_idx+s]*(1-h[n_idx])
                  for s<0,  sum_{n_idx=-s}^{n-1} h[n_idx+s]*(1-h[n_idx])
    where n_idx is the index in the "1-h" array.
    """
    g = np.zeros(n)
    if s >= 0:
        # Terms: h[i+s] * (1-h[i]) for i = 0..n-1-s
        # d/dh[j] of h[i+s]*(1-h[i]):
        #   If j = i+s (i.e., i = j-s): contributes (1-h[j-s]) for i in [0, n-1-s], so j in [s, n-1]
        #   If j = i (i.e., i = j): contributes -h[j+s] for i in [0, n-1-s], so j in [0, n-1-s]
        g[s:] += (1.0 - h[:n-s])   # (1-h[i]) for i=0..n-1-s, added to positions s..n-1

        g[m:] -= h[:n-m]           # -h[i-m] for i=m..n-1, subtracted from positions m..n-1

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _smooth_obj(x, alpha, top_k):
    n = len(x)
    c = _corr_raw(x) * (2.0 / n)
    m = c.max()
    idx = np.argsort(c)[::-1][:top_k]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sum_fn(x, half_n):
    return x.sum() - half_n

def _sum_jac(x, half_n):
    return np.ones(len(x))

def _sls_run(h, alpha, top_k, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h.copy(), args=(alpha, top_k),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

def _grad_shift(h, s, n):
    g = np.zeros(n)
    if s >= 0:
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    else:
        m = -s
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    return g

def _sl_optimize(h, n_active, max_iter, t_timeout):
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    best_c5 = _eval(h)
    best_h = h.copy()
    stagnation = 0
    for it in range(max_iter):
        if time.time() > t_timeout:
            break
        c = _corr_raw(h)
        c5_raw = c.max()
        idx = np.argsort(c)[::-1][:n_active]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
        for k in idx:
            s = k - (n - 1)
            g = _grad_shift(h, s, n)
            rhs = c[k] - g @ h
            cons.append(g @ h_var <= t_var + rhs)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        solved = False
        for solver in [cp.GLOP, cp.CBC, cp.GLPK, cp.CVXOPT]:
            try:
                prob.solve(solver=solver)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
        if not solved or h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).flatten(), 0.0, 1.0)
        h_new = _repair(h_new, n)
        c5_new = _eval(h_new)
        if abs(h_new - h).max() < 1e-12:
            stagnation += 1
        else:
            stagnation = 0
        if stagnation > 5:
            break
        c_new = _corr_raw(h_new)
        if c_new.max() > c5_raw * 1.01 + 0.5:
            n_active = min(n_active + 10, 2 * n - 1)
        h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5

def _coord_search(h, n_iter, rng, t_timeout):
    n = len(h)
    dx = 2.0 / n
    c5 = _eval(h)
    best_h = h.copy()
    best_c5 = c5
    h = h.copy()
    for _ in range(n_iter):

                        best_h = h.copy()
                else:

    return best_h, best_c5

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _proj_grad_step(h, alpha, top_k, steps, step0):
    n = len(h)
    h = h.copy()
    cur_c = _eval(h)
    v = np.zeros(n)
    sq = np.zeros(n)
    step = step0
    mom = 0.9
    for _ in range(steps):
        c = _corr_raw(h) * (2.0 / n)
        m = c.max()
        idx = np.argsort(c)[::-1][:top_k]
        s = np.clip(alpha * (c[idx] - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii, k in enumerate(idx):
            if w[ii] < 1e-14:
                continue
            sh = k - (n - 1)
            gg = _grad_shift(h, sh, n) * w[ii] * (2.0 / n)
            g += gg
        g = g - g.mean()
        v = mom * v + (1.0 - mom) * g
        sq = 0.95 * sq + 0.05 * g * g
        direction = v / (np.sqrt(sq) + 1e-8)
        taken = False
        for _t in range(12):
            h_new = _proj(h - step * direction, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:
                h = h_new
                cur_c = c_new
                step = min(step * 1.1, step0 * 40.0)
                taken = True
                break
            step *= 0.4

        n_iter = int(remaining * 1000)
        best_h, best_c5 = _coord_search(best_h, n_iter, rng, budget_s - 3)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
