                a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_dx(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _smooth_obj(x, alpha, m_lags):
    c = _corr_dx(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = _corr_dx(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _con_fun(x, t):
    return x.sum() - t

def _con_jac(x, t):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _lbfgs(h, alpha, m_lags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    res = minimize(_smooth_obj, h, jac=_smooth_grad,
                   args=(alpha, m_lags), method='L-BFGS-B',
                   bounds=bnds,
                   options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-16, 'maxls': 40})
    return _repair(res.x, n)

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _proj_grad(h, alpha, m_lags, steps, step0, rng):
    n = len(h)
    v = np.zeros(n)
    mu = 0.88
    step = step0
    cur_c = _eval(h)
    for i in range(steps):
        g = _smooth_grad(h, alpha, m_lags)
        v = mu * v + (1.0 - mu) * g
        ng = np.linalg.norm(v)
        if ng < 1e-16:
            break
        v /= ng
        taken = False
        for _t in range(12):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-17:
                h = h_new; cur_c = c_new
                step = min(step * 1.08, step0 * 30.0)
                taken = True
                break
            step *= 0.38
        if not taken:
            step = min(max(step * 1.4, step0 * 0.5), step0 * 30.0)
            if i % 80 == 40:
                idx = rng.randint(0, n, size=8)
                h[idx] += 0.003 * rng.randn(8)
                h = _repair(h, n)
                cur_c = _eval(h)
    return _repair(h, n), cur_c

def _anneal(h, n_iter, rng, temp0, temp_end):
    n = len(h)
    cur_c = _eval(h)
    best_h = h.copy()
    best_c = cur_c
    n_tries = int(np.ceil(n_iter / 300.0))
    for i in range(n_tries):
        actual_iter = min(300, n_iter - i * 300)
        for _ in range(actual_iter):
            temp = temp0 * (temp_end / temp0) ** (i * 300.0 / max(1, n_iter))
            k = rng.randint(3, 12)
            idx = rng.randint(0, n, size=k)
            d = 0.008 * rng.randn(k)
            d -= d.mean()
            h_new = h.copy()
            h_new[idx] += d
            if h_new.min() < -0.05 or h_new.max() > 1.05:
                continue
            h_new = _repair(h_new, n)
            c_new = _eval(h_new)
            if c_new < cur_c or rng.rand() < np.exp(-(c_new - cur_c) / max(temp, 1e-12)):
                h = h_new; cur_c = c_new
                if c_new < best_c:
                    best_c = c_new; best_h = h.copy()
    return best_h, best_c

def _polish(h, n_iter, rng):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        ks = rng.randint(2, 5)
        idx = rng.randint(0, n, size=ks)
        d = (1e-9 + 2e-8 * rng.rand()) * rng.randn(ks)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < -0.01 or h_new.max() > 1.01:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-17:
            h = h_new; c5c = c5n
            if c5n < best_c:
                best_c = c5n; best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP moderate alpha sweep
    for alpha in [30000, 100000, 500000, 3000000]:
        if _el() > 150:
            break
        lift = min(60, max(12, int(7 * np.log10(alpha) - 16)))
        h_new = _slsqp(best_h.copy(), alpha, lift, 600, 1e-16)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n; best_h = h_new

    # Phase 2: L-BFGS-B (different optimizer, handles bounds natively)
    if _el() < budget_s - 350:
        for alpha in [200000, 2000000, 20000000, 200000000]:
            if _el() > budget_s - 300:
                break
            lift = min(50, max(10, int(5 * np.log10(alpha) - 15)))
            h_new = _lbfgs(best_h.copy(), alpha, lift, 400)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n; best_h = h_new

    # Phase 3: Multi-start with structured perturbations
    if _el() < budget_s - 300:
        for trial in range(8):
            if _el() > budget_s - 250:
                break
            scale = 0.015 * (1 + trial * 0.3)
            # Mix random and Fourier-like perturbations
            freq = rng.randint(1, n // 10 + 1)
            phase = rng.rand() * 2 * np.pi
            fourier_pert = np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            h_pert = best_h + scale * (0.5 * rng.randn(n) + 0.5 * fourier_pert)
            h_pert = _repair(h_pert, n)
            h_new = _slsqp(h_pert, 500000, 25, 250, 1e-15)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n; best_h = h_new

    # Phase 4: Projected gradient with momentum
    if _el() < budget_s - 200:
        h_pg, c_pg = _proj_grad(best_h.copy(), 3000000, 50, 600, 0.003, rng)
        if c_pg < best_c5:
            best_c5 = c_pg; best_h = h_pg
        h_pg, c_pg = _proj_grad(best_h.copy(), 50000000, 80, 500, 0.0015, rng)
        if c_pg < best_c5:
            best_c5 = c_pg; best_h = h_pg

    # Phase 5: Simulated annealing (global escape)
    if _el() < budget_s - 100:
        rem = int((budget_s - 100 - _el()) * 120)
        rem = max(rem, 5000)
        best_h, best_c5 = _anneal(best_h, rem, rng, 0.004, 0.00005)

    # Phase 6: Re-optimize after potential escape
    if _el() < budget_s - 40:
        h_new = _slsqp(best_h.copy(), 5000000, 35, 400, 1e-16)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n; best_h = h_new
        h_new = _lbfgs(best_h.copy(), 10000000, 40, 300)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n; best_h = h_new

    # Phase 7: Coarse-to-fine polish
    if _el() < budget_s - 10:
        rem = int((budget_s - 10 - _el()) * 500)
        rem = max(rem, 2000)
        best_h, best_c5 = _polish(best_h, rem, rng)

    # Phase 8: Final SLSQP squeeze
    if _el() < budget_s - 3:
        h_new = _slsqp(best_h.copy(), 100000000, 30, 200, 1e-17)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n; best_h = h_new

    # Final ultra-fine polish
    if _el() < budget_s - 1:
        rem = int((budget_s - 1 - _el()) * 900)
        rem = max(rem, 1000)
        best_h, best_c5 = _polish(best_h, rem, rng)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
