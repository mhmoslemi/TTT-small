def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

        return np.full(n, 0.5)

        h *= target / s

        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

import numpy as np
from scipy.optimize import minimize
import time


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h


def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)


def _cd_pass(h, rng, n_bis=50):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = 2 * n - 1
    nm1 = n - 1
    for j in rng.permutation(n):
        j = int(j)
        beta = np.zeros(cmax)
        klo1 = max(0, nm1 - j)
        khi1 = min(cmax - 1, 2 * nm1 - j)
        ks1 = np.arange(klo1, khi1 + 1)
        beta[ks1] += (1.0 - h[j + ks1 - nm1])
        klo2 = j
        khi2 = min(cmax - 1, j + nm1)
        ks2 = np.arange(klo2, khi2 + 1)
        beta[ks2] -= h[j + nm1 - ks2]
        g = c - beta * h[j]
        lo, hi = 0.0, 1.0
        for _ in range(n_bis):
            t1 = lo + (hi - lo) / 3.0
            t2 = hi - (hi - lo) / 3.0
            if np.max(g + beta * t1) <= np.max(g + beta * t2):
                hi = t2
            else:
                lo = t1
        tn = 0.5 * (lo + hi)
        if abs(tn - h[j]) > 1e-15:
            c += beta * (tn - h[j])
            h[j] = tn
    return h


def _sm_fun(x, alpha, aidx):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    ca = c[aidx]
    m = ca.max()
    s = np.clip(alpha * (ca - m), -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha


def _sm_jac(x, alpha, aidx):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    ca = c[aidx]
    m = ca.max()
    s = np.clip(alpha * (ca - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(len(w)):
        wk = w[ii]
        if wk < 1e-15:
            continue
        wk *= dx
        k = int(aidx[ii])
        L = k - nm1
        if L >= 0:
            jj = np.arange(0, n - L)
            g[jj] += wk * (1.0 - x[jj + L])
            jj2 = np.arange(L, n)
            g[jj2] -= wk * x[jj2 - L]
        else:
            M = -L
            jj = np.arange(M, n)
            g[jj] += wk * (1.0 - x[jj - M])
            jj2 = np.arange(0, n - M)
            g[jj2] -= wk * x[jj2 + M]
    return g


def _sum_fun(x, target):
    return x.sum() - target


def _sum_jac(x, target):
    return np.ones(len(x))


def _slsqp(h, alpha, aidx, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    args = (alpha, aidx)
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_sm_fun, h, jac=_sm_jac, args=args,
                       bounds=bnds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return h


def _local_search(h, n, rng, n_iter):
    best_h = h.copy()
    best_c5 = _eval(best_h)
    c5c = best_c5
    for _ in range(n_iter):
        sc = 5.0e-6 * (10.0 ** (rng.randint(0, 4) - 1.5))
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 30)) % n
        d = sc * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)

                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5
