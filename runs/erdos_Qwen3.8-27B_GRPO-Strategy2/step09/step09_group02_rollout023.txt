def _hard_subgrad(h, n_active=10):

def _slsqp_smooth(h, alpha, n_active, maxiter, ftol):

        m = c.max()

        s -= s.max()

            else:
                M = -L
                g[M:] += wk * (1.0 - x[:n-M])
                g[:n-M] -= wk * x[M:]
        return g

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    for _ in range(200):
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _eval(h):
    return float(np.max(_corr(h)))

def _grad_lag(h, k):
    """Gradient of correlation at specific lag k (index in full correlate output)."""
    n = len(h)
    g = np.zeros(n)
    L = k - (n - 1)
    if L >= 0:
        g[:n - L] += (1.0 - h[L:])
        g[L:] -= h[:n - L]
    else:
        M = -L
        g[M:] += (1.0 - h[:n - M])
        g[:n - M] -= h[M:]
    return g

def _subgrad_top(h, m):
    """Subgradient using top m lags equally weighted."""
    n = len(h)
    c = _corr(h)
    idx = np.argsort(c)[-m:]
    g = np.zeros(n)
    for k in idx:
        g += _grad_lag(h, k)
    g /= m
    g -= g.mean()
    return g

def _proj(h, target):
    lo, hi = -10.0, 10.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp(h, alpha, m, maxiter, ftol):
    n = len(h)
    dx = 2.0 / n
    
    def obj(x):
        c = _corr(x)
        mx = c.max()
        idx = np.argsort(c)[-m:]
        s = np.clip(alpha * (c[idx] - mx), -400, 400)
        s -= s.max()
        return mx + np.log(np.exp(s).sum()) / alpha
    
    def jac(x):
        c = _corr(x)
        mx = c.max()
        idx = np.argsort(c)[-m:]
        s = np.clip(alpha * (c[idx] - mx), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii, k in enumerate(idx):
            if w[ii] < 1e-15:
                continue
            wk = w[ii] * dx
            L = k - (n - 1)
            if L >= 0:
                g[:n - L] += wk * (1.0 - x[L:])
                g[L:] -= wk * x[:n - L]
            else:
                M = -L
                g[M:] += wk * (1.0 - x[:n - M])
                g[:n - M] -= wk * x[M:]
        return g
    
    bnds = [(0.0, 1.0)] * n
    tgt = n * 0.5
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt,
             'jac': lambda x: np.ones(len(x))}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _grad_desc(h, n_steps, step0, m, t0, tmax):
    n = len(h)
    v = np.zeros(n)
    step = step0
    cur = _eval(h)
    for _ in range(n_steps):
        if time.time() > tmax:
            break
        g = _subgrad_top(h, m)
        ng = np.linalg.norm(g)
        if ng < 1e-16:
            break
        v = 0.85 * v + 0.15 * g
        vn = np.linalg.norm(v)
        if vn < 1e-16:
            break
        v /= vn
        for f in [1.0, 0.5, 0.25, 0.125, 0.06]:
            sn = step * f
            h2 = _proj(h - sn * v, n * 0.5)
            c2 = _eval(h2)
            if c2 < cur - 1e-16:
                h, cur = h2, c2
                step = min(step * 1.05, step0 * 25)
                break
        else:
            step = min(max(step * 1.4, step0), step0 * 25)
    return h, cur

def _guided_polish(h, n_iter, rng, t0, tmax, base_range=60):
    n = len(h)
    c5c = _eval(h)
    best_h, best_c = h.copy(), c5c
    for _ in range(n_iter):
        if time.time() > tmax:
            break
        c = _corr(h)
        k_star = np.argmax(c)
        g = _grad_lag(h, k_star)
        g -= g.mean()
        # Find i (most negative g) and j (most positive g)
        i = int(np.argmin(g))
        j = int(np.argmax(g))
        if i == j:
            j = (i + 1) % n
        d = (1e-8 + 3e-7 * rng.rand()) * rng.randn()
        # Ensure we're moving in the right direction
        if g[i] * d < 0:  # we want to decrease h where g>0, increase where g<0
            d = -d
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c2 = _eval(h)
            if c2 < c5c - 1e-17:
                c5c = c2
                if c2 < best_c:
                    best_c, best_h = c2, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    for _ in range(200):
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _eval(h):
    return float(np.max(_corr(h)))

def _grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    L = k - (n - 1)
    if L >= 0:
        g[L:] += (1.0 - h[:n - L])
        g[:n - L] -= h[L:]
    else:
        M = -L
        g[:n - M] += (1.0 - h[M:])
        g[M:] -= h[:n - M]
    return g

def _subgrad(h, m):
    c = _corr(h)
    idx = np.argsort(c)[-m:]
    g = np.zeros(len(h))
    for k in idx:
        g += _grad_lag(h, k)
    g /= m
    g -= g.mean()
    return g, c

def _proj(h, target):
    lo, hi = -10.0, 10.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp(h, alpha, m, maxiter, ftol):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        c = _corr(x)
        mx = c.max()
        idx = np.argsort(c)[-m:]
        s = np.clip(alpha * (c[idx] - mx), -400, 400)
        s -= s.max()
        return mx + np.log(np.exp(s).sum()) / alpha
    def jcn(x):
        c = _corr(x)
        mx = c.max()
        idx = np.argsort(c)[-m:]
        s = np.clip(alpha * (c[idx] - mx), -400, 400)
        s -= s.max()
        w = np.exp(s); w /= w.sum()
        g = np.zeros(n)
        for ii, k in enumerate(idx):
            if w[ii] < 1e-15: continue
            wk = w[ii] * dx
            L = k - (n - 1)
            if L >= 0:
                g[L:] += wk * (1.0 - x[:n-L])
                g[:n-L] -= wk * x[L:]
            else:
                M = -L
                g[:n-M] += wk * (1.0 - x[M:])
                g[M:] -= wk * x[:n-M]
        return g
    bnds = [(0.0, 1.0)] * n
    tgt = n * 0.5
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt,
             'jac': lambda x: np.ones(len(x))}]
    try:
        res = minimize(obj, h, jac=jcn, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except:
        return h

def _gd(h, nsteps, s0, m, t0, tmax):
    n = len(h)
    v = np.zeros(n); step = s0; cur = _eval(h)
    for _ in range(nsteps):
        if time.time() > tmax: break
        g, _ = _subgrad(h, m)
        ng = np.linalg.norm(g)
        if ng < 1e-16: break
        v = 0.85*v + 0.15*g
        vn = np.linalg.norm(v)
        if vn < 1e-16: break
        v /= vn
        taken = False
        for f in [1.0, 0.5, 0.25, 0.125, 0.06]:
            sn = step * f
            h2 = _proj(h - sn*v, n*0.5)
            c2 = _eval(h2)
            if c2 < cur - 1e-16:
                h, cur = h2, c2
                step = min(step*1.05, s0*30)
                taken = True; break
        if not taken:
            step = min(max(step*1.4, s0), s0*30)
    return h, cur

def _gpolish(h, rng, t0, tmax, kp=25):
    """Gradient-guided polish: pick pairs from top-kp gradient positions."""
    n = len(h)
    c5c = _eval(h)
    c_cache = _corr(h)
    k_star = int(np.argmax(c_cache))
    g = _grad_lag(h, k_star); g -= g.mean()
    best_h, best_c = h.copy(), c5c
    while time.time() < tmax:
        neg_i = np.argsort(g)[:kp]
        pos_i = np.argsort(g)[::-1][:kp]
        i = int(neg_i[rng.randint(kp)])
        j = int(pos_i[rng.randint(kp)])
        if i == j: j = (i+1) % n
        d = (1e-8 + 4e-7*rng.rand()) * rng.randn()
        if g[i]*d < 0: d = -d
        ni, nj = h[i]+d, h[j]-d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c2 = _eval(h)
            if c2 < c5c - 1e-17:
                c5c = c2
                c_cache = _corr(h)
                k_star = int(np.argmax(c_cache))
                g = _grad_lag(h, k_star); g -= g.mean()
                if c2 < best_c: best_c, best_h = c2, h.copy()
            else:
                h[i], h[j] = ni-d, nj+d
    return best_h, best_c

def _rpolish(h, rng, t0, tmax, mxr=100):
    """Random polish over wide range."""
    n = len(h)
    c5c = _eval(h)
    best_h, best_c = h.copy(), c5c
    while time.time() < tmax:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, mxr)) % n
        if i == j: j = (i+1) % n
        d = (1e-8 + 5e-7*rng.rand()) * rng.randn()
        ni, nj = h[i]+d, h[j]-d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c2 = _eval(h)
            if c2 < c5c - 1e-17:
                c5c = c2
                if c2 < best_c: best_c, best_h = c2, h.copy()
            else:
                h[i], h[j] = ni-d, nj+d
    return best_h, best_c
