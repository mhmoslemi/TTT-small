def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _smoothed_obj(x, alpha, m, n, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    top = np.argsort(c)[-m:]
    s = alpha * (c[top] - c.max())
    s = np.clip(s - s.max(), -500, 500)
    return c.max() + np.log(np.exp(s).sum()) / alpha + lam * (x.sum() - n * 0.5) ** 2

def _proj_onto_simplex_sum(h, target):
    # Project h onto {x: 0<=x<=1, sum(x)=target}
    lo, hi = -5.0, 5.0
    for _ in range(60):

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _subgrad_step(h, step, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    k_star = int(np.argmax(c))
    # gradient of c[k_star] w.r.t. h
    L = k_star - (n - 1)
    g = np.zeros(n)
    if L >= 0:

        for i in range(n - L):
            g[i + L] += (1.0 - h[i])
            g[i] -= h[i + L]
    else:
        M = -L
        for i in range(n - M):
            g[i] += (1.0 - h[i + M])
            g[i + M] -= h[i]
    g *= dx
    return h - step * g

        return _smoothed_obj(x, alpha, m, n, lam)
    res = minimize(obj, h0, method='L-BFGS-B', bounds=[(0,1)]*n,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
    return _repair(res.x, n)

def _run_slsqp(h0, alpha, m, n, maxiter):

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx

        s = alpha * (c[top] - c.max())

        return np.ones(n)

            cur = cn
            h = hn
            if cn < bc:
                bc = cn
                bh = h.copy()
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.asarray(initial_h_values, dtype=float), n)
    bh = h.copy()
    bc5 = _eval(bh)

    # Phase 1: SLSQP geometric alpha ramp
    cfg1 = [
        (2000, 15, 500), (10000, 25, 550), (50000, 40, 550),
        (250000, 60, 500), (1000000, 80, 450), (5000000, 100, 400),
        (20000000, 120, 350), (100000000, 150, 300),
    ]
    stall = 0
    for i in range(len(cfg1)):
        if time.time() - t0 > 350:
            break
        a, m, mi = cfg1[i]
        hn = _slsqp(bh, a, m, n, mi)
        cn = _eval(hn)
        if cn < bc5 - 1e-12:
            bc5 = cn
            bh = hn.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 2:
                bh = _repair(bh + 0.002 * rng.randn(n) - 0.002 * np.mean(bh + 0.002 * rng.randn(n)), n)
                stall = 0
                bc5 = _eval(bh)

    # Phase 2: Momentum projected gradient
    if time.time() - t0 < budget_s - 420:
        dl = budget_s - 420
        for a, m, ns, s0 in [
            (1e8, 100, 1000, 4e-4),
            (5e8, 150, 800, 2e-4),
            (2e9, 200, 600, 8e-5),
            (1e10, 250, 400, 3e-5),
        ]:
            if time.time() > dl:
                break
            hn, cn = _momentum_grad(bh, a, m, n, ns, s0, t0, dl)
            if cn < bc5:
                bc5 = cn
                bh = hn.copy()

    # Phase 3: Final SLSQP at very high alpha
    if time.time() - t0 < budget_s - 280:
        for a, m, mi in [(5e8, 200, 400), (2e9, 250, 350), (1e10, 300, 300), (5e10, 350, 250)]:
            if time.time() - t0 > budget_s - 270:
                break
            hn = _slsqp(bh, a, m, n, mi)
            cn = _eval(hn)
            if cn < bc5:
                bc5 = cn
                bh = hn.copy()

    # Phase 4: Block refinement
    if time.time() - t0 < budget_s - 170:
        nt = int((budget_s - 170 - (time.time() - t0)) / 0.004)
        bh, bc5 = _block_refine(bh, nt, rng, n, t0, budget_s - 170)

    # Phase 5: Fine coordinate descent
    if time.time() - t0 < budget_s - 2:
        nt = int((budget_s - 2 - (time.time() - t0)) / 0.001)
        bh, bc5 = _fine_coord(bh, nt, rng, n, t0, budget_s - 2)

    bh = _repair(bh, n)
    bc5 = _eval(bh)
    return (bh, bc5, n)
