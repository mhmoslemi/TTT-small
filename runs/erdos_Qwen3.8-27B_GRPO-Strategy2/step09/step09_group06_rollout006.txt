import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        if abs(s - target) < 1e-14:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _proj(h, target):
    lo, hi = -2.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _soft_obj(x, alpha, m, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s - s.max(), -500, 500)
    return c.max() + np.log(np.exp(s).sum()) / alpha

def _soft_grad(x, alpha, m, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s - s.max(), -500, 500)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fun(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, mlags, maxiter, t_end):
    n = len(h)
    target = n * 0.5
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (target,)}]
    try:
        res = minimize(_soft_obj, h, jac=_soft_grad, args=(alpha, mlags, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if time.time() > t_end:
            return h
        return _repair(res.x, n)
    except Exception:
        return h

def _fista(h, alpha, m, t_end):
    n = len(h)
    target = n * 0.5
    x = h.copy()
    y = h.copy()
    t = 1.0
    step = 0.01
    cur_c = _eval(x)
    best_x = x.copy()
    best_c = cur_c
    no_improve = 0
    while time.time() < t_end:
        grad = _soft_grad(y, alpha, m, n)
        taken = False
        for _bt in range(15):
            y_new = _proj(y - step * grad, target)
            c_new = _eval(y_new)
            if c_new < cur_c + 1e-14:
                x = y_new
                t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
                y = x + ((t - 1.0) / t_new) * (x - y)
                y = _proj(y, target)
                t = t_new
                cur_c = c_new
                step = min(step * 1.03, 0.2)
                taken = True
                if c_new < best_c:
                    best_c = c_new
                    best_x = x.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                break
            step *= 0.5
        if not taken:
            step = min(step * 1.5, 0.2)
            no_improve += 1
        if no_improve > 500:
            break
    return _repair(best_x, n), best_c

def _grad_local_search(h, n_iter, rng, base_step, alpha, m, t_end):
    n = len(h)
    target = n * 0.5
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    for it in range(n_iter):
        if time.time() > t_end:
            break
        frac = it / max(n_iter, 1)
        step = base_step * (1.0 + 1.5 * (1.0 - frac))
        # Gradient-guided: mix negative gradient with random
        g = _soft_grad(h, alpha, m, n)
        g_dir = -g / (np.linalg.norm(g) + 1e-15)
        k = rng.choice([2, 3, 4])
        idx = rng.randint(0, n, size=k)
        d = step * (0.7 * g_dir[idx] + 0.3 * rng.randn(k))
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5cur - 1e-15:
            c5cur = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    return _repair(np.repeat(h, f), len(h) * f)

def run(seed=42, budget_s=1100, **kwargs):

    h0 = np.array(initial_h_values, dtype=float).copy()

    if time.time() < t_end - 2:

        nc2 = len(h2)

        while time.time() < t_end - 2:

            j = (i + 1 + rng.randint(0, 9)) % nc2
            d = 1e-8 * rng.randn()

                    h2[i], h2[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
