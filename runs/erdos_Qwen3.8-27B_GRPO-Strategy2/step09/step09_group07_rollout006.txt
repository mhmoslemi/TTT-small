    rng = np.random.RandomState(seed)
    h_init = np.array(initial_h_values, dtype=float).copy()
    n_init = len(h_init)

    best_h = _repair(h_init.copy(), n_init)
    best_c5 = _eval(best_h)
    best_n = n_init

    # Phase 1: Multi-resolution exploration (12% budget)
    phase1_end = t0 + budget_s * 0.12
    n_list = [280, 360, 560, 720, 1000]
    if n_init not in n_list:
        n_list.append(n_init)
    n_list = sorted(set(n_list))
    cands = []
    for n in n_list:
        if time.time() > phase1_end:
            break
        if n == n_init:
            h = _repair(h_init.copy(), n)
        else:
            h = _repair(_interp_to(h_init, n), n)
        h = _cd_pass(h, rng)
        h = _repair(h, n)
        c5 = _eval(h)
        cands.append((c5, h, n))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
    cands.sort(key=lambda x: x[0])

    # Phase 2: Intensive CD with restarts on top-3 (38% budget)
    phase2_end = t0 + budget_s * 0.50
    for ci in range(min(3, len(cands))):
        if time.time() > phase2_end:
            break
        _, h_cand, n = cands[ci]
        if cands[ci][0] > best_c5 + 0.002:
            continue
        h = h_cand.copy()
        c5 = _eval(h)
        stall = 0
        for it in range(200):
            if time.time() > phase2_end:
                break
            h = _cd_pass(h, rng)
            h = _repair(h, n)
            c5n = _eval(h)
            if c5n < best_c5 - 1e-7:
                best_c5 = c5n
                best_h = h.copy()
                best_n = n
                stall = 0
            elif c5n < c5:
                c5 = c5n
                stall = 0
            else:
                stall += 1
                if stall >= 8:
                    h = _perturb(h, n, rng, 0.03)
                    c5 = _eval(h)
                    stall = 0
        h = _cd_pass(h, rng)
        h = _repair(h, n)
        c5n = _eval(h)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h.copy()
            best_n = n

    # Phase 3: SLSQP with increasing smoothness (22% budget)
    phase3_end = t0 + budget_s * 0.72
    alphas = [2000, 10000, 50000, 200000, 1000000]
    for alpha in alphas:
        if time.time() > phase3_end:
            break
        h_new = _slsqp(best_h.copy(), alpha, 40, 500)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 4: Alternating CD + SLSQP (14% budget)
    phase4_end = t0 + budget_s * 0.86
    h_work = best_h.copy()
    n = best_n
    for it in range(30):
        if time.time() > phase4_end:
            break
        h_work = _cd_pass(h_work, rng)
        h_work = _repair(h_work, n)
        c5 = _eval(h_work)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_work.copy()
        if it % 8 == 7 and time.time() < phase4_end - 10:
            h_new = _slsqp(best_h.copy(), 2000000, 50, 300)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()
                h_work = best_h.copy()

    # Phase 5: Final CD + SLSQP (6% budget)
    phase5_end = t0 + budget_s * 0.92
    while time.time() < phase5_end:
        h_work = _cd_pass(best_h, rng)
        h_work = _repair(h_work, n)
        c5 = _eval(h_work)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_work.copy()
    if time.time() < t0 + budget_s * 0.96:
        h_new = _slsqp(best_h.copy(), 5000000, 60, 300)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 6: Final local search (4% budget)
    phase6_end = t0 + budget_s * 0.995
    if time.time() < phase6_end:
        n_iter = int((phase6_end - time.time()) / 0.0025)
        n_iter = min(n_iter, 150000)
        if n_iter > 1000:
            best_h, best_c5 = _local_search(best_h, n, rng, n_iter)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
