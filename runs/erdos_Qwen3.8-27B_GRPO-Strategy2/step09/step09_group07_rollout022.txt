                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _subgrad_step(h, eta, m_lags, n_step=5):
    n = len(h)
    h_cur = h.copy()
    c = _corr(h_cur)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = alpha_s = 200.0
    s_vals = np.clip(s * (c[idx] - m), -500, 500)
    s_vals -= s_vals.max()
    w = np.exp(s_vals)
    w /= w.sum()
    grad = np.zeros(n)
    dx = 2.0 / n
    nm1 = n - 1
    for ii in range(len(idx)):
        wk = w[ii] * dx
        if wk < 1e-15:
            continue
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:
            grad[L:] += wk * (1.0 - h_cur[:n - L])
            grad[:n - L] -= wk * h_cur[L:]
        else:
            M = -L
            grad[:n - M] += wk * (1.0 - h_cur[M:])
            grad[M:] -= wk * h_cur[:n - M]
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h_cur
    for _ in range(n_step):
        step = eta / max(gn, 1e-10)
        step = min(step, 0.1)
        h_new = h_cur - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        c5c = _eval(h_cur)
        if c5n < c5c:
            h_cur = h_new
            c = _corr(h_cur)
            m = c.max()
            idx = np.argsort(c)[-m_lags:]
            s_vals = np.clip(s * (c[idx] - m), -500, 500)
            s_vals -= s_vals.max()
            w = np.exp(s_vals)
            w /= w.sum()
            grad = np.zeros(n)
            for ii in range(len(idx)):
                wk = w[ii] * dx
                if wk < 1e-15:
                    continue
                k = int(idx[ii])
                L = k - nm1
                if L >= 0:
                    grad[L:] += wk * (1.0 - h_cur[:n - L])
                    grad[:n - L] -= wk * h_cur[L:]
                else:
                    M = -L
                    grad[:n - M] += wk * (1.0 - h_cur[M:])
                    grad[M:] -= wk * h_cur[:n - M]
            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                break
            step = eta / max(gn, 1e-10)
            step = min(step, 0.05)
        else:
            eta *= 0.5
            if eta < 1e-4:
                break
    return h_cur

def _local_search(h, rng, n_iter):
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval(best_h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        d = 3e-6 * rng.randn()
        ni, nj = best_h[i] + d, best_h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            best_h[i], best_h[j] = ni, nj
            c5n = _eval(best_h)
            if c5n < best_c5:
                best_c5 = c5n
            else:
                best_h[i], best_h[j] = ni - d, nj + d
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Intensive CD with stall detection (45% budget)
    p1_end = t0 + budget_s * 0.45
    stall = 0
    while time.time() < p1_end:
        h, c = _cd_pass(h, rng, tert=20)
        h = _repair(h, n)
        c5 = _eval(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 50:
            h = best_h.copy()
            n_pert = max(1, n // 8)
            idx = rng.choice(n, n_pert, replace=False)
            h[idx] += rng.randn(n_pert) * 0.02
            h = _repair(h, n)
            stall = 0

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)

    # Phase 2: Subgradient + SLSQP continuation (20% budget)
    p2_end = t0 + budget_s * 0.65
    if time.time() < p2_end:
        h_sub = _subgrad_step(best_h, 0.05, 30, n_step=8)
        h_sub = _repair(h_sub, n)
        c5s = _eval(h_sub)
        if c5s < best_c5:
            best_h = h_sub
            best_c5 = c5s
    if time.time() < t0 + budget_s - 250:
        for alpha, ml in [(2000, 30), (8000, 40), (30000, 50), (100000, 60), (400000, 70)]:
            if time.time() > t0 + budget_s - 200:
                break
            h_new = _slsqp_refine(best_h.copy(), alpha, ml, 250)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 3: More CD from best (15% budget)
    p3_end = t0 + budget_s * 0.80
    if time.time() < p3_end:
        h2 = best_h.copy()
        while time.time() < p3_end:
            h2, c = _cd_pass(h2, rng, tert=20)
            h2 = _repair(h2, n)
            c5 = _eval(h2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h2.copy()

    # Phase 4: Final SLSQP at high alpha
    if time.time() < t0 + budget_s - 60:
        h_new = _slsqp_refine(best_h.copy(), 2000000, 80, 400)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 5: Local search (remaining time)
    if time.time() < t0 + budget_s - 3:
        left = max(0.0, t0 + budget_s - 3 - time.time())
        n_iter = min(int(left / 0.0025), 200000)
        best_h, best_c5 = _local_search(best_h, rng, n_iter)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
