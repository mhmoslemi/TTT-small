    def _el(): return time.time() - t0
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0); h = _repair(h0, n)
    best_h, best_c5 = h.copy(), _eval(h)
    for ratio in [4, 7]:
        if _el() > 12: break
        nc = max(50, n // ratio); f = n // nc
        hc = _repair(h0[:nc*f].reshape(nc, f).mean(axis=1).copy(), nc)
        for a, ml in [(5e3, 20), (5e4, 40), (5e5, 60)]:
            if _el() > 20: break
            hc = _slsqp(hc, a, ml, 200)
        hu = _repair(np.repeat(hc, f), n)
        cu = _eval(hu)
        if cu < best_c5: best_c5, best_h = cu, hu.copy()
    for a, ml, mi in [(2e3, 20, 400), (1e4, 30, 450), (5e4, 40, 500),
                       (2e5, 50, 500), (1e6, 55, 500), (5e6, 60, 450),
                       (2e7, 65, 400), (1e8, 70, 350), (5e8, 75, 300),
                       (2e9, 80, 250), (1e10, 85, 200), (5e10, 90, 150),
                       (5e11, 95, 100)]:
        if _el() > 430: break
        hn = _slsqp(best_h, a, ml, mi); cn = _eval(hn)
        if cn < best_c5: best_c5, best_h = cn, hn.copy()
    if _el() < budget_s - 260:
        for a, ml, ns, s0 in [(2e4, 50, 1000, .006), (2e5, 60, 800, .004),
                               (2e6, 70, 600, .002), (2e7, 80, 500, .001),
                               (2e8, 90, 400, .0005)]:
            if _el() > budget_s - 200: break
            hn = _pgd(best_h.copy(), a, ml, ns, s0); cn = _eval(hn)
            if cn < best_c5: best_c5, best_h = cn, hn.copy()
    if _el() < budget_s - 100:
        nt = int((budget_s - 100 - _el()) / 0.0016)
        best_h, best_c5 = _ms_search(best_h, nt, rng)
    if _el() < budget_s - 35:
        for a, ml in [(1e10, 120), (1e11, 130)]:
            if _el() > budget_s - 20: break
            hn = _slsqp(best_h, a, ml, 150); cn = _eval(hn)
            if cn < best_c5: best_c5, best_h = cn, hn.copy()
    if _el() < budget_s - 8:
        nt = int((budget_s - 8 - _el()) / 0.0012)
        best_h, best_c5 = _ms_search(best_h, nt, rng)
    if _el() < budget_s - 1.5:
        nt = int((budget_s - 1.5 - _el()) / 0.0003)
        best_h, best_c5 = _polish(best_h, nt, rng)
    best_h = _repair(best_h, len(best_h)); best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
