                D = _delta_matrix(h)

def _anneal(h, C, D, rng, deadline, T0=0.005):

    best_h = h.copy()
    best_c = cur_c
    total = deadline - t0

        T = T0 * (0.01 + 0.99 * frac)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        new_C = C - D[i] + D[j]
        new_c = float(new_C.max())
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-10)):
            h[i] = 0.0
            h[j] = 1.0
            C = _corr(h)
            D = _delta_matrix(h)
            cur_c = C.max()
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def _block_move(h, C, rng, deadline):

        h[b:b+L] = tmp
        new_c = _c5(h)

            best_h = h.copy()
            C = _corr(h)
        else:
            tmp = h[a:a+L].copy()
            h[a:a+L] = h[b:b+L]
            h[b:b+L] = tmp
    return best_h, best_c

import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _delta_matrix(h):
    n = len(h)
    k = np.arange(-(n - 1), n)
    ii = np.arange(n)[:, None]
    ik = ii + k[None, :]
    v1 = (ik >= 0) & (ik < n)
    t1 = np.where(v1, 1.0 - h[np.clip(ik, 0, n - 1)], 0.0)
    ik2 = ii - k[None, :]
    v2 = (ik2 >= 0) & (ik2 < n)
    t2 = np.where(v2, h[np.clip(ik2, 0, n - 1)], 0.0)
    return t1 - t2

def _binarize(h):
    n = len(h)
    t = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:t]] = 1.0
    return out

def _steepest(h, C, D, rng, deadline, n_try=80):
