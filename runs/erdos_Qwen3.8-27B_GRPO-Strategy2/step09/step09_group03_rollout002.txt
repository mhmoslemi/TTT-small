        else:

Actually, let me try a different strategy for the fine-tuning phase. Instead of random moves, use a "steepest descent" in the space of 2-point moves:

For each pair (i, j), the change in c_k when we add ε to h[i] and subtract ε from h[j] is:
Δc_k = ε * [∂c_k/∂h[i] - ∂c_k/∂h[j]]

We want to choose (i, j, ε) that minimizes the max c_k. This is a complex combinatorial problem, but we can approximate:

1. Compute all ∂c_k/∂h[i] for the active lags
2. Find the pair (i, j) that gives the most negative Δ(max c_k)
3. Make that move

But computing all ∂c_k/∂h[i] for all pairs is O(n² * |active|) which is too slow.

A compromise: for each i, compute the "sensitivity" s[i] = sum of ∂c_k/∂h[i] over active k (weighted). Then the best 2-point move is to increase h at the position with minimum s and decrease at maximum s.

This is essentially what the subgradient does! The subgradient gives s[i] = ∂max/∂h[i], and the gradient descent step is h ← h - lr * s (projected). The 2-point move version would be:
- i* = argmin s[i] (most beneficial to increase)
- j* = argmax s[i] (most beneficial to decrease)
- h[i*] += ε, h[j*] -= ε

This is a "steepest 2-point descent" and should be more effective than random 2-point moves.

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n * 0.5
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            slack = 1.0 - h
            tot = slack.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * slack
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
            else:
                a = max(diff, -h[i])
            h[i] += a
            diff -= a
            if abs(diff) < 1e-15:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj(h, target):
    lo, hi = -4.0, 4.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _subgrad(h, c, frac):
    n = len(h)
    m = c.max()
    thresh = m - frac * m
    active = np.where(c >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    w = c[active].copy()
    ws = w.sum()
    if ws < 1e-15:
        w[:] = 1.0 / len(w)
    else:
        w /= ws
    g = np.zeros(n)
    for ii in range(len(active)):
        if w[ii] < 1e-12:
            continue
        L = active[ii] - (n - 1)
        wt = w[ii]
        if L >= 0:
            g[L:] += wt * (1.0 - h[:n - L])
            g[:n - L] -= wt * h[L:]
        else:
            M = -L
            g[:n - M] += wt * (1.0 - h[M:])
            g[M:] -= wt * h[:n - M]
    return g

def _pg_loop(h, steps, lr, mom, frac, deadline):
    n = len(h)
    v = np.zeros(n)
    best_c = _eval(h)
    best_h = h.copy()
    c = _eval_vec(h)
    for it in range(steps):
        if time.time() > deadline:
            break
        g = _subgrad(h, c, frac)
        v = mom * v - lr * g
        h_new = _proj(h + v, n * 0.5)
        c_new = _eval_vec(h_new)
        val = float(c_new.max())
        h = h_new
        c = c_new
        if val < best_c:
            best_c = val
            best_h = h.copy()
    return best_h, best_c

def _steepest2(h, n_iter, rng, deadline):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    c_vec = _eval_vec(h)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        g = _subgrad(h, c_vec, 0.005)
        i_st = np.argmin(g)
        j_st = np.argmax(g)
        moved = False
        for eps in (5e-5, 2e-5, 8e-6, 3e-6, 1e-6, 4e-7):
            h[i_st] += eps
            h[j_st] -= eps
            if h[i_st] > 1.0 or h[j_st] < 0.0:
                h[i_st] -= eps
                h[j_st] += eps
                continue
            c_new = _eval(h)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h.copy()
                c_vec = _eval_vec(h)
                moved = True
                break
            h[i_st] -= eps
            h[j_st] += eps
        if it % 50 == 0:
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                d = 1e-5 * rng.randn()
                h[i] += d
                h[j] -= d
                if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                    c_new = _eval(h)
                    if c_new < best_c - 1e-16:
                        best_c = c_new
                        best_h = h.copy()
                        c_vec = _eval_vec(h)
                    else:
                        h[i] -= d
                        h[j] += d
    return best_h, best_c

def _random2(h, n_iter, rng, deadline):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 2e-6 * rng.randn()
        h[i] += d
        h[j] -= d
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
            h[i] -= d
            h[j] += d
            continue
        c_new = _eval(h)
        if c_new < best_c - 1e-16:
            best_c = c_new
            best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d
    return best_h, best_c

def _spread(h, sig):
    n = len(h)
    x = np.arange(n)
    k = np.zeros(n)
    for sh in (-n, 0, n):
        k += np.exp(-0.5 * ((x - sh) / sig) ** 2)
    k /= k.sum()
    from scipy.signal import fftconvolve
    return fftconvolve(h, k, mode='wrap')

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)

        if c_new < best_c5:

            best_h = h_new

        h_sp = _spread(best_h, sig)

    return (best_h, best_c5, n)
