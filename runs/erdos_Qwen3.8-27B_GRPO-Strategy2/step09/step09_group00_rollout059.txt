def _proj_excess(v, S):
    """Project v onto {x: 0<=x[i]<=1, sum(x)=S}"""
    n = len(v)
    lo, hi = -v.max(), 1.0 - v.min()
    for _ in range(60):
        mid = 0.5*(lo+hi)
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > S:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + 0.5*(lo+hi), 0.0, 1.0)
    # Fix residual
    d = S - x.sum()
    if abs(d) > 1e-14:
        if d > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += d/cnt
            np.clip(x, 0.0, 1.0, out=x)
    return x
