import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    for _ in range(200):
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_all(h):

def _active_lags(c, band):
    m = c.max()
    return np.where(c >= m - band)[0]

def _grad_for_lags(h, lags):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in lags:
        L = k - (n - 1)
        if L >= 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    g *= dx
    g -= g.mean()
    return g

    for _ in range(80):

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid

    cur_c = _eval(h)

    mu = 0.82
    stall_count = 0

        g = _grad_for_lags(h, lags)
        ng = np.linalg.norm(g)
