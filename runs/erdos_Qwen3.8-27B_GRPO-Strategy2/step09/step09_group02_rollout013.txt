    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    for _ in range(100):
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:

        else:

                diff -= a

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    for _ in range(100):
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj_sum(h, target):
    lo, hi = -10.0, 10.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(x, alpha, ml):
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n-L])
            g[:n-L] -= wk * x[L:]
        else:
            M = -L
            g[:n-M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n-M]
    return g

def _sum_fun(x):
    return x.sum() - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp_opt(h, alpha, ml, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, ml), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _nesterov_proj(h, alpha, ml, steps, step0):
    n = len(h)
    x = h.copy()
    y = h.copy()
    best_x = h.copy()
    best_c = _eval(h)
    step = step0
    for t in range(1, steps + 1):
        beta = (3.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (beta - 1.0) / beta
        g = _smooth_grad(y, alpha, ml)
        gn = np.linalg.norm(g)
        if gn < 1e-18:
            break
        g /= gn
        cur_c = _eval(y)
        taken = False
        for _ in range(8):
            x_new = _proj_sum(y - step * g, n * 0.5)
            c_new = _eval(x_new)
            if c_new < cur_c - 1e-17:
                y_new = x_new + theta * (x_new - x)
                y_new = np.clip(y_new, 0.0, 1.0)
                y = y_new
                x = x_new
                cur_c = c_new
                step = min(step * 1.05, step0 * 40.0)
                taken = True
                if c_new < best_c:
                    best_c = c_new
                    best_x = x_new.copy()
                break
            step *= 0.4
        if not taken:
            step = max(step * 0.7, step0 * 0.1)
            y = x.copy()
    return _repair(best_x, n), best_c

def _exact_swap(h, i, j, c_cur, n):
    """Find optimal d for h[i]+=d, h[j]-=d via 1D subgradient bisection."""
    d_lo = max(-h[i], h[j] - 1.0)
    d_hi = min(1.0 - h[i], h[j])
    if d_hi - d_lo < 1e-14:
        return 0.0
    eps = 1e-8
    h[i] += eps
    h[j] -= eps
    c_p = _corr(h)
    h[i] -= 2.0 * eps
    h[j] += 2.0 * eps
    c_m = _corr(h)
    h[i] += eps
    h[j] -= eps
    slopes = (c_p - c_m) / (2.0 * eps)
    h[i] -= eps
    h[j] += eps
    top_idx = np.argsort(c_cur)[-120:]
    a = slopes[top_idx]
    b = c_cur[top_idx]
    for _ in range(30):
        d_mid = 0.5 * (d_lo + d_hi)
        vals = a * d_mid + b
        active = np.argwhere(vals >= vals.max() - 1e-15)
        g_sub = a[active].sum()
        if g_sub > 1e-12:
            d_hi = d_mid
        elif g_sub < -1e-12:
            d_lo = d_mid
        else:
            break
    d_opt = 0.5 * (d_lo + d_hi)
    cands = [d_opt, d_lo, d_hi, 0.0]
    best_d = 0.0
    best_val = float('inf')
    for d in cands:
        if d < d_lo - 1e-14 or d > d_hi + 1e-14:
            continue
        d = np.clip(d, d_lo, d_hi)
        h[i] += d
        h[j] -= d
        cv = float(np.max(_corr(h)))
        h[i] -= d
        h[j] += d
        if cv < best_val:
            best_val = cv
            best_d = d
    return best_d

def _local_search(h, n, rng, niter, base_step, c5_init):
    best_h = h.copy()
    best_c = c5_init
    cur_c = c5_init
    for it in range(niter):
        i = rng.randint(n)
        dist = 1 + rng.randint(0, max(1, n // 3))
        j = (i + dist) % n
        if i == j:
            continue
        c_cur = _corr(h)
        d = _exact_swap(h, i, j, c_cur, n)
        if d == 0.0:
            d = base_step * (0.2 + 0.8 * rng.rand()) * (1 if rng.rand() < 0.5 else -1)
        h[i] += d
        h[j] -= d
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
            h[i] -= d
            h[j] += d
            continue
        h = _repair(h, n)
        c_new = _eval(h)
        if c_new < cur_c - 1e-16:
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d
            h = _repair(h, n)
    return best_h, best_c

def _fine_polish(h, n, rng, niter):
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(niter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 16)) % n
        d = (5e-9 + 8e-8 * rng.rand()) * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-17:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _group_polish(h, n, rng, niter, gs):
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(niter):
        base = rng.randint(n)
        idx = np.sort((base + rng.randint(0, 30, size=gs)) % n)
        d = (5e-9 + 1e-7 * rng.rand()) * rng.randn(gs)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < -1e-6 or h_new.max() > 1.0 + 1e-6:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-17:
            c5c = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP with alpha sweep
    alpha_seq = [50000, 500000, 5000000, 50000000, 500000000, 5000000000]
    for alpha in alpha_seq:
        if _el() > 200:
            break
        ml = min(180, max(15, int(10 * np.log10(alpha) - 20)))
        h_new = _slsqp_opt(best_h.copy(), alpha, ml, 1200, 1e-16)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 2: Nesterov projected gradient
    if _el() < budget_s - 400:
        configs = [
            (10000, 30, 2500, 0.006),
            (100000, 45, 2000, 0.004),
            (500000, 60, 1500, 0.0025),
            (2000000, 80, 1200, 0.0015),
            (10000000, 100, 1000, 0.0008),
            (50000000, 120, 800, 0.0005),
        ]
        for alpha, ml, steps, s0 in configs:
            if _el() > budget_s - 300:
                break
            h_new, c_new = _nesterov_proj(best_h.copy(), alpha, ml, steps, s0)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new

    # Phase 3: SLSQP high alpha
    if _el() < budget_s - 250:
        for alpha in [20000000000, 100000000000]:
            if _el() > budget_s - 200:
                break
            h_new = _slsqp_opt(best_h, alpha, 200, 600, 1e-17)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 4: Exact 2-swap local search
    if _el() < budget_s - 80:
        rem = budget_s - 80 - _el()
        ntries = int(rem * 350)
        best_h, best_c5 = _local_search(best_h, n, rng, ntries, 3e-5, best_c5)

    # Phase 5: Fine polish
    if _el() < budget_s - 30:
        rem = budget_s - 30 - _el()
        ntries = int(rem * 900)
        best_h, best_c5 = _fine_polish(best_h, n, rng, ntries)

    # Phase 6: Group polish
    if _el() < budget_s - 10:
        rem = budget_s - 10 - _el()
        ntries = int(rem * 400)
        best_h, best_c5 = _group_polish(best_h, n, rng, ntries, 4)

    # Phase 7: Ultra-fine final polish
    if _el() < budget_s - 2:
        rem = budget_s - 2 - _el()
        ntries = int(rem * 1000)
        best_h, best_c5 = _fine_polish(best_h, n, rng, ntries)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
