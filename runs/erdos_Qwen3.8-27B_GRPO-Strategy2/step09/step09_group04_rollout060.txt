        d = 3e-5 * rng.randn(bs)
        d -= d.mean()
        hn = h.copy()
        hn[i:i + bs] += d
        if hn.min() < -0.3 or hn.max() > 1.3:
            continue
        hn = _repair(hn, n)
        cn = _eval(hn)
        dl = cn - cc
        if dl < 0 or (t > 1e-10 and rng.rand() < np.exp(-dl / t)):
            h, cc = hn, cn
            if cn < bc:
                bc, bh = cn, h.copy()
    return bh, bc

def _polish(h, n, nit, rng, step0, stepend):
    cc = _eval(h)
    bh, bc = h.copy(), cc
    h2 = h.copy()
    for it in range(nit):
        frac = it / max(1, nit)
        step = step0 * (stepend / step0) ** frac
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = step * rng.randn()
        h2[i] += d
        h2[j] -= d
        if 0.0 <= h2[i] <= 1.0 and 0.0 <= h2[j] <= 1.0:
            cn = _eval(h2)
            if cn < cc:
                cc = cn
                if cn < bc:
                    bc, bh = cn, h2.copy()
        else:
            h2[i] -= d
            h2[j] += d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    bh, bc5 = h.copy(), _eval(h)

    def _el():
        return time.time() - t0

    # Phase 1: SLSQP smoothmax sweep
    for a in [5000, 20000, 100000, 500000, 2000000, 10000000, 50000000]:
        if _el() > 200:
            break
        ml = min(80, max(15, int(15 * np.log10(a / 5000.0) + 15)))
        hn = _do_slsqp(bh, a, ml, 1500)
        cn = _eval(hn)
        if cn < bc5:
            bc5, bh = cn, hn

    # Phase 2: Adam projected gradient
    if _el() < budget_s - 380:
        for a, ml, lr, ni in [(20000, 40, 0.005, 600),
                              (200000, 60, 0.003, 500),
                              (2000000, 80, 0.001, 400),
                              (20000000, 100, 0.0005, 300)]:
            if _el() > budget_s - 330:
                break
            hn, cn = _adam_h(bh, a, ml, ni, lr)
            if cn < bc5:
                bc5, bh = cn, hn

    # Phase 3: Binary GA for diverse topologies
    if _el() < budget_s - 280:
        gt = min(80, int((budget_s - 280 - _el()) * 0.6))
        ga_h, ga_c = _ga_binary(n, n // 2, 24, 2000, rng, gt)
        ga_h = _repair(ga_h, n)
        ga_c = _eval(ga_h)
        ga_h2 = _do_slsqp(ga_h, 200000, 50, 500)
        ga_c2 = _eval(ga_h2)
        if ga_c2 < bc5:
            bc5, bh = ga_c2, ga_h2
        elif ga_c < bc5:
            bc5, bh = ga_c, ga_h

    # Phase 4: SLSQP refinement after GA
    if _el() < budget_s - 200:
        for a in [500000, 5000000]:
            if _el() > budget_s - 160:
                break
            hn = _do_slsqp(bh, a, 60, 800)
            cn = _eval(hn)
            if cn < bc5:
                bc5, bh = cn, hn

    # Phase 5: Block simulated annealing
    if _el() < budget_s - 120:
        blocks = np.array([2, 3, 4, 5, 6, 8, 12])
        nt = min(int((budget_s - 120 - _el()) / 0.004), 80000)
        bh, bc5 = _sa_blocks(bh, n, nt, rng, blocks, 5e-5, 0.99993)

    # Phase 6: Extended 2-point polish with annealed step
    if _el() < budget_s - 2:
        remaining = budget_s - 2 - _el()
        npol = min(int(remaining / 0.0011), 900000)
        bh, bc5 = _polish(bh, n, npol, rng, 5e-6, 5e-9)

    # Phase 7: Final ultra-fine polish
    if _el() < budget_s - 1:
        nh = min(int((budget_s - 1 - _el()) / 0.0011), 50000)
        if nh > 100:
            bh, bc5 = _polish(bh, n, nh, rng, 1e-7, 1e-10)

    bh = _repair(bh, n)
    bc5 = _eval(bh)
    return (bh, bc5, n)
