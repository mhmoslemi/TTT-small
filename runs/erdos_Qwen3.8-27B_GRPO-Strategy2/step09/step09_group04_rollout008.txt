def _repair(h, n):

    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a

def _coord_opt(h, i):
    """Find optimal value of h[i] minimizing max correlation."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')  # raw correlation
    # Compute slope of each c[k] w.r.t. h[i]
    slopes = np.zeros(2*n-1)
    for k in range(2*n-1):
        L = k - (n-1)
        s = 0.0
        if L >= 0:
            if i <= n-1-L:
                s += (1.0 - h[i+L])
            if i >= L:
                s -= h[i-L]
        else:
            M = -L
            if i >= M:
                s += (1.0 - h[i-M])
            if i <= n-1-M:
                s -= h[i+M]
        slopes[k] = s

    xi = h[i]
    offsets = c - slopes * xi

    # Evaluate at several points

    s -= s.max()

        h1, c1 = _coord_sweep(best_h, rng, 500, tlim)
        if c1 < best_c5:
            best_c5, best_h = c1, h1
    
    # Phase 2: Gradient descent with soft-max
    if _el() < budget_s - 150:
        for alpha, ml, steps, s0 in [(50000, 50, 500, 0.005),
                                      (500000, 80, 400, 0.003),
                                      (5000000, 100, 300, 0.001)]:
            if _el() > budget_s - 150:
                break
            hn, cn = _grad_descend(best_h.copy(), alpha, ml, steps, s0)
            if cn < best_c5:
                best_c5, best_h = cn, hn
    
    # Phase 3: More coordinate descent
    if _el() < budget_s - 80:
        tlim = min(100, budget_s - 130)
        h2, c2 = _coord_sweep(best_h, rng, 1000, tlim)
        if c2 < best_c5:
            best_c5, best_h = c2, h2
    
    # Phase 4: Try different n values
    for n_try in [400, 480, 520, 560, 600, 640]:
        if _el() > budget_s - 60:
            break
        if n_try == n0:
            continue
        # Create initial h for this n by interpolation
        h_init = np.interp(np.linspace(0, 2, n_try, endpoint=False),
                          np.linspace(0, 2, n0, endpoint=False),
                          best_h)
        h_init = _repair(h_init, n_try)
        c_init = _eval(h_init)
        # Quick coordinate descent
        tlim = min(30, (budget_s - 60 - _el()) / 2)
        if tlim > 5:
            hn, cn = _coord_sweep(h_init, rng, 200, tlim)
            if cn < c_init:
                h_init, c_init = hn, cn
        if c_init < best_c5:
            best_c5, best_h = c_init, h_init
    
    # Phase 5: Fine gradient polish
    if _el() < budget_s - 30:
        n = len(best_h)
        for alpha, ml, steps, s0 in [(20000000, 120, 200, 0.0005),
                                      (100000000, 150, 150, 0.0003)]:
            if _el() > budget_s - 30:
                break
            hn, cn = _grad_descend(best_h.copy(), alpha, ml, steps, s0)
            if cn < best_c5:
                best_c5, best_h = cn, hn
    
    # Phase 6: Ultra-fine coordinate polish
    if _el() < budget_s - 5:
        n = len(best_h)
        h2 = best_h.copy()
        c2 = best_c5
        order = rng.permutation(n)
        max_iter = int((budget_s - 5 - _el()) / 0.02)  # ~20ms per coord op
        for it in range(max_iter):
            if _el() > budget_s - 5:
                break
            i = order[it % n]
