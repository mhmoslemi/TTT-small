            if L >= 0:
                valid_n = n - L
                contrib = cur[:valid_n] * (1.0 - cur[L:L+valid_n])
                im = np.argmax(contrib)
                p1, p2 = im, im + L
            else:
                M = -L
                valid_n = n - M
                contrib = cur[M:M+valid_n] * (1.0 - cur[:valid_n])
                im = np.argmax(contrib)
                p1, p2 = im + M, im
            step = step_base * (0.5 + rng.rand())
            if cur[p1] > 0.4 and cur[p2] > 0.4:
                hi_i, lo_i = (p1, p2) if cur[p1] >= cur[p2] else (p2, p1)
                cur[hi_i] -= step
                cur[lo_i] += step
            elif cur[p1] < 0.6 and cur[p2] < 0.6:
                hi_i, lo_i = (p1, p2) if cur[p1] <= cur[p2] else (p2, p1)
                cur[hi_i] += step
                cur[lo_i] -= step
            else:
                d = step * (1.0 if rng.rand() > 0.5 else -1.0)
                cur[p1] += d
                cur[p2] -= d
            if cur.min() < -1e-9 or cur.max() > 1+1e-9:
                cur[p1] = np.clip(cur[p1], 0, 1)
                cur[p2] = np.clip(cur[p2], 0, 1)
            cur = _repair(cur, n)
            c_new = _eval(cur)
            if c_new < cmax:
                cmax = c_new
                improved = True
                if cmax < best_c:
                    best_c = cmax
                    best_h = cur.copy()
                stuck = 0
                break
            else:
                # Undo by re-repairing won't work; just accept the worse state occasionally
                if rng.rand() < 0.02:
                    stuck = 0
                else:
                    stuck += 1
        if stuck > 80:
            # Perturb to escape
            idx = rng.choice(n, size=4, replace=False)
            cur[idx] += 0.005 * rng.randn(4)
            cur = _repair(cur, n)
            stuck = 0
        if cmax < 0.3808:
            break
    return best_h, best_c

def _proj_grad_desc(h, alpha, ml, steps, step0):
    n = len(h)
    v = np.zeros(n)
    sq = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    for it in range(steps):
        g = _sm_grad(h, alpha, ml)
        gp = g - g.mean()
        v = 0.9 * v + 0.1 * gp
        sq = 0.95 * sq + 0.05 * g * g
        direction = v / (np.sqrt(sq) + 1e-8)
        taken = False
        for _t in range(16):
            h_new = _proj(h - step * direction, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:
                h = h_new
                cur_c = c_new
                step = min(step * 1.15, step0 * 50.0)
                taken = True
                break
            step *= 0.45
        if not taken:
            step = min(step * 1.3, step0 * 50.0)
    return _repair(h, n)

def _fine_polish(h, n_iter, rng, step):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(best_h)
    cur_c = best_c
    cur = h.copy()
    for it in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        if i == j:
            continue
        d = step * rng.randn()
        ni, nj = cur[i] + d, cur[j] - d
        if not (0 <= ni <= 1 and 0 <= nj <= 1):
            continue
        cur[i], cur[j] = ni, nj
        c_new = _eval(cur)
        if c_new < cur_c:
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = cur.copy()
        else:
            cur[i], cur[j] = ni - d, nj + d
    return best_h, best_c

def _make_start(n, seed):
    rng = np.random.RandomState(seed)
    h = np.full(n, 0.5)
    h += 0.15 * np.sin(2 * np.pi * 3 * np.arange(n) / n)
    h += 0.08 * np.sin(2 * np.pi * 7 * np.arange(n) / n + 1.3)
    h += 0.05 * rng.randn(n)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Also try alternative starts
    alt_starts = [h.copy()]
    for s in [seed+1, seed+2]:
        alt_starts.append(_make_start(n, s))

    # Phase 1: Quick SLSQP on best start
    for start in alt_starts[:2]:
        if time.time() - t0 > 25:
            break
        h_new = _slsqp(start, 5000, 40, 300, 1e-14)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 2: SLSQP alpha sweep
    alphas = [10000, 50000, 200000, 1000000, 5000000, 30000000, 200000000]
    for a in alphas:
        if time.time() - t0 > 200:
            break
        lift = min(120, max(20, int(25*np.log10(a/10000) + 25)))
        h_new = _slsqp(best_h, a, lift, 800, 1e-15)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 3: Projected gradient descent
    if time.time() - t0 < budget_s - 500:
        configs = [(5000, 40, 800, 0.004), (20000, 60, 600, 0.003),
                   (100000, 80, 500, 0.002), (500000, 100, 400, 0.001),
                   (2000000, 150, 300, 0.0005)]
        for alpha, ml, steps, s0 in configs:
            if time.time() - t0 > budget_s - 450:
                break
            h_new = _proj_grad_desc(best_h.copy(), alpha, ml, steps, s0)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 4: Bottleneck-aware local search (main novel phase)
    if time.time() - t0 < budget_s - 200:
        remaining = budget_s - 200 - (time.time() - t0)
        n_iter = int(remaining / 0.006)
        best_h, best_c5 = _bottleneck_ls(best_h, n_iter, rng, topk=8, step_base=0.0025)

    # Phase 5: Fine polish with decreasing steps
    if time.time() - t0 < budget_s - 30:
        remaining = budget_s - 30 - (time.time() - t0)
        for step in [1e-5, 5e-6, 1e-6, 5e-7]:
            val = int(remaining * 0.2 / 0.001)
            if val < 100:
                val = 100
            best_h, best_c5 = _fine_polish(best_h, val, rng, step)
            remaining = budget_s - 30 - (time.time() - t0)
            if remaining < 2:
                break

    # Phase 6: Final ultra-fine
    if time.time() - t0 < budget_s - 1:
        remaining = budget_s - 1 - (time.time() - t0)
        val = int(remaining / 0.0008)
        best_h, best_c5 = _fine_polish(best_h, val, rng, 2e-7)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
