import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-11:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _cd_pass(h, rng, n):
    c = _corr(h)
    cmax = 2 * n - 1
    nm1 = n - 1
    offs = np.arange(cmax) - nm1
    jlist = np.arange(n)
    rng.shuffle(jlist)
    for p in range(n):
        j = int(jlist[p])
        beta = np.zeros(cmax)
        lo, hi = j, min(cmax - 1, j + nm1)
        beta[lo:hi+1] += 1.0 - h[j - offs[lo:hi+1]]
        lo2, hi2 = max(0, nm1 - j), min(cmax - 1, cmax - 1 - j)
        beta[lo2:hi2+1] -= h[j + offs[lo2:hi2+1]]
        g = c - beta * h[j]
        a, b = 0.0, 1.0
        for _ in range(22):
            if b - a < 1e-11:
                break
            t1 = a + (b - a) * 0.33333
            t2 = b - (b - a) * 0.33333
            if np.max(g + beta * t1) <= np.max(g + beta * t2):
                b = t1
            else:
                a = t2
        tn = 0.5 * (a + b)
        if abs(tn - h[j]) > 1e-12:
            c += beta * (tn - h[j])
            h[j] = tn
    return h

def _sob(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = np.clip(alpha * (c[idx] - m), -500.0, 500.0)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sf(x, t):
    return x.sum() - t

def _sj(x, t):
    return np.ones(len(x))

def _slsqp(h, alpha, ml, mi):
    n = len(h)
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (n * 0.5,)}]
    try:
        r = minimize(_sob, h, method='SLSQP', args=(alpha, ml),
                     bounds=bd, constraints=cs,
                     options={'maxiter': mi, 'ftol': 1e-14})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _local(h, n, rng, ni):
    c5 = _eval(h)
    bh = h.copy()
    b5 = c5
    for _ in range(ni):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 10)) % n
        d = 2e-6 * rng.randn()
        ni2, nj2 = h[i] + d, h[j] - d
        if 0.0 <= ni2 <= 1.0 and 0.0 <= nj2 <= 1.0:
            h[i], h[j] = ni2, nj2
            c5n = _eval(h)
            if c5n < c5:
                c5 = c5n
                if c5n < b5:
                    b5 = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni2 - d, nj2 + d
    return bh, b5
