def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    for _ in range(200):
        if abs(s - target) < 1e-14:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-15:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

    m = c.max()

    d_max = min(1.0 - h[i], h[j])
    
    if d_max <= d_min:
        return 0.0, c_max
    
    # Binary search or try several values
    # Actually, the optimal d is where the maximum correlation is minimized
    # This is a 1D optimization problem
    
    # Quick approach: try logarithmically spaced values
    for d in np.linspace(-d_max, d_max, 21):
        if abs(d) < 1e-15:
            continue
        h[i] += d
        h[j] -= d
        c_new = _corr(h)
        c_new_max = c_new.max()
        if c_new_max < best_c:
            best_c = c_new_max
            best_d = d
        h[i] -= d
        h[j] += d
    
    return best_d, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    for _ in range(200):
        if abs(s - target) < 1e-14:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-15:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(x, alpha, m_lags):
    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _cons_fun(x, t):
    return x.sum() - t

def _cons_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    target = n * 0.5
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (target,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _fista(h0, alpha, m_lags, n_iter, L):
    n = len(h0)
    h = h0.copy()
    z = h0.copy()
    t = 1.0
    for _ in range(n_iter):
        g = _smooth_grad(z, alpha, m_lags)
        g -= g.mean()
        y = np.clip(z - g / L, 0, 1)
        h_new = _proj(y, n * 0.5)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = h_new + ((t - 1.0) / t_new) * (h_new - h)
        z = np.clip(z, 0, 1)
        h = h_new
        t = t_new
    return h, _eval(h)

def _grad2point(h, n_iter, rng, m_lags=30, temp0=1e-6, temp_end=1e-12):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    g = _smooth_grad(h, 1e6, m_lags)
    g -= g.mean()
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        g = np.zeros(n)
        gn = 1.0
    else:
        g /= gn
    for it in range(n_iter):
        frac = it / max(1, n_iter)
        temp = temp0 * (temp_end / temp0) ** frac
        r = rng.rand()
        if r < 0.6:
            neg = g < -0.01
            pos = g > 0.01
            if neg.any() and pos.any():
                i = rng.choice(np.where(neg)[0])
                j = rng.choice(np.where(pos)[0])
            else:
                i, j = _rand_pair(n, rng)
        else:
            i, j = _rand_pair(n, rng)
        d_max = min(1.0 - h[i], h[j])
        d_min = max(0.0, h[j] - 1.0)
        if d_max - d_min < 1e-14:
            continue
        scale = 0.002 * (1.0 - 0.7 * frac)
        ds = np.linspace(-scale, scale, 13)
        best_d = 0.0
        best_c5 = c5c
        for d in ds:
            if d * (1.0 - h[i]) < -1e-12 or d + h[j] < -1e-12 or d > 0 and d > 1 - h[i] or d < 0 and -d > h[j]:
                continue
            h[i] += d
            h[j] -= d
            if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
                h[i] -= d
                h[j] += d
                continue
            c5n = _eval(h)
            if c5n < best_c5 - 1e-15 or (c5n - c5c < temp and rng.rand() < 0.3):
                best_c5 = c5n
                best_d = d
            h[i] -= d
            h[j] += d
        if best_d != 0.0 or best_c5 < c5c:
            h[i] += best_d
            h[j] -= best_d
            h = _repair(h, n)
            if _eval(h) < c5c + temp:
                c5c = _eval(h)
                if c5c < best_c:
                    best_c = c5c
                    best_h = h.copy()
        if (it + 1) % 500 == 0:
            g = _smooth_grad(h, 1e6, m_lags)
            g -= g.mean()
            gn = np.linalg.norm(g)
            if gn > 1e-15:
                g /= gn
            else:
                g = np.zeros(n)
    return best_h, best_c

def _rand_pair(n, rng):
    i = rng.randint(n)
    j = (i + 1 + rng.randint(0, n - 1)) % n
    return i, j

def _group_search(h, n_iter, rng, gs=4, scale=5e-5):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        base = rng.randint(n)
        idx = np.sort((base + rng.randint(0, 20, size=gs)) % n)
        d = scale * rng.randn(gs)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < -0.02 or h_new.max() > 1.02:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-16:
            c5c = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def _cos_init(n, f):
    return np.clip(0.5 + 0.45 * np.cos(2.0 * np.pi * f * np.arange(n) / n), 0, 1)
