    if np.linalg.norm(d) < 1e-12: return None

def _dense_line_search(h,C,curC,d,step):
    mabs = float(np.abs(d).max())
    if mabs <= 1e-15 or step <= 0: return h,C,curC,False
    d = d / mabs
    d -= d.mean()
    pos = d > 0; neg = d < 0
    af = np.inf
    if np.any(pos): af = min(af, float(np.min((1.0 - h[pos]) / d[pos])) * 0.999999)
    if np.any(neg): af = min(af, float(np.min(h[neg] / (-d[neg]))) * 0.999999)
    if not np.isfinite(af) or af <= 1e-14: return h,C,curC,False
    factors = (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125)
    for f in factors:
       a = step * f
       if a > af: a = af
       if a <= 1e-14: break
       hn = h + a * d
       if hn.min() < 0.0 or hn.max() > 1.0: continue
       Cn = _corr(hn); cn = float(Cn.max())
       if cn < curC - 1e-15:
          return hn, Cn, cn, True

    s = h.sum()
    if s < 1e-14:
        h.fill(0.5)
        return h
    for _ in range(80):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
                if abs(diff) < 1e-13:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
                if abs(diff) < 1e-13:
                    break
    return h


def _default_h(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 + 0.45 * np.sin(np.pi * x) + 0.03 * rng.randn(n)
    return _repair(h, n)


def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def _sum_fn(x, target):
    return x.sum() - target


def _sum_jac(x, target):
    return np.ones(len(x))


def _active_idx(C, K):
    K = max(1, min(int(K), C.size))
    if C.size <= K:
        return np.arange(C.size)
    return np.argpartition(C, -K)[-K:]


def _add_grad(g, h, L, w):
    n = len(h)
    m = n - abs(L)
    if w == 0.0 or m <= 0:
        return
    if L >= 0:
        g[:m] += w * (1.0 - h[:m])
        g[L:L + m] -= w * h[:m]
    else:
        M = -L
        g[M:M + m] += w * (1.0 - h[:m])
        g[:m] -= w * h[M:M + m]


def _smooth_obj(x, alpha, m_lags):
    C = _corr(x)
    m = float(C.max())
    idx = _active_idx(C, m_lags)
    z = alpha * (C[idx] - m)
    z -= z.max()
    return m + float(np.log(np.exp(z).sum())) / alpha


def _smooth_grad(x, alpha, m_lags):
    C = _corr(x)
    m = float(C.max())
    idx = _active_idx(C, m_lags)
    z = alpha * (C[idx] - m)
    z -= z.max()
    w = np.exp(z)
    sw = w.sum()
    g = np.zeros(len(x))
    if sw <= 0.0:
        return g
    w /= sw
    base = len(x) - 1
    for a in range(idx.size):
        wa = float(w[a])
        if wa > 1e-16:
            _add_grad(g, x, int(idx[a]) - base, wa)
    return g


def _slsqp_run(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _sum_fn, "jac": _sum_jac, "args": (n * 0.5,)}]
    try:
        res = minimize(
            _smooth_obj, h, jac=_smooth_grad, args=(alpha, m_lags),
            method="SLSQP", bounds=bnds, constraints=cons,
            options={"maxiter": maxiter, "ftol": ftol}
        )
        xv = np.asarray(res.x, dtype=float)
    except Exception:
        return h
    return _repair(xv, n)


def _grad_sum(h, active, weights=None):
    n = len(h)
    g = np.zeros(n)
    base = n - 1
    if weights is None:
        weights = np.ones(len(active))
    for a in range(len(active)):
        wa = float(weights[a])
        if wa != 0.0:
            _add_grad(g, h, int(active[a]) - base, wa)
    return g


def _grad_matrix(h, active):
    n = len(h)
    G = np.zeros((len(active), n))
    base = n - 1
    for a in range(len(active)):
        _add_grad(G[a], h, int(active[a]) - base, 1.0)
    return G


def _lp_direction(G, delta=1.0):
    A, n = G.shape
    if A == 0:
        return None
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.empty((A, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(A)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    lb = np.empty(n + 1)
    ub = np.empty(n + 1)
    lb[:n] = -delta
    ub[:n] = delta
    lb[-1] = -1e6
    ub[-1] = 1e6
    res = None
    try:
        res = linprog(c, A_ub, b_ub, A_eq, b_eq, bounds=(lb, ub), method="highs")
    except Exception:
        try:
            res = linprog(c, A_ub, b_ub, A_eq, b_eq, bounds=(lb, ub))
        except Exception:
            res = None
    if res is None or not getattr(res, "success", False) or not np.isfinite(res.fun):
        return None
    d = np.asarray(res.x[:n], dtype=float)
    d -= d.mean()
    if np.linalg.norm(d) < 1e-12:
        return None
    return d


def _dense_line_search(h, C, curC, d, step):
    mabs = float(np.abs(d).max())
    if mabs <= 1e-15 or step <= 0.0:
        return h, C, curC, False
    d = np.asarray(d, dtype=float)
    d = d / mabs
    d -= d.mean()
    pos = d > 0.0
    neg = d < 0.0
    af = np.inf
    if np.any(pos):
        af = min(af, float(np.min((1.0 - h[pos]) / d[pos])) * 0.999999)
    if np.any(neg):
        af = min(af, float(np.min(h[neg] / (-d[neg]))) * 0.999999)
    if not np.isfinite(af) or af <= 1e-14:
        return h, C, curC, False
    a0 = min(step, af)
    factors = (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125)
    for f in factors:
        a = a0 * f
        if a <= 1e-15:
            break
        hn = h + a * d
        if hn.min() < 0.0 or hn.max() > 1.0:
            continue
        Cn = _corr(hn)
        cn = float(Cn.max())
        if cn < curC - 1e-15:
            return hn, Cn, cn, True
    return h, C, curC, False


def _lp_phase(h, C, curC, t_end):
    for K in (6, 10, 4):
        if time.time() >= t_end:
            break
        step = 5e-4
        calls = 0
        noimp = 0
        while time.time() < t_end and calls < 70:
            calls += 1
            active = _active_idx(C, K)
            if active.size == 0:
                break
            G = _grad_matrix(h, active)
            d = _lp_direction(G, 1.0)
            if d is None:
                g = _grad_sum(h, active)
                d = -(g - g.mean())
                dn = float(np.linalg.norm(d))
                if dn > 1e-14:
                    d = d / dn
            d = np.asarray(d, dtype=float)
            d -= d.mean()
            mabs = float(np.abs(d).max())
            if not np.isfinite(mabs) or mabs <= 1e-14:
                noimp += 1
                if noimp > 30:
                    break
                continue
            h2, C2, c2, ok = _dense_line_search(h, C, curC, d, step)
            if ok:
                h, C, curC = h2, C2, c2
                noimp = 0
                if step < 2e-3:
                    step = min(step * 1.1, 2e-3)
            else:
                noimp += 1
                step *= 0.55
                if noimp > 25:
                    break
    return h, C, curC


def _best_from_variants(h, n):
    best_h = _repair(h, n)
    best_C = _corr(best_h)
    bestc = float(best_C.max())
    if n % 2 == 0:
        m = n // 2
        cand = 0.5 * (h + 1.0 - h[::-1])
        hv = _repair(cand, n)
        Cc = _corr(hv)
        c = float(Cc.max())
        if c < bestc:
            best_h, best_C, bestc = hv, Cc, c
        cand = np.empty(n)
        a = 0.5 * (h[:m] + 1.0 - h[m:])
        cand[:m] = a
        cand[m:] = 1.0 - a
        hv = _repair(cand, n)
        Cc = _corr(hv)
        c = float(Cc.max())
        if c < bestc:
            best_h, best_C, bestc = hv, Cc, c
    hv = _repair(1.0 - h, n)
    Cc = _corr(hv)
    c = float(Cc.max())
    if c < bestc:
        best_h, best_C, bestc = hv, Cc, c
    return best_h, best_C, bestc


def _coef_into(h, i, j, coef):
    n = len(h)
    coef.fill(0.0)
    om = 1.0 - h
    hrev = h[::-1]
    coef[n - 1 - j:2 * n - 1 - j] += om
    coef[j:j + n] -= hrev
    coef[n - 1 - i:2 * n - 1 - i] -= om
    coef[i:i + n] += hrev


def _pair_scan(h, C, curC, i, j, ts, prod, coef):
    if i == j or len(ts) == 0:
        return curC, False
    n = len(h)
    base = n - 1
    c0 = base
    _coef_into(h, i, j, coef)
    p1 = base + j - i
    p2 = base + i - j
    bestc = float(curC)
    bestt = 0.0
    for t in ts:
        t = float(t)
        if t == 0.0 or not np.isfinite(t):
            continue
        np.multiply(coef, t, out=prod)
        C += prod
        t2 = t * t
        C[c0] -= 2.0 * t2
        if p1 == p2:
            C[p1] += t2
        else:
            C[p1] += t2
            C[p2] += t2
        c = float(C.max())
        if c < bestc:
            bestc = c
            bestt = t
        if p1 == p2:
            C[p1] -= t2
        else:
            C[p1] -= t2
            C[p2] -= t2
        C[c0] += 2.0 * t2
        C -= prod
    if bestt == 0.0:
        return curC, False
    np.multiply(coef, bestt, out=prod)
    C += prod
    t2 = bestt * bestt
    C[c0] -= 2.0 * t2
    if p1 == p2:
        C[p1] += t2
    else:
        C[p1] += t2
        C[p2] += t2
    h[i] -= bestt
    h[j] += bestt
    return bestc, True


def _pair_phase(h, C, curC, t_end, rng):
    n = len(h)
    N = 2 * n - 1
    prod = np.empty(N, dtype=float)
    coef = np.empty(N, dtype=float)
    step = 1e-4
    K = min(10, max(1, n // 2))
    active = _active_idx(C, K)
    gp = _grad_sum(h, active)
    gp -= gp.mean()
    order = np.argsort(gp)
    lsize = max(1, min(8, n // 2))
    low_idx = order[:lsize]
    high_idx = order[-lsize:]
    noimp = 0
    it = 0
    while time.time() < t_end:
        it += 1
        if it % 20 == 1:
            active = _active_idx(C, K)
            gp = _grad_sum(h, active)
            gp -= gp.mean()
            order = np.argsort(gp)
            low_idx = order[:lsize]
            high_idx = order[-lsize:]
        if noimp < 4:
            i = int(high_idx[-1])
            j = int(low_idx[0])
        elif noimp < 40:
            i = int(high_idx[rng.randint(len(high_idx))])
            j = int(low_idx[rng.randint(len(low_idx))])
        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        if i == j:
            i = int(rng.randint(n))
            j = (i + 1) % n
        if gp[i] > gp[j]:
            maxf = min(step, max(0.0, h[i]), max(0.0, 1.0 - h[j]))
            maxb = min(0.5 * step, max(0.0, h[j]), max(0.0, 1.0 - h[i]))
        else:
            maxf = min(0.5 * step, max(0.0, h[i]), max(0.0, 1.0 - h[j]))
            maxb = min(step, max(0.0, h[j]), max(0.0, 1.0 - h[i]))
        ts = []
        if maxf > 1e-13:
            ts = [maxf, 0.35 * maxf]
        if maxb > 1e-13:
            if not ts:
                ts = []
            ts.extend((-maxb, -0.35 * maxb))
        if not ts:
            noimp += 1
            if noimp > 60:
                step *= 0.5
                if step < 1e-9:
                    break
            continue
        newc, ok = _pair_scan(h, C, curC, i, j, ts, prod, coef)
        if ok:
            curC = newc
            noimp = 0
            if step < 4e-4:
                step = min(step * 1.05, 4e-4)
        else:
            noimp += 1
            if noimp % 25 == 0:
                step *= 0.55
                if step < 1e-9:
                    break
            if noimp > 200:
                noise = rng.randn(n)
                noise -= noise.mean()
                h2 = np.clip(h + 1e-5 * noise, 0.0, 1.0)
                h2 = _repair(h2, n)
                C2 = _corr(h2)
                c2 = float(C2.max())
                if c2 < curC:
                    h, C, curC = h2, C2, c2
                    noimp = 0
                    step = max(step, 2e-5)
                else:
                    noimp = 0
                    if step < 1e-8:
                        break
        if it % 4000 == 0:
            h = _repair(h, n)
            C = _corr(h)
            curC = float(C.max())
    return h, C, curC


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s)
    rng = np.random.RandomState(seed)
    try:
        h = np.asarray(initial_h_values, dtype=float).ravel().copy()
    except NameError:
        h = _default_h(560, seed)
    n = len(h)
    h = _repair(h, n)

    h, C, curC = _best_from_variants(h, n)
    best_h = h.copy()
    best_C = C.copy()
    bestc = float(curC)

    slsqp_end = t0 + min(240.0, max(90.0, budget_s - 650.0))
    alpha_seq = (
        (1000, 32, 100),
        (5000, 24, 100),
        (20000, 18, 100),
        (100000, 12, 80),
        (500000, 10, 80),
        (2000000, 8, 60),
        (10000000, 6, 50),
    )
    for alpha, m_lags, iters in alpha_seq:
        if time.time() >= slsqp_end:
            break
        hv = _slsqp_run(h, alpha, m_lags, iters, 1e-14)
        Cc = _corr(hv)
        cc = float(Cc.max())
        if cc < curC - 1e-15:
            h, C, curC = hv, Cc, cc
        elif cc < curC + 1e-8:
            h, C, curC = hv, Cc, cc
        if curC < bestc:
            bestc = curC
            best_h = h.copy()
            best_C = C.copy()

    if time.time() < deadline - 120:
        h, C, curC = _best_from_variants(h, n)
        if curC < bestc:
            bestc = curC
            best_h = h.copy()
            best_C = C.copy()
        if curC >= bestc:
            h = best_h.copy()
            C = best_C.copy()
            curC = bestc

    lp_end = t0 + min(500.0, max(60.0, budget_s - 350.0))
    if time.time() < lp_end:
        h, C, curC = _lp_phase(h, C, curC, lp_end)
        if curC < bestc:
            bestc = curC
            best_h = h.copy()
            best_C = C.copy()

    pair_end = deadline - 20.0
    if time.time() < pair_end:
        h, C, curC = _pair_phase(h, C, curC, pair_end, rng)
        if curC < bestc:
            bestc = curC
            best_h = h.copy()
            best_C = C.copy()

    if time.time() < deadline - 5.0:
        h2, C2, c2 = _best_from_variants(h, n)
        if c2 < bestc:
            bestc = c2
            best_h = h2
            best_C = C2

    final_h = _repair(best_h, n)
    final_C = _corr(final_h)
    final_c = float(final_C.max() * (2.0 / n))
    return final_h, final_c, n
