def _exact_2pt(h, i, j, c_cur):
    """Try exact 2-point move and return best new h and c."""
    n = len(h)
    best_d = 0.0
    best_c = c_cur
    # Try a range of d values via bisection on the 1D function
    lo = max(-h[i], h[j] - 1) + 1e-12  # d range: h[i]+d in [0,1], h[j]-d in [0,1]
    hi = min(1.0 - h[i], h[j]) - 1e-12
    if hi <= lo:
        return h, c_cur
    # Evaluate at several points
    for t in np.linspace(-1, 1, 13):
        d = lo + t * (hi - lo) / 2.0
        if abs(d) < 1e-15:
            continue
        h[i] += d
        h[j] -= d
        c = _eval(h)
        h[i] -= d
        h[j] += d
        if c < best_c:
            best_c = c
            best_d = d
    if best_d == 0:
        return h, c_cur
    h[i] += best_d
    h[j] -= best_d
    return h, best_c

def _grad_guided_npt(h, n_iter, rng, m_lags, n_trial):

    n = len(h)

    best_h = h.copy()
    best_c = c5c
    for it in range(n_iter):

        top = np.argsort(c)[-m_lags:]

        dx = 2.0 / n
        for k in top:

            if L >= 0:
                g[L:] += (1.0 - h[:n - L])
                g[:n - L] -= h[L:]
            else:
                M = -L
                g[:n - M] += (1.0 - h[M:])
                g[M:] -= h[:n - M]
        g_proj = g - g.mean()
        # Find i with max g_proj, j with min
        i = int(np.argmax(g_proj))
        j = int(np.argmin(g_proj))
        if i == j:
            continue
        h, c5c = _exact_2pt(h, i, j, c5c)
        if c5c < best_c:
            best_c = c5c
            best_h = h.copy()
        # Also try random pairs for diversity
        if it % n_trial == 0:
            ii = rng.randint(n)
            jj = (ii + 1 + rng.randint(0, 16)) % n
            if ii != jj:
                h, c5c = _exact_2pt(h, ii, jj, c5c)
                if c5c < best_c:
                    best_c = c5c
                    best_h = h.copy()
    return best_h, best_c

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)

def _smooth_obj(x, alpha, m_lags):
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_lags):

    m = c.max()

    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii, k in enumerate(idx):

            continue
