        return np.full(n, 0.5)

        h *= target / s
        h = np.clip(h, 0.0, 1.0)

    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    s = h.sum()

        return np.full(n, 0.5)

        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)

        if abs(s - tgt) < 1e-12:
            break

        for i in od:
            if d > 0:

            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:

def _down(h, f):
    n = len(h)

    o = np.empty(n * f)

def _sa2(h, ni, rng, T0, cool, jit):
    n = len(h)
    c5 = _eval(h)
    bh = h.copy()
    bc = c5
    for it in range(ni):

        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 20)) % n
        d = rng.randn() * jit
        ni_, nj_ = h[i] + d, h[j] - d
        if ni_ < 0 or ni_ > 1 or nj_ < 0 or nj_ > 1:

            if c5n < bc:
                bc = c5n
                bh = h.copy()
        else:

def _sab(h, ni, rng, T0, cool, jit, bs):

def _grd2(h, ni, a, nt):
    n = len(h)
    c5 = _eval(h)
    bh = h.copy()
    bc = c5

        j = int(np.argmin(gp))

            continue

        bc_ = c5

            h[j] -= d

                h[i] -= d
                h[j] += d
                continue

                bc_ = c5n
                bd = d
            else:

                h[j] += d

            if c5 < bc:
                bc = c5
                bh = h.copy()
        else:
            h[i] -= bd
            h[j] += bd
    return bh, bc
