                    else:
                        h[i:i+bs] = t1
                        h[j:j+bs] = t2
        
        if improved:
            stall = 0
            if c < bc:
                bc = c
                bh = h.copy()
        else:
            stall += 1
            T *= 0.995
            if T < 1e-5:
                T = 1e-5
        
        if stall > 300:
            # Escape: random perturbation
            n_esc = rng.randint(3, 10)
            for _ in range(n_esc):
                i = rng.randint(n)
                j = rng.randint(n)
                if i != j:
                    h[i], h[j] = h[j], h[i]
            c = _c5(h)
            stall = 0
            T = 0.003
    
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    
    n0 = len(h0)
    best = _repair(h0, n0)
    best_c = _c5(best)
    
    # Try symmetries and variations
    cands = [best]
    for v in [1.0 - best, best[::-1]]:
        cands.append(_proj(v, n0 * 0.5))
    # Antisymmetric and symmetric parts
    cands.append(_proj(0.5 + 0.5 * (best - best[::-1]), n0 * 0.5))
    cands.append(_proj(0.5 + 0.5 * (best + best[::-1] - 1.0), n0 * 0.5))
    
    for cv in cands:
        cc = _c5(cv)
        if cc < best_c:
            best, best_c = cv, cc
    
    # Phase 1: Main local search at current resolution (600s)
    end1 = t0 + min(600, budget_s * 0.55)
    best, best_c = _local(best, rng, end1)
    
    # Phase 2: Try at half resolution, optimize, then upsample (200s)
    end2 = t0 + min(800, budget_s * 0.73)
    if end2 - time.time() > 30:
        nh = _resample(best, n0 // 2)
        hn, cn = _local(nh, rng, end2 - 20)
        hu = _resample(hn, n0)
