import numpy as np
from scipy.optimize import minimize

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):

    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(corr.max()), corr

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:

        g[L:] += (1.0 - h[:n - L])
        g[:n - L] -= h[L:]
    else:
        M = -L
        g[M:] += (1.0 - h[:n - M])
        g[:n - M] -= h[M:]
    return g * dx

    return x.sum() - half

    s -= s.max()

def _smooth_grad(x, alpha, ml):

    g = np.zeros(n)

        else:
