    return float(np.max(_corr(h)))

def _cd_pass(h, rng):
    n = len(h)
    nm1 = n - 1
    cmax = 2 * n - 1
    c = _corr(h)
    jlist = np.arange(n)
    rng.shuffle(jlist)
    for j in jlist:
        j = int(j)
        beta = np.zeros(cmax)
        k1_lo = j
        k1_hi = min(cmax - 1, j + nm1)
        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax - 1, cmax - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
        beta[k2_lo:k2_hi + 1] -= h[j + lags2]
        g = c - beta * h[j]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(20):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(g + beta * t1) <= np.max(g + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-15:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

def _fw_grad(h, alpha, m_lags):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    idx = np.argsort(c)[-m_lags:]
    c_top = c[idx]
    s = alpha * c_top
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    nm1 = n - 1
    grad = np.zeros(n)
    for ii in range(len(idx)):
        wk = w[ii] * dx
        k = int(idx[ii])
        L = k - nm1
        j1_lo = max(0, -L)
        j1_hi = min(n, n - L)
        if j1_hi > j1_lo:
            grad[j1_lo:j1_hi] += wk * (1.0 - h[j1_lo + L:j1_hi + L])
        j2_lo = max(0, L)
        j2_hi = min(n, n + L)
        if j2_hi > j2_lo:
            grad[j2_lo:j2_hi] -= wk * h[j2_lo - L:j2_hi - L]
    return grad

def _fw_vertex(g, n):
    """argmin_{x in S} <g,x> where S={0<=x<=1, sum(x)=n/2}"""
    # Sort indices by g value
    order = np.argsort(g)
    x = np.zeros(n)
    x[order[:n // 2]] = 1.0
    return x

    best_h = h.copy()
    best_c5 = _eval(best_h)
    for it in range(n_iters):
        if time.time() > t_end:
            break
        g = _fw_grad(h, alpha, m_lags)
        v = _fw_vertex(g, n)
        # Line search for step size
        game = 2.0 / (it + 2)
        h_new = (1 - game) * h + game * v
        h_new = _repair(h_new, n)
        c5_new = _eval(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[i] -= d
                h[j] += d
    return best_h, best_c5
