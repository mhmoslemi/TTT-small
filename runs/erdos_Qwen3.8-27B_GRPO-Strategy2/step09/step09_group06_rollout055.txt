    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _repair(h, n):
    return _proj(np.clip(h, 0.0, 1.0).copy(), n * 0.5)

def _smooth_obj(x, alpha, m, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    cm = c.max()
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - cm)
    s = np.clip(s - s.max(), -500, 500)
    return cm + np.sum(np.exp(s)) / alpha

def _smooth_grad(x, alpha, m, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s - s.max(), -500, 500)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

    target = n * 0.5

        return _smooth_obj(x, alpha, m, n)
    def grad(x):
        return _smooth_grad(x, alpha, m, n)

        return x.sum() - target
    def sjac(x):
        return np.ones(n)

    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except:
        return h

    x = h.copy()
    y = h.copy()

    best_x = h.copy()

        g = _smooth_grad(y, alpha, m, n)
        taken = False

            if _smooth_obj(x_new, alpha, m, n) <= _smooth_obj(y, alpha, m, n):

        t = t_new

            best_x = x.copy()

        for _t in range(14):
            x_new = _proj(y - step * g, target)
            if _smooth_obj(x_new, alpha, m, n) <= _smooth_obj(y, alpha, m, n) + 1e-15:
                taken = True
                break
            step *= 0.4
        if not taken:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        x = x_new
        t = t_new
        step = min(step * 1.04, step0 * 10.0)
        c = _eval(x)
        if c < best_c:
            best_c = c
            best_x = x.copy()
    return _repair(best_x, n), best_c

def _subgrad_max(h, m, n):
    c = _eval_all(h)
    top = np.argsort(c)[-m:]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in top:
        L = k - (n - 1)
        if L >= 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    return g * dx / m

def _local_search(h, n_iters, rng, t0=0.03, t1=0.001):
    n = len(h)
    c5 = _eval(h)
    best_h = h.copy()
    best_c = c5
    for it in range(n_iters):
        frac = it / max(n_iters, 1)
        temp = t0 * (t1 / t0) ** frac
        r = rng.rand()
        if r < 0.35:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 12)) % n
            d = temp * 0.03 * rng.randn()
            a, b = h[i] + d, h[j] - d
            if 0.0 <= a <= 1.0 and 0.0 <= b <= 1.0:
                h[i], h[j] = a, b
                c5n = _eval(h)
                if c5n < c5:
                    c5 = c5n
                    if c5 < best_c:
                        best_c, best_h = c5, h.copy()
                elif rng.rand() < np.exp(-(c5n - c5) / (temp * 0.003)):
                    pass
                else:
                    h[i], h[j] = a - d, b + d
        elif r < 0.65:
            bs = rng.randint(3, 16)
            s = rng.randint(0, n - bs)
            d = temp * 0.08 * rng.randn(bs)
            d -= d.mean()
            old = h[s:s+bs].copy()
            h[s:s+bs] += d
            if h.min() >= -0.1 and h.max() <= 1.1:
                h = _repair(h, n)
                c5n = _eval(h)
                if c5n < c5:
                    c5 = c5n
                    if c5 < best_c:
                        best_c, best_h = c5, h.copy()
                elif rng.rand() < np.exp(-(c5n - c5) / (temp * 0.003)):
                    pass
                else:
                    h[s:s+bs] = old
            else:
                h[s:s+bs] = old
        else:
            m = 5
            g = _subgrad_max(h, m, n)
            gn = np.linalg.norm(g)
            if gn < 1e-12:
                continue
            g = g / gn
            for step in [temp * 0.3, temp * 0.1, temp * 0.03]:
                h_new = _proj(h - step * g, n * 0.5)
                c5n = _eval(h_new)
                if c5n < c5:
                    h = h_new
                    c5 = c5n
                    if c5 < best_c:
                        best_c, best_h = c5, h.copy()
                    break
    return _repair(best_h, n), best_c

def _polish(h, n_iters, rng):
    n = len(h)
    c5 = _eval(h)
    for _ in range(n_iters):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        d = 5e-9 * rng.randn()
        a, b = h[i] + d, h[j] - d
        if 0.0 <= a <= 1.0 and 0.0 <= b <= 1.0:
            h[i], h[j] = a, b
            c5n = _eval(h)
            if c5n < c5:
                c5 = c5n
            else:
                h[i], h[j] = a - d, b + d
    return h, c5

def _gradient_polish(h, n_steps, rng):
    n = len(h)
    c5 = _eval(h)
    step = 0.005
    for _ in range(n_steps):
        c = _eval_all(h)
        m = min(8, n * 2 - 1)
        top = np.argsort(c)[-m:]
        g = np.zeros(n)
        for k in top:
            L = k - (n - 1)
            if L >= 0:
                g[L:] += (1.0 - h[:n - L])
                g[:n - L] -= h[L:]
            else:
                M = -L
                g[:n - M] += (1.0 - h[M:])
                g[M:] -= h[:n - M]
        g *= (2.0 / n) / m
        for s in [step, step * 0.3, step * 0.1]:
            h_new = _proj(h - s * g, n * 0.5)
            c5n = _eval(h_new)
            if c5n < c5 - 1e-15:
                h = h_new
                c5 = c5n
                step = min(step * 1.1, 0.02)
                break
        else:
            step *= 0.6
    return _repair(h, n), c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = _repair(h, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP with smooth max (homotopy)
    alpha_sweep = [
        (5000, 30, 700), (20000, 40, 600), (100000, 50, 500),
        (500000, 60, 450), (2000000, 70, 400), (10000000, 80, 350),
        (50000000, 90, 300), (200000000, 100, 250),
    ]
    for alpha, m, mi in alpha_sweep:
        if _el() > budget_s * 0.28:
            break
        h_new = _run_slsqp(best_h, alpha, m, mi)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 2: FISTA (Nesterov accelerated)
    if _el() < budget_s * 0.45:
        for alpha, m, steps, s0 in [
            (100000, 50, 5000, 0.004),
            (2000000, 60, 4000, 0.002),
            (20000000, 70, 3000, 0.001),
        ]:
            if _el() > budget_s * 0.45:
                break
            h_new, c5n = _run_fista(best_h.copy(), alpha, m, steps, s0)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 3: High-alpha SLSQP
    if _el() < budget_s * 0.55:
        for alpha, m, mi in [(5000000000, 110, 300), (50000000000, 120, 200)]:
            if _el() > budget_s * 0.55:
                break
            h_new = _run_slsqp(best_h, alpha, m, mi)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 4: Gradient polish (subgradient descent on max)
    if _el() < budget_s * 0.62:
        h_new, c5n = _gradient_polish(best_h.copy(), 3000, rng)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 5: Simulated annealing local search
    if _el() < budget_s * 0.82:
        rem = budget_s * 0.82 - _el()
        n_ls = min(int(rem / 0.0018), 400000)
        h_new, c5n = _local_search(best_h.copy(), n_ls, rng, t0=0.02, t1=0.0005)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 6: Denser gradient polish
    if _el() < budget_s * 0.88:
        h_new, c5n = _gradient_polish(best_h.copy(), 2000, rng)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 7: Fine 2-point polish
    if _el() < budget_s - 2:
        rem = budget_s - 2 - _el()
        n_pol = min(int(rem / 0.0001), 5000000)
        h_new, c5n = _polish(best_h.copy(), n_pol, rng)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
