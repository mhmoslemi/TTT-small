    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

    """One step of projected gradient descent on max of c[k]"""
    c = _corr(h)
    kmax = np.argmax(c)
    # gradient of c[kmax] w.r.t. h
    L = kmax - (n - 1)
    grad = np.zeros(n)

            grad[i + L] -= h[i]
    else:

            grad[i] -= h[i + M]

                best_overall_c5 = c5_tr
                best_overall = h_tr.copy()

    # Phase 5: Final intensive CD (8% budget)
    p5_end = time.time() + budget_s * 0.08
    h = best_overall.copy()
    n_fin = len(h)
    while time.time() < p5_end:
        h = _cd_pass(h, rng, rng.randint(4))
        h = _repair(h, n_fin)
        c5 = _eval(h)
        if c5 < best_overall_c5:
            best_overall_c5 = c5
            best_overall = h.copy()

    # Phase 6: Ultra-fine local search (remaining time)
    if time.time() < t0 + budget_s - 2:
        h = best_overall.copy()
        n_curr = len(h)
        c5_curr = best_overall_c5
        time_avail = t0 + budget_s - 3 - time.time()
        n_iter = int(time_avail * 200)
        for _ in range(min(n_iter, 120000)):
            i = rng.randint(n_curr)
            j = (i + 1 + rng.randint(0, 6)) % n_curr
            d = 3e-6 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5_curr - 1e-14:
                    c5_curr = c5n
                    if c5n < best_overall_c5:
                        best_overall_c5 = c5n
                        best_overall = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_overall = _repair(best_overall, len(best_overall))
    best_overall_c5 = _eval(best_overall)
    return (best_overall, best_overall_c5, len(best_overall))
