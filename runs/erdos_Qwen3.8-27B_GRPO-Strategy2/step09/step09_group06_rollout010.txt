import numpy as np

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _full_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    dx = 2.0 / n
    g = np.zeros(n)
    if L >= 0:
        g[:n - L] += dx * (1.0 - h[L:])
        g[L:] -= dx * h[:n - L]
    else:
        M = -L
        g[M:] += dx * (1.0 - h[:n - M])
        g[:n - M] -= dx * h[M:]
    return g

def _grad_smooth(h, alpha, m):
    n = len(h)
    c = _full_corr(h)
    top_m = np.argsort(c)[-m:]
    cm = c[top_m].max()
    s = alpha * (c[top_m] - cm)
    s = np.clip(s, -500, 500)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)

        if w[ii] < 1e-15:
            continue
        L = k_idx - (n - 1)
        g += w[ii] * _grad_lag(h, L, n)
    return g

def _obj_slsqp(x, alpha, m, n):
    c = _full_corr(x)
    top = np.sort(c)[-m:]
    s = alpha * (top - top.max())
    s = np.clip(s, -500, 500)
    return top.max() + np.log(np.exp(s).sum()) / alpha

def _grad_slsqp(x, alpha, m, n):
    return _grad_smooth(x, alpha, m)

    return x.sum() - target

    return np.ones(len(x))

def _slsqp_opt(h, alpha, m, maxiter, n):

                       options={'maxiter': maxiter, 'ftol': 1e-20})
