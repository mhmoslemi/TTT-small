def _repair(h, n):

    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()

            break

        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break

def _to_binary(h, n, rng):

    n_ones = int(b.sum())
    if n_ones > target:

        at_thresh = np.where(b == 1)[0]
        rng.shuffle(at_thresh)

            b[idx] = 0
    elif n_ones < target:

        at_thresh = at_thresh[np.argsort(-h[at_thresh])]
        for idx in at_thresh[:target - n_ones]:

    target = n // 2
    ones = np.where(b == 1)[0]

        # Try a subset of swaps for efficiency
        n_trials = min(len(ones) * len(zeros), 20000)

        j_idx = rng.randint(0, len(zeros), n_trials)

            b[i] = 0
            b[j] = 1
            v = _eval(b)
            b[i] = 1
            b[j] = 0
            if v < best_val:
                best_val = v
                best_swap = (i, j)
            # Aspirin rule: accept if improves global best
            elif v < best_c5 and best_swap is None:
                best_swap = (i, j)
                best_val = v
        
        if best_swap is None:
            break
        
        i, j = best_swap
        b[i] = 0
        b[j] = 1
        # Update ones/zeros sets
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        # Add to tabu
        tabu[(min(i,j), max(i,j))] = it + tabu_len
        # Remove old tabu entries
        if len(tabu) > 500:
            tabu = {k: v for k, v in tabu.items() if v > it}
        
        cur_c5 = best_val
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_b = b.copy()
        it += 1
    
    return best_b, best_c5

    ones_set = set(np.where(b == 1)[0].tolist())
    zeros_set = set(np.where(b == 0)[0].tolist())
    ones = np.array(list(ones_set))
    zeros = np.array(list(zeros_set))

    best_c5 = cur_c5
    tabu = {}
    no_improve = 0
    
    for it in range(n_iters):
        # Sample candidate swaps
        n_t = min(n_trials, len(ones) * len(zeros))
        i_idx = rng.randint(0, len(ones), n_t)
        j_idx = rng.randint(0, len(zeros), n_t)

        best_val = cur_c5 + 0.01  # Allow slight worsening (aspiration)
        best_asp = None
        best_asp_val = cur_c5

            j = zeros[j_idx[t]]
            key = (min(int(i), int(j)), max(int(i), int(j)))
            if key in tabu and tabu[key] > it:
                continue
            b[i] = 0; b[j] = 1
            v = _eval(b)
            b[i] = 1; b[j] = 0
            if v < best_val:
                best_val = v
                best_swap = (i, j)
            if v < best_asp_val:
                best_asp_val = v
                best_asp = (i, j)

            if best_asp is not None:
                best_swap = best_asp
                best_val = best_asp_val
            else:
                break
        
        i, j = best_swap
        b[i] = 0; b[j] = 1
        ones_set.discard(int(i)); ones_set.add(int(j))
        zeros_set.discard(int(j)); zeros_set.add(int(i))
        ones = np.array(list(ones_set))
        zeros = np.array(list(zeros_set))

            best_b = b.copy()
