- If s > 0, decrease t; if s < 0, increase t; if s = 0, we're at optimum

            best_t = t_test

    h[j] = s - best_t

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for ii in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[ii])
                h[ii] += a
                diff -= a
            else:
                a = min(-diff, h[ii])
                h[ii] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    idx = np.argsort(c)[-mlags:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, n, mlags):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    idx = np.argsort(c)[-mlags:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)

            continue

            g[:n-L] -= wk * x[L:]
        else:
            M = -L
            g[:n-M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n-M]

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for ii in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[ii])
                h[ii] += a; diff -= a
            else:
                a = min(-diff, h[ii])
                h[ii] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _s_obj(x, alpha, n, ml):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _s_grad(x, alpha, n, ml):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-14:
            continue
        wk = w[ii] * (2.0 / n)
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sf(x, tgt):
    return x.sum() - tgt

def _sj(x, tgt):
    return np.ones(len(x))

def _slsqp(h, alpha, ml, nit, ftol, n):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (n * 0.5,)}]
    try:
        res = minimize(_s_obj, h, jac=_s_grad, args=(alpha, n, ml), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': nit, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _pair_opt(h, i, j, n):
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    ti = h[i]
    if hi - lo < 1e-14:
        return _eval(h)
    best_t, best_v = ti, _eval(h)
    for t in np.linspace(lo, hi, 15):
        h[i] = t; h[j] = s - t
        v = _eval(h)
        if v < best_v:
            best_v = v; best_t = t
    lo2 = max(lo, best_t - (hi - lo) / 14.0)
    hi2 = min(hi, best_t + (hi - lo) / 14.0)
    for t in np.linspace(lo2, hi2, 15):
        h[i] = t; h[j] = s - t
        v = _eval(h)
        if v < best_v:
            best_v = v; best_t = t
    h[i] = best_t; h[j] = s - best_t
    return best_v

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = np.array(initial_h_values, dtype=float).copy()

        if time.time() - t0 - last_imp > 45:
