def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - t) < 1e-14:
            break
    d = t - h.sum()
    if abs(d) > 1e-14:
        ordr = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in ordr:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-14:
                break
    return h

def _sobj(x, alpha, ml):
    n = len(x); dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max(); idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m); s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sgrad(x, alpha, ml):

    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max(); idx = np.argsort(c)[-ml:]

    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15: continue
        wk = w[ii] * dx; L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]

    return x.sum() - t

    return np.ones(len(x))

                     options={'maxiter': mi, 'ftol': 1e-16})
