So we want infl[i,k*] - infl[j,k*] ≤ 0, i.e., infl[j,k*] ≥ infl[i,k*].

This means: we want to move mass from position i (where it hurts the active lag) to position j (where it helps).

So the optimal move is:
  i = argmax_k* infl[., k*]  (position most harmful to active lag)
  j = argmin_k* infl[., k*]  (position most helpful to active lag)

But we need to check all lags, not just the active one. Let me just use the gradient-guided approach.

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0, 1).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0, 1)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(_corr(h).max())

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sm_obj(x, alpha, mags):
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-mags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sm_grad(x, alpha, mags):
    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-mags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15: continue
        wk = w[ii] * (2.0 / n)
        k = idx[ii]
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n-L])
            g[:n-L] -= wk * x[L:]
        else:
            M = -L
            g[:n-M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n-M]
    return g

def _sum_f(x, hn):
    return x.sum() - hn

def _sum_j(x, hn):
    return np.ones(len(x))

def _slsqp(h, alpha, mags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_f, 'jac': _sum_j, 'args': (n*0.5,)}]
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, args=(alpha, mags),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except:
        return h

def _infl(h):
    n = len(h)
    scale = 2.0 / n
    k = np.arange(2 * n - 1)
    L = (k - (n - 1))[None, :]
    I = np.arange(n)[:, None]
    iL = I - L
    iL2 = I + L
    m1 = (iL >= 0) & (iL <= n - 1)
    m2 = (iL2 >= 0) & (iL2 <= n - 1)
    h_iL = np.where(m1, h[np.clip(iL, 0, n-1)], 0.0)
    h_iL2 = np.where(m2, h[np.clip(iL2, 0, n-1)], 0.0)
    return scale * ((1.0 - h_iL) * m1 - h_iL2 * m2)

def _accel_grad(h, alpha, mags, steps, step0):
    n = len(h)
    v = np.zeros(n); sq = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    for _ in range(steps):
        g = _sm_grad(h, alpha, mags)
        v = 0.9 * v + 0.1 * g
        sq = 0.95 * sq + 0.05 * g * g
        d = v / (np.sqrt(sq) + 1e-8)
        taken = False
        for _t in range(14):
            h_new = _proj(h - step * d, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:

                step = min(step * 1.1, step0 * 50)
                taken = True; break
            step *= 0.4
        if not taken:

    return _repair(h, n)

                    cur_c = float(nc.max()); c = nc; break
    if cur_c < best_c:
        best_c = cur_c; best_h = h.copy()
    return best_h, best_c

def _grad2pt(h, n, max_it, alpha, mags):
    c5c = _eval(h)
    best_h = h.copy(); best_c = c5c
    for _ in range(max_it):
        g = _sm_grad(h, alpha, mags)
        gp = g - g.mean()
        i = int(np.argmax(gp)); j = int(np.argmax(-gp))
        if i == j: continue
        best_d = 0; best_c5 = c5c
        for frac in [0.05, 0.2, 0.5]:

                if not (0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1):

                h[i] += d; h[j] -= d

                if c5n < best_c5:

                else:

            if c5c < best_c:
                best_c = c5c; best_h = h.copy()
        else:
            h[i] -= best_d; h[j] += best_d
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0, 1).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0, 1)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(_corr(h).max())

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sm_obj(x, alpha, mags):
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-mags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sm_grad(x, alpha, mags):
    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-mags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * (2.0 / n)
        L = idx[ii] - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_f(x, hn):
    return x.sum() - hn

def _sum_j(x, hn):
    return np.ones(len(x))

def _slsqp(h, alpha, mags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_f, 'jac': _sum_j, 'args': (n * 0.5,)}]
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, args=(alpha, mags),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _infl(h):
    n = len(h)
    scale = 2.0 / n
    k = np.arange(2 * n - 1)
    L = (k - (n - 1))[None, :]
    I = np.arange(n)[:, None]
    iL = I - L
    iL2 = I + L
    m1 = (iL >= 0) & (iL <= n - 1)
    m2 = (iL2 >= 0) & (iL2 <= n - 1)
    h_iL = np.where(m1, h[np.clip(iL, 0, n - 1)], 0.0)
    h_iL2 = np.where(m2, h[np.clip(iL2, 0, n - 1)], 0.0)
    return scale * ((1.0 - h_iL) * m1 - h_iL2 * m2)

def _accel_grad(h, alpha, mags, steps, step0):
    n = len(h)
    v = np.zeros(n); sq = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    for _ in range(steps):
        g = _sm_grad(h, alpha, mags)
        v = 0.88 * v + 0.12 * g
        sq = 0.95 * sq + 0.05 * g * g
        d = v / (np.sqrt(sq) + 1e-8)
        taken = False
        for _t in range(16):
            h_new = _proj(h - step * d, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:
                h = h_new; cur_c = c_new
                step = min(step * 1.08, step0 * 60.0)
                taken = True
                break
            step *= 0.38
        if not taken:
            step = min(step * 1.25, step0 * 60.0)
    return _repair(h, n)

def _fine2(h, n, max_it, rng, recompute=3000, stall_limit=50000):
    infl = _infl(h)
    c = _corr(h)
    best_h = h.copy()
    best_c = float(c.max())
    cur_c = best_c
    last_improve = 0
    st = 0
    for it in range(max_it):
        if it % recompute == 0:
            infl = _infl(h)
            c = _corr(h)
            cur_c = float(c.max())
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 15)) % n
        if i == j:
            continue
        diff = infl[i] - infl[j]
        found = False
        for d in [4e-6, 1.5e-6, 6e-7, 2e-7, 8e-8]:
            nc = c + d * diff
            if nc.max() < cur_c:
                if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
                    h[i] += d; h[j] -= d
                    c = nc; cur_c = float(nc.max())
                    found = True
                    break
            nc = c - d * diff
            if nc.max() < cur_c:
                if 0 <= h[i] - d <= 1 and 0 <= h[j] + d <= 1:
                    h[i] -= d; h[j] += d
                    c = nc; cur_c = float(nc.max())
                    found = True
                    break
        if found:
            st = 0
        else:
            st += 1
        if st > stall_limit:
            h = _repair(h + 1e-5 * rng.randn(n), n) - h.mean() + (n * 0.5) / n
            h = _repair(h, n)
            infl = _infl(h); c = _corr(h); cur_c = float(c.max())
            st = 0
    if cur_c < best_c:
        best_c = cur_c; best_h = h.copy()
    return best_h, best_c

def _binary_search(h, n, max_swaps, rng):
    hb = np.zeros(n)
    idx = np.argsort(h)[-n // 2:]
    hb[idx] = 1.0
    infl = _infl(hb)
    c = _corr(hb)
    best_c = float(c.max())
    best_hb = hb.copy()
    ones = np.where(hb == 1)[0]
    zeros = np.where(hb == 0)[0]
    for it in range(max_swaps):
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        dc = infl[j] - infl[i]
        nc = c + dc
        nc_max = float(nc.max())
        if nc_max < best_c:
            hb[i] = 0; hb[j] = 1
            c = nc; best_c = nc_max
            best_hb = hb.copy()
            ones = np.where(hb == 1)[0]
            zeros = np.where(hb == 0)[0]
            infl = _infl(hb)
    return best_hb, best_c

def _grad2pt(h, n, max_it, alpha, mags):
    c5c = _eval(h)
    best_h = h.copy(); best_c = c5c
    for _ in range(max_it):
        g = _sm_grad(h, alpha, mags)
        gp = g - g.mean()
        i = int(np.argmax(gp)); j = int(np.argmax(-gp))
        if i == j:
            continue
        found = False
        for frac in [0.04, 0.15, 0.4]:
            for sgn in [1, -1]:
                d = sgn * frac * (gp[i] - gp[j]) * 0.01
                if not (0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1):
                    continue
                h[i] += d; h[j] -= d
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    found = True
                    break
                else:
                    h[i] -= d; h[j] += d
                    break
            if found:
                break
        if c5c < best_c:
            best_c = c5c; best_h = h.copy()
    return best_h, best_c
