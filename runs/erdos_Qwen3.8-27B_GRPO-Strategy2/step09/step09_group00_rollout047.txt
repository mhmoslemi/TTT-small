        c = m.sum()
        if c == 0:

        x[m] += d / c
    return x

    g = np.zeros(n)

            g += (1.0 - 2.0 * h)

            g[k:] -= h[:n-k]

import numpy as np
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h))) * (2.0 / n)

def _proj(h, S):
    x = np.clip(h, 0.0, 1.0).copy()
    for _ in range(80):
        d = S - x.sum()
        if abs(d) < 1e-14:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _coord_sweep(h, S, rng, max_sweeps, time_limit, n_cand=60):
    n = len(h)
    h = h.copy()
    corr = _corr(h)
    best_c = _c5(h)
    best_h = h.copy()
    t0 = time.time()
    
    for sv in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        order = rng.permutation(n)
        for i in order:
            hi = h[i]
            m_idx = np.arange(2 * n - 1)
            p = i + m_idx - (n - 1)
            q = i - m_idx + (n - 1)
            mask_p = (p >= 0) & (p < n)
            mask_q = (q >= 0) & (q < n)
            slope = np.zeros(2 * n - 1)
            slope[mask_p] = 1.0 - h[p[mask_p]]
            slope[mask_q] -= h[q[mask_q]]
            
            cands = np.linspace(0.0, 1.0, n_cand)
            deltas = cands - hi
            vals = corr[None, :] + deltas[:, None] * slope[None, :]
            vals[:, n - 1] = corr[n - 1] + cands * (1.0 - cands) - hi * (1.0 - hi)
            mx_vals = vals.max(axis=1)
            bi = int(np.argmin(mx_vals))
            bv = cands[bi]
            
            if bv != hi and mx_vals[bi] < np.max(corr) - 1e-12:
                lo = max(0.0, bv - 0.1 / (n_cand - 1))
                hi_bound = min(1.0, bv + 0.1 / (n_cand - 1))
                cands2 = np.linspace(lo, hi_bound, 40)
                d2 = cands2 - hi
                v2 = corr[None, :] + d2[:, None] * slope[None, :]
                v2[:, n - 1] = corr[n - 1] + cands2 * (1.0 - cands2) - hi * (1.0 - hi)
                mx2 = v2.max(axis=1)
                bi2 = int(np.argmin(mx2))
                if mx2[bi2] < mx_vals[bi]:
                    bv = cands2[bi2]
                h[i] = bv
                corr = _corr(h)
        
        h = _proj(h, S)
        c = _c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def _subgrad_step(h, rng, deadline, S, n_active=30):
    n = len(h)
    h = h.copy()
    corr = _corr(h)
    c_cur = float(corr.max()) * (2.0 / n)
    best_c = c_cur
    best_h = h.copy()
    t0 = time.time()
    mom = np.zeros(n)
    beta = 0.6
    step = 0.4
    stale = 0
    
    while time.time() < deadline:
        idx_s = np.argsort(corr)[::-1]
        active_k = sorted(set(int(i - (n - 1)) for i in idx_s[:n_active]))
        
        g = np.zeros(n)
        for k in active_k:
            if k == 0:
                g += 1.0 - 2.0 * h
            else:
                g[:n - k] += 1.0 - h[k:]
                g[k:] -= h[:n - k]
        g /= max(len(active_k), 1)
        
        mom = beta * mom - step * g
        h_new = _proj(h + mom, S)
        corr_new = _corr(h_new)
        c_new = float(corr_new.max()) * (2.0 / n)
        
        if c_new < c_cur - 1e-15:
            h = h_new
            corr = corr_new
            c_cur = c_new
            stale = 0
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
        else:
            stale += 1
            if stale > 30:
                step *= 0.6
                mom *= 0.3
                if step < 0.005:
                    break
                stale = 0
    return best_h, best_c

                            best_h = h.copy()
                    else:
                        h[i] -= d
                        h[j] += d
                    continue
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = (rng.random() - 0.5) * 0.04
        h[i] += d
        h[j] -= d
        h = np.clip(h, 0, 1)
        corr2 = _corr(h)
        mx2 = float(corr2.max())
        if mx2 - mx < -1e-14 or (mx2 > mx and rng.random() < np.exp(-(mx2 - mx) / max(T, 1e-15))):
            mx = mx2
            corr = corr2
            if mx * (2.0 / n) < best_c:
                best_c = mx * (2.0 / n)
                best_h = h.copy()
        else:
            h[i] -= d
            h[j] += d
            h = np.clip(h, 0, 1)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    S = n0 * 0.5
    h0 = _proj(h0, S)
    
    best = h0.copy()
    best_c = _c5(best)
    
    for hh in (1.0 - h0, h0[::-1]):
        h2 = _proj(hh, S)
        c = _c5(h2)
        if c < best_c:
            best = h2.copy()
            best_c = c
    
    # Phase 1: Subgradient warmup (60s)
    end1 = t0 + min(80.0, 0.08 * budget_s)
    if end1 > time.time() + 3:
        best, best_c = _subgrad_step(best, rng, end1, S, n_active=50)
    
    # Phase 2: Coordinate descent main (500s)
    end2 = t0 + min(580.0, 0.58 * budget_s)
    if end2 > time.time() + 5:
        best, best_c = _coord_sweep(best, S, rng, 999999, end2 - time.time(), n_cand=80)
    
    # Phase 3: Subgradient for escape (80s)
    end3 = t0 + min(680.0, 0.68 * budget_s)
    if end3 > time.time() + 3:
        bh, bc = _subgrad_step(best, rng, end3, S, n_active=15)
        if bc < best_c:
            best = bh
            best_c = bc
    
    # Phase 4: More coordinate descent (150s)
    end4 = t0 + min(850.0, 0.85 * budget_s)
    if end4 > time.time() + 5:
        best, best_c = _coord_sweep(best, S, rng, 999999, end4 - time.time(), n_cand=100)
    
    # Phase 5: SA polish (100s)
    end5 = t0 + min(960.0, 0.96 * budget_s)
    if end5 > time.time() + 3:
        best, best_c = _sa_polish(best, rng, end5)
    
    # Phase 6: Final coordinate refinement
    end6 = t0 + budget_s - 3.0
    if end6 > time.time() + 3:
        best, best_c = _coord_sweep(best, S, rng, 999999, end6 - time.time(), n_cand=120)
    
    # Final variants
    for hh in (1.0 - best, best[::-1]):
        h2 = _proj(hh, S)
        c = _c5(h2)
        if c < best_c:
            best = h2
            best_c = c
    
    best = _proj(best, S)
    best_c = _c5(best)
    return best, best_c, n0
