    idx = np.argsort(h)[::-1][:n // 2]
    out = np.zeros(n)
    out[idx] = 1.0
    return out

def _score_vec(h, lags):
    n = len(h)
    s = np.zeros(n)
    for k in lags:
        k = abs(int(k))
        if k == 0 or k >= n:
            continue
        s[:n - k] += h[k:]
        s[k:] += h[:n - k]
    return s

def _bin_search(h0, rng, deadline, top_m=5):
    n = len(h0)
    h = _to_binary(h0, n)
    c = _c5(h)
    best_c = c
    best_h = h.copy()
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    stall = 0
    max_stall = 2500
    it = 0
    ov = _c5_vec(h)
    s = np.zeros(n)
    lags = [0]
    temp = 0.002
    while time.time() < deadline - 0.5:
        if it % 100 == 0:
            ov = _c5_vec(h)
            top_idx = np.argsort(ov)[::-1][:top_m]
            lags = (top_idx - (n - 1)).tolist()
            s = _score_vec(h, lags)
        if rng.random() < 0.82:
            n_cand = max(15, n // 10)
            si = np.array([s[i] for i in ones])
            sj = np.array([s[j] for j in zeros])
            ti = np.argsort(-si)[:n_cand]
            bj = np.argsort(sj)[:n_cand]
            i = ones[ti[rng.randint(n_cand)]]
            j = zeros[bj[rng.randint(n_cand)]]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        if i == j:
            it += 1
            continue
        h[i] = 0.0
        h[j] = 1.0
        cn = _c5(h)
        if cn < c - 1e-15:
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            c = cn
            stall = 0
            if cn < best_c:
                best_c = cn
                best_h = h.copy()
        elif cn - c < temp:
            ones.remove(i); ones.append(j)
            zeros.remove(j); zeros.append(i)
            c = cn
            stall += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
        temp *= 0.9997
        if stall > max_stall:
            h[:] = best_h
            c = best_c
            ones = list(np.where(h > 0.5)[0])
            zeros = list(np.where(h <= 0.5)[0])
            for _ in range(rng.randint(4, 18)):
                ii = ones[rng.randint(len(ones))]
                jj = zeros[rng.randint(len(zeros))]
                if ii != jj:
                    h[ii] = 0.0; h[jj] = 1.0
                    ones.remove(ii); ones.append(jj)
                    zeros.remove(jj); zeros.append(ii)
            c = _c5(h)
            stall = 0
            max_stall = min(25000, max_stall + 2000)
        it += 1
    return best_h, best_c

def _cont_grad(h0, deadline, lr0=0.02):
    n = len(h0)
    S = n * 0.5
    h = _proj(h0.astype(float), S)
    c = _c5(h)
    best_c = c
    best_h = h.copy()
    lr = lr0
    stale = 0
    while time.time() < deadline - 0.5:
        ov = _c5_vec(h)
        mx = ov.max()
        if mx < 1e-12:
            break
        thresh = mx - 0.04 * mx
        active = np.where(ov >= thresh)[0]
        lags = (active - (n - 1)).tolist()
        g = np.zeros(n)
        for k in lags:
            kk = abs(int(k))
            if kk == 0:
                g += 1.0 - 2.0 * h
                continue
            if kk < n:
                g[:n - kk] += (1.0 - h[kk:])
                g[kk:] += (-h[:n - kk])
        g /= max(1, len(lags))
        h_new = _proj(h - lr * g, S)
        cn = _c5(h_new)
        if cn < c - 1e-15:
            h = h_new
            c = cn
            lr = min(lr * 1.05, 0.08)
            stale = 0
            if cn < best_c:
                best_c = cn
                best_h = h.copy()
        else:
            lr *= 0.6
            stale += 1
            if stale > 30 or lr < 1e-9:
                break
    return best_h, best_c

def _two_swap(h, ones, zeros, rng, n):
    h_orig = h.copy()
    i1 = ones[rng.randint(len(ones))]
    i2 = ones[rng.randint(len(ones))]
    j1 = zeros[rng.randint(len(zeros))]
    j2 = zeros[rng.randint(len(zeros))]
    if i1 == i2 or j1 == j2 or i1 == j1 or i1 == j2 or i2 == j1 or i2 == j2:
        return False
    h[i1] = 0.0; h[i2] = 0.0; h[j1] = 1.0; h[j2] = 1.0
    cn = _c5(h)
    return cn, (i1, j1, i2, j2), h_orig

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n = len(h0)
    S = n * 0.5
    h0 = _proj(h0, S)
    best_h = h0.copy()
    best_c = _c5(best_h)

    end1 = t0 + 50.0
    hc, cc = _cont_grad(h0, end1)
    if cc < best_c:
        best_h, best_c = hc, cc

    end2 = t0 + 500.0
    hb1 = _to_binary(hc, n)
    h_b1, c_b1 = _bin_search(hb1, rng, end2, top_m=5)
    if c_b1 < best_c:
        best_h, best_c = h_b1, c_b1

    end3 = t0 + 800.0
    hb2 = _to_binary(h0, n)
    h_b2, c_b2 = _bin_search(hb2, rng, end3, top_m=7)
    if c_b2 < best_c:
        best_h, best_c = h_b2, c_b2

    end4 = t0 + budget_s - 30.0
    hr, cr = _cont_grad(best_h, end4, lr0=0.01)
    if cr < best_c:
        best_h, best_c = hr, cr

    hb_f = _to_binary(best_h, n)
    cb_f = _c5(hb_f)
    if cb_f < best_c:
        best_h, best_c = hb_f, cb_f

    best_h = _proj(best_h, S)
    best_c = _c5(best_h)
    return best_h, best_c, n
