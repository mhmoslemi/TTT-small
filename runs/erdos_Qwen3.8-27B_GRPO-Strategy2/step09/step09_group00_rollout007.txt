def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _rep(h, n):
    S = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(60):
        h *= S / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - S) < 1e-14:
            break
    d = S - h.sum()
    if abs(d) > 1e-12:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            a = min(abs(d), 1.0 - h[i]) if d > 0 else min(-d, h[i])
            h[i] += a if d > 0 else -a
            d -= a if d > 0 else -a
            if abs(d) < 1e-13:
                break
    return h

def _prj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(60):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x
