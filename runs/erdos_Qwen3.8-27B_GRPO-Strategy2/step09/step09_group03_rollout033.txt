import numpy as np
from scipy.optimize import minimize
import time

def _clip_sum(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}"""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        h *= target / s if s > 0 else 0.5
        h = np.clip(h, 0.0, 1.0)
    # Fix residual
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(-h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _full_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _smooth_obj(x, alpha, ml):
    c = _full_corr(x)
    m = c.max()
    idx = np.argpartition(-c, ml)[:ml]
    s = alpha * (c[idx] - m)
    s = np.clip(s, -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    cx = c * dx
    m = cx.max()
    idx = np.argpartition(-cx, ml)[:ml]
    s = alpha * (cx[idx] - m)
    s = np.clip(s, -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _slsqp(h, alpha, ml, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    half = n * 0.5
    cons = [{'type': 'eq',
             'fun': lambda x: x.sum() - half,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_obj, h0=h, jac=_smooth_grad,
                       args=(alpha, ml), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 50})
        return _clip_sum(res.x, half)
    except Exception:
        return h

def _clip_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        s = h.sum()
        if s < 1e-15:
            return np.full(len(h), target/len(h))
        h = np.clip(h * target/s, 0.0, 1.0)
        if abs(h.sum()-target) < 1e-13:
            break
    d = target - h.sum()
    if abs(d) > 1e-13:
        idx = np.argsort(-h) if d > 0 else np.argsort(h)

                a = min(d, 1.0-h[i]); h[i] += a; d -= a
            else:

    cc = _c5(h); bc = cc; bh = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def _clip_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        s = h.sum()
        if s < 1e-15:
            return np.full(len(h), target / len(h))
        h = np.clip(h * (target / s), 0.0, 1.0)
        if abs(h.sum() - target) < 1e-13:
            break
    d = target - h.sum()
    if abs(d) > 1e-13:
        idx = np.argsort(-h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-14:
                break
    return h

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _obj(x, al, ml):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    ix = np.argpartition(-c, ml)[:ml]
    s = np.clip(al * (c[ix] - m), -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / al

def _grad(x, al, ml):
    n = len(x)
    dx = 2.0 / n
    cr = np.correlate(x, 1.0 - x, mode='full')
    mx = cr.max() * dx
    ix = np.argpartition(-cr, ml)[:ml]
    s = np.clip(al * (cr[ix] * dx - mx), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(ix):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sf(x, hl):
    return x.sum() - hl

def _sj(x, hl):
    return np.ones(len(x))

def _slsqp(h, al, ml, mx, ft):
    n = len(h)
    hl = n * 0.5
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (hl,)}]
    try:
        r = minimize(_obj, h, jac=_grad, args=(al, ml), method='SLSQP',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': mx, 'ftol': ft, 'maxls': 50})
        return _clip_sum(r.x, hl)
    except Exception:
        return h

def _proj(h, tg):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > tg:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

        else:
            i = rng.randint(n)
            j = rng.randint(n)
        if i == j:
            j = (j + 1) % n
        d = sc * rng.randn()
        ho, jo = h[i], h[j]
        h[i] = ho + d
        h[j] = jo - d
        if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
            cn = _c5(h)
            if cn < cc:
                cc = cn
                gc = _grad(h, alpha, ml)
                if cn < bc:
                    bc = cn
                    bh = h.copy()
            else:
                h[i], h[j] = ho, jo
        else:
            h[i], h[j] = ho, jo
    return bh, bc

def _micro(h, niter, rng):
    n = len(h)
    cc = _c5(h)
    bc = cc
    bh = h.copy()
    for it in range(niter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 15)) % n
        d = 1.5e-7 * rng.randn()
        ho, jo = h[i], h[j]
        ni, nj = ho + d, jo - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            cn = _c5(h)
            if cn < cc:
                cc = cn
                if cn < bc:
                    bc = cn
                    bh = h.copy()
            else:
                h[i], h[j] = ho, jo
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _clip_sum(h0, n * 0.5)
    bh = h.copy()
    bc = _c5(bh)

    for ratio in [4, 2]:
        if time.time() - t0 > 40:
            break
        hc = _downsample(h, ratio)
        for ac in [5e4, 5e5, 5e6]:
            if time.time() - t0 > 55:
                break
            hc = _slsqp(hc, ac, 30, 600, 1e-14)
        hu = _upsample(hc, ratio)
        if _c5(hu) < bc:
            bc = _c5(hu)
            bh = hu.copy()

    alpha_seq = [5e4, 2e5, 1e6, 5e6, 2e7, 1e8, 5e8, 2e9]
    for a in alpha_seq:
        if time.time() - t0 > 330:
            break
        ml = min(100, max(20, int(30 * np.log10(a / 5e4) + 20)))
        hn = _slsqp(bh, a, ml, 3000, 1e-16)
        cn = _c5(hn)
        if cn < bc:
            bc = cn
            bh = hn.copy()

    if time.time() - t0 < budget_s - 400:
        pgs = [(5e6, 60, 1000, 0.005, 0.85),
               (5e7, 80, 800, 0.003, 0.90),
               (5e8, 100, 500, 0.002, 0.92),
               (5e9, 120, 400, 0.001, 0.94)]
        for a, ml, st, lr0, mo in pgs:
            if time.time() - t0 > budget_s - 280:
                break
            hn, cn = _npg(bh.copy(), a, ml, st, lr0, mo)
            if cn < bc:
                bc = cn
                bh = hn.copy()

    if time.time() - t0 < budget_s - 180:
        for a in [5e9, 2e10, 1e11]:
            if time.time() - t0 > budget_s - 80:
                break
            hn = _slsqp(bh, a, 200, 2500, 1e-18)
            cn = _c5(hn)
            if cn < bc:
                bc = cn
                bh = hn.copy()

    if time.time() - t0 < budget_s - 120:
        for att in range(10):
            if time.time() - t0 > budget_s - 60:
                break
            ns = 4e-5 * (10 - att)
            hp = _clip_sum(bh + ns * rng.randn(n), n * 0.5)
            hn = _slsqp(hp, 5e7, 100, 600, 1e-15)
            cn = _c5(hn)
            if cn < bc:
                bc = cn
                bh = hn.copy()

    if time.time() - t0 < budget_s - 55:
        hc2 = _downsample(bh, 2)
        for ac in [1e8, 1e9]:
            if time.time() - t0 > budget_s - 30:
                break
            hc2 = _slsqp(hc2, ac, 60, 800, 1e-15)
        hu2 = _upsample(hc2, 2)
        if _c5(hu2) < bc:
            bc = _c5(hu2)
            bh = hu2.copy()

    if time.time() - t0 < budget_s - 10:
        rem = budget_s - 10 - (time.time() - t0)
        nt = max(50000, int(rem / 0.001))
        bh, bc = _glocal(bh, nt, rng, 2e8, 150)

    if time.time() - t0 < budget_s - 1:
        rem = budget_s - 1 - (time.time() - t0)
        nt = max(50000, int(rem / 0.0005))
        bh, bc = _micro(bh, nt, rng)

    bh = _clip_sum(bh, n)
    bc = _c5(bh)
    return (bh, bc, n)
