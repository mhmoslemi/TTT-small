So Q = Σ_k w_k * dx * (A_0 + A_k + A_{-k}) (roughly). For k=0: A_0 = I (identity). For k>0: A_k is a banded matrix.

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _eval_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _obj_fn(x, alpha, m, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s - s.max(), -500, 500)
    return c.max() + np.log(np.exp(s).sum()) / alpha

def _grad_fn(x, alpha, m, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s - s.max(), -500, 500)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fn(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_obj_fn, h, jac=_grad_fn, args=(alpha, m_lags, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _targeted_opt(h, n_rounds, n_per_round, rng):
    """Actively reduce top lags by targeted mass transfers."""
    n = len(h)
    best_h = h.copy()
    best_c = _eval(best_h)
    for round in range(n_rounds):
        c = _eval_full(best_h)
        cmax = c.max()
        # Find lags within 1e-5 of max
        top_lags = np.where(c > cmax - 2e-5)[0]
        if len(top_lags) == 0:
            top_lags = np.argsort(c)[-5:]
        for attempt in range(n_per_round):
            k = top_lags[rng.randint(len(top_lags))]
            L = k - (n - 1)
            if L >= 0:
                contrib = best_h[:n-L] * (1.0 - best_h[L:])
                rng_order = np.argsort(contrib)[::-1][:5]
                for pos in rng_order:
                    d = rng.uniform(2e-5, 8e-5)
                    ni = best_h[pos] - d
                    nj = best_h[pos+L] + d
                    if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                        best_h[pos] = ni
                        best_h[pos+L] = nj
                        cn = _eval(best_h)
                        if cn < best_c - 1e-15:
                            best_c = cn
                            break
                        best_h[pos] = ni + d
                        best_h[pos+L] = nj - d

                            break

                        best_h[pos] = nj - d

def _proj_grad(h, alpha, m_lags, nsteps, step0):
    n = len(h)
    v = np.zeros(n)
    mu = 0.65
    step = step0
    cur_c = _eval(h)
    for _ in range(nsteps):

        v = mu * v + (1.0 - mu) * g
        gn = np.linalg.norm(v)
        if gn < 1e-16:
            break
        v /= gn
        taken = False
        for _t in range(12):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-16:
                h = h_new
                cur_c = c_new
                step = min(step * 1.08, step0 * 15.0)
                taken = True
                break
            step *= 0.45
        if not taken:
            step = min(step * 1.5, step0 * 15.0)
    return _repair(h, n)

def _local_search(h, n_iter, rng, base_step):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    stall = 0
    for it in range(n_iter):
        c = _eval_full(h)
        cmax = c.max()
        top_lags = np.where(c > cmax - 3e-5)[0]
        frac = it / max(n_iter, 1)
        # Informed move: target active lags
        if rng.rand() < 0.5 and len(top_lags) > 0:
            k = top_lags[rng.randint(len(top_lags))]
            L = k - (n - 1)
            if L >= 0:
                contrib = h[:n-L] * (1.0 - h[L:])
                pos = int(np.argmax(contrib))
                d = base_step * (1.5 - frac) * rng.uniform(0.5, 2.0)
                d = min(d, h[pos] * 0.5, (1.0 - h[pos+L]) * 0.5)
                if d < 1e-8:

                h[pos] -= d
                h[pos+L] += d
            else:
                M2 = -L
                contrib = h[M2:] * (1.0 - h[:n-M2])

                d = base_step * (1.5 - frac) * rng.uniform(0.5, 2.0)
                d = min(d, h[pos+M2] * 0.5, (1.0 - h[pos]) * 0.5)

                h[pos+M2] -= d
                h[pos] += d

            idx = rng.randint(0, n, size=kk)

            d -= d.mean()
            h[idx] += d
            if h.min() < -0.05 or h.max() > 1.05:
                h = _repair(best_h, n)
                continue
        h = _repair(h, n)
        c5n = _eval(h)
        if c5n < c5cur - 1e-15:
            c5cur = c5n
            stall = 0
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        else:
            stall += 1
            if stall > 50:
                h = best_h.copy()
                c5cur = best_c
                stall = 0
    return best_h, best_c
