    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _binary_proj(h, n):
    """Project to binary with exactly n/2 ones."""
    target = n // 2
    h = h.copy()
    # Threshold: take top n/2 values as 1
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def _swap_search(h, n, nit, rng, elite=10):
    """Local search via swaps for binary h."""
    target = n // 2
    h = _binary_proj(h, n)
    c = _eval(h)
    bh = h.copy()
    bc = c
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    for it in range(nit):
        i = rng.choice(list(ones)) if rng.rand() < 0.5 else rng.choice(list(zeros))
        # Find j to swap with
        if h[i] == 1:
            j = rng.choice(list(zeros))
        else:
            j = rng.choice(list(ones))
        
        h[i], h[j] = h[j], h[i]
        cn = _eval(h)
        if cn < c:
            c = cn
            if cn < bc:
                bc, bh = cn, h.copy()
            if h[i] == 1:
                ones.add(i); zeros.discard(i); zeros.add(j); ones.discard(j)
            else:
                zeros.add(i); ones.discard(i); ones.add(j); zeros.discard(j)
        else:
            h[i], h[j] = h[j], h[i]
            # swap back the sets
            if h[i] == 1:
                ones.discard(i); zeros.add(i); zeros.discard(j); ones.add(j)
            else:
                zeros.discard(i); ones.add(i); ones.discard(j); zeros.add(j)
    
    return bh, bc

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _sobj(x, alpha, mk):
    n = len(x); dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-mk:]
    s = alpha * (c[idx] - m); s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sgrad(x, alpha, mk):
    n = len(x); dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-mk:]
    s = alpha * (c[idx] - m); s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _cfun(x, hn):
    return x.sum() - hn

def _cjac(x, hn):
    return np.ones(len(x))

def _slsqp(h, alpha, mk, maxiter):
    n = len(h)
    try:
        r = minimize(_sobj, h, jac=_sgrad, args=(alpha, mk), method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _cfun, 'jac': _cjac, 'args': (n * 0.5,)}],
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return h

    t = n // 2
    h = h.copy()
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:t]] = 1.0
    return b
