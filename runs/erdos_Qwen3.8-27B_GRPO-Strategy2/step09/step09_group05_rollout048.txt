        h = _repair(h, n)
        c_new = _eval(h)
        if c_new < cur_c or rng.rand() < np.exp(-(c_new - cur_c) / max(temp, 1e-10)):
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h = best_h.copy() if rng.rand() < 0.01 else h
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Multi-resolution warmup
    for ratio in [4, 8]:
        if elapsed() > 35:
            break
        nc = max(50, n // ratio)
        f = n // nc
        hc = _downsample(h, f)
        for alpha in [5000, 50000, 500000]:
            if elapsed() > 50:
                break
            hc = _slsqp(hc, alpha, 50, 500, 1e-13)
        h_up = _upsample(hc, f)
        c5u = _eval(h_up)
        if c5u < best_c5:
            best_c5 = c5u
            best_h = h_up.copy()

    # Phase 2: Careful SLSQP alpha sweep
    alpha_seq = [8000, 40000, 200000, 1000000, 5000000, 25000000, 100000000, 500000000, 2000000000]
    for i, alpha in enumerate(alpha_seq):
        if elapsed() > 420:
            break
        lift = int(30 + 12 * i)
        lift = min(130, lift)
        h_new = _slsqp(best_h.copy(), alpha, lift, 1000, 1e-15)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 3: Nesterov projected gradient
    if elapsed() < budget_s - 280:
        configs = [
            (50000, 60, 500, 0.004),
            (300000, 80, 400, 0.002),
            (1500000, 100, 350, 0.001),
            (8000000, 120, 300, 0.0004),
        ]
        for alpha, ml, steps, s0 in configs:
            if elapsed() > budget_s - 200:
                break
            h_new = _nesterov(best_h.copy(), alpha, ml, steps, s0)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new
            # Re-run SLSQP after gradient step
            if elapsed() < budget_s - 210:
                h2 = _slsqp(h_new, 500000, 80, 400, 1e-15)
                c5n2 = _eval(h2)
                if c5n2 < best_c5:
                    best_c5 = c5n2
                    best_h = h2

    # Phase 4: Short anneal + SLSQP to escape local min
    if elapsed() < budget_s - 120:
        h_ann, c_ann = _anneal_refine(best_h.copy(), 3e-4, 1e-6, 5000, rng)
        if c_ann < best_c5:
            best_c5 = c_ann
            best_h = h_ann
        h_new = _slsqp(best_h.copy(), 2000000, 100, 600, 1e-15)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 5: Gradient-guided pair search
    if elapsed() < budget_s - 40:
        nitr = min(int((budget_s - 40 - elapsed()) / 0.012), 50000)
        best_h, best_c5 = _pair_search(best_h, nitr, rng, 90)

    # Phase 6: Ultra-fine random polish
    if elapsed() < budget_s - 3:
        h2 = best_h.copy()
        nc2 = len(h2)
        c5c = _eval(h2)
        npol = min(int((budget_s - 3 - elapsed()) / 0.0012), 400000)
        for _ in range(npol):
            if elapsed() > budget_s - 1:
                break
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 10)) % nc2
            d = 1.2e-7 * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
