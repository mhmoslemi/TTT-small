def _slsqp_obj(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1

        if wk < 1e-15:

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        h[0] += diff
    return h

        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax - 1, cmax - 1 - j)

            else:

        if abs(t_new - h[j]) > 1e-15:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(len(idx)):
        wk = w[ii]
        if wk < 1e-15:
            continue
        wk *= dx
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:

            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

    return x.sum() - target

    return np.ones(len(x))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        h[0] += diff
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _cd_pass(h, rng, nit=40):
    n = len(h)
    c = _corr(h)
    cmax = 2 * n - 1
    jlist = np.arange(n).copy()
    rng.shuffle(jlist)
    nm1 = n - 1
    for j in jlist:
        j = int(j)
        beta = np.zeros(cmax)
        k1lo, k1hi = j, min(cmax - 1, j + nm1)
        lags1 = np.arange(k1lo, k1hi + 1) - nm1
        beta[k1lo:k1hi + 1] += 1.0 - h[j - lags1]
        k2lo, k2hi = max(0, nm1 - j), min(cmax - 1, cmax - 1 - j)
        lags2 = np.arange(k2lo, k2hi + 1) - nm1
        beta[k2lo:k2hi + 1] -= h[j + lags2]
        g = c - beta * h[j]
        lo, hi = 0.0, 1.0
        for _ in range(nit):
            t1 = lo + (hi - lo) / 3.0
            t2 = hi - (hi - lo) / 3.0
            if np.max(g + beta * t1) <= np.max(g + beta * t2):
                hi = t2
            else:
                lo = t1
        tnew = 0.5 * (lo + hi)
        if abs(tnew - h[j]) > 1e-15:
            c += beta * (tnew - h[j])
            h[j] = tnew
    return h

def _sq_obj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sq_grad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(len(idx)):
        wk = w[ii]
        if wk < 1e-15:
            continue
        wk *= dx
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sf(x, t):
    return x.sum() - t

def _sj(x, t):
    return np.ones(len(x))

def _slsqp(h, alpha, ml, mi):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (n * 0.5,)}]
    try:
        res = minimize(_sq_obj, h, jac=_sq_grad, args=(alpha, ml),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': mi, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _lsearch(h, n, rng, ni):

    best_h, best_c5 = h.copy(), c5c
    for _ in range(ni):

                    best_c5, best_h = c5n, h.copy()
            else:

        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        h[0] += diff
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _cd_pass(h, rng, nit=45):
    n = len(h)
    c = _corr(h)
    cmax = 2 * n - 1
    nm1 = n - 1
    jlist = np.arange(n).copy()
    rng.shuffle(jlist)
    for j in jlist:
        j = int(j)
        beta = np.zeros(cmax)
        k1lo, k1hi = j, min(cmax - 1, j + nm1)
        lags1 = np.arange(k1lo, k1hi + 1) - nm1
        beta[k1lo:k1hi + 1] += 1.0 - h[j - lags1]
        k2lo, k2hi = max(0, nm1 - j), min(cmax - 1, cmax - 1 - j)
        lags2 = np.arange(k2lo, k2hi + 1) - nm1
        beta[k2lo:k2hi + 1] -= h[j + lags2]
        g = c - beta * h[j]
        oj = h[j]
        lo, hi = 0.0, 1.0
        for _ in range(nit):
            t1 = lo + (hi - lo) / 3.0
            t2 = hi - (hi - lo) / 3.0
            c1 = g + beta * t1
            c1[nm1] -= (t1 - oj) ** 2
            c2 = g + beta * t2
            c2[nm1] -= (t2 - oj) ** 2
            if np.max(c1) <= np.max(c2):
                hi = t2
            else:
                lo = t1
        tnew = 0.5 * (lo + hi)
        if abs(tnew - oj) > 1e-15:
            c += beta * (tnew - oj)
            c[nm1] -= (tnew - oj) ** 2
            h[j] = tnew
    return h

def _sq_obj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sq_grad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(len(idx)):
        wk = w[ii]
        if wk < 1e-15:
            continue
        wk *= dx
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sf(x, t):
    return x.sum() - t

def _sj(x, t):
    return np.ones(len(x))

def _slsqp(h, alpha, ml, mi):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (n * 0.5,)}]
    try:
        res = minimize(_sq_obj, h, jac=_sq_grad, args=(alpha, ml),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': mi, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _lsearch(h, n, rng, ni):
    c5c = _eval(h)
    best_h, best_c5 = h.copy(), c5c
    for _ in range(ni):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        d = 2e-6 * rng.randn()
        ni_, nj_ = h[i] + d, h[j] - d
        if 0 <= ni_ <= 1 and 0 <= nj_ <= 1:
            h[i], h[j] = ni_, nj_
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c5:
                    best_c5, best_h = c5n, h.copy()
            else:
                h[i], h[j] = ni_ - d, nj_ + d
    return best_h, best_c5

def _balance_lags(h, rng, n_passes=5):
    n = len(h)
    dx = 2.0 / n
    for _ in range(n_passes):
        c = _corr(h)
        cmax = c.max()
        tol = 0.02
        active = np.where(c > cmax - tol)[0]
        if len(active) < 2:
            break
        n_active = len(active)
        delta = np.zeros(n)
        nm1 = n - 1
        for k_idx in active:
            L = k_idx - nm1
            if L >= 0:
                delta[L:] += (1.0 - h[:n - L])
                delta[:n - L] -= h[L:]
            else:
                M = -L
                delta[:n - M] += (1.0 - h[M:])
                delta[M:] -= h[:n - M]
        delta /= (n_active * dx)
        step = 0.001
        h = h - step * delta
        h = _repair(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: CD with adaptive perturbation (40% time)
    p1 = t0 + budget_s * 0.40
    stuck, eps = 0, 0.012
    while time.time() < p1:
        h = _cd_pass(h, rng)
        h = _repair(h, n)
        c5 = _eval(h)
        if c5 < best_c5 - 1e-9:
            best_c5, best_h, stuck, eps = c5, h.copy(), 0, max(0.003, eps * 0.96)
        else:
            stuck += 1
            if stuck >= 6:
                h = _repair(h + eps * rng.randn(n), n)
                stuck, eps = 0, max(0.003, eps * 0.75)

    # Phase 2: SLSQP with multiple alphas (25% time)
    p2 = t0 + budget_s * 0.65
    alphas = [200, 1000, 5000, 20000, 100000, 500000, 2000000]
    for alpha in alphas:
        if time.time() > p2 - 15:
            break
        hn = _slsqp(best_h.copy(), alpha, 50, 500)
        c5n = _eval(hn)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn
        # Also try from a perturbed version
        if alpha >= 5000 and time.time() < p2 - 30:
            hp = _repair(best_h + 0.008 * rng.randn(n), n)
            hn2 = _slsqp(hp, alpha * 0.5, 50, 300)
            c5n2 = _eval(hn2)
            if c5n2 < best_c5:
                best_c5, best_h = c5n2, hn2

    # Phase 3: CD + balance from best (20% time)
    p3 = t0 + budget_s * 0.85
    h2 = best_h.copy()
    while time.time() < p3:
        h2 = _cd_pass(h2, rng)
        h2 = _repair(h2, n)
        h2 = _balance_lags(h2, rng, 3)
        h2 = _repair(h2, n)
        c5 = _eval(h2)
        if c5 < best_c5:
            best_c5, best_h = c5, h2.copy()

    # Phase 4: Final SLSQP + local search (10% time)
    if time.time() < budget_s - 40:
        hn = _slsqp(best_h.copy(), 4000000, 20, 350)
        c5n = _eval(hn)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn
    if time.time() < budget_s - 3:
        rem = budget_s - 3 - (time.time() - t0)
        ntr = min(int(rem / 0.005), 180000)
        best_h, best_c5 = _lsearch(best_h, n, rng, ntr)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
