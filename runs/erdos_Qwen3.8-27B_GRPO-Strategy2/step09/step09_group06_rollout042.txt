import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sum_fn(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _obj_fn(x, alpha, m, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s, -500, 500)
    return c.max() + np.log(np.exp(s).sum()) / alpha

def _grad_fn(x, alpha, m, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s, -500, 500)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _slsqp(h, alpha, m_lags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_obj_fn, h, jac=_grad_fn, args=(alpha, m_lags, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _pgd_nesterov(h, alpha, topm, nsteps, step0, momentum):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    y = h.copy()
    prev_y = h.copy()
    for it in range(nsteps):
        beta = (1.0 + np.sqrt(1.0 + 4.0 * (momentum ** 2) * it)) / 2.0
        theta = (1.0 - beta) * momentum
        y_new = h + theta * (h - prev_y)
        prev_y = y.copy()
        y = y_new
        g = _grad_fn(y, alpha, topm, n)
        v = momentum * v + (1.0 - momentum) * g
        taken = False
        for _t in range(14):
            h_new = _proj(y - step * v, target)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-16:
                h = h_new
                cur_c = c_new
                step = min(step * 1.06, step0 * 80.0)
                taken = True
                break
            step *= 0.45
        if not taken:
            step = min(step * 1.4, step0 * 80.0)
    return _repair(h, n), cur_c

def _lag_dir(h, n, k_lag):
    g = np.zeros(n)
    L = k_lag - (n - 1)
    if L >= 0:
        g[L:] += (1.0 - h[:n - L])
        g[:n - L] -= h[L:]
    else:
        M = -L
        g[:n - M] += (1.0 - h[M:])
        g[M:] -= h[:n - M]
    return g

    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    for it in range(niter):
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        top_lags = np.argsort(c)[-6:]
        k = top_lags[rng.randint(len(top_lags))]
        d = _lag_dir(h, n, k)
        d -= d.mean()
        scale = 3e-5 * (1.0 - 0.5 * it / max(niter, 1)) * rng.uniform(0.3, 1.0)
        h_new = h - scale * d
        if h_new.min() < -0.05 or h_new.max() > 1.05:
            scale *= 0.1
            h_new = h - scale * d
        h_new = _proj(h_new, n * 0.5)
        c5n = _eval(h_new)
        if c5n < c5cur:
            c5cur = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

    n = len(h)

    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    return _repair(np.repeat(h, f), len(h) * f)
