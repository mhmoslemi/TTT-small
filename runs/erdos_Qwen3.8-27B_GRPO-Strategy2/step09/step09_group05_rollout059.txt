def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)

        if abs(s - target) < 1e-12:

    diff = target - h.sum()
    if abs(diff) > 1e-12:

                diff -= a

                h[idx] -= a
                diff += a
