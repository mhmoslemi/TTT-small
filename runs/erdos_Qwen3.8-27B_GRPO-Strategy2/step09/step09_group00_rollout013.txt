    for f in [lambda x: 1.0 - x, lambda x: x[::-1],
              lambda x: 0.5 + 0.5 * (x - x[::-1]),
              lambda x: 0.5 + 0.5 * (x + x[::-1] - 1.0)]:
        hv = _proj(f(h0.copy()), n0 * 0.5)
        c = _c5(hv)
        if c < best_c:
            best = hv
            best_c = c

    # Phase 1: FISTA top-m (180s)
    end1 = t0 + min(180.0, 0.16 * budget_s)
    if time.time() < end1:
        hcur = best.copy()
        for mt, tol in [(5, 0.025), (3, 0.012), (2, 0.006)]:
            slot = (end1 - time.time()) / 3.0
            if slot < 5:
                break
            hf, bh, bc = _fista_topm(hcur, mt, tol, slot, 5.0)
            cf = _c5(hf)
            if bc < best_c:
                best = bh
                best_c = bc
            if cf < best_c:
                best = hf
                best_c = cf
            hcur = hf if cf <= bc else bh

    # Phase 2: Multi-resolution FISTA (220s)
    end2 = t0 + min(400.0, 0.36 * budget_s)
    if time.time() < end2:
        for m in [512, 640, 800]:
            slot = (end2 - time.time()) / 3.0
            if slot < 10:
                break
            hm = _resample(best, m)
            for mt in [3, 8]:
                if time.time() >= end2 - 5:
                    break
                tt = (end2 - time.time()) / 2.0
                hf, bh, bc = _fista_topm(hm, mt, 0.015, tt, 8.0)
                hb = _resample(bh, n0) if len(bh) != n0 else bh
                cb = _c5(hb)
                if cb < best_c:
                    best = hb
                    best_c = cb
                hf_r = _resample(hf, n0) if len(hf) != n0 else hf
                cf = _c5(hf_r)
                if cf < best_c:
                    best = hf_r
                    best_c = cf

    # Phase 3: Local search (380s)
    end3 = t0 + min(780.0, 0.71 * budget_s)
    if time.time() < end3:
        best, best_c = _local_search(best, rng, end3)

    # Phase 4: FISTA refinement (150s)
    end4 = t0 + budget_s - 50.0
    if time.time() < end4:
        for mt, tol in [(2, 0.004), (3, 0.008)]:
            if time.time() >= end4 - 5:
                break
            hf, bh, bc = _fista_topm(best, mt, tol, (end4 - time.time()) / 2.0, 20.0)
            cf = _c5(hf)
            if bc < best_c:
                best = bh
                best_c = bc
            if cf < best_c:
                best = hf
                best_c = cf

    # Phase 5: Final local polish
    end5 = t0 + budget_s - 3.0
    if time.time() < end5:
        best, best_c = _local_search(best, rng, end5)

    for f in [lambda x: 1.0 - x, lambda x: x[::-1],
              lambda x: 0.5 + 0.5 * (x - x[::-1])]:
        hv = _proj(f(best.copy()), len(best) * 0.5)
        c = _c5(hv)
        if c < best_c:
            best = hv
            best_c = c
    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
