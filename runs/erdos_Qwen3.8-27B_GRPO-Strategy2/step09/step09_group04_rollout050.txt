        for trial in range(8):
            if _el() > budget_s - 115:
                break
            hn = _diversify_binary_blend(best_h, rng2, n)
            cn = _eval(hn)
            if cn < best_c5 + 0.002:
                hn = _slsqp(hn, 500000, 50, 2500)
                cn = _eval(hn)
                if cn < best_c5:
                    best_c5, best_h = cn, hn

    # Phase 5: High-precision SLSQP
    if _el() < budget_s - 70:
        for a in [100000000, 500000000, 2000000000]:
            if _el() > budget_s - 45:
                break
            ml = min(200, 50 + int(22 * np.log10(a / 1e8 + 1)))
            hn = _slsqp(best_h, a, ml, 6000)
            cn = _eval(hn)
            if cn < best_c5:
                best_c5, best_h = cn, hn

    # Phase 6: Projected gradient with adaptive step
    if _el() < budget_s - 25:
        h6 = best_h.copy()
        c6 = best_c5
        alpha6 = 2000000
        ml6 = 80
        nsteps = min(int((budget_s - 25 - _el()) / 0.003), 80000)
        mu = 0.88
        v = np.zeros(n)
        step = 0.002
        for _ in range(nsteps):
            if _el() > budget_s - 20:
                break
            g = _sgrad(h6, alpha6, ml6)
            v = mu * v + (1.0 - mu) * g
            gn = np.linalg.norm(v)
            if gn < 1e-16:
                break
            v /= gn
            improved = False
            for sf in [1.0, 0.5, 0.25, 0.1, 0.05]:
                hn = _proj_sum(h6 - step * sf * v, n * 0.5)
                cn = _eval(hn)
                if cn < c6 - 1e-18:
                    h6, c6 = hn, cn
                    if cn < best_c5:
                        best_c5, best_h = cn, h6.copy()
                    step = min(step * 1.05, 0.04)
                    improved = True
                    break
            if not improved:
                step *= 0.6

    # Phase 7: Ultra-fine coordinate descent
    if _el() < budget_s - 3:
        h7 = best_h.copy()
        c7 = best_c5
        rng3 = np.random.RandomState(seed + 4242)
        nsteps = min(int((budget_s - 3 - _el()) / 0.0008), 900000)
        for _ in range(nsteps):
            if _el() > budget_s - 2:
                break
            i = rng3.randint(n)
            j = (i + 1 + rng3.randint(0, 5)) % n
            d = 2e-7 * rng3.randn()
            ni, nj = h7[i] + d, h7[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h7[i], h7[j] = ni, nj
                cn = _eval(h7)
                if cn < c7:
                    c7 = cn
                    if cn < best_c5:
                        best_c5, best_h = cn, h7.copy()
                else:
                    h7[i], h7[j] = ni - d, nj + d

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
