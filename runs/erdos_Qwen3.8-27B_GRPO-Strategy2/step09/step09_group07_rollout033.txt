import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-15:
                break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _cd_pass(h, order, n_ternary=50):
    n = len(h)
    c = _corr(h)
    cmax = 2 * n - 1
    nm1 = n - 1
    for j in order:
        j = int(j)
        beta = np.zeros(cmax)
        k1_lo = j
        k1_hi = min(cmax - 1, j + nm1)
        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax - 1, cmax - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
        beta[k2_lo:k2_hi + 1] -= h[j + lags2]
        g = c - beta * h[j]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(n_ternary):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(g + beta * t1) <= np.max(g + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-15:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

def _slsqp_obj(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(len(idx)):
        wk = w[ii]
        if wk < 1e-15:
            continue
        wk *= dx
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:
            if L < n:
                g[L:] += wk * (1.0 - x[:n - L])
                g[:n - L] -= wk * x[L:]
        else:
            M = -L
            if M < n:
                g[:n - M] += wk * (1.0 - x[M:])
                g[M:] -= wk * x[:n - M]
    return g

def _sum_fun(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:

                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-
18})
        return _repair(res.x, n)
    except Exception:
        return h

def _local_search(h, n, rng, n_iter, base_step):
    c5c = _eval(h)
    best_h = h.copy()
    best_c5 = c5c
    for i in range(n_iter):
        scale = base_step * (1.0 + 0.5 * np.cos(2.0 * np.pi * i / max(1, n_iter // 10)))
        idx_a = rng.randint(n)
        idx_b = (idx_a + 1 + rng.randint(0, 12)) % n
        if rng.rand() < 0.2:
            idx_b = rng.randint(n)
        d = scale * rng.randn()
        ni = h[idx_a] + d
        nj = h[idx_b] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[idx_a], h[idx_b] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[idx_a], h[idx_b] = ni - d, nj + d
    return best_h, best_c5

            best_h = h.copy()
        if abs(prev_c5 - c5) < 1e-12:
