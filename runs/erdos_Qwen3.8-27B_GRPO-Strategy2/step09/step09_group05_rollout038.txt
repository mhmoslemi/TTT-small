import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _swap_delta(h, a, b, n):
    L = np.arange(-(n - 1), n)
    am, ap = a - L, a + L
    bm, bp = b - L, b + L
    vm, vp = (am >= 0) & (am < n), (ap >= 0) & (ap < n)
    vm2, vp2 = (bm >= 0) & (bm < n), (bp >= 0) & (bp < n)
    d = np.zeros(2 * n - 1)
    d[vm] += h[am[vm]]
    d[vp] += h[ap[vp]]
    d[vm2] -= h[bm[vm2]]
    d[vp2] -= h[bp[vp2]]
    return d

    h = h.copy()
    c = _corr_raw(h)
    best_h = h.copy()
    best_c5 = c.max() * 2.0 / n
    no_imp = 0
    temp = 3.0
    t0 = time.time()
    m1 = h >= 0.5
    m0 = ~m1
    while time.time() - t0 < t_limit:
        K = k_lags
        top = np.argsort(c)[-K:]
        lags = top - (n - 1)
        w = np.ones(K)
        w[0] = 5.0
        if K > 1:
            w[1] = 3.0
        if K > 2:
            w[2] = 2.0
        F = np.zeros(n)
        for k in range(K):
            L = lags[k]
            if L >= 0:
                F[L:n] += w[k] * h[:n - L]
                F[:n - L] += w[k] * h[L:]
            else:
                F[:n + L] += w[k] * h[-L:]
                F[-L:] += w[k] * h[:-L]
        F_1 = np.where(m1, F, 1e18)
        F_0 = np.where(m0, F, -1e18)
        a = int(np.argmin(F_1))
        b = int(np.argmax(F_0))
        if not m1[a] or m1[b]:
            continue
        delta = _swap_delta(h, a, b, n)
        new_max = np.max(c + delta)
        cur_max = c.max()
        accept = new_max < cur_max - 1e-12
        if not accept and new_max - cur_max < temp:
            if rng.rand() < np.exp(-(new_max - cur_max) / max(temp, 0.01)):
                accept = True
        if accept:

def _fine_polish(h, n, t_limit, rng):
    c5c = _eval(h)
    best_h = h.copy()
    best_c5 = c5c
    t0 = time.time()
    step = 5e-5
    while time.time() - t0 < t_limit:
        i = rng.randint(n)
        j = (i + rng.randint(1, min(15, n))) % n
        d = step * rng.randn() * 0.3
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                step = min(step * 1.02, 2e-4)
                if c5c < best_c5:
                    best_c5 = c5c
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                step = max(step * 0.98, 1e-6)
    return best_h, best_c5

def _make_periodic(n, p, rng):
    h = np.zeros(n)
    n_per = p // 2
    for start in range(0, n, p):
        end = min(start + p, n)
        length = end - start
        picks = rng.choice(length, size=min(n_per, length), replace=False)
        h[start + picks] = 1.0
    return _repair(h, n)
