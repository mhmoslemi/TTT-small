import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s < 1e-15:
            return np.full(n, 0.5)
        h *= target / s
        np.clip(h, 0.0, 1.0, out=h)
    d = target - h.sum()
    if abs(d) > 1e-13:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if abs(d) < 1e-13:
                break
            a = min(abs(d), 1.0 - h[i]) if d > 0 else min(abs(d), h[i])
            h[i] += a if d > 0 else -a
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(30):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-12:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _fista_fast(h0, a, tl):
    t0 = time.time()
    n = len(h0)
    S = n * 0.5
    h = _repair(h0, n)
    y = h.copy()
    tv = 1.0
    Lc = 10.0

    def _val(z):
        cr = _corr(z)
        mx = cr.max()
        return mx + np.log(np.sum(np.exp(np.clip(a * (cr - mx), -400, 0)))) / a

    def _grad(z):
        cr = _corr(z)
        mx = cr.max()
        w = np.exp(np.clip(a * (cr - mx), -400, 0))
        w /= w.sum()
        # Gradient of sum w_k * corr[k] w.r.t. h
        # corr[k] for lag L=k-(n-1): d/dh[m] = (1-h[m+L]) for m+L<n or similar
        g = np.zeros(n)
        for k_idx in range(2 * n - 1):
            if w[k_idx] < 1e-15:
                continue
            L = k_idx - (n - 1)
            if L >= 0:
                lo, hi = 0, n - L
                g[lo:hi] += w[k_idx] * (1.0 - h[lo + L:hi + L])
                g[hi:hi + L] += 0  # no contribution
            else:
                La = -L
                lo, hi = La, n
                g[lo:hi] -= w[k_idx] * h[lo - La:hi - La]
        return 2.0 / n * g

    fy = _val(y)
    bh = h.copy()
    bc = _c5(h)
    for it in range(60000):
        if time.time() - t0 > tl:
            break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tv * tv))
            hp = h.copy()
            y = h + ((tv - 1.0) / tn) * (h - hp)
            tv = tn
            y = _proj(y, S)
            fy = _val(y)
        else:
            hp = h.copy()
        g = _grad(y)
        step = 1.0 / max(Lc, 1e-12)
        for _ in range(10):
            xn = _proj(y - step * g, S)
            fx = _val(xn)
            d = xn - y
            rhs = fy + np.dot(g, d) + 0.5 * (1.0 / step) * np.dot(d, d)
            if fx <= rhs + 1e-9:
                break
            step *= 0.5
        Lc = 1.0 / max(step, 1e-12)
        h = xn
        Lc = max(Lc * 0.9, 1.0)
        if it % 20 == 0:
            c = _c5(h)
            if c < bc:
                bc = c
                bh = h.copy()
    return bh, bc

def _greedy_binary(n, rng):
    """Greedy: place n/2 ones to minimize max overlap"""
    half = n // 2
    h = np.zeros(n)
    for _ in range(half):
        best_pos = 0
        best_score = np.inf
        trial = h.copy()
        indices = rng.permutation(n)
        for idx in indices[:80]:
            trial[idx] = 1.0
            sc = _corr(trial).max()
            if sc < best_score:
                best_score = sc
                best_pos = idx
            trial[idx] = 0.0
        h[best_pos] = 1.0

def _swap_anneal(h0, rng, deadline):
    """Lag-targeted swap search with simulated annealing"""
    n = len(h0)
    h = h0.copy()
    c = _c5(h)
    bc = c
    bh = h.copy()
    T = 5e-5
    T_min = 1e-7
    stale = 0
    cool_rate = 1.0 / max(1, int((deadline - time.time()) * 2.0))

    while time.time() < deadline:
        T = max(T_min, T * (1.0 - cool_rate))
        corr = _corr(h)
        n_corr = len(corr)
        worst_idx = np.argsort(corr)[-8:][::-1]

        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        found_improve = False
        attempts = 0
        max_attempts = 300

        while attempts < max_attempts:
            attempts += 1
            # Choose target lag
            wi = worst_idx[rng.randint(len(worst_idx))]
            L = int(wi - (n - 1))
            if L == 0:
                i = int(rng.choice(ones))
                j = int(rng.choice(zeros))
            elif L > 0:
                gi = ones[(ones + L < n) & (h[ones + L] < 0.5)]
                gj = zeros[(zeros + L < n) & (h[zeros + L] > 0.5)]
                if len(gi) == 0 or len(gj) == 0:
                    i, j = int(rng.choice(ones)), int(rng.choice(zeros))
                else:
                    i, j = int(rng.choice(gi)), int(rng.choice(gj))
            else:
                La = -L
                gi = ones[(ones >= La) & (h[ones - La] < 0.5)]
                gj = zeros[(zeros >= La) & (h[zeros - La] > 0.5)]
                if len(gi) == 0 or len(gj) == 0:
                    i, j = int(rng.choice(ones)), int(rng.choice(zeros))
                else:
                    i, j = int(rng.choice(gi)), int(rng.choice(gj))

            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            c_new = _c5(h)
            delta = c_new - c
            if c_new < c - 1e-14:
                c = c_new
                if c < bc:
                    bc = c
                    bh = h.copy()
                    stale = 0
                found_improve = True
            elif rng.random() < np.exp(-delta / max(T, 1e-10)):
                c = c_new
            else:
                h[i], h[j] = h[j], h[i]

        if not found_improve:
            stale += 1
            if stale % 200 == 0:
                # Perturbation: random block of swaps
                n_swap = rng.randint(10, 40)
                for _ in range(n_swap):
                    ii = int(rng.choice(ones if len(ones) else np.arange(n)))
                    jj = int(rng.choice(zeros if len(zeros) else np.arange(n)))
                    if ii != jj and h[ii] != h[jj]:
                        h[ii], h[jj] = h[jj], h[ii]
                c = _c5(h)
                stale = 0

    return bh, bc

def _block_swap_search(h0, rng, deadline):
    """Try moving contiguous blocks of 1s"""
    n = len(h0)
    h = h0.copy()
    c = _c5(h)
    bc = c
    bh = h.copy()

    while time.time() < deadline:
        ones = np.where(h > 0.5)[0]
        if len(ones) < 2:
            break
        # Find a contiguous block of 1s to move
        starts = []
        prev = ones[0] - 1
        for p in ones:
            if p != prev + 1:
                if prev >= 0:
                    starts.append(prev)
            prev = p
        if prev >= 0:
            starts.append(prev)

        if len(starts) > 0:
            blk_end = int(rng.choice(starts))
            # Find block start
            blk_start = blk_end
            while blk_start > 0 and h[blk_start - 1] > 0.5:
                blk_start -= 1
            blk_len = blk_end - blk_start + 1
            if 2 <= blk_len <= 15:
                # Find target position (a block of 0s of same length)
                zeros = np.where(h < 0.5)[0]
                if len(zeros) > blk_len:
                    tstart = int(rng.choice(zeros[:len(zeros)-blk_len]))
                    # Check if path is clear (0s at target)
                    if np.all(h[tstart:tstart + blk_len] < 0.5):
                        block_vals = h[blk_start:blk_start + blk_len].copy()
                        h[blk_start:blk_start + blk_len] = h[tstart:tstart + blk_len].copy()
                        h[tstart:tstart + blk_len] = block_vals
                        c_new = _c5(h)
                        if c_new < bc - 1e-14:
                            bc = c_new
                            bh = h.copy()
                            c = c_new
                        else:
                            h[tstart:tstart + blk_len] = h[blk_start:blk_start + blk_len].copy()
                            h[blk_start:blk_start + blk_len] = block_vals

    return bh, bc
