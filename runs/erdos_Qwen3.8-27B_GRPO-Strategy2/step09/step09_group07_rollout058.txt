def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _corr(h):
    n = len(h)

    return float(np.max(_corr(h)))

def _smooth_obj(h, alpha, m_lags):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    # Use top m_lags for efficiency
    idx = np.argsort(c)[-m_lags:]
    c_top = c[idx]
    m = c_top.max()
    s = alpha * (c_top - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

    w = np.ones(len(idx)) / len(idx)
    g = np.zeros(n)
    dx = 2.0 / n

        k = int(idx[ii])

        else:
            M = -L
            g[:n-M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n-M]
    # Momentum update
    v = momentum * v - step_size * g
    h_new = h + v
    return h_new, v

                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _eval(h):
    return float(np.max(_corr(h)))

def _cd_pass(h, rng):

    cmax = 2 * n - 1
    jlist = np.arange(n)
    rng.shuffle(jlist)
    nm1 = n - 1

        beta = np.zeros(cmax)
        k1_lo = j
        k1_hi = min(cmax - 1, j + nm1)
        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax - 1, cmax - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
        beta[k2_lo:k2_hi + 1] -= h[j + lags2]
        g = c - beta * h[j]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(30):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(g + beta * t1) <= np.max(g + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-14:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

def _lgbfgs_smooth(h, alpha, m_lags, maxiter):
    n = len(h)
    dx = 2.0 / n
    nm1 = n - 1
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        idx = np.argsort(c)[-m_lags:]
        ct = c[idx]
        mx = ct.max()
        s = alpha * (ct - mx)
        s -= s.max()
        return mx + np.log(np.exp(s).sum()) / alpha
    
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        idx = np.argsort(c)[-m_lags:]
        ct = c[idx]
        mx = ct.max()
        s = alpha * (ct - mx)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            wk = w[ii] * dx
            if wk < 1e-15:
                continue
            k = int(idx[ii])
            L = k - nm1
            if L >= 0:
                g[:n-L] += wk * (1.0 - x[L:])
                g[L:] -= wk * x[:n-L]
            else:
                M = -L
                g[:n-M] += wk * (1.0 - x[M:])
                g[M:] -= wk * x[:n-M]
        return g
    
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-15})
    return res.x

    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a

        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax - 1, cmax - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1

        g = c - beta * h[j]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(30):

            t2 = hi_t - (hi_t - lo_t) / 3.0

                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-14:
            c += beta * (t_new - h[j])
            h[j] = t_new

def _lgbfgs(h_init, alpha, m_lags, maxiter):
    n = len(h_init)
    dx = 2.0 / n
    nm1 = n - 1
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        idx = np.argsort(c)[-m_lags:]
        ct = c[idx]
        mx = ct.max()
        s = alpha * (ct - mx)
        s -= s.max()
        return mx + np.log(np.exp(s).sum()) / alpha + 0.5 * alpha * (x.sum() - n/2.0)**2
    
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        idx = np.argsort(c)[-m_lags:]
        ct = c[idx]
        mx = ct.max()
        s = alpha * (ct - mx)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            wk = w[ii] * dx
            if wk < 1e-15:
                continue
            k = int(idx[ii])
            L = k - nm1
            if L >= 0:
                g[:n-L] += wk * (1.0 - x[L:])
                g[L:] -= wk * x[:n-L]
            else:
                M = -L
                g[:n-M] += wk * (1.0 - x[M:])
                g[M:] -= wk * x[:n-M]
        g += 2.0 * alpha * (x.sum() - n/2.0)
        return g
    
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h_init, jac=grad, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 50})
    return res.x
