            fn_ = mcn + np.log(np.exp(shn).sum()) / alpha
            dv = xn - y
            if fn_ <= fy + float(np.dot(g, dv)) + 0.5 * (1.0 / step) * float(np.dot(dv, dv)) + 1e-10:
                ok = True; break
            step *= 0.5
        L = max(2.0 / max(step, 1e-12), 1.0)
        if ok:
            hp = h; h = xn; L = max(L * 0.85, 1.0)
            if it % 20 == 0:
                c = _c5(h)
                if c < best_c:
                    best_c = c; best_h = h.copy()
        else:
            break
    c = _c5(h)
    if c < best_c:
        best_c = c; best_h = h.copy()
    return h, best_h, best_c

def _grad_momentum(h0, deadline, n_active=3):
    """Direct gradient on active lags with momentum"""
    n = len(h0); S = n * 0.5
    h = h0.copy(); vel = np.zeros(n)
    beta = 0.9; lr = 0.02
    best_h = h.copy(); best_c = _c5(h)
    corr = _corr_vec(h)
    t0 = time.time()
    for it in range(500000):
        if time.time() > deadline:
            break
        mc = float(np.max(corr))
        tol = mc - 0.005
        active = np.nonzero(corr >= tol)[0]
        if len(active) > n_active:
            active = active[:n_active]
        g = np.zeros(n)
        for k in active:
            g += _grad_lag(h, int(k))
        g /= len(active)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-12:
            break
        vel = beta * vel - lr * g / g_norm
        h_new = _proj(h + vel, S)
        corr_new = _corr_vec(h_new)
        mc_new = float(np.max(corr_new))
        if mc_new < mc + 1e-10:
            h = h_new; corr = corr_new; mc = mc_new
            if mc * 1 < best_c - 1e-14:
                best_c = mc; best_h = h.copy()
            if it % 200 == 0:
                lr *= 0.97; beta = 0.9 - 0.1 * (it / 5000)
        else:
            vel = 0.5 * vel
            lr *= 0.95
            if it > 100:
                h = _proj(h + 0.01 * (rng_global.random(n) - 0.5), S)
                corr = _corr_vec(h); mc = float(np.max(corr))
    c = _c5(h)
    if c < best_c:
        best_c = c; best_h = h.copy()
    return best_h, best_c

def _sa_targeted(h0, rng, deadline):
    """SA focusing on top-2 active lags with O(1) checks"""
    n = len(h0); S = n * 0.5
    h = h0.copy()
    corr = _corr_vec(h)
    mx = float(np.max(corr))
    best_c = mx; best_h = h.copy()
    t0 = time.time()
    total = max(1.0, deadline - t0)
    recompute = 0
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t0) / total)
        T = 0.3 * (0.001) ** frac
        if recompute % 50 == 0:
            corr = _corr_vec(h); mx = float(np.max(corr))
            k1 = int(np.argmax(corr))
            k2 = k1
            c2s = corr.copy(); c2s[k1] = -1
            k2 = int(np.argmax(c2s))
            top_lags = [k1, k2]
            g1 = _grad_lag(h, k1); g2 = _grad_lag(h, k2)
            max_c = mx
            recompute = -50
        recompute += 1
        k_use = k1 if rng.random() < 0.6 else k2
        g_use = g1 if k_use == k1 else g2
        neg = np.nonzero(g_use < -0.005)[0]
        pos = np.nonzero(g_use > 0.005)[0]
        if len(neg) == 0 or len(pos) == 0:
            i = rng.randint(n); j = rng.randint(n)
            if i == j: continue
            s = 1 if rng.random() < 0.5 else -1
        else:
            i = int(neg[rng.randint(len(neg))])
            j = int(pos[rng.randint(len(pos))])
            s = 1
        avail = min(1.0 - h[i], h[j]) if s > 0 else min(h[i], 1.0 - h[j])
        if avail < 1e-12:
            continue
        base = 0.005 + 0.02 * (1.0 - frac)
        d = s * min(avail, base * (0.3 + 1.4 * rng.random()))
        # Check against top lags
        delta1 = d * (g1[i] - g1[j])
        delta2 = d * (g2[i] - g2[j])
        new_c1 = corr[k1] + delta1
        new_c2 = corr[k2] + delta2
        new_mx = max(new_c1, new_c2, max_c)
        if new_mx <= max_c + T:
            h[i] += d; h[j] -= d
            h = np.clip(h, 0.0, 1.0)
            h = _proj(h, S)
            cn = max_c if new_mx <= max_c + 1e-14 else new_mx
            if cn < best_c - 1e-14:
                best_c = cn; best_h = h.copy()
        else:
            if rng.random() < np.exp(-(new_mx - max_c) / max(T, 1e-15)):
                h[i] += d; h[j] -= d
                h = np.clip(h, 0.0, 1.0)
                h = _proj(h, S)
    corr_f = _corr_vec(h); cf = float(np.max(corr_f))
    if cf < best_c:
        best_c = cf; best_h = h.copy()
    return best_h, best_c

def _fine_tune(h0, rng, deadline):
    """Very fine gradient-directed coordinate search"""
    n = len(h0); S = n * 0.5
    h = h0.copy()
    best_c = _c5(h); best_h = h.copy()
    t0 = time.time()
    while time.time() < deadline:
        corr = _corr_vec(h); mx = float(np.max(corr))
        k_a = int(np.argmax(corr))
        g = _grad_lag(h, k_a)
        order = np.argsort(g)
        q = max(3, n // 10)
        low = order[:q]; high = order[-q:]
        i = int(low[rng.randint(len(low))])
        j = int(high[rng.randint(len(high))])
        if i == j:
            continue
        s = 1.0 if g[i] < g[j] else -1.0
        avail = min(1.0 - h[i], h[j]) if s > 0 else min(h[i], 1.0 - h[j])
        if avail < 1e-13:
            continue
        frac = min(1.0, (time.time() - t0) / max(1e-9, deadline - t0))
        step = max(5e-6, 2e-4 * (0.01) ** frac)
        d = s * min(avail, step * (0.5 + rng.random()))
        h[i] += d; h[j] -= d
        h = _proj(h, S)
        c = _c5(h)
        if c < mx - 1e-15:
            if c < best_c - 1e-15:
                best_c = c; best_h = h.copy()
        else:
            h[i] -= d; h[j] += d
            h = _proj(h, S)
    c = _c5(h)
    if c < best_c:
        best_c = c; best_h = h.copy()
    return best_h, best_c

rng_global = np.random.RandomState(42)

def run(seed=42, budget_s=1100, **kwargs):
    global rng_global
    t0 = time.time()
    rng_global = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _proj(h0, n0 * 0.5)
    best = h0.copy(); best_c = _c5(best)
    # Symmetry variants
    for hh in [1.0 - h0.copy(), h0[::-1].copy()]:
        h2 = _proj(hh, n0 * 0.5)
        c = _c5(h2)
        if c < best_c:
            best = h2; best_c = c
    # Phase 1: Quick FISTA
    end1 = t0 + min(100.0, 0.10 * budget_s)
    hcur = best.copy()
    for a in [50, 200, 1000]:
        if time.time() >= end1 - 2:
            break
        tt = min(30.0, end1 - time.time() - 1)
        hf, bh, bc = _fista(hcur, a, tt, 20.0)
        cf = _c5(hf)
        if bc < best_c:
            best = bh; best_c = bc
        if cf < best_c:
            best = hf; best_c = cf
        hcur = hf if cf <= bc else bh
    # Phase 2: High-alpha FISTA
    end2 = t0 + min(350.0, 0.35 * budget_s)
    hcur = best.copy()
    alphas = [5000, 20000, 100000, 500000]
    tt_per = max(1.0, (end2 - time.time()) / len(alphas))
    for a in alphas:
        if time.time() >= end2 - 2:
            break
        hf, bh, bc = _fista(hcur, a, min(tt_per, end2 - time.time() - 1), 50.0)
        cf = _c5(hf)
        if bc < best_c:
            best = bh; best_c = bc
        if cf < best_c:
            best = hf; best_c = cf
        hcur = hf if cf <= bc else bh
    # Phase 3: Gradient momentum
    end3 = t0 + min(550.0, 0.55 * budget_s)
    if end3 > time.time() + 5:
        hcur, c3 = _grad_momentum(best, end3, n_active=3)
        if c3 < best_c:
            best = hcur; best_c = c3
    # Phase 4: Targeted SA
    end4 = t0 + min(820.0, 0.82 * budget_s)
    if end4 > time.time() + 5:
        best, best_c = _sa_targeted(best, rng_global, end4)
    # Phase 5: Fine tune
    end5 = t0 + budget_s - 30.0
    if end5 > time.time() + 3:
        best, best_c = _fine_tune(best, rng_global, end5)
    # Phase 6: Final SA polish
    end6 = t0 + budget_s - 3.0
    if end6 > time.time() + 2:
        best, best_c = _sa_targeted(best, rng_global, end6)
    # Final variants and repair
    for hh in [1.0 - best, best[::-1].copy()]:
        h2 = _proj(hh, len(hh) * 0.5)
        c = _c5(h2)
        if c < best_c:
            best = h2; best_c = c
    best = _proj(best, len(best) * 0.5)
    best_c = _c5(best)
    return best, best_c, len(best)
