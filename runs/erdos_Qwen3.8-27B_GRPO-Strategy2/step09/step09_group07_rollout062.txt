                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _cd_pass(h, rng, temp=0.0):

    c = _corr(h)
    cmax = 2 * n - 1
    jlist = np.arange(n)
    rng.shuffle(jlist)
    nm1 = n - 1
    for j in jlist:

        beta = np.zeros(cmax)
        k1_lo = j
        k1_hi = min(cmax - 1, j + nm1)
        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax - 1, cmax - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
        beta[k2_lo:k2_hi + 1] -= h[j + lags2]
        g = c - beta * h[j]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(50):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(g + beta * t1) <= np.max(g + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)

        if abs(t_new - h[j]) > 1e-14:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

def _pair_anneal(h, n, rng, n_it, T0, T1):

    c5c = best_c5
    for it in range(n_it):

            h[j] += d

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _cd_pass(h, rng, temp=0.0):
    n = len(h)
    c = _corr(h)
    cmax = 2 * n - 1
    jlist = np.arange(n)
    rng.shuffle(jlist)
    nm1 = n - 1
    for j in jlist:
        j = int(j)
        beta = np.zeros(cmax)
        k1_lo = j
        k1_hi = min(cmax - 1, j + nm1)
        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax - 1, cmax - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
        beta[k2_lo:k2_hi + 1] -= h[j + lags2]
        g = c - beta * h[j]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(50):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(g + beta * t1) <= np.max(g + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if temp > 0.0 and rng.random() < 0.3:
            t_new = np.clip(h[j] + temp * rng.randn(), 0.0, 1.0)
        if abs(t_new - h[j]) > 1e-14:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

def _top_grad(h, n, alpha, m):
    c = _corr(h)
    idx = np.argsort(c)[-m:]
    s = np.clip(alpha * (c[idx] - c.max()), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    dx_v = 2.0 / n
    for ii in range(len(idx)):
        wk = w[ii] * dx_v
        if wk < 1e-15:
            continue
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:
            g[L:] += wk * (1.0 - h[:n - L])
            g[:n - L] -= wk * h[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n - M]
    return g

def _loc_search(h, n, rng, n_it):

def _loc_search(h, n, rng, n_it):
    best_h = h.copy()
    best_c5 = _eval(best_h)
    c5c = best_c5
    for it in range(n_it):
        i = rng.randint(n)
        jj = (i + 1 + rng.randint(0, n - 1)) % n
        d = 0.004 * rng.randn()
        hi, hj = h[i] + d, h[jj] - d
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        h[i], h[jj] = hi, hj
        c5n = _eval(h)
        if c5n < c5c:
            c5c = c5n
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h.copy()
        else:
            h[i], h[jj] = hi - d, hj + d
    return best_h, best_c5

def _cd_block(h, rng, n, t_end, max_temp):
    best_h = h.copy()
    best_c5 = _eval(best_h)
    passes = 0
    while time.time() < t_end:
        frac = min(1.0, passes / 200.0)
        T = max_temp * (1.0 - frac) ** 2
        h = _cd_pass(h, rng, T)
        h = _repair(h, n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        passes += 1
        if passes % 60 == 0 and time.time() < t_end - 3.0:
            hp = h + 0.02 * rng.randn(n)
            hp = _repair(hp, n)
            for _ in range(2):
                hp = _cd_pass(hp, rng)
                hp = _repair(hp, n)
            c5p = _eval(hp)
            if c5p < best_c5:
                best_c5 = c5p
                best_h = hp.copy()
                h = hp.copy()
    return best_h, best_c5
