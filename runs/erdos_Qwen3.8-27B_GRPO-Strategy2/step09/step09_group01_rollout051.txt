                h = h_new; c = c_new; lstuck = 0
                if c < bc:
                    bc = c; bh = h.copy()
            else:
                lstuck += 1
                # random nudge
                if lstuck > 50:
                    i = rng.randint(n); j = rng.randint(n)
                    h[i] += step * 0.5 * (rng.random() - 0.5)
                    h[j] -= step * 0.5 * (rng.random() - 0.5)
                    h = _proj(h, S)
                    c = _c5(h)
                    lstuck = 0
    c = _c5(h)
    if c < bc:
        bc = c; bh = h.copy()
    return bh, bc

def _sa_bin(h0, dl, T0, T1, seed):
    rng = np.random.RandomState(seed)
    n = len(h0)
    t = n // 2
    h = np.zeros(n)
    h[np.argsort(h0)[::-1][:t]] = 1.0
    c = _c5(h)
    bc = c; bh = h.copy()
    t0 = time.time()
    stuck = 0
    while time.time() - t0 < dl:
        elapsed = time.time() - t0
        frac = min(1.0, elapsed / max(dl, 1e-9))
        T = T0 * (T1 / T0) ** frac
        pos1 = np.where(h == 1.0)[0]
        pos0 = np.where(h == 0.0)[0]
        i = pos1[rng.randint(len(pos1))]
        j = pos0[rng.randint(len(pos0))]
        h[i], h[j] = 0.0, 1.0
        c_new = _c5(h)
        d = c_new - c
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-15)):
            c = c_new; stuck = 0
            if c < bc:
                bc = c; bh = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
            stuck += 1
        if stuck > 8000:
            stuck = 0
            for _ in range(10):
                i = pos1[rng.randint(len(pos1))]
                j = pos0[rng.randint(len(pos0))]
                h[i], h[j] = 0.0, 1.0
                cn2 = _c5(h)
                if cn2 < c:
                    c = cn2
                    if c < bc:
                        bc = c; bh = h.copy()
                else:
                    h[i], h[j] = 1.0, 0.0
    return bh, bc

def _swap_ls(h0, dl, s0, seed):
    rng = np.random.RandomState(seed)
    n = len(h0)
    S = n * 0.5
    h = _repair(h0.copy(), n)
    c = _c5(h); bc = c; bh = h.copy()
    t0 = time.time()
    it = 0
    g = _agrad(h)
    recomp = 300
    while time.time() - t0 < dl:
        if it >= recomp:
            g = _agrad(h)
            recomp = it + 300
        elapsed = (time.time() - t0) / max(dl, 1e-9)
        step = s0 * max(1e-7, 0.05 ** elapsed)
        order = np.argsort(g)
        q = max(20, n // 6)
        i = order[rng.randint(q)]
        j = order[n - 1 - rng.randint(q)]
        if i == j:
            it += 1; continue
        sgn = 1.0 if g[i] <= g[j] else -1.0
        avail = min(1.0 - h[i], h[j]) if sgn > 0 else min(h[i], 1.0 - h[j])
        d = sgn * min(avail, step * (0.5 + rng.random() * 1.5))
        if abs(d) < 1e-15:
            it += 1; continue
        h[i] += d; h[j] -= d
        h = _proj(h, S)
        c_new = _c5(h)
        if c_new < c - 1e-15:
            c = c_new
            if c < bc:
                bc = c; bh = h.copy()
        else:
            h[i] -= d; h[j] += d
            h = _proj(h, S)
        it += 1
    c = _c5(h)
    if c < bc:
        bc = c; bh = h.copy()
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)

    best = _repair(h0, n0)
    best_c = _c5(best)

    # Symmetry
    for var in [1.0 - best, best[::-1]]:
        hv = _proj(var, n0 * 0.5)
        cv = _c5(hv)
        if cv < best_c:
            best, best_c = hv, cv

    # Try alt grid sizes
    for nn in [n0 - 12, n0 + 12]:
        if nn < 100 or nn > 1000 or (time.time() - t0) > 30:
            continue
        x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
        x1 = (np.arange(nn) + 0.5) * (2.0 / nn)
        ht = _proj(np.interp(x1, x0, best), nn * 0.5)
        ht, ct = _gdescent(ht, min(20.0, 40 - (time.time() - t0)), 2e-4, seed + 1)
        if ct < best_c:
            best, best_c = ht, ct

    n = len(best)

    # Phase 1: Gradient descent (100s)
    end1 = t0 + 100.0
    t_rem = budget_s - 5.0
    end1 = min(end1, t0 + 0.12 * t_rem)
    best, best_c = _gdescent(best, end1 - time.time(), 3e-5, seed + 100)

    # Phase 2: Binary SA coarse (250s)
    end2 = min(end1 + 250.0, t0 + 0.40 * t_rem)
    best, best_c = _sa_bin(best, end2 - time.time(), 0.025, 1e-5, seed + 200)

    # Phase 3: Gradient descent fine (150s)
    end3 = min(end2 + 150.0, t0 + 0.58 * t_rem)
    best, best_c = _gdescent(best, end3 - time.time(), 8e-6, seed + 300)

    # Phase 4: Swap local search (250s)
    end4 = min(end3 + 250.0, t0 + 0.82 * t_rem)
    best, best_c = _swap_ls(best, end4 - time.time(), 4e-5, seed + 400)

    # Phase 5: Binary SA fine (200s)
    end5 = min(end4 + 200.0, t0 + t_rem)
    best, best_c = _sa_bin(best, end5 - time.time(), 0.008, 1e-7, seed + 500)

    # Phase 6: Final gradient micro (remaining)
    end6 = t0 + budget_s - 5.0
    best, best_c = _gdescent(best, max(1.0, end6 - time.time()), 1e-6, seed + 600)

    # Final symmetry
    nf = len(best)
    for var in [1.0 - best, best[::-1]]:
        hv = _proj(var, nf * 0.5)
        cv = _c5(hv)
        if cv < best_c:
            best, best_c = hv, cv

    best = _repair(best, nf)
    best_c = _c5(best)
    return best, best_c, nf
