        idx = np.argsort(1.0-h) if d > 0 else np.argsort(h)
        for i in idx:

            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-14: break

    oh = 1.0 - h
    S1 = np.convolve(w, oh)[:n]
    S2 = np.convolve(w, h[::-1])[n-1::-1][:n]
    return (S1 - S2) / dx
