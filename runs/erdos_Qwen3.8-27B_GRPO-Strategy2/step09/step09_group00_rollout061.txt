def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full'))) * (2.0/n)

def _proj(v, S):
    x = np.clip(np.array(v, dtype=float, copy=True), 0.0, 1.0)
    for _ in range(100):
        d = S - x.sum()
        if abs(d) < 1e-14:
            break
        m = (x < 0.9999999) if d > 0 else (x > 1e-7)
        c = m.sum()
        if c == 0:
            break

        np.clip(x, 0.0, 1.0, out=x)
    return x

def _fft_size(n):
    N = 1
    while N < 2*n - 1:
        N <<= 1
    return N

def _pn(h):
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n+1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    O_pos = pref[n:] - ac
    O_neg = (pref[n] - pref[:n]) - ac

    wN = np.zeros(n)

import numpy as np

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _proj(v, S):
    x = np.clip(np.array(v, dtype=float, copy=True), 0.0, 1.0)
    for _ in range(80):
        d = S - x.sum()
        if abs(d) < 1e-14:
            break
        m = (x < 0.9999999) if d > 0 else (x > 1e-7)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
        np.clip(x, 0.0, 1.0, out=x)
    return x

def _fft_size(n):
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    return N

def _pn(h):
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    O_pos = pref[n:] - ac
    O_neg = (pref[n] - pref[:n]) - ac
    mx = max(float(O_pos.max()), float(O_neg.max()))
    return O_pos, O_neg, mx

        e += np.exp(np.clip(a * (O_neg[1:] - mx), -400, 0)).sum()
    return dx * (mx + np.log(e) / a)

def _grad_w(h, wP, wN):

    linP = np.zeros(n)

    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

    wN = np.zeros(n)

        en = np.exp(np.clip(a * (O_neg[1:] - mx), -400, 0))

        if tot < 1e-300:

            wP[0] = 1.0
        else:
            wP[0] = e0 / tot
            wP[1:] = ep / tot
            wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _grad_w(h, wP, wN)
