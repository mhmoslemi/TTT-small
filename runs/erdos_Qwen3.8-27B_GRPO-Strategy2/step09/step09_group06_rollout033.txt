                g[:n - L] -= wk * x[L:]
            else:
                M = -L
                g[:n - M] += wk * (1.0 - x[M:])
                g[M:] -= wk * x[:n - M]
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h, jac=_jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _local_search(h, n_iters, rng, base_scale):
    n = len(h)
    c5 = _eval(h)
    best_h = h.copy()
    best_c = c5
    for it in range(n_iters):
        ptype = it % 4
        h_new = h.copy()
        if ptype == 0:
            w = rng.randint(3, 25)
            st = rng.randint(0, n - w)
            h_new[st:st + w] += base_scale * rng.randn()
        elif ptype == 1:
            freq = rng.choice([2, 3, 5, 7, 11, 13])
            ph = rng.rand() * 6.2832
            h_new += base_scale * rng.randn() * np.sin(6.2832 * freq * np.arange(n) / n + ph)
        elif ptype == 2:
            k = rng.randint(3, 12)
            idx = rng.choice(n, size=k, replace=False)
            d = base_scale * rng.randn(k)
            d -= d.mean()
            h_new[idx] += d
        else:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 10)) % n
            d = base_scale * 0.6 * rng.randn()
            h_new[i] += d
            h_new[j] -= d
        h_new = _repair(h_new, n)
        c_new = _eval(h_new)
        if c_new < c5:
            c5 = c_new
            h = h_new
            if c5 < best_c:
                best_c = c5
                best_h = h.copy()
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    return _repair(np.repeat(h, f), len(h) * f)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _el():
        return time.time() - t0

    # Phase 1: Multi-resolution warm-up
    for ratio in [4, 8]:
        if _el() > 50:
            break
        nc = max(50, n // ratio)
        f = n // nc
        hc = _downsample(h, f)
        for alpha, ml, mi in [(5000, 10, 300), (50000, 20, 300)]:
            if _el() > 55:
                break
            hc = _slsqp_opt(hc, alpha, ml, mi)
        hc, _ = _proj_grad(hc, 12, 2, 500, 0.01, 0.55, 100)
        h_up = _upsample(hc, f)
        c5u = _eval(h_up)
        if c5u < best_c5:
            best_c5 = c5u
            best_h = h_up.copy()

    # Phase 2: Main projected gradient (large budget)
    if _el() < budget_s - 350:
        rounds = [
            (40, 8, 1500, 0.006, 0.55, 150),
            (25, 5, 2000, 0.005, 0.60, 200),
            (15, 3, 2500, 0.004, 0.65, 200),
            (8, 2, 3000, 0.003, 0.70, 250),
            (3, 1, 4000, 0.002, 0.78, 200),
            (2, 1, 4000, 0.001, 0.85, 300),
        ]
        for m_s, m_e, steps, s0, mu, re in rounds:
            if _el() > budget_s - 300:
                break
            h_new, c_new = _proj_grad(best_h.copy(), m_s, m_e, steps, s0, mu, re)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()

    # Phase 3: SLSQP polish
    if _el() < budget_s - 150:
        for alpha, ml, mi in [(200000, 25, 400), (2000000, 40, 350), (20000000, 60, 300)]:
            if _el() > budget_s - 100:
                break
            h_new = _slsqp_opt(best_h, alpha, ml, mi)
            c_new = _eval(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()

    # Phase 4: Another projected gradient pass at m=1
    if _el() < budget_s - 100:
        h_new, c_new = _proj_grad(best_h.copy(), 1, 1, 3000, 0.001, 0.9, 400)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 5: Structured local search
    if _el() < budget_s - 50:
        rem = budget_s - 50 - _el()
        ntries = max(100, int(rem / 0.0025))
        best_h, best_c5 = _local_search(best_h, ntries, rng, 1.5e-5)

    # Phase 6: Ultra-fine 2-point polish
    if _el() < budget_s - 2:
        h2 = best_h.copy()
        nc2 = len(h2)
        c5c = _eval(h2)
        npol = int(max(100, (budget_s - 2 - _el()) / 0.0004))
        for _ in range(npol):
            if _el() > budget_s - 2:
                break
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 6)) % nc2
            d = 3e-8 * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if -0.01 <= ni <= 1.01 and -0.01 <= nj <= 1.01:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
