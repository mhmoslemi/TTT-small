def _repair(h, n):
    # Ensure h is in [0,1] and sum(h) = n/2
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        diff = target - s
        # Distribute diff to entries not at boundary
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            avail = 1.0 - h[mask]
            # Proportional distribution
            amt = min(diff, avail.sum())
            h[mask] += amt * avail / avail.sum()
        else:
            mask = h > 0.0
            if not mask.any():
                break
            avail = h[mask]
            amt = min(-diff, avail.sum())
            h[mask] -= amt * avail / avail.sum()
    # Final exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-15:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

    n = len(h)

    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_exact(h, top_lags):
    """Compute gradient of max c_k using only top lags (weighted subgradient)"""
    n = len(h)
    dx = 2.0 / n
    c = _corr_full(h)
    # Get top lags
    idx = np.argsort(c)[-top_lags:]
    g = np.zeros(n)
    for k in idx:
        L = k - (n - 1)
        if L >= 0:
            g[L:] += dx * (1.0 - h[:n - L])
            g[:n - L] -= dx * h[L:]
        else:
            M = -L
            g[:n - M] += dx * (1.0 - h[M:])
            g[M:] -= dx * h[:n - M]
    # Remove mean to respect sum constraint
    g -= g.mean()
    return g, c.max()

def _slsqp_smooth(h, alpha, m_lags, maxiter, ftol):

        m = c.max()
        idx = np.argsort(c)[-m_lags:]
        s = np.clip(alpha * (c[idx] - m), -500, 500)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        idx = np.argsort(c)[-m_lags:]
        s = np.clip(alpha * (c[idx] - m), -500, 500)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii, k in enumerate(idx):
            if w[ii] < 1e-15:
                continue
            wk = w[ii] * dx
            L = k - (n - 1)
            if L >= 0:
                g[L:] += wk * (1.0 - x[:n - L])
                g[:n - L] -= wk * x[L:]
            else:
                M = -L
                g[:n - M] += wk * (1.0 - x[M:])
                g[M:] -= wk * x[:n - M]
        return g
    
    bnds = [(0.0, 1.0)] * n

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        diff = target - s
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            avail = 1.0 - h[mask]
            amt = min(diff, avail.sum())
            h[mask] += amt * avail / avail.sum()
        else:
            mask = h > 0.0
            if not mask.any():
                break
            avail = h[mask]
            amt = min(-diff, avail.sum())
            h[mask] -= amt * avail / avail.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-15:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _obj_smooth(x, alpha, ml):
    c = _corr_full(x)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _grad_smooth(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = _corr_full(x)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _slsqp(h0, alpha, ml, maxiter, ftol):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    def _cf(x):
        return x.sum() - n * 0.5
    def _cg(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cg}]
    try:
        res = minimize(_obj_smooth, h0, jac=_grad_smooth,
                       args=(alpha, ml), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h0

def _proj_c(x, target):
    lo, hi = -10.0, 10.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def _nesterov_loop(h0, n_steps, step0, top_k, mu):
    n = len(h0)
    target = n * 0.5
    dx = 2.0 / n
    x = h0.copy()
    v = np.zeros(n)
    cur_c = _eval(x)
    step = step0
    for _ in range(n_steps):
        x_n = x - mu * v
        c = _corr_full(x_n)
        m = c.max()
        idx = np.argsort(c)[-top_k:]
        g = np.zeros(n)
        for k in idx:
            L = k - (n - 1)
            w = dx / top_k
            if L >= 0:
                g[L:] += w * (1.0 - x_n[:n - L])
                g[:n - L] -= w * x_n[L:]
            else:
                M = -L
                g[:n - M] += w * (1.0 - x_n[M:])
                g[M:] -= w * x_n[:n - M]
        g -= g.mean()
        taken = False
        for _t in range(12):
            v_new = x_n - step * g
            hp = _proj_c(v_new, target)
            c_new = _eval(hp)
            if c_new < cur_c - 1e-18:
                v = hp - x
                x = hp
                cur_c = c_new
                step = min(step * 1.04, step0 * 40.0)
                taken = True
                break
            step *= 0.38
        if not taken:

            v = _proj_c(x - v, target) - x
            step = max(step * 1.3, step0 * 0.3)
    return _repair(x, n), cur_c

    c5c = _eval(h)
    best_h = h.copy()

    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 25)) % n
        d = (scale_lo + (scale_hi - scale_lo) * rng.rand()) * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-17:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _group_polish(h, n_iter, rng, gs=4, scale=3e-8):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        base = rng.randint(n)
        idx = np.sort((base + rng.randint(0, 35, size=gs)) % n)
        d = scale * (0.5 + rng.rand()) * rng.randn(gs)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < 0 or h_new.max() > 1:

        c5n = _eval(h_new)
        if c5n < c5c - 1e-17:
            c5c = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def _multi_restart(h0, rng, n_rest, time_budget, alpha, ml, maxiter):
