```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _subgrad(h, m_lags=20):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    g = np.zeros(n)
    top = np.argsort(c)[-m_lags:]
    nm1 = n - 1
    for k in top:
        m = int(k) - nm1
        if m >= 0:
            g[m:] += (1.0 - h[:n-m])
            g[:n-m] -= h[m:]
        else:
            mm = -m
            g[:n-mm] += (1.0 - h[mm:])
            g[mm:] -= h[:n-mm]
    g *= dx / m_lags
    return g, c

def _cg_step(h, lr, m_lags):
    g, c = _subgrad(h, m_lags)
    h_new = np.clip(h - lr * g, 0.0, 1.0)
    h_new = _repair(h_new, len(h))
    return h_new, c
```

```python
def _smooth_max_grad(h, alpha, m_lags):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(c)[-m_lags:]
    vals = c[top_idx]
    # Softmax weights
    max_val = vals.max()
    exps = np.exp(alpha * (vals - max_val))
    w = exps / exps.sum()
    # Weighted gradient
    g = np.zeros(n)
    nm1 = n - 1
    for ii, k in enumerate(top_idx):
        if w[ii] < 1e-15:
            continue
        m = int(k) - nm1
        if m >= 0:
            g[m:] += w[ii] * (1.0 - h[:n-m])
            g[:n-m] -= w[ii] * h[m:]
        else:
            mm = -m
            g[:n-mm] += w[ii] * (1.0 - h[mm:])
            g[mm:] -= w[ii] * h[:n-mm]
    f = (1.0/alpha) * (np.log(np.exp(alpha*(vals-max_val)).sum()) + alpha*max_val)
    return g * dx, f, c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _cd_pass(h, rng, ls_iters=22):
    n = len(h)
    c = _corr(h)
    cmax = 2 * n - 1
    jlist = np.arange(n)
    rng.shuffle(jlist)
    nm1 = n - 1
    for j in jlist:
        j = int(j)
        beta = np.zeros(cmax)
        k1_lo = j
        k1_hi = min(cmax - 1, j + nm1)
        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax - 1, cmax - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
        beta[k2_lo:k2_hi + 1] -= h[j + lags2]
        gvec = c - beta * h[j]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(ls_iters):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(gvec + beta * t1) <= np.max(gvec + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-12:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

def _sm_max(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    top = np.argpartition(c, -m_lags)[-m_lags:]
    v = c[top]
    mv = v.max()
    e = np.exp(np.clip(alpha * (v - mv), -500, 500))
    f = mv + np.log(e.sum()) / alpha
    w = e / e.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(len(top)):
        wi = w[ii]
        if wi < 1e-15:
            continue
        k = int(top[ii])
        m = k - nm1
        if m >= 0:
            g[m:] += wi * (1.0 - x[:n - m])
            g[:n - m] -= wi * x[m:]
        else:
            mm = -m
            g[:n - mm] += wi * (1.0 - x[mm:])
            g[mm:] -= wi * x[:n - mm]
    g *= dx
    return f, g

def _lbgfsm(h0, alpha, m_lags, pen, maxiter):
    n = len(h0)
    tgt = n * 0.5

    def _ofg(x):
        f_sw, g_sw = _sm_max(x, alpha, m_lags)
        se = x.sum() - tgt
        f = f_sw + pen * se * se
        g = g_sw + 2.0 * pen * se
        return f, g

    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_ofg, h0, jac=True, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return _repair(res.x, n)
    except Exception:
        return h0.copy()

def _psgd(h0, n, time_limit, m_lags, alpha):
    h = h0.copy()
    best_c5 = _eval(h)
    best = h.copy()
    lr = 0.03
    stuck = 0
    t_s = time.time()
    step = 0
    while time.time() - t_s < time_limit:
        _f, g = _sm_max(h, alpha, m_lags)
        h_new = np.clip(h - lr * g, 0.0, 1.0)
        h_new = _repair(h_new, n)
        c5 = _eval(h_new)
        c5_old = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h_new.copy()
            stuck = 0
        if c5 <= c5_old + 1e-10:
            h = h_new
        else:
            stuck += 1
            if stuck > 40:
                lr *= 0.85
                stuck = 0
        step += 1
        if step % 500 == 0:
            lr *= 0.97
    return best, best_c5

def _local_search(h, n, rng, n_iter):
    c5c = _eval(h)
    best_h = h.copy()
    best_c5 = c5c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        d = 3e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

                best_h = hl2

    # Phase 3: CD intensive (25%)
    p3_end = t0 + budget_s * 0.65
    h = best_h.copy()
    while time.time() < p3_end:
        h = _cd_pass(h, rng)
        h = _repair(h, n)
        c5 = _eval(h)
        if c5 < best_c5:

            best_h = h.copy()

            best_h = bp

        c5 = _eval(hl)
        if c5 < best_c5:

    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
```
