def _repair(h, n):

    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                h[idx] += a
                diff -= a

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_vals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _proj_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, k, n):
    dx = 2.0 / n
    g = np.zeros(n)
    L = k - (n - 1)
    if L >= 0:
        g[L:] += dx * (1.0 - h[:n - L])
        g[:n - L] -= dx * h[L:]
    else:
        M = -L
        g[:n - M] += dx * (1.0 - h[M:])
        g[M:] -= dx * h[:n - M]
    return g

def _smooth_obj(x, alpha, ml):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, ml):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        g += w[ii] * _grad_lag(x, k, n)
    return g

def _csum(x):
    return x.sum() - len(x) * 0.5

def _cjac(x):
    return np.ones(len(x))

def _do_slsqp(h, alpha, ml, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _csum, 'jac': _cjac}]
    try:
        r = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha, ml),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return h

def _do_lbfgs(h, alpha, ml, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _csum, 'jac': _cjac}]
    try:
        r = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha, ml),
                     method='L-BFGS-B', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'maxfun': maxiter * 3})
        return _repair(r.x, n)
    except Exception:
        return h

def _subgrad_descent(h, n_iters, step0, top_m, t0, deadline):
    n = len(h)
    cur = _eval(h)
    best_h, best_c = h.copy(), cur
    step = step0
    for _ in range(n_iters):
        if time.time() > deadline:
            break
        c = _corr_vals(h)
        idx_top = np.argsort(c)[-top_m:]
        g = np.zeros(n)
        for k in idx_top:
            g += _grad_lag(h, k, n)
        g /= top_m
        taken = False
        for _ in range(12):

            cn = _eval(hn)
            if cn < cur - 1e-15:

                step = min(step * 1.06, step0 * 25.0)
                taken = True

        if not taken:

        k = idx_top[rng.randint(top_m)]
        L = k - (n - 1)
        if L >= 0:
            nvalid = n - L
        else:
            nvalid = n + L
        i = rng.randint(nvalid)
        if L >= 0:
            ii, jj = i + L, i
        else:
            ii, jj = i, i - L
        d = 2e-5 * rng.randn()
        ni, nj = h[ii] + d, h[jj] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[ii], h[jj] = ni, nj
            cn = _eval(h)
            if cn < cur:
                cur = cn
                if cn < best_c:
                    best_c, best_h = cn, h.copy()
            else:
                h[ii], h[jj] = ni - d, nj + d
    return best_h, best_c

def _local_perturb(h, nit, rng, scale, npert, t0, deadline):
