            d = 3e-5 * rng.randn()
        else:
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = h[j] - h[i]
            h[i], h[j] = h[j], h[i]
            c5n = _eval(h)
            if c5n < c5_cur:
                c5_cur = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
                    g = _smooth_grad(h, alpha, ml)
            else:
                h[i], h[j] = h[j], h[i]
            continue
        h_new = h.copy()
        h_new[i] += d
        h_new[j] -= d
        if h_new[i] < -1e-10 or h_new[i] > 1 + 1e-10 or h_new[j] < -1e-10 or h_new[j] > 1 + 1e-10:
            continue
        h_new[i] = np.clip(h_new[i], 0, 1)
        h_new[j] = np.clip(h_new[j], 0, 1)
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5_cur:
            h = h_new
            c5_cur = c5n
            g = _smooth_grad(h, alpha, ml)
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return _repair(out, n * f)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    def _remaining():
        return budget_s - 30 - _el()

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Multi-resolution warm-up
    for ratio in [4, 6, 8]:
        if _remaining() < 200:
            break
        nc = max(32, n // ratio)
        hc = _downsample(h, n // nc)
        for a in [20000, 100000, 500000]:
            if _remaining() < 180:
                break
            hc = _slsqp_opt(hc, a, 15, 400, 1e-15)
        h_up = _upsample(hc, n // nc)
        c_up = _eval(h_up)
        if c_up < best_c5:
            best_c5 = c_up
            best_h = h_up.copy()

    # Phase 2: SLSQP alpha sweep
    alpha_seq = [5000, 20000, 100000, 500000, 2000000, 10000000, 50000000, 200000000]
    for alpha in alpha_seq:
        if _remaining() < 300:
            break
        ml = min(100, max(20, int(25 * np.log10(alpha / 5000.0) + 25)))
        h_new = _slsqp_opt(best_h, alpha, ml, 3000, 1e-16)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 3: Adam optimization
    if _remaining() > 300:
        for alpha, steps, lr in [(1000000, 500, 0.01), (10000000, 400, 0.005),
                                  (100000000, 300, 0.002), (1000000000, 250, 0.001)]:
            if _remaining() < 200:
                break
            ml = min(150, max(30, int(20 * np.log10(alpha / 1e6 + 1) + 40)))
            h_new = _adam_opt(best_h.copy(), alpha, ml, steps, lr, lambda: _remaining() - 200 - 0)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 4: High-alpha SLSQP squeeze
    if _remaining() > 150:
        for alpha in [5e8, 2e9, 5e9, 2e10]:
            if _remaining() < 60:
                break
            h_new = _slsqp_opt(best_h, alpha, 200, 2000, 1e-18)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 5: Reheat + re-optimize
    if _remaining() > 100:
        for attempt in range(8):
            if _remaining() < 40:
                break
            noise = 5e-5 * (9 - attempt) * 0.5
            h_p = best_h + noise * rng.randn(n)
            h_p = _repair(h_p, n)
            h_new = _slsqp_opt(h_p, 1e8, 120, 800, 1e-15)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 6: Gradient-informed local search
    if _remaining() > 20:
        nt = int(_remaining() * 0.7 / 0.001)
        nt = min(nt, 1200000)
        best_h, best_c5 = _grad_search(best_h, nt, rng, 5e8, 200,
                                        lambda: _remaining() * 0.7 - 10)

    # Phase 7: Ultra-fine 2-point polish
    if _remaining() > 5:
        h2 = best_h.copy()
        nc2 = len(h2)
        c5c = _eval(h2)
        c5b = best_c5
        bh2 = best_h.copy()
        n_iter = int((_remaining() - 2) / 0.0005)
        n_iter = min(n_iter, 800000)
        for _ in range(n_iter):
            if _remaining() < 2:
                break
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 12)) % nc2
            d = 1e-7 * rng.randn() * 20
            ni, nj = h2[i] + d, h2[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < c5b:
                        c5b = c5n
                        bh2 = h2.copy()
                else:
                    h2[i], h2[j] = ni - d, nj + d
        if c5b < best_c5:
            best_c5 = c5b
            best_h = bh2

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
