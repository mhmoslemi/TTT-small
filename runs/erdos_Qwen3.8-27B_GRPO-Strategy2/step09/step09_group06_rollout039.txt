def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)

        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-15:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _soft_obj(x, alpha, m, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s - s.max(), -500, 500)
    return c.max() + np.log(np.exp(s).sum()) / alpha

def _soft_grad(x, alpha, m, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s - s.max(), -500, 500)
    w = np.exp(s)

    dx = 2.0 / n

                       method='SLSQP', bounds=bnds, constraints=cons,
