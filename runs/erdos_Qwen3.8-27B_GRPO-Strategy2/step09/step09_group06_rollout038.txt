                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sum_fn(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _obj_fn(x, alpha, m, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s - s.max(), -600, 600)
    return c.max() + np.log(np.exp(s).sum()) / alpha

def _grad_fn(x, alpha, m, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s - s.max(), -600, 600)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii]
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g * dx

def _slsqp(h, alpha, m, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_obj_fn, h, jac=_grad_fn, args=(alpha, m, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _grad_step(h, alpha, m, nsteps, step0, mom):
    n = len(h)
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    for _ in range(nsteps):
        g = _grad_fn(h, alpha, m, n)
        v = mom * v + (1.0 - mom) * g
        taken = False
        for _t in range(15):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-16:
                h = h_new
                cur_c = c_new
                step = min(step * 1.03, step0 * 20.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.5, step0 * 20.0)
    return _repair(h, n)

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    return _repair(np.repeat(h, f), len(h) * f)

def _local_search(h, n_iter, rng, base_step):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    stuck = 0
    temp = 1e-4
    for it in range(n_iter):
        frac = it / max(n_iter, 1)
        temp = 1e-4 * (1.0 - frac) + 1e-7
        step = base_step * (2.0 - 1.5 * frac) * (0.5 + rng.rand())
        k = rng.choice([2, 3, 4, 5, 6, 8, 10])
        idx = rng.randint(0, n, size=k)
        d = step * rng.randn(k)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < -0.1 or h_new.max() > 1.1:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        delta = c5n - c5cur
        if delta < 0 or rng.rand() < np.exp(-delta / temp):
            h = h_new
            c5cur = c5n
            stuck = 0
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        else:
            stuck += 1
    return best_h, best_c

def _fourier_refine(h, n_modes, n_iter, rng):
    n = len(h)
    x = np.arange(n, dtype=float)
    best_h = h.copy()
    best_c = _eval(best_h)
    for it in range(n_iter):
        mode = rng.randint(1, n_modes + 1)
        phase = rng.rand() * 2.0 * np.pi
        amp = rng.uniform(0.002, 0.08) * (1.0 - 0.5 * it / n_iter)
        d = amp * np.cos(2.0 * np.pi * mode * x / n + phase)
        d -= d.mean()
        h_try = _repair(h + d, n)
        c_new = _eval(h_try)
        if c_new < best_c:
            h = h_try
            best_h = h.copy()
            best_c = c_new
    return best_h, best_c
