import numpy as np
import time
from scipy.optimize import minimize, differential_evolution


def _proj(h, target):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if target <= 0.0:
        return np.zeros(n)
    if target >= float(n):
        return np.ones(n)
    lo = float(np.min(h)) - 10.0
    hi = float(np.max(h)) + 10.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def _repair(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    target = n * 0.5
    s = float(h.sum())
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(120):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = float(h.sum())
        if abs(s - target) < 1e-13:
            break
    diff = target - float(h.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h


def _cvec(h):
    return np.correlate(h, 1.0 - h, mode="full")


def _eval(h):
    n = len(h)
    return float(np.max(_cvec(h)) * (2.0 / n))


def _soft_info(x, alpha, ml):
    c = _cvec(x)
    mval = float(c.max())
    lz = c.size
    if ml < lz:
        if ml <= 0:
            return mval, np.array([], dtype=int), np.array([], dtype=float), 0.0
        idx = np.argpartition(-c, ml - 1)[:ml]
    else:
        idx = np.arange(lz)
    s = alpha * (c[idx] - mval)
    smax = float(s.max())
    e = np.exp(s - smax)
    se = float(e.sum())
    if se <= 0.0:
        w = np.full(idx.size, 1.0 / idx.size)
        logsum = smax
    else:
        w = e / se
        logsum = smax + np.log(se)
    return mval, idx, w, float(logsum)


def _grad_lags(x, idx, w):
    n = len(x)
    off = n - 1
    g = np.zeros(n, dtype=float)
    for ii in range(idx.size):
        wk = float(w[ii])
        if wk <= 1e-16:
            continue
        L = int(idx[ii]) - off
        if L == 0:
            g += wk * (1.0 - 2.0 * x)
        else:
            M = -L if L < 0 else L
            mlen = n - M
            if L > 0:
                g[M:] += wk * (1.0 - x[:mlen])
                g[:mlen] -= wk * x[M:]
            else:
                g[:mlen] += wk * (1.0 - x[M:])
                g[M:] -= wk * x[:mlen]
    return g


def _smooth_obj(x, alpha, ml):
    mval, idx, w, logsum = _soft_info(x, alpha, ml)
    return float(mval + logsum / alpha)


def _smooth_grad(x, alpha, ml):
    mval, idx, w, logsum = _soft_info(x, alpha, ml)
    return _grad_lags(x, idx, w)


def _eq_sum(u, target):
    return float(u.sum() - target)


def _eq_sum_jac(u):
    return np.ones(len(u), dtype=float)


def _slsqp_full(h, alpha, ml, maxiter, ftol):
    n = len(h)
    target = n * 0.5
    h = np.asarray(h, dtype=float).copy()
    bnds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _eq_sum, "jac": _eq_sum_jac, "args": (target,)}]
    try:
        res = minimize(
            _smooth_obj, h, jac=_smooth_grad, args=(alpha, ml),
            method="SLSQP", bounds=bnds, constraints=cons,
            options={"maxiter": maxiter, "ftol": ftol, "maxls": 60}
        )
        if res.x is not None:
            return _proj(res.x, target)
    except Exception:
        pass
    return h


def _block_eq(u, target):
    return float(u.sum() - target)


def _block_eq_jac(u):
    return np.ones(len(u), dtype=float)


def _block_obj(u, base, idx, alpha, ml):
    full = base.copy()
    full[idx] = u
    return _smooth_obj(full, alpha, ml)


def _block_grad(u, base, idx, alpha, ml):
    full = base.copy()
    full[idx] = u
    return _smooth_grad(full, alpha, ml)[idx]


def _viol_grad(x, T):
    c = _cvec(x)
    v = c - T
    act = v > 1e-14
    if np.logical_and(act.sum() > 0, v[act].sum() > 1e-12):
        idx = np.where(act)[0]
        w = v[idx] / float(v[idx].sum())
        return _grad_lags(x, idx, w)
    m = float(c.max())
    ntop = min(32, c.size)
    idx = np.argpartition(-c, ntop - 1)[:ntop]
    diff = c[idx] - m
    if float((diff - diff.max())) > -1e-12:
        w = np.ones(ntop, dtype=float) / ntop
    else:
        w = np.exp(8.0 * diff)
        w /= w.sum()
    return _grad_lags(x, idx, w)


def _block_optimize(h, rng, K, alpha, ml, maxiter, mix):

    target = n * 0.5
    base = _repair(h, n)
    c = _cvec(base)
    R = float(c.max())

    if alpha < 50.0:
        _, sidx, sw, _ = _soft_info(base, alpha, 192)
        g = _grad_lags(base, sidx, sw)
    else:
        T = R - max(0.5, 0.004 * R)
        g = _viol_grad(base, T)

    K = int(min(K, n))
    scores = np.abs(g - g.mean())
    idx = np.argsort(-scores)[:K].copy()
    if mix > 0 and K > 1:
        rr = int(round(mix * K))
        for _ in range(rr):
            idx[rng.randint(K)] = rng.randint(n)
    idx = np.unique(idx)
    if idx.size < 2:

    basec = _eval(base)
    tf = target - (base.sum() - base[idx].sum())
    if tf < -1e-7 or tf > idx.size + 1e-7:
        return base, basec
    tf = float(np.clip(tf, 0.0, idx.size))

    x0 = base[idx].copy()
    bnds = [(0.0, 1.0)] * idx.size
    cons = [{"type": "eq", "fun": _block_eq, "jac": _block_eq_jac, "args": (tf,)}]
    best = base

            cand[idx] = res.x

            cand = _repair(cand, n)
            cc = _eval(cand)
            if cc < bestc:
                best = cand
                bestc = cc
    except Exception:
        pass
    return best, bestc


def _line_search(cur, d, step, target, base_c, maxtries=10):
    best = cur.copy()
    bestc = base_c
    st = step
    for _ in range(maxtries):
        cand = _proj(cur + st * d, target)
        c = _eval(cand)
        if c < bestc:
            best = cand
            bestc = float(c)
        st *= 0.5
        if st < 1e-12:
            break
    return best, bestc


def _gradient_local(h, end_time, step0, gap0):

    target = n * 0.5

    cc = _eval(cur)
    best = cur.copy()
    bc = cc
    step = float(step0)
    gap = float(max(gap0, 1e-6))
    stall = 0
    while time.time() < end_time:

        d = -(g - g.mean())
        norm = float(np.linalg.norm(d))
        if norm < 1e-12:
            break
        d /= norm
        cand, c2 = _line_search(cur, d, step, target, cc, 10)

def _pair_local(h, rng, end_time, scale):
    n = len(h)
    cur = _repair(h, n)
    cc = _eval(cur)
    best = cur.copy()
    bc = cc
    scale = float(scale)
    while time.time() < end_time:
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            j = (i + 1) % n
        d = scale * float(rng.randn())
        a = cur[i] + d
        b = cur[j] - d
        if a < 0.0 or a > 1.0 or b < 0.0 or b > 1.0:
            continue
        cur[i] = a
        cur[j] = b
        c = _eval(cur)
        if c < cc:
            cc = c
            if c < bc:
                bc = c
                best = cur.copy()
        else:
            cur[i] = a - d
            cur[j] = b + d
    return best, bc


def _resample(h, m):
    n = len(h)
    if m == n:
        return _proj(h, n * 0.5)
    old = np.linspace(0.0, 2.0, n, endpoint=False)
    new = np.linspace(0.0, 2.0, m, endpoint=False)
    y = np.interp(new, old, h)
    return _proj(y, m * 0.5)


def _sym(h):
    n = len(h)
    s = 0.5 * (h + h[::-1])
    return _proj(s, n * 0.5)
