import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_one(h, L):
    """Gradient of correlation at lag L (0-indexed, where L=0 means no shift)."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if L >= 0:
        g[:n-L] += (1.0 - h[L:])
        g[L:] -= h[:n-L]
    else:
        M = -L
        g[:n-M] += (1.0 - h[M:])
        g[M:] -= h[:n-M]
    return g * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _topk_grad(h, k):
    """Gradient of sum of top-k correlations."""
    n = len(h)
    c = _corr(h)
    idx = np.argpartition(c, -k)[-k:]
    g = np.zeros(n)
    for ki in idx:
        L = ki - (n - 1)
        g += _grad_one(h, L)
    return g / k

def _run_slsqp(h, alpha, ml, maxiter):
    n = len(h)
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        idx = np.argsort(c)[-ml:]
        s = alpha * (c[idx] - m)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        idx = np.argsort(c)[-ml:]
        s = alpha * (c[idx] - m)
        s -= s.max()
        w = np.exp(s); w /= w.sum()
        g = np.zeros(n)
        for ii, k in enumerate(idx):
            if w[ii] < 1e-15:
                continue
            wk = w[ii] * dx
            L = k - (n - 1)
            if L >= 0:
                g[L:] += wk * (1.0 - x[:n - L])
                g[:n - L] -= wk * x[L:]
            else:
                M = -L
                g[:n - M] += wk * (1.0 - x[M:])
                g[M:] -= wk * x[:n - M]
        return g
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5, 
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(r.x, n)
    except:
        return h

def _topk_descent(h, k, nsteps, base_step, rng):
    """Projected gradient descent on top-k mean objective."""
    n = len(h)
    cur = _eval(h)
    v = np.zeros(n)
    step = base_step
    for _ in range(nsteps):
        g = _topk_grad(h, k)
        v = 0.8 * v + (1 - 0.8) * g  # momentum
        taken = False
        for _ in range(10):
            hn = _proj(h - step * v, n * 0.5)
            cn = _eval(hn)
            if cn < cur:
                h, cur, taken = hn, cn, True
                step = min(step * 1.05, base_step * 20)
                break
            step *= 0.5
        if not taken:
            step = base_step
    return h, cur

    cur_c5 = _eval(h)

    for it in range(niter):

```python
def _build_grad_matrix(h, n):
    """Build G[k, j] = dc_k/dh_j for all k, j."""
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        L = k - (n-1)
        if L >= 0:
            G[k, :n-L] += (1.0 - h[L:])
            G[k, L:] -= h[:n-L]
        else:
            M = -L
            G[k, :n-M] += (1.0 - h[M:])
            G[k, M:] -= h[:n-M]
    return G * dx
```

Wait, but G depends on h! When h changes, G changes. So I can't precompute it once.

However, for a small perturbation d: h_new = h + d*e_i - d*e_j

For the O(d²) term: c_k = dx * sum h[a]*(1-h[b]) where (a,b) are the pairs.
The second-order term involves products of the changes, which are O(d²). For d ~ 1e-4, d² ~ 1e-8, which is negligible compared to the first-order term.

So for small perturbations, I can use:
c_k_new ≈ c_k + d * (G[k,i] - G[k,j])

This is a linear function of d! So the optimal d for a given (i,j) pair can be found analytically:
- We want to minimize max_k (c_k + d * g_k) where g_k = G[k,i] - G[k,j]
- This is a 1D min-max problem
