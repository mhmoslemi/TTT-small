        improved = False
        for base_step in [5e-5, 1e-5, 2e-6, 5e-7, 1e-7]:
            for sign in [1, -1]:
                d = sign * base_step
                ni = best[i] + d
                nj = best[j] - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                best[i] = ni
                best[j] = nj
                c5n = _eval(best)
                if c5n < best_c - 1e-16:
                    best_c = c5n
                    improved = True
                    break
                else:
                    best[i] = ni - d
                    best[j] = nj + d
            if improved:
                break
        # Also try a few random pairs
        for _ in range(3):
            a, b = rng.randint(0, n, 2)
            if a == b:
                continue
            for base_step in [1e-5, 1e-6, 1e-7]:
                d = base_step * rng.randn()
                na = best[a] + d
                nb = best[b] - d
                if na < 0 or na > 1 or nb < 0 or nb > 1:
                    continue
                best[a] = na
                best[b] = nb
                c5n = _eval(best)
                if c5n < best_c - 1e-16:
                    best_c = c5n
                    break
                else:
                    best[a] = na - d
                    best[b] = nb + d
        # Periodic wider moves
        if count % 1000 == 0:
            for _ in range(5):
                a, b = rng.randint(0, n, 2)
                if a == b:
                    continue
                d = 1e-4 * rng.randn()
                na = best[a] + d
                nb = best[b] - d
                if na < 0 or na > 1 or nb < 0 or nb > 1:
                    continue
                best[a] = na
                best[b] = nb
                c5n = _eval(best)
                if c5n < best_c - 1e-16:
                    best_c = c5n
                else:
                    best[a] = na - d
                    best[b] = nb + d
    return best, best_c

def _proj_grad_step(h, alpha, m, nsteps, step0, budget_end):
    n = len(h)
    v = np.zeros(n)
    mu = 0.5
    step = step0
    cur_c = _eval(h)
    for _ in range(nsteps):
        if time.time() > budget_end:
            break
        g = _obj_grad(h, alpha, m, n)
        v = mu * v + (1.0 - mu) * g
        taken = False
        for _t in range(10):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-15:
                h = h_new
                cur_c = c_new
                step = min(step * 1.1, step0 * 20.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.5, step0 * 20.0)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _el():
        return time.time() - t0

    # Phase 1: Multi-start exploration (60s)
    starts = [h.copy()]
    # Perturbed start
    p = h.copy() + rng.randn(n) * 0.05
    starts.append(_repair(p, n))
    # Sinusoidal modulation
    s2 = h.copy()
    s2 += 0.1 * np.sin(np.pi * np.arange(n) / n)
    starts.append(_repair(s2, n))

    for hs in starts:
        if _el() > 50:
            break
        for alpha, m, mi in [(50000, 50, 300), (500000, 80, 200)]:
            if _el() > 55:
                break
            hs = _slsqp_opt(hs, alpha, m, mi, n)
            c5n = _eval(hs)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = hs.copy()

    # Phase 2: SLSQP sweep (200s)
    alpha_seq = [(1000000, 80, 400), (5000000, 100, 350),
                 (20000000, 120, 300), (100000000, 150, 250),
                 (500000000, 200, 200), (2000000000, 250, 150)]
    for alpha, m, mi in alpha_seq:
        if _el() > 250:
            break
        h_new = _slsqp_opt(best_h, alpha, m, mi, n)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 3: Projected gradient (100s)
    if _el() < budget_s - 500:
        for alpha, m, steps, s0 in [(2000000000, 100, 500, 0.001),
                                     (10000000000, 150, 300, 0.0005)]:
            if _el() > budget_s - 450:
                break
            h_new = _proj_grad_step(best_h.copy(), alpha, m, steps, s0, budget_s - 100)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 4: Simulated annealing (300s)
    if _el() < budget_s - 400:
        sa_end = min(budget_s - 300, _el() + 250)
        n_iter = 200000
        best_h, best_c5 = _sim_anneal(best_h, n_iter, rng, 0.05, 0.001, sa_end)

    # Phase 5: Quick SLSQP after SA
    if _el() < budget_s - 250:
        for alpha, m, mi in [(5000000000, 200, 200), (50000000000, 300, 100)]:
            if _el() > budget_s - 100:
                break
            h_new = _slsqp_opt(best_h, alpha, m, mi, n)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 6: Extended gradient polish (remaining time minus 5s)
    if _el() < budget_s - 10:
        best_h, best_c5 = _gradient_polish(best_h, budget_s - 5, rng)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
