    c = np.correlate(h, 1.0 - h, 'full') * dx
    m_eff = min(m, 2 * n - 1)
    idx = np.argpartition(-c, m_eff - 1)[:m_eff] if m_eff > 1 else np.array([c.argmax()])
    vals = c[idx]
    mx = vals.max()
    s = alpha * (vals - mx)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(m_eff):
        wi = w[ii]
        if wi < 1e-16:
            continue
        k = idx[ii] - (n - 1)
        if k >= 0:
            g[:n-k] += wi * (1.0 - h[k:])
            if k > 0:
                g[k:] -= wi * h[:n-k]
        else:
            k2 = -k
            g[k2:] += wi * (1.0 - h[:n-k2])
            g[:n-k2] -= wi * h[k2:]
    return g * dx

def _project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)
