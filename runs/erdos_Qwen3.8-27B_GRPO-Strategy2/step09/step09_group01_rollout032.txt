```python
        G[x, x : x+n] -= h[::-1]
    return G
```

```python
import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(50):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            for i in np.argsort(1.0 - h):
                if d <= 1e-12: break
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
        else:
            for i in np.argsort(h):
                if -d <= 1e-12: break
                a = min(-d, h[i]); h[i] -= a; d += a
    return h

def _ov_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _build_G(h):
    n = len(h)
    G = np.zeros((n, 2 * n - 1))
    rh = h[::-1]
    for x in range(n):
        G[x, n - 1 - x: n + 1 - x] += 1.0 - h
        G[x, x: x + n] -= rh
    return G

def _update_G(G, i, j, d, n):
    idx = np.arange(n)
    s1 = i - idx + n - 1
    m1 = (s1 >= 0) & (s1 < 2 * n - 1)
    G[idx[m1], s1[m1]] += d
    s2 = idx - i + n - 1
    m2 = (s2 >= 0) & (s2 < 2 * n - 1)
    G[idx[m2], s2[m2]] += d
    s3 = j - idx + n - 1
    m3 = (s3 >= 0) & (s3 < 2 * n - 1)
    G[idx[m3], s3[m3]] -= d
    s4 = idx - j + n - 1
    m4 = (s4 >= 0) & (s4 < 2 * n - 1)
    G[idx[m4], s4[m4]] -= d

def _subgrad(G, O, tol=2e-4):
    mx = O.max()
    active = O >= mx - tol
    return G[active].mean(axis=0) if active.sum() > 0 else np.zeros(len(G))

def _anneal(h0, dl, rng, s0=5e-5, beta0=200.0, beta1=20000.0, s_block=True):
    t0 = time.time()
    n = len(h0)
    h = _repair(h0, n).copy()
    O = _ov_vec(h).copy()
    G = _build_G(h)
    cmax = O.max()
    best_c = cmax
    best_h = h.copy()
    g = _subgrad(G, O)
    order_g = np.argsort(g)
    lo = order_g[:n // 4]
    hi = order_g[-n // 4:]
    recomp = 500
    it = 0
    beta = beta0
    while time.time() - t0 < dl:
        fr = (time.time() - t0) / dl
        beta = beta0 * (beta1 / beta0) ** fr
        base_s = s0 * (1e-4) ** (fr ** 1.5)
        if it >= recomp:
            O = _ov_vec(h)
            G = _build_G(h)
            cmax = O.max()
            g = _subgrad(G, O)
            order_g = np.argsort(g)
            lo = order_g[:n // 4]
            hi = order_g[-n // 4:]
            recomp = it + 500
        do_block = s_block and rng.random() < 0.12
        if do_block:
            L = int(rng.randint(3, min(20, n // 20)))
            i = int(lo[rng.randint(len(lo))])
            j = int(hi[rng.randint(len(hi))])
            i_b = i % (n - L)
            j_b = j % (n - L)
            if i_b == j_b or i_b + L > n or j_b + L > n:
                it += 1
                continue
            b1 = np.arange(i_b, i_b + L)
            b2 = np.arange(j_b, j_b + L)
            d_eff = 1.0
            dO = d_eff * (G[b2].sum(axis=0) - G[b1].sum(axis=0))
            new_O = O + dO
            new_max = new_O.max()
            if new_max < cmax or rng.random() < np.exp(-beta * max(0, new_max - cmax)):
                tmp = h[b1].copy()
                h[b1] = h[b2]
                h[b2] = tmp
                for x in b1:
                    _update_G(G, int(x), int(x), 0.0, n)
                # Recompute G for block (simpler for blocks)
                G = _build_G(h)
                O = new_O
                cmax = new_max
                if cmax < best_c:
                    best_c = cmax
                    best_h = h.copy()
        else:
            if rng.random() < 0.85:
                i = int(lo[rng.randint(len(lo))])
                j = int(hi[rng.randint(len(hi))])
            else:
                i = int(rng.randint(n))
                j = int(rng.randint(n))
            if i == j:
                it += 1
                continue
            sgn = 1 if g[i] >= g[j] else -1
            avail = min(1.0 - h[j], h[i]) if sgn > 0 else min(h[i], 1.0 - h[j])
            if avail < 1e-14:
                it += 1
                continue
            mult = 0.3 + 1.7 * rng.random()
            if rng.random() < 0.03:
                mult *= 5.0
            d = sgn * min(avail, base_s * mult)
            dO = d * (G[j] - G[i])
            new_max = (O + dO).max()
            if new_max < cmax - 1e-15:
                h[i] -= d
                h[j] += d
                O += dO
                cmax = new_max
                _update_G(G, i, j, -d, n)
                if cmax < best_c:
                    best_c = cmax
                    best_h = h.copy()
            elif rng.random() < np.exp(-beta * max(0.0, new_max - cmax)):
                h[i] -= d
                h[j] += d
                O += dO
                cmax = new_max
                _update_G(G, i, j, -d, n)
        it += 1
    return best_h, best_c

def _smooth_max(h0, alpha, dl):
    t0 = time.time()
    n = len(h0)
    S = n * 0.5
    h = _repair(h0, n)
    hp = h.copy()
    y = h.copy()
    tv = 1.0
    Lc = 5.0
    for it in range(50000):
        if time.time() - t0 > dl:
            break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tv * tv))
            y = h + ((tv - 1.0) / tn) * (h - hp)
            tv = tn
            y = np.clip(y, -0.1, 1.1)
            y += (S - y.sum()) / n
        else:
            y = h.copy()
        O = _ov_vec(y)
        mx = O.max()
        w = np.exp(np.clip(alpha * (O - mx), -500.0, 0.0))
        w /= max(w.sum(), 1e-30)
        G = _build_G(y)
        g = (w * G).sum(axis=1)
        step = 1.0 / max(Lc, 1e-6)
        xn = np.clip(y - step * g, 0.0, 1.0)
        xn += (S - xn.sum()) / n
        xn = np.clip(xn, 0.0, 1.0)
        O2 = _ov_vec(xn)
        if O2.max() <= O.max() + 1e-8:
            hp = h
            h = xn
            Lc = max(Lc * 0.85, 1.0)
        else:
            Lc *= 2.0
            for _ in range(8):
                step /= 2
                xn = np.clip(y - step * g, 0.0, 1.0)
                xn += (S - xn.sum()) / n
                xn = np.clip(xn, 0.0, 1.0)
                if _ov_vec(xn).max() <= O.max() + 1e-8:
                    hp = h
                    h = xn
                    Lc = 1.0 / step
                    break
            else:
                hp = h
                y = h.copy()
                tv = 1.0
                if Lc > 1e6:
                    break
    return h

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _repair(out, m)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    best = _repair(h0, n0)
    best_c = _c5(best)

    # Symmetry trial
    for var in [1.0 - best, best[::-1], 0.5 + 0.5 * (best - best[::-1])]:
        hv = _repair(var, n0)
        cv = _c5(hv)
        if cv < best_c:
            best, best_c = hv, cv

    # Phase 1: Short smooth pass (40s)
    rem = budget_s - (time.time() - t0) - 900.0
    rem = max(rem, 30.0)
    for a in [100, 1000]:
        if time.time() - t0 > budget_s - 900:
            break
        hf = _smooth_max(best, a, min(20.0, 0.4 * rem))
        cf = _c5(hf)
        if cf < best_c:
            best, best_c = hf, cf

    # Phase 2: Main annealing (600s)
    end2 = t0 + budget_s - 350.0
    best, best_c = _anneal(best, end2 - time.time(), rng, s0=4e-5,
                           beta0=150.0, beta1=50000.0, s_block=True)

    # Phase 3: Finer annealing (200s)
    end3 = t0 + budget_s - 130.0
    best, best_c = _anneal(best, end3 - time.time(), rng, s0=1.5e-5,
                           beta0=500.0, beta1=80000.0, s_block=True)

    # Phase 4: Very fine (100s)
    end4 = t0 + budget_s - 30.0
    best, best_c = _anneal(best, end4 - time.time(), rng, s0=5e-6,
                           beta0=2000.0, beta1=200000.0, s_block=False)

    # Symmetry + repair
    n = len(best)
    for var in [1.0 - best, best[::-1], 0.5 + 0.5 * (best - best[::-1])]:
        hv = _repair(var, n)
        cv = _c5(hv)
        if cv < best_c:
            best, best_c = hv, cv

    best = _repair(best, n)
    best_c = _c5(best)
    return best, best_c, n
```
