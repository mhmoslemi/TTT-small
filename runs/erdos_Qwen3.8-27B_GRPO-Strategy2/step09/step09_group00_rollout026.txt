```python
def _gsoft(h, a):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    scaled = a * (corr - mx)
    e = np.exp(np.clip(scaled, -40, 0))
    w = e / e.sum()
    
    u = np.zeros(n)
    u[0] = w[n-1]
    if n > 1:
        u[1:] = w[n:] + w[:n-1][::-1]
    
    # Fast gradient using FFT
    Nf = 1
    while Nf < 2*n:
        Nf <<= 1
    H = np.fft.rfft(h, Nf)
    U = np.fft.rfft(u, Nf)
    u_h_conv = np.fft.irfft(H * U, Nf).real[:n]
    
    # upper[m] = sum_{j=1}^{m} u[j]*(1-h[m-j])
    cum_u = np.cumsum(u)  # cum_u[m] = u[0]+...+u[m]
    upper = (cum_u - u[0]) - u_h_conv + u[0]*h
    
    # lower[m] = sum_{j=1}^{n-1-m} u[j]*h[m+j]
    if n > 1:
        corr_hu = np.correlate(h, u, mode='full')
        lower = corr_hu[n-1:] - u[0]*h
    else:
        lower = np.zeros(n)
    
    g = dx * (u[0]*(1.0 - 2.0*h) + upper - lower)
    return g
```
