            h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _fine_polish(h, n_iter, rng):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 15)) % n
        d = (5e-10 + 2e-9 * rng.rand()) * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _group_opt(h, n_iter, rng, gs=5):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        base = rng.randint(n)
        idx = np.sort((base + rng.randint(0, 40, size=gs)) % n)
        d = (5e-9 + 3e-8 * rng.rand()) * rng.randn(gs)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < 0 or h_new.max() > 1:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-18:
            c5c = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Nesterov PG with alpha sweep
    alphas = [8000, 30000, 150000, 800000, 5000000, 50000000]
    for alpha in alphas:
        if _el() > budget_s * 0.32:
            break
        tk = min(n * 2 - 1, max(80, int(8 * np.log10(alpha))))
        sp = max(0.001, 0.02 / (alpha ** 0.12))
        h_new = _nesterov(best_h.copy(), alpha, tk, 3000, sp, 0.85, sp * 80)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 2: SLSQP at high alpha
    if _el() < budget_s * 0.55:
        for alpha in [100000000, 2000000000]:
            if _el() > budget_s * 0.53:
                break
            h_new = _slsqp_opt(best_h, alpha, 250, 400)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 3: More Nesterov with very high alpha, fine steps
    if _el() < budget_s * 0.7:
        for alpha in [5000000000, 50000000000]:
            if _el() > budget_s * 0.68:
                break
            tk = min(n * 2 - 1, 350)
            h_new = _nesterov(best_h.copy(), alpha, tk, 4000, 0.0003, 0.9, 0.05)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 4: Perturbation + re-optimize (exploit different basin)
    if _el() < budget_s * 0.78:
        h_pert = best_h + 0.002 * rng.randn(n)
        h_pert = _repair(h_pert, n)
        h_new = _nesterov(h_pert, 100000000, 200, 2000, 0.002, 0.85, 0.1)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 5: Gradient local search
    if _el() < budget_s * 0.88:
        rem = budget_s * 0.88 - _el()
        ntry = max(500, int(rem * 400))
        best_h, best_c5 = _local_search(best_h, ntry, rng, 40)

    # Phase 6: Group coordination
    if _el() < budget_s * 0.94:
        rem = budget_s * 0.94 - _el()
        ntry = max(300, int(rem * 350))
        best_h, best_c5 = _group_opt(best_h, ntry, rng, 5)

    # Phase 7: Ultra-fine 2-point polish
    if _el() < budget_s - 3:
        rem = budget_s - 3 - _el()
        ntry = max(1000, int(rem * 1200))
        best_h, best_c5 = _fine_polish(best_h, ntry, rng)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
