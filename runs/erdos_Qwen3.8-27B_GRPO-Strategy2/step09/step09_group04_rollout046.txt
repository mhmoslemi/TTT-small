def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s < 1e-15:
            h[:] = 0.5
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
    return h

def _sobj(x, alpha, ml):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sgrad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0 and L < n:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        elif L < 0:
            M = -L
            if M < n:
                g[:n - M] += wk * (1.0 - x[M:])
                g[M:] -= wk * x[:n - M]
    return g

def _sf(x, t):
    return x.sum() - t

def _sj(x, t):
    return np.ones(len(x))

def _do_slsqp(h, alpha, ml, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n

                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(r.x, n)
    except:
        return h

def _do_lbfgs(h, alpha, ml, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (n * 0.5,)}]

    except:
        return h

        else:

    cc = _eval(h)

            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:

                cn = _eval(h)
                if cn < cc:
                    cc = cn
                else:
                    h[i], h[j] = ni - d, nj + d
    return h, cc
