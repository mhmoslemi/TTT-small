```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

        else:
            hi = mid
```

```python
def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(x, alpha, m_lags):
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_lags):
    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii, k in enumerate(idx):

        wk = w[ii] * dx
        L = k - (n - 1)

            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fn(x, half_n):
    return x.sum() - half_n

    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter, ftol):

    bnds = [(0.0, 1.0)] * n

                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,

    except Exception:
        return h
```

Now, the golden section search. For each coordinate i:
- a[k] = c[k] - old_i * coef[k] (the intercept)
- b[k] = coef[k] (the slope)
- g(x) = max_k(a[k] + b[k]*x)
- Find x* in [0,1] minimizing g(x)

```python
def _min_1d_convex_pl(a, b):
    """Minimize max_k(a[k] + b[k]*x) for x in [0,1]."""
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c1 = hi - phi * (hi - lo)
    d1 = lo + phi * (hi - lo)
    f1 = np.max(a + b * c1)
    f2 = np.max(a + b * d1)
    for _ in range(45):
        if f1 < f2:
            hi = d1
            d1 = c1
            f2 = f1
            c1 = hi - phi * (hi - lo)
            f1 = np.max(a + b * c1)
        else:
            lo = c1
            c1 = d1
            f1 = f2
            d1 = lo + phi * (hi - lo)
            f2 = np.max(a + b * d1)
    return (lo + hi) * 0.5
```

Now the full coordinate descent:

```python
def _coord_descent(h, n_sweeps, rng, t0, budget):
    n = len(h)
    dx = 2.0 / n
    h = _repair(h, n)
    c = _corr(h)
    best_c5 = float(c.max())
    best_h = h.copy()
    stall = 0
    PHI = (np.sqrt(5.0) - 1.0) / 2.0
    
    for sweep in range(n_sweeps):
        if time.time() - t0 > budget - 100:
            break
        perm = rng.permutation(n)
        for i in perm:
            old_i = h[i]
            coef = _compute_coef(h, i, n)
            a = c - old_i * coef
            b = coef
            # Golden section
            lo, hi = 0.0, 1.0
            c1 = hi - PHI * (hi - lo)
            d1 = lo + PHI * (hi - lo)
            f1 = np.max(a + b * c1)
            f2 = np.max(a + b * d1)
            for _ in range(45):
                if f1 < f2:
                    hi = d1; d1 = c1; f2 = f1
                    c1 = hi - PHI * (hi - lo)
                    f1 = np.max(a + b * c1)
                else:
                    lo = c1; c1 = d1; f1 = f2
                    d1 = lo + PHI * (hi - lo)
                    f2 = np.max(a + b * d1)
            x_opt = (lo + hi) * 0.5
            h[i] = x_opt
            c += (x_opt - old_i) * coef
        
        h = _repair(h, n)
        c = _corr(h)
        cur = float(c.max())
        if cur < best_c5 - 1e-12:
            best_c5 = cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30 and stall % 10 == 0:
                # Perturb to escape
                npert = max(5, n // 50)
                idx = rng.randint(0, n, size=npert)
                h[idx] += 0.02 * rng.randn(npert)
                h = _repair(h, n)
                c = _corr(h)
    
    return best_h, best_c5
```

The issue is: will coordinate descent actually improve over the SLSQP result? Let me think...

The SLSQP result is a local minimum of the smooth max. As alpha → ∞, the smooth max → max. So the SLSQP result should be close to a local minimum of the max. But coordinate descent finds a coordinate-wise minimum, which is a stronger condition (every coordinate is optimized given the others). So it should find a point that's at least as good.

But there's a subtlety: the sum constraint. The SLSQP handles it directly (as an equality constraint). My coordinate descent ignores it during the sweep and projects at the end. The projection might move the point away from the optimum.

To handle this better, I could:
1. Alternate between two coordinates: change h[i] by +d and h[j] by -d. This maintains the sum exactly.
2. Use a Lagrange multiplier for the sum.

Option 1 is cleaner. Let me modify the coordinate descent to do paired updates:

For each pair (i, j):
- Minimize g(d) = max_k (c[k] + d*(coef_i[k] - coef_j[k])) for d in some range.
- Update: h[i] += d, h[j] -= d.
- Constraint: 0 ≤ h[i]+d ≤ 1 and 0 ≤ h[j]-d ≤ 1.

The range of d is [max(-h[i], h[j]-1), min(1-h[i], h[j])].

This is still a 1D convex piecewise linear minimization! And it exactly preserves the sum.

But I need to compute coef_i - coef_j, which is two coefficient vectors. And the golden section is over the range of d, not [0,1].

Let me modify:
