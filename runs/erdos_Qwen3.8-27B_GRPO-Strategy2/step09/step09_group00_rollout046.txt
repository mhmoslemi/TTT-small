def _fft_size(n):
    m = 2 * n - 1
    N = 1
    while N < m:

def _repair(h, n):
    S = n * 0.5

    s = h.sum()

        return np.full(n, 0.5)
    for _ in range(50):
        h *= S / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - S) < 1e-14:
            break
    d = S - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a

def _proj(v, S):

    # Phase 5: Fine SA polish
    end5 = t0 + budget_s - 5.0
    if end5 > time.time():
        best, best_c = _sa_polish(best, rng, end5)
    # Final variants
    for hh in (1.0 - best, best[::-1], 0.5 + 0.5 * (best - best[::-1])):
        h2 = _proj(hh, len(hh) * 0.5)
        c = _c5(h2)
        if c < best_c:
            best = h2; best_c = c
    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
