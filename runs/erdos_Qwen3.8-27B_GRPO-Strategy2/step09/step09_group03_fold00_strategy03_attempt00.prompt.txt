def validate_solution(h, n_points):
    dx = 2.0 / n_points
    h_sum = np.sum(h) * dx
    # Check integral
    if h_sum < 0.9999 or h_sum > 1.0001:

    if np.min(h) < 0 or np.max(h) > 1:

    try:
