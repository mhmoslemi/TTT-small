def _repair(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Water-filling projection
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    h = np.clip(h - lam, 0, 1)
    # Fix residual
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        # Distribute evenly among non-boundary points
        free = (h > 1e-10) & (h < 1 - 1e-10)
        if free.sum() > 0:
            adj = diff / free.sum()
            h[free] = np.clip(h[free] + adj, 0, 1)
        # Fix remaining
        diff = target - h.sum()
        if abs(diff) > 1e-15:

                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-15:
                    break
    return h

    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _overlap_grad(h, L, n):
    """Gradient of ∑ h[i]*(1-h[i+L]) w.r.t. h, for shift L (in samples)"""
    dx = 2.0 / n
    g = np.zeros(n)
    if L >= 0:

        g[:m] += (1.0 - h[L:]) * dx
        g[L:] -= h[:m] * dx
    else:
        M = -L
        m = n - M
        g[:m] += (1.0 - h[M:]) * dx
        g[M:] -= h[:m] * dx
    return g

    """Smooth max gradient focusing on top_k shifts"""
    n = len(h)

    m = c.max()
    idx = np.argpartition(-c, min(top_k, len(c)-1))[:top_k]

            continue
        L = k - (n - 1)  # shift in samples
        gk = _overlap_grad(h, L, n)
        g += w[ii] * gk

    return m + np.log(np.exp(s).sum()) / alpha

- Second term: (1-h)[m-k+n-1] with m-k+n-1=i, i.e., m=i+k-n+1: contributes h[i+k-n+1]*(-1) if 0≤i+k-n+1<n

So S[i,k] = (1-h[i-k+n-1]) * [0≤i-k+n-1<n] - h[i+k-n+1] * [0≤i+k-n+1<n]

Let me define: for k in [0, 2n-2], let L = k - (n-1) (the lag, in [-(n-1), n-1]).
Then i-k+n-1 = i-L and i+k-n+1 = i+L.

S[i,k] = (1-h[i-L]) * [0≤i-L<n] - h[i+L] * [0≤i+L<n]

This is correct! And it's O(1) per entry, so O(n*(2n-1)) = O(n²) total to precompute S.

Let me verify with a small example. n=3, h=[a,b,c].
c_full[k] for k=0,1,2,3,4 (lags -2,-1,0,1,2):

k=0, L=-2: c = h[0]*(1-h[0-(-2)]) = h[0]*(1-h[2]) = a*(1-c)
  S[0,0] = (1-h[0-(-2)])*[0≤0+2<3] - h[0+(-2)]*[0≤0-2<3] = (1-h[2])*1 - 0 = 1-c ✓ (d/dc of a*(1-c) w.r.t. a = 1-c)
  S[1,0] = (1-h[1+2])*[0≤3<3] - h[1-2]*[0≤-1<3] = 0 - 0 = 0 ✓

k=1, L=-1: c = h[0]*(1-h[1]) + h[1]*(1-h[2]) = a*(1-b) + b*(1-c)
  S[0,1] = (1-h[0+1])*[0≤1<3] - h[0-1]*[0≤-1<3] = (1-b)*1 - 0 = 1-b ✓

Great, the formula works!

Now, for the 1D optimization:
new_c = c + δ*(S[i,:] - S[j,:])

f(δ) = max(new_c) = max(c + δ*(S[i,:] - S[j,:]))

This is a convex PL function of δ. The slope at any δ is one of the entries of (S[i,:] - S[j,:]).

I can find the optimal δ by:
1. Let v = S[i,:] - S[j,:]
2. f(δ) = max_k (c[k] + v[k]*δ)
3. At δ=0, the active set is A = {k : c[k] = max(c)}
4. The subgradient at 0 contains {v[k] : k ∈ A}
5. If 0 ∈ subgradient, δ*=0
6. Otherwise, we need to move. The direction is determined by the most extreme v[k] in A.

For a fast 1D convex PL minimization, I can use:
- The function f(δ) = max_k (c[k] + v[k]*δ)
- This is minimized at δ* where the upper envelope has a local minimum
- The "breakpoints" are where c[i] + v[i]*δ = c[j] + v[j]*δ, i.e., δ = (c[j]-c[i])/(v[i]-v[j])
- There are at most O(n²) breakpoints, but I don't need to enumerate them all

Simpler approach: use the fact that the optimal δ satisfies:
max_{v[k]>0} (c[k] + v[k]*δ) ≤ min_{v[k]<0} (c[k] + v[k]*δ') ... no that's not right.

Actually, the cleanest approach for 1D convex PL minimization:
- f(δ) = max_k (c[k] + v[k]*δ)
- The minimum is where the "slope" of f changes from negative to positive
- Slope of f at δ is max_{k: c[k]+v[k]*δ = f(δ)} v[k] (upper Dini derivative)
- and min_{k: c[k]+v[k]*δ = f(δ)} v[k] (lower Dini derivative)
- f is minimized where 0 ∈ [lower_slope, upper_slope]

Alternative: since f is the pointwise max of linear functions, it's the support function of the set {(v[k], c[k])}. The minimum of f(δ) over δ is... actually, I can just do a few evaluations.

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(np.abs(h - 0.5))
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _compute_S(h):
    """Compute sensitivity matrix S[i,k] = d(c[k])/d(h[i]) where c = correlate(h,1-h,'full')*dx"""
    n = len(h)
    dx = 2.0 / n
    S = np.zeros((n, 2*n-1))
    L = np.arange(2*n-1) - (n-1)  # lags
    for i in range(n):
        # S[i,k] = (1-h[i-L[k]])*[0<=i-L[k]<n] - h[i+L[k]]*[0<=i+L[k]<n]
        iL = i - L  # indices for (1-h) factor
        iP = i + L  # indices for -h factor
        mask1 = (iL >= 0) & (iL < n)
        mask2 = (iP >= 0) & (iP < n)
        S[i, mask1] += (1 - h[iL[mask1]]) * dx
        S[i, mask2] -= h[iP[mask2]] * dx
    return S

def _smooth_obj(x, alpha, ml):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
    m = c.max()
    idx = np.argpartition(-c, ml-1)[:ml]
    s = np.clip(alpha*(c[idx]-m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum())/alpha

def _smooth_grad(x, alpha, ml):
    n = len(x)
    dx = 2.0/n
    c = np.correlate(x, 1.0-x, mode='full')*dx
    m = c.max()
    idx = np.argpartition(-c, ml-1)[:ml]
    s = np.clip(alpha*(c[idx]-m), -400, 400)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-16: continue
        wk = w[ii]*dx
        L = k - (n-1)
        if L >= 0:
            g[L:] += wk*(1.0-x[:n-L])
            g[:n-L] -= wk*x[L:]
        else:
            M = -L
            g[:n-M] += wk*(1.0-x[M:])
            g[M:] -= wk*x[:n-M]
    return g

def _sum_fn(x, half):
    return x.sum() - half

def _slsqp(h, alpha, ml, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)]*n
    cons = [{'type':'eq', 'fun':_sum_fn, 'jac':lambda x: np.ones(n), 'args':(n*0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,ml),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter':maxiter, 'ftol':ftol, 'maxls':40})
        return _repair(res.x, n)
    except:
        return h

def _coord_descent(h, n, S, n_iter, rng):
    """1D coordinate descent using sensitivity matrix."""
    best_c = _eval(h)
    best_h = h.copy()
    dx = 0  # already in S
    for it in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, n-1)) % n
        if i == j:
            j = (i + 1) % n
        lo_d = max(-h[i], h[j] - 1.0)
        hi_d = min(1.0 - h[i], h[j])
        if hi_d - lo_d < 1e-14:
            continue
        v = S[i] - S[j]  # slope vector, shape (2n-1,)
        c = np.correlate(h, 1.0-h, mode='full') * (2.0/n)
        # Golden section on f(δ) = max(c + v*δ)
        gr = (np.sqrt(5.0)-1.0)/2.0
        a, b = lo_d, hi_d
        m1 = b - gr*(b-a)
        m2 = a + gr*(b-a)
        f1 = np.max(c + v*m1)
        f2 = np.max(c + v*m2)
        for _ in range(30):
            if f1 < f2:
                b, m2, f2 = m2, m1, f1
                m1 = b - gr*(b-a)
                f1 = np.max(c + v*m1)
            else:
                a, m1, f1 = m1, m2, f2
                m2 = a + gr*(b-a)
                f2 = np.max(c + v*m2)
        d_opt = (a+b)/2.0
        h[i] += d_opt
        h[j] -= d_opt
        h = _repair(h, n)
        S = _compute_S(h)  # Update S (expensive!)
        c_new = _eval(h)
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
    return best_h, best_c
```

        else:
            a, c1, f1 = c1, c2, f2
            c2 = a + gr * (b - a)
            h2 = h.copy(); h2[i] += c2; h2[j] -= c2
            f2 = np.max(np.correlate(h2, 1.0 - h2, 'full'))
    return (a + b) / 2.0

def _coord_descent(h, n, n_iter, rng, grad_info=None):
    """Coordinate descent with exact 1D optimization. Gradient-informed pair selection."""
    best_c = _eval(h)
    best_h = h.copy()
    g = _smooth_grad(h, 1e8, 50) if grad_info is None else grad_info
    for it in range(n_iter):
        if it % 200 == 0:
            g = _smooth_grad(h, 1e8, 50)
        # Gradient-informed: pick pair with max |g[i]-g[j]|
        if it % 5 == 0:
            order = np.argpartition(-np.abs(g), 10)[:10]
            i = order[rng.randint(10)]
            j = rng.randint(n)
            if i == j:
                j = (i + 1) % n
        else:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, n - 2)) % n
            if i == j:
                j = (i + 1) % n
        d = _coord_1d(h, i, j, n)
        if abs(d) < 1e-15:
            continue
        h[i] += d
        h[j] -= d
        h = _repair(h, n)
        c_new = _eval(h)
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP warmup
    for alpha in [50000, 500000]:
        if _el() > 60:
            break
        h_new = _slsqp(best_h, alpha, 40, 800, 1e-14)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 2: Coordinate descent (key new component)
    if _el() < budget_s - 400:
        nt = min(int((budget_s - 400 - _el()) / 0.004), 50000)
        best_h, best_c5 = _coord_descent(best_h, n, nt, rng)

    # Phase 3: High-alpha SLSQP
    for alpha in [5e6, 5e7, 5e8]:
        if _el() > budget_s - 250:
            break
        ml = min(100, max(30, int(20 * np.log10(alpha / 5e6) + 30)))
        h_new = _slsqp(best_h, alpha, ml, 1500, 1e-16)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 4: More coordinate descent
    if _el() < budget_s - 150:
        nt2 = min(int((budget_s - 150 - _el()) / 0.004), 40000)
        best_h, best_c5 = _coord_descent(best_h, n, nt2, rng)

    # Phase 5: Ultra-high alpha SLSQP
    if _el() < budget_s - 80:
        for alpha in [5e9, 5e10]:
            if _el() > budget_s - 40:
                break
            h_new = _slsqp(best_h, alpha, 200, 1000, 1e-18)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 6: Final fine coordinate descent
    if _el() < budget_s - 5:
        nt3 = min(int((budget_s - 5 - _el()) / 0.004), 30000)
        best_h, best_c5 = _coord_descent(best_h, n, nt3, rng)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
