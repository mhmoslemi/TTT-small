- If f'(t) < 0: t* > t, so lo_t = t
- If f'(t) = 0: t* = t (for a convex function, this means we're at the global min)

In practice, checking just the argmax's beta works because:
- If beta[argmax] > 0: the max is achieved at a point where the function is increasing, so the overall max is increasing, and the min of the max is to the left.
- Similarly for < 0.
- If beta[argmax] = 0: we might be at the min, or at a kink where the max is flat. Either way, we can stop.

But there's a subtlety: what if the argmax has positive beta, but there's another k with the same value and negative beta? Then 0 ∈ ∂f(t) and t IS the minimizer. But in practice, exact ties have measure zero, so this won't happen.

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _slsqp_obj(x, alpha, n, m_lags):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, n, m_lags):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        k = int(idx[ii])
        wj = w[ii] * dx
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo > j_hi:

        jj = np.arange(j_lo, j_hi + 1)
        g[j_lo:j_hi + 1] += wj * (1.0 - 2.0 * x[k - jj])

    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter):
    n = len(h)

    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n * 0.5,)}]

                       options={'maxiter': maxiter, 'ftol': 1e-18})
