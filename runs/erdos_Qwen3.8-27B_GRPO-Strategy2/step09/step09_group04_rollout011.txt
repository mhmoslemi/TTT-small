    n = len(h)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)

        h *= target / s

            break

    if abs(diff) > 1e-9:

        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

    n = len(h0); m = len(lags)
    A = np.zeros((m, n)); b = np.zeros(m)
    for ci in range(m):
        L = lags[ci]
        if L >= 0:
            c_val = np.dot(h0[L:n], 1.0 - h0[0:n-L])
            A[ci, L:n] += (1.0 - h0[0:n-L])
            A[ci, 0:n-L] -= h0[L:n]
        else:
            M = -L
            c_val = np.dot(h0[M:n], 1.0 - h0[0:n-M])
            A[ci, M:n] += (1.0 - h0[0:n-M])
            A[ci, 0:n-M] -= h0[M:n]
        b[ci] = c_val - A[ci] @ h0
    return A, b

def _slp_lp(h0, lags, damping=1.0):
    n = len(h0)
    A, b = _compute_grads(h0, lags)
    m = len(lags)
    c_obj = np.zeros(n + 1); c_obj[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = -b
    A_eq = np.zeros((1, n + 1)); A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        h_new = res.x[:n] * damping + h0 * (1.0 - damping)
        return _repair(h_new, n), float(res.x[n])
    return h0.copy(), float('inf')

def _slsqp_obj(x, alpha, ml):
    n = len(x)
    c = _eval_vec(x)
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - c.max())
    s -= s.max()
    return c.max() + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, ml):
    n = len(x)
    c = _eval_vec(x)
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - c.max())
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    lags = idx - (n - 1)
    A, _ = _compute_grads(x, lags)
    return (w @ A) * (2.0 / n)

def _slsqp_run(h, alpha, ml, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def _cfunc(x): return x.sum() - n * 0.5
    cons = [{'type': 'eq', 'fun': _cfunc, 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(_slsqp_obj, h, jac=_slsqp_grad, args=(alpha, ml),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(r.x, n)
    except:
        return h

def _local_cd(h, n, budget, rng):
    c = _eval(h)
    bh, bc = h.copy(), c
    deadline = time.time() + budget
    step_sz = 2e-5
    while time.time() < deadline:
        for round_i in range(200):
            if time.time() > deadline:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, n)) % n
            d = step_sz * rng.randn()
            if 0 <= h[i]+d <= 1 and 0 <= h[j]-d <= 1:
                h[i] += d; h[j] -= d
                cn = _eval(h)
                if cn < c:
                    c = cn
                    if c < bc: bc, bh = c, h.copy()
                    step_sz = min(step_sz * 1.1, 5e-4)
                else:
                    h[i] -= d; h[j] += d
                    step_sz *= 0.95
            else:
                step_sz *= 0.97
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = _repair(h, n)
    best_h, best_c5 = h.copy(), _eval(h)

    def _el(): return time.time() - t0

    # Phase 1: SLP top-80 lags (exploration)
    m1 = min(80, 2*n-1)
    for it in range(500):
        if _el() > 120: break
        corr = _eval_vec(h)
        top_idx = np.argsort(corr)[-m1:]
        lags = (top_idx - (n - 1)).astype(float)
        h_new, _ = _slp_lp(h, lags, damping=1.0)
        c_new = _eval(h_new)
        if c_new < best_c5: best_c5, best_h = c_new, h_new.copy()
        h = h_new

    # Phase 2: SLP top-200 lags
    m2 = min(200, 2*n-1)
    for it in range(500):
        if _el() > 320: break
        corr = _eval_vec(h)
        top_idx = np.argsort(corr)[-m2:]
        lags = (top_idx - (n - 1)).astype(float)
        h_new, _ = _slp_lp(h, lags, damping=0.8)
        c_new = _eval(h_new)
        if c_new < best_c5: best_c5, best_h = c_new, h_new.copy()
        h = h_new

    # Phase 3: SLP ALL lags
    all_lags = np.arange(-(n-1), n, dtype=float)
    for it in range(500):
        if _el() > 550: break
        h_new, _ = _slp_lp(h, all_lags, damping=0.7)
        c_new = _eval(h_new)
        if c_new < best_c5: best_c5, best_h = c_new, h_new.copy()
        h = h_new

    # Phase 4: SLSQP polish with multiple alphas
    if _el() < budget_s - 200:
        for alpha in [5000, 50000, 500000, 5000000]:
            if _el() > budget_s - 180: break
            ml = min(60, n)
            hn = _slsqp_run(h, alpha, ml, 500)
            cn = _eval(hn)
            if cn < best_c5: best_c5, best_h = cn, hn.copy()
            h = bn = hn

    # Phase 5: SLP again from refined point
    if _el() < budget_s - 150:
        for it in range(300):
            if _el() > budget_s - 130: break
            lags = all_lags
            h_new, _ = _slp_lp(h, lags, damping=0.5)
            c_new = _eval(h_new)
            if c_new < best_c5: best_c5, best_h = c_new, h_new.copy()
            h = h_new

    # Phase 6: Coordinate descent local search
    if _el() < budget_s - 5:
        rem = budget_s - 5 - _el()
        best_h, best_c5 = _local_cd(best_h, n, rem, rng)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
