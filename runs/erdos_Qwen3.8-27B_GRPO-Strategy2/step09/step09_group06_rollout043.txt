import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _eval_all(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.1)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    # Exact fix
    diff = target - h.sum()
    h = np.clip(h, 0.0, 1.0)
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _proj(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target}"""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _obj_logsumexp(x, alpha, m, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s - s.max(), -500, 500)
    return c.max() + np.log(np.exp(s).sum()) / alpha

def _grad_logsumexp(x, alpha, m, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s - s.max(), -500, 500)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _slsqp_step(h, alpha, m_lags, maxiter, n):
    bnds = [(0.0, 1.0)] * n
    target = n * 0.5
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj_logsumexp, h.copy(), jac=_grad_logsumexp,
                       args=(alpha, m_lags, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 50})
        return _repair(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def _nesterov_pgd(h, alpha, m_lags, nsteps, step0, n):
    """Nesterov-accelerated projected gradient descent."""
    cur_c = _eval(h)
    y = h.copy()
    v = np.zeros(n)
    for t in range(nsteps):
        # Nesterov interpolation
        t1 = t + 1
        beta = (1.0 + np.sqrt(1.0 + 4.0 * t1 * t1)) / 2.0
        alpha_nest = (t1 - 1.0) / beta
        y = h + alpha_nest * (h - v)
        # Gradient at y
        g = _grad_logsumexp(y, alpha, m_lags, n)
        # Line search
        step = step0
        best_h = h
        best_c = cur_c
        for _ in range(8):
            h_new = _proj(h - step * g, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:
                best_h = h_new
                best_c = c_new
                break
            step *= 0.5
        if best_h is not h:
            v = h.copy()
            h = best_h
            cur_c = best_c
        else:
            step = step0
            v = h.copy()
    return _repair(h, n)

import numpy as np
from scipy.optimize import minimize
import time


def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h


def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)


def _lsq_obj(x, alpha, m, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s - s.max(), -400, 400)
    return c.max() + np.log(np.exp(s).sum()) / alpha


def _lsq_grad(x, alpha, m, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s - s.max(), -400, 400)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g


def _sum_fn(x, target):
    return x.sum() - target


def _sum_jac(x, target):
    return np.ones(len(x))


def _slsqp(h, alpha, ml, maxiter, n):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_lsq_obj, h.copy(), jac=_lsq_grad, args=(alpha, ml, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-19, 'maxls': 40})
        return _repair(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        return h


def _nest_pgd(h, alpha, ml, nsteps, step0, n):
    cur_c = _eval(h)
    v = np.zeros(n)
    mu = 0.7
    step = step0
    for t in range(nsteps):
        g = _lsq_grad(h, alpha, ml, n)
        v = mu * v + (1.0 - mu) * g
        taken = False
        for _ in range(10):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-16:
                h = h_new
                cur_c = c_new
                step = min(step * 1.03, step0 * 20.0)
                taken = True
                break
            step *= 0.45
        if not taken:
            step = max(step * 1.5, step0 * 0.1)
            v *= 0.5
    return _repair(h, n)


def _grad_guided_search(h, n_iter, rng, n):
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    base = 5e-5
    for it in range(n_iter):
        frac = it / max(n_iter, 1)
        step = base * (2.5 - 2.0 * frac) * (0.3 + 0.7 * rng.rand())
        # Compute gradient to guide direction
        g = _lsq_grad(h, 50000, 40, n)
        # Pick i from top-of-gradient (increase) and j from bottom (decrease)
        gi = np.argmax(g)
        gj = np.argmin(g)
        if rng.rand() < 0.4:
            i, j = gi, gj
        else:
            k = rng.choice([2, 2, 3, 3, 4, 5, 7])
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 20)) % n
        d = step * rng.randn()
        h_new = h.copy()
        h_new[i] += d
        h_new[j] -= d
        if h_new.min() < -0.05 or h_new.max() > 1.05:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5cur:
            c5cur = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c


def _fine_polish(h, n, n_iter, rng):
    c5c = _eval(h)
    h2 = h.copy()
    nc = len(h2)
    for _ in range(n_iter):
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 15)) % nc
        d = (1e-7 + 5e-8 * rng.rand()) * rng.randn()
        ni, nj = h2[i] + d, h2[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h2[i], h2[j] = ni, nj
            c5n = _eval(h2)
            if c5n < c5c:
                c5c = c5n
            else:
                h2[i], h2[j] = ni - d, nj + d
    return h2, c5c


def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)


def _upsample(h, f):
    return _repair(np.repeat(h, f), len(h) * f)

    best_c5 = _eval(best_h)
