                h[j] -= d
                c_nw = _eval(h)
                if c_nw < c5cur:
                    c5cur = c_nw
                    if c5cur < best_c:
                        best_c = c5cur
                        best_h = h.copy()
                else:
                    h[i] -= d
                    h[j] += d
    return best_h, best_c

def _2pt_search(h, n_iter, rng, alpha, ml):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        g = _lse_grad(h, alpha, ml)
        gp = g - g.mean()
        i = int(np.argmax(gp))
        j = int(np.argmin(gp))
        if i == j:
            continue
        best_d = 0.0
        best_c5 = c5c
        for frac in [0.05, 0.15, 0.3, 0.5, 0.7, 0.9]:
            d = frac * (gp[i] - gp[j]) * 0.005
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < best_c5:
                best_c5 = c5n
                best_d = d
            else:
                h[i], h[j] = h[i] - d, h[j] + d
                break
        if best_d != 0.0:
            c5c = best_c5
            if c5c < best_c:
                best_c = c5c
                best_h = h.copy()
    return best_h, best_c

def _fine_polish(h, n_iter, rng):
    nc = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 12)) % nc
        d = 2.0e-7 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _multi_start(h, n, rng, alpha, ml, maxiter, n_starts):
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(n_starts):
        pert = rng.randn(n) * 0.02
        pert -= pert.mean()
        h_try = _repair(h + pert, n)
        h_try = _slsqp(h_try, alpha, ml, maxiter, 1e-14)
        c_try = _eval(h_try)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _update(h_new):
        nonlocal best_h, best_c5
        c = _eval(h_new)
        if c < best_c5:
            best_c5 = c
            best_h = h_new.copy()

    # Phase 1: Quick SLSQP warmup
    for alpha in [5000, 50000, 500000]:
        if time.time() - t0 > 60:
            break
        h = _slsqp(best_h, alpha, 40, 500, 1e-14)
        _update(h)

    # Phase 2: Trust-constr (different algorithm, may find different basin)
    for alpha in [100000, 1000000, 10000000]:
        if time.time() - t0 > 200:
            break
        h_new = _trustcon(best_h, alpha, 50, 800, 1e-14)
        _update(h_new)

    # Phase 3: Nesterov projected gradient
    if time.time() - t0 < budget_s - 300:
        h_new = _nest_grad(best_h.copy(), 200000, 60, 800, 0.003)
        _update(h_new)
        if time.time() - t0 < budget_s - 250:
            h_new = _nest_grad(best_h.copy(), 2000000, 80, 600, 0.001)
            _update(h_new)

    # Phase 4: More SLSQP at very high alpha
    if time.time() - t0 < budget_s - 150:
        for alpha in [100000000, 500000000, 2000000000]:
            if time.time() - t0 > budget_s - 100:
                break
            h_new = _slsqp(best_h, alpha, 80, 600, 1e-16)
            _update(h_new)

    # Phase 5: Coordinate descent sweeps
    if time.time() - t0 < budget_s - 80:
        rem = int(budget_s - 80 - (time.time() - t0))
        n_sw = max(1, rem // 30)
        best_h, best_c5 = _coord_descent(best_h, n_sw, rng)

    # Phase 6: Gradient-guided 2-point search
    if time.time() - t0 < budget_s - 40:
        nitr = min(int((budget_s - 40 - (time.time() - t0)) / 0.01), 50000)
        best_h, best_c5 = _2pt_search(best_h, nitr, rng, 5000000, 60)

    # Phase 7: Multi-start perturbation + quick SLSQP
    if time.time() - t0 < budget_s - 30:
        rem = int(budget_s - 30 - (time.time() - t0))
        ns = max(1, min(5, rem // 15))
        best_h, best_c5 = _multi_start(best_h, n, rng, 5000000, 50, 300, ns)

    # Phase 8: Fine polish
    if time.time() - t0 < budget_s - 3:
        nitr = min(int((budget_s - 3 - (time.time() - t0)) / 0.0012), 400000)
        best_h, best_c5 = _fine_polish(best_h, nitr, rng)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
