    active = c >= (m - tol)
    # If too many, take top n_active
    if active.sum() > n_active:
        idx = np.argsort(c)[-n_active:]
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True

    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)

            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]

            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]

    return np.correlate(h, 1.0 - h, mode='full') * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if s < 1e-15:
            return np.full(n, 0.5)
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-15:
                break
    return h

def _sobj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sgrad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            if n - L > 0:
                g[L:n] += wk * (1.0 - x[:n-L])
            if L > 0:
                g[:n-L] -= wk * x[L:n]
        else:
            M = -L
            if n - M > 0:
                g[:n-M] += wk * (1.0 - x[M:n])
            if M > 0:
                g[M:n] -= wk * x[:n-M]
    return g

def _sf(x, hn):
    return x.sum() - hn

def _sj(x, hn):
    return np.ones(len(x))

def _do_slsqp(h, alpha, ml, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (n * 0.5,)}]
    try:
        r = minimize(_sobj, h, jac=_sgrad, args=(alpha, ml),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(r.x, n)
    except:
        return h

def _do_lbfgs(h, alpha, ml, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (n * 0.5,)}]
    try:
        r = minimize(_sobj, h, jac=_sgrad, args=(alpha, ml),
                     method='L-BFGS-B', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'maxfun': maxiter * 4})
        return _repair(r.x, n)
    except:
        return h

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _subgrad_step(h, n_active, tol):
    """Compute subgradient of max correlation and take a projected step."""
    n = len(h)
    c = _corr(h)
    m = c.max()
    # Find active lags
    active_mask = c >= (m - tol)
    idx = np.where(active_mask)[0]
    if len(idx) == 0:
        idx = np.argsort(c)[-1:]
    elif len(idx) > n_active:
        idx = np.argsort(c)[-n_active:]
    g = np.zeros(n)
    dx = 2.0 / n
    for k in idx:
        L = k - (n - 1)
        if L >= 0:
            if n - L > 0:
                g[L:n] += (1.0 - h[:n-L]) * dx
            if L > 0:
                g[:n-L] -= h[L:n] * dx
        else:
            M = -L
            if n - M > 0:
                g[:n-M] += (1.0 - h[M:n]) * dx
            if M > 0:
                g[M:n] -= h[:n-M] * dx
    g /= len(idx)
    return g

def _proj_subgrad(h, n_steps, start_tol, end_tol, n_active):
    """Projected subgradient with decreasing tol."""
    n = len(h)
    target = n * 0.5
    best_h = h.copy()
    best_c = _eval(best_h)
    step = 0.005
    tol = start_tol
    for i in range(n_steps):
        tol = start_tol * (end_tol / start_tol) ** (i / max(n_steps, 1))
        g = _subgrad_step(h, n_active, tol)
        # Backtracking line search
        taken = False
        for _ in range(15):
            h_new = _proj(h - step * g, target)
            c_new = _eval(h_new)
            if c_new < _eval(h):
                h = h_new
                step = min(step * 1.05, 0.05)
                taken = True
                break
            step *= 0.5
        if not taken:
            step *= 1.1
        if _eval(h) < best_c:
            best_c = _eval(h)
            best_h = h.copy()
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        s = h.sum()
        if s < 1e-15:
            return np.full(n, 0.5)
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-15:
                break
    return h

def _sobj(x, alpha, ml):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sgrad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15: continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            if n - L > 0: g[L:n] += wk * (1.0 - x[:n-L])
            if L > 0: g[:n-L] -= wk * x[L:n]
        else:
            M = -L
            if n - M > 0: g[:n-M] += wk * (1.0 - x[M:n])
            if M > 0: g[M:n] -= wk * x[:n-M]
    return g

def _sf(x, hn): return x.sum() - hn
def _sj(x, hn): return np.ones(len(x))

def _do_slsqp(h, alpha, ml, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (n*0.5,)}]
    try:
        r = minimize(_sobj, h, jac=_sgrad, args=(alpha, ml),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(r.x, n)
    except: return h

def _do_lbfgs(h, alpha, ml, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (n*0.5,)}]
    try:
        r = minimize(_sobj, h, jac=_sgrad, args=(alpha, ml),
                     method='L-BFGS-B', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'maxfun': maxiter*4})
        return _repair(r.x, n)
    except: return h

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _proj_subgrad(h, n_steps, start_tol=1e-3, end_tol=1e-8, n_act=4):
    n = len(h)
    target = n * 0.5
    best_h, best_c = h.copy(), _eval(h)
    step = 0.004
    for i in range(n_steps):
        tol = start_tol * (end_tol / start_tol) ** (i / max(n_steps - 1, 1))
        c = _corr(h)
        m = c.max()
        idx = np.argsort(c)[-n_act:]
        active = c >= (m - tol)
        acts = np.where(active)[0]
        if len(acts) == 0: acts = idx
        elif len(acts) > n_act: acts = idx
        dx = 2.0 / n
        g = np.zeros(n)
        for k in acts:
            L = k - (n - 1)
            if L >= 0:
                if n - L > 0: g[L:n] += (1.0 - h[:n-L])
                if L > 0: g[:n-L] -= h[L:n]
            else:
                M = -L
                if n - M > 0: g[:n-M] += (1.0 - h[M:n])
                if M > 0: g[M:n] -= h[:n-M]
        g *= dx / len(acts)
        taken = False
        for _ in range(18):
            hnew = _proj(h - step * g, target)
            cn = _eval(hnew)
            ch = _eval(h)
            if cn < ch:
                h, step = hnew, min(step * 1.04, 0.04)
                taken = True
                if cn < best_c: best_c, best_h = cn, h.copy()
                break
            step *= 0.5
        if not taken: step = min(step * 1.2, 0.04)
    return best_h, best_c

def _cycling_local(h, n_rounds, rng):
    n = len(h)
    best_h, best_c = h.copy(), _eval(h)
    cur = h.copy()
    cur_c = best_c
    for rnd in range(n_rounds):
        improved = False
        order = rng.permutation(n)
        for i in order:
            j_pool = [j for j in [ (i+1)%n, (i+7)%n, (i+n//3)%n, (i+n//2)%n ] if j != i]
            rng.shuffle(j_pool)
            for j in j_pool[:3]:
                for d_scale in [5e-6, 2e-6, 8e-7]:
                    d = d_scale * rng.randn()
                    ni, nj = cur[i] + d, cur[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        cur[i], cur[j] = ni, nj
                        cn = _eval(cur)
                        if cn < cur_c:
                            cur_c = cn; improved = True
                            if cn < best_c: best_c, best_h = cn, cur.copy()
                        else:
                            cur[i], cur[j] = ni - d, nj + d
            if cur_c < best_c - 1e-9:
                break
        if not improved: break
    return best_h, best_c

def _final_polish(h, n_iter, rng):
    n = len(h)
    best_h, best_c = h.copy(), _eval(h)
    cur = h.copy()
    cur_c = best_c

                ni, nj = cur[i] + dd, cur[j] - dd

                    cur[i], cur[j] = ni, nj
                    cn = _eval(cur)
                    if cn < cur_c:
                        cur_c = cn
                        if cn < best_c: best_c, best_h = cn, cur.copy()
                    else:
                        cur[i], cur[j] = ni - dd, nj + dd
    return best_h, best_c

def _binary_init(h, n, rng):
    hb = (h > 0.5).astype(float)
    s = hb.sum()
    target = n // 2
    if s != target:
        if s > target:
            ones = np.where(hb == 1)[0]
            for idx in ones[np.argsort(-h[ones])]:
                if s == target: break
                hb[idx] = 0; s -= 1
        else:
            zeros = np.where(hb == 0)[0]
            for idx in zeros[np.argsort(-h[zeros])]:
                if s == target: break
                hb[idx] = 1; s += 1
    return hb
