import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    for _ in range(50):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _to_bin(h, n, rng):
    b = (h + 0.25 * rng.randn(n) >= 0.5).astype(float)
    target = n // 2
    s = int(b.sum())
    if s > target:
        ones = np.where(b == 1)[0]
        perm = rng.permutation(ones)
        b[perm[:s - target]] = 0.0
    elif s < target:
        zeros = np.where(b == 0)[0]
        perm = rng.permutation(zeros)
        b[perm[:target - s]] = 1.0
    return b

    best_sc = _c5(b)
    cur_sc = best_sc
    for it in range(niter):

            break
        t = t0_val * (t1_val / t0_val) ** (float(it) / niter)
        # Random swap
        i1 = rng.randint(n)
        i2 = rng.randint(n)
        while i2 == i1:
            i2 = rng.randint(n)
        if b[i1] == b[i2]:
            continue
        b[i1], b[i2] = b[i2], b[i1]
        ns = _c5(b)

def _continuous_opt(h, n, t0, deadline, maxiter=2000):
    """SLSQP with soft-max objective."""
    alpha = 20000
    ml = 60
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        k = c.max()
        idx = np.argsort(c)[-ml:]
        s = alpha * (c[idx] - k)
        s -= s.max()
        return k + np.log(np.exp(s).sum()) / alpha
    
    def jac(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        k = c.max()
        idx = np.argsort(c)[-ml:]
        s = alpha * (c[idx] - k)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii, ki in enumerate(idx):
            if w[ii] < 1e-15:
                continue
            wk = w[ii] * dx
            L = ki - (n - 1)
            if L >= 0:
                g[L:] += wk * (1.0 - x[:n-L])
                g[:n-L] -= wk * x[L:]
            else:
                M = -L
                g[:n-M] += wk * (1.0 - x[M:])
                g[M:] -= wk * x[:n-M]
        return g
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5, 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(r.x, n)
    except:
        return h

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(60):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a

def _to_bin(h, n, rng):
    b = (h + 0.25 * rng.randn(n) >= 0.5).astype(float)
    t = n // 2
    s = int(b.sum())
    if s > t:

    elif s < t:

    return b
