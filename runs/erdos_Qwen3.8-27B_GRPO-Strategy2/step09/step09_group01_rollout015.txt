    return idx, corr[idx].copy()

def _compute_AB(h, ks, n):
    A = np.zeros(n)
    B = np.zeros(n)
    idx = np.arange(n)
    for k in ks:
        mf = (idx + k >= 0) & (idx + k < n)
        mb = (idx - k >= 0) & (idx - k < n)
        cf = np.zeros(n)
        cb = np.zeros(n)
        cf[mf] = 1.0 - h[idx[mf] + k]
        cb[mb] = h[idx[mb] - k]
        A += -cf + cb
        B += cf - cb
    return A, B

def _focused_search(h0, rng, tl, T=40, restarts=0):
    n = len(h0)
    h = _to_binary(h0, n).copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    tag, vals = _topN(corr, T)
    c5 = float(np.max(corr) * dx)
    best_c5 = c5
    best_h = h.copy()
    t0 = time.time()
    it = 0
    stag = 0
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    while time.time() - t0 < tl:
        if stag > 3000:
            h = best_h.copy()
            corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
            tag, vals = _topN(corr, T)
            c5 = float(np.max(corr) * dx)
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            stag = 0
            restarts += 1
            if restarts > 50:
                break
        ks = (tag - (n - 1)).astype(int)
        A, B = _compute_AB(h, ks, n)
        ai = np.full(n, 999.0)
        ai[ones] = A[ones]
        bj = np.full(n, 999.0)
        bj[zeros] = B[zeros]
        cand_count = min(8, len(ones) * len(zeros))
        best_delta = 999.0
        best_ii, best_jj = -1, -1
        order_i = np.argsort(ai[ones])[:min(30, len(ones))]
        order_j = np.argsort(bj[zeros])[:min(30, len(zeros))]
        checked = set()
        for oi in order_i:
            for oj in order_j:
                ii = int(ones[oi])
                jj = int(zeros[oj])
                if ii == jj or (ii, jj) in checked:
                    continue
                checked.add((ii, jj))
                d = A[ii] + B[jj]
                if d < best_delta:
                    best_delta = d
                    best_ii, best_jj = ii, jj
        fr = (time.time() - t0) / max(tl, 1e-9)
        temp = 3.0 * (1.0 - fr) ** 2
        improved = False
        trial = False
        if best_delta < -1e-12 or (temp > 0.1 and rng.random() < temp * 0.3):
            trial = best_ii >= 0 and best_jj >= 0
        if not trial and cand_count > 1:
            ii = int(ones[rng.randint(len(ones))])
            jj = int(zeros[rng.randint(len(zeros))])
            if ii != jj:
                trial = True
                best_ii, best_jj = ii, jj
        if trial:
            h[best_ii] = 0.0
            h[best_jj] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
            new_c5 = float(np.max(corr) * dx)
            if new_c5 <= c5 + 1e-15:
                c5 = new_c5
                tag, vals = _topN(corr, T)
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    stag = 0
                    improved = True
            else:
                h[best_ii] = 1.0
                h[best_jj] = 0.0
            stag += 1
        else:
            stag += 2
        it += 1
    c5f = _c5(h)
    if c5f < best_c5:
        best_c5 = c5f
        best_h = h.copy()
    return best_h, best_c5

def _fista_simple(h0, a, tl, T=30):
    n = len(h0)
    S = n * 0.5
    h = _repair(h0, n)
    bh = h.copy()
    bc = _c5(h)
    t0 = time.time()
    hp = h.copy()
    y = h.copy()
    tv = 1.0
    for it in range(200000):
        if time.time() - t0 > tl:
            break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tv * tv))
            y = h + ((tv - 1.0) / tn) * (h - hp)
            tv = tn
        corr = np.correlate(y, 1.0 - y, mode='full')
        tag, vals = _topN(corr, T)
        mx = vals[0]
        dx = 2.0 / n
        w = np.zeros(n)
        for m in tag:
            k = m - (n - 1)
            idx = np.arange(n)
            mf = (idx + k >= 0) & (idx + k < n)
            mb = (idx - k >= 0) & (idx - k < n)
            f = np.zeros(n)
            f[mf] = y[idx[mf] + k] * (1.0 - y[idx[mf]])
            f[mb] += y[idx[mb] - k] * (1.0 - y[idx[mb] - k])
            w += vals[0] * f / mx
        w *= dx / len(tag)
        step = 1.0 / max(a, 1e-6)
        for _ in range(6):
            xn = _proj(y - step * w, S)
            cx = _c5(xn)
            if cx <= _c5(y) + 0.5 * step * np.dot(w, xn - y) + 1e-10:
                break
            step *= 0.5
        hp = h.copy()
        h = xn
        c = _c5(h)
        if c < bc:
            bc = c
            bh = h.copy()
    return h, bh, bc

def _try_symmetries(h, n):
    best = h.copy()
    bc = _c5(best)
    for v in [1.0 - h, h[::-1],
              0.5 + 0.5 * (h - h[::-1]),
              0.5 + 0.5 * (h + h[::-1] - 1.0)]:
        hv = _proj(v, n * 0.5)
        cv = _c5(hv)
        if cv < bc:
            best, bc = hv, cv
    return best, bc

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    best = _to_binary(_repair(h0, n0), n0)
    best_c = _c5(best)

    best, best_c = _try_symmetries(best, n0)

    end_fista = t0 + 80.0
    if best_c > 0.3805:
        hf, bh, bc = _fista_simple(best, 200.0, end_fista - time.time(), T=40)
        if bc < best_c:
            best, best_c = bh, bc
        hf, bh, bc = _fista_simple(hf, 800.0, min(end_fista - time.time(), 40.0), T=50)
        if bc < best_c:
            best, best_c = bh, bc
        best = _to_binary(best, n0)
        best_c = _c5(best)

    end1 = t0 + 350.0
    if time.time() < end1:
        bh, bc = _focused_search(best, rng, end1 - time.time(), T=40)
        if bc < best_c:
            best, best_c = bh, bc

    end2 = t0 + 650.0
    if time.time() < end2:
        bh, bc = _focused_search(best, rng, end2 - time.time(), T=50)
        if bc < best_c:
            best, best_c = bh, bc

    end3 = t0 + 900.0
    if time.time() < end3:
        bh, bc = _focused_search(best, rng, end3 - time.time(), T=60)
        if bc < best_c:
            best, best_c = bh, bc

    end4 = t0 + budget_s - 10.0
    if time.time() < end4:
        bh, bc = _focused_search(best, rng, min(80.0, end4 - time.time()), T=60)
        if bc < best_c:
            best, best_c = bh, bc

    best, best_c = _try_symmetries(best, n0)
    best = _repair(best, n0)
    best_c = _c5(best)
    return best, best_c, n0
