def _repair(h, n):
    S = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        d = S - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    d = S - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            a = min(abs(d), h[i]) if d < 0 else min(abs(d), 1.0 - h[i])
            h[i] += np.sign(d) * a
            d -= np.sign(d) * a
            if abs(d) < 1e-13:
                break
    return h

def _c5(h):
