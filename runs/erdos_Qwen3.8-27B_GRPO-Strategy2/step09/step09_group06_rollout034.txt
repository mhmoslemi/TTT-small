```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_k_orig(h, k, n):
    """Gradient of result[k] (before 2/n scaling) w.r.t. h."""
    L = k - (n - 1)
    gk = np.zeros(n)
    if L >= 0:
        gk[:n - L] += (1.0 - h[L:])
        gk[L:] -= h[:n - L]
    else:
        M = -L
        gk[M:] += (1.0 - h[:n - M])
        gk[:n - M] -= h[M:]
    return gk

def _grad_max(h, n, ml, alpha):
    c = _corr(h, n)
    top = np.argsort(c)[-ml:]
    s = alpha * (c[top] - c.max())
    s = np.clip(s - s.max(), -500, 500)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(top)):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        gk = _grad_k_orig(h, top[ii], n)
        g += wk * gk
    return g

# SCA LP step
def _sca_lp(h, n):
    c = _corr(h, n)  # (2/n) * result[k]
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n * 0.5]
    for k in range(2 * n - 1):
        gk = _grad_k_orig(h, k, n)
        a = (2.0 / n) * gk
        const = c[k] - np.dot(a, h)
        cons.append(t_var >= const + a @ h_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if h_var.value is not None:
            return _repair(h_var.value, n)
    except Exception:
        pass
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        if h_var.value is not None:
            return _repair(h_var.value, n)
    except Exception:
        pass
    return h
```
