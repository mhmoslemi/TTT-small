import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _cd_pass(h, rng, niter_bin=50):
    n = len(h)
    c = _corr(h)
    cmax = 2 * n - 1
    nm1 = n - 1
    order = rng.permutation(n)
    for j in order:
        j = int(j)
        beta = np.zeros(cmax)
        k1_lo = j
        k1_hi = min(cmax - 1, j + nm1)
        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax - 1, cmax - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
        beta[k2_lo:k2_hi + 1] -= h[j + lags2]
        g = c - beta * h[j]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(niter_bin):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(g + beta * t1) <= np.max(g + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-15:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h, c

def _slsqp_obj(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    c -= c.max()
    c = np.clip(alpha * c, -500, 500)
    c -= c.max()
    return c.max() + np.log(np.exp(c).sum()) / alpha

def _slsqp_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(2 * n - 1):
        wk = w[ii] * dx
        if wk < 1e-15:
            continue
        L = ii - nm1
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fun(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp_opt(h, alpha, maxiter=500):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac,
             'args': (n * 0.5,)}]
    try:
        res = minimize(_slsqp_obj, h.copy(), jac=_slsqp_grad,
                       args=(alpha,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _fine_search(h, n, rng, niter, scale):
    best_h = h.copy()
    best_c5 = _eval(best_h)
    for _ in range(niter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        d = scale * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i] = ni
            h[j] = nj
            c5n = _eval(h)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h.copy()
            else:
                h[i] = ni - d
                h[j] = nj + d
    return best_h, best_c5

def _multi_cd(h, n, rng, t_end, niter_bin=50):
    best_h = h.copy()
    best_c5 = _eval(best_h)
    while time.time() < t_end:
        h, _ = _cd_pass(h, rng, niter_bin)
        h = _repair(h, n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5
