import numpy as np
import time

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-14:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(50):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _active_lags(corr, m=8):
    """Find indices of top m correlation values"""
    idx = np.argsort(corr)[::-1][:m]
    return idx

def _grad_c5_active(h, active_idx):
    """Compute gradient of max over active lags (equal weight)"""
    n = len(h)
    corr = _corr(h)
    vals = corr[active_idx]
    mx = vals.max()
    # Soft weights: exponential of (val - mx) / scale
    scale = max(1e-6, (mx - vals.min()) if len(vals) > 1 else 1.0)
    w = np.exp((vals - mx) / scale)
    w /= w.sum()
    
    # For each active lag, compute gradient contribution
    # corr[n-1+k] = sum_{i=0}^{n-k-1} h[i](1-h[i+k]) for k >= 0
    # corr[n-1-k] = sum_{i=k}^{n-1} h[i](1-h[i-k]) for k >= 0
    # gradient of corr[n-1+k] w.r.t. h[j]:
    #   = (1-h[j+k]) for j < n-k  (from h[j] term)
    #   - h[j-k] for j >= k       (from 1-h[j] term in h[j-k](1-h[j]))
    #   for k > 0
    # For k=0: gradient = 1 - 2*h[j]
    
    g = np.zeros(n)
    for wi, ki in zip(w, active_idx):
        k = ki - (n - 1)  # actual lag
        if k == 0:
            g += wi * (1.0 - 2.0 * h)
        elif k > 0:
            # corr[n-1+k] = sum_{i=0}^{n-k-1} h[i](1-h[i+k])
            # d/dh[j]: (1-h[j+k]) if j < n-k, minus h[j-k] if j >= k
            g[:n-k] += wi * (1.0 - h[k:])
            g[k:] -= wi * h[:n-k]
        else:
            # k < 0, let k' = -k
            kp = -k
            # corr[n-1-k'] = sum_{i=k'}^{n-1} h[i](1-h[i-k'])

            g[kp:] += wi * (1.0 - h[:n-kp])
            g[:n-kp] -= wi * h[kp:]
    return g

One more thought: the coord descent phase. For each coordinate j:
1. Fix all h[i] for i ≠ j
2. Find the optimal h[j] that minimizes C₅

The optimal h[j] can be found by noting that C₅ depends on h[j] through:
- c(0): h[j](1-h[j]) → quadratic, minimized at h[j] = 0 or 1
- c(k) for k ≠ 0: h[j] appears linearly in terms like h[j](1-h[j+k]) and h[j-k](1-h[j])

1. Identifying which lag gives the max
2. Finding h[j] that minimizes that lag's contribution
3. Checking if other lags become the max
4. Repeating (at most n iterations)
