        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _hard_grad(h, m_lags):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    idx = np.argsort(c)[-m_lags:]
    g = np.zeros(n)
    for k in idx:
        L = k - (n - 1)
        if L >= 0:
            g[L:] += dx * (1.0 - h[:n - L])
            g[:n - L] -= dx * h[L:]
        else:
            M = -L
            g[:n - M] += dx * (1.0 - h[M:])
            g[M:] -= dx * h[:n - M]
    g -= g.mean()
    return g

def _smooth_grad(h, alpha, m_lags):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - h[:n - L])
            g[:n - L] -= wk * h[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n - M]
    g -= g.mean()
    return g

def _smooth_obj(x, alpha, m_lags):
    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _adam_proj(h, alpha, m_lags, steps, lr0, t_end):
    n = len(h)
    v = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    cur = _eval(h)
    lr = lr0
    for it in range(steps):
        if time.time() > t_end:
            break
        g = _smooth_grad(h, alpha, m_lags)
        v = b1 * v + (1.0 - b1) * g
        m2 = b2 * m2 + (1.0 - b2) * g * g
        vhat = v / (1.0 - b1 ** (it + 1))
        mhat = m2 / (1.0 - b2 ** (it + 1))
        step_dir = vhat / (np.sqrt(mhat) + eps)
        taken = False
        s = step_dir * lr
        for _ in range(7):
            h_new = _proj(h - s, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur - 1e-16:
                h = h_new
                cur = c_new
                lr = min(lr * 1.04, lr0 * 40.0)
                taken = True
                break
            s *= 0.3
        if not taken:
            lr = max(lr * 0.6, lr0 * 0.005)
    return _repair(h, n), cur

def _targeted_search(h, t_end, seed):
    n = len(h)
    rng = np.random.RandomState(seed)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    m_lags = 20
    stuck = 0
    while time.time() < t_end:
        g = _hard_grad(best_h, m_lags)
        gn = np.linalg.norm(g)
        if gn < 1e-18:
            break
        # Pick indices with highest |g|
        top_g = np.abs(g)
        n_pert = rng.randint(2, 7)
        top_idx = np.argsort(top_g)[-min(n_pert * 3, n):][::-1]
        idx = top_idx[:n_pert]
        d = (2e-8 + 8e-8 * rng.rand()) * rng.randn(n_pert)
        # Bias toward gradient descent direction
        for i, ix in enumerate(idx):
            d[i] -= 0.5 * (2e-8 + 8e-8) * g[ix] / gn * 0.1
        d -= d.mean()
        h_new = best_h.copy()
        h_new[idx] += d
        if h_new.min() < -0.15 or h_new.max() > 1.15:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5c - 5e-16:
            c5c = c5n
            best_h = h_new
            stuck = 0
            if c5n < best_c:
                best_c = c5n
        else:
            stuck += 1
            if stuck % 3000 == 0:
                d2 = 8e-6 * rng.randn(n)
                d2 -= d2.mean()
                cand = _repair(best_h + d2, n)
                cc = _eval(cand)
                if cc < best_c + 5e-4:
                    best_h = cand
                    c5c = cc
                stuck = 0
    return best_h, best_c

def _fine_polish(h, t_end, seed):
    n = len(h)
    rng = np.random.RandomState(seed)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    while time.time() < t_end:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 25)) % n
        d = (3e-9 + 2e-8 * rng.rand()) * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 3e-17:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _group_polish(h, t_end, seed, gs=4):
    n = len(h)
    rng = np.random.RandomState(seed)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    while time.time() < t_end:
        base = rng.randint(n)
        idx = np.sort((base + rng.randint(0, 35, size=gs)) % n)
        d = (3e-9 + 4e-8 * rng.rand()) * rng.randn(gs)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < -0.05 or h_new.max() > 1.05:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5c - 3e-17:
            c5c = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def _slsqp_quick(h, alpha, m_lags, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, t: x.sum() - t,
             'jac': lambda x, t: np.ones(len(x)), 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=lambda x: _smooth_grad(x, alpha, m_lags),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.success or res.x is not None:
            return _repair(res.x, n)
    except Exception:
        pass
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Try 1-h as alternative start
    h_alt = _repair(1.0 - h0, n)
    c_alt = _eval(h_alt)
    if c_alt < best_c5:
        best_c5, best_h = c_alt, h_alt.copy()

    # Phase 1: Quick SLSQP polish (40s)
    t1 = min(t0 + 40, t0 + budget_s * 0.05)
    h_new = _slsqp_quick(best_h.copy(), 2000000, 30, 300, t1)
    c5n = _eval(h_new)
    if c5n < best_c5:
        best_c5, best_h = c5n, h_new

    # Phase 2: Adam projected gradient - ramping alpha (180s)
    configs = [(300000, 30, 700, 0.008), (3000000, 35, 500, 0.004),
               (30000000, 45, 400, 0.002), (300000000, 55, 300, 0.001)]
    t_phase2_end = t0 + 220
    for alpha, ml, steps, lr0 in configs:
        if time.time() > t_phase2_end:
            break
        t_end = min(time.time() + 50, t_phase2_end)
        h_new, c5n = _adam_proj(best_h.copy(), alpha, ml, steps, lr0, t_end)
        if c5n < best_c5:
            best_c5, best_h = c5n, h_new

    # Phase 3: More Adam at very high alpha (100s)
    t_phase3_end = min(t0 + 320, t0 + budget_s * 0.35)
    h_new, c5n = _adam_proj(best_h.copy(), 5000000000, 60, 400, 0.0008, t_phase3_end)
    if c5n < best_c5:
        best_c5, best_h = c5n, h_new

    # Phase 4: Targeted gradient search (350s)
    t_phase4_end = min(t0 + 670, t0 + budget_s * 0.72)
    best_h, best_c5 = _targeted_search(best_h, t_phase4_end, seed)

    # Phase 5: Fine 2-point polish (200s)
    t_phase5_end = min(t0 + 870, t0 + budget_s * 0.87)
    best_h, best_c5 = _fine_polish(best_h, t_phase5_end, seed + 1)

    # Phase 6: Group coordination (150s)
    t_phase6_end = min(t0 + 1020, t0 + budget_s - 80)
    best_h, best_c5 = _group_polish(best_h, t_phase6_end, seed + 2, 4)

    # Phase 7: Final ultra-fine
    t_final = t0 + budget_s - 10
    best_h, best_c5 = _fine_polish(best_h, t_final, seed + 3)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
