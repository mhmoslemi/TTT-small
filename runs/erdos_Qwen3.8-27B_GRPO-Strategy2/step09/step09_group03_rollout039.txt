import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _coord_optimize(h, n, n_sweeps=5):
    """Coordinate-wise optimization with ternary search."""
    dx = 2.0 / n
    best_c5 = _eval(h)
    best_h = h.copy()
    
    for sweep in range(n_sweeps):
        for i in range(n):
            # Compute base correlation and slope for coordinate i
            # c[k] as function of h[i]: c[k] = base[k] + slope[k] * h[i]
            # First, set h[i] = 0 to get base
            saved = h[i]
            h[i] = 0.0
            base = _corr_full(h)  # O(n) via correlate
            h[i] = 1.0
            at_one = _corr_full(h)  # O(n) via correlate
            h[i] = saved
            slope = at_one - base  # c[k](1) - c[k](0) = slope
            
            # Now find h_i in [0,1] minimizing max_k (base[k] + slope[k]*h_i)
            # Ternary search
            lo, hi = 0.0, 1.0
            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(base + slope * m1)
                f2 = np.max(base + slope * m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            
            h_opt = (lo + hi) / 2.0
            h[i] = h_opt
        
        h = _repair(h, n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _slope(h, n, p):
    dx = 2.0 / n
    sl = np.zeros(2 * n - 1)
    lo1 = max(0, n - 1 - p)
    hi1 = min(2 * n - 2, n - 1 + p)
    for m in range(lo1, hi1 + 1):
        sl[m] += dx * (1.0 - h[p + m - n + 1])
    lo2 = max(0, p)
    hi2 = min(2 * n - 2, p + n - 1)
    for m in range(lo2, hi2 + 1):
        idx = p - m + n - 1
        if 0 <= idx < n:
            sl[m] -= dx * h[idx]
    return sl

def _coord_sweep(h, C, n, order, tsteps=50):
    dx = 2.0 / n
    length = 2 * n - 1
    for p in order:
        sl = np.zeros(length)
        lo1 = max(0, n - 1 - p)
        hi1 = min(2 * n - 2, n - 1 + p)
        for m in range(lo1, hi1 + 1):
            sl[m] += dx * (1.0 - h[p + m - n + 1])
        lo2 = max(0, p)
        hi2 = min(2 * n - 2, p + n - 1)
        for m in range(lo2, hi2 + 1):
            idx = p - m + n - 1
            if 0 <= idx < n:
                sl[m] -= dx * h[idx]
        hp = h[p]
        base = C - sl * hp
        lo, hi = 0.0, 1.0
        for _ in range(tsteps):
            m1 = lo + (hi - lo) * 0.3333
            m2 = hi - (hi - lo) * 0.3333
            if np.max(base + sl * m1) < np.max(base + sl * m2):
                hi = m2
            else:
                lo = m1
        h_new = (lo + hi) * 0.5
        delta = h_new - hp
        if abs(delta) > 1e-15:
            C += sl * delta
            h[p] = h_new
    return h, C

def _pair_sweep(h, C, n, pairs, tsteps=50):
    dx = 2.0 / n
    length = 2 * n - 1
    for (i, j) in pairs:
        S = h[i] + h[j]
        sl_i = np.zeros(length)
        lo1 = max(0, n - 1 - i)
        hi1 = min(2 * n - 2, n - 1 + i)
        for m in range(lo1, hi1 + 1):
            sl_i[m] += dx * (1.0 - h[i + m - n + 1])
        lo2 = max(0, i)
        hi2 = min(2 * n - 2, i + n - 1)
        for m in range(lo2, hi2 + 1):
            idx2 = i - m + n - 1
            if 0 <= idx2 < n:
                sl_i[m] -= dx * h[idx2]
        sl_j = np.zeros(length)
        lo1 = max(0, n - 1 - j)
        hi1 = min(2 * n - 2, n - 1 + j)
        for m in range(lo1, hi1 + 1):
            sl_j[m] += dx * (1.0 - h[j + m - n + 1])
        lo2 = max(0, j)
        hi2 = min(2 * n - 2, j + n - 1)
        for m in range(lo2, hi2 + 1):
            idx2 = j - m + n - 1
            if 0 <= idx2 < n:
                sl_j[m] -= dx * h[idx2]
        # C[m] = base[m] + sl_i[m]*h[i] + sl_j[m]*h[j], h[j]=S-h[i]
        # C[m](t) = base[m] + sl_j[m]*S + (sl_i[m]-sl_j[m])*t
        hi_i, hi_j = h[i], h[j]
        base = C - sl_i * hi_i - sl_j * hi_j + sl_j * S
        dsl = sl_i - sl_j
        lo, hi = max(0.0, S - 1.0), min(1.0, S)
        if lo >= hi:
            continue
        for _ in range(tsteps):
            m1 = lo + (hi - lo) * 0.3333
            m2 = hi - (hi - lo) * 0.3333
            if np.max(base + dsl * m1) < np.max(base + dsl * m2):
                hi = m2
            else:
                lo = m1
        t_new = (lo + hi) * 0.5
        hi_i, hi_j = h[i], h[j]
        if abs(t_new - hi_i) > 1e-15:
            C += sl_i * (t_new - hi_i) + sl_j * (S - t_new - hi_j)
            h[i] = t_new
            h[j] = S - t_new
    return h, C

def _smooth_obj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, min(ml, 2 * n - 2))[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sum_fn(x, half):
    return x.sum() - half

def _slsqp(h, alpha, ml, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _smooth_obj if False else _sum_fn, 'args': (n * 0.5,)}]
    # Use gradient-free SLSQP with smooth objective
    from scipy.optimize import minimize as _min
    def _obj(x):
        return _smooth_obj(x, alpha, ml)
    def _grad(x):
        n2 = len(x)
        dx2 = 2.0 / n2
        c = np.correlate(x, 1.0 - x, mode='full') * dx2
        m = c.max()
        idx = np.argpartition(-c, min(ml, 2 * n2 - 2))[:ml]
        s = np.clip(alpha * (c[idx] - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n2)
        for ii, k in enumerate(idx):
            if w[ii] < 1e-15:
                continue
            wk = w[ii] * dx2
            L = k - (n2 - 1)
            if L >= 0:
                g[L:] += wk * (1.0 - x[:n2 - L])
                g[:n2 - L] -= wk * x[L:]
            else:
                M = -L
                g[:n2 - M] += wk * (1.0 - x[M:])
                g[M:] -= wk * x[:n2 - M]
        return g
    try:
        res = _min(_obj, h, jac=_grad, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 30})
        return _repair(res.x, n)
    except Exception:
        return h
