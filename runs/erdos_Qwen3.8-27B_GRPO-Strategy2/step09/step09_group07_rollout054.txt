```python
def _alternating_lp(h, n):
    """Given h, solve LP for g = 1-h to minimize max correlation."""
    # c[k] = sum_i h[i * (1-h[i+shift])]
    # For fixed h, this is linear in g: c[k] = sum_i h[i * g[i+shift_k]
    # where shift depends on k.
    # 
    # Actually, let me think about this more carefully.
    # np.correlate(h, g, 'full')[k] = sum_i h[i] * g[i + k - (n-1)]
    # 
    # For k = nm1 + L (L >= 0): terms are h[i]*g[i+L] for i=0..n-1-L
    # For k = nm1 + L (L < 0): terms are h[i]*g[i+L] for valid i
    #
    # So c[k] = sum_i h[i] * g[i + L] where L = k - (n-1)
    # This is linear in g with coefficients h[i] at appropriate positions.
    
    # LP: minimize t
    # subject to: for each k, sum_{valid i} h[i] * g[i+L_k] <= t
    #             0 <= g[i] <= 1
    #             sum(g) = n/2
    #             t free
    
    nm1 = n - 1
    cmax = 2 * n - 1
    
    # Build A matrix: (cmax + 2) rows, (n + 1) cols
    # Row 0..cmax-1: correlation constraints
    # Row cmax: sum constraint (equality, handled separately)
    # Actually for linprog, equalities are separate.
    
    # A_ineq * x <= b_ineq
    # x = [g[0], ..., g[n-1], t]
    
    A = np.zeros((cmax, n + 1))
    b = np.zeros(cmax)
    
    for k in range(cmax):
        L = k - nm1
        if L >= 0:
            # g indices: L to n-1, h indices: 0 to n-1-L
            for i in range(n - L):
                A[k, i + L] = h[i]
        else:
            M = -L
            # g indices: 0 to n-1-M, h indices: M to n-1
            for i in range(n - M):
                A[k, i] = h[i + M]
        A[k, n] = -1.0  # -t on the left side: sum - t <= 0
    
    b[:] = 0.0
    
    # Equality: sum(g) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 <= g[i] <= 1, t free
    bounds = [(0, 1)] * n + [(None, None)]
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return None
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _cd_pass(h, rng):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    cmax = 2 * n - 1
    jlist = np.arange(n)
    rng.shuffle(jlist)
    nm1 = n - 1
    for j in jlist:
        j = int(j)
        beta = np.zeros(cmax)
        k1_lo = j
        k1_hi = min(cmax - 1, j + nm1)
        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax - 1, cmax - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
        beta[k2_lo:k2_hi + 1] -= h[j + lags2]
        g = c - beta * h[j]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(30):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(g + beta * t1) <= np.max(g + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-14:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

def _alt_lp(fixed, n, rng):
    """Given 'fixed' (either h or g=1-h), solve LP to optimize the other."""
    nm1 = n - 1
    cmax = 2 * n - 1
    rows, cols, data = [], [], []
    for k in range(cmax):
        L = k - nm1
        if L >= 0:
            for i in range(n - L):
                rows.append(k); cols.append(i + L); data.append(fixed[i])
        else:
            M = -L
            for i in range(n - M):
                rows.append(k); cols.append(i); data.append(fixed[i + M])
        rows.append(k); cols.append(n); data.append(-1.0)
    A = csr_matrix((data, (rows, cols)), shape=(cmax, n + 1))
    b = np.zeros(cmax)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x[:n]
    return None

def _alternating(h, n, n_iters, rng, t0, budget_s, frac):
    best_h = h.copy()
    best_c5 = _eval(best_h)
    for it in range(n_iters):

            h_candidate = _repair(h_candidate, n)

                best_h = h_candidate.copy()

    return best_h, best_c5

def _slsqp_obj(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(len(idx)):
        wk = w[ii]
        if wk < 1e-15:
            continue
        wk *= dx
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fun(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _local_search(h, n, rng, n_iter):
    c5c = _eval(h)
    best_h = h.copy()
    best_c5 = c5c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 15)) % n
        d = 1e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

    h = best_h.copy()
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _cd_pass(h, rng, n_iterations=35):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = 2 * n - 1
    jlist = np.arange(n)
    rng.shuffle(jlist)
    nm1 = n - 1
    for j in jlist:
        j = int(j)
        beta = np.zeros(cmax)
        k1_lo = j
        k1_hi = min(cmax - 1, j + nm1)
        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax - 1, cmax - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
        beta[k2_lo:k2_hi + 1] -= h[j + lags2]
        g = c - beta * h[j]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(n_iterations):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(g + beta * t1) <= np.max(g + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-14:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

def _slsqp_obj(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(len(idx)):
        wk = w[ii]
        if wk < 1e-15:
            continue
        wk *= dx
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fun(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _local_search(h, n, rng, n_iter):
    c5c = _eval(h)
    best_h = h.copy()
    best_c5 = c5c
    step_sizes = [5e-5, 1e-5, 2e-6, 5e-6]
    for it in range(n_iter):
        ss = step_sizes[it % len(step_sizes)]
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 20)) % n
        d = ss * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def _perturb_cd(h, n, rng, n_pert, cd_passes, deadline):
    h = h.copy()
    idx = rng.choice(n, size=n_pert, replace=False)
    h[idx] = rng.uniform(0, 1, size=n_pert)
    h = _repair(h, n)
    for _ in range(cd_passes):
        if time.time() > deadline:
            break
        h = _cd_pass(h, rng)
        h = _repair(h, n)
    return h, _eval(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: CD passes (50% budget)
    deadline = t0 + budget_s * 0.50
    passes = 0
    while time.time() < deadline:
        h = _cd_pass(h, rng)
        h = _repair(h, n)
        passes += 1
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if passes % 25 == 0 and time.time() > deadline:
            break

    # Phase 2: SLSQP refinement (20% budget)
    deadline = t0 + budget_s * 0.70
    for alpha in [10000, 100000, 1000000, 5000000, 20000000]:
        if time.time() > deadline:
            break
        h_new = _slsqp(best_h.copy(), alpha, 60, 1000)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 3: CD from SLSQP result (10% budget)
    h = best_h.copy()
    deadline = t0 + budget_s * 0.80
    while time.time() < deadline:
        h = _cd_pass(h, rng)
        h = _repair(h, n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Perturbation + CD (5% budget)
    deadline = t0 + budget_s * 0.85
    while time.time() < deadline:
        h_p, c5_p = _perturb_cd(best_h, n, rng, n // 10, 10, deadline)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p

    # Phase 5: Final SLSQP (5% budget)
    deadline = t0 + budget_s * 0.90
    if time.time() < deadline:
        h_new = _slsqp(best_h.copy(), 50000000, 100, 800)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 6: Local search (10% budget)
    deadline = t0 + budget_s * 0.98
    n_iter = max(5000, int((deadline - time.time()) / 0.002))
    n_iter = min(n_iter, 500000)
    best_h, best_c5 = _local_search(best_h, n, rng, n_iter)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
```
