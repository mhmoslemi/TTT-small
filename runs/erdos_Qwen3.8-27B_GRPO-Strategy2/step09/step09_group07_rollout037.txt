def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:

            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)

            else:

                diff += a

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h


def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def _eval(h):
    return float(np.max(_corr(h)))


def _cd_pass(h, rng):
    n = len(h)
    c = _corr(h)
    cmax = 2 * n - 1
    jlist = np.arange(n)
    rng.shuffle(jlist)
    nm1 = n - 1
    for j in jlist:
        j = int(j)
        lo = max(0, nm1 - j)
        hi = min(cmax - 1, nm1 + j)
        mr = np.arange(lo, hi + 1)
        beta = (1.0 - h[j + mr - nm1]) - h[j - mr + nm1]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(50):
            tm = 0.5 * (lo_t + hi_t)
            ms = int(np.argmax(c + beta * tm))
            if beta[ms] > 1e-15:
                hi_t = tm
            elif beta[ms] < -1e-15:
                lo_t = tm
            else:
                lo_t = hi_t = tm
                break
        tn = 0.5 * (lo_t + hi_t)
        if abs(tn - h[j]) > 1e-14:
            c += beta * (tn - h[j])
            h[j] = tn
    return h


def _sq_obj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha


def _sq_grad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = idx[ii] - nm1
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g


def _sfun(x, tgt):
    return x.sum() - tgt


def _sjac(x, tgt):
    return np.ones(len(x))


def _slsqp(h, alpha, ml, mx):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sfun, 'jac': _sjac, 'args': (n * 0.5,)}]
    try:
        r = minimize(_sq_obj, h, jac=_sq_grad, args=(alpha, ml),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': mx, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return h


def _lsearch(h, n, rng, nit, ds):
    c5 = _eval(h)
    bh = h.copy()
    bc5 = c5
    for _ in range(nit):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, max(2, n // 30))) % n
        d = ds * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5:
                c5 = c5n
                if c5n < bc5:
                    bc5 = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc5


def _resize(h, nn):
    no = len(h)
    xo = np.linspace(0.0, 2.0, no, endpoint=False) + 1.0 / no
    xn = np.linspace(0.0, 2.0, nn, endpoint=False) + 1.0 / nn
    f = interp1d(xo, h, kind='linear', fill_value='extrapolate')
    return _repair(f(xn), nn)

    h0 = np.array(initial_h_values, dtype=float).copy()

        h2 = best_h + 5e-4 * rng.randn(n_b)

            h2 = _repair(h2, n_b)
        c5 = _eval(h2)
        if c5 < best_c5:

    # Phase 3: SLSQP (~15%)

    n_b = len(best_h)
    for al, ml, mx in [(5000, 40, 500), (50000, 60, 500), (500000, 80, 400)]:
        if time.time() > p3e:
            break
        hn = _slsqp(best_h.copy(), al, ml, mx)
        c5n = _eval(hn)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn

    # Phase 4: More CD (~20%)
    p4e = t0 + budget_s * 0.80
    h3 = best_h.copy()
    n3 = len(h3)
    while time.time() < p4e:
        h3 = _cd_pass(h3, rng)
        h3 = _repair(h3, n3)
        c5 = _eval(h3)
        if c5 < best_c5:
            best_c5, best_h = c5, h3.copy()

    # Phase 5: High-alpha SLSQP
    if time.time() < t0 + budget_s - 60:
        hn = _slsqp(best_h.copy(), 5000000, 100, 400)
        c5n = _eval(hn)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn

    # Phase 6: Final local search
    if time.time() < t0 + budget_s - 5:
        n_ls = len(best_h)
        for ds in [2e-4, 5e-5, 1e-5, 5e-6]:
            rem = t0 + budget_s - 10 - time.time()
            if rem < 10:
                break
            nt = min(int(rem / 0.005), 60000)
            best_h, best_c5 = _lsearch(best_h, n_ls, rng, nt, ds)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
