def _kelly_lp(h, m_lags, max_iter, prev_cuts=None):
    """Cutting plane (Kelley) method for min max_k c[k]."""
    n = len(h)
    target = n * 0.5
    cuts = prev_cuts if prev_cuts is not None else []
    
    for it in range(max_iter):
        c = _corr_dx(h)
        m = c.max()
        idx = np.argsort(c)[-m_lags:]
        
        # Add new cuts from current h
        for k_idx in idx:
            L = k_idx - (n - 1)
            g = _corr_grad_single(h, L)
            val = c[k_idx]
            # Cut: val + g·(x - h) <= t  =>  g·x <= t - val + g·h
            rhs = -val + np.dot(g, h)  # g·x <= t + rhs... 
            # Actually: c_k(h + d) ≈ c_k(h) + g·d
            # We want c_k(h') <= t for all active k
            # c_k(h) + g·(h' - h) <= t
            # g·h' <= t - c_k(h) + g·h
            # Since g·h is computable: but actually g is the gradient, and c_k(h) is the value
            # So: g·h' <= t + (g·h - c_k(h))
            # Note: g·h is NOT c_k(h) in general (it's for a bilinear form, g·h != c_k(h) unless it's linear)
            # For bilinear form x^T A x: ∇f = 2Ax (or (A+A^T)x), and g·x = 2x^TAx = 2f(x) for symmetric A
            # But our form is not a standard quadratic...
            
            # Let me just use: f_k(x) ≈ f_k(h) + ∇f_k(h)·(x-h)
            # So: f_k(h) + g·(x-h) <= t
            # g·x <= t - f_k(h) + g·h
            rhs_const = -val + np.dot(g, h)
            cuts.append((g.copy(), rhs_const))
        
        # Solve LP: min t s.t. g_i·x <= t + rhs_i for all cuts, sum(x)=target, 0<=x<=1
        # This is: min t s.t. g_i·x - t <= rhs_i, sum(x)=target, 0<=x<=1
        
        # Use scipy linprog or cvxpy
        import cvxpy as cp
        x_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [x_var >= 0, x_var <= 1, cp.sum(x_var) == target]
        for g_cut, rhs_cut in cuts:
            constraints.append(g_cut @ x_var - t_var <= rhs_cut)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        
        try:
            prob.solve(solver=cp.GLOP)  # or CLP, or CBC
        except:
            prob.solve(solver=cp.CBC)
        
        if x_var.value is None:
            break
        
        h_new = np.clip(x_var.value, 0, 1)
        h_new = _repair(h_new, n)
        c_new = _eval(h_new)
        
        if c_new < c.max() - 1e-12:
            # Check if LP solution satisfies all cuts (it should by construction)
            # Evaluate true objective
            if c_new < _eval(h):
                h = h_new
            else:
                # LP gave a point with worse true objective - add more cuts
                continue
        else:
            # Converged or no improvement
            # Check if we're at a fixed point
            break
        
        # Limit number of cuts for tractability
        if len(cuts) > 200:
            # Keep only the most violated/recent cuts
            c_check = _corr_dx(h)
            # Keep m_lags + some buffer
            cuts = cuts[-150:]
    
    return h, _eval(h)

import numpy as np
from scipy.optimize import minimize, linprog

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    for _ in range(100):
        h *= target / max(s, 1e-30)
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-15:
                break
            a = min(abs(diff), max(1.0 - h[idx], 0) if diff > 0 else max(h[idx], 0))
            h[idx] += a if diff > 0 else -a
            diff += -a if diff > 0 else a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_dx(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _corr_grad_at(h, L):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if L >= 0:
        m = n - L
        g[L:] += dx * (1.0 - h[:m])
        g[:m] -= dx * h[L:]
    else:
        M = -L
        m = n - M
        g[:m] += dx * (1.0 - h[M:])
        g[M:] -= dx * h[:m]
    return g

def _top_grad(h, m):
    n = len(h)
    c = _corr_dx(h)
    idx = np.argsort(c)[-m:]
    g = np.zeros(n)
    for k in idx:
        g += _corr_grad_at(h, k - (n - 1))
    g -= g.mean()
    return g, c[idx]

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _fista(h, m, n_iter, step0):
    n = len(h)
    target = n * 0.5
    x = h.copy()
    y = h.copy()
    k = 1.0
    best_x = h.copy()
    best_c = _eval(h)
    step = step0
    for it in range(n_iter):
        g, _ = _top_grad(y, m)
        y_new = _proj(y - step * g, target)
        c_new = _eval(y_new)
        if c_new < best_c:
            best_c = c_new
            best_x = y_new.copy()
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * k * k)) * 0.5
        y = y_new + ((k - 1.0) / t_new) * (y_new - x)
        x = y_new
        k = t_new
        if it > 0 and it % 200 == 0:
            if _eval(x) > best_c + 1e-9:
                step *= 0.7
    return _repair(best_x, n), best_c

def _mom_grad(h, m, n_iter, step0, mu=0.9):
    n = len(h)
    target = n * 0.5
    best_h = h.copy()
    best_c = _eval(h)
    v = np.zeros(n)
    step = step0
    for it in range(n_iter):
        g, _ = _top_grad(h, m)
        v = mu * v + (1.0 - mu) * g
        ng = np.linalg.norm(v)
        if ng < 1e-18:
            break
        v /= ng
        taken = False
        for _t in range(12):
            h_new = _proj(h - step * v, target)
            c_new = _eval(h_new)
            if c_new < _eval(h) - 1e-16:
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                step = min(step * 1.04, step0 * 25.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(max(step * 1.2, step0 * 0.3), step0 * 25.0)
    return _repair(best_h, n), best_c

def _grad_pairs(h, n_iter, rng, m=20):
    """Gradient-guided pair swaps."""
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    c = _corr_dx(h)
    active = np.argsort(c)[-m:]

        i = rng.randint(n)
        # Choose j to maximize |g[i] - g[j]|
        diff_g = np.abs(g[i] - g)
        diff_g[i] = 0
        j = np.argmax(diff_g)
        d = (2e-6 + 8e-6 * rng.rand()) * rng.randn()
        # Make d decrease the max
        if g[i] < g[j]:

        else:

        ni, nj = h[i] + d, h[j] - d

                    best_c = c5n
                    best_h = h.copy()

def _slsqp_obj(x, alpha, m):
    n = len(x)
    c = _corr_dx(x)
    mx = c.max()
    idx = np.argsort(c)[-m:]
    s = np.clip(alpha * (c[idx] - mx), -400, 400)
    s -= s.max()
    return mx + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, m):
    n = len(x)
    dxv = 2.0 / n
    c = _corr_dx(x)
    mx = c.max()
    idx = np.argsort(c)[-m:]
    s = np.clip(alpha * (c[idx] - mx), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dxv
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _cons_fun(x, tgt):
    return x.sum() - tgt

def _cons_jac(x, tgt):
    return np.ones(len(x))

def _do_slsqp(h, alpha, m, maxiter):
    n = len(h)
    tgt = n * 0.5
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (tgt,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad,
                       args=(alpha, m), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-17})
        return _repair(res.x, n)
    except:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    for _ in range(100):
        h *= target / max(s, 1e-30)
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-15:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_dx(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _cg(h, L):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if L >= 0:
        m = n - L
        g[L:] += dx * (1.0 - h[:m])
        g[:m] -= dx * h[L:]
    else:
        M = -L
        m = n - M
        g[:m] += dx * (1.0 - h[M:])
        g[M:] -= dx * h[:m]
    return g

def _topg(h, m):
    n = len(h)
    c = _corr_dx(h)
    idx = np.argsort(c)[-m:]
    g = np.zeros(n)
    for k in idx:
        g += _cg(h, k - (n - 1))
    g -= g.mean()
    return g, c

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _fista(h, m, n_it, s0):
    n = len(h)
    tgt = n * 0.5
    x = h.copy()
    y = h.copy()
    kk = 1.0
    bx = h.copy()
    bc = _eval(h)
    step = s0
    for it in range(n_it):
        g, _ = _topg(y, m)
        yn = _proj(y - step * g, tgt)
        cn = _eval(yn)
        if cn < bc:
            bc = cn
            bx = yn.copy()
        tn = (1.0 + np.sqrt(1.0 + 4.0 * kk * kk)) * 0.5
        y = yn + ((kk - 1.0) / tn) * (yn - x)
        x = yn
        kk = tn
        if it > 0 and it % 300 == 0:
            if _eval(x) > bc + 1e-9:
                step *= 0.7
                y = x.copy()
                kk = 1.0
    return _repair(bx, n), bc

def _momg(h, m, n_it, s0):
    n = len(h)
    tgt = n * 0.5
    bh = h.copy()
    bc = _eval(h)
    v = np.zeros(n)
    step = s0
    mu = 0.92
    for it in range(n_it):
        g, _ = _topg(h, m)
        v = mu * v + (1.0 - mu) * g
        ng = np.linalg.norm(v)
        if ng < 1e-18:
            break
        v /= ng
        taken = False
        for _t in range(12):
            hn = _proj(h - step * v, tgt)
            cn = _eval(hn)
            if cn < _eval(h) - 1e-16:
                h = hn
                if cn < bc:
                    bc = cn
                    bh = h.copy()
                step = min(step * 1.05, s0 * 30.0)
                taken = True
                break
            step *= 0.38
        if not taken:
            step = min(max(step * 1.3, s0 * 0.4), s0 * 30.0)
    return _repair(bh, n), bc

def _gpair(h, n_it, rng, m=20):
    n = len(h)
    cc = _eval(h)
    bh = h.copy()
    bc = cc
    c = _corr_dx(h)
    act = np.argsort(c)[-m:]

        if it % 40 == 0:

        g = np.zeros(n)
        for k in act:
            g += _cg(h, k - (n - 1))
        i = rng.randint(n)
        dg = np.abs(g[i] - g)
        dg[i] = 0
        j = np.argmax(dg)
        d = (2e-6 + 6e-6 * rng.rand())

            d = abs(d)
        else:
            d = -abs(d)
        ni, nj = h[i] + d, h[j] - d

                    bc = cn
                    bh = h.copy()
            else:

    return bh, bc

def _sobj(x, alpha, m):
    n = len(x)
    c = _corr_dx(x)
    mx = c.max()
    idx = np.argsort(c)[-m:]
    s = np.clip(alpha * (c[idx] - mx), -400, 400)
    s -= s.max()
    return mx + np.log(np.exp(s).sum()) / alpha

def _sgrad(x, alpha, m):
    n = len(x)
    dv = 2.0 / n
    c = _corr_dx(x)
    mx = c.max()
    idx = np.argsort(c)[-m:]
    s = np.clip(alpha * (c[idx] - mx), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dv
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _cfun(x, t):
    return x.sum() - t

def _cjac(x, t):
    return np.ones(len(x))

def _slsqp(h, alpha, m, maxit):
    n = len(h)
    t = n * 0.5
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac, 'args': (t,)}]
    try:
        res = minimize(_sobj, h, jac=_sgrad, args=(alpha, m),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-17})
        return _repair(res.x, n)
    except Exception:
        return h
