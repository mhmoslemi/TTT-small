def _proj(v, S):

        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    best = _repair(h0, n0)
    best_c = _c5(best)
    for var in [1.0 - best, best[::-1]]:
        hv = _proj(var, n0 * 0.5)
        cv = _c5(hv)
        if cv < best_c:
            best, best_c = hv, cv

    # Phase 1: FISTA (120s)
    end1 = t0 + 120.0
    hc = best.copy()
    for a in [50, 200, 800, 3000, 10000, 40000]:
        rem = end1 - time.time()
        if rem < 5:
            break
        hf, bh, bc = _fista(hc, a, rem * 0.18)
        if bc < best_c:
            best, best_c = bh, bc
        hc = hf

    # Phase 2: Gradient-guided local search (350s)
    end2 = t0 + 470.0
    best, best_c = _loc_search(best, rng, end2, s0=2e-4)

    # Phase 3: Resolution exploration (180s)
    end3 = t0 + 650.0
    for nm in [480, 640, 720, 840, 560]:
        rem = end3 - time.time()
        if rem < 40:
            break
        hr = _resample(best, nm)
        cr = _c5(hr)
        if cr < best_c - 0.0001:
            hf, bh, bc = _fista(hr, 3000, min(30.0, rem * 0.2))
            if bc < cr:
                cr, hr = bc, bh
        if cr < best_c:
            best, best_c = _repair(hr, nm), cr
        rem = end3 - time.time()
        if rem > 60:
            bh2, bc2 = _loc_search(best if len(best) == nm else _resample(hr, len(best)), rng, min(rem * 0.4, 60.0), s0=1e-4)
            if bc2 < best_c:
                best, best_c = bh2, bc2

    # Phase 4: Fine local search (300s)
    end4 = t0 + budget_s - 100.0
    best, best_c = _loc_search(best, rng, end4, s0=5e-5)

    # Phase 5: Micro polish (90s)
    end5 = t0 + budget_s - 10.0
    best, best_c = _loc_search(best, rng, end5, s0=1e-5)

    # Final: symmetry and repair
    n = len(best)
    for var in [1.0 - best, best[::-1], 0.5 + 0.5 * (best - best[::-1])]:
        hv = _proj(var, n * 0.5)
        cv = _c5(hv)
        if cv < best_c:
            best, best_c = hv, cv
    best = _repair(best, n)
    best_c = _c5(best)
    return best, best_c, n
