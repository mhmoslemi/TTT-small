import time

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    for _ in range(200):
        if abs(s - target) < 1e-14:

        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()

                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
    return h

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    c = _corr_vec(x)
    top = np.partition(c, -m)[-m:]

    s -= s.max()
    lse = mx + np.log(np.exp(s).sum()) / alpha
    # Penalty for constraint violation
    sv = x.sum() - n * 0.5
    lse += pen * sv * sv
    return lse

    c_max = c.max()

    # Add constraint penalty gradient

    g += 2.0 * pen * sv * np.ones(n)

        if np.clip(h - mid, 0, 1).sum() > target:

        else:

            else:

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    for _ in range(200):
        if abs(s - target) < 1e-14:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-15:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smooth_obj(x, alpha, m, pen):
    n = len(x)
    c = _corr_vec(x)
    idx = np.argpartition(c, -m)[-m:]
    top = c[idx]
    mx = top.max()
    s = alpha * (top - mx)
    s -= s.max()
    lse = mx + np.log(np.exp(s).sum()) / alpha
    sv = x.sum() - n * 0.5
    return lse + pen * sv * sv

def _smooth_grad(x, alpha, m, pen):
    n = len(x)
    dx = 2.0 / n
    c = _corr_vec(x)
    idx = np.argsort(c)[-m:]
    c_idx = c[idx]
    c_max = c.max()
    s = alpha * (c_idx - c_max)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-14:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    sv = x.sum() - n * 0.5
    g += 2.0 * pen * sv
    return g

def _slsqp_run(h, alpha, m, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)},)
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, m, 0.0), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-17})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj_thresh(h, target):
    lo, hi = -10.0, 10.0
    for _ in range(90):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _proj_grad(h, alpha, m, steps, step0, momentum=0.88):
    n = len(h)
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    for it in range(steps):
        g = _smooth_grad(h, alpha, m, 0.0)
        v = momentum * v + (1.0 - momentum) * g
        ng = np.linalg.norm(v)
        if ng < 1e-16:
            break
        v /= ng
        taken = False
        for _t in range(14):
            h_new = _proj_thresh(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-17:
                h = h_new; cur_c = c_new
                step = min(step * 1.04, step0 * 40.0)
                taken = True
                break
            step *= 0.38
        if not taken:
            step = max(step * 1.25, step0 * 0.3)
            step = min(step, step0 * 40.0)
    return _repair(h, n)

def _nesterov(h, alpha, m, steps, step0):
    n = len(h)
    y = h.copy()
    v = np.zeros(n)
    cur_c = _eval(h)
    for it in range(steps):
        t_k = 2.0 / (it + 3.0)
        g = _smooth_grad(y, alpha, m, 0.0)
        v = (it / (it + t_k)) * v - step0 * t_k * g
        y_new = _proj_thresh(y + v, n * 0.5)
        c_new = _eval(y_new)
        if c_new <= cur_c + 1e-16:
            y = y_new
            cur_c = c_new
    return _repair(y, n)

def _coord_search(h, n_iter, rng, step_base):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for it in range(n_iter):
        grp = rng.randint(1, 4)
        idx = rng.randint(0, n, size=grp)
        if np.any(np.diff(np.sort(idx)) == 0):
            continue
        d = step_base * (0.4 + 0.6 * rng.rand()) * rng.randn(grp)
        d -= d.mean()
        h_new = h.copy()
        ok = True
        for i, ix in enumerate(idx):
            nv = h[ix] + d[i]
            if nv < 0 or nv > 1:
                ok = False
                break
            h_new[ix] = nv
        if not ok:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-16:
            c5c = c5n; h = h_new
            if c5n < best_c:
                best_c = c5n; best_h = h.copy()
    return best_h, best_c

def _ultra_fine(h, n_iter, rng):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 20)) % n
        d = (0.5e-8 + 2e-8 * rng.rand()) * rng.randn()
        hi, hj = h[i] + d, h[j] - d
        if 0 <= hi <= 1 and 0 <= hj <= 1:
            h[i], h[j] = hi, hj
            c5n = _eval(h)
            if c5n < c5c - 1e-17:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n; best_h = h.copy()
            else:
                h[i], h[j] = hi - d, hj + d
    return best_h, best_c

    t0 = time.time()
    rng = np.random.RandomState(seed)
