def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(50):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-14:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-13:

    d = t - h.sum()

            a = min(abs(d), h[i]) if d < 0 else min(abs(d), 1.0 - h[i])

import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(50):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-14:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-13:
            break
    d = t - h.sum()
    if abs(d) > 1e-13:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if abs(d) < 1e-13:
                break
            a = min(abs(d), h[i]) if d < 0 else min(abs(d), 1.0 - h[i])
            h[i] += a if d > 0 else -a
    return h

def _resample(h, m):
    n = len(h)
    if n == m:
        return _proj(h.copy(), m * 0.5)
    x0 = (np.arange(n) + 0.5) * 2.0 / n
    x1 = (np.arange(m) + 0.5) * 2.0 / m
    return _proj(np.interp(x1, x0, h), m * 0.5)

def _to_bin(h, n):
    b = np.zeros(n)
    b[np.argsort(h)[::-1][:n // 2]] = 1.0
    return b

def _sa(h0, rng, tl, T0, guided):
    n = len(h0)
    h = h0.copy()
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    c = _c5(h)
    bc, bh = c, h.copy()
    t0, stall = time.time(), 0
    while time.time() - t0 < tl:
        if guided > 0 and rng.random() < guided:
            corr = np.correlate(h, 1.0 - h, mode='full')
            k = int(np.argmax(corr)) - (n - 1)
            i, j = None, None
            if 0 <= k < n:
                co = [x for x in range(n - k) if h[x] > 0.5 and h[x + k] <= 0.5]
                if co:
                    i = co[rng.randint(len(co))]
                    cz = [x for x in zeros if x != i and 0 <= x + k < n and h[x + k] > 0.5]
                    j = cz[rng.randint(len(cz))] if cz else zeros[rng.randint(len(zeros))]
            if i is None:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        cn = _c5(h)
        d = cn - c
        fr = min((time.time() - t0) / max(tl, 1e-9), 1.0)
        T = max(T0 * (1.0 - fr) ** 3, 1e-7)
        if d < 0 or rng.random() < np.exp(-d / T):
            c = cn
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c < bc:
                bc, bh = c, h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i], h[j] = 1.0, 0.0
        if stall > 3000:
            for _ in range(20):
                a = ones[rng.randint(len(ones))]
                b = zeros[rng.randint(len(zeros))]
                h[a], h[b] = 0.0, 1.0
            c = _c5(h)
            ones = np.where(h > 0.5)[0].tolist()
            zeros = np.where(h <= 0.5)[0].tolist()
            stall = 0
    return bh, bc

def _refine(h0, rng, tl):
    n = len(h0)
    h = _repair(h0.copy(), n)
    c, bc, bh = _c5(h), _c5(h), h.copy()
    t0, step = time.time(), 0.02
    while time.time() - t0 < tl:
        ok = False
        for _ in range(40):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = step * (2.0 * rng.random() - 1.0)
            hi, hj = h[i] + d, h[j] - d
            if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                h[i], h[j] = hi, hj
                cn = _c5(h)
                if cn < c - 1e-15:
                    c, ok = cn, True
                    if c < bc:
                        bc, bh = c, h.copy()
                else:
                    h[i], h[j] = hi - d, hj + d
        if not ok:
            step *= 0.7
            if step < 1e-9:
                break
        else:
            step = min(step * 1.02, 0.05)
    h = _repair(h, n)
    c = _c5(h)
    if c < bc:
        bc, bh = c, h.copy()
    return bh, bc

def _periodic(n, blk, rng):
    p = np.zeros(n)
    for i in range(0, n, 2 * blk):
        p[i:i + blk] = 1.0
    p = np.roll(p, rng.randint(n))
    cnt = int(p.sum())
    if cnt > n // 2:
        idx = np.where(p > 0.5)[0]
        sel = rng.choice(idx, cnt - n // 2, replace=False)
        p[sel] = 0.0
    elif cnt < n // 2:
        idx = np.where(p <= 0.5)[0]
        sel = rng.choice(idx, n // 2 - cnt, replace=False)
        p[sel] = 1.0
    return p

def run(seed=42, budget_s=1100, **kwargs):
    ts = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        h0 = np.full(400, 0.5)
    best = _repair(h0.copy(), len(h0))
    bc = _c5(best)
    for v in [1.0 - best, best[::-1], 0.5 + 0.5 * (best - best[::-1])]:
        hv = _proj(v, len(best) * 0.5)
        cv = _c5(hv)
        if cv < bc:
            best, bc = hv, cv
    deadline = ts + budget_s - 8.0
    n_list = [200, 240, 280, 320, 360, 400, 480, 560, 640]
    for ni, n in enumerate(n_list):
        rem = deadline - time.time()
        if rem < 20:
            break
        alloc = rem / (len(n_list) - ni)
        h_s = _resample(best, n)
        starts = [_to_bin(h_s, n)]
        rb = np.zeros(n)
        rb[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(rb)
        starts.append(_to_bin(_periodic(n, 2, rng), n))
        starts.append(_to_bin(_periodic(n, 5, rng), n))
        for s in starts[:4]:
            rem2 = deadline - time.time()
            if rem2 < 8:
                break
            for T0, gd in [(0.05, 0.2), (0.02, 0.3)]:
                rem3 = deadline - time.time()
                if rem3 < 5:
                    break
                tt = min(rem3 * 0.35, alloc * 0.4 / len(starts))
                hb, cb = _sa(s, rng, tt, T0, gd)
                if cb < bc:
                    best, bc = hb, cb
            rem4 = deadline - time.time()
            if rem4 < 5:
                break
            best, bc = _refine(best, rng, min(rem4 * 0.2, alloc * 0.15 / len(starts)))
    rem = deadline - time.time()
    if rem > 20:
        n = len(best)
        hb, cb = _sa(_to_bin(best, n), rng, rem * 0.35, 0.01, 0.3)
        if cb < bc:
            best, bc = hb, cb
        best, bc = _refine(best, rng, min(rem * 0.15, deadline - time.time()))
        rem3 = deadline - time.time()
        if rem3 > 10:
            hb, cb = _sa(_to_bin(best, len(best)), rng, rem3 * 0.5, 0.005, 0.25)
            if cb < bc:
                best, bc = hb, cb
            rem4 = deadline - time.time()
            if rem4 > 3:
                best, bc = _refine(best, rng, rem4)
    n = len(best)
    best = _repair(best, n)
    bc = _c5(best)
    return best, bc, n
