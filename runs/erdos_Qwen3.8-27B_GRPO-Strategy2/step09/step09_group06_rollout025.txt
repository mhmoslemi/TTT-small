from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _obj_fn(x, n, m, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    idx = np.argsort(c)[-m:]
    cm = c[idx].max()
    s = alpha * (c[idx] - cm)
    s = np.clip(s - s.max(), -500, 500)
    return cm + np.log(np.exp(s).sum()) / alpha

def _slsqp(h, alpha, m_lags, maxiter, n):
    target = n * 0.5
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj_fn, h, method='SLSQP', bounds=bnds, constraints=cons,
                       args=(n, m_lags, alpha),
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _guided_pair_search(h, n_iter, rng):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(best_h)
    c5cur = best_c
    for it in range(n_iter):
        c = np.correlate(best_h, 1.0 - best_h, mode='full') * (2.0 / n)
        cmax = c.max()
        active = np.where(c > cmax - 0.005)[0]
        # Pick a pair weighted by active lag structure
        for _ in range(4):
            if len(active) > 0:
                ak = active[rng.randint(len(active))]
                L = ak - (n - 1)
                # Coordinates most affected by lag L
                if L >= 0:
                    coords = np.arange(max(0, n - L - 3), min(n, n - L + 3))
                else:
                    M = -L
                    coords = np.arange(max(0, M - 3), min(n, M + 3))
                if len(coords) < 2:
                    coords = np.arange(n)
                i = coords[rng.randint(len(coords))]
            else:
                i = rng.randint(n)
            j = (i + 1 + rng.randint(0, n // 3)) % n
            if i == j:
                continue
            s_val = best_h[i] + best_h[j]
            t_min = max(0.0, s_val - 1.0)
            t_max = min(1.0, s_val)
            if t_max - t_min < 1e-6:
                continue
            # Try several splits
            best_tc = c5cur
            best_vi = best_h[i]
            for vi in np.linspace(t_min, t_max, 9):
                vj = s_val - vi
                if vi < 0 or vi > 1 or vj < 0 or vj > 1:
                    continue
                best_h[i] = vi
                best_h[j] = vj
                tc = _eval(best_h)
                if tc < best_tc:
                    best_tc = tc
                    best_vi = vi
            best_h[i] = best_vi
            best_h[j] = s_val - best_vi
            if best_tc < c5cur - 1e-15:
                c5cur = best_tc
                if c5cur < best_c:
                    best_c = c5cur
        best_h = _repair(best_h, n)
    return best_h, best_c

def _micro_polish(h, n_iter, rng):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(best_h)
    c5cur = best_c
    for it in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 15)) % n
        d = 4e-7 * rng.randn()
        ni, nj = best_h[i] + d, best_h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            best_h[i], best_h[j] = ni, nj
            c5n = _eval(best_h)
            if c5n < c5cur - 1e-16:
                c5cur = c5n
                if c5n < best_c:
                    best_c = c5n
            else:
                best_h[i], best_h[j] = ni - d, nj + d
    return best_h, best_c

def _proj_simplex(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _proj_grad(h, alpha, m_lags, nsteps, step0, n):
    v = np.zeros(n)
    mu = 0.5
    step = step0
    c5cur = _eval(h)
    for _ in range(nsteps):
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        cmax = c.max()
        idx = np.argsort(c)[-m_lags:]
        s = alpha * (c[idx] - cmax)
        w = np.exp(np.clip(s - s.max(), -500, 500))
        w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for ii, k in enumerate(idx):
            if w[ii] < 1e-15:
                continue
            wk = w[ii] * dx
            L = k - (n - 1)
            if L >= 0:
                g[:n-L] += wk * (1.0 - h[L:])
                g[n-L:] -= wk * h[:L] if L > 0 else np.array([0.0])
            else:
                M = -L
                g[M:] += wk * (1.0 - h[:n-M])
                g[:M] -= wk * h[M:]
        v = mu * v + (1.0 - mu) * g
        taken = False
        for _t in range(12):
            h_new = _proj_simplex(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < c5cur - 1e-15:
                h = h_new
                c5cur = c_new
                step = min(step * 1.05, step0 * 8.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.4, step0 * 8.0)
    return _repair(h, n), c5cur
