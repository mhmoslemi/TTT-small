        def _proj(hv, target):
            lo,hi=-3.0,3.0
            for _ in range(60):
                mid=(lo+hi)*0.5
                if np.clip(hv-mid,0,1).sum()>target: lo=mid
                else: hi=mid
            return np.clip(hv-(lo+hi)*0.5,0,1)
        for al,ml,st,s0,mu in [(50000,40,300,0.003,0.85),(500000,60,300,0.001,0.90),(5000000,80,250,0.0005,0.92)]:
            if _el()>budget_s-250: break
            hh=best_h.copy(); cur=_eval(hh); v=np.zeros(n); step=s0
            for _ in range(st):
                g=_gbf(hh,al,ml)
                v=mu*v+(1-mu)*g
                taken=False
                for _ in range(10):
                    hn=_proj(hh-step*v,n*0.5)
                    cn=_eval(hn)
                    if cn<cur:
                        hh,cur,taken=hn,cn,True
                        step=min(step*1.05,s0*12)
                        break
                    step*=0.45
                if not taken: step=min(step*1.2,s0*12)
            hc=_repair(hh,n); cc=_eval(hc)
            if cc<best_c: best_c,best_h=cc,hc

    # Phase 4: Active-set coordinate descent (main work)
    if _el()<budget_s-80:
        best_h, best_c = _coord_descent(best_h, n, rng, _el, budget_s-80)

    # Phase 5: Another SLSQP pass after coordinate descent
    if _el()<budget_s-30:
        for al,ml in [(1000000,60),(10000000,80)]:
            if _el()>budget_s-10: break
            try:
                r=minimize(_obf,best_h,jac=_gbf,args=(al,ml),method='SLSQP',
                           bounds=bnds,constraints=cons,options={'maxiter':3000,'ftol':1e-18})
                hc=_repair(r.x,n); cc=_eval(hc)
                if cc<best_c: best_c,best_h=cc,hc
            except: pass

    # Phase 6: Ultra-fine polish
    if _el()<budget_s-2:
        h2=best_h.copy(); c2=best_c
        niter=min(int((budget_s-2-_el())/0.0004),3000000)
        for _ in range(niter):
            if _el()>budget_s-2: break
            i=rng.randint(n)
            j=(i+1+rng.randint(0,4))%n
            d=5e-7*rng.randn()
            ni,nj=h2[i]+d,h2[j]-d
            if 0<=ni<=1 and 0<=nj<=1:
                h2[i],h2[j]=ni,nj
                cn=_eval(h2)
                if cn<c2:
                    c2=cn
                    if cn<best_c: best_c,best_h=cn,h2.copy()
                else:
                    h2[i],h2[j]=ni-d,nj+d

    best_h=_repair(best_h,n)
    best_c=_eval(best_h)
    return (best_h, best_c, n)
