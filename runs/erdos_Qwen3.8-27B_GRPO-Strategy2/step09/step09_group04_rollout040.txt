            if cn < _eval(bh):
                bh = h.copy()
    return bh, _eval(bh)

def _swap_polish(h, n, rng, niter, scale):
    h2 = h.copy()
    c2 = _eval(h2)
    bs, bc = h2.copy(), c2
    for _ in range(niter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, max(1, n // 3))) % n
        d = scale * rng.randn()
        ni, nj = h2[i] + d, h2[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h2[i], h2[j] = ni, nj
            cn = _eval(h2)
            if cn < c2:
                c2 = cn
                if cn < bc:
                    bc, bs = cn, h2.copy()
            else:
                h2[i], h2[j] = h2[i] - d, h2[j] + d
    return bs, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h, best_c5 = h.copy(), _eval(h)

    def _el():
        return time.time() - t0

    # Phase 1: Quick SLSQP polish (establish baseline)
    for a, ml, mi in [(5000, 25, 300), (50000, 40, 400), (500000, 60, 400)]:
        if _el() > 60:
            break
        hn = _slsqp(best_h, a, ml, mi)
        cn = _eval(hn)
        if cn < best_c5:
            best_c5, best_h = cn, hn

    # Phase 2: Basin hopping - explore multiple basins
    # This is the key difference: perturb + re-optimize to escape local minima
    if _el() < budget_s - 200:
        hop_budget = budget_s - 200 - _el()
        # First round: moderate alpha, more iterations per hop
        lim1 = hop_budget * 0.4
        best_h, best_c5 = _basin_hop(best_h, n, rng, 50000, 50, 350, lim1)
        # Second round: higher alpha, fewer iterations
        if _el() < budget_s - 180:
            lim2 = budget_s - 180 - _el()
            best_h, best_c5 = _basin_hop(best_h, n, rng, 500000, 65, 250, lim2)

    # Phase 3: Focused high-alpha SLSQP
    if _el() < budget_s - 120:
        for a, ml, mi in [(5e6, 70, 600), (5e7, 90, 500), (5e8, 100, 400)]:
            if _el() > budget_s - 100:
                break
            hn = _slsqp(best_h, a, ml, mi)
            cn = _eval(hn)
            if cn < best_c5:
                best_c5, best_h = cn, hn

    # Phase 4: Gradient-guided climb for fine adjustments
    if _el() < budget_s - 80:
        rem = budget_s - 80 - _el()
        nt = min(int(rem / 0.008), 80000)
        best_h, best_c5 = _grad_climb(best_h, n, rng, 200000, 70, nt, 0.003)

    # Phase 5: Graduated swap polish (decreasing scale)
    if _el() < budget_s - 5:
        for scale in [5e-6, 1e-6, 3e-7, 1e-7, 3e-8]:
            remaining = budget_s - 5 - _el()
            if remaining < 5:
                break
            nt = min(int(remaining / (len([5e-6,1e-6,3e-7,1e-7,3e-8]) * 0.002)), 80000)
            best_h, best_c5 = _swap_polish(best_h, n, rng, nt, scale)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
