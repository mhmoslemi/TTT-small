def _subgradient_step(h, step_size, n_lags=5):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = 2 * n - 1
    nm1 = n - 1
    
    # Pick top n_lags lags
    idx = np.argsort(c)[-n_lags:]
    
    # Compute subgradient for each of these lags
    grad = np.zeros(n)
    for k in idx:
        k = int(k)
        L = k - nm1
        w = 1.0 / n_lags
        if L >= 0:
            # c(nm1+L) = sum_{i=0}^{n-1-L} h[i]*(1-h[i+L])
            # dc/dh[i] = (1-h[i+L]) for i < n-L (first factor)
            #            -h[i-L] for i >= L (second factor)
            for i in range(n - L):
                grad[i] += w * (1.0 - h[i + L])
            for i in range(L, n):
                grad[i] -= w * h[i - L]
        else:
            M = -L
            # c(nm1-M) = sum_{i=M}^{n-1} h[i]*(1-h[i-M])
            # dc/dh[i] = (1-h[i-M]) for i >= M (first factor)
            #            -h[i+M] for i < n-M (second factor)
            for i in range(M, n):
                grad[i] += w * (1.0 - h[i - M])
            for i in range(n - M):
                grad[i] -= w * h[i + M]
    
    # Take step
    h_new = h - step_size * grad
    # Project to [0,1]^n
    h_new = np.clip(h_new, 0.0, 1.0)
    # Project to sum = n/2
    h_new = _repair(h_new, n)
    return h_new

                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-14:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

def _subgrad_step(h, step, n_top):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    nm1 = n - 1
    idx = np.argsort(c)[-n_top:]
    grad = np.zeros(n)
    for k in idx:
        k = int(k)
        L = k - nm1
        w = 1.0 / n_top
        if L >= 0:
            grad[:n-L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n-L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n-M])
            grad[:n-M] -= w * h[M:]
    h_new = np.clip(h - step * grad, 0.0, 1.0)
    return _repair(h_new, n)

def _slsqp_obj(x, alpha, m_lags):

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, m_lags):

        else:

f(t) = max_k (a_k + b_k * t), t ∈ [0,1]

The minimum is at:
- t = 0 if f'(0+) ≥ 0 (all active constraints have b_k ≤ 0... wait no)
  Actually, f'(0+) = max{b_k : a_k = max_j a_j} (the max slope among the active constraints at t=0)
  If f'(0+) ≥ 0, the minimum is at t=0.
- t = 1 if f'(1-) ≤ 0
  f'(1-) = max{b_k : a_k + b_k = max_j (a_j + b_j)}
  If f'(1-) ≤ 0, the minimum is at t=1.
- Otherwise, the minimum is at an intersection of two lines a_i + b_i t = a_j + b_j t where the upper envelope changes.

The exact minimum can be found by:
1. Computing the upper envelope of the lines {a_k + b_k t} for t ∈ [0,1]
2. Finding the minimum of the envelope

The upper envelope of n lines can be computed in O(n log n) time using a convex hull trick. But implementing this correctly is tricky.

A simpler approach: since the function is convex, we can use bisection on the value:
- Binary search for the minimum value v
- For a given v, check if min_t max_k (a_k + b_k t - v) ≤ 0
  This is equivalent to: does there exist t ∈ [0,1] such that a_k + b_k t ≤ v for all k?
  For b_k > 0: t ≤ (v - a_k) / b_k
  For b_k < 0: t ≥ (v - a_k) / b_k
  For b_k = 0: need a_k ≤ v
  So we need: max_{b_k<0} (v-a_k)/b_k ≤ min(1, min_{b_k>0} (v-a_k)/b_k) and a_k ≤ v for b_k=0

This is O(n) per check, and bisection takes O(log(1/ε)) iterations. With ε = 1e-12, that's about 40 iterations × O(n) = O(n log(1/ε)).

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - tgt) < 1e-14:
            break
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-14: break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _cd_pass(h, rng, n_iter=50):
    n = len(h)
    c = _corr(h)
    cm = 2 * n - 1
    jl = np.arange(n)
    rng.shuffle(jl)
    nm = n - 1
    for j in jl:
        j = int(j)
        bt = np.zeros(cm)
        k1l, k1h = j, min(cm-1, j+nm)
        l1 = np.arange(k1l, k1h+1) - nm
        bt[k1l:k1h+1] += (1.0 - h[j-l1])
        k2l, k2h = max(0, nm-j), min(cm-1, cm-1-j)
        l2 = np.arange(k2l, k2h+1) - nm
        bt[k2l:k2h+1] -= h[j+l2]
        g = c - bt * h[j]
        lo, hi = 0.0, 1.0
        for _ in range(n_iter):
            t1 = lo + (hi-lo)/3.0
            t2 = hi - (hi-lo)/3.0
            if np.max(g + bt*t1) <= np.max(g + bt*t2): hi = t2
            else: lo = t1
        tn = 0.5*(lo+hi)
        if abs(tn - h[j]) > 1e-14:
            c += bt*(tn - h[j])
            h[j] = tn
    return h

    idx = np.argsort(c)[-ntop:]
    gr = np.zeros(n)
    for k in idx:
        k = int(k); L = k - nm; w = 1.0/ntop
        if L >= 0:
            gr[:n-L] += w*(1.0-h[L:]); gr[L:] -= w*h[:n-L]
        else:
            M = -L; gr[M:] += w*(1.0-h[:n-M]); gr[:n-M] -= w*h[M:]
    return _repair(np.clip(h - step*gr, 0.0, 1.0), n)

def _lo_func(x, a, ml):
    n = len(x); dx = 2.0/n
    c = np.correlate(x, 1.0-x, mode='full')*dx
    m = c.max()
    ix = np.argsort(c)[-ml:]
    s = np.clip(a*(c[ix]-m), -500, 500); s -= s.max()
    return m + np.log(np.exp(s).sum())/a

def _lo_grad(x, a, ml):
    n = len(x); dx = 2.0/n
    c = np.correlate(x, 1.0-x, mode='full')*dx
    m = c.max()
    ix = np.argsort(c)[-ml:]
    s = np.clip(a*(c[ix]-m), -500, 500); s -= s.max()
    w = np.exp(s); w /= w.sum()
    nm = n-1; gr = np.zeros(n)
    for ii in range(len(ix)):
        wk = w[ii]*dx
        if wk < 1e-15: continue

        else:

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - tgt) < 1e-14:
            break
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-14: break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _cd_pass(h, rng, nit):
    n = len(h)
    c = _corr(h)
    cm = 2 * n - 1
    jl = np.arange(n)
    rng.shuffle(jl)
    nm = n - 1
    for j in jl:
        j = int(j)
        bt = np.zeros(cm)
        k1l = j
        k1h = min(cm - 1, j + nm)
        l1 = np.arange(k1l, k1h + 1) - nm
        bt[k1l:k1h+1] += (1.0 - h[j - l1])
        k2l = max(0, nm - j)
        k2h = min(cm - 1, cm - 1 - j)
        l2 = np.arange(k2l, k2h + 1) - nm
        bt[k2l:k2h+1] -= h[j + l2]
        g = c - bt * h[j]
        lo, hi = 0.0, 1.0
        for _ in range(nit):
            t1 = lo + (hi - lo) / 3.0
            t2 = hi - (hi - lo) / 3.0
            if np.max(g + bt * t1) <= np.max(g + bt * t2):
                hi = t2
            else:
                lo = t1
        tn = 0.5 * (lo + hi)
        if abs(tn - h[j]) > 1e-14:
            c += bt * (tn - h[j])
            h[j] = tn
    return h

def _subgrad(h, step, ntop):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    nm = n - 1
    idx = np.argsort(c)[-ntop:]
    gr = np.zeros(n)
    for k in idx:
        k = int(k); L = k - nm; w = 1.0 / ntop
        if L >= 0:
            gr[:n-L] += w * (1.0 - h[L:]); gr[L:] -= w * h[:n-L]
        else:
            M = -L; gr[M:] += w * (1.0 - h[:n-M]); gr[:n-M] -= w * h[M:]
    return _repair(np.clip(h - step * gr, 0.0, 1.0), n)

def _lo_func(x, a, ml):
    n = len(x); dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    ix = np.argsort(c)[-ml:]
    s = np.clip(a * (c[ix] - m), -500, 500); s -= s.max()
    return m + np.log(np.exp(s).sum()) / a

def _lo_grad(x, a, ml):
    n = len(x); dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    ix = np.argsort(c)[-ml:]
    s = np.clip(a * (c[ix] - m), -500, 500); s -= s.max()
    w = np.exp(s); w /= w.sum()
    nm = n - 1; gr = np.zeros(n)
    for ii in range(len(ix)):
        wk = w[ii] * dx
        if wk < 1e-15: continue
        k = int(ix[ii]); L = k - nm
        if L >= 0:
            gr[:n-L] += wk * (1.0 - x[L:]); gr[L:] -= wk * x[:n-L]
        else:
            M = -L; gr[M:] += wk * (1.0 - x[:n-M]); gr[:n-M] -= wk * x[M:]
    return gr

def _sj(x, t): return np.ones(len(x))

    n = len(h)
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (n * 0.5,)}]
    try:

                     method='SLSQP', bounds=bd, constraints=cs,
                     options={'maxiter': mi, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return h
