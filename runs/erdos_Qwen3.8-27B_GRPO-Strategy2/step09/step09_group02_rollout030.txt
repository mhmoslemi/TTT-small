        xg = np.arange(n) / n

        pert -= pert.mean()
        hn = repair(h + pert, n)
        cn = eval_c5(hn)
        if cn < best_c:
            best_c = cn
            best_h = hn
    return best_h, best_c

def block_search(h, n_iter, rng):
    sizes = np.array([4, 6, 8, 12, 16, 24, 32])
    scales = np.array([0.004, 0.002, 0.001, 0.0005, 0.0002])
    n = len(h)
    cur = eval_c5(h)
    best_h = h.copy()
    best_c = cur
    for _ in range(n_iter):
        bs = int(rng.choice(sizes))
        sc = float(rng.choice(scales))
        start = rng.randint(0, max(1, n - bs))
        idx = np.arange(start, min(start + bs, n))
        d = sc * rng.randn(len(idx))
        d -= d.mean()
        hn = h.copy()
        hn[idx] += d
        if hn.min() < -0.2 or hn.max() > 1.2:
            continue
        hn = repair(hn, n)
        cn = eval_c5(hn)
        if cn < cur - 1e-15:
            cur = cn
            h = hn
            if cn < best_c:
                best_c = cn
                best_h = h.copy()
    return best_h, best_c

def fine_polish(h, n_iter, rng):
    n = len(h)
    cur = eval_c5(h)
    best_h = h.copy()
    best_c = cur
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 40)) % n
        d = (2e-9 + 1e-7 * rng.rand()) * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            cn = eval_c5(h)
            if cn < cur - 1e-16:
                cur = cn
                if cn < best_c:
                    best_c = cn
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def el():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = repair(h0, n)
    best_h = h.copy()
    best_c5 = eval_c5(best_h)

    # Phase 1: SLSQP with graded smoothness (small k to large k)
    seq1 = [(20000, 6, 2000, 1e-15), (100000, 12, 1500, 1e-15),
            (500000, 20, 1200, 1e-16), (5000000, 35, 1000, 1e-16),
            (50000000, 55, 800, 1e-17), (500000000, 90, 600, 1e-17)]
    for alpha, k, mi, ft in seq1:
        if el() > budget_s * 0.35:
            break
        hn = slsqp_opt(best_h.copy(), alpha, k, mi, ft)
        cn = eval_c5(hn)
        if cn < best_c5:
            best_c5 = cn
            best_h = hn

    # Phase 2: L-BFGS-B (different optimizer code path)
    seq2 = [(200000, 15, 4000, 1e-14), (5000000, 40, 3000, 1e-14),
            (100000000, 80, 2000, 1e-15)]
    for alpha, k, mi, gt in seq2:
        if el() > budget_s * 0.50:
            break
        hn = lbfgs_opt(best_h.copy(), alpha, k, mi, gt)
        cn = eval_c5(hn)
        if cn < best_c5:
            best_c5 = cn
            best_h = hn

    # Phase 3: Nesterov projected gradient (multiple configs)
    seq3 = [(200000, 25, 1000, 0.005, 0.65),
            (2000000, 50, 800, 0.003, 0.72),
            (20000000, 80, 600, 0.0015, 0.78),
            (200000000, 130, 500, 0.0006, 0.84)]
    for alpha, k, steps, s0, mom in seq3:
        if el() > budget_s - 280:
            break
        hn = nest_grad(best_h.copy(), alpha, k, steps, s0, mom)
        cn = eval_c5(hn)
        if cn < best_c5:
            best_c5 = cn
            best_h = hn

    # Phase 4: Sinusoidal restarts + gradient follow-up
    if el() < budget_s - 180:
        for _ in range(4):
            if el() > budget_s - 120:
                break
            hn, cn = sin_restart(best_h, rng, 10, 0.012)
            if cn < best_c5:
                best_c5 = cn
                best_h = hn
            hn2 = nest_grad(hn.copy(), 5000000, 50, 500, 0.002, 0.75)
            cn2 = eval_c5(hn2)
            if cn2 < best_c5:
                best_c5 = cn2
                best_h = hn2
            hn3 = slsqp_opt(hn.copy(), 10000000, 60, 400, 1e-16)
            cn3 = eval_c5(hn3)
            if cn3 < best_c5:
                best_c5 = cn3
                best_h = hn3

    # Phase 5: Block search (structured neighborhood exploration)
    if el() < budget_s - 50:
        rem = budget_s - 50 - el()
        ntr = int(min(rem * 120, 4000))
        best_h, best_c5 = block_search(best_h, ntr, rng)

    # Phase 6: Final SLSQP polish (very high alpha, many active lags)
    if el() < budget_s - 20:
        hn = slsqp_opt(best_h.copy(), 1000000000, 200, 400, 1e-17)
        cn = eval_c5(hn)
        if cn < best_c5:
            best_c5 = cn
            best_h = hn

    # Phase 7: Fine 2-point polish
    if el() < budget_s - 2:
        rem = budget_s - 2 - el()
        ntr = int(rem * 1200)
        best_h, best_c5 = fine_polish(best_h, ntr, rng)

    best_h = repair(best_h, n)
    best_c5 = eval_c5(best_h)
    return (best_h, best_c5, n)
