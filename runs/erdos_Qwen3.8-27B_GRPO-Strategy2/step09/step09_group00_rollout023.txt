import numpy as np
import time

def _repair(h, n):
    # Clip and fix sum to n/2
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    for _ in range(50):
        if abs(s - t) < 1e-12: break
        h *= t / s if s > 0 else 1.0
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
    d = t - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13: break
    return h

def _proj_box_sum(v, S):
    x = np.array(v, dtype=float).copy()
    for _ in range(50):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-12: break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0: break
        x[m] += d / c
    return x

def _c5(h):
    return float(np.max(np.correlate(h, 1.0-h, mode='full'))) * (2.0/len(h))

def _fft_size(n):
    m = 2*n - 1
    N = 1
    while N < m: N <<= 1
    return N

def _pn(h):
    """Compute positive and negative lag correlations"""
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.zeros(n + 1)
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.zeros(n)  # Positive lags: C[k] = sum h[i]*(1-h[i+k])
    Nv = np.zeros(n)  # Negative lags
    P[0] = total - ac[0]
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
    mx = max(float(P[0]), float(np.max(P[1:])), float(np.max(Nv[1:])))
    return P, Nv, mx

def _soft_val(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    dx = 2.0/n
    e = np.exp(np.clip(a*(P[0]-mx), -300, 0))
    if n > 1:
        e += np.exp(np.clip(a*(P[1:]-mx), -300, 0)).sum()
        e += np.exp(np.clip(a*(Nv[1:]-mx), -300, 0)).sum()
    return dx * (mx + np.log(max(e,1e-300)) / a)

def _grad_weights(h, wP, wN):
    n = len(h)
    dx = 2.0/n
    u = np.zeros(n)
    if n > 1: u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n); cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n-1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)
        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)
        q = -(R[n-1:2*n-1].real + L[:n].real)
    return dx * (wP[0]*(1.0-2.0*h) + linP + linN + q)

def _soft_grad(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    wP = np.zeros(n); wN = np.zeros(n)
    e0 = np.exp(np.clip(a*(P[0]-mx), -300, 0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(a*(P[1:]-mx), -300, 0))
        en = np.exp(np.clip(a*(Nv[1:]-mx), -300, 0))
        tot += ep.sum() + en.sum()
        if tot < 1e-300: tot = 1.0; e0 = 1.0
        wP[0] = e0/tot; wP[1:] = ep/tot; wN[1:] = en/tot
    else:
        wP[0] = 1.0
    return _grad_weights(h, wP, wN)

def _fista(h0, alpha, max_t, L0=10.0):

        accepted = False
        for _ in range(10):

            fx = _soft_val(xn, alpha)

            if fx <= rhs + 1e-10:

For n where neither n+k-n+1 nor n equals a or b: no change.
For n = a: h'[a+k-n+1](1-h'[a]) - h[a+k-n+1](1-h[a])
  = h[a+k-n+1](1-h[a]-da) - h[a+k-n+1](1-h[a])  [assuming a+k-n+1 ≠ b]
  = -da * h[a+k-n+1]
  Plus if a+k-n+1 = b: (h[b]-da)(1-h[a]-da) - h[b](1-h[a]) = -da(1-h[a]) - da*h[b] + da²

For n+k-n+1 = a (i.e., n = a+k-n+1): h'[a](1-h'[n]) - h[a](1-h[n])
  = (h[a]+da)(1-h[n]) - h[a](1-h[n])  [assuming n ≠ b]
  = da*(1-h[n])
  Plus if n = b: (h[a]+da)(1-h[b]+da) - h[a](1-h[b]) = da(1-h[b]) + da*h[a] + da²

So the cross terms (da²) exist when both a and b are "active" for the same k. But for small da, da² is negligible. And in practice, for the O(n) update, we're using:
Δoutput[k] ≈ da*(1-2*h[a+k-n+1]) + (-da)*(1-2*h[b+k-n+1])

This is the first-order approximation, which is exact when a and b don't interact in the same term. When they do (a+k-n+1 = b or b+k-n+1 = a), there's a da² correction. For |da| ≤ 0.05, da² ≤ 0.0025, which is negligible compared to the other terms. So the O(n) update is accurate to first order, which is fine for our accept/reject criterion.

import numpy as np
import time

def _fft_size(n):
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    return N

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    for _ in range(80):
        if s < 1e-15:
            return np.full(n, 0.5)
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-13:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float).copy()
    for _ in range(60):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-12:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _c5(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / len(h))

def _pn(h):
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.zeros(n + 1)
    np.cumsum(h, out=pref[1:])
    P = np.zeros(n)
    Nv = np.zeros(n)
    P[0] = pref[n] - ac[0]
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = pref[n] - pref[idx] - ac[idx]
    mx = float(P[0])
    if n > 1:
        mx = max(mx, float(np.max(P[1:])), float(np.max(Nv[1:])))
    return P, Nv, mx

def _soft_val(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    dx = 2.0 / n
    e = np.exp(np.clip(a * (P[0] - mx), -300, 0))
    if n > 1:
        e += np.exp(np.clip(a * (P[1:] - mx), -300, 0)).sum()
        e += np.exp(np.clip(a * (Nv[1:] - mx), -300, 0)).sum()
    return dx * (mx + np.log(max(e, 1e-300)) / a)

def _grad_weights(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)
        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)
        q = -(R[n - 1:2 * n - 1].real + L[:n].real)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _soft_grad(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(a * (P[0] - mx), -300, 0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(a * (P[1:] - mx), -300, 0))
        en = np.exp(np.clip(a * (Nv[1:] - mx), -300, 0))
        tot += ep.sum() + en.sum()
        if tot < 1e-300:
            tot = 1.0
            e0 = 1.0
        wP[0] = e0 / tot
        wP[1:] = ep / tot
        wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _grad_weights(h, wP, wN)

def _fista(h0, alpha, max_t, L0=10.0):
    n = len(h0)
    S = n * 0.5
    h = _repair(h0, n)
    hp = h.copy()
    y = h.copy()
    t = 1.0
    fy = _soft_val(h, alpha)
    best_h = h.copy()
    best_c = _c5(h)
    stale = 0
    Lc = L0
    t0 = time.time()
    for it in range(1000000):
        if time.time() - t0 > max_t:
            break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp)
            t = tn
            if y.min() < -0.1 or y.max() > 1.1 or abs(y.sum() - S) > 0.01:
                y = _proj(y, S)
            fy = _soft_val(y, alpha)
        g = _soft_grad(y, alpha)
        step = 1.0 / max(Lc, 1e-10)
        accepted = False
        for _ in range(12):
            xn = _proj(y - step * g, S)
            fx = _soft_val(xn, alpha)
            dv = xn - y
            rhs = fy + float(np.dot(g, dv)) + 0.5 * (1.0 / step) * float(np.dot(dv, dv))
            if fx <= rhs + 1e-10:
                accepted = True
                break
            step *= 0.5
        Lc = 1.0 / max(step, 1e-10)
        if accepted:
            hp = h
            h = xn
            stale = 0
            Lc = max(Lc * 0.88, 1.0)
            if it % 8 == 0:
                c = _c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
        else:
            stale += 1
            hp = h
            y = h
            t = 1.0
            fy = _soft_val(h, alpha)
            Lc = max(Lc * 0.5, 1.0)
            if stale > 15:
                break
    c = _c5(h)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return h, best_h, best_c

def _fast_local_search(h0, rng, deadline):
    n = len(h0)

    dx = 2.0 / n
    mx = float(corr.max())

    best_h = h.copy()
    t_start = time.time()
    total_t = max(1.0, deadline - t_start)
    it = 0
    one_minus_2h = 1.0 - 2.0 * h
    while time.time() < deadline - 0.2:
        frac = min(1.0, (time.time() - t_start) / total_t)
        T = 0.003 * (0.008) ** frac
        step_max = 0.06 * (0.15) ** frac
        if it % 2000 == 0:
            corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
            mx = float(corr.max())
            one_minus_2h = 1.0 - 2.0 * h
            if mx * dx < best_c:
                best_c = mx * dx
                best_h = h.copy()
        k_act = int(np.argmax(corr)) - (n - 1)
        if rng.random() < 0.72 and abs(k_act) > 0 and abs(k_act) < n:
            k = abs(k_act)

            if len(pos) > 2:

                j = i + k if k_act > 0 else i - k
                if rng.random() < 0.5:
                    d = -rng.random() * step_max
                else:
                    d = rng.random() * step_max
            else:
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    it += 1
                    continue
                d = (rng.random() - 0.5) * step_max
        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))

                it += 1

        if h[i] + d < 0.0:

            one_minus_2h = 1.0 - 2.0 * h
            cn = mx * dx
            if cn < best_c - 1e-15:

                best_h = h.copy()
        else:
            corr[lo_a:hi_a + 1] -= delta
            corr[lo_b:hi_b + 1] += delta
            h[i] -= d
            h[j] += d
        it += 1
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    if n % m == 0:
        out = h.reshape(m, n // m).mean(axis=1)
    else:
        x0 = (np.arange(n) + 0.5) * (2.0 / n)
        x1 = (np.arange(m) + 0.5) * (2.0 / m)
        out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)
