    t = n // 2
    order = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[order[:t]] = 1.0
    return hb

def _fft_corr(h, N, n):
    H = np.fft.rfft(h, N)
    return np.fft.irfft(H * np.conj(H), N).real[:n]

def _smooth_fn(h, alpha, N, n):
    ac = _fft_corr(h, N, n)
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    tot = pref[n]
    P0 = tot - ac[0]
    mx = P0
    if n > 1:
        idx = np.arange(1, n)
        P = pref[n - idx] - ac[idx]
        Nv = tot - pref[idx] - ac[idx]
        mx = max(mx, float(P.max()), float(Nv.max()))
    e = np.exp(np.clip(alpha * (P0 - mx), -400.0, 0.0))
    if n > 1:
        e += np.exp(np.clip(alpha * (P - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(alpha * (Nv - mx), -400.0, 0.0)).sum()
    return (2.0/n) * (mx + np.log(max(e, 1e-300)) / alpha)

def _smooth_grad(h, alpha, N, n):
    ac = _fft_corr(h, N, n)
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    tot = pref[n]
    P0 = tot - ac[0]
    mx = P0
    if n > 1:
        idx = np.arange(1, n)
        P = pref[n - idx] - ac[idx]
        Nv = tot - pref[idx] - ac[idx]
        mx = max(mx, float(P.max()), float(Nv.max()))
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(alpha * (P0 - mx), -400.0, 0.0))
    te = e0
    if n > 1:
        ep = np.exp(np.clip(alpha * (P - mx), -400.0, 0.0))
        en = np.exp(np.clip(alpha * (Nv - mx), -400.0, 0.0))
        te += ep.sum() + en.sum()
    if te < 1e-300:
        te, e0 = 1.0, 1.0
    wP[0] = e0 / te
    if n > 1:
        wP[1:] = ep / te
        wN[1:] = en / te
    # Gradient of the smooth function
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n-1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Hh = np.fft.rfft(h, N)
        R = np.fft.irfft(Hh * np.fft.rfft(u[::-1], N), N)[n-1:2*n-1].real
        L = np.fft.irfft(Hh * np.fft.rfft(u, N), N)[:n].real
        q = -(R + L)
    val = _smooth_fn(h, alpha, N, n)
    g = (2.0/n) * (wP[0]*(1.0-2.0*h) + linP + linN + q)
    return val, g

                break
            step *= 0.5

        x0 = (np.arange(n) + 0.5) * (2.0 / n)
        x1 = (np.arange(m) + 0.5) * (2.0 / m)
        h_m = np.interp(x1, x0, h)
        h_m = _proj(h_m, m * 0.5)
        cm = _c5(h_m)
        if cm < best_c:
            best, best_c = h_m, cm
            n_new = m
        # Quick local search at this resolution
        rem2 = tl - (time.time() - t0)
        if rem2 > 15:
            h_m2, c_m2 = _cont_ls(h_m.copy(), rng, min(rem2 * 0.3, 25.0), m, 3e-5)
            if c_m2 < best_c:
                best, best_c = h_m2, c_m2
                n_new = m
    return best, best_c, len(best)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n = len(h0)
    h0 = _repair(h0, n)
    best = h0.copy()
    best_c = _c5(best)

    # Phase 1: FISTA with multiple alphas (180s)
    end1 = t0 + 180.0
    for a in [80, 300, 1200, 6000, 30000, 100000]:
        rem = end1 - time.time()
        if rem < 5:
            break
        fh, fc = _fista(best, a, min(rem * 0.18, 30.0), n)
        if fc < best_c:
            best, best_c = fh, fc

    # Phase 2: Binarize + block local search (220s)
    end2 = t0 + 400.0
    hb = _binarize(best, n)
    cb = _c5(hb)
    if cb < best_c:
        best, best_c = hb, cb
    bh, bc = _block_ls(hb.copy(), rng, min(180.0, end2 - time.time()), n)
    if bc < best_c:
        best, best_c = bh, bc

    # Phase 3: FISTA from binary result (120s)
    end3 = t0 + 520.0
    for a in [500, 3000, 15000]:
        rem = end3 - time.time()
        if rem < 5:
            break
        fh, fc = _fista(best, a, min(rem * 0.3, 40.0), n)
        if fc < best_c:
            best, best_c = fh, fc

    # Phase 4: Second binary round (130s)
    end4 = t0 + 650.0
    hb2 = _binarize(best, n)
    if _c5(hb2) < best_c:
        best, best_c = hb2, _c5(hb2)
    bh2, bc2 = _block_ls(hb2.copy(), rng, min(110.0, end4 - time.time()), n)
    if bc2 < best_c:
        best, best_c = bh2, bc2

    # Phase 5: Continue FISTA + continuous LS (200s)
    end5 = t0 + 850.0
    for a in [2000, 10000, 50000]:
        rem = end5 - time.time()
        if rem < 5:
            break
        fh, fc = _fista(best, a, min(rem * 0.25, 40.0), n)
        if fc < best_c:
            best, best_c = fh, fc
    for s0 in [5e-5, 1e-5, 3e-6]:
        rem = end5 - time.time()
        if rem < 5:
            break
        fh, fc = _cont_ls(best.copy(), rng, min(rem * 0.25, 40.0), n, s0)
        if fc < best_c:
            best, best_c = fh, fc

    # Phase 6: Try different resolution (150s)
    end6 = t0 + budget_s - 10.0
    if end6 - time.time() > 30:
        br, bc_r, nr = _explore_resolutions(best, n, rng, min(140.0, end6 - time.time()))
        if bc_r < best_c:
            best, best_c, n = br, bc_r, nr

    # Final micro-LS
    rem = budget_s - 3.0 - (time.time() - t0)
    if rem > 5:
        fh, fc = _cont_ls(best.copy(), rng, rem, len(best), 5e-6)
        if fc < best_c:
            best, best_c = fh, fc

    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
