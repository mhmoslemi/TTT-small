import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
            else:
                a = min(-diff, h[idx])
            h[idx] += np.sign(diff) * a
            diff -= np.sign(diff) * a
            if abs(diff) < 1e-14:
                break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _cd_pass(h, rng):
    n = len(h)
    c = _corr(h)
    cmax_len = 2 * n - 1
    nm1 = n - 1
    jlist = np.arange(n)
    rng.shuffle(jlist)
    for j in jlist:
        j = int(j)
        beta = np.zeros(cmax_len)
        k1_lo = j
        k1_hi = min(cmax_len - 1, j + nm1)
        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax_len - 1, cmax_len - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
        beta[k2_lo:k2_hi + 1] -= h[j + lags2]
        g = c - beta * h[j]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(32):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(g + beta * t1) <= np.max(g + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-15:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

def _slsqp_obj(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(len(idx)):
        wk = w[ii]
        if wk < 1e-15:
            continue
        wk *= dx
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fun(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _local_search(h, n, rng, n_iter, step_scale):
    c5c = _eval(h)
    best_h = h.copy()
    best_c5 = c5c
    base_step = 8e-6 * step_scale
    for i in range(n_iter):
        frac = 1.0 - i / max(n_iter, 1)
        d = base_step * (0.05 + 0.95 * frac) * rng.randn()
        a = rng.randint(n)
        b = (a + 1 + rng.randint(0, 15)) % n
        na, nb = h[a] + d, h[b] - d
        if 0 <= na <= 1 and 0 <= nb <= 1:
            h[a], h[b] = na, nb
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[a], h[b] = na - d, nb + d
    return best_h, best_c5

def _subgrad_polish(h, n, rng, budget_end):
    h = h.copy()
    n_active = 15
    for it in range(5000):
        if time.time() > budget_end:
            break
        c = _corr(h)
        m = c.max()
        tol = 0.005 + 0.005 * np.exp(-it / 100.0)
        active = np.where(c > m - tol)[0]
        if len(active) == 0:
            active = np.argmax(c)
            active = np.array([active])
        g = np.zeros(n)
        nm1 = n - 1
        dx = 2.0 / n
        for k in active[:n_active]:
            L = k - nm1
            if L >= 0:
                g[:n-L] += (1.0 - h[L:]) * dx
                g[L:] -= h[:n-L] * dx
            else:
                M = -L
                g[:n-M] += (1.0 - h[M:]) * dx
                g[M:] -= h[:n-M] * dx
        g /= min(len(active), n_active)
        gn = np.dot(g, g)
        if gn < 1e-30:
            break
        step = 0.1 / np.sqrt(gn)
        h = _repair(h - step * g, n)
        if it % 20 == 19:
            h = _cd_pass(h, rng)
            h = _repair(h, n)
    return h
