  from scipy.optimize import minimize
  import time

  def _eval_c5(h):
      n = len(h)
      return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0 / n)

  def _repair(h, n):
      h = np.clip(h, 0.0, 1.0).copy()
      target = n / 2.0
      s = h.sum()
      if s < 1e-15:
          return np.full(n, 0.5)
      for _ in range(300):
          h *= target / s
          h = np.clip(h, 0.0, 1.0)
          s = h.sum()
          if abs(s - target) < 1e-14:
              break
      diff = target - h.sum()
      if abs(diff) > 1e-14:
          idx = np.argsort(1.0-h) if diff > 0 else np.argsort(h)
          for i in idx:
              if diff > 0:
                  a = min(diff, 1.0-h[i])
                  h[i] += a; diff -= a
              else:
                  a = min(-diff, h[i])
                  h[i] -= a; diff += a
              if abs(diff) < 1e-15:
                  break
      return h

  def _solve_1d(c0, a, n):
      """Minimize max_{L≠0}(c0[k]+a[k]*t) over t∈[0,1] via subgradient bisection."""
      k0 = n - 1
      m = 2*n - 1
      # Mask out L=0
      c0m = np.delete(c0, k0)
      am = np.delete(a, k0)
      lo, hi = 0.0, 1.0

          v = am*mid + c0m
          vmax = v.max()
          act = np.where(v >= vmax - 1e-10)[0]
          sl = am[act]
          if sl.min() <= 0.0 <= sl.max():
              break
          if sl.max() < 0.0:
              lo = mid
          else:
              hi = mid
      return 0.5*(lo+hi)

  def _cd_pass(h, n, order):
      """One pass of coordinate descent with given coordinate order."""
      m = 2*n - 1
      L_all = np.arange(m) - (n-1)
      max_change = 0.0
      for i in order:
          old = h[i]
          h[i] = 0.0
          c0 = np.correlate(h, 1.0-h, mode='full')
          h[i] = old

          a = np.zeros(m)

          t_new = _solve_1d(c0, a, n)
          h[i] = t_new
