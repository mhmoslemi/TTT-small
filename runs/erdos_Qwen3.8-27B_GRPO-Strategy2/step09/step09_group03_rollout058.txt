    n = len(h0)
    target = n * 0.5
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best_c = _eval(x)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_smooth(h, alpha, topk):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
    m = c.max()
    tk = min(topk, 2 * n - 1)
    idx = np.argpartition(-c, tk - 1)[:tk]
    s = alpha * (c[idx] - m)
    s = np.clip(s, -500, 500)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii in range(tk):
        if w[ii] < 1e-16:
            continue
        wk = w[ii] * 2.0 / n
        L = idx[ii] - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - h[:n - L])
            g[:n - L] -= wk * h[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n - M]
    return g

def _obj_smooth(h, alpha, topk):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
    m = c.max()
    tk = min(topk, 2 * n - 1)
    idx = np.argpartition(-c, tk - 1)[:tk]
    s = alpha * (c[idx] - m)
    s = np.clip(s, -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _fista(h0, alpha, topk, t_end, lr_init):
    n = len(h0)
    target = n * 0.5
    x = h0.copy(); y = h0.copy(); x_prev = h0.copy()
    t_mom = 1.0
    best_c = _eval(x); best_x = x.copy()
    lr = lr_init
    for it in range(5000):
        if time.time() > t_end:
            break
        g = _grad_smooth(y, alpha, topk)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = x + ((t_mom - 1.0) / t_new) * (x - x_prev)
        y = _proj(y, target)
        x_prev = x.copy()
        x_new = _proj(x - lr * g, target)
        c_new = _eval(x_new)
        if c_new < best_c:
            best_c = c_new
            best_x = x_new.copy()
        if c_new > _eval(x) * 1.001:
            lr *= 0.5
            x_new = x
        else:
            lr = min(lr * 1.05, lr_init * 5.0)
        x = x_new
        t_mom = t_new
    return best_x, best_c

def _slsqp(h, alpha, topk, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj_smooth, h, jac=_grad_smooth, args=(alpha, topk),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 30})
        return _repair(res.x, n)
    except Exception:
        return h

def _local_search(h, t_end, rng):
    n = len(h)
    best_c = _eval(h); best_h = h.copy()
    h_cur = h.copy()
    step = 5e-5
    while time.time() < t_end:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 30)) % n
        if i == j:
            j = (j + 1) % n
        d = step * rng.randn()
        ni, nj = h_cur[i] + d, h_cur[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h_cur[i], h_cur[j] = ni, nj
            c = _eval(h_cur)
            if c < best_c:
                best_c = c; best_h = h_cur.copy()
                step = min(step * 1.08, 5e-3)
            else:
                h_cur[i], h_cur[j] = ni - d, nj + d
                step = max(step * 0.92, 1e-8)
        if rng.rand() < 0.005:
            h_cur = best_h + 0.3 * (h_cur - best_h)
    return best_h, best_c

    h_cur = h.copy()

        j = rng.randint(0, max(1, n - b))

            continue
        d = 1e-5 * rng.randn()
        for k in range(b):
            ni = h_cur[i + k] + d
            nj = h_cur[j + k] - d
            if not (0 <= ni <= 1 and 0 <= nj <= 1):

            h_cur[i + k] = ni; h_cur[j + k] = nj
        else:
            c = _eval(h_cur)
            if c < best_c:
                best_c = c; best_h = h_cur.copy()
            else:
                for k in range(b):
                    h_cur[i + k] -= d; h_cur[j + k] += d

    n = len(h); m = n // f

    for i in range(n):
        out[i*f:(i+1)*f] = h[i]
