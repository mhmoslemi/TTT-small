                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _sum_fn(x, half_n):
    return x.sum() - half_n

def _sum_jac(x, half_n):
    return np.ones(len(x))

def _proj_grad(h, alpha, m_lags, steps, step0):
    n = len(h)
    v = np.zeros(n); sq = np.zeros(n)
    step = step0; cur_c = _eval(h)
    for _it in range(steps):
        g = _smooth_grad(h, alpha, m_lags)
        v = 0.85 * v + 0.15 * g
        sq = 0.95 * sq + 0.05 * g * g
        d = v / (np.sqrt(sq) + 1e-8)
        taken = False
        for _t in range(12):
            h_new = _proj(h - step * d, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:
                h = h_new; cur_c = c_new
                step = min(step * 1.15, step0 * 50.0)
                taken = True; break
            step *= 0.4
        if not taken:
            step = min(step * 1.3, step0 * 50.0)
    return _repair(h, n)

def _grad_2pt(h, n_iter, alpha, m_lags):
    n = len(h)
    c5c = _eval(h); best_h = h.copy(); best_c = c5c
    for _ in range(n_iter):
        g = _smooth_grad(h, alpha, m_lags)
        gp = g - g.mean()
        i = int(np.argmax(gp)); j = int(np.argmin(gp))
        if i == j:
            continue
        orig_i, orig_j = h[i], h[j]
        best_d = 0.0; best_c5 = c5c
        for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
            d = frac * (gp[i] - gp[j]) * 0.01
            ni, nj = orig_i + d, orig_j - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < best_c5:
                best_c5 = c5n; best_d = d
            else:
                h[i], h[j] = orig_i, orig_j
                break
        h[i], h[j] = orig_i, orig_j
        if best_d != 0.0:
            h[i] += best_d; h[j] -= best_d
            c5c = best_c5
            if c5c < best_c:
                best_c = c5c; best_h = h.copy()
    return best_h, best_c

def _structured_polish(h, n_iter, alpha, m_lags):
    n = len(h)
    c5c = _eval(h); best_h = h.copy(); best_c = c5c
    for _ in range(n_iter):
        g = _smooth_grad(h, alpha, m_lags)
        gp = g - g.mean()
        order = np.argsort(-np.abs(gp))
        for k in range(min(30, len(order))):
            i = int(order[k])
            j = int(np.argmin(gp)) if gp[i] > 0 else int(np.argmax(gp))
            if i == j:
                continue
            for d in [5e-6, 2e-6, 8e-7, 3e-7, 1e-7]:
                si = 1 if gp[i] > 0 else -1
                sj = -si
                ni = h[i] + si * d
                nj = h[j] + sj * d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    if c5c < best_c:
                        best_c = c5c; best_h = h.copy()
                    break
                else:
                    h[i], h[j] = h[i] - si * d, h[j] - sj * d
    return best_h, best_c

def _local_search(h, n_iter, rng, base_step, n_pert):
    n = len(h)
    c5cur = _eval(h); best_h = h.copy(); best_c = c5cur
    for _ in range(n_iter):
        idx = rng.randint(0, n, size=n_pert)
        d = base_step * rng.randn(n_pert)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < 0 or h_new.max() > 1:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5cur:
            c5cur = c5n; h = h_new
            if c5n < best_c:
                best_c = c5n; best_h = h.copy()
    return best_h, best_c

def _downsample(h, f):
    n = len(h); m = n // f
    return _repair(h[:m*f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    n = len(h); out = np.empty(n * f)
    for i in range(n):
        out[i*f:(i+1)*f] = h[i]
    return _repair(out, n * f)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy(); best_c5 = _eval(h)

    # Phase 1: Multi-res exploration + perturbations
    candidates = [h.copy()]
    for _p in range(3):
        hp = h + 0.02 * rng.randn(n)
        hp = _proj(hp, n * 0.5)
        hp = _repair(hp, n)
        candidates.append(hp)
    for ratio in [4, 8]:
        if time.time() - t0 > 40:
            break
        nc = max(40, n // ratio); f = n // nc
        for c0 in candidates[:2]:
            hc = _downsample(c0, f)
            for alpha in [1000, 5000, 50000]:
                if time.time() - t0 > 60:
                    break
                hc = _slsqp(hc, alpha, 30, 300, 1e-14)
            hu = _upsample(hc, f)
            cu = _eval(hu)
            if cu < best_c5:
                best_c5 = cu; best_h = hu.copy()
            candidates.append(hu)

    # Phase 2: SLSQP alpha sweep at full resolution
    alpha_seq = [2000, 8000, 30000, 100000, 400000, 1500000, 5000000, 20000000, 80000000, 300000000]
    for alpha in alpha_seq:
        if time.time() - t0 > 400:
            break
        lift = min(100, max(20, int(30 * np.log10(alpha / 2000) + 25)))
        h_new = _slsqp(best_h, alpha, lift, 1000, 1e-16)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n; best_h = h_new

    # Phase 3: Projected gradient descent
    if time.time() - t0 < budget_s - 280:
        configs = [
            (3000, 40, 500, 0.005),
            (20000, 50, 450, 0.003),
            (100000, 70, 350, 0.002),
            (500000, 100, 300, 0.001),
            (2000000, 130, 250, 0.0005),
        ]
        for alpha, ml, steps, s0 in configs:
            if time.time() - t0 > budget_s - 200:
                break
            h_new = _proj_grad(best_h.copy(), alpha, ml, steps, s0)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n; best_h = h_new

    # Phase 4: High-alpha SLSQP
    if time.time() - t0 < budget_s - 150:
        for alpha in [1000000000, 5000000000, 20000000000]:
            if time.time() - t0 > budget_s - 100:
                break
            h_new = _slsqp(best_h, alpha, 120, 500, 1e-17)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n; best_h = h_new

    # Phase 5: Gradient-guided 2-point
    if time.time() - t0 < budget_s - 60:
        nitr = min(int((budget_s - 60 - (time.time() - t0)) / 0.008), 80000)
        best_h, best_c5 = _grad_2pt(best_h, nitr, 2000000, 80)

    # Phase 6: Local random search
    if time.time() - t0 < budget_s - 30:
        ntries = min(int((budget_s - 30 - (time.time() - t0)) / 0.002), 150000)
        best_h, best_c5 = _local_search(best_h, ntries, rng, 1.5e-6, 3)

    # Phase 7: Structured gradient polish
    if time.time() - t0 < budget_s - 5:
        nitr = min(int((budget_s - 5 - (time.time() - t0)) / 0.004), 120000)
        best_h, best_c5 = _structured_polish(best_h, nitr, 5000000, 60)

    # Phase 8: Final ultra-fine random pairs
    if time.time() - t0 < budget_s - 1:
        h2 = best_h.copy(); nc2 = len(h2); c5c = _eval(h2)
        npol = min(int((budget_s - 1 - (time.time() - t0)) / 0.001), 400000)
        for _ in range(npol):
            if time.time() - t0 > budget_s - 0.5:
                break
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 6)) % nc2
            d = 2e-7 * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n; best_h = h2.copy()
                else:
                    h2[i], h2[j] = h2[i] - d, h2[j] + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
