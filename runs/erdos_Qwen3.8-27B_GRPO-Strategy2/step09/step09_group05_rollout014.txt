import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sobj(x, alpha, ml):
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sgrad(x, alpha, ml):
    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * (2.0 / n)
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n-L])
            g[:n-L] -= wk * x[L:]
        else:
            M = -L
            g[:n-M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n-M]
    return g

def _sf(x, half):
    return x.sum() - half

def _sj(x, half):
    return np.ones(len(x))

def _slsqp(h, alpha, ml, maxit, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (n*0.5,)}]
    try:
        res = minimize(_sobj, h, jac=_sgrad, args=(alpha, ml),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-16})
        if time.time() < t_end:
            return _repair(res.x, n)
    except Exception:
        pass
    return h

def _binarize(h, n):
    t = n // 2
    b = (h > 0.5).astype(float)
    cnt = int(b.sum())
    if cnt > t:
        idx = np.where(b == 1)[0]
        b[idx[np.argsort(h[idx])][:cnt - t]] = 0
    elif cnt < t:
        idx = np.where(b == 0)[0]
        b[idx[np.argsort(-h[idx])][:t - cnt]] = 1
    return b

def _bin_swap(b, t_end, max_passes=500):
    n = len(b)
    b = b.astype(float).copy()
    c5 = _eval(b)
    for _ in range(max_passes):
        if time.time() > t_end:
            break
        c = _corr(b)
        advanced = False
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        top_idx = np.argsort(c)[-60:]
        for i in ones:
            if time.time() > t_end or b[i] != 1:
                continue
            dists = np.abs(zeros - i)
            for jj in np.argsort(dists)[:40]:
                j = int(zeros[jj])
                if b[j] != 0:
                    continue
                mx = 0.0
                for ki in top_idx:
                    L = ki - (n - 1)
                    a1 = b[i + L] if 0 <= i + L < n else 0.0
                    a2 = b[i - L] if 0 <= i - L < n else 0.0
                    b1 = b[j + L] if 0 <= j + L < n else 0.0
                    b2 = b[j - L] if 0 <= j - L < n else 0.0
                    nc = c[ki] + (2.0 / n) * (a1 + a2 - b1 - b2)
                    if nc > mx:
                        mx = nc
                if mx < c5 - 1e-14:
                    b[i] = 0.0; b[j] = 1.0
                    c5n = _eval(b)
                    if c5n < c5:
                        c5 = c5n; zeros = np.where(b == 0)[0]; advanced = True
                        break
                    else:
                        b[i] = 1.0; b[j] = 0.0
        if not advanced:
            break
    return b, c5

def _grad_2pt(h, rng, t_end):
    n = len(h)
    c5 = _eval(h)
    best_h = h.copy(); best_c5 = c5
    stalled = 0
    for _ in range(20000000):
        if time.time() > t_end:
            break
        if stalled > 200000:
            ki = rng.randint(0, n, size=6)
            for kk in ki:
                d = 0.02 * rng.randn()
                if 0 <= h[kk] + d <= 1:
                    h[kk] += d
            h = _repair(h, n)
            c5 = _eval(h)
            stalled = 0
        g = _sgrad(h, 500000, 80)
        gp = g - g.mean()
        i = int(np.argmax(gp)); j = int(np.argmin(gp))
        if rng.rand() < 0.12:
            i = int(rng.randint(n)); j = int(rng.randint(n))
        if i == j:
            continue
        moved = False
        for sc in [0.5, 0.8, 1.0, 0.2, 0.1]:
            d = sc * min(h[i], 1.0 - h[j])
            if d < 1e-10:
                continue
            h[i] -= d; h[j] += d
            c_new = _eval(h)
            if c_new < c5:
                c5 = c_new; moved = True; stalled = 0
                if c5 < best_c5:
                    best_c5 = c5; best_h = h.copy()
                break
            h[i] += d; h[j] -= d
        if not moved:
            i, j = j, i
            for sc in [0.5, 0.8, 1.0]:
                d = sc * min(h[i], 1.0 - h[j])
                if d < 1e-10:
                    continue
                h[i] -= d; h[j] += d
                c_new = _eval(h)
                if c_new < c5:
                    c5 = c_new; moved = True; stalled = 0
                    if c5 < best_c5:
                        best_c5 = c5; best_h = h.copy()
                    break
                h[i] += d; h[j] -= d
        if not moved:
            stalled += 1
    return best_h, best_c5

def _ultra_fine(h, rng, t_end):
    n = len(h)
    c5 = _eval(h)
    for _ in range(10**9):
        if time.time() > t_end:
            break
        i = int(rng.randint(n))
        j = int((i + 1 + rng.randint(0, 20)) % n)
        d = 2e-7 * rng.randn()
        if 0 <= h[i] + d <= 1 and 0 <= h[j] - d <= 1:
            h[i] += d; h[j] -= d
            c_new = _eval(h)
            if c_new < c5:
                c5 = c_new
            else:
                h[i] -= d; h[j] += d
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    t_p1 = min(t0 + 200, t0 + budget_s - 550)
    for alpha in [5000, 50000, 500000, 5000000, 50000000]:
        if time.time() > t_p1:
            break
        ml = min(100, max(15, int(15 * np.log10(alpha / 1000) + 15)))
        h_new = _slsqp(best_h, alpha, ml, 600, t_p1)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n; best_h = h_new

    t_p2 = min(t0 + 350, t0 + budget_s - 400)
    if time.time() < t_p2:
        b = _binarize(best_h, n)
        c5b = _eval(b)
        if c5b < best_c5:
            best_c5 = c5b; best_h = b.copy()
        best_h, best_c5 = _bin_swap(best_h.copy(), t_p2)

    t_p3 = min(t0 + 450, t0 + budget_s - 300)
    if time.time() < t_p3:
        for alpha in [200000, 2000000, 20000000, 200000000]:
            if time.time() > t_p3:
                break
            h_new = _slsqp(best_h, alpha, 60, 400, t_p3)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n; best_h = h_new

    t_p4 = t0 + budget_s - 50
    best_h, best_c5 = _grad_2pt(best_h.copy(), rng, t_p4)

    if time.time() < t0 + budget_s - 3:
        best_h, best_c5 = _ultra_fine(best_h.copy(), rng, t0 + budget_s - 1)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
