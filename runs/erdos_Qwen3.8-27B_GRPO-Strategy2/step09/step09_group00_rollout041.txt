import numpy as np

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * (2.0/n))

def _repair(h, n):
    S = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(50):
        h *= S/s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s-S) < 1e-13:
            break
    d = S - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1-h) if d>0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1-h[i]); h[i]+=a; d-=a
            else:
                a = min(-d, h[i]); h[i]-=a; d+=a
            if abs(d) < 1e-13: break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum())/x.size
    for _ in range(20):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-13: break
        m = (x<1.0) if d>0 else (x>0.0)
        c = m.sum()
        if c == 0: break
        x[m] += d/c
    return x

def _pn(h):
    n = len(h)
    N = 1
    m = 2*n-1
    while N < m: N <<= 1
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H*np.conj(H), N).real[:n]
    pref = np.empty(n+1); pref[0]=0; np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n); Nv = np.zeros(n)
    P[0] = total - ac[0]
    if n > 1:
        idx = np.arange(1,n)
        P[idx] = pref[n-idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
        mx = max(float(np.max(P)), float(np.max(Nv)))
    else:
        mx = float(P[0])
    return P, Nv, mx

def _soft_val(h, a):
    P, Nv, mx = _pn(h)
    n = len(h); dx = 2.0/n
    e = np.exp(np.clip(a*(P[0]-mx), -400, 0))
    if n>1:
        e += np.exp(np.clip(a*(P[1:]-mx), -400, 0)).sum()
        e += np.exp(np.clip(a*(Nv[1:]-mx), -400, 0)).sum()
    return dx*(mx + np.log(e)/a)

def _grad_weights(h, wP, wN):
    n = len(h); dx = 2.0/n
    u = np.zeros(n)
    if n>1: u[1:] = wP[1:]+wN[1:]
    cp = np.zeros(n); cn = np.zeros(n)
    if n>1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n-1::-1]; linN = cn
    q = np.zeros(n)
    if n>1 and np.any(u[1:]):
        Nf = 1; m=2*n-1
        while Nf<m: Nf<<=1
        H = np.fft.rfft(h, Nf)
        R = np.fft.irfft(H*np.fft.rfft(u[::-1], Nf), Nf)
        L = np.fft.irfft(H*np.fft.rfft(u, Nf), Nf)
        q = -(R[n-1:2*n-1].real + L[:n].real)
    return dx*(wP[0]*(1-2*h) + linP + linN + q)

def _soft_grad(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    wP = np.zeros(n); wN = np.zeros(n)
    e0 = np.exp(np.clip(a*(P[0]-mx), -400, 0))
    tot = e0
    if n>1:
        ep = np.exp(np.clip(a*(P[1:]-mx), -400, 0))
        en = np.exp(np.clip(a*(Nv[1:]-mx), -400, 0))
        tot += ep.sum() + en.sum()
        if tot <= 0: tot=1.0; e0=1.0
        wP[0]=e0/tot; wP[1:]=ep/tot; wN[1:]=en/tot
    else:
        wP[0]=1.0
    return _grad_weights(h, wP, wN)

def _fista(h0, alpha, max_t, L0=10.0):

    n = len(h0); S = n*0.5
    t0 = time.time()
    h = _repair(h0, n)
    hp = h.copy(); y = h.copy(); t = 1.0
    fy = _soft_val(h, alpha); fh = fy
    best_h = h.copy(); best_c = _c5(h)
    stale = 0; Lc = L0
    for it in range(500000):
        if time.time()-t0 > max_t: break
        if it:
            tn = 0.5*(1.0+np.sqrt(1.0+4.0*t*t))
            y = h + ((t-1.0)/tn)*(h-hp)
            t = tn
            if y.min()<-1e-5 or y.max()>1+1e-5 or abs(y.sum()-S)>1e-8:
                y = _proj(y, S)
            fy = _soft_val(y, alpha)
        else:
            y = h; fy = fh
        g = _soft_grad(y, alpha)
        step = 1.0/max(Lc, 1e-12)
        accepted = False
        for _ in range(14):
            xn = _proj(y - step*g, S)
            fx = _soft_val(xn, alpha)
            dv = xn-y
            rhs = fy + float(np.dot(g,dv)) + 0.5*(1.0/step)*float(np.dot(dv,dv))
            if fx <= rhs + 1e-11:
                accepted = True; break
            step *= 0.5
        Lc = 1.0/max(step, 1e-12)
        if accepted:
            hp = h; h = xn; fh = fx; stale = 0
            Lc = max(Lc*0.85, 1.0)
            if it%5 == 0:
                c = _c5(h)
                if c < best_c: best_c = c; best_h = h.copy()
        else:
            stale += 1; hp = h; y = h; t = 1.0; fy = fh
            Lc = max(Lc*0.5, 1.0)
            if stale > 12: break
    c = _c5(h)
    if c < best_c: best_c = c; best_h = h.copy()
    return h, best_h, best_c

def _active_grad(h, P, Nv):
    n = len(h); dx = 2.0/n
    mx = float(P[0])
    if n>1: mx = max(mx, float(np.max(P[1:])), float(np.max(Nv[1:])))
    tol = max(1e-9, 15.0*dx)
    wP = np.zeros(n); wN = np.zeros(n)
    cnt = 0
    if P[0] >= mx-tol: wP[0]=1.0; cnt+=1
    if n>1:
        mP = P[1:]>=mx-tol
        if np.any(mP):
            pos = np.nonzero(mP)[0]+1
            wP[pos]=1.0; cnt+=int(pos.size)

            wN[pos]=1.0; cnt+=int(pos.size)
    if cnt==0: return np.zeros(n)
    inv = 1.0/cnt
    wP*=inv; wN*=inv
    return _grad_weights(h, wP, wN)

import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * (2.0/n))

def _repair(h, n):
    S = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15: return np.full(n, 0.5)
    for _ in range(50):
        h *= S/s; np.clip(h, 0.0, 1.0, out=h); s = h.sum()
        if abs(s-S) < 1e-13: break
    d = S - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1-h) if d>0 else np.argsort(h)
        for i in order:
            if d > 0: a=min(d,1-h[i]); h[i]+=a; d-=a
            else: a=min(-d,h[i]); h[i]-=a; d+=a
            if abs(d)<1e-13: break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S-x.sum())/x.size
    for _ in range(20):
        np.clip(x, 0.0, 1.0, out=x); d = S-x.sum()
        if abs(d)<1e-13: break
        m = (x<1.0) if d>0 else (x>0.0); c=m.sum()
        if c==0: break
        x[m]+=d/c
    return x

def _pn(h):
    n=len(h); N=1; m=2*n-1
    while N<m: N<<=1
    H=np.fft.rfft(h,N); ac=np.fft.irfft(H*np.conj(H),N).real[:n]
    pref=np.empty(n+1); pref[0]=0; np.cumsum(h,out=pref[1:])
    total=pref[n]; P=np.empty(n); Nv=np.zeros(n)
    P[0]=total-ac[0]
    if n>1:
        idx=np.arange(1,n); P[idx]=pref[n-idx]-ac[idx]
        Nv[idx]=total-pref[idx]-ac[idx]
        mx=max(float(np.max(P)),float(np.max(Nv)))
    else: mx=float(P[0])
    return P,Nv,mx

def _soft_val(h,a):
    P,Nv,mx=_pn(h); n=len(h); dx=2.0/n
    e=np.exp(np.clip(a*(P[0]-mx),-400,0))
    if n>1:
        e+=np.exp(np.clip(a*(P[1:]-mx),-400,0)).sum()
        e+=np.exp(np.clip(a*(Nv[1:]-mx),-400,0)).sum()
    return dx*(mx+np.log(e)/a)

def _grad_weights(h,wP,wN):
    n=len(h); dx=2.0/n; u=np.zeros(n)
    if n>1: u[1:]=wP[1:]+wN[1:]
    cp=np.zeros(n); cn=np.zeros(n)
    if n>1: cp[1:]=np.cumsum(wP[1:]); cn[1:]=np.cumsum(wN[1:])
    linP=cp[n-1::-1]; linN=cn; q=np.zeros(n)
    if n>1 and np.any(u[1:]):
        Nf=1; m=2*n-1
        while Nf<m: Nf<<=1
        H=np.fft.rfft(h,Nf)
        R=np.fft.irfft(H*np.fft.rfft(u[::-1],Nf),Nf)
        L=np.fft.irfft(H*np.fft.rfft(u,Nf),Nf)
        q=-(R[n-1:2*n-1].real+L[:n].real)
    return dx*(wP[0]*(1-2*h)+linP+linN+q)

def _soft_grad(h,a):
    P,Nv,mx=_pn(h); n=len(h)
    wP=np.zeros(n); wN=np.zeros(n)
    e0=np.exp(np.clip(a*(P[0]-mx),-400,0)); tot=e0
    if n>1:
        ep=np.exp(np.clip(a*(P[1:]-mx),-400,0))
        en=np.exp(np.clip(a*(Nv[1:]-mx),-400,0))
        tot+=ep.sum()+en.sum()
        if tot<=0: tot=1.0; e0=1.0
        wP[0]=e0/tot; wP[1:]=ep/tot; wN[1:]=en/tot
    else: wP[0]=1.0
    return _grad_weights(h,wP,wN)

def _fista(h0,alpha,max_t,L0=10.0):
    if max_t<=0: return h0.copy(),h0.copy(),_c5(h0)
    n=len(h0); S=n*0.5; t0=time.time()
    h=_repair(h0,n); hp=h.copy(); y=h.copy(); t=1.0
    fy=_soft_val(h,alpha); fh=fy
    best_h=h.copy(); best_c=_c5(h); stale=0; Lc=L0
    for it in range(500000):
        if time.time()-t0>max_t: break
        if it:
            tn=0.5*(1.0+np.sqrt(1.0+4.0*t*t))
            y=h+((t-1.0)/tn)*(h-hp); t=tn
            if y.min()<-1e-5 or y.max()>1+1e-5 or abs(y.sum()-S)>1e-8: y=_proj(y,S)
            fy=_soft_val(y,alpha)
        else: y=h; fy=fh
        g=_soft_grad(y,alpha); step=1.0/max(Lc,1e-12); accepted=False
        for _ in range(14):
            xn=_proj(y-step*g,S); fx=_soft_val(xn,alpha)
            dv=xn-y; rhs=fy+float(np.dot(g,dv))+0.5*(1.0/step)*float(np.dot(dv,dv))
            if fx<=rhs+1e-11: accepted=True; break
            step*=0.5
        Lc=1.0/max(step,1e-12)
        if accepted:
            hp=h; h=xn; fh=fx; stale=0; Lc=max(Lc*0.85,1.0)
            if it%5==0:
                c=_c5(h)
                if c<best_c: best_c=c; best_h=h.copy()
        else:
            stale+=1; hp=h; y=h; t=1.0; fy=fh; Lc=max(Lc*0.5,1.0)
            if stale>12: break
    c=_c5(h)
    if c<best_c: best_c=c; best_h=h.copy()
    return h,best_h,best_c

def _active_grad(h,P,Nv):
    n=len(h); dx=2.0/n; mx=float(P[0])
    if n>1: mx=max(mx,float(np.max(P[1:])),float(np.max(Nv[1:])))
    tol=max(1e-9,15.0*dx); wP=np.zeros(n); wN=np.zeros(n); cnt=0
    if P[0]>=mx-tol: wP[0]=1.0; cnt+=1
    if n>1:
        mP=P[1:]>=mx-tol
        if np.any(mP):
            pos=np.nonzero(mP)[0]+1; wP[pos]=1.0; cnt+=int(pos.size)
        mN=Nv[1:]>=mx-tol

    if cnt==0: return np.zeros(n)
    inv=1.0/cnt; wP*=inv; wN*=inv

def _resample(h,m):
    n=len(h)
    if n==m: return _repair(h,n)
    if n%m==0: out=h.reshape(m,n//m).mean(axis=1)
    else:

        out=np.interp(x1,x0,h)
    return _proj(out,m*0.5)

def _grad_local(h0,rng,deadline):
