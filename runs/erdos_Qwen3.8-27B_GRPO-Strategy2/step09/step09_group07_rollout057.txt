import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _cd_pass(h, rng):

    nm1 = n - 1
    cmax = 2 * n - 1
    c = _corr(h)
    jlist = np.arange(n)
    rng.shuffle(jlist)
    for j in jlist:
        j = int(j)
        p_lo = max(0, 2 * j - nm1)
        p_hi = min(n - 1, 2 * j)
        if p_lo > p_hi:
            continue
        p = np.arange(p_lo, p_hi + 1)
        q = 2 * j - p
        k_idx = p + nm1 - j
        beta = np.zeros(cmax)
        beta[k_idx] = 1.0 - h[p] - h[q]
        c_prime = c - beta * h[j]
        lo_t, hi_t = 0.0, 1.0
        bv = -beta * h[j]
        for _ in range(32):
            t1 = lo_t + (hi_t - lo_t) * 0.33333
            t2 = hi_t - (hi_t - lo_t) * 0.33333
            if np.max(c_prime + beta * t1) <= np.max(c_prime + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        delta = t_new - h[j]
        if abs(delta) > 1e-13:
            c += beta * delta
            h[j] = t_new
    return h

    m = c.max()

    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sm_grad(x, alpha, ml):
    n = len(x)
    nm1 = n - 1
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        k = int(idx[ii])
        L = k - nm1
        wk = w[ii] * dx
        if L >= 0:
            g[:n - L] += wk * (1.0 - x[L:])
            g[L:] -= wk * x[:n - L]
        else:
            M = -L
            g[M:] += wk * (1.0 - x[:n - M])
            g[:n - M] -= wk * x[M:]
    return g

def _sum_fn(x, tgt):
    return x.sum() - tgt

def _sum_jc(x, tgt):
    return np.ones(len(x))

def _slsqp(h, alpha, ml, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jc, 'args': (n * 0.5,)}]
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, args=(alpha, ml),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

    c5c = best_c5

        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:

            c5n = _eval(h)

                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5
