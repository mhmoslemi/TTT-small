        active = np.where(c > cmax - tol)[0]
        n_act = min(len(active), 40)
        if n_act == 0:
            break
        idx = active[:n_act]
        g = np.zeros(n)
        for k in idx:
            g += _grad_shift(best_h, k, n, dx)
        g /= n_act
        i_pos = np.argmin(g)
        i_neg = np.argmax(g)
        if i_pos == i_neg:
            i_pos2 = np.argsort(g)[1]
            i_neg2 = np.argsort(-g)[1]
            i_pos, i_neg = i_pos2, i_neg2
        improved = False
        for d_scale in [1e-5, 5e-6, 1e-6, 5e-5, 1e-4]:
            d = d_scale
            hi = best_h[i_pos] + d
            hj = best_h[i_neg] - d
            if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                continue
            trial = best_h.copy()
            trial[i_pos] = hi
            trial[i_neg] = hj
            c_try = _eval(trial)
            if c_try < best_c - 1e-15:
                best_h = trial
                best_c = c_try
                improved = True
                break
        if not improved:
            for _t in range(8):
                ip = rng.randint(n)
                jn = (ip + 1 + rng.randint(0, n - 1)) % n
                d = 1e-5 * rng.randn()
                hi = best_h[ip] + d
                hj = best_h[jn] - d
                if 0 <= hi <= 1 and 0 <= hj <= 1:
                    trial = best_h.copy()
                    trial[ip] = hi
                    trial[jn] = hj
                    c_try = _eval(trial)
                    if c_try < best_c - 1e-15:
                        best_h = trial
                        best_c = c_try
                        improved = True
                        break
        total_iters += 1
        if total_iters % 2000 == 0 and not improved:
            break
    return best_h, best_c

    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    dx = 2.0 / n
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Quick SLSQP refinement
    for alpha, ml, iters in [(50000, 30, 400), (500000, 50, 400), (5000000, 80, 300)]:
        if _el() > 80:
            break
        h_new = _slsqp_step(best_h, alpha, ml, iters, 1e-14)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 2: Gradient descent with active set
    if _el() < budget_s - 300:
        for alpha in [500000, 5000000, 50000000]:
            if _el() > budget_s - 250:
                break
            ml = min(100, int(20 * np.log10(max(alpha, 1) / 1e5) + 30))
            h_new, c5n = _grad_descent(best_h.copy(), n, dx, alpha, ml,
                                       400, 0.002, 0.8)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 3: Second SLSQP pass at high alpha
    if _el() < budget_s - 150:
        for alpha in [100000000, 1000000000]:
            if _el() > budget_s - 60:
                break
            h_new = _slsqp_step(best_h, alpha, 150, 800, 1e-16)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 4: Reheat perturbations
    if _el() < budget_s - 80:
        for attempt in range(4):
            if _el() > budget_s - 40:
                break
            ns = 5e-5 * (4 - attempt)
            h_pert = _repair(best_h + ns * rng.randn(n), n)
            h_new = _slsqp_step(h_pert, 100000000, 100, 400, 1e-15)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 5: Fine local search (most of remaining time)
    budget_end_ls = t0 + budget_s - 5
    best_h, best_c5 = _local_search(best_h, n, dx, budget_end_ls, rng)

    # Phase 6: Ultra-fine polish
    if _el() < budget_s - 0.5:
        h2 = best_h.copy()
        c2 = best_c5
        for d in [2e-6, 1e-6, 5e-7, 2e-7]:
            for _ in range(min(int((budget_s - 0.5 - _el()) / 0.002), 100000)):
                if _el() > budget_s - 0.5:
                    break
                c = np.correlate(h2, 1.0 - h2, mode='full') * dx
                cmax = c.max()
                active = np.where(c > cmax - 0.01 * cmax)[0]
                g = np.zeros(n)
                for k in active[:30]:
                    g += _grad_shift(h2, k, n, dx)
                g /= max(len(active), 1)
                ip = np.argmin(g)
                jn = np.argmax(g)
                if ip == jn:
                    continue
                hp = h2[ip] + d
                hj = h2[jn] - d
                if 0 <= hp <= 1 and 0 <= hj <= 1:
                    h2[ip], h2[jn] = hp, hj
                    c_try = _eval(h2)
                    if c_try < c2 - 1e-16:
                        c2 = c_try
                        if c2 < best_c5:
                            best_c5 = c2
                            best_h = h2.copy()
                    else:
                        h2[ip], h2[jn] = hp - d, hj + d

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
