import numpy as np

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _corr_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(50):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-12:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _fft_size(n):
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _pn(h):
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.zeros(n + 1)
    np.cumsum(h, out=pref[1:])
    tot = pref[n]
    P = np.empty(n)
    P[0] = tot - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = tot - pref[idx] - ac[idx]
        mx = float(max(np.max(P), np.max(Nv)))
    else:
        mx = float(P[0])
    return P, Nv, mx

def _sval(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    dx = 2.0 / n
    e = np.exp(np.clip(a * (P[0] - mx), -400.0, 0.0))

        cn[1:] = np.cumsum(wN[1:])

        q = -(R + L)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

    n = len(h)
    wP = np.zeros(n)
    wN = np.zeros(n)

def _binary_local(h0, rng, dl):
    """Local search on binary (0/1) representations using swaps."""
    t0 = time.time()
    n = len(h0)
    # Round to binary
    S = int(n * 0.5)
    h = (h0 > 0.5).astype(float)
    if h.sum() != S:
        # Adjust
        if h.sum() > S:
            # Need to turn some 1s to 0s
            order = np.argsort(h0)[::-1]  # most confident 1s first
            idx = order[S:]  # these should be 0
            h[idx] = 0.0
        else:
            order = np.argsort(h0)  # least confident 0s first (should be 1)
            idx = order[:S - int(h.sum())]
            h[idx] = 1.0
    c = _c5(h)
    bc = c
    bh = h.copy()
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    it = 0
    while time.time() < dl:
        # Rebuild lists periodically
        if it % 5000 == 0:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        
        fr = (time.time() - t0) / max(1e-9, dl - t0)
        
        if rng.random() < 0.7:
            # Single swap
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            h[i] = 0.0
            h[j] = 1.0
            cn = _c5(h)
            if cn < c - 1e-14:
                c = cn
                if c < bc:
                    bc = c
                    bh = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
        else:
            # Block swap: swap a contiguous block of 1s with a contiguous block of 0s
            blen = 1 + rng.randint(0, 4)  # 1 to 4
            i = int(rng.randint(n - blen))
            block = h[i:i+blen]
            if np.all(block == 1.0) or np.all(block == 0.0):
                target = 0.0 if block[0] == 1.0 else 1.0
                # Find a block of the opposite value
                j = int(rng.randint(n - blen))
                oblock = h[j:j+blen]
                if h[i+blen-1] < j or h[j+blen-1] < i:  # no overlap
                    if np.all(oblock == target):
                        h[i:i+blen] = target
                        h[j:j+blen] = block[0]
                        cn = _c5(h)
                        if cn < c - 1e-14:
                            c = cn
                            if c < bc:
                                bc = c
                                bh = h.copy()
                        else:
                            h[i:i+blen] = 1.0 - target
                            h[j:j+blen] = 1.0 - block[0]
        
        it += 1
    
    c = _c5(h)
    if c < bc:
        bc = c
        bh = h.copy()
    return bh, bc

def _cont_local(h0, rng, dl, s0=5e-5):

            offset = int(rng.randint(1, n//4)) * (-1 if rng.random() < 0.5 else 1)
            j = (i + offset) % n

                    bc = c
                    bh = h.copy()

    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)[n-1:2*n-1].real
        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)[:n].real
        q = -(R + L)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _sg(h, a):
    P, Nv, mx = _pn(h)
    n = len(h)
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(a * (P[0] - mx), -400.0, 0.0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(a * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(a * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0:
            tot, e0 = 1.0, 1.0
        wP[0] = e0 / tot
        wP[1:] = ep / tot
        wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _gW(h, wP, wN)

def _fista(h0, a, tl, L=8.0):
    t0 = time.time()
    n = len(h0)
    S = n * 0.5
    h = _proj(h0.copy(), S)
    hp = h.copy()
    y = h.copy()
    tv = 1.0
    fy = _sval(h, a)
    bh = h.copy()
    bc = _c5(h)
    Lc = L
    for it in range(200000):
        if time.time() - t0 > tl:
            break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tv * tv))
            y = h + ((tv - 1.0) / tn) * (h - hp)
            tv = tn
            if y.min() < -1e-5 or y.max() > 1 + 1e-5 or abs(y.sum() - S) > 1e-8:
                y = _proj(y, S)
            fy = _sval(y, a)
        g = _sg(y, a)
        step = 1.0 / max(Lc, 1e-12)
        ok = False
        for _ in range(10):
            xn = _proj(y - step * g, S)
            fx = _sval(xn, a)
            d = xn - y
            rhs = fy + float(np.dot(g, d)) + 0.5 * (1.0 / step) * float(np.dot(d, d))
            if fx <= rhs + 1e-10:
                ok = True
                break
            step *= 0.5
        Lc = 1.0 / max(step, 1e-12)
        if ok:
            hp = h
            h = xn
            Lc = max(Lc * 0.9, 1.0)
            if it % 16 == 0:
                c = _c5(h)
                if c < bc:
                    bc = c
                    bh = h.copy()
        else:
            hp, y, tv = h, h, 1.0
            Lc = max(Lc * 0.5, 1.0)
    c = _c5(h)
    if c < bc:
        bc = c
        bh = h.copy()
    return h, bh, bc

def _bin_local(h0, rng, dl):
    t0 = time.time()
    n = len(h0)
    S = int(n * 0.5)
    h = (h0 > 0.5).astype(float)
    if h.sum() > S:
        order = np.argsort(h0)[::-1]
        h[order[S:]] = 0.0
    elif h.sum() < S:
        order = np.argsort(h0)
        h[order[:S - int(h.sum())]] = 1.0
    c = _c5(h)
    bc = c
    bh = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    it = 0
    while time.time() < dl:
        if it % 3000 == 0:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if rng.random() < 0.65:
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            h[i] = 0.0
            h[j] = 1.0
            cn = _c5(h)
            if cn < c - 1e-14:
                c = cn
                if c < bc:
                    bc = c
                    bh = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
        elif rng.random() < 0.5:
            blen = 1 + rng.randint(0, 3)
            i = int(rng.randint(max(1, n - blen)))
            j = int(rng.randint(max(1, n - blen)))
            if i + blen > j and j + blen > i:
                it += 1
                continue
            if np.all(h[i:i+blen] == 1.0) and np.all(h[j:j+blen] == 0.0):
                h[i:i+blen] = 0.0
                h[j:j+blen] = 1.0
                cn = _c5(h)
                if cn < c - 1e-14:
                    c = cn
                    if c < bc:
                        bc = c
                        bh = h.copy()
                else:
                    h[i:i+blen] = 1.0
                    h[j:j+blen] = 0.0
            elif np.all(h[i:i+blen] == 0.0) and np.all(h[j:j+blen] == 1.0):
                h[i:i+blen] = 1.0
                h[j:j+blen] = 0.0
                cn = _c5(h)
                if cn < c - 1e-14:
                    c = cn
                    if c < bc:
                        bc = c
                        bh = h.copy()
                else:
                    h[i:i+blen] = 0.0
                    h[j:j+blen] = 1.0
        else:
            corr = _corr_vec(h)
            bk = int(np.argmax(corr))
            lag = bk - (n - 1)
            i = int(rng.randint(n))
            j = (i + abs(lag)) % n
            if h[i] == 1.0 and h[j] == 0.0:
                h[i] = 0.0
                h[j] = 1.0
                cn = _c5(h)
                if cn < c - 1e-14:
                    c = cn
                    if c < bc:
                        bc = c
                        bh = h.copy()
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        it += 1
    c = _c5(h)
    if c < bc:
        bc = c
        bh = h.copy()
    return bh, bc

def _cont_local(h0, rng, dl, s0=5e-5):
    t0 = time.time()
    n = len(h0)
    h = h0.copy()
    c = _c5(h)
    bc = c
    bh = h.copy()
    it = 0
    while time.time() < dl:
        fr = (time.time() - t0) / max(1e-9, dl - t0)
        base = max(1e-11, s0 * (1e-5) ** fr)
        r = rng.random()
        if r < 0.55:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        elif r < 0.8:
            corr = _corr_vec(h)
            bk = int(np.argmax(corr))
            lag = bk - (n - 1)
            i = int(rng.randint(n))
            j = (i + abs(lag)) % n
        else:
            i = int(rng.randint(n))
            off = int(rng.randint(1, max(2, n // 8))) * (1 if rng.random() < 0.5 else -1)
            j = (i + off) % n
        if i != j and h[i] > 1e-8 and h[j] < 1.0 - 1e-8:
            d = min(h[i], 1.0 - h[j], base * (0.5 + 1.5 * rng.random()))
            h[i] -= d
            h[j] += d
            cn = _c5(h)
            if cn < c - 1e-14:
                c = cn
                if c < bc:
                    bc = c
                    bh = h.copy()
            else:
                h[i] += d
                h[j] -= d
        it += 1
    c = _c5(h)
    if c < bc:
        bc = c
        bh = h.copy()
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n = len(h0)
    h0 = _proj(h0, n * 0.5)
    best = h0.copy()
    best_c = _c5(best)

    # Phase 1: FISTA (90s)
    end1 = t0 + 90.0
    hc = best.copy()
    for a in [50, 200, 800, 3000, 10000]:
        rem = end1 - time.time()
        if rem < 4:
            break
        hf, bh, bc = _fista(hc, a, rem * 0.20)
        if bc < best_c:
            best, best_c = bh, bc
        hc = hf

    # Phase 2: Binary local search (220s)
    end2 = t0 + 310.0
    bh, bc = _bin_local(hc, rng, (end2 - time.time()) * 0.55)
    if bc < best_c:
        best, best_c = bh, bc
    bh2, bc2 = _bin_local(best, rng, min(60.0, end2 - time.time() - 3))
    if bc2 < best_c:
        best, best_c = bh2, bc2

    # Phase 3: FISTA from best binary (70s)
    end3 = t0 + 380.0
    for a in [3000, 15000, 60000]:
        rem = end3 - time.time()
        if rem < 4:
            break
        hf, bh, bc = _fista(best.copy(), a, rem * 0.30)
        if bc < best_c:
            best, best_c = bh, bc

    # Phase 4: Continuous local search (380s)
    end4 = t0 + 760.0
    best, best_c = _cont_local(best, rng, end4 - 5, s0=4e-5)

    # Phase 5: Binary local again on continuous result (80s)
    end5 = t0 + 840.0
    bh, bc = _bin_local(best, rng, end5 - 5)
    if bc < best_c:
        best, best_c = bh, bc

    # Phase 6: Fine continuous (160s)
    end6 = t0 + 1000.0
    best, best_c = _cont_local(best, rng, end6 - 5, s0=8e-6)

    # Phase 7: Micro (90s)
    end7 = t0 + budget_s - 8.0
    if end7 > time.time() + 5:
        best, best_c = _cont_local(best, rng, end7 - 5, s0=2e-6)

    # Final binary sweep
    if time.time() < t0 + budget_s - 8:
        bh, bc = _bin_local(best, rng, min(15.0, t0 + budget_s - 10 - time.time()))
        if bc < best_c:
            best, best_c = bh, bc

    best = _proj(best, len(best) * 0.5)
    best_c = _c5(best)
    return best, best_c, len(best)
