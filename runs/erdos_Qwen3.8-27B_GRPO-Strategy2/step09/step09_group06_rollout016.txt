                diff += a
    return h

def _obj(x, alpha, M, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    top = np.partition(c, -M)[-M:]
    s = alpha * (top - top[-1])
    s = np.clip(s, -500, 500)
    return c.max() + float(np.log(np.exp(s).sum())) / alpha

def _grad(x, alpha, M, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argpartition(c, -M)[-M:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s, -500, 500)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(M):
        if w[ii] < 1e-15:
            continue
        k = int(idx[ii])
        L = k - (n - 1)
        if L >= 0:
            g[L:] += w[ii] * (1.0 - x[:n-L])
            g[:n-L] -= w[ii] * x[L:]
        else:
            Mv = -L
            g[:n-Mv] += w[ii] * (1.0 - x[Mv:])
            g[Mv:] -= w[ii] * x[:n-Mv]
    return g * dx

def _sum_fn(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, M, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_obj, h, jac=_grad, args=(alpha, M, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'maxls': 80})
        return _repair(res.x, n)
    except Exception:
        return h

def _targeted_move(h, c_vec, step, rng):
    n = len(h)
    h2 = h.copy()
    top_idx = np.argpartition(c_vec, -4)[-4:]
    for k in top_idx:
        L = int(k) - (n - 1)
        if L > 0:
            prods = h2[L:] * (1.0 - h2[:n-L])
            i_max = int(np.argmax(prods))
            d = step * (0.4 + 0.6 * rng.rand())
            if h2[i_max + L] > d and h2[i_max] < 1.0 - d:
                h2[i_max + L] -= d
                h2[i_max] += d
        elif L < 0:
            t = -L
            prods = h2[:n-t] * (1.0 - h2[t:])
            i_max = int(np.argmax(prods))
            d = step * (0.4 + 0.6 * rng.rand())
            if h2[i_max] > d and h2[i_max + t] < 1.0 - d:
                h2[i_max] -= d
                h2[i_max + t] += d
    return h2

def _anneal(h, n_iter, rng, base_step, T0, T1):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    h_cur = h.copy()
    for it in range(n_iter):
        T = T0 * (T1 / T0) ** (it / max(n_iter, 1))
        c_vec = _eval_vec(h_cur)
        step = base_step * (0.5 + rng.rand())
        h_new = h_cur.copy()
        r = rng.rand()
        if r < 0.65:
            h_new = _targeted_move(h_new, c_vec, step, rng)
        elif r < 0.85:
            k = rng.randint(2, 5)
            idx = rng.choice(n, size=k, replace=False)
            d = step * 0.7 * rng.randn(k)
            d -= d.sum() / k
            h_new[idx] += d
        else:
            i = rng.randint(n)
            j = (i + rng.randint(1, n // 4)) % n
            d = step * rng.randn()
            h_new[i] += d
            h_new[j] -= d
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        delta = c5n - c5cur
        if delta < 0 or (T > 1e-12 and rng.rand() < np.exp(-max(delta, 0) / T)):
            h_cur = h_new
            c5cur = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = h_cur.copy()
    return best_h, best_c

def _fine_search(h, n_iter, rng, base_step):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    h_cur = h.copy()
    for it in range(n_iter):
        c_vec = _eval_vec(h_cur)
        step = base_step * (0.5 + rng.rand())
        h_new = _targeted_move(h_cur, c_vec, step, rng)
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5c:
            c5c = c5n
            h_cur = h_new
            if c5c < best_c:
                best_c = c5c
                best_h = h_cur.copy()
        else:
            k = rng.randint(2, 3)
            idx = rng.choice(n, size=k, replace=False)
            d = base_step * 0.3 * rng.randn(k)
            d -= d.sum() / k
            h_try = h_cur.copy()
            h_try[idx] += d
            h_try = _repair(h_try, n)
            c5t = _eval(h_try)
            if c5t < c5c:
                c5c = c5t
                h_cur = h_try
                if c5c < best_c:
                    best_c = c5c
                    best_h = h_cur.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _el():
        return time.time() - t0

    # Phase 1: SLSQP warm-up
    p1 = [(3000, 25, 500), (15000, 35, 500), (75000, 45, 500),
          (400000, 55, 450), (2000000, 65, 450),
          (10000000, 75, 400), (50000000, 85, 350)]
    for alpha, M, mi in p1:
        if _el() > budget_s * 0.38:
            break
        h_new = _slsqp(best_h, alpha, M, mi)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 2: Simulated annealing (exploration)
    if _el() < budget_s - 400:
        rem = budget_s - 400 - _el()
        ntries = max(100, int(rem / 0.005))
        best_h, best_c5 = _anneal(best_h, ntries, rng, 1.2e-4, 2e-5, 5e-9)

    # Phase 3: SLSQP high-alpha polish
    if _el() < budget_s - 180:
        p3 = [(300000000, 90, 400), (2000000000, 100, 350),
              (15000000000, 110, 300), (100000000000, 120, 250)]
        for alpha, M, mi in p3:
            if _el() > budget_s - 140:
                break
            h_new = _slsqp(best_h, alpha, M, mi)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 4: Fine targeted search
    if _el() < budget_s - 60:
        rem = budget_s - 60 - _el()
        ntries = max(50, int(rem / 0.003))
        best_h, best_c5 = _fine_search(best_h, ntries, rng, 2e-5)

    # Phase 5: Ultra-fine 2-point polish
    if _el() < budget_s - 1:
        h2 = best_h.copy()
        nc2 = len(h2)
        c5c = _eval(h2)
        npol = int((budget_s - 1 - _el()) / 0.0005)
        for _ in range(npol):
            if _el() > budget_s - 1:
                break
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 4)) % nc2
            d = 4e-8 * rng.randn()
            ni = h2[i] + d
            nj = h2[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
