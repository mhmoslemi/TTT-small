import numpy as np
import time

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    for _ in range(200):
        s = h.sum()
        if s < 1e-12:
            return np.full(n, 0.5)
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - tgt) < 1e-14:
            break
    d = tgt - h.sum()
    if abs(d) > 1e-14:
        idx = np.argsort(1 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
    return h

def _proj(h, tgt):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _ls_obj(h, alpha, topm, n):
    c = _corr(h)
    idx = np.argsort(c)[-topm:]
    ct = c[idx]
    s = alpha * (ct - c.max())
    s = np.clip(s - s.max(), -500, 500)
    return c.max() + np.log(np.exp(s).sum()) / alpha

def _ls_grad(h, alpha, topm, n):
    dx = 2.0 / n
    c = _corr(h)
    idx = np.argsort(c)[-topm:]
    ct = c[idx]
    s = alpha * (ct - c.max())
    s = np.clip(s - s.max(), -500, 500)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - h[:n - L])
            g[:n - L] -= wk * h[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n - M]
    return g

def _fista(h, alpha, topm, n, niter, step0):
    x = h.copy()
    y = h.copy()
    t = 1.0
    step = step0
    for k in range(niter):
        g = _ls_grad(y, alpha, topm, n)
        z = y - step * g
        x_new = _proj(z, n * 0.5)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + (t - 1.0) / t_new * (x_new - x)
        y = np.clip(y, 0.0, 1.0)
        x = x_new
        t = t_new
        if k % 50 == 0 and k > 0:
            step = min(step * 1.08, step0 * 50.0)
    return _repair(x, n)

def _targeted_fista(h, niter, step0, rng, n):
    x = h.copy()
    y = h.copy()
    t = 1.0
    step = step0
    for k in range(niter):
        if k % 200 == 0:
            alpha = 50000 * (1 + k / niter * 5)
            topm = 40
        g = _ls_grad(y, alpha, topm, n)
        if k % 100 == 50:
            pert = rng.randn(n) * (step0 * 0.1)
            g += pert / (1 + k * 1e-4)
        z = y - step * g
        x_new = _proj(z, n * 0.5)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + (t - 1.0) / t_new * (x_new - x)
        y = np.clip(y, 0.0, 1.0)
        x = x_new
        t = t_new
        if k % 100 == 0 and k > 0:
            step = min(step * 1.05, step0 * 30.0)
    return _repair(x, n)

def _block_search(h, n_iter, rng, base_step, n):
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    for it in range(n_iter):
        frac = it / max(n_iter, 1)
        step = base_step * (3.0 - 2.0 * frac) * (0.5 + rng.rand())
        blk_len = int(rng.choice([5, 10, 20, 40, 80, 140]))
        start = rng.randint(0, n - blk_len)
        idx = np.arange(start, start + blk_len)
        d = step * rng.randn(blk_len)
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < -0.05 or h_new.max() > 1.05:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5cur - 1e-15:
            c5cur = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def _point_search(h, n_iter, rng, base_step, n):
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    sizes = [2, 2, 3, 3, 4, 5, 7]
    for it in range(n_iter):
        frac = it / max(n_iter, 1)
        step = base_step * (4.0 - 3.0 * frac) * (0.5 + rng.rand())
        k = int(rng.choice(sizes))
        idx = rng.randint(0, n, size=k)
        d = step * rng.randn(k)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < -0.02 or h_new.max() > 1.02:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5cur - 1e-16:
            c5cur = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def _push_binary(h, n, strength):
    h_new = h.copy()
    mask_lo = h_new < 0.5
    mask_hi = ~mask_lo
    h_new[mask_lo] -= strength * h_new[mask_lo]
    h_new[mask_hi] += strength * (1.0 - h_new[mask_hi])
    return _repair(np.clip(h_new, 0, 1), n)
