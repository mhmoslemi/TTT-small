                else:

                    h[j] += d
            else:
                d = cur_s * (0.5 + rng.random())
                if d > min(h[i], 1.0 - h[j]):
                    d = min(h[i], 1.0 - h[j]) * 0.8
                if d < 1e-10:
                    continue
                h[i] -= d
                h[j] += d
                cn = _c5(h)
                if cn < c - 1e-13:
                    c = cn
                    if c < best_c:
                        best_c, best_h = c, h.copy()
                else:
                    h[i] += d
                    h[j] -= d
        elif r < 0.8:
            # Block move: perturb a contiguous segment
            bl = max(2, int(rng.randint(2, n // 15)))
            start = int(rng.randint(0, n - bl))
            amt = cur_s * 0.5 * (rng.random() - 0.3)
            h[start:start + bl] += amt
            np.clip(h, 0.0, 1.0, out=h)
            h = _proj(h, S)
            cn = _c5(h)
            if cn < c - 1e-13:
                c = cn
                if c < best_c:
                    best_c, best_h = c, h.copy()
            else:
                h[start:start + bl] -= amt
                np.clip(h, 0.0, 1.0, out=h)
                h = _proj(h, S)
        else:
            # Lag-directed: pick positions contributing to worst lag
            if len(worst_lags) > 0:
                k = int(worst_lags[rng.randint(len(worst_lags))])
                a = k - (n - 1)
                lo = max(0, -a)
                hi = min(n, n - a)
                if hi - lo > 1:
                    i = int(rng.randint(lo, hi))
                    j = int(rng.randint(n))
                    if i != j:
                        if h[i] < h[j]:
                            d = cur_s * (0.3 + rng.random() * 0.7)
                            d = min(d, 1.0 - h[i], h[j])
                            if d > 1e-10:
                                h[i] += d
                                h[j] -= d
                                cn = _c5(h)
                                if cn < c - 1e-13:
                                    c = cn
                                    if c < best_c:
                                        best_c, best_h = c, h.copy()
                                else:
                                    h[i] -= d
                                    h[j] += d
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _proj(h, m * 0.5)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def _symmetries(h, S):
    n = len(h)
    cands = [h, 1.0 - h, h[::-1]]
    cands.append(0.5 + 0.5 * (h - h[::-1]))
    cands.append(0.5 + 0.5 * (h + h[::-1] - 1.0))
    best_c = _c5(h)
    best = h.copy()
    for v in cands:
        hv = _proj(v, S)
        cv = _c5(hv)
        if cv < best_c:
            best_c, best = cv, hv
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        n0 = 560
        h0 = np.zeros(n0)
        ones = np.arange(0, n0, 2)
        h0[ones] = 1.0
    n0 = len(h0)
    S0 = n0 * 0.5
    h0 = _proj(h0, S0)
    best, best_c = _symmetries(h0, S0)

    def remaining():
        return budget_s - 5.0 - (time.time() - t0)

    # Phase 1: Nesterov subgradient (90s)
    e1 = t0 + 90.0
    rem = e1 - time.time()
    if rem > 10:
        bh, bc = _nesterov(best, rem * 0.6, tol=1e-3)
        if bc < best_c:
            best, best_c = bh, bc
        bh, bc = _nesterov(best, min(rem * 0.3, 30), tol=1e-5)
        if bc < best_c:
            best, best_c = bh, bc

    # Phase 2: Local search coarse (200s)
    e2 = t0 + 290.0
    rem = e2 - time.time()
    if rem > 20:
        best, best_c = _loc(best, rng, rem, s0=2e-3)

    # Phase 3: Try different grid size
    e3 = t0 + 400.0
    rem = e3 - time.time()
    if rem > 30 and best_c > 0.3808:
        for n_try in [420, 720]:
            if rem < 20:
                break
            h_try = _resample(best, n_try)
            S_try = n_try * 0.5
            bh, bc = _nesterov(h_try, min(rem * 0.3, 30), tol=1e-3)
            bh, bc = _loc(bh, rng, min(rem * 0.4, 40), s0=1e-3)
            if bc < best_c:
                best, best_c = bh, bc
            rem = e3 - time.time()

    # Phase 4: Nesterov refinement (100s)
    e4 = t0 + 500.0
    rem = e4 - time.time()
    if rem > 15:
        bh, bc = _nesterov(best, rem * 0.5, tol=1e-5)
        if bc < best_c:
            best, best_c = bh, bc
        bh, bc = _nesterov(best, rem * 0.4, tol=1e-6)
        if bc < best_c:
            best, best_c = bh, bc

    # Phase 5: Local search medium (250s)
    e5 = t0 + 750.0
    rem = e5 - time.time()
    if rem > 20:
        best, best_c = _loc(best, rng, rem, s0=5e-4)

    # Phase 6: Nesterov fine (100s)
    e6 = t0 + 850.0
    rem = e6 - time.time()
    if rem > 10:
        bh, bc = _nesterov(best, rem, tol=1e-7)
        if bc < best_c:
            best, best_c = bh, bc

    # Phase 7: Local search fine (remaining)
    e7 = budget_s - 5.0
    rem = e7 - time.time()
    if rem > 10:
        best, best_c = _loc(best, rng, rem, s0=1e-4)

    # Final symmetry + repair
    n = len(best)
    best, best_c = _symmetries(best, n * 0.5)
    best = _proj(best, n * 0.5)
    best_c = _c5(best)
    return best, best_c, n
