        it += 1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n = len(h0)
    h0 = _repair(h0, n)
    best_h = h0.copy()
    best_c = _c5(h0)
    n_starts = 6
    t_mid = t0 + budget_s * 0.55
    t_start_each = (t_mid - t0) / n_starts
    starts = [
        h0,
        _tm(n, rng),
        _rnd(n, rng),
        _bin(np.roll(h0, n // 7), rng),
        _bin(1.0 - h0, rng),
    ]
    while len(starts) < n_starts:
        pert = h0 + 0.3 * rng.randn(n)
        starts.append(_bin(pert, rng))
    for s in range(n_starts):
        te = min(t_mid, t0 + (s + 1) * t_start_each)
        if time.time() > te - 1:
            continue
        h_sa, c_sa = _sa(starts[s], rng, te)
        if c_sa < best_c:
            best_c = c_sa
            best_h = h_sa.copy()
        if time.time() > t_mid - 2:
            break
    # local search phase
    t_ls1_end = t0 + budget_s * 0.72
    best_h, best_c = _ls2(best_h, rng, t_ls1_end)
    # binary local search
    t_ls2_end = t0 + budget_s * 0.82
    hb = _bin(best_h, rng)
    cb = _c5(hb)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()
    best_h, best_c = _sa(best_h, rng, t_ls2_end)
    # final fine local search
    t_end = t0 + budget_s - 10
    best_h, best_c = _ls2(best_h, rng, t_end)
    # repair
    best_h = _repair(best_h, n)
    best_c = _c5(best_h)
    return best_h, best_c, n
