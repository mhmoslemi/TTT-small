def _eval(h):

def _eval_vec(h):

    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(80):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            step = min(abs(diff), 1.0 - h[idx]) if diff > 0 else min(-diff, h[idx])
            h[idx] += step if diff > 0 else -step
            diff += -step if diff > 0 else step
            if abs(diff) < 1e-12:

def _smobj(x, alpha, m):
    c = _eval_vec(x)
    top = np.sort(c)[-m:]
    s = alpha * top
    s -= s.max()
    return top[-1] + np.log(np.exp(s).sum()) / alpha

def _smgrad(x, alpha, m):

    dx = 2.0 / n
    c = _eval_vec(x)
    idx = np.argsort(c)[-m:]
    ctop = c[idx]
    s = alpha * ctop
    s -= s.max()

                h[i], h[j] = h[i] - d, h[j] + d
    return h, cc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = _repair(h, n)
    best_h, best_c5 = h.copy(), _eval(h)

    def _left():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP with increasing alpha
    alphas = [3000, 12000, 50000, 250000, 1000000, 5000000, 25000000]
    for a in alphas:
        if _left() < 280:
            break
        m = min(30, max(10, int(4 * np.log10(a / 3000.0) + 10)))
        hn = _slsqp_pass(best_h, a, m, 3000)
        cn = _eval(hn)
        if cn < best_c5:
            best_c5, best_h = cn, hn

    # Phase 2: Active-set steepest descent
    if _left() > 200:
        maxit = min(int(_left() / 3), 800)
        hn, cn = _active_descent(best_h.copy(), n, maxit)
        if cn < best_c5:
            best_c5, best_h = cn, hn

    # Phase 3: Momentum gradient with multiple alpha settings
    if _left() > 180:
        for a, mu in [(20000000, 0.88), (200000, 0.85), (5000000, 0.90)]:
            if _left() < 120:
                break
            m = min(25, max(12, int(3 * np.log10(a / 2e5) + 12)))
            maxit = min(int(_left() / 8), 6000)
            hn, cn = _momentum_grad(best_h.copy(), n, a, m, maxit, mu)
            if cn < best_c5:
                best_c5, best_h = cn, hn

    # Phase 4: Second SLSQP pass from improved point
    if _left() > 100:
        for a in [50000000, 200000000]:
            if _left() < 80:
                break
            m = 20
            hn = _slsqp_pass(best_h, a, m, 2000)
            cn = _eval(hn)
            if cn < best_c5:
                best_c5, best_h = cn, hn

    # Phase 5: Fine 2-point search (coarse then fine)
    if _left() > 20:
        hn, cn = _fine_search(best_h, n, rng, 30000, 5e-6)
        if cn < best_c5:
            best_c5, best_h = cn, hn
    if _left() > 10:
        hn, cn = _fine_search(best_h, n, rng, 30000, 5e-7)
        if cn < best_c5:
            best_c5, best_h = cn, hn
    if _left() > 5:
        hn, cn = _fine_search(best_h, n, rng, 20000, 2e-7)
        if cn < best_c5:
            best_c5, best_h = cn, hn

    # Phase 6: Final active descent polish
    if _left() > 5:
        hn, cn = _active_descent(best_h.copy(), n, min(int(_left() / 4), 200))
        if cn < best_c5:
            best_c5, best_h = cn, hn

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
