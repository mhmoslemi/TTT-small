    b1, b2, eps = 0.85, 0.999, 1e-8
    best_c = _eval(h)
    best_h = h.copy()
    for t in range(steps):
        if time.time() > deadline:
            break
        g, c = _grad(h, alpha, act)
        mv = b1 * mv + (1.0 - b1) * g
        vv = b2 * vv + (1.0 - b2) * g * g
        mh = mv / (1.0 - b1 ** (t + 1))
        vh = vv / (1.0 - b2 ** (t + 1))
        h_new = _proj(h - lr * mh / (np.sqrt(vh) + eps), n * 0.5)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def _polish2(h, n_steps, rng, deadline):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    scales = np.array([1e-4, 5e-5, 2e-5, 1e-5, 5e-6, 1e-6,
                       -1e-4, -5e-5, -2e-5, -1e-5, -5e-6, -1e-6])
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + rng.randint(1, 15)) % n
        base = best_c
        h_try = h.copy()
        improved = False
        for sc in scales:
            d = sc * (0.5 + rng.rand())
            hi, hj = h_try[i] + d, h_try[j] - d
            if 0 <= hi <= 1 and 0 <= hj <= 1:
                h_try[i], h_try[j] = hi, hj
                h_try = _repair(h_try, n)
                c_try = _eval(h_try)
                if c_try < base - 1e-15:
                    base = c_try
                    improved = True
                    best_h = h_try.copy()
                    best_c = c_try
                    h = h_try
        if not improved:
            h = h.copy()  # reset
    return best_h, best_c
