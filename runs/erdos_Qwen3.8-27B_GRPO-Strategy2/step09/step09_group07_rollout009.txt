        h_pert = _repair(h, n)
        h_pert = _cd_pass(h_pert, rng, n_subsets=1)
        h_pert = _repair(h_pert, n)
        c5_new = _eval(h_pert)
        delta = c5_new - h_cur_c5
        if delta < 0 or (temp > 1e-6 and rng.random() < np.exp(-delta / (temp * 0.01))):
            h = h_pert
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1
        if stagnation > 80:
            temp *= 0.7
            stagnation = 0
        if temp < 0.0005:
            temp = 0.0005
    return best_h, best_c5

def _local_search(h, n, rng, n_iter):
    c5c = _eval(h)
    best_h = h.copy()
    best_c5 = c5c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        d = 1e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = np.array(initial_h_values, dtype=float).copy()
    n = len(h)
    h = _repair(h, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Gradient descent with smooth max (20% budget)
    p1_end = t0 + budget_s * 0.20
    h_gd, c5_gd = _grad_descent(h.copy(), n, rng, t0, p1_end)
    if c5_gd < best_c5:
        best_c5 = c5_gd
        best_h = h_gd

    # Phase 2: Simulated annealing with CD (35% budget)
    p2_end = t0 + budget_s * 0.55
    h_an, c5_an = _anneal(best_h.copy(), n, rng, t0, p2_end)
    if c5_an < best_c5:
        best_c5 = c5_an
        best_h = h_an

    # Phase 3: Multiple CD passes (25% budget)
    p3_end = t0 + budget_s * 0.80
    h_cd = best_h.copy()
    while time.time() < p3_end:
        h_cd = _cd_pass(h_cd, rng, n_subsets=1)
        h_cd = _repair(h_cd, n)
        c5 = _eval(h_cd)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cd.copy()

    # Phase 4: Second anneal from improved solution (15% budget)
    if time.time() < t0 + budget_s - 80:
        p4_end = t0 + budget_s * 0.95
        h_an2, c5_an2 = _anneal(best_h.copy(), n, rng, t0, p4_end)
        if c5_an2 < best_c5:
            best_c5 = c5_an2
            best_h = h_an2

    # Phase 5: Final CD + local search polishing
    if time.time() < t0 + budget_s - 10:
        h_fin = best_h.copy()
        h_fin = _cd_pass(h_fin, rng)
        h_fin = _repair(h_fin, n)
        c5 = _eval(h_fin)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_fin
        n_iter = int((budget_s - 12 - (time.time() - t0)) / 0.004)
        n_iter = max(500, min(n_iter, 150000))
        best_h, best_c5 = _local_search(best_h, n, rng, n_iter)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
