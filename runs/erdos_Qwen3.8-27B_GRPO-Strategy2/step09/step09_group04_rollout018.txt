def _coord_descent(h, n, passes=10):
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full') * dx  # current C values
    
    for p in range(passes):
        for j in range(n):
            old_hj = h[j]
            if old_hj < 1e-10 or old_hj > 1 - 1e-10:
                # Check if we can improve
                pass
            
            # Compute A(L) for all L = -(n-1) to (n-1)
            # A(L) = (1-h[j-L]) if 0<=j-L<n, else 0
            #        - h[j+L] if 0<=j+L<n, else 0
            
            # L ranges from -(n-1) to (n-1), idx = L+(n-1) ranges from 0 to 2n-2
            A = np.zeros(2*n-1)
            # First term: 1-h[j-L] for 0<=j-L<n → j-n+1<=L<=j
            lo1 = max(-(n-1), j-(n-1))
            hi1 = min(n-1, j)
            if lo1 <= hi1:
                # L from lo1 to hi1: A[L+(n-1)] += 1-h[j-L]
                L_vals = np.arange(lo1, hi1+1)
                A[L_vals+(n-1)] += 1.0 - h[j-L_vals]
            
            # Second term: -h[j+L] for 0<=j+L<n → -j<=L<=n-1-j
            lo2 = max(-(n-1), -j)
            hi2 = min(n-1, n-1-j)
            if lo2 <= hi2:
                L_vals = np.arange(lo2, hi2+1)
                A[L_vals+(n-1)] -= h[j+L_vals]
            
            # B(L) = corr[L] - A(L)*old_hj
            B = corr - A * old_hj
            
            # Minimize max_L (A[L]*t + B[L]) over t in [0,1] with sum constraint
            # The sum constraint: sum(h) = target
            # If I change h[j] from old_hj to t, I need to compensate elsewhere
            # For now, ignore the constraint and project later
            
            # 1D optimization: minimize f(t) = max_k (A[k]*t + B[k]) for t in [0,1]
            # f(t) is convex piecewise linear
            # Use ternary search
            lo, hi = 0.0, 1.0
            for _ in range(60):
                m1 = lo + (hi-lo)/3.0
                m2 = hi - (hi-lo)/3.0
                if np.max(A*m1 + B) < np.max(A*m2 + B):
                    hi = m2
                else:
                    lo = m1
            t_new = (lo+hi)/2.0
            
            # But we need to maintain sum(h) = target
            # Changing h[j] by (t_new - old_hj) breaks the constraint
            # We need to compensate. Simple approach: distribute the change over all other coordinates
            # Or: just do the change and repair at the end
            h[j] = t_new
            # Update correlation
            diff = t_new - old_hj
            corr += diff * A
    
        # Repair sum constraint
        h = _repair(h, n)
        # Recompute correlation
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
    
    return h
