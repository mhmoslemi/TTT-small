def _gradient_step(h, step_scale):

    # Compute gradient of the actual max (using subgradient)
    c = np.correlate(h, 1-h, mode='full') * 2.0/n
    m = c.max()
    # Active set: shifts where c is close to max
    tol = 1e-8
    active = np.where(c >= m - tol)[0]
    # Subgradient: average of gradients at active shifts
    g = np.zeros(n)
    for k in active:

        if L >= 0:
            g[L:] += (1.0 - h[:n-L])
            g[:n-L] -= h[L:]
        else:

    g /= len(active)

    h_new = _proj(h - step_scale * g, n * 0.5)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, ml)[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argpartition(-c, ml)[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _subgrad(h, tol):
    """Subgradient of max_k c_k using active set."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
    m = c.max()
    active = np.where(c >= m - tol)[0]
    g = np.zeros(n)
    for k in active:
        L = k - (n - 1)
        if L >= 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    g /= max(len(active), 1)
    return g, m, active

def _sum_fn(x, half):
    return x.sum() - half

def _sum_jac(x, half):
    return np.ones(len(x))

def _slsqp(h, alpha, ml, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha, ml),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 50})
        return _repair(res.x, n)
    except Exception:
        return h

def _pg_long(h, alpha, ml, steps, lr0, momentum):
    n = len(h)
    v = np.zeros(n)
    best_c = _eval(h)
    best_h = h.copy()

        v = momentum * v - lr * g

        c_new = _eval(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        taken = False
        step = lr
        for _t in range(12):

            c_try = _eval(h_try)
            if c_try < _eval(h) - 1e-15:
                h = h_try
                v = momentum * v
                lr = min(step * 1.08, lr0 * 25.0)
                taken = True
                break
            step *= 0.5
        if not taken:
            h = h_new
            lr = min(lr * 1.04, lr0 * 25.0)
    return best_h

def _subgrad_opt(h, n_steps, tol_init, tol_min):

    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    step = 0.005
    for it in range(n_steps):

        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            step = min(step * 1.03, 0.02)
        else:
            step *= 0.7
        h = h_new
        if step < 1e-8:
            break
    return best_h

def _grad_local(h, n_iter, rng, alpha, ml):
    n = len(h)
    c5c = _eval(h)
    best_c = c5c
    best_h = h.copy()
    g = _smooth_grad(h, alpha, ml)
    for _ in range(n_iter):
        top3 = np.argpartition(-np.abs(g), 3)[:3]
        i, j = int(top3[0]), int(top3[1])

        h_try = h.copy()
        h_try[i] += d
        h_try[j] -= d
        if h_try.min() < -1e-10 or h_try.max() > 1.0 + 1e-10:

            h_try[j] -= d

        h_try = _repair(h_try, n)

            h = h_try
            g = _smooth_grad(h, alpha, ml)
            if c5c < best_c:

def _coarse_upsample(h_coarse, n_target, f):
    out = np.empty(n_target)
    nc = len(h_coarse)
