        (500000000, 120, 600, 0.0004),
    ]
    for alpha, ml, steps, s0 in pg_configs:
        if time.time() > dl2:
            break
        h2, c2 = _pgd(best_h.copy(), alpha, ml, steps, s0, dl2)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2

    # Phase 3: Subgradient walk (explores landscape)
    dl3 = t0 + min(700, budget_s - 200)
    if time.time() < dl3:
        h2, c2 = _subgrad_walk(best_h, 50000, rng, 80, dl3)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2

    # Phase 4: Coordinate descent (long)
    dl4 = t0 + min(950, budget_s - 100)
    if time.time() < dl4:
        rem = dl4 - time.time()
        n_it = int(rem * 3500)
        h2, c2 = _coord_desc(best_h, n_it, rng, dl4)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2

    # Phase 5: Pair optimization
    dl5 = t0 + budget_s - 20
    if time.time() < dl5:
        h2, c2 = _pair_opt(best_h, dl5)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2

    # Phase 6: Final ultra-fine coordinate descent
    dl6 = t0 + budget_s - 0.5
    if time.time() < dl6:
        rem = dl6 - time.time()
        n_it = int(rem * 5000)
        h2, c2 = _coord_desc(best_h, max(n_it, 1000), rng, dl6)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
