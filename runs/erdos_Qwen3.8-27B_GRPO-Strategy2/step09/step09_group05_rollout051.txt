                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _all_lags(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _subgrad_lag(h, M):
    """Subgradient of c[±M] w.r.t. h. M = |lag|."""
    n = len(h)
    dx = 2.0 / n
    if M >= n:
        return np.zeros(n)
    g = np.zeros(n)
    g[M:] += dx * (1.0 - h[:n - M])
    g[:n - M] -= dx * h[M:]
    return g

def _smooth_obj(x, alpha, m_lags):
    c = _all_lags(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = alpha * (c[idx] - m)
    s = np.clip(s, -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_lags):
    n = len(x)
    c = _all_lags(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * (2.0 / n)
        L = k - (n - 1)
        M = abs(L)
        if M >= n:
            continue
        g[M:] += wk * (1.0 - x[:n - M])
        g[:n - M] -= wk * x[M:]
    return g

def _sum_fn(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:

def _nesterov_proj_grad(h, alpha, m_lags, n_steps, step0):

    n = len(h)
    h_curr = h.copy()
    h_prev = h.copy()
    step = step0
