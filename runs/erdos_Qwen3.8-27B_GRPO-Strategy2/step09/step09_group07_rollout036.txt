def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _corr(h):
    n = len(h)

    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    return m + np.log(np.sum(np.exp(c - m))) / alpha

    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    w = np.exp(c - m)
    w /= w.sum()
    w *= dx / alpha

        else:

    return g

    return x.sum() - target

    return np.ones(len(x))

    def obj(x):

        s = np.clip(alpha * (c[idx] - m), -500, 500)

        return m + np.log(np.exp(s).sum()) / alpha

        c = np.correlate(x, 1.0 - x, mode='full') * dx

            else:
                M = -L
                g[:n-M] += wk * (1.0 - x[M:])
                g[M:] -= wk * x[:n-M]
        return g

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _cd_pass(h, rng):
    n = len(h)
    c = _corr(h)
    cmax = 2 * n - 1
    nm1 = n - 1
    jlist = np.arange(n)
    rng.shuffle(jlist)
    for j in jlist:
        j = int(j)
        beta = np.zeros(cmax)
        Lmax1 = min(j, nm1)
        if Lmax1 >= 0:
            beta[nm1:nm1+Lmax1+1] += 1.0 - h[j-Lmax1:j+1][::-1]
        Lmax2 = nm1 - j
        if Lmax2 >= 0:
            beta[nm1:nm1+Lmax2+1] -= h[j:j+Lmax2+1]
        if j < nm1:
            beta[0:nm1-j] += 1.0 - h[j+1:n][::-1]
        if j >= 1:
            beta[nm1-j:nm1] -= h[0:j]
        g = c - beta * h[j]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(22):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(g + beta * t1) <= np.max(g + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-14:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

def _lsq_obj(x, alpha, m_lags, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-m_lags:]
    m = c.max()
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _lsq_grad(x, alpha, m_lags, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-m_lags:]
    m = c.max()
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(len(idx)):
        wk = w[ii] * dx
        if wk < 1e-15:
            continue
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n-L])
            g[:n-L] -= wk * x[L:]
        else:
            M = -L
            g[:n-M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n-M]
    return g

def _sum_con(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, maxiter, n_lags, n):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_lsq_obj, h, jac=_lsq_grad,
                       args=(alpha, n_lags, n), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _perturb(h, n, rng, scale):
    h2 = h.copy()
    mask = rng.random(n) < 0.35
    cnt = mask.sum()
    if cnt > 0:
        h2[mask] += rng.randn(cnt) * scale
    return _repair(h2, n)
