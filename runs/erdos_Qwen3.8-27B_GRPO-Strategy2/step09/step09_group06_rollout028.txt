    except Exception:
        return h

def _pgrad(h, alpha, m_lags, nsteps, step0, mu):
    n = len(h)
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    for _ in range(nsteps):
        g = _lsgrad(h, alpha, m_lags, n)
        g -= g.mean()
        v = mu * v + (1.0 - mu) * g
        gnorm = np.linalg.norm(v)
        if gnorm < 1e-14:
            break
        taken = False
        for _t in range(12):
            h_new = _proj(h - step * v / gnorm, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-15:
                h = h_new
                cur_c = c_new
                step = min(step * 1.15, step0 * 40.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.3, step0 * 40.0)
    return _repair(h, n), cur_c

def _gls(h, n_trials, rng):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(best_h)
    cur_c = best_c
    T = 3e-7
    stall = 0
    for t in range(n_trials):
        cvec = _cvec(best_h)
        top_m = 14
        top_idx = np.argsort(cvec)[-top_m:]
        top_vals = cvec[top_idx]
        a = 1000.0
        s = a * (top_vals - top_vals.max())
        s = np.clip(s, -500, 0)
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        dx = 2.0 / n
        for ii, k in enumerate(top_idx):
            wk = w[ii] * dx
            L = k - (n - 1)
            if L >= 0:
                g[L:] += wk * (1.0 - best_h[:n - L])
                g[:n - L] -= wk * best_h[L:]
            else:
                M = -L
                g[:n - M] += wk * (1.0 - best_h[M:])
                g[M:] -= wk * best_h[:n - M]
        g -= g.mean()
        kw = np.abs(g) + 1e-12
        kw /= kw.sum()
        k = rng.choice([2, 2, 2, 3, 3, 4])
        idx = rng.choice(n, size=k, replace=False, p=kw)
        d = -np.sign(g[idx]) * (0.4e-5 + 1.6e-5 * rng.rand(k))
        d -= d.mean()
        h_new = best_h.copy()
        h_new[idx] += d
        h_new = _repair(h_new, n)
        c_new = _eval(h_new)
        delta = c_new - cur_c
        if delta < -1e-15 or (T > 1e-14 and rng.rand() < np.exp(-max(delta, 0.0) / T)):
            best_h = h_new
            cur_c = c_new
            if c_new < best_c - 1e-12:
                best_c = c_new
                stall = 0
                T = max(T * 0.99, 5e-10)
            else:
                stall += 1
        else:
            stall += 1
        if stall > 3000:
            pert = 5e-5 * rng.randn(n)
            pert -= pert.mean()
            best_h = _repair(best_h + pert, n)
            cur_c = _eval(best_h)
            stall = 0
    return best_h, best_c

def _twopoint(h, n_trials, rng):
    n = len(h)
    best_h = h.copy()
    cur_c = _eval(best_h)
    best_c = cur_c
    for t in range(n_trials):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 10)) % n
        d = 8e-8 * rng.randn()
        ni = best_h[i] + d
        nj = best_h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        best_h[i] = ni
        best_h[j] = nj
        c_new = _eval(best_h)
        if c_new < cur_c - 1e-16:
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
        else:
            best_h[i] -= d
            best_h[j] += d
    return _repair(best_h, n), best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    return _repair(np.repeat(h, f), len(h) * f)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 0: Multi-resolution warmup (20s)
    for ratio in [4, 8]:
        if time.time() - t0 > 20:
            break
        nc = max(50, n // ratio)
        f = n // nc
        hc = _downsample(best_h, f)
        hc = _slsqp(hc, 10000, 30, 400)
        hc = _slsqp(hc, 100000, 50, 400)
        h_up = _upsample(hc, f)
        c5u = _eval(h_up)
        if c5u < best_c5:
            best_c5 = c5u
            best_h = h_up.copy()

    # Phase 1: SLSQP sweep (120s)
    slsqp_configs = [
        (5000, 25, 600), (20000, 35, 600), (80000, 45, 550),
        (300000, 55, 500), (1000000, 65, 450), (5000000, 75, 400),
        (20000000, 85, 350), (100000000, 100, 300),
    ]
    for alpha, ml, mi in slsqp_configs:
        if time.time() - t0 > 140:
            break
        h_new = _slsqp(best_h, alpha, ml, mi)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 2: Projected gradient with momentum (200s)
    pg_configs = [
        (5000, 50, 6000, 0.004, 0.6),
        (30000, 60, 5000, 0.003, 0.65),
        (100000, 70, 4000, 0.002, 0.68),
        (500000, 80, 3500, 0.001, 0.70),
        (2000000, 90, 3000, 0.0006, 0.72),
        (10000000, 100, 2500, 0.0003, 0.74),
        (50000000, 110, 2000, 0.0002, 0.75),
    ]
    for alpha, ml, steps, s0, mu in pg_configs:
        if time.time() - t0 > 340:
            break
        h_new, c_new = _pgrad(best_h.copy(), alpha, ml, steps, s0, mu)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 3: High-alpha SLSQP polish (80s)
    for alpha, ml, mi in [(100000000, 110, 350), (500000000, 120, 300),
                          (2000000000, 130, 250)]:
        if time.time() - t0 > 420:
            break
        h_new = _slsqp(best_h, alpha, ml, mi)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 4: Gradient-informed local search (500s)
    if time.time() - t0 < budget_s - 80:
        rem = min(budget_s - 80 - (time.time() - t0), 500.0)
        ntries = int(rem / 0.0018)
        best_h, best_c5 = _gls(best_h, ntries, rng)

    # Phase 5: 2-point ultra-fine polish (remaining)
    if time.time() - t0 < budget_s - 1:
        rem = budget_s - 1 - (time.time() - t0)
        ntries = int(rem / 0.00007)
        best_h, best_c5 = _twopoint(best_h, ntries, rng)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
