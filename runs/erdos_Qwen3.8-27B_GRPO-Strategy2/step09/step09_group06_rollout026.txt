import time

    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(-h) if diff < 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _to_binary(h, n):

    idx = np.argpartition(-h, target)[:target]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _binary_sa(hb, n_iter, rng, t0, t_max):
    """Simulated annealing on binary sequence"""
    n = len(hb)
    c_cur = _eval(hb)
    best = hb.copy()
    best_c = c_cur
    cur = hb.copy()
    T = 0.01
    T_min = 1e-6
    cooling = (T_min / T) ** (1.0 / max(n_iter, 1))
    for it in range(n_iter):
        if time.time() > t_max:
            break
        T = max(T * cooling, T_min)
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        cur[i], cur[j] = 0.0, 1.0
        c_new = _eval(cur)
        delta = c_new - c_cur
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
                best = cur.copy()
        else:
            cur[i], cur[j] = 1.0, 0.0

        if np.clip(h - mid, 0, 1).sum() > target:

    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-m_lags:]
    s = alpha * (c[idx] - c[idx].max())
    s = np.clip(s, -500, 500)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):

        L = k - (n - 1)

            g[M:] -= wk * h[:n-M] * dx

    v = np.zeros(n)

        g = _grad_soft(h, alpha, m_lags, n)
        v = momentum * v + (1 - momentum) * g
        taken = False

            c_new = _eval(h_new)

                taken = True

    n = len(h)

    m_lags_list = [10, 20, 30, 40, 50, 60]
    steps_list = [500, 400, 300, 200, 150, 100]
    step0_list = [0.01, 0.008, 0.005, 0.003, 0.002, 0.001]

**Alternative: Use the autocorrelation structure**

For binary h, the problem is: maximize min_k A(k) where A(k) = autocorrelation.

This is related to "optimal difference sets" in combinatorics. For a set S of size n/2 in Z_n, the difference multiset D = {s1-s2 : s1,s2 ∈ S} has the property that |A(k)| = number of pairs (s1,s2) ∈ S×S with s1-s2 = k.

The sum of A(k) over all k is |S|² = (n/2)². So the average A(k) is n/4. We want to make A(k) as close to n/4 as possible for all k.

This is achieved by "perfect difference sets" (when they exist) or "near-perfect" ones.

For n=560, n/2=280, n/4=140. We want A(k) ≈ 140 for all k.

The maximum deviation from 140 determines C₅. This is a combinatorial optimization problem.

**Spectral approach**: The autocorrelation A(k) is related to the Fourier transform of h. Specifically, if H(ω) = FFT(h), then A(k) = IFFT(|H(ω)|²)(k). 

This means: minimize max_k (n/2 - A(k)) is equivalent to making |H(ω)|² as flat as possible.

For binary h, |H(ω)|² is determined by the positions of the 1s. We want the "periodogram" of the indicator set to be as flat as possible.

This connection to flat spectra is interesting but doesn't immediately give an algorithm.

import numpy as np
import time

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(-h) if diff < 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14: break
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _to_binary(h, n):
    target = n // 2
    idx = np.argpartition(-h, target)[:target]
    hb = np.zeros(n); hb[idx] = 1.0
    return hb

def _binary_sa(hb, n_iter, rng, T0, T_min, t_max):
    n = len(hb)
    c_cur = _eval(hb)
    best = hb.copy(); best_c = c_cur; cur = hb.copy()
    T = T0
    cool = (T_min / max(T0, 1e-12)) ** (1.0 / max(n_iter, 1))
    for _ in range(n_iter):
        if time.time() > t_max: break
        T = max(T * cool, T_min)
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0: break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        cur[i], cur[j] = 0.0, 1.0
        c_new = _eval(cur)
        d = c_new - c_cur
        if d < 0 or (T > 1e-10 and rng.rand() < np.exp(-d / T)):
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur; best = cur.copy()
        else:
            cur[i], cur[j] = 1.0, 0.0
    return best, best_c

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_soft(h, alpha, m_lags, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-m_lags:]
    s = alpha * (c[idx] - c[idx].max())
    s = np.clip(s, -500, 500)
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15: continue
        L = k - (n - 1)
        if L >= 0:
            g[:n-L] += w[ii] * (1.0 - h[L:]) * dx
            g[L:] -= w[ii] * h[:n-L] * dx
        else:
            M = -L
            g[:n-M] -= w[ii] * h[M:] * dx
            g[M:] += w[ii] * (1.0 - h[:n-M]) * dx
    return g

def _pgrad(h, alpha, m_lags, steps, step0, t_max):
    n = len(h)
    v = np.zeros(n); step = step0; c_cur = _eval(h)
    for _ in range(steps):
        if time.time() > t_max: break
        g = _grad_soft(h, alpha, m_lags, n)
        v = 0.5 * v + 0.5 * g
        taken = False
        for _ in range(15):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < c_cur - 1e-16:
                h = h_new; c_cur = c_new
                step = min(step * 1.1, step0 * 50)
                taken = True; break
            step *= 0.4
        if not taken: step = min(step * 1.5, step0 * 50)
    return _repair(h, n), c_cur

def _continuation(h, t_max):
    n = len(h)
    cfgs = [(500,10,500,0.01),(2000,15,400,0.008),(10000,20,350,0.005),
            (50000,30,300,0.003),(200000,40,250,0.002),(1000000,50,200,0.001),
            (5000000,60,150,0.0005),(20000000,80,100,0.0003)]
    c_cur = _eval(h)
    for alpha, ml, st, s0 in cfgs:
        if time.time() > t_max: break
        h, c_new = _pgrad(h, alpha, ml, st, s0, t_max)
        if c_new < c_cur: c_cur = c_new
    return _repair(h, n), c_cur

def _hard_threshold(h, n, eps=0.1):
    ht = h.copy()
    ht[ht > 1.0 - eps] = 1.0
    ht[ht < eps] = 0.0
    return _repair(ht, n)

def _local_search(h, n_iter, rng, bstep, t_max):
    n = len(h); c5c = _eval(h)
    best_h = h.copy(); best_c = c5c
    for it in range(n_iter):
        if time.time() > t_max: break
        step = bstep * (3.0 - 2.0 * it / max(n_iter, 1)) * (0.5 + rng.rand())
        k = rng.choice([2,3,4,5,6])
        idx = rng.randint(0, n, size=k)
        d = step * rng.randn(k); d -= d.mean()
        h_new = h.copy(); h_new[idx] += d
        if h_new.min() < -0.05 or h_new.max() > 1.05: continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-15:
            c5c = c5n; h = h_new
            if c5n < best_c: best_c = c5n; best_h = h.copy()
    return best_h, best_c

def _polish(h, n_iter, rng, t_max):
    n = len(h); c5c = _eval(h); best_c = c5c; best_h = h.copy()
    for _ in range(n_iter):
        if time.time()>t_max: break
        i = rng.randint(n); j = (i + 1 + rng.randint(0,10)) % n
        d = 2e-7 * rng.randn()
        ni, nj = h[i]+d, h[j]-d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c: best_c = c5n; best_h = h.copy()
            else:
                h[i], h[j] = ni-d, nj+d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy(); best_c5 = _eval(best_h)

    t1 = t0 + budget_s * 0.22
    hb = _to_binary(h, n)
    best_bin = hb.copy(); best_bin_c = _eval(hb)
    for r in range(10):
        if time.time() > t1: break
        T0 = 0.025 * max(1.0 - 0.08*r, 0.3)
        hb, c = _binary_sa(best_bin, 12000, rng, T0, 1e-8, t1)
        if c < best_bin_c: best_bin_c = c; best_bin = hb.copy()
        if best_bin_c < best_c5:
            best_c5 = best_bin_c; best_h = best_bin.copy()
    
    # Phase 2: Continuation from best
    t2 = t0 + budget_s * 0.45

    if c_w < best_c5: best_c5 = c_w; best_h = h_w.copy()
    
    # Phase 3: Threshold + Binary SA
    t3 = t0 + budget_s * 0.60
    ht = _hard_threshold(h_w, n, 0.05)
    hb3 = _to_binary(ht, n)
    best_bin3 = hb3.copy(); best_bin3_c = _eval(hb3)
    for r in range(5):

        hb3, c = _binary_sa(best_bin3, 10000, rng, T0, 1e-8, t3)
        if c < best_bin3_c: best_bin3_c = c; best_bin3 = hb3.copy()
        if best_bin3_c < best_c5:
            best_c5 = best_bin3_c; best_h = best_bin3.copy()
    
    # Phase 4: Final continuation
    t4 = t0 + budget_s * 0.80

    return (best_h, best_c5, n)
