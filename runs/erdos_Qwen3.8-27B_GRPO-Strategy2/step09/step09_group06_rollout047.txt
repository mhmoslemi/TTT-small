def _eval(h):

def _full_corr(h):

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:

    else:

    for idx in order:

            break
        if diff > 0:

            h[idx] += a
            diff -= a
        else:

def _lsqp_obj(x, alpha, m_lags, n):
    c = _full_corr(x)
    top_m = np.argpartition(c, -m_lags)[-m_lags:]
    c_top = c[top_m]
    s = alpha * (c_top - c_top.max())
    s = np.clip(s - s.max(), -500, 0)
    return c.max() + np.log(np.exp(s).sum()) / alpha

def _lsqp_grad(x, alpha, m_lags, n):
    c = _full_corr(x)
    top_m = np.argpartition(c, -m_lags)[-m_lags:]
    c_top = c[top_m]
    s = alpha * (c_top - c_top.max())
    s = np.clip(s - s.max(), -500, 0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(top_m):
        if w[ii] < 1e-15:
            continue
        L = abs(k - (n - 1))
        if L == 0:
            g += w[ii] * (1.0 - 2.0 * x)
        else:
            g[L:] += w[ii] * (1.0 - x[:n-L])
            g[:n-L] -= w[ii] * x[L:]
    return g

def _sum_con(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp_opt(h, alpha, m_lags, maxiter):
    n = len(h)
    target = n * 0.5
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (target,)}]
    try:
        res = minimize(_lsqp_obj, h, jac=_lsqp_grad, args=(alpha, m_lags, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except:
        return h

def _coord_descent(h, n_iter, rng, base_step):
    """Coordinate pair descent for fine tuning."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c = _eval(h)
    for it in range(n_iter):
        frac = it / max(n_iter, 1)
        step = base_step * (1.0 - 0.5*frac)
        k = rng.randint(2, 6)
        idx = rng.choice(n, size=k, replace=False)
        d = step * rng.randn(k)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < -0.05 or h_new.max() > 1.05:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < best_c - 1e-15:
            best_c = c5n
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def _fine_polish(h, n_iter, rng):
    """Ultra-fine 2-point polish."""
    n = len(h)
    h = h.copy()
    best_c = _eval(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 10)) % n
        d = 5e-8 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < best_c:
                best_c = c5n
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, best_c
