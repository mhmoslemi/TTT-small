import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _all_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _proj(v, S):
    x = np.clip(np.array(v, dtype=float), 0.0, 1.0).copy()
    x += (S - x.sum()) / x.size
    for _ in range(50):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-14:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _grad_active(h, k_vals, n):
    """Gradient of max over active k values"""
    dx = 2.0 / n
    g = np.zeros(n)
    for k in k_vals:
        idx = abs(k - (n - 1))
        if idx == 0:
            g += (1.0 - 2.0 * h) * dx
        else:
            # g[j] = (1-h[j-idx])*[idx<=j] - h[j+idx]*[j<n-idx]
            g[idx:n] += (1.0 - h[:n-idx]) * dx
            g[:n-idx] += -h[idx:n] * dx
    return g / len(k_vals)

def _zeleny(h, rng, deadline, n_active=10, beta=50.0):
    n = len(h)
    S = n * 0.5
    dx = 2.0 / n
    h = _proj(h, S)
    c = _c5(h)
    best_c = c
    best_h = h.copy()
    alpha = 1.0 / n  # initial step size
    t0 = time.time()
    
    while time.time() < deadline:
        corr = _all_corr(h)
        k_all = np.arange(2 * n - 1)
        # Find active set
        mx = corr.max()
        active = k_all[corr >= mx - beta * dx]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        active = active[:n_active]  # cap at n_active
        
        g = _grad_active(h, active, n)
        
        # Projected gradient step
        for _ in range(5):  # backtracking

                    best_h = h.copy()
                alpha = min(alpha * 1.05, 1.0/n*2)
                break
            alpha *= 0.5
        
        if alpha < 1e-12:
            alpha = 1.0 / n * 0.1

    mx = corr.max()
    tol = 8.0 * (2.0 / n)

    if len(active) == 0:

    for wl, wg in zip(lags, w):
