def _lag_targeted(h0, rng, t_budget):
    """Targeted moves to reduce critical lags."""
    n = len(h0)
    h = h0.astype(float).copy()
    c = _c5(h)
    bc = c
    bh = h.copy()
    t0 = time.time()
    recomp = 0
    while time.time() - t0 < t_budget:
        if time.time() - t0 > recomp:
            vec = _c5_vec(h)
            mx = vec.max()
            # Find critical lags (within 1% of max)
            crit = np.where(vec > 0.99 * mx)[0]
            recomp = time.time() + 5.0
        
        # For each critical lag, find a pair to fix
        # Lag k in "full" mode corresponds to shift k - (n-1)
        # vec[k] = sum_i h[i] * (1-h[i+k-(n-1)]) for valid i
        # For shift s = k-(n-1): overlap = sum_{i=max(0,-s)}^{min(n-1, n-1-s)} h[i] * (1-h[i+s])
        
        # Pick a random critical lag
        k = crit[rng.randint(len(crit))]
        s = k - (n - 1)  # actual shift
        
        # Find i where h[i]=1 and h[i+s]=0 (or h[i]=0 and h[i+s]=1)
        if s >= 0:
            valid_i = np.arange(0, n - s)
        else:
            valid_i = np.arange(-s, n)
        
        # h[i] * (1-h[i+s]) > 0 means h[i]=1, h[i+s]=0
        # We want to swap h[i] with some h[j] to reduce this
        mismatch = np.where((h[valid_i] > 0.5) & (h[valid_i + s] < 0.5))[0]
        if len(mismatch) == 0:
            mismatch = np.where((h[valid_i] < 0.5) & (h[valid_i + s] > 0.5))[0]
        
        if len(mismatch) > 0:
            ii = valid_i[mismatch[rng.randint(len(mismatch))]]
            # Find a j to swap with that doesn't hurt other lags
            # Simple: pick random j with opposite value
            if h[ii] > 0.5:
                zeros = np.where(h < 0.5)[0]
            else:
                zeros = np.where(h > 0.5)[0]
            j = zeros[rng.randint(len(zeros))]
            if j != ii:
                h[ii], h[j] = h[j], h[ii]
                cn = _c5(h)
                if cn < bc:
                    bc = cn; bh = h.copy()
                else:
                    h[ii], h[j] = h[j], h[ii]
    
    return bh, bc

import numpy as np

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _c5_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    S = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(50):
        h *= S / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - S) < 1e-13:
            break
    d = S - h.sum()
    if abs(d) > 1e-13:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if abs(d) < 1e-13:
                break
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
    return h

def _proj(h, S):
    x = h.astype(float).copy()
    x += (S - x.sum()) / x.size
    for _ in range(30):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-12:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x
