def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _eval_full(h):

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()

        return h
    # Proportional scaling with clipping iteration

            h[:] = 0.5
            return h

        h_s = np.clip(h_s, 0.0, 1.0)

            break

        s = s2

    diff = target - h.sum()

                remaining -= amt

                remaining += amt

def _grad_overlap(x, alpha, m, n):
