def _to_binary(h, rng):

    S = n * 0.5
    # Sort by distance to nearest of {0,1}
    d0 = h
    d1 = 1.0 - h
    d = np.minimum(d0, d1)
    # Flip the ones closest to boundary
    order = np.argsort(d)
    hb = h.copy()

            hb[i] = 0.0
        else:
            hb[i] = 1.0

        it += 1
    
    return bh, bc

import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(50):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _repair(h, n):
    S = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= S / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - S) < 1e-14:
            break
    d = S - h.sum()
    if abs(d) > 1e-14:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[i])
            else:
                a = min(-d, h[i])
            h[i] += a * (1 if d > 0 else -1)
    return h

def _to_binary(h, rng):
    n = len(h)
    S = int(n * 0.5)
    hb = (h > 0.5).astype(float)
    cnt = int(hb.sum())
    if cnt > S:
        oi = np.nonzero(hb)[0]
        rng.shuffle(oi)
        hb[oi[:cnt-S]] = 0.0
    elif cnt < S:
        zi = np.nonzero(1-hb)[0]
        rng.shuffle(zi)
        hb[zi[:S-cnt]] = 1.0
    return hb

def _grad_step(h, lr, n_active=10, tol=0.004):
    n = len(h)
    corr = np.correlate(h, 1.0-h, mode='full') * (2.0/n)
    mx = corr.max()
    active = np.where(corr >= mx - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        active = active[np.argsort(corr[active])[::-1][:n_active]]
    g = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        if lag > 0:
            g[lag:] += (1.0 - h[:n-lag])
            g[:n-lag] -= h[lag:]
        elif lag < 0:
            m = -lag
            g[:n-m] += (1.0 - h[m:])
            g[m:] -= h[:n-m]
        else:
            g += (1.0 - 2.0*h)
    g /= len(active)
    return _proj(h - lr*g, n*0.5), mx

def _grad_descent(h0, deadline, rng):
    n = len(h0)
    h = h0.copy()
    c = _c5(h)
    bh = h.copy()
    bc = c
    lr = 0.8
    while time.time() < deadline:
        h_new, mx = _grad_step(h, lr)
        nc = _c5(h_new)
        if nc < c:
            h = h_new
            c = nc
            lr = min(lr*1.08, 3.0)
            if c < bc:
                bc = c
                bh = h.copy()
        else:
            lr *= 0.85
            if lr < 0.02:
                lr = 0.5
                h = _proj(h + lr*(np.random.RandomState(int(time.time()*1e6)).rand(n)-0.5)*0.02, n*0.5)
                c = _c5(h)
    return bh, bc

def _binary_sa(h0, rng, deadline):
    t0 = time.time()
    n = len(h0)
    h = _to_binary(h0, rng)
    c = _c5(h)
    bh = h.copy()
    bc = c
    T = 0.04
    while time.time() < deadline:
        frac = min(1.0, (time.time()-t0)/max(1e-9, deadline-t0))
        T = max(1e-5, 0.04*(1e-5/0.04)**frac)
        ones = np.nonzero(h)[0]
        zeros = np.nonzero(1-h)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        i = ones[int(rng.randint(len(ones)))]
        j = zeros[int(rng.randint(len(zeros)))]
        h[i]=0.0; h[j]=1.0
        nc = _c5(h)
        dE = nc-c
        if dE<0 or rng.random()<np.exp(-dE/max(T,1e-10)):
            c=nc
            if c<bc: bc=c; bh=h.copy()
        else:
            h[i]=1.0; h[j]=0.0
    return bh, bc

def _cont_sa(h0, rng, deadline, s_max=0.08):
    t0 = time.time()
    n = len(h0)
    S = n*0.5
    h = h0.copy()
    c = _c5(h)
    bh = h.copy()
    bc = c
    T = 0.02
    while time.time() < deadline:
        frac = min(1.0, (time.time()-t0)/max(1e-9, deadline-t0))

            if i==j: continue

            else: amt=max(amt,-(1.0-h[j]),-h[i])
            if abs(amt)<1e-12: continue
            h[i]+=amt; h[j]-=amt
            nc=_c5(h)
            dE=nc-c
            if dE<0 or rng.random()<np.exp(-dE/max(T,1e-10)):
                c=nc
                if c<bc: bc=c; bh=h.copy()
            else:
                h[i]-=amt; h[j]+=amt
        elif r<0.8:
            i=int(rng.randint(n)); j=int(rng.randint(n))
            if i==j: continue
            amt=scale*3*(rng.random()*2-1)
            if amt>0: amt=min(amt,h[j],1.0-h[i])
            else: amt=max(amt,-(1.0-h[j]),-h[i])
            if abs(amt)<1e-12: continue
            h[i]+=amt; h[j]-=amt

def _structured_starts(n, rng):

    alpha = (1+5**0.5)/2.0
