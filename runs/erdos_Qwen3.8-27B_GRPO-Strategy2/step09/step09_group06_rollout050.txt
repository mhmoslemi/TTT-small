def _grad_one(x, k, n):
    """Gradient of f_k(h) = dx * sum h[i]*(1-h[i+k_effective])"""
    dx = 2.0 / n
    L = k - (n - 1)
    g = np.zeros(n)
    if L >= 0:

        g[:n - L] -= x[L:]
    else:
        M = -L
        g[:n - M] += (1.0 - x[M:])
        g[M:] -= x[:n - M]
    return g * dx
