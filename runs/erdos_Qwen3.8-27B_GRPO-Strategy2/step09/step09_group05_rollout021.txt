import numpy as np
from scipy.optimize import minimize
import time


def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h


def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def _sobj(x, alpha, ml):
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha


def _sgrad(x, alpha, ml):
    n = len(x)
    c = _corr(x)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * (2.0 / n)
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g


def _sfn(x, hn):
    return x.sum() - hn


def _sjac(x, hn):
    return np.ones(len(x))


def _slsqp(h, alpha, ml, mi, ft):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sfn, 'jac': _sjac, 'args': (n * 0.5,)}]
    try:
        r = minimize(_sobj, h, jac=_sgrad, method='SLSQP',
                     bounds=bnds, constraints=cons, args=(alpha, ml),
                     options={'maxiter': mi, 'ftol': ft})
        return _repair(r.x, n)
    except Exception:
        return h


def _binarize(h, n):
    idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb


def _bswap(h, ks):
    n = len(h)
    oi = np.where(h > 0.5)[0]
    zi = np.where(h < 0.5)[0]
    if len(oi) == 0 or len(zi) == 0:
        return -1, -1
    t1 = np.zeros(len(oi))
    ip = oi + ks
    vp = (ip >= 0) & (ip < n)
    t1[vp] -= (1.0 - h[ip[vp]])
    im = oi - ks
    vm = (im >= 0) & (im < n)
    t1[vm] += h[im[vm]]
    t2 = np.zeros(len(zi))
    jp = zi + ks
    vp2 = (jp >= 0) & (jp < n)
    t2[vp2] += (1.0 - h[jp[vp2]])
    jm = zi - ks
    vm2 = (jm >= 0) & (jm < n)
    t2[vm2] -= h[jm[vm2]]
    return int(oi[np.argmin(t1)]), int(zi[np.argmin(t2)])


def _bopt(h, ni, T0, rng, tl, tk):
    n = len(h)
    h = h.copy()
    c = _corr(h)
    c5 = float(c.max())
    bh = h.copy()
    bc5 = c5
    ts = time.time()
    for it in range(ni):
        if time.time() - ts > tl:
            break
        tidx = np.argsort(c)[-tk:]
        ki = int(tidx[rng.randint(tk)]) - (n - 1)
        i, j = _bswap(h, ki)
        if i < 0:
            break
        c_prev = c.copy()
        h[i], h[j] = h[j], h[i]
        c = _corr(h)
        cn = float(c.max())
        T = T0 * (0.01) ** (it / max(ni, 1))
        if cn < c5 or (T > 1e-12 and rng.rand() < np.exp(-(cn - c5) / T)):
            c5 = cn
            if c5 < bc5:
                bc5 = c5
                bh = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
            c = c_prev
    return bh, bc5
