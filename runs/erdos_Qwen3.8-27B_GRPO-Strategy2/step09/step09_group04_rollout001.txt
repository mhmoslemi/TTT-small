    s = h.sum()

def _soft_max_obj(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _soft_max_grad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

            hn = _run_slsqp(h, a, ml, mi, dl2)
            cn = _eval(hn)
            if cn < best_c5:
                best_c5, best_h = cn, hn

    # Phase 3: Projected gradient
    dl3 = t0 + min(920, budget_s * 0.84)
    for a, ml, st, s0, mu in [(5000, 30, 8000, 0.005, 0.85),
                                (50000, 50, 6000, 0.003, 0.90),
                                (500000, 70, 5000, 0.001, 0.92),
                                (5000000, 90, 4000, 0.0005, 0.93)]:
        if time.time() > dl3:
            break
        hn = _run_proj_grad(best_h.copy(), a, ml, st, s0, mu, dl3)
        cn = _eval(hn)
        if cn < best_c5:
            best_c5, best_h = cn, hn

    # Phase 4: Fine 2-point polish
    h2 = best_h.copy()
    c2 = best_c5
    dl4 = deadline - 3
    while time.time() < dl4:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = 2e-7 * rng.randn()
        ni, nj = h2[i] + d, h2[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h2[i], h2[j] = ni, nj
            cn = _eval(h2)
            if cn < c2:
                c2 = cn
                if cn < best_c5:
                    best_c5, best_h = cn, h2.copy()
            else:
                h2[i], h2[j] = ni - d, nj + d

    best_h = _repair_cont(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
