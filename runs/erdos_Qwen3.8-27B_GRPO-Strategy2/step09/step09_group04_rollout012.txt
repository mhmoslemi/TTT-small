import cvxpy as cp

def _lp_active_set(h, active_lags, n):
    """Solve LP: min t s.t. linearized active constraints, sum=h_target, 0<=h<=1"""
    h_new = cp.Variable(n)
    t = cp.Variable()
    target_sum = n / 2.0
    constraints = [t >= 0, cp.sum(h_new) == target_sum, h_new >= 0, h_new <= 1]
    
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    
    for L in active_lags:
        # Compute gradient of c_L at h
        gL = np.zeros(n)
        if L >= 0:
            for i in range(n - L):
                gL[i] += (1.0 - h[i+L])
                gL[i+L] += (-h[i])
        else:
            M = -L
            for i in range(n - M):
                gL[i] += (1.0 - h[i+M])
                gL[i+M] += (-h[i])
        gL *= (2.0 / n)
        
        # Linear constraint: t >= c[L] + gL @ (h_new - h)
        constraints += [t >= c[L] + gL @ (h_new - h)]
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.GLOP)  # or CPLEX, GUROBI
    if h_new.value is not None:
        return h_new.value
    return None

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13: break
    return h

def _sobj(x, alpha, ml, lam):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    pen = lam * (np.sum(x) - n / 2.0) ** 2
    return m + np.log(np.exp(s).sum()) / alpha + pen

def _sobj_grad(x, alpha, ml, lam):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15: continue
        wk = w[ii] * dx
        L = k - (n - 1)

            g[L:] += wk * (1.0 - x[:n - L])

        else:

            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g
