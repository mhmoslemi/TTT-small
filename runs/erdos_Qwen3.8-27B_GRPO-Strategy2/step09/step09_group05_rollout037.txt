import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= t / s
        np.clip(h, 0, 1, out=h)
        s = h.sum()
        if abs(s - t) < 1e-13:
            break
    d = t - h.sum()
    if abs(d) > 1e-13:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, t):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if np.clip(h - m, 0, 1).sum() > t:
            lo = m
        else:
            hi = m
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _og(h, al, ml):
    n = len(h)
    dx = 2.0 / n
    cr = np.correlate(h, 1.0 - h, mode='full')
    c = cr * dx
    m = c.max()
    ix = np.argsort(cr)[-ml:]
    s = np.clip(al * (c[ix] - m), -400, 400)
    sm = s.max()
    s -= sm
    w = np.exp(s)
    ws = w.sum()
    w /= ws
    v = m + (sm + np.log(ws)) / al
    g = np.zeros(n)
    for ii in range(len(ix)):
        if w[ii] < 1e-15:
            continue
        L = ix[ii] - (n - 1)
        wk = w[ii]
        if L >= 0:
            g[:n - L] += wk * (1.0 - h[L:])
            g[L:] -= wk * h[:n - L]
        else:
            M = -L
            g[M:] += wk * (1.0 - h[:n - M])
            g[:n - M] -= wk * h[M:]
    return v, g * dx

def _sf(x, t):
    return x.sum() - t

def _sj(x, t):
    return np.ones(len(x))

def _of(x, al, ml):
    return _og(x, al, ml)[0]

def _oj(x, al, ml):
    return _og(x, al, ml)[1]

def _slsqp(h, al, ml, mi, ft):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (n * 0.5,)}]
    try:
        r = minimize(_of, h, jac=_oj, args=(al, ml), method='SLSQP',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': mi, 'ftol': ft})
        return _repair(r.x, n)
    except Exception:
        return h

def _fista(h, n, ni, a0, a1, ml0, t0, tl):
    x = h.copy()
    y = h.copy()
    pk = 1.0
    bc = _eval(x)
    bx = x.copy()
    ratio = (a1 / a0) ** (1.0 / max(1, ni - 1)) if a1 > a0 else 1.0
    al = a0
    for it in range(ni):
        if time.time() - t0 > tl:
            break
        al *= ratio
        mlc = min(int(ml0 * (1.0 + 2.0 * al / a0)), 2 * n)
        v, g = _og(y, al, mlc)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * pk * pk)) / 2.0
        y = x + ((pk - 1) / tn) * (x - y)
        pk = tn
        st = 1.0 / max(al * 0.01, 1e-6)
        for _ in range(25):
            xn = _proj(y - st * g, n * 0.5)
            vn, _ = _og(xn, al, mlc)
            if vn <= v + 1e-12:
                break
            st *= 0.4
        c = _eval(xn)
        if c < bc:
            bc = c
            bx = xn.copy()
        x = xn
    return _repair(bx, n), bc

def _exact_step(h, n, t0, tl, nsteps=5000):
    x = h.copy()
    bc = _eval(x)
    bx = x.copy()
    step = 0.01
    for _ in range(nsteps):
        if time.time() - t0 > tl:
            break
        cr = np.correlate(x, 1.0 - x, mode='full')
        ki = int(np.argmax(cr))
        L = ki - (n - 1)
        dx = 2.0 / n
        g = np.zeros(n)
        if L >= 0:
            g[:n - L] = (1.0 - x[L:]) * dx
            g[L:] -= x[:n - L] * dx
        else:
            M = -L
            g[M:] = (1.0 - x[:n - M]) * dx
            g[:n - M] -= x[M:] * dx
        taken = False
        for _t in range(20):
            xn = _proj(x - step * g, n * 0.5)
            c = _eval(xn)
            if c < _eval(x) - 1e-15:
                x = xn
                taken = True
                break
            step *= 0.5
        if not taken:
            step = min(step * 1.5, 0.5)
        else:
            step = min(step * 1.05, 0.5)
        c = _eval(x)
        if c < bc:
            bc = c
            bx = x.copy()
    return _repair(bx, n), bc

def _gdelta(h, i, j):
    n = len(h)
    dx = 2.0 / n
    m = np.arange(2 * n - 1)
    idx1i = i + m - (n - 1)
    v1i = (idx1i >= 0) & (idx1i < n)
    t1i = np.where(v1i, 1.0 - h[np.clip(idx1i, 0, n - 1)], 0.0)
    idx2i = i - m + (n - 1)
    v2i = (idx2i >= 0) & (idx2i < n)
    t2i = np.where(v2i, h[np.clip(idx2i, 0, n - 1)], 0.0)
    gi = dx * (t1i - t2i)
    idx1j = j + m - (n - 1)
    v1j = (idx1j >= 0) & (idx1j < n)
    t1j = np.where(v1j, 1.0 - h[np.clip(idx1j, 0, n - 1)], 0.0)
    idx2j = j - m + (n - 1)
    v2j = (idx2j >= 0) & (idx2j < n)
    t2j = np.where(v2j, h[np.clip(idx2j, 0, n - 1)], 0.0)
    gj = dx * (t1j - t2j)
    return gi - gj

def _gold2(h, t0, tl):
    n = len(h)
    x = h.copy()
    bc = _eval(x)
    bx = x.copy()
    gr = 0.618033988749895
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    cont = True
    while cont:
        if time.time() - t0 > tl:
            break
        _, g = _og(x, 5000000, 100)
        gp = g - g.mean()
        if gp.max() - gp.min() < 1e-14:
            break
        i = int(np.argmax(gp))
        j = int(np.argmin(gp))
        if i == j:
            break
        delta = _gdelta(x, i, j)
        dlo = max(-x[i] + 1e-13, x[j] - 1.0 + 1e-13)
        dhi = min(1.0 - x[i] - 1e-13, x[j] - 1e-13)
        if dhi <= dlo:
            break
        a, b = dlo, dhi
        c_ = b - gr * (b - a)
        d_ = a + gr * (b - a)
        fc = np.max(c + c_ * delta)
        fd = np.max(c + d_ * delta)
        for _ in range(50):
            if b - a < 1e-12:
                break
            if fc < fd:
                b = d_
                d_ = c_
                fd = fc
                c_ = b - gr * (b - a)
                fc = np.max(c + c_ * delta)
            else:
                a = c_
                c_ = d_
                fc = fd
                d_ = a + gr * (b - a)
                fd = np.max(c + d_ * delta)
        dopt = 0.5 * (a + b)
        x[i] += dopt
        x[j] -= dopt
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        cm = float(np.max(c))
        if cm < bc:
            bc = cm
            bx = x.copy()
        else:
            x = bx.copy()
            c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
            break
    return _repair(bx, n), bc

def _fine(h, t0, tl, rng):
    n = len(h)
    x = h.copy()

    bx = x.copy()

                bx = x.copy()

                x[i] = ni - d
