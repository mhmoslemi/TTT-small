import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    for _ in range(100):
        s = h.sum()
        if s < 1e-15:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-15:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _smax_obj(x, alpha, ml):
    c = _cvals(x)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smax_grad(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n-L])
            g[:n-L] -= wk * x[L:]
        else:
            M = -L
            g[:n-M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n-M]
    return g

def _proj(h, target):
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _lbfgs_opt(h0, alpha, ml, maxiter):
    n = len(h0)
    # Remove equality constraint by parameterization:
    # Use n-1 free variables, last determined by sum constraint
    # Actually, let's just use penalty + L-BFGS-B
    def obj_penalty(x):
        base = _smax_obj(x, alpha, ml)
        pen = 1e4 * (x.sum() - n * 0.5) ** 2
        return base + pen
    
    def grad_penalty(x):
        base_g = _smax_grad(x, alpha, ml)
        pen_g = 2e4 * (x.sum() - n * 0.5) * np.ones(n)
        return base_g + pen_g
    
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj_penalty, h0, jac=grad_penalty, method='L-BFGS-B',
                   bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-18})
    return _repair(res.x, n)

def _slsqp_opt(h, alpha, ml, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
            'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smax_obj, h, jac=_smax_grad,
                       args=(alpha, ml), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except:
        return h

def _pgrad(h0, alpha, ml, steps, s0):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    mu = 0.8
    step = s0
    cur = _eval(h)
    for _ in range(steps):
        g = _smax_grad(h, alpha, ml)
        v = mu * v + (1 - mu) * g
        ng = np.linalg.norm(v)
        if ng < 1e-18:
            break
        v /= ng
        ok = False
        for _ in range(12):
            h2 = _proj(h - step * v, n * 0.5)
            c2 = _eval(h2)
            if c2 < cur - 1e-18:
                h, cur = h2, c2
                step = min(step * 1.04, s0 * 50)
                ok = True
                break
            step *= 0.35
        if not ok:
            step = min(max(step * 1.3, s0 * 0.5), s0 * 50)
    return _repair(h, n), cur

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    for _ in range(100):
        s = h.sum()
        if s < 1e-15:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-15:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _smax_obj(x, alpha, ml):
    c = _cvals(x)
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smax_grad(x, alpha, ml):

    dx = 2.0 / n
    c = _cvals(x)
    m = c.max()

    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n-L])
            g[:n-L] -= wk * x[L:]
        else:
            M = -L
            g[:n-M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n-M]
    return g

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

        v = mu * v + (1 - mu) * g
        ng = np.linalg.norm(v)
