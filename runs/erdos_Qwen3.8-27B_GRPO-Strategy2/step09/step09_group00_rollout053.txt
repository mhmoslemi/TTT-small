    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if abs(s - S) < 1e-14:
        return h
    for _ in range(50):
        h *= S / max(s, 1e-15)
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - S) < 1e-14:
            break
    d = S - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _grad_max(h, top_k=5):
    """Compute gradient of soft-max of top_k correlations"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top k indices
    top_idx = np.argsort(corr)[::-1][:top_k]
    # Soft weights (use very peaked distribution)
    vals = corr[top_idx]
    mx = vals.max()
    w = np.exp(vals - mx)
    w /= w.sum()
    
    g = np.zeros(n)
    for idx, wk in zip(top_idx, w):
        k = int(idx) - (n - 1)
        ak = abs(k)
        if ak == 0:
            g += wk * (1.0 - 2.0 * h)
        else:
            g[ak:] += wk * (1.0 - h[:n-ak])
            g[:n-ak] -= wk * h[ak:]
    g *= dx
    return g

        mx = top.max()

def _micro_search(h0, rng, deadline):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_k = 5
    act = np.argsort(corr)[::-1][:top_k]
    shifts = (act - (n - 1)).astype(int)
    mx = float(corr[act[0]])
    best_c = mx * dx
    best_h = h.copy()
    t0 = time.time()
    it = 0
    while time.time() < deadline:
        if it % 40 == 0:
            corr = np.correlate(h, 1.0 - h, mode='full')
            act = np.argsort(corr)[::-1][:top_k]
            shifts = (act - (n - 1)).astype(int)
            mx = float(corr[act[0]])
            if mx * dx < best_c:
                best_c = mx * dx
                best_h = h.copy()
        frac = min(1.0, (time.time() - t0) / max(1e-9, deadline - t0))
        base = max(1e-8, 4e-5 * (1e-6) ** frac)
        improved = False
        for ai in range(len(shifts)):
            k = shifts[ai]
            ak = abs(k)
            if ak == 0:
                continue
            valid = np.arange(0, n - ak) if k >= 0 else np.arange(ak, n)
            if k >= 0:
                contrib = h[valid] * (1.0 - h[valid + ak])
            else:
                contrib = h[valid] * (1.0 - h[valid - ak])
            top_p = np.argsort(contrib)[::-1][:6]
            for pi in top_p:
                i = int(valid[pi]) if k >= 0 else int(valid[pi])
                j = i + ak if k >= 0 else i - ak
                d = base * (0.3 + 1.7 * rng.random())
                if rng.random() < 0.5:
                    d = -d
                if h[i] + d < 0 or h[i] + d > 1 or h[j] - d < 0 or h[j] - d > 1:
                    continue
                h[i] += d
                h[j] -= d
                mv = float(np.max(_corr_at_shifts(h, shifts)))
                if mv < mx - 1e-15:
                    mx = mv
                    improved = True
                    if mx * dx < best_c:
                        best_c = mx * dx
                        best_h = h.copy()
                    break
                h[i] -= d
                h[j] += d
        if not improved and rng.random() < 0.3:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i != j:
                d = (rng.random() - 0.5) * 2.0 * base
                if h[i] + d >= 0 and h[i] + d <= 1 and h[j] - d >= 0 and h[j] - d <= 1:
                    h[i] += d
                    h[j] -= d
                    mv = float(np.max(_corr_at_shifts(h, shifts)))
                    if mv < mx - 1e-15:
                        mx = mv
                        if mx * dx < best_c:
                            best_c = mx * dx
                            best_h = h.copy()
                    else:
                        h[i] -= d
                        h[j] += d
        it += 1
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        n = 560
        x = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        h0 = 0.5 + 0.35 * np.sin(3.0 * np.pi * x) + 0.15 * np.sin(7.0 * np.pi * x + 0.5)
        h0 = _proj(h0, n * 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)
    best = h0.copy()
    best_c = _c5(best)
    for hh in [1.0 - h0, h0[::-1], 0.5 + 0.5 * (h0 - h0[::-1])]:
        h2 = _proj(hh.copy(), n0 * 0.5)
        c = _c5(h2)
        if c < best_c:
            best = h2
            best_c = c
    # Phase 1: FISTA moderate alpha (100s)
    end1 = t0 + 100.0
    if end1 > time.time():
        hcur = best.copy()
        for a in [80, 400, 2000, 10000]:
            if time.time() >= end1 - 2:
                break
            tt = (end1 - time.time() - 1) / 4
            hf, bh, bc = _fista_top(hcur, a, min(end1 - 2, time.time() + tt), 5)
            cf = _c5(hf)
            if bc < best_c:
                best = bh
                best_c = bc
            if cf < best_c:
                best = hf
                best_c = cf
            hcur = hf if cf <= bc else bh
    # Phase 2: Multi-resolution (80s)
    end2 = t0 + 180.0
    if end2 > time.time():
        for m in [280, 512, 640, n0]:
            if time.time() >= end2 - 5:
                break
            slot = min(18.0, end2 - time.time() - 3)
            if slot < 4:
                break
            hm = _resample(best, m)
            hcur = hm.copy()
            for a in [500, 5000, 50000]:
                if time.time() >= end2 - 3:
                    break
                tt_p = (end2 - time.time() - 2) / 3
                hf, bh, bc = _fista_top(hcur, a, min(end2 - 3, time.time() + tt_p), 4)
                cf = _c5(hf)
                if bc < best_c:
                    hb = _resample(_repair(bh, m), n0) if m != n0 else bh
                    best = hb
                    best_c = _c5(hb)
                if cf < best_c:
                    hb = _resample(_repair(hf, m), n0) if m != n0 else hf
                    best = hb
                    best_c = _c5(hb)
                hcur = _repair(hf if cf <= bc else bh, m)
    # Phase 3: Very high alpha FISTA (150s)
    end3 = t0 + 330.0
    if end3 > time.time():
        hcur = best.copy()
        for a in [5000, 50000, 500000, 5000000]:
            if time.time() >= end3 - 5:
                break
            tt_p = (end3 - time.time() - 3) / 4
            hf, bh, bc = _fista_top(hcur, a, min(end3 - 5, time.time() + tt_p), 3)
            cf = _c5(hf)
            if bc < best_c:
                best = bh
                best_c = bc
            if cf < best_c:
                best = hf
                best_c = cf
            hcur = hf if cf <= bc else bh
    # Phase 4: Fast micro-search (300s)
    end4 = t0 + 630.0
    if end4 > time.time():
        best, best_c = _micro_search(best, rng, end4)
    # Phase 5: FISTA + micro alternation (250s)
    end5 = t0 + 880.0
    if end5 > time.time():
        hcur = best.copy()
        for a in [100000, 1000000, 10000000]:
            if time.time() >= end5 - 60:
                break
            tt_p = min(60.0, (end5 - 60 - time.time()) / 3)
            hf, bh, bc = _fista_top(hcur, a, min(end5 - 60, time.time() + tt_p), 3)
            cf = _c5(hf)
            if bc < best_c:
                best = bh
                best_c = bc
            if cf < best_c:
                best = hf
                best_c = cf
            hcur = hf if cf <= bc else bh
        if time.time() < end5 - 10:
            best, best_c = _micro_search(best, rng, end5)
    # Phase 6: Final polish (remaining)
    end6 = t0 + budget_s - 3.0
    if end6 > time.time():
        best, best_c = _micro_search(best, rng, end6)
    # Final variants
    for hh in [1.0 - best, best[::-1], 0.5 + 0.5 * (best - best[::-1]), 0.5 + 0.5 * (best + best[::-1] - 1.0)]:
        h2 = _proj(hh.copy(), len(hh) * 0.5)
        c = _c5(h2)
        if c < best_c:
            best = h2
            best_c = c
    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
