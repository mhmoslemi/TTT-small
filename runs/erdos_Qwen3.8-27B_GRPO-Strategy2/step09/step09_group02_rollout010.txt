from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    for _ in range(200):
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_dx(h):

def _smooth_obj(x, alpha, m_lags):
    n = len(x)
    c = _corr_dx(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = _corr_dx(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:

    hl = _repair(hl, nl)
    for a in [100000, 1000000, 10000000]:
        hl = _slsqp(hl, a, 40, 400, 1e-14)
    hn = np.zeros(560)
    hn[0::2] = hl
    hn[1::2] = hl
    hn = _repair(hn, 560)
    for a in [100000, 1000000]:
        hn = _slsqp(hn, a, 50, 300, 1e-14)
    return hn

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    bh = h.copy()
    bc = _eval(bh)

    # Phase 1: SLSQP sweep
    alphas = [30000, 200000, 1500000, 12000000, 80000000]
    for a in alphas:
        if _el() > 160:
            break
        ml = min(120, max(15, int(9 * np.log10(a) - 18)))
        hn = _slsqp(bh.copy(), a, ml, 500, 1e-16)
        c = _eval(hn)
        if c < bc:
            bc = c
            bh = hn

    # Phase 2: Simulated annealing
    if _el() < budget_s - 350:
        hn, c = _anneal(bh.copy(), 300000, 55, 2500, 0.0008, 1e-9, rng)
        if c < bc:
            bc = c
            bh = hn

    # Phase 3: SLSQP refinement
    if _el() < budget_s - 250:
        for a in [200000000, 2000000000, 20000000000]:
            if _el() > budget_s - 230:
                break
            hn = _slsqp(bh.copy(), a, 100, 400, 1e-17)
            c = _eval(hn)
            if c < bc:
                bc = c
                bh = hn

    # Phase 4: Try low-res upsample alternative
    if _el() < budget_s - 200:
        try:
            hl = _lowres_upsample(seed)
            cl = _eval(hl)
            if cl < bc + 0.005:
                hn = _slsqp(hl, 5000000, 80, 400, 1e-16)
                c = _eval(hn)
                if c < bc:
                    bc = c
                    bh = hn
        except Exception:
            pass

    # Phase 5: Informed local search (gradient + random)
    if _el() < budget_s - 120:
        rem = budget_s - 120 - _el()
        nat = int(rem * 200)
        bh, bc = _informed_search(bh, nat, rng)

    # Phase 6: Second annealing pass (shorter, cooler)
    if _el() < budget_s - 80:
        hn, c = _anneal(bh.copy(), 800000, 70, 1500, 0.0002, 1e-10, rng)
        if c < bc:
            bc = c
            bh = hn

    # Phase 7: Final SLSQP
    if _el() < budget_s - 40:
        for a in [50000000000, 500000000000]:
            if _el() > budget_s - 35:
                break
            hn = _slsqp(bh.copy(), a, 120, 300, 1e-18)
            c = _eval(hn)
            if c < bc:
                bc = c
                bh = hn

    # Phase 8: Fine polish
    if _el() < budget_s - 5:
        rem = budget_s - 5 - _el()
        nat = int(rem * 900)
        bh, bc = _fpolish(bh, nat, rng)

    bh = _repair(bh, len(bh))
    bc = _eval(bh)
    return (bh, bc, len(bh))
