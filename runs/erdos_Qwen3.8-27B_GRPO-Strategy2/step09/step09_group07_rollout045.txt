    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0

        s = h.sum()

    diff = target - h.sum()

        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:

                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a

    cmax_val = c.max()
    dx = 2.0 / n
    nm1 = n - 1
    
    idx = np.argsort(c)[-top_k:]
    c_top = c[idx]
    weights = np.exp(50 * (c_top - cmax_val))  # softmax-like
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for ii in range(len(idx)):
        k = int(idx[ii])
        L = k - nm1
        w = weights[ii] * dx
        if L >= 0:
            # grad[j] += w * (1-h[j+L]) for j in [0, n-1-L]
            # grad[j] -= w * h[j-L] for j in [L, n-1]
            grad[:n-L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n-L]

            # grad[j] += w * (1-h[j-M]) for j in [M, n-1]
            # grad[j] -= w * h[j+M] for j in [0, n-1-M]
            grad[M:] += w * (1.0 - h[:n-M])
            grad[:n-M] -= w * h[M:]
    
    return grad, cmax_val

def _subgrad_descent(h, n, t0, budget_frac, top_k_list, base_lr):
    """Polish with projected subgradient + line search"""
    best_h = h.copy()
    best_c5 = _eval(best_h)
    deadline = t0 + budget_frac
    
    step_count = 0
    for top_k in top_k_list:
        if time.time() > deadline:
            break
        for _ in range(200):
            if time.time() > deadline:
                break
            c = _corr(best_h)
            cmax_val = c.max()
            dx = 2.0 / n
            nm1 = n - 1
            
            idx = np.argsort(c)[-top_k:]
            c_top = c[idx]
            # Soft weights emphasizing the max
            scale = 100.0
            weights = np.exp(scale * (c_top - cmax_val))
            wsum = weights.sum()
            if wsum < 1e-30:
                weights = np.ones(top_k) / top_k
            else:
                weights /= wsum

                else:
                    M = -L
                    grad[M:] += w * (1.0 - best_h[:n-M])
                    grad[:n-M] -= w * best_h[M:]
            
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            
            # Polyak-like step size
            lr = base_lr * cmax_val / (gnorm * gnorm + 1e-10)
            lr = min(lr, 0.1)

                    best_h = trial
                    if c5_trial < best_c5:
                        best_c5 = c5_trial
                    step_count += 1
                    break
            else:
                # No improvement, try smaller
                pass
    
    return best_h, best_c5, step_count

def _local_search(h, n, rng, n_iter):
    c5c = _eval(h)
    best_h = h.copy()
    best_c5 = c5c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = 5e-5 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:

import numpy as np
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _cd_pass(h, rng):
    n = len(h)
    c = _corr(h)
    cmax_size = 2 * n - 1
    jlist = np.arange(n)
    rng.shuffle(jlist)
    nm1 = n - 1
    for j in jlist:
        j = int(j)
        beta = np.zeros(cmax_size)
        k1_lo = j
        k1_hi = min(cmax_size - 1, j + nm1)
        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax_size - 1, cmax_size - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
        beta[k2_lo:k2_hi + 1] -= h[j + lags2]
        g = c - beta * h[j]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(25):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(g + beta * t1) <= np.max(g + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-14:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

    n = len(h)

    cmax_size = 2 * n - 1

        beta = np.zeros(cmax_size)
        k1_lo = j
        k1_hi = min(cmax_size - 1, j + nm1)
        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax_size - 1, cmax_size - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
        beta[k2_lo:k2_hi + 1] -= h[j + lags2]

def _compute_subgrad(h, n, top_k, beta_scale):
    c = _corr(h)
    cmax_val = c.max()
    dx = 2.0 / n
    nm1 = n - 1
    idx = np.argsort(c)[-top_k:]
    c_top = c[idx]
    weights = np.exp(beta_scale * (c_top - cmax_val))
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(top_k) / top_k
    else:
        weights /= wsum
    grad = np.zeros(n)
    for ii in range(len(idx)):
        k = int(idx[ii])
        L = k - nm1
        w = weights[ii] * dx
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
        else:
            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return grad, c

                    best_h = trial

                    break
