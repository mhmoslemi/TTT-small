def _corr(h):

def _eval(h):

def _repair(h, n):

    target = 0.5 * n

    for _ in range(45):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:

        else:

def _top_weights(c, alpha, topm):
    nc = len(c)
    topm = int(min(max(1, int(topm)), nc))
    if topm == nc:

    else:

    v = c[idx]

        x = alpha * (v - m)
        x -= x.max()
        w = np.exp(x)
        s = w.sum()
        if s <= 0.0 or not np.isfinite(s):

def _grad_weighted(h, idx, w, dx):
    n = len(h)

            M = -L
            g[:n - M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n - M]

            else:
                dlt = m - bm
                if dlt < 0.006 and rng.rand() < math.exp(-max(0.0, dlt) / temp):
                    accept = True
            if accept:
                bm, bts, c = m, ts, cc
                improved = True
                age = 100
                temp = max(1e-6, temp * 0.997)
            else:
                b[i] = oi
                b[j] = oj
        temp = max(1e-6, temp * 0.9999)
        if not improved and temp < 1e-4:
            break
    return b, bm


def _block_sa(h, deadline, rng, size, amp, temp):
    n = len(h)
    target = 0.5 * n
    h = np.array(h, dtype=float).copy()
    cur = _eval(h)
    best = h.copy()
    bc = cur
    g = None
    age = 100
    while time.time() < deadline:
        if age > 25:
            c = _corr(h)
            idx, w = _top_weights(c, 800, 80)
            g = _grad_weighted(h, idx, w, 2.0 / n)
            age = 0
        age += 1
        supp = rng.choice(n, size, replace=False)
        if g is not None and size > 4:
            top = np.argpartition(np.abs(g), -size // 2)[-size // 2:]
            supp = np.unique(np.concatenate((top, supp[:size - size // 2])))
        d = amp * rng.randn(len(supp))
        delta = np.zeros(n)
        delta[supp] = d
        delta -= delta.mean()
        cand = _proj(h + delta, target)
        cn = _eval(cand)
        if cn < cur or (temp > 1e-10 and rng.rand() < math.exp(-max(0.0, cn - cur) / temp)):
            h = cand
            cur = cn
            if cn < bc - 1e-14:
                bc = cn
                best = h.copy()
        temp = max(1e-10, temp * 0.999)
        amp = max(1e-5, amp * 0.9995)
    return best, bc


def _pair_polish(h, deadline, rng, start_scale, max_scale):
    n = len(h)
    h = np.array(h, dtype=float).copy()
    cbest = _eval(h)
    scale = start_scale
    g = None
    age = 100
    bad = 0
    grid = np.array([-0.95, -0.7, -0.45, -0.25, -0.12,
                     -0.05, -0.02, 0.02, 0.05,
                     0.12, 0.25, 0.45, 0.7, 0.95])
    extras = np.array([0.15, 0.35, 0.65, 0.85])
    while time.time() < deadline:
        if age > 20:
            c = _corr(h)
            idx, w = _top_weights(c, 1500, 80)
            g = _grad_weighted(h, idx, w, 2.0 / n)
            age = 0
        age += 1
        ok = False
        for _ in range(35):
            if time.time() >= deadline:
                break
            if g is not None and rng.rand() < 0.8:
                order = np.argsort(-g)
                hi = order[:16]
                lo = order[-16:]
                i = hi[rng.randint(len(hi))]
                j = lo[rng.randint(len(lo))]
            else:
                i = rng.randint(n)
                j = rng.randint(n)
            if i == j:
                continue
            hi0, hj0 = h[i], h[j]
            lo_d = max(-hi0, hj0 - 1.0)
            up_d = min(1.0 - hi0, hj0)
            span = up_d - lo_d
            if span < 1e-12:
                continue
            base = min(scale, 0.9 * span)
            ds = grid * base
            if span > 4.0 * base:
                ds = np.concatenate((ds, lo_d + extras * span))
            best_d = 0.0
            best_cv = cbest
            for d in ds:
                if d < lo_d or d > up_d:
                    continue
                ni = hi0 + d
                nj = hj0 - d
                if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                    continue
                h[i] = ni
                h[j] = nj
                cv = _eval(h)
                if cv < best_cv:
                    best_cv = cv
                    best_d = d
            h[i] = hi0
            h[j] = hj0
            if best_d != 0.0:
                h[i] = hi0 + best_d
                h[j] = hj0 - best_d
                cv = _eval(h)
                if cv < cbest - 1e-14:
                    cbest = cv
                    ok = True
                    age = 100
                    scale = min(scale * 1.25, max_scale)
                else:
                    h[i] = hi0
                    h[j] = hj0
        if ok:
            bad = 0
        else:
            bad += 1
            scale = max(1e-7, scale * 0.6)
            if bad > 35:
                break
    return h, cbest


def _local_refine(h, deadline, rng):
    n = len(h)
    best = np.array(h, dtype=float).copy()
    bc = _eval(best)
    t = time.time()
    rem = max(0.0, deadline - t)

    gdl = t + min(rem * 0.38, 200.0)
    gs = [(800, 128, 0.06, 0.80),
          (4000, 96, 0.03, 0.85),
          (20000, 80, 0.015, 0.90),
          (100000, 64, 0.006, 0.93),
          (500000, 48, 0.002, 0.95)]
    if gdl > t + 3.0:
        chunk = max(1.0, (gdl - t) / max(1, len(gs)))
        for a, tm, s0, mu in gs:
            if time.time() >= gdl:
                break
            h2, c2 = _proj_grad(best, a, tm, 600, s0, mu,
                                min(gdl, time.time() + chunk + 2.0))
            if c2 < bc - 1e-14:
                best, bc = h2, c2

    rem = deadline - time.time()
    if rem > 5.0:
        pdl = time.time() + min(rem * 0.30, 120.0)
        h2, c2 = _pair_polish(best, pdl, rng, 5e-4, 0.12)
        if c2 < bc - 1e-14:
            best, bc = h2, c2

    rem = deadline - time.time()
    if rem > 5.0:
        bdl = time.time() + min(rem * 0.25, 100.0)
        h2, c2 = _block_sa(best, bdl, rng, 14, 0.04, 0.005)
        if c2 < bc - 1e-14:
            best, bc = h2, c2

    rem = deadline - time.time()
    if rem > 250.0 and n < 900:
        h2 = _slsqp(best, 50000, 80, 12)
        h2 = _repair(h2, n)
        c2 = _eval(h2)
        if c2 < bc - 1e-14:
            best, bc = h2, c2

    rem = deadline - time.time()
    if rem > 100.0 and n % 2 == 0:
        bb = _make_binary(best, rng)
        bdl = time.time() + min(rem * 0.35, 120.0)
        bb, bbm = _binary_local(bb, bdl, rng)
        if bbm < bc - 1e-14:
            best, bc = bb, bbm

    rem = deadline - time.time()
    if rem > 5.0:
        h2, c2 = _pair_polish(best, deadline - 1, rng, 1e-4, 0.05)
        if c2 < bc - 1e-14:
            best, bc = h2, c2

    best = _repair(best, n)
    bc = _eval(best)
    return best, bc


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s)
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float).copy()
    else:
        base = (np.arange(560) % 2 == 0).astype(float)
    n = len(base)
    base = _repair(base, n)

    best = base.copy()
    bc = _eval(best)

    rem = max(0.0, deadline - time.time())
    pdl = time.time() + max(0.0, rem * 0.58)
    best, bc = _local_refine(best, pdl, rng)

    for kind in range(4):
        if time.time() > deadline - 80.0:
            break
        if kind == 0:
            cand = _symcomp(best)
        elif kind == 1 and n % 2 == 0:
            cand = _make_binary(best, rng)
        elif kind == 2:
            cand = best[::-1].copy()
        else:
            continue
        cand = _repair(cand, n)
        c0 = _eval(cand)
        if c0 <= bc + 0.01:
            qdl = time.time() + min(70.0, max(20.0, (deadline - time.time()) * 0.25))
            cand, c2 = _local_refine(cand, qdl, rng)
            if c2 < bc - 1e-14:
                best, bc = cand, c2

    if time.time() < deadline - 10.0:
        best, bc = _local_refine(best, deadline - 1, rng)

    best = _repair(best, n)
    bc = _eval(best)
    return best, float(bc), n
