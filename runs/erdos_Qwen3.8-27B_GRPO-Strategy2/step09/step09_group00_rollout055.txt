    if s == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    if shift > 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    else:
        g[s:] += (1.0 - h[:n-s])
        g[:n-s] -= h[s:]
    return g

def _active_shifts(corr, n, ktop=5):
    """Get the ktop shifts with highest overlap"""
    idx = np.argsort(corr)[::-1][:ktop]
    return idx - (n - 1)  # convert to shift values

def _nesterov_subgrad(h0, n, max_time, topk=5, alpha0=0.05):
    """Nesterov-accelerated projected subgradient on true max"""
    S = n * 0.5
    t0 = time.time()
    h = h0.copy()
    y = h.copy()
    h_prev = h.copy()
    t = 1.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stale = 0
    
    for it in range(100000):
        if time.time() - t0 > max_time:
            break
        if it > 0:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / t_new) * (h - h_prev)
            t = t_new
            y = _proj(y, S)
            corr = np.correlate(y, 1.0 - y, mode='full')
        
        # Get active shifts
        act = _active_shifts(corr, n, topk)
        # Average subgradient over active set
        g = np.zeros(n)
        for s in act:
            g += _sg(y, s)
        g /= len(act)
        
        # Step size
        alpha = alpha0 / np.sqrt(it + 1)
        alpha = min(alpha, 0.5)  # cap
        
        # Projected step
        h_new = _proj(y - alpha * g, S)
        
        # Check improvement
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c_new = float(corr_new.max()) * (2.0 / n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            stale = 0
        elif c_new < float(corr.max()) * (2.0 / n):
            stale = 0
        else:
            stale += 1
        
        h_prev = h
        h = h_new
        corr = corr_new
        
        if stale > 500:
            break
    
    return h, best_h, best_c

def _repair(h, n):
    S = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        s = h.sum()
        if abs(s - S) < 1e-14:
            break

        np.clip(h, 0.0, 1.0, out=h)
    d = S - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    for _ in range(50):
        x += (S - x.sum()) / x.size
        np.clip(x, 0.0, 1.0, out=x)
        if abs(x.sum() - S) < 1e-13:
            break
    d = S - x.sum()
    if abs(d) > 1e-12:
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c > 0:
            x[m] += d / c
        np.clip(x, 0.0, 1.0, out=x)
    return x

def _pn(h):
    """Compute all overlaps P[k] for positive shifts k=0..n-1"""
    n = len(h)
    # Use FFT for speed
    N = 1
    m_len = 2 * n - 1
    while N < m_len:
        N <<= 1
    H = np.fft.rfft(h, N)
    # autocorrelation of h
    ac_h = np.fft.irfft(H * np.conj(H), N).real[:n]
    # overlap at shift 0: sum(h) - sum(h^2) = n/2 - sum(h^2)
    # overlap at shift k>0: sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
    #                      = sum_{i=0}^{n-1-k} h[i] - sum h[i]*h[i+k]
    # The first term varies with k
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n)
    P[0] = total - ac_h[0]  # sum(h) - sum(h^2)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac_h[idx]
        # Also need "negative" shifts: sum h[i+k]*(1-h[i]) for k>0
        # = sum_{i=0}^{n-1-k} h[i+k] - sum h[i+k]*h[i]
        # = (total - pref[k]) - ac_h[k]
        Nv = np.empty(n)
        Nv[0] = 0.0
        Nv[idx] = (total - pref[idx]) - ac_h[idx]
    else:
        Nv = np.zeros(1)
    return P, Nv

        score = 0.0
        for ai in active_idx:

            # Nv[k] = sum_{i=k}^{n-1} h[i]*(1-h[i-k]) for shift -k

            if shift == 0:

                delta = (1-2*h[i]) - (1-2*h[j])  # = 2*(h[j]-h[i])

                di = 0.0
                if i < n - shift: di += (1.0 - h[i+shift])
                if i >= shift: di -= h[i-shift]
                dj = 0.0

                if j >= shift: dj -= h[j-shift]
                delta = di - dj
            else:

                s2 = -shift
                di = 0.0
                if i >= s2: di += (1.0 - h[i-s2])
                if i < n - s2: di -= h[i+s2]
                dj = 0.0
                if j >= s2: dj += (1.0 - h[j-s2])

            sgn = -1.0
        else:
            sgn = 1.0 if rng.random() < 0.5 else -1.0

        if avail < 1e-12:

            continue
        
        # Step size: small, with occasional large jumps
        step = rng.uniform(0.001, 0.01)
        if rng.random() < 0.01:
            step *= 10.0
        d = sgn * min(step, avail)
        
        h[i] += d
        h[j] -= d
        
        # Quick check: evaluate only active overlaps
        improved = True
        for ai in active_idx[:3]:  # check top 3 only
            shift = int(ai)
            if shift == 0:
                ov = float(np.sum(h * (1-h)))

                break

            it += 1
        else:

            h[j] += d
            it += 1

    P, Nv = _pn(h)
    all_ov = np.concatenate([P, Nv[1:]])
    c = float(all_ov.max()) * dx
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c

            c = mx * dx
            if c < best_c - 1e-14:
                best_c = c; best_h = h.copy()
            active = np.argsort(allv)[::-1][:topk]
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            it += 1; continue
        score = 0.0
        for ai in active:
            s = int(ai)
            di = 0.0; dj = 0.0
            if s == 0:
                di = 1.0 - 2.0 * h[i]; dj = 1.0 - 2.0 * h[j]
            elif s > 0:
                if i < n - s: di += 1.0 - h[i+s]
                if i >= s: di -= h[i-s]
                if j < n - s: dj += 1.0 - h[j+s]
                if j >= s: dj -= h[j-s]
            else:
                s2 = -s
                if i >= s2: di += 1.0 - h[i-s2]
                if i < n - s2: di -= h[i+s2]
                if j >= s2: dj += 1.0 - h[j-s2]
                if j < n - s2: dj -= h[j+s2]
            score += di - dj
        if score < 0:
            sgn = 1.0
        elif score > 0:
            sgn = -1.0
        else:
            sgn = 1.0 if rng.random() < 0.5 else -1.0
        avail = min(1.0 - h[i], h[j]) if sgn > 0 else min(h[i], 1.0 - h[j])
        if avail < 1e-12:
            it += 1; continue
        frac = min(1.0, (time.time() - t0) / max(1e-9, deadline - t0))
        base = 0.005 * (0.01) ** frac
        d = sgn * min(avail, base * (0.5 + rng.random() * 2.0))
        h[i] += d; h[j] -= d
        ov0 = 0.0
        for ai in active[:3]:
            s = int(ai)
            if s == 0:
                ov0 = max(ov0, float(np.dot(h, 1.0-h)))
            elif s > 0:
                ov0 = max(ov0, float(np.dot(h[:n-s], 1.0-h[s:])))
            else:
                s2 = -s
                ov0 = max(ov0, float(np.dot(h[s2:], 1.0-h[:n-s2])))
        if ov0 <= mx + 0.002:
            pass
        else:
            h[i] -= d; h[j] += d
        it += 1
    P, Nv = _pn(h)
    allv = np.concatenate([P, Nv[1:]])
    c = float(allv.max()) * dx
    if c < best_c:
        best_c = c; best_h = h.copy()
    return best_h, best_c

def _sa_search(h0, rng, deadline):
    n = len(h0)
    dx = 2.0 / n
    h = _repair(h0.copy(), n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(corr.max())
    best_c = mx * dx
    best_h = h.copy()
    t0 = time.time()
    total_t = max(1.0, deadline - t0)
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t0) / total_t)
        T = 0.003 * (0.001) ** frac
        ka = int(np.argmax(corr)) - (n - 1)
        if rng.random() < 0.7:
            k = abs(ka)
            if 0 < k < n:
                if ka > 0:
                    valid = (h[:n-k] > 0.01) & (h[k:] < 0.99)
                else:
                    valid = (h[k:] > 0.01) & (h[:n-k] < 0.99)
                if valid.sum() > 2:
                    pos = np.nonzero(valid)[0]
                    i = int(pos[rng.randint(len(pos))])
                    j = i + k if ka > 0 else i - k
                    if 0 <= j < n:
                        d = (rng.random() - 0.5) * 0.05
                        if h[i]+d < 0: d = -h[i]
                        if h[i]+d > 1: d = 1.0-h[i]
                        if h[j]-d < 0: d = h[j]
                        if h[j]-d > 1: d = h[j]-1.0
                        if abs(d) > 1e-10:
                            h[i]+=d; h[j]-=d
                            c2 = np.correlate(h, 1.0-h, mode='full')
                            mx2 = float(c2.max())
                            delta = mx2 - mx
                            if delta < -1e-14 or (delta > 0 and rng.random() < np.exp(-delta/max(T,1e-15))):
                                mx=mx2; corr=c2
                                if mx*dx < best_c: best_c=mx*dx; best_h=h.copy()
                            else:
                                h[i]-=d; h[j]+=d
                    it = 0
                    continue
        i = rng.randint(n); j = rng.randint(n)
        if i == j: continue
        d = (rng.random()-0.5) * 0.04
        if h[i]+d<0: d=-h[i]
        if h[i]+d>1: d=1.0-h[i]
        if h[j]-d<0: d=h[j]
        if h[j]-d>1: d=h[j]-1.0
        if abs(d)<1e-10: continue
        h[i]+=d; h[j]-=d
        c2 = np.correlate(h, 1.0-h, mode='full')
        mx2 = float(c2.max())
        delta = mx2-mx
        if delta<-1e-14 or (delta>0 and rng.random()<np.exp(-delta/max(T,1e-15))):
            mx=mx2; corr=c2
            if mx*dx<best_c: best_c=mx*dx; best_h=h.copy()
        else:
            h[i]-=d; h[j]+=d
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    if n % m == 0:
        out = h.reshape(m, n//m).mean(axis=1)
    elif m % n == 0:
        out = np.tile(h, m//n)
    else:
        x0 = (np.arange(n)+0.5)*(2.0/n)
        x1 = (np.arange(m)+0.5)*(2.0/m)
        out = np.interp(x1, x0, h)
    return _proj(out, m*0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)
    candidates = [h0]
    for hh in [1.0-h0.copy(), h0[::-1].copy(),
               0.5+0.5*(h0-h0[::-1]), 0.5+0.5*(h0+h0[::-1]-1.0)]:
        h2 = _proj(hh, n0*0.5)
        c = _c5(h2)
        if c < _c5(candidates[0]) or len(candidates) < 3:
            candidates.append(h2)
        if time.time()-t0 > 15:
            break
    if time.time()-t0 < 30:
        x = np.linspace(0, 2, n0, endpoint=False)
        for f in [2.0, 3.0, 4.0, 7.0, 11.0]:
            phase = rng.uniform(0, 2*np.pi)
            hh = 0.5 + 0.45*np.sin(2*np.pi*f*x + phase)
            hh = _proj(hh, n0*0.5)
            c = _c5(hh)
            if c < _c5(candidates[0]):
                candidates.append(hh)
            if time.time()-t0 > 30:
                break
    best = min(candidates, key=_c5)
    best_c = _c5(best)
    end1 = t0 + min(250.0, 0.25*budget_s)
    if end1 > time.time():
        for tk in [3, 5, 8]:
            if time.time() >= end1 - 5:
                break
            slot = min(70.0, end1 - time.time() - 2)
            if slot < 5:
                break
            hf, bh, bc = _nesterov_tophk(best, slot, topk=tk, alpha0=0.06)
            cf = _c5(hf)
            if bc < best_c:
                best = bh; best_c = bc
            if cf < best_c:
                best = hf; best_c = cf
            best = _repair(best, len(best))
    end2 = t0 + min(400.0, 0.40*budget_s)
    if end2 > time.time():
        best, best_c = _local_search(best, rng, end2, topk=5)
    end3 = t0 + min(600.0, 0.60*budget_s)
    if end3 > time.time():
        best, best_c = _nesterov_tophk(best, min(150.0, (end3-time.time())*0.5), topk=4, alpha0=0.02)
        best = _repair(best, len(best))
        best_c = _c5(best)
        if end3 > time.time():
            best, best_c = _local_search(best, rng, end3, topk=4)
    end4 = t0 + min(850.0, 0.85*budget_s)
    if end4 > time.time():
        best, best_c = _sa_search(best, rng, end4)
    end5 = t0 + budget_s - 30.0
    if end5 > time.time():
        best, best_c = _local_search(best, rng, end5, topk=3)
    final_candidates = [best]
    for hh in [1.0-best.copy(), best[::-1].copy()]:
        h2 = _proj(hh, len(best)*0.5)
        final_candidates.append(h2)
    best = min(final_candidates, key=_c5)
    best_c = _c5(best)
    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
