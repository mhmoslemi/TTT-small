                if cn < cur - 1e-16:
                    h, cur, taken = hn, cn, True
                    gscale = min(gscale * 1.05, 0.02)
                    break
            if not taken:
                gscale = max(gscale * 0.7, 1e-5)
        if cur < bc:
            bc, bh = cur, h.copy()
        if it > 0 and it % 5000 == 0:
            h = _repair(bh + 1e-4 * rng.randn(n), n)
            cur = _eval(h)
    return bh, bc

def _fine_polish(h, niter, rng):
    n = len(h)
    h2 = h.copy()
    c2 = _eval(h2)
    for _ in range(niter):
        mode = rng.rand()
        if mode < 0.6:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 4)) % n
        elif mode < 0.85:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 10)) % n
        else:
            i, j = rng.randint(n, size=2)
            if i == j:
                continue
        d = 3e-7 * rng.randn()
        ni, nj = h2[i] + d, h2[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h2[i], h2[j] = ni, nj
            cn = _eval(h2)
            if cn < c2:
                c2 = cn
            else:
                h2[i], h2[j] = ni - d, nj + d
    return h2, c2

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h0.copy(), _eval(h0)

    def el():
        return time.time() - t0

    # Phase 1: SLSQP with increasing alpha
    params1 = [(1e5, 25, 1500), (1e6, 35, 2000), (1e7, 45, 2000),
               (5e7, 60, 2500), (1e8, 70, 2500), (5e8, 80, 3000), (1e9, 90, 3000)]
    for alpha, ml, miter in params1:
        if el() > 220:
            break
        hn = _do_slsqp(best_h, alpha, ml, miter, miter * 2)
        cn = _eval(hn)
        if cn < best_c5:
            best_c5, best_h = cn, hn

    # Phase 2: Local search with greedy pairs
    if el() < budget_s - 400:
        best_h, best_c5 = _local_search(best_h, 40000, rng, 1e7, 50, 8)

    # Phase 3: Multi-restart with perturbation
    if el() < budget_s - 250:
        for rs in [0.015, 0.008, 0.004, 0.002]:
            if el() > budget_s - 200:
                break
            hp = _repair(best_h + rs * rng.randn(n), n)
            hp = _do_slsqp(hp, 1e7, 50, 1500, 3000)
            cn = _eval(hp)
            if cn < best_c5:
                best_c5, best_h = cn, hp
            best_h, best_c5 = _local_search(best_h, 15000, rng, 1e7, 50, 8)

    # Phase 4: Binary-structure exploration
    if el() < budget_s - 150:
        for _ in range(4):
            if el() > budget_s - 120:
                break
            hb = np.zeros(n)
            hb[rng.choice(n, size=n // 2, replace=False)] = 1.0
            hb = _do_slsqp(hb, 1e6, 40, 2000, 4000)
            cn = _eval(hb)
            if cn < best_c5 + 0.01:
                best_h, best_c5 = _local_search(hb, 15000, rng, 1e7, 50, 8)

    # Phase 5: Final high-alpha SLSQP + local search
    if el() < budget_s - 80:
        for alpha, ml in [(1e9, 100), (5e9, 120), (1e10, 150)]:
            if el() > budget_s - 60:
                break
            hn = _do_slsqp(best_h, alpha, ml, 2000, 4000)
            cn = _eval(hn)
            if cn < best_c5:
                best_c5, best_h = cn, hn
        best_h, best_c5 = _local_search(best_h, 20000, rng, 1e8, 80, 6)

    # Phase 6: Fine polish
    if el() < budget_s - 3:
        nrem = int(min((budget_s - 3 - el()) / 0.0015, 400000))
        best_h, best_c5 = _fine_polish(best_h, max(nrem, 5000), rng)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
