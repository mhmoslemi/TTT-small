```python
def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:

    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smooth_obj_and_grad(x, alpha, topk):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    # Use top-k for efficiency
    if topk < len(c):
        idx = np.argpartition(-c, topk)[:topk]
        c_top = c[idx]
    else:
        idx = np.arange(len(c))
        c_top = c
    m = c_top.max()
    s = alpha * (c_top - m)
    s = np.clip(s, -500, 500)
    s -= s.max()
    w = np.exp(s)
    w_sum = w.sum()
    obj = m + np.log(w_sum) / alpha
    
    # Gradient: d/dx_i of sum_k w_k * c_k / w_sum + m
    # = sum_k (w_k/w_sum) * dc_k/dx_i
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        wi = w[ii] / w_sum * dx
        if wi < 1e-15:
            continue
        L = k - (n - 1)  # actual lag
        if L >= 0:
            # c_k = sum_{j=0}^{n-1-L} h[j]*(1-h[j+L]) * dx
            # dc_k/dx[j] for j in [0, n-1-L]: (1-x[j+L]) * dx
            # dc_k/dx[j+L] for j in [0, n-1-L]: -x[j] * dx
            g[L:] += wi * (1.0 - x[:n-L])
            g[:n-L] -= wi * x[L:]

            g[M:] += wi * (1.0 - x[:n-M])
```

Alternative: use a Lagrangian. Define:

Actually, this is elegant! For λ > 0, the Lagrangian penalizes sum(h) > n/2, pushing h towards smaller values. For λ < 0, it's the opposite. By adjusting λ, we can hit the constraint exactly.

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _smooth_obj_grad(x, alpha, topk):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    if topk < len(c):
        idx = np.argpartition(-c, topk)[:topk]
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    wsum = w.sum()
    obj = m + np.log(wsum) / alpha
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        wi = w[ii] / wsum * dx
        if wi < 1e-15:
            continue
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wi * (1.0 - x[:n - L])
            g[:n - L] -= wi * x[L:]
        else:
            M = -L
            g[M:] += wi * (1.0 - x[:n - M])
            g[:n - M] -= wi * x[M:]
    return obj, g

def _slsqp(h, alpha, topk, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    def _o(x):
        return _smooth_obj_grad(x, alpha, topk)[0]
    def _j(x):
        return _smooth_obj_grad(x, alpha, topk)[1]
    try:
        res = minimize(_o, h.copy(), jac=_j, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _cd_sweep(h, rng, max_iter=20):
    n = len(h)
    gr = 0.6180339887
    order = rng.permutation(n)
    improved = False
    for i in order:
        a, b = 0.0, 1.0
        c_pt = b - gr * (b - a)
        d_pt = a + gr * (b - a)
        h[i] = c_pt
        fc = _eval(h)
        h[i] = d_pt
        fd = _eval(h)
        for _ in range(max_iter):
            if (b - a) < 1e-9:
                break
            if fc < fd:
                b = d_pt; d_pt = c_pt; fd = fc
                c_pt = b - gr * (b - a)
                h[i] = c_pt; fc = _eval(h)
            else:
                a = c_pt; c_pt = d_pt; fc = fd
                d_pt = a + gr * (b - a)
                h[i] = d_pt; fd = _eval(h)
        h[i] = 0.0
        f0 = _eval(h)
        h[i] = 1.0
        f1 = _eval(h)
        cands = [(f0, 0.0), (fc, c_pt), (fd, d_pt), (f1, 1.0)]
        cands.sort(key=lambda t: t[0])
        h[i] = cands[0][1]
    return improved

def _coord_descent(h, n_sweeps, rng, patience=12, max_iter=20):

    best_c = _eval(h)
    best_h = h.copy()
    stall = 0
    for sweep in range(n_sweeps):

            c_pt = b - gr * (b - a)

            h[i] = c_pt; fc = _eval(h)
            h[i] = d_pt; fd = _eval(h)
            for _ in range(max_iter):

                if fc < fd:
                    b = d_pt; d_pt = c_pt; fd = fc
                    c_pt = b - gr * (b - a)
                    h[i] = c_pt; fc = _eval(h)
                else:
                    a = c_pt; c_pt = d_pt; fc = fd
                    d_pt = a + gr * (b - a)
                    h[i] = d_pt; fd = _eval(h)
            h[i] = 0.0; f0 = _eval(h)
            h[i] = 1.0; f1 = _eval(h)
            cands = [(f0, 0.0), (fc, c_pt), (fd, d_pt), (f1, 1.0)]
            cands.sort(key=lambda t: t[0])
            h[i] = cands[0][1]
        h = _repair(h, n)
        cur = _eval(h)

        else:
            stall += 1
            if stall >= patience:

    return best_h, best_c

def _pair_search(h, n_trials, rng, scale_init=0.05):
    n = len(h)
    h = h.copy()
    best_c = _eval(h)
    best_h = h.copy()
    scale = scale_init
    for t in range(n_trials):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, n - 1)) % n
        d = rng.randn() * scale
        hi, hj = h[i] + d, h[j] - d
        if 0 <= hi <= 1 and 0 <= hj <= 1:
            h[i], h[j] = hi, hj
            c = _eval(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
                scale = min(scale * 1.01, 0.1)
            else:
                h[i], h[j] = hi - d, hj + d
                if t % 500 == 0:
                    scale *= 0.97
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _el():
        return time.time() - t0

    # Phase 1: SLSQP warmup
    for alpha in [50000, 500000, 5000000]:

    # Phase 2: Coordinate descent (main workhorse)
    if _el() < budget_s * 0.55:

        if c5_cd < best_c5:

            best_h = h_cd

    # Phase 3: High-alpha SLSQP polish
    if _el() < budget_s * 0.70:
        for alpha in [50000000, 500000000, 5000000000]:
            if _el() > budget_s * 0.70:
                break
            h_new = _slsqp(best_h, alpha, 100, 600, 1e-16)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 4: More coordinate descent with restarts
    if _el() < budget_s * 0.88:

            if restart > 0:

            else:

    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
```
