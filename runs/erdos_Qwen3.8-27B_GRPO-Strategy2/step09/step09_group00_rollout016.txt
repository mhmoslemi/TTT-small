            if cn < c - 1e-15:
                h = hn
                corr = cn_raw
                c = cn
                accepted = True
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
                break
            lr *= 0.5
        if not accepted:
            stall += 1
            if stall > 80:
                noise_scale = 0.008 * min(1.0, 50.0 / (stall - 70))
                h = _proj(h + rng.randn(n) * noise_scale, S)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c = float(corr.max()) * dx
                lr = lr_init * 0.4
                stall = 0
        else:
            lr = min(0.1, lr * 1.03)
            if stall > 0:
                stall = max(0, stall - 10)
    return best_h, best_c

def _sa_polish(h0, n, budget, rng):
    h = h0.copy()
    S = n * 0.5
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = float(corr.max()) * dx
    best_c = c
    best_h = h.copy()
    t0 = time.time()
    T0, Tend = 0.008, 0.00005
    while time.time() - t0 < budget:
        frac = min(1.0, (time.time() - t0) / max(budget, 1e-9))
        T = T0 * (Tend / T0) ** frac
        step = 0.008 * (T / T0) ** 0.3
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = (rng.random() - 0.5) * 2.0 * step
        if h[i] + d < 0:
            d = -h[i]
        if h[i] + d > 1:
            d = 1.0 - h[i]
        if h[j] - d < 0:
            d = h[j]
        if h[j] - d > 1:
            d = h[j] - 1.0
        if abs(d) < 1e-13:
            continue
        h[i] += d
        h[j] -= d
        cn = _c5(h)
        delta = cn - c
        if delta < -1e-15 or (delta >= 0 and rng.random() < np.exp(-delta / max(T, 1e-15))):
            c = cn
            if c < best_c:
                best_c = c
                best_h = h.copy()
                corr = np.correlate(h, 1.0 - h, mode='full')
        else:
            h[i] -= d
            h[j] += d
    return best_h, best_c

def _block_sa(h0, n, budget, rng, block_size=4):
    h = h0.copy()
    S = n * 0.5
    c = _c5(h)
    best_c = c
    best_h = h.copy()
    t0 = time.time()
    while time.time() - t0 < budget:
        i = rng.randint(0, n - block_size + 1)
        j = rng.randint(0, n - block_size + 1)
        if i == j or abs(i - j) < block_size:
            continue
        d = (rng.random() - 0.5) * 0.03
        h[i:i+block_size] += d
        h[j:j+block_size] -= d
        cn = _c5(h)
        if cn < c:
            c = cn
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h[i:i+block_size] -= d
            h[j:j+block_size] += d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _proj(h0, n0 * 0.5)
    best_h = h0.copy()
    best_c = _c5(h0)

    for hh in (1.0 - h0.copy(), h0[::-1].copy()):
        c = _c5(hh)
        if c < best_c:
            best_h = hh.copy()
            best_c = c

    # Phase 1: Main gradient descent at n0 (40% budget)
    t1 = t0 + 0.40 * budget_s
    if time.time() < t1:
        h1, c1 = _grad_desc(best_h, n0, t1 - time.time() - 2, rng, lr_init=0.04, top_m=5)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Gradient descent with smaller top_m (15% budget)
    t2 = t0 + 0.55 * budget_s
    if time.time() < t2:
        h2, c2 = _grad_desc(best_h, n0, t2 - time.time() - 2, rng, lr_init=0.015, top_m=3)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Try n=280 (15% budget)
    t3 = t0 + 0.70 * budget_s
    if time.time() < t3 and n0 >= 280:
        n2 = n0 // 2
        h2s = best_h.reshape(n2, 2).mean(axis=1)
        h2s = _proj(h2s, n2 * 0.5)
        c2s = _c5(h2s)
        if c2s < best_c * 1.005:
            h2s, c2s = _grad_desc(h2s, n2, t3 - time.time() - 2, rng, lr_init=0.03, top_m=4)
            if c2s < best_c:
                best_h, best_c = h2s, c2s
    del t3

    # Phase 4: Fine gradient at n0 (15% budget)
    t4 = t0 + 0.85 * budget_s
    if time.time() < t4:
        if len(best_h) != n0:
            x_fine = np.linspace(0, 1, n0)
            x_coarse = np.linspace(0, 1, len(best_h))
            best_h = _proj(np.interp(x_fine, x_coarse, best_h), n0 * 0.5)
            best_c = _c5(best_h)
        slot = (t4 - time.time()) / 3.0
        hcur = best_h.copy()
        for top_m in (2, 3, 5):
            if time.time() >= t4 - 1:
                break
            hh, cc = _grad_desc(hcur, n0, slot, rng, lr_init=0.008, top_m=top_m)
            if cc < best_c:
                best_h, best_c = hh, cc
            hcur = _proj(hh + rng.randn(n0) * 0.004, n0 * 0.5)

    # Phase 5: SA polish (8% budget)
    t5 = t0 + budget_s - 60
    if time.time() < t5:
        best_h, best_c = _sa_polish(best_h, len(best_h), t5 - time.time() - 2, rng)

    # Phase 6: Block SA + final gradient (7% budget)
    t6 = t0 + budget_s - 15
    if time.time() < t6:
        best_h, best_c = _block_sa(best_h, len(best_h), t6 - time.time() - 10, rng, block_size=3)
        if time.time() < t6 - 3:
            best_h, best_c = _grad_desc(best_h, len(best_h), t6 - time.time() - 4, rng, lr_init=0.005, top_m=3)

    # Final micro-SA
    t7 = t0 + budget_s - 3
    if time.time() < t7:
        best_h, best_c = _sa_polish(best_h, len(best_h), t7 - time.time() - 1, rng)

    best_h = _proj(best_h, len(best_h) * 0.5)
    best_c = _c5(best_h)
    return best_h, best_c, len(best_h)
