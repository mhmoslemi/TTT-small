    dx = 2.0 / n

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _eval_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _soft_grad(h, alpha, m_lags, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-m_lags:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s - s.max(), -500, 500)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - h[:n - L])
            g[:n - L] -= wk * h[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n - M]
    return g

def _sum_fn(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _obj_fn(x, alpha, m_lags, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    idx = np.argsort(c)[-m_lags:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s - s.max(), -500, 500)
    return c.max() + np.log(np.exp(s).sum()) / alpha

def _slsqp_opt(h, alpha, m_lags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_obj_fn, h, jac=_soft_grad, args=(alpha, m_lags, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return h

def _nesterov_pg(h, alpha, m_lags, nsteps, step0, mu=0.75):
    n = len(h)
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    for _ in range(nsteps):
        y = _proj(h + mu * v, n * 0.5)
        g = _soft_grad(y, alpha, m_lags, n)
        v = mu * v - step * g
        h_new = _proj(h + v, n * 0.5)
        c_new = _eval(h_new)
        if c_new < cur_c:
            h = h_new
            cur_c = c_new
            step = min(step * 1.03, step0 * 30.0)
        else:
            for _bt in range(6):
                step *= 0.5
                v = v * 0.8
                h_new = _proj(h + v, n * 0.5)
                c_new = _eval(h_new)
                if c_new < cur_c:
                    h = h_new
                    cur_c = c_new
                    break
            else:
                step = max(step, step0 * 0.01)
    return _repair(h, n)

def _block_search(h, n_iter, rng, base_bs):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    stale = 0
    for it in range(n_iter):
        frac = it / max(n_iter, 1)
        bs = max(4, int(base_bs * (1.5 - 0.5 * frac) * rng.uniform(0.6, 1.8)))
        bs = min(bs, n // 4)
        start = rng.randint(0, n - bs)
        idx = np.arange(start, start + bs)
        step = base_bs * 4e-5 * (2.0 - frac) * rng.uniform(0.3, 2.5)
        d = step * rng.randn(bs)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < -0.05 or h_new.max() > 1.05:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5cur - 1e-16:
            c5cur = c5n
            h = h_new
            stale = 0
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        else:
            stale += 1
            if stale % 80 == 0:
                kick_n = rng.randint(10, 40)
                kick_idx = rng.randint(0, n, size=kick_n)
                h[kick_idx] += 3e-5 * rng.randn(kick_n)
                h = _repair(h, n)
                c5cur = _eval(h)
                stale = 0
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    return _repair(np.repeat(h, f), len(h) * f)

def _fine_polish(h, nsteps, rng):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    for _ in range(nsteps):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        if rng.rand() < 0.15:
            k = (j + 1 + rng.randint(0, 12)) % n
            d1 = 4e-8 * rng.randn()
            d2 = 4e-8 * rng.randn()
            ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
            if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
                h[i], h[j], h[k] = ni, nj, nk
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c:
                        best_c = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
        else:
            d = 4e-8 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c:

                    if c5n < best_c:
                        best_c = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return best_h, best_c
