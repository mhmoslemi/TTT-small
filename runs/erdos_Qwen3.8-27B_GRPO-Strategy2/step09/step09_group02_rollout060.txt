import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    for _ in range(200):
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _corr_vals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _pnorm_obj(x, p, m_lags):
    c = _corr_vals(x)
    idx = np.argsort(c)[-m_lags:]
    vals = c[idx]
    return float(np.sum(vals**p))

def _pnorm_grad(x, p, m_lags):
    n = len(x)
    dx = 2.0 / n
    c_raw = np.correlate(x, 1.0 - x, mode='full')
    c = c_raw * dx
    idx = np.argsort(c)[-m_lags:]
    # gradient of sum c[k]^p = p * sum c[k]^(p-1) * dc[k]/dx
    g = np.zeros(n)
    for k in idx:
        val = c[k]
        if val <= 0:
            w = p * val**(p-1)
        else:
            w = p * val**(p-1)
        L = k - (n - 1)
        # dc[k]/dh[m] = dx * (1 - h[m-L] - h[m+L]) with h[j]=0 OOB
        for m in range(n):
            gm = 1.0
            if 0 <= m - L < n:
                gm -= x[m - L]
            if 0 <= m + L < n:
                gm -= x[m + L]
            g[m] += w * dx * gm
    return g

def _pnorm_grad_fast(x, p, m_lags):
    n = len(x)
    dx = 2.0 / n
    c_raw = np.correlate(x, 1.0 - x, mode='full')
    c = c_raw * dx
    idx = np.argsort(c)[-m_lags:]
    g = np.zeros(n)
    for k in idx:
        L = k - (n - 1)
        # weight for this lag
        w = p * (c[k])**(p-1) * dx
        # gradient contribution: w * (1 - x[m-L] - x[m+L])
        g += w  # the "1" part
        if L >= 0:
            g[:n-L] -= w * x[L:]
            g[L:] -= w * x[:n-L]
        else:
            M = -L
            g[M:] -= w * x[:n-M]
            g[:n-M] -= w * x[M:]
    return g

def _active_grad(h, m_lags, n_steps, step0):

    target = n * 0.5
    h = h.copy()
    cur = _eval(h)
    step = step0
    momentum = np.zeros(n)
    for _ in range(n_steps):
        c = _corr_vals(h)
        idx = np.argsort(c)[-m_lags:]
        # Compute gradient for active lags
        g = np.zeros(n)
        dx = 2.0/n
        for k in idx:
            L = k - (n-1)
            g += dx
            if L >= 0:
                g[:n-L] -= dx * h[L:]
                g[L:] -= dx * h[:n-L]
            else:
                M = -L
                g[M:] -= dx * h[:n-M]
                g[:n-M] -= dx * h[M:]
        # Remove mean (for sum constraint)
        g -= g.mean()
        # Momentum
        momentum = 0.7 * momentum + 0.3 * g
        norm = np.linalg.norm(momentum)
        if norm < 1e-16:

        direction = -momentum / norm
        # Line search with projection

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    for _ in range(200):
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _corr_vals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lags(h, lags):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in lags:
        L = k - (n - 1)
        if L >= 0:
            g[L:] += dx * (1.0 - h[:n - L])
            g[:n - L] -= dx * h[L:]
        else:
            M = -L
            g[:n - M] += dx * (1.0 - h[M:])
            g[M:] -= dx * h[:n - M]
    return g

def _lsq_logsumexp(h, alpha, ml, maxit):
    n = len(h)
    target = n * 0.5
    def _obj(x):
        c = _corr_vals(x)
        idx = np.argsort(c)[-ml:]
        m = c.max()
        s = np.clip(alpha * (c[idx] - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def _grd(x):
        c = _corr_vals(x)
        idx = np.argsort(c)[-ml:]
        m = c.max()
        s = np.clip(alpha * (c[idx] - m), -400, 400)
        s -= s.max()
        w = np.exp(s); w /= w.sum()
        g = np.zeros(n)
        for ii, k in enumerate(idx):
            if w[ii] < 1e-15: continue
            wk = w[ii] * 2.0 / n
            L = k - (n - 1)
            if L >= 0:
                g[L:] += wk * (1.0 - x[:n - L]); g[:n - L] -= wk * x[L:]
            else:
                M = -L; g[:n - M] += wk * (1.0 - x[M:]); g[M:] -= wk * x[:n - M]
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h, jac=_grd, method='SLSQP', args=(),
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(res.x, n)
    except:
        return h

def _lsq_pnorm(h, p, ml, maxit):
    n = len(h)
    target = n * 0.5
    def _obj(x):
        c = _corr_vals(x)
        idx = np.argsort(c)[-ml:]
        return float(np.sum(c[idx] ** p))
    def _grd(x):
        c = _corr_vals(x)
        idx = np.argsort(c)[-ml:]
        g = np.zeros(n)
        for k in idx:
            w = p * c[k] ** (p - 1)
            L = k - (n - 1)
            if L >= 0:
                g[L:] += w * (1.0 - x[:n - L]); g[:n - L] -= w * x[L:]
            else:
                M = -L; g[:n - M] += w * (1.0 - x[M:]); g[M:] -= w * x[:n - M]
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h, jac=_grd, method='SLSQP', args=(),
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(res.x, n)
    except:
        return h

def _lag_cycle(h, n_rounds, n_sel, rng):
    n = len(h)
    h = h.copy()
    cur = _eval(h)
    best_h = h.copy(); best_c = cur
    all_lags = np.arange(2 * n - 1)
    for _ in range(n_rounds):
        c = _corr_vals(h)
        sel = rng.choice(all_lags, size=n_sel, replace=False)
        wts = c[sel] ** 6
        if wts.sum() < 1e-30:
            wts = np.ones(len(sel))
        wts /= wts.sum()
        g = np.zeros(n)
        for ii, k in enumerate(sel):
            gg = _grad_lags(h, [k])
            g += wts[ii] * gg
        g -= g.mean()
        ng = np.linalg.norm(g)
        if ng < 1e-16: break
        step = 2e-4
        for _ in range(10):
            trial = h - step * g / ng
            trial = np.clip(trial, 0.0, 1.0)
            trial = _repair(trial, n)
            cn = _eval(trial)
            if cn < cur - 1e-15:
                h = trial; cur = cn
                if cur < best_c: best_c = cur; best_h = h.copy()
                break
            step *= 0.45
    return best_h, best_c

def _peak_shave(h, n_iter, rng, ml=40):
    n = len(h)
    h = h.copy()
    cur = _eval(h)
    best_h = h.copy(); best_c = cur
    for _ in range(n_iter):
        c = _corr_vals(h)
        top = np.argsort(c)[-ml:]
        g = _grad_lags(h, top)
        g -= g.mean()
        ng = np.linalg.norm(g)
        if ng < 1e-16: break
        nsub = rng.choice([6, 10, 14, 18, 24])
        wts = np.abs(g)
        if wts.sum() < 1e-16: wts = np.ones(n)
        probs = wts / wts.sum()
        sub = rng.choice(n, size=nsub, replace=False, p=probs)
        gs = g[sub]
        s = gs @ gs
        if s < 1e-16: continue
        d = -gs / s
        md = np.abs(d).max()
        if md < 1e-16: continue
        d = d / md * 1.5e-4
        d -= d.mean()
        trial = h.copy()
        trial[sub] += d
        trial = _repair(trial, n)
        cn = _eval(trial)
        if cn < cur - 1e-16:
            h = trial; cur = cn
            if cur < best_c: best_c = cur; best_h = h.copy()
    return best_h, best_c

def _sim_anneal(h, n_iter, rng, t0=5e-7, t1=1e-11):
    n = len(h)
    h = h.copy()
    cur = _eval(h)
    best_h = h.copy(); best_c = cur
    for it in range(n_iter):
        t = t0 * (t1 / t0) ** (it / max(1, n_iter - 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j: continue
        amp = 2e-7 + 3e-7 * (1.0 - it / max(1, n_iter))
        d = rng.randn() * amp
        h[i] += d; h[j] -= d
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
            h[i] -= d; h[j] += d; continue
        h = _repair(h, n)
        cn = _eval(h)
        delta = cn - cur
        if delta < 0 or rng.rand() < np.exp(-max(delta / t, -500)):
            h = h; cur = cn
            if cur < best_c: best_c = cur; best_h = h.copy()
        else:
            h[i] -= d; h[j] += d
            h = _repair(h, n)
    return best_h, best_c

def _proj_grad(h, alpha, ml, steps, s0, rng):
    n = len(h)
    h = h.copy()
    cur = _eval(h)
    v = np.zeros(n)
    step = s0
    mu = 0.8
    for _ in range(steps):
        c = _corr_vals(h)
        idx = np.argsort(c)[-ml:]
        m = c.max()
        s = np.clip(alpha * (c[idx] - m), -400, 400)
        s -= s.max()
        w = np.exp(s); w /= w.sum()
        g = np.zeros(n)
        for ii, k in enumerate(idx):
            if w[ii] < 1e-15: continue
            gg = _grad_lags(h, [k])
            g += w[ii] * gg
        g -= g.mean()
        v = mu * v + (1 - mu) * g
        ng = np.linalg.norm(v)
        if ng < 1e-16: break
        v /= ng
        taken = False
        for _ in range(12):
            trial = _repair(np.clip(h - step * v, 0, 1), n)
            cn = _eval(trial)
            if cn < cur - 1e-15:
                h = trial; cur = cn
                step = min(step * 1.07, s0 * 30)
                taken = True
                break
            step *= 0.38
        if not taken:
            step = min(max(step * 1.4, s0 * 0.5), s0 * 30)
    return _repair(h, n), cur

    t0 = time.time()

        if _el() > 180: break
        ml = min(150, 20 + 10 * i)
        h_new = _lsq_logsumexp(best_h.copy(), a, ml, 600)
