        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _cd_pass(h, rng):
    n = len(h)
    c = _corr(h)
    cmax = 2 * n - 1
    nm1 = n - 1
    jlist = np.arange(n)
    rng.shuffle(jlist)
    for j in jlist:
        j = int(j)
        beta = np.zeros(cmax)
        k1_lo = j
        k1_hi = min(cmax - 1, j + nm1)
        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax - 1, cmax - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
        beta[k2_lo:k2_hi + 1] -= h[j + lags2]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(30):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(c + beta * t1) <= np.max(c + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-14:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

# L-BFGS-B with penalty for sum constraint
def _lfsb_obj(x, alpha, penalty, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    e = np.exp(s)
    lse = m + np.log(e.sum()) / alpha
    pen = penalty * (x.sum() - 0.5 * n) ** 2
    return lse + pen

def _lfsb_grad(x, alpha, penalty, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    e = np.exp(s)
    w = e / e.sum()
    nm1 = n - 1
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-18:
            continue
        L = k - nm1
        if L >= 0:
            g[:n - L] += wk * (1.0 - x[L:])
            g[L:] -= wk * x[:n - L]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    g += 2.0 * penalty * (x.sum() - 0.5 * n)
    return g

def _lfsb_optimize(h, alpha, penalty, maxiter, n):
    n2 = n
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lfsb_obj, h, jac=_lfsb_grad,
                       args=(alpha, penalty, n2), method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n2)
    except Exception:
        return h
