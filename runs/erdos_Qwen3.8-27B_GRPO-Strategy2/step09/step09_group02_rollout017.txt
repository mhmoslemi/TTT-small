def _pair_optimal_d(h, c, i, j, n):
    """Find optimal transfer d from j to i"""
    dx = 2.0/n
    d_lo = -h[j]  # h[j] can decrease to 0
    d_hi = 1.0 - h[i]  # h[i] can increase to 1
    
    # Compute a_L for each lag L
    # c has 2n-1 entries, indexed 0 to 2n-2, corresponding to L = -(n-1) to n-1
    a = np.zeros(2*n-1)
    for L in range(-(n-1), n):
        idx = L + (n-1)  # index into c
        # h[i] contribution: (1-h[i+L]) if 0 <= i+L < n, minus h[i-L] if 0 <= i-L < n
        ai = 0.0
        if 0 <= i+L < n:
            ai += (1.0 - h[i+L])
        if 0 <= i-L < n:
            ai -= h[i-L]
        # h[j] contribution with -d
        aj = 0.0
        if 0 <= j+L < n:
            aj -= (1.0 - h[j+L])
        if 0 <= j-L < n:
            aj += h[j-L]
        a[idx] = ai + aj
    
    # f(d) = max_k (c[k] + d*a[k]) for d in [d_lo, d_hi]
    # Find the minimum of this convex piecewise linear function
    
    # Strategy: evaluate at d=0, d_lo, d_hi, and at a few critical points
    candidates = [0.0, d_lo, d_hi]
    
    # Add intersection points of top lags
    top_idx = np.argsort(c)[-min(10, 2*n-1):]
    for p in range(len(top_idx)):
        for q in range(p+1, len(top_idx)):
            ip, iq = top_idx[p], top_idx[q]
            if abs(a[ip] - a[iq]) > 1e-15:
                d_int = (c[iq] - c[ip]) / (a[ip] - a[iq])
                if d_lo <= d_int <= d_hi:
                    candidates.append(d_int)
    
    best_d = 0.0
    best_val = np.inf
    for d in candidates:
        d = max(d_lo, min(d_hi, d))
        val = np.max(c + d * a)
        if val < best_val:
            best_val = val
            best_d = d
    return best_d

def _compute_a(h, i, j, n):
    """Compute a_L for all L = -(n-1) to n-1"""
    a = np.zeros(2*n-1)
    Ls = np.arange(-(n-1), n)  # shape (2n-1,)
    
    # h[i+L] valid when 0 <= i+L < n, i.e., -i <= L < n-i
    mask1 = (Ls >= -i) & (Ls < n-i)
    a[mask1] += (1.0 - h[i + Ls[mask1]])
    
    # h[i-L] valid when 0 <= i-L < n, i.e., i-n < L <= i
    mask2 = (Ls > i-n) & (Ls <= i)
    a[mask2] -= h[i - Ls[mask2]]
    
    # -(1-h[j+L]) valid when 0 <= j+L < n
    mask3 = (Ls >= -j) & (Ls < n-j)
    a[mask3] -= (1.0 - h[j + Ls[mask3]])
    
    # h[j-L] valid when 0 <= j-L < n
    mask4 = (Ls > j-n) & (Ls <= j)
    a[mask4] += h[j - Ls[mask4]]
    
    return a
