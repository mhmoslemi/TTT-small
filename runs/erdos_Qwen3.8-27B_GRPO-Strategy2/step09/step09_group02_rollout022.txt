import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    for _ in range(200):
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _pnorm_obj(x, p, top_k):
    c = _corr(x)
    idx = np.argsort(c)[-top_k:]
    cp = c[idx]
    s = np.sum(np.power(cp, p))
    return s ** (1.0 / p)

def _pnorm_grad(x, p, top_k):
    n = len(x)
    c = _corr(x)
    idx = np.argsort(c)[-top_k:]
    cp = c[idx]
    denom = np.sum(np.power(cp, p)) ** ((p - 1.0) / p)
    w = np.power(cp, p - 1.0) / denom
    dx = 2.0 / n
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-20:
            continue
        ww = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += ww * (1.0 - x[:n - L])
            g[:n - L] -= ww * x[L:]
        else:
            M = -L
            g[:n - M] += ww * (1.0 - x[M:])
            g[M:] -= ww * x[:n - M]
    return g

def _slsqp_run(h, p, top_k, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    tgt = n * 0.5
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt,
             'jac': lambda x: np.ones(len(x))}]
    try:
        res = minimize(_pnorm_obj, h, jac=_pnorm_grad, args=(p, top_k),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _coord_update(h, i, top_idx, c_top):
    n = len(h)
    cur = h[i]
    L = top_idx - (n - 1)
    m = np.abs(L)
    A = np.zeros(len(top_idx))
    mask_hi = i >= m
    mask_lo = (i + m) <= (n - 1)
    idx_hi = i - m[mask_hi]
    idx_lo = i + m[mask_lo]
    A[mask_hi] += (1.0 - h[idx_hi])
    A[mask_lo] -= h[idx_lo]
    m0 = (m == 0)
    B = c_top - A * cur

    def f(t):
        vals = A * t + B
        vals[m0] = t * (1.0 - t) + B[m0]
        return np.max(vals)

    lo, hi = 0.0, 1.0
    for _ in range(38):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if f(m1) < f(m2):
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5

def _coord_descent(h, n_top, n_sweeps, rng, time_limit, t0):

    best_h = h.copy()

            else:

                h[i] -= 2 * d

def _sa_polish(h, n_iter, rng, t0, t_start, t_end, time_limit, base_step):

    c5c = _eval(h)
    best_h = h.copy()

        ni, nj = h[i] + d, h[j] - d
