This is a 1D piecewise linear minimization! I can solve it exactly.

2. C_k(d) ≈ C_k + d·s_k (ignoring second-order)
3. Find d that minimizes max_k (C_k + d·s_k)

This is the 1D problem: min_d max_k (C_k + d·s_k)

def _local_search(h, n_iter, rng, base_step, mlags=20):
    n = len(h)
    c5c = _eval(h)
    best_h, best_c = h.copy(), c5c
    step = base_step
    for it in range(n_iter):
        c = _corr(h)
        top = np.argsort(c)[-mlags:]
        dxv = 2.0 / n
        gh = np.zeros(n)
        for k in top:
            L = k - (n - 1)
            if L >= 0:
                gh[L:] += dxv * (1.0 - h[:n - L])
                gh[:n - L] -= dxv * h[L:]
            else:
                M = -L
                gh[:n - M] += dxv * (1.0 - h[M:])
                gh[M:] -= dxv * h[:n - M]
        sh = h * (1 - h)
        S = sh.sum() + 1e-30
        gdir = sh * (gh - (gh * sh).sum() / S)
        npert = rng.randint(2, 7)
        idx = rng.randint(0, n, size=npert)

        ng = np.linalg.norm(gdir) + 1e-30
        d -= 0.4 * step * gdir[idx] / ng
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < -0.15 or h_new.max() > 1.15:
            step *= 0.7
            continue

        c5n = _eval(h_new)
        if c5n < c5c - 1e-15:
            c5c, h = c5n, h_new
            step = min(step * 1.04, base_step * 5)
            if c5n < best_c:
                best_c, best_h = c5n, h.copy()
        else:
            step *= 0.93
        step = max(step, 1e-10)
    return best_h, best_c

def _fine_polish(h, n_iter, rng):
    n = len(h)
    c5c = _eval(h)
    best_h, best_c = h.copy(), c5c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 25)) % n
        d = (1e-9 + 5e-8 * rng.rand()) * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:

                    best_c, best_h = c5n, h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _group_polish(h, n_iter, rng, gs=5):

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()

    for _ in range(150):
        if abs(s - t) < 1e-13: break
        h *= t / s; h = np.clip(h, 0, 1); s = h.sum()
    d = t - h.sum()

            if abs(d) < 1e-14: break

            else:

    sh = h * (1 - h)
    S = sh.sum() + 1e-30
    return sh * (gh - (gh * sh).sum() / S)
