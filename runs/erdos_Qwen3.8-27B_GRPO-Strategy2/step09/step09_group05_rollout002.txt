    p_seq = [10, 30, 100, 400, 1500]
    for p in p_seq:
        if time.time() - t0 > 280:
            break
        h_new = _slsqp_pm(best_h.copy(), p, 40, 600)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 3: Adam projected gradient with power mean
    adam_cfgs = [(30, 40, 500, 0.004), (100, 50, 400, 0.002), (500, 70, 300, 0.001)]
    for p, ml, steps, lr0 in adam_cfgs:
        if time.time() - t0 > budget_s - 350:
            break
        h_new = _adam_pm(best_h.copy(), p, ml, steps, lr0)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 4: Another SLSQP round with high p
    if time.time() - t0 < budget_s - 250:
        for p in [5000, 20000]:
            if time.time() - t0 > budget_s - 220:
                break
            h_new = _slsqp_pm(best_h.copy(), p, 50, 400)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 5: Block search
    if time.time() - t0 < budget_s - 150:
        remaining = budget_s - 150 - (time.time() - t0)
        niter = int(remaining / 0.007)
        niter = min(niter, 25000)
        best_h, best_c5 = _block_search(best_h, niter, rng, [3, 5, 8, 15], 0.004)

    # Phase 6: Swap search
    if time.time() - t0 < budget_s - 70:
        remaining = budget_s - 70 - (time.time() - t0)
        niter = int(remaining / 0.004)
        niter = min(niter, 40000)
        best_h, best_c5 = _swap_search(best_h, niter, rng, 18)

    # Phase 7: Fine 2-point polish (medium step)
    if time.time() - t0 < budget_s - 30:
        remaining = budget_s - 30 - (time.time() - t0)
        niter = int(remaining / 0.0012)
        niter = min(niter, 80000)
        best_h, best_c5 = _fine_2pt(best_h, niter, rng, 5e-7)

    # Phase 8: Ultra-fine 2-point polish
    if time.time() - t0 < budget_s - 2:
        remaining = budget_s - 2 - (time.time() - t0)
        niter = int(remaining / 0.001)
        niter = min(niter, 100000)
        best_h, best_c5 = _fine_2pt(best_h, niter, rng, 1e-7)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
