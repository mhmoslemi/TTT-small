def _proj(v,S):

    x += (S-x.sum())/x.size

import numpy as np

def _proj(v, S):
    x = np.asarray(v, float).copy()
    x += (S - x.sum()) / x.size
    for _ in range(80):
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = x < 1.0
            if not m.any():
                break
            x[m] += d / m.sum()
            np.minimum(x, 1.0, out=x)
        else:
            m = x > 0.0
            if not m.any():
                break
            x[m] += d / m.sum()
            np.maximum(x, 0.0, out=x)
    d = S - x.sum()
    if abs(d) > 1e-12:
        for i in range(x.size):
            if d > 0:
                add = min(d, 1.0 - x[i])
                x[i] += add
                d -= add
            else:
                rem = min(-d, x[i])
                x[i] -= rem
                d += rem
            if abs(d) < 1e-13:
                break
    return x

def _repair(h, n):
    h = np.clip(np.asarray(h, float), 0.0, 1.0).copy()
    return _proj(h, 0.5 * n)

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * (2.0 / n)

def _add_lag_grad(g, h, k, neg, wg=1.0):
    n = len(h)
    if k == 0:
        g += wg * (1.0 - 2.0 * h)
    else:
        nk = n - k
        if not neg:
            g[:nk] += wg * (1.0 - h[k:])
            g[k:] -= wg * h[:nk]
        else:
            g[k:] += wg * (1.0 - h[:nk])
            g[:nk] -= wg * h[k:]
    return g

def _resample(h, m):
    n = len(h)
    if n == m:
        return _proj(h, 0.5 * m)
    if n % m == 0:
        out = h.reshape(m, n // m).mean(axis=1)
    else:
        x0 = (np.arange(n) + 0.5) * (2.0 / n)
        x1 = (np.arange(m) + 0.5) * (2.0 / m)
        out = np.interp(x1, x0, h)
    return _proj(out, 0.5 * m)

def _swap_bin(hb, vb, i, j):
    if hb[i] == hb[j]:
        return
    if hb[i]:
        hb[i] = 0; vb[i] = 1
        hb[j] = 1; vb[j] = 0
    else:
        hb[i] = 1; vb[i] = 0
        hb[j] = 0; vb[j] = 1

def _binary_polish(h0, rng, deadline):
    n = len(h0)
    S = n // 2
    if S <= 0 or S >= n:
        h = _repair(h0, n)
        return h, _c5(h)
    b = np.zeros(n, dtype=np.int32)
    b[np.argsort(h0)[::-1][:S]] = 1
    best = b.copy()
    bestmax = int(np.correlate(b, 1 - b, mode="full").max())
    alt = np.zeros(n, dtype=np.int32)
    alt[np.argsort(h0)[:S]] = 1
    am = int(np.correlate(alt, 1 - alt, mode="full").max())
    if am < bestmax:
        best = alt
        bestmax = am
    hb = best.copy()
    vb = 1 - hb
    cm = bestmax
    t0 = time.time()
    total = max(1e-9, deadline - t0)
    last = t0
    while time.time() < deadline:
        C = np.correlate(hb, vb, mode="full")
        cm = int(C.max())
        if cm < bestmax:
            bestmax = cm
            best = hb.copy()
            last = time.time()
        fr = min(1.0, (time.time() - t0) / total)
        if time.time() - last > 20.0:
            hb = best.copy()
            vb = 1 - hb
            cm = bestmax
            last = time.time()
            continue
        T = 5.0 * (0.05 ** fr) + 0.3
        ctr = n - 1
        idx = np.flatnonzero(C >= cm - 1)
        if idx.size == 0:
            idx = np.array([int(C.argmax())])
        elif idx.size > 32:
            vals = C[idx]
            p = np.argpartition(vals, -32)[-32:]
            idx = idx[p]
            vals = vals[p]
        else:
            vals = C[idx]
        w = np.exp(vals - (cm - 1))
        ws = float(w.sum())
        if ws <= 0.0:
            w = np.ones(len(idx), dtype=float)
            ws = float(len(idx))
        w /= ws
        m = int(idx[rng.choice(len(idx), p=w)])
        k = abs(m - ctr)
        neg = m > ctr
        si = sj = -1
        if k:

            if not neg:

                if pos.size:

                    _swap_bin(hb, vb, a, a + k)
                    si, sj = a, a + k
            else:
                pos = np.flatnonzero((hb[:nk] == 0) & (hb[k:] == 1))
                if pos.size:
                    a = int(pos[rng.randint(pos.size)])
                    _swap_bin(hb, vb, a, a + k)

        if si < 0:

                a = int(o[rng.randint(o.size)])

                _swap_bin(hb, vb, a, b)
                si, sj = a, b
        if si >= 0:

                    last = time.time()
            else:

    hf = best.astype(float)
    return hf, _c5(hf)
