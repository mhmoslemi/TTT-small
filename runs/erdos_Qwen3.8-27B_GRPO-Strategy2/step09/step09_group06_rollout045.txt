            best_h = h_up.copy()

    # Phase 2: SLSQP top-m sweep
    slsqp_configs = [(120, 500), (60, 500), (40, 400), (25, 350), (15, 300), (10, 250)]
    for m, mi in slsqp_configs:
        if _el() > 380:
            break
        h_new = _run_slsqp(best_h, m, mi)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 3: PGD with momentum
    pgd_configs = [(50, 600, 0.004, 0.5), (30, 500, 0.002, 0.6),
                   (20, 400, 0.001, 0.7), (12, 300, 0.0005, 0.75),
                   (8, 200, 0.0003, 0.8)]
    for m, steps, s0, mu in pgd_configs:
        if _el() > 680:
            break
        h_new, _ = _pgd(best_h.copy(), m, steps, s0, mu)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 4: SLSQP refinement after PGD
    if _el() < budget_s - 200:
        for m in [15, 10, 8]:
            if _el() > budget_s - 150:
                break
            h_new = _run_slsqp(best_h, m, 300)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()

    # Phase 5: Lag-targeted local search
    if _el() < budget_s - 80:
        rem = int((budget_s - 80 - _el()) * 250)
        best_h, best_c5 = _lag_search(best_h, n, rng, rem)

    # Phase 6: Ultra-fine polish
    if _el() < budget_s - 2:
        npol = int((budget_s - 2 - _el()) * 1200)
        best_h, best_c5 = _polish(best_h, n, rng, npol)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
