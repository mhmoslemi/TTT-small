import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    target = n // 2
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_obj(x, alpha, topk):
    n = len(x)
    c = _corr_vec(x)
    m = c.max()
    idx = np.argsort(c)[-topk:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, topk):
    n = len(x)
    dx = 2.0 / n
    c = _corr_vec(x)
    m = c.max()
    idx = np.argsort(c)[-topk:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

            lo = mid
        else:
            hi = mid

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    target = n // 2
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(80):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_obj(x, alpha, topk):
    c = _corr_vec(x)
    m = c.max()
    idx = np.argsort(c)[-topk:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, topk):
    n = len(x)
    dx = 2.0 / n
    c = _corr_vec(x)
    m = c.max()
    idx = np.argsort(c)[-topk:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:

            g[:n - M] += wk * (1.0 - x[M:])

def _thue_morse_bin(n, rng):
    t = np.zeros(n)
    for i in range(1, n):
        t[i] = (t[i >> 1] ^ (i & 1))
    target = n // 2

    if s > target:

        idx = rng.choice(len(ones), size=s - target, replace=False)
        t[ones[idx]] = 0
    elif s < target:
        zeros = np.where(t == 0)[0]
        idx = rng.choice(len(zeros), size=target - s, replace=False)
        t[zeros[idx]] = 1
    return t
