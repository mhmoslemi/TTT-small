    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(c)) * (2.0 / float(n))

    return _c5(_corr(h), len(h))

def _proj(h, target):
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    s = float(h.sum())
    if abs(s - target) <= 1e-14:
        return h
    lo = -float(h.max())
    hi = 1.0 - float(h.min())
    for _ in range(70):

        if ss > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

        lam = -5.0
    return h, c, lam

def _cd_opt(h, n, rng, t_end, grid, lam0=0.0):
    target = 0.5 * n
    h = _proj(h, target)
    c = _corr(h)
    bh = h.copy()
    bc = _c5(c, n)
    lam = float(lam0)
    passes = 0
    while time.time() < t_end - 1.0:
        passes += 1
        h, c, lam = _cd_pass(h, c, lam, grid, time.time(), rng)
        if abs(float(h.sum()) - target) > 1.0e-12 or (passes & 1) == 0:
            h = _proj(h, target)
            c = _corr(h)
            if abs(float(h.sum()) - target) > 1.0e-9:
                lam = 0.0
            else:
                lam *= 0.5
        c5 = _c5(c, n)
        if c5 < bc:
            bc = c5
            bh = h.copy()
        elif c5 > bc + 0.02:
            h = bh.copy()
            c = _corr(h)
            lam = 0.0
    bh = _proj(bh, target)
    bc = _eval(bh)
    return bh, bc

def _sactive(c, alpha, ml):
    m = float(c.max())
    if ml >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argpartition(c, -ml)[-ml:]
    s = np.clip(alpha * (c[idx] - m), -700.0, 700.0)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    return m + np.log(w.sum()) / alpha, idx, w

def _sobj(x, alpha, ml, dx):
    c = _corr(x)
    v, _, _ = _sactive(c, alpha, ml)
    return v * dx

def _sgrad(x, alpha, ml, dx):
    c = _corr(x)
    _, idx, w = _sactive(c, alpha, ml)
    n = len(x)
    g = np.zeros(n)
    lm = n - 1
    for ii in range(len(idx)):
        wk = w[ii] * dx
        if wk < 1.0e-18:
            continue
        k = int(idx[ii])
        L = k - lm
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sf(x, target):
    return x.sum() - target

def _sj(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, ml, it):
    n = len(h)
    try:
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (0.5 * n,)}]
        res = minimize(_sobj, h, jac=_sgrad, args=(alpha, ml, 2.0 / n), method='SLSQP',
                       bounds=bnds, constraints=cons, options={'maxiter': int(it), 'ftol': 1.0e-14})
        x = np.asarray(res.x, dtype=float)
        if np.all(np.isfinite(x)):
            return _proj(x, 0.5 * n)
    except Exception:
        pass
    return h

def run(seed=42, budget_s=1100, **kwargs):
    budget_s = float(budget_s)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    try:
        parent = np.asarray(initial_h_values, dtype=float)
    except Exception:
        parent = None
    if parent is None or parent.size == 0:
        parent = _fallback()
    n = int(parent.size)
    target = 0.5 * n
    best = _proj(parent, target)
    bc = _eval(best)

    qend = t0 + max(3.0, 0.02 * budget_s)
    variants = [best[::-1].copy(), 1.0 - best, 0.5 * (best + best[::-1])]
    for sh in (1, 2, 4, max(1, n // 8), max(1, n // 4)):
        variants.append(np.roll(best, sh))
    for vv in variants:
        if time.time() > qend:
            break
        best, bc = _cons(vv, n, target, best, bc)

    main_end = t0 + 0.78 * budget_s
    cend = t0 + 0.06 * budget_s
    if budget_s > 300 and time.time() < cend:
        for m in (4, 2):
            if n % m == 0:
                nc = n // m
                if nc >= 40 and nc % 2 == 0:
                    hc = _down(best, m)
                    hb, hyp = _up(_cd(hc, nc, rng, cend, _GC)[0], m)
                    hb = _proj(hb, target)
                    cup = _eval(hb)
                    if cup < bc:
                        best, bc = hb, cup
                    if cup < bc + 0.02 and time.time() < min(t0 + 0.13 * budget_s, main_end - 30.0):
                        hb, hc5 = _cd(hb, n, rng, min(t0 + 0.13 * budget_s, main_end - 30.0), _GC)
                        best, bc = _cons(hb, n, target, best, bc)
                    break

    if time.time() < main_end:
        hb, hc5 = _cd(best, n, rng, main_end, _GC)
        best, bc = _cons(hb, n, target, best, bc)

    rend = t0 + 0.87 * budget_s
    sigs = (0.01, 0.03, 0.08, 0.2, 0.5)
    for i, sg in enumerate(sigs):
        if time.time() > rend - 3.0:
            break
        cand = best + sg * rng.standard_normal(n)
        cand = np.clip(cand, 0.0, 1.0)
        cand = _proj(cand, target)
        cc = _eval(cand)
        if cc < bc + 0.05:
            take = max(4.0, (rend - 3.0 - time.time()) / max(1, len(sigs) - i))
            cand, cc = _cd(cand, n, rng, min(rend - 2.0, time.time() + take), _GC)
            best, bc = _cons(cand, n, target, best, bc)

    if time.time() < t0 + budget_s - 90.0:
        hh = _slsqp(best, 50000.0, 60, 35)
        best, bc = _cons(hh, n, target, best, bc)
        if time.time() < t0 + budget_s - 45.0:
            hh = _slsqp(best, 300000.0, 100, 25)
            best, bc = _cons(hh, n, target, best, bc)

    fend = t0 + budget_s - 2.0
    if time.time() < fend - 10.0:
        hf, cf = _cd(best, n, rng, fend, _GF)
        best, bc = _cons(hf, n, target, best, bc)

    if time.time() < t0 + budget_s - 0.5:
        best, bc = _cons(0.5 * (best + best[::-1]), n, target, best, bc)
        best, bc = _cons(1.0 - best, n, target, best, bc)

    best = _proj(best, target)
    bc = _eval(best)
    return best, float(bc), int(n)
