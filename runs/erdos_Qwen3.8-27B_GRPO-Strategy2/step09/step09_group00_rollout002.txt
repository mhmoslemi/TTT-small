def _repair(h, n):
    S = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= S / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - S) < 1e-14:
            break
    d = S - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:

def _proj(v, S):

def _overlap_at_shift(h, k):

def _grad_one_shift(h, k, dx):

    n = len(h)

        # d/dh[i] for i in [0, n-k): (1-h[i+k])
        # d/dh[i] for i in [k, n): -h[i-k]
        g[:n-k] += (1.0 - h[k:])
        g[k:] += -h[:n-k]
    else:

        # overlap = sum h[i+k]*(1-h[i]) for i in [0, n-k)
        # same as above but roles swapped
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] += -h[k:]
    return dx * g

def _proj(v, S):
