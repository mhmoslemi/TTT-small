    n = len(h)

    return b

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(30):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-12:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _pn(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.zeros(n + 1)
    np.cumsum(h, out=pref[1:])
    tot = pref[n]
    P = np.zeros(n)
    Nv = np.zeros(n)
    P[0] = tot - ac[0]
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = tot - pref[idx] - ac[idx]
        mx = float(max(np.max(P), np.max(Nv)))
    else:
        mx = float(P[0])
    return P, Nv, mx

        e += np.exp(np.clip(a * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(max(e, 1e-300)) / a)

    wN = np.zeros(n)

        en = np.exp(np.clip(a * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0:

        wP[0] = e0 / tot
        wP[1:] = ep / tot
        wN[1:] = en / tot

    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

            k = im - n + 1
            m = abs(k)
            if 0 < m < n:
                if k > 0:
                    v = np.where((h[:n-m] == 1) & (h[m:] == 0))[0]
                else:
                    v = np.where((h[m:] == 0) & (h[:n-m] == 1))[0] + m
                if len(v):
                    i = int(v[rng.randint(len(v))])
                else:
                    i = int(op[rng.randint(len(op))])
            else:
                i = int(op[rng.randint(len(op))])
            j = int(zp[rng.randint(len(zp))])
        elif r < 0.36 and n > 20:
            a = rng.randint(n - 5)
            L = rng.randint(2, min(35, n - a))
            s = h[a:a+L].copy()
            h[a:a+L] = s[::-1]
            nc = _c5(h)
            d = nc - cc
            if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):
                cc = nc
                if nc < bc:
                    bc, bh, stale = nc, h.copy(), 0
                else:
                    stale += 1
                op = np.where(h == 1)[0]
                zp = np.where(h == 0)[0]
            else:
                h[a:a+L] = s
            continue
        else:
            i = int(op[rng.randint(len(op))])
            j = int(zp[rng.randint(len(zp))])
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        nc = _c5(h)
        d = nc - cc
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):
            cc = nc
            if nc < bc:
                bc, bh, stale = nc, h.copy(), 0
            else:
                stale += 1
            op[op == i] = j
            zp[zp == j] = i
        else:
            h[i] = 1.0
            h[j] = 0.0
        if stale > 60000:
            h = bh.copy()
            cc = bc
            op = np.where(h == 1)[0].copy()
            zp = np.where(h == 0)[0].copy()
            stale = 0
    return bh, _c5(bh)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    te = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n = len(h0)
    S = int(n * 0.5)

    starts = [_to_binary(h0, n)]
    # Bit-reversal structured
    hs = np.zeros(n)
    for i in range(n):
        t = i
        rv = 0
        for _ in range(n.bit_length()):
            rv = (rv << 1) | (t & 1)
            t >>= 1
        hs[i] = 1.0 if rv % 2 == 0 else 0.0
    starts.append(_to_binary(hs, n))
    # Random
    hr = np.zeros(n)
    hr[rng.choice(n, S, replace=False)] = 1.0
    starts.append(hr)
    # Fibonacci
    hf = np.zeros(n)
    phi = 0.618033988749895
    for i in range(n):
        hf[i] = 1.0 if ((i + 1) * phi) % 1.0 < 0.5 else 0.0
    starts.append(_to_binary(hf, n))

    best_h = min(starts, key=_c5).copy()
    best_c = _c5(best_h)

    # Phase 1: FISTA smooth (warm start, 80s)
    t1 = t0 + 80.0
    hf, bf, bc = _fista(best_h, 200, t1 - t0 - 2)
    if bc < best_c:
        best_h, best_c = bf, bc
    hb2 = _to_binary(hf, n)
    c2 = _c5(hb2)
    if c2 < best_c:
        best_h, best_c = hb2, c2

    # Phase 2: FISTA high-alpha (100s)
    t2 = t0 + 180.0
    hf, bf, bc = _fista(hf, 5000, t2 - t0 - 2)
    if bc < best_c:
        best_h, best_c = bf, bc
    hb3 = _to_binary(hf, n)
    c3 = _c5(hb3)
    if c3 < best_c:
        best_h, best_c = hb3, c3

    # Phase 3: SA warm (300s)
    t3 = t0 + 480.0
    best_h, best_c = _sa(best_h, rng, t3, 0.06, 0.01)

    # Phase 4: SA medium (250s)
    t4 = t0 + 730.0
    best_h, best_c = _sa(best_h, rng, t4, 0.01, 0.003)

    # Phase 5: FISTA fine polish (120s)
    t5 = t0 + 850.0
    hfloat = best_h.copy()
    hf, bf, bc = _fista(hfloat, 20000, t5 - t0 - 2)
    if bc < best_c:
        best_h, best_c = bf.copy(), bc
    hb4 = _to_binary(hf, n)
    c4 = _c5(hb4)
    if c4 < best_c:
        best_h, best_c = hb4, c4

    # Phase 6: SA fine (240s)
    best_h, best_c = _sa(best_h, rng, te, 0.003, 0.0003)

    # Final: try FISTA micro
    rem = te - time.time()
    if rem > 20:
        hf, bf, bc = _fista(best_h.copy(), 50000, max(rem - 2, 5))
        if bc < best_c:
            best_h, best_c = bf.copy(), bc

    nc = _c5(best_h)
    if abs(best_h.sum() - S) > 0.01:
        best_h = _proj(best_h, S)
        nc = _c5(best_h)
    return best_h, nc, n
