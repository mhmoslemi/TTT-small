    if s < 1e-12:

        h = np.clip(h, 0.0, 1.0)
        s = h.sum()

                diff -= a
            else:

                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _corr_grad(m, h, n):
    g = np.zeros(n)
    if m >= 0:
        g[m:] += (1.0 - h[:n-m])
        g[:n-m] -= h[m:]
    else:
        p = -m
        g[:n-p] += (1.0 - h[p:])
        g[p:] -= h[:n-p]
    return g

def _lp_step(h, top_m, max_steps=10):
    """Iterative linear programming step."""
    n = len(h)
    c = _corr_full(h)
    dx = 2.0 / n
    # Get top lags
    top_idx = np.argsort(c)[-top_m:]
    # Build linearized constraints
    # Variable: h (length n)
    h_var = cp.Variable(n)
    t = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    for k in top_idx:
        m = k - (n - 1)
        grad = _corr_grad(m, h, n)
        cval = c[k]
        constraints.append(cval + grad @ (h_var - h) <= t)
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h

        return m + np.log(np.exp(s).sum()) / alpha

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _corr_grad(m, h, n):
    g = np.zeros(n)
    if m >= 0:
        g[m:] += (1.0 - h[:n - m])
        g[:n - m] -= h[m:]
    else:
        p = -m
        g[:n - p] += (1.0 - h[p:])
        g[p:] -= h[:n - p]
    return g

def _lp_solve(h, M, solver):
    n = len(h)
    corr = _corr_full(h)
    top = np.argsort(corr)[-M:]
    h_var = cp.Variable(n)
    t = cp.Variable()
    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2]
    for k in top:
        m = k - (n - 1)
        g = _corr_grad(m, h, n)
        cval = corr[k]
        rhs = t - cval + g @ h
        cons.append(g @ h_var <= rhs)
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=solver)
    except Exception:
        return h
    if h_var.value is None:
        return h
    return _repair(h_var.value.flatten().copy(), n)

def _slsqp_solver(h, alpha, ml, maxiter, ftol):
    n = len(h)
    def obj(x):
        c = _corr_full(x)
        m = c.max()
        idx = np.argsort(c)[-ml:]
        s = np.clip(alpha * (c[idx] - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def grad(x):
        c = _corr_full(x)
        m = c.max()
        idx = np.argsort(c)[-ml:]
        s = np.clip(alpha * (c[idx] - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii, k in enumerate(idx):
            if w[ii] < 1e-15: continue
            g += w[ii] * _corr_grad(k - (n - 1), x, n)
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
