                g[M:] -= wk * x[:n-M]

def _p_norm_opt(h, p, m, n, maxiter=500):
    """Optimize p-norm of top-m correlations using L-BFGS-B"""
    dx = 2.0 / n
    target = n * 0.5
    
    def obj(x):
        c = np.correlate(x, 1.0-x, mode='full') * dx
        idx = np.argsort(c)[-m:]
        vals = c[idx]
        # p-norm
        return np.sum(vals**p) ** (1.0/p)
    
    def grad(x):
        c = np.correlate(x, 1.0-x, mode='full') * dx
        idx = np.argsort(c)[-m:]
        vals = c[idx]
        norm_p = np.sum(vals**p) ** (1.0/p)
        g = np.zeros(n)
        if norm_p < 1e-15:
            return g
        for ii, k in enumerate(idx):
            w = (vals[ii]**(p-1)) / norm_p**(p-1)
            L = k - (n-1)
            if L >= 0:
                g[L:] += w * (1.0 - x[:n-L]) * dx
                g[:n-L] -= w * x[L:] * dx
            else:
                M = -L
                g[:n-M] += w * (1.0 - x[M:]) * dx
                g[M:] -= w * x[:n-M] * dx
        return g
    
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12})
        return _repair(res.x, n)
    except:
        return h

            h_new[start:start+B] = 1.0 - h_new[start:start+B]

            cur_c = c_new

        g += 2.0 * penalty * (x.sum() - target)
    return g

def _pnorm_obj(x, p, m, n, target, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-m:]
    vals = c[idx]
    base = np.sum(vals**p) ** (1.0 / p)
    if penalty > 0:
        base += penalty * (x.sum() - target)**2
    return base

def _pnorm_grad(x, p, m, n, target, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-m:]
    vals = c[idx]
    norm_p = np.sum(vals**p) ** (1.0 / p)
    g = np.zeros(n)
    if norm_p > 1e-15:
        for ii, k in enumerate(idx):
            w = (vals[ii]**(p - 1)) / (norm_p**(p - 1))
            L = k - (n - 1)
            if L >= 0:
                g[L:] += w * (1.0 - x[:n - L]) * dx
                g[:n - L] -= w * x[L:] * dx
            else:
                M = -L
                g[:n - M] += w * (1.0 - x[M:]) * dx
                g[M:] -= w * x[:n - M] * dx
    if penalty > 0:
        g += 2.0 * penalty * (x.sum() - target)
    return g

def _opt_lbfgs(x0, obj_fn, grad_fn, args, n, maxiter):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, x0, jac=grad_fn, args=args, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-10})
        return _repair(res.x, n)
    except:
        return x0

def _opt_slsqp(x0, obj_fn, grad_fn, args, n, target, maxiter):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (target,)}]
    try:
        res = minimize(obj_fn, x0, jac=grad_fn, args=args, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except:
        return x0

def _sa_search(h, n_it, rng, beta0):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    cur_h = h.copy()
    cur_c = best_c
    target = n * 0.5
    for it in range(n_it):
        frac = it / max(n_it, 1)
        beta = beta0 * (1.0 - 0.99 * frac)
        r = rng.rand()
        if r < 0.25:
            # Random small perturbation
            k = rng.randint(3, 20)
            idx = rng.randint(0, n, size=k)
            h_new = cur_h.copy()
            amp = 0.03 * (1.0 - frac) + 0.005
            h_new[idx] += amp * rng.randn(k)
        elif r < 0.45:
            # Block shift
            B = rng.randint(4, 25)
            start = rng.randint(0, n - B)
            h_new = cur_h.copy()
            block = h_new[start:start + B].copy()
            shift = rng.randint(-B//2, B//2 + 1)
            h_new[start:start + B] = np.roll(block, shift)
        elif r < 0.65:
            # Sinusoidal perturbation
            k = rng.randint(1, 6)
            amp = 0.04 * (1.0 - frac) + 0.005
            phase = rng.rand() * 2.0 * np.pi
            h_new = cur_h + amp * np.sin(2.0 * np.pi * k * np.arange(n) / n + phase)
        elif r < 0.80:
            # Block inversion
            B = rng.randint(5, 30)
            start = rng.randint(0, n - B)
            h_new = cur_h.copy()
            h_new[start:start + B] = 1.0 - h_new[start:start + B]
        else:
            # Local smoothing
            B = rng.randint(3, 10)
            start = rng.randint(0, n - 2 * B)
            h_new = cur_h.copy()
            h_new[start:start + 2*B] = 0.3 * h_new[start:start + 2*B] + \
                0.7 * np.convolve(h_new[start:start + 2*B], np.ones(B)/B, 'valid')[:B]
            # Hmm, this is getting complicated. Let me simplify.
            h_new = cur_h.copy()
            start = rng.randint(0, n - B)
            h_new[start:start+B] = np.convolve(h_new[start:start+B], np.ones(B)/B, mode='same')
        
        h_new = _proj(h_new, target)
        c_new = _eval(h_new)
        if c_new < cur_c:
            cur_h = h_new
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        else:
            # Simulated annealing acceptance
            if beta > 0:
                delta = c_new - cur_c
                if delta < 0.01 and rng.rand() < np.exp(-delta / (beta * 1e-3)):
                    cur_h = h_new
                    cur_c = c_new
    return best_h, best_c

def _block_cd(h, n, n_passes, rng):
    """Block coordinate descent with Nelder-Mead"""
    best_h = h.copy()
    best_c = _eval(h)
    B = 10
    for _pass in range(n_passes):
        for start in range(0, n, B):
            end = min(start + B, n)
            h_block = h[start:end].copy()
            # Try a few random perturbations of the block
            for _trial in range(3):
                amp = 0.02 * rng.rand() + 0.005
                h_trial_block = h_block + amp * rng.randn(end - start)
                h_new = h.copy()
                h_new[start:end] = h_trial_block
                h_new = _repair(h_new, n)
                c_new = _eval(h_new)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                    h_block = h[start:end].copy()
    return best_h, best_c
