def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    S = n * 0.5
    s = h.sum()
    if abs(s - S) < 1e-14:
        return h
    for _ in range(80):
        mask = (h < 1.0) if s < S else (h > 0.0)
        c = mask.sum()
        if c == 0:
            break

        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - S) < 1e-13:
            break
    d = S - h.sum()
    if abs(d) > 1e-13:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a

def _subgrad_step(h, step, S):

    while time.time() < deadline:
        frac = min(1.0, (time.time() - t_start) / total_t)
        T = 0.004 * (0.3) ** (frac * 2.5)
        k_act = int(np.argmax(corr)) - (n - 1)
        ak = abs(k_act)
        if ak > 0 and ak < n and rng.random() < 0.8:
            if k_act > 0:
                pairs_i = np.arange(n - ak)
                contrib = h[pairs_i] * (1.0 - h[pairs_i + ak])
            else:
                pairs_i = np.arange(ak, n)
                contrib = h[pairs_i] * (1.0 - h[pairs_i - ak])
            tc = contrib.sum()
            if tc > 1e-12:
                w = contrib / tc
                cumw = np.cumsum(w)
                r = rng.random()
                idx = min(int(np.searchsorted(cumw, r)), len(pairs_i) - 1)
                i0 = int(pairs_i[idx])
                j0 = i0 + ak if k_act > 0 else i0 - ak
            else:
                i0 = rng.randint(n); j0 = (i0 + ak) % n
            d = (rng.random() - 0.5) * 0.025 * (1.0 - 0.5 * frac)
            if h[i0] > 0.5 and h[j0] < 0.5:
                d = -abs(d)
            h[i0] += d; h[j0] -= d
            h = _proj(h, S)
            c2 = np.correlate(h, 1.0 - h, mode='full')
            mx2 = float(c2.max())
            delta = mx2 - mx
            if delta < -1e-14 or (delta > 0 and rng.random() < np.exp(-delta / max(T, 1e-15))):
                mx = mx2; corr = c2
                cn = mx * (2.0 / n)
                if cn < best_c:
                    best_c = cn; best_h = h.copy()
            else:
                h[i0] -= d; h[j0] += d
                h = _proj(h, S)
        else:
            i = rng.randint(n); j = rng.randint(n)
            if i == j:
                continue
            d = (rng.random() - 0.5) * 0.04
            h[i] += d; h[j] -= d
            h = _proj(h, S)
            c2 = np.correlate(h, 1.0 - h, mode='full')
            mx2 = float(c2.max())
            delta = mx2 - mx
            if delta < -1e-14 or (delta > 0 and rng.random() < np.exp(-delta / max(T, 1e-15))):
                mx = mx2; corr = c2
                cn = mx * (2.0 / n)
                if cn < best_c:
                    best_c = cn; best_h = h.copy()
            else:
                h[i] -= d; h[j] += d
                h = _proj(h, S)
    return best_h, best_c

def _coord_descent(h, rng, deadline):
    n = len(h)
    S = n * 0.5
    h = h.copy()
    c = _c5(h)
    best_c = c; best_h = h.copy()
    t0 = time.time()
    stall = 0
    while time.time() < deadline and stall < 300:
        corr = np.correlate(h, 1.0 - h, mode='full')
        top3 = np.argsort(corr)[::-1][:3]
        improved = False
        for k_idx in top3:
            k = int(k_idx) - (n - 1)
            if k == 0:
                continue
            ak = abs(k)
            if k > 0:
                pairs = np.arange(n - ak)
                ci = h[pairs]; cj = h[pairs + ak]
            else:
                pairs = np.arange(ak, n)
                ci = h[pairs]; cj = h[pairs - ak]
            top_pairs = np.argsort(ci * (1.0 - cj))[::-1][:15]
            for pi in top_pairs:
                i = int(pairs[pi])
                j = i + ak if k > 0 else i - ak
                for d in [0.005, -0.005, 0.01, -0.01, 0.02, -0.02, 0.003, -0.003]:
                    h2 = h.copy()
                    h2[i] += d; h2[j] -= d
                    if h2[i] < 0 or h2[i] > 1 or h2[j] < 0 or h2[j] > 1:
                        continue
                    h2 = _proj(h2, S)
                    c2 = _c5(h2)
                    if c2 < c - 1e-15:
                        h = h2; c = c2; improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if improved:
            stall = 0
            if c < best_c:
                best_c = c; best_h = h.copy()
        else:
            stall += 1
            i = rng.randint(n); j = rng.randint(n)
            if i != j:
                d = (rng.random() - 0.5) * 0.02
                h[i] += d; h[j] -= d
                h = _proj(h, S)
    return best_h, best_c

def _sgd_phase(h, S, deadline):
    n = len(h)
    h = h.copy()
    step = 0.08
    best_c = _c5(h)
    best_h = h.copy()
    last_imp = time.time()
    while time.time() < deadline - 1:
        hn, mx = _subgrad_step(h, step, S)
        c_new = _c5(hn)
        if c_new < best_c:
            best_c = c_new
            best_h = hn.copy()
            last_imp = time.time()
            step = min(step * 1.05, 0.3)
        else:
            step *= 0.92
        h = hn
        if time.time() - last_imp > 15:
            step *= 0.6
            h = _proj(h + np.random.randn(n) * 0.015, S)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)

    n0 = len(h0)
    h0 = _repair(h0, n0)
    best = h0.copy()
    best_c = _c5(best)

    for tf in [lambda x: 1.0 - x, lambda x: x[::-1],
               lambda x: 0.5 + 0.5*(x - x[::-1]),
               lambda x: 0.5 + 0.5*(x + x[::-1] - 1.0)]:
        h2 = _proj(tf(h0.copy()), n0 * 0.5)
        c2 = _c5(h2)
        if c2 < best_c:
            best = h2; best_c = c2

    # Phase A: Subgradient at n0
    endA = t0 + 140.0
    if endA > time.time():
        best, best_c = _sgd_phase(best, n0 * 0.5, endA)

    # Phase B: Multi-resolution exploration
    endB = t0 + 380.0
    if endB > time.time():
        for n_low in [128, 192, 256, 384]:
            if time.time() > endB - 15:
                break
            slot = min(30.0, endB - time.time() - 2)
            if slot < 5:
                break
            hl = _resample(best, n_low)
            hl, cl = _sgd_phase(hl, n_low * 0.5, time.time() + slot * 0.5)
            hl, cl = _sa_weighted(hl, rng, time.time() + slot * 0.4)
            hu = _resample(hl, n0)
            cu = _c5(hu)
            if cu < best_c:
                best = hu; best_c = cu
                best, best_c = _sa_weighted(best, rng, time.time() + 8.0)

    # Phase C: Full-res SGD
    endC = t0 + 580.0
    if endC > time.time():
        best, best_c = _sgd_phase(best, n0 * 0.5, endC)

    # Phase D: Weighted SA
    endD = t0 + 880.0
    if endD > time.time():
        best, best_c = _sa_weighted(best, rng, endD)

    # Phase E: Coordinate descent
    endE = t0 + budget_s - 30.0
    if endE > time.time():
        best, best_c = _coord_descent(best, rng, endE)

    # Phase F: Final SA
    endF = t0 + budget_s - 3.0
    if endF > time.time():
        best, best_c = _sa_weighted(best, rng, endF)

    # Final transforms
    for tf in [lambda x: 1.0 - x, lambda x: x[::-1],
               lambda x: 0.5 + 0.5*(x - x[::-1]),
               lambda x: 0.5 + 0.5*(x + x[::-1] - 1.0)]:
        h2 = _proj(tf(best.copy()), len(best) * 0.5)
        c2 = _c5(h2)
        if c2 < best_c:
            best = h2; best_c = c2

    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
