        if abs(s - target) < 1e-15:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-15:
                break
    return h

def _project(y, n):
    target = n * 0.5
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(y - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-15:
                break
    return h

def _cd_pass(h, rng, bisection_iters=40):
    n = len(h)
    c = _corr(h)
    cmax = 2 * n - 1
    jlist = np.arange(n)
    rng.shuffle(jlist)
    nm1 = n - 1
    for j in jlist:
        j = int(j)
        beta = np.zeros(cmax)
        k1_lo = j
        k1_hi = min(cmax - 1, j + nm1)
        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax - 1, cmax - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
        beta[k2_lo:k2_hi + 1] -= h[j + lags2]
        g = c - beta * h[j]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(bisection_iters):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(g + beta * t1) <= np.max(g + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-15:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

def _pg_gradient(h, w, n):
    """Gradient of weighted sum of correlations."""
    dx = 2.0 / n
    nm1 = n - 1
    cmax = 2 * n - 1
    grad = np.zeros(n)
    for j in range(n):
        k1_lo = j
        k1_hi = min(cmax - 1, j + nm1)
        lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
        grad[j] += w[k1_lo:k1_hi+1] @ (1.0 - h[j - lags1]) * dx
        k2_lo = max(0, nm1 - j)
        k2_hi = min(cmax - 1, cmax - 1 - j)
        lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
        grad[j] -= w[k2_lo:k2_hi+1] @ h[j + lags2] * dx
    return grad
