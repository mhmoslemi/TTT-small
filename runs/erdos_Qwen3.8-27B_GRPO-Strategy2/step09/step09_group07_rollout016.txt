import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _beta_j(h, j, n):
    cmax = 2 * n - 1
    nm1 = n - 1
    beta = np.zeros(cmax)
    k1_lo = j
    k1_hi = min(cmax - 1, j + nm1)
    lags1 = np.arange(k1_lo, k1_hi + 1) - nm1
    beta[k1_lo:k1_hi + 1] += (1.0 - h[j - lags1])
    k2_lo = max(0, nm1 - j)
    k2_hi = min(cmax - 1, cmax - 1 - j)
    lags2 = np.arange(k2_lo, k2_hi + 1) - nm1
    beta[k2_lo:k2_hi + 1] -= h[j + lags2]
    return beta

def _cd_paired_pass(h, rng, n):
    c = _corr(h)
    cmax = 2 * n - 1
    for _ in range(n):
        j = rng.randint(n)
        jp = rng.randint(n)
        while jp == j:
            jp = rng.randint(n)
        s = h[j] + h[jp]
        lo = max(0.0, s - 1.0)
        hi = min(1.0, s)
        if hi - lo < 1e-14:
            continue
        bj = _beta_j(h, j, n)
        bjp = _beta_j(h, jp, n)
        bd = bj - bjp
        g = c - bj * h[j] - bjp * h[jp]
        lo_t, hi_t = lo, hi
        for _ in range(22):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(g + bd * t1) <= np.max(g + bd * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        delta = t_new - h[j]
        if abs(delta) > 1e-13:
            c += bd * delta
            h[j] = t_new
            h[jp] = s - t_new
    return h

def _cd_sequential_pass(h, rng, n):
    c = _corr(h)
    cmax = 2 * n - 1
    jlist = np.arange(n)
    rng.shuffle(jlist)
    for j in jlist:
        j = int(j)
        beta = _beta_j(h, j, n)
        g = c - beta * h[j]
        lo_t, hi_t = 0.0, 1.0
        for _ in range(20):
            t1 = lo_t + (hi_t - lo_t) / 3.0
            t2 = hi_t - (hi_t - lo_t) / 3.0
            if np.max(g + beta * t1) <= np.max(g + beta * t2):
                hi_t = t2
            else:
                lo_t = t1
        t_new = 0.5 * (lo_t + hi_t)
        if abs(t_new - h[j]) > 1e-13:
            c += beta * (t_new - h[j])
            h[j] = t_new
    return h

def _proj_sum(h, target):
    def s(tau):
        return np.clip(h - tau, 0.0, 1.0).sum()
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if s(mid) > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    hp = np.clip(h - tau, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - hp)
        else:
            order = np.argsort(hp)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - hp[idx])
                hp[idx] += a
                diff -= a
            else:
                a = min(-diff, hp[idx])
                hp[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return hp

def _grad_lse(h, c, alpha, m_lags):
    n = len(h)
    dx = 2.0 / n
    nm1 = n - 1
    idx = np.argsort(c)[-m_lags:]
    s = alpha * (c[idx] - c[idx].max())
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        wk = w[ii] * alpha / m_lags
        if wk < 1e-15:
            continue
        wk *= dx
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:
            if L == 0:
                g[:] += wk * (1.0 - 2.0 * h)
            else:
                g[L:] += wk * (1.0 - h[:n - L])
                g[:n - L] -= wk * h[L:]
        else:
            M = -L
            if M == 0:
                g[:] += wk * (1.0 - 2.0 * h)
            else:
                g[M:] += wk * (1.0 - h[:n - M])
                g[:n - M] -= wk * h[M:]
    return g

def _subgrad_phase(h, n, t_end, rng):
    best_h = h.copy()
    best_c5 = _eval(best_h)
    target = n / 2.0
    alpha = 80.0
    m_lags = 15
    lr = 0.004
    stall = 0
    while time.time() < t_end:
        c = _corr(h)
        g = _grad_lse(h, c, alpha, m_lags)
        h_new = _proj_sum(h - lr * g, target)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                h_new = _proj_sum(h + 0.008 * rng.randn(n), target)
                stall = 0
        h = h_new
        alpha *= 1.008
        lr *= 0.9985
        m_lags = min(m_lags + (1 if stall % 100 == 0 else 0), 35)
    return best_h, best_c5

def _slsqp_obj(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _slsqp_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for ii in range(len(idx)):
        wk = w[ii]
        if wk < 1e-15:
            continue
        wk *= dx
        k = int(idx[ii])
        L = k - nm1
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fun(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h
