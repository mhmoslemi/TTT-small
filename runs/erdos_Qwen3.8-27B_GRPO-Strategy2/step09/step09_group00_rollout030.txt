import numpy as np
import time

def _fft_size(n):
    m = 2*n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _repair(h, n):
    t = 0.5 * n
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    for _ in range(50):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-14:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(30):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max()) * (2.0 / n)

def _pn(h):
    """Compute all overlap values P[k] and N[k] and max."""
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n)
    P[0] = total - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
        mx = max(float(np.max(P)), float(np.max(Nv)))
    else:
        mx = float(P[0])
    return P, Nv, mx

    P, Nv, mx = _pn(h)
    n = len(h)
    dx = 2.0 / n
    e = np.exp(np.clip(a * (P[0] - mx), -400, 0))
    if n > 1:
        e += np.exp(np.clip(a * (P[1:] - mx), -400, 0)).sum()
        e += np.exp(np.clip(a * (Nv[1:] - mx), -400, 0)).sum()
    return dx * (mx + np.log(e) / a)

def _grad_weights(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)
        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)
        q = -(R[n - 1:2*n - 1].real + L[:n].real)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

    P, Nv, mx = _pn(h)
    n = len(h)
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(a * (P[0] - mx), -400, 0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(a * (P[1:] - mx), -400, 0))
        en = np.exp(np.clip(a * (Nv[1:] - mx), -400, 0))
        tot += ep.sum() + en.sum()
        if tot <= 0:
            tot = 1.0; e0 = 1.0
        wP[0] = e0 / tot
        wP[1:] = ep / tot
        wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _grad_weights(h, wP, wN)

def _fista(h0, alpha, max_t, L0=None):
    if max_t <= 0:
        h = h0.copy()
        return h, h.copy(), _c5(h)
    n = len(h0)
    S = n * 0.5
    t0 = time.time()
    h = _repair(h0, n)
    hp = h.copy(); y = h.copy(); t = 1.0
    fy = _soft_val(h, alpha); fh = fy
    best_h = h.copy(); best_c = _c5(h)
    stale = 0
    Lc = L0 if L0 else 2.0 * alpha + 1.0
    for it in range(500000):
        if time.time() - t0 > max_t:
            break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp)
            t = tn
            if y.min() < -1e-5 or y.max() > 1.0 + 1e-5 or abs(y.sum() - S) > 1e-8:
                y = _proj(y, S)
            fy = _soft_val(y, alpha)
        else:
            y = h; fy = fh
        g = _soft_grad(y, alpha)
        step = 1.0 / max(Lc, 1e-12)
        accepted = False
        for _ in range(20):
            xn = _proj(y - step * g, S)
            fx = _soft_val(xn, alpha)
            dv = xn - y
            rhs = fy + float(np.dot(g, dv)) + 0.5 * (1.0 / step) * float(np.dot(dv, dv))
            if fx <= rhs + 1e-11:
                accepted = True
                break
            step *= 0.5
        Lc = 1.0 / max(step, 1e-12)
        if accepted:
            hp = h; h = xn; fh = fx; stale = 0
            Lc = max(Lc * 0.9, 1.0)
            if it % 3 == 0:
                c = _c5(h)
                if c < best_c:
                    best_c = c; best_h = h.copy()
        else:
            stale += 1; hp = h; y = h; t = 1.0; fy = fh
            Lc = max(Lc * 0.5, 1.0)
            if stale > 15:
                break
    c = _c5(h)
    if c < best_c:
        best_c = c; best_h = h.copy()
    return h, best_h, best_c

def _subgrad_minimax(h0, max_t):
    """Direct subgradient method on the max of overlaps."""
    n = len(h0)
    S = n * 0.5
    h = _repair(h0, n)
    t0 = time.time()
    best_h = h.copy(); best_c = _c5(h)
    P, Nv, mx = _pn(h)
    # Active set: indices where P or Nv is within tol of mx
    dx = 2.0 / n
    tol = 5.0 * dx
    all_vals = np.concatenate([P, Nv])
    all_idx = np.concatenate([np.arange(n), np.arange(n)])
    # P[0] is at index 0, P[k] at index k (k>=1), N[k] at index n+k (k>=1)
    # Actually let's just track active k values
    eta = 0.01
    for it in range(100000):
        if time.time() - t0 > max_t:
            break
        P, Nv, mx = _pn(h)
        # Determine active set
        act_P = P >= mx - tol
        act_N = Nv >= mx - tol
        wP = act_P.astype(float)
        wN = act_N.astype(float)
        cnt = wP.sum() + wN.sum()
        if cnt < 1:
            wP[0] = 1.0; cnt = 1
        wP /= cnt; wN /= cnt
        g = _grad_weights(h, wP, wN)
        # Barzilai-Borwein or fixed step
        h_new = _proj(h - eta * g, S)
        c_new = _c5(h_new)
        if c_new < best_c:
            best_c = c_new; best_h = h_new.copy()
        # Adaptive step
        if c_new < mx * dx:
            eta *= 1.05
        else:
            eta *= 0.7
            h_new = h  # reject
        h = h_new
        if mx * dx < 0.380:
            break
    return best_h, best_c

def _sa_search(h0, rng, deadline):
    n = len(h0)
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(corr.max())
    best_c = mx * (2.0 / n)
    best_h = h.copy()
    t_start = time.time()
    total_t = max(1.0, deadline - t_start)
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t_start) / total_t)
        T = 3.0 * (0.003) ** frac
        k_act = int(np.argmax(corr)) - (n - 1)
        # Targeted move toward active constraint (70%)
        if rng.random() < 0.7:
            k = abs(k_act)
            if k > 0 and k < n:
                if k_act > 0:
                    valid = (h[:n-k] > 0.005) & (h[k:] < 0.995)
                else:
                    valid = (h[k:] > 0.005) & (h[:n-k] < 0.995)
                if valid.sum() > 2:
                    pos = np.nonzero(valid)[0]
                    i = int(pos[rng.randint(len(pos))])
                    j = i + k if k_act > 0 else i - k
                    if j < 0 or j >= n:
                        j = rng.randint(n)
                    d = (rng.random() - 0.4 * (1 if k_act > 0 else -1)) * 0.05
                    h[i] += d; h[j] -= d
                    c2 = np.correlate(h, 1.0 - h, mode='full')
                    mx2 = float(c2.max())
                    delta = mx2 - mx
                    if delta < -1e-14 or (delta > 0 and rng.random() < np.exp(-delta / max(T, 1e-15))):
                        mx = mx2; corr = c2
                        cn = mx * (2.0 / n)
                        if cn < best_c:
                            best_c = cn; best_h = h.copy()
                    else:
                        h[i] -= d; h[j] += d
                    continue
        # Random pair move
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = (rng.random() - 0.5) * 0.08
        d = max(-h[i], min(d, 1.0 - h[i]))
        d = max(d - h[j], min(d, 1.0 - h[j] - d + h[j]))  # This is getting complicated
        # Simpler:
        d = (rng.random() - 0.5) * 0.08
        hi = h[i] + d
        hj = h[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        h[i] = hi; h[j] = hj
        c2 = np.correlate(h, 1.0 - h, mode='full')
        mx2 = float(c2.max())
        delta = mx2 - mx
        if delta < -1e-14 or (delta > 0 and rng.random() < np.exp(-delta / max(T, 1e-15))):
            mx = mx2; corr = c2
            cn = mx * (2.0 / n)
            if cn < best_c:
                best_c = cn; best_h = h.copy()
        else:
            h[i] = hi - d; h[j] = hj + d
    return best_h, best_c

def _grad_local(h0, rng, deadline):
    n = len(h0)
    h = h0.copy()
    c = _c5(h)
    best_c = c; best_h = h.copy()
    t0 = time.time()
    recompute_every = 512
    it = 0
    while time.time() < deadline:
        if it % recompute_every == 0:
            c = _c5(h)
            if c < best_c:
                best_c = c; best_h = h.copy()
            P, Nv, mx = _pn(h)
            dx = 2.0 / n
            tol = max(2e-8, 10.0 * dx)
            wP = (P >= mx - tol).astype(float)
            wN = (Nv >= mx - tol).astype(float)
            cnt = wP.sum() + wN.sum()
            if cnt > 0:
                wP /= cnt; wN /= cnt
            else:
                wP[0] = 1.0; cnt = 1
            g = _grad_weights(h, wP, wN)
            order = np.argsort(g)
            q = max(8, n // 10)
            low = order[:q]; high = order[-q:]
        frac = min(1.0, (time.time() - t0) / max(1e-9, deadline - t0))
        base = max(5e-6, 5e-5 * (1e-8) ** frac)
        if rng.random() < 0.75:
            i = int(low[rng.randint(len(low))])
            j = int(high[rng.randint(len(high))])
        else:
            i = rng.randint(n); j = rng.randint(n)
        if i != j:
            s = 1 if g[i] <= g[j] else -1
            avail = min(1.0 - h[i], h[j]) if s > 0 else min(h[i], 1.0 - h[j])
            if avail > 1e-14:
                d = s * rng.random() * base * 2.0
                h[i] += d; h[j] -= d
                cn = _c5(h)
                if cn < c:
                    c = cn
                    if c < best_c:
                        best_c = c; best_h = h.copy()
                else:
                    h[i] -= d; h[j] += d
        it += 1
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

                if bc < best_c:
