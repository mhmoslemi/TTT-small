from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    for _ in range(100):
        if abs(s - target) < 1e-13:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_dx(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _corr_grad(h, k_lags):
    """Gradient of sum of correlations at given full-correlation indices, times dx."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in k_lags:
        M = abs(k - (n - 1))
        if M == 0:
            g += dx * (1.0 - 2.0 * h)
        else:
            g[M:] += dx * (1.0 - h[:n - M])
            g[:n - M] -= dx * h[M:]
    return g

def _corr_grad_w(h, k_lags, weights):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for i, k in enumerate(k_lags):
        if weights[i] < 1e-15:
            continue
        w = weights[i] * dx
        M = abs(k - (n - 1))
        if M == 0:
            g += w * (1.0 - 2.0 * h)
        else:
            g[M:] += w * (1.0 - h[:n - M])
            g[:n - M] -= w * h[M:]
    return g

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp(h, alpha, n_lags, maxiter):
    n = len(h)
    target = n * 0.5
    
    def obj(x):
        c = _corr_dx(x)
        m = c.max()
        idx = np.argsort(c)[-n_lags:]
        s = np.clip(alpha * (c[idx] - m), -300, 300)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    
    def grad(x):
        c = _corr_dx(x)
        m = c.max()
        idx = np.argsort(c)[-n_lags:]
        s = np.clip(alpha * (c[idx] - m), -300, 300)
        s -= s.max()
        w = np.exp(s); w /= w.sum()
        return _corr_grad_w(x, idx, w)
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(len(x))}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except:
        return h

def _subgrad(h, n_iter, alpha0, n_lags, eta0, mu=0.85):
    n = len(h)
    v = np.zeros(n)
    h_c = h.copy()

    eta = eta0; alpha = alpha0

        w = np.exp(s); w /= w.sum()

        else:

def _fine_polish(h, n_iter, rng, scale, maxdist):
