            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = ni - d, nj + d

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
