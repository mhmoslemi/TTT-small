from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-11:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-11:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _smooth_obj(x, alpha, ml):
    n = len(x)
    c = _corr_vec(x)
    m = c.max()
    idx = np.argpartition(-c, ml)[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, ml):
    n = len(x)
    c = _corr_vec(x)
    m = c.max()
    idx = np.argpartition(-c, ml)[:ml]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii]
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_constraint(x, half):
    return x.sum() - half

def _sum_jac(x, half):
    return np.ones(len(x))

def _slsqp(h, alpha, ml, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, ml), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 30})
        return _repair(res.x, n)
    except Exception:
        return h

def _lbfgs_aug(x, alpha, ml, lam_sum, half):
    # Augmented Lagrangian: smooth_obj + lam * (sum - half)^2 / 2
    n = len(x)
    base = _smooth_obj(x, alpha, ml)
    s = x.sum() - half
    return base + 0.5 * lam_sum * s * s

def _lbfgs_aug_grad(x, alpha, ml, lam_sum, half):
    n = len(x)
    g = _smooth_grad(x, alpha, ml)
    s = x.sum() - half
    g += lam_sum * s * np.ones(n)
    return g

def _lbfgs_opt(h, alpha, ml, lam_seq, maxiter):
