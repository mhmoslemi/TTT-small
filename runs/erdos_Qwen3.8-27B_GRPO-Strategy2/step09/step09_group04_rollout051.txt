import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-15:
                break
    return h

def _proj_box_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _sobj_all(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sgrad_all(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ki in range(len(w)):
        if w[ki] < 1e-8:
            continue
        wk = w[ki] * dx
        L = ki - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sobj_top(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sgrad_top(x, alpha, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = alpha * (c[idx] - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-8:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _cf(x, hn):
    return x.sum() - hn

def _cj(x, hn):
    return np.ones(len(x))

def _slsqp_top(h, a, ml, n, mi):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n * 0.5,)}]
    try:
        r = minimize(_sobj_top, h.copy(), jac=_sgrad_top, args=(a, ml),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': mi, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return h

def _slsqp_all(h, a, n, mi):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n * 0.5,)}]
    try:
        r = minimize(_sobj_all, h.copy(), jac=_sgrad_all, args=(a,),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': mi, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return h

def _grad_desc(h, n, grad_fn, gargs, steps, s0, mu):
    cc = _eval(h)
    v = np.zeros(n)
    step = s0
    for _ in range(steps):
        g = grad_fn(h, *gargs)
        v = mu * v + (1.0 - mu) * g
        taken = False
        for _ in range(14):
            hn = _proj_box_sum(h - step * v, n * 0.5)
            cn = _eval(hn)
            if cn < cc - 1e-17:
                h, cc, taken = hn, cn, True
                step = min(step * 1.06, s0 * 25.0)
                break
            step *= 0.42
        if not taken:
            step = min(step * 1.2, s0 * 25.0)
    return _repair(h, n)

def _block_perturb(h, n, rng, nblk, blmin, blmax, amp):
    h2 = h.copy()
    for _ in range(nblk):
        bl = rng.randint(blmin, blmax + 1)
        st = rng.randint(0, n - bl)
        d = amp * rng.randn(bl)
        d -= d.mean()
        h2[st:st + bl] += d
    return _repair(np.clip(h2, 0.0, 1.0), n)

def _xchg_perturb(h, n, rng, nsw, amp):
    h2 = h.copy()
    for _ in range(nsw):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, n - 2)) % n
        d = amp * rng.randn()
        if 0 <= h2[i] + d <= 1 and 0 <= h2[j] - d <= 1:
            h2[i] += d
            h2[j] -= d
    return _repair(h2, n)

def _polish2(h, c5, n, rng, npol, scale):
    h2 = h.copy()
    c2 = c5
    for _ in range(npol):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        d = scale * rng.randn()
        ni, nj = h2[i] + d, h2[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h2[i], h2[j] = ni, nj
            cn = _eval(h2)
            if cn < c2:
                c2 = cn
            else:
                h2[i], h2[j] = ni - d, nj + d
    return h2, c2

def _polish3(h, c5, n, rng, npol, scale):
