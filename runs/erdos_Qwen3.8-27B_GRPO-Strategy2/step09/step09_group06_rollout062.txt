import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_max_obj(h, alpha, m, n):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s - s.max(), -500, 500)
    return c.max() + np.log(np.exp(s).sum()) / alpha

def _smooth_max_grad(h, alpha, m, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s - s.max(), -500, 500)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - h[:n - L])
            g[:n - L] -= wk * h[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n - M]
    return g

def _adam_pgd(h, alpha, m_lags, n_steps, lr, t0):
    n = len(h)
    target = n / 2.0
    beta1, beta2, eps = 0.75, 0.9, 1e-8
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    t = 0
    h = h.copy()
    for step in range(n_steps):
        if time.time() - t0 > 5:  # time check every step (rough)
            break
        t += 1
        g = _smooth_max_grad(h, alpha, m_lags, n)
        m1 = beta1 * m1 + (1 - beta1) * g
        m2 = beta2 * m2 + (1 - beta2) * g * g
        m1h = m1 / (1 - beta1**t)
        m2h = m2 / (1 - beta2**t)
        h_new = _proj(h - lr * m1h / (np.sqrt(m2h) + eps), target)
        h = h_new

def _cd_sweep(h, n_passes=5, n_cand=20):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = c.max()
    L = np.arange(-(n - 1), n)
    cands = np.linspace(0, 1, n_cand)
    for _pass in range(n_passes):
        for j0 in range(n):
            old = h[j0]
            term1 = np.zeros(2 * n - 1)
            v1 = (j0 + L >= 0) & (j0 + L < n)
            term1[v1] = 1.0 - h[j0 + L[v1]]
            v2 = (j0 - L >= 0) & (j0 - L < n)
            term2 = np.zeros(2 * n - 1)
            term2[v2] = h[j0 - L[v2]]
            fac = (term1 - term2) * dx
            best_val = old
            best_cm = cmax + 1.0
            for v in cands:
                delta = v - old
                if abs(delta) < 1e-12:
                    continue
                nm = (c + delta * fac).max()
                if nm < best_cm:
                    best_cm = nm
                    best_val = v
            if best_val != old:
                delta = best_val - old
                h[j0] = best_val
                c += delta * fac
                cmax = c.max()
        # Project to maintain constraint
        h = _repair(h, n)
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        cmax = c.max()
    return h

    target = n / 2.0

    if s < 1e-12:

        h = np.clip(h, 0.0, 1.0)

                break

                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _ev(h):

def _rp(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = 0.5 * n
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)

        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - tgt) < 1e-13:
            break

    if abs(d) > 1e-13:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            if abs(d) < 1e-13:
                break

                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a

def _pj(h, tgt):
    lo, hi = -5.0, 5.0
    for _ in range(55):

        if np.clip(h - m, 0.0, 1.0).sum() > tgt:
            lo = m
        else:
            hi = m
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _sg(h, al, ml, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    ix = np.argsort(c)[-ml:]
    s = al * (c[ix] - c.max())
    s = np.clip(s - s.max(), -400.0, 400.0)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)

        else:

        h = _pj(h - lr * m1 / (1.0 - b1 ** t) / (np.sqrt(m2 / (1.0 - b2 ** t)) + 1e-8), tgt)
