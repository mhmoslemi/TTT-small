import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        if s < 1e-12:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                amt = min(diff, 1.0 - h[idx])
                h[idx] += amt
                diff -= amt
            else:
                amt = min(-diff, h[idx])
                h[idx] -= amt
                diff += amt
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lag(h, L):
    """Gradient of correlation at lag L w.r.t. h (scaled by 2/n)."""
    n = len(h)
    g = np.zeros(n)
    if L >= 0:
        g[:n-L] += (1.0 - h[L:])
        g[L:] -= h[:n-L]
    else:
        M = -L
        g[M:] += (1.0 - h[:n-M])
        g[:n-M] -= h[M:]
    return g * (2.0 / n)

def _hinge_obj(x, C, lam, top_k):
    c = _corr(x)
    if top_k < len(c):
        idx = np.argpartition(-c, top_k)[:top_k]
        e = c[idx] - C
    else:
        e = c - C
    e = np.clip(e, 0, None)
    return C + lam * np.dot(e, e)

def _hinge_grad(x, C, lam, top_k):
    n = len(x)
    c = _corr(x)
    dk = len(c)
    idx = np.argpartition(-c, top_k)[:top_k] if top_k < dk else np.arange(dk)

        else:
            M = -L
            g[M:] += wk * (1.0 - x[:n-M])

                h_new, c_new = _momentum_opt(best_h.copy(), C, lam, 80, 300, 0.001, 0.9, lambda: _remain() - 10)
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_new.copy()
            C = max(best_c5 - 0.001, 0.37)

    # Phase 5: Targeted pair search
    if _remain() > 60:
        n_iter = int(min(_remain() * 0.7, 200000))
        best_h, best_c5 = _pair_search(best_h, n_iter, rng, lambda: _remain() - 10)

    # Phase 6: Final fine-tuning
    if _remain() > 20:
        C = best_c5 - 0.0003
        h_new, c_new = _momentum_opt(best_h.copy(), C, 100000, 120, int(min(_remain()*0.5, 1000)), 0.0005, 0.9, lambda: _remain() - 5)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
