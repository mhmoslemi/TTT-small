import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _proj(h, S):
    x = np.array(h, dtype=float, copy=True)
    n = len(x)
    x += (S - x.sum()) / n
    for _ in range(200):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-14:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _resample(h, m):
    n = len(h)
    if n == m:
        return h.copy()
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    return _proj(np.interp(x1, x0, h), m * 0.5)

def _subgrad_dir(h, top_k=5):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(corr.max())
    idx = np.argsort(corr)[-top_k:]
    mid = n - 1
    g = np.zeros(n)
    for k in idx:
        lag = k - mid
        wk = 1.0 / top_k
        if lag == 0:
            g += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += wk * (1.0 - h[:n - lag])
            g[:n - lag] -= wk * h[lag:]
        else:
            la = -lag
            g[:n - la] += wk * (1.0 - h[la:])
            g[la:] -= wk * h[:n - la]
    return g, mx, corr

def _proj_subgrad(h0, t_limit, top_k=5, step0=0.05):
    n = len(h0)
    S = n * 0.5
    t0 = time.time()
    h = _proj(h0, S)
    best_h = h.copy()
    best_c = _c5(h)
    step = step0
    while time.time() - t0 < t_limit:
        g, mx, _ = _subgrad_dir(h, top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        h_new = _proj(h - step * g / gn, S)
        c_new = _c5(h_new)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        if c_new <= _c5(h) + 1e-12:
            h = h_new
            step = min(step * 1.04, 0.3)
        else:
            step *= 0.82
            if step < 1e-10:
                break
    c_final = _c5(h)
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()
    return best_h, best_c
