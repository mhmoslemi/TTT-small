import numpy as np
import numpy.random as npr
import scipy.optimize as optimize

def run(seed=42, budget_s=600, **kwargs):
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 0.5)
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
    
    def objective(h_):
        h_ = np.clip(h_, 0.0, 1.0)
        h_ = h_ * (target_sum / np.sum(h_)) if np.sum(h_) != 0 else h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    def constraint(h_):
        return np.sum(h_) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    def de_optimize():
        N_pop = 3 * n_points
        pop = [np.clip(npr.uniform(0, 1, n_points), 0.0, 1.0) * (target_sum / np.sum(np.clip(npr.uniform(0, 1, n_points), 0.0, 1.0))) for _ in range(N_pop)]
        best_h = h.copy()
        best_c5 = objective(best_h)
        for gen in range(200):
            if gen > 150:
                break
            mutant = np.zeros(n_points)
            a, b, c = npr.choice(range(N_pop), 3, replace=False)
            mutant = pop[a] + 0.8 * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = mutant * (target_sum / np.sum(mutant)) if np.sum(mutant) != 0 else mutant
            for i in range(n_points):
                if npr.rand() < 0.7:
                    mutant[i] = pop[npr.choice(range(N_pop))] [i]
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = mutant * (target_sum / np.sum(mutant)) if np.sum(mutant) != 0 else mutant
            c5 = objective(mutant)
            if c5 < best_c5:
                best_h = mutant.copy()
                best_c5 = c5
            pop.append(mutant)
            if len(pop) > N_pop:
                pop.pop(np.argmax([objective(p) for p in pop]))
        return best_h, best_c5
    
    h_opt, c5_bound = de_optimize()
    h_opt = np.clip(h_opt, 0.0, 1.0)
    h_opt = h_opt * (target_sum / np.sum(h_opt)) if np.sum(h_opt) != 0 else h_opt
    return (h_opt.tolist(), c5_bound, n_points)
