```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = np.roll(h, n_points//4)
        h = np.clip(h, 0, 1)
        h = h * n_points / 2
        h = h / np.sum(h) * n_points / 2
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-5, 'gtol': 1e-5, 'warm_start': True}
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h, objective(h), n_points)
    
    best_h = result.x
    best_c5 = result.fun
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
```
