import numpy as np
import scipy.optimize as optimize
import numpy.random as npr

def run(seed=42, budget_s=600, **kwargs):
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(100, 1.0 / 100)
    h = np.clip(h, 0.0, 1.0)
    h = h * (50.0 / np.sum(h)) if np.sum(h) != 0 else h

    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    pop_size = 50
    max_gen = 200
    mut_factor = 0.5
    cross_rate = 0.7
    best_h = h.copy()
    best_c5 = np.inf

    def objective(h_):
        h_ = np.clip(h_, 0.0, 1.0)
        h_ = h_ * (target_sum / np.sum(h_)) if np.sum(h_) != 0 else h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    def constraint(h_):
        return np.sum(h_) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def de_optimize():
        nonlocal best_h, best_c5
        pop = np.random.rand(pop_size, n_points)
        pop = pop * (target_sum / np.sum(pop, axis=1)[:, None]) if np.sum(pop, axis=1) != 0 else pop
        pop = np.clip(pop, 0.0, 1.0)

        for gen in range(max_gen):
            for i in range(pop_size):
                a, b, c = npr.choice(pop_size, 3, replace=False)
                mutant = pop[a] + mut_factor * (pop[b] - pop[c])
                mutant = np.clip(mutant, 0.0, 1.0)
                mutant = mutant * (target_sum / np.sum(mutant)) if np.sum(mutant) != 0 else mutant
                mask = np.random.rand(n_points) < cross_rate
                trial = np.where(mask, mutant, pop[i])
                trial = np.clip(trial, 0.0, 1.0)
                trial = trial * (target_sum / np.sum(trial)) if np.sum(trial) != 0 else trial
                obj = objective(trial)
                if obj < objective(pop[i]):
                    pop[i] = trial.copy()
            current_h = pop[np.argmin([objective(h_) for h_ in pop])]
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
            if best_c5 < 0.3807:
                break

    start_time = np.datetime64('now')
    while np.datetime64('now') - start_time < np.timedelta64(budget_s, 's'):
        de_optimize()

    h_opt = best_h
    h_opt = np.clip(h_opt, 0.0, 1.0)
    h_opt = h_opt * (target_sum / np.sum(h_opt)) if np.sum(h_opt) != 0 else h_opt
    c5_bound = objective(h_opt)

    return (h_opt.tolist(), c5_bound, n_points)
