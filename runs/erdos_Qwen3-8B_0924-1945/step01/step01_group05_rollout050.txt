import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * 2 - 1
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    result = opt.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-6,
        disp=False
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        c5 = objective(h)
        return (h, c5, n_points)
    
    best_h = result.x
    best_c5 = result.fun
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
