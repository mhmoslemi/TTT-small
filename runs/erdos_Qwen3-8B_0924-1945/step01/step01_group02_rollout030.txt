```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h = h * scale

    def objective(C):
        return C

    def constraint_k(C, k):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return corr[k] * dx - (1 - C)

    start_time = time.time()
    best_c5 = np.inf
    best_h = h.copy()
    best_h_sum = np.sum(best_h)

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        C = cp.Variable()
        constraints = [0 <= cp.Variable(n_points) <= 1.0, cp.sum(cp.Variable(n_points)) == target_sum]
        for k in range(1, n_points):
            h_var = cp.Variable(n_points)
            constraints.append(cp.sum(cp.multiply(h_var, h_var[k:])) * dx >= 1 - C)
        problem = cp.Problem(cp.Minimize(C), constraints)
        try:
            problem.solve(solver=cp.SCS, eps=1e-6, max_iter=100)
            if problem.status == 'optimal':
                current_c5 = C.value
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_var.value
                    best_h_sum = np.sum(best_h)
        except:
            pass
        h = best_h.copy()
        sum_h = best_h_sum
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h = h * scale

    c5_bound = np.max(np.correlate(h, 1.0 - h, mode="full") * dx)
    return h.tolist(), c5_bound, n_points
```
