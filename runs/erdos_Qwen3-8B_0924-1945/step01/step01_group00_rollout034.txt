import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points // 2] = 1
        h = h / np.sum(h) * n_points / 2

    def normalize(h):
        h = np.clip(h, 0, 1)
        return h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def objective(h):
        h = normalize(h)
        return compute_c5(h)

    bounds = [(0, 1) for _ in range(n_points)]
    population_size = 15
    mutation_factor = 0.5
    crossover_prob = 0.9
    max_iter = 100

    start_time = time()
    result = differential_evolution(objective, bounds, strategy='best1bin', 
                                   popsize=population_size, mutation=mutation_factor, 
                                   recombination=crossover_prob, tol=1e-5, 
                                   maxiter=max_iter, seed=seed, polish=False, 
                                   callback=lambda x: None, disp=False)

    best_h = result.x
    best_c5 = compute_c5(best_h)

    if time() - start_time < budget_s:
        for _ in range(50):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = normalize(h)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5

    best_h = normalize(best_h)
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n_points)
