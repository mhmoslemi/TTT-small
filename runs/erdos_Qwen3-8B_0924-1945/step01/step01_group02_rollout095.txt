```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))
    
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h = h * scale

    n_lags = n_points - 1
    A = []
    for k in range(1, n_lags + 1):
        A_k = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = (i + k) % n_points
            A_k[i, j] = dx
            j = (i - k) % n_points
            A_k[i, j] = dx
        A.append(A_k)

    start_time = time.time()
    best_h = h.copy()
    best_c5 = float('inf')
    best_eval_time = 0

    for _ in range(30):
        if time.time() - start_time > budget_s - 10:
            break
        h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
        C = cp.Variable()
        constraints = [
            cp.sum(h_var) == target_sum,
            cp.sum(cp.multiply(h_var, h_var)) >= 1.0 - C,
        ]
        for k in range(n_lags):
            constraints.append(cp.quad_form(h_var, A[k]) >= 1.0 - C)
        prob = cp.Problem(cp.Minimize(C), constraints)
        prob.solve(solver=cp.CVXPY, verbose=False)

        if prob.status == 'optimal':
            h_opt = np.clip(h_var.value, 0.0, 1.0)
            sum_h_opt = np.sum(h_opt)
            scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
            h_opt = h_opt * scale
            c5 = np.max(np.correlate(h_opt, 1.0 - h_opt, mode="full") * dx)
            if c5 < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5
                best_eval_time = time.time() - start_time
                print(f"New best C5: {best_c5:.6f}")

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        h_opt = best_h.copy()
        c5 = np.max(np.correlate(h_opt, 1.0 - h_opt, mode="full") * dx)
        best_c5 = c5

    return best_h.tolist(), best_c5, n_points
```
