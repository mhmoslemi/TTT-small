import numpy as np
import scipy.optimize as optimize
import numpy.random as npr

def run(seed=42, budget_s=600, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 0.5)
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
    
    def objective(h_):
        h_ = np.clip(h_, 0.0, 1.0)
        h_ = h_ * (target_sum / np.sum(h_)) if np.sum(h_) != 0 else h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    def constraint(h_):
        return np.sum(h_) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    def optimize_solution():
        result = optimize.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-6}
        )
        h_opt = result.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        h_opt = h_opt * (target_sum / np.sum(h_opt)) if np.sum(h_opt) != 0 else h_opt
        return h_opt, objective(h_opt)
    
    def generate_initial_population(pop_size):
        pop = []
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h = np.clip(h, 0.0, 1.0)
            h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
            pop.append(h)
        return pop
    
    def differential_evolution():
        pop_size = 50
        mutation_factor = 0.5
        crossover_rate = 0.7
        max_generations = 200
        pop = generate_initial_population(pop_size)
        best_h = pop[0]
        best_c5 = objective(best_h)
        
        for gen in range(max_generations):
            if gen * pop_size * n_points > 600 * 1e6:
                break
            new_pop = []
            for i in range(pop_size):
                a, b, c = npr.choice(pop_size, 3, replace=False)
                mutant = pop[a] + mutation_factor * (pop[b] - pop[c])
                mutant = np.clip(mutant, 0.0, 1.0)
                mutant = mutant * (target_sum / np.sum(mutant)) if np.sum(mutant) != 0 else mutant
                trial = np.zeros_like(mutant)
                for j in range(n_points):
                    if npr.rand() < crossover_rate:
                        trial[j] = mutant[j]
                    else:
                        trial[j] = pop[i][j]
                trial = np.clip(trial, 0.0, 1.0)
                trial = trial * (target_sum / np.sum(trial)) if np.sum(trial) != 0 else trial
                trial_c5 = objective(trial)
                if trial_c5 < objective(pop[i]):
                    new_pop.append(trial)
                else:
                    new_pop.append(pop[i])
            pop = new_pop
            current_best = min(pop, key=objective)
            current_c5 = objective(current_best)
            if current_c5 < best_c5:
                best_h = current_best
                best_c5 = current_c5
        return best_h, best_c5
    
    if 'initial_h_values' in globals():
        result = optimize_solution()
        h_opt, c5_bound = result
        return (h_opt.tolist(), c5_bound, n_points)
    else:
        result = differential_evolution()
        h_opt, c5_bound = result
        return (h_opt.tolist(), c5_bound, n_points)
