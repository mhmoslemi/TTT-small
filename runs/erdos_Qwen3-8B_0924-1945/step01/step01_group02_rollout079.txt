```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        h[::5] = 1.0
        h = h * (target_sum / np.sum(h))

    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    if sum_h != 0:
        h = h * (target_sum / sum_h)

    def build_constraints(n_points, dx):
        constraints = []
        for k in range(1, n_points):
            A = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = (i + k) % n_points
                A[i, j] = dx
            A = A.T @ A
            constraints.append(A)
        return constraints

    constraints = build_constraints(n_points, dx)
    n_constraints = len(constraints)

    h = cp.Variable(n_points)
    C = cp.Variable()

    objective = cp.Minimize(C)
    constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum]
    for i in range(n_constraints):
        constraints.append(cp.quad_form(h, constraints[i]) >= 1 - C)

    problem = cp.Problem(objective, constraints)
    start_time = time.time()
    try:
        problem.solve(solver=cp.SCS, eps=1e-6, max_iter=100)
    except:
        pass

    h_opt = np.clip(h.value, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    if sum_h_opt != 0:
        h_opt = h_opt * (target_sum / sum_h_opt)

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        if sum_h_opt != 0:
            h_opt = h_opt * (target_sum / sum_h_opt)

        def objective(h_flat):
            h = h_flat.reshape(n_points)
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            corr = np.correlate(h, 1.0 - h, mode="full")
            c5 = np.max(corr * dx)
            return c5

        def constraint(h_flat):
            h = h_flat.reshape(n_points)
            sum_h = np.sum(h)
            return sum_h - target_sum

        cons = [{'type': 'eq', 'fun': constraint}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        result = opt.minimize(objective, h_opt.flatten(), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'max iterations': 100})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        if sum_h_opt != 0:
            h_opt = h_opt * (target_sum / sum_h_opt)

    c5_bound = np.max(np.correlate(h_opt, 1.0 - h_opt, mode="full") * dx)
    return h_opt.tolist(), c5_bound, n_points
```
