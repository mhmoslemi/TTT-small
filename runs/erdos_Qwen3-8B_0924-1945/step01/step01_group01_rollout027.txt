import numpy as np
import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def evaluate(h_vec, n_points):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def objective(h_vec):
        return evaluate(h_vec, n_points)
    
    # Initial configuration
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=np.float64)
    else:
        initial_h = np.random.rand(45)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (45 / 2) / np.sum(initial_h)
    
    # Coarse grid optimization
    n_points = 45
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_coarse = opt.differential_evolution(
        objective, bounds, popsize=40, mutation=(0.8, 1.0), 
        recombination=0.9, strategy='best1bin', tol=1e-5, 
        maxiter=300, disp=False, polish=True, seed=seed)
    best_h = result_coarse.x
    best_c5 = result_coarse.fun
    
    # Fine grid refinement
    elapsed = time.time() - start_time
    if elapsed > budget_s - 300:
        print("Time budget exceeded, returning coarse solution")
        final_h = best_h
        final_c5 = best_c5
    else:
        n_points = 90
        dx = 2.0 / n_points
        h_coarse = best_h
        h_fine = np.zeros(n_points)
        indices = np.linspace(0, n_points - 1, n_points, dtype=int)
        for i in range(n_points):
            idx = int(np.round(i * 45 / 90))
            h_fine[i] = h_coarse[idx]
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
        
        bounds_fine = [(0.0, 1.0) for _ in range(n_points)]
        result_fine = opt.differential_evolution(
            objective, bounds_fine, popsize=20, mutation=(1.0, 1.0),
            recombination=0.9, strategy='best1bin', tol=1e-6,
            maxiter=200, disp=False, polish=True, seed=seed)
        best_h_fine = result_fine.x
        best_c5_fine = result_fine.fun
        
        elapsed = time.time() - start_time
        if elapsed > budget_s - 100:
            print("Time budget exceeded, returning fine solution")
            final_h = best_h_fine
            final_c5 = best_c5_fine
        else:
            # Local refinement with L-BFGS-B
            def objective_lbfgs(h_vec):
                return evaluate(h_vec, n_points)
            
            result_local = opt.minimize(
                objective_lbfgs, best_h_fine, method='L-BFGS-B',
                bounds=bounds_fine, tol=1e-6, options={'maxiter': 100})
            final_h = result_local.x
            final_c5 = result_local.fun
    
    # Final feasibility check
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate(final_h, n_points)
    
    return (final_h.tolist(), final_c5, n_points)
