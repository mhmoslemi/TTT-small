import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 90
    h = np.zeros(n_points)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def obj_and_cons(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        n_corr = len(np.correlate(h, 1 - h, mode="full"))
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = []
        A = np.zeros((n_corr, n_points))
        for k in range(n_corr):
            for i in range(n_points):
                j = (i + k) % n_points
                A[k, i] += h[i] * h[j]
        A = A * (2.0 / n_points)
        cons.append({'type': 'ineq', 'fun': lambda x: 1 - A @ x})
        return np.max(A @ h), cons
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        n_corr = len(np.correlate(h, 1 - h, mode="full"))
        return np.max(np.correlate(h, 1 - h, mode="full") * (2.0 / n_points))
    
    def run_qp():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        A = np.zeros((n_points * 2 - 1, n_points))
        for k in range(n_points * 2 - 1):
            for i in range(n_points):
                j = (i + k) % n_points
                A[k, i] += 1
        A = A * (2.0 / n_points)
        c = np.zeros(n_points)
        c[0] = 1
        G = np.zeros((n_points * 2 - 1, n_points))
        for k in range(n_points * 2 - 1):
            for i in range(n_points):
                j = (i + k) % n_points
                G[k, i] += 1
        G = G * (2.0 / n_points)
        h0 = h.copy()
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        try:
            sol = opt.slsqp(objective, h0, bounds=[(0, 1)] * n_points)
            return sol.x
        except:
            return h0
    
    try:
        result = opt.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1)] * n_points, tol=1e-5, options={'maxiter': 500})
        best_h = result.x
        best_c5 = result.fun
    except:
        best_h = run_qp()
        best_c5 = objective(best_h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning initial solution")
        h = initial_h_values if 'initial_h_values' in globals() else h
        best_c5 = objective(h)
    
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points)
