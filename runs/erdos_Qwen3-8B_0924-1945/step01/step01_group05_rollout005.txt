import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 150
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Use a binary Barker code pattern
        barker_lengths = [1, 2, 4, 5, 7, 11, 13, 17, 23, 29, 31, 41]
        barker_len = max([bl for bl in barker_lengths if bl <= n_points])
        barker_code = np.array([1] * barker_len)
        for i in range(1, barker_len):
            if np.abs(np.correlate(barker_code[:i], barker_code[:i], mode='full')[1]) != 0:
                barker_code = np.array([1] * (barker_len - 1))
                break
        # Interpolate to n_points
        h = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, barker_len), barker_code)
        h = (h * n_points / 2).astype(np.float64)
        h = h / np.sum(h) * n_points / 2
    
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-5, 'gtol': 1e-5, 'warm_start': True}
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h, objective(h), n_points)
    
    best_h = result.x
    best_c5 = result.fun
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
