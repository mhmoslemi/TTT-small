import numpy as np
from scipy.optimize import minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    h = np.round(initial_h_values).astype(float)
    ones = np.sum(h)
    if ones != n_points // 2:
        diff = n_points // 2 - ones
        idxs = np.where(h == 0)[0]
        if diff > 0:
            h[idxs[:diff]] = 1
        else:
            h[idxs[-diff:]] = 0
    h = h.astype(float)
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        ones = np.sum(h)
        if abs(ones - n_points / 2) > 1e-6:
            h = h / ones * (n_points / 2)
        return compute_c5(h)
    
    def fitness(h):
        return evaluate_candidate(h)
    
    def objective(x):
        return fitness(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time()
    
    def run_ga():
        pop_size = 30
        elite_size = 5
        mut_rate = 0.1
        cross_rate = 0.8
        max_gen = 100
        best_h = h.copy()
        best_c5 = evaluate_candidate(best_h)
        
        def init_population():
            pop = np.random.rand(pop_size, n_points)
            pop = (pop > 0.5).astype(float)
            counts = np.sum(pop, axis=1)
            for i in range(pop_size):
                if abs(counts[i] - n_points / 2) > 1:
                    diff = n_points // 2 - counts[i]
                    idxs = np.where(pop[i] == 0)[0]
                    if diff > 0:
                        pop[i][idxs[:diff]] = 1
                    else:
                        pop[i][idxs[-diff:]] = 0
            return pop
        
        pop = init_population()
        for gen in range(max_gen):
            if time() - start_time > budget_s:
                break
            fit = np.array([evaluate_candidate(ind) for ind in pop])
            elite_idx = np.argsort(fit)[:elite_size]
            elite = pop[elite_idx]
            new_pop = elite.copy()
            for i in range(pop_size - elite_size):
                p1, p2 = np.random.choice(elite_size, 2, replace=False)
                child = np.copy(pop[elite_idx[p1]])
                if np.random.rand() < cross_rate:
                    split = np.random.randint(1, n_points - 1)
                    child[:split] = pop[elite_idx[p2]][:split]
                mut_mask = np.random.rand(n_points) < mut_rate
                child[mut_mask] = 1 - child[mut_mask]
                counts = np.sum(child)
                if abs(counts - n_points / 2) > 1:
                    diff = n_points // 2 - counts
                    idxs = np.where(child == 0)[0]
                    if diff > 0:
                        child[idxs[:diff]] = 1
                    else:
                        child[idxs[-diff:]] = 0
                new_pop.append(child)
            pop = new_pop
            current_best = pop[np.argmin(fit)]
            current_c5 = evaluate_candidate(current_best)
            if current_c5 < best_c5:
                best_h = current_best.copy()
                best_c5 = current_c5
        return best_h, best_c5
    
    best_h, best_c5 = run_ga()
    
    if time() - start_time < budget_s:
        for _ in range(50):
            if time() - start_time > budget_s:
                break
            h = np.random.rand(n_points)
            h = (h > 0.5).astype(float)
            counts = np.sum(h)
            if abs(counts - n_points / 2) > 1:
                diff = n_points // 2 - counts
                idxs = np.where(h == 0)[0]
                if diff > 0:
                    h[idxs[:diff]] = 1
                else:
                    h[idxs[-diff:]] = 0
            h = h.astype(float)
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    def refine_solution(h):
        h = np.clip(h, 0, 1)
        ones = np.sum(h)
        if abs(ones - n_points / 2) > 1e-6:
            h = h / ones * (n_points / 2)
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            return res.x
        return h
    
    best_h = refine_solution(best_h)
    final_c5 = evaluate_candidate(best_h)
    return (best_h, final_c5, n_points)
