import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        initial_guesses = [np.copy(initial_h_values)]
    else:
        initial_guesses = []

    # Generate additional structured guesses
    for shift in range(3):
        guess = np.zeros(n_points)
        for i in range(n_points):
            if (i + shift) % 5 == 0:
                guess[i] = 1.0
        guess = np.clip(guess, 0, 1)
        guess = guess * target_sum / np.sum(guess)
        initial_guesses.append(guess)

    # Add perturbed and reversed guesses
    for i in range(2):
        guess = np.copy(initial_guesses[0])
        guess += np.random.uniform(-0.1, 0.1, size=n_points)
        guess = np.clip(guess, 0, 1)
        guess = guess * target_sum / np.sum(guess)
        initial_guesses.append(guess)

    # Add reversed guess
    reversed_guess = np.flip(initial_guesses[0])
    reversed_guess = np.clip(reversed_guess, 0, 1)
    reversed_guess = reversed_guess * target_sum / np.sum(reversed_guess)
    initial_guesses.append(reversed_guess)

    best_h = None
    best_c5 = float('inf')

    def objective(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    time_used = 0
    for guess in initial_guesses:
        if time_used >= budget_s:
            break
        x0 = guess.copy()
        result = opt.minimize(
            objective,
            x0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
        )
        if result.success:
            h_vals = result.x.copy()
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals * target_sum / np.sum(h_vals)
            c5 = objective(h_vals)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_vals
        time_used += 100

    # Fallback to last guess if no improvement
    if best_h is None:
        best_h = initial_guesses[-1].copy()
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        best_c5 = objective(best_h)

    return (best_h, best_c5, n_points)
