import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    target_sum = n_points / 2
    
    # Initialize population
    if 'initial_h_values' in globals():
        h_pop = np.array(initial_h_values, dtype=np.float64)
        h_pop = np.clip(h_pop, 0, 1)
        h_pop = h_pop * target_sum / np.sum(h_pop)
        h_pop = np.tile(h_pop, (50, 1))
    else:
        base_pattern = np.zeros(n_points)
        base_pattern[::(n_points // 3 + 1)] = 1
        base_pattern = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(base_pattern)), base_pattern)
        base_pattern = np.clip(base_pattern, 0, 1)
        base_pattern = base_pattern * target_sum / np.sum(base_pattern)
        h_pop = np.tile(base_pattern, (50, 1))
    
    def evaluate(h_flat):
        h = np.clip(h_flat, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    pop_size = 50
    gen_max = 80
    mutation_rate = 0.1
    crossover_rate = 0.8
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    
    start_time = time.time()
    
    for gen in range(gen_max):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate population
        fitness = np.array([evaluate(h) for h in h_pop])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = h_pop[best_idx].copy()
        
        # Select parents
        parents = h_pop[np.argsort(fitness)][:int(pop_size * 0.3)]
        
        # Generate offspring
        offspring = []
        for _ in range(pop_size - len(parents)):
            if np.random.rand() < crossover_rate:
                p1, p2 = np.random.choice(parents, size=2, replace=False)
                child = np.zeros(n_points)
                split = np.random.randint(1, n_points - 1)
                child[:split] = p1[:split]
                child[split:] = p2[split:]
            else:
                child = np.random.rand(n_points)
            
            # Mutate
            if np.random.rand() < mutation_rate:
                idxs = np.random.choice(n_points, size=int(n_points * 0.05), replace=False)
                child[idxs] = np.random.rand(len(idxs))
            
            # Normalize
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            offspring.append(child)
        
        # Replace population
        h_pop = np.vstack([parents, offspring])
        h_pop = h_pop[np.random.rand(pop_size) < 0.5]
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = evaluate(best_h)
    
    return (best_h, final_c5, n_points)
