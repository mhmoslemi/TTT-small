import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = np.copy(initial_h_values)
        h = np.clip(h, 0, 1)
        h /= np.sum(h)
        h *= target_sum
    else:
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        base = 0.5 + 0.5 * np.sin(np.pi * x)
        h = base
        h = np.clip(h, 0, 1)
        h /= np.sum(h)
        h *= target_sum

    def objective(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = objective(best_h)
    time_used = 0

    def objective_wrapper(x):
        nonlocal time_used
        start_time = time_used
        result = objective(x)
        time_used += 1
        if time_used > budget_s:
            raise RuntimeError("Time budget exceeded")
        return result

    def constraint_wrapper(x):
        return constraint(x)

    result = opt.minimize(
        objective_wrapper,
        best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_wrapper},
        options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h /= np.sum(best_h)
    best_h *= target_sum
    best_c5 = objective(best_h)

    return (best_h, best_c5, n_points)
