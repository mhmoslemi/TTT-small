import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 90
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Precompute lags
    lags = []
    dx = 2.0 / n_points
    for i in range(n_points):
        for j in range(n_points):
            k = (i - j) % n_points
            if k <= n_points - 1:
                lags.append((i, j))
    
    # Objective function
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        quad_terms = np.zeros(len(lags))
        for idx, (i, j) in enumerate(lags):
            quad_terms[idx] = h[i] * h[j] * dx
        max_overlap = np.max(quad_terms)
        return max_overlap
    
    # QP formulation
    def qp_formulation(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        quad_terms = np.zeros(len(lags))
        for idx, (i, j) in enumerate(lags):
            quad_terms[idx] = h[i] * h[j] * dx
        return np.max(quad_terms)
    
    # Solve using CVXOPT
    try:
        from cvxopt import matrix, solvers
        solvers.options['show_progress'] = False
        solvers.options['abstol'] = 1e-8
        solvers.options['reltol'] = 1e-8
        solvers.options['feastol'] = 1e-8
        
        c = matrix([0.0] * n_points)
        G = matrix([[-h[i] for i in range(n_points)] for _ in range(n_points)])
        h_vec = matrix([0.0] * n_points)
        for i in range(n_points):
            h_vec[i] = 1.0
        A = matrix([[1.0] * n_points])
        b = matrix([n_points / 2])
        
        # Quadratic constraints
        Q = matrix([[0.0] * n_points for _ in range(n_points)])
        for idx, (i, j) in enumerate(lags):
            Q[i, j] += 1.0 * dx
            Q[j, i] += 1.0 * dx
        
        qp = solvers.qp(c, Q, G, h_vec, A, b)
        if 'status' in qp and qp['status'] == 'optimal':
            best_h = np.array(qp['x']).flatten()
            best_c5 = qp['primal objective']
            elapsed = time.time() - start_time
            if elapsed > budget_s:
                print("Time budget exceeded, returning initial solution")
                h = initial_h_values if 'initial_h_values' in globals() else h
                best_c5 = qp_formulation(h)
                return (h.tolist(), best_c5, n_points)
            return (best_h.tolist(), best_c5, n_points)
    except:
        pass
    
    # Fallback to differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.differential_evolution(objective, bounds, popsize=40, mutation=(0.5, 1.0), 
                                       recombination=0.9, strategy='best1bin', tol=1e-5,
                                       maxiter=200, disp=False, polish=True, seed=seed)
    
    best_h = result.x
    best_c5 = result.fun
    
    # Final check
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning initial solution")
        h = initial_h_values if 'initial_h_values' in globals() else h
        best_c5 = objective(h)
    
    # Final feasibility check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points)
