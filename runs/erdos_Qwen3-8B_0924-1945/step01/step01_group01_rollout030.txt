import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize or load initial configuration
    n_points = 90
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Objective function
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Gradient-based refinement using L-BFGS-B
    def objective_gradient(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_index = np.argmax(correlate_vals)
        lag = max_index - n_points + 1
        if lag < 0:
            lag = 0
        elif lag >= n_points:
            lag = n_points - 1
        grad = np.zeros(n_points)
        for i in range(n_points):
            x = i * 2.0 / n_points
            x_plus_k = x + lag * 2.0 / n_points
            if x_plus_k >= 2.0:
                x_plus_k = 2.0 - 1e-12
            val = 1 - h[int(x_plus_k * n_points / 2.0)]
            grad[i] = val * 2.0 / n_points
        return np.max(correlate_vals), grad
    
    # Bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Function to enforce constraint
    def constraint(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint}
    
    # Optimization with L-BFGS-B
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5, options={'maxiter': 500, 'disp': False})
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Check if we need to refine with higher points
    if time.time() - start_time < budget_s - 100:
        n_points_refine = 120
        h_refine = np.interp(np.linspace(0, 2, n_points_refine), np.linspace(0, 2, n_points), best_h)
        h_refine = np.clip(h_refine, 0, 1)
        h_refine = h_refine * (n_points_refine / 2) / np.sum(h_refine)
        
        def objective_refine(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points_refine / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points_refine)
            return np.max(correlate_vals)
        
        def objective_gradient_refine(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points_refine / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points_refine)
            max_index = np.argmax(correlate_vals)
            lag = max_index - n_points_refine + 1
            if lag < 0:
                lag = 0
            elif lag >= n_points_refine:
                lag = n_points_refine - 1
            grad = np.zeros(n_points_refine)
            for i in range(n_points_refine):
                x = i * 2.0 / n_points_refine
                x_plus_k = x + lag * 2.0 / n_points_refine
                if x_plus_k >= 2.0:
                    x_plus_k = 2.0 - 1e-12
                val = 1 - h[int(x_plus_k * n_points_refine / 2.0)]
                grad[i] = val * 2.0 / n_points_refine
            return np.max(correlate_vals), grad
        
        bounds_refine = [(0.0, 1.0) for _ in range(n_points_refine)]
        
        def constraint_refine(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points_refine / 2) / np.sum(h)
            return np.sum(h) - n_points_refine / 2
        
        cons_refine = {'type': 'eq', 'fun': constraint_refine}
        
        result_refine = opt.minimize(objective_refine, h_refine, method='L-BFGS-B', bounds=bounds_refine, constraints=cons_refine, tol=1e-5, options={'maxiter': 500, 'disp': False})
        best_h = result_refine.x
        best_c5 = objective_refine(best_h)
    
    # Final check
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution so far")
        h = best_h
        best_c5 = objective(best_h)
    
    # Final feasibility check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points)
