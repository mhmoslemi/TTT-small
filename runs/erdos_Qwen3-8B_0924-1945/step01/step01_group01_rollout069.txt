import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 90
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def gradient(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_index = np.argmax(correlate_vals)
        k = max_index - n_points + 1
        if k < 0 or k >= n_points:
            return np.zeros_like(h)
        shift = np.roll(h, k)
        grad = (1 - shift) * (2.0 / n_points)
        return grad
    
    # Optimization using L-BFGS-B
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(
        fun=objective,
        x0=h,
        method='L-BFGS-B',
        jac=gradient,
        bounds=bounds,
        tol=1e-5,
        options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-5, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    # Check if optimization improved
    if np.isnan(best_c5) or best_c5 >= objective(h):
        best_h = h
        best_c5 = objective(h)
    
    # Refine with higher resolution
    n_points = 90
    h = best_h.copy()
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = objective(h)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
    
    return (h.tolist(), final_c5, n_points)
