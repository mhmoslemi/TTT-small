```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_restarts = 20
    early_stop = 5
    best_c5 = float('inf')
    best_h = None

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = (h > 0.5).astype(float)
        h = h * (target_sum / np.sum(h))

    def compute_corr(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def compute_quad_constraints(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        C = compute_corr(h_scaled)
        return C

    def objective_and_constraints(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            scale = 1.0
        else:
            scale = target_sum / sum_h
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        C = np.max(corr * dx)
        return C

    start_time = time.time()
    for restart in range(max_restarts):
        if time.time() - start_time > budget_s:
            break
        h = np.random.rand(n_points)
        h = (h > 0.5).astype(float)
        h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0.0, 1.0)

        h_flat = h.flatten()
        c5 = objective_and_constraints(h_flat)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_flat.copy()

        if restart % 5 == 0:
            h = np.random.normal(0.5, 0.1, n_points)
            h = np.clip(h, 0.0, 1.0)
            h = h * (target_sum / np.sum(h))

        h_flat = h.flatten()
        c5 = objective_and_constraints(h_flat)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_flat.copy()

    if time.time() - start_time > budget_s:
        return best_h.tolist(), best_c5, n_points

    h_flat = best_h
    h = h_flat.reshape(n_points)
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h_scaled = h * scale
    c5 = compute_corr(h_scaled)
    return h_scaled.tolist(), c5, n_points
```
