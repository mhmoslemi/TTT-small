import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial solution (if available, otherwise create a structured pattern)
    n_points = 200
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[::n_points // 3] = 1.0
        h = np.clip(h, 0, 1)
        h = h * n_points / 2 / np.sum(h)
    
    # Ensure sum(h) = n_points / 2
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Genetic algorithm parameters
    pop_size = 50
    max_generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_ratio = 0.2
    mutation_steps = 10
    best_c5 = float('inf')
    best_h = h
    
    # Initial population
    population = [h.copy() for _ in range(pop_size)]
    for ind in population:
        ind = np.clip(ind, 0, 1)
        ind = ind * target_sum / np.sum(ind)
    
    start_time = time.time()
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = []
        for ind in population:
            f = objective(ind)
            fitness.append((f, ind))
        
        # Sort by fitness
        fitness.sort()
        fit_values, fit_population = zip(*fitness)
        best_fitness = fit_values[0]
        best_individual = fit_population[0]
        
        if best_fitness < best_c5:
            best_c5 = best_fitness
            best_h = best_individual
        
        # Select next generation
        new_population = []
        for _ in range(pop_size // 2):
            # Crossover
            if np.random.rand() < crossover_rate:
                p1, p2 = np.random.choice(population, size=2, replace=False)
                child = np.concatenate([p1[:n_points // 2], p2[n_points // 2:]])
            else:
                child = np.random.rand(n_points)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] = np.random.uniform(0, 1)
            
            # Normalize
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        
        # Add elites
        for i in range(int(pop_size * elite_ratio)):
            new_population.append(population[i])
        
        # Replace population
        population = new_population
        
        # Early termination
        if time.time() - start_time > 500:
            mutation_rate += 0.05
        if time.time() - start_time < 200:
            crossover_rate -= 0.1
        
        # Periodic save
        if gen % 10 == 0:
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * target_sum / np.sum(best_h)
    
    # Final checks
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
