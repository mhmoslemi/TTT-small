import numpy as np
import numpy.random as npr
import scipy.optimize as optimize

def run(seed=42, budget_s=600, **kwargs):
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 0.5)
    
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
    
    def objective(h_):
        h_ = np.clip(h_, 0.0, 1.0)
        sum_h = np.sum(h_)
        if sum_h == 0:
            return np.inf
        h_ = h_ * (target_sum / sum_h)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    pop_size = 3 * n_points
    pop = []
    for _ in range(pop_size):
        candidate = np.random.rand(n_points)
        candidate = np.clip(candidate, 0.0, 1.0)
        sum_candidate = np.sum(candidate)
        if sum_candidate == 0:
            candidate = np.full(n_points, 0.5)
        else:
            candidate = candidate * (target_sum / sum_candidate)
        pop.append(candidate)
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    for gen in range(200):
        if gen == 0:
            new_pop = pop.copy()
        else:
            new_pop = []
        
        for i in range(pop_size):
            a, b, c = npr.choice(pop, 3, replace=False)
            mutant = a + 0.8 * (b - c)
            mutant = np.clip(mutant, 0.0, 1.0)
            sum_mutant = np.sum(mutant)
            if sum_mutant == 0:
                mutant = np.full(n_points, 0.5)
            else:
                mutant = mutant * (target_sum / sum_mutant)
            mutant = np.clip(mutant, 0.0, 1.0)
            
            trial = mutant.copy()
            for j in range(n_points):
                if npr.rand() < 0.7:
                    trial[j] = mutant[j]
            
            trial = np.clip(trial, 0.0, 1.0)
            sum_trial = np.sum(trial)
            if sum_trial == 0:
                trial = np.full(n_points, 0.5)
            else:
                trial = trial * (target_sum / sum_trial)
            trial = np.clip(trial, 0.0, 1.0)
            
            c5 = objective(trial)
            if c5 < best_c5 or (np.abs(c5 - best_c5) < 1e-8 and np.random.rand() < 0.1):
                best_c5 = c5
                best_h = trial.copy()
            
            if len(new_pop) < pop_size:
                new_pop.append(trial)
        
        pop = new_pop
    
    best_h = np.clip(best_h, 0.0, 1.0)
    best_sum = np.sum(best_h)
    if best_sum != 0:
        best_h = best_h * (target_sum / best_sum)
    best_h = np.clip(best_h, 0.0, 1.0)
    c5_bound = objective(best_h)
    
    return (best_h.tolist(), c5_bound, n_points)
