```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Random binary vector with about half ones
        h = np.zeros(n_points, dtype=np.float64)
        num_ones = int(np.ceil(n_points / 2))
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        h[indices] = 1.0
        h = h * (target_sum / np.sum(h))

    def compute_c5(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def evaluate_solution(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def construct_quadratic_constraints(n_points, dx, k_max):
        A = []
        for k in range(1, k_max + 1):
            A_k = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    A_k[i][j] = dx
                j = i - k
                if j >= 0:
                    A_k[i][j] = dx
            A.append(A_k)
        return A

    A = construct_quadratic_constraints(n_points, dx, n_points - 1)
    num_constraints = len(A)

    variables = [cp.Variable(n_points), cp.Variable()]
    h_var, C_var = variables

    constraints = [
        cp.sum(h_var) == target_sum,
        0 <= h_var,
        h_var <= 1.0
    ]

    for k in range(num_constraints):
        A_k = A[k]
        quadratic_expr = cp.quad_form(h_var, A_k)
        constraints.append(quadratic_expr >= 1.0 - C_var)

    objective = cp.Minimize(C_var)
    prob = cp.Problem(objective, constraints)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    elapsed = 0
    restarts = 0
    max_restarts = 20

    start_time = time.time()
    while restarts < max_restarts and elapsed < budget_s:
        try:
            prob.solve(solver=cp.ECOS, eps=1e-8, verbose=False)
            if prob.status == 'OPTIMAL':
                h_candidate = h_var.value
                c5_candidate = evaluate_solution(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
                restarts += 1
                elapsed = time.time() - start_time
            else:
                break
        except:
            break

    # Final check and scale
    h_final = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(h_final)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h_final = h_final * scale

    elapsed = time.time() - start_time
    return h_final.tolist(), best_c5, n_points
```
