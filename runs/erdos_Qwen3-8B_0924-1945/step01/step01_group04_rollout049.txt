import numpy as np
import scipy.optimize as optimize
import numpy.random as npr

def run(seed=42, budget_s=600, **kwargs):
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 0.5)
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h

    def objective(h_):
        h_ = np.clip(h_, 0.0, 1.0)
        h_ = h_ * (target_sum / np.sum(h_)) if np.sum(h_) != 0 else h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def de_optimizer():
        np.random.seed(seed)
        pop_size = 3 * n_points
        pop = np.random.rand(pop_size, n_points)
        pop = np.clip(pop, 0.0, 1.0)
        pop = pop * (target_sum / np.sum(pop, axis=1, keepdims=True)) if np.sum(pop, axis=1) != 0 else pop

        best_h = h.copy()
        best_c5 = objective(best_h)
        gen = 0
        max_gen = 200
        F = 0.8
        P_crossover = 0.7

        while gen < max_gen and np.alltrue(np.abs(best_c5 - 0.385) < 1e-4):
            gen += 1
            new_pop = []
            for _ in range(pop_size):
                a, b, c = np.random.choice(pop, 3, replace=False)
                mutant = a + F * (b - c)
                mutant = np.clip(mutant, 0.0, 1.0)
                mutant = mutant * (target_sum / np.sum(mutant)) if np.sum(mutant) != 0 else mutant

                trial = mutant.copy()
                for i in range(n_points):
                    if np.random.rand() < P_crossover:
                        trial[i] = mutant[i]
                trial = np.clip(trial, 0.0, 1.0)
                trial = trial * (target_sum / np.sum(trial)) if np.sum(trial) != 0 else trial

                c5 = objective(trial)
                if c5 < objective(pop[np.random.choice(range(pop_size))]):
                    new_pop.append(trial)
                else:
                    new_pop.append(pop[np.random.choice(range(pop_size))])

            pop = np.array(new_pop)
            current_c5 = objective(pop[np.argmin([objective(h) for h in pop])])
            if current_c5 < best_c5:
                best_h = pop[np.argmin([objective(h) for h in pop])]
                best_c5 = current_c5

        return best_h, best_c5

    h_opt, c5_bound = de_optimizer()
    h_opt = np.clip(h_opt, 0.0, 1.0)
    h_opt = h_opt * (target_sum / np.sum(h_opt)) if np.sum(h_opt) != 0 else h_opt
    return (h_opt.tolist(), c5_bound, n_points)
