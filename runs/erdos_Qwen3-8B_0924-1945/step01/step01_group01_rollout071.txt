import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Function to evaluate C5 and clip/rescale h
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Coarse-grid DE optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_de = opt.differential_evolution(
        objective, bounds, popsize=50, mutation=(0.8, 1.0),
        recombination=0.9, strategy='best1bin', tol=1e-5,
        maxiter=200, disp=False, polish=True, seed=seed
    )
    best_h_de = result_de.x
    best_c5_de = result_de.fun
    
    # Gradient refinement with L-BFGS-B
    def objective_with_gradient(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(correlate_vals)
        k_star = np.argmax(correlate_vals)
        gradient = np.zeros_like(h)
        if k_star < len(h):
            gradient[:-k_star] += (1 - h[k_star:]) * (2.0 / n_points)
        elif k_star >= len(h):
            gradient[k_star - len(h):] += (1 - h[:len(h) - k_star]) * (2.0 / n_points)
        return c5, gradient
    
    result_l_bfgs = opt.minimize(
        lambda x: objective_with_gradient(x)[0],
        best_h_de,
        bounds=bounds,
        method='L-BFGS-B',
        jac=lambda x: objective_with_gradient(x)[1],
        options={'maxiter': 300, 'ftol': 1e-6}
    )
    best_h_l_bfgs = result_l_bfgs.x
    best_c5_l_bfgs = result_l_bfgs.fun
    
    # Multi-scale refinement
    n_points_refined = 90
    refined_h = np.array([best_h_l_bfgs[i % n_points] for i in range(n_points_refined)])
    refined_h = refined_h * (n_points_refined / 2) / np.sum(refined_h)
    refined_h = np.clip(refined_h, 0, 1)
    
    result_de_refined = opt.differential_evolution(
        objective, [(0.0, 1.0) for _ in range(n_points_refined)],
        popsize=30, mutation=(0.9, 1.0), recombination=0.7,
        strategy='best1bin', tol=1e-5, maxiter=100, disp=False, polish=True, seed=seed
    )
    best_h_de_refined = result_de_refined.x
    best_c5_de_refined = result_de_refined.fun
    
    # Final evaluation and fallback
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
        best_h = best_h_l_bfgs
        best_c5 = best_c5_l_bfgs
    
    # Final check and result
    h = np.clip(best_h_l_bfgs, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points)
