import numpy as np
import scipy.optimize as optimize
import numpy.random as npr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    max_restarts = 5
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    best_h = None
    best_c5 = float('inf')
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.random.rand(n_points)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = h0 * (target_sum / np.sum(h0)) if np.sum(h0) != 0 else h0
    
    def objective(h_):
        h_ = np.clip(h_, 0.0, 1.0)
        if np.sum(h_) == 0:
            return float('inf')
        h_ = h_ * (target_sum / np.sum(h_))
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    def constraint(h_):
        return np.sum(h_) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    for _ in range(max_restarts):
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values, dtype=float)
        else:
            h = np.random.rand(n_points)
            h = np.clip(h, 0.0, 1.0)
            h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
        
        result = optimize.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-6}
        )
        
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0.0, 1.0)
            h_opt = h_opt * (target_sum / np.sum(h_opt)) if np.sum(h_opt) != 0 else h_opt
            c5 = objective(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
    
    if best_h is None:
        best_h = h0
        best_c5 = objective(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
