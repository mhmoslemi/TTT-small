```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    h_flat = h.flatten()
    h_cp = cp.Variable(n_points, bounds=(0.0, 1.0))
    constraints = [cp.sum(h_cp) == target_sum]

    max_lag = n_points - 1
    C = cp.Variable()

    for k in range(1, max_lag + 1):
        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            j1 = (i + k) % n_points
            j2 = (i - k + n_points) % n_points
            A[i, j1] = dx
            A[i, j2] = dx
        constraint = cp.quad_form(h_cp, A) >= 1.0 - C
        constraints.append(constraint)

    problem = cp.Problem(cp.Minimize(C), constraints)
    start_time = time.time()

    try:
        problem.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8, max_iter=1000)
    except:
        problem.solve(solver=cp.SCS, abstol=1e-8, reltol=1e-8, max_iter=1000)

    h_opt = h_cp.value
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for k in range(1, max_lag + 1):
            A = np.zeros((n_points, n_points))
            for i in range(n_points):
                j1 = (i + k) % n_points
                j2 = (i - k + n_points) % n_points
                A[i, j1] = dx
                A[i, j2] = dx
            constraint = cp.quad_form(h_cp, A) >= 1.0 - C
            constraints.append(constraint)
        problem = cp.Problem(cp.Minimize(C), constraints)
        try:
            problem.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8, max_iter=1000)
        except:
            problem.solve(solver=cp.SCS, abstol=1e-8, reltol=1e-8, max_iter=1000)
        h_opt = h_cp.value
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale

    c5_bound = np.max(np.correlate(h_opt, 1.0 - h_opt, mode="full")) * dx
    return h_opt.tolist(), c5_bound, n_points
```
