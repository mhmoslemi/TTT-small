import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    n_points = 45

    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Objective function
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Stage 2: Critical Lag Identification and Gradient Refinement
    def gradient_refinement(h_start, n_points):
        h = h_start.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_idx = np.argmax(correlate_vals)
        k = max_idx - n_points + 1
        if k < 0:
            k += n_points
        grad = np.zeros_like(h)
        for i in range(n_points):
            x_idx = i
            k_idx = (x_idx + k) % n_points
            grad[i] = (1 - h[k_idx]) * (2.0 / n_points)
        return grad
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run gradient-based optimization
    result = opt.minimize(objective, h, method='L-BFGS-B', jac=gradient_refinement, bounds=bounds, tol=1e-6, maxiter=100)
    best_h = result.x
    best_c5 = result.fun
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = objective(best_h)
    
    # Stage 3: Multi-Scale Refinement
    n_points_ref = 90
    dx = 2.0 / n_points_ref
    h_ref = best_h.copy()
    h_ref = np.repeat(h_ref, 2)
    h_ref = np.interp(np.linspace(0, 2, n_points_ref), np.linspace(0, 2, n_points), h_ref)
    h_ref = np.clip(h_ref, 0, 1)
    h_ref = h_ref * (n_points_ref / 2) / np.sum(h_ref)
    
    # Run gradient-based optimization on refined grid
    result = opt.minimize(objective, h_ref, method='L-BFGS-B', jac=gradient_refinement, bounds=bounds, tol=1e-6, maxiter=100)
    best_h = result.x
    best_c5 = result.fun
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points_ref / 2) / np.sum(best_h)

    h = np.clip(best_h, 0, 1)
    h = h * (n_points_ref / 2) / np.sum(h)
    final_c5 = objective(h)
    
    # Ensure we do not exceed time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution found")
    
    return (h.tolist(), final_c5, n_points_ref)
