import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.round(h).astype(int)
        h = h / np.sum(h) * n_points / 2
        h = np.clip(h, 0, 1)

    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / total
        return np.clip(h * scale, 0, 1)

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def objective(x):
        h = normalize(x)
        return compute_c5(h)

    bounds = [(0, 1) for _ in range(n_points)]
    result = differential_evolution(objective, bounds, strategy='best1bin', 
                                   popsize=15, mutation=0.5, recombination=0.9,
                                   tol=1e-5, maxiter=100, callback=None, 
                                   polish=True, disp=False, vectorized=True)

    best_h = normalize(result.x)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
