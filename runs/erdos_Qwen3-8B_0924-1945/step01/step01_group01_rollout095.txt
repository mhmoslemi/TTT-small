import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def gradient_descent(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_index = np.argmax(correlate_vals)
        if max_index >= len(h):
            max_index -= len(h)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            x = i * (2.0 / n_points)
            k = max_index
            if k >= 0:
                if x + k < 2.0:
                    grad[i] = (1 - h[int((x + k) * n_points / 2)])
        return grad
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    # Gradient-based refinement
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(objective, h, method='L-BFGS-B', jac=gradient_descent,
                          bounds=bounds, tol=1e-6, maxiter=100)
    best_h = result.x
    best_c5 = result.fun
    
    # Check feasibility and update
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = objective(best_h)
    
    # Multi-scale refinement
    n_points = 90
    dx = 2.0 / n_points
    h = best_h.copy()
    h = np.resize(h, n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective_high_res(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    result = opt.minimize(objective_high_res, h, method='L-BFGS-B', jac=gradient_descent,
                          bounds=bounds, tol=1e-6, maxiter=100)
    best_h = result.x
    best_c5 = result.fun
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = objective_high_res(best_h)
    
    # Fallback to differential evolution if needed
    def objective_full_res(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    result = opt.differential_evolution(objective_full_res, bounds, popsize=30,
                                       mutation=(0.9, 1.0), recombination=0.7,
                                       strategy='best1bin', tol=1e-5, maxiter=200,
                                       disp=False, polish=True, seed=seed)
    best_h = result.x
    best_c5 = result.fun
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = objective_full_res(best_h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
    
    return (best_h.tolist(), best_c5, n_points)
