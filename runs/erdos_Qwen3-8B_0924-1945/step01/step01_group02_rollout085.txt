```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def construct_A(k):
        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            j1 = (i - k + n_points) % n_points
            j2 = (i + k + n_points) % n_points
            A[i, j1] += dx
            A[i, j2] += dx
        return A

    A_list = [construct_A(k) for k in range(1, n_points)]

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    best_h = h.copy()
    best_c5 = evaluate(h)

    start_time = time.time()
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        h = np.clip(h + np.random.uniform(-0.05, 0.05, size=n_points), 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale

        C = cp.Variable()
        h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
        constraints = [
            cp.sum(h_var) == target_sum,
            cp.sum(cp.multiply(h_var, h_var)) >= 1.0 - C,
        ]
        for A in A_list:
            constraints.append(cp.quad_form(h_var, A) >= 1.0 - C)
        obj = cp.Minimize(C)
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-8)
            if prob.status == 'OPTIMAL':
                h_opt = h_var.value
                c5_opt = evaluate(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass

    return best_h.tolist(), best_c5, n_points
```
