import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', None)
    if initial_h_values is not None:
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
    else:
        x = np.linspace(0, 2, n_points, endpoint=False)
        h = (1 + np.cos(np.pi * x / 2)) / 2
        h = h * (n_points / 2) / (np.sum(h) * 2 / n_points)

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate(best_h)

    def de_objective(x):
        return evaluate(x)

    def de_constraint(x):
        return np.sum(x) - n_points / 2

    def de_bounds():
        return [(0, 1) for _ in range(n_points)]

    def de_constraints():
        return [{'type': 'eq', 'fun': de_constraint}]

    def de_run():
        result = differential_evolution(
            de_objective,
            bounds=de_bounds(),
            constraints=de_constraints(),
            popsize=20,
            tol=1e-5,
            mutation=(0.8, 1),
            recombination=0.9,
            maxiter=50,
            polish=True,
            disp=False,
            vectorized=True
        )
        return result.x

    def slsqp_refine(x):
        res = minimize(objective, x, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        return res.x

    def run_optimization():
        nonlocal best_h, best_c5
        for _ in range(10):
            if time() - start_time > budget_s:
                break
            candidate = de_run()
            candidate = np.clip(candidate, 0, 1)
            candidate = candidate / np.sum(candidate) * n_points / 2
            current_c5 = compute_c5(candidate)
            if current_c5 < best_c5:
                best_h = candidate.copy()
                best_c5 = current_c5
        for _ in range(5):
            if time() - start_time > budget_s:
                break
            refined = slsqp_refine(best_h)
            refined = np.clip(refined, 0, 1)
            refined = refined / np.sum(refined) * n_points / 2
            current_c5 = compute_c5(refined)
            if current_c5 < best_c5:
                best_h = refined.copy()
                best_c5 = current_c5

    run_optimization()

    for _ in range(10):
        if time() - start_time > budget_s:
            break
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
        current_c5 = evaluate(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n_points)
