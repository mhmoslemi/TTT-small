```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = np.zeros(n_points)
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = (h < 0.5).astype(np.float64)
        h *= target_sum / h.sum()
    h = np.clip(h, 0.0, 1.0)

    def calc_c5(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def build_constraints(h):
        constraints = []
        for k in range(1, n_points):
            A = np.zeros((n_points, n_points))
            for i in range(n_points):
                j1 = (i + k) % n_points
                j2 = (i - k + n_points) % n_points
                A[i, j1] += dx
                A[i, j2] += dx
            cons = cp.bmat([[cp.Variable(n_points, nonneg=True), cp.Variable()]], 
                          [cp.sum(A @ h) >= 1.0 - cp.Variable()])
            constraints.append(cons)
        return constraints

    start_time = time.time()
    best_c5 = calc_c5(h)
    best_h = h.copy()

    for _ in range(20):
        if time.time() - start_time > budget_s - 50:
            break
        h = np.random.rand(n_points)
        h = (h < 0.5).astype(np.float64)
        h *= target_sum / h.sum()
        h = np.clip(h, 0.0, 1.0)

        h_var = cp.Variable(n_points, nonneg=True)
        c_var = cp.Variable()
        constraints = [cp.sum(h_var) == target_sum]
        for k in range(1, n_points):
            A = np.zeros((n_points, n_points))
            for i in range(n_points):
                j1 = (i + k) % n_points
                j2 = (i - k + n_points) % n_points
                A[i, j1] += dx
                A[i, j2] += dx
            constraints.append(cp.sum(A @ h_var) >= 1.0 - c_var)
        obj = cp.Minimize(c_var)
        problem = cp.Problem(obj, constraints)
        try:
            problem.solve(solver=cp.ECOS, eps=1e-8)
            if problem.status == 'OPTIMAL':
                h_opt = h_var.value
                c5 = c_var.value
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
        except:
            pass

    h = best_h
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h))
    c5_bound = calc_c5(h)
    return h.tolist(), c5_bound, n_points
```
