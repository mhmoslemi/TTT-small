import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Feasibility check
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlate_vals * (2.0 / n_points))
        return c5
    
    # Stage 2: Critical lag optimization
    def gradient_refinement(h):
        h = h.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(correlate_vals)
        k = max_idx - n_points + 1
        if k < 0:
            k += n_points
        if k >= n_points:
            k -= n_points
        grad = np.zeros_like(h)
        for i in range(n_points):
            x = i * (2.0 / n_points)
            if x + k >= 2.0:
                val = 0.0
            else:
                val = 1.0 - h[np.argmin(np.abs(np.linspace(0, 2, n_points) - (x + k)))]
            grad[i] = val * (2.0 / n_points)
        return grad
    
    def refine_lagged(h):
        h = h.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        for _ in range(100):
            grad = gradient_refinement(h)
            res = opt.line_search.minimize(
                fun=objective,
                x0=h,
                jac=grad,
                method='L-BFGS-B',
                bounds=[(0.0, 1.0) for _ in range(n_points)]
            )
            if res.success:
                h = res.x
            else:
                break
        return h
    
    h = refine_lagged(h)
    best_c5 = objective(h)
    best_h = h
    
    # Stage 3: Refinement with double the resolution
    n_points = 90
    dx = 2.0 / n_points
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, 45), best_h)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def obj_90(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlate_vals * dx)
        return c5
    
    h = refine_lagged(h)
    best_c5 = min(best_c5, obj_90(h))
    best_h = h
    
    # Stage 4: Fallback to differential evolution
    if time.time() - start_time > budget_s * 0.5:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(obj_90, bounds, popsize=30, mutation=(0.9, 1.0), 
                                           recombination=0.7, strategy='best1bin', tol=1e-5,
                                           maxiter=200, disp=False, polish=True, seed=seed)
        best_h = result.x
        best_c5 = result.fun
    
    # Final check and return
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = obj_90(h)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
    
    return (h.tolist(), final_c5, n_points)
