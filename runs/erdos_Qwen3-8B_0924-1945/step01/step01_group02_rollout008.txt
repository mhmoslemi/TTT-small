```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        num_ones = int(np.ceil(n_points / 2))
        indices = np.random.choice(n_points, num_ones, replace=False)
        h[indices] = 1.0
        h = h * (target_sum / np.sum(h))
    
    def evaluate_c5(h):
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    def construct_qp(h0):
        h = cp.Variable(n_points)
        C = cp.Variable()
        constraints = []
        constraints.append(cp.sum(h) * dx == 1.0)
        constraints.append(h >= 0.0)
        constraints.append(h <= 1.0)
        for k in range(1, n_points):
            A = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = (i + k) % n_points
                A[i, j] = dx
                j = (i - k + n_points) % n_points
                A[i, j] = dx
            constraints.append(h @ A @ h >= 1.0 - C)
        obj = cp.Minimize(C)
        prob = cp.Problem(obj, constraints)
        try:
            result = prob.solve(solver=cp.ECOS, eps=1e-8, verbose=False)
        except:
            result = prob.solve(solver=cp.SCS, max_iter=100, verbose=False)
        if result is None or result not in [cp.OPTIMAL, cp.OPTIMAL_INFEASIBLE]:
            return h0, evaluate_c5(h0)
        return h.value, evaluate_c5(h.value)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    elapsed = 0
    
    start_time = time.time()
    for _ in range(20):
        if elapsed >= budget_s:
            break
        h_start = h.copy()
        h_start += 0.1 * np.random.randn(n_points)
        h_start = np.clip(h_start, 0.0, 1.0)
        h_start = h_start * (target_sum / np.sum(h_start))
        h_opt, c5_opt = construct_qp(h_start)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
        elapsed = time.time() - start_time
    
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    best_h = best_h * scale
    final_c5 = evaluate_c5(best_h)
    
    return best_h.tolist(), final_c5, n_points
```
