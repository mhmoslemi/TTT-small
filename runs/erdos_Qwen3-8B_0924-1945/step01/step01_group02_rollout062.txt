```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h[:int(n_points / 2)] = 1.0
        h = np.clip(h, 0.0, 1.0)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        scale = target_sum / h_sum if h_sum != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h.copy()
    best_c5 = objective(h)
    elapsed_time = 0.0

    start_time = time.time()
    for _ in range(10):
        if elapsed_time >= budget_s:
            break
        result = opt.minimize(objective, h.flatten(), method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        h = result.x.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        scale = target_sum / h_sum if h_sum != 0 else 1.0
        h = h * scale
        c5 = objective(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        elapsed_time = time.time() - start_time

    return best_h.tolist(), best_c5, n_points
```
