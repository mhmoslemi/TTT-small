import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial solution (if available)
    n_points = 200
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Generate a base pattern with evenly spaced 1s and 0s
        base = np.zeros(n_points)
        base[::(n_points // 3 + 1)] = 1
        # Interpolate and normalize
        h = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, len(base)), base)
        h = np.clip(h, 0, 1)
        h = h * n_points / 2 / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Genetic algorithm setup
    population_size = 50
    max_generations = 80
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    population = [h.copy() for _ in range(population_size)]
    
    def mutate(individual):
        idxs = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
        individual[idxs] = np.random.rand(len(idxs))
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        return individual
    
    def crossover(parent1, parent2):
        if np.random.rand() < crossover_rate:
            split = np.random.randint(1, n_points - 1)
            child = np.concatenate((parent1[:split], parent2[split:]))
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            return child
        return parent1
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        # Select top performers
        elite_indices = np.argsort(fitness)[:int(population_size * 0.3)]
        elites = [population[i] for i in elite_indices]
        
        # Generate new population
        new_population = []
        while len(new_population) < population_size:
            # Select two parents
            parent1, parent2 = np.random.choice(elites, 2, replace=False)
            child = crossover(parent1, parent2)
            if np.random.rand() < mutation_rate:
                child = mutate(child)
            new_population.append(child)
        
        population = new_population
        # Update best solution
        for ind in population:
            current_c5 = objective(ind)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = ind.copy()
        
        # Adjust mutation rate based on time remaining
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 500:
            mutation_rate += 0.05
        if remaining < 200:
            crossover_rate -= 0.1
    
    # Final checks
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
