import numpy as np
import scipy.optimize as optimize
import numpy.random as npr

def run(seed=42, budget_s=600, **kwargs):
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.random.rand(100)
        h = h * (99.0 / np.sum(h)) if np.sum(h) != 0 else h
    
    h = np.clip(h, 0.0, 1.0)
    h = h * (99.0 / np.sum(h)) if np.sum(h) != 0 else h
    
    def objective(h_):
        h_ = np.clip(h_, 0.0, 1.0)
        h_ = h_ * (99.0 / np.sum(h_)) if np.sum(h_) != 0 else h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * 2.0 / 100.0
    
    def constraint(h_):
        return np.sum(h_) - 99.0
    
    bounds = [(0.0, 1.0) for _ in range(100)]
    cons = {'type': 'eq', 'fun': constraint}
    
    result = optimize.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-6}
    )
    
    h_opt = result.x
    h_opt = np.clip(h_opt, 0.0, 1.0)
    h_opt = h_opt * (99.0 / np.sum(h_opt)) if np.sum(h_opt) != 0 else h_opt
    c5_bound = objective(h_opt)
    
    return (h_opt.tolist(), c5_bound, 100)
