import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = np.copy(initial_h_values)
    h = np.clip(h, 0, 1)
    h /= np.sum(h)
    h *= target_sum

    def objective(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def optimize_guess(guess):
        guess = np.clip(guess, 0, 1)
        guess /= np.sum(guess)
        guess *= target_sum
        result = opt.minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-6, 'eps': 1e-6}
        )
        result.x = np.clip(result.x, 0, 1)
        result.x /= np.sum(result.x)
        result.x *= target_sum
        return result.x, objective(result.x)

    initial_guesses = []
    initial_guesses.append(h)
    initial_guesses.append(np.roll(h, 10))
    initial_guesses.append(np.random.uniform(0.3, 0.7, n_points))
    initial_guesses.append(np.fliplr([h])[0])
    initial_guesses.append((h + np.roll(h, 10)) / 2)

    best_c5 = float('inf')
    best_h = h

    for guess in initial_guesses:
        h_opt, c5 = optimize_guess(guess)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt

    for _ in range(5):
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        guess = best_h + perturb
        guess = np.clip(guess, 0, 1)
        guess /= np.sum(guess)
        guess *= target_sum
        h_opt, c5 = optimize_guess(guess)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt

    best_h = np.clip(best_h, 0, 1)
    best_h /= np.sum(best_h)
    best_h *= target_sum
    final_c5 = objective(best_h)
    return (best_h, final_c5, n_points)
