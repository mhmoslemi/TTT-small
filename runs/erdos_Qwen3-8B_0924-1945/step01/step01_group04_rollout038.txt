import numpy as np
import scipy.optimize as optimize

def run(seed=42, budget_s=600, **kwargs):
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with a structured binary sequence to minimize autocorrelation
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Greedy binary sequence construction with minimal autocorrelation
        h = np.zeros(n_points, dtype=float)
        ones_needed = int(np.ceil(target_sum))
        for i in range(ones_needed):
            # Place a 1 at the position with minimal overlap
            overlap = np.sum(h) + np.sum(h[:n_points - i] * h[i:])
            idx = np.argmin(overlap)
            h[idx] = 1.0
    
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
    
    def objective(h_):
        h_ = np.clip(h_, 0.0, 1.0)
        h_ = h_ * (target_sum / np.sum(h_)) if np.sum(h_) != 0 else h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    def constraint(h_):
        return np.sum(h_) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    # Optimization parameters
    max_iter = 100
    perturbation_scale = 0.05
    best_c5 = float('inf')
    best_h = h.copy()
    
    for _ in range(5):
        result = optimize.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-8, 'eps': 1e-6}
        )
        
        h_opt = result.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        h_opt = h_opt * (target_sum / np.sum(h_opt)) if np.sum(h_opt) != 0 else h_opt
        
        c5_bound = objective(h_opt)
        
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
        
        # Perturb the solution for diversity
        h = best_h + np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
    
    # Final check and return
    h_final = np.clip(best_h, 0.0, 1.0)
    h_final = h_final * (target_sum / np.sum(h_final)) if np.sum(h_final) != 0 else h_final
    c5_bound = objective(h_final)
    
    return (h_final.tolist(), c5_bound, n_points)
