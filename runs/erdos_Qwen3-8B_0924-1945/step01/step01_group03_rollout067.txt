import numpy as np

import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = np.copy(initial_h_values)
    else:
        x = np.linspace(0, 2, n_points)
        base = 0.5 + 0.5 * np.sin(np.pi * x)
        h = np.clip(base, 0, 1)
        h = h / np.sum(h) * target_sum
    
    def objective(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - target_sum

    def clip_and_scale(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x == 0:
            return x
        scale = target_sum / sum_x
        return x * scale

    def optimize_guess(guess):
        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.differential_evolution(
            objective,
            bounds,
            args=(),
            strategy='best1bin',
            mutation=(0.5, 1),
            recombination=0.8,
            population_size=15,
            atol=1e-6,
            tol=1e-6,
            maxiter=200,
            disp=False,
            polish=True,
            init=guess,
            workers=-1
        )
        result = clip_and_scale(result.x)
        return result

    best_h = h
    best_c5 = objective(best_h)

    guess_variants = []
    x = np.linspace(0, 2, n_points)
    base = 0.5 + 0.5 * np.sin(np.pi * x)
    guess_variants.append(base)
    guess_variants.append(base + 0.15 * np.random.randn(n_points))
    guess_variants.append(base + 0.15 * np.random.randn(n_points))
    guess_variants.append(base[::-1])
    guess_variants.append(0.5 * base + 0.5 * base[::-1])
    guess_variants.append(0.5 + 0.5 * np.cos(np.pi * x))
    guess_variants.append(0.5 + 0.5 * np.abs(np.sin(np.pi * x)))
    guess_variants.append(0.5 + 0.5 * np.sin(2 * np.pi * x))
    guess_variants.append(0.5 + 0.5 * np.sin(3 * np.pi * x))
    guess_variants.append(0.5 + 0.5 * np.sin(2 * np.pi * x / 3))

    for guess in guess_variants:
        guess = np.clip(guess, 0, 1)
        guess = guess / np.sum(guess) * target_sum
        guess = optimize_guess(guess)
        c5 = objective(guess)
        if c5 < best_c5:
            best_h = guess
            best_c5 = c5

    best_h = clip_and_scale(best_h)
    best_c5 = objective(best_h)

    return (best_h, best_c5, n_points)
