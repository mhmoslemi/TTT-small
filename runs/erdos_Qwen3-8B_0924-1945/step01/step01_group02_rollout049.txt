```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        num_ones = int(np.ceil(n_points / 2))
        indices = np.random.choice(n_points, num_ones, replace=False)
        h[indices] = 1.0
        h = h * (target_sum / np.sum(h))

    def compute_C5(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def compute_constraints(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return corr * dx

    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_C5(best_h)

    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        h = np.random.rand(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale

        A_constraints = []
        for k in range(1, n_points):
            A = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = (i + k) % n_points
                A[i, j] = dx
                j = (i - k + n_points) % n_points
                A[i, j] = dx
            A_constraints.append(A)

        variables = cp.Variable(n_points)
        C = cp.Variable(n_points)
        objective = cp.Minimize(C[0])
        constraints = [
            cp.sum(variables) == target_sum,
            cp.constraints.Box(0.0, 1.0, variables),
            cp.sum([cp.quad_form(variables, A) for A in A_constraints]) >= 1 - C[0]
        ]

        try:
            problem = cp.Problem(objective, constraints)
            problem.solve(solver=cp.ECOS, eps=1e-8)
            if problem.status == 'OPTIMAL':
                h_opt = variables.value
                c5 = compute_C5(h_opt)
                if c5 < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5
        except:
            pass

    h_final = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(h_final)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h_final = h_final * scale
    c5_final = compute_C5(h_final)

    return h_final.tolist(), c5_final, n_points
```
