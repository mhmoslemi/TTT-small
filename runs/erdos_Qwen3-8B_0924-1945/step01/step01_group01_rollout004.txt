import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 90
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Objective and gradient function
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def gradient(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_idx = np.argmax(correlate_vals)
        k = max_idx - (n_points - 1)
        if k < 0:
            k = 0
        if k >= n_points:
            k = n_points - 1
        grad = np.zeros(n_points)
        for i in range(n_points):
            x = i * (2.0 / n_points)
            x_k = x + k * (2.0 / n_points)
            if x_k >= 2.0:
                x_k = 2.0
            if x_k < 0.0:
                x_k = 0.0
            idx_k = int(x_k * n_points / 2.0)
            if idx_k >= n_points:
                idx_k = n_points - 1
            if idx_k < 0:
                idx_k = 0
            grad[i] = (1 - h[idx_k]) * (2.0 / n_points)
        return grad
    
    # Optimization using L-BFGS-B
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(objective, h, method='L-BFGS-B', jac=gradient, bounds=bounds, tol=1e-5, maxiter=500)
    best_h = result.x
    best_c5 = result.fun
    
    # Fallback to differential evolution if needed
    if np.isnan(best_c5) or np.isinf(best_c5):
        best_c5 = objective(best_h)
    
    # Grid refinement
    n_points = 120
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    result = opt.minimize(objective, h, method='L-BFGS-B', jac=gradient, bounds=bounds, tol=1e-5, maxiter=500)
    best_h = result.x
    best_c5 = result.fun
    
    # Final feasibility check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    # Time check
    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        print("Time budget exceeded, returning best solution")
    
    return (h.tolist(), final_c5, n_points)
