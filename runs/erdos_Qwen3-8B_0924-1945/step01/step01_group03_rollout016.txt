import numpy as np
import scipy.optimize as opt
import scipy.signal as signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h_values if 'initial_h_values' in locals() else np.zeros(n_points)
    
    def generate_initial_guesses(base_guess):
        guesses = []
        guesses.append(base_guess.copy())
        for phase in range(5):
            guess = np.zeros(n_points)
            interval = n_points // 5
            for i in range(5):
                start = (phase + i * interval) % n_points
                guess[start::interval] = 1.0
            guess = guess.copy()
            guess[guess > 1.0] = 1.0
            guess[guess < 0.0] = 0.0
            guess = guess * target_sum / np.sum(guess)
            guesses.append(guess)
        for _ in range(2):
            noise = np.random.normal(0, 0.1, n_points)
            guess = base_guess + noise
            guess = np.clip(guess, 0, 1)
            guess = guess * target_sum / np.sum(guess)
            guesses.append(guess)
        guesses.append(np.flip(base_guess))
        guesses.append(0.5 * base_guess + 0.5 * np.flip(base_guess))
        return guesses
    
    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = signal.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    def constraint(x):
        return np.sum(x) - target_sum
    
    guesses = generate_initial_guesses(initial_h)
    best_h = np.copy(guesses[0])
    best_c5 = objective(best_h)
    
    for guess in guesses:
        x0 = guess.copy()
        result = opt.minimize(
            objective,
            x0,
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-6, 'eps': 1e-6}
        )
        if result.success:
            x = result.x.copy()
            x = np.clip(x, 0, 1)
            x = x * target_sum / np.sum(x)
            c5 = objective(x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = x
    
    return (best_h, best_c5, n_points)
