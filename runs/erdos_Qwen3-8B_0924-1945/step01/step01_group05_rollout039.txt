import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[::3] = 1.0
        h = np.interp(h, [0, 1], [0, 1])
        h = h * target_sum / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        
        population = [best_h.copy() for _ in range(50)]
        for i in range(50):
            if np.random.rand() < 0.1:
                mutation = np.random.rand(n_points)
                population[i] = mutation * target_sum / np.sum(mutation)
            else:
                if i < 25:
                    parent1 = np.random.choice(population, size=1)[0]
                    parent2 = np.random.choice(population, size=1)[0]
                    crossover = np.random.rand(n_points) < 0.5
                    population[i] = np.where(crossover, parent1, parent2)
                else:
                    population[i] = np.random.rand(n_points)
        
        for i in range(50):
            population[i] = np.clip(population[i], 0, 1)
            population[i] = population[i] * target_sum / np.sum(population[i])
        
        fitness = np.array([objective(h) for h in population])
        top_indices = np.argsort(fitness)[:15]
        best_h = population[top_indices[0]]
        best_c5 = fitness[top_indices[0]]
        
        if gen % 10 == 0:
            if time.time() - start_time > budget_s:
                break
            if np.random.rand() < 0.2:
                best_h = np.random.rand(n_points)
                best_h = best_h * target_sum / np.sum(best_h)
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
