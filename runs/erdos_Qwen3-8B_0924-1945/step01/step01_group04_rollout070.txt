import numpy as np
import scipy.optimize as optimize

def run(seed=42, budget_s=600, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 1.0 / n_points)
    
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
    
    def objective(h_):
        h_ = np.clip(h_, 0.0, 1.0)
        h_ = h_ * (target_sum / np.sum(h_)) if np.sum(h_) != 0 else h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def differential_evolution(population, mutation_factor=0.5, crossover_rate=0.7, max_generations=200):
        best_solution = population[0].copy()
        best_objective = objective(best_solution)
        population = [np.clip(p * (target_sum / np.sum(p)), 0.0, 1.0) if np.sum(p) != 0 else p for p in population]
        population = [p * (target_sum / np.sum(p)) if np.sum(p) != 0 else p for p in population]
        for _ in range(max_generations):
            new_population = []
            for i in range(len(population)):
                a, b, c = np.random.choice(population, 3, replace=False)
                mutant = a + mutation_factor * (b - c)
                mutant = np.clip(mutant, 0.0, 1.0)
                mutant = mutant * (target_sum / np.sum(mutant)) if np.sum(mutant) != 0 else mutant
                trial = np.copy(mutant)
                for j in range(len(trial)):
                    if np.random.rand() < crossover_rate:
                        trial[j] = mutant[j]
                    else:
                        trial[j] = population[i][j]
                trial = np.clip(trial, 0.0, 1.0)
                trial = trial * (target_sum / np.sum(trial)) if np.sum(trial) != 0 else trial
                obj = objective(trial)
                if obj < best_objective:
                    best_solution = trial.copy()
                    best_objective = obj
                new_population.append(trial)
            population = new_population
        return best_solution
    
    population_size = 50
    population = [np.random.rand(n_points) for _ in range(population_size)]
    population = [np.clip(p * (target_sum / np.sum(p)), 0.0, 1.0) if np.sum(p) != 0 else p for p in population]
    population = [p * (target_sum / np.sum(p)) if np.sum(p) != 0 else p for p in population]
    
    best_h = differential_evolution(population)
    
    h_opt = best_h
    h_opt = np.clip(h_opt, 0.0, 1.0)
    h_opt = h_opt * (target_sum / np.sum(h_opt)) if np.sum(h_opt) != 0 else h_opt
    
    c5_bound = objective(h_opt)
    
    return (h_opt.tolist(), c5_bound, n_points)
