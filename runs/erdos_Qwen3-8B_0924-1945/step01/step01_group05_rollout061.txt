import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[::n_points//3] = 1.0
        h = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, len(h)), h)
        h = h * n_points / 2
        h = h / np.sum(h) * n_points / 2

    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h_norm = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    start_time = time.time()
    
    def evolve_population(population):
        offspring = []
        for i in range(population_size):
            if np.random.rand() < crossover_rate:
                parent1 = np.random.choice(population)
                parent2 = np.random.choice(population)
                child = np.copy(parent1)
                idx = np.random.randint(1, n_points - 1)
                child[:idx] = parent2[:idx]
            else:
                child = np.copy(population[np.random.randint(population_size)])
            
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[j] = np.random.rand()
            
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            offspring.append(child)
        
        population.extend(offspring)
        population = np.array(population)
        population = np.vstack([population, np.random.rand(population_size, n_points)])
        population = np.vstack([population, initial_h_values[:population_size] if 'initial_h_values' in globals() else np.random.rand(population_size, n_points)])
        population = population[:population_size]
        return population
    
    population = [h]
    best_h = h
    best_c5 = objective(h)
    
    for gen in range(generations):
        population = evolve_population(population)
        for individual in population:
            c5 = objective(individual)
            if c5 < best_c5:
                best_c5 = c5
                best_h = individual
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        if gen % 10 == 0:
            mutation_rate += 0.05
            if mutation_rate > 0.5:
                mutation_rate = 0.5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
