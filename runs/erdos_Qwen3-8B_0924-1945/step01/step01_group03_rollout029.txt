import numpy as np

import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_guesses = []

    if 'initial_h_values' in locals():
        initial_guess = np.copy(initial_h_values)
        initial_guess = np.clip(initial_guess, 0, 1)
        initial_guess /= np.sum(initial_guess)
        initial_guess *= target_sum
        initial_guesses.append(initial_guess)

    def generate_row_guess():
        step = n_points // 5
        guess = np.zeros(n_points)
        for i in range(n_points):
            if (i % step) == 0:
                guess[i] = 1
        guess = np.clip(guess, 0, 1)
        guess /= np.sum(guess)
        guess *= target_sum
        return guess

    def generate_perturbed_guess(base):
        noise = np.random.uniform(-0.2, 0.2, size=n_points)
        guess = base + noise
        guess = np.clip(guess, 0, 1)
        guess /= np.sum(guess)
        guess *= target_sum
        return guess

    def generate_reversed_guess(base):
        guess = np.flip(base)
        guess = np.clip(guess, 0, 1)
        guess /= np.sum(guess)
        guess *= target_sum
        return guess

    def generate_blended_guess(base, structured):
        guess = 0.5 * base + 0.5 * structured
        guess = np.clip(guess, 0, 1)
        guess /= np.sum(guess)
        guess *= target_sum
        return guess

    initial_guesses.append(generate_row_guess())
    initial_guesses.append(generate_perturbed_guess(initial_guesses[0]))
    initial_guesses.append(generate_reversed_guess(initial_guesses[0]))
    initial_guesses.append(generate_blended_guess(initial_guesses[0], initial_guesses[1]))

    best_h = None
    best_c5 = float('inf')
    max_iterations = 200
    tolerance = 1e-6
    early_stop = 50
    candidates = []

    for guess in initial_guesses:
        x = guess.copy()
        x = np.clip(x, 0, 1)
        x /= np.sum(x)
        x *= target_sum
        result = opt.minimize(
            lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
            x,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
            options={'maxiter': max_iterations, 'ftol': tolerance, 'eps': tolerance}
        )
        if result.success:
            final_x = np.clip(result.x, 0, 1)
            final_x /= np.sum(final_x)
            final_x *= target_sum
            c5 = np.max(np.correlate(final_x, 1 - final_x, mode='full')) * dx
            candidates.append((final_x, c5))
        else:
            candidates.append((x, np.max(np.correlate(x, 1 - x, mode='full')) * dx))

    for c in candidates:
        if c[1] < best_c5:
            best_c5 = c[1]
            best_h = c[0]

    best_h = np.clip(best_h, 0, 1)
    best_h /= np.sum(best_h)
    best_h *= target_sum
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    return (best_h, final_c5, n_points)
