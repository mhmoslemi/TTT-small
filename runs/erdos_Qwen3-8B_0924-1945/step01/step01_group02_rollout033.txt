import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_restarts = 20
    num_evals = 3
    best_c5 = float('inf')
    best_h = None

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        num_ones = int(n_points / 2)
        idxs = np.random.choice(n_points, size=num_ones, replace=False)
        h[idxs] = 1.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    if initial_h_values is None:
        h = np.zeros(n_points, dtype=np.float64)
        num_ones = int(n_points / 2)
        idxs = np.random.choice(n_points, size=num_ones, replace=False)
        h[idxs] = 1.0
        h = h * (target_sum / np.sum(h))

    start_time = time.time()
    for _ in range(max_restarts):
        if time.time() - start_time > budget_s:
            break
        if initial_h_values is not None:
            h = np.array(initial_h_values, dtype=np.float64)
        else:
            h = np.zeros(n_points, dtype=np.float64)
            num_ones = int(n_points / 2)
            idxs = np.random.choice(n_points, size=num_ones, replace=False)
            h[idxs] = 1.0
            h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))

        A = []
        for k in range(1, n_points):
            A_k = np.zeros((n_points, n_points))
            for i in range(n_points):
                j1 = (i - k + n_points) % n_points
                j2 = (i + k) % n_points
                A_k[i, j1] += dx
                A_k[i, j2] += dx
            A.append(A_k)

        h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
        C = cp.Variable()
        cons = []
        cons.append(cp.sum(h_var) == target_sum)
        for k in range(len(A)):
            cons.append(cp.quad_form(h_var, A[k]) >= 1.0 - C)

        prob = cp.Problem(cp.Minimize(C), cons)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-8, verbose=False)
            if prob.status == 'OPTIMAL':
                h_opt = h_var.value
                h_opt = np.clip(h_opt, 0.0, 1.0)
                h_opt = h_opt * (target_sum / np.sum(h_opt))
                if np.sum(h_opt) * dx < 1.0 - 1e-7 or np.sum(h_opt) * dx > 1.0 + 1e-7:
                    h_opt = h_opt * (target_sum / np.sum(h_opt))
                c5 = evaluate(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
        except:
            pass
        if time.time() - start_time > budget_s:
            break

    h = best_h
    c5_bound = evaluate(h)
    return h.tolist(), c5_bound, n_points
