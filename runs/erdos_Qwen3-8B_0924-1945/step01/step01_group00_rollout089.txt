import numpy as np
import scipy.optimize as opt
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    h = initial_h_values.copy()
    
    def normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 1.0 / n_points)
        return h * (n_points / 2) / total

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = normalize(h)
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()

    def run_de():
        pop_size = 20
        pop = np.random.rand(pop_size, n_points)
        pop = normalize(pop)
        best_h = pop[0]
        best_c5 = evaluate_candidate(best_h)
        for gen in range(100):
            if time() - start_time > budget_s:
                break
            new_pop = []
            for i in range(pop_size):
                a, b, c = np.random.choice(pop_size, 3, replace=False)
                mutant = pop[a] + 0.5 * (pop[b] - pop[c])
                mutant = np.clip(mutant, 0, 1)
                mutant = normalize(mutant)
                trial = mutant
                if np.random.rand() < 0.9:
                    trial = pop[i]
                new_pop.append(trial)
            pop = np.array(new_pop)
            for h in pop:
                curr_c5 = evaluate_candidate(h)
                if curr_c5 < best_c5:
                    best_h = h
                    best_c5 = curr_c5
        return best_h, best_c5

    best_h, best_c5 = run_de()
    
    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.rand(n_points)
            h = normalize(h)
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5

    best_h = normalize(best_h)
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
