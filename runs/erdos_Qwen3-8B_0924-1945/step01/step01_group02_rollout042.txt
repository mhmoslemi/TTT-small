```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        h[::5] = 1.0
        h = h * (target_sum / np.sum(h))

    h = np.clip(h, 0.0, 1.0)
    h = h * target_sum / np.sum(h)

    def build_constraints():
        constraints = []
        for k in range(1, n_points):
            A = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = (i + k) % n_points
                A[i, j] = dx
                j = (i - k + n_points) % n_points
                A[i, j] = dx
            constraints.append(A)
        return constraints

    constraints = build_constraints()

    h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
    C = cp.Variable()
    obj = cp.Minimize(C)
    constraints = [
        cp.sum(h_var) == target_sum,
        cp.sum(cp.multiply(h_var, 1.0 - h_var)) <= 1.0 - C
    ]
    for A in constraints:
        constraints.append(cp.quad_form(h_var, A) >= 1.0 - C)

    start_time = time.time()
    prob = cp.Problem(obj, constraints)
    result = prob.solve(solver=cp.ECOS, verbose=False, max_iter=1000)
    elapsed_time = time.time() - start_time

    h_opt = h_var.value
    h_opt = np.clip(h_opt, 0.0, 1.0)
    h_opt = h_opt * target_sum / np.sum(h_opt)

    if elapsed_time < budget_s:
        for k in range(1, n_points):
            A = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = (i + k) % n_points
                A[i, j] = dx
                j = (i - k + n_points) % n_points
                A[i, j] = dx
            constraints.append(cp.quad_form(h_var, A) >= 1.0 - C)
        prob = cp.Problem(obj, constraints)
        result = prob.solve(solver=cp.ECOS, verbose=False, max_iter=1000)
        h_opt = h_var.value
        h_opt = np.clip(h_opt, 0.0, 1.0)
        h_opt = h_opt * target_sum / np.sum(h_opt)

    c5_bound = np.max(np.correlate(h_opt, 1.0 - h_opt, mode="full")) * dx

    return h_opt.tolist(), c5_bound, n_points
```
