import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        idx = np.arange(n_points)
        h[idx % 3 == 0] = 1.0
        h = (h * n_points / 2) / np.sum(h)
    
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    max_generations = 100
    population_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    start_time = time.time()
    population = [h.copy() for _ in range(population_size)]
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        offspring = []
        for _ in range(population_size):
            parent1, parent2 = np.random.choice(population, 2, replace=False)
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:split], parent2[split:]])
            else:
                child = parent1.copy()
            mutation_mask = np.random.rand(n_points) < mutation_rate
            child[mutation_mask] = np.random.rand(np.sum(mutation_mask))
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            offspring.append(child)
        population = np.array(offspring)
        current_c5 = np.array([objective(ind) for ind in population])
        indices = np.argsort(current_c5)
        population = population[indices[:population_size]]
        current_c5 = current_c5[indices[:population_size]]
        if np.min(current_c5) < best_c5:
            best_h = population[np.argmin(current_c5)].copy()
            best_c5 = np.min(current_c5)
        if gen % 10 == 0:
            if time.time() - start_time > budget_s:
                break
            if np.random.rand() < 0.1:
                new_individual = np.random.rand(n_points)
                new_individual = new_individual * target_sum / np.sum(new_individual)
                new_individual = np.clip(new_individual, 0, 1)
                population = np.vstack([population, new_individual])
                population = np.array([population[i] for i in np.argsort([objective(p) for p in population])[:population_size]])
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
