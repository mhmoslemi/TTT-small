import scipy.optimize as optimize

    max_restarts = 5
    best_c5 = float('inf')
    best_h = np.full(n_points, 0.5)

    def objective(h_):
        h_ = np.clip(h_, 0.0, 1.0)
        h_ = h_ * (target_sum / np.sum(h_)) if np.sum(h_) != 0 else h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    def constraint(h_):
        return np.sum(h_) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    for restart in range(max_restarts):
        if 'initial_h_values' in globals() and restart == 0:
            h0 = np.array(initial_h_values, dtype=float)
        else:
            h0 = np.random.rand(n_points)
            h0 = np.clip(h0, 0.0, 1.0)
            h0 = h0 * (target_sum / np.sum(h0)) if np.sum(h0) != 0 else h0
        
        result = optimize.minimize(
            objective,
            h0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-6}
        )
        
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0.0, 1.0)
            h_opt = h_opt * (target_sum / np.sum(h_opt)) if np.sum(h_opt) != 0 else h_opt

            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
