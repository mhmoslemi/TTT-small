import numpy as np
from scipy.optimize import differential_evolution, Bounds
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    
    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h / total * n_points / 2
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def objective(h):
        return compute_c5(h)
    
    bounds = Bounds(0, 1)
    constraints = []
    start_time = time()
    
    def run_optimization():
        best_h = initial_h_values.copy()
        best_c5 = compute_c5(best_h)
        
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.9,
            maxiter=100,
            disp=False,
            polish=True,
            callback=None,
            seed=seed
        )
        
        best_h = result.x
        best_c5 = compute_c5(best_h)
        
        return best_h, best_c5
    
    best_h, best_c5 = run_optimization()
    
    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = normalize(h)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    best_h = normalize(best_h)
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n_points)
