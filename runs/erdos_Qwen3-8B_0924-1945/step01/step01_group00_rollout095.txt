import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 50  # Improved resolution
    initial_h_values = kwargs.get('initial_h_values', None)

    if initial_h_values is not None:
        h = np.clip(initial_h_values, 0, 1)
    else:
        x = np.linspace(0, 2, n_points, endpoint=False)
        h = (1 + np.cos(np.pi * x / 2)) * 0.5
        h = h * (n_points / 2) / (np.sum(h) * 2.0 / n_points)

    dx = 2.0 / n_points

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return np.max(np.correlate(h, 1 - h, mode="full") * dx)

    def constraint(h):
        return np.sum(h) - n_points / 2

    def objective(x):
        return compute_c5(x)

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()

    def normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if np.isclose(total, 0):
            return np.full(n_points, 0.5)
        return h / total * n_points / 2

    def evaluate(h):
        return compute_c5(normalize(h))

    # Global search with differential evolution
    def de_search():
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=20,
            mutation=(0.8, 1.0),
            recombination=0.9,
            tol=1e-6,
            maxiter=50,
            disp=False,
            vectorized=True,
            polish=False
        )
        return normalize(result.x)

    best_h = h.copy()
    best_c5 = evaluate(best_h)

    # Run global search
    for _ in range(2):
        if time() - start_time > budget_s:
            break
        candidate = de_search()
        candidate_c5 = evaluate(candidate)
        if candidate_c5 < best_c5:
            best_h = candidate
            best_c5 = candidate_c5

    # Local refinement with SLSQP
    def refine(h):
        h = h.copy()
        def slsqp_objective(x):
            x = normalize(x)
            return evaluate(x)
        cons = [{'type': 'eq', 'fun': constraint}]
        result = minimize(slsqp_objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        return normalize(result.x)

    for _ in range(5):
        if time() - start_time > budget_s:
            break
        refined = refine(best_h)
        refined_c5 = evaluate(refined)
        if refined_c5 < best_c5:
            best_h = refined
            best_c5 = refined_c5

    # Final evaluation
    best_h = normalize(best_h)
    best_c5 = evaluate(best_h)

    return (best_h, best_c5, n_points)
