import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define parameters
    n_points = 90
    dx = 2.0 / n_points
    initial_h = np.array([], dtype=np.float64)
    
    # Initialize h using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Function to evaluate C5 and ensure constraints
    def evaluate(h_vec):
        h = np.clip(h_vec, 0, 1)
        total = np.sum(h)
        if not np.isclose(total, n_points / 2, atol=1e-6):
            h = h * (n_points / 2) / total
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    # Gradient-based optimization using L-BFGS-B
    def objective(h_vec):
        return evaluate(h_vec)
    
    def gradient(h_vec):
        h = np.clip(h_vec, 0, 1)
        total = np.sum(h)
        h = h * (n_points / 2) / total
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        max_idx = np.argmax(correlate_vals)
        # Compute derivative for the max lag
        k = max_idx - n_points + 1
        if k < 0:
            k += n_points
        derivative = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            x_k = x + k * dx
            if x_k < 2.0:
                derivative[i] = (1 - h[int(x_k / dx)]) * dx
        return derivative
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization stage
    result = opt.minimize(
        objective,
        h,
        method='L-BFGS-B',
        jac=gradient,
        bounds=bounds,
        options={
            'maxiter': 500,
            'ftol': 1e-6,
            'gtol': 1e-5,
            'eps': 1e-8
        }
    )
    
    best_h = result.x
    best_c5 = evaluate(best_h)
    
    # Check time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning initial solution")
        h = initial_h_values if 'initial_h_values' in globals() else h
        best_c5 = evaluate(h)
        best_h = h
    
    # Final feasibility check
    h = np.clip(best_h, 0, 1)
    total = np.sum(h)
    if not np.isclose(total, n_points / 2, atol=1e-6):
        h = h * (n_points / 2) / total
    final_c5 = evaluate(h)
    
    return (h.tolist(), final_c5, n_points)
