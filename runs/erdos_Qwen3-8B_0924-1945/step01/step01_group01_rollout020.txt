import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Stage 2: Coarse-Grid Global Optimization with DE
    def de_optimize(h, n_iter=200):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(
            objective, bounds, popsize=50, mutation=(0.8, 1.0), 
            recombination=0.9, strategy='best1bin', tol=1e-5,
            maxiter=n_iter, disp=False, polish=True, seed=seed
        )
        best_h = result.x
        best_c5 = result.fun
        return best_h, best_c5
    
    best_h, best_c5 = de_optimize(h)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning initial solution")
        h = initial_h_values if 'initial_h_values' in globals() else h
        best_c5 = objective(h)
        return (h.tolist(), best_c5, n_points)
    
    # Stage 3: Adaptive Gradient Refinement with L-BFGS-B
    def compute_gradient(h):
        k_star = np.argmax(np.correlate(h, 1 - h, mode="full"))
        dx = 2.0 / n_points
        grad = np.zeros_like(h)
        for i in range(n_points):
            if (i + k_star) < n_points:
                grad[i] = dx * (1 - h[(i + k_star) % n_points])
        return grad
    
    def grad_objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        k_star = np.argmax(correlate_vals)
        dx = 2.0 / n_points
        grad = np.zeros_like(h)
        for i in range(n_points):
            if (i + k_star) < n_points:
                grad[i] = dx * (1 - h[(i + k_star) % n_points])
        return np.max(correlate_vals), grad
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(
        lambda h_vec: grad_objective(h_vec)[0],
        best_h, method='L-BFGS-B', jac=lambda h_vec: grad_objective(h_vec)[1],
        bounds=bounds, tol=1e-6, maxiter=300, options={'ftol': 1e-6, 'gtol': 1e-6}
    )
    best_h = result.x
    best_c5 = grad_objective(best_h)[0]
    
    # Stage 4: Multi-Scale Refinement with n_points=90
    n_points = 90
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:len(best_h)] = best_h
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    def objective_new(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    bounds_new = [(0.0, 1.0) for _ in range(n_points)]
    result_new = opt.differential_evolution(
        objective_new, bounds_new, popsize=30, mutation=(0.9, 1.0), 
        recombination=0.7, strategy='best1bin', tol=1e-5,
        maxiter=100, disp=False, polish=True, seed=seed
    )
    best_h_new = result_new.x
    best_c5_new = result_new.fun
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, using previous solution")
        h = best_h
        best_c5 = best_c5
        n_points = 45
    else:
        h = best_h_new
        best_c5 = best_c5_new
        n_points = 90
    
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective_new(h)
    
    return (h.tolist(), final_c5, n_points)
