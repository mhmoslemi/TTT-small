import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def objective(C):
        return C

    def constraints(h, C):
        constraints = []
        # Integral constraint
        constraints.append(cp.sum(h) == target_sum)
        # Bounds
        constraints.append(cp.constraints.NonNeg(h))
        constraints.append(cp.constraints.NonPos(-h))
        # Overlap constraints
        for k in range(1, n_points):
            A = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = (i + k) % n_points
                A[i, j] = dx
                j = (i - k) % n_points
                A[i, j] = dx
            overlap = cp.bmat([[h, cp.reshape(cp.multiply(h, h), (n_points, 1))],
                               [cp.reshape(cp.multiply(h, h), (1, n_points)), cp.Constant(0.0)]])

            # Convex constraint: h^T A h >= 1 - C
            # Since A is positive semidefinite, this is convex
            # To implement this, we convert it to a QP constraint
            # (h^T A h) >= 1 - C => C >= 1 - h^T A h
            constraints.append(C >= 1.0 - cp.quad_form(h, A))
        return constraints

    h_vars = cp.Variable(n_points, bounds=(0.0, 1.0))
    C = cp.Variable()

    # Build constraints
    cons = constraints(h_vars, C)

    # Define the optimization problem
    prob = cp.Problem(cp.Minimize(C), cons)

    # Set up solver options
    solver_args = {
        'eps': 1e-6,
        'max_iters': 100,
        'verbose': False
    }

    start_time = time.time()
    result = prob.solve(solver=cp.ECOS, **solver_args)
    elapsed_time = time.time() - start_time

    # Clip and rescale
    h_opt = np.clip(h_vars.value, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale

    # Validate and evaluate
    if elapsed_time < budget_s:
        for _ in range(2):
            start_time = time.time()
            result = prob.solve(solver=cp.ECOS, **solver_args)
            elapsed_time = time.time() - start_time
            if elapsed_time >= budget_s:
                break

    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale

    # Evaluate C5
    corr = np.correlate(h_opt, 1.0 - h_opt, mode="full")
    c5_bound = np.max(corr * dx)

    return h_opt.tolist(), c5_bound, n_points
