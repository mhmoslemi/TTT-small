import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        idx = np.arange(n_points)
        h[idx % 3 == 0] = 1
        h = np.clip(h, 0, 1)
        h = (h * target_sum) / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = (h * target_sum) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def run_ga():
        pop_size = 50
        gen_max = 100
        mut_rate = 0.1
        cross_rate = 0.8
        best_h = h.copy()
        best_c5 = objective(best_h)
        elapsed = 0
        
        for gen in range(gen_max):
            population = np.random.rand(pop_size, n_points)
            population = (population * target_sum) / np.sum(population, axis=1, keepdims=True)
            population = np.clip(population, 0, 1)
            
            for i in range(pop_size):
                if np.random.rand() < mut_rate:
                    mutation = np.random.rand(n_points) * 0.2 - 0.1
                    population[i] += mutation
                    population[i] = np.clip(population[i], 0, 1)
                    population[i] = (population[i] * target_sum) / np.sum(population[i])
            
            for i in range(pop_size):
                if np.random.rand() < cross_rate:
                    j = np.random.randint(pop_size)
                    point = np.random.randint(n_points)
                    population[i], population[j] = population[j].copy(), population[i].copy()
            
            fitness = np.array([objective(ind) for ind in population])
            idx = np.argsort(fitness)[:int(pop_size * 0.3)]
            population = population[idx]
            population = np.vstack([population, np.random.rand(int(pop_size * 0.1), n_points)])

            for i in range(pop_size):
                population[i] = (population[i] * target_sum) / np.sum(population[i])
            
            idx = np.argmin(fitness)
            current_h = population[idx]
            current_c5 = fitness[idx]
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
            
            elapsed = time.time() - time.time()
            if elapsed >= budget_s:
                break
        
        return best_h, best_c5
    
    start_time = time.time()
    best_h, best_c5 = run_ga()

    best_h = (best_h * target_sum) / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
