import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        idx = np.arange(n_points)
        h[idx % 3 == 0] = 1.0
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - (n_points / 2)
    
    bounds = [(0, 1) for _ in range(n_points)]
    max_gen = 100
    pop_size = 50
    mut_rate = 0.1
    cross_rate = 0.8
    best_h = h.copy()
    best_c5 = objective(h)
    time_used = 0
    start_time = time.time()
    
    for gen in range(max_gen):
        if time_used >= budget_s:
            break
        pop = np.random.rand(pop_size, n_points)
        pop = pop * (n_points / 2) / np.sum(pop, axis=1, keepdims=True)
        pop = np.clip(pop, 0, 1)
        for i in range(pop_size):
            if np.random.rand() < mut_rate:
                idx = np.random.choice(n_points, 10)
                pop[i, idx] = np.random.rand(10)
                pop[i] = pop[i] * (n_points / 2) / np.sum(pop[i])
                pop[i] = np.clip(pop[i], 0, 1)
        
        for i in range(pop_size):
            if np.random.rand() < cross_rate and i > 0:
                j = np.random.randint(0, i)
                p1, p2 = pop[i], pop[j]
                crossover_point = np.random.randint(1, n_points - 1)
                pop[i] = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
                pop[i] = pop[i] * (n_points / 2) / np.sum(pop[i])
                pop[i] = np.clip(pop[i], 0, 1)
        
        for i in range(pop_size):
            current_c5 = objective(pop[i])
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = pop[i].copy()
        
        time_used = time.time() - start_time
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = objective(best_h)
    return (best_h, final_c5, n_points)
