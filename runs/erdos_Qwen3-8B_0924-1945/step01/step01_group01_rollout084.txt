import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage 1: Initialization
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Stage 2: Gradient-based refinement
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def gradient(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_idx = np.argmax(correlate_vals)
        # Compute gradient of the max lag
        k = max_idx - n_points + 1
        if k < 0:
            k += n_points
        gradient = np.zeros_like(h)
        for i in range(n_points):
            x = i * (2.0 / n_points)
            if x + k < 2.0:
                gradient[i] = (1 - h[i]) * (2.0 / n_points)
        return gradient
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(objective, h, method='L-BFGS-B', jac=gradient, bounds=bounds, 
                          tol=1e-6, options={'maxiter': 100})
    best_h = result.x
    best_c5 = result.fun
    
    # Stage 3: Multi-scale refinement with higher resolution
    n_points = 90
    dx = 2.0 / n_points
    best_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, 45), best_h)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    def objective_high_res(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    # Stage 4: Fallback to differential evolution if needed
    def fallback_de():
        nonlocal best_h, best_c5
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(objective_high_res, bounds, 
                                           popsize=30, mutation=(0.9, 1.0), 
                                           recombination=0.7, strategy='best1bin',
                                           tol=1e-5, maxiter=200, 
                                           disp=False, seed=seed)
        best_h = result.x
        best_c5 = result.fun
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        final_c5 = objective_high_res(best_h)
        return final_c5
    
    # Stage 5: Final evaluation
    elapsed = time.time() - start_time
    if elapsed > budget_s - 100:
        final_c5 = fallback_de()
    else:
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        final_c5 = objective_high_res(best_h)
    
    # Final checks
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective_high_res(h)
    
    return (h.tolist(), final_c5, n_points)
