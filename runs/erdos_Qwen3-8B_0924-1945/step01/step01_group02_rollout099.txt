```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def objective(C):
        return C

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    def lag_constraints(h_flat, C):
        h = h_flat.reshape(n_points)
        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            for k in range(1, n_points):
                j = (i + k) % n_points
                A[i, j] += dx
                j = (i - k + n_points) % n_points
                A[i, j] += dx
        quad = h @ A @ h
        return quad >= 1.0 - C

    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()

    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        h = np.random.rand(n_points) * 2 - 1
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))

        C = cp.Variable()
        h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
        constraints = [constraint(h_var), lag_constraints(h_var, C)]
        prob = cp.Problem(cp.Minimize(C), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-8)
            if prob.status == 'optimal' and prob.value < best_c5:
                best_c5 = prob.value
                best_h = h_var.value.copy()
        except:
            pass

    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    best_h = best_h * (target_sum / sum_h) if sum_h != 0 else best_h

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(10):
            if time.time() - start_time > budget_s:
                break
            h = np.random.rand(n_points) * 2 - 1
            h = np.clip(h, 0.0, 1.0)
            h = h * (target_sum / np.sum(h))

            C = cp.Variable()
            h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
            constraints = [constraint(h_var), lag_constraints(h_var, C)]
            prob = cp.Problem(cp.Minimize(C), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-8)
                if prob.status == 'optimal' and prob.value < best_c5:
                    best_c5 = prob.value
                    best_h = h_var.value.copy()
            except:
                pass

    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    best_h = best_h * (target_sum / sum_h) if sum_h != 0 else best_h

    return best_h.tolist(), best_c5, n_points
```
