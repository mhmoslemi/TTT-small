import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Objective function
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Stage 1: Critical Lag Identification and Gradient Refinement
    def gradient_refinement(h, max_iter=100, tol=1e-6):
        best_h = h.copy()
        best_c5 = objective(best_h)
        for _ in range(max_iter):
            correlate_vals = np.correlate(best_h, 1 - best_h, mode="full") * (2.0 / n_points)
            k_star = np.argmax(correlate_vals)
            grad = np.zeros_like(best_h)
            if k_star < n_points:
                grad[:-k_star] += (1 - best_h[k_star:])
            elif k_star >= n_points:
                grad[k_star - n_points + 1:] += (1 - best_h[:k_star - n_points + 1])
            grad *= 2.0 / n_points
            result = opt.line_search.scalar_search_line_opt(
                lambda x: objective(x), 
                lambda x: grad,
                best_h, 
                maxiter=10,
                tolerance=tol
            )
            if result[0] is None:
                break
            best_h = result[1]
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            current_c5 = objective(best_h)
            if abs(current_c5 - best_c5) < tol:
                break
            best_c5 = current_c5
        return best_h, best_c5
    
    h, best_c5 = gradient_refinement(h)
    
    # Stage 2: Multi-Scale Refinement
    n_points = 90
    dx = 2.0 / n_points
    h_new = np.zeros(n_points)
    h_new[:len(h)] = h
    h_new = h_new * (n_points / 2) / np.sum(h_new)
    h_new = np.clip(h_new, 0, 1)
    h = h_new
    
    def objective_new(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    h, best_c5 = gradient_refinement(h, max_iter=100, tol=1e-6)
    best_c5 = objective_new(h)
    
    # Stage 3: Fallback to Differential Evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.differential_evolution(objective_new, bounds, popsize=30, mutation=(0.9, 1.0),
                                       recombination=0.7, strategy='best1bin', tol=1e-5,
                                       maxiter=200, disp=False, polish=True, seed=seed)
    
    best_h = result.x
    best_c5 = result.fun
    
    # Final check
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
        h = best_h
        best_c5 = objective_new(h)
    
    # Final feasibility check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective_new(h)
    
    return (h.tolist(), final_c5, n_points)
