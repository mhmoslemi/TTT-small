import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_guesses = []

    if 'initial_h_values' in globals():
        base_guess = np.copy(initial_h_values)
        base_guess = np.clip(base_guess, 0, 1)
        base_guess /= np.sum(base_guess)
        base_guess *= target_sum
        initial_guesses.append(base_guess)
    else:
        base_guess = np.zeros(n_points)
        for i in range(n_points):
            base_guess[i] = 1 if (i % 5 == 0) else 0
        base_guess = np.clip(base_guess, 0, 1)
        base_guess /= np.sum(base_guess)
        base_guess *= target_sum
        initial_guesses.append(base_guess)

    def generate_structured_guess():
        period = n_points // 5
        guess = np.zeros(n_points)
        for i in range(n_points):
            if (i % period == 0):
                guess[i] = 1
        guess = np.clip(guess, 0, 1)
        guess /= np.sum(guess)
        guess *= target_sum
        return guess

    initial_guesses.append(generate_structured_guess())
    initial_guesses.append(np.random.uniform(0, 1, n_points))
    initial_guesses.append(np.flip(base_guess))
    initial_guesses.append(0.5 * base_guess + 0.5 * generate_structured_guess())

    initial_guesses = [np.clip(g, 0, 1) for g in initial_guesses]
    initial_guesses = [g * (target_sum / np.sum(g)) for g in initial_guesses]

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    for guess in initial_guesses:
        x0 = guess.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

        def objective(x):
            x = np.clip(x, 0, 1)
            corr = np.correlate(x, 1 - x, mode='full')
            c5 = np.max(corr) * dx
            return c5

        result = opt.minimize(
            objective,
            x0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-6, 'eps': 1e-6}
        )

        if result.success:
            h_vals = result.x
            h_vals = np.clip(h_vals, 0, 1)
            h_vals /= np.sum(h_vals)
            h_vals *= target_sum
            current_c5 = objective(h_vals)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_vals
        else:
            h_vals = np.clip(x0, 0, 1)
            h_vals /= np.sum(h_vals)
            h_vals *= target_sum
            current_c5 = objective(h_vals)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_vals

    best_h = np.clip(best_h, 0, 1)
    best_h /= np.sum(best_h)
    best_h *= target_sum
    final_c5 = objective(best_h)

    return (best_h, final_c5, n_points)
