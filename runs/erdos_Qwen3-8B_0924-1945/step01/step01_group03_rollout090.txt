import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial = initial_h_values if 'initial_h_values' in locals() else np.random.rand(n_points)
    
    def generate_guesses():
        guesses = []
        base = np.copy(initial)
        base = np.clip(base, 0, 1)
        base = base / np.sum(base) * target_sum
        guesses.append(base)
        
        spaced = np.zeros(n_points)
        step = n_points // 5
        for i in range(5):
            spaced[i::step] = 1
        spaced = np.clip(spaced, 0, 1)
        spaced = spaced / np.sum(spaced) * target_sum
        guesses.append(spaced)
        
        perturbed = np.copy(base)
        perturbed += np.random.uniform(-0.2, 0.2, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed / np.sum(perturbed) * target_sum
        guesses.append(perturbed)
        
        reversed_ = np.copy(base)[::-1]
        reversed_ = np.clip(reversed_, 0, 1)
        reversed_ = reversed_ / np.sum(reversed_) * target_sum
        guesses.append(reversed_)
        
        blended = (base + spaced) / 2
        blended = np.clip(blended, 0, 1)
        blended = blended / np.sum(blended) * target_sum
        guesses.append(blended)
        
        return guesses
    
    def objective(x):
        x = np.clip(x, 0, 1)
        x = x / np.sum(x) * target_sum
        corr = np.correlate(x, 1 - x, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(x):
        x = np.clip(x, 0, 1)
        x = x / np.sum(x) * target_sum
        return np.sum(x) - target_sum
    
    best_h = None
    best_c5 = float('inf')
    guesses = generate_guesses()
    
    for guess in guesses:
        x0 = guess.copy()
        result = opt.minimize(
            objective,
            x0,
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-6, 'eps': 1e-6}
        )
        if result.success:
            final = result.x
            final = np.clip(final, 0, 1)
            final = final / np.sum(final) * target_sum
            c5 = objective(final)
            if c5 < best_c5:
                best_c5 = c5
                best_h = final
    
    if best_h is None:
        best_h = guesses[0]
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
