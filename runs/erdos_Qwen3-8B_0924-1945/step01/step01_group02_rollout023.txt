```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def compute_A(k):
        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = (i + k) % n_points
            A[i, j] = dx
            j = (i - k) % n_points
            A[i, j] = dx
        return A

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0.0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def optimize_and_return():
        best_obj = float('inf')
        best_h = np.copy(h)
        for restart in range(20):
            if time.time() - start_time > budget_s:
                break
            
            h_curr = np.copy(h)
            h_curr += np.random.uniform(-0.05, 0.05, size=n_points)
            h_curr = np.clip(h_curr, 0.0, 1.0)
            sum_h_curr = np.sum(h_curr)
            h_curr = h_curr * (target_sum / sum_h_curr) if sum_h_curr != 0.0 else h_curr
            
            A_list = [compute_A(k) for k in range(1, n_points)]
            variables = [cp.Variable(n_points), cp.Variable()]
            h_var, C_var = variables

            constraints = [
                cp.sum(h_var) == target_sum,
                cp.Minimize(C_var),
                h_var >= 0,
                h_var <= 1
            ]
            for A in A_list:
                constraints.append(h_var.T @ A @ h_var >= 1.0 - C_var)

            try:
                prob = cp.Problem(cp.Minimize(C_var), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-8)
                if prob.status == 'OPTIMAL':
                    h_current = h_var.value
                    current_obj = C_var.value
                    if current_obj < best_obj:
                        best_obj = current_obj
                        best_h = h_current
            except:
                pass

        return best_h

    start_time = time.time()
    best_h = optimize_and_return()
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    best_h = best_h * (target_sum / sum_h) if sum_h != 0.0 else best_h
    c5_bound = evaluate(best_h)
    return best_h.tolist(), c5_bound, n_points
```
