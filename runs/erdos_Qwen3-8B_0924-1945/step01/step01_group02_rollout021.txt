import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150  # Higher resolution for better bound
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    # Precompute the shifted kernel for efficient correlation
    h_kernel = 1.0 - h
    corr_length = 2 * n_points - 1

    # Prepare constraints for quadratic program
    A_list = []
    for k in range(1, n_points):
        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = i + k
            if j < n_points:
                A[i, j] = dx
            j = i - k
            if j >= 0:
                A[i, j] = dx
        A_list.append(A)

    # Optimization variables
    h_var = cp.Variable(n_points)
    C = cp.Variable()

    # Objective: minimize C
    objective = cp.Minimize(C)

    # Constraints
    constraints = [
        cp.sum(h_var) == target_sum,
        cp.multiply(h_var, h_var) <= 1.0,
        cp.multiply(h_var, h_var) >= 0.0
    ]

    # Add quadratic constraints for each k
    for i, A in enumerate(A_list):
        constr = cp.quad_form(h_var, A) >= 1.0 - C
        constraints.append(constr)

    # Assemble and solve
    prob = cp.Problem(objective, constraints)
    best_c5 = float('inf')
    best_h = h.copy()
    best_h_val = h.copy()
    best_c5_val = float('inf')
    num_restarts = 20
    timeout = budget_s * 0.75  # Reserve 25% for final checks

    start_time = time.time()
    for r in range(num_restarts):
        if time.time() - start_time > timeout:
            break
        # Randomize initial guess
        h_guess = h.copy()
        noise = np.random.normal(0, 0.05, size=n_points)
        h_guess += noise
        h_guess = np.clip(h_guess, 0.0, 1.0)
        h_guess = h_guess * (target_sum / np.sum(h_guess))

        # Solve QP
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver='ECOS', eps=1e-8, verbose=False)
        except:
            prob.solve(solver='SCS', max_iter=100, verbose=False)

        if prob.status == 'OPTIMAL':
            current_c5 = 1.0 - C.value
            current_h = h_var.value
            current_h = np.clip(current_h, 0.0, 1.0)
            current_h = current_h * (target_sum / np.sum(current_h))
            corr = np.correlate(current_h, h_kernel, mode='full')
            c5 = np.max(corr * dx)
            if c5 < best_c5_val:
                best_c5_val = c5
                best_h_val = current_h.copy()
                best_h = current_h.copy()

    # Final evaluation
    corr = np.correlate(best_h_val, h_kernel, mode='full')
    c5_bound = np.max(corr * dx)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (target_sum / np.sum(best_h))
    elapsed = time.time() - start_time
    return best_h.tolist(), c5_bound, n_points
