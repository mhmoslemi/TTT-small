import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def refine_with_gradient(h, max_iter=300, tol=1e-6):
        k_star = np.argmax(np.correlate(h, 1 - h, mode="full"))
        dx = 2.0 / n_points
        grad = (1 - h[(k_star + np.arange(n_points)) % n_points]) * dx
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.minimize(objective, h, method='L-BFGS-B', jac=grad, bounds=bounds, tol=tol, options={'maxiter': max_iter})
        return result.x
    
    def refine_with_de(h, max_iter=200):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(objective, bounds, popsize=50, mutation=(0.8, 1.0), 
                                           recombination=0.9, strategy='best1bin', tol=1e-5,
                                           maxiter=max_iter, disp=False, polish=True, seed=seed)
        return result.x
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    # Stage 1: Coarse-Grid Global Optimization with DE
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.5:
        best_h = refine_with_de(h, max_iter=200)
        best_c5 = objective(best_h)
    
    # Stage 2: Gradient-Based Refinement
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.7:
        best_h = refine_with_gradient(best_h, max_iter=300, tol=1e-6)
        best_c5 = objective(best_h)
    
    # Stage 3: Multi-Scale Refinement (increase resolution)
    n_points = 90
    dx = 2.0 / n_points
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, best_h.size), best_h)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    best_c5 = objective(h)
    h = refine_with_gradient(h, max_iter=300, tol=1e-6)
    best_c5 = objective(h)
    
    # Final check and fallback if needed
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        best_c5 = objective(h)
    else:
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        best_c5 = objective(h)
    
    return (h.tolist(), best_c5, n_points)
