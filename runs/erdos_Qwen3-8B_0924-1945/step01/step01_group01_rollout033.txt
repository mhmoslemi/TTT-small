import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Helper function to evaluate C5
    def evaluate_c5(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Stage 1: Differential Evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.differential_evolution(
        evaluate_c5, bounds, popsize=50, mutation=(0.8, 1.0),
        recombination=0.9, strategy='best1bin', tol=1e-5, maxiter=200,
        disp=False, polish=True, seed=seed
    )
    best_h = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (best_h.tolist(), best_c5, n_points)
    
    # Stage 2: Gradient-Based Refinement
    def objective(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_k = np.argmax(correlate_vals)
        return correlate_vals[max_k]
    
    def gradient(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_k = np.argmax(correlate_vals)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + max_k < n_points:
                grad[i] = (1 - h[i + max_k]) * (2.0 / n_points)
        return grad
    
    result = opt.minimize(
        objective, best_h, bounds=bounds, method='L-BFGS-B',
        jac=gradient, tol=1e-6, maxiter=300, options={'maxfun': 1000}
    )
    best_h = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (best_h.tolist(), best_c5, n_points)
    
    # Stage 3: Multi-scale refinement
    n_points = 90
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:len(best_h)] = best_h
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def evaluate_scaled_c5(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    result = opt.differential_evolution(
        evaluate_scaled_c5, bounds, popsize=30, mutation=(0.9, 1.0),
        recombination=0.7, strategy='best1bin', tol=1e-5, maxiter=100,
        disp=False, polish=True, seed=seed
    )
    best_h = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (best_h.tolist(), best_c5, n_points)
    
    # Final check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = evaluate_scaled_c5(h)
    
    return (h.tolist(), final_c5, n_points)
