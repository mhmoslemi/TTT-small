import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 90
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def gradient(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_index = np.argmax(correlate_vals)
        if max_index >= len(h):
            max_index -= len(h)
        if max_index < 0:
            max_index += len(h)
        dx = 2.0 / n_points
        grad = np.zeros_like(h)
        for i in range(n_points):
            x = i * dx
            if max_index + i < len(h):
                val = 1 - h[max_index + i]
            else:
                val = 0
            grad[i] = val * dx
        return grad
    
    # Stage 2: Gradient-based refinement with L-BFGS-B
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(objective, h, method='L-BFGS-B', jac=gradient,
                          bounds=bounds, tol=1e-5, options={'maxiter': 500, 'disp': False})
    best_h = result.x
    best_c5 = result.fun
    
    # Stage 3: Adaptive grid refinement
    elapsed = time.time() - start_time
    if elapsed < budget_s - 400:
        n_points_new = 120
        dx_new = 2.0 / n_points_new
        h_new = np.clip(best_h, 0, 1)
        h_new = h_new * (n_points_new / 2) / np.sum(h_new)
        h_new = np.concatenate([h_new[:n_points_new // 2], np.zeros(n_points_new - n_points_new // 2)])
        h_new = h_new * (n_points_new / 2) / np.sum(h_new)
        def objective_new(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points_new / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full") * dx_new
            return np.max(correlate_vals)
        
        def gradient_new(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points_new / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full") * dx_new
            max_index = np.argmax(correlate_vals)
            if max_index >= len(h):
                max_index -= len(h)
            if max_index < 0:
                max_index += len(h)
            dx_new = 2.0 / n_points_new
            grad = np.zeros_like(h)
            for i in range(n_points_new):
                x = i * dx_new
                if max_index + i < len(h):
                    val = 1 - h[max_index + i]
                else:
                    val = 0
                grad[i] = val * dx_new
            return grad
        
        result_new = opt.minimize(objective_new, h_new, method='L-BFGS-B', jac=gradient_new,
                                 bounds=[(0.0, 1.0) for _ in range(n_points_new)], tol=1e-5,
                                 options={'maxiter': 300, 'disp': False})
        best_h = result_new.x
        best_c5 = result_new.fun
        n_points = n_points_new
    
    # Stage 4: Fallback to differential evolution if needed
    if best_c5 > 0.3808 or (time.time() - start_time) > budget_s - 100:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result_de = opt.differential_evolution(objective, bounds, popsize=20, mutation=(0.9, 1.0),
                                               recombination=0.7, strategy='best1bin', tol=1e-5,
                                               maxiter=300, disp=False, polish=True, seed=seed)
        best_h = result_de.x
        best_c5 = result_de.fun
    
    # Final check
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best candidate")
    
    # Final feasibility check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points)
