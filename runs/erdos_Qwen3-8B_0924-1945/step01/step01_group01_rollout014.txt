import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Objective function
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def refine_solution(h_vec, n_points_final):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points_final / 2) / np.sum(h)
        # Use L-BFGS-B on the refined grid
        bounds = [(0.0, 1.0) for _ in range(n_points_final)]
        def obj_func(x):
            x = x * (n_points_final / 2) / np.sum(x)
            x = np.clip(x, 0, 1)
            correlate_vals = np.correlate(x, 1 - x, mode="full") * (2.0 / n_points_final)
            return np.max(correlate_vals)
        result = opt.minimize(obj_func, h, method='L-BFGS-B', bounds=bounds, tol=1e-8, maxiter=100)
        return result.x
    
    # Coarse optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_coarse = opt.differential_evolution(
        objective, bounds, popsize=40, mutation=(0.8, 1.0), 
        recombination=0.9, strategy='best1bin', tol=1e-5, 
        maxiter=300, disp=False, polish=True, seed=seed
    )
    best_h = result_coarse.x
    best_c5 = result_coarse.fun
    
    # Refine to 90 points
    n_points_final = 90
    h_refined = np.zeros(n_points_final)
    h_refined[:n_points] = best_h
    h_refined = refine_solution(h_refined, n_points_final)
    
    # Final evaluation
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning coarse solution")
        h = initial_h_values if 'initial_h_values' in globals() else h
        best_c5 = objective(h)
    
    # Final check and return
    h = np.clip(h_refined, 0, 1)
    h = h * (n_points_final / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points_final)
