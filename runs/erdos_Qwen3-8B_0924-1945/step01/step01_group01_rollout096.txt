import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial configuration
    n_points = 45
    h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)

    # Objective function for gradient step
    def obj_grad(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_k = np.argmax(correlate_vals)
        return correlate_vals[max_k], [1 - h[i + max_k] * (2.0 / n_points) if i + max_k < len(h) else 0 for i in range(len(h))]

    # Objective function for DE
    def obj_de(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)

    # L-BFGS-B optimizer
    def optimize_lbfgs(h_start, max_iter=100, tol=1e-6):
        h = h_start.copy()
        best_h = h.copy()
        best_c5 = obj_de(h)
        for _ in range(max_iter):
            grad = obj_grad(h)[1]
            h_new = opt.minimize(lambda x: obj_de(x), h, jac=grad, method='L-BFGS-B', bounds=[(0,1)]*n_points).x
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5 = obj_de(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5
                h = h_new
            if abs(c5 - best_c5) < tol:
                break
        return best_h, best_c5

    # Differential Evolution optimizer
    def optimize_de(h_start, popsize=30, max_iter=200):
        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.differential_evolution(obj_de, bounds, popsize=popsize, mutation=(0.9, 1.0), recombination=0.7,
                                            strategy='best1bin', tol=1e-5, maxiter=max_iter, disp=False, seed=seed)
        return result.x, result.fun

    # Main optimization
    h, c5 = optimize_lbfgs(h)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning initial solution")
        h = initial_h_values if 'initial_h_values' in globals() else h
        c5 = obj_de(h)

    # Refinement
    n_points_new = 90
    h_new = np.interp(np.linspace(0, 2, n_points_new), np.linspace(0, 2, n_points), h)
    h_new = h_new * (n_points_new / 2) / np.sum(h_new)
    h_new, c5_new = optimize_lbfgs(h_new, max_iter=50, tol=1e-6)
    if c5_new < c5:
        h, c5 = h_new, c5_new

    # Fallback to DE
    if c5 > 0.383652 and c5_new > 0.383652:
        h, c5 = optimize_de(h, popsize=60, max_iter=100)

    # Final check and return
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = obj_de(h)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
    return (h.tolist(), final_c5, n_points)
