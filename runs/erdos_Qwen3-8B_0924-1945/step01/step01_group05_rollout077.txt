import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        period = n_points // 3
        h[::period] = 1.0
        h = np.clip(h, 0, 1)
        h = h * n_points / 2 / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * n_points / 2 / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    initial_guess = h
    
    def genetic_algorithm():
        pop_size = 50
        max_gen = 100
        mutation_rate = 0.1
        crossover_rate = 0.8
        elite_ratio = 0.2
        mut_var = 0.1
        best_h = initial_guess.copy()
        best_c5 = objective(best_h)
        current_gen = 0
        time_start = time.time()
        while current_gen < max_gen and time.time() - time_start < budget_s:
            pop = np.random.rand(pop_size, n_points)
            pop = pop * n_points / 2 / np.sum(pop, axis=1, keepdims=True)
            pop = np.clip(pop, 0, 1)
            pop_fitness = np.array([objective(ind) for ind in pop])
            indices = np.argsort(pop_fitness)
            pop = pop[indices]
            pop_fitness = pop_fitness[indices]
            elite_size = int(pop_size * elite_ratio)
            elites = pop[:elite_size]
            selected = pop[:int(pop_size * 0.6)]
            offspring = []
            for _ in range(pop_size - elite_size):
                if np.random.rand() < crossover_rate:
                    parent1, parent2 = np.random.choice(selected, 2, replace=False)
                    child = np.concatenate([parent1[:n_points // 2], parent2[n_points // 2:]])
                else:
                    child = np.random.rand(n_points)
                    child = child * n_points / 2 / np.sum(child)
                    child = np.clip(child, 0, 1)
                if np.random.rand() < mutation_rate:
                    idxs = np.random.choice(n_points, int(n_points * 0.05), replace=False)
                    child[idxs] += np.random.uniform(-mut_var, mut_var, size=len(idxs))
                    child = np.clip(child, 0, 1)
                    child = child * n_points / 2 / np.sum(child)
                offspring.append(child)
            pop = np.vstack([elites, offspring])
            pop = np.clip(pop, 0, 1)
            pop = pop * n_points / 2 / np.sum(pop, axis=1, keepdims=True)
            pop_fitness = np.array([objective(ind) for ind in pop])
            indices = np.argsort(pop_fitness)
            pop = pop[indices]
            pop_fitness = pop_fitness[indices]
            current_gen += 1
            if pop_fitness[0] < best_c5:
                best_h = pop[0]
                best_c5 = pop_fitness[0]
        return best_h, best_c5
    
    h, c5 = genetic_algorithm()
    h = np.clip(h, 0, 1)
    h = h * n_points / 2 / np.sum(h)
    final_c5 = objective(h)
    return (h, final_c5, n_points)
