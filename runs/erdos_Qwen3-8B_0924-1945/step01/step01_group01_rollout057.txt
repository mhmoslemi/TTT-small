import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def refine_with_l_bfgs(h):
        def grad_func(x):
            x = np.clip(x, 0, 1)
            x = x * (n_points / 2) / np.sum(x)
            corr = np.correlate(x, 1 - x, mode="full") * (2.0 / n_points)
            k_star = np.argmax(corr)
            grad = np.zeros_like(x)
            for i in range(n_points):
                if i + k_star < n_points:
                    grad[i] = (1 - x[i + k_star]) * (2.0 / n_points)
                if i - k_star >= 0:
                    grad[i] += (1 - x[i - k_star]) * (2.0 / n_points)
            return grad
        
        result = opt.minimize(objective, h, method='L-BFGS-B', jac=grad_func, bounds=[(0, 1)]*n_points, tol=1e-6, options={'maxiter': 300})
        return result.x
    
    def evolve_with_de(h, n_iters=200):
        bounds = [(0.0, 1.0) for _ in range(len(h))]
        result = opt.differential_evolution(objective, bounds, popsize=50, mutation=(0.8, 1.0), recombination=0.9, strategy='best1bin', tol=1e-5, maxiter=n_iters, disp=False, polish=True, seed=seed)
        return result.x
    
    best_h = h
    best_c5 = objective(h)
    time_used = time.time() - start_time
    
    if time_used < budget_s:
        # Coarse optimization with DE
        best_h = evolve_with_de(h)
        best_c5 = objective(best_h)
        time_used = time.time() - start_time
        
        # Refinement with L-BFGS-B
        if time_used < budget_s:
            best_h = refine_with_l_bfgs(best_h)
            best_c5 = objective(best_h)
            time_used = time.time() - start_time
        
        # Multi-scale refinement
        if time_used < budget_s:
            n_points = 90
            dx = 2.0 / n_points
            h_new = np.zeros(n_points)
            h_new[:len(best_h)] = best_h
            h_new = np.random.rand(n_points - len(best_h))
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_new = np.concatenate((h_new, best_h[len(best_h):]))
            best_h = refine_with_l_bfgs(h_new)
            best_c5 = objective(best_h)
            time_used = time.time() - start_time
    
    # Final check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
    
    return (h.tolist(), final_c5, n_points)
