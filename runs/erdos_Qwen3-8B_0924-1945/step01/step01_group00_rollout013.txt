import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize, Bounds
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if np.random.rand() < 0.5 else 0
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    def de_search(population, budget):
        pop_size = len(population)
        mut_factor = 0.5
        cross_prob = 0.9
        gen = 0
        while time() - start_time < budget:
            gen += 1
            new_population = []
            for i in range(pop_size):
                a, b, c = np.random.choice(population, 3, replace=False)
                mutant = a + mut_factor * (b - c)
                mutant = np.clip(mutant, 0, 1)
                mutant = mutant / np.sum(mutant) * n_points / 2
                trial = mutant
                if np.random.rand() < cross_prob:
                    trial = np.random.choice([mutant, population[i]], size=n_points)
                trial = np.clip(trial, 0, 1)
                trial = trial / np.sum(trial) * n_points / 2
                trial_c5 = evaluate_candidate(trial)
                if trial_c5 < compute_c5(population[i]):
                    new_population.append(trial)
                else:
                    new_population.append(population[i])
            population = new_population
            current_c5 = compute_c5(population[0])
            if current_c5 < best_c5:
                best_h = population[0].copy()
                best_c5 = current_c5
            if gen > 100:
                break
        return best_h, best_c5

    best_h, best_c5 = de_search([h.copy() for _ in range(20)], budget_s)

    for _ in range(100):
        if time() - start_time >= budget_s:
            break
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
        current_c5 = evaluate_candidate(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    best_c5 = evaluate_candidate(best_h)
    return (best_h, best_c5, n_points)
