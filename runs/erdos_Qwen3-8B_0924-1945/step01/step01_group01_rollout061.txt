import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Objective function
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Optimization using L-BFGS-B and differential evolution
    def optimize_func(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        k_max = np.argmax(correlate_vals)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if k_max < n_points and k_max + i < n_points:
                grad[i] += (1 - h[i + k_max]) * (2.0 / n_points)
        return np.max(correlate_vals), grad
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, jac=None, tol=1e-6, options={'maxiter': 100})
    best_h = result.x
    best_c5 = result.fun
    
    if time.time() - start_time > budget_s * 0.5:
        # Fallback to differential evolution if time is running out
        result = opt.differential_evolution(objective, bounds, popsize=30, mutation=(0.9, 1.0), 
                                           recombination=0.7, strategy='best1bin', tol=1e-5,
                                           maxiter=200, disp=False, polish=True, seed=seed)
        best_h = result.x
        best_c5 = result.fun
    
    # Final check
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution found")
        h = best_h
        best_c5 = objective(h)
    
    # Final feasibility check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points)
