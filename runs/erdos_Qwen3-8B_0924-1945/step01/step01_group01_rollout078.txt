import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def refine_solution(h, n_points_new):
        dx = 2.0 / n_points_new
        h_new = np.zeros(n_points_new)
        for i in range(len(h)):
            pos = int(i * n_points_new / n_points)
            h_new[pos] += h[i]
        h_new = h_new * (n_points_new / 2) / np.sum(h_new)
        h_new = np.clip(h_new, 0, 1)
        return h_new
    
    def l_bfgs_refinement(h, n_points):
        def l_bfgs_objective(x):
            x = np.clip(x, 0, 1)
            x = x * (n_points / 2) / np.sum(x)
            correlate_vals = np.correlate(x, 1 - x, mode="full") * (2.0 / n_points)
            return np.max(correlate_vals)
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.minimize(l_bfgs_objective, h, bounds=bounds, method='L-BFGS-B', tol=1e-6, max_iter=100)
        return result.x
    
    def optimize(h, n_points):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(
            objective,
            bounds,
            popsize=20,
            mutation=(0.8, 1.0),
            recombination=0.9,
            strategy='best1bin',
            tol=1e-6,
            maxiter=200,
            disp=False,
            polish=True,
            seed=seed
        )
        return result.x
    
    # Coarse optimization
    best_h = h
    best_c5 = objective(h)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (best_h.tolist(), best_c5, n_points)
    
    best_h = optimize(best_h, n_points)
    best_c5 = objective(best_h)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (best_h.tolist(), best_c5, n_points)
    
    # Refine to 90 points
    n_points_new = 90
    h_refined = refine_solution(best_h, n_points_new)
    h_refined = optimize(h_refined, n_points_new)
    h_refined = l_bfgs_refinement(h_refined, n_points_new)
    best_c5 = objective(h_refined)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (h_refined.tolist(), best_c5, n_points_new)
    
    # Final feasibility check
    h_final = np.clip(h_refined, 0, 1)
    h_final = h_final * (n_points_new / 2) / np.sum(h_final)
    final_c5 = objective(h_final)
    
    return (h_final.tolist(), final_c5, n_points_new)
