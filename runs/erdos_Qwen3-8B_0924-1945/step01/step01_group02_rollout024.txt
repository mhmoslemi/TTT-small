```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        num_ones = int(np.ceil(n_points / 2))
        indices = np.random.choice(n_points, num_ones, replace=False)
        h[indices] = 1.0
        h = h * (target_sum / np.sum(h))

    def compute_constraints(h, k):
        res = np.zeros_like(h)
        for i in range(n_points):
            if i + k < n_points:
                res[i + k] += h[i]
            if i - k >= 0:
                res[i - k] += h[i]
        return res * dx

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    n_lags = n_points - 1
    A = []
    for k in range(1, n_lags + 1):
        A_k = np.zeros((n_points, n_points))
        for i in range(n_points):
            if i + k < n_points:
                A_k[i, i + k] = dx
            if i - k >= 0:
                A_k[i, i - k] = dx
        A.append(A_k)

    variables = []
    for k in range(n_lags):
        variables.append(cp.Variable(n_points))
    C = cp.Variable()

    prob = cp.Problem(cp.Minimize(C),
                      [cp.sum(variables[k]) == 1 for k in range(n_lags)] +
                      [cp.sum(variables[k] * A[k][i][j] * variables[k][j] for j in range(n_points)) >= 1 - C for k in range(n_lags)] +
                      [cp.sum(variables[k]) == 1.0 for k in range(n_lags)] +
                      [variables[k] >= 0 for k in range(n_lags)] +
                      [variables[k] <= 1 for k in range(n_lags)])

    result = prob.solve(solver=cp.ECOS, eps=1e-8, verbose=False)
    h_opt = variables[0].value
    c5_bound = objective(h_opt.flatten())

    return h_opt.tolist(), c5_bound, n_points
```
