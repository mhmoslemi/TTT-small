```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h = h * scale

    def build_constraints(n_points, dx):
        lags = np.arange(1, n_points)
        A = []
        for k in lags:
            row = np.zeros(n_points)
            for i in range(n_points - k):
                row[i] += dx
                row[i + k] += dx
            A.append(row)
        return A

    A = build_constraints(n_points, dx)
    C = cp.Variable()
    h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
    constraints = [
        cp.sum(h_var) == target_sum,
        C >= 1.0 - cp.sum(cp.multiply(h_var, h_var)) + cp.sum(cp.multiply(h_var, h_var))
    ]
    for i in range(len(A)):
        constraints.append(A[i] @ h_var >= 1.0 - C)

    prob = cp.Problem(cp.Minimize(C), constraints)
    start_time = time.time()
    try:
        prob.solve(solver=cp.ECOS, verbose=False, eps=1e-6, max_iter=1000)
    except:
        prob.solve(solver=cp.SCS, verbose=False, eps=1e-6, max_iter=1000)

    h_opt = h_var.value
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(3):
            new_h = np.clip(h_opt + np.random.normal(0, 0.05, n_points), 0.0, 1.0)
            new_sum = np.sum(new_h)
            scale = target_sum / new_sum if new_sum != 0 else 1.0
            new_h = new_h * scale
            h_opt = new_h

    c5_bound = np.max(np.correlate(h_opt, 1.0 - h_opt, mode="full")) * dx
    return h_opt.tolist(), c5_bound, n_points
```
