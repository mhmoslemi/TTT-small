import numpy as np
import numpy.random as npr
import scipy.optimize as optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = h * (target_sum / np.sum(h))
    
    h = np.clip(h, 0.0, 1.0)
    
    def objective(h_):
        h_ = np.clip(h_, 0.0, 1.0)
        h_ = h_ * (target_sum / np.sum(h_)) if np.sum(h_) != 0 else h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    N_pop = 3 * n_points
    F = 0.8
    P_crossover = 0.7
    max_generations = 200
    pop = []
    for _ in range(N_pop):
        h = np.random.rand(n_points)
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))
        pop.append(h)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    for gen in range(max_generations):
        new_pop = []
        for _ in range(N_pop):
            a, b, c = np.random.choice(pop, 3, replace=False)
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = mutant * (target_sum / np.sum(mutant))
            mutant = np.clip(mutant, 0.0, 1.0)
            
            child = mutant.copy()
            for i in range(n_points):
                if np.random.rand() < P_crossover:
                    child[i] = mutant[i]
            child = np.clip(child, 0.0, 1.0)
            child = child * (target_sum / np.sum(child))
            child = np.clip(child, 0.0, 1.0)
            
            new_pop.append(child)
        
        pop = new_pop
        for h_ in pop:
            c5 = objective(h_)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_.copy()
        
        if gen % 50 == 0:
            print(f"Generation {gen}, Best C5: {best_c5}")
        
        if best_c5 < 0.3807:
            break
    
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (target_sum / np.sum(best_h))
    best_h = np.clip(best_h, 0.0, 1.0)
    c5_bound = objective(best_h)
    return (best_h.tolist(), c5_bound, n_points)
