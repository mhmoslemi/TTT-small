```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_restarts = 20
    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                h[i] = 1.0
        h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0.0, 1.0)

    for restart in range(max_restarts):
        h_flat = h.copy()
        np.random.seed(seed + restart)
        if restart > 0:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            h_flat += noise
            h_flat = np.clip(h_flat, 0.0, 1.0)
            h_flat = h_flat * (target_sum / np.sum(h_flat))

        A = []
        for k in range(1, n_points):
            A_k = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = (i + k) % n_points
                A_k[i, j] = dx
                j = (i - k + n_points) % n_points
                A_k[i, j] = dx
            A.append(A_k)

        h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
        C = cp.Variable(pos=True)
        constraints = [cp.sum(h_var) == target_sum]
        for k in range(len(A)):
            lhs = cp.quad_form(h_var, A[k])
            constraints.append(lhs >= 1.0 - C)

        prob = cp.Problem(cp.Minimize(C), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-8, verbose=False)
        if prob.status in ['OPTIMAL', 'OPTIMAL_INACCURATE']:
            c5 = C.value
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_var.value
        else:
            h_flat = h_flat.copy()
            h_flat = np.clip(h_flat, 0.0, 1.0)
            h_flat = h_flat * (target_sum / np.sum(h_flat))
            h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
            C = cp.Variable(pos=True)
            constraints = [cp.sum(h_var) == target_sum]
            for k in range(len(A)):
                lhs = cp.quad_form(h_var, A[k])
                constraints.append(lhs >= 1.0 - C)
            prob = cp.Problem(cp.Minimize(C), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-8, verbose=False)
            if prob.status in ['OPTIMAL', 'OPTIMAL_INACCURATE']:
                c5 = C.value
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_var.value

        elapsed = time.time() - time.time()
        if elapsed >= budget_s:
            break

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (target_sum / np.sum(best_h))
    c5_bound = np.max(np.correlate(best_h, 1.0 - best_h, mode="full") * dx)
    return best_h.tolist(), c5_bound, n_points
```
