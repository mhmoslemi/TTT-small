```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_lag = n_points - 1

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        h[::5] = 1.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def construct_constraints():
        A = []
        for k in range(1, max_lag + 1):
            row = np.zeros(n_points)
            for i in range(n_points - k):
                row[i] += dx
                row[i + k] += dx
            A.append(row)
        return np.array(A)

    A = construct_constraints()
    n_constraints = len(A)

    h_var = cp.Variable(n_points)
    C = cp.Variable()

    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target_sum,
        [cp.sum(cp.multiply(A[i, :], h_var)) >= 1 - C for i in range(n_constraints)]
    ]

    prob = cp.Problem(cp.Minimize(C), constraints)
    start_time = time.time()

    try:
        result = prob.solve(solver=cp.ECOS, eps=1e-6, max_iter=1000)
        h_opt = h_var.value
    except:
        h_opt = h.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        try:
            result = prob.solve(solver=cp.ECOS, eps=1e-6, max_iter=2000)
            h_opt = h_var.value
        except:
            h_opt = h.copy()
    else:
        h_opt = h.copy()

    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale

    c5_bound = evaluate(h_opt)
    return h_opt.tolist(), c5_bound, n_points
```
