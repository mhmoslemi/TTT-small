import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def initial_guess():
        if 'initial_h_values' in globals():
            h = np.copy(initial_h_values)
            h = np.clip(h, 0, 1)
            h /= np.sum(h)
            h *= target_sum
            return h
        else:
            h = np.zeros(n_points)
            for i in range(n_points):
                if i % 5 == 0:
                    h[i] = 1
            h = np.clip(h, 0, 1)
            h /= np.sum(h)
            h *= target_sum
            return h

    h = initial_guess()

    def objective(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.9,
        tol=1e-6,
        maxiter=200,
        disp=False,
        vectorized=True
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h /= np.sum(best_h)
    best_h *= target_sum
    c5_bound = objective(best_h)

    return (best_h, c5_bound, n_points)
