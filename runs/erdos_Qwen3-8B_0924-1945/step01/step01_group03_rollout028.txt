import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_guesses = []

    if 'initial_h_values' in globals():
        base_guess = np.copy(initial_h_values)
        base_guess = np.clip(base_guess, 0, 1)
        base_guess /= np.sum(base_guess)
        base_guess *= target_sum
        initial_guesses.append(base_guess)

    # Generate structured guess
    spacing = n_points // 5
    structured_guess = np.zeros(n_points)
    for i in range(5):
        idx = (i * spacing) % n_points
        structured_guess[idx] = 1.0
    structured_guess = np.clip(structured_guess, 0, 1)
    structured_guess /= np.sum(structured_guess)
    structured_guess *= target_sum
    initial_guesses.append(structured_guess)

    # Generate perturbed guess
    perturbed_guess = np.copy(initial_guesses[0])
    noise = np.random.uniform(-0.2, 0.2, n_points)
    perturbed_guess += noise
    perturbed_guess = np.clip(perturbed_guess, 0, 1)
    perturbed_guess /= np.sum(perturbed_guess)
    perturbed_guess *= target_sum
    initial_guesses.append(perturbed_guess)

    # Generate reversed guess
    reversed_guess = np.copy(initial_guesses[0])
    reversed_guess = reversed_guess[::-1]
    reversed_guess = np.clip(reversed_guess, 0, 1)
    reversed_guess /= np.sum(reversed_guess)
    reversed_guess *= target_sum
    initial_guesses.append(reversed_guess)

    # Generate blended guess
    blended_guess = 0.5 * initial_guesses[0] + 0.5 * structured_guess
    blended_guess = np.clip(blended_guess, 0, 1)
    blended_guess /= np.sum(blended_guess)
    blended_guess *= target_sum
    initial_guesses.append(blended_guess)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    for guess in initial_guesses:
        x0 = guess.copy()
        def objective(x):
            h_vals = x.copy()
            h_vals = np.clip(h_vals, 0, 1)
            corr = np.correlate(h_vals, 1 - h_vals, mode='full')
            c5 = np.max(corr) * dx
            return c5

        def constraint(x):
            return np.sum(x) - target_sum

        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.minimize(
            objective,
            x0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-6, 'disp': False}
        )

        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0, 1)
            final_h /= np.sum(final_h)
            final_h *= target_sum
            current_c5 = objective(final_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = final_h
        else:
            final_h = x0.copy()
            final_h = np.clip(final_h, 0, 1)
            final_h /= np.sum(final_h)
            final_h *= target_sum
            current_c5 = objective(final_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = final_h

    best_h = np.clip(best_h, 0, 1)
    best_h /= np.sum(best_h)
    best_h *= target_sum
    final_c5 = objective(best_h)
    return (best_h, final_c5, n_points)
