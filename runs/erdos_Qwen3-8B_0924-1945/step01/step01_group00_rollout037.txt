import numpy as np
from scipy.optimize import minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy() if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    def run_de():
        pop_size = 20
        mutation_factor = 0.5
        crossover_prob = 0.9
        max_gen = 100
        best_h = h.copy()
        best_c5 = evaluate_candidate(best_h)
        pop = np.random.uniform(0, 1, (pop_size, n_points))
        pop = pop / np.sum(pop, axis=1, keepdims=True) * n_points / 2
        for gen in range(max_gen):
            if time() - start_time > budget_s:
                break
            new_pop = []
            for i in range(pop_size):
                a, b, c = np.random.choice(pop_size, 3, replace=False)
                mutant = pop[a] + mutation_factor * (pop[b] - pop[c])
                trial = np.clip(mutant, 0, 1)
                trial = trial / np.sum(trial) * n_points / 2
                if np.random.rand() < crossover_prob:
                    trial = np.random.choice([pop[i], trial], size=n_points, p=[0.5, 0.5])
                trial = np.clip(trial, 0, 1)
                trial = trial / np.sum(trial) * n_points / 2
                new_pop.append(trial)
            pop = np.array(new_pop)
            for cand in pop:
                cand_c5 = evaluate_candidate(cand)
                if cand_c5 < best_c5:
                    best_h = cand.copy()
                    best_c5 = cand_c5
        return best_h, best_c5

    start_time = time()
    best_h, best_c5 = run_de()

    def run_sqp(h):
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            return res.x
        return h

    for _ in range(15):
        if time() - start_time > budget_s:
            break
        h = run_sqp(h)
        current_c5 = evaluate_candidate(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    return (best_h, evaluate_candidate(best_h), n_points)
