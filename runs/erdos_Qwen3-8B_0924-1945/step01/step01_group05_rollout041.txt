import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    target_sum = n_points / 2.0
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % (n_points // 3 + 1) == 0:
                h[i] = 1.0
        h = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, n_points), h)
        h = h * target_sum / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h
    best_c5 = objective(h)
    elapsed = 0
    
    for gen in range(100):
        if elapsed >= budget_s:
            break
        population = [best_h.copy() for _ in range(50)]
        for i in range(50):
            if np.random.rand() < 0.1:
                mutation = np.random.rand(n_points) * 2 - 1
                population[i] = np.clip(population[i] + mutation, 0, 1)
            if np.random.rand() < 0.8:
                j = np.random.randint(0, 50)
                crossover_point = np.random.randint(1, n_points - 1)
                population[i] = np.concatenate((population[i][:crossover_point], population[j][crossover_point:]))
            population[i] = population[i] * target_sum / np.sum(population[i])
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        elapsed = time.time() - time.time()
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
