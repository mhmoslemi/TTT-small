```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    h = h.astype(np.float64)
    h = np.clip(h, 0.0, 1.0)

    n_lags = n_points - 1
    n_vars = n_points
    h_var = cp.Variable(n_vars, bounds=(0.0, 1.0))
    C = cp.Variable()

    constraints = []
    for k in range(1, n_lags + 1):
        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = (i + k) % n_points
            A[i, j] = dx
            j = (i - k + n_points) % n_points
            A[i, j] = dx
        expr = cp.quad_form(h_var, A)
        constraints.append(expr >= 1.0 - C)

    constraints.append(cp.sum(h_var) == target_sum)
    prob = cp.Problem(cp.Minimize(C), constraints)

    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()

    for _ in range(2):
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iter=200)
            if prob.status in ['OPTIMAL', 'OPT'] and C.value < best_c5:
                best_c5 = C.value
                best_h = h_var.value
        except:
            pass

        if time.time() - start_time > budget_s:
            break

    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    best_h = best_h * scale

    c5_bound = np.max(np.correlate(best_h, 1.0 - best_h, mode="full") * dx)
    return best_h.tolist(), c5_bound, n_points
```
