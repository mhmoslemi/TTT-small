import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Default number of points and initial setup
    n_points = 90
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Precompute lag indices for valid cross-correlation
    def get_valid_lags(n_points):
        lags = []
        for i in range(n_points):
            for j in range(n_points):
                lag = j - i
                if lag >= 0 and lag < 2 * n_points:
                    if lag < n_points or lag > n_points - 1:
                        lags.append((i, j))
        return lags
    
    valid_lags = get_valid_lags(n_points)
    
    # Prepare QP matrices
    def prepare_qp(n_points, valid_lags):
        n_vars = n_points
        n_constraints = len(valid_lags)
        
        # Objective: minimize mu
        c = np.zeros(n_vars + 1)
        c[-1] = 1.0
        
        # Q matrix (quadratic terms)
        Q = np.zeros((n_vars + 1, n_vars + 1))
        for i, j in valid_lags:
            Q[i, j] += 1.0 / n_points
            Q[j, i] += 1.0 / n_points
        Q = 0.5 * Q  # Ensure symmetry
        
        # A matrix (linear constraints)
        A = np.zeros((n_constraints, n_vars + 1))
        for idx, (i, j) in enumerate(valid_lags):
            A[idx, i] = 1.0 / n_points
            A[idx, j] = 1.0 / n_points
            A[idx, -1] = -1.0
        
        # B vector (right-hand side)
        B = np.zeros(n_constraints)
        B[:] = 1.0 - 1e-8  # Slight slack
        
        return Q, A, B
    
    Q, A, B = prepare_qp(n_points, valid_lags)
    
    # Solve QP
    try:
        from cvxopt import matrix, solvers
        solvers.options['maxiters'] = 1000
        solvers.options['show_progress'] = False
        solvers.options['abstol'] = 1e-8
        solvers.options['reltol'] = 1e-8
        
        c_mat = matrix(c)
        Q_mat = matrix(Q)
        A_mat = matrix(A)
        B_mat = matrix(B)
        
        sol = solvers.qp(Q_mat, c_mat, A_mat, B_mat)
        h_qp = np.array(sol['x'][:n_points])
        mu_qp = sol['x'][-1]
        
        # Check feasibility and normalize
        h_qp = np.clip(h_qp, 0, 1)
        h_qp = h_qp * (n_points / 2) / np.sum(h_qp)
        
        # Re-evaluate C5
        h_values = h_qp
        correlate_vals = np.correlate(h_values, 1 - h_values, mode="full") * (2.0 / n_points)
        c5_bound = np.max(correlate_vals)
        
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            print("Time budget exceeded, returning initial solution")
            h_values = initial_h_values if 'initial_h_values' in globals() else h
            correlate_vals = np.correlate(h_values, 1 - h_values, mode="full") * (2.0 / n_points)
            c5_bound = np.max(correlate_vals)
            return (h_values.tolist(), c5_bound, n_points)
        
        return (h_values.tolist(), c5_bound, n_points)
    
    except Exception:
        # Fallback to differential evolution
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(
            lambda h_vec: np.max(np.correlate(h_vec, 1 - h_vec, mode="full") * (2.0 / n_points)),
            bounds, popsize=40, mutation=(0.5, 1.0), recombination=0.9,
            strategy='best1bin', tol=1e-5, maxiter=200, disp=False, polish=True, seed=seed
        )
        
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        correlate_vals = np.correlate(best_h, 1 - best_h, mode="full") * (2.0 / n_points)
        c5_bound = np.max(correlate_vals)
        
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            print("Time budget exceeded, returning initial solution")
            h_values = initial_h_values if 'initial_h_values' in globals() else h
            correlate_vals = np.correlate(h_values, 1 - h_values, mode="full") * (2.0 / n_points)
            c5_bound = np.max(correlate_vals)
            return (h_values.tolist(), c5_bound, n_points)
        
        return (best_h.tolist(), c5_bound, n_points)
