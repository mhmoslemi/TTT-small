import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    h = h.astype(np.float64)
    h = np.clip(h, 0.0, 1.0)

    def compute_overlap_matrix():
        n = n_points
        A = []
        for k in range(1, n):
            row = np.zeros(n)
            for i in range(n):
                j = (i + k) % n
                row[i] = dx
                j = (i - k + n) % n
                row[i] = dx
            A.append(row)
        return np.array(A)

    A = compute_overlap_matrix()
    n_constraints = A.shape[0]

    h_var = cp.Variable(n_points, low=0.0, up=1.0)
    C = cp.Variable()

    constraints = [
        cp.sum(h_var * dx) == 1.0,
        cp.sum(A[i] * h_var) >= 1.0 - C for i in range(n_constraints)
    ]

    problem = cp.Problem(cp.Minimize(C), constraints)
    solver = cp.SCS() if n_points < 200 else cp.ECOS()
    start_time = time.time()
    result = problem.solve(solver=solver, eps=1e-6, verbose=False)
    elapsed = time.time() - start_time

    h_opt = h_var.value
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt) * dx
    if abs(sum_h_opt - 1.0) > 1e-6:
        scale = 1.0 / sum_h_opt
        h_opt = h_opt * scale

    c5_bound = np.max(np.correlate(h_opt, 1.0 - h_opt, mode="full")) * dx

    if elapsed < budget_s:
        h_opt = np.clip(h_opt, 0.0, 1.0)
        h_var = cp.Variable(n_points, low=0.0, up=1.0)
        constraints = [
            cp.sum(h_var * dx) == 1.0,
            cp.sum(A[i] * h_var) >= 1.0 - C for i in range(n_constraints)
        ]
        problem = cp.Problem(cp.Minimize(C), constraints)
        result = problem.solve(solver=solver, eps=1e-6, verbose=False)
        h_opt = h_var.value
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt) * dx
        if abs(sum_h_opt - 1.0) > 1e-6:
            scale = 1.0 / sum_h_opt
            h_opt = h_opt * scale

        c5_bound = np.max(np.correlate(h_opt, 1.0 - h_opt, mode="full")) * dx

    return h_opt.tolist(), c5_bound, n_points
