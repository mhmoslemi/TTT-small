import numpy as np
import scipy.optimize as opt
import time
from scipy.stats import norm

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define initial configuration
    n_points = 90
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2) / np.sum(h)
    
    # Define lag indices
    lags = np.arange(1, n_points + 1)
    valid_lags = []
    for lag in lags:
        idx = np.arange(n_points - lag + 1)
        if idx.size > 0:
            valid_lags.append((idx, lag))
    valid_lags += [(np.arange(n_points), lag) for lag in range(n_points, 2 * n_points)]
    
    # Define linear and quadratic terms for QP
    A = np.zeros((len(valid_lags), n_points))
    for i, (idx, lag) in enumerate(valid_lags):
        A[i, idx] = 1
    
    # Objective: minimize μ
    # Constraints: A^T h <= 1 - μ
    # Also: np.sum(h) == n_points / 2
    # Define QP problem
    # Since μ is scalar, we minimize μ with constraints
    # For QP, we solve for h and μ
    # Objective: μ
    # Constraints: A^T h + μ <= 1
    # Also: np.sum(h) == n_points / 2
    # Make the problem convex by minimizing μ
    # QP is: minimize μ, subject to A^T h + μ <= 1, sum(h) == n_points/2, 0 <= h <= 1
    # Rewriting: minimize μ, s.t. A^T h + μ <= 1, sum(h) == n_points / 2, 0 <= h <= 1
    
    # Convert to standard QP form: min 0.5 * x^T P x + q^T x
    # Here the objective function is μ, which is a scalar. We need to transform it
    # to involve μ in both the quadratic and linear terms
    # We define x as a concatenation of h and μ, i.e., x = [h; μ]
    # Objective is μ
    # New objective = 0.5 * [0 1]^T [0 0; 0 0] [h; μ] + [0, 1]^T [h; μ] = μ
    # P = [[0, 0], [0, 0]]
    # q = [0, 1]
    # A * x = A^T h + μ <= 1
    # Since μ is part of x, write A * h + μ <= 1
    # New constraint matrix for inequality is [[A, 1]]
    # The equality constraint is sum(h) == n_points / 2 => [1, 0] * x == n_points / 2
    # Bounds: 0 <= h_i <= 1, 0 <= μ <= 1
    
    # Construct problem
    P = np.zeros((n_points + 1, n_points + 1))
    q = np.zeros(n_points + 1)
    q[-1] = 1.0
    G = np.zeros((len(valid_lags), n_points + 1))
    for i, (idx, lag) in enumerate(valid_lags):
        G[i, idx] = 1.0
        G[i, -1] = 1.0
    h_bounds = [(0.0, 1.0) for _ in range(n_points)]
    mu_bounds = [(0.0, 1.0)]
    
    # Solve QP using CVXOPT
    try:
        from cvxopt import matrix, solvers
        solvers.options['show_progress'] = False
        solvers.options['maxiters'] = 100
        solvers.options['abstol'] = 1e-8
        solvers.options['reltol'] = 1e-8
        solvers.options['feastol'] = 1e-8
        
        c = matrix(q)
        G_qp = matrix(G)
        h = matrix(h)
        mu = matrix(0.0)
        A_qp = matrix(np.vstack([np.ones(n_points), np.zeros(n_points)]))
        b_qp = matrix([n_points / 2.0, 0.0])
        A_ineq = matrix(np.hstack([A, np.ones((len(valid_lags), 1))]))
        b_ineq = matrix(np.ones(len(valid_lags)))
        
        solution = solvers.qp(c, G_qp, A_qp, b_qp, A_ineq, b_ineq)
        x = np.array(solution['x']).flatten()
        h_opt = x[:n_points]
        mu_opt = x[-1]
        if np.any(np.isnan(h_opt)) or np.any(np.isinf(h_opt)):
            raise ValueError("Invalid solution found")
        
        # Clip and normalize
        h_opt = np.clip(h_opt, 0.0, 1.0)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        final_c5 = mu_opt
    except:
        # Fallback to differential evolution
        def objective(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0.0, 1.0)
            h = h * (n_points / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(correlate_vals)
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(
            objective, bounds, popsize=40, mutation=(0.5, 1.0), 
            recombination=0.9, strategy='best1bin', tol=1e-5,
            maxiter=200, disp=False, polish=True, seed=seed
        )
        best_h = result.x
        best_c5 = result.fun
        
        # Final check
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            print("Time budget exceeded, returning initial solution")
            h = initial_h_values if 'initial_h_values' in globals() else h
            best_c5 = objective(h)
        
        # Final feasibility check
        h = np.clip(best_h, 0.0, 1.0)
        h = h * (n_points / 2) / np.sum(h)
        final_c5 = objective(h)
        h_opt = h
        
    # Ensure final solution is feasible
    h_opt = np.clip(h_opt, 0.0, 1.0)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    
    # Final evaluation
    correlate_vals = np.correlate(h_opt, 1 - h_opt, mode="full") * dx
    final_c5 = np.max(correlate_vals)
    
    # Time check
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning solution")
    
    return (h_opt.tolist(), final_c5, n_points)
