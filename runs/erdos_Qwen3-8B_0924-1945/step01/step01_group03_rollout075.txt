def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h_values if 'initial_h_values' in locals() else np.random.rand(n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = (initial_h / np.sum(initial_h)) * target_sum
    best_h = initial_h.copy()
    best_c5 = float('inf')
    candidates = []

    def evaluate(h):
        h_vals = np.clip(h, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5

    for _ in range(3):
        h_rand = np.random.rand(n_points)
        h_rand = (h_rand / np.sum(h_rand)) * target_sum
        h_rand = np.clip(h_rand, 0, 1)
        c5_rand = evaluate(h_rand)
        if c5_rand < best_c5:
            best_c5 = c5_rand
            best_h = h_rand.copy()
        candidates.append((h_rand.copy(), c5_rand))

    def objective(x):
        return evaluate(x)

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 200, 'ftol': 1e-6, 'eps': 1e-6}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = (best_h / np.sum(best_h)) * target_sum
    final_c5 = evaluate(best_h)

    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = best_h

    return (best_h, best_c5, n_points)
