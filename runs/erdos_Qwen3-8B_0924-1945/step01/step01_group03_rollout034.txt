import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.copy(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[::5] = 1.0
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * target_sum

    def objective(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def evaluate_candidate(x):
        try:
            c5 = objective(x)
            return c5
        except:
            return np.inf

    def de_callback(x):
        nonlocal best_h, best_c5
        c5 = evaluate_candidate(x)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x.copy()

    best_h = np.copy(h)
    best_c5 = objective(best_h)

    from scipy.optimize import differential_evolution

    result = differential_evolution(
        objective,
        bounds,
        args=(),
        strategy='rand',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-6,
        maxiter=200,
        disp=False,
        polish=True,
        callback=de_callback,
        vectorized=True
    )

    final_h = result.x
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h / np.sum(final_h) * target_sum
    final_c5 = evaluate_candidate(final_h)

    if final_c5 < best_c5:
        best_h = final_h
        best_c5 = final_c5

    return (best_h, best_c5, n_points)
