import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec, n_points):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Coarse grid optimization
    def coarse_optimize(h, n_points):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(
            lambda x: objective(x, n_points),
            bounds,
            popsize=40,
            mutation=(0.8, 1.0),
            recombination=0.9,
            strategy='best1bin',
            tol=1e-5,
            maxiter=300,
            disp=False,
            polish=True,
            seed=seed
        )
        return result.x
    
    # Fine grid refinement
    def fine_optimize(h, n_points):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(
            lambda x: objective(x, n_points),
            bounds,
            popsize=20,
            mutation=(1.0, 1.0),
            recombination=0.9,
            strategy='best1bin',
            tol=1e-6,
            maxiter=200,
            disp=False,
            polish=True,
            seed=seed
        )
        return result.x
    
    # Local search refinement
    def local_search(h, n_points):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.minimize(
            lambda x: objective(x, n_points),
            h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        return result.x
    
    # Stage 1: Coarse grid optimization
    h = coarse_optimize(h, n_points)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return (h.tolist(), objective(h, n_points), n_points)
    
    # Stage 2: Fine grid refinement
    n_points_fine = 90
    h_fine = np.linspace(0, 2, n_points_fine)
    h_interp = np.interp(h_fine, np.linspace(0, 2, n_points), h)
    h_interp = np.clip(h_interp, 0, 1)
    h_interp = h_interp * (n_points_fine / 2) / np.sum(h_interp)
    h = fine_optimize(h_interp, n_points_fine)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        h = np.clip(h, 0, 1)
        h = h * (n_points_fine / 2) / np.sum(h)
        return (h.tolist(), objective(h, n_points_fine), n_points_fine)
    
    # Final local search
    h = local_search(h, n_points_fine)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        h = np.clip(h, 0, 1)
        h = h * (n_points_fine / 2) / np.sum(h)
        return (h.tolist(), objective(h, n_points_fine), n_points_fine)
    
    # Final checks
    h = np.clip(h, 0, 1)
    h = h * (n_points_fine / 2) / np.sum(h)
    final_c5 = objective(h, n_points_fine)
    
    return (h.tolist(), final_c5, n_points_fine)
