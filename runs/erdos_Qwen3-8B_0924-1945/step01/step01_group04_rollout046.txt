import numpy as np
import scipy.optimize as optimize
import numpy.random as npr

def run(seed=42, budget_s=600, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 1.0 / n_points)
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
    
    def objective(h_):
        h_ = np.clip(h_, 0.0, 1.0)
        h_ = h_ * (target_sum / np.sum(h_)) if np.sum(h_) != 0 else h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    def constraint(h_):
        return np.sum(h_) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    def optimize_with_restart(h0):
        result = optimize.minimize(
            objective,
            h0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        h_opt = result.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        h_opt = h_opt * (target_sum / np.sum(h_opt)) if np.sum(h_opt) != 0 else h_opt
        return h_opt, objective(h_opt)
    
    h_opt, c5_bound = optimize_with_restart(h)
    best_h = h_opt
    best_c5 = c5_bound
    time_used = 0
    
    while time_used < budget_s - 50:
        perturbation = npr.uniform(-0.05, 0.05, size=n_points)
        h_perturbed = h_opt + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (target_sum / np.sum(h_perturbed)) if np.sum(h_perturbed) != 0 else h_perturbed
        h_opt_new, c5_new = optimize_with_restart(h_perturbed)
        if c5_new < best_c5:
            best_h = h_opt_new
            best_c5 = c5_new
        time_used += 100
    
    return (best_h.tolist(), best_c5, n_points)
