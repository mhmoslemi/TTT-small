import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 90
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def grad_objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_idx = np.argmax(correlate_vals)
        k = max_idx - (n_points - 1)
        if k < 0 or k >= n_points:
            return np.zeros_like(h)
        h_shifted = np.roll(h, k)
        grad = (1 - h_shifted) * (2.0 / n_points)
        return grad
    
    # Optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(objective, h, bounds=bounds, method='L-BFGS-B', jac=grad_objective,
                          options={'maxiter': 500, 'ftol': 1e-5, 'eps': 1e-8})
    
    best_h = result.x
    best_c5 = result.fun
    
    # Adaptive grid refinement
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        n_points = 120
        h = np.zeros(n_points)
        h[:len(best_h)] = best_h
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        def objective_new(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(correlate_vals)
        
        def grad_objective_new(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            max_idx = np.argmax(correlate_vals)
            k = max_idx - (n_points - 1)
            if k < 0 or k >= n_points:
                return np.zeros_like(h)
            h_shifted = np.roll(h, k)
            grad = (1 - h_shifted) * (2.0 / n_points)
            return grad
        
        result = opt.minimize(objective_new, h, bounds=bounds, method='L-BFGS-B', jac=grad_objective_new,
                              options={'maxiter': 300, 'ftol': 1e-5, 'eps': 1e-8})
        best_h = result.x
        best_c5 = result.fun
    
    # Final check and return
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best candidate")
        h = best_h
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        final_c5 = objective(h)
        return (h.tolist(), final_c5, n_points)
    
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points)
