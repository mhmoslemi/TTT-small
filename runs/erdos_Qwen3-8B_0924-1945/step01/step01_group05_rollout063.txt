import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        idxs = np.round(np.linspace(0, n_points, 6)).astype(int)
        h[idxs] = 1
        h = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, len(idxs)), h[idxs])
        h = (h * n_points / 2).astype(np.float64)
        h = h / np.sum(h) * n_points / 2

    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.inf

    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5

    bounds = [(0, 1) for _ in range(n_points)]

    pop_size = 50
    max_gen = 100
    mut_rate = 0.1
    cross_rate = 0.8
    elite_frac = 0.3
    mut_frac = 0.1

    pop = np.random.rand(pop_size, n_points)
    pop = (pop * target_sum / np.sum(pop, axis=1, keepdims=True)).astype(np.float64)
    pop = np.clip(pop, 0, 1)

    start_time = time.time()
    for gen in range(max_gen):
        if time.time() - start_time > budget_s - 10:
            break
        fit = np.array([objective(ind) for ind in pop])
        elite_idx = np.argsort(fit)[:int(elite_frac * pop_size)]
        best_idx = elite_idx[0]
        best_c5 = fit[best_idx]
        best_h = pop[best_idx].copy()

        new_pop = []
        for _ in range(pop_size // 2):
            p1, p2 = np.random.choice(elite_idx, size=2, replace=False)
            child = pop[p1].copy()
            if np.random.rand() < cross_rate:
                split = np.random.randint(1, n_points - 1)
                child[:split] = pop[p2][:split]
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
            if np.random.rand() < mut_rate:
                mask = np.random.rand(n_points) < mut_frac
                child[mask] = np.random.rand(np.sum(mask))
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
            new_pop.append(child)
        pop = np.array(new_pop)
        pop = np.clip(pop, 0, 1)
        pop = pop * target_sum / np.sum(pop, axis=1, keepdims=True)

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
