import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', None)
    
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        x = np.linspace(0, 2, n_points, endpoint=False)
        h = (1 + np.cos(np.pi * x / 2)) * 0.5
        h = h * (n_points / 2) / (np.sum(h) * 2.0 / n_points)
    
    def normalize_h(h):
        total = np.sum(h)
        if total == 0:
            h += 1e-6
        scale = (n_points / 2) / total
        h = h * scale
        return np.clip(h, 0, 1)
    
    h = normalize_h(h)
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def objective(x):
        h = normalize_h(x)
        return compute_c5(h)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time()
    
    def differential_evolution_stage():
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=20,
            mutation=(0.8, 1.0),
            recombination=0.9,
            tol=1e-6,
            polish=True,
            disp=False,
            maxiter=50
        )
        return result.x
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    if time() - start_time < budget_s:
        de_result = differential_evolution_stage()
        de_h = normalize_h(de_result)
        de_c5 = compute_c5(de_h)
        if de_c5 < best_c5:
            best_h = de_h
            best_c5 = de_c5
    
    if time() - start_time < budget_s:
        cons = [{'type': 'eq', 'fun': constraint}]
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            best_h = normalize_h(res.x)
            best_c5 = compute_c5(best_h)
    
    if time() - start_time < budget_s:
        for _ in range(20):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = normalize_h(h)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5
    
    best_h = normalize_h(best_h)
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n_points)
