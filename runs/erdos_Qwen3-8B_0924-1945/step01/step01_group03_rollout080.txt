import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = np.copy(initial_h_values)
    else:
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sin(np.pi * x)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    
    def objective(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - target_sum

    def evaluate_candidate(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5, h_vals

    bounds = [(0, 1) for _ in range(n_points)]

    def generate_variants(base):
        variants = []
        variants.append(base)
        noise = np.random.uniform(-0.15, 0.15, size=n_points)
        variants.append(base + noise)
        variants.append(np.roll(base, 1))
        variants.append(np.flip(base))
        variants.append(0.5 * base + 0.5 * np.flip(base))
        variants.append(0.5 + 0.5 * np.abs(np.sin(np.pi * np.linspace(0, 2, n_points))))
        variants.append(0.5 + 0.5 * (np.sin(np.pi * np.linspace(0, 2, n_points)) > 0))
        variants.append(0.5 + 0.5 * np.cos(np.pi * np.linspace(0, 2, n_points)))
        variants.append(np.exp(-0.5 * (np.linspace(0, 2, n_points) - 1)**2) / np.sqrt(2 * np.pi))
        variants.append(np.zeros(n_points))
        for i in range(n_points):
            variants.append(np.zeros(n_points))
            variants[i] = 1
        return variants

    variants = generate_variants(h)
    variants = [np.clip(v, 0, 1) * (target_sum / np.sum(np.clip(v, 0, 1))) for v in variants]
    candidates = variants[:10]

    best_c5 = float('inf')
    best_h = None
    best_candidate = None
    start_time = time.time()

    for candidate in candidates:
        result = opt.minimize(
            objective,
            candidate,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-6}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
            best_candidate = candidate
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    if best_h is None:
        best_h = best_candidate

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h))
    final_c5, _ = evaluate_candidate(best_h)

    return (best_h, final_c5, n_points)
