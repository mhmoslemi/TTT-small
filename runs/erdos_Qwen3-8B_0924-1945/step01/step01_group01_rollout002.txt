import numpy as np
import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Function to compute the objective
    def objective(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Function to validate the solution
    def validate(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h, objective(h)
    
    # Coarse-grid optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_coarse = opt.differential_evolution(
        objective, bounds, popsize=40, mutation=(0.8, 1.0), 
        recombination=0.9, strategy='best1bin', tol=1e-5,
        maxiter=300, disp=False, polish=True, seed=seed)
    best_h_coarse, best_c5_coarse = validate(result_coarse.x)
    
    # Refinement to finer grid
    n_points_refine = 90
    dx_refine = 2.0 / n_points_refine
    h_refine = np.interp(np.linspace(0, 2, n_points_refine), np.linspace(0, 2, n_points), best_h_coarse)
    h_refine = np.clip(h_refine, 0, 1)
    h_refine = h_refine * (n_points_refine / 2) / np.sum(h_refine)
    
    # Function for fine-grid optimization
    def objective_refine(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points_refine / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx_refine
        return np.max(correlate_vals)
    
    # Fine-grid optimization
    result_refine = opt.differential_evolution(
        objective_refine, bounds, popsize=20, mutation=(1.0, 1.2), 
        recombination=0.9, strategy='best1bin', tol=1e-6,
        maxiter=200, disp=False, polish=True, seed=seed)
    best_h_refine, best_c5_refine = validate(result_refine.x)
    
    # Local search with L-BFGS-B
    def objective_local(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points_refine / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx_refine
        return np.max(correlate_vals)
    
    result_local = opt.minimize(
        objective_local, result_refine.x, method='L-BFGS-B',
        bounds=bounds, tol=1e-6, maxiter=100, options={'disp': False})
    best_h_local, best_c5_local = validate(result_local.x)
    
    # Final solution selection
    best_h = best_h_local
    final_c5 = best_c5_local
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best refinement solution")
        best_h = best_h_refine
        final_c5 = best_c5_refine
    
    # Final checks
    h = np.clip(best_h, 0, 1)
    h = h * (n_points_refine / 2) / np.sum(h)
    final_c5 = objective_local(h)
    
    return (h.tolist(), final_c5, n_points_refine)
