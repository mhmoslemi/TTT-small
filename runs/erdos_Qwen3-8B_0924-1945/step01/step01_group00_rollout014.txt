import numpy as np
import scipy.optimize as opt
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h * n_points / 2 / np.sum(h)
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h * n_points / 2 / np.sum(h)
        return compute_c5(h)
    
    def objective(x):
        return evaluate_candidate(x)
    
    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * n_points / 2 / np.sum(x)
        return x
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    
    pop_size = 20
    mut_factor = 0.5
    cross_prob = 0.9
    gen_max = 100
    gen_count = 0
    
    def de_step(pop):
        new_pop = []
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + mut_factor * (pop[b] - pop[c])
            trial = np.clip(mutant, 0, 1)
            trial = trial * n_points / 2 / np.sum(trial)
            trial_c5 = compute_c5(trial)
            if trial_c5 < compute_c5(pop[i]):
                new_pop.append(trial)
            else:
                new_pop.append(pop[i])
        return new_pop
    
    pop = [normalize(np.random.rand(n_points)) for _ in range(pop_size)]
    for _ in range(gen_max):
        if time() - start_time > budget_s:
            break
        gen_count += 1
        pop = de_step(pop)
        current_c5 = min([compute_c5(h) for h in pop])
        if current_c5 < best_c5:
            best_h = np.array([h for h in pop if compute_c5(h) == current_c5][0])
            best_c5 = current_c5
    
    if time() - start_time < budget_s:
        for _ in range(50):
            if time() - start_time > budget_s:
                break
            h = np.random.rand(n_points)
            h = normalize(h)
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    for _ in range(15):
        if time() - start_time > budget_s:
            break
        res = opt.minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            best_h = res.x
            best_c5 = evaluate_candidate(best_h)
    
    best_h = normalize(best_h)
    return (best_h, best_c5, n_points)
