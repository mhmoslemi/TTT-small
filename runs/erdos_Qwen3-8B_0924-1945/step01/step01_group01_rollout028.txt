import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 90
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Function to compute C5
    def compute_c5(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Gradient-based optimization (L-BFGS-B)
    def objective(h_vec):
        return compute_c5(h_vec)
    
    def gradient(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_k = np.argmax(correlate_vals)
        grad = np.zeros_like(h)
        for i in range(n_points):
            x = i * (2.0 / n_points)
            if max_k < n_points:
                x_k = x
            else:
                x_k = x - (max_k - n_points + 1) * (2.0 / n_points)
            if 0 <= x_k <= 2:
                grad[i] = (1 - h[int(x_k * n_points / 2) % n_points]) * (2.0 / n_points)
        return grad
    
    # Initial optimization with gradient
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(
        objective,
        h,
        method='L-BFGS-B',
        jac=gradient,
        bounds=bounds,
        options={'ftol': 1e-5, 'gtol': 1e-5, 'maxiter': 500, 'disp': False}
    )
    best_h = result.x
    best_c5 = result.fun
    
    # Adaptive grid refinement
    n_points = 120
    h = np.zeros(n_points)
    h[:len(best_h)] = best_h
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Second optimization with refined grid
    result = opt.minimize(
        objective,
        h,
        method='L-BFGS-B',
        jac=gradient,
        bounds=bounds,
        options={'ftol': 1e-5, 'gtol': 1e-5, 'maxiter': 500, 'disp': False}
    )
    best_h = result.x
    best_c5 = result.fun
    
    # Final check and fallback to DE if needed
    elapsed = time.time() - start_time
    if elapsed > budget_s - 100:
        print("Time budget exceeded, returning refined solution")
        h = np.clip(best_h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        best_c5 = compute_c5(h)
    
    # Final evaluation
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = compute_c5(h)
    
    return (h.tolist(), final_c5, n_points)
