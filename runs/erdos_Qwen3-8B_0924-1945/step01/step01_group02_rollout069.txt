import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    h = h / (np.sum(h) / target_sum)
    h = np.clip(h, 0.0, 1.0)

    h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
    C = cp.Variable()

    constraints = []
    constraints.append(cp.sum(h_var * dx) == 1.0)

    for k in range(1, n_points):
        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            j1 = (i + k) % n_points
            j2 = (i - k + n_points) % n_points
            A[i, j1] = dx
            A[i, j2] = dx
        A = A.astype(np.float64)
        overlap = cp.quad_form(h_var, A)
        constraints.append(overlap >= 1.0 - C)

    prob = cp.Problem(cp.Minimize(C), constraints)

    start_time = time.time()
    try:
        prob.solve(solver=cp.ECOS, eps=1e-6, verbose=False, time_limit=budget_s - (time.time() - start_time))
    except:
        prob.solve(solver=cp.SCS, eps=1e-6, verbose=False, time_limit=budget_s - (time.time() - start_time))

    h_opt = h_var.value
    h_opt = np.clip(h_opt, 0.0, 1.0)
    h_opt = h_opt * (target_sum / np.sum(h_opt * dx))

    corr = np.correlate(h_opt, 1.0 - h_opt, mode="full")
    c5_bound = np.max(corr * dx)

    return h_opt.tolist(), c5_bound, n_points
