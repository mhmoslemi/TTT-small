import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_guesses = []

    if 'initial_h_values' in globals():
        base_guess = np.copy(initial_h_values)
        base_guess = np.clip(base_guess, 0, 1)
        base_guess /= np.sum(base_guess)
        base_guess *= target_sum
        initial_guesses.append(base_guess)

        row_guess = np.zeros(n_points)
        spacing = n_points // 5
        for i in range(n_points):
            if (i % spacing) == 0:
                row_guess[i] = 1.0
        row_guess = np.clip(row_guess, 0, 1)
        row_guess /= np.sum(row_guess)
        row_guess *= target_sum
        initial_guesses.append(row_guess)

        perturbed_guess = base_guess + np.random.uniform(-0.2, 0.2, size=n_points)
        perturbed_guess = np.clip(perturbed_guess, 0, 1)
        perturbed_guess /= np.sum(perturbed_guess)
        perturbed_guess *= target_sum
        initial_guesses.append(perturbed_guess)

        reversed_guess = 1.0 - base_guess
        reversed_guess = np.clip(reversed_guess, 0, 1)
        reversed_guess /= np.sum(reversed_guess)
        reversed_guess *= target_sum
        initial_guesses.append(reversed_guess)

        blended_guess = 0.5 * base_guess + 0.5 * row_guess
        blended_guess = np.clip(blended_guess, 0, 1)
        blended_guess /= np.sum(blended_guess)
        blended_guess *= target_sum
        initial_guesses.append(blended_guess)

    else:
        base_guess = np.zeros(n_points)
        base_guess[0::2] = 1.0
        base_guess = np.clip(base_guess, 0, 1)
        base_guess /= np.sum(base_guess)
        base_guess *= target_sum
        initial_guesses.append(base_guess)

        row_guess = np.zeros(n_points)
        spacing = n_points // 5
        for i in range(n_points):
            if (i % spacing) == 0:
                row_guess[i] = 1.0
        row_guess = np.clip(row_guess, 0, 1)
        row_guess /= np.sum(row_guess)
        row_guess *= target_sum
        initial_guesses.append(row_guess)

        perturbed_guess = base_guess + np.random.uniform(-0.2, 0.2, size=n_points)
        perturbed_guess = np.clip(perturbed_guess, 0, 1)
        perturbed_guess /= np.sum(perturbed_guess)
        perturbed_guess *= target_sum
        initial_guesses.append(perturbed_guess)

        reversed_guess = 1.0 - base_guess
        reversed_guess = np.clip(reversed_guess, 0, 1)
        reversed_guess /= np.sum(reversed_guess)
        reversed_guess *= target_sum
        initial_guesses.append(reversed_guess)

        blended_guess = 0.5 * base_guess + 0.5 * row_guess
        blended_guess = np.clip(blended_guess, 0, 1)
        blended_guess /= np.sum(blended_guess)
        blended_guess *= target_sum
        initial_guesses.append(blended_guess)

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    start_time = opt.time.time()

    for guess in initial_guesses:
        x0 = guess.copy()
        x0 = np.clip(x0, 0, 1)
        x0 /= np.sum(x0)
        x0 *= target_sum

        def objective(x):
            x = np.clip(x, 0, 1)
            x /= np.sum(x)
            x *= target_sum
            corr = np.correlate(x, 1 - x, mode='full')
            c5 = np.max(corr) * dx
            return c5

        def constraint(x):
            return np.sum(x) - target_sum

        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.minimize(objective, x0, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 200, 'ftol': 1e-6, 'eps': 1e-6})

        if opt.time.time() - start_time > budget_s * 0.8:
            break

        if result.success:
            x = result.x
            x = np.clip(x, 0, 1)
            x /= np.sum(x)
            x *= target_sum
            c5 = objective(x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = x.copy()

    if best_c5 == np.inf:
        best_h = initial_guesses[0]
        best_h = np.clip(best_h, 0, 1)
        best_h /= np.sum(best_h)
        best_h *= target_sum
        best_c5 = objective(best_h)

    best_h = np.clip(best_h, 0, 1)
    best_h /= np.sum(best_h)
    best_h *= target_sum
    best_c5 = objective(best_h)

    return (best_h, best_c5, n_points)
