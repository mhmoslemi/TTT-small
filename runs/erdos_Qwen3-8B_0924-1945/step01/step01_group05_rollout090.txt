import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[::3] = 1
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 0, 1)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def evolve():
        population = [np.copy(h) for _ in range(50)]
        best = np.copy(h)
        best_c5 = objective(h)
        gen = 0
        start_time = time.time()
        
        while gen < 100 and time.time() - start_time < budget_s:
            gen += 1
            new_population = []
            for _ in range(50):
                parent1 = np.random.choice(population)
                parent2 = np.random.choice(population)
                child = np.copy(parent1)
                if np.random.rand() < 0.8:
                    split = np.random.randint(1, n_points - 1)
                    child[:split] = parent2[:split]
                if np.random.rand() < 0.1:
                    idxs = np.random.choice(n_points, 5, replace=False)
                    child[idxs] = np.random.rand(5)
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
                new_population.append(child)
            population = new_population
            for indiv in population:
                c5 = objective(indiv)
                if c5 < best_c5:
                    best = np.copy(indiv)
                    best_c5 = c5
            if (time.time() - start_time) > budget_s - 10:
                break
        return best, best_c5
    
    best_h, best_c5 = evolve()
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
