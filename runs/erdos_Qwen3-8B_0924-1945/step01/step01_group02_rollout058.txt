import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = np.zeros(n_points, dtype=np.float64)
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        num_ones = int(np.ceil(n_points / 2))
        h[:num_ones] = 1.0
        h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h))

    def compute_inequalities(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        res = np.zeros(n_points - 1, dtype=np.float64)
        for k in range(1, n_points):
            corr = np.correlate(h, 1 - h, mode='full')
            res[k - 1] = np.max(corr * dx)
        return res

    def evaluate_c5(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def optimize(h_initial, c5_prev, max_iter=200):
        h = cp.Variable(n_points)
        c = cp.Variable()
        constraints = [
            cp.sum(h) == target_sum,
            0 <= h <= 1
        ]
        for k in range(1, n_points):
            corr = np.correlate(h.value, 1 - h.value, mode='full') if h.value is not None else np.zeros(n_points * 2 - 1)
            lb = np.min(corr * dx) if h.value is not None else 0
            constraints.append(cp.sum(cp.multiply(h, cp.roll(h, k))) >= 1 - c - lb)
        prob = cp.Problem(cp.Minimize(c), constraints)
        result = prob.solve(solver=cp.ECOS, eps=1e-8, max_iter=max_iter)
        h_opt = h.value
        c_opt = c.value
        if h_opt is not None and c_opt is not None:
            h_opt = np.clip(h_opt, 0.0, 1.0)
            h_opt = h_opt * (target_sum / np.sum(h_opt))
        return h_opt, c_opt

    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    elapsed_time = 0
    for _ in range(30):
        if elapsed_time >= budget_s:
            break
        h_initial = h.copy()
        np.random.shuffle(h_initial)
        h_initial = h_initial * (target_sum / np.sum(h_initial))
        h_initial = np.clip(h_initial, 0.0, 1.0)
        h_initial = h_initial * (target_sum / np.sum(h_initial))
        h_opt, c_opt = optimize(h_initial, best_c5, max_iter=200)
        if c_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c_opt
        elapsed_time = time.time() - time.time()
    return best_h.tolist(), best_c5, n_points
