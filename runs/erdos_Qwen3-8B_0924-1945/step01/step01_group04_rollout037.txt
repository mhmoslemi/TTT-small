import numpy as np
import scipy.optimize as optimize
import numpy.random as npr

def run(seed=42, budget_s=600, **kwargs):
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 0.5)
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
    
    def objective(h_):
        h_ = np.clip(h_, 0.0, 1.0)
        h_ = h_ * (target_sum / np.sum(h_)) if np.sum(h_) != 0 else h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def constraint(h_):
        return np.sum(h_) - target_sum
    
    def de_objective(v):
        v = np.clip(v, 0.0, 1.0)
        v = v * (target_sum / np.sum(v)) if np.sum(v) != 0 else v
        corr = np.correlate(v, 1 - v, mode='full')
        return np.max(corr) * dx
    
    def de_constraint(v):
        return np.sum(v) - target_sum
    
    def de_mutate(a, b, c, F):
        mutant = a + F * (b - c)
        mutant = np.clip(mutant, 0.0, 1.0)
        return mutant
    
    def de_crossover(parent, mutant, P_crossover):
        child = np.copy(parent)
        for i in range(n_points):
            if np.random.rand() < P_crossover:
                child[i] = mutant[i]
        return child
    
    def de_enforce_constraints(v):
        v = np.clip(v, 0.0, 1.0)
        v = v * (target_sum / np.sum(v)) if np.sum(v) != 0 else v
        return v
    
    population_size = 3 * n_points
    population = []
    for _ in range(population_size):
        candidate = np.random.rand(n_points)
        candidate = np.clip(candidate, 0.0, 1.0)
        candidate = candidate * (target_sum / np.sum(candidate)) if np.sum(candidate) != 0 else candidate
        population.append(candidate)
    
    best = np.copy(h)
    best_c5 = objective(h)
    
    start_time = np.datetime64('now')
    
    for gen in range(200):
        if np.datetime64('now') - start_time > np.datetime64(f'{budget_s}s'):
            break
        new_population = []
        for _ in range(population_size // 2):
            a, b, c = np.random.choice(population, 3, replace=False)
            mutant = de_mutate(a, b, c, 0.8)
            child = de_crossover(a, mutant, 0.7)
            child = de_enforce_constraints(child)
            new_population.append(child)
        population = new_population + population[:population_size // 2]
        for candidate in population:
            c5 = objective(candidate)
            if c5 < best_c5:
                best = np.copy(candidate)
                best_c5 = c5
    
    best = np.clip(best, 0.0, 1.0)
    best = best * (target_sum / np.sum(best)) if np.sum(best) != 0 else best
    
    return (best.tolist(), best_c5, n_points)
