import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45
    h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def l_bfgs_step(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Identify the dominant k
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_k = np.argmax(correlate_vals)
        max_val = correlate_vals[max_k]
        
        # Compute gradient for dominant k
        grad = np.zeros(n_points)
        dx = 2.0 / n_points
        for i in range(n_points):
            x = i * dx
            if (x + max_k * dx) < 2:
                k = int(np.round((x + max_k * dx) / dx))
                if 0 <= k < n_points:
                    grad[i] = (1 - h[k]) * dx
        
        # Use L-BFGS-B with gradient
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.minimize(objective, h, method='L-BFGS-B', jac=grad, bounds=bounds, tol=1e-6, options={'maxiter': 300})
        return result.x
    
    best_h = h
    best_c5 = objective(h)
    
    # Differential evolution stage
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.differential_evolution(objective, bounds, popsize=50, mutation=(0.8, 1.0), 
                                       recombination=0.9, strategy='best1bin', tol=1e-5,
                                       maxiter=200, disp=False, polish=True, seed=seed)
    
    best_h = result.x
    best_c5 = result.fun
    
    # Gradient refinement
    for _ in range(3):
        h = l_bfgs_step(best_h)
        c5 = objective(h)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        else:
            break
    
    # Multi-scale refinement
    elapsed = time.time() - start_time
    if elapsed < budget_s - 100:

        dx_new = 2.0 / n_points_new
        h_new = np.zeros(n_points_new)

        h_new = h_new * (n_points_new / 2) / np.sum(h_new)
        
        def objective_new(h_vec):

            h = h * (n_points_new / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full") * dx_new
            return np.max(correlate_vals)
        
        result = opt.differential_evolution(objective_new, [(0.0, 1.0) for _ in range(n_points_new)], 
                                           popsize=30, mutation=(0.9, 1.0), recombination=0.7, 
                                           strategy='best1bin', tol=1e-5, maxiter=100, disp=False, polish=True, seed=seed)
        best_h_new = result.x
        best_c5_new = result.fun
        if best_c5_new < best_c5:
            best_h = best_h_new
            best_c5 = best_c5_new
    
    # Final check
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
    
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points)
