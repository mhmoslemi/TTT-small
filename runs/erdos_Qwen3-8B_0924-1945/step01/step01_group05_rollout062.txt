import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        idxs = np.arange(n_points)
        h[idxs % 3 == 0] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = h
    best_c5 = objective(h)
    
    for gen in range(100):
        population = []
        for _ in range(50):
            h_new = np.copy(best_h)
            mask = np.random.rand(n_points) < 0.1
            h_new[mask] = np.random.rand(np.sum(mask))
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            population.append(h_new)
        
        fitness = [objective(h) for h in population]
        sorted_idx = np.argsort(fitness)
        population = [population[i] for i in sorted_idx[:30]]
        
        new_population = []
        for _ in range(25):
            p1, p2 = np.random.choice(population, 2, replace=False)
            crossover = np.random.rand(n_points) < 0.8
            child = p1.copy()
            child[crossover] = p2[crossover]
            child = child * (n_points / 2) / np.sum(child)
            new_population.append(child)
        
        population += new_population
        population = [np.clip(p, 0, 1) for p in population]
        population = [p * (n_points / 2) / np.sum(p) for p in population]
        
        fitness = [objective(p) for p in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
