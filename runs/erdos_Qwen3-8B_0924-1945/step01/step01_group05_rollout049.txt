import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial solution (if available, otherwise create a random one)
    n_points = 100
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = (h - np.min(h)) / (np.max(h) - np.min(h))
        h = np.clip(h, 0, 1)
        h = (h * n_points / 2).astype(np.float64)
        h = h / np.sum(h) * n_points / 2
    
    # Ensure sum(h) = n_points / 2
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Differential evolution setup
    pop_size = 15
    max_iter = 100
    mutation = (0.5, 1)
    recombination = 0.8
    
    # Optimization settings
    start_time = time.time()
    result = opt.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=pop_size,
        mutation=mutation,
        recombination=recombination,
        maxiter=max_iter,
        tol=1e-6,
        seed=seed,
        disp=False
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h, objective(h), n_points)
    
    best_h = result.x
    best_c5 = result.fun
    
    # Final sanity checks
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
