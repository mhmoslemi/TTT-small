import numpy as np
import time
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', None)
    
    def _round_to_binary(h):
        return np.round(h).astype(int)
    
    def _ensure_half_ones(h):
        ones = np.sum(h)
        if ones < n_points / 2:
            diff = int(n_points / 2 - ones)
            indices = np.where(h == 0)[0]
            h[indices[:diff]] = 1
        elif ones > n_points / 2:
            diff = int(ones - n_points / 2)
            indices = np.where(h == 1)[0]
            h[indices[:diff]] = 0
        return h
    
    if initial_h_values is not None:
        h = _round_to_binary(initial_h_values.copy())
        h = _ensure_half_ones(h)
    else:
        h = np.zeros(n_points, dtype=int)
        h[:int(n_points / 2)] = 1
        h = _ensure_half_ones(h)
    
    dx = 2.0 / n_points
    
    def compute_c5(h):
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        h = _ensure_half_ones(h)
        h = h.astype(float)
        return compute_c5(h)
    
    def fitness(h):
        return evaluate_candidate(h)
    
    def objective(x):
        return fitness(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    
    def _ga_step(population, best_h, best_c5):
        new_population = []
        for i in range(len(population)):
            for j in range(i + 1, len(population)):
                parent1 = population[i]
                parent2 = population[j]
                cross_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:cross_point], parent2[cross_point:]])
                child2 = np.concatenate([parent2[:cross_point], parent1[cross_point:]])
                child1 = _ensure_half_ones(child1)
                child2 = _ensure_half_ones(child2)
                new_population.append(child1)
                new_population.append(child2)
        new_population = np.array(new_population)
        np.random.shuffle(new_population)
        return new_population[:len(population)]
    
    def _mutate(individual):
        mutated = individual.copy()
        num_mutations = np.random.poisson(1)
        for _ in range(num_mutations):
            idx = np.random.randint(n_points)
            mutated[idx] = 1 - mutated[idx]
        mutated = _ensure_half_ones(mutated)
        return mutated
    
    population = np.array([h.copy() for _ in range(30)])
    best_h = population[0].copy()
    best_c5 = evaluate_candidate(best_h)
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        population = _ga_step(population, best_h, best_c5)
        for i in range(len(population)):
            if time.time() - start_time > budget_s:
                break
            individual = _mutate(population[i])
            current_c5 = evaluate_candidate(individual)
            if current_c5 < best_c5:
                best_h = individual.copy()
                best_c5 = current_c5
    
    def _local_refinement(h):
        h_cont = h.astype(float)
        def local_objective(x):
            x = np.clip(x, 0, 1)
            x = x / np.sum(x) * n_points / 2
            return evaluate_candidate(x)
        res = minimize(local_objective, h_cont, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        return res.x
    
    best_h = _local_refinement(best_h)
    best_c5 = evaluate_candidate(best_h)
    
    if time.time() - start_time < budget_s:
        for _ in range(15):
            if time.time() - start_time > budget_s:
                break
            h = np.random.randint(0, 2, n_points)
            h = _ensure_half_ones(h)
            h = h.astype(float)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    
    return (best_h, current_c5, n_points)
