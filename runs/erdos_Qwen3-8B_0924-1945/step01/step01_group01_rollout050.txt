import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 90
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Precompute valid indices for each lag
    def valid_indices(n_points):
        indices = []
        for s in range(n_points):
            row = []
            for i in range(n_points):
                j = (i + s) % n_points
                if j < n_points:
                    row.append(j)
            indices.append(row)
        for s in range(n_points, 2 * n_points):
            row = []
            for i in range(n_points):
                j = i + s - n_points
                if j < n_points:
                    row.append(j)
            indices.append(row)
        return indices
    
    valid_indices_cache = valid_indices(n_points)
    
    # Objective function for QP
    def qp_objective(mu, h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        constraints = []
        for lag in range(2 * n_points):
            indices = valid_indices_cache[lag]
            quad_term = np.sum(h[indices] * h[indices]) * dx
            constraints.append(quad_term <= 1 - mu)
        return mu
    
    # QP formulation
    def solve_qp(h_initial):
        h = h_initial.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        G = []
        hvec = h.copy()
        for lag in range(2 * n_points):
            indices = valid_indices_cache[lag]
            G.append(np.zeros(n_points))
            for i in indices:
                G[lag][i] = 2 * hvec[i]
        G = np.array(G)
        hvec = hvec.copy()
        c = np.zeros(n_points)
        for lag in range(2 * n_points):
            indices = valid_indices_cache[lag]
            for i in indices:
                c[i] += hvec[i]
        A = np.ones(n_points)
        b = np.array([n_points / 2])
        lb = np.zeros(n_points)
        ub = np.ones(n_points)
        
        # Initial mu guess based on current h
        current_mu = 0.0
        for lag in range(2 * n_points):
            indices = valid_indices_cache[lag]
            quad_term = np.sum(h[indices] * h[indices]) * dx
            current_mu = max(current_mu, 1 - quad_term)
        
        try:
            from cvxopt import matrix, solvers
            solvers.options['show_progress'] = False
            solvers.options['abstol'] = 1e-8
            solvers.options['reltol'] = 1e-8
            solvers.options['feastol'] = 1e-8
            solvers.options['maxiters'] = 1000
            P = matrix(G.T @ G, (n_points, n_points))
            q = matrix(-c)
            G_qp = matrix(G.T, (2 * n_points, n_points))
            h_qp = matrix(np.array([-1.0 for _ in range(2 * n_points)]), (2 * n_points, 1))
            A_qp = matrix(A, (1, n_points))
            b_qp = matrix(b)
            sol = solvers.qp(P, q, G_qp, h_qp, A_qp, b_qp)
            h_opt = np.array(sol['x']).flatten()
            mu = current_mu
            if 'x' in sol:
                mu = 1 - np.min(sol['x'])
            return h_opt, mu
        except:
            return h, current_mu
    
    # Solve QP
    h_qp, mu_qp = solve_qp(h)
    
    # Fallback to differential evolution if QP fails
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.differential_evolution(objective, bounds, popsize=40, mutation=(0.5, 1.0), 
                                       recombination=0.9, strategy='best1bin', tol=1e-5,
                                       maxiter=200, disp=False, polish=True, seed=seed)
    
    best_h = result.x
    best_c5 = result.fun
    
    # Final check
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning QP solution")
        h = h_qp
        best_c5 = objective(h)
    
    # Final feasibility check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points)
