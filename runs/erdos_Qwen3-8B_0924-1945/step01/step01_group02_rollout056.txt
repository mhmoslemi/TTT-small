```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[::3] = 1.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def build_constraints(h):
        n = n_points
        A = []
        for k in range(1, n):
            a = np.zeros((n, n))
            for i in range(n):
                j = (i + k) % n
                a[i, j] = dx
                j = (i - k + n) % n
                a[i, j] = dx
            A.append(a)
        return A

    A = build_constraints(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    elapsed = 0

    for _ in range(20):
        h = np.clip(np.random.rand(n_points), 0.0, 1.0)
        h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))

        variables = [cp.Variable(n_points), cp.Variable()]
        constraints = [
            cp.sum(variables[0]) == target_sum,
            0 <= variables[0],
            variables[0] <= 1.0
        ]
        for k in range(len(A)):
            constraints.append(cp.quad_form(variables[0], A[k]) >= 1.0 - variables[1])

        prob = cp.Problem(cp.Minimize(variables[1]), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-8, max_iter=100, warn=False)
            if prob.status == 'optimal':
                h_opt = variables[0].value
                c5 = variables[1].value
                h_opt = np.clip(h_opt, 0.0, 1.0)
                sum_h = np.sum(h_opt)
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                h_opt = h_opt * scale
                c5_val = evaluate(h_opt)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt.copy()
                elapsed += time.time() - time.time()
        except:
            pass

        if elapsed >= budget_s:
            break

    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    best_h = best_h * scale
    c5_bound = evaluate(best_h)
    return best_h.tolist(), c5_bound, n_points
```
