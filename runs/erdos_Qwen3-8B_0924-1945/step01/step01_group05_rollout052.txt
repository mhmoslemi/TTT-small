import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Determine n_points based on budget and performance feedback
    n_points = 200
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Create base pattern: evenly spaced 1s and 0s
        h = np.zeros(n_points)
        step = n_points // 3
        for i in range(step, n_points, step):
            h[i] = 1.0
        # Refine by linear interpolation and normalization
        h = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * n_points / 2 / np.sum(h)
    
    # Ensure sum(h) = n_points / 2
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # GA settings
    pop_size = 50
    max_gen = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = np.random.rand(pop_size, n_points)
    population = np.clip(population, 0, 1)
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    best_h = h
    best_c5 = objective(h)
    best_gen = 0
    
    start_time = time.time()
    
    for gen in range(max_gen):
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        # Select top performers
        elite_idx = np.argsort(fitness)[:int(pop_size * 0.3)]
        elite = population[elite_idx]
        # Generate offspring
        offspring = []
        for _ in range(pop_size - len(elite)):
            # Crossover
            if np.random.rand() < crossover_rate:
                p1, p2 = np.random.choice(elite, 2, replace=False)
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
            else:
                child = np.random.rand(n_points)
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation_mask = np.random.rand(n_points) < 0.1
                child[mutation_mask] = np.random.rand(np.sum(mutation_mask))
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            offspring.append(child)
        # Combine elite and offspring
        population = np.vstack([elite, offspring])
        # Update best solution
        current_best_idx = np.argmin(fitness)
        current_best = population[current_best_idx]
        current_c5 = fitness[current_best_idx]
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
            best_gen = gen
        # Early termination check
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        # Adjust mutation rate based on time and progress
        if elapsed < 200:
            mutation_rate += 0.05
        elif elapsed < 500:
            mutation_rate += 0.025
        # Add random solutions to maintain diversity
        if np.random.rand() < 0.1:
            random_sol = np.random.rand(n_points)
            random_sol = np.clip(random_sol, 0, 1)
            random_sol = random_sol * target_sum / np.sum(random_sol)
            population = np.vstack([population, random_sol])
    
    # Final sanity check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
