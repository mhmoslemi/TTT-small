import numpy as np
from scipy.optimize import minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    h = np.round(initial_h_values).copy()
    while np.sum(h) != n_points // 2:
        if np.sum(h) < n_points // 2:
            idx = np.where(h == 0)[0]
            if idx.size > 0:
                h[idx[0]] = 1
        else:
            idx = np.where(h == 1)[0]
            if idx.size > 0:
                h[idx[0]] = 0

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.round(h)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time()

    def swap_segments(a, b):
        m = np.random.randint(1, n_points - 1)
        n = np.random.randint(m, n_points)
        return np.concatenate([a[:m], b[m:n], a[n:]])

    def mutate(h):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < 0.1:
                h[i] = 1 - h[i]
        while np.sum(h) != n_points // 2:
            if np.sum(h) < n_points // 2:
                idx = np.where(h == 0)[0]
                if idx.size > 0:
                    h[idx[0]] = 1
            else:
                idx = np.where(h == 1)[0]
                if idx.size > 0:
                    h[idx[0]] = 0
        return h

    def crossover(a, b):
        if np.random.rand() < 0.8:
            m = np.random.randint(1, n_points - 1)
            n = np.random.randint(m, n_points)
            return np.concatenate([a[:m], b[m:n], a[n:]])
        return a

    def run_ga():
        pop = [np.random.choice([0, 1], size=n_points, p=[0.5, 0.5]) for _ in range(30)]
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            fit = np.array([fitness(ind) for ind in pop])
            idxs = np.argsort(fit)[:int(len(pop) * 0.2)]
            selected = [pop[i] for i in idxs]
            new_pop = []
            while len(new_pop) < len(pop):
                a, b = np.random.choice(selected, size=2, replace=False)
                child = crossover(a, b)
                child = mutate(child)
                new_pop.append(child)
            pop = new_pop
        best_h = min(pop, key=fitness)
        best_c5 = fitness(best_h)
        return best_h, best_c5

    best_h, best_c5 = run_ga()

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
            while np.sum(h) != n_points // 2:
                if np.sum(h) < n_points // 2:
                    idx = np.where(h == 0)[0]
                    if idx.size > 0:
                        h[idx[0]] = 1
                else:
                    idx = np.where(h == 1)[0]
                    if idx.size > 0:
                        h[idx[0]] = 0
            current_c5 = fitness(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.round(best_h).copy()
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = compute_c5(best_h)

    return (best_h, current_c5, n_points)
