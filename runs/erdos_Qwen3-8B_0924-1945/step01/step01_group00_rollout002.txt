import numpy as np
import scipy.optimize as opt
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    def run_DE():
        pop_size = 20
        pop = np.random.uniform(0, 1, (pop_size, n_points))
        pop = pop / np.sum(pop, axis=1, keepdims=True) * n_points / 2
        for _ in range(100):
            if time() - start_time > budget_s:
                return best_h, best_c5
            new_pop = []
            for i in range(pop_size):
                a, b, c = np.random.choice(pop_size, 3, replace=False)
                mutant = pop[a] + 0.5 * (pop[b] - pop[c])
                mutant = np.clip(mutant, 0, 1)
                mutant = mutant / np.sum(mutant) * n_points / 2
                trial = mutant
                if np.random.rand() < 0.9:
                    trial = pop[i]
                trial = np.clip(trial, 0, 1)
                trial = trial / np.sum(trial) * n_points / 2
                new_pop.append(trial)
            pop = np.array(new_pop)
            for candidate in pop:
                current_c5 = evaluate_candidate(candidate)
                if current_c5 < best_c5:
                    best_h = candidate.copy()
                    best_c5 = current_c5
        return best_h, best_c5

    best_h, best_c5 = run_DE()

    for _ in range(15):
        if time() - start_time > budget_s:
            break
        res = opt.minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            current_h = res.x
            current_c5 = evaluate_candidate(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            break

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
