import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if hasattr(initial_h_values, '__iter__'):
        h = np.copy(initial_h_values)
        h = np.clip(h, 0, 1)
        h /= np.sum(h)
        h *= target_sum
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sin(np.pi * x)
        h = np.clip(h, 0, 1)
        h /= np.sum(h)
        h *= target_sum

    def obj(x):
        x = np.clip(x, 0, 1)
        x = x / np.sum(x) * target_sum
        corr = np.correlate(x, 1 - x, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def cons(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': cons}, options={'maxiter': 200, 'ftol': 1e-6, 'eps': 1e-6})
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * target_sum
    c5_bound = obj(best_h)
    return (best_h, c5_bound, n_points)
