import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def find_critical_lag(h):
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_index = np.argmax(correlate_vals)
        if max_index <= len(h) - 1:
            return max_index - len(h) + 1
        else:
            return max_index - len(h)
    
    def gradient(h):
        k_star = find_critical_lag(h)
        dx = 2.0 / n_points
        gradient = np.zeros_like(h)
        for i in range(n_points):
            if i + k_star < n_points:
                gradient[i] = (1 - h[i + k_star]) * dx
        return gradient
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    for _ in range(100):
        grad = gradient(h)
        result = opt.line_search.scalar_search_line_newton(
            f=objective, x0=h, grad=grad, bounds=bounds, method='BFGS'
        )
        if result.success:
            h = result.x
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        else:
            break
    
    if time.time() - start_time < budget_s:
        h = best_h.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        result = opt.differential_evolution(
            objective, bounds, popsize=30, mutation=(0.9, 1.0), 
            recombination=0.7, strategy='best1bin', tol=1e-5,
            maxiter=200, disp=False, polish=True, seed=seed
        )
        best_h = result.x
        best_c5 = result.fun
    
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
    
    return (h.tolist(), final_c5, n_points)
