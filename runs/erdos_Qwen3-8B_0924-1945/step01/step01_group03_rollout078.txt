import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_initial_guesses = 5
    max_iterations = 200
    tolerance = 1e-6
    early_stop = 100
    candidates = []

    # Initial guess from provided values
    if 'initial_h_values' in globals():
        h_base = np.copy(initial_h_values)
    else:
        # Fallback to uniform distribution
        h_base = np.full(n_points, 0.5)

    h_base = np.clip(h_base, 0, 1)
    h_base = h_base * target_sum / np.sum(h_base)

    # Generate initial guesses
    def generate_guess(base, variation):
        guess = base.copy()
        if variation == 'structured':
            spacing = n_points // 5
            indices = np.arange(0, n_points, spacing)
            guess[indices] = 1.0
            guess = np.clip(guess, 0, 1)
            guess = guess * target_sum / np.sum(guess)
        elif variation == 'perturbed':
            noise = np.random.normal(0, 0.2, n_points)
            guess += noise
            guess = np.clip(guess, 0, 1)
            guess = guess * target_sum / np.sum(guess)
        elif variation == 'reversed':
            guess = guess[::-1]
        elif variation == 'blended':
            guess = 0.5 * base + 0.5 * np.random.rand(n_points)
        else:
            guess = base
        return guess

    initial_guesses = [
        generate_guess(h_base, 'structured'),
        generate_guess(h_base, 'perturbed'),
        generate_guess(h_base, 'reversed'),
        generate_guess(h_base, 'blended'),
        h_base
    ]

    # Optimization loop
    best_c5 = np.inf
    best_h = np.zeros(n_points)

    for guess in initial_guesses:
        h = guess.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        def objective(x):
            x = np.clip(x, 0, 1)
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx

        def constraint(x):
            return np.sum(x) - target_sum

        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': max_iterations, 'ftol': tolerance, 'eps': tolerance}
        )

        if result.success:
            final_h = np.clip(result.x, 0, 1)
            final_h = final_h * target_sum / np.sum(final_h)
            current_c5 = objective(final_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = final_h

    # Final evaluation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    return (best_h, final_c5, n_points)
