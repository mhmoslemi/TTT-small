import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def compute_gradient(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_idx = np.argmax(correlate_vals)
        k = max_idx - n_points + 1
        if k < 0:
            k = 0
        elif k >= n_points:
            k = n_points - 1
        gradient = np.zeros_like(h)
        for i in range(n_points):
            x = i * 2.0 / n_points
            if x + k < 2.0:
                gradient[i] = (1 - h[int((x + k) * n_points / 2)]) * 2.0 / n_points
        return gradient
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.differential_evolution(objective, bounds, popsize=30, mutation=(0.7, 1.0), 
                                       recombination=0.9, strategy='best1bin', tol=1e-6,
                                       maxiter=200, disp=False, polish=True, seed=seed)
    best_h = result.x
    best_c5 = result.fun
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning initial solution")
        h = initial_h_values if 'initial_h_values' in globals() else h
        best_c5 = objective(h)
    
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    # Multi-scale refinement
    for _ in range(3):
        if elapsed > budget_s:
            break
        n_points = min(90, n_points * 2)
        dx = 2.0 / n_points
        h_new = np.zeros(n_points)
        h_new[:len(h)] = h
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        def objective_new(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(correlate_vals)
        
        result_new = opt.minimize(objective_new, h_new, method='L-BFGS-B', jac=compute_gradient, 
                                 bounds=[(0.0, 1.0) for _ in range(n_points)], tol=1e-6, 
                                 options={'maxiter': 100, 'disp': False})
        best_h = result_new.x
        best_c5 = result_new.fun
        h = np.clip(best_h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        final_c5 = objective_new(h)
        elapsed = time.time() - start_time
    
    return (h.tolist(), final_c5, n_points)
