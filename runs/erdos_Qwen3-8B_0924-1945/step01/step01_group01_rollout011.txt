import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Default settings
    n_points = 90
    dx = 2.0 / n_points
    
    # Initial configuration
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Function to compute overlap integral
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    # Precompute valid indices for each lag
    lags = np.arange(n_points)
    valid_lags = []
    for s in range(1, 2 * n_points - 1):
        valid = np.zeros(n_points, dtype=bool)
        for i in range(n_points):
            j = (i + s) % n_points
            valid[i] = True
        valid_lags.append(valid)
    
    # QP formulation
    def qp_objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return np.max([np.sum(h[i] * h[j] for i, j in enumerate(range(n_points)) if valid_lags[s][i]) * dx for s in range(len(valid_lags))])
    
    # QP constraints
    def qp_constraints(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return [np.sum(h[i] * h[j] for i, j in enumerate(range(n_points)) if valid_lags[s][i]) * dx - 1 for s in range(len(valid_lags))]
    
    # Optimization using CVXOPT
    def solve_qp():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        try:
            G = np.zeros((len(valid_lags), n_points))
            for s in range(len(valid_lags)):
                for i in range(n_points):
                    if valid_lags[s][i]:
                        G[s, i] = 1
            G = -G
            h0 = np.random.rand(n_points)
            h0 = np.clip(h0, 0, 1)
            h0 = h0 * (n_points / 2) / np.sum(h0)
            c = np.zeros(n_points)
            A = np.ones((1, n_points))
            b = np.array([n_points / 2])
            P = np.zeros((n_points, n_points))
            for s in range(len(valid_lags)):
                for i in range(n_points):
                    if valid_lags[s][i]:
                        P[i, i] += 1
            P = P * dx
            sol = opt.slsqp(lambda x: np.dot(x, np.dot(P, x)), 
                            lambda x: np.dot(A, x) - b, 
                            bounds=[(0, 1) for _ in range(n_points)], 
                            eqcons=[lambda x: np.dot(A, x) - b], 
                            jac=lambda x: np.dot(P, x) * 2, 
                            disp=False)
            return sol.x
        except:
            return h
    
    # Try solving QP
    try:
        best_h = solve_qp()
        best_c5 = compute_c5(best_h)
    except:
        best_h = h
        best_c5 = compute_c5(best_h)
    
    # Fallback to DE if needed
    if best_c5 > 0.38:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(compute_c5, bounds, popsize=40, mutation=(0.5, 1.0), 
                                           recombination=0.9, strategy='best1bin', tol=1e-5,
                                           maxiter=200, disp=False, polish=True, seed=seed)
        best_h = result.x
        best_c5 = result.fun
    
    # Final checks
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
    
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = compute_c5(h)
    
    return (h.tolist(), final_c5, n_points)
