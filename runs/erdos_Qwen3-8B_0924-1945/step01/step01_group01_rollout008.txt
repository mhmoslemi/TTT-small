import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def grad_estimate(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        k_star = np.argmax(correlate_vals)
        if k_star < n_points:
            grad = np.zeros(n_points)
            for i in range(n_points):
                if i + k_star < n_points:
                    grad[i] = (1 - h[i + k_star]) * (2.0 / n_points)
                if i - k_star >= 0:
                    grad[i] -= (1 - h[i - k_star]) * (2.0 / n_points)
        else:
            grad = np.zeros(n_points)
        return grad
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result_de = opt.differential_evolution(objective, bounds, popsize=50, mutation=(0.8, 1.0), 
                                          recombination=0.9, strategy='best1bin', tol=1e-5,
                                          maxiter=200, disp=False, polish=True, seed=seed)
    best_h = result_de.x
    best_c5 = result_de.fun
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning initial solution")
        h = initial_h_values if 'initial_h_values' in globals() else h
        best_c5 = objective(h)
    
    def l_bfgs_step(h_vec, max_iter=300, tol=1e-6):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, jac=grad_estimate, 
                              tol=tol, options={'maxiter': max_iter, 'ftol': 1e-9, 'gtol': 1e-9})
        return result.x, result.fun
    
    h_refined, c5_refined = l_bfgs_step(best_h)
    best_h = h_refined
    best_c5 = c5_refined
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning refined solution")
        h = best_h
        best_c5 = objective(h)
    
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points)
