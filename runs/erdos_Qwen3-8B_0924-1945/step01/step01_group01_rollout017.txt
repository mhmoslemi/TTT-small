import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration with optional parent
    n_points = 90
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Objective function
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Gradient-based optimization
    def grad_objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_idx = np.argmax(correlate_vals)
        k = max_idx - n_points + 1
        if k < 0 or k >= n_points:
            return np.zeros_like(h)
        grad = np.zeros_like(h)
        for i in range(n_points):
            x = i * (2.0 / n_points)
            x_k = x + k * (2.0 / n_points)
            if 0 <= x_k < 2:
                grad[i] = (1 - h[np.clip(int(x_k / (2.0 / n_points)), 0, n_points - 1)]) * (2.0 / n_points)
        return grad
    
    # LBFGS-B optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(objective, h, method='L-BFGS-B', jac=grad_objective, bounds=bounds,
                          tol=1e-5, options={'maxiter': 500, 'ftol': 1e-6, 'gtol': 1e-5, 'maxfun': 1000})
    
    best_h = result.x
    best_c5 = result.fun
    
    # Adaptive grid refinement
    elapsed = time.time() - start_time
    if elapsed < budget_s - 100:
        n_points = 120
        h = np.zeros(n_points)
        h[:len(best_h)] = best_h
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.minimize(objective, h, method='L-BFGS-B', jac=grad_objective, bounds=bounds,
                              tol=1e-5, options={'maxiter': 300, 'ftol': 1e-6, 'gtol': 1e-5, 'maxfun': 500})
        best_h = result.x
        best_c5 = result.fun
    
    # Fallback to differential evolution if stuck
    if best_c5 > 0.383 and time.time() - start_time < budget_s - 100:
        def de_objective(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(correlate_vals)
        result = opt.differential_evolution(de_objective, bounds, popsize=30, mutation=(0.9, 1.0),
                                            recombination=0.7, strategy='best1bin', tol=1e-5,
                                            maxiter=200, disp=False, polish=True, seed=seed)
        best_h = result.x
        best_c5 = result.fun
    
    # Final checks
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
    return (h.tolist(), final_c5, n_points)
