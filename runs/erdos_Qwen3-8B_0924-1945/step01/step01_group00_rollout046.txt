import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
from scipy.stats import truncnorm
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    best_c5 = compute_c5(best_h)

    def run_cmaes():
        nonlocal best_h, best_c5
        population_size = 20
        sigma = 0.5
        for _ in range(500):
            if time() - start_time > budget_s:
                break
            candidates = np.random.normal(0, sigma, (population_size, n_points))
            for i in range(population_size):
                candidate = best_h + candidates[i]
                candidate = np.clip(candidate, 0, 1)
                candidate = candidate / np.sum(candidate) * n_points / 2
                c5 = compute_c5(candidate)
                if c5 < best_c5:
                    best_h = candidate.copy()
                    best_c5 = c5
            sigma *= 1.01 if np.std(candidates) < 0.5 else 0.99

    run_cmaes()

    def refine_with_slsqp():
        nonlocal best_h, best_c5
        for _ in range(50):
            if time() - start_time > budget_s:
                break
            res = minimize(objective, best_h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-5)
            if res.success:
                candidate = res.x
                candidate = np.clip(candidate, 0, 1)
                candidate = candidate / np.sum(candidate) * n_points / 2
                c5 = compute_c5(candidate)
                if c5 < best_c5:
                    best_h = candidate.copy()
                    best_c5 = c5
            else:
                break

    refine_with_slsqp()

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n_points)
