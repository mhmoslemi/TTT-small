import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_guesses = []

    if 'initial_h_values' in globals():
        base = np.copy(initial_h_values)
        base = np.clip(base, 0, 1)
        base /= np.sum(base)
        base *= target_sum
        initial_guesses.append(base)

    # Generate structured guesses
    for shift in range(5):
        spaced = np.zeros(n_points)
        for i in range(n_points):
            spaced[i] = 1 if (i + shift) % 5 == 0 else 0
        spaced = np.clip(spaced, 0, 1)
        spaced /= np.sum(spaced)
        spaced *= target_sum
        initial_guesses.append(spaced)

    # Generate random perturbed guesses
    for _ in range(2):
        perturbed = np.copy(initial_guesses[0])
        noise = np.random.uniform(-0.2, 0.2, size=n_points)
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed /= np.sum(perturbed)
        perturbed *= target_sum
        initial_guesses.append(perturbed)

    # Generate reversed and blended guesses
    reversed_guess = np.copy(initial_guesses[0])
    reversed_guess = reversed_guess[::-1]
    reversed_guess = np.clip(reversed_guess, 0, 1)
    reversed_guess /= np.sum(reversed_guess)
    reversed_guess *= target_sum
    initial_guesses.append(reversed_guess)

    blended = 0.5 * initial_guesses[0] + 0.5 * initial_guesses[1]
    blended = np.clip(blended, 0, 1)
    blended /= np.sum(blended)
    blended *= target_sum
    initial_guesses.append(blended)

    best_c5 = float('inf')
    best_h = None

    for guess in initial_guesses:
        x0 = guess.copy()
        x0 = np.clip(x0, 0, 1)
        x0 /= np.sum(x0)
        x0 *= target_sum

        def objective(x):
            x = np.clip(x, 0, 1)
            corr = np.correlate(x, 1 - x, mode='full')
            c5 = np.max(corr) * dx
            return c5

        def constraint(x):
            return np.sum(x) - target_sum

        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.minimize(
            objective,
            x0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-6, 'eps': 1e-6}
        )

        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate /= np.sum(h_candidate)
            h_candidate *= target_sum
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate

        # Early termination check
        if np.isclose(best_c5, float('inf')):
            break

    if best_h is None:
        best_h = initial_guesses[0]

    best_h = np.clip(best_h, 0, 1)
    best_h /= np.sum(best_h)
    best_h *= target_sum
    c5_bound = objective(best_h)

    return (best_h, c5_bound, n_points)
