```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iterations = 30
    iteration_time = budget_s / max_iterations

    def generate_initial_guess():
        if initial_h_values is not None:
            return np.array(initial_h_values, dtype=np.float64)
        else:
            h = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
            h = h * (target_sum / np.sum(h))
            return h

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    best_h = None
    best_c5 = float('inf')

    for i in range(max_iterations):
        start_time = time.time()
        h = generate_initial_guess()
        result = opt.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})
        elapsed = time.time() - start_time
        if elapsed > iteration_time or i == max_iterations - 1:
            break
        h_opt = result.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5 = objective(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt

    return best_h.tolist(), best_c5, n_points
```
