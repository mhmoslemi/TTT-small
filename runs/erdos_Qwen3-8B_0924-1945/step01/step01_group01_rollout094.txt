import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Objective function
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Optimization using gradient-based method with lag focus
    def gradient_objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_idx = np.argmax(correlate_vals)
        k = max_idx - n_points + 1
        grad = np.zeros(n_points)
        for i in range(n_points):
            x = i * 2.0 / n_points
            if k + i < n_points:
                grad[i] = (1 - h[i]) * (2.0 / n_points)
            if k - i >= 0:
                grad[i] -= (1 - h[k - i]) * (2.0 / n_points)
        return np.max(correlate_vals), grad
    
    # Stage 1: Gradient-based refinement
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, jac=None, tol=1e-6, max_iter=100)
    best_h = result.x
    best_c5 = result.fun
    
    # Stage 2: Multi-scale refinement
    n_points = 90
    dx = 2.0 / n_points
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, 45), best_h)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Final objective function
    def final_objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    # Optimization using differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.differential_evolution(final_objective, bounds, popsize=30, mutation=(0.9, 1.0),
                                       recombination=0.7, strategy='best1bin', tol=1e-5,
                                       maxiter=200, disp=False, polish=True, seed=seed)
    best_h = result.x
    best_c5 = result.fun
    
    # Final checks
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution found")
        h = best_h
        best_c5 = final_objective(h)
    
    # Final feasibility check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = final_objective(h)
    
    return (h.tolist(), final_c5, n_points)
