import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def construct_inequalities(h):
        n = n_points
        A = np.zeros((n, n), dtype=np.float64)
        for i in range(n):
            for k in range(1, n):
                j = (i + k) % n
                A[i, j] += dx
        return A

    A = construct_inequalities(h)

    h_vars = cp.Variable(n_points, bounds=(0.0, 1.0))
    C = cp.Variable()

    constraints = [
        cp.sum(h_vars) == target_sum,
        cp.sum(cp.multiply(A, cp.multiply(h_vars, h_vars))) >= (1 - C),
    ]

    prob = cp.Problem(cp.Minimize(C), constraints)
    prob.solve(solver=cp.SCS, verbose=False, max_iters=100)

    h_opt = h_vars.value
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale

    elapsed = time.time() - time.time()
    c5_bound = evaluate(h_opt)

    return h_opt.tolist(), c5_bound, n_points
