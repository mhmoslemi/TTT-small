import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def objective(C):
        return C

    def constraint(h, C):
        lag_values = np.arange(1, n_points)
        constraints = []
        for k in lag_values:
            overlap = np.zeros_like(h)
            for i in range(n_points):
                j = (i + k) % n_points
                overlap[i] += h[i] * h[j] * dx
            constraints.append(overlap >= 1 - C)
        return constraints

    h = cp.Variable(n_points, lower=0.0, upper=1.0)
    C = cp.Variable()

    cons = [
        cp.sum(h * dx) == target_sum,
        *constraint(h, C)
    ]

    prob = cp.Problem(cp.Minimize(C), cons)
    solver_options = {'max_iter': 100, 'eps': 1e-6}
    if 'ECOS' in cp.installed_solvers():
        prob.solve(solver=cp.ECOS, **solver_options)
    elif 'SCS' in cp.installed_solvers():
        prob.solve(solver=cp.SCS, **solver_options)

    h_opt = h.value
    if h_opt is None:
        h_opt = np.copy(h)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale

    c5_bound = np.max(np.correlate(h_opt, 1.0 - h_opt, mode='full') * dx)
    return h_opt.tolist(), c5_bound, n_points
