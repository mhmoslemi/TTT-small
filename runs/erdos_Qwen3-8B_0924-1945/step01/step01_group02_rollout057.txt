```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        h[::3] = 1.0
        h = h * (target_sum / np.sum(h))

    def compute_convolution(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        h = h * (target_sum / sum_h)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def build_constraints(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        h = h * (target_sum / sum_h)
        A = []
        for k in range(1, n_points):
            A_k = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    A_k[i, j] = dx
                j = i - k
                if j >= 0:
                    A_k[i, j] = dx
            A.append(A_k)
        return A

    def solve_qp(h_initial):
        h = cp.Variable(n_points)
        C = cp.Variable()
        A_list = build_constraints(h_initial)
        constraints = [
            cp.sum(h) == target_sum,
            0 <= h,
            h <= 1,
        ]
        for A_k in A_list:
            constraints.append(h @ A_k @ h >= 1 - C)
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver='SCS', eps=1e-8, max_iter=100)
            return h.value, C.value
        except:
            return h_initial, compute_convolution(h_initial)

    best_h = h
    best_c5 = compute_convolution(h)
    elapsed = 0

    for _ in range(20):
        if elapsed >= budget_s:
            break
        h_initial = np.copy(best_h)
        np.random.seed(int(time.time()))
        noise = np.random.uniform(-0.05, 0.05, n_points)
        h_initial += noise
        h_initial = np.clip(h_initial, 0.0, 1.0)
        h_initial = h_initial * (target_sum / np.sum(h_initial))
        h_opt, c5 = solve_qp(h_initial)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        h_opt = h_opt * (target_sum / np.sum(h_opt))
        c5_current = compute_convolution(h_opt)
        if c5_current < best_c5:
            best_h = h_opt
            best_c5 = c5_current
        elapsed = time.time() - time.time()

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (target_sum / np.sum(best_h))
    return best_h.tolist(), best_c5, n_points
```
