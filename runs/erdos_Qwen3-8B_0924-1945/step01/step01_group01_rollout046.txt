import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def gradient_descent(h_vec, max_iter=100, tol=1e-6):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        for _ in range(max_iter):
            correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            k_max = np.argmax(correlate_vals)
            grad = np.zeros(n_points)
            for i in range(n_points):
                if k_max < n_points and k_max >= 0:
                    if i + k_max < n_points:
                        grad[i] = (1 - h[i + k_max]) * (2.0 / n_points)
                    if i - k_max >= 0:
                        grad[i] = (1 - h[i - k_max]) * (2.0 / n_points)
            h -= 0.01 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            if np.linalg.norm(grad) < tol:
                break
        return h
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s * 0.6:
        h = best_h
        final_c5 = objective(h)
        return (h.tolist(), final_c5, n_points)
    
    h = gradient_descent(h, max_iter=100, tol=1e-6)
    final_c5 = objective(h)
    best_c5 = final_c5
    best_h = h.copy()
    
    elapsed = time.time() - start_time
    if elapsed > budget_s * 0.8:
        h = best_h
        final_c5 = objective(h)
        return (h.tolist(), final_c5, n_points)
    
    n_points = 90
    h = np.zeros(n_points)
    h[:len(best_h)] = best_h
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def de_objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.differential_evolution(de_objective, bounds, popsize=30, mutation=(0.9, 1.0), 
                                       recombination=0.7, strategy='best1bin', tol=1e-5,
                                       maxiter=200, disp=False, polish=True, seed=seed)
    
    best_h = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        h = best_h
        final_c5 = objective(h)
        return (h.tolist(), final_c5, n_points)
    
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points)
