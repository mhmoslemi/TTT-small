import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[::(n_points//3 + 1)] = 1.0
        h = np.clip(h, 0, 1)
        h = h * n_points / 2.0 / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * n_points / 2.0 / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h
    best_c5 = objective(h)
    
    pop_size = 50
    generations = 80
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_ratio = 0.2
    diversity_ratio = 0.1
    
    start_time = time.time()
    
    for gen in range(generations):
        population = [np.random.rand(n_points) for _ in range(pop_size)]
        for p in population:
            p = np.clip(p, 0, 1)
            p = p * n_points / 2.0 / np.sum(p)
        population.append(best_h)
        
        fitness = [objective(p) for p in population]
        sorted_idx = np.argsort(fitness)
        population = [population[i] for i in sorted_idx[:int(pop_size * (1 - elite_ratio))]]
        
        new_population = []
        for _ in range(int(pop_size * (1 - elite_ratio))):
            parent1, parent2 = np.random.choice(population, 2, replace=False)
            if np.random.rand() < crossover_rate:
                idx = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:idx], parent2[idx:]])
            else:
                child = parent1[:]
            child = np.clip(child, 0, 1)
            child = child * n_points / 2.0 / np.sum(child)
            new_population.append(child)
        
        for _ in range(int(pop_size * diversity_ratio)):
            mutant = np.random.rand(n_points)
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * n_points / 2.0 / np.sum(mutant)
            new_population.append(mutant)
        
        population = new_population
        fitness = [objective(p) for p in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx]
            best_c5 = fitness[best_idx]
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        if elapsed < 500:
            mutation_rate += 0.05
        elif elapsed < 200:
            crossover_rate -= 0.1
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * n_points / 2.0 / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
