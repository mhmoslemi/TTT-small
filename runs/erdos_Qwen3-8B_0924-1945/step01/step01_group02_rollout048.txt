```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        ones = np.random.rand(n_points) > 0.5
        h[ones] = 1.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def build_constraints(h):
        n = n_points
        A = []
        for k in range(1, n):
            A_k = np.zeros((n, n))
            for i in range(n):
                j = i + k
                if j < n:
                    A_k[i, j] = dx
                j = i - k
                if j >= 0:
                    A_k[i, j] = dx
            A.append(A_k)
        return A

    def solve_qp(h0):
        h = h0.copy()
        A = build_constraints(h)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale

        n = n_points
        C = cp.Variable()
        h_var = cp.Variable(n, lb=0.0, ub=1.0)
        objective = C
        constraints = [
            cp.sum(h_var) == target_sum,
            C >= 1.0 - cp.sum(cp.multiply(h_var, h_var))
        ]
        for A_k in A:
            constraints.append(C >= 1.0 - cp.quad_form(h_var, A_k))

        prob = cp.Problem(cp.Minimize(C), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-8)
            if prob.status not in ['OPTIMAL', 'OPTIMAL_INFEASIBLE']:
                return h
            h_opt = h_var.value
            return h_opt
        except:
            return h

    best_h = h.copy()
    best_c5 = evaluate(best_h)
    elapsed = 0
    start_time = time.time()

    for _ in range(20):
        if elapsed >= budget_s:
            break
        h = np.random.rand(n_points)
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))
        h = solve_qp(h)
        c5 = evaluate(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        elapsed = time.time() - start_time

    h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h = h * scale
    c5_bound = evaluate(h)
    return h.tolist(), c5_bound, n_points
```
