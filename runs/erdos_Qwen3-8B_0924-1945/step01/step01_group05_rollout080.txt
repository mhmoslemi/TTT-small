import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values if available or generate a structured pattern
    n_points = 150
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Use a Barker code pattern with linear interpolation
        barker_lengths = [1, 2, 4, 5, 7, 11, 13, 17, 23, 29, 31, 41, 47, 59, 61, 73, 83, 97, 109, 127]
        target_length = n_points
        barker_len = min(barker_lengths, key=lambda x: abs(x - target_length))
        barker = np.zeros(barker_len, dtype=np.int8)
        barker[0] = 1
        for i in range(1, barker_len):
            barker[i] = barker[i - 1] - 1 if barker[i - 1] > 0 else 0
        barker = barker[:target_length]
        h = barker.astype(np.float64)
    
    # Ensure sum(h) = n_points / 2
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization settings
    maxiter = 200
    tol = 1e-5
    
    # Optimization
    start_time = time.time()
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol}
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h, objective(h), n_points)
    
    best_h = result.x
    best_c5 = result.fun
    
    # Final sanity checks
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
