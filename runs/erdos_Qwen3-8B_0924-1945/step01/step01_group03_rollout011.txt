import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if hasattr(globals(), 'initial_h_values'):
        h = np.copy(initial_h_values)
        h = np.clip(h, 0, 1)
        h /= np.sum(h)
        h *= target_sum
    else:
        h = np.zeros(n_points)
        h[::5] = 1.0
        h = np.clip(h, 0, 1)
        h /= np.sum(h)
        h *= target_sum

    def objective(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = opt.time.time()
    result = opt.differential_evolution(
        objective,
        bounds,
        args=(),
        strategy='best1bin',
        popsize=15,
        maxiter=200,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        disp=False,
        polish=True,
        vectorized=True
    )
    
    elapsed = opt.time.time() - start_time
    if elapsed >= budget_s:
        best_h = h
        best_c5 = objective(best_h)
    else:
        best_h = result.x
        best_c5 = result.fun
    
    best_h = np.clip(best_h, 0, 1)
    best_h /= np.sum(best_h)
    best_h *= target_sum
    
    return (best_h, best_c5, n_points)
