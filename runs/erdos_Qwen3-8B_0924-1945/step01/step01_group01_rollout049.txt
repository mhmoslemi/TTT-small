import numpy as np
import scipy.optimize as opt
import time
import cvxopt
import cvxopt.solvers

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 90
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Precompute lag indices
    lags = np.arange(-n_points + 1, n_points)
    valid_lags = np.where((lags >= 0) & (lags < n_points))[0]
    valid_lags = lags[valid_lags]
    
    # Objective function: minimize max overlap
    def objective(mu):
        Q = np.zeros((n_points, n_points))
        for lag in valid_lags:
            i = np.arange(n_points)
            j = (i + lag) % n_points
            Q[i, j] += 1.0
        Q /= dx
        c = np.zeros(n_points)
        A = np.zeros((len(valid_lags) + 1, n_points))
        b = np.zeros(len(valid_lags) + 1)
        
        for idx, lag in enumerate(valid_lags):
            i = np.arange(n_points)
            j = (i + lag) % n_points
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = 1.0 - mu
        
        A[-1, :] = 1.0
        b[-1] = n_points / 2.0
        
        # Convert to cvxopt format
        P = cvxopt.matrix(Q)
        q = cvxopt.matrix(np.zeros(n_points))
        G = cvxopt.matrix(np.block([-A, -A]))
        h = cvxopt.matrix(np.block([-b, -b]))
        G = cvxopt.matrix(np.block([[-A, -A], [A, A]]))
        h = cvxopt.matrix(np.block([b, -b]))
        A = cvxopt.matrix(np.block([[A], [-A]]))
        b = cvxopt.matrix(np.block([b, -b]))
        
        # Optimization
        sol = cvxopt.solvers.qp(P, q, G, h, A, b)
        if sol['status'] != 'optimal':
            return np.inf
        return mu
    
    # Solve for optimal mu
    mu_bounds = (0.35, 0.45)
    result = opt.minimize_scalar(objective, bounds=mu_bounds, method='bounded', tol=1e-8)
    final_mu = result.fun
    h = np.array(sol['x']).flatten()
    
    # Final check
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning initial solution")
        h = initial_h_values if 'initial_h_values' in globals() else h
        final_mu = objective(final_mu)
    
    # Final feasibility check
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    correlates = np.correlate(h, 1 - h, mode="full") * dx
    final_c5 = np.max(correlates)
    
    return (h.tolist(), final_c5, n_points)
