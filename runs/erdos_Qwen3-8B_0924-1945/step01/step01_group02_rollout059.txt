```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))
    
    def compute_c5(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    def compute_constraints(h_flat, C):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx) - C <= 0
    
    def build_constraints(C):
        h = cp.Variable(n_points)
        constraints = []
        constraints.append(cp.sum(h) == target_sum)
        constraints.append(cp.constraints.only_enforce_if(cp.sum(h) == target_sum))
        for k in range(1, n_points):
            A = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = (i + k) % n_points
                A[i][j] = dx
                j = (i - k + n_points) % n_points
                A[i][j] = dx
            constraints.append(cp.QuadForm(h, A) >= 1 - C)
        return constraints
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        if np.random.rand() < 0.1:
            h = np.random.rand(n_points) * 2 - 1
            h = np.clip(h, 0.0, 1.0)
            h = h * (target_sum / np.sum(h))
        
        C = cp.Variable()
        prob = cp.Problem(cp.Minimize(C), build_constraints(C))
        try:
            result = prob.solve(solver=cp.ECOS, eps=1e-8, max_iter=100)
            if result is not None and result < best_c5:
                best_c5 = result
                best_h = h.copy()
        except:
            pass
    
    if time.time() - start_time < budget_s:
        h = best_h.copy()
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        c5_bound = compute_c5(h)
    else:
        c5_bound = best_c5
    
    return best_h.tolist(), c5_bound, n_points
```
