import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_lag = n_points - 1

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def compute_overlap(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def build_constraints(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        constraints = []
        for k in range(1, max_lag + 1):
            if k > n_points - k:
                continue
            i_range = np.arange(n_points - k)
            overlap = np.sum(h[i_range] * h[i_range + k]) * dx
            constraints.append(overlap >= 1 - C)
        return constraints

    h_var = cp.Variable(n_points, lb=0.0, ub=1.0)
    C = cp.Variable()

    # Objective: minimize C
    objective = cp.Minimize(C)

    # Constraints
    sum_constraint = cp.sum(h_var) * dx == 1
    constraints = [sum_constraint]
    for k in range(1, max_lag + 1):
        if k > n_points - k:
            continue
        i_range = np.arange(n_points - k)
        overlap = np.sum(h_var[i_range] * h_var[i_range + k]) * dx
        constraints.append(overlap >= 1 - C)

    prob = cp.Problem(objective, constraints)
    start_time = time.time()
    try:
        result = prob.solve(solver=cp.ECOS, verbose=False)
    except:
        result = prob.solve(solver=cp.SCS, verbose=False)

    h_opt = h_var.value
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = prob.solve(solver=cp.ECOS, verbose=False)
        h_opt = h_var.value
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale

    c5_bound = compute_overlap(h_opt.flatten())
    return h_opt.tolist(), c5_bound, n_points
