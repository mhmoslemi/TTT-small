import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def refine_solution(h, n_points_fine):
        dx = 2.0 / n_points_fine
        h_fine = np.zeros(n_points_fine)
        for i in range(n_points):
            h_fine[i * n_points_fine // n_points] += h[i]
        h_fine = h_fine * (n_points_fine / 2) / np.sum(h_fine)
        h_fine = np.clip(h_fine, 0, 1)
        bounds = [(0.0, 1.0) for _ in range(n_points_fine)]
        result = opt.differential_evolution(lambda x: objective(x), bounds, 
                                           popsize=20, mutation=(0.9, 1.0), 
                                           recombination=0.9, strategy='best1bin', 
                                           tol=1e-6, maxiter=200, disp=False, polish=True, seed=seed)
        return result.x
    
    best_h = h
    best_c5 = objective(h)
    elapsed = time.time() - start_time
    
    if elapsed < budget_s:
        h_fine = refine_solution(h, 90)
        best_h = h_fine
        best_c5 = objective(h_fine)
        elapsed = time.time() - start_time
    
    if elapsed < budget_s:
        h_fine = refine_solution(best_h, 135)
        best_h = h_fine
        best_c5 = objective(h_fine)
        elapsed = time.time() - start_time
    
    if elapsed < budget_s:
        result = opt.minimize(objective, best_h, method='L-BFGS-B', bounds=[(0.0, 1.0)] * len(best_h), tol=1e-6)
        best_h = result.x
        best_c5 = objective(best_h)
    
    # Final check
    h = np.clip(best_h, 0, 1)
    h = h * (len(h) / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, len(h))
