import scipy.optimize as opt
import time

    def evaluate_c5(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Gradient of C₅ with respect to h
    def gradient_c5(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full")
        max_k = np.argmax(correlate_vals)
        grad = np.zeros_like(h)
        for i in range(n_points):
            x = i * (2.0 / n_points)
            if max_k < n_points:
                x_plus_k = x
                if x_plus_k < 2.0:
                    grad[i] = -(1 - h[int(x_plus_k * n_points / 2.0)])
            elif max_k >= n_points:
                x_plus_k = x - (max_k - n_points)
                if x_plus_k >= 0:
                    grad[i] = -(1 - h[int(x_plus_k * n_points / 2.0)])
        return grad * (2.0 / n_points)
    
    # Gradient-based refinement with L-BFGS-B
    result = opt.minimize(evaluate_c5, h, method='L-BFGS-B', jac=gradient_c5,
                          bounds=[(0.0, 1.0) for _ in range(n_points)],
                          options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6})
    best_h = result.x
    best_c5 = result.fun

    if evaluate_c5(best_h) < evaluate_c5(h):

    h_refine = np.clip(h_refine, 0, 1)
    h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
