import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', None)
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        x = np.linspace(0, 2, n_points, endpoint=False)
        h = (1 + np.cos(np.pi * x / 2)) / 2
        h = h * (n_points / 2) / (np.sum(h) * 2 / n_points)
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    start_time = time()

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def de_objective(x):
        return evaluate_candidate(x)

    def de_constraint(x):
        return np.sum(x) - n_points / 2

    de_bounds = [(0, 1) for _ in range(n_points)]
    de_cons = [{'type': 'eq', 'fun': de_constraint}]

    def run_de():
        result = differential_evolution(
            de_objective, de_bounds, constraints=de_cons,
            popsize=20, mutation=(0.8, 1.0), recombination=0.9,
            tol=1e-5, maxiter=50, disp=False)
        return result.x

    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    start_time = time()
    for _ in range(3):
        if time() - start_time > budget_s:
            break
        candidate = run_de()
        candidate_c5 = evaluate_candidate(candidate)
        if candidate_c5 < best_c5:
            best_h = candidate.copy()
            best_c5 = candidate_c5

    for _ in range(10):
        if time() - start_time > budget_s:
            break
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
        current_c5 = evaluate_candidate(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    h_opt = best_h.copy()
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt / np.sum(h_opt) * n_points / 2
    current_c5 = evaluate_candidate(h_opt)
    return (h_opt, current_c5, n_points)
