```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = np.zeros(n_points, dtype=np.float64)
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.zeros(n_points)
        h[::5] = 1.0
        h = h * (target_sum / np.sum(h))

    def objective(x):
        h_flat = x
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        scale = target_sum / h_sum if h_sum != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(x):
        h_flat = x
        h = h_flat.reshape(n_points)
        h_sum = np.sum(h)
        return h_sum - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    start_time = time.time()
    result = opt.minimize(objective, h.flatten(), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'max_iter': 200})
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    h_sum = np.sum(h_opt)
    scale = target_sum / h_sum if h_sum != 0 else 1.0
    h_opt = h_opt * scale

    if time.time() - start_time < budget_s:
        result = opt.minimize(objective, h_opt.flatten(), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'max_iter': 100})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        h_sum = np.sum(h_opt)
        scale = target_sum / h_sum if h_sum != 0 else 1.0
        h_opt = h_opt * scale

    c5_bound = objective(h_opt.flatten())
    return h_opt.tolist(), c5_bound, n_points
```
