import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = np.round(initial_h_values).astype(int)
    else:
        h = np.zeros(n_points, dtype=int)
        for i in range(n_points):
            h[i] = 1 if np.random.rand() < 0.5 else 0
        if np.sum(h) != n_points // 2:
            h[np.random.choice(n_points, size=abs(np.sum(h) - n_points // 2), replace=False)] ^= 1

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.round(h).astype(int)
        if np.sum(h) != n_points // 2:
            h[np.random.choice(n_points, size=abs(np.sum(h) - n_points // 2), replace=False)] ^= 1
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    def run_optimization():
        nonlocal best_h, best_c5
        pop_size = 30
        mutation_rate = 0.1
        crossover_rate = 0.8
        max_generations = 100
        for gen in range(max_generations):
            if time() - start_time > budget_s:
                break
            population = np.random.rand(pop_size, n_points)
            population = np.round(population).astype(int)
            population = [np.concatenate([np.ones(n_points // 2), np.zeros(n_points // 2)]) for _ in range(pop_size)]
            for i in range(pop_size):
                if np.sum(population[i]) != n_points // 2:
                    population[i][np.random.choice(n_points, size=abs(np.sum(population[i]) - n_points // 2), replace=False)] ^= 1
            population_scores = [evaluate_candidate(p) for p in population]
            population = [population[i] for i in np.argsort(population_scores)]
            population_scores = np.sort(population_scores)
            elite = population[:int(pop_size * 0.2)]
            new_population = elite[:]
            while len(new_population) < pop_size:
                p1, p2 = np.random.choice(len(population), 2, replace=False)
                child = population[p1].copy()
                if np.random.rand() < crossover_rate:
                    split = np.random.randint(1, n_points - 1)
                    child[:split] = population[p2][:split]
                    cnt = np.sum(child)
                    if cnt < n_points // 2:
                        child[np.random.choice(n_points - cnt, size=n_points - cnt, replace=False)] = 1
                    elif cnt > n_points // 2:
                        child[np.random.choice(cnt, size=cnt - n_points // 2, replace=False)] = 0
                cnt = np.sum(child)
                if cnt != n_points // 2:
                    child[np.random.choice(n_points, size=abs(cnt - n_points // 2), replace=False)] ^= 1
                new_population.append(child)
            population = new_population
            population_scores = [evaluate_candidate(p) for p in population]
            population = [population[i] for i in np.argsort(population_scores)]
            population_scores = np.sort(population_scores)
            best_candidate = population[0]
            best_score = population_scores[0]
            if best_score < best_c5:
                best_h = best_candidate.copy()
                best_c5 = best_score
        return best_h, best_c5

    best_h, best_c5 = run_optimization()

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.rand(n_points)
            h = np.round(h).astype(int)
            if np.sum(h) != n_points // 2:
                h[np.random.choice(n_points, size=abs(np.sum(h) - n_points // 2), replace=False)] ^= 1
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.round(best_h).astype(int)
    if np.sum(best_h) != n_points // 2:
        best_h[np.random.choice(n_points, size=abs(np.sum(best_h) - n_points // 2), replace=False)] ^= 1
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = compute_c5(best_h)
    return (best_h, current_c5, n_points)
