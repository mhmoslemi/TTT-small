import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 90
    dx = 2.0 / n_points

    # Initialize h using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def objective(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    # Gradient-based optimization
    def grad_func(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        if max_idx < n_points:
            lag = max_idx - n_points
        else:
            lag = max_idx - n_points
        lags = np.arange(-n_points + 1, n_points)
        corr_vals = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr_vals)
        gradient = np.zeros_like(h)
        for i in range(n_points):
            x = i * dx
            x_k = x + lag * dx
            if 0 <= x_k < 2:
                gradient[i] = (1 - h[np.argmin(np.abs(np.linspace(0, 2, n_points) - x_k))]) * dx
        return gradient

    # Initial optimization with L-BFGS-B
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, jac=grad_func, tol=1e-5, options={'maxiter': 500})
    best_h = result.x
    best_c5 = result.fun

    # Adaptive grid refinement
    n_points = 120
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:len(best_h)] = best_h
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, jac=grad_func, tol=1e-5, options={'maxiter': 300})
    best_h = result.x
    best_c5 = result.fun

    # Fallback to differential evolution if needed
    elapsed = time.time() - start_time
    if elapsed > budget_s - 100:
        result = opt.differential_evolution(objective, bounds, popsize=20, mutation=(0.9, 1.0), recombination=0.7, strategy='best1bin', tol=1e-5, maxiter=200, disp=False, polish=True, seed=seed)
        best_h = result.x
        best_c5 = result.fun

    # Final check and return
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    return (h.tolist(), final_c5, n_points)
