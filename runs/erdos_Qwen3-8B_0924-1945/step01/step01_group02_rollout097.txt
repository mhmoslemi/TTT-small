```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def compute_c5(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def build_constraints(k):
        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = (i + k) % n_points
            A[i, j] = dx
            j = (i - k) % n_points
            A[i, j] = dx
        return A

    best_h = h.copy()
    best_c5 = compute_c5(h)
    elapsed_time = 0
    start_time = time.time()

    for _ in range(20):
        if elapsed_time >= budget_s:
            break
        if np.random.rand() < 0.1:
            h = np.random.rand(n_points) * 2 - 1
            h = np.clip(h, 0.0, 1.0)
            h = h * (target_sum / np.sum(h))
        h = h.copy()

        variables = []
        for i in range(n_points):
            variables.append(cp.Variable(n_points, name=f'h_{i}'))
        h_vars = np.array(variables)

        C = cp.Variable()
        constraints = [
            cp.sum(h_vars) == target_sum,
            cp.sum(h_vars) == target_sum,
            C >= 0
        ]

        for k in range(1, n_points):
            A = build_constraints(k)
            quad_expr = cp.quad_form(h_vars, A)
            constraints.append(quad_expr >= 1 - C)

        prob = cp.Problem(cp.Minimize(C), constraints)
        try:
            result = prob.solve(solver=cp.ECOS, eps=1e-8, max_iter=100)
            if result is not None and result >= 0:
                h_opt = np.array([var.value for var in h_vars])
                h_opt = np.clip(h_opt, 0.0, 1.0)
                sum_h_opt = np.sum(h_opt)
                scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
                h_opt = h_opt * scale
                current_c5 = compute_c5(h_opt)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_opt.copy()
                elapsed_time = time.time() - start_time
        except:
            pass

    h_final = np.clip(best_h, 0.0, 1.0)
    sum_h_final = np.sum(h_final)
    scale = target_sum / sum_h_final if sum_h_final != 0 else 1.0
    h_final = h_final * scale
    final_c5 = compute_c5(h_final)
    return h_final.tolist(), final_c5, n_points
```
