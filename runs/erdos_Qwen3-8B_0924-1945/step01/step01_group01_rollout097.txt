import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 90
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Precompute lag indices and build quadratic constraints
    def build_qp_constraints(h, n_points):
        dx = 2.0 / n_points
        constraints = []
        for s in range(n_points):
            valid_indices = []
            for i in range(n_points):
                j = (i + s) % n_points
                if j < n_points:
                    valid_indices.append((i, j))
            if valid_indices:
                A = np.zeros((len(valid_indices), n_points))
                for idx, (i, j) in enumerate(valid_indices):
                    A[idx, i] = 1
                    A[idx, j] = 1
                constraints.append(A)
        return constraints
    
    constraints = build_qp_constraints(h, n_points)
    if not constraints:
        return (h.tolist(), 0.0, n_points)
    
    # Objective function for QP
    def objective(mu):
        return mu
    
    # QP constraints setup
    P = np.zeros((n_points, n_points))
    for s in range(n_points):
        valid_indices = []
        for i in range(n_points):
            j = (i + s) % n_points
            if j < n_points:
                valid_indices.append((i, j))
        if valid_indices:
            for i, j in valid_indices:
                P[i, j] += 1
    
    # QP problem setup
    c = np.zeros(n_points)
    G = np.zeros((len(constraints), n_points))
    h_vec = np.zeros(len(constraints))
    for i, constraint in enumerate(constraints):
        G[i, :] = constraint.sum(axis=0)
        h_vec[i] = 1 - 1.0  # Target is <= 1 - mu
    
    # Formulate and solve QP
    try:
        from cvxopt import matrix, solvers
        solvers.options['abstol'] = 1e-8
        solvers.options['reltol'] = 1e-8
        solvers.options['maxiters'] = 10000
        P_cvx = matrix(P, tc='d')
        c_cvx = matrix(c, tc='d')
        G_cvx = matrix(G, tc='d')
        h_cvx = matrix(h_vec, tc='d')
        sol = solvers.qp(P_cvx, c_cvx, G_cvx, h_cvx)
        if sol['status'] != 'optimal':
            raise Exception("QP solver failed")
        h_qp = np.array(sol['x']).flatten()
        h_qp = np.clip(h_qp, 0, 1)
        h_qp = h_qp * (n_points / 2) / np.sum(h_qp)
        best_c5 = np.max(np.correlate(h_qp, 1 - h_qp, mode="full") * (2 / n_points))
        return (h_qp.tolist(), best_c5, n_points)
    except:
        pass
    
    # Fallback to differential evolution
    def objective_de(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.differential_evolution(objective_de, bounds, popsize=40, mutation=(0.5, 1.0), 
                                       recombination=0.9, strategy='best1bin', tol=1e-5,
                                       maxiter=200, disp=False, polish=True, seed=seed)
    
    best_h = result.x
    best_c5 = result.fun
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning initial solution")
        h = initial_h_values if 'initial_h_values' in globals() else h
        best_c5 = objective_de(h)
    
    # Final check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective_de(h)
    
    return (h.tolist(), final_c5, n_points)
