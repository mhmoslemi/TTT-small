import numpy as np
import scipy.optimize as optimize
import numpy.random as npr

def run(seed=42, budget_s=600, **kwargs):
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(100, 1.0 / 100)
    h = np.clip(h, 0.0, 1.0)
    h = h * (50.0 / np.sum(h)) if np.sum(h) != 0 else h

    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h_):
        h_ = np.clip(h_, 0.0, 1.0)
        h_ = h_ * (target_sum / np.sum(h_)) if np.sum(h_) != 0 else h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    def constraint(h_):
        return np.sum(h_) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def differential_evolution(population, generations=200, mutation_factor=0.5, crossover_rate=0.7):
        best_solution = population.copy()
        best_value = np.inf
        for _ in range(generations):
            new_population = []
            for i in range(len(population)):
                a, b, c = npr.choice(len(population), 3, replace=False)
                mutant = population[a] + mutation_factor * (population[b] - population[c])
                mutant = np.clip(mutant, 0.0, 1.0)
                mutant = mutant * (target_sum / np.sum(mutant)) if np.sum(mutant) != 0 else mutant
                trial = np.copy(population[i])
                for j in range(len(trial)):
                    if npr.rand() < crossover_rate:
                        trial[j] = mutant[j]
                trial = np.clip(trial, 0.0, 1.0)
                trial = trial * (target_sum / np.sum(trial)) if np.sum(trial) != 0 else trial
                trial_value = objective(trial)
                if trial_value < best_value:
                    best_value = trial_value
                    best_solution = trial.copy()
                new_population.append(trial)
            population = new_population
            if best_value < 0.38085:
                break
        return best_solution, best_value

    population_size = 50
    population = np.array([np.clip(npr.uniform(0, 1, n_points), 0, 1) * (target_sum / np.sum(np.clip(npr.uniform(0, 1, n_points), 0, 1))) if np.sum(np.clip(npr.uniform(0, 1, n_points), 0, 1)) != 0 else np.clip(npr.uniform(0, 1, n_points), 0, 1) for _ in range(population_size)])
    population = np.clip(population, 0.0, 1.0)
    population = population * (target_sum / np.sum(population)) if np.sum(population) != 0 else population

    best_h, best_c5 = differential_evolution(population)

    return (best_h.tolist(), best_c5, n_points)
