```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h_initial = np.array(initial_h_values, dtype=np.float64)
    else:
        h_initial = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h_initial[i] = 1.0 if i % 5 == 0 else 0.0
        h_initial = h_initial * (target_sum / np.sum(h_initial))

    def build_A(k):
        A = np.zeros((n_points, n_points), dtype=np.float64)
        for i in range(n_points):
            j1 = (i + k) % n_points
            j2 = (i - k) % n_points
            A[i, j1] = dx
            A[i, j2] = dx
        return A

    A_list = [build_A(k) for k in range(1, n_points)]

    best_h = h_initial.copy()
    best_c5 = float('inf')

    def objective_and_constraints(h_flat, C):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def solve_qp():
        nonlocal best_h, best_c5
        h = cp.Variable(n_points, bounds=(0.0, 1.0))
        C = cp.Variable()
        constraints = [
            cp.sum(h) == target_sum,
        ]
        for A in A_list:
            constraint = cp.quad_form(h, A) >= 1 - C
            constraints.append(constraint)
        prob = cp.Problem(cp.Minimize(C), constraints)
        try:
            result = prob.solve(solver=cp.ECOS, eps=1e-8, verbose=False)
            if result is not None and result < best_c5:
                best_h = h.value
                best_c5 = result
        except:
            pass

    start_time = time.time()
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        if initial_h_values is None:
            perturbation = np.random.normal(0, 0.05, n_points)
            h_initial = h_initial + perturbation
            h_initial = np.clip(h_initial, 0.0, 1.0)
            h_initial = h_initial * (target_sum / np.sum(h_initial))
        solve_qp()
        h_initial = best_h.copy()
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0.0, 1.0)
    sum_h = np.sum(h_final)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h_final = h_final * scale
    h_final = np.clip(h_final, 0.0, 1.0)
    sum_h = np.sum(h_final)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h_final = h_final * scale
    c5_final = objective_and_constraints(h_final, 0)
    return h_final.tolist(), c5_final, n_points
```
