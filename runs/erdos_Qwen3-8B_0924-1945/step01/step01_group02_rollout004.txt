```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        h[:n_points//2] = 1.0
        h = h * (target_sum / np.sum(h))

    def build_overlap_matrix():
        A = []
        for k in range(1, n_points):
            A_k = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = (i + k) % n_points
                A_k[i, j] = dx
                j = (i - k + n_points) % n_points
                A_k[i, j] = dx
            A.append(A_k)
        return np.stack(A)

    A_overlap = build_overlap_matrix()
    C = cp.Variable()
    h_var = cp.Variable(n_points, bounds=(0.0, 1.0))

    constraints = [
        cp.sum(h_var) == target_sum,
        cp.sum(cp.multiply(A_overlap[i], cp.outer(h_var, h_var))) >= 1 - C for i in range(len(A_overlap))
    ]
    objective = cp.Minimize(C)

    start_time = time.time()
    result = cp.Problem(objective, constraints).solve(solver='ECOS', eps=1e-6, verbose=False)
    h_opt = np.clip(h_var.value, 0.0, 1.0)
    if np.sum(h_opt) < target_sum - 1e-6:
        h_opt = h_opt * (target_sum / np.sum(h_opt))
    elif np.sum(h_opt) > target_sum + 1e-6:
        h_opt = h_opt * (target_sum / np.sum(h_opt))

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 10:
        h_opt = np.random.rand(n_points) * 0.5 + 0.5
        h_opt = h_opt * (target_sum / np.sum(h_opt))
        h_opt = np.clip(h_opt, 0.0, 1.0)

    c5_bound = np.max(np.correlate(h_opt, 1.0 - h_opt, mode="full")) * dx
    return h_opt.tolist(), c5_bound, n_points
```
