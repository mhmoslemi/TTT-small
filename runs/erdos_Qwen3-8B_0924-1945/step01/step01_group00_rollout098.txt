import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    start_time = time()

    def run_cma_es():
        population_size = 20
        dim = n_points
        best_h = h.copy()
        best_c5 = evaluate_candidate(best_h)
        for _ in range(200):
            if time() - start_time > budget_s:
                break
            mean = np.mean([best_h] * population_size, axis=0)
            sigma = 0.5 * np.ones(dim)
            candidates = np.random.normal(mean, sigma, (population_size, dim))
            candidates = np.clip(candidates, 0, 1)
            candidates = [c / np.sum(c) * n_points / 2 for c in candidates]
            current_c5s = [evaluate_candidate(c) for c in candidates]
            best_idx = np.argmin(current_c5s)
            best_h = candidates[best_idx]
            best_c5 = current_c5s[best_idx]
        return best_h, best_c5

    best_h, best_c5 = run_cma_es()
    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    h = best_h.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2
    current_c5 = evaluate_candidate(h)
    return (h, current_c5, n_points)
