import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Function to evaluate and enforce constraints
    def evaluate(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Objective function for optimization
    def objective(h_vec):
        return evaluate(h_vec)
    
    # Coarse optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_coarse = opt.differential_evolution(
        objective, bounds, popsize=40, mutation=(0.8, 1.0),
        recombination=0.9, strategy='best1bin', tol=1e-5,
        maxiter=300, disp=False, polish=True, seed=seed)
    
    best_h = result_coarse.x
    best_c5 = result_coarse.fun
    
    # Fine refinement with increased n_points
    n_points_fine = 90
    dx_fine = 2.0 / n_points_fine
    h_fine = np.interp(np.linspace(0, 2, n_points_fine), np.linspace(0, 2, n_points), best_h)
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (n_points_fine / 2) / np.sum(h_fine)
    
    def evaluate_fine(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points_fine / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx_fine
        return np.max(correlate_vals)
    
    def objective_fine(h_vec):
        return evaluate_fine(h_vec)
    
    result_fine = opt.differential_evolution(
        objective_fine, bounds, popsize=20, mutation=(1.0, 1.0),
        recombination=0.9, strategy='best1bin', tol=1e-6,
        maxiter=200, disp=False, polish=True, seed=seed)
    
    best_h_fine = result_fine.x
    best_c5_fine = result_fine.fun
    
    # Local refinement with L-BFGS-B
    result_local = opt.minimize(
        evaluate_fine, best_h_fine, method='L-BFGS-B', bounds=bounds,
        options={'maxiter': 100, 'gtol': 1e-6})
    
    best_h_local = result_local.x
    best_c5_local = result_local.fun
    
    # Select the best solution
    best_c5 = min(best_c5, best_c5_fine, best_c5_local)
    best_h = best_h_local
    
    # Final checks
    h = np.clip(best_h, 0, 1)
    h = h * (n_points_fine / 2) / np.sum(h)
    final_c5 = evaluate_fine(h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
    
    # Validate constraints
    assert np.all(0 <= h <= 1), "h values out of bounds"
    assert np.isclose(np.sum(h), n_points_fine / 2, atol=1e-6), "Integral constraint violated"
    assert np.isfinite(final_c5), "Final C5 is not finite"
    
    return (h.tolist(), final_c5, n_points_fine)
