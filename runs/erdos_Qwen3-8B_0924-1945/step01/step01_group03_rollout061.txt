import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = np.copy(initial_h_values)
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * target_sum
    
    def objective(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(x):
        return np.sum(x) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        result = opt.differential_evolution(
            objective,
            bounds,
            args=(),
            strategy='best1bin',
            popsize=15,
            maxiter=10,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            disp=False,
            x0=best_h.copy()
        )
        if result.success:
            h_vals = result.x
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals / np.sum(h_vals) * target_sum
            c5 = objective(h_vals)
            if c5 < best_c5:
                best_h = h_vals.copy()
                best_c5 = c5
        else:
            break
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * target_sum
    c5_bound = objective(best_h)
    return (best_h, c5_bound, n_points)
