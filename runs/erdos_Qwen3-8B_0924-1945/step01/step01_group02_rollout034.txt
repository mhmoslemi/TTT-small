```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def compute_C5(h):
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def build_A_matrix(k):
        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            j1 = (i + k) % n_points
            j2 = (i - k) % n_points
            A[i, j1] = dx
            A[i, j2] = dx
        return A

    def optimize_with_relaxation(h_initial):
        h = cp.Variable(n_points)
        C = cp.Variable()
        constraints = [
            cp.sum(h) == target_sum,
            0 <= h <= 1,
            cp.sum([h[i] * h[j] for i, j in zip(range(n_points), (range(n_points) + k) % n_points)]) >= 1 - C for k in range(1, n_points)
        ]
        prob = cp.Problem(cp.Minimize(C), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-8, verbose=False)
        if prob.status not in ['OPTIMAL', 'OPTIMAL_INACCURATE']:
            return h_initial
        h_opt = h.value
        return h_opt

    start_time = time.time()
    best_h = h
    best_c5 = compute_C5(h)
    elapsed = 0

    for _ in range(20):
        if elapsed >= budget_s:
            break
        h_initial = h.copy()
        np.random.shuffle(h_initial)
        h_initial = h_initial * (target_sum / np.sum(h_initial))
        h_initial = np.clip(h_initial, 0.0, 1.0)
        h_opt = optimize_with_relaxation(h_initial)
        c5 = compute_C5(h_opt)
        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5
        elapsed = time.time() - start_time

    return best_h.tolist(), best_c5, n_points
```
