```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h))

    def compute_constraints(h):
        n = len(h)
        A = []
        for k in range(1, n):
            row = np.zeros(n)
            for i in range(n):
                j = (i + k) % n
                row[i] += dx
                row[j] += dx
            A.append(row)
        return np.array(A)

    A = compute_constraints(h)
    n_lags = len(A)

    h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
    C = cp.Variable()

    constraints = [
        cp.sum(h_var) == target_sum,
        cp.sum(cp.multiply(h_var, h_var)) >= 1.0 - C
    ]

    for i in range(n_lags):
        constraints.append(cp.quad_form(h_var, A[i]) >= 1.0 - C)

    prob = cp.Problem(cp.Minimize(C), constraints)
    start_time = time.time()
    result = prob.solve(solver=cp.SCS, eps=1e-6, max_iters=300)
    h_opt = h_var.value
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        A = compute_constraints(h_opt)
        n_lags = len(A)
        h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
        C = cp.Variable()

        constraints = [
            cp.sum(h_var) == target_sum,
            cp.sum(cp.multiply(h_var, h_var)) >= 1.0 - C
        ]

        for i in range(n_lags):
            constraints.append(cp.quad_form(h_var, A[i]) >= 1.0 - C)

        prob = cp.Problem(cp.Minimize(C), constraints)
        result = prob.solve(solver=cp.SCS, eps=1e-6, max_iters=300)
        h_opt = h_var.value
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale

    c5_bound = np.max(np.correlate(h_opt, 1.0 - h_opt, mode="full")) * dx
    return h_opt.tolist(), c5_bound, n_points
```
