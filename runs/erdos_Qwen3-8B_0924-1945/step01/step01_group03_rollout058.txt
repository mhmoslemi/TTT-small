import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h_values if 'initial_h_values' in locals() else np.zeros(n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h / np.sum(initial_h) * target_sum

    def generate_guess(base):
        guess = base.copy()
        guess = np.clip(guess, 0, 1)
        guess = guess / np.sum(guess) * target_sum
        return guess

    base = 0.5 + 0.5 * np.sin(np.linspace(0, 2*np.pi, n_points))
    guesses = []
    for _ in range(10):
        noise = np.random.uniform(-0.15, 0.15, n_points)
        guess = base + noise
        guess = np.clip(guess, 0, 1)
        guess = guess / np.sum(guess) * target_sum
        guesses.append(guess)

    def objective(x):
        x = np.clip(x, 0, 1)
        x = x / np.sum(x) * target_sum
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    for guess in guesses:
        bounds = [(0, 1)] * n_points
        result = opt.minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-6, 'maxiter': 100}
        )
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * target_sum
    return (best_h, best_c5, n_points)
