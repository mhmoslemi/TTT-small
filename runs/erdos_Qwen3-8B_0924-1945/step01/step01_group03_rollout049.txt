    def objective(x):

        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.differential_evolution(

        bounds,
        constraints={'type': 'eq', 'fun': constraint},

        popsize=15,

        disp=False
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    c5_bound = objective(best_h)

    return (best_h, c5_bound, n_points)
