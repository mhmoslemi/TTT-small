import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Objective function
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Coarse-grid global optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_de = opt.differential_evolution(objective, bounds, popsize=50, mutation=(0.8, 1.0), 
                                          recombination=0.9, strategy='best1bin', tol=1e-5,
                                          maxiter=200, disp=False, polish=True, seed=seed)
    best_h = result_de.x
    best_c5 = result_de.fun
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning initial solution")
        h = initial_h_values if 'initial_h_values' in globals() else h
        best_c5 = objective(h)
    
    # Adaptive gradient refinement
    def gradient_objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_val = np.max(correlate_vals)
        max_idx = np.argmax(correlate_vals)
        if max_idx < n_points:
            lag = max_idx - n_points + 1
        else:
            lag = max_idx - 2 * n_points + 1
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + lag >= 0 and i + lag < n_points:
                grad[i] = (1 - h[i + lag]) * (2.0 / n_points)
        return max_val
    
    result_lbgfs = opt.minimize(gradient_objective, best_h, bounds=bounds, method='L-BFGS-B',
                                jac=None, tol=1e-6, options={'maxiter': 300, 'disp': False})
    best_h = result_lbgfs.x
    best_c5 = gradient_objective(best_h)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution from DE")
        best_h = result_de.x
        best_c5 = result_de.fun
    
    # Multi-scale refinement
    n_points = 90
    dx = 2.0 / n_points
    h_refined = np.zeros(n_points)
    h_refined[:len(best_h)] = best_h
    h_refined = np.clip(h_refined, 0, 1)
    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
    
    def refined_objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    result_de_refined = opt.differential_evolution(refined_objective, bounds, popsize=30, mutation=(0.9, 1.0),
                                                  recombination=0.7, strategy='best1bin', tol=1e-5,
                                                  maxiter=100, disp=False, polish=True, seed=seed)
    best_h_refined = result_de_refined.x
    best_c5_refined = result_de_refined.fun
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution from L-BFGS-B")
        best_h = best_h
        best_c5 = best_c5
    
    # Final check
    h = np.clip(best_h_refined, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = refined_objective(h)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Final refinement time exceeded, using best solution")
        h = best_h
        final_c5 = best_c5
    
    return (h.tolist(), final_c5, n_points)
