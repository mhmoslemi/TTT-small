import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 90
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def gradient(h_vec, dx):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        max_idx = np.argmax(correlate_vals)
        k = max_idx - n_points + 1
        if k < 0 or k >= n_points:
            return np.zeros_like(h)
        grad = np.zeros_like(h)
        for i in range(n_points):
            x = i * dx
            x_k = x + k * dx
            if x_k >= 2:
                grad[i] = 0
            else:
                grad[i] = (1 - h[np.clip(np.floor(x_k / dx), 0, n_points - 1)])
        return grad
    
    # Stage 2: Gradient-based refinement
    result = opt.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        jac=lambda x: gradient(x, 2.0 / n_points),
        options={'maxiter': 500, 'ftol': 1e-5, 'eps': 1e-6, 'disp': False}
    )
    best_h = result.x
    best_c5 = result.fun
    
    # Stage 3: Adaptive grid refinement
    n_points_new = 120
    dx_new = 2.0 / n_points_new
    h_new = np.interp(np.linspace(0, 2, n_points_new), np.linspace(0, 2, n_points), best_h)
    h_new = np.clip(h_new, 0, 1)
    h_new = h_new * (n_points_new / 2) / np.sum(h_new)
    
    # Stage 2 with new grid
    result = opt.minimize(
        lambda x: objective(x),
        h_new,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0) for _ in range(n_points_new)],
        jac=lambda x: gradient(x, dx_new),
        options={'maxiter': 200, 'ftol': 1e-5, 'eps': 1e-6, 'disp': False}
    )
    best_h = result.x
    best_c5 = result.fun
    
    # Final check
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
        h = best_h
        best_c5 = objective(h)
    
    # Final feasibility check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points_new / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points_new)
