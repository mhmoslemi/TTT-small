def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Initial configuration
    n_points = 90
    if 'initial_h_values' in globals():

    else:

        h = np.clip(h, 0, 1)

    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Precompute lag indices
    def valid_indices(n_points):
        indices = []
        for i in range(n_points):
            for j in range(n_points):
                if j - i >= 0 and j - i < n_points:
                    indices.append((i, j))
                if i - j >= 0 and i - j < n_points:
                    indices.append((j, i))
        return np.array(indices)
    
    valid = valid_indices(n_points)
    lags = np.unique(valid[:, 1] - valid[:, 0])
    
    # Objective function
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Quadratic program formulation
    def qp_formulation(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        dx = 2.0 / n_points
        n = n_points
        Q = np.zeros((n, n))
        for i in range(n):
            for j in range(n):
                if i + j < n or i + j >= 2 * n:
                    continue
                for k in range(len(lags)):
                    lag = lags[k]
                    if i + j == n + lag:
                        Q[i, j] += 1.0
        c = np.zeros(n)
        A_eq = np.zeros((1, n))
        A_eq[0, :] = 1.0
        b_eq = n_points / 2
        bounds = [(0.0, 1.0) for _ in range(n)]
        G = np.zeros((len(lags), n))
        for k in range(len(lags)):
            lag = lags[k]
            for i in range(n):
                for j in range(n):
                    if i + j == n + lag:
                        G[k, i] += h[j]
        h_vec = np.zeros(n)
        for k in range(len(lags)):
            G[k, :] = h
        h_vec = np.linalg.lstsq(G, np.ones(len(lags)), rcond=None)[0]
        return np.max(h_vec)
    
    try:
        result = minimize(qp_formulation, h, method='L-BFGS-B', bounds=[(0,1)]*n_points, tol=1e-8)

        best_c5 = qp_formulation(best_h)
    except:
        best_h = h
        best_c5 = objective(h)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning initial solution")
        h = initial_h_values if 'initial_h_values' in globals() else h
        best_c5 = objective(h)
    
    # Final feasibility check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points)
