import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 90
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Precompute valid indices for all lags
    dx = 2.0 / n_points
    valid_lags = []
    for s in range(1, n_points):
        indices = []
        for i in range(n_points):
            j = i + s
            if j < n_points:
                indices.append((i, j))
            else:
                j -= n_points
                if j >= 0:
                    indices.append((i, j))
        valid_lags.append(indices)
    
    # Define objective and constraints
    def objective(mu):
        return mu
    
    def constraints(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        # Compute overlap for all lags
        overlaps = []
        for lag_indices in valid_lags:
            overlap = 0.0
            for i, j in lag_indices:
                overlap += h[i] * h[j]
            overlaps.append(overlap * dx)
        return np.array(overlaps) - (1 - mu)
    
    # QP formulation
    def solve_qp():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        mu = 1.0
        for _ in range(100):
            try:
                c = np.array([0.0] * n_points + [1.0])
                G = np.zeros((len(valid_lags), n_points + 1))
                for idx, lag_indices in enumerate(valid_lags):
                    for i, j in lag_indices:
                        G[idx, i] += h[j]
                        G[idx, j] += h[i]
                    G[idx, -1] = -1.0
                h_vec = np.random.rand(n_points)
                h_vec = np.clip(h_vec, 0, 1)
                h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
                h_prev = h_vec.copy()
                mu_prev = mu
                constraints_val = constraints(h_prev)
                mu = 1.0 - np.max(constraints_val)
                # Update h with a small step
                h_vec = h_prev + (mu - mu_prev) * np.array([0.0] * n_points)
                h_vec = np.clip(h_vec, 0, 1)
                h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
                h = h_vec
                if np.abs(mu - mu_prev) < 1e-8:
                    break
            except:
                break
        return h, mu
    
    # Run QP solver
    best_h, best_c5 = solve_qp()
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = 1.0 - np.max(np.array([sum(best_h[i] * best_h[j] for i, j in lag_indices) * dx for lag_indices in valid_lags]))
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning initial solution")
        h = initial_h_values if 'initial_h_values' in globals() else h
        best_c5 = 1.0 - np.max(np.array([sum(h[i] * h[j] for i, j in lag_indices) * dx for lag_indices in valid_lags]))
    
    # Final check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = 1.0 - np.max(np.array([sum(h[i] * h[j] for i, j in lag_indices) * dx for lag_indices in valid_lags]))
    
    return (h.tolist(), final_c5, n_points)
