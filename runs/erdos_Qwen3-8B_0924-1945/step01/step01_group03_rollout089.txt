        x0 = guess.copy()

        x0 = x0 / np.sum(x0) * target_sum

        def objective(x):
            x = np.clip(x, 0, 1)
            x = x / np.sum(x) * target_sum
            corr = np.correlate(x, 1 - x, mode='full')
            c5 = np.max(corr) * dx
            return c5

        def constraint(x):
            return np.sum(x) - target_sum

        bounds = [(0, 1) for _ in range(n_points)]

        result = opt.minimize(
            objective,
            x0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-6, 'eps': 1e-6}
        )

        if result.success:
            final_x = result.x
            final_x = np.clip(final_x, 0, 1)
            final_x = final_x / np.sum(final_x) * target_sum
            c5 = objective(final_x)

                best_c5 = c5
