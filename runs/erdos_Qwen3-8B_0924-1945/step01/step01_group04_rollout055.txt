import numpy as np
import scipy.optimize as optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Start with parent if available, otherwise use random initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.random.rand(100)
        h = h * (50.0 / np.sum(h)) if np.sum(h) != 0 else h
    
    # Enforce constraints
    h = np.clip(h, 0.0, 1.0)
    h = h * (50.0 / np.sum(h)) if np.sum(h) != 0 else h
    
    def objective(h_):
        h_ = np.clip(h_, 0.0, 1.0)
        h_ = h_ * (50.0 / np.sum(h_)) if np.sum(h_) != 0 else h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * (2.0 / 100)
    
    def constraint(h_):
        return np.sum(h_) - 50.0
    
    bounds = [(0.0, 1.0) for _ in range(100)]
    cons = {'type': 'eq', 'fun': constraint}
    
    # Use differential evolution for global optimization
    result = optimize.differential_evolution(
        objective,
        bounds,
        constraints=cons,
        strategy='best1bin',
        popsize=50,
        mut=0.5,
        recombination=0.7,
        maxiter=200,
        tol=1e-6,
        seed=seed,
        polish=True,
        disp=False,
        vectorized=True
    )
    
    h_opt = result.x
    h_opt = np.clip(h_opt, 0.0, 1.0)
    h_opt = h_opt * (50.0 / np.sum(h_opt)) if np.sum(h_opt) != 0 else h_opt
    c5_bound = objective(h_opt)
    
    return (h_opt.tolist(), c5_bound, 100)
