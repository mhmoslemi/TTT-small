import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h_initial = np.copy(initial_h_values)
    else:
        h_initial = np.zeros(60)
        x = np.linspace(0, 2, 60)
        h_initial = 0.5 + 0.5 * np.sin(np.pi * x)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (60 / 2) / np.sum(h_initial)
    
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (target_sum / np.sum(h_initial))
    
    def objective(x):
        h_vals = np.copy(x)
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(
        objective,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h))
    c5_bound = objective(best_h)
    
    return (best_h, c5_bound, n_points)
