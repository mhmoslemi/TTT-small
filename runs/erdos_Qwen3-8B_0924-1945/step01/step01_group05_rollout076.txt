import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[::n_points//3] = 1
        h = np.clip(h, 0, 1)
        h = h * n_points / 2 / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * n_points / 2 / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    population_size = 50
    max_generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    population = np.random.rand(population_size, n_points)
    population = np.clip(population, 0, 1)
    population = population * n_points / 2 / np.sum(population, axis=1, keepdims=True)
    
    best_h = h
    best_c5 = objective(h)
    
    start_time = time.time()
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        
        offspring = []
        
        for _ in range(population_size):
            parent1, parent2 = np.random.choice(population, 2, replace=False)
            
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:split], parent2[split:]])
            else:
                child = parent1.copy()
            
            if np.random.rand() < mutation_rate:
                mutate_indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
                child[mutate_indices] = np.random.rand(len(mutate_indices))
            
            child = np.clip(child, 0, 1)
            child = child * n_points / 2 / np.sum(child)
            offspring.append(child)
        
        population = np.array(offspring)
        population = np.clip(population, 0, 1)
        population = population * n_points / 2 / np.sum(population, axis=1, keepdims=True)
        
        for ind in population:
            current_c5 = objective(ind)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = ind
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * n_points / 2 / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
