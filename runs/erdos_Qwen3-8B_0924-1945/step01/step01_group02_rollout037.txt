```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def evaluate_h(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def make_constraints(h):
        num_lags = n_points - 1
        H = h.reshape(-1, 1)
        constraints = []
        for k in range(1, num_lags + 1):
            A = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    A[i, j] = dx
                j = i - k
                if j >= 0:
                    A[i, j] = dx
            quad_form = cp.QuadForm(H, A)
            constraints.append(quad_form >= 1 - cp.Variable())
        return constraints

    best_h = h.copy()
    best_c5 = evaluate_h(h)
    elapsed_time = 0

    for _ in range(20):
        if elapsed_time >= budget_s:
            break
        h_rand = np.random.rand(n_points)
        h_rand = (h_rand - 0.5) * 2
        h_rand = np.clip(h_rand, 0.0, 1.0)
        h_rand = h_rand * (target_sum / np.sum(h_rand))
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        constraints = [
            cp.sum(h_var) == target_sum,
            cp.Minimize(C),
            cp.constraints.NonNeg(h_var),
            cp.constraints.NonPos(h_var)
        ] + make_constraints(h_var)
        prob = cp.Problem(cp.Minimize(C), constraints)
        try:
            start_time = time.time()
            prob.solve(solver=cp.ECOS, eps=1e-8, verbose=False)
            elapsed_time += time.time() - start_time
            if prob.status == 'OPTIMAL':
                h_opt = h_var.value
                h_opt = np.clip(h_opt, 0.0, 1.0)
                sum_h_opt = np.sum(h_opt)
                scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
                h_opt = h_opt * scale
                c5 = evaluate_h(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
        except:
            pass

    h_final = best_h
    sum_h = np.sum(h_final)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h_final = h_final * scale
    final_c5 = evaluate_h(h_final)
    return h_final.tolist(), final_c5, n_points
```
