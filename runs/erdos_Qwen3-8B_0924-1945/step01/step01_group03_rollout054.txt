import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h_values if 'initial_h_values' in locals() else np.zeros(n_points)

    def generate_guesses(base):
        guesses = []
        base = np.clip(base, 0, 1)
        base = base * (target_sum / np.sum(base))
        guesses.append(base.copy())
        for i in range(5):
            shift = np.random.randint(0, n_points)
            spaced = np.zeros(n_points)
            step = n_points // 5
            for j in range(n_points):
                if (j + shift) % step < 2:
                    spaced[j] = 1
            spaced = np.clip(spaced, 0, 1)
            spaced = spaced * (target_sum / np.sum(spaced))
            guesses.append(spaced.copy())
        for i in range(5):
            noise = np.random.uniform(-0.2, 0.2, n_points)
            perturbed = base + noise
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * (target_sum / np.sum(perturbed))
            guesses.append(perturbed.copy())
        for i in range(5):
            flipped = 1 - base
            flipped = np.clip(flipped, 0, 1)
            flipped = flipped * (target_sum / np.sum(flipped))
            guesses.append(flipped.copy())
        return [np.clip(g, 0, 1) for g in guesses]

    guesses = generate_guesses(initial_h)

    best_c5 = np.inf
    best_h = np.zeros(n_points)
    candidates = []

    for guess in guesses:
        x0 = guess.copy()
        x0 = np.clip(x0, 0, 1)
        x0 = x0 * (target_sum / np.sum(x0))

        def objective(x):
            x = np.clip(x, 0, 1)
            x = x * (target_sum / np.sum(x))
            corr = np.correlate(x, 1 - x, mode='full')
            c5 = np.max(corr) * dx
            return c5

        def constraint(x):
            return np.sum(x) - target_sum

        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.minimize(
            objective,
            x0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
        )

        x = result.x
        x = np.clip(x, 0, 1)
        x = x * (target_sum / np.sum(x))
        c5 = objective(x)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x.copy()
        candidates.append((x, c5))

        if len(candidates) > 5:
            candidates.pop(np.argmax([c[1] for c in candidates]))

    candidates.sort(key=lambda x: x[1])
    best_h = candidates[0][0]
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h))
    final_c5 = objective(best_h)
    return (best_h, final_c5, n_points)
