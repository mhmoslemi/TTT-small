import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        initial_h = np.copy(initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h / np.sum(initial_h) * target_sum
    else:
        x = np.linspace(0, 2, n_points)
        # Base guess: sinusoidal with clipping and rescaling
        base_h = 0.5 + 0.5 * np.sin(np.pi * x)
        base_h = np.clip(base_h, 0, 1)
        base_h = base_h / np.sum(base_h) * target_sum
        initial_h = base_h

    def objective(x):
        h = np.clip(x, 0, 1)
        h = h / np.sum(h) * target_sum
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    best_h = initial_h
    best_c5 = objective(best_h)

    # Generate variations
    def generate_guesses(base):
        variations = []
        x = np.linspace(0, 2, n_points)
        for i in range(10):
            if i == 0:
                # Base
                var = base.copy()
            elif i == 1:
                # Add noise
                var = base + np.random.uniform(-0.15, 0.15, size=n_points)
            elif i == 2:
                # Phase shift
                var = 0.5 + 0.5 * np.sin(np.pi * (x - 0.5))
            elif i == 3:
                # Reverse
                var = base[::-1]
            elif i == 4:
                # Blend base and reverse
                var = 0.5 * base + 0.5 * base[::-1]
            elif i == 5:
                # Triangular wave
                var = 0.5 + 0.5 * np.abs(np.sin(np.pi * x) + 1)
            elif i == 6:
                # Square wave with 1/3 duty cycle
                var = 0.5 + 0.5 * (np.sin(np.pi * x) > 0.5)
            elif i == 7:
                # Modulate with cosine
                var = 0.5 + 0.5 * np.sin(np.pi * x) * (1 + 0.5 * np.cos(2 * np.pi * x))
            elif i == 8:
                # Gaussian kernel
                var = np.exp(-0.5 * (x - 1)**2)
            elif i == 9:
                # Piecewise linear
                x_segments = np.linspace(0, 2, 5)
                y_segments = [0, 1, 0, 1, 0]
                var = np.interp(x, x_segments, y_segments)
            var = np.clip(var, 0, 1)
            var = var / np.sum(var) * target_sum
            variations.append(var)
        return variations

    guesses = generate_guesses(initial_h)

    # Run optimization
    elapsed = 0
    best_c5 = float('inf')
    best_h = initial_h
    for guess in guesses:
        x0 = guess.copy()
        result = opt.minimize(
            objective,
            x0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-6, 'eps': 1e-6}
        )
        if result.success and result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
            elapsed += result.message.count('success')
            if elapsed > budget_s:
                break

    # Fallback: use best from guesses
    if best_c5 == float('inf'):
        best_h = guesses[0]
        best_c5 = objective(best_h)

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * target_sum
    best_c5 = objective(best_h)

    return (best_h, best_c5, n_points)
