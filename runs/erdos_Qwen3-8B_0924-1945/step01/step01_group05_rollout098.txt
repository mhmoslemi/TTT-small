import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[::2] = 1
        h = (h - np.min(h)) / (np.max(h) - np.min(h))
        h = h * n_points / 2
        h = h / np.sum(h) * n_points / 2
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)

    def obj(x):
        h_flat = np.clip(x, 0, 1)
        h_flat = h_flat * target_sum / np.sum(h_flat)
        corr = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def cons(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': cons}, options={'maxiter': 200, 'ftol': 1e-6})
    elapsed = time.time() - time.time()  # dummy placeholder
    best_h = result.x
    best_c5 = result.fun
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = obj(best_h)
    return (best_h, final_c5, n_points)
