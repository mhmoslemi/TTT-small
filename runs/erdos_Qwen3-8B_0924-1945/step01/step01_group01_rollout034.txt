import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parent initialization
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Feasibility check
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.inf
    
    # Objective and gradient function
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def gradient(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_k = np.argmax(correlate_vals)
        grad = np.zeros(n_points)
        for i in range(n_points):
            if max_k - i >= 0 and max_k - i < n_points:
                grad[i] = (1 - h[max_k - i]) * (2.0 / n_points)
        return grad
    
    # Stage 1: Gradient-based refinement
    def lbfgs_refinement(h, max_iter=100, tol=1e-6):
        for _ in range(max_iter):
            grad = gradient(h)
            h -= 0.01 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h[:] = h[:]
            if np.max(np.abs(grad)) < tol:
                break
        return h
    
    h = lbfgs_refinement(h)
    current_c5 = objective(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h[:] = h[:]
    
    # Stage 2: Increase resolution and continue refinement
    n_points = 90
    dx = 2.0 / n_points
    h_new = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, 45), best_h)
    h_new = np.clip(h_new, 0, 1)
    h_new = h_new * (n_points / 2) / np.sum(h_new)
    
    def objective_new(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def gradient_new(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        max_k = np.argmax(correlate_vals)
        grad = np.zeros(n_points)
        for i in range(n_points):
            if max_k - i >= 0 and max_k - i < n_points:
                grad[i] = (1 - h[max_k - i]) * dx
        return grad
    
    h = lbfgs_refinement(h_new, max_iter=150, tol=1e-5)
    current_c5 = objective_new(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h[:] = h[:]
    
    # Stage 3: Fallback to DE if needed
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.differential_evolution(objective_new, bounds, 
                                       popsize=30, mutation=(0.9, 1.0), 
                                       recombination=0.7, strategy='best1bin', 
                                       tol=1e-5, maxiter=200, disp=False, 
                                       polish=True, seed=seed)
    h = result.x
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    current_c5 = objective_new(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h[:] = h[:]
    
    # Final check against time
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
    
    # Final feasibility and evaluation
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective_new(h)
    
    # Ensure correct n_points and dx
    if n_points == 90:
        final_c5 = objective_new(h)
    else:
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        final_c5 = np.max(correlate_vals)
    
    return (h.tolist(), final_c5, n_points)
