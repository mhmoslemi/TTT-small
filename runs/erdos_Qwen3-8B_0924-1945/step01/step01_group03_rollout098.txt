import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = np.copy(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * target_sum
    else:
        h = np.zeros(n_points)
        h[::5] = 1.0
        h = h / np.sum(h) * target_sum

    def objective(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1)] * n_points
    result = opt.differential_evolution(
        objective, bounds, constraints={'type': 'eq', 'fun': constraint},
        strategy='best1bin', popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8,
        maxiter=200, mutation_size=0.1, seed=seed, disp=False
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * target_sum
    best_c5 = objective(best_h)

    return (best_h, best_c5, n_points)
