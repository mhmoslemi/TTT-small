```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def get_overlap_matrix(k):
        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = (i + k) % n_points
            A[i, j] += dx
            j = (i - k) % n_points
            A[i, j] += dx
        return A

    h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
    C = cp.Variable()

    constraints = [
        cp.sum(h_var) == target_sum,
        cp.sum(cp.multiply(h_var, 1.0 - h_var)) <= 1.0 - C
    ]

    for k in range(1, n_points):
        A_k = get_overlap_matrix(k)
        constraints.append(cp.quad_form(h_var, A_k) >= 1.0 - C)

    prob = cp.Problem(cp.Minimize(C), constraints)
    solver_opts = {
        'max_iter': 500,
        'eps': 1e-6,
        'verbose': False,
        'use_gpu': False
    }

    try:
        start_time = time.time()
        result = prob.solve(solver=cp.ECOS, **solver_opts)
        elapsed_time = time.time() - start_time
    except:
        elapsed_time = budget_s

    if elapsed_time < budget_s:
        h_opt = h_var.value
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale

        c5_bound = np.max(np.correlate(h_opt, 1.0 - h_opt, mode="full") * dx)
    else:
        h_opt = h_var.value
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = np.max(np.correlate(h_opt, 1.0 - h_opt, mode="full") * dx)

    return h_opt.tolist(), c5_bound, n_points
```
