import scipy.optimize as opt

        guesses.append(base_guess)
        structured = np.zeros(n_points)
        spacing = n_points // 5
        for i in range(5):
            structured[i * spacing % n_points] = 1
        structured = np.clip(structured, 0, 1)
        structured /= np.sum(structured)
        structured *= target_sum
        guesses.append(structured)
        perturbed = base_guess * (1 + 0.2 * (np.random.rand(n_points) - 0.5))
        perturbed = np.clip(perturbed, 0, 1)
        perturbed /= np.sum(perturbed)
        perturbed *= target_sum
        guesses.append(perturbed)
        reversed_guess = 1 - base_guess
        reversed_guess = np.clip(reversed_guess, 0, 1)
        reversed_guess /= np.sum(reversed_guess)
        reversed_guess *= target_sum
        guesses.append(reversed_guess)
        combined = (base_guess + structured) / 2
        combined = np.clip(combined, 0, 1)
        combined /= np.sum(combined)
        combined *= target_sum
        guesses.append(combined)
        return guesses

    def objective(x):
        x_clipped = np.clip(x, 0, 1)
        corr = np.correlate(x_clipped, 1 - x_clipped, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - target_sum
