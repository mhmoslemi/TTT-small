import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_guesses = []

    # Initial guess 1: use parent construction
    if 'initial_h_values' in globals():
        h = np.copy(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[::7] = 1.0
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    initial_guesses.append(h)

    # Initial guess 2: structured pattern with phase shift
    h = np.zeros(n_points)
    step = n_points // 5
    for i in range(5):
        h[::step] = 1.0
    h = np.roll(h, np.random.randint(n_points))
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    initial_guesses.append(h)

    # Initial guess 3: perturbed version of parent
    if 'initial_h_values' in globals():
        h = np.copy(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[::7] = 1.0
    h = np.clip(h, 0, 1)
    noise = np.random.uniform(-0.2, 0.2, n_points)
    h = np.clip(h + noise, 0, 1)
    h = h * target_sum / np.sum(h)
    initial_guesses.append(h)

    # Initial guess 4: reversed version
    if 'initial_h_values' in globals():
        h = np.copy(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[::7] = 1.0
    h = np.clip(h, 0, 1)
    h = h[::-1]
    h = h * target_sum / np.sum(h)
    initial_guesses.append(h)

    # Initial guess 5: blended version
    if 'initial_h_values' in globals():
        h = np.copy(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[::7] = 1.0
    h1 = h * 0.5
    h2 = np.zeros(n_points)
    h2[::7] = 1.0
    h = 0.5 * h1 + 0.5 * h2
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    initial_guesses.append(h)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = opt.time.time()

    for guess in initial_guesses:
        x = guess.copy()
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        x = x.copy()

        def objective(x_vec):
            x_vec = np.clip(x_vec, 0, 1)
            corr = np.correlate(x_vec, 1 - x_vec, mode='full')
            return np.max(corr) * dx

        def constraint(x_vec):
            return np.sum(x_vec) - target_sum

        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.minimize(
            objective,
            x,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            tol=1e-6,
            options={'maxiter': 200, 'ftol': 1e-6}
        )

        if opt.time.time() - start_time > budget_s - 10:
            break

        if result.success:
            x_opt = result.x
            x_opt = np.clip(x_opt, 0, 1)
            x_opt = x_opt * target_sum / np.sum(x_opt)
            c5 = objective(x_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = x_opt

    return (best_h, best_c5, n_points)
