import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial solution (if available, otherwise create a random one)
    n_points = 100
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = (h - np.min(h)) / (np.max(h) - np.min(h))
        h = np.clip(h, 0, 1)
        h = (h * n_points / 2).astype(np.float64)
        h = h / np.sum(h) * n_points / 2
    
    # Ensure sum(h) = n_points / 2
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.full(n_points, target_sum / n_points)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.full(n_points, target_sum / n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimize using differential evolution
    result = opt.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-6,
        seed=seed
    )
    
    # Return the best solution found
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h) if np.sum(best_h) > 0 else np.full(n_points, target_sum / n_points)
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
