import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with initial_h_values if available, otherwise use a random guess
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(90)
        h = np.clip(h, 0, 1)
        h = h * (90 / 2) / np.sum(h)
    
    # Define function to evaluate C5
    def evaluate_c5(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (90 / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / 90)
        return np.max(correlate_vals)
    
    # Function to compute the quadratic form for the QP
    def compute_quad_form(h, lags):
        n = len(h)
        quad = np.zeros(len(lags))
        for i, lag in enumerate(lags):
            if 0 <= lag < n:
                quad[i] = np.dot(h[:n - lag], h[lag:])
        return quad
    
    # Precompute valid lags for the QP formulation
    def get_valid_lags(n_points):
        valid_lags = []
        for k in range(1, 2 * n_points):
            if k <= n_points:
                valid_lags.append(k)
            else:
                valid_lags.append(k - 2 * n_points)
        return valid_lags
    
    valid_lags = get_valid_lags(90)
    
    # QP formulation with CVXOPT
    def cvxopt_qp_formulation(h):
        n = len(h)
        m = len(valid_lags)
        c = np.zeros(m + 1)
        c[-1] = 1  # Objective is to minimize mu
        G = np.zeros((m + 1, n))
        h_clipped = np.clip(h, 0, 1)
        quad_form = compute_quad_form(h_clipped, valid_lags)
        G[:m, :] = -np.eye(n)
        G[-1, :] = np.ones(n)
        h_normalized = h_clipped * (n / 2) / np.sum(h_clipped)
        lhs = -np.eye(n)
        rhs = np.zeros(n)
        rhs[-1] = n / 2
        A = np.hstack((np.eye(m), np.zeros((m, 1))))
        b = -quad_form
        return c, G, h_normalized, A, b, rhs
    
    # QP solver using CVXOPT
    def solve_qp(h):
        try:
            c, G, h_normalized, A, b, rhs = cvxopt_qp_formulation(h)
            import cvxopt
            cvxopt.solvers.options['maxiter'] = 100
            cvxopt.solvers.options['abstol'] = 1e-8
            cvxopt.solvers.options['reltol'] = 1e-8
            cvxopt.solvers.options[' feastol'] = 1e-8
            P = cvxopt.matrix(np.eye(len(h)))
            q = cvxopt.matrix(c)
            G_mat = cvxopt.matrix(G)
            h_mat = cvxopt.matrix(h)
            A_mat = cvxopt.matrix(A)
            b_mat = cvxopt.matrix(b)
            rhs_mat = cvxopt.matrix(rhs)
            sol = cvxopt.solvers.qp(P, q, G_mat, h_mat, A_mat, b_mat)
            if sol['status'] == 'optimal':
                x = np.array(sol['x'])
                return x, evaluate_c5(x)
            else:
                return h, evaluate_c5(h)
        except:
            return h, evaluate_c5(h)
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    elapsed = 0
    
    while elapsed < budget_s - 10:
        h, current_c5 = solve_qp(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break
    
    # Fallback to differential evolution if needed
    if best_c5 > 0.38:
        bounds = [(0.0, 1.0) for _ in range(90)]
        def objective(h_vec):
            h = np.clip(h_vec, 0, 1)
            h = h * (90 / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / 90)
            return np.max(correlate_vals)
        
        result = opt.differential_evolution(objective, bounds, popsize=30, mutation=(0.5, 1.0),
                                           recombination=0.9, strategy='best1bin', tol=1e-5,
                                           maxiter=100, disp=False, polish=True, seed=seed)
        best_h = result.x
        best_c5 = result.fun
    
    # Final validation
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (90 / 2) / np.sum(h_final)
    final_c5 = evaluate_c5(h_final)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
    
    return (h_final.tolist(), final_c5, 90)
