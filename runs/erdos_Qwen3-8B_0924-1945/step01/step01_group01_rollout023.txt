import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 90
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def formulate_qp(h, n_points, dx):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        n = n_points
        lags = np.arange(1, n - 1)
        P = np.zeros((n, n))
        for i in range(n):
            for s in lags:
                j = (i + s) % n
                P[i, j] += dx * h[i] * h[j]
        P = 0.5 * (P + P.T)  # Symmetrize
        q = np.zeros(n)
        A = np.zeros((len(lags), n))
        b = np.ones(len(lags)) * (1 - 1e-8)
        for idx, s in enumerate(lags):
            for i in range(n):
                j = (i + s) % n
                A[idx, i] = h[i] * dx
        return P, q, A, b, h

    def solve_qp(P, q, A, b, h, dx):
        from cvxopt import matrix, solvers
        solvers.options['show_progress'] = False
        solvers.options['abstol'] = 1e-8
        solvers.options['reltol'] = 1e-8
        solvers.options['feastol'] = 1e-8
        try:
            P_c = matrix(P)
            q_c = matrix(q)
            A_c = matrix(A)
            b_c = matrix(b)
            sol = solvers.qp(P_c, q_c, A_c, b_c)
            h_opt = np.array(sol['x'])
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            return h_opt
        except:
            return h

    def evaluate_c5(h, n_points, dx):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    best_h = h
    best_c5 = evaluate_c5(h, n_points, dx)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning initial solution")
        return (h.tolist(), best_c5, n_points)
    
    try:
        for _ in range(3):
            if elapsed > budget_s:
                break
            P, q, A, b, _ = formulate_qp(h, n_points, dx)
            h = solve_qp(P, q, A, b, h, dx)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            c5 = evaluate_c5(h, n_points, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
            elapsed = time.time() - start_time
        return (best_h.tolist(), best_c5, n_points)
    except:
        return (h.tolist(), best_c5, n_points)
