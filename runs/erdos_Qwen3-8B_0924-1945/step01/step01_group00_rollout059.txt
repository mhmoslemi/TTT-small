import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else None
    
    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h / total * n_points / 2
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        h = normalize(h)
        return compute_c5(h)
    
    bounds = [(0, 1) for _ in range(n_points)]
    population_size = 15
    mutation_factor = 0.5
    crossover_prob = 0.9
    
    if initial_h_values is not None:
        h0 = normalize(initial_h_values)
    else:
        h0 = np.random.rand(n_points)
        h0 = normalize(h0)
    
    def objective(h):
        return evaluate_candidate(h)
    
    start_time = time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=population_size,
        mutation=mutation_factor,
        recombination=crossover_prob,
        tol=1e-5,
        maxiter=100,
        polish=True,
        disp=False,
        seed=seed
    )
    
    best_h = result.x
    best_c5 = evaluate_candidate(best_h)
    
    if time() - start_time < budget_s:
        for _ in range(50):
            if time() - start_time > budget_s:
                break
            h = np.random.rand(n_points)
            h = normalize(h)
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    best_h = normalize(best_h)
    current_c5 = evaluate_candidate(best_h)
    
    return (best_h, current_c5, n_points)
