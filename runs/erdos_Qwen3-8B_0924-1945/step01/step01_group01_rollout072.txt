import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Objective function
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Coarse-grid optimization
    def coarse_optimize(h_init, max_iter=200):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(
            objective, bounds, popsize=50, mutation=(0.8, 1.0), 
            recombination=0.9, strategy='best1bin', tol=1e-5, 
            maxiter=max_iter, disp=False, polish=True, seed=seed
        )
        return result.x, result.fun
    
    # Gradient refinement
    def refine_with_lbfgs(h_init, max_iter=300):
        def objective_with_grad(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            c5 = np.max(correlate_vals)
            # Find the lag that maximizes the correlation
            k_max = np.argmax(correlate_vals)
            # Compute gradient for that lag
            dx = 2.0 / n_points
            grad = np.zeros_like(h)
            for i in range(n_points):
                if k_max < n_points and k_max >= 0:
                    if i + k_max < n_points:
                        grad[i] += (1 - h[i + k_max]) * dx
                    if i - k_max >= 0:
                        grad[i] += (1 - h[i - k_max]) * dx
            return c5, grad
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.minimize(
            lambda x: objective_with_grad(x)[0],
            h_init.copy(), bounds=bounds, method='L-BFGS-B', 
            jac=lambda x: objective_with_grad(x)[1], tol=1e-6, 
            max_iter=max_iter
        )
        return result.x, result.fun
    
    # Multi-scale refinement
    def refine_with_higher_res(h_init, new_n_points):
        dx = 2.0 / new_n_points
        h_rescaled = np.zeros(new_n_points)
        for i in range(new_n_points):
            h_rescaled[i] = h_init[int(i * n_points / new_n_points)]
        h_rescaled = np.clip(h_rescaled, 0, 1)
        h_rescaled = h_rescaled * (new_n_points / 2) / np.sum(h_rescaled)
        return h_rescaled
    
    # Stage 1: Coarse optimization
    h_coarse, c5_coarse = coarse_optimize(h.copy(), max_iter=200)
    best_h, best_c5 = h_coarse, c5_coarse
    
    # Stage 2: Gradient refinement
    h_refined, c5_refined = refine_with_lbfgs(h_coarse, max_iter=300)
    best_h, best_c5 = h_refined, c5_refined
    
    # Stage 3: Multi-scale refinement
    h_higher_res = refine_with_higher_res(best_h, 90)
    h_higher_res = np.clip(h_higher_res, 0, 1)
    h_higher_res = h_higher_res * (90 / 2) / np.sum(h_higher_res)
    h_refined_final, c5_final = refine_with_lbfgs(h_higher_res, max_iter=300)
    best_h, best_c5 = h_refined_final, c5_final
    
    # Final check
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
    
    # Final feasibility check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
