```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        h[::3] = 1.0
        h = h * (target_sum / np.sum(h))

    def build_A_k(k):
        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = (i + k) % n_points
            A[i, j] = dx
            j = (i - k + n_points) % n_points
            A[i, j] = dx
        return A

    def compute_C5(h):
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def solve_qp(h0):
        h = cp.Variable(n_points)
        C = cp.Variable()
        constraints = [
            cp.sum(h * dx) == target_sum,
            0 <= h,
            h <= 1,
            cp.sum([h[i] * h[j] * dx for i in range(n_points) for j in range(n_points) if abs(i - j) == k]) >= 1 - C for k in range(1, n_points)
        ]
        prob = cp.Problem(cp.Minimize(C), constraints)
        try:
            result = prob.solve(solver='ECOS', eps=1e-8, max_iters=100)
            if result is not None and result < np.inf:
                return h.value, C.value
        except:
            pass
        return h0, compute_C5(h0)

    best_h = h
    best_c5 = compute_C5(h)
    elapsed = 0.0

    for _ in range(20):
        if elapsed >= budget_s:
            break
        h = np.random.uniform(0.0, 1.0, size=n_points)
        h = h * (target_sum / np.sum(h))
        h, c5 = solve_qp(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
        elapsed = time.time() - time.time()

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (target_sum / np.sum(best_h))
    return best_h.tolist(), best_c5, n_points
```
