import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Stage 1: Coarse-Grid Global Optimization (DE)
    result = opt.differential_evolution(objective, bounds, popsize=50, mutation=(0.8, 1.0),
                                       recombination=0.9, strategy='best1bin', tol=1e-5,
                                       maxiter=200, disp=False, polish=True, seed=seed)
    best_h = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning initial solution")
        h = initial_h_values if 'initial_h_values' in globals() else h
        best_c5 = objective(h)
        return (h.tolist(), best_c5, n_points)
    
    # Stage 2: Adaptive Gradient Refinement (L-BFGS-B)
    def grad_f(h_vec, k):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_idx = np.argmax(correlate_vals)
        k_star = max_idx - n_points + 1
        if k_star < 0 or k_star >= n_points:
            return np.zeros_like(h)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + k_star < n_points:
                grad[i] = (1 - h[i + k_star]) * (2.0 / n_points)
        return grad
    
    def objective_with_grad(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_val = np.max(correlate_vals)
        return max_val
    
    options = {'maxiter': 300, 'ftol': 1e-6, 'eps': 1e-8}
    result = opt.minimize(objective_with_grad, best_h, method='L-BFGS-B', bounds=bounds, jac=lambda x: grad_f(x, 0), options=options)
    best_h = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution from stage 1")
        return (best_h.tolist(), best_c5, n_points)
    
    # Stage 3: Multi-Scale Refinement
    n_points = 90
    dx = 2.0 / n_points
    h_refined = np.zeros(n_points)
    for i in range(n_points):
        if i < len(best_h):
            h_refined[i] = best_h[i]
        else:
            h_refined[i] = np.random.rand() * 0.5
    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
    h_refined = np.clip(h_refined, 0, 1)
    
    def objective_refined(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    result_refined = opt.differential_evolution(objective_refined, bounds, popsize=30, mutation=(0.9, 1.0),
                                               recombination=0.7, strategy='best1bin', tol=1e-5,
                                               maxiter=100, disp=False, polish=True, seed=seed)
    best_h = result_refined.x
    best_c5 = result_refined.fun
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution from stage 2")
        h = best_h
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        final_c5 = objective_refined(h)
        return (h.tolist(), final_c5, n_points)
    
    # Final feasibility check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective_refined(h)
    
    return (h.tolist(), final_c5, n_points)
