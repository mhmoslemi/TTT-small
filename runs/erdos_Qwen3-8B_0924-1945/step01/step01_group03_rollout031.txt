import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Generate initial guess based on sinusoidal function
    x = np.linspace(0, 2, n_points)
    base_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    base_guess = np.clip(base_guess, 0, 1)
    base_guess = base_guess * (target_sum / np.sum(base_guess))

    def objective(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Optimization loop
    best_h = base_guess.copy()
    best_c5 = objective(best_h)
    elapsed = 0

    # Generate some structured guess variations
    guess_variants = []
    for i in range(10):
        if i == 0:
            guess = base_guess.copy()
        elif i == 1:
            guess = base_guess.copy() + np.random.uniform(-0.15, 0.15, n_points)
        elif i == 2:
            guess = base_guess.copy()
            guess = np.roll(guess, int(n_points / 4))
        elif i == 3:
            guess = base_guess.copy()[::-1]
        elif i == 4:
            guess = 0.5 * base_guess + 0.5 * base_guess[::-1]
        elif i == 5:
            guess = 0.5 * base_guess + 0.5 * np.roll(base_guess, int(n_points / 3))
        elif i == 6:
            guess = np.zeros(n_points)
            for k in range(n_points // 5):
                guess[k] = 1
                guess[k + n_points // 3] = 1
                guess[k + 2 * n_points // 3] = 1
            guess = np.clip(guess, 0, 1)
            guess = guess * (target_sum / np.sum(guess))
        elif i == 7:
            guess = base_guess.copy()
            guess = 0.5 * guess + 0.5 * np.cos(np.pi * x)
            guess = np.clip(guess, 0, 1)
            guess = guess * (target_sum / np.sum(guess))
        elif i == 8:
            guess = np.zeros(n_points)
            guess[0::3] = 1
            guess = np.clip(guess, 0, 1)
            guess = guess * (target_sum / np.sum(guess))
        else:
            guess = base_guess.copy()
            guess = 0.5 * guess + 0.5 * np.roll(base_guess, i)
            guess = np.clip(guess, 0, 1)
            guess = guess * (target_sum / np.sum(guess))
        guess_variants.append(guess)

    for guess in guess_variants:
        guess = np.clip(guess, 0, 1)
        guess = guess * (target_sum / np.sum(guess))
        result = opt.minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-6}
        )
        if result.success:
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (target_sum / np.sum(current_h))
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
                elapsed = 0
            else:
                elapsed += 1
        else:
            elapsed += 1

        if elapsed > 10:
            break

    # Final evaluation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h))
    c5_bound = objective(best_h)

    return (best_h, c5_bound, n_points)
