import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial solution (if available, otherwise create a structured pattern)
    n_points = 200
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Generate a Barker code-based initial pattern
        def generate_barker(n):
            barker = [1]
            for l in range(1, n):
                if l == 1:
                    barker.append(1)
                else:
                    s = sum(barker[-l+1:-1])
                    if s == 0:
                        barker.append(1)
                    else:
                        barker.append(0)
            return barker

        # Find the nearest Barker code length
        L = n_points
        for l in range(1, L + 1):
            if l in [1, 2, 3, 4, 5, 7, 11, 13, 17, 23, 29, 31]:
                break
        barker = generate_barker(l)
        # Interpolate to n_points samples
        x = np.linspace(0, 2, len(barker))
        xp = np.linspace(0, 2, n_points)
        h = np.interp(xp, x, barker)
        h = np.clip(h, 0, 1)
        h = h * n_points / 2 / np.sum(h)
    
    # Ensure sum(h) = n_points / 2
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum
    
    # Bounds and optimization settings
    bounds = [(0, 1) for _ in range(n_points)]
    maxiter = 200
    tol = 1e-5
    
    start_time = time.time()
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol}
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h, objective(h), n_points)
    
    best_h = result.x
    best_c5 = result.fun
    
    # Final sanity checks
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
