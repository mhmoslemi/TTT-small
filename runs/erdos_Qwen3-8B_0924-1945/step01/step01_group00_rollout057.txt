import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 50
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy() if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    def run_cma_es():
        nonlocal best_h, best_c5
        pop_size = 20
        mu = 10
        sigma = 0.5
        population = np.random.uniform(0, 1, (pop_size, n_points))
        population = population / np.sum(population, axis=1, keepdims=True) * n_points / 2
        for _ in range(200):
            if time() - start_time > budget_s:
                break
            for i in range(pop_size):
                population[i] = np.clip(population[i], 0, 1)
                population[i] = population[i] / np.sum(population[i]) * n_points / 2
            energies = np.array([evaluate_candidate(p) for p in population])
            idxs = np.argsort(energies)
            best_pop = population[idxs[0]]
            best_energy = energies[idxs[0]]
            if best_energy < best_c5:
                best_h = best_pop.copy()
                best_c5 = best_energy
            population = population[idxs[:mu]]
            mean = np.mean(population, axis=0)
            cov = np.cov(population, rowvar=False)
            sigma *= np.std(population, axis=0).mean()
            new_population = np.random.multivariate_normal(mean, cov, pop_size)
            new_population = np.clip(new_population, 0, 1)
            new_population = new_population / np.sum(new_population, axis=1, keepdims=True) * n_points / 2
            population = np.vstack([population, new_population])
        return best_h, best_c5

    best_h, best_c5 = run_cma_es()

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
