import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    h = h.astype(np.float64)
    h = np.clip(h, 0.0, 1.0)

    def build_constraints():
        n = n_points
        A = []
        for k in range(1, n):
            A_k = np.zeros((n, n))
            for i in range(n):
                j1 = (i + k) % n
                j2 = (i - k + n) % n
                A_k[i, j1] = dx
                A_k[i, j2] = dx
            A.append(A_k)
        return A

    A = build_constraints()
    n_constraints = len(A)

    h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
    C = cp.Variable()

    constraints = [
        cp.sum(h_var) == target_sum,
        cp.sum(cp.multiply(h_var, h_var)) <= cp.Variable(n_points, nonneg=True) + 1.0,
    ]

    for i in range(n_constraints):
        constraint = cp.bmat([[h_var, cp.Constant(np.zeros((n_points, 1)))],
                             [cp.Constant(np.zeros((1, n_points))), cp.Constant(1.0)]] *
                             [A[i]] +
                             [cp.Constant(np.zeros((1, 1)))])
        constraints.append(cp.trace(constraint) >= 1.0 - C)

    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)

    start_time = time.time()
    result = prob.solve(solver=cp.ECOS, eps=1e-6, verbose=False)
    elapsed_time = time.time() - start_time

    h_opt = h_var.value
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale

    if elapsed_time < budget_s:
        for _ in range(3):
            if elapsed_time + 10 > budget_s:
                break
            new_h = h_opt + np.random.normal(0, 0.05, n_points)
            new_h = np.clip(new_h, 0.0, 1.0)
            new_sum = np.sum(new_h) * dx
            new_h = new_h * (target_sum / new_sum) if new_sum != 0.0 else new_h
            h_opt = new_h

    c5_bound = np.max(np.correlate(h_opt, 1.0 - h_opt, mode="full")) * dx
    return h_opt.tolist(), c5_bound, n_points
