import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = np.copy(initial_h_values)
    h = np.clip(h, 0, 1)
    h /= np.sum(h)
    h *= target_sum
    
    def obj_func(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(x):
        return np.sum(x) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def generate_guess():
        x = np.linspace(0, 2, n_points)
        base = 0.5 + 0.5 * np.sin(np.pi * x)
        base = np.clip(base, 0, 1)
        base = base * target_sum / np.sum(base)
        noise = np.random.uniform(-0.15, 0.15, size=n_points)
        guess = base + noise
        guess = np.clip(guess, 0, 1)
        guess = guess * target_sum / np.sum(guess)
        return guess
    
    def optimize_guess(guess):
        result = opt.minimize(
            obj_func,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        return best_h, obj_func(best_h)
    
    best_h = None
    best_c5 = float('inf')
    time_used = 0
    
    # Generate initial structured guesses
    guesses = [generate_guess() for _ in range(10)]
    
    # Run optimization on each guess
    for guess in guesses:
        if time_used >= budget_s:
            break
        start_time = time_used
        h_opt, c5 = optimize_guess(guess)
        time_used = time_used + 100
        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5
    
    # Final refinement with another guess
    if time_used < budget_s:
        guess = generate_guess()
        h_opt, c5 = optimize_guess(guess)
        time_used = time_used + 100
        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5
    
    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = obj_func(best_h)
    return (best_h, final_c5, n_points)
