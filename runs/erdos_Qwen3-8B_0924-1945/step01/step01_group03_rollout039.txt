import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = np.copy(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = (h / np.sum(h)) * target_sum
        h = np.clip(h, 0, 1)

    def objective(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    
    def optimize_guess(guess):
        h_vals = guess.copy()
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = (h_vals / np.sum(h_vals)) * target_sum
        result = opt.minimize(
            objective,
            h_vals,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-6, 'eps': 1e-6}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = (best_h / np.sum(best_h)) * target_sum
        c5_bound = objective(best_h)
        return best_h, c5_bound

    best_h, best_c5 = optimize_guess(h)
    for _ in range(3):
        h_rand = np.random.rand(n_points)
        h_rand = (h_rand / np.sum(h_rand)) * target_sum
        h_rand = np.clip(h_rand, 0, 1)
        current_h, current_c5 = optimize_guess(h_rand)
        if current_c5 < best_c5:
            best_h, best_c5 = current_h, current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = (best_h / np.sum(best_h)) * target_sum
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
