import numpy as np
from scipy.optimize import minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else None
    if initial_h_values is not None:
        h = np.round(initial_h_values).astype(float)
    else:
        h = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    start_time = time()

    def run_evolution():
        population_size = 30
        mutation_rate = 0.1
        crossover_rate = 0.8
        generations = 100
        elite_ratio = 0.2
        local_refinement = 10
        best_h = h.copy()
        best_c5 = evaluate_candidate(best_h)
        population = [np.random.choice([0, 1], size=n_points, p=[0.5, 0.5]) for _ in range(population_size)]
        for indiv in population:
            indiv = indiv / np.sum(indiv) * n_points / 2
        for _ in range(generations):
            if time() - start_time > budget_s:
                break
            fitness_scores = [evaluate_candidate(indiv) for indiv in population]
            sorted_indices = np.argsort(fitness_scores)
            population = [population[i] for i in sorted_indices[:int(population_size * elite_ratio)]]
            new_population = []
            for _ in range(int(population_size * (1 - elite_ratio))):
                if np.random.rand() < crossover_rate:
                    parent1, parent2 = np.random.choice(population, 2, replace=False)
                    cross_point = np.random.randint(1, n_points - 1)
                    child = np.concatenate([parent1[:cross_point], parent2[cross_point:]])
                else:
                    child = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
                child = child / np.sum(child) * n_points / 2
                new_population.append(child)
            population += new_population
            for indiv in population:
                if np.random.rand() < mutation_rate:
                    mut_pos = np.random.randint(n_points)
                    indiv[mut_pos] = 1 - indiv[mut_pos]
                    indiv = indiv / np.sum(indiv) * n_points / 2
            current_best = min(population, key=evaluate_candidate)
            if evaluate_candidate(current_best) < best_c5:
                best_h = current_best.copy()
                best_c5 = evaluate_candidate(best_h)
        return best_h, best_c5

    best_h, best_c5 = run_evolution()

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
