import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
    
    def generate_initial_guess():
        x = np.linspace(0, 2, n_points)
        sin_guess = 0.5 + 0.5 * np.sin(np.pi * x)
        sin_guess = np.clip(sin_guess, 0, 1)
        sin_guess = sin_guess / np.sum(sin_guess) * target_sum
        return sin_guess
    
    def objective(x):
        h_vals = np.clip(x, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = generate_initial_guess()
    best_c5 = objective(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * target_sum
    best_c5 = objective(best_h)
    
    def evaluate_candidate(start):
        x0 = start.copy()
        result = opt.differential_evolution(
            objective,
            bounds,
            popsize=15,
            strategy='best1bin',
            tol=1e-6,
            maxiter=200,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            args=(),
            callback=None,
            disp=False,
            polish=True,
            xtol=1e-6,
            constraints=[{'type': 'eq', 'fun': constraint}]
        )
        h_vals = np.clip(result.x, 0, 1)
        h_vals = h_vals / np.sum(h_vals) * target_sum
        c5 = objective(h_vals)
        return h_vals, c5
    
    best_c5 = float('inf')
    for _ in range(10):
        guess = generate_initial_guess()
        guess += np.random.uniform(-0.15, 0.15, size=n_points)
        guess = np.clip(guess, 0, 1)
        guess = guess / np.sum(guess) * target_sum
        h_vals, c5 = evaluate_candidate(guess)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_vals
    
    return (best_h, best_c5, n_points)
