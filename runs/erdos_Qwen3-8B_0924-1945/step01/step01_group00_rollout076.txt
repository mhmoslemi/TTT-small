import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()

    def de_run():
        pop_size = 20
        mutation = 0.5
        crossover = 0.9
        max_gen = 100
        best_h = h.copy()
        best_c5 = evaluate_candidate(best_h)
        population = np.random.rand(pop_size, n_points)
        population = np.clip(population, 0, 1)
        population = population / np.sum(population, axis=1, keepdims=True) * n_points / 2
        for gen in range(max_gen):
            if time() - start_time > budget_s:
                break
            new_population = []
            for i in range(pop_size):
                a, b, c = np.random.randint(0, pop_size, 3)
                mutant = population[a] + mutation * (population[b] - population[c])
                mutant = np.clip(mutant, 0, 1)
                mutant = mutant / np.sum(mutant) * n_points / 2
                cross_points = np.random.rand(n_points) < crossover
                trial = np.where(cross_points, mutant, population[i])
                trial = np.clip(trial, 0, 1)
                trial = trial / np.sum(trial) * n_points / 2
                new_population.append(trial)
            population = np.array(new_population)
            for indiv in population:
                c5 = evaluate_candidate(indiv)
                if c5 < best_c5:
                    best_h = indiv.copy()
                    best_c5 = c5
        return best_h, best_c5

    best_h, best_c5 = de_run()

    if time() - start_time < budget_s:
        for _ in range(50):
            if time() - start_time > budget_s:
                break
            h = np.random.rand(n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
