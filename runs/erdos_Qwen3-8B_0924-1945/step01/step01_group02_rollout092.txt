import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def obj_and_cons(h_flat):
        h = h_flat.reshape(n_points)
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clipped)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h_clipped * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        c5 = np.max(corr * dx)
        return c5

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj_and_cons(h)

    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        h_new = np.copy(h)
        i = np.random.randint(n_points)
        h_new[i] = np.random.uniform(0.0, 1.0)
        h_new = np.clip(h_new, 0.0, 1.0)
        sum_h_new = np.sum(h_new)
        scale = target_sum / sum_h_new if sum_h_new != 0 else 1.0
        h_scaled = h_new * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        c5_new = np.max(corr * dx)
        if c5_new < best_c5:
            best_h = h_scaled.copy()
            best_c5 = c5_new
            h = h_scaled.copy()
    h = best_h.copy()
    c5_bound = best_c5
    return h.tolist(), c5_bound, n_points
