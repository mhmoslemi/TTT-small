import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h_values = np.array([1, 0, 0, 0, 1, 0, 0, 0, 1, 0, 0, 0, 1, 0, 0, 0, 1, 0, 0, 0, 1, 0, 0, 0, 1, 0, 0, 0, 1, 0, 0, 0, 1, 0, 0, 0, 1, 0, 0, 0, 1, 0])
    h = np.copy(initial_h_values)
    h = np.clip(h, 0, 1)
    h /= np.sum(h)
    h *= target_sum

    def objective(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    best_h = np.copy(h)
    best_c5 = objective(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_h /= np.sum(best_h)
    best_h *= target_sum
    best_c5 = objective(best_h)

    for _ in range(5):
        perturbed = np.random.uniform(0.8, 1.2, size=n_points)
        guess = h * perturbed
        guess = np.clip(guess, 0, 1)
        guess /= np.sum(guess)
        guess *= target_sum
        result = opt.minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-6, 'eps': 1e-6}
        )
        h_candidate = result.x
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate /= np.sum(h_candidate)
        h_candidate *= target_sum
        c5_candidate = objective(h_candidate)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate

    return (best_h, best_c5, n_points)
