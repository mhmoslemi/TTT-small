import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    
    def normalize_h(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        scale = (n_points / 2) / total
        return h * scale
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def objective(h):
        h_norm = normalize_h(h)
        return compute_c5(h_norm)
    
    def bounds_func(x):
        return [(0, 1) for _ in range(n_points)]
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def de_func(x):
        h = normalize_h(x)
        return compute_c5(h)
    
    def run_de():
        result = differential_evolution(de_func, bounds, strategy='best1bin', 
                                       popsize=20, tol=1e-5, mutation=(0.5, 1), 
                                       recombination=0.9, disp=False, 
                                       maxiter=100, polish=False)
        best_h = normalize_h(result.x)
        return best_h, compute_c5(best_h)
    
    def run_sqp(h):
        def sqp_objective(x):
            h_norm = normalize_h(x)
            return compute_c5(h_norm)
        res = minimize(sqp_objective, h, method='SLSQP', bounds=bounds, 
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(normalize_h(x)) - n_points / 2}], 
                       tol=1e-5)
        return normalize_h(res.x), compute_c5(normalize_h(res.x))
    
    start_time = time()
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)
    
    if 'initial_h_values' in kwargs:
        best_h = normalize_h(best_h)
        best_c5 = compute_c5(best_h)
    
    # Differential evolution
    if time() - start_time < budget_s:
        best_h_de, best_c5_de = run_de()
        if best_c5_de < best_c5:
            best_h, best_c5 = best_h_de, best_c5_de
    
    # Local refinement
    if time() - start_time < budget_s:
        best_h, best_c5 = run_sqp(best_h)
    
    # Final refinement
    if time() - start_time < budget_s:
        h = np.random.uniform(0, 1, n_points)
        h = normalize_h(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h, best_c5 = h, current_c5
    
    best_h = normalize_h(best_h)
    return (best_h, best_c5, n_points)
