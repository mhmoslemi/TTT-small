import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = np.round(initial_h_values).astype(int)
    else:
        h = np.zeros(n_points, dtype=int)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                h[i] = 1
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2
    h = np.round(h).astype(int)
    h = h / np.sum(h) * n_points / 2
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)
    
    def objective(x):
        return evaluate_candidate(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    
    def run_ga():
        pop_size = 30
        mut_rate = 0.1
        cross_rate = 0.8
        generations = 100
        elite_frac = 0.2
        local_ref_steps = 10
        
        def generate_binary_vector():
            vec = np.zeros(n_points, dtype=int)
            vec[:n_points//2] = 1
            np.random.shuffle(vec)
            return vec
        
        population = np.array([generate_binary_vector() for _ in range(pop_size)])
        population = population / np.sum(population, axis=1, keepdims=True) * n_points / 2
        population = population.astype(float)
        
        for gen in range(generations):
            if time() - start_time > budget_s:
                break
            scores = np.array([evaluate_candidate(ind) for ind in population])
            elite_mask = scores <= np.sort(scores)[::-1][int(elite_frac * pop_size):]
            elites = population[elite_mask]
            
            new_population = elites.copy()
            while len(new_population) < pop_size:
                if np.random.rand() < cross_rate:
                    p1, p2 = np.random.choice(elite_mask, size=2, replace=False)
                    child = np.copy(population[p1])
                    split = np.random.randint(1, n_points - 1)
                    child[:split] = population[p2][:split]
                    child[split:] = population[p1][split:]
                    child = np.clip(child, 0, 1)
                    child = child / np.sum(child) * n_points / 2
                else:
                    p = np.random.choice(elite_mask)
                    child = np.copy(population[p])
                    for i in range(n_points):
                        if np.random.rand() < mut_rate:
                            child[i] = 1 - child[i]
                    child = np.clip(child, 0, 1)
                    child = child / np.sum(child) * n_points / 2
                new_population = np.vstack((new_population, child))
            population = new_population[:pop_size]
            
            if np.min(scores) < best_c5:
                best_c5 = np.min(scores)
                best_h = population[np.argmin(scores)].copy()
        
        for _ in range(local_ref_steps):
            if time() - start_time > budget_s:
                break
            h = best_h.copy()
            h = h / np.sum(h) * n_points / 2
            res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
            if res.success:
                h = res.x
                current_c5 = evaluate_candidate(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
    
    run_ga()
    
    for _ in range(50):
        if time() - start_time > budget_s:
            break
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
        current_c5 = evaluate_candidate(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
