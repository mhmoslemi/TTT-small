import numpy as np
import scipy.optimize as optimize
import numpy.random as npr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Random initial guess
        h = np.random.rand(100)
        h = h * (50.0 / np.sum(h)) if np.sum(h) != 0 else h
    
    h = np.clip(h, 0.0, 1.0)
    h = h * (50.0 / np.sum(h)) if np.sum(h) != 0 else h
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def objective(h_):
        h_ = np.clip(h_, 0.0, 1.0)
        h_norm = h_ * (target_sum / np.sum(h_)) if np.sum(h_) != 0 else h_
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx
    
    def constraint(h_):
        return np.sum(h_) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    # Differential Evolution optimization
    result = optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=200,
        popsize=50,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.7,
        disp=False,
        polish=True,
        workers=-1,
        vectorized=True
    )
    
    h_opt = result.x
    h_opt = np.clip(h_opt, 0.0, 1.0)
    h_opt = h_opt * (target_sum / np.sum(h_opt)) if np.sum(h_opt) != 0 else h_opt
    c5_bound = objective(h_opt)
    
    return (h_opt.tolist(), c5_bound, n_points)
