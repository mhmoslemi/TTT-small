import numpy as np
import scipy.optimize as optimize

def run(seed=42, budget_s=600, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 1.0 / n_points)
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h

    def objective(h_):
        h_ = np.clip(h_, 0.0, 1.0)
        h_ = h_ * (target_sum / np.sum(h_)) if np.sum(h_) != 0 else h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def differential_evolution(population, max_iter=200, seed=42):
        np.random.seed(seed)
        pop_size = len(population)
        mutation_factor = 0.5
        crossover_rate = 0.7
        best_solution = population[0]
        best_score = objective(best_solution)
        for _ in range(max_iter):
            for i in range(pop_size):
                a, b, c = np.random.choice(pop_size, 3, replace=False)
                mutant = population[a] + mutation_factor * (population[b] - population[c])
                mutant = np.clip(mutant, 0.0, 1.0)
                mutant = mutant * (target_sum / np.sum(mutant)) if np.sum(mutant) != 0 else mutant
                trial = population[i].copy()
                for j in range(n_points):
                    if np.random.rand() < crossover_rate:
                        trial[j] = mutant[j]
                trial = np.clip(trial, 0.0, 1.0)
                trial = trial * (target_sum / np.sum(trial)) if np.sum(trial) != 0 else trial
                score = objective(trial)
                if score < best_score:
                    best_score = score
                    best_solution = trial
                if score < objective(population[i]):
                    population[i] = trial
        return best_solution, best_score

    population_size = 50
    population = [np.clip(h + np.random.normal(0, 0.1, n_points), 0.0, 1.0) for _ in range(population_size)]
    population = [h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h for h in population]
    best_h, best_c5 = differential_evolution(population, max_iter=200, seed=seed)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (target_sum / np.sum(best_h)) if np.sum(best_h) != 0 else best_h
    final_c5 = objective(best_h)

    return (best_h.tolist(), final_c5, n_points)
