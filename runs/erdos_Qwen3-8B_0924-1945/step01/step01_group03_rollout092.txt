import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float)
        base = np.clip(base, 0, 1)
        base /= np.sum(base)
        base *= target_sum
    else:
        x = np.linspace(0, 2, n_points)
        base = 0.5 + 0.5 * np.sin(np.pi * x)
        base = np.clip(base, 0, 1)
        base /= np.sum(base)
        base *= target_sum

    def objective(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(
        objective,
        base,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'ftol': 1e-6, 'gtol': 1e-6, 'maxiter': 200}
    )
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h /= np.sum(best_h)
    best_h *= target_sum
    c5_bound = objective(best_h)

    return (best_h, c5_bound, n_points)
