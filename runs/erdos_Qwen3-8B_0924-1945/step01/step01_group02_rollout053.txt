```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def build_constraints(n_points, dx):
        constraints = []
        for k in range(1, n_points):
            A = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    A[i, j] = dx
                j = i - k
                if j >= 0:
                    A[i, j] = dx
            constraints.append(A)
        return constraints

    def objective(C, h):
        return C

    def constraint_1(h):
        return np.sum(h) - target_sum

    def constraint_2(C, h, A, rhs):
        return C - h @ A @ h

    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()
    h_flat = h.copy()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        constraints = build_constraints(n_points, dx)
        rhs = np.ones(len(constraints)) * 1.0
        C = cp.Variable()
        h = cp.Variable(n_points, bounds=(0.0, 1.0))
        problem = cp.Problem(cp.Minimize(C), [
            constraint_1(h),
            *[constraint_2(C, h, A, rhs[i]) >= 0 for i, A in enumerate(constraints)]
        ])
        try:
            problem.solve(solver=cp.SCS, eps=1e-6, max_iters=100)
            if problem.status in ['OPTIMAL', 'OPTIMAL_INACCURATE', 'SUCCESS']:
                h_sol = h.value
                c5 = evaluate(h_sol)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_sol.copy()
                h_flat = h_sol.copy()
        except:
            pass

    h_opt = best_h
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = evaluate(h_opt)
    return h_opt.tolist(), c5_bound, n_points
```
