import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        idx = np.linspace(0, n_points, 3, endpoint=False).astype(int)
        h[idx] = 1
        h = np.interp(np.arange(n_points), idx, h)
        h = np.clip(h, 0, 1)
        h = h * n_points / 2 / np.sum(h)
    
    target_sum = n_points / 2
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(best_h)
    remaining_time = budget_s
    
    for _ in range(100):
        if remaining_time < 10:
            break
        population = []
        for _ in range(50):
            if np.random.rand() < 0.2:
                h_candidate = np.random.rand(n_points)
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            else:
                h_candidate = best_h.copy()
                mutation = np.random.rand(n_points) < 0.1
                h_candidate[mutation] = np.random.rand(np.sum(mutation))
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            population.append(h_candidate)
        
        fitness = [objective(h) for h in population]
        population = [p for _, p in sorted(zip(fitness, population))]
        population = population[:30]
        
        for _ in range(20):
            if remaining_time < 10:
                break
            if len(population) < 2:
                break
            h1, h2 = np.random.choice(population, 2, replace=False)
            h_offspring = h1.copy()
            idx = np.random.choice(n_points, 2, replace=False)
            h_offspring[idx[0]:idx[1]] = h2[idx[0]:idx[1]]
            h_offspring = np.clip(h_offspring, 0, 1)
            h_offspring = h_offspring * target_sum / np.sum(h_offspring)
            population.append(h_offspring)
        
        population = np.array(population)
        fitness = objective(population)
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        remaining_time = budget_s - (time.time() - start_time)
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
