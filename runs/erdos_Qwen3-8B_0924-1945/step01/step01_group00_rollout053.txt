import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = np.round(initial_h_values).astype(int)
    else:
        h = np.zeros(n_points, dtype=int)
        for i in range(n_points):
            h[i] = 1 if np.random.rand() < 0.5 else 0
        while np.sum(h) != n_points // 2:
            idx = np.random.randint(n_points)
            h[idx] = 1 - h[idx]
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)
    
    def fitness(h):
        return evaluate_candidate(h)
    
    def objective(x):
        return fitness(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time()
    
    def run_optimization():
        nonlocal best_h, best_c5
        population_size = 30
        num_generations = 100
        mutation_rate = 0.1
        crossover_rate = 0.8
        elite_fraction = 0.2
        
        population = np.array([np.random.choice([0, 1], size=n_points, p=[0.5, 0.5]) for _ in range(population_size)])
        for i in range(population_size):
            while np.sum(population[i]) != n_points // 2:
                idx = np.random.randint(n_points)
                population[i][idx] = 1 - population[i][idx]
        
        for gen in range(num_generations):
            if time() - start_time > budget_s:
                break
            fitness_scores = np.array([compute_c5(ind) for ind in population])
            elite_indices = np.argsort(fitness_scores)[:int(elite_fraction * population_size)]
            elites = population[elite_indices]
            
            new_population = elites.copy()
            for _ in range(population_size - len(elites)):
                parents = np.random.choice(elite_indices, size=2, replace=False)
                parent1 = population[parents[0]]
                parent2 = population[parents[1]]
                if np.random.rand() < crossover_rate:
                    split = np.random.randint(1, n_points - 1)
                    child = np.concatenate([parent1[:split], parent2[split:]])
                else:
                    child = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
                while np.sum(child) != n_points // 2:
                    idx = np.random.randint(n_points)
                    child[idx] = 1 - child[idx]
                new_population = np.vstack([new_population, child])
            population = new_population
            current_best = np.argmin(fitness_scores)
            current_c5 = fitness_scores[current_best]
            if current_c5 < best_c5:
                best_h = population[current_best].copy()
                best_c5 = current_c5
        
        for _ in range(10):
            if time() - start_time > budget_s:
                break
            h = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
            while np.sum(h) != n_points // 2:
                idx = np.random.randint(n_points)
                h[idx] = 1 - h[idx]
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    run_optimization()
    
    for _ in range(10):
        if time() - start_time > budget_s:
            break
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
        current_c5 = evaluate_candidate(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    
    return (best_h, current_c5, n_points)
