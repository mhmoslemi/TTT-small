import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def grad_objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_k = np.argmax(correlate_vals)
        grad = np.zeros_like(h)
        for i in range(n_points):
            x = i * (2.0 / n_points)
            if x + max_k <= 2.0:
                grad[i] += (1 - h[int((x + max_k) * n_points / 2)])
            if x - max_k >= 0:
                grad[i] += (1 - h[int((x - max_k) * n_points / 2)])
        return grad * (2.0 / n_points)
    
    def lbfgs_step(h_vec):
        result = opt.minimize(
            objective,
            h_vec,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            jac=grad_objective,
            options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-6}
        )
        h = result.x
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def de_step(h_vec):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(
            objective,
            bounds,
            popsize=30,
            mutation=(0.9, 1.0),
            recombination=0.7,
            strategy='best1bin',
            tol=1e-5,
            maxiter=200,
            disp=False,
            polish=True,
            seed=seed
        )
        h = result.x
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning initial solution")
        h = initial_h_values if 'initial_h_values' in globals() else h
        best_c5 = objective(h)
        return (h.tolist(), best_c5, n_points)
    
    for _ in range(10):
        h = lbfgs_step(h)
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if current_c5 < best_c5 - 1e-6:
            break
        
        if elapsed > budget_s:
            break
        
        h = de_step(h)
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if current_c5 < best_c5 - 1e-6:
            break
        
        if elapsed > budget_s:
            break
    
    # Final evaluation
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points)
