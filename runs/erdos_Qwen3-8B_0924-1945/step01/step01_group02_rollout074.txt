```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        counts = np.ceil(n_points / 2).astype(int)
        indices = np.random.choice(n_points, counts, replace=False)
        h[indices] = 1.0
        h = h * (target_sum / np.sum(h))

    def build_constraints(h, n_points, dx):
        constraints = []
        for k in range(1, n_points):
            A = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = (i + k) % n_points
                A[i, j] = dx
                j = (i - k + n_points) % n_points
                A[i, j] = dx
            constraints.append(A)
        return constraints

    def optimize_with_cvxpy(h_initial, n_points, dx, target_sum):
        h = cp.Variable(n_points)
        C = cp.Variable()
        constraints = []
        constraints.append(cp.sum(h) == target_sum)
        constraints.append(0 <= h)
        constraints.append(h <= 1)
        for k in range(1, n_points):
            A = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = (i + k) % n_points
                A[i, j] = dx
                j = (i - k + n_points) % n_points
                A[i, j] = dx
            quad = cp.quad_form(h, A)
            constraints.append(quad >= 1 - C)
        obj = cp.Minimize(C)
        prob = cp.Problem(obj, constraints)
        result = prob.solve(solver='ECOS', eps=1e-8, verbose=False)
        if result is None:
            return h_initial
        return h.value

    best_h = h.copy()
    best_c5 = float('inf')
    elapsed = 0
    start_time = time.time()

    for _ in range(20):
        if elapsed >= budget_s:
            break
        h_initial = h.copy()
        h_initial += np.random.uniform(-0.05, 0.05, size=n_points)
        h_initial = np.clip(h_initial, 0.0, 1.0)
        h_initial = h_initial * (target_sum / np.sum(h_initial))
        h_opt = optimize_with_cvxpy(h_initial, n_points, dx, target_sum)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        h_opt = h_opt * (target_sum / sum_h_opt)
        corr = np.correlate(h_opt, 1.0 - h_opt, mode='full')
        c5 = np.max(corr * dx)
        if c5 < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5
        elapsed = time.time() - start_time

    h_opt = best_h
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    h_opt = h_opt * (target_sum / sum_h_opt)
    corr = np.correlate(h_opt, 1.0 - h_opt, mode='full')
    c5_bound = np.max(corr * dx)
    return h_opt.tolist(), c5_bound, n_points
```
