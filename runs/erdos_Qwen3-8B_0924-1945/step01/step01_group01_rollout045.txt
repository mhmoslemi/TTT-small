import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h with provided initial configuration or random
    def initialize_h(n_points):
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values, dtype=np.float64)
        else:
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        return h
    
    n_points = 45
    h = initialize_h(n_points)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Stage 1: Coarse global optimization with DE
    def stage1():
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(
            objective, bounds, popsize=50, mutation=(0.8, 1.0),
            recombination=0.9, strategy='best1bin', tol=1e-5,
            maxiter=200, disp=False, polish=True, seed=seed
        )
        best_h = result.x
        best_c5 = result.fun
        return best_h, best_c5
    
    # Stage 2: Gradient-based refinement
    def stage2(h_init, n_points):
        def grad_func(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            max_k = np.argmax(correlate_vals)
            grad = np.zeros_like(h)
            for i in range(n_points):
                if i + max_k < n_points:
                    grad[i] += (1 - h[i + max_k]) * (2.0 / n_points)
                if i - max_k >= 0:
                    grad[i] += (1 - h[i - max_k]) * (2.0 / n_points)
            return grad
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.minimize(
            objective, h_init, method='L-BFGS-B', jac=grad_func,
            bounds=bounds, tol=1e-6, maxiter=300, options={'ftol': 1e-6}
        )
        best_h = result.x
        best_c5 = result.fun
        return best_h, best_c5
    
    # Stage 3: Multi-scale refinement
    def stage3(h_init, new_n_points):
        dx = 2.0 / new_n_points
        h = np.interp(np.linspace(0, 2, new_n_points), np.linspace(0, 2, n_points), h_init)
        h = np.clip(h, 0, 1)
        h = h * (new_n_points / 2) / np.sum(h)
        return stage2(h, new_n_points)
    
    # Execute stages
    best_h, best_c5 = stage1()
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning initial solution")
        h = initial_h_values if 'initial_h_values' in globals() else h
        best_c5 = objective(h)
    else:
        # Stage 2
        h_stage2, c5_stage2 = stage2(best_h, n_points)
        if c5_stage2 < best_c5:
            best_h, best_c5 = h_stage2, c5_stage2
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            print("Time budget exceeded, returning stage2 solution")
        else:
            # Stage 3: increase resolution
            new_n_points = 90
            h_stage3, c5_stage3 = stage3(best_h, new_n_points)
            if c5_stage3 < best_c5:
                best_h, best_c5 = h_stage3, c5_stage3
            elapsed = time.time() - start_time
            if elapsed > budget_s:
                print("Time budget exceeded, returning stage3 solution")
    
    # Final clipping and rescaling
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
