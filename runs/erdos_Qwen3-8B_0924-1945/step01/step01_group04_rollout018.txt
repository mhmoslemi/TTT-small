import numpy as np
import scipy.optimize as optimize

def run(seed=42, budget_s=600, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 1.0 / n_points)
    
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
    
    def objective(h_):
        h_ = np.clip(h_, 0.0, 1.0)
        h_ = h_ * (target_sum / np.sum(h_)) if np.sum(h_) != 0 else h_
        corr = np.correlate(h_, 1 - h_, mode="full")
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def constraint(h_):
        return np.sum(h_) - target_sum
    
    def de_optimize():
        from scipy.optimize import differential_evolution
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=50,
            tol=1e-6,
            mutation=(0.5, 1.0),
            recombination=0.7,
            maxiter=200,
            polish=True,
            disp=False,
            seed=seed
        )
        best_h = result.x
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = best_h * (target_sum / np.sum(best_h)) if np.sum(best_h) != 0 else best_h
        return best_h, objective(best_h)
    
    try:
        h_opt, c5_bound = de_optimize()
    except:
        h_opt = h
        c5_bound = objective(h)
    
    if c5_bound > 0.385:
        h_opt = h
        c5_bound = objective(h)
    
    return (h_opt.tolist(), c5_bound, n_points)
