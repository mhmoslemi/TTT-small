import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def gradient(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_idx = np.argmax(correlate_vals)
        k = max_idx - n_points + 1
        if k < 0 or k >= n_points:
            return np.zeros_like(h)
        grad = np.zeros_like(h)
        for i in range(n_points):
            x = i * (2.0 / n_points)
            x_shifted = x + k * (2.0 / n_points)
            if 0 <= x_shifted < 2:
                idx = int(np.round(x_shifted * n_points / 2))
                if 0 <= idx < n_points:
                    grad[i] = (1 - h[idx]) * (2.0 / n_points)
        return grad
    
    def lbfgs_step(h, max_iter=100, tol=1e-6):
        for _ in range(max_iter):
            grad = gradient(h)
            h_new = opt.linesearch.scalar_search_armijo(lambda x: objective(x), h, grad, bounds=[(0,1)]*n_points)
            if np.linalg.norm(h_new - h) < tol:
                break
            h = h_new
        return h
    
    def de_refine(h, n_points_new, max_iter=200, pop_size=30):
        h = np.clip(h, 0, 1)
        h = h * (n_points_new / 2) / np.sum(h)
        bounds = [(0.0, 1.0) for _ in range(n_points_new)]
        result = opt.differential_evolution(lambda x: objective(x), bounds, 
                                           popsize=pop_size, mutation=(0.9, 1.0), 
                                           recombination=0.7, strategy='best1bin', 
                                           tol=1e-5, maxiter=max_iter, disp=False, 
                                           polish=True, seed=seed)
        return result.x
    
    best_h = h.copy()
    best_c5 = objective(h)
    elapsed = time.time() - start_time
    
    # Stage 1: Gradient refinement
    h_refined = lbfgs_step(h)
    c5_refined = objective(h_refined)
    if c5_refined < best_c5:
        best_h, best_c5 = h_refined, c5_refined
    
    # Stage 2: Refinement with higher resolution
    n_points_new = 90
    h_high = de_refine(h_refined, n_points_new)
    c5_high = objective(h_high)
    if c5_high < best_c5:
        best_h, best_c5 = h_high, c5_high
    
    # Stage 3: Final refinement or fallback
    if elapsed < budget_s - 200:
        h_final = de_refine(best_h, n_points, max_iter=200, pop_size=30)
        c5_final = objective(h_final)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final
    
    # Final checks
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
    
    return (h.tolist(), final_c5, n_points)
