import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = np.round(initial_h_values).astype(int)
        h = np.where(h == 1, 1, 0)
        if np.sum(h) != n_points // 2:
            diff = n_points // 2 - np.sum(h)
            h[np.argsort(np.abs(h - 0.5))[:diff]] = 1
    else:
        h = np.zeros(n_points, dtype=int)
        indices = np.random.choice(n_points, size=n_points // 2, replace=False)
        h[indices] = 1
    h = h.astype(float)
    dx = 2.0 / n_points
    correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
    c5_bound = np.max(correlate_vals)

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    best_h = h.copy()
    best_c5 = c5_bound
    start_time = time()

    def run_optimization():
        nonlocal best_h, best_c5
        pop_size = 30
        mut_rate = 0.1
        cross_rate = 0.8
        max_gen = 100
        for gen in range(max_gen):
            if time() - start_time > budget_s:
                break
            pop = np.random.randint(0, 2, size=(pop_size, n_points))
            pop = np.where(np.sum(pop, axis=1)[:, np.newaxis] != n_points // 2, pop, np.where(np.random.rand(pop_size, n_points) < 0.5, 1 - pop, pop))
            fitness_vals = np.array([compute_c5(p) for p in pop])
            idxs = np.argsort(fitness_vals)[:int(pop_size * 0.2)]
            elite = pop[idxs]
            new_pop = []
            for i in range(pop_size // 2):
                p1 = elite[np.random.choice(len(elite))]
                p2 = elite[np.random.choice(len(elite))]
                child = np.copy(p1)
                if np.random.rand() < cross_rate:
                    idx = np.random.choice(n_points)
                    child[idx:] = p2[idx:]
                child = np.where(np.sum(child) != n_points // 2, np.where(np.random.rand(n_points) < 0.5, 1 - child, child), child)
                new_pop.append(child)
            pop = np.vstack((elite, new_pop))
            pop = pop[np.random.permutation(pop_size)]
            for p in pop:
                if np.random.rand() < mut_rate:
                    idx = np.random.choice(n_points)
                    p[idx] = 1 - p[idx]
                    if np.sum(p) != n_points // 2:
                        diff = n_points // 2 - np.sum(p)
                        p[np.argsort(np.abs(p - 0.5))[:diff]] = 1
            fitness_vals = np.array([compute_c5(p) for p in pop])
            best_idx = np.argmin(fitness_vals)
            if fitness_vals[best_idx] < best_c5:
                best_h = pop[best_idx].copy()
                best_c5 = fitness_vals[best_idx]
        return best_h, best_c5

    best_h, best_c5 = run_optimization()

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
