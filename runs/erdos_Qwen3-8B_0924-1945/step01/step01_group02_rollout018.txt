```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        h[:n_points // 2] = 1.0
        h = h * (target_sum / np.sum(h))

    h = h * (target_sum / np.sum(h))

    def build_overlap_matrix(k):
        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = (i + k) % n_points
            A[i, j] = dx
            j = (i - k + n_points) % n_points
            A[i, j] = dx
        return A

    A_list = [build_overlap_matrix(k) for k in range(1, n_points)]

    h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
    C = cp.Variable()

    constraints = []
    constraints += [cp.sum(h_var * dx) == target_sum]
    for A in A_list:
        constraints += [cp.quad_form(h_var, A) >= 1.0 - C]

    prob = cp.Problem(cp.Minimize(C), constraints)

    solver_options = {
        'ECOS': {
            'max_iters': 1000,
            'atol': 1e-8,
            'rtol': 1e-8
        },
        'SCS': {
            'max_iters': 1000,
            'eps': 1e-6
        }
    }

    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()
    for solver in ['ECOS', 'SCS']:
        try:
            prob.solve(solver=solver, **solver_options[solver])
            if prob.status == 'OPTIMAL' and C.value < best_c5:
                best_c5 = C.value
                best_h = h_var.value.copy()
        except:
            pass
        if time.time() - start_time > budget_s:
            break

    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    best_h = best_h * scale
    best_c5 = np.max(np.correlate(best_h, 1.0 - best_h, mode="full") * dx)
    return best_h.tolist(), best_c5, n_points
```
