import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = np.copy(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = (h / np.sum(h)) * target_sum
    
    def objective(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(x):
        return np.sum(x) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h
    best_c5 = objective(h)
    best_h = np.clip(best_h, 0, 1)
    best_h = (best_h / np.sum(best_h)) * target_sum
    best_c5 = objective(best_h)
    
    for _ in range(3):
        h_rand = np.random.rand(n_points)
        h_rand = (h_rand / np.sum(h_rand)) * target_sum
        h_rand = np.clip(h_rand, 0, 1)
        c5 = objective(h_rand)
        if c5 < best_c5:
            best_h = h_rand
            best_c5 = c5
    
    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 200, 'ftol': 1e-6, 'eps': 1e-6}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = (best_h / np.sum(best_h)) * target_sum
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
