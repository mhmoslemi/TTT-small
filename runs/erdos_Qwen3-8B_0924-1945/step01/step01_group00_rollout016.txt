import numpy as np
from scipy.optimize import minimize
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    h = np.round(initial_h_values).copy()
    ones_needed = n_points // 2
    ones_current = np.sum(h)
    if ones_current < ones_needed:
        diff = ones_needed - ones_current
        h[:diff] = 1
    elif ones_current > ones_needed:
        diff = ones_current - ones_needed
        h[-diff:] = 0
    h = h.astype(float)
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        ones_current = np.sum(h)
        if ones_current < n_points // 2:
            diff = (n_points // 2) - ones_current
            h[:diff] = 1
        elif ones_current > n_points // 2:
            diff = ones_current - (n_points // 2)
            h[-diff:] = 0
        h = h.astype(float)
        return compute_c5(h)
    
    def objective(x):
        return evaluate_candidate(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    best_h = h.copy()
    best_c5 = evaluate_candidate(h)
    start_time = time()
    
    for _ in range(100):
        if time() - start_time > budget_s:
            break
        x0 = best_h.copy()
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            current_h = res.x
            current_c5 = evaluate_candidate(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            break
    
    for _ in range(50):
        if time() - start_time > budget_s:
            break
        h = np.random.uniform(0, 1, n_points)
        ones_current = np.sum(h)
        if ones_current < n_points // 2:
            diff = (n_points // 2) - ones_current
            h[:diff] = 1
        elif ones_current > n_points // 2:
            diff = ones_current - (n_points // 2)
            h[-diff:] = 0
        h = h.astype(float)
        current_c5 = evaluate_candidate(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    best_h = np.clip(best_h, 0, 1)
    ones_current = np.sum(best_h)
    if ones_current < n_points // 2:
        diff = (n_points // 2) - ones_current
        best_h[:diff] = 1
    elif ones_current > n_points // 2:
        diff = ones_current - (n_points // 2)
        best_h[-diff:] = 0
    best_h = best_h.astype(float)
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
