```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(h)

    for _ in range(20):
        if np.random.rand() < 0.5:
            h = initial_h_values.copy() if initial_h_values is not None else np.zeros(n_points)
            for i in range(n_points):
                h[i] = 1.0 if i % 3 == 0 else 0.0
            h = h * (target_sum / np.sum(h))
        else:
            h = np.random.uniform(0, 1, n_points)
            h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0.0, 1.0)

        h_flat = h.flatten()
        x = cp.Variable(n_points)
        c = cp.Variable()
        constraints = []
        constraints.append(cp.sum(x) == target_sum)
        for k in range(1, n_points):
            A = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = (i + k) % n_points
                A[i, j] = dx
                j = (i - k + n_points) % n_points
                A[i, j] = dx
            constraint_expr = cp.quad_form(x, A) >= 1 - c
            constraints.append(constraint_expr)
        constraints.append(cp.Bounds(0, 1, x))
        prob = cp.Problem(cp.Minimize(c), constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-8, max_iter=200)
            if prob.status == 'optimal':
                h_opt = x.value
                h_opt = np.clip(h_opt, 0.0, 1.0)
                sum_h_opt = np.sum(h_opt)
                scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
                h_opt = h_opt * scale
                c5 = objective(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
        except:
            pass

        if time.time() - start_time > budget_s - 50:
            break

    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    best_h = best_h * scale

    return best_h.tolist(), best_c5, n_points
```
