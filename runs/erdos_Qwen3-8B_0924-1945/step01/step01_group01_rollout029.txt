import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 90
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    n_points = len(h)
    dx = 2.0 / n_points
    best_c5 = np.inf
    best_h = h
    
    def objective(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def gradient(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        max_lag = np.argmax(correlate_vals)
        lag = max_lag - (n_points - 1)
        if lag < 0:
            lag = 0
        elif lag > n_points - 1:
            lag = n_points - 1
        grad = np.zeros_like(h)
        for i in range(n_points):
            if lag + i < n_points:
                grad[i] = (1 - h[lag + i]) * dx
        return grad
    
    def constraint_func(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return np.sum(h) - n_points / 2
    
    def constraint_func_jac(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        sum_h = np.sum(h)
        jac = np.zeros_like(h)
        for i in range(n_points):
            jac[i] = (n_points / 2) / sum_h - (h[i] * (n_points / 2)) / (sum_h ** 2)
        return jac
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Gradient-based optimization
    result = opt.minimize(objective, h, method='L-BFGS-B', jac=gradient, bounds=bounds,
                          constraints={'type': 'eq', 'fun': constraint_func, 'jac': constraint_func_jac},
                          tol=1e-5, options={'maxiter': 500, 'ftol': 1e-8})
    
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        best_c5 = objective(h)
    
    # Adaptive refinement
    for _ in range(3):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        n_points *= 2
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        h[:len(best_h)] = best_h
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        def new_objective(h_vec):
            h = np.clip(h_vec, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(correlate_vals)
        
        def new_gradient(h_vec):
            h = np.clip(h_vec, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
            max_lag = np.argmax(correlate_vals)
            lag = max_lag - (n_points - 1)
            if lag < 0:
                lag = 0
            elif lag > n_points - 1:
                lag = n_points - 1
            grad = np.zeros_like(h)
            for i in range(n_points):
                if lag + i < n_points:
                    grad[i] = (1 - h[lag + i]) * dx
            return grad
        
        def new_constraint_func(h_vec):
            h = np.clip(h_vec, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            return np.sum(h) - n_points / 2
        
        def new_constraint_func_jac(h_vec):
            h = np.clip(h_vec, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            sum_h = np.sum(h)
            jac = np.zeros_like(h)
            for i in range(n_points):
                jac[i] = (n_points / 2) / sum_h - (h[i] * (n_points / 2)) / (sum_h ** 2)
            return jac
        
        new_bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        result = opt.minimize(new_objective, h, method='L-BFGS-B', jac=new_gradient, bounds=new_bounds,
                              constraints={'type': 'eq', 'fun': new_constraint_func, 'jac': new_constraint_func_jac},
                              tol=1e-5, options={'maxiter': 500, 'ftol': 1e-8})
        
        if result.success:
            best_h = result.x
            best_c5 = result.fun
        else:
            best_c5 = new_objective(h)
    
    # Final check
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
        h = best_h
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        best_c5 = objective(h)
    
    # Final feasibility check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points)
