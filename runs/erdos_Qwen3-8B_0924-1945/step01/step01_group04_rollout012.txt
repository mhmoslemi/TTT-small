import numpy as np
import scipy.optimize as optimize
import numpy.random as npr

def run(seed=42, budget_s=600, **kwargs):
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                h[i] = 1.0
        h = h * (target_sum / np.sum(h))
    
    h = np.clip(h, 0.0, 1.0)
    
    def objective(h_):
        h_ = np.clip(h_, 0.0, 1.0)
        h_ = h_ * (target_sum / np.sum(h_)) if np.sum(h_) != 0 else h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    def constraint(h_):
        return np.sum(h_) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    best_h = h.copy()
    best_c5 = objective(h)
    iter_count = 0
    max_iter = 100
    perturbation = 0.05
    
    for _ in range(max_iter):
        result = optimize.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1, 'ftol': 1e-8, 'eps': 1e-6}
        )
        
        h_opt = result.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        h_opt = h_opt * (target_sum / np.sum(h_opt)) if np.sum(h_opt) != 0 else h_opt
        c5 = objective(h_opt)
        
        if c5 < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5
            perturbation = min(perturbation * 1.1, 0.15)
        else:
            perturbation = max(perturbation * 0.9, 0.01)
        
        h_perturbed = best_h + np.random.uniform(-perturbation, perturbation, size=n_points)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (target_sum / np.sum(h_perturbed)) if np.sum(h_perturbed) != 0 else h_perturbed
        c5_perturbed = objective(h_perturbed)
        
        if c5_perturbed < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = c5_perturbed
        
        iter_count += 1
        if iter_count >= max_iter or best_c5 < 0.3808:
            break
    
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (target_sum / np.sum(best_h)) if np.sum(best_h) != 0 else best_h
    c5_bound = objective(best_h)
    
    return (best_h.tolist(), c5_bound, n_points)
