import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in kwargs:
        initial_h = kwargs['initial_h_values']
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h / np.sum(initial_h) * n_points / 2
    else:
        initial_h = np.random.uniform(0, 1, n_points)
        initial_h = initial_h / np.sum(initial_h) * n_points / 2

    dx = 2.0 / n_points
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate(h):
        return objective(h)

    bounds = [(0, 1) for _ in range(n_points)]
    pop_size = 20
    max_gen = 100
    mutation = 0.5
    recombination = 0.9

    start = time()
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)

    def de_func(pop):
        nonlocal best_h, best_c5
        new_pop = []
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + mutation * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant / np.sum(mutant) * n_points / 2
            trial = mutant
            if np.random.rand() < recombination:
                trial = pop[i] * (1 - recombination) + mutant * recombination
            trial = np.clip(trial, 0, 1)
            trial = trial / np.sum(trial) * n_points / 2
            trial_c5 = evaluate(trial)
            if trial_c5 < best_c5:
                best_h = trial.copy()
                best_c5 = trial_c5
            new_pop.append(trial)
        return new_pop

    pop = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
    for _ in range(max_gen):
        if time() - start > budget_s:
            break
        pop = de_func(pop)

    def local_refinement(h):
        res = minimize(objective, h, method='L-BFGS-B', bounds=bounds)
        return res.x

    best_h = local_refinement(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    best_c5 = evaluate(best_h)
    return (best_h, best_c5, n_points)
