```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = np.zeros(n_points)

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
    else:
        h = np.random.rand(n_points)
        h = np.sort(h)
        h = np.interp(h, np.linspace(0, 1, n_points), np.linspace(0, 1, n_points))
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))

    def compute_overlap(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale

        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def build_constraints():
        n = n_points
        A = []
        for k in range(1, n):
            row = np.zeros(n)
            for i in range(n - k):
                row[i] += dx
                row[i + k] += dx
            A.append(row)
        return np.array(A)

    A = build_constraints()
    n_constraints = len(A)

    h_var = cp.Variable(n_points)
    c_var = cp.Variable()

    constraints = [
        cp.Minimize(c_var),
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target_sum,
        [cp.sum(cp.multiply(A[i], cp.multiply(h_var, h_var))) >= 1 - c_var for i in range(n_constraints)]
    ]

    start_time = time.time()
    result = cp.Problem(cp.Minimize(c_var), constraints)
    result.solve(solver='SCS', eps=1e-6, max_iterations=200, verbose=False)

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result.solve(solver='ECOS', eps=1e-6, max_iterations=200, verbose=False)

    h_opt = np.clip(h_var.value, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale

    c5_bound = compute_overlap(h_opt.flatten())
    return h_opt.tolist(), c5_bound, n_points
```
