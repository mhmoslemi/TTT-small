import numpy as np
import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration with n_points = 45
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Define objective function
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Coarse optimization with DE
    def coarse_optimize(h_initial):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(
            objective, bounds, popsize=40, mutation=(0.8, 1.0), 
            recombination=0.9, strategy='best1bin', tol=1e-5, 
            maxiter=300, disp=False, polish=True, seed=seed)
        best_h = result.x
        best_c5 = result.fun
        return best_h, best_c5
    
    # Fine optimization with DE and local search
    def fine_optimize(h_initial, n_points_fine):
        h_fine = np.interp(np.linspace(0, 2, n_points_fine), np.linspace(0, 2, n_points), h_initial)
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_points_fine / 2) / np.sum(h_fine)
        bounds = [(0.0, 1.0) for _ in range(n_points_fine)]
        result = opt.differential_evolution(
            lambda x: objective(x), bounds, popsize=20, mutation=(1.0, 1.2), 
            recombination=0.9, strategy='best1bin', tol=1e-6, 
            maxiter=200, disp=False, polish=True, seed=seed)
        best_h = result.x
        best_c5 = result.fun
        # Local refinement with L-BFGS-B
        result_lbgfs = opt.minimize(lambda x: objective(x), best_h, method='L-BFGS-B', bounds=bounds, tol=1e-6)
        final_h = result_lbgfs.x
        final_c5 = result_lbgfs.fun
        return final_h, final_c5
    
    # Run coarse optimization
    best_h_coarse, best_c5_coarse = coarse_optimize(h)
    
    # Refine with finer grid
    n_points_fine = 90
    best_h_fine, best_c5_fine = fine_optimize(best_h_coarse, n_points_fine)
    
    # Ensure final feasibility
    final_h = best_h_fine
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points_fine / 2) / np.sum(final_h)
    
    # Final evaluation
    final_c5 = objective(final_h)
    
    # Check time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning fine-tuned solution")
        final_h = best_h_fine
        final_c5 = best_c5_fine
    
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points_fine / 2) / np.sum(final_h)
    final_c5 = objective(final_h)
    
    return (final_h.tolist(), final_c5, n_points_fine)
