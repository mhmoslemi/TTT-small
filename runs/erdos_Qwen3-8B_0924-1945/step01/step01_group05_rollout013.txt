import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Default number of points
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial_h_values if provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = (h - np.min(h)) / (np.max(h) - np.min(h))
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    
    # Objective function
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Constraints
    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Differential Evolution settings
    start_time = time.time()
    result = opt.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-6,
        seed=seed,
        polish=True,
        disp=False
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        h_final = np.clip(h, 0, 1)
        h_final = h_final * target_sum / np.sum(h_final)
        c5_final = objective(h_final)
        return (h_final, c5_final, n_points)
    
    h_final = result.x
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * target_sum / np.sum(h_final)
    c5_final = objective(h_final)
    
    return (h_final, c5_final, n_points)
