import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlate_vals) * (2.0 / n_points)
        return c5
    
    def gradient(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full")
        max_index = np.argmax(correlate_vals)
        k = max_index - n_points + 1
        if k < 0:
            k = 0
        elif k >= n_points:
            k = n_points - 1
        grad = np.zeros_like(h)
        if k < n_points:
            grad = 1 - h[k:]
        grad = grad * (2.0 / n_points)
        return grad
    
    def optimize_with_lbfgs(h_start, max_iter=300, tol=1e-6):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.minimize(
            objective,
            h_start,
            method='L-BFGS-B',
            jac=gradient,
            bounds=bounds,
            tol=tol,
            options={'maxiter': max_iter, 'disp': False}
        )
        if result.success:
            return result.x
        return h_start
    
    def optimize_with_de(h_start, max_iter=200):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(
            objective,
            bounds,
            popsize=50,
            mutation=(0.8, 1.0),
            recombination=0.9,
            strategy='best1bin',
            tol=1e-5,
            maxiter=max_iter,
            disp=False,
            polish=True,
            seed=seed
        )
        return result.x
    
    best_h = h.copy()
    best_c5 = objective(h)
    elapsed = time.time() - start_time
    
    # Stage 1: Coarse Global Optimization (DE)
    if elapsed < budget_s - 200:
        h = optimize_with_de(h, max_iter=200)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = objective(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        elapsed = time.time() - start_time
    
    # Stage 2: Local Gradient Refinement (L-BFGS-B)
    if elapsed < budget_s - 250:
        h = optimize_with_lbfgs(h, max_iter=300, tol=1e-6)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = objective(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        elapsed = time.time() - start_time

    if elapsed < budget_s - 100:

        if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
            h = np.random.rand(n_points)

            h = h * (n_points / 2) / np.sum(h)
        
        def objective_scaled(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(correlate_vals) * dx
            return c5
        
        def gradient_scaled(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full")
            max_index = np.argmax(correlate_vals)
            k = max_index - n_points + 1
            if k < 0:
                k = 0
            elif k >= n_points:
                k = n_points - 1
            grad = np.zeros_like(h)
            if k < n_points:
                grad = 1 - h[k:]
            grad = grad * dx
            return grad
        
        def optimize_with_lbfgs_scaled(h_start, max_iter=300, tol=1e-6):
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            result = opt.minimize(
                objective_scaled,
                h_start,
                method='L-BFGS-B',
                jac=gradient_scaled,
                bounds=bounds,
                tol=tol,
                options={'maxiter': max_iter, 'disp': False}
            )
            if result.success:
                return result.x
            return h_start
        
        h = optimize_with_lbfgs_scaled(h, max_iter=300, tol=1e-6)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = objective_scaled(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        elapsed = time.time() - start_time
    
    # Stage 4: Fallback if needed
    if elapsed < budget_s:
        if best_c5 > 0.385:
            h = optimize_with_de(best_h, max_iter=100)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            c5 = objective(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    
    # Final feasibility check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    # Return best solution
    return (h.tolist(), final_c5, n_points)
