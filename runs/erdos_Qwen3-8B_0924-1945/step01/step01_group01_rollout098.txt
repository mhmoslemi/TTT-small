import numpy as np
import scipy.optimize as opt
import time
from cvxopt import matrix, solvers

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Default number of points
    n_points = 90
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within bounds and sum is correct
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(mu):
        # Formulate the QP: minimize mu with constraints
        # Objective: mu
        # Constraints: sum(h_i * h_j) <= 1 - mu for each lag
        # More precisely, we cast it as: sum(h_i * h_j) + mu <= 1
        # This is a convex QP with variables h and mu
        
        # Prepare quadratic terms and constraints
        quad_matrix = np.zeros((n_points, n_points))
        for i in range(n_points):
            for j in range(n_points):
                if 0 <= i - j < n_points:
                    quad_matrix[i, j] += 1.0
                if 0 <= j - i < n_points:
                    quad_matrix[i, j] += 1.0
        
        # Convert to matrix form for CVXOPT
        P = matrix(quad_matrix, tc='d')
        q = matrix(np.zeros(n_points), tc='d')
        G = matrix(-np.eye(n_points), tc='d')
        h = matrix(np.ones(n_points), tc='d')
        A = matrix(np.ones(n_points), (1, n_points), tc='d')
        b = matrix(n_points / 2, tc='d')
        c = matrix(0.0, tc='d')
        
        # Objective is to minimize mu, so we add mu to the constraints
        # We reformulate the problem to minimize mu with sum(h) = n_points/2
        # and constraints: sum(h_i * h_j) + mu <= 1 for all i, j
        # This is equivalent to minimizing mu with constraints:
        # 1. sum(h) = n_points / 2
        # 2. sum(h_i * h_j) <= 1 - mu for all lags
        # We rewrite the constraint as sum(h_i * h_j) + mu <= 1
        
        # Create constraints for all lags
        # Use a matrix that represents sum(h_i * h_j) for all possible lags
        # We need to construct a matrix where entry (i, j) corresponds to a lag
        # and we have a constraint for each lag
        # This can be done by building a matrix of coefficients
        
        constraints = []
        for lag in range(1, n_points):
            row = np.zeros(n_points)
            for i in range(n_points):
                j = (i + lag) % n_points
                row[i] += 1.0
                row[j] += 1.0
            constraints.append(row)
        
        # Build the constraint matrix
        C = np.array(constraints)
        # Each constraint is sum(C[i] * h) <= 1 - mu
        # We rewrite this as: sum(C[i] * h) + mu <= 1, so mu <= 1 - sum(C[i] * h)
        # The constraint is: -sum(C[i] * h) <= mu - 1
        # Or equivalently: sum(C[i] * h) + mu <= 1
        # We can combine this into the QP form
        
        # For the CVXOPT QP, we have the form: min 0.5 * x^T P x + q^T x
        # where x is the vector of variables (h and mu)
        # So we need to combine h and mu into a vector of size n_points + 1
        # To do this, we expand all constraints to include mu
        
        # Formulate the QP with variables [h_0, h_1, ..., h_n, mu]
        # P is modified to include the mu terms
        # This is a full design decision
        # We focus on the h part and then handle the mu implicitly as part of the constraints
        
        # For the purpose of this formulation, we use the lag-based constraints
        # and minimize mu
        # So the main constraints are:
        # sum(h_i * h_j) <= 1 - mu for all lags
        # sum(h_i) = n_points / 2
        
        # CVXOPT expects the constraints in the form G * x <= h
        # where x is the vector [h_0, h_1, ..., h_n, mu]
        # So we build G and h for all the constraints and the sum constraint
        
        # Build all constraints into a single matrix
        G = np.zeros((n_points + 1, n_points + 1))
        h_vec = np.zeros(n_points + 1)
        
        # First, the constraints sum(h_i * h_j) + mu <= 1 for each lag
        # Each constraint corresponds to a row in G
        for i in range(n_points):
            for j in range(n_points):
                if 0 <= i - j < n_points:
                    G[i, j] += 1.0
                if 0 <= j - i < n_points:
                    G[i, j] += 1.0
        # Add mu column to each constraint
        for i in range(n_points):
            G[i, n_points] = 1.0
        
        # Second, add the sum constraint: sum(h) = n_points / 2
        # This becomes: sum(h_i) + mu = n_points / 2
        # So we have a constraint: sum(h_i) + mu = n_points / 2
        # This is a different form, so we need to add a separate row
        G[n_points, 0:n_points] = 1.0
        G[n_points, n_points] = 1.0
        h_vec[n_points] = n_points / 2
        
        # Now solve the QP: minimize 0.5 * x^T P x + q^T x
        # where P is a matrix of 0s, because the objective is to minimize mu
        # So the objective is to minimize 0.5 * x^T P x + q^T x
        # We have P = 0, but we have to consider mu in the constraints
        
        # The P matrix is 0 for the h part and 0 for mu
        # q is a column vector where q[n_points] = 1
        q = np.zeros(n_points + 1)
        q[n_points] = 1.0
        
        P = np.zeros((n_points + 1, n_points + 1))
        c = matrix(0.0, (n_points + 1, 1), tc='d')
        
        # Now solve the QP
        sol = solvers.qp(matrix(P), matrix(q), matrix(G), matrix(h_vec))
        if 'status' in sol and sol['status'] != 'optimal':
            return np.inf
        
        x = np.array(sol['x'])
        h_opt = x[:n_points]
        mu_opt = x[n_points]
        
        # Now compute the actual C5 value
        # Convert h_opt back to a step function and compute the overlap
        h_values = np.clip(h_opt, 0, 1)
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        correlate_vals = np.correlate(h_values, 1 - h_values, mode="full") * (2.0 / n_points)
        c5_bound = np.max(correlate_vals)
        
        return c5_bound
    
    # Try to solve the QP
    # We expect the solver to take a significant amount of time
    # If we run out of time, we fall back to a simpler optimization
    try:
        c5 = objective(0.0)
    except:
        c5 = np.inf
    
    # If QP fails or is too slow, fall back to differential evolution
    if np.isinf(c5) or (time.time() - start_time) > budget_s * 0.7:
        n_points = 45
        h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Objective function using differential evolution
        def objective_de(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(correlate_vals)
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(objective_de, bounds, popsize=20, mutation=(0.5, 1.0), 
                                           recombination=0.9, strategy='best1bin', tol=1e-5,
                                           maxiter=500, disp=False, polish=True, seed=seed)
        
        best_h = result.x
        best_c5 = result.fun
        
        # Final check
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            print("Time budget exceeded, returning initial solution")
            h = initial_h_values if 'initial_h_values' in globals() else h
            best_c5 = objective_de(h)
        
        # Final feasibility check
        h = np.clip(best_h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        final_c5 = objective_de(h)
        
        return (h.tolist(), final_c5, n_points)
    
    # Final check on the solution
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning initial solution")
        h = initial_h_values if 'initial_h_values' in globals() else h
        best_c5 = objective(0.0)
    
    # Final feasibility check
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(0.0)
    
    return (h.tolist(), final_c5, n_points)
