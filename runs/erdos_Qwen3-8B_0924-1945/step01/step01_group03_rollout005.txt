import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

        base_guess = np.copy(initial_h_values)
    else:
        base_guess = np.zeros(n_points)
        base_guess[::7] = 1.0
        base_guess = np.clip(base_guess, 0, 1)
        base_guess /= np.sum(base_guess)
        base_guess *= target_sum

    def generate_guesses(base_guess):
        guesses = []
        guesses.append(base_guess.copy())
        shift = int(n_points / 5)
        row_guess = np.zeros(n_points)
        row_guess[::shift] = 1.0
        row_guess = np.clip(row_guess, 0, 1)
        row_guess /= np.sum(row_guess)
        row_guess *= target_sum
        guesses.append(row_guess)
        perturbed_guess = base_guess.copy()
        perturbed_guess += np.random.uniform(-0.2, 0.2, size=n_points)
        perturbed_guess = np.clip(perturbed_guess, 0, 1)
        perturbed_guess /= np.sum(perturbed_guess)
        perturbed_guess *= target_sum
        guesses.append(perturbed_guess)
        reversed_guess = 1.0 - base_guess.copy()
        reversed_guess = np.clip(reversed_guess, 0, 1)
        reversed_guess /= np.sum(reversed_guess)
        reversed_guess *= target_sum
        guesses.append(reversed_guess)
        blended_guess = 0.5 * base_guess + 0.5 * row_guess
        blended_guess = np.clip(blended_guess, 0, 1)
        blended_guess /= np.sum(blended_guess)
        blended_guess *= target_sum
        guesses.append(blended_guess)
        return guesses

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    def objective(x):
        x = np.clip(x, 0, 1)
        x /= np.sum(x)
        x *= target_sum
        return evaluate_c5(x)

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    start_time = time.time()
    guesses = generate_guesses(base_guess)
    for guess in guesses:
        x0 = guess.copy()
        result = opt.minimize(
            objective,
            x0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-6, 'disp': False}
        )
        if result.success:
            current_h = result.x.copy()
            current_h = np.clip(current_h, 0, 1)
            current_h /= np.sum(current_h)
            current_h *= target_sum
            current_c5 = evaluate_c5(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        if time.time() - start_time > 500:
            break

    if best_c5 == float('inf'):
        best_h = base_guess.copy()
        best_h = np.clip(best_h, 0, 1)
        best_h /= np.sum(best_h)
        best_h *= target_sum
        best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, n_points)
