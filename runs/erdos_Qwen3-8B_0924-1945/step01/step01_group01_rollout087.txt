import numpy as np
import scipy.optimize as opt
import time
import cvxopt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 90
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    # Precompute lag indices
    max_lag = 2 * n_points - 2
    lag_indices = []
    for lag in range(1, max_lag + 1):
        indices = []
        for i in range(n_points):
            j = (i + lag) % n_points
            if j < n_points:
                indices.append((i, j))
        lag_indices.append(indices)
    
    # Objective function for QP
    def objective(mu):
        return mu
    
    # Constraints for QP
    A = []
    b = []
    for lag in range(1, max_lag + 1):
        indices = lag_indices[lag - 1]
        row = np.zeros(n_points)
        for i, j in indices:
            row[i] += 1
            row[j] += 1
        A.append(row)
        b.append(1 - mu)
    
    # QP setup
    P = np.zeros((n_points, n_points))
    for lag in range(1, max_lag + 1):
        indices = lag_indices[lag - 1]
        for i, j in indices:
            P[i, j] += 1
    
    # Convert to CVXOPT format
    P = cvxopt.matrix(P)
    q = cvxopt.matrix(np.zeros(n_points))
    G = cvxopt.matrix(-np.eye(n_points))
    h = cvxopt.matrix(np.zeros(n_points))
    A = cvxopt.matrix(np.vstack(A))
    b = cvxopt.matrix(np.array(b))
    
    # Solve QP
    try:
        sol = cvxopt.solvers.qp(P, q, G, h, A, b, options={'abstol': 1e-8, 'reltol': 1e-8, 'feastol': 1e-8, 'maxiter': 1000})
        if sol['status'] == 'optimal':
            h_qp = np.array(sol['x']).flatten()
            h_qp = np.clip(h_qp, 0, 1)
            h_qp = h_qp * (n_points / 2) / np.sum(h_qp)
            final_c5 = np.max(np.correlate(h_qp, 1 - h_qp, mode="full") * dx)
            return (h_qp.tolist(), final_c5, n_points)
    except:
        pass
    
    # Fallback to differential evolution
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.differential_evolution(objective, bounds, popsize=40, mutation=(0.5, 1.0), 
                                       recombination=0.9, strategy='best1bin', tol=1e-5,
                                       maxiter=200, disp=False, polish=True, seed=seed)
    
    best_h = result.x
    best_c5 = result.fun
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning initial solution")
        h = initial_h_values if 'initial_h_values' in globals() else h
        best_c5 = objective(h)
    
    # Final check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points)
