import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', None)
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        x = np.linspace(0, 2, n_points, endpoint=False)
        h = (1 + np.cos(np.pi * x / 2)) / 2
        h = h * (n_points / 2) / (np.sum(h) * 2.0 / n_points)

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    bounds = [(0, 1) for _ in range(n_points)]

    def objective(x):
        return evaluate_candidate(x)

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    def de_callback(x, convergence):
        nonlocal best_h, best_c5
        current_c5 = evaluate_candidate(x)
        if current_c5 < best_c5:
            best_h = x.copy()
            best_c5 = current_c5

    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        mutation=(0.8, 1.0),
        recombination=0.9,
        tol=1e-6,
        maxiter=100,
        disp=False,
        polish=True,
        callback=de_callback,
        workers=-1
    )

    if time() - start_time < budget_s:
        best_h = result.x
        best_c5 = evaluate_candidate(best_h)

    def slsqp_refine(x):
        def objective_func(x):
            return evaluate_candidate(x)
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        res = minimize(
            objective_func,
            x,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            tol=1e-6
        )
        return res.x

    for _ in range(5):
        if time() - start_time >= budget_s:
            break
        x0 = best_h.copy()
        refined_h = slsqp_refine(x0)
        current_c5 = evaluate_candidate(refined_h)
        if current_c5 < best_c5:
            best_h = refined_h
            best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    final_c5 = evaluate_candidate(best_h)
    return (best_h, final_c5, n_points)
