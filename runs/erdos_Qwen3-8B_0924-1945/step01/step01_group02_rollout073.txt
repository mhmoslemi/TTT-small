import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_lag = n_points - 1

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def build_constraints(h, dx, n_points, max_lag):
        n = n_points
        constraints = []
        for k in range(1, max_lag + 1):
            A_k = np.zeros((n, n))
            for i in range(n):
                j = i + k
                if j < n:
                    A_k[i, j] = dx
                j = i - k
                if j >= 0:
                    A_k[i, j] = dx
            constraints.append(A_k)
        return constraints

    def compute_c5(h, dx, n_points):
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def solve_qp(h, dx, n_points, max_lag, target_sum, max_restarts=20, max_iter=100, eps=1e-8, solver='ECOS'):
        constraints = build_constraints(h, dx, n_points, max_lag)
        best_c5 = float('inf')
        best_h = h.copy()
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = best_h * (target_sum / np.sum(best_h)) if np.sum(best_h) != 0 else best_h

        start_time = time.time()
        for restart in range(max_restarts):
            if time.time() - start_time > budget_s:
                break
            h_flat = h.copy()
            np.random.normal(0, 0.05, size=h_flat.shape, out=h_flat)
            h_flat = np.clip(h_flat, 0.0, 1.0)
            h_flat = h_flat * (target_sum / np.sum(h_flat)) if np.sum(h_flat) != 0 else h_flat

            c = cp.Variable()
            h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
            constraints = [cp.sum(h_var) == target_sum]
            for A_k in constraints:
                constraints.append(h_var.T @ A_k @ h_var >= 1.0 - c)
            objective = cp.Minimize(c)
            prob = cp.Problem(objective, constraints)

            try:
                result = prob.solve(solver=solver, eps=eps, max_iter=max_iter)
                if result is not None:
                    new_h = h_var.value
                    new_h = np.clip(new_h, 0.0, 1.0)
                    new_h = new_h * (target_sum / np.sum(new_h)) if np.sum(new_h) != 0 else new_h
                    new_c5 = compute_c5(new_h, dx, n_points)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = new_h
            except:
                pass

        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = best_h * (target_sum / np.sum(best_h)) if np.sum(best_h) != 0 else best_h
        return best_h, best_c5

    h_opt, c5_bound = solve_qp(h, dx, n_points, max_lag, target_sum, max_restarts=20, max_iter=100, eps=1e-8)
    return h_opt.tolist(), c5_bound, n_points
