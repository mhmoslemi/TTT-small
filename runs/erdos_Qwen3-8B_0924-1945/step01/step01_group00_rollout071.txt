import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if np.random.rand() < 0.5 else 0
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    n_iter = 0
    max_iter = 100

    while n_iter < max_iter and time() - start_time < budget_s:
        n_iter += 1
        # Perturb the best solution
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        candidate = best_h + perturbation
        candidate = np.clip(candidate, 0, 1)
        candidate = candidate / np.sum(candidate) * n_points / 2
        current_c5 = evaluate_candidate(candidate)
        if current_c5 < best_c5:
            best_h = candidate.copy()
            best_c5 = current_c5

    for _ in range(100):
        if time() - start_time > budget_s:
            break
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
        current_c5 = evaluate_candidate(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
