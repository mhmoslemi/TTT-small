import numpy as np
import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Function to evaluate the overlap integral
    def evaluate_c5(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Function to compute gradient (only for dominant k)
    def compute_gradient(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        k_max = np.argmax(correlate_vals)
        gradient = np.zeros_like(h)
        for i in range(n_points):
            if i + k_max < n_points:
                gradient[i] = (1 - h[i + k_max]) * (2.0 / n_points)
        return gradient
    
    # Coarse grid optimization with differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_de = opt.differential_evolution(
        evaluate_c5,
        bounds,
        popsize=50,
        mutation=(0.8, 1.0),
        recombination=0.9,
        strategy='best1bin',
        tol=1e-5,
        maxiter=200,
        disp=False,
        polish=True,
        seed=seed
    )
    best_h = result_de.x
    best_c5 = result_de.fun
    
    # Refinement with L-BFGS-B
    result_lbfgs = opt.minimize(
        evaluate_c5,
        best_h,
        method='L-BFGS-B',
        bounds=bounds,
        jac=compute_gradient,
        options={'maxiter': 300, 'ftol': 1e-6, 'gtol': 1e-6}
    )
    best_h = result_lbfgs.x
    best_c5 = result_lbfgs.fun
    
    # Multi-scale refinement
    n_points_refine = 90
    dx = 2.0 / n_points_refine
    h_refine = np.zeros(n_points_refine)
    for i in range(n_points):
        h_refine[i * n_points_refine // n_points] += best_h[i]
    h_refine = h_refine * (n_points_refine / 2) / np.sum(h_refine)
    h_refine = np.clip(h_refine, 0, 1)
    
    def evaluate_c5_refine(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points_refine / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    result_de_refine = opt.differential_evolution(
        evaluate_c5_refine,
        [(0.0, 1.0) for _ in range(n_points_refine)],
        popsize=30,
        mutation=(0.9, 1.0),
        recombination=0.7,
        strategy='best1bin',
        tol=1e-5,
        maxiter=100,
        disp=False,
        polish=True,
        seed=seed
    )
    best_h_refine = result_de_refine.x
    best_c5_refine = result_de_refine.fun
    
    # Final check and selection
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
        best_h = best_h if evaluate_c5(best_h) < evaluate_c5_refine(best_h_refine) else best_h_refine
        best_c5 = evaluate_c5(best_h)
    else:
        best_h = best_h_refine
        best_c5 = best_c5_refine
    
    # Final feasibility checks
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points_refine / 2) / np.sum(best_h)
    final_c5 = evaluate_c5_refine(best_h)
    
    return (best_h.tolist(), final_c5, n_points_refine)
