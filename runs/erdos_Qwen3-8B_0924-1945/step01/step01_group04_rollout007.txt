import numpy as np
import scipy.optimize as optimize
import numpy.random as npr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 1.0 / n_points)
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
    
    def objective(h_):
        h_ = np.clip(h_, 0.0, 1.0)
        h_ = h_ * (target_sum / np.sum(h_)) if np.sum(h_) != 0 else h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    def constraint(h_):
        return np.sum(h_) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    def optimize_once(h_start):
        result = optimize.minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        h_opt = result.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        h_opt = h_opt * (target_sum / np.sum(h_opt)) if np.sum(h_opt) != 0 else h_opt
        return h_opt, objective(h_opt)
    
    best_h, best_c5 = h, objective(h)
    for _ in range(30):
        perturbation = np.random.normal(0, 0.05, n_points)
        h_pert = h + perturbation
        h_pert = np.clip(h_pert, 0.0, 1.0)
        h_pert = h_pert * (target_sum / np.sum(h_pert)) if np.sum(h_pert) != 0 else h_pert
        h_opt, c5 = optimize_once(h_pert)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5
            if best_c5 < 0.3808:
                break
    
    return (best_h.tolist(), best_c5, n_points)
