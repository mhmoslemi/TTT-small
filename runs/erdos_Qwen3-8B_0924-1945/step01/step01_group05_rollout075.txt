import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        idx = np.round(np.linspace(0, n_points - 1, n_points // 3 + 1))
        h[idx.astype(int)] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    
    best_h = h
    best_c5 = objective(best_h)
    
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    for gen in range(generations):
        population = [np.random.rand(n_points) for _ in range(population_size)]
        for i in range(population_size):
            population[i] = np.clip(population[i], 0, 1)
            population[i] = population[i] * target_sum / np.sum(population[i])
        
        new_population = []
        for i in range(population_size // 2):
            p1, p2 = population[i], population[population_size - 1 - i]
            if np.random.rand() < crossover_rate:
                c1 = np.concatenate([p1[:n_points//2], p2[n_points//2:]])
                c2 = np.concatenate([p2[:n_points//2], p1[n_points//2:]])
            else:
                c1, c2 = p1, p2
            for c in [c1, c2]:
                c = np.clip(c, 0, 1)
                c = c * target_sum / np.sum(c)
                new_population.append(c)
        
        for i in range(population_size // 2):
            p = population[i]
            if np.random.rand() < mutation_rate:
                idx = np.random.choice(n_points, size=5, replace=False)
                p[idx] = np.random.rand(5)
            p = np.clip(p, 0, 1)
            p = p * target_sum / np.sum(p)
            new_population.append(p)
        
        new_population = [np.clip(p, 0, 1) for p in new_population]
        new_population = [p * target_sum / np.sum(p) for p in new_population]
        
        fitness = [objective(p) for p in new_population]
        indices = np.argsort(fitness)
        new_population = [new_population[i] for i in indices[:population_size]]
        
        if np.min(fitness) < best_c5:
            best_h = new_population[0]
            best_c5 = np.min(fitness)
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
