        h = h * (n_points / 2) / np.sum(h)
    
    # Objective function
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Optimization using CVXOPT with quadratic programming
    def qp_objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Setup QP constraints
    n_vars = n_points
    A = np.zeros((n_points, n_vars))
    b = np.zeros(n_points)
    
    for i in range(n_points):
        for j in range(n_points):
            if i + j < n_points:
                A[i, j] = 1
            elif i + j >= n_points:
                A[i, j] = 1
            else:
                A[i, j] = 1
    
    # Validate the constraints for all lag values
    dx = 2.0 / n_points
    Q = np.zeros((n_vars, n_vars))

            Q[i, j] = dx ** 2 * 1.0  # Quadratic term coefficient
    
    # Mu is the objective to minimize
    # Formulate QP: minimize mu s.t. h^T Q h + C <= mu
    # For each lag, we have constraints h^T Q_h h <= mu, where Q_h is a matrix with 1s at positions that satisfy lag
