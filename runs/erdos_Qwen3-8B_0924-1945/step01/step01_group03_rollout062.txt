import scipy.optimize as opt

    np.random.seed(seed)
    
    # Initial guess based on parent construction
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    h = np.clip(h, 0, 1)
    h /= np.sum(h)
    h *= target_sum

        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(x):
        return np.sum(x) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Differential evolution setup

        maxiter=max_gen,
        tol=1e-6,

        recombination=0.8,
        disp=False,
        polish=True,
        vectorized=True,
        x0=h
    )

    # Check time and finalize
    elapsed = time.time() - start_time
    if elapsed > time_limit:

        return (best_h, c5_bound, n_points)
