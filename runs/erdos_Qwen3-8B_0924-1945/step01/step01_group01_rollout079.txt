import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 90
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Objective function
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Gradient-based optimization (L-BFGS-B)
    def objective_gradient(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_k = np.argmax(correlate_vals)
        if max_k >= len(h):
            return np.zeros_like(h)
        k = max_k - len(h) + 1
        if k < 0 or k >= len(h):
            return np.zeros_like(h)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            x = i * (2.0 / n_points)
            xk = (k + i) * (2.0 / n_points)
            if 0 <= xk < 2:
                grad[i] = (1 - h[int(xk / (2.0 / n_points))]) * (2.0 / n_points)
        return grad
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(objective, h, method='L-BFGS-B', jac=objective_gradient,
                          bounds=bounds, tol=1e-5, options={'maxiter': 500})
    
    best_h = result.x
    best_c5 = result.fun
    
    # Adaptive grid refinement
    n_points_refined = 120
    h_refined = np.zeros(n_points_refined)
    h_refined[:n_points] = best_h
    h_refined = np.random.permutation(h_refined)
    h_refined = np.clip(h_refined, 0, 1)
    h_refined = h_refined * (n_points_refined / 2) / np.sum(h_refined)
    
    def objective_refined(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points_refined / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points_refined)
        return np.max(correlate_vals)
    
    def objective_gradient_refined(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points_refined / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points_refined)
        max_k = np.argmax(correlate_vals)
        if max_k >= len(h):
            return np.zeros_like(h)
        k = max_k - len(h) + 1
        if k < 0 or k >= len(h):
            return np.zeros_like(h)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            x = i * (2.0 / n_points_refined)
            xk = (k + i) * (2.0 / n_points_refined)
            if 0 <= xk < 2:
                grad[i] = (1 - h[int(xk / (2.0 / n_points_refined))]) * (2.0 / n_points_refined)
        return grad
    
    result_refined = opt.minimize(objective_refined, h_refined, method='L-BFGS-B',
                                 jac=objective_gradient_refined, bounds=bounds,
                                 tol=1e-5, options={'maxiter': 500})
    
    best_h_refined = result_refined.x
    best_c5_refined = result_refined.fun
    
    # Final evaluation
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning refined solution")
        h = best_h_refined
        best_c5 = best_c5_refined
    else:
        h = best_h_refined
        best_c5 = best_c5_refined
    
    # Final feasibility check
    h = np.clip(h, 0, 1)
    h = h * (n_points_refined / 2) / np.sum(h)
    final_c5 = objective_refined(h)
    
    return (h.tolist(), final_c5, n_points_refined)
