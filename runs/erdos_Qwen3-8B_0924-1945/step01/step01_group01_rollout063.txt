import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import linprog, quadratic_program

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Default configuration
    n_points = 90
    dx = 2.0 / n_points
    
    # Load initial values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    # Function to build QP problem
    def build_qp_problem(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        n = n_points
        dx = 2.0 / n
        n_lags = 2 * n - 1
        G = np.zeros((n_lags, n))
        h_i = h
        for s in range(n_lags):
            if s < n:
                i = 0
                j = s
            else:
                i = s - n
                j = n - 1
            # Since we are using the correlate method, we only need to enforce constraints for s in [0, n)
            # For s in [n, 2n-1), the overlap is symmetric
            if s < n:
                for i in range(n):
                    j = (i + s) % n
                    G[s, i] = h[i] * h[j]
        # Use the maximum of the quadratic terms as the objective
        # We aim to minimize the maximum of these terms
        # Construct the QP problem to minimize μ subject to G*s <= 1 - μ
        # Convert to standard form: minimize c^T x subject to Gx <= b
        # Here, x = h, c = 0, G = G, and b = 1 - μ
        # We fix μ as a variable and optimize it along with h
        # This is a convex quadratic program: we minimize μ subject to G*h <= 1 - μ
        # Rewrite as μ + G*h <= 1
        # This becomes a convex QP with variables (h, μ)
        # Objective: minimize μ
        # Constraints: μ + G*h <= 1
        # Additionally, the constraints h >= 0, h <= 1, and sum(h) = n/2
        # Formulate as standard QP with H=0, c=[1, 0, ... 0], A = [G, 1], b = 1
        # We introduce a new variable μ and solve for μ
        # Total variables: n + 1
        # We need to construct A and b accordingly
        # Formulate the QP problem
        n_vars = n + 1
        H = np.zeros((n_vars, n_vars))
        c = np.zeros(n_vars)
        c[0] = 1.0  # Minimize μ
        A = np.zeros((n_lags, n_vars))
        for i in range(n_lags):
            for j in range(n):
                A[i, j] = G[i, j]
            A[i, n] = 1.0
        b = np.ones(n_lags)
        # Define the QP problem
        qp = quadratic_program(c=c, G=H, A=A, b=b, bounds=(0, 1))
        return qp
    
    # Try solving the QP problem
    try:
        if 'initial_h_values' in globals():
            h_candidate = np.array(initial_h_values, dtype=np.float64)
        else:
            h_candidate = np.random.rand(n_points)
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Build the QP and solve
        qp = build_qp_problem(h_candidate)
        result = qp.solve()
        mu = result.x[0]
        h_candidate = result.x[1:n_points+1]
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Evaluate the C5 bound
        final_c5 = compute_objective(h_candidate)
        best_h = h_candidate
        best_c5 = final_c5

    except:
        # Fallback to differential evolution if QP fails
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(compute_objective, bounds, popsize=40, mutation=(0.5, 1.0), 
                                           recombination=0.9, strategy='best1bin', tol=1e-5,
                                           maxiter=200, disp=False, polish=True, seed=seed)
        best_h = result.x
        best_c5 = result.fun
    
    # Final checks
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best found solution")
    
    # Final feasibility check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = compute_objective(h)
    
    return (h.tolist(), final_c5, n_points)
