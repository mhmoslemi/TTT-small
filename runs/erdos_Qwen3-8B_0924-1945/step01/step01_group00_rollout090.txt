import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy() if 'initial_h_values' in kwargs else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x / np.sum(x) * n_points / 2
        return x

    def fitness(x):
        return evaluate_candidate(x)

    start_time = time()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    bounds = [(0, 1) for _ in range(n_points)]

    def de_objective(x):
        return evaluate_candidate(x)

    def de_bounds():
        return [(0, 1) for _ in range(n_points)]

    def de_fitness(x):
        return evaluate_candidate(x)

    def de_constraint(x):
        return np.sum(x) - n_points / 2

    def de_objective_norm(x):
        x = normalize(x)
        return evaluate_candidate(x)

    def de_bounds_norm():
        return [(0, 1) for _ in range(n_points)]

    def de_fitness_norm(x):
        return evaluate_candidate(x)

    def de_objective_norm(x):
        x = normalize(x)
        return evaluate_candidate(x)

    def de_fitness_norm(x):
        return evaluate_candidate(x)

    def de_constraint_norm(x):
        return np.sum(x) - n_points / 2

    if time() - start_time < budget_s:
        result = differential_evolution(de_objective_norm, de_bounds_norm, strategy='best1bin', popsize=20, tol=1e-5, mutation=(0.5, 1), recombination=0.9, maxiter=100)
        best_h = result.x
        best_c5 = evaluate_candidate(best_h)

    if time() - start_time < budget_s:
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, tol=1e-5)
        if res.success:
            best_h = res.x
            best_c5 = evaluate_candidate(best_h)

    best_h = normalize(best_h)
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
