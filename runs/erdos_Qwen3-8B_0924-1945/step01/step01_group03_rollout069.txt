import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h_initial = np.copy(initial_h_values)
    else:
        x = np.linspace(0, 2, n_points)
        h_initial = 0.5 + 0.5 * np.sin(np.pi * x)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * target_sum / np.sum(h_initial)

    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def objective(x):
        return evaluate_c5(x)

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def optimize_guess(guess):
        x0 = guess.copy()
        x0 = np.clip(x0, 0, 1)
        x0 = x0 * target_sum / np.sum(x0)
        result = opt.minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-6}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        c5 = evaluate_c5(best_h)
        return best_h, c5

    def generate_guesses():
        base = np.linspace(0, 2, n_points)
        base_vals = 0.5 + 0.5 * np.sin(np.pi * base)
        base_vals = np.clip(base_vals, 0, 1)
        base_vals = base_vals * target_sum / np.sum(base_vals)
        guesses = [base_vals]
        noise = 0.15 * np.random.rand(n_points)
        for i in range(10):
            guess = base_vals + (np.random.rand(n_points) - 0.5) * noise
            guess = np.clip(guess, 0, 1)
            guess = guess * target_sum / np.sum(guess)
            guesses.append(guess)
        return guesses

    guesses = generate_guesses()
    best_h, best_c5 = guesses[0], evaluate_c5(guesses[0])
    for guess in guesses:
        h, c5 = optimize_guess(guess)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    return (best_h, best_c5, n_points)
