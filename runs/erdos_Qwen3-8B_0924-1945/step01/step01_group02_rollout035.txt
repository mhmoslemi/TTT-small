```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_restarts = 20
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    elapsed_time = 0

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                h[i] = 1.0
        h = h * (target_sum / np.sum(h))

    def compute_Ak(k, n_points):
        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = (i + k) % n_points
            A[i, j] = dx
            j = (i - k) % n_points
            A[i, j] = dx
        return A

    for restart in range(max_restarts):
        if elapsed_time >= budget_s:
            break
        h_flat = h.copy()
        np.random.seed(seed + restart)
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h_flat += perturbation
        h_flat = np.clip(h_flat, 0.0, 1.0)
        h_flat = h_flat * (target_sum / np.sum(h_flat))

        A_constraints = [compute_Ak(k, n_points) for k in range(1, n_points)]

        variables = [cp.Variable(n_points), cp.Variable()]
        h, C = variables

        constraints = [
            cp.sum(h) == target_sum,
            0 <= h <= 1.0,
            cp.sum(cp.multiply(h, h)) >= 1.0 - C
        ]

        for A in A_constraints:
            constraints.append(cp.quad_form(h, A) >= 1.0 - C)

        prob = cp.Problem(cp.Minimize(C), constraints)
        try:
            start = time.time()
            prob.solve(solver=cp.ECOS, eps=1e-8)
            elapsed_time += time.time() - start
            if prob.status == 'optimal':
                current_h = h.value
                current_c5 = np.max(np.correlate(current_h, 1.0 - current_h, mode="full")) * dx
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
        except:
            pass

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (target_sum / np.sum(best_h))
    final_c5 = np.max(np.correlate(best_h, 1.0 - best_h, mode="full")) * dx
    return best_h.tolist(), final_c5, n_points
```
