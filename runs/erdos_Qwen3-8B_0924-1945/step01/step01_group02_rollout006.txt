```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = np.zeros(n_points, dtype=np.float64)
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = (h < 0.5).astype(np.float64)
        h = h * (target_sum / np.sum(h))
    
    def build_overlap_matrix(k):
        mat = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = (i + k) % n_points
            mat[i, j] = dx
            j = (i - k) % n_points
            mat[i, j] = dx
        return mat
    
    A = []
    for k in range(1, n_points):
        A.append(build_overlap_matrix(k))
    A = np.vstack(A)
    
    h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
    C = cp.Variable()
    prob = cp.Problem(cp.Minimize(C),
                      [
                          cp.sum(h_var * dx) == 1,
                          cp.sum(cp.multiply(A, h_var * h_var.T)) >= (n_points - 1) * (1 - C),
                          C >= 0
                      ])
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = 0.5
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        try:
            result = prob.solve(solver=cp.ECOS, eps=1e-6, verbose=False)
            if result is not None:
                h_opt = h_var.value
                if h_opt is not None:
                    h_opt = np.clip(h_opt, 0.0, 1.0)
                    sum_h = np.sum(h_opt) * dx
                    scale = target_sum / sum_h if sum_h != 0 else 1.0
                    h_opt = h_opt * scale
                    corr = np.correlate(h_opt, 1.0 - h_opt, mode="full")
                    c5 = np.max(corr * dx)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_opt.copy()
        except:
            pass
    
    h_final = best_h
    sum_h = np.sum(h_final) * dx
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h_final = h_final * scale
    corr = np.correlate(h_final, 1.0 - h_final, mode="full")
    c5_final = np.max(corr * dx)
    return h_final.tolist(), c5_final, n_points
```
