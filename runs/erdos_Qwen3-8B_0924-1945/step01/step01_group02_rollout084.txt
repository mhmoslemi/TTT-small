```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.where(np.random.rand(n_points) < 0.5, 1.0, 0.0)
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def build_qp(h_init):
        n = n_points
        variables = cp.Variable(n)
        C = cp.Variable(nonneg=True)
        constraints = []
        constraints.append(cp.sum(variables) == target_sum)
        constraints.append(cp.SOC(1, variables))
        for k in range(1, n):
            A = np.zeros((n, n))
            for i in range(n):
                j = (i + k) % n
                A[i, j] = dx
                j = (i - k) % n
                A[i, j] = dx
            constraints.append(cp.quad_form(variables, A) >= 1 - C)
        obj = cp.Minimize(C)
        prob = cp.Problem(obj, constraints)
        result = prob.solve(solver=cp.ECOS, eps=1e-8)
        if result is not None and result.status in [cp.OPTIMAL, cp.OPTIMAL_INFEASIBLE]:
            return variables.value, C.value
        return h_init, np.inf

    best_h = h.copy()
    best_c5 = evaluate(h)
    start_time = time.time()

    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        h_init = np.copy(h)
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0.0, 1.0)
        h_init = h_init * (target_sum / np.sum(h_init))
        h_opt, c5 = build_qp(h_init)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = evaluate(h_opt)
        if c5_bound < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_bound

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(10):
            if time.time() - start_time > budget_s:
                break
            h_init = np.copy(best_h)
            h_init += np.random.normal(0, 0.02, n_points)
            h_init = np.clip(h_init, 0.0, 1.0)
            h_init = h_init * (target_sum / np.sum(h_init))
            h_opt, c5 = build_qp(h_init)
            h_opt = np.clip(h_opt, 0.0, 1.0)
            sum_h_opt = np.sum(h_opt)
            scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
            h_opt = h_opt * scale
            c5_bound = evaluate(h_opt)
            if c5_bound < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_bound

    return best_h.tolist(), best_c5, n_points
```
