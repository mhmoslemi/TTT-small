import scipy.optimize as optimize
import numpy.random as npr

def run(seed=42, budget_s=600, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    np.random.seed(seed)
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
        h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
    
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
    
    def objective(h_):
        h_ = np.clip(h_, 0.0, 1.0)
        h_ = h_ * (target_sum / np.sum(h_)) if np.sum(h_) != 0 else h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    def constraint(h_):
        return np.sum(h_) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    def optimize_with_restart(h0):
        result = optimize.minimize(
            objective,
            h0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-6, 'disp': False}
        )
        if result.success:
            h_opt = np.clip(result.x, 0.0, 1.0)
            h_opt = h_opt * (target_sum / np.sum(h_opt)) if np.sum(h_opt) != 0 else h_opt
            return h_opt, objective(h_opt)
        return h0, objective(h0)
    
    best_h, best_c5 = h, objective(h)
    for _ in range(5):
        if 'initial_h_values' in globals():
            h0 = initial_h_values.copy()
        else:
            h0 = np.zeros(n_points)

        if c5 < best_c5:
