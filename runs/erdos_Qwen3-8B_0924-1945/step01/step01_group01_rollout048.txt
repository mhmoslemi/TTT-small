import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def gradient_refinement(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(correlate_vals)
        k = max_idx - n_points + 1
        if k < 0:
            k += n_points
        h_grad = np.zeros_like(h)
        for i in range(n_points):
            x = i * (2.0 / n_points)
            x_shift = x + k * (2.0 / n_points)
            if 0 <= x_shift <= 2:
                h_grad[i] = (1 - h[np.floor(x_shift * n_points / 2).astype(int)]) * (2.0 / n_points)
        return h_grad
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Gradient-based refinement
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, jac=gradient_refinement, tol=1e-6, options={'maxiter': 100})
    
    best_h = result.x
    best_c5 = result.fun
    
    # Check time
    elapsed = time.time() - start_time
    if elapsed > budget_s - 300:
        h = initial_h_values if 'initial_h_values' in globals() else h
        best_c5 = objective(h)
        return (h.tolist(), best_c5, n_points)
    
    # Multi-scale refinement
    n_points = 90
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, 45), best_h)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, jac=gradient_refinement, tol=1e-6, options={'maxiter': 100})
    
    best_h = result.x
    best_c5 = result.fun
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 200:
        h = best_h
        best_c5 = objective(h)
        return (h.tolist(), best_c5, n_points)
    
    # Fallback to differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.differential_evolution(objective, bounds, popsize=30, mutation=(0.9, 1.0), recombination=0.7, strategy='best1bin', tol=1e-5, maxiter=200, disp=False)
    
    best_h = result.x
    best_c5 = result.fun
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        h = best_h
        best_c5 = objective(h)
        return (h.tolist(), best_c5, n_points)
    
    # Final check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points)
