import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 90
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Objective function and gradient
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def gradient(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_idx = np.argmax(correlate_vals)
        k = max_idx - n_points + 1
        if k < 0 or k >= n_points:
            return np.zeros_like(h)
        # Compute derivative with respect to h
        grad = np.zeros_like(h)
        for i in range(n_points):
            x = i * (2.0 / n_points)
            x_k = x + k * (2.0 / n_points)
            if x_k >= 2.0:
                x_k -= 2.0
            if x_k < 0:
                x_k += 2.0
            if 0 <= x_k < 2.0:
                grad[i] = (1 - h[int(x_k * n_points / 2.0)]) * (2.0 / n_points)
        return grad
    
    # Optimization using L-BFGS-B
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, jac=gradient, tol=1e-5, options={'maxiter': 500})
    best_h = result.x
    best_c5 = result.fun
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 200:
        h = np.clip(best_h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        best_c5 = objective(h)
        return (h.tolist(), best_c5, n_points)
    
    # Adaptive grid refinement
    n_points = 120
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, jac=gradient, tol=1e-5, options={'maxiter': 300})
    best_h = result.x
    best_c5 = result.fun
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 100:
        h = np.clip(best_h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        best_c5 = objective(h)
        return (h.tolist(), best_c5, n_points)
    
    # Fallback to DE if convergence is stalled
    result = opt.differential_evolution(objective, bounds, strategy='best1bin', popsize=30, mutation=(0.9, 1.0), recombination=0.7,
                                        tol=1e-5, maxiter=200, seed=seed)
    best_h = result.x
    best_c5 = result.fun
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        h = np.clip(best_h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        best_c5 = objective(h)
    
    # Final check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points)
