```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    h = np.clip(h, 0.0, 1.0)

    def generate_lag_constraints(n_points):
        n_lags = n_points - 1
        constraints = []
        for k in range(1, n_points):
            A = np.zeros((n_points, n_points))
            for i in range(n_points):
                if i + k < n_points:
                    A[i, i + k] = dx
                if i - k >= 0:
                    A[i, i - k] = dx
            constraints.append(A)
        return constraints

    lag_constraints = generate_lag_constraints(n_points)

    h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
    C = cp.Variable()

    constraints = [
        cp.sum(h_var) == target_sum,
        h_var >= 0,
        h_var <= 1
    ]

    for k in range(len(lag_constraints)):
        A = lag_constraints[k]
        constraint = cp.quad_form(h_var, A) >= 1 - C
        constraints.append(constraint)

    prob = cp.Problem(cp.Minimize(C), constraints)
    start_time = time.time()
    while time.time() - start_time < budget_s:
        try:
            prob.solve(solver=cp.SCS, eps=1e-5, max_iter=100, verbose=False)
            if prob.status == 'OPTIMAL':
                h_opt = h_var.value
                c5_bound = 1.0 - C.value
                h_opt = np.clip(h_opt, 0.0, 1.0)
                sum_h_opt = np.sum(h_opt)
                scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
                h_opt = h_opt * scale
                return h_opt.tolist(), c5_bound, n_points
        except:
            pass
    h_opt = h_var.value
    c5_bound = 1.0 - C.value
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    return h_opt.tolist(), c5_bound, n_points
```
