import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial solution (if available, otherwise create a base pattern)
    n_points = 200
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Base pattern: evenly spaced 1s with spacing
        spacing = n_points // 3
        base_h = np.zeros(n_points)
        for i in range(n_points):
            if i % spacing == 0:
                base_h[i] = 1
        # Refine base_h with linear interpolation
        base_h = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, n_points//spacing), base_h)
        h = base_h
    
    # Normalize to ensure sum equals n_points / 2
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    # Genetic algorithm parameters
    pop_size = 50
    n_generations = 80
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = np.random.rand(pop_size, n_points)
    population = np.clip(population, 0, 1)
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    for gen in range(n_generations):
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        # Select top 30% for next generation
        elite_size = int(0.3 * pop_size)
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = population[elite_indices]
        # Generate offspring
        offspring = []
        for _ in range(pop_size - elite_size):
            # Crossover
            parent1, parent2 = np.random.choice(elite_size, 2, replace=False)
            child = np.copy(elite[parent1])
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points - 1)
                child[:split] = elite[parent2][:split]
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation = np.random.uniform(-0.1, 0.1, n_points)
                child += mutation
                child = np.clip(child, 0, 1)
                child = child * target_sum / np.sum(child)
            offspring.append(child)
        # Update population
        population = np.vstack((elite, offspring))
        # Update best solution
        current_best = population[np.argmin(fitness)]
        current_c5 = objective(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
        # Check time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        # Adjust mutation rate based on time
        if elapsed < 200:
            mutation_rate += 0.05
        elif elapsed < 500:
            mutation_rate += 0.02
    
    # Final checks
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
