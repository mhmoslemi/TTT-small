import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def objective(x):
        x = np.clip(x, 0, 1)
        x = x / np.sum(x) * n_points / 2
        return compute_c5(x)

    def bounds_func(x):
        return np.clip(x, 0, 1)

    def evaluate_candidate(x):
        x = bounds_func(x)
        x = x / np.sum(x) * n_points / 2
        return compute_c5(x)

    def fitness(x):
        return evaluate_candidate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    def de_objective(x):
        return fitness(x)

    def de_bounds_func(x):
        return np.clip(x, 0, 1)

    def de_evaluate_candidate(x):
        x = de_bounds_func(x)
        x = x / np.sum(x) * n_points / 2
        return compute_c5(x)

    def de_fitness(x):
        return de_evaluate_candidate(x)

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    # Differential Evolution
    bounds = [(0, 1) for _ in range(n_points)]
    result = differential_evolution(de_fitness, bounds, strategy='best1bin',
                                   popsize=20, tol=1e-5, mutation=(0.5, 1), recombination=0.9,
                                   maxiter=50, disp=False, polish=False, callback=None,
                                   options={'mutation': (0.5, 1), 'recombination': 0.9})

    if result.success:
        best_h = result.x
        best_c5 = evaluate_candidate(best_h)

    # Local refinement with SLSQP
    def objective_lsqp(x):
        x = np.clip(x, 0, 1)
        x = x / np.sum(x) * n_points / 2
        return compute_c5(x)

    cons_lsqp = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    bounds_lsqp = [(0, 1) for _ in range(n_points)]

    for _ in range(15):
        if time() - start_time > budget_s:
            break
        x0 = best_h.copy()
        res = minimize(objective_lsqp, x0, method='SLSQP', bounds=bounds_lsqp, constraints=cons_lsqp,
                       tol=1e-5, options={'maxiter': 50})
        if res.success:
            current_h = res.x
            current_c5 = evaluate_candidate(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        else:
            break

    # Random restarts if needed
    for _ in range(10):
        if time() - start_time > budget_s:
            break
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
        current_c5 = evaluate_candidate(h)
        if current_c5 < best_c5:
            best_h = h
            best_c5 = current_c5

    final_h = np.clip(best_h, 0, 1)
    final_h = final_h / np.sum(final_h) * n_points / 2
    final_c5 = evaluate_candidate(final_h)
    return (final_h, final_c5, n_points)
