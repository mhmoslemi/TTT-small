import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.copy(initial_h_values)
        h = np.clip(h, 0, 1)
        h /= np.sum(h)
        h *= target_sum
    else:
        h = np.zeros(n_points)
        h[::5] = 1.0
        h = np.clip(h, 0, 1)
        h /= np.sum(h)
        h *= target_sum

    def objective(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    def evaluate_with_timeout(f, args, timeout):
        import signal
        signal.signal(signal.SIGALRM, lambda *a: None)
        signal.alarm(timeout)
        try:
            result = f(*args)
            signal.alarm(0)
            return result
        except:
            signal.alarm(0)
            return None

    result = None
    best_h = h.copy()
    best_c5 = objective(best_h)
    elapsed = 0

    for gen in range(200):
        bounds = [(0, 1) for _ in range(n_points)]
        res = evaluate_with_timeout(opt.differential_evolution, (objective, bounds, {'constraints': {'type': 'eq', 'fun': constraint}, 'popsize': 15, 'mutation': (0.5, 1), 'recombination': 'blend'},), budget_s - elapsed)
        elapsed += 10
        if res is None:
            break
        if res.success:
            candidate = res.x
            candidate = np.clip(candidate, 0, 1)
            candidate /= np.sum(candidate)
            candidate *= target_sum
            c5 = objective(candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate.copy()
        if elapsed >= budget_s:
            break

    best_h = np.clip(best_h, 0, 1)
    best_h /= np.sum(best_h)
    best_h *= target_sum
    c5_bound = objective(best_h)

    return (best_h, c5_bound, n_points)
