import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.copy(initial_h_values)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 5 == 0:
                h[i] = 1
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

    def objective(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.differential_evolution(
        objective,
        bounds,
        args=(),
        strategy='best1bin',
        popsize=15,
        maxiter=200,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        disp=False
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    c5_bound = objective(best_h)

    return (best_h, c5_bound, n_points)
