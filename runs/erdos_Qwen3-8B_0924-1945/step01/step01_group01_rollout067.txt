import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define initial points
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Coarse grid optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.differential_evolution(objective, bounds, popsize=40, mutation=(0.8, 1.0),
                                       recombination=0.9, strategy='best1bin', tol=1e-5,
                                       maxiter=300, disp=False, polish=True, seed=seed)
    best_h = result.x
    best_c5 = result.fun
    
    # Refine on finer grid
    n_points_new = 90
    h_new = np.interp(np.linspace(0, 2, n_points_new), np.linspace(0, 2, n_points), best_h)
    h_new = np.clip(h_new, 0, 1)
    h_new = h_new * (n_points_new / 2) / np.sum(h_new)
    
    def objective_fine(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points_new / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points_new)
        return np.max(correlate_vals)
    
    result_fine = opt.differential_evolution(objective_fine, bounds, popsize=20, mutation=(1.0, 1.2),
                                            recombination=0.9, strategy='best1bin', tol=1e-6,
                                            maxiter=200, disp=False, polish=True, seed=seed)
    best_h_fine = result_fine.x
    best_c5_fine = result_fine.fun
    
    # Local refinement with L-BFGS-B
    def objective_local(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points_new / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points_new)
        return np.max(correlate_vals)
    
    bounds_local = [(0.0, 1.0) for _ in range(n_points_new)]
    result_local = opt.minimize(objective_local, best_h_fine, method='L-BFGS-B', bounds=bounds_local,
                               tol=1e-8, options={'maxiter': 100, 'disp': False})
    best_h_local = result_local.x
    best_c5_local = result_local.fun
    
    # Final check and return
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning fine stage solution")
        h = best_h_fine
        c5 = best_c5_fine
    else:
        h = best_h_local
        c5 = best_c5_local
    
    h = np.clip(h, 0, 1)
    h = h * (n_points_new / 2) / np.sum(h)
    final_c5 = objective_local(h)
    
    return (h.tolist(), final_c5, n_points_new)
