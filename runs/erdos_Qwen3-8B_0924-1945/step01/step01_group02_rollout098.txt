```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h = h * scale

    def compute_c5(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    num_restarts = 20
    best_h = h.copy()
    best_c5 = compute_c5(h)

    start_time = time.time()

    for restart in range(num_restarts):
        if time.time() - start_time > budget_s:
            break
        h = np.clip(h + np.random.uniform(-0.05, 0.05, size=n_points), 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale

        h_var = cp.Variable(n_points)
        C = cp.Variable()

        constraints = []
        constraints.append(cp.sum(h_var) == target_sum)
        constraints.append(cp.constraints.Box(0.0, 1.0, h_var))

        A = []
        for k in range(1, n_points):
            A_k = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    A_k[i, j] = dx
                j = i - k
                if j >= 0:
                    A_k[i, j] = dx
            A.append(A_k)

        for k in range(len(A)):
            constraints.append(h_var.T @ A[k] @ h_var >= 1.0 - C)

        prob = cp.Problem(cp.Minimize(C), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-8, verbose=False)
            if prob.status == 'OPTIMAL' and C.value < best_c5:
                best_c5 = C.value
                best_h = h_var.value.copy()
        except:
            pass

    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    best_h = best_h * scale
    c5_bound = compute_c5(best_h)
    return best_h.tolist(), c5_bound, n_points
```
