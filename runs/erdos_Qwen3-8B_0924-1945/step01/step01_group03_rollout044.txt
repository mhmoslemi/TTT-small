import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def generate_initial_guesses():
        base = np.sin(np.linspace(0, np.pi, n_points)) * 0.5 + 0.5
        base = np.clip(base, 0, 1)
        base = base * target_sum / np.sum(base)
        guesses = [base]
        
        # Add noise
        noise = np.random.normal(0, 0.15, n_points)
        guesses.append(np.clip(base + noise, 0, 1))
        guesses.append(np.clip(base - noise, 0, 1))
        
        # Phase shift
        shifted = np.sin(np.linspace(0, np.pi, n_points) + np.pi / 4) * 0.5 + 0.5
        shifted = np.clip(shifted, 0, 1)
        shifted = shifted * target_sum / np.sum(shifted)
        guesses.append(shifted)
        
        # Reverse
        reversed_base = base[::-1]
        reversed_base = np.clip(reversed_base, 0, 1)
        reversed_base = reversed_base * target_sum / np.sum(reversed_base)
        guesses.append(reversed_base)
        
        # Blend base and reverse
        blended = 0.5 * base + 0.5 * reversed_base
        blended = np.clip(blended, 0, 1)
        blended = blended * target_sum / np.sum(blended)
        guesses.append(blended)
        
        # Triangular wave
        triangular = np.abs(np.sin(np.linspace(0, np.pi, n_points)) * 0.5 + 0.5)
        triangular = np.clip(triangular, 0, 1)
        triangular = triangular * target_sum / np.sum(triangular)
        guesses.append(triangular)
        
        # Square wave
        square = np.zeros(n_points)
        square[:n_points // 3] = 1
        square = np.clip(square, 0, 1)
        square = square * target_sum / np.sum(square)
        guesses.append(square)
        
        # Modulated sinusoidal
        modulated = np.sin(np.linspace(0, np.pi, n_points) + np.cos(np.linspace(0, np.pi, n_points)) * 0.1) * 0.5 + 0.5
        modulated = np.clip(modulated, 0, 1)
        modulated = modulated * target_sum / np.sum(modulated)
        guesses.append(modulated)
        
        # Gaussian kernel
        x = np.linspace(0, 2, n_points)
        distances = np.abs(x - 1)
        gaussian = np.exp(-distances ** 2 / 0.5)
        gaussian = gaussian / np.sum(gaussian)
        gaussian = gaussian * target_sum
        guesses.append(gaussian)
        
        # Piecewise linear
        segments = [0.5, 1, 0.5, 0, 0.5]
        piecewise = np.repeat(segments, n_points // len(segments))
        piecewise = np.clip(piecewise, 0, 1)
        piecewise = piecewise * target_sum / np.sum(piecewise)
        guesses.append(piecewise)
        
        return guesses
    
    def objective(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    def constraint(x):
        return np.sum(x) - target_sum
    
    guesses = generate_initial_guesses()
    best_c5 = np.inf
    best_h = None
    
    for guess in guesses:
        x0 = guess.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            maxiter=200,
            disp=False,
            polish=True,
            args=(),
            vectorized=True
        )
        h = result.x
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        c5 = objective(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Final evaluation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
