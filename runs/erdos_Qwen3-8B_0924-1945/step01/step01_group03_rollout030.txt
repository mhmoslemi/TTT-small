import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Base initial guess: sinusoidal function
    x = np.linspace(0, 2, n_points)
    base_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    h_initial = np.clip(base_guess, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial)

    # Function to create initial guesses
    def create_guesses(base):
        guesses = []
        for i in range(10):
            noise = np.random.uniform(-0.15, 0.15, n_points)
            guess = base + noise
            guess = np.clip(guess, 0, 1)
            guess = guess * target_sum / np.sum(guess)
            guesses.append(guess)
        return guesses

    initial_guesses = create_guesses(h_initial)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    runtime = 0

    for guess in initial_guesses:
        h = np.copy(guess)
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        def objective(x):
            nonlocal runtime
            if runtime >= budget_s:
                return float('inf')
            h_vals = np.clip(x, 0, 1)
            corr = np.correlate(h_vals, 1 - h_vals, mode='full')
            c5 = np.max(corr) * dx
            return c5
        result = opt.minimize(
            objective, h, method='L-BFGS-B', bounds=bounds,
            constraints=constraints, tol=1e-6, options={'disp': False}
        )
        if result.success:
            h_final = np.clip(result.x, 0, 1)
            h_final = h_final * target_sum / np.sum(h_final)
            corr = np.correlate(h_final, 1 - h_final, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_final

        runtime += 100

    # Fallback to best guess if optimization fails
    if best_c5 == float('inf'):
        best_h = h_initial
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx

    return (best_h, best_c5, n_points)
