import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_guesses = []

    # Initial guess from parent
    if 'initial_h_values' in globals():
        h_base = np.copy(initial_h_values)
        h_base = np.clip(h_base, 0, 1)
        h_base /= np.sum(h_base)
        h_base *= target_sum
        initial_guesses.append(h_base)
    else:
        h_base = np.zeros(n_points)
        h_base[::7] = 1.0
        h_base = np.clip(h_base, 0, 1)
        h_base /= np.sum(h_base)
        h_base *= target_sum
        initial_guesses.append(h_base)

    # Generate additional initial guesses
    for _ in range(4):
        h = np.zeros(n_points)
        idx = np.random.choice(n_points, size=5, replace=False)
        h[idx] = 1.0
        h = np.clip(h, 0, 1)
        h /= np.sum(h)
        h *= target_sum
        initial_guesses.append(h)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    for guess in initial_guesses:
        x0 = np.copy(guess)
        x0 = np.clip(x0, 0, 1)
        x0 /= np.sum(x0)
        x0 *= target_sum

        def objective(x):
            x = np.clip(x, 0, 1)
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx

        def constraint(x):
            return np.sum(x) - target_sum

        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.minimize(
            objective,
            x0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-6, 'eps': 1e-6}
        )

        if result.success:
            x = result.x
            x = np.clip(x, 0, 1)
            x /= np.sum(x)
            x *= target_sum
            c5 = objective(x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = x

    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    return (best_h, best_c5, n_points)
