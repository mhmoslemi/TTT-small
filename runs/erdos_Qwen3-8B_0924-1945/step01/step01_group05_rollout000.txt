import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Define parameters
    n_points = 100
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = (h - np.min(h)) / (np.max(h) - np.min(h))
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    
    # Define objective function
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Define constraint function
    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Differential evolution setup
    method = 'differential_evolution'
    strategy = 'best1bin'
    pop_size = 15
    maxiter = 100
    mutation = (0.5, 1)
    recombination = 0.8
    tol = 1e-6
    
    start_time = time.time()
    result = opt.differential_evolution(
        objective,
        bounds,
        strategy=strategy,
        popsize=pop_size,
        maxiter=maxiter,
        mutation=mutation,
        recombination=recombination,
        tol=tol,
        seed=seed,
        disp=False
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        final_c5 = objective(h)
        return (h, final_c5, n_points)
    
    best_h = result.x
    best_c5 = result.fun
    
    # Final validation and normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
