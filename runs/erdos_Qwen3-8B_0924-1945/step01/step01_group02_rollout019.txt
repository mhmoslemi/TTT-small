import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_lag = n_points - 1

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0.0, 1.0)

    def get_overlap_matrix(k):
        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = (i + k) % n_points
            A[i, j] = dx
            j = (i - k) % n_points
            A[i, j] = dx
        return A

    A_list = [get_overlap_matrix(k) for k in range(1, max_lag + 1)]

    h_vars = cp.Variable(n_points, bounds=(0.0, 1.0))
    C = cp.Variable()

    constraints = []
    for A in A_list:
        constraints.append(cp.quad_form(h_vars, A) >= 1 - C)
    constraints.append(cp.sum(h_vars) == target_sum)

    prob = cp.Problem(cp.Minimize(C), constraints)
    start_time = time.time()
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except:
        prob.solve(solver=cp.SCS, verbose=False)

    h_opt = np.clip(h_vars.value, 0.0, 1.0)
    h_opt = h_opt * (target_sum / np.sum(h_opt))
    c5_bound = np.max(np.correlate(h_opt, 1.0 - h_opt, mode="full")) * dx
    return h_opt.tolist(), c5_bound, n_points
