```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            if i % 5 == 0:
                h[i] = 1.0
        h = h * (target_sum / np.sum(h))

    def compute_correlation(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def optimize_h():
        h = cp.Variable(n_points)
        C = cp.Variable()
        constraints = [
            cp.sum(h) == target_sum,
            0 <= h <= 1
        ]
        for k in range(1, n_points):
            A = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    A[i, j] = dx
                j = i - k
                if j >= 0:
                    A[i, j] = dx
            quad_const = cp.quad_form(h, A)
            constraints.append(quad_const >= 1 - C)
        obj = cp.Minimize(C)
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-8, verbose=False)
            if prob.status == ' OPTIMAL':
                return h.value, C.value
        except:
            pass
        return None, None

    best_h = h
    best_c5 = compute_correlation(h)
    start_time = time.time()
    for _ in range(20):
        if time.time() - start_time > budget_s - 50:
            break
        perturb = np.random.normal(0, 0.05, size=n_points)
        h_perturb = h + perturb
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        h_perturb = h_perturb * (target_sum / np.sum(h_perturb))
        h_opt, c5_opt = optimize_h()
        if h_opt is not None and c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h_opt = np.sum(best_h)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    best_h = best_h * scale
    c5_bound = compute_correlation(best_h)
    return best_h.tolist(), c5_bound, n_points
```
