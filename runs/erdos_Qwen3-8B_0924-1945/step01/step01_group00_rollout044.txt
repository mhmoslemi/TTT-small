import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 90  # Increased resolution for better precision
    dx = 2.0 / n_points
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    
    # Fallback to cosine initialization if initial_h_values not provided
    if 'initial_h_values' not in kwargs:
        x = np.linspace(0, 2, n_points, endpoint=False)
        h = (1 + np.cos(np.pi * x / 2)) / 2  # Cosine wave between 0 and 1
        h = h * (n_points / 2) / (np.sum(h) * dx)  # Normalize to satisfy ∑h = n_points/2
        initial_h_values = h
    
    def compute_c5(h):
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)
    
    # Differential Evolution setup
    bounds = [(0, 1) for _ in range(n_points)]
    popsize = 20
    mutation = 0.8
    crossover = 0.9
    maxiter = 50
    tol = 1e-5
    
    def objective(x):
        return evaluate_candidate(x)
    
    # DE with constraint handling through post-normalization
    def de_objective(x):
        return objective(x)
    
    def de_constraint(x):
        return np.sum(x) - n_points / 2
    
    start_time = time()
    
    # Global search with DE
    result = differential_evolution(de_objective, bounds, strategy='best1bin', 
                                   mutation=mutation, recombination=crossover, 
                                   maxiter=maxiter, tol=1e-5, popsize=popsize, 
                                   disp=False, vectorized=True)
    
    best_h = result.x
    best_c5 = evaluate_candidate(best_h)
    
    # Local refinement with SLSQP
    def slsqp_objective(x):
        return evaluate_candidate(x)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    bounds_slsqp = [(0, 1) for _ in range(n_points)]
    
    # Refine top 5 candidates
    for _ in range(5):
        if time() - start_time > budget_s:
            break
        x0 = best_h.copy()
        res = minimize(slsqp_objective, x0, method='SLSQP', bounds=bounds_slsqp, constraints=cons, tol=1e-5)
        if res.success:
            current_h = res.x
            current_c5 = evaluate_candidate(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        else:
            break
    
    # Final checks
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    final_c5 = evaluate_candidate(best_h)
    
    return (best_h, final_c5, n_points)
