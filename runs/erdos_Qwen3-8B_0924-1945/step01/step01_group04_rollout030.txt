def run(seed=42, budget_s=600, **kwargs):

        sum_h = np.sum(h_)
        if sum_h == 0:
            return np.inf
        h_ = h_ * (target_sum / sum_h)
        corr = np.correlate(h_, 1 - h_, mode="full")
        return np.max(corr) * dx
    
    def constraint(h_):
        return np.sum(h_) - target_sum

        population = np.random.uniform(0, 1, (population_size, n_points))

                if obj_trial < obj_current:
                    population[i] = trial
            
            current_best = np.min([objective(p) for p in population])

                best_solution = population[np.argmin([objective(p) for p in population])]
