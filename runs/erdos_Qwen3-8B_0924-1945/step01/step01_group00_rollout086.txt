import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    
    def compute_correlate(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            return 1e10
        h = h / h_sum * n_points / 2
        return compute_correlate(h)
    
    bounds = [(0, 1) for _ in range(n_points)]
    start_time = time()
    
    result = differential_evolution(objective, bounds, strategy='best1bin', 
                                   popsize=20, tol=1e-5, mutation=(0.5, 1), 
                                   recombination=0.9, maxiter=100, 
                                   polish=True, disp=False, 
                                   callback=lambda x: None)
    
    best_h = result.x
    best_c5 = compute_correlate(best_h)
    
    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum == 0:
                continue
            h = h / h_sum * n_points / 2
            current_c5 = compute_correlate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = compute_correlate(best_h)
    
    return (best_h, current_c5, n_points)
