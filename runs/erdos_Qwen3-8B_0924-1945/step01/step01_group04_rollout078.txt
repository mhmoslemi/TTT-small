        h_ = np.clip(h_, 0, 1)
        h_ = h_ * (target_sum / np.sum(h_)) if np.sum(h_) != 0 else h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    def constraint(h_):
        return np.sum(h_) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    max_restarts = 5
    best_h = h.copy()
    best_c5 = objective(h)
    
    for _ in range(max_restarts):
        if 'initial_h_values' in globals():

            h = np.sort(h)
            h = h - np.mean(h)
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
        
        result = optimize.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
