import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    
    def normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / total
        return h * scale
    
    def evaluate(h):
        h = normalize(h)
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def objective(h):
        return evaluate(h)
    
    def bounds_func(x):
        return np.clip(x, 0, 1)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time()
    best_h = initial_h_values.copy()
    best_c5 = evaluate(best_h)
    
    def de_objective(x):
        return evaluate(x)
    
    def de_bounds_func(x):
        return bounds_func(x)
    
    if 'initial_h_values' in kwargs:
        initial_pop = [initial_h_values.copy() for _ in range(20)]
    else:
        initial_pop = [np.random.rand(n_points) for _ in range(20)]
        for i in range(len(initial_pop)):
            initial_pop[i] = normalize(initial_pop[i])
    
    def de_fitness(x):
        return de_objective(x)
    
    def de_crossover(parent1, parent2):
        child = parent1.copy()
        for i in range(n_points):
            if np.random.rand() < 0.9:
                child[i] = parent2[i]
        return child
    
    def de_mutate(ind):
        mutant = np.random.rand(n_points)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                mutant[i] = parent1[i]
        return mutant
    
    def de_run():
        nonlocal best_h, best_c5
        pop = [normalize(np.random.rand(n_points)) for _ in range(20)]
        for gen in range(100):
            if time() - start_time > budget_s:
                break
            new_pop = []
            for i in range(20):
                a, b, c = np.random.choice(range(20), 3, replace=False)
                parent1, parent2, parent3 = pop[a], pop[b], pop[c]
                mutant = de_mutate(parent1)
                child = de_crossover(parent1, parent2)
                child = normalize(child)
                new_pop.append(child)
            pop = new_pop
            for candidate in pop:
                c5 = evaluate(candidate)
                if c5 < best_c5:
                    best_h = candidate.copy()
                    best_c5 = c5
    
    def sqp_run(h):
        nonlocal best_h, best_c5
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum})
        if res.success:
            h_new = normalize(res.x)
            c5_new = evaluate(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
    
    de_run()
    for _ in range(15):
        if time() - start_time > budget_s:
            break
        sqp_run(best_h)
    
    best_h = normalize(best_h)
    best_c5 = evaluate(best_h)
    
    return (best_h, best_c5, n_points)
