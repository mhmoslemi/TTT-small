```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def build_constraints(n_points, dx):
        constraints = []
        for k in range(1, n_points):
            A = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = (i + k) % n_points
                A[i, j] = dx
                j = (i - k) % n_points
                A[i, j] = dx
            constraints.append(A)
        return constraints

    def solve_qp(h_initial, constraints, n_points, dx, target_sum):
        h = cp.Variable(n_points)
        C = cp.Variable()
        objective = cp.Minimize(C)
        constraints = [
            cp.sum(h * dx) == target_sum,
            0 <= h <= 1,
            cp.sum((h @ A) * dx) >= 1 - C for A in constraints
        ]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-8)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                h_opt = h.value
                c5 = evaluate(h_opt.flatten())
                return h_opt, c5
        except:
            pass
        return h_initial, evaluate(h_initial.flatten())

    constraints = build_constraints(n_points, dx)
    best_h = h.copy()
    best_c5 = evaluate(h.flatten())

    start_time = time.time()
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        h_initial = np.random.rand(n_points) * 2 - 1
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (target_sum / np.sum(h_initial))
        h_opt, c5 = solve_qp(h_initial, constraints, n_points, dx, target_sum)
        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5

    return best_h.tolist(), best_c5, n_points
```
