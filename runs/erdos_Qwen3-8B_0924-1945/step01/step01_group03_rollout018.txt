import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Configuration
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate initial guess based on sinusoidal pattern
    def generate_initial_guess():
        x_vals = np.linspace(0, 2, n_points)
        base_guess = 0.5 + 0.5 * np.sin(np.pi * x_vals)
        base_guess = np.clip(base_guess, 0, 1)
        base_guess = base_guess * (target_sum / np.sum(base_guess))
        return base_guess
    
    # Fallback initial guess
    def fallback_guess():
        return np.full(n_points, 0.5)
    
    # Generate structured guesses
    structured_guesses = []
    base_guess = generate_initial_guess()
    structured_guesses.append(base_guess)
    noise_guess = base_guess * (1 + 0.15 * np.random.randn(n_points))
    noise_guess = np.clip(noise_guess, 0, 1)
    noise_guess = noise_guess * (target_sum / np.sum(noise_guess))
    structured_guesses.append(noise_guess)
    shifted_guess = base_guess * np.roll(np.cos(np.pi * x_vals), 1)
    shifted_guess = np.clip(shifted_guess, 0, 1)
    shifted_guess = shifted_guess * (target_sum / np.sum(shifted_guess))
    structured_guesses.append(shifted_guess)
    
    # Optimize each guess
    best_h = np.full(n_points, np.nan)
    best_c5 = float('inf')
    time_used = 0
    
    for guess in structured_guesses:
        guess = np.clip(guess, 0, 1)
        guess = guess * (target_sum / np.sum(guess))
        guess = guess.astype(float)
        
        def objective(x):
            x = np.clip(x, 0, 1)
            x = x * (target_sum / np.sum(x))
            corr = np.correlate(x, 1 - x, mode='full')
            c5 = np.max(corr) * dx
            return c5
        
        constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        bounds = [(0, 1) for _ in range(n_points)]
        
        start_time = opt.time.time()
        result = opt.differential_evolution(
            objective,
            bounds,
            constraints=constraints,
            strategy='best1bin',
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            maxiter=200,
            disp=False
        )
        time_used += opt.time.time() - start_time
        
        if time_used >= budget_s:
            break
        
        result_x = result.x
        result_x = np.clip(result_x, 0, 1)
        result_x = result_x * (target_sum / np.sum(result_x))
        
        final_c5 = objective(result_x)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = result_x
    
    # Fallback to fallback guess if no improvement
    if np.isnan(best_h).all():
        best_h = fallback_guess()
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (target_sum / np.sum(best_h))
        best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
