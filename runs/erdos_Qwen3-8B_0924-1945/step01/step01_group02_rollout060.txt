```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        ones = np.ceil(n_points / 2)
        h[:ones] = 1.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    def constraint_lag(h_flat, k, C):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return 1.0 - C - corr[k]

    variables = cp.Variable(n_points)
    C = cp.Variable()

    constraints = [
        cp.sum(variables) == target_sum,
        0 <= variables,
        variables <= 1.0
    ]

    for k in range(1, n_points):
        constraints.append(cp.sum(cp.multiply(variables, variables[k:])) >= 1.0 - C)

    prob = cp.Problem(cp.Minimize(C), constraints)
    result = prob.solve(solver=cp.ECOS, verbose=False)
    h_opt = variables.value
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale

    c5_bound = objective(h_opt.flatten())
    return h_opt.tolist(), c5_bound, n_points
```
