import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = np.copy(initial_h_values)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h /= np.sum(initial_h)
    initial_h *= target_sum

    def generate_initial_guesses(base_guess):
        guesses = []
        guesses.append(base_guess.copy())
        n = n_points
        step = n // 5
        row_guess = np.zeros(n)
        for i in range(5):
            row_guess[i * step] = 1
        row_guess = np.roll(row_guess, np.random.randint(n))
        row_guess = np.clip(row_guess, 0, 1)
        row_guess /= np.sum(row_guess)
        row_guess *= target_sum
        guesses.append(row_guess)
        guesses.append(np.clip(base_guess + np.random.uniform(-0.2, 0.2, n), 0, 1))
        guesses.append(np.clip(1 - base_guess, 0, 1))
        guesses.append(0.5 * base_guess + 0.5 * row_guess)
        return guesses

    def objective(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    guesses = generate_initial_guesses(initial_h)
    best_h = None
    best_c5 = float('inf')

    for guess in guesses:
        x0 = guess.copy()
        result = opt.minimize(
            objective,
            x0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-6, 'disp': False}
        )
        if result.success:
            h_vals = result.x
            h_vals = np.clip(h_vals, 0, 1)
            h_vals /= np.sum(h_vals)
            h_vals *= target_sum
            c5 = objective(h_vals)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_vals

    if best_h is None:
        best_h = initial_h.copy()
        best_h = np.clip(best_h, 0, 1)
        best_h /= np.sum(best_h)
        best_h *= target_sum
        best_c5 = objective(best_h)

    return (best_h, best_c5, n_points)
