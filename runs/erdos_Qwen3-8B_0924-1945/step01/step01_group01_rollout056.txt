import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Objective function
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Gradient-based optimization
    def gradient_approx(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_idx = np.argmax(correlate_vals)
        if max_idx < n_points:
            lag = max_idx
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_test = h.copy()
                h_test[i] += 1e-6
                h_test = np.clip(h_test, 0, 1)
                h_test = h_test * (n_points / 2) / np.sum(h_test)
                corr = np.correlate(h_test, 1 - h_test, mode="full") * (2.0 / n_points)
                grad[i] = (corr[max_idx] - correlate_vals[max_idx]) / 1e-6
        else:
            lag = max_idx - n_points
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_test = h.copy()
                h_test[i] += 1e-6
                h_test = np.clip(h_test, 0, 1)
                h_test = h_test * (n_points / 2) / np.sum(h_test)
                corr = np.correlate(h_test, 1 - h_test, mode="full") * (2.0 / n_points)
                grad[i] = (corr[max_idx] - correlate_vals[max_idx]) / 1e-6
        return grad
    
    # Setup for optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, jac=gradient_approx,
                          tol=1e-5, maxiter=500, options={'disp': False})
    
    best_h = result.x
    best_c5 = result.fun
    
    # Adaptive grid refinement
    n_points = 90
    h = best_h.copy()
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_ref = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, jac=gradient_approx,
                              tol=1e-5, maxiter=500, options={'disp': False})
    
    best_h = result_ref.x
    best_c5 = result_ref.fun
    
    # Fallback to differential evolution if needed
    elapsed = time.time() - start_time
    if elapsed > budget_s - 100:
        print("Time budget exceeded, returning refined solution")
        h = best_h.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        final_c5 = objective(h)
        return (h.tolist(), final_c5, n_points)
    
    # Final evaluation
    h = best_h.copy()
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points)
