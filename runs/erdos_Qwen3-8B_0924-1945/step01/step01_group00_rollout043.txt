import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', None)
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        x = np.linspace(0, 2, n_points, endpoint=False)
        h = (1 + np.cos(np.pi * x / 2)) / 2
        h = h * (n_points / 2) / (np.sum(h) * 2 / n_points)
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)
    
    def objective(x):
        return evaluate_candidate(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time()
    
    def de_run():
        def de_fitness(x):
            return evaluate_candidate(x)
        
        result = differential_evolution(
            de_fitness,
            bounds,
            strategy='best1bin',
            popsize=20,
            tol=1e-5,
            mutation=(0.8, 1.0),
            recombination=0.9,
            maxiter=50,
            disp=False,
            polish=True,
            init='latinhypercube',
            vectorize=True,
            workers=1
        )
        best_h = result.x
        best_c5 = evaluate_candidate(best_h)
        return best_h, best_c5
    
    best_h, best_c5 = de_run()
    
    if time() - start_time < budget_s:
        for _ in range(10):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    for _ in range(5):
        if time() - start_time > budget_s:
            break
        res = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-5
        )
        if res.success:
            current_h = res.x
            current_c5 = evaluate_candidate(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            break
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    final_c5 = evaluate_candidate(best_h)
    return (best_h, final_c5, n_points)
