import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 90
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Objective function
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Gradient approximation
    def gradient(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_lag = np.argmax(correlate_vals)
        dx = 2.0 / n_points
        grad = np.zeros_like(h)
        for i in range(n_points):
            x = i * dx
            k = max_lag - n_points
            if 0 <= x + k < 2:
                grad[i] = (1 - h[int((x + k) / dx)]) * dx
        return grad
    
    # Optimization with L-BFGS-B
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, jac=gradient,
                          options={'maxiter': 500, 'ftol': 1e-5, 'eps': 1e-6, 'disp': False})
    
    best_h = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time
    
    # Check time and apply fallback if needed
    if elapsed > budget_s - 100:
        print("Time budget exceeded, returning best solution")
        h = best_h
        best_c5 = objective(h)
    else:
        # Refinement with more points
        n_points = 120
        h = best_h
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        def objective_refine(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(correlate_vals)
        
        def gradient_refine(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            max_lag = np.argmax(correlate_vals)
            dx = 2.0 / n_points
            grad = np.zeros_like(h)
            for i in range(n_points):
                x = i * dx
                k = max_lag - n_points
                if 0 <= x + k < 2:
                    grad[i] = (1 - h[int((x + k) / dx)]) * dx
            return grad
        
        result_refine = opt.minimize(objective_refine, h, method='L-BFGS-B', bounds=bounds, jac=gradient_refine,
                                     options={'maxiter': 300, 'ftol': 1e-5, 'eps': 1e-6, 'disp': False})
        best_h = result_refine.x
        best_c5 = result_refine.fun
    
    # Final clipping and rescaling
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective_refine(h)
    
    return (h.tolist(), final_c5, n_points)
