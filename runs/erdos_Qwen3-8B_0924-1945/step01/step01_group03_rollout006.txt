import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def generate_initial_guesses():
        base_guess = np.copy(initial_h_values)
        base_guess = np.clip(base_guess, 0, 1)
        base_guess /= np.sum(base_guess)
        base_guess *= target_sum

        structured_guess = np.zeros(n_points)
        spacing = n_points // 5
        for i in range(5):
            start = np.random.randint(0, spacing)
            for j in range(n_points):
                if (j - start) % spacing == 0:
                    structured_guess[j] = 1
        structured_guess = np.clip(structured_guess, 0, 1)
        structured_guess /= np.sum(structured_guess)
        structured_guess *= target_sum

        perturbed_guess = base_guess * (1 + 0.2 * np.random.rand(n_points))
        perturbed_guess = np.clip(perturbed_guess, 0, 1)
        perturbed_guess /= np.sum(perturbed_guess)
        perturbed_guess *= target_sum

        reversed_guess = base_guess[::-1]
        reversed_guess = np.clip(reversed_guess, 0, 1)
        reversed_guess /= np.sum(reversed_guess)
        reversed_guess *= target_sum

        blended_guess = 0.5 * base_guess + 0.5 * structured_guess
        blended_guess = np.clip(blended_guess, 0, 1)
        blended_guess /= np.sum(blended_guess)
        blended_guess *= target_sum

        return [base_guess, structured_guess, perturbed_guess, reversed_guess, blended_guess]

    initial_guesses = generate_initial_guesses()

    best_c5 = np.inf
    best_h = np.zeros(n_points)

    for guess in initial_guesses:
        x0 = guess.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

        def objective(x):
            x_vals = x.copy()
            x_vals = np.clip(x_vals, 0, 1)
            corr = np.correlate(x_vals, 1 - x_vals, mode='full')
            c5 = np.max(corr) * dx
            return c5

        result = opt.minimize(
            objective,
            x0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-6, 'eps': 1e-6}
        )

        if result.success and np.isfinite(result.fun):
            optimized = np.clip(result.x, 0, 1)
            optimized /= np.sum(optimized)
            optimized *= target_sum
            c5_val = objective(optimized)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = optimized

    return (best_h, best_c5, n_points)
