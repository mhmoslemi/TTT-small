import numpy as np
from scipy.optimize import differential_evolution, minimize
from numpy.random import choice, uniform, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(x):
        return evaluate_candidate(x)

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x / np.sum(x) * n_points / 2
        return x

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()

    def differential_evolution_run():
        pop_size = 20
        mutation = 0.5
        recombination = 0.9
        max_gen = 100
        best_h = h.copy()
        best_c5 = evaluate_candidate(best_h)
        pop = [normalize(np.random.rand(n_points)) for _ in range(pop_size)]
        for gen in range(max_gen):
            if time() - start_time > budget_s:
                break
            new_pop = []
            for i in range(pop_size):
                a, b, c = choice(pop_size, 3, replace=False)
                mutant = pop[a] + mutation * (pop[b] - pop[c])
                trial = np.clip(mutant, 0, 1)
                trial = trial / np.sum(trial) * n_points / 2
                trial_c5 = compute_c5(trial)
                if trial_c5 < compute_c5(pop[i]):
                    new_pop.append(trial)
                else:
                    new_pop.append(pop[i])
            pop = new_pop
            current_best = min(pop, key=evaluate_candidate)
            if evaluate_candidate(current_best) < best_c5:
                best_h = current_best.copy()
                best_c5 = evaluate_candidate(best_h)
        return best_h, best_c5

    best_h, best_c5 = differential_evolution_run()

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = normalize(best_h)
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
