```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_restarts = 20
    max_evaluations = 50
    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        num_ones = int(np.ceil(n_points / 2))
        h[:num_ones] = 1.0
        h = h * (target_sum / np.sum(h))

    def get_A_k(k):
        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            j1 = (i + k) % n_points
            j2 = (i - k) % n_points
            A[i, j1] = dx
            A[i, j2] = dx
        return A

    for restart in range(max_restarts):
        start_time = time.time()
        A_list = [get_A_k(k) for k in range(1, n_points)]

        h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
        C = cp.Variable()

        constraints = [
            cp.sum(h_var) == target_sum,
            cp.sum(cp.multiply(h_var, h_var) @ A_list[0]) >= 1 - C
        ]
        for A in A_list[1:]:
            constraints.append(cp.sum(cp.multiply(h_var, h_var) @ A) >= 1 - C)

        prob = cp.Problem(cp.Minimize(C), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-8, max_iter=100)
            if prob.status == 'OPTIMAL':
                current_h = np.round(h_var.value, 5)
                current_h = np.clip(current_h, 0.0, 1.0)
                current_sum = np.sum(current_h)
                current_h = current_h * (target_sum / current_sum) if current_sum != 0 else current_h
                c5 = np.max(np.correlate(current_h, 1 - current_h, mode="full") * dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current_h
        except:
            pass

        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            break

    final_h = best_h
    final_c5 = np.max(np.correlate(final_h, 1 - final_h, mode="full") * dx)
    final_h = np.clip(final_h, 0.0, 1.0)
    final_sum = np.sum(final_h)
    final_h = final_h * (target_sum / final_sum) if final_sum != 0 else final_h
    return final_h.tolist(), final_c5, n_points
```
