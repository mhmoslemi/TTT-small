import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def clip_and_scale(x):
        x = np.clip(x, 0, 1)
        x = x / np.sum(x) * n_points / 2
        return x

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    dx = 2.0 / n_points

    def objective(x):
        x = clip_and_scale(x)
        return compute_c5(x)

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    best_c5 = best_c5

    for _ in range(150):
        if time() - start_time > budget_s:
            break
        x0 = best_h.copy()
        res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, tol=1e-5)
        if res.success:
            current_h = res.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            break

    for _ in range(100):
        if time() - start_time > budget_s:
            break
        h = np.random.uniform(0, 1, n_points)
        h = clip_and_scale(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    best_h = clip_and_scale(best_h)
    current_c5 = compute_c5(best_h)
    return (best_h, current_c5, n_points)
