import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import truncnorm
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    while time() - start_time < budget_s:
        population_size = 20
        population = np.random.uniform(0, 1, (population_size, n_points))
        population = population / np.sum(population, axis=1, keepdims=True) * n_points / 2
        population = np.clip(population, 0, 1)
        population_c5 = np.array([compute_c5(p) for p in population])
        best_idx = np.argmin(population_c5)
        best_candidate = population[best_idx]
        best_c5 = population_c5[best_idx]
        if np.allclose(best_candidate, best_h, atol=1e-6):
            break
        best_h = best_candidate.copy()
        if best_c5 < best_c5:
            best_c5 = best_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    best_c5 = evaluate_candidate(best_h)
    return (best_h, best_c5, n_points)
