import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 90
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Precompute lag indices
    dx = 2.0 / n_points
    lags = np.arange(1, n_points)
    total_lags = 2 * n_points - 1
    valid_lags = np.concatenate([lags, lags - n_points])
    
    # Objective function for QP
    def qp_objective(mu):
        return mu
    
    # Constraint function for QP
    def qp_constraints(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return corr - (1 - mu)
    
    # Choose solver based on time
    if time.time() - start_time > budget_s * 0.7:
        # Fallback to differential evolution if time is limited
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        def objective(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(correlate_vals)
        result = opt.differential_evolution(objective, bounds, popsize=20, mutation=(0.5, 1.0), 
                                           recombination=0.9, strategy='best1bin', tol=1e-5,
                                           maxiter=300, disp=False, polish=True, seed=seed)
        best_h = result.x
        best_c5 = result.fun
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            print("Time budget exceeded, returning initial solution")
            h = initial_h_values if 'initial_h_values' in globals() else h
            best_c5 = objective(h)
        # Final check
        h = np.clip(best_h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        final_c5 = objective(h)
        return (h.tolist(), final_c5, n_points)
    
    # CVXOPT QP formulation
    try:
        import cvxopt
        cvxopt.solvers.options['maxiter'] = 100
        cvxopt.solvers.options['abstol'] = 1e-8
        cvxopt.solvers.options['reltol'] = 1e-8
        
        P = np.zeros((n_points, n_points))
        q = np.zeros(n_points)
        G = np.zeros((total_lags, n_points))
        h_bounds = np.zeros(total_lags)
        mu = 1.0
        
        for i in range(n_points):
            for j in range(n_points):
                P[i, j] = dx * dx * 0.5 * (i == j)
        
        for lag_idx, lag in enumerate(valid_lags):
            if lag < 0:
                j = lag + n_points
            else:
                j = lag
            for i in range(n_points):
                if 0 <= i + lag < n_points:
                    G[lag_idx, i] += dx * dx * 1.0
                else:
                    G[lag_idx, i] += 0.0
            h_bounds[lag_idx] = (1 - mu)
        
        P = cvxopt.matrix(P)
        q = cvxopt.matrix(q)
        G = cvxopt.matrix(G)
        h_bounds = cvxopt.matrix(h_bounds)
        
        A = cvxopt.matrix(np.ones(n_points), (1, n_points))
        b = cvxopt.matrix(n_points / 2)
        
        # Minimize mu
        sol = cvxopt.solvers.qp(P, q, G, h_bounds, A, b)
        
        if sol['status'] != 'optimal':
            raise Exception("QP solver failed")
        
        h = np.array(sol['x']).flatten()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        best_c5 = np.max(np.correlate(h, 1 - h, mode="full") * dx)
        
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            print("Time budget exceeded, returning initial solution")
            h = initial_h_values if 'initial_h_values' in globals() else h
            best_c5 = np.max(np.correlate(h, 1 - h, mode="full") * dx)
        
        # Final check and return
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        final_c5 = np.max(np.correlate(h, 1 - h, mode="full") * dx)
        return (h.tolist(), final_c5, n_points)
    except:
        # Fallback to differential evolution if QP is not working
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        def objective(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(correlate_vals)
        result = opt.differential_evolution(objective, bounds, popsize=20, mutation=(0.5, 1.0), 
                                           recombination=0.9, strategy='best1bin', tol=1e-5,
                                           maxiter=300, disp=False, polish=True, seed=seed)
        best_h = result.x
        best_c5 = result.fun
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            print("Time budget exceeded, returning initial solution")
            h = initial_h_values if 'initial_h_values' in globals() else h
            best_c5 = objective(h)
        # Final check
        h = np.clip(best_h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        final_c5 = objective(h)
        return (h.tolist(), final_c5, n_points)
