```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_restarts = 20
    improvement_threshold = 1e-4
    best_c5 = float('inf')
    best_h = None

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def evaluate_c5(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def build_qp(h_initial):
        h = cp.Variable(n_points)
        C = cp.Variable()
        constraints = []
        constraints.append(cp.sum(h) * dx == 1)
        constraints.append(cp.constraints.Box(0.0, 1.0, h))
        for k in range(1, n_points):
            A = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = (i + k) % n_points
                A[i, j] += dx
                j = (i - k + n_points) % n_points
                A[i, j] += dx
            constraints.append(cp.quad_form(h, A) >= 1 - C)
        obj = cp.Minimize(C)
        prob = cp.Problem(obj, constraints)
        return prob, h, C

    start_time = time.time()
    for restart in range(max_restarts):
        if time.time() - start_time > budget_s:
            break
        if initial_h_values is None:
            h = np.random.rand(n_points) * 2 - 1
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
        else:
            h = np.copy(h)
        if np.random.rand() < 0.1:
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
        prob, h_var, C_var = build_qp(h)
        try:
            result = prob.solve(solver=cp.ECOS, eps=1e-8, max_iter=100)
            if result is not None:
                c5 = C_var.value
                if c5 < best_c5 - improvement_threshold:
                    best_c5 = c5
                    best_h = h_var.value
        except:
            pass
        if np.random.rand() < 0.5 and restart < max_restarts - 1:
            h = np.copy(best_h)
    if best_h is None:
        best_h = h
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    best_h = best_h * scale
    c5_bound = evaluate_c5(best_h)
    return best_h.tolist(), c5_bound, n_points
```
