import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def refine_solution(h, n_points):
        h = h.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(
            objective, bounds, popsize=20, mutation=(0.9, 1.0),
            recombination=0.9, strategy='best1bin', tol=1e-6, maxiter=200,
            disp=False, polish=True, seed=seed)
        best_h = result.x
        final_c5 = result.fun
        return best_h, final_c5
    
    best_h = h
    best_c5 = objective(h)
    elapsed = time.time() - start_time
    
    if elapsed < budget_s - 300:
        n_points_refine = 90
        h_refined, c5_refined = refine_solution(h, n_points_refine)
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points_refine / 2) / np.sum(h_refined)
        final_c5 = objective(h_refined)
        best_h = h_refined
        best_c5 = final_c5
    
    if elapsed < budget_s - 100:
        result = opt.minimize(
            objective, best_h, method='L-BFGS-B', bounds=[(0.0, 1.0)]*len(best_h),
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6, 'eps': 1e-6}
        )
        best_h = result.x
        best_c5 = result.fun
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
    
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points)
