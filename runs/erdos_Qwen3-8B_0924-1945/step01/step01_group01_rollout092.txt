import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def gradient_estimate(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        k_star = np.argmax(correlate_vals)
        grad = np.zeros_like(h)
        if k_star < n_points:
            grad[:-k_star] += (1 - h[k_star:]) * (2.0 / n_points)
        if k_star > 0:
            grad[k_star:] += (1 - h[:n_points - k_star]) * (2.0 / n_points)
        return grad
    
    # Stage 1: Coarse global optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.differential_evolution(
        objective, bounds, popsize=50, mutation=(0.8, 1.0), 
        recombination=0.9, strategy='best1bin', tol=1e-5,
        maxiter=200, disp=False, polish=True, seed=seed
    )
    best_h = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning initial solution")
        h = initial_h_values if 'initial_h_values' in globals() else h
        best_c5 = objective(h)
        return (h.tolist(), best_c5, n_points)
    
    # Stage 2: Gradient-based refinement
    try:
        result = opt.minimize(
            objective, best_h, method='L-BFGS-B', bounds=bounds,
            jac=gradient_estimate, tol=1e-6, options={'maxiter': 300}
        )
        best_h = result.x
        best_c5 = result.fun
    except:
        best_h = best_h
        best_c5 = objective(best_h)
    
    # Final check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (h.tolist(), best_c5, n_points)
