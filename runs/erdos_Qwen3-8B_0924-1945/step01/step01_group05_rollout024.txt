import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with the initial construction if available
    n_points = 100
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = np.concatenate([h, np.zeros(n_points//2)])
        h = np.random.permutation(h)
    
    # Normalize to ensure sum(h) == n_points / 2
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization settings
    maxiter = 200
    tol = 1e-6
    
    # Optimization
    start_time = time.time()
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol}
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h, objective(h), n_points)
    
    best_h = result.x
    best_c5 = result.fun
    
    # Final sanity checks
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
