import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_lag = n_points - 1

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h = h * scale
    h = np.clip(h, 0.0, 1.0)

    def build_lag_constraints(lag):
        indices = []
        for i in range(n_points):
            j = i + lag
            if j < n_points:
                indices.append((i, j))
            j = i - lag
            if j >= 0:
                indices.append((j, i))
        return indices

    lag_constraints = [build_lag_constraints(k) for k in range(1, max_lag + 1)]

    A = []
    for k in range(1, max_lag + 1):
        row = np.zeros(n_points)
        for i, j in lag_constraints[k - 1]:
            row[i] += dx
            row[j] += dx
        A.append(row)

    h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
    C = cp.Variable()

    constraints = [
        cp.sum(h_var) == target_sum,
        C >= 1.0 - cp.sum(A[k] * h_var) for k in range(len(A))
    ]

    obj = cp.Minimize(C)
    prob = cp.Problem(obj, constraints)

    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        try:
            result = prob.solve(solver=cp.SCS, eps=1e-6, max_iter=100)
            if result is not None and result < best_c5:
                best_c5 = result
                best_h = h_var.value
        except:
            pass
        if np.allclose(h_var.value, h_var.value, atol=1e-6):
            break
        h = h_var.value
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
        constraints = [
            cp.sum(h_var) == target_sum,
            C >= 1.0 - cp.sum(A[k] * h_var) for k in range(len(A))
        ]
        obj = cp.Minimize(C)
        prob = cp.Problem(obj, constraints)

    h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h = h * scale
    h = np.clip(h, 0.0, 1.0)

    c5_bound = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
    return h.tolist(), c5_bound, n_points
