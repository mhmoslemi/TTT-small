import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Use initial guess if available
    if hasattr(initial_h_values, '__iter__'):
        h = np.copy(initial_h_values)
        valid = np.isfinite(h).all()
        if not valid:
            h = np.zeros(n_points)
    else:
        h = np.zeros(n_points)
        h[::5] = 1.0
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * target_sum

    def objective(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        c5 = np.max(corr) * dx
        if np.isnan(c5):
            return np.inf
        return c5

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]

    # Differential evolution setup
    pop_size = 15
    max_gen = 200
    start_time = time.time()

    def de_objective(x):
        return objective(x)

    result = opt.differential_evolution(
        de_objective,
        bounds,
        constraints={'type': 'eq', 'fun': constraint},
        strategy='best1bin',
        popsize=pop_size,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-6,
        maxiter=max_gen,
        polish=True,
        disp=False,
        vectorized=True,
        tol_relativemetric=1e-6,
        seed=seed
    )

    if time.time() - start_time > budget_s - 10:
        print("Time budget exceeded, returning best candidate so far")
        best_h = h
    else:
        best_h = result.x

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * target_sum
    c5_bound = objective(best_h)

    return (best_h, c5_bound, n_points)
