import numpy as np
import scipy.optimize as optimize
import numpy.random as npr

def run(seed=42, budget_s=600, **kwargs):
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 100
        h = np.full(n_points, 1.0 / n_points)
    h = np.clip(h, 0.0, 1.0)
    target_sum = n_points / 2
    h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h

    def objective(h_):
        h_ = np.clip(h_, 0.0, 1.0)
        total = np.sum(h_)
        h_ = h_ * (target_sum / total) if total != 0 else h_
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * (2.0 / n_points)

    bounds = [(0.0, 1.0) for _ in range(len(h))]

    def de_mutate(pop, idx, F):
        a, b, c = np.random.choice(pop, 3, replace=False)
        mutant = a + F * (b - c)
        mutant[idx] = h[idx]
        return mutant

    def de_crossover(pop, mutant, cr):
        trial = np.copy(pop)
        for i in range(len(mutant)):
            if np.random.rand() < cr:
                trial[i] = mutant[i]
        return trial

    n_points = len(h)
    pop_size = 50
    F = 0.5
    cr = 0.7
    max_gen = 200
    tol = 1e-5
    best_h = h.copy()
    best_c5 = objective(best_h)

    for gen in range(max_gen):
        pop = np.random.rand(pop_size, n_points)
        pop = np.clip(pop, 0.0, 1.0)
        for i in range(pop_size):
            pop[i] = pop[i] * (target_sum / np.sum(pop[i])) if np.sum(pop[i]) != 0 else pop[i]
        for i in range(pop_size):
            mutant = de_mutate(pop, i, F)
            trial = de_crossover(pop, mutant, cr)
            for j in range(n_points):
                trial[j] = np.clip(trial[j], 0.0, 1.0)
                trial[j] = trial[j] * (target_sum / np.sum(trial[j])) if np.sum(trial[j]) != 0 else trial[j]
            c5 = objective(trial)
            if c5 < best_c5:
                best_c5 = c5
                best_h = trial.copy()
        if best_c5 < tol:
            break

    h_opt = best_h.copy()
    h_opt = np.clip(h_opt, 0.0, 1.0)
    total = np.sum(h_opt)
    h_opt = h_opt * (target_sum / total) if total != 0 else h_opt
    c5_bound = objective(h_opt)
    return (h_opt.tolist(), c5_bound, n_points)
