import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x / np.sum(x) * n_points / 2
        return x

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    def de_objective(x):
        return evaluate_candidate(normalize(x))

    result = differential_evolution(de_objective, bounds, strategy='best1bin', popsize=20, mutation=(0.5, 1), recombination=0.9, tol=1e-5, maxiter=100, disp=False, polish=True, init=best_h)
    best_h = normalize(result.x)
    best_c5 = result.fun

    if time() - start_time < budget_s:
        for _ in range(10):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5

    best_h = normalize(best_h)
    return (best_h, best_c5, n_points)
