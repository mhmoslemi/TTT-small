import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 50
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = np.round(initial_h_values)
        h = np.clip(h, 0, 1)
        if np.sum(h) != n_points // 2:
            h[np.random.choice(n_points, n_points - np.sum(h), replace=False)] = 1
    else:
        h = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
        h = np.clip(h, 0, 1)
        if np.sum(h) != n_points // 2:
            h[np.random.choice(n_points, n_points - np.sum(h), replace=False)] = 1

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    def run_optimization():
        nonlocal best_h, best_c5
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            x0 = best_h.copy()
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
            if res.success:
                current_h = res.x
                current_c5 = evaluate_candidate(current_h)
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
            else:
                break
        return best_h, best_c5

    best_h, best_c5 = run_optimization()

    if time() - start_time < budget_s:
        for _ in range(50):
            if time() - start_time > budget_s:
                break
            h = np.random.choice([0, 1], size=n_points)
            h = np.clip(h, 0, 1)
            if np.sum(h) != n_points // 2:
                h[np.random.choice(n_points, n_points - np.sum(h), replace=False)] = 1
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
