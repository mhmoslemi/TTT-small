import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 90
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Objective function
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Gradient-based optimization
    def objective_with_gradient(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_index = np.argmax(correlate_vals)
        k = max_index - n_points + 1
        if k < 0:
            k = 0
        if k >= n_points:
            k = n_points - 1
        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(n_points):
            if k + i < n_points:
                grad[i] = (1 - h[k + i]) * (2.0 / n_points)
        return np.max(correlate_vals), grad
    
    # Optimization using L-BFGS-B
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, jac=False, tol=1e-5, options={'maxiter': 500})
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Adaptive refinement
    n_points = 120
    h_refined = np.zeros(n_points)
    step = n_points // len(best_h)
    for i in range(len(best_h)):
        h_refined[i * step] = best_h[i]
    h_refined = np.clip(h_refined, 0, 1)
    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
    
    # Objective function for refined grid
    def objective_refined(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Gradient-based optimization with refined grid
    result_refined = opt.minimize(objective_refined, h_refined, method='L-BFGS-B', bounds=bounds, jac=False, tol=1e-5, options={'maxiter': 500})
    best_h_refined = result_refined.x
    best_c5_refined = objective_refined(best_h_refined)
    
    # Final check
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning refined solution")
        h = best_h_refined
        best_c5 = best_c5_refined
    
    # Final feasibility check
    h = np.clip(best_h_refined, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective_refined(h)
    
    return (h.tolist(), final_c5, n_points)
