```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def compute_corr(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return corr * dx

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = compute_corr(h)
        return np.max(corr)

    def build_qp(h_initial):
        h = cp.Variable(n_points)
        C = cp.Variable()
        constraints = []
        constraints.append(cp.sum(h) == target_sum)
        constraints.append(cp.minimize(C))
        for k in range(1, n_points):
            A = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = (i + k) % n_points
                A[i, j] = dx
                j = (i - k + n_points) % n_points
                A[i, j] = dx
            constraints.append(h @ A @ h >= 1 - C)
        prob = cp.Problem(cp.Minimize(C), constraints)
        result = prob.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8, max_iter=100)
        h_opt = np.clip(h.value, 0.0, 1.0)
        sum_h = np.sum(h_opt)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_opt = h_opt * scale
        return h_opt, evaluate(h_opt)

    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate(h)
    num_restarts = 20
    for _ in range(num_restarts):
        if np.random.rand() < 0.1:
            h = np.random.rand(n_points)
            h = h * (target_sum / np.sum(h))
        h = h.copy()
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))
        h_opt, c5 = build_qp(h)
        if c5 < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5
        if time.time() - start_time > budget_s - 50:
            break
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        h = best_h.copy()
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        best_c5 = evaluate(h)
    return best_h.tolist(), best_c5, n_points
```
