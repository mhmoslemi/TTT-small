```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h.copy()
    best_c5 = objective(h)

    start_time = time.time()
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        if np.random.rand() < 0.2:
            perturb = np.random.uniform(-0.05, 0.05, n_points)
            h = np.clip(h + perturb, 0.0, 1.0)
        else:
            h = np.random.uniform(0.0, 1.0, n_points)
        h = h * (target_sum / np.sum(h))

        x = cp.Variable(n_points)
        c = cp.Variable()
        obj = cp.Minimize(c)
        constraints = [
            cp.sum(x) == target_sum,
            0 <= x,
            x <= 1,
        ]
        for k in range(1, n_points - 1):
            A_k = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    A_k[i, j] = dx
                j = i - k
                if j >= 0:
                    A_k[i, j] = dx
            lhs = cp.quad_form(x, A_k)
            constraints.append(lhs >= 1 - c)
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-6, max_iter=100)
            if prob.status == 'OPTIMAL':
                h_opt = x.value
                h_opt = np.clip(h_opt, 0.0, 1.0)
                sum_h_opt = np.sum(h_opt)
                scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
                h_opt = h_opt * scale
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
        except:
            pass
    elapsed_time = time.time() - start_time
    h_opt = best_h
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt)
    return h_opt.tolist(), c5_bound, n_points
```
