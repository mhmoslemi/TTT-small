import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def refine_h(h, n_points_new):
        h_new = np.zeros(n_points_new)
        dx = 2.0 / n_points_new
        scale = 1.0
        if n_points_new != n_points:
            # Interpolate from previous h to new grid
            h_new = np.interp(np.linspace(0, 2, n_points_new), np.linspace(0, 2, n_points), h)
            h_new = np.clip(h_new, 0, 1)
            scale = (n_points_new / 2) / np.sum(h_new)
        h_new = h_new * scale
        h_new = np.clip(h_new, 0, 1)
        return h_new
    
    def local_search(h, n_points):
        h = h.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, tol=1e-6, options={'maxiter': 100})
        h = result.x
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Coarse grid optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.differential_evolution(objective, bounds, popsize=40, mutation=(0.8, 1.0), 
                                       recombination=0.9, strategy='best1bin', tol=1e-5,
                                       maxiter=300, disp=False, polish=True, seed=seed)
    best_h = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time
    
    # Fine grid refinement
    if elapsed < budget_s:
        n_points_new = 90
        h_fine = refine_h(best_h, n_points_new)
        bounds_fine = [(0.0, 1.0) for _ in range(n_points_new)]
        result_fine = opt.differential_evolution(objective, bounds_fine, popsize=20, mutation=(1.0, 1.0), 
                                               recombination=0.9, strategy='best1bin', tol=1e-6,
                                               maxiter=200, disp=False, polish=True, seed=seed)
        best_h = result_fine.x
        best_c5 = result_fine.fun
        elapsed = time.time() - start_time
    
    # Local search refinement
    if elapsed < budget_s:
        h_local = local_search(best_h, n_points)
        best_c5 = objective(h_local)
        best_h = h_local
    
    # Final check
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
        h = best_h
        best_c5 = objective(h)
    
    # Final feasibility check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points)
