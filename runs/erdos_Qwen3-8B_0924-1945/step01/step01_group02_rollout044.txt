```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    h = h.astype(np.float64)

    def build_A(k):
        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = (i + k) % n_points
            A[i, j] += dx
            j = (i - k) % n_points
            A[i, j] += dx
        return A

    A_list = [build_A(k) for k in range(1, n_points)]

    h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
    C = cp.Variable()

    constraints = []
    constraints.append(cp.sum(h_var) * dx == 1.0)
    for A in A_list:
        constraints.append(h_var.T @ A @ h_var >= 1.0 - C)

    prob = cp.Problem(cp.Minimize(C), constraints)
    start_time = time.time()
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
    except:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=100)

    h_opt = h_var.value
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale

    corr = np.correlate(h_opt, 1.0 - h_opt, mode="full")
    c5_bound = np.max(corr * dx)

    return h_opt.tolist(), c5_bound, n_points
```
