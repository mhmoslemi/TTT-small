import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.stats import uniform

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constrain_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    def optimize_with_de(pop_size=20, max_gen=100, mutation=0.5, crossover=0.9):
        population = np.random.uniform(0, 1, (pop_size, n_points))
        population = population / np.sum(population, axis=1, keepdims=True) * n_points / 2
        for gen in range(max_gen):
            if time.time() - start_time > budget_s:
                break
            new_population = []
            for i in range(pop_size):
                a, b, c = np.random.choice(pop_size, 3, replace=False)
                mutant = population[a] + mutation * (population[b] - population[c])
                mutant = np.clip(mutant, 0, 1)
                mutant = mutant / np.sum(mutant) * n_points / 2
                if np.random.rand() < crossover:
                    mutant = np.where(np.random.rand(n_points) < 0.5, mutant, population[i])
                new_population.append(mutant)
            population = np.array(new_population)
            population = np.vstack([population, best_h])
            population = np.unique(population, axis=0)
            population = population[:pop_size]
            for candidate in population:
                current_c5 = evaluate_candidate(candidate)
                if current_c5 < best_c5:
                    best_h = candidate.copy()
                    best_c5 = current_c5
        return best_h, best_c5

    best_h, best_c5 = optimize_with_de()
    if time.time() - start_time < budget_s:
        for _ in range(10):
            if time.time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    def refine_with_sqp(iterations=15):
        h = best_h.copy()
        for _ in range(iterations):
            if time.time() - start_time > budget_s:
                break
            def obj_func(x):
                x = np.clip(x, 0, 1)
                x = x / np.sum(x) * n_points / 2
                return evaluate_candidate(x)

            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            res = minimize(obj_func, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = res.x
                current_c5 = evaluate_candidate(h)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
            else:
                break
        return best_h, best_c5

    best_h, best_c5 = refine_with_sqp()
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
