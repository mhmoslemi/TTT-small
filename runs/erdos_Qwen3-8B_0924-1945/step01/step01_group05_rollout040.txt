import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        idx = np.linspace(0, n_points-1, n_points // 3 + 1).astype(int)
        h[idx] = 1.0
        h = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, len(idx)), h[idx])
        h = (h * n_points / 2).astype(np.float64)
        h = h / np.sum(h) * n_points / 2
    
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    population_size = 50
    generations = 80
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_ratio = 0.3
    mutation_samples = 10
    best_h = h.copy()
    best_c5 = objective(h)
    
    for gen in range(generations):
        population = [np.random.rand(n_points) for _ in range(population_size)]
        for p in population:
            p = (p - np.min(p)) / (np.max(p) - np.min(p))
            p = (p * n_points / 2).astype(np.float64)
            p = p / np.sum(p) * n_points / 2
        population = [np.clip(p, 0, 1) for p in population]
        population = [p * target_sum / np.sum(p) for p in population]
        
        for p in population:
            if np.random.rand() < mutation_rate:
                idx = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
                p[idx] = np.random.rand(len(idx))
        
        for p in population:
            if np.random.rand() < crossover_rate:
                idx = np.random.choice(n_points, size=int(n_points * 0.2), replace=False)
                p[idx] = np.random.rand(len(idx))
        
        fitness = [objective(p) for p in population]
        sorted_idx = np.argsort(fitness)
        population = [population[i] for i in sorted_idx]
        fitness = [fitness[i] for i in sorted_idx]
        population = population[:int(population_size * elite_ratio)] + [np.random.rand(n_points) for _ in range(mutation_samples)]
        population = [np.clip(p, 0, 1) for p in population]
        population = [p * target_sum / np.sum(p) for p in population]
        
        for p in population:
            if np.random.rand() < mutation_rate:
                idx = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
                p[idx] = np.random.rand(len(idx))
        
        for p in population:
            if np.random.rand() < crossover_rate:
                idx = np.random.choice(n_points, size=int(n_points * 0.2), replace=False)
                p[idx] = np.random.rand(len(idx))
        
        for p in population:
            if objective(p) < best_c5:
                best_c5 = objective(p)
                best_h = p.copy()
        
        elapsed = time.time() - time.time()
        if elapsed >= budget_s:
            break
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
