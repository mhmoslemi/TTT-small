import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Initial guess
    if 'initial_h_values' in globals():
        h = np.copy(initial_h_values)
    else:
        h = np.zeros(42)
        h[::5] = 1.0
        h = np.clip(h, 0, 1)
        h = h * (42 / 2) / np.sum(h)
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Function to evaluate C5
    def objective(x):
        h_vals = np.clip(x, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Constraint function
    def constraint(x):
        return np.sum(x) - target_sum
    
    # Bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization using differential evolution
    result = opt.differential_evolution(
        objective,
        bounds,
        args=(),
        popsize=15,
        strategy='best1bin',
        maxiter=200,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        disp=False,
        polish=True,
        callback=None
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    c5_bound = objective(best_h)
    
    return (best_h, c5_bound, n_points)
