import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial solution (if available, otherwise create a structured one)
    n_points = 100
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Generate a structured initial guess: alternating 1s and 0s with a small perturbation
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = h * (n_points / 2) / np.sum(h)
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure sum(h) = n_points / 2
    target_sum = n_points / 2
    h = h * target_sum / np.sum(h)
    
    def objective(x):
        h_flat = x.copy()
        h_flat = np.clip(h_flat, 0, 1)
        h_flat = h_flat * target_sum / np.sum(h_flat)
        corr = np.correlate(h_flat, 1 - h_flat, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    # Bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Differential Evolution setup
    result = opt.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=100,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        polish=True,
        disp=False
    )
    
    elapsed = time.time() - time.time()
    if elapsed >= budget_s:
        return (h, objective(h), n_points)
    
    best_h = result.x
    best_c5 = result.fun
    
    # Post-processing
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
