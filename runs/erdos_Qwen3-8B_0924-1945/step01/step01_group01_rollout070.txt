import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with initial_h_values if available, else generate a random sample
    n_points = 90
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure constraints are satisfied
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Define objective and gradient
    dx = 2.0 / n_points
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def gradient(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        max_index = np.argmax(correlate_vals)
        lag = max_index - n_points + 1
        if lag < 0 or lag >= n_points:
            return np.zeros_like(h_vec)
        grad = np.zeros_like(h_vec)
        for i in range(n_points):
            x = i * dx
            x_plus_k = x + lag * dx
            if x_plus_k >= 2:
                grad[i] = 0
            else:
                idx = int(x_plus_k / dx)
                if idx < 0 or idx >= n_points:
                    grad[i] = 0
                else:
                    grad[i] = (1 - h[idx]) * dx
        return grad
    
    # Optimize with L-BFGS-B
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(objective, h, method='L-BFGS-B', jac=gradient, bounds=bounds,
                          tol=1e-5, options={'maxiter': 500, 'disp': False})
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Fallback to differential evolution if no improvement
    if best_c5 > objective(h) + 1e-6:
        result = opt.differential_evolution(objective, bounds, strategy='best1bin',
                                          popsize=30, mutation=(0.9, 1.0), recombination=0.7,
                                          maxiter=300, tol=1e-5, disp=False, seed=seed)
        best_h = result.x
        best_c5 = result.fun
    
    # Final check and evaluation
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
    
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points)
