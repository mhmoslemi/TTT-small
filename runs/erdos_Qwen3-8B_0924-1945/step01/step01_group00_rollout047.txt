import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
        h = h / np.sum(h) * n_points / 2
        h = np.round(h).astype(float)

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    bounds = [(0, 1) for _ in range(n_points)]

    def objective(x):
        x = np.clip(x, 0, 1)
        x = x / np.sum(x) * n_points / 2
        return compute_c5(x)

    start_time = time()
    result = differential_evolution(objective, bounds, strategy='best1bin', popsize=15, mutation=0.5, recombination=0.9, tol=1e-5, maxiter=500, callback=lambda x: None, disp=False, polish=True, seed=seed)

    best_h = result.x
    best_c5 = compute_c5(best_h)

    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 0:
        for _ in range(10):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    return (best_h, best_c5, n_points)
