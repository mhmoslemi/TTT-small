import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    h = h.astype(np.float64)
    h = np.clip(h, 0.0, 1.0)

    t0 = time.time()

    def compute_constraints(h, dx, n_points):
        n_lags = n_points - 1
        A = []
        for k in range(1, n_points):
            A_k = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = (i + k) % n_points
                A_k[i, j] = dx
                j = (i - k) % n_points
                A_k[i, j] = dx
            A.append(A_k)
        return A

    A_list = compute_constraints(h, dx, n_points)

    h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
    C = cp.Variable()

    constraints = [
        cp.sum(h_var * dx) == target_sum,
        cp.sum(cp.multiply(h_var, h_var)) >= 1.0 - C,
    ]

    for A_k in A_list:
        constraints.append(cp.quad_form(h_var, A_k) >= 1.0 - C)

    prob = cp.Problem(cp.Minimize(C), constraints)
    prob.solve(solver=cp.ECOS, verbose=False)

    elapsed = time.time() - t0
    if elapsed >= budget_s:
        h_opt = h
        c5_bound = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
        return h.tolist(), c5_bound, n_points

    h_opt = h_var.value
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale

    c5_bound = np.max(np.correlate(h_opt, 1.0 - h_opt, mode="full")) * dx
    return h_opt.tolist(), c5_bound, n_points
