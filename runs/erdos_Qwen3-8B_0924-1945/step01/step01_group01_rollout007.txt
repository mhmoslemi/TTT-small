import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Objective function
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Optimization using differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.differential_evolution(objective, bounds, popsize=50, mutation=(0.8, 1.0), 
                                       recombination=0.9, strategy='best1bin', tol=1e-5,
                                       maxiter=200, disp=False, polish=True, seed=seed)
    
    best_h = result.x
    best_c5 = result.fun
    
    # Gradient-based refinement
    def grad_objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        best_corr = np.max(correlate_vals)
        best_k = np.argmax(correlate_vals)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if best_k + i < len(correlate_vals):
                grad[i] = (1 - h[i]) * (2.0 / n_points)
        return best_corr, grad
    
    def refine_with_lbfgs(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        best_corr, grad = grad_objective(h)
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.minimize(lambda x: grad_objective(x)[0], h, method='L-BFGS-B', jac=grad_objective, bounds=bounds)
        return result.x
    
    best_h = refine_with_lbfgs(best_h)
    best_c5 = grad_objective(best_h)[0]
    
    # Check time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
        h = best_h
        best_c5 = grad_objective(h)[0]
    
    # Final check
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = grad_objective(h)[0]
    
    return (h.tolist(), final_c5, n_points)
