import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Start with initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check feasibility
    def is_feasible(h_vec):
        return np.all(h_vec >= 0) and np.all(h_vec <= 1) and np.isclose(np.sum(h_vec), n_points / 2, atol=1e-6)
    
    if not is_feasible(h):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    def objective(h_vec):
        nonlocal best_c5
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(correlate_vals)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    def gradient(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_idx = np.argmax(correlate_vals)
        k = max_idx - n_points + 1
        if k < 0 or k >= n_points:
            return np.zeros_like(h)
        grad = np.zeros_like(h)
        grad[np.arange(n_points) - k] += (1 - h[np.arange(n_points) - k]) * (2.0 / n_points)
        return grad
    
    def optimize_with_lbfgs():
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.minimize(objective, best_h, method='L-BFGS-B', jac=gradient, bounds=bounds, tol=1e-6,
                              options={'maxiter': 100, 'disp': False})
        if result.success:
            best_h = result.x
            best_c5 = compute_c5(best_h)
        return result.success
    
    def optimize_with_de():
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(objective, bounds, popsize=30, mutation=(0.9, 1.0), 
                                           recombination=0.7, strategy='best1bin', tol=1e-5,
                                           maxiter=200, disp=False, seed=seed)
        best_h = result.x
        best_c5 = result.fun
        return True
    
    def refine_resolution():
        nonlocal n_points, best_h, best_c5
        n_points *= 2
        dx = 2.0 / n_points
        h_refined = np.zeros(n_points)
        h_refined[:len(best_h)] = best_h * (n_points / 2) / (len(best_h) / 2)
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        best_h = h_refined
        best_c5 = compute_c5(best_h)
        return True
    
    # Perform gradient-based optimization
    success = optimize_with_lbfgs()
    if success:
        pass
    else:
        success = optimize_with_de()
    
    # Refine with higher resolution
    if success and time.time() - start_time < budget_s - 200:
        if refine_resolution():
            pass
    
    # Fallback to DE if needed
    if not success or time.time() - start_time > budget_s - 100:
        success = optimize_with_de()
    
    # Final check
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
    
    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
