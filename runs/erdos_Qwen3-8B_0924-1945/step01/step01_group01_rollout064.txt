import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define the number of points and sample size
    n_points = 120
    dx = 2.0 / n_points
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Function to evaluate C5
    def evaluate_c5(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    # Precompute valid lags
    max_lag = 2 * n_points - 2
    valid_lags = np.arange(max_lag + 1)
    
    # Function to formulate QP problem
    def formulate_qp(h):
        n = n_points
        A = np.zeros((len(valid_lags), n))
        b = np.zeros(len(valid_lags))
        
        for i, lag in enumerate(valid_lags):
            for j in range(n):
                k = (j + lag) % n
                A[i, j] = h[j] * h[k]
        
        A = A.T
        c = np.zeros(n)
        c[0] = 1
        c[1] = -1
        c = np.concatenate([c, np.zeros(len(valid_lags))])
        G = np.zeros((n + len(valid_lags), n))
        h_bounds = np.zeros(n)
        h_bounds[0] = 1
        h_bounds[1] = -1
        h_bounds = np.concatenate([h_bounds, np.zeros(len(valid_lags))])
        G[0, :] = 1
        G[1, :] = -1
        G[2:, :] = A.T
        
        return G, h_bounds, c, A, b
    
    # Optimization function
    def optimize():
        nonlocal h, start_time
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            return
        
        try:
            G, h_bounds, c, A, b = formulate_qp(h)
            sol = opt.linprog(c, G, h_bounds, bounds=[(0, 1) for _ in range(n_points)], method='highs')
            if sol.success:
                h = sol.x
        except:
            pass
        
        # Fallback to differential evolution if needed
        if not np.isfinite(evaluate_c5(h)):
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            result = opt.differential_evolution(evaluate_c5, bounds, popsize=20, mutation=(0.5, 1.0),
                                               recombination=0.9, strategy='best1bin', tol=1e-5,
                                               maxiter=200, disp=False, polish=True, seed=seed)
            h = result.x
            c5 = result.fun
        
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            return
    
    # Run the optimization
    optimize()
    
    # Final checks
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    c5 = evaluate_c5(h)
    
    return (h.tolist(), c5, n_points)
