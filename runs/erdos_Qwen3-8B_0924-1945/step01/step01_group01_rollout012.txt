import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Default parameters
    n_points = 90
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def build_constraints(n_points):
        num_lags = 2 * n_points - 1
        idx = np.indices((n_points, n_points))
        lag_idx = idx[0] - idx[1]
        valid_lags = np.where((lag_idx >= 0) & (lag_idx < n_points) & (lag_idx > 0))
        valid_lags = valid_lags[0]
        return valid_lags
    
    valid_lags = build_constraints(n_points)
    
    def evaluate(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def qp_objective_and_constraints(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        n = len(h)
        G = np.zeros((len(valid_lags), n))
        for i, lag in enumerate(valid_lags):
            pos = np.arange(n)
            idx = np.mod(pos - lag, n)
            G[i, idx] = h[idx]
        
        h_transposed = np.outer(h, h)
        h_transposed = h_transposed.flatten()
        h_transposed = h_transposed[valid_lags]
        
        mu = 1.0 - np.min(h_transposed)
        return mu, G, -np.ones((len(valid_lags), 1)), np.ones((len(valid_lags), 1)), np.zeros(n), 1.0
    
    def cvxqp_solver(n_points):
        n = n_points
        h = np.random.rand(n)
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        
        try:
            G, c, A, b, l, u = qp_objective_and_constraints(h)
            c = np.array([1.0])
            G = G.T
            A = A.T
            b = b.T
            l = l.T
            u = u.T
            h = np.zeros(n)
            sol = opt.minimize(lambda x: 0.5 * x @ G @ x + c @ x, h, bounds=[(0, 1) for _ in range(n)], method='SLSQP', constraints=[{'type': 'ineq', 'fun': lambda x: A @ x - b, 'jac': lambda x: A.T @ np.ones(n)}])
            h = sol.x
            h = np.clip(h, 0, 1)
            h = h * (n / 2) / np.sum(h)
            return h
        except:
            return h
    
    # Try QP solver
    try:
        h = cvxqp_solver(n_points)
        result = evaluate(h)
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            print("Time budget exceeded, returning initial solution")
            h = initial_h_values if 'initial_h_values' in globals() else h
            result = evaluate(h)
        return (h.tolist(), result, n_points)
    except:
        # Fallback to differential evolution
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(lambda h: evaluate(h), bounds, popsize=40, mutation=(0.5, 1.0), 
                                            recombination=0.9, strategy='best1bin', tol=1e-5,
                                            maxiter=200, disp=False, polish=True, seed=seed)
        best_h = result.x
        best_c5 = result.fun
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            print("Time budget exceeded, returning initial solution")
            h = initial_h_values if 'initial_h_values' in globals() else best_h
            best_c5 = evaluate(h)
        return (best_h.tolist(), best_c5, n_points)
