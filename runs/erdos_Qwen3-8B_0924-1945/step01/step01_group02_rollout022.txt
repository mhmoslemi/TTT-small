```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            scale = 1.0
        else:
            scale = target_sum / sum_h
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5_val = np.max(corr * dx)
        return c5_val

    def build_qp(h_initial):
        h = cp.Variable(n_points)
        C = cp.Variable()
        constraints = []
        constraints.append(cp.sum(h) == target_sum)
        constraints.append(cp.constraints.NonNeg(h))
        constraints.append(cp.constraints.NonPos(-h))
        for k in range(1, n_points):
            A = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = (i + k) % n_points
                A[i, j] = dx
                j = (i - k + n_points) % n_points
                A[i, j] = dx
            constraints.append(h @ A @ h >= 1.0 - C)
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        try:
            result = prob.solve(solver=cp.ECOS, eps=1e-8)
            if result is not None:
                h_opt = h.value
                C_opt = C.value
                return h_opt, C_opt
        except:
            pass
        return None, None

    best_h = h
    best_c5 = evaluate(h)
    time_used = 0
    num_restarts = 60

    for i in range(num_restarts):
        if time_used >= budget_s:
            break
        start_time = time.time()
        perturbation = np.random.normal(0, 0.05, n_points)
        h_perturbed = h + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (target_sum / np.sum(h_perturbed))
        h_opt, c5_opt = build_qp(h_perturbed)
        if h_opt is not None:
            c5_val = evaluate(h_opt)
            if c5_val < best_c5:
                best_h = h_opt
                best_c5 = c5_val
        time_used += time.time() - start_time

    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    if sum_h != 0:
        best_h = best_h * (target_sum / sum_h)
    best_c5 = evaluate(best_h)
    return best_h.tolist(), best_c5, n_points
```
