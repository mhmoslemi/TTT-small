```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h))
    
    def calculate_c5(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    def compute_constraints(h):
        n = n_points
        lags = np.arange(1, n)
        n_lags = len(lags)
        A = []
        for k in lags:
            row = np.zeros(n)
            for i in range(n):
                j = (i + k) % n
                row[i] = dx
                row[j] = dx
            A.append(row)
        return np.array(A)
    
    A = compute_constraints(h)
    h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
    C = cp.Variable()
    constraints = [
        cp.sum(h_var) == target_sum,
        cp.sum(cp.multiply(A[i], h_var) @ h_var) >= 1 - C for i in range(len(A))
    ]
    obj = cp.Minimize(C)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, eps=1e-8, max_iter=1000)
    
    best_h = h_var.value
    best_c5 = calculate_c5(best_h)
    
    for _ in range(19):
        h_perturb = best_h + np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        h_perturb = h_perturb * (target_sum / np.sum(h_perturb))
        A_perturb = compute_constraints(h_perturb)
        h_var_perturb = cp.Variable(n_points, bounds=(0.0, 1.0))
        C_perturb = cp.Variable()
        constraints_perturb = [
            cp.sum(h_var_perturb) == target_sum,
            cp.sum(cp.multiply(A_perturb[i], h_var_perturb) @ h_var_perturb) >= 1 - C_perturb for i in range(len(A_perturb))
        ]
        obj_perturb = cp.Minimize(C_perturb)
        prob_perturb = cp.Problem(obj_perturb, constraints_perturb)
        try:
            prob_perturb.solve(solver=cp.ECOS, eps=1e-8, max_iter=1000)
            if prob_perturb.status == "OPTIMAL":
                h_perturb_opt = h_var_perturb.value
                c5_perturb = calculate_c5(h_perturb_opt)
                if c5_perturb < best_c5:
                    best_h = h_perturb_opt
                    best_c5 = c5_perturb
        except:
            pass
    
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (target_sum / np.sum(best_h))
    return best_h.tolist(), best_c5, n_points
```
