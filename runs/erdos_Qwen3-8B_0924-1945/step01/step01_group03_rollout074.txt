import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if hasattr(initial_h_values, '__len__') and len(initial_h_values) == n_points:
        h = np.copy(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = (h / np.sum(h)) * target_sum
        h = np.clip(h, 0, 1)
    
    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        return compute_c5(h_vals)

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    best_h = h
    best_c5 = compute_c5(best_h)
    best_c5 = np.inf if np.isnan(best_c5) else best_c5
    for _ in range(3):
        h_rand = np.random.rand(n_points)
        h_rand = (h_rand / np.sum(h_rand)) * target_sum
        h_rand = np.clip(h_rand, 0, 1)
        c5_rand = compute_c5(h_rand)
        if c5_rand < best_c5:
            best_h = h_rand
            best_c5 = c5_rand

    result = opt.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 200, 'ftol': 1e-6, 'eps': 1e-6}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = (best_h / np.sum(best_h)) * target_sum
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_points)
