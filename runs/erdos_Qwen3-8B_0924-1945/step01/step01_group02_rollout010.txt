import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        ones_needed = np.ceil(n_points / 2)
        indices = np.random.choice(n_points, size=int(ones_needed), replace=False)
        h[indices] = 1.0
        h = h * (target_sum / np.sum(h))

    def evaluate_c5(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def build_constraints(n_points, dx):
        n_lags = n_points - 1
        constraints = []
        for k in range(1, n_points):
            A = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    A[i][j] = dx
                j = i - k
                if j >= 0:
                    A[i][j] = dx
            constraints.append(A)
        return constraints

    constraints = build_constraints(n_points, dx)

    def solve_qp(h_guess, c5_guess, max_iter=100):
        h = cp.Variable(n_points, bounds=(0.0, 1.0))
        c = cp.Variable(numeric=True)
        obj = cp.Minimize(c)
        cons = []
        cons.append(cp.sum(h) == target_sum)
        for A in constraints:
            cons.append(h.T @ A @ h >= 1.0 - c)
        prob = cp.Problem(obj, cons)
        try:
            result = prob.solve(solver=cp.ECOS, max_iter=max_iter, eps=1e-8)
            if result is not None:
                return h.value, c.value
        except:
            pass
        return h_guess, c5_guess

    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    elapsed = 0
    start_time = time.time()

    for _ in range(20):
        if elapsed >= budget_s:
            break
        h_guess = h.copy()
        h_guess += np.random.uniform(-0.05, 0.05, size=n_points)
        h_guess = np.clip(h_guess, 0.0, 1.0)
        h_guess = h_guess * (target_sum / np.sum(h_guess))
        h_opt, c5_opt = solve_qp(h_guess, evaluate_c5(h_guess))
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = evaluate_c5(h_opt)
        if c5_bound < best_c5:
            best_h = h_opt
            best_c5 = c5_bound
        elapsed = time.time() - start_time

    return best_h.tolist(), best_c5, n_points
