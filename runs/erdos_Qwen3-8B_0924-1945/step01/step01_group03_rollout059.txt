import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 64
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if hasattr(globals(), 'initial_h_values'):
        h = np.copy(initial_h_values)
        h = np.clip(h, 0, 1)
        h /= np.sum(h)
        h *= target_sum
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sin(np.pi * x)
        h = np.clip(h, 0, 1)
        h /= np.sum(h)
        h *= target_sum

    def evaluate_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        return evaluate_c5(h_vals)

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 200, 'ftol': 1e-9, 'eps': 1e-6}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h /= np.sum(best_h)
    best_h *= target_sum
    c5_bound = evaluate_c5(best_h)
    return (best_h, c5_bound, n_points)
