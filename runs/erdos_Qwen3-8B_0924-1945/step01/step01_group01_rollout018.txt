import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with provided initial_h_values or generate random if not available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(90)
        h = np.clip(h, 0, 1)
        h = h * 45.0 / np.sum(h)
    
    # Ensure constraints
    def clip_rescale(h):
        h = np.clip(h, 0, 1)
        h = h * 45.0 / np.sum(h)
        return h
    
    h = clip_rescale(h)
    
    def objective(h_vec):
        h = clip_rescale(h_vec)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / len(h))
        return np.max(correlate_vals)
    
    # Gradient-based optimization with L-BFGS-B
    def gradient(h_vec):
        h = clip_rescale(h_vec)
        correlate_vals = np.correlate(h, 1 - h, mode="full")
        max_index = np.argmax(correlate_vals)
        k = max_index - len(h) + 1
        if k >= len(h):
            k = len(h) - 1
        if k < 0:
            k = 0
        grad = np.zeros_like(h)
        dx = 2.0 / len(h)
        for i in range(len(h)):
            x = i * dx
            if k >= 0 and k < len(h):
                val = 1 - h[k]
                grad[i] = val * dx
        return grad
    
    bounds = [(0.0, 1.0) for _ in range(90)]
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, jac=gradient,
                          options={'maxiter': 500, 'ftol': 1e-5, 'gtol': 1e-5, 'eps': 1e-8})
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best found")
        return (best_h.tolist(), best_c5, 90)
    
    # Final check and evaluation
    best_h = clip_rescale(best_h)
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, 90)
