import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def gradient(h_vec, k):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        grad = np.zeros_like(h)
        x_indices = np.arange(n_points)
        k_indices = (x_indices + k) % n_points
        grad[x_indices] = (1 - h[k_indices]) * (2.0 / n_points)
        return grad
    
    def update_h(h, delta, k):
        h_new = h.copy()
        x_indices = np.arange(n_points)
        k_indices = (x_indices + k) % n_points
        h_new[x_indices] = np.clip(h_new[x_indices] + delta * (1 - h[k_indices]), 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Coarse-grid global optimization with DE
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.differential_evolution(objective, bounds, popsize=50, mutation=(0.8, 1.0), 
                                       recombination=0.9, strategy='best1bin', tol=1e-5,
                                       maxiter=200, disp=False, polish=True, seed=seed)
    best_h = result.x
    best_c5 = result.fun
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = objective(best_h)
    elapsed = time.time() - start_time
    
    # Adaptive gradient refinement
    if elapsed < budget_s:
        k_max = np.argmax(np.correlate(best_h, 1 - best_h, mode="full")) - n_points + 1
        k = np.clip(k_max, 0, n_points - 1)
        for _ in range(300):
            grad = gradient(best_h, k)
            h_update = best_h - 0.01 * grad
            h_update = np.clip(h_update, 0, 1)
            h_update = h_update * (n_points / 2) / np.sum(h_update)
            new_c5 = objective(h_update)
            if new_c5 < best_c5:
                best_h = h_update
                best_c5 = new_c5
            else:
                break
            elapsed = time.time() - start_time
            if elapsed > budget_s:
                break
    
    # Multi-scale refinement
    if elapsed < budget_s:
        n_points = 90
        dx = 2.0 / n_points
        h_new = np.zeros(n_points)
        for i in range(n_points):
            h_new[i] = best_h[int(i * n_points / n_points)]
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        best_c5 = objective(h_new)
        best_h = h_new
        k_max = np.argmax(np.correlate(best_h, 1 - best_h, mode="full")) - n_points + 1
        k = np.clip(k_max, 0, n_points - 1)
        for _ in range(300):
            grad = gradient(best_h, k)
            h_update = best_h - 0.01 * grad
            h_update = np.clip(h_update, 0, 1)
            h_update = h_update * (n_points / 2) / np.sum(h_update)
            new_c5 = objective(h_update)
            if new_c5 < best_c5:
                best_h = h_update
                best_c5 = new_c5
            else:
                break
            elapsed = time.time() - start_time
            if elapsed > budget_s:
                break
    
    # Final feasibility check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
