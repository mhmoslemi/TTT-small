import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial solution if available
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Constraint handling
    def clip_and_rescale(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Objective function for correlation-based C₅
    def objective(h_vec):
        h = clip_and_rescale(h_vec)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Stage 1: Coarse global optimization with differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.differential_evolution(
        objective, bounds, popsize=50, mutation=(0.8, 1.0),
        recombination=0.9, strategy='best1bin', tol=1e-5,
        maxiter=200, disp=False, polish=True, seed=seed
    )
    
    best_h = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time
    
    # Stage 2: Adaptive gradient refinement with L-BFGS-B
    def grad_func(h_vec):
        h = clip_and_rescale(h_vec)
        correlate_vals = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(correlate_vals)
        dx = 2.0 / n_points
        grad = np.zeros_like(h)
        if max_idx < len(h):
            grad[max_idx - len(h) + 1: max_idx + 1] += (1 - h[max_idx - len(h) + 1: max_idx + 1]) * dx
        return grad
    
    result = opt.minimize(
        objective, best_h, method='L-BFGS-B', bounds=bounds,
        jac=grad_func, tol=1e-6, options={'maxiter': 300, 'disp': False}
    )
    best_h = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time
    
    # Stage 3: Multi-scale refinement with increased resolution
    n_points = 90
    dx = 2.0 / n_points
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, 45), best_h)
    h = clip_and_rescale(h)
    
    def objective_new(h_vec):
        h = clip_and_rescale(h_vec)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    result = opt.minimize(
        objective_new, h, method='L-BFGS-B', bounds=[(0.0, 1.0) for _ in range(n_points)],
        jac=lambda x: grad_func(x) * dx, tol=1e-6, options={'maxiter': 300, 'disp': False}
    )
    best_h = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time
    
    # Final check for time budget
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution from refinement")
        h = best_h
        best_c5 = objective_new(h)
    
    # Final feasibility and evaluation
    h = clip_and_rescale(best_h)
    final_c5 = objective_new(h)
    
    return (h.tolist(), final_c5, n_points)
