import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_guess = np.sin(np.linspace(0, 2 * np.pi, n_points)) * 0.5 + 0.5
    initial_guess = np.clip(initial_guess, 0, 1)
    initial_guess /= np.sum(initial_guess)
    initial_guess *= target_sum
    
    def objective(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - target_sum

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h /= np.sum(h)
        h *= target_sum
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5, h

    def generate_guesses(base_guess):
        guesses = []
        for i in range(10):
            noise = np.random.uniform(-0.15, 0.15, n_points)
            guess = base_guess + noise
            guess = np.clip(guess, 0, 1)
            guess /= np.sum(guess)
            guess *= target_sum
            guesses.append(guess)
        return guesses

    guesses = generate_guesses(initial_guess)
    best_h = initial_guess
    best_c5 = evaluate(best_h)[0]

    from scipy.optimize import differential_evolution

    def optimize_guess(guess):
        bounds = [(0, 1) for _ in range(n_points)]
        result = differential_evolution(
            objective,
            bounds,
            args=(),
            strategy='best1bin',
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            maxiter=200,
            callback=None,
            disp=False,
            polish=True,
            vectorized=True,
            x0=guess
        )
        result = result.x
        result = np.clip(result, 0, 1)
        result /= np.sum(result)
        result *= target_sum
        c5, _ = evaluate(result)
        return result, c5

    for i, guess in enumerate(guesses):
        elapsed = 0
        start = np.datetime64('now')
        try:
            h, c5 = optimize_guess(guess)
            if c5 < best_c5:
                best_h, best_c5 = h, c5
            elapsed = np.datetime64('now') - start
            if elapsed.total_seconds() > 250:
                break
        except:
            continue

    return (best_h, best_c5, n_points)
