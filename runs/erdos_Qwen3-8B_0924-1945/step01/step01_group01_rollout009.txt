import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Stage 1: Parent Initialization
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Stage 2: Critical Lag Identification and Gradient Refinement
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def compute_gradient(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_index = np.argmax(correlate_vals)
        lag = max_index - n_points + 1
        grad = np.zeros(n_points)
        for i in range(n_points):
            dx = 2.0 / n_points
            if 0 <= i - lag < n_points:
                grad[i] = dx * (1 - h[i - lag])
        return grad
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.LBFGS(objective, bounds=bounds, jac=compute_gradient, 
                      maxiter=100, tol=1e-6, disp=False, 
                      options={'gtol': 1e-6, 'maxcor': 50})
    best_h = result.x
    best_c5 = objective(best_h)
    current_c5 = best_c5
    elapsed = time.time() - start_time

    # Stage 3: Multi-scale Refinement
    if elapsed < budget_s - 200:
        n_points = 90
        dx = 2 / n_points
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, 45), best_h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        def refine_objective(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(correlate_vals)

        def refine_gradient(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
            max_index = np.argmax(correlate_vals)
            lag = max_index - n_points + 1
            grad = np.zeros(n_points)
            for i in range(n_points):
                dx = 2.0 / n_points
                if 0 <= i - lag < n_points:
                    grad[i] = dx * (1 - h[i - lag])
            return grad

        result = opt.LBFGS(refine_objective, bounds=bounds, jac=refine_gradient, 
                          maxiter=100, tol=1e-6, disp=False, 
                          options={'gtol': 1e-6, 'maxcor': 50})
        best_h = result.x
        best_c5 = refine_objective(best_h)
        current_c5 = best_c5
        elapsed = time.time() - start_time

    # Stage 4: Fallback to Differential Evolution
    if elapsed < budget_s - 100:
        n_points = 90
        dx = 2 / n_points
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, 45), best_h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        def de_objective(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(correlate_vals)

        result = opt.differential_evolution(de_objective, bounds, 
                                           popsize=30, mutation=(0.9, 1.0),
                                           recombination=0.7, strategy='best1bin', 
                                           tol=1e-5, maxiter=200, disp=False, 
                                           polish=True, seed=seed)
        best_h = result.x
        best_c5 = de_objective(best_h)
        current_c5 = best_c5

    # Final checks and return
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    return (h.tolist(), final_c5, n_points)
