import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage 1: Parent Initialization
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check feasibility
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def grad_objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_idx = np.argmax(correlate_vals)
        k = max_idx - n_points + 1
        k = np.clip(k, 0, n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            x = i * (2.0 / n_points)
            if k + i <= n_points - 1:
                grad[i] = (1 - h[i]) * (2.0 / n_points)
            if i - k >= 0:
                grad[i] -= h[i - k] * (2.0 / n_points)
        return grad
    
    # Stage 2: Gradient refinement
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(objective, h, method='L-BFGS-B', jac=grad_objective, bounds=bounds,
                          tol=1e-6, max_iter=100, options={'maxfun': 1000})
    best_h = result.x
    best_c5 = result.fun
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = objective(best_h)
    
    # Stage 3: Multi-scale refinement
    n_points = 90
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[::2] = 1.0
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    def objective_90(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def grad_objective_90(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        max_idx = np.argmax(correlate_vals)
        k = max_idx - n_points + 1
        k = np.clip(k, 0, n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if k + i <= n_points - 1:
                grad[i] = (1 - h[i]) * dx
            if i - k >= 0:
                grad[i] -= h[i - k] * dx
        return grad
    
    result = opt.minimize(objective_90, h, method='L-BFGS-B', jac=grad_objective_90, bounds=bounds,
                          tol=1e-6, max_iter=100, options={'maxfun': 1000})
    best_h = result.x
    best_c5 = result.fun
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = objective_90(best_h)
    
    # Stage 4: Fallback to differential evolution
    if best_c5 > 0.383:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.differential_evolution(objective_90, bounds, popsize=30, mutation=(0.9, 1.0),
                                           recombination=0.7, strategy='best1bin', tol=1e-5,
                                           maxiter=200, disp=False, polish=True, seed=seed)
        best_h = result.x
        best_c5 = result.fun
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        best_c5 = objective_90(best_h)
    
    # Final checks
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution")
    
    # Final validation
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective_90(h)
    
    return (h.tolist(), final_c5, n_points)
