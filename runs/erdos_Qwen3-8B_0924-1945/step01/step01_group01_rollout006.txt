import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    if not (np.all(h >= 0) and np.all(h <= 1) and np.isclose(np.sum(h), n_points / 2, atol=1e-6)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def evaluate(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def objective(h_vec):
        return evaluate(h_vec)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Stage 2: Coarse Global Search with DE
    result_de = opt.differential_evolution(
        objective,
        bounds,
        popsize=50,
        mutation=(0.8, 1.0),
        recombination=0.9,
        strategy='best1bin',
        tol=1e-5,
        maxiter=200,
        disp=False,
        polish=True,
        seed=seed
    )
    best_c5 = result_de.fun
    best_h = result_de.x
    
    # Stage 3: Local Refinement with L-BFGS-B
    def gradient(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_index = np.argmax(correlate_vals)
        k = max_index - n_points + 1
        if k < 0:
            k = 0
        grad = np.zeros_like(h)
        for i in range(n_points):
            x = i * (2.0 / n_points)
            if x + k < 2.0:
                grad[i] = (1 - h[int((x + k) * (n_points / 2))]) * (2.0 / n_points)
        return grad
    
    result_lbfgs = opt.minimize(
        objective,
        best_h,
        method='L-BFGS-B',
        bounds=bounds,
        jac=gradient,
        maxiter=300,
        tol=1e-6,
        options={'disp': False}
    )
    best_c5 = result_lbfgs.fun
    best_h = result_lbfgs.x
    
    # Stage 4: Multi-scale Refinement
    n_points_refine = 90
    dx_refine = 2.0 / n_points_refine
    h_refine = np.interp(np.linspace(0, 2, n_points_refine), np.linspace(0, 2, n_points), best_h)
    h_refine = np.clip(h_refine, 0, 1)
    h_refine = h_refine * (n_points_refine / 2) / np.sum(h_refine)
    
    def objective_refine(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points_refine / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx_refine
        return np.max(correlate_vals)
    
    result_de_refine = opt.differential_evolution(
        objective_refine,
        [(0.0, 1.0) for _ in range(n_points_refine)],
        popsize=30,
        mutation=(0.9, 1.0),
        recombination=0.7,
        strategy='best1bin',
        tol=1e-5,
        maxiter=100,
        disp=False,
        polish=True,
        seed=seed
    )
    best_h_refine = result_de_refine.x
    best_c5_refine = result_de_refine.fun
    
    if best_c5_refine < best_c5:
        best_h = best_h_refine
        best_c5 = best_c5_refine
    
    # Final check
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best solution so far")
    
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = evaluate(h)
    
    return (h.tolist(), final_c5, n_points)
