import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 90
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial feasibility
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Objective function
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    # Gradient approximation
    def gradient(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_idx = np.argmax(correlate_vals)
        k = max_idx - n_points + 1
        if k < 0 or k >= n_points:
            return np.zeros_like(h)
        grad = np.zeros_like(h)
        for i in range(n_points):
            x = i * 2.0 / n_points
            x_plus_k = (i + k) * 2.0 / n_points
            if x_plus_k >= 2.0:
                grad[i] = 0.0
            else:
                grad[i] = 1.0 - h[np.int64(x_plus_k / 2.0 * n_points)]
        return grad
    
    # Optimization using L-BFGS-B
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, jac=gradient,
                          options={'maxiter': 500, 'ftol': 1e-5, 'eps': 1e-6})
    
    best_h = result.x
    best_c5 = result.fun
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = objective(best_h)
    
    # Adaptive grid refinement
    n_points = 120
    h = np.linspace(0, 2, n_points) * 0.0
    h = np.random.rand(n_points)
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, jac=gradient,
                          options={'maxiter': 300, 'ftol': 1e-5, 'eps': 1e-6})
    
    best_h = result.x
    best_c5 = result.fun
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = objective(best_h)
    
    # Final evaluation
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Time budget exceeded, returning best candidate")
        h = best_h
        final_c5 = objective(h)
    
    # Final checks
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return (h.tolist(), final_c5, n_points)
