```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    h = h.astype(np.float64)
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h))

    def compute_overlap(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def build_constraints(h, n_points, dx):
        A = []
        for k in range(1, n_points):
            A_k = np.zeros((n_points, n_points))
            for i in range(n_points):
                j = (i + k) % n_points
                A_k[i, j] = dx
                j = (i - k + n_points) % n_points
                A_k[i, j] = dx
            A.append(A_k)
        return A

    A = build_constraints(h, n_points, dx)

    h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
    C = cp.Variable()

    constraints = []
    for k in range(len(A)):
        constraints.append(cp.quad_form(h_var, A[k]) >= 1.0 - C)

    constraints.append(cp.sum(h_var) == target_sum)

    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)

    best_h = h.copy()
    best_c5 = compute_overlap(best_h)
    start_time = time.time()
    elapsed = 0

    for _ in range(10):
        try:
            result = prob.solve(solver=cp.ECOS, verbose=False)
            if result is None:
                break
            h_opt = h_var.value
            h_opt = np.clip(h_opt, 0.0, 1.0)
            h_opt = h_opt * (target_sum / np.sum(h_opt))
            c5 = compute_overlap(h_opt)
            if c5 < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
        except:
            break

    return best_h.tolist(), best_c5, n_points
```
