import numpy as np
import random
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = np.round(initial_h_values).astype(int)
    else:
        h = np.zeros(n_points, dtype=int)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                h[i] = 1
        h = h[:n_points//2] + h[n_points//2:]
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def objective(x):
        h = np.clip(x, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()

    def run_evolution():
        population = [np.random.rand(n_points) for _ in range(30)]
        for p in population:
            p = p / np.sum(p) * n_points / 2
        best_h = np.random.rand(n_points)
        best_h = best_h / np.sum(best_h) * n_points / 2
        best_c5 = compute_c5(best_h)
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            next_pop = []
            for i in range(30):
                if i < 10:
                    p = population[i]
                else:
                    p = np.random.rand(n_points)
                p = p / np.sum(p) * n_points / 2
                next_pop.append(p)
            next_pop.sort(key=lambda x: compute_c5(x))
            population = next_pop
            for p in population:
                p = p / np.sum(p) * n_points / 2
            if compute_c5(population[0]) < best_c5:
                best_h = population[0].copy()
                best_c5 = compute_c5(best_h)
        return best_h, best_c5

    best_h, best_c5 = run_evolution()

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.rand(n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = compute_c5(best_h)
    return (best_h, current_c5, n_points)
