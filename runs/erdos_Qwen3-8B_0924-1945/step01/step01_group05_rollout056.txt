import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    target_sum = n_points / 2

    # Initialize with available initial_h_values or a structured pattern
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Construct a binary Barker code of length closest to n_points
        barker_lengths = np.array([1, 2, 3, 4, 5, 7, 11, 13, 17, 23, 29, 31, 41, 43, 47, 59, 61, 67, 71, 73, 79, 83, 89, 97])
        L = barker_lengths[barker_lengths <= n_points][-1]
        barker_code = np.array([1] * L)
        for i in range(1, L):
            barker_code[i] = np.sum(barker_code[:i]) % 2
        barker_code = np.roll(barker_code, (L - n_points % L) % L)[:n_points]
        h = barker_code.astype(np.float64)
        h = h * target_sum / np.sum(h)

    # Clip and normalize to meet the constraint
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)

    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5

    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}

    start_time = time.time()
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 200, 'ftol': 1e-5, 'gtol': 1e-5, 'warm_start': True}
    )
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h, objective(h), n_points)

    best_h = result.x
    best_c5 = result.fun

    # Final sanitation and validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = objective(best_h)

    return (best_h, final_c5, n_points)
