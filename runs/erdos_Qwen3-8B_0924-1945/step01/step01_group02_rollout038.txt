```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def compute_c5(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def get_constraints(C):
        n = n_points
        A = np.zeros((n, n))
        for i in range(n):
            A[i, i] = dx
        constraints = []
        for k in range(1, n):
            A_k = np.zeros((n, n))
            for i in range(n):
                j = (i + k) % n
                A_k[i, j] = dx
                j = (i - k + n) % n
                A_k[i, j] = dx
            constraints.append(cp.bmat([[A_k, -np.ones((n, 1))], [np.ones((1, n)), -C]]) >= 0)
        return constraints

    best_h = h
    best_c5 = compute_c5(h)
    elapsed = 0

    for _ in range(20):
        if elapsed >= budget_s:
            break
        h_init = h.copy()
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        h_init += perturb
        h_init = np.clip(h_init, 0.0, 1.0)
        sum_h = np.sum(h_init)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_init = h_init * scale

        C = cp.Variable()
        h_var = cp.Variable(n_points, lb=0.0, ub=1.0)
        constraints = [
            cp.sum(h_var) == target_sum,
            C >= 0
        ]
        constraints += get_constraints(C)

        try:
            prob = cp.Problem(cp.Minimize(C), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-8, max_iter=100, warn=True)
            if prob.status in ["OPTIMAL", "OPTIMAL_INFEASIBLE"]:
                c5 = C.value
                h_opt = h_var.value
                h_opt = np.clip(h_opt, 0.0, 1.0)
                sum_h_opt = np.sum(h_opt)
                scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
                h_opt = h_opt * scale
                current_c5 = compute_c5(h_opt)
                if current_c5 < best_c5:
                    best_h = h_opt
                    best_c5 = current_c5
                elapsed += time.time() - elapsed
        except:
            pass

    h_opt = best_h
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = compute_c5(h_opt)
    return h_opt.tolist(), c5_bound, n_points
```
