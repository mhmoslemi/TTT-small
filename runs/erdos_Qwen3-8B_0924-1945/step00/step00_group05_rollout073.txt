import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h with small random deviation from initial_h_values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = (h - 0.5) * 2
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

    # Rescale to meet the sum constraint
    h = h * target_sum / np.sum(h)

    def objective(x):
        h = x.reshape(n_points)
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        # Rescale to meet the sum constraint
        h = h * target_sum / np.sum(h)
        # Compute the overlap integrals
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Define constraints: sum(h) == target_sum and 0 <= h <= 1
    def constraints(x):
        h = x.reshape(n_points)
        return [np.sum(h) - target_sum]

    # Define bounds [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]

    # Set up the optimization
    result = opt.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)

    best_h = result.x
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
