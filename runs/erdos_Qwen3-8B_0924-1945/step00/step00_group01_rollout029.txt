import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 47
    if 'initial_h_values' in kwargs:
        initial_h_values = kwargs['initial_h_values']
        h_vals = (initial_h_values * (n_points / 2)) / np.sum(initial_h_values)
    else:
        h_vals = np.random.uniform(0, 1, n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    
    dx = 2.0 / n_points
    max_iterations = 100
    no_improvement_threshold = 5
    best_c5 = np.inf
    best_h_vals = h_vals.copy()
    last_improvement = 0
    
    start_time = time.time()
    
    for iter in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Compute correlations
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5_values = corr * dx
        current_max_c5 = np.max(c5_values)
        
        if current_max_c5 < best_c5:
            best_c5 = current_max_c5
            best_h_vals = h_vals.copy()
            last_improvement = 0
        else:
            last_improvement += 1
        
        if last_improvement >= no_improvement_threshold:
            break
        
        # Select worst lag
        max_index = np.argmax(c5_values)
        lag = max_index - (n_points - 1)
        
        # Define objective function for the worst lag
        def objective(x):
            h = x.reshape(n_points)
            if np.any(h < 0) or np.any(h > 1):
                return np.inf
            overlap = 0
            for i in range(n_points):
                j = (i + lag) % n_points
                overlap += h[i] * (1 - h[j])
            return overlap
        
        # Define constraints
        def constraint_sum(x):
            return np.sum(x) - n_points / 2
        
        # Bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimization
        try:
            result = opt.minimize(
                objective,
                h_vals,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'ftol': 1e-8, 'maxiter': 100}
            )
            h_vals = result.x
        except:
            pass
        
        # Clamp and normalize
        h_vals = np.clip(h_vals, 0, 1)
        sum_h = np.sum(h_vals)
        if sum_h != 0:
            h_vals = h_vals * (n_points / 2) / sum_h
        
        # Check termination
        if np.max(np.abs(h_vals - best_h_vals)) < 1e-6:
            break
    
    # Final check
    corr = np.correlate(best_h_vals, 1 - best_h_vals, mode='full')
    c5_bound = np.max(corr) * dx
    
    # Ensure constraints
    if not np.all((best_h_vals >= 0) & (best_h_vals <= 1)):
        best_h_vals = np.clip(best_h_vals, 0, 1)
        sum_h = np.sum(best_h_vals)
        if sum_h != 0:
            best_h_vals = best_h_vals * (n_points / 2) / sum_h
    
    return (best_h_vals, c5_bound, n_points)
