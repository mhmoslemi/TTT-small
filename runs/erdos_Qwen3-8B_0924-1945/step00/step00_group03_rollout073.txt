import numpy as np
import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = len(initial_h_values) if initial_h_values is not None else 70
    dx = 2.0 / n_points
    h = np.copy(initial_h_values) if initial_h_values is not None else np.zeros(n_points)
    if initial_h_values is None:
        h = np.zeros(n_points)
        for i in range(n_points // 2):
            h[i] = 1.0
        for i in range(n_points // 2, n_points):
            h[i] = 0.0
        h = np.roll(h, n_points // 2 - 16)
        h = np.roll(h, -10)
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if total != n_points / 2:
        h = h * (n_points / 2) / total
    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(h, dx)
    start_time = time.time()
    t_total = int(budget_s * 0.75)
    T0 = 0.7 * (1.0 + best_c5) / 3.0
    Tmax = int(0.1 * t_total)
    beta = 0.75
    gamma = 0.65
    lambda0 = 0.15 / dx

    for t in range(t_total):
        if time.time() - start_time > budget_s:
            break
        if t % 100 == 0:
            lambda0 *= 0.8 if t < t_total // 2 else 0.95
        T = T0 * (1 - t / Tmax) ** (1 / beta)
        noise = np.random.normal(0, lambda0, n_points)
        perturbed = h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        delta = (n_points / 2 - current_sum) / (n_points * 0.95)
        perturbed += delta * np.random.choice([-1, 1], size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed) if np.sum(perturbed) != 0 else perturbed
        new_c5 = evaluate_erdos_solution(perturbed, dx)
        if new_c5 < best_c5 or (np.random.rand() < np.exp(-(new_c5 - best_c5) / T)):
            h = np.copy(perturbed)
            best_c5 = new_c5
            best_h = np.copy(h)
        if best_c5 < 0.3807:
            break
    final_c5 = evaluate_erdos_solution(best_h, dx)
    assert (best_h >= 0).all() and (best_h <= 1).all()
    assert np.isclose(np.sum(best_h), n_points / 2, atol=1e-6)
    assert np.isfinite(final_c5)
    return (best_h, final_c5, n_points)
