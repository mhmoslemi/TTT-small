import numpy as np
import scipy.optimize as opt
import time
import warnings

# Pre-imported functions
evaluate_erdos_solution = None
initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    h = initial_h_values if initial_h_values is not None else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(correlate_vals)
    
    def objective(h):
        nonlocal best_c5
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    def constraints(h):
        return np.sum(h) - n_points / 2
    
    def bounds_func(h):
        return np.clip(h, 0, 1)
    
    for _ in range(1000):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        h = best_h.copy()
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode='full') * dx
        worst_lag = np.argmax(correlate_vals)
        gradients = np.zeros(n_points)
        
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * dx
        
        # Simple gradient descent step
        step_size = 0.05
        h = h - step_size * gradients
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Add noise for exploration
        noise = np.random.normal(0, 0.05, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Update best solution
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5, n_points
