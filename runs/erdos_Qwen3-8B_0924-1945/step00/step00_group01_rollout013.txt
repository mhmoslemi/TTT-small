import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    h = h * n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2.0

    def objective(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals / np.sum(h_vals) * n_points / 2.0
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def gradient(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals / np.sum(h_vals) * n_points / 2.0
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        max_corr = np.max(corr)
        max_index = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            h_vals[i] += 1e-6
            h_vals = h_vals / np.sum(h_vals) * n_points / 2.0
            corr = np.correlate(h_vals, 1 - h_vals, mode='full')
            new_c5 = np.max(corr) * dx
            h_vals[i] -= 1e-6
            grad[i] = (new_c5 - objective(x)) / 1e-6
        return grad

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}]
    start_time = time.time()
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        jac=gradient,
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h, objective(h), n_points)
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2.0
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
