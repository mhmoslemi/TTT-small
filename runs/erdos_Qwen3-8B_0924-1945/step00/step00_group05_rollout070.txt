import numpy as np
import scipy.optimize as opt
from functools import partial

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points

    # Get initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Fallback: uniform distribution with adjusted values
        h = np.linspace(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    learning_rate = 0.01
    decay_rate = 0.999
    iteration = 0
    elapsed_time = 0

    def evaluate(h):
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)

    while elapsed_time < budget_s:
        start_time = np.datetime64('now')
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        k_max = np.argmax(overlaps)

        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx

        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (n_points / 2) / sum_h

        new_c5 = evaluate(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            iteration = 0
        else:
            iteration += 1

        if iteration >= 100:
            learning_rate *= decay_rate
            iteration = 0

        elapsed_time = np.datetime64('now') - start_time
        elapsed_time = np.timedelta64(elapsed_time, 's').astype(np.int64)

    return best_h, best_c5, n_points
