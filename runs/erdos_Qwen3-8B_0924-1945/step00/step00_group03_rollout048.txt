def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    def update_h(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - n_points / 2) > 1e-7:
            h = h * (n_points / 2) / total
        return h
    
    def compute_c5(h):
        return max(np.correlate(h, 1 - h, mode='full')) * dx
    
    for _ in range(50):
        start_time = time.time()
        if time.time() - start_time > budget_s - 0.5:
            break
        h = update_h(h)
        best_c5 = compute_c5(h)
        if best_c5 < 0.3807:
            break
        
        n = len(h)
        t = cp.Variable()
        h_var = cp.Variable(n)
        constraints = []
        
        for k in range(n - 1):
            idx = (np.arange(n) + k) % n
            overlap = h_var[idx] * (1 - h_var)
            constraints.append(t >= cp.sum(overlap) * dx)
        
        constraints += [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2]
        
        obj = t + 1e-3 * cp.sum((h_var - 0.5) ** 2)
        prob = cp.Problem(cp.Minimize(obj), constraints)
        result = prob.solve()
        if result is not None:
            h = update_h(h_var.value)
            best_c5 = compute_c5(h)
            best_h = h.copy()
            
        if best_c5 < 0.3807:
            break
        
        h += np.random.normal(0, 0.05, n)
        h = update_h(h)
        best_c5 = compute_c5(h)
        best_h = h.copy()
        
        if best_c5 < 0.3807:
            break
    
    return (best_h, best_c5, n_points)
