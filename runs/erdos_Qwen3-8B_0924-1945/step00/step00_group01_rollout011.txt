import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = evaluate_erdos_solution()  # Assume this function is defined elsewhere
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Normalize initial_h_values to satisfy the integral constraint
    sum_h = np.sum(initial_h_values)
    if abs(sum_h - n_points / 2) > 1e-6:
        initial_h_values = initial_h_values * (n_points / 2) / sum_h

    def objective(h_vals):
        h = h_vals / 2  # Since h_vals is scaled to sum to n_points / 2, h ranges [0, 1]
        shifted_h = np.roll(h, 1)
        corr = np.correlate(h, 1 - shifted_h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def gradient(h_vals):
        h = h_vals / 2
        shifted_h = np.roll(h, 1)
        corr = np.correlate(h, 1 - shifted_h, mode="full")
        max_corr = np.max(corr)
        max_index = np.argmax(corr)

        # Finite difference to approximate gradient of the max correlation
        grad = np.zeros_like(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += 1e-6
            h_plus = h_plus / 2  # Ensure h ranges [0, 1]
            shifted_h_plus = np.roll(h_plus, 1)
            corr_plus = np.correlate(h_plus, 1 - shifted_h_plus, mode="full")
            grad_plus = np.max(corr_plus) * dx

            h_minus = h_vals.copy()
            h_minus[i] -= 1e-6
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus / 2  # Ensure h ranges [0, 1]
            shifted_h_minus = np.roll(h_minus, 1)
            corr_minus = np.correlate(h_minus, 1 - shifted_h_minus, mode="full")
            grad_minus = np.max(corr_minus) * dx

            grad[i] = (grad_plus - grad_minus) / (2 * 1e-6)
        return grad

    # Constraints
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2},
            {'type': 'bounds', 'fun': lambda x: np.clip(x, 0, 1)})

    # Optimization
    start_time = time.time()
    result = opt.minimize(
        fun=objective,
        x0=initial_h_values,
        method='L-BFGS-B',
        jac=gradient,
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 100}
    )

    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        print("Time budget exceeded. Using initial solution.")
        return (initial_h_values, objective(initial_h_values), n_points)

    h_opt = result.x
    h_opt = np.clip(h_opt, 0, 1)
    sum_h = np.sum(h_opt)
    if abs(sum_h - n_points / 2) > 1e-6:
        # Normalize if constraints are violated
        h_opt = h_opt * (n_points / 2) / sum_h
    final_c5 = objective(h_opt)
    return (h_opt, final_c5, n_points)
