import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.clip(initial_h_values, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (n_points / 2) / sum_h
        else:
            h = np.full(n_points, 0.5)
    else:
        h = np.full(n_points, 0.5)
    
    best_c5 = np.inf
    best_h = h.copy()
    max_iterations = 500
    learning_rate = 0.01
    momentum = 0.9
    decay_rate = 0.999
    perturb_prob = 0.05
    perturb_std = 0.05
    time_limit = budget_s
    
    start_time = time.time()
    
    for _ in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > time_limit:
            break
        
        # Compute overlap integrals
        conv = np.correlate(h, 1 - h, mode="full")
        c5_values = conv * dx
        c5 = np.max(c5_values)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the worst lag
        max_idx = np.argmax(c5_values)
        offset = max_idx - n_points + 1
        
        # Compute gradient for this lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + offset
            if 0 <= j < n_points:
                grad[i] += (1 - h[j])  # contribution from h[i] to (1-h[j])
                grad[j] -= h[i]         # contribution from h[j] to -h[i]
        
        # Update with momentum and learning rate
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        
        # Enforce sum constraint
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (n_points / 2) / sum_h
        else:
            h = np.full(n_points, 0.5)
        
        # Random perturbation
        if np.random.rand() < perturb_prob:
            h += np.random.normal(0, perturb_std, n_points)
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 0:
                h = h * (n_points / 2) / sum_h
            else:
                h = np.full(n_points, 0.5)
        
        # Decay learning rate
        learning_rate *= decay_rate
    
    # Final evaluation
    conv = np.correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(conv * dx)
    return (best_h, final_c5, n_points)
