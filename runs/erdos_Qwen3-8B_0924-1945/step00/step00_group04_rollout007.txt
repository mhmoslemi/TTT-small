import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h_values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[0::2] = 1.0
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0, 1)
        cross_corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(cross_corr) * dx
        return c5

    def objective(h):
        return -evaluate(h)

    def fitness(h):
        return -evaluate(h)

    def mutate(parent):
        mutation = np.random.uniform(-0.05, 0.05, size=n_points)
        child = parent + mutation
        child = np.clip(child, 0, 1)
        child = child * (target_sum / np.sum(child))
        child = np.clip(child, 0, 1)
        child = child * (target_sum / np.sum(child))
        return child

    def crossover(parent1, parent2):
        alpha = np.random.rand(n_points)
        child = alpha * parent1 + (1 - alpha) * parent2
        child = np.clip(child, 0, 1)
        child = child * (target_sum / np.sum(child))
        child = np.clip(child, 0, 1)
        child = child * (target_sum / np.sum(child))
        return child

    population = [mutate(best_h) for _ in range(50)]
    population_fitness = [fitness(ind) for ind in population]
    population = [p for _, p in sorted(zip(population_fitness, population), reverse=True)]
    population_fitness = [f for _, f in sorted(zip(population_fitness, population_fitness), reverse=True)]

    start_time = time.time()
    no_improvement_count = 0
    while time.time() - start_time < budget_s:
        new_population = []
        for _ in range(50):
            parent1, parent2 = np.random.choice(population, 2, replace=False)
            child = crossover(parent1, parent2)
            new_population.append(child)
        new_population_fitness = [fitness(ind) for ind in new_population]
        combined_population = population + new_population
        combined_fitness = population_fitness + new_population_fitness
        combined_population, combined_fitness = zip(*sorted(zip(combined_fitness, combined_population), reverse=True))
        combined_population = list(combined_population)
        combined_fitness = list(combined_fitness)
        population = combined_population[:50]
        population_fitness = combined_fitness[:50]
        current_best = population[0]
        current_c5 = evaluate(current_best)
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        if no_improvement_count >= 50:
            break
    return (best_h, best_c5, n_points)
