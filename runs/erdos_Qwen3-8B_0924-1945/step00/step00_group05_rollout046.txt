import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial guess
    n_points = 500
    dx = 2.0 / n_points
    h = np.array(initial_h_values[:n_points], dtype=np.float64)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h if sum_h != 0 else np.zeros(n_points)
    
    # Optimization parameters
    max_iter = 500
    learning_rate = 0.01
    momentum = 0.9
    decay_rate = 0.999
    perturb_prob = 0.05
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Compute overlap integrals
        conv = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(conv)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the worst lag
        k = np.argmax(conv)
        
        # Compute gradient for the worst lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] += (1 - h[j]) * dx
            j = (i - k) % n_points
            grad[i] -= h[j] * dx
        
        # Gradient descent update
        grad = grad / np.max(np.abs(grad)) if np.max(np.abs(grad)) > 1e-10 else grad
        h = h - learning_rate * grad
        
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h if sum_h != 0 else np.zeros(n_points)
        
        # Apply perturbation
        if np.random.rand() < perturb_prob:
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            h = h * (n_points / 2) / sum_h if sum_h != 0 else np.zeros(n_points)
        
        # Learning rate decay and momentum
        learning_rate *= decay_rate
        momentum *= decay_rate
        
    return (best_h, best_c5, n_points)
