import numpy as np
import scipy.signal as signal
import time
import cvxpy as cp
import random

initial_h_values = None  # Will be provided by the parent context

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45  # Starting with the initial construction
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    if np.abs(sum_h - n_points / 2) > 1e-8:
        h = h * (n_points / 2) / sum_h
    h = np.clip(h, 0, 1)
    best_c5 = float('inf')
    best_h = h.copy()
    best_n = n_points
    best_time = time.time()
    max_iter = int(budget_s / 0.1)
    learning_rate = 0.1
    momentum = 0.9
    noise_scale = 0.05
    iter_count = 0
    v = np.zeros(n_points)
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        correlate_vals = signal.correlate(h, 1 - h, mode="full") * dx
        worst_lag = np.argmax(correlate_vals)
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * dx
        v = momentum * v + learning_rate * gradients
        h = h - v
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h
        if iter_count % 100 == 0:
            noise = noise_scale * np.random.normal(0, 1, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            h = h * (n_points / 2) / sum_h
        current_c5 = np.max(correlate_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            best_n = n_points
            best_time = time.time()
        if iter_count % 50 == 0 and np.abs(current_c5 - best_c5) < 1e-5:
            break
    return best_h, best_c5, best_n
