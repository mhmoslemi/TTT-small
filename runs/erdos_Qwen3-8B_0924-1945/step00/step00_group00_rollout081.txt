```python
import numpy as np
import time
from scipy.optimize import minimize
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    
    n_points = 45
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    
    def compute_c5(h):
        h_array = np.array(h)
        corr = np.correlate(h_array, 1 - h_array, mode='full') * dx
        return np.max(corr)
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        h = h * (n_points / 2) / total
        return compute_c5(h)
    
    def constraint(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        h = h * (n_points / 2) / total
        return h - h
    
    cons = {'type': 'eq', 'fun': constraint}
    
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        
        result = minimize(objective, h, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-8, 'maxiter': 10})
        h = result.x
        
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if np.abs(current_c5 - best_c5) < 1e-5:
            break
    
    return best_h, best_c5, n_points
```
