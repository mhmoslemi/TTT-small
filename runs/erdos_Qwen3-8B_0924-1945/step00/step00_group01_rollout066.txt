import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = globals().get('initial_h_values', None)
    n_points = 100 if initial_h_values is None else len(initial_h_values)
    if initial_h_values is not None:
        h_vals = np.array(initial_h_values, dtype=np.float64)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    else:
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    
    dx = 2.0 / n_points
    best_h = h_vals.copy()
    best_c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * dx
    max_iterations = 100
    no_improvement_count = 0
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 100:
            break
        
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        corr = corr * dx
        max_corr_idx = np.argmax(corr)
        max_corr_val = corr[max_corr_idx]
        
        def objective(x):
            h = x.copy()
            h[(h < 0)] = 0
            h[(h > 1)] = 1
            h = (h * (n_points / 2)) / np.sum(h)
            corr = np.correlate(h, 1 - h, mode='full')
            return corr[max_corr_idx]
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{
            'type': 'eq',
            'fun': lambda x: np.sum(x) - n_points / 2
        }]
        
        result = opt.minimize(
            objective,
            h_vals,
            bounds=bounds,
            constraints=constraints,
            method='L-BFGS-B'
        )
        
        if result.success:
            h_vals = result.x
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
            new_c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * dx
            if new_c5 < best_c5:
                best_h = h_vals.copy()
                best_c5 = new_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= 5:
            break
    
    h_vals = np.clip(best_h, 0, 1)
    h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    final_c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * dx
    return (h_vals, final_c5, n_points)
