import numpy as np
import random
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if total != 0:
        h = h / total * (n_points / 2)
    else:
        h = np.full(n_points, 0.5)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    start = time.time()
    for _ in range(300):
        if time.time() - start > budget_s:
            break
        indices = np.random.choice(n_points, 5, replace=False)
        noise = np.random.normal(0, 0.1, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total != 0:
            h = h / total * (n_points / 2)
        else:
            h = np.full(n_points, 0.5)
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    shifted = np.zeros_like(h)
    for i in range(len(h)):
        shifted[i] = 1 - h[i]
    corr = correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
