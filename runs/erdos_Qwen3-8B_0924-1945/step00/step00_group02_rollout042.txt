```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    initial_h_values = np.random.rand(n_points)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values *= n_points / 2
    initial_h_values = np.clip(initial_h_values, 0, n_points)
    initial_h_values /= n_points
    h = initial_h_values.copy()
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr * 2.0 / n_points
    
    def clip_and_scale(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scale = (n_points / 2) / total if total != 0 else 1.0
        h *= scale
        return h
    
    def perturb(h, target_indices, scale):
        h_candidate = h.copy()
        for i in target_indices:
            h_candidate[i] = 1 - h_candidate[i]
            h_candidate[i] += np.random.uniform(-scale, scale)
            h_candidate[i] = np.clip(h_candidate[i], 0, 1)
        h_candidate = clip_and_scale(h_candidate)
        return h_candidate
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    
    for _ in range(int(budget_s // 0.1)):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode="full")
        lag = np.argmax(corr)
        overlap = h * np.roll(1 - h, lag)
        max_overlap = np.max(overlap)
        threshold = max_overlap * 0.8
        target_indices = np.where(overlap > threshold)[0]
        target_indices = np.random.choice(target_indices, size=min(30, len(target_indices)), replace=False)
        
        if len(target_indices) == 0:
            continue
        
        candidates = []
        for idx in target_indices:
            h_candidate = h.copy()
            h_candidate[idx] = 1 - h_candidate[idx]
            h_candidate = clip_and_scale(h_candidate)
            candidates.append(h_candidate)
        
        for idx in target_indices:
            h_candidate = h.copy()
            h_candidate[idx] += np.random.uniform(-0.05, 0.05)
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = clip_and_scale(h_candidate)
            candidates.append(h_candidate)
        
        for candidate in candidates:
            c5 = compute_c5(candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate.copy()
        
        h = best_h.copy()
    
    h = clip_and_scale(h)
    best_c5 = compute_c5(h)
    
    return (h, best_c5, n_points)
```
