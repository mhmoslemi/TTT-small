import numpy as np
import numpy.random as npr
import scipy.signal as signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in globals() else np.zeros(70)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.copy(initial_h_values)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Precompute all lags
    lags = np.arange(-n_points + 1, n_points)
    
    # Initial parameters
    t_total = int(budget_s * 0.75)
    T0 = 0.7 * (1.0 + best_c5) / 3.0
    Tmax = int(0.1 * t_total)
    beta = 0.75
    gamma = 0.65
    lambda0 = 0.15 / dx
    
    start_time = time.time()
    
    for t in range(t_total):
        if time.time() - start_time > budget_s:
            break
        if t % 100 == 0:
            lambda0 *= 0.8 if t < t_total // 2 else 0.95
        
        # Generate perturbation
        noise = npr.normal(0, lambda0, size=n_points)
        perturbed = h + noise
        perturbed = np.clip(perturbed, 0.0, 1.0)
        
        # Constraint restoration
        current_sum = np.sum(perturbed)
        delta = (n_points / 2 - current_sum) / (n_points * 0.95)
        perturbed += delta * npr.choice([-1, 1], size=n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        perturbed = perturbed * (n_points / 2 / np.sum(perturbed)) if np.sum(perturbed) != 0 else perturbed
        
        # Evaluate new solution
        try:
            new_c5 = evaluate_erdos_solution(perturbed, dx)
        except:
            continue
        
        # Acceptance criteria
        if new_c5 < best_c5 or (np.random.rand() < np.exp(-(new_c5 - best_c5) / (T0 * (1 - t / Tmax) ** (1 / beta)))):
            h = perturbed.copy()
            best_h = h.copy()
            best_c5 = new_c5
        
        # Cool temperature
        if t > Tmax:
            break
    
    # Final checks
    assert (best_h >= 0).all() and (best_h <= 1).all()
    assert np.isclose(np.sum(best_h), n_points / 2, atol=1e-6)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (best_h, final_c5, n_points)
