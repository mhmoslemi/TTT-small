import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45  # Start with the initial n_points
    dx = 2.0 / n_points
    initial_h = initial_h_values if 'initial_h_values' in kwargs else np.random.rand(n_points)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = np.inf
    n_iter = 0
    learning_rate = 0.05
    noise_scale = 0.05
    max_iter = 500
    early_stop = 50
    no_improve = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s and n_iter < max_iter:
        # Compute correlation
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        current_c5 = np.max(correlate_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= early_stop:
                break
        
        # Identify worst lag
        worst_lag = np.argmax(correlate_vals)
        
        # Compute gradient for worst lag
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * dx
        
        # Update h
        h = h - learning_rate * gradients
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Add structured noise
        if n_iter % 100 == 0:
            noise = noise_scale * np.random.normal(0, 1, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        n_iter += 1
    
    # Final check
    final_correlate = np.correlate(best_h, 1 - best_h, mode="full") * dx
    final_c5 = np.max(final_correlate)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    return (best_h, final_c5, n_points)
