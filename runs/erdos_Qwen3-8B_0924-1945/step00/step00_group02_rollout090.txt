import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300  # Optimal for balance between performance and precision
    dx = 2.0 / n_points
    initial_h_values_available = 'initial_h_values' in globals()
    
    if initial_h_values_available:
        h = np.copy(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = (h - 0.5) * 2  # Scale to [0,1]
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Rescale to ensure sum(h) == n_points / 2
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = np.copy(h)
    best_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
    
    start_time = time.time()
    for _ in range(1200):  # 1200 iterations within 600s
        if time.time() - start_time > budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        lag = np.argmax(corr)
        
        # Calculate overlap_x for the most significant lag
        overlap_x = h * np.roll(1 - h, lag)
        threshold = max_corr * 0.8
        target_indices = np.where(overlap_x > threshold)[0]
        
        # Generate candidates
        candidates = []
        for i in np.random.choice(target_indices, min(5, len(target_indices)), replace=False):
            # Flip
            h_flip = h.copy()
            h_flip[i] = 1 - h_flip[i]
            h_flip = np.clip(h_flip, 0, 1)
            h_flip = h_flip * (n_points / 2) / np.sum(h_flip)
            candidates.append(h_flip)
            
            # Add noise
            h_noise = h.copy()
            h_noise[i] += np.random.uniform(-0.05, 0.05)
            h_noise = np.clip(h_noise, 0, 1)
            h_noise = h_noise * (n_points / 2) / np.sum(h_noise)
            candidates.append(h_noise)
        
        # Evaluate candidates
        for candidate in candidates:
            corr_candidate = np.correlate(candidate, 1 - candidate, mode="full")
            c5_candidate = np.max(corr_candidate) * dx
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = np.copy(candidate)
        
        # Update current h
        h = best_h
    
    # Final check
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    best_h = np.copy(h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx
    
    return (best_h, best_c5, n_points)
