import numpy as np
import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 450
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 * (1 + np.random.uniform(-1, 1))
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    h_values = h.copy()

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * 2.0 / n_points

    best_c5 = compute_c5(h)
    best_h = h.copy()

    start_time = time.time()
    dt = 0.1
    iterations = int((budget_s - (time.time() - start_time)) / dt)
    for t in range(iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        max_val = corr[max_k]

        # Targeted mutation
        h_mut = h.copy()
        n_mut = 10
        for i in range(n_mut):
            idx = max_k - 10 + i
            if 0 <= idx < len(corr):
                contrib = corr[idx] - np.mean(corr)
                delta = -np.sign(contrib) * 0.1 * (max_val - 0.38)
                h_mut[idx] += delta
                h_mut[idx] = np.clip(h_mut[idx], 0, 1)
        h_mut = h_mut * (n_points / 2) / np.sum(h_mut)

        # Global mutation
        h_mut2 = h.copy()
        for _ in range(5):
            idx = np.random.randint(0, len(h_mut2))
            h_mut2[idx] += np.random.normal(0, 0.2 * np.exp(-t / 300))
            h_mut2[idx] = np.clip(h_mut2[idx], 0, 1)
        h_mut2 = h_mut2 * (n_points / 2) / np.sum(h_mut2)

        c5_mut = compute_c5(h_mut)
        c5_mut2 = compute_c5(h_mut2)
        if c5_mut < best_c5:
            best_c5 = c5_mut
            best_h = h_mut.copy()
        if c5_mut2 < best_c5:
            best_c5 = c5_mut2
            best_h = h_mut2.copy()
        h = best_h

    h_values = best_h
    c5_bound = best_c5
    return (h_values, c5_bound, n_points)
