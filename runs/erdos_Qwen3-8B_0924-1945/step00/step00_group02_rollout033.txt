import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 450
    initial_h_values = kwargs.get('initial_h_values', None)
    
    if initial_h_values is None:
        h = 0.5 * (1 + np.random.uniform(-1, 1, n_points))
    else:
        h = initial_h_values.copy()
    
    # Clip and normalize
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    dx = 2.0 / n_points

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()
    
    for t in range(6000):  # 6000 iterations with ~0.1s each
        if time.time() - start_time > budget_s:
            break
        
        current_c5 = evaluate(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        corr = np.correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        
        # Targeted mutation
        h_t = h.copy()
        for k in range(max_k - 10, max_k + 11):
            if 0 <= k < len(corr):
                contrib = (h * (1 - h[k % len(h)])) if k < len(h) else (h * (1 - h[k - len(h)]))
                contrib = np.sum(contrib)
                if contrib > 0:
                    h_t[k % len(h)] = max(0, min(1, h_t[k % len(h)] - 0.01))
                else:
                    h_t[k % len(h)] = max(0, min(1, h_t[k % len(h)] + 0.01))
        h_t = np.clip(h_t, 0, 1)
        h_t = h_t * (n_points / 2) / np.sum(h_t)

        # Global mutation
        h_g = h.copy()
        indices = np.random.choice(len(h), 5, replace=False)
        for i in indices:
            h_g[i] = max(0, min(1, h_g[i] + np.random.normal(0, 0.1)))
        h_g = np.clip(h_g, 0, 1)
        h_g = h_g * (n_points / 2) / np.sum(h_g)

        # Select better solution
        c5_t = evaluate(h_t)
        c5_g = evaluate(h_g)
        if c5_t < c5_g:
            h = h_t
        else:
            h = h_g

    return best_h, best_c5, n_points
