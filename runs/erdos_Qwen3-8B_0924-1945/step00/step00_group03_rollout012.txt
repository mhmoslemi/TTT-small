        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx

    bounds = [(0, 1) for _ in range(n_points)]

    result = optimize.minimize(
        fun=objective,
        x0=initial_h,
        method='SLSQP',
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        bounds=bounds,
        options={'maxiter': 100, 'ftol': 1e-5}
    )
    if result.success:
        h_opt = result.x
    else:
        h_opt = initial_h.copy()

            noisy_h = initial_h * (1 + 0.1 * np.random.randn(n_points))
            noisy_h = np.clip(noisy_h, 0, 1)
            noisy_h /= np.sum(noisy_h) * dx
            noisy_h *= sum_h / dx
            result = optimize.minimize(
                fun=objective,
                x0=noisy_h,
                method='SLSQP',
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                bounds=bounds,
                options={'maxiter': 100, 'ftol': 1e-5}
            )
            if result.success:
                h_opt = result.x
                break

    h_opt = np.clip(h_opt, 0, 1)
    h_opt /= np.sum(h_opt) * dx
