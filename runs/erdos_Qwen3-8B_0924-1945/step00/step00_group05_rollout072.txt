import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in globals() else np.random.rand(500) * 1000 / 500
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlaps)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    learning_rate = 0.1
    iteration = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        k_max = np.argmax(overlaps)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (n_points / 2) / sum_h
        
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            iteration = 0
        else:
            iteration += 1
            if iteration > 200:
                learning_rate *= 0.95
                iteration = 0
    
    h_values = best_h
    c5_bound = best_c5
    return (h_values, c5_bound, n_points)
