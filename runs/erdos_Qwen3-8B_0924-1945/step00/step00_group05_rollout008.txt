import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h.npy') if 'initial_h_values' in kwargs else None
    if initial_h_values is not None:
        h = np.clip(initial_h_values, 0, 1)
        current_sum = np.sum(h)
        if current_sum != 0:
            h = h * (len(h) / 2) / current_sum
    else:
        n_points = 100
        h = np.zeros(n_points)
        h[:50] = 1.0
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum != 0:
            h = h * (n_points / 2) / current_sum

    n_points = len(h)
    dx = 2.0 / n_points
    max_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    max_stagnation = 50
    stagnation = 0
    mutation_noise = 0.05
    mutation_rate = 0.1
    pop_size = 50
    n_generations = 0

    while time.time() - start_time < budget_s:
        population = [h.copy() for _ in range(pop_size)]
        for i in range(pop_size):
            candidate = h.copy()
            noise = np.random.normal(0, mutation_noise, n_points)
            candidate += noise
            candidate = np.clip(candidate, 0, 1)
            current_sum = np.sum(candidate)
            if current_sum != 0:
                candidate = candidate * (n_points / 2) / current_sum
            population[i] = candidate

        evaluations = []
        for candidate in population:
            overlaps = np.correlate(candidate, 1 - candidate, mode="full") * dx
            c5 = np.max(overlaps)
            evaluations.append((c5, candidate))
        evaluations.sort()
        best_c5, best_candidate = evaluations[0]
        if best_c5 < max_c5:
            max_c5 = best_c5
            best_h = best_candidate.copy()
            stagnation = 0
        else:
            stagnation += 1

        if stagnation >= max_stagnation:
            mutation_noise *= 0.9
            stagnation = 0

        n_generations += 1
        if n_generations % 10 == 0:
            mutation_noise *= 1.1

        if best_c5 < 0.3807:
            break

    if best_c5 >= 0.3807:
        best_h = h.copy()
        overlaps = np.correlate(best_h, 1 - best_h, mode="full") * dx
        max_c5 = np.max(overlaps)

    if np.sum(best_h) != n_points / 2:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    overlaps = np.correlate(best_h, 1 - best_h, mode="full") * dx
    c5_bound = np.max(overlaps)
    return best_h, c5_bound, n_points
