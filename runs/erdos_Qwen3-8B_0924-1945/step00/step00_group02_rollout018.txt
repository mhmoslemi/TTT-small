import numpy as np
import time
from scipy.stats import bernoulli

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def scale_h(h):
        total = np.sum(h)
        if abs(total) < 1e-10:
            return np.zeros_like(h)
        return h * (n_points / 2) / total
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr * 2.0 / n_points
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_iter = 0
    start_time = time.time()
    for _ in range(int((budget_s - (time.time() - start_time)) // 0.1) + 1):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        lag = np.argmax(corr)
        if lag >= len(h):
            lag = len(h) - 1
        overlap = h * (1 - np.roll(h, lag))
        indices = np.argsort(overlap)[::-1]
        threshold = np.max(overlap) * 0.8
        target_indices = indices[overlap[indices] > threshold]
        candidates = []
        for idx in target_indices[:int(len(target_indices) * 0.1)]:
            for delta in [-0.05, 0.05]:
                h_candidate = h.copy()
                h_candidate[idx] += delta
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = scale_h(h_candidate)
                c5 = compute_c5(h_candidate)
                candidates.append((c5, h_candidate))
        for _ in range(5):
            idx = np.random.randint(0, n_points)
            delta = np.random.uniform(-0.05, 0.05)
            h_candidate = h.copy()
            h_candidate[idx] += delta
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = scale_h(h_candidate)
            c5 = compute_c5(h_candidate)
            candidates.append((c5, h_candidate))
        if candidates:
            candidates.sort()
            if candidates[0][0] < best_c5:
                best_c5 = candidates[0][0]
                best_h = candidates[0][1].copy()
                best_iter = _
        if _ % 200 == 0 and best_c5 < compute_c5(h):
            h = best_h.copy()
            best_iter = _
        if _ == 0 or (time.time() - start_time) > budget_s - 10:
            break
    return (best_h, best_c5, n_points)
