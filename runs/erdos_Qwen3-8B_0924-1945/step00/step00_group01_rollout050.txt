import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction if available
    if 'initial_h_values' in globals():
        h_vals = np.array(initial_h_values, dtype=np.float64)
    else:
        # Default construction if no parent is provided
        h_vals = np.zeros(47)
        h_vals[::2] = 1
        h_vals = h_vals / np.sum(h_vals) * (47 / 2)
    
    n_points = len(h_vals)
    dx = 2.0 / n_points
    
    # Normalize to meet the constraint
    h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
    
    def objective(x):
        x = x.reshape(n_points)
        x = np.clip(x, 0, 1)
        x = x / np.sum(x) * (n_points / 2)
        corr = np.correlate(x, 1 - x, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def gradient(x):
        x = x.reshape(n_points)
        x = np.clip(x, 0, 1)
        x = x / np.sum(x) * (n_points / 2)
        corr = np.correlate(x, 1 - x, mode="full")
        max_corr = np.max(corr)
        max_index = np.argmax(corr)
        
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        for i in range(n_points):
            x_plus = x.copy()
            x_plus[i] += 1e-6
            x_plus = x_plus / np.sum(x_plus) * (n_points / 2)
            corr_plus = np.correlate(x_plus, 1 - x_plus, mode="full")
            grad[i] = (np.max(corr_plus) * dx - max_corr * dx) / 1e-6
        
        return grad.flatten()
    
    # Setup optimization
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
    
    result = optimize.minimize(
        objective,
        h_vals,
        bounds=bounds,
        constraints=cons,
        method='L-BFGS-B',
        jac=gradient,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    best_h_vals = result.x
    best_h_vals = np.clip(best_h_vals, 0, 1)
    best_h_vals = best_h_vals / np.sum(best_h_vals) * (n_points / 2)
    
    # Final evaluation
    corr = np.correlate(best_h_vals, 1 - best_h_vals, mode="full")
    c5_bound = np.max(corr) * dx
    
    return (best_h_vals, c5_bound, n_points)
