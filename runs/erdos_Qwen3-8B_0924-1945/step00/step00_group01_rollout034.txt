import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 100
    dx = 2.0 / n_points
    
    # Initial construction
    if 'initial_h_values' in globals():
        h_vals = (initial_h_values * (n_points / 2)) / np.sum(initial_h_values)
    else:
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    
    def evaluate(h):
        if np.any(h < 0) or np.any(h > 1):
            return np.inf
        if not np.isclose(np.sum(h), n_points / 2, atol=1e-6):
            return np.inf
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = h_vals.copy()
    best_c5 = evaluate(h_vals)
    best_time = time.time()
    
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    mutation_amp = 0.01
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate population
        population = [h_vals.copy() for _ in range(population_size)]
        fitness = [evaluate(h) for h in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        
        # Selection
        parents = []
        for _ in range(population_size // 2):
            idx1 = np.random.randint(population_size)
            idx2 = np.random.randint(population_size)
            if fitness[idx1] < fitness[idx2]:
                parents.append(population[idx1])
            else:
                parents.append(population[idx2])
        
        # Crossover
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 >= len(parents):
                break
            parent1 = parents[i]
            parent2 = parents[i + 1]
            crossover_point = np.random.randint(1, n_points - 1)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            offspring.append(child1)
            offspring.append(child2)
        
        # Mutation
        for i in range(len(offspring)):
            h = offspring[i]
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    h[j] += np.random.uniform(-mutation_amp, mutation_amp)
                    h[j] = np.clip(h[j], 0, 1)
                    h = (h * (n_points / 2)) / np.sum(h)
            offspring[i] = h
        
        # Replace worst
        for i in range(len(population)):
            if np.random.rand() < 0.2:
                population[i] = offspring[i % len(offspring)]
        
        # Update best
        for h in population:
            c = evaluate(h)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
    
    # Final check
    final_time = time.time()
    elapsed = final_time - start_time
    if elapsed > budget_s:
        best_h = h_vals.copy()
        best_c5 = evaluate(h_vals)
    
    return (best_h, best_c5, n_points)
