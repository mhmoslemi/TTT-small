import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load("initial_h_values.npy")  # Assume this is preloaded
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = np.clip(initial_h_values, 0, 1)
    h *= target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    elapsed_time = 0
    learning_rate = 0.05
    max_iterations = 1000
    improvement_threshold = 1e-6
    no_improve_counter = 0
    
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        cross_corr = np.correlate(h, 1 - h, mode="full") * dx
        idx = np.argmax(cross_corr)
        lag_k = idx - (n_points - 1)
        grad = np.zeros(n_points)
        for j in range(n_points):
            if j + lag_k < n_points:
                grad[j] += (1 - h[j + lag_k]) * dx
            if j - lag_k >= 0:
                grad[j] -= h[j - lag_k] * dx
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum < 1e-6:
            h = np.zeros(n_points)
        else:
            h = np.clip(h, 0, 1) * (target_sum / current_sum)
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve_counter = 0
        else:
            no_improve_counter += 1
        if no_improve_counter > 50:
            break
        if np.abs(current_c5 - best_c5) < improvement_threshold:
            learning_rate *= 0.995
        elapsed_time = time.time() - start_time
    
    return (best_h, best_c5, n_points)
