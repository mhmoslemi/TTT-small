import numpy as np
import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in globals() else None
    
    n_points = 500
    dx = 2.0 / n_points
    max_lag = n_points - 1
    pop_size = 50
    mutation_noise = 0.05
    mutation_rate = 0.1
    selection_size = 20
    stagnation_threshold = 50
    
    def normalize(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / current_sum
        h = np.clip(h * scale, 0, 1)
        return h
    
    def evaluate(h):
        h = normalize(h)
        if np.any(h < 0) or np.any(h > 1):
            return np.inf
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlaps)
    
    def crossover(parent1, parent2):
        idx = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:idx], parent2[idx:]])
        return normalize(child)
    
    def mutate(h):
        h = h.copy()
        noise = np.random.normal(0, mutation_noise, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = normalize(h)
        if np.random.rand() < mutation_rate:
            i, j = np.random.choice(n_points, 2, replace=False)
            h[i], h[j] = h[j], h[i]
        return h
    
    def evaluate_population(pop):
        return [evaluate(h) for h in pop]
    
    def tournament_selection(pop, fitnesses):
        indices = np.random.choice(len(pop), 5, replace=False)
        best_idx = np.argmin(fitnesses[indices])
        return pop[indices[best_idx]]
    
    def evolve_population(pop):
        new_pop = []
        for _ in range(pop_size):
            parent1 = tournament_selection(pop, evaluate_population(pop))
            parent2 = tournament_selection(pop, evaluate_population(pop))
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_pop.append(child)
        return new_pop
    
    def track_best(pop, best_h, best_c):
        fitnesses = evaluate_population(pop)
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c:
            best_c = fitnesses[best_idx]
            best_h = pop[best_idx]
        return best_h, best_c
    
    def adaptive_mutation(mutate, mutation_noise, mutation_rate):
        if np.random.rand() < 0.5:
            mutation_noise *= 0.9
        else:
            mutation_noise *= 1.1
        if np.random.rand() < 0.5:
            mutation_rate *= 0.9
        else:
            mutation_rate *= 1.1
        return mutate, mutation_noise, mutation_rate
    
    def check_constraints(h):
        if np.any(h < 0) or np.any(h > 1):
            return False
        if not np.isclose(np.sum(h), n_points / 2):
            return False
        return True
    
    h = np.zeros(n_points)
    if initial_h_values is not None:
        h = np.clip(initial_h_values, 0, 1)
        h = normalize(h)
    best_h = h
    best_c = evaluate(h)
    pop = [h.copy() for _ in range(pop_size)]
    start_time = time.time()
    stagnation = 0
    
    while time.time() - start_time < budget_s:
        pop = evolve_population(pop)
        best_h, best_c = track_best(pop, best_h, best_c)
        if best_c < 0.38085:
            stagnation = 0
        else:
            stagnation += 1
        if stagnation >= stagnation_threshold:
            mutation_noise *= 0.5
            mutation_rate *= 0.5
            stagnation = 0
        mutate, mutation_noise, mutation_rate = adaptive_mutation(mutate, mutation_noise, mutation_rate)
    
    if not check_constraints(best_h):
        best_h = normalize(h)
    
    return best_h, best_c, n_points
