import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 200  # Chosen to balance speed and resolution
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
    if 'initial_h_values' in globals():
        initial_h_values = np.array(initial_h_values)
        # Normalize initial_h_values to meet the integral constraint
        initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    
    # Initialize population
    population_size = 50
    population = []
    for _ in range(population_size):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    def evaluate_fitness(h):
        # Compute cross-correlation and find max overlap
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    best_fitness = float('inf')
    best_h = np.copy(population[0])
    
    # Run for 100 generations or until time is up
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = [evaluate_fitness(h) for h in population]
        # Find the best individual
        best_idx = np.argmin(fitness)
        best_fitness = fitness[best_idx]
        best_h = np.copy(population[best_idx])
        if best_fitness < 0.38085:
            break
        # Selection: tournament selection
        tournament_size = 5
        parents = []
        for _ in range(population_size // 2):
            participants = np.random.choice(range(population_size), tournament_size, replace=False)
            winner = participants[np.argmin([fitness[i] for i in participants])]
            parents.append(population[winner])
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            if i + 1 < population_size:
                parent1 = parents[i]
                parent2 = parents[i + 1]
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
                child2 = np.concatenate((parent2[:crossover_point], parent1[crossover_point:]))
                offspring.append(child1)
                offspring.append(child2)
            else:
                offspring.append(parents[i])
        # Mutation
        mutation_rate = 0.8
        for i in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                idx = np.random.randint(n_points)
                delta = np.random.uniform(-0.05, 0.05)
                offspring[i][idx] += delta
                offspring[i][idx] = np.clip(offspring[i][idx], 0, 1)
                # Re-normalize to meet the integral constraint
                offspring[i] = offspring[i] * (n_points / 2) / np.sum(offspring[i])
        # Replace the population
        population = offspring
    
    final_fitness = evaluate_fitness(best_h)
    return (best_h, final_fitness, n_points)
