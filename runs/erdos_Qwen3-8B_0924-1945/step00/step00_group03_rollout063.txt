import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_required = n_points / 2
    
    def objective(x):
        h_current = x.copy()
        h_current = np.clip(h_current, 0, 1)
        conv = np.correlate(h_current, 1 - h_current, mode='full')
        c5 = np.max(conv) * dx
        return c5
    
    def constraint_sum(x):
        return np.sum(x) - sum_required
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h
    best_c5 = objective(h)
    
    start_time = time.time()
    
    try:
        result = optimize.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-5,
            options={'maxiter': 100}
        )
        h_current = np.clip(result.x, 0, 1)
        if np.abs(np.sum(h_current) - sum_required) < 1e-5:
            c5 = objective(h_current)
            if c5 < best_c5:
                best_h = h_current
                best_c5 = c5
    except:
        pass
    
    if time.time() - start_time < budget_s - 10:
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = np.clip(h + perturbation, 0, 1)
        h_perturbed = h_perturbed * (sum_required / np.sum(h_perturbed))
        h_perturbed = np.clip(h_perturbed, 0, 1)
        c5_perturbed = objective(h_perturbed)
        if c5_perturbed < best_c5:
            best_h = h_perturbed
            best_c5 = c5_perturbed
    
    best_h = np.clip(best_h, 0, 1)
    if np.abs(np.sum(best_h) - sum_required) > 1e-5:
        best_h = best_h * (sum_required / np.sum(best_h))
        best_h = np.clip(best_h, 0, 1)
    c5 = objective(best_h)
    
    return (best_h, c5, n_points)
