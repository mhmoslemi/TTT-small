    best_h = h.copy()

        # Compute full correlation
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        worst_lag = np.argmax(correlate_vals)
        
        # Compute gradient for worst lag
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * (2.0 / n_points)
        
        # Update momentum and h
        v = momentum_factor * v + learning_rate * gradients
        h = h - v
        
        # Clip and normalize
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Add noise
        if iteration % 100 == 0 and iteration > 0:
            noise = noise_scale * np.random.normal(0, 1, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Compute current C5
        current_c5 = np.max(correlate_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5

        # Early stopping
        if iteration > 0 and current_c5 - best_c5 < 1e-5:

    final_correlate_vals = np.correlate(best_h, 1 - best_h, mode="full") * (2.0 / n_points)

    best_h = best_h * (n_points / 2) / np.sum(best_h)
