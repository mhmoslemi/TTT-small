        return np.sum(h_flat) - n_points / 2

    cons = {'type': 'eq', 'fun': constraint_sum}
    result = opt.minimize(objective, h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-5, options={'maxiter': 100})
    h_best = result.x
    c5_best = result.fun

    # Perturbation and re-optimization
    for _ in range(2):
        noise = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturb = np.clip(h_best + noise, 0, 1)
        result = opt.minimize(objective, h_perturb, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-5, options={'maxiter': 100})
        h_best = result.x
        c5_best = result.fun

    h_best = np.clip(h_best, 0, 1)
    sum_h = np.sum(h_best)
    if abs(sum_h - n_points / 2) > 1e-6:
        h_best = h_best / sum_h * (n_points / 2)
    conv = np.correlate(h_best, 1 - h_best, mode='full')
    c5_final = np.max(conv) * dx
    return (h_best, c5_final, n_points)
