import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    target_sum = n_points / 2
    initial_h = initial_h_values.copy()
    h = np.clip(initial_h, 0, 1)
    h *= target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    elapsed_time = 0
    learning_rate = 0.05
    decay_rate = 0.995
    perturb_rate = 0.1
    max_iter = 1000
    no_improve_count = 0
    start_time = time.time()

    for i in range(max_iter):
        if elapsed_time >= budget_s:
            break
        cross_corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        idx = np.argmax(cross_corr)
        lag_k = idx - (n_points - 1)
        grad = np.zeros(n_points)
        for j in range(n_points):
            if j + lag_k < n_points:
                grad[j] += (1 - h[j + lag_k]) * (2.0 / n_points)
            if j - lag_k >= 0:
                grad[j] -= h[j - lag_k] * (2.0 / n_points)
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            h = np.zeros(n_points)
        else:
            h = np.clip(h, 0, 1) * (target_sum / current_sum)
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        if no_improve_count >= 50:
            break
        elapsed_time = time.time() - start_time
        if i % 100 == 0 and np.random.rand() < perturb_rate:
            noise = np.random.normal(0, 0.05, size=n_points)
            h += noise
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            if current_sum == 0:
                h = np.zeros(n_points)
            else:
                h = np.clip(h, 0, 1) * (target_sum / current_sum)
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        learning_rate *= decay_rate

    return best_h, best_c5, n_points
