import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Assuming this is provided
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    elapsed_time = 0
    max_iterations = 500
    step_size = 0.1
    decay_factor = 0.95
    epsilon = 1e-8
    improvement_threshold = 1e-4
    start_time = time.time()
    
    for iteration in range(max_iterations):
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
        
        cross_corr = np.correlate(h, 1 - h, mode="full") * dx
        max_index = np.argmax(cross_corr)
        if np.isnan(cross_corr[max_index]):
            break
        
        perturb_h = h.copy()
        perturb_h[max_index] += epsilon
        forward_diff = (evaluate_erdos_solution(perturb_h) - best_c5) / epsilon
        perturb_h[max_index] -= 2 * epsilon
        backward_diff = (best_c5 - evaluate_erdos_solution(perturb_h)) / epsilon
        grad = (forward_diff + backward_diff) / 2
        
        h -= step_size * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (target_sum / h_sum)
        
        new_c5 = evaluate_erdos_solution(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        if iteration % 10 == 0:
            if np.abs(new_c5 - best_c5) < improvement_threshold:
                break
        
        step_size *= decay_factor
    
    return (best_h, best_c5, n_points)
