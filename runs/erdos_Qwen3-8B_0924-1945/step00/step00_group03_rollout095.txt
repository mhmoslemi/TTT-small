import numpy as np
import random
import time
import scipy.signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = globals().get('initial_h_values', np.random.rand(70) * 2 - 1)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.clip(initial_h_values, 0, 1)
    h = h * (n_points/2) / np.sum(h)
    best_h = h.copy()
    best_c5 = scipy.signal.correlate(h, 1 - h, mode='full') * dx
    best_c5 = np.max(best_c5)
    initial_c5 = best_c5
    t_total = int(budget_s * 0.75)
    T0 = max(1.0, 1.0 / (1.0 + initial_c5 * 2.0)) * 0.5
    Tmax = int(t_total * 0.1)
    beta = 0.75
    gamma = 0.65
    lambda0 = 0.15 / dx
    start_time = time.time()
    
    for t in range(t_total):
        if time.time() - start_time > budget_s:
            break
        if t % 100 == 0:
            lambda0 *= 0.8 if t < t_total // 2 else 0.95
        noise = np.random.normal(0, lambda0, n_points)
        perturbed = h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        delta = (n_points / 2 - current_sum) / (n_points * 0.95)
        perturbed += delta * np.random.choice([-1, 1], size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed) if np.sum(perturbed) != 0 else perturbed
        
        new_c5 = scipy.signal.correlate(perturbed, 1 - perturbed, mode='full') * dx
        new_c5 = np.max(new_c5)
        
        if new_c5 < best_c5 or (np.random.rand() < np.exp(-(new_c5 - best_c5) / (T0 * (1 - t / Tmax)**(1 / beta)))):
            h = perturbed.copy()
            best_c5 = new_c5
            best_h = h.copy()
        
        if best_c5 < 0.3807:
            break
    
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = scipy.signal.correlate(h, 1 - h, mode='full') * dx
    final_c5 = np.max(final_c5)
    return (h, final_c5, n_points)
