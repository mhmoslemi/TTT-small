import numpy as np
import scipy.optimize as optimize
import time

initial_h_values = np.array([...])  # Placeholder for the initial values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    max_iter = int(budget_s / 0.1)  # 600s / 0.1s per iteration
    best_c5 = np.inf
    best_h = np.zeros(n_points)
    dt = 2.0 / n_points

    # Initialize from initial_h_values if available
    if initial_h_values.size > 0:
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        h = h * n_points / 2.0 / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * n_points / 2.0 / np.sum(h)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr * dt

    def update_h(h, idx, val):
        h_new = h.copy()
        h_new[idx] = val
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * n_points / 2.0 / np.sum(h_new)
        return h_new

    for t in range(max_iter):
        start_time = time.time()
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        lag = np.argmax(corr)
        overlap_x = h * (1 - np.roll(h, lag))
        threshold = np.max(overlap_x) * 0.8
        target_indices = np.where(overlap_x > threshold)[0]
        if len(target_indices) < 1:
            break

        candidates = []
        for idx in np.random.choice(target_indices, min(10, len(target_indices)), replace=False):
            # Flip
            h_candidate = update_h(h, idx, 1 - h[idx])
            candidates.append((h_candidate, compute_c5(h_candidate)))
            # Add noise
            h_candidate = update_h(h, idx, h[idx] + np.random.uniform(-0.05, 0.05))
            candidates.append((h_candidate, compute_c5(h_candidate)))

        if not candidates:
            break
        best_candidate, best_candidate_c5 = min(candidates, key=lambda x: x[1])
        if best_candidate_c5 < best_c5:
            best_c5 = best_candidate_c5
            best_h = best_candidate
        h = best_h
        elapsed = time.time() - start_time
        if elapsed > 0.1:
            break

    # Ensure feasibility
    h = np.clip(h, 0, 1)
    h = h * n_points / 2.0 / np.sum(h)
    c5 = compute_c5(h)
    return (h, c5, n_points)
