    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def clip_and_scale(h):
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
        if current_sum == 0.0:
            return np.zeros_like(h)
        scaling = target_sum / current_sum
        return h * scaling
    
    initial_h = np.array(initial_h_values, dtype=np.float64)
    initial_h = clip_and_scale(initial_h)
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    elapsed_time = 0.0
    
    def fitness(h):
        h = clip_and_scale(h)
        c5 = evaluate_erdos_solution(h)
        return -c5
    
    population_size = 50
    population = [np.copy(best_h) for _ in range(population_size)]
    for i in range(population_size):
        perturb = np.random.uniform(-0.05, 0.05, size=n_points)
        population[i] += perturb
        population[i] = clip_and_scale(population[i])
    
    best_fitness = fitness(best_h)
    no_improvement_count = 0
    start_time = time.time()
    
    while elapsed_time < budget_s:
        new_population = []
        for _ in range(population_size // 2):
            p1, p2 = np.random.choice(population, 2, replace=False)
            mutant = p1 + 0.5 * (p2 - p1)
            mutant = clip_and_scale(mutant)
            new_population.append(mutant)
        
        for candidate in new_population:
            c5 = evaluate_erdos_solution(candidate)
            if -c5 < best_fitness:
                best_fitness = -c5
                best_h = np.copy(candidate)
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        population = new_population
        if no_improvement_count >= 50:
            break
        
        elapsed_time = time.time() - start_time
    
    best_c5 = -best_fitness
    best_h = clip_and_scale(best_h)
    assert np.allclose(np.sum(best_h), target_sum, atol=1e-6)
    assert np.all(0 <= best_h <= 1)
    return (best_h, best_c5, n_points)
