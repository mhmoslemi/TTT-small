import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45  # starting with the provided initial_h_values
    h_values = initial_h_values
    n_points = len(h_values)
    dx = 2.0 / n_points
    sum_h = n_points / 2
    h = h_values.copy()
    h = h / np.sum(h) * sum_h

    def compute_c5(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * sum_h
        corr = correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def objective(x):
        x = np.clip(x, 0, 1)
        x = x / np.sum(x) * sum_h
        return compute_c5(x)

    def constraint(x):
        x = np.clip(x, 0, 1)
        x = x / np.sum(x) * sum_h
        return np.sum(x) - sum_h

    def generate_initial_population(size):
        population = []
        for _ in range(size):
            h = np.random.rand(n_points)
            h = h / np.sum(h) * sum_h
            population.append(h)
        return population

    def evaluate_population(population):
        scores = []
        for h in population:
            score = compute_c5(h)
            scores.append(score)
        return scores

    def update_population(population, scores, best_index):
        best = population[best_index]
        for i in range(len(population)):
            if random.random() < 0.3:
                population[i] = best + np.random.normal(0, 0.05, n_points)
                population[i] = np.clip(population[i], 0, 1)
                population[i] = population[i] / np.sum(population[i]) * sum_h
        return population

    def optimize_c5():
        population = generate_initial_population(50)
        best_score = float('inf')
        best_h = np.random.rand(n_points)
        best_h = best_h / np.sum(best_h) * sum_h
        best_score = compute_c5(best_h)
        start_time = time.time()
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            scores = evaluate_population(population)
            for i, score in enumerate(scores):
                if score < best_score:
                    best_score = score
                    best_h = population[i].copy()
            population = update_population(population, scores, np.argmin(scores))
        return best_h, best_score

    h_opt, c5_opt = optimize_c5()
    return h_opt, c5_opt, n_points
