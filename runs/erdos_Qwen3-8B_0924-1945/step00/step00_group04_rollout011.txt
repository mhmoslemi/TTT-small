import numpy as np
import scipy.optimize as opt
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    initial_h_values = np.array([0.4444444444444444, 0.4444444444444444, 0.5555555555555556, 0.4444444444444444, 0.5555555555555556, 0.5555555555555556, 0.5555555555555556, 0.4444444444444444, 0.5555555555555556, 0.5555555555555556, 0.5555555555555556, 0.4444444444444444, 0.5555555555555556, 0.5555555555555556, 0.5555555555555556, 0.5555555555555556, 0.5555555555555556, 0.5555555555555556, 0.5555555555555556, 0.5555555555555556, 0.5555555555555556, 0.5555555555555556, 0.5555555555555556, 0.4444444444444444, 0.5555555555555556, 0.5555555555555556, 0.5555555555555556, 0.5555555555555556, 0.4444444444444444, 0.5555555555555556, 0.5555555555555556, 0.5555555555555556, 0.5555555555555556, 0.4444444444444444, 0.4444444444444444, 0.5555555555555556, 0.5555555555555556, 0.5555555555555556, 0.5555555555555556, 0.4444444444444444, 0.4444444444444444, 0.5555555555555556, 0.5555555555555556, 0.5555555555555556, 0.4444444444444444, 0.5555555555555556])
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(x):
        h_val = x.copy()
        h_val = np.clip(h_val, 0, 1)
        h_val = h_val / np.sum(h_val) * target_sum
        corr = correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1)] * n_points

    result = opt.minimize(
        objective,
        h,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        method='SLSQP',
        options={'ftol': 1e-6, 'maxiter': 100}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * target_sum
    c5_bound = objective(best_h)

    return (best_h, c5_bound, n_points)
