import numpy as np
import scipy.optimize
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h.npy')
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    best_h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    def clip_and_scale(h, target_sum):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        h = h * scale
        return h
    
    def evaluate_candidate(h):
        h = clip_and_scale(h, target_sum)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def generate_population(mean_h, size=100):
        pop = np.random.rand(size, n_points)
        pop = mean_h + pop * 0.1
        pop = np.clip(pop, 0, 1)
        for i in range(size):
            h = pop[i]
            current_sum = np.sum(h)
            if current_sum == 0:
                h = np.random.rand(n_points)
            else:
                scale = target_sum / current_sum
                h = h * scale
            pop[i] = h
        return pop
    
    start_time = time.time()
    
    # Initial repair
    best_h = clip_and_scale(best_h, target_sum)
    best_c5 = evaluate_candidate(best_h)
    
    for _ in range(100):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 5:
            break
        
        population = generate_population(best_h)
        fitness = np.array([evaluate_candidate(h) for h in population])
        top_indices = np.argsort(fitness)[:50]
        top_population = population[top_indices]
        
        # Crossover
        children = []
        for i in range(50):
            if i % 2 == 0:
                parent1 = top_population[i // 2]
                parent2 = top_population[i // 2 + 1]
                child = np.zeros_like(parent1)
                mid = n_points // 2
                child[:mid] = parent1[:mid]
                child[mid:] = parent2[mid:]
                children.append(child)
            else:
                child = np.random.rand(n_points)
                child = clip_and_scale(child, target_sum)
                children.append(child)
        
        # Mutation
        mutated = []
        for child in children:
            if np.random.rand() < 0.1:
                perturbation = np.random.uniform(-0.05, 0.05, n_points)
                child += perturbation
                child = np.clip(child, 0, 1)
                current_sum = np.sum(child)
                if current_sum > 0:
                    child = child * (target_sum / current_sum)
            mutated.append(child)
        
        for candidate in mutated:
            c5 = evaluate_candidate(candidate)
            if c5 < best_c5:
                best_h = candidate.copy()
                best_c5 = c5
    
    # Local refinement
    def objective(h):
        h = clip_and_scale(h, target_sum)
        corr = np.correlate(h, 1 - h, mode="full")
        return -np.max(corr) * dx
    
    bounds = [(0, 1) for _ in range(n_points)]
    result = scipy.optimize.minimize(
        objective, best_h, method='SLSQP',
        bounds=bounds, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
        tol=1e-6, options={'maxiter': 100}
    )
    
    best_h = result.x
    best_c5 = evaluate_candidate(best_h)
    
    return best_h, best_c5, n_points
