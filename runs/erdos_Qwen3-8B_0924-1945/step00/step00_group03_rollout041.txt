import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Assuming initial construction is available
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    # Objective function to minimize
    def compute_c5(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    # Initial evaluation
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Time tracking
    start_time = time.time()
    elapsed_time = 0
    
    # Optimization parameters
    max_iterations = int(budget_s / 10)  # 10 iterations per second
    learning_rate = 0.05
    lambda_penalty = 1e-3
    num_lags = n_points - 1
    num_kernels = 10  # Number of kernel functions to consider
    
    # Function to create kernel arrays for all lags
    def create_kernels(h_vals, n_points):
        kernels = []
        for k in range(1, n_points):
            kernel = np.zeros(n_points)
            for i in range(n_points):
                kernel[i] = h_vals[i] * (1 - h_vals[(i + k) % n_points])
            kernels.append(kernel)
        return kernels
    
    # Run optimization
    for iter in range(max_iterations):
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
        
        # Create kernel arrays
        kernels = create_kernels(h, n_points)
        
        # Set up optimization problem
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        constraints = [t >= cp.sum(cp.multiply(kernels[k], h_var)) * dx for k in range(num_lags)]
        constraints += [0 <= h_var, h_var <= 1]
        constraints += [cp.sum(h_var) == n_points / 2]
        
        # Objective: minimize t + lambda * sum((h - 0.5)^2)
        objective = t + lambda_penalty * cp.pnorm(h_var - 0.5, 2)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == 'optimal':
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                np.sum(h_new) = n_points / 2
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            else:
                print("Optimal solution not found, using previous best")
        except:
            print("Optimization error, using previous best")
    
        # Gradient descent refinement
        for _ in range(5):
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Random perturbation
            perturbation = np.random.normal(0, 0.05, size=n_points)
            h += perturbation
            h = np.clip(h, 0, 1)
            np.sum(h) = n_points / 2
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Final checks
    best_h = np.clip(best_h, 0, 1)
    np.sum(best_h) = n_points / 2
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_points
