        h_clip = np.clip(h_candidate, 0, 1)

        c5 = np.max(corr) * (2.0 / n_points)

        h_clip = np.clip(h_candidate, 0, 1)
        h_sum = np.sum(h_clip)
        if h_sum == 0:
            return np.zeros_like(h_clip)
        h_norm = h_clip * (target_sum / h_sum)
        return h_norm

        start = time.time()

            if np.random.rand() < 0.3:

                mutant += np.random.normal(0, 0.05, n_points)
                mutant = clip_and_scale(mutant)
                new_population.append(mutant)
            else:
                parent1 = population[np.random.randint(population_size)]
                parent2 = population[np.random.randint(population_size)]
                child = (parent1 + parent2) / 2
                child = clip_and_scale(child)
                new_population.append(child)
