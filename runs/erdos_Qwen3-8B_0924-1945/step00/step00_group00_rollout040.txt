import numpy as np
import numpy.random as npr
import time
import scipy.optimize as sco

def run(seed=42, budget_s=600, **kwargs):
    npr.seed(seed)
    n_points = 80
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    if 'initial_h_values' in globals():
        initial_h_values = np.array(globals()['initial_h_values'])
    else:
        initial_h_values = np.random.rand(n_points)
        initial_h_values /= np.sum(initial_h_values)
        initial_h_values *= n_points / 2
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def gradient_step(h, m):
        gradients = np.zeros_like(h)
        for i in range(n_points):
            j = i - m
            if 0 <= j < n_points:
                gradients[i] = (1 - h[j]) * dx
        return gradients
    
    learning_rate = 0.1
    decay_rate = 0.995
    noise_scale = 0.05
    p_noise = 0.05
    max_iterations = 1000
    early_stop = 100
    best_c5 = float('inf')
    
    for it in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        current_c5 = np.max(correlate_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if current_c5 < 1e-8:
            break
        
        max_index = np.argmax(correlate_vals)
        m = max_index - (n_points - 1)
        if m < 0:
            m = 0
        elif m > n_points - 1:
            m = n_points - 1
        
        gradients = gradient_step(h, m)
        h -= learning_rate * gradients
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        if npr.random() < p_noise:
            noise = npr.normal(0, noise_scale, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        learning_rate *= decay_rate
    
    h = best_h
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    c5_bound = evaluate_c5(h)
    
    return (h.tolist(), c5_bound, n_points)
