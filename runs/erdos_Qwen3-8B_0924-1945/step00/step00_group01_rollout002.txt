import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Define initial h_values or create a random one
    if 'initial_h_values' in globals():
        h_vals = initial_h_values.copy()
        n_points = len(h_vals)
        # Normalize to satisfy sum constraint
        sum_h = np.sum(h_vals)
        h_vals = (h_vals * (n_points / 2)) / sum_h
    else:
        n_points = 47  # Default starting point
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    
    dx = 2.0 / n_points
    max_iterations = 100
    improvement_threshold = 1e-4
    early_stop_iterations = 5
    best_c5 = np.inf
    best_h = h_vals.copy()
    start_time = time.time()
    
    for iteration in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Compute all correlations
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5_values = corr * dx
        max_c5 = np.max(c5_values)
        
        # Early stop if no improvement for several iterations
        if max_c5 >= best_c5 - improvement_threshold:
            if iteration > early_stop_iterations:
                break
            best_c5 = max_c5
            best_h = h_vals.copy()
        
        # Choose a lag with high overlap
        lag_idx = np.argmax(c5_values)
        lag = lag_idx - (n_points - 1)
        
        # Clip values to [0, 1] and re-normalize
        h_clipped = np.clip(h_vals, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped != 0:
            h_clipped = (h_clipped * (n_points / 2)) / sum_clipped
        else:
            h_clipped = np.zeros(n_points)
        
        # Define the subproblem objective
        def objective(x):
            h = x.copy()
            if np.any(h < 0) or np.any(h > 1):
                return np.inf
            c = np.correlate(h, 1 - h, mode='full')[lag_idx] * dx
            return c
        
        # Define constraints
        def constraint_sum(x):
            return np.sum(x) - n_points / 2
        
        # Optimization
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        try:
            result = opt.minimize(
                objective,
                h_clipped,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-8, 'maxiter': 100}
            )
            h_vals = result.x
        except:
            h_vals = h_clipped.copy()
        
        # Clip again and re-normalize in case of numerical issues
        h_vals = np.clip(h_vals, 0, 1)
        sum_h = np.sum(h_vals)
        if sum_h != 0:
            h_vals = (h_vals * (n_points / 2)) / sum_h
        else:
            h_vals = np.zeros(n_points)
    
    # Final evaluation
    corr = np.correlate(h_vals, 1 - h_vals, mode='full')
    c5_final = np.max(corr) * dx
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_vals.copy()
    
    # Ensure feasible solution
    h_vals = np.clip(best_h, 0, 1)
    sum_h = np.sum(h_vals)
    if sum_h != 0:
        h_vals = (h_vals * (n_points / 2)) / sum_h
    else:
        h_vals = np.zeros(n_points)
    
    return (h_vals, c5_final, n_points)
