```python
import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 120
    dx = 2.0 / n_points
    initial_h_values = np.random.rand(n_points)
    initial_h_values = np.clip(initial_h_values, 0.0, 1.0)
    total = np.sum(initial_h_values)
    initial_h_values = initial_h_values * (n_points / 2.0) / total
    h = initial_h_values.copy()
    best_c5 = np.inf
    best_h = h.copy()
    
    start_time = time.time()
    
    for _ in range(300):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Randomly perturb 5 indices
        indices = np.random.choice(n_points, 5, replace=False)
        noise = np.random.normal(0, 0.1, 5)
        h[indices] += noise
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        h = h * (n_points / 2.0) / total
        
        # Evaluate the overlap integral
        try:
            shifted = np.correlate(h, 1.0 - h, mode="full")
            c5 = np.max(np.abs(shifted)) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        except:
            pass
    
    return (best_h, best_c5, n_points)
```
