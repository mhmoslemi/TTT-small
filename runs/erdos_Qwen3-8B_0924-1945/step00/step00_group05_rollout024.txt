import numpy as np
import random
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else None
    n_points = 500
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h = np.clip(initial_h_values, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (n_points / 2) / current_sum
        else:
            h = np.zeros(n_points)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                h[i] = 1
        h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlaps)
    
    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        return evaluate(x)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint}
    result = minimize(objective, h, method='SLSQP', bounds=[(0, 1) for _ in range(n_points)], constraints=cons, tol=1e-8)
    h_opt = result.x
    c5_bound = evaluate(h_opt)
    return (h_opt, c5_bound, n_points)
