import numpy as np
import random
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import spsolve
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    import os
    import sys
    import math
    import numpy as np
    import random
    import time
    import scipy.signal as signal
    import scipy.optimize as optimize
    
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial setup functions
    def create_initial_h(n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 if np.random.rand() < 0.5 else 0.0
        h /= np.sum(h)
        h *= (n_points / 2)
        return h
    
    def evaluate_c5(h, n_points):
        dx = 2.0 / n_points
        h = h.astype(np.float64)
        corr = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Parameters
    n_points = 45
    max_iterations = 100
    swarm_size = 20
    inertia_weight_start = 0.7
    inertia_weight_end = 0.4
    c1 = 1.5
    c2 = 1.5
    velocity_scale = 0.5
    
    # Initial swarm
    if 'initial_h_values' in kwargs and kwargs['initial_h_values'] is not None:
        initial_h = kwargs['initial_h_values']
        h_values = initial_h.copy()
    else:
        h_values = create_initial_h(n_points)
    
    # Initialize particles
    particles = np.array([h_values + np.random.rand(n_points) * 0.01 - 0.005 for _ in range(swarm_size)])
    particles = np.vstack([particles, h_values])
    velocities = np.random.rand(swarm_size + 1, n_points) * 0.1 - 0.05
    
    # Personal best and global best
    pbest = particles.copy()
    pbest_fitness = np.array([evaluate_c5(p, n_points) for p in particles])
    gbest = particles[np.argmin(pbest_fitness)]
    gbest_fitness = np.min(pbest_fitness)
    
    best_h = gbest
    best_c5 = gbest_fitness
    start_time = time.time()
    
    for iter in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Update inertia weight
        inertia_weight = inertia_weight_start - (inertia_weight_start - inertia_weight_end) * iter / max_iterations
        
        for i in range(len(particles)):
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (inertia_weight * velocities[i] +
                             c1 * r1 * (pbest[i] - particles[i]) +
                             c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Clip to [0, 1]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            
            # Normalize to ensure sum is n_points / 2
            particles[i] = (particles[i] * (n_points / 2)) / np.sum(particles[i])
        
        # Evaluate new positions
        for i in range(len(particles)):
            fitness = evaluate_c5(particles[i], n_points)
            if fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fitness
        
        # Update global best
        current_best = np.argmin(pbest_fitness)
        if pbest_fitness[current_best] < gbest_fitness:
            gbest = pbest[current_best].copy()
            gbest_fitness = pbest_fitness[current_best]
        
        # Track best solution
        if gbest_fitness < best_c5:
            best_h = gbest.copy()
            best_c5 = gbest_fitness
    
    # Final evaluation
    best_c5 = evaluate_c5(best_h, n_points)
    
    # Validate
    assert np.all(best_h >= 0.0) and np.all(best_h <= 1.0)
    assert np.isclose(np.sum(best_h), n_points / 2, atol=1e-8)
    
    return (best_h, best_c5, n_points)
