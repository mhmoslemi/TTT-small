import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points)
    initial_h = np.clip(initial_h, 0, 1)
    total = np.sum(initial_h)
    initial_h = initial_h * (n_points / 2) / total if total != 0 else np.full(n_points, 0.5)
    
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = float('inf')
    learning_rate = 0.1
    decay_rate = 0.999
    noise_scale = 0.05
    p_noise = 0.05
    max_iter = 1000
    early_stop = 100
    iter_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        correlate_vals = np.correlate(best_h, 1 - best_h, mode="full") * (2.0 / n_points)
        current_c5 = np.max(correlate_vals)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
        
        if current_c5 == best_c5:
            if iter_count % early_stop == 0:
                best_h = np.random.rand(n_points)
                best_h = np.clip(best_h, 0, 1)
                total = np.sum(best_h)
                best_h = best_h * (n_points / 2) / total if total != 0 else np.full(n_points, 0.5)
            iter_count += 1
            continue
        
        best_h = best_h.copy()
        max_idx = np.argmax(correlate_vals)
        lag = max_idx - n_points + 1
        
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = i - lag
            if 0 <= j < n_points:
                gradients[i] = (1 - best_h[j]) * (2.0 / n_points)
        
        best_h -= learning_rate * gradients
        best_h = np.clip(best_h, 0, 1)
        total = np.sum(best_h)
        best_h = best_h * (n_points / 2) / total if total != 0 else np.full(n_points, 0.5)
        
        if np.random.random() < p_noise:
            noise = np.random.normal(0, noise_scale, n_points)
            best_h += noise
            best_h = np.clip(best_h, 0, 1)
            total = np.sum(best_h)
            best_h = best_h * (n_points / 2) / total if total != 0 else np.full(n_points, 0.5)
        
        learning_rate *= decay_rate
        iter_count += 1
    
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h)
    best_h = best_h * (n_points / 2) / total if total != 0 else np.full(n_points, 0.5)
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full") * (2.0 / n_points))
    
    return best_h, final_c5, n_points
