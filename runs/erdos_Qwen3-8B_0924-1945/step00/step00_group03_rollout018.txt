        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Evaluate
        current_c5 = evaluate_erdos_solution(h_perturbed, dx)

            best_c5 = current_c5

        for _ in range(10):
            # Compute the overlap
            correlate = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')

            grad = np.zeros(n_points)
