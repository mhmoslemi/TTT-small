import numpy as np
import random
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Pre-imported initial solution if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(initial_h)
        # Scale to satisfy the integral constraint
        sum_initial = np.sum(initial_h)
        h_vals = (initial_h * (n_points / 2)) / sum_initial
    else:
        n_points = 100  # Start with 100 points, can be increased if needed
        h_vals = np.random.uniform(0, 1, n_points)
        h_vals = (h_vals * (n_points // 2)) / np.sum(h_vals)
    
    # Function to compute the max overlap
    def evaluate_overlap(h):
        dx = 2.0 / n_points
        shifted = np.roll(h, shift=1, axis=0)
        shifted[-1] = 0
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    def mutate(h, mutation_rate=0.01):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.uniform(-0.05, 0.05)
                h[i] = np.clip(h[i], 0.0, 1.0)
        # Re-normalize to satisfy the integral constraint
        h = (h * (n_points // 2)) / np.sum(h)
        return h
    
    def crossover(parent1, parent2):
        split = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:split], parent2[split:]])
        child = (child * (n_points // 2)) / np.sum(child)
        return child
    
    pop_size = 50
    population = [mutate(h_vals) for _ in range(pop_size)]
    best_h = h_vals
    best_c5 = evaluate_overlap(h_vals)
    generations = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s and generations < 100:
        # Evaluate fitness
        fitness = [evaluate_overlap(ind) for ind in population]
        # Select parents using tournament selection
        parents = []
        for _ in range(pop_size // 2):
            tournament = np.random.choice(range(pop_size), size=5)
            winner = tournament[np.argmin(fitness[tournament])]
            parents.append(population[winner])
        # Crossover
        offspring = []
        for i in range(0, pop_size // 2, 2):
            parent1 = parents[i]
            parent2 = parents[i + 1]
            child1 = crossover(parent1, parent2)
            child2 = crossover(parent2, parent1)
            offspring.append(child1)
            offspring.append(child2)
        # Mutate
        for i in range(len(offspring)):
            offspring[i] = mutate(offspring[i])
        # Replace the worst with the best offspring
        combined = population + offspring
        fitness_combined = [evaluate_overlap(ind) for ind in combined]
        sorted_indices = np.argsort(fitness_combined)
        new_population = [combined[i] for i in sorted_indices[:pop_size]]
        # Update best solution
        current_best = new_population[0]
        current_c5 = evaluate_overlap(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
            no_improvement = 0
        else:
            no_improvement += 1
        # Early stopping
        if no_improvement >= 10:
            break
        population = new_population
        generations += 1
    
    # Final evaluation
    final_c5 = evaluate_overlap(best_h)
    return best_h, final_c5, n_points
