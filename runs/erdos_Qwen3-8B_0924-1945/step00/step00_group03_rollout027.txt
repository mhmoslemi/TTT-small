import numpy as np
import scipy.optimize as opt
import time
import copy

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    h_values = np.array(initial_h_values, dtype=float) if initial_h_values is not None else np.random.rand(n_points)
    h_values = np.clip(h_values, 0, 1)
    sum_h = np.sum(h_values)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        h_values += (n_points / 2 - sum_h) / n_points
        h_values = np.clip(h_values, 0, 1)
    h_values = np.random.permutation(h_values)
    dx = 2.0 / n_points
    c5_bound = evaluate_erdos_solution(h_values, dx)
    best_h = h_values.copy()
    best_c5 = c5_bound
    start_time = time.time()
    max_iter = 100
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Perturb and evaluate
        perturb = np.random.normal(0, 0.1, n_points)
        h_values = h_values + perturb
        h_values = np.clip(h_values, 0, 1)
        sum_h = np.sum(h_values)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            h_values += (n_points / 2 - sum_h) / n_points
            h_values = np.clip(h_values, 0, 1)
        c5 = evaluate_erdos_solution(h_values, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_values.copy()
    # Local optimization
    def objective_func(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if np.abs(sum_x - n_points / 2) > 1e-6:
            x += (n_points / 2 - sum_x) / n_points
            x = np.clip(x, 0, 1)
        return evaluate_erdos_solution(x, dx)
    result = opt.minimize(objective_func, best_h, method='L-BFGS-B', options={'maxiter': 100})
    h_values = result.x
    h_values = np.clip(h_values, 0, 1)
    sum_h = np.sum(h_values)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        h_values += (n_points / 2 - sum_h) / n_points
        h_values = np.clip(h_values, 0, 1)
    c5 = evaluate_erdos_solution(h_values, dx)
    return (h_values, c5, n_points)

def evaluate_erdos_solution(h, dx):
    h_fft = np.fft.fft(h)
    one_minus_h = 1.0 - h
    one_minus_h_fft = np.fft.fft(one_minus_h)
    cross = np.fft.ifft(h_fft * np.conj(one_minus_h_fft))
    cross = np.abs(cross)
    max_overlap = np.max(cross)
    return max_overlap * dx
