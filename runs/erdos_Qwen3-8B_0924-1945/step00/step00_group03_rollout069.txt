import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

initial_h_values = None  # Placeholder for actual initial values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Use 200 points for sufficient resolution
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        # Use a uniform distribution as an initial guess
        h = np.random.uniform(0, 1, n_points)
        h = (h - h.min()) / (h.max() - h.min())
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Define the function to evaluate C5
    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Initial deterministic perturbation
    h_perturb = h.copy()
    h_perturb += np.random.uniform(-0.05, 0.05, n_points)
    h_perturb = np.clip(h_perturb, 0, 1)
    h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
    current_c5 = evaluate_c5(h_perturb)
    if current_c5 < best_c5:
        best_h = h_perturb.copy()
        best_c5 = current_c5
    
    # Optimization loop
    start_time = time.time()
    for _ in range(500):  # Limit iterations to stay within budget
        if time.time() - start_time > budget_s:
            break
        
        # Convex relaxation with bounded variables
        h_var = cp.Variable(n_points, lb=0, ub=1)
        t = cp.Variable(n=1)
        
        # Define the constraints for all lags
        constraints = [t >= 0]
        for k in range(n_points):
            for i in range(n_points):
                if i + k < n_points:
                    idx = i + k
                else:
                    idx = i + k - n_points
                expr = h_var[i] * (1 - h_var[idx])
                constraints.append(t >= expr)
        
        # Objective function with quadratic penalty
        lambda_penalty = 1e-3
        obj = t + lambda_penalty * cp.sum_squares(h_var - 0.5)
        
        # Solve the problem
        prob = cp.Problem(cp.Minimize(obj), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == 'OPTIMAL':
                h_new = np.array(h_var.value).flatten()
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                current_c5 = evaluate_c5(h_new)
                if current_c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = current_c5
        except:
            pass
        
        # Gradient descent refinement
        h_refine = best_h.copy()
        for _ in range(10):
            grad = np.zeros(n_points)
            for k in range(n_points):
                for i in range(n_points):
                    if i + k < n_points:
                        idx = i + k
                    else:
                        idx = i + k - n_points
                    grad[i] += (1 - h_refine[idx]) * (1 - h_refine[i])
                    grad[idx] -= h_refine[i] * (1 - h_refine[idx])
            grad = grad * dx
            h_refine -= 0.01 * grad
            h_refine = np.clip(h_refine, 0, 1)
            h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
            current_c5 = evaluate_c5(h_refine)
            if current_c5 < best_c5:
                best_h = h_refine.copy()
                best_c5 = current_c5
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h)
    
    # Return best solution
    return (best_h, final_c5, n_points)
