import numpy as np
import random
import time
import scipy.optimize as opt
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.clip(initial_h_values, 0, 1)
        current_sum = np.sum(h)
        if current_sum != 0:
            h = h * (n_points / 2) / current_sum
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = np.clip(h, 0, 1)
    
    def evaluate(h):
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        return max(overlaps)
    
    def evaluate_fitness(h):
        return evaluate(h)
    
    def normalize(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / current_sum
    
    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        return normalize(h)
    
    def crossover(parent1, parent2):
        idx = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:idx], parent2[idx:]])
        return clip_and_normalize(child)
    
    def mutate(h, mutation_rate=0.1, mutation_noise=0.05):
        if np.random.rand() < mutation_rate:
            noise = np.random.normal(0, mutation_noise, size=n_points)
            h = h + noise
        h = clip_and_normalize(h)
        return h
    
    def tournament_selection(population, k=5):
        selected = np.random.choice(population, size=k, replace=False)
        return selected[np.argmin([evaluate_fitness(ind) for ind in selected])]
    
    def evolution(population, mutation_rate=0.1, mutation_noise=0.05):
        new_population = []
        for _ in range(len(population) // 2):
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            child = crossover(parent1, parent2)
            child = mutate(child, mutation_rate, mutation_noise)
            new_population.append(child)
        new_population += [tournament_selection(population)]
        return new_population
    
    def fitness(h):
        return -evaluate_fitness(h)
    
    def constraints(h):
        return [np.sum(h) - n_points / 2]
    
    def objective(h):
        return -evaluate_fitness(h)
    
    def bounds_func(x):
        return [0.0, 1.0]
    
    start_time = time.time()
    current_best = clip_and_normalize(h)
    best_c5 = evaluate_fitness(current_best)
    
    population = [clip_and_normalize(np.random.rand(n_points)) for _ in range(50)]
    population = [clip_and_normalize(h) for h in population]
    best_h = current_best.copy()
    
    for gen in range(1000):
        if time.time() - start_time > budget_s:
            break
        population = evolution(population)
        evals = [evaluate_fitness(h) for h in population]
        best_idx = np.argmin(evals)
        best_candidate = population[best_idx]
        if evaluate_fitness(best_candidate) < best_c5:
            best_c5 = evaluate_fitness(best_candidate)
            best_h = best_candidate.copy()
    
    best_h = clip_and_normalize(best_h)
    if evaluate_fitness(best_h) < best_c5:
        best_c5 = evaluate_fitness(best_h)
    
    return (best_h, best_c5, n_points)
