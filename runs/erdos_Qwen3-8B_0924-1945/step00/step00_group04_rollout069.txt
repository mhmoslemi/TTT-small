import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=np.float64)
    else:
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (target_sum / np.sum(initial_h))

    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (target_sum / np.sum(initial_h))
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    elapsed_time = 0

    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]

    def mutate(h):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h = h + perturbation
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        return h

    for _ in range(10000):
        start_time = time.time()
        new_population = []
        for i in range(population_size):
            h = population[i]
            h_mut = mutate(h)
            new_population.append(h_mut)
        new_population = np.array(new_population)
        fitness = np.array([evaluate(h) for h in new_population])
        best_idx = np.argmin(fitness)
        best_new = new_population[best_idx]
        best_c5_new = fitness[best_idx]
        if best_c5_new < best_c5:
            best_h = best_new.copy()
            best_c5 = best_c5_new
        population = new_population
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            break

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h))
    assert np.all(0 <= best_h <= 1)
    assert np.isclose(np.sum(best_h), target_sum, atol=1e-6)
    return (best_h, best_c5, n_points)
