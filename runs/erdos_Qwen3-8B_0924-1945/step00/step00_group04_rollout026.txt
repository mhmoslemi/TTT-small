    best_h = h.copy()
    max_iter = 100
    tolerance = 1e-6

    def objective(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x) * dx
        if not np.isclose(sum_x, 1):
            x = x / sum_x * 1
        x = np.clip(x, 0, 1)

        return c5

    def constraint(x):
        sum_x = np.sum(x) * dx
        return sum_x - 1

    bounds = [(0, 1) for _ in range(n_points)]

                best_h = h.copy()

                break
