import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Choose a manageable number of samples
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Construct initial h values (if available) or a random one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.random.uniform(0, 1, n_points)
    
    # Ensure the sum constraint is satisfied
    h = h * (target_sum / np.sum(h))

    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (target_sum / sum_h)
        full_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(full_corr * dx)
        return c5

    def constraint(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (target_sum / sum_h)
        return h - h  # Ensures values are in [0, 1]

    # Set up bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint, 'jac': lambda h_vec: np.ones_like(h_vec)}]

    # Optimization setup
    start_time = time.time()
    result = opt.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        tol=1e-5,
        options={'maxiter': 1000}
    )

    # Ensure the result is within budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        return (h, objective(h), n_points)

    # Extract the best solution
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h != 0:
        best_h = best_h * (target_sum / sum_h)
    c5_bound = objective(best_h)
    return (best_h, c5_bound, n_points)
