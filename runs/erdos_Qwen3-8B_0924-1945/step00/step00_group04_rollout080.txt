import numpy as np
import scipy.optimize as opt
import random
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h_values = np.array([0.71428571, 0.71428571, 0.71428571, 0.71428571, 0.71428571,
                                 0.71428571, 0.71428571, 0.71428571, 0.71428571, 0.71428571,
                                 0.71428571, 0.71428571, 0.71428571, 0.71428571, 0.71428571,
                                 0.71428571, 0.71428571, 0.71428571, 0.71428571, 0.71428571,
                                 0.71428571, 0.71428571, 0.71428571, 0.71428571, 0.71428571,
                                 0.71428571, 0.71428571, 0.71428571, 0.71428571, 0.71428571,
                                 0.71428571, 0.71428571, 0.71428571, 0.71428571, 0.71428571,
                                 0.71428571, 0.71428571, 0.71428571, 0.71428571, 0.71428571,
                                 0.71428571, 0.71428571, 0.71428571, 0.71428571, 0.71428571,
                                 0.71428571, 0.71428571, 0.71428571, 0.71428571, 0.71428571,
                                 0.71428571, 0.71428571, 0.71428571, 0.71428571, 0.71428571,
                                 0.71428571, 0.71428571, 0.71428571, 0.71428571, 0.71428571])

    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return h * scale

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    def objective(h):
        return -evaluate(h)

    def constraint_sum(h):
        return np.sum(h) - target_sum

    def constraint_bounds(h):
        return np.all((0 <= h) & (h <= 1))

    def initial_h():
        h = repair(initial_h_values.copy())
        return h

    h = initial_h()
    best_h = h.copy()
    best_c5 = evaluate(h)
    start_time = time()

    population_size = 100
    mutation_rate = 0.1
    max_generations = 200

    for gen in range(max_generations):
        if time() - start_time > budget_s:
            break
        population = []
        for _ in range(population_size):
            candidate = h.copy()
            mutation = np.random.uniform(-0.05, 0.05, size=n_points)
            candidate += mutation
            candidate = repair(candidate)
            population.append(candidate)
        
        population.sort(key=lambda x: evaluate(x))
        top_candidates = population[:50]

        new_population = []
        for i in range(0, 50, 2):
            if i + 1 < 50:
                parent1 = top_candidates[i]
                parent2 = top_candidates[i + 1]
                child = np.zeros_like(parent1)
                split = np.random.randint(1, n_points - 1)
                child[:split] = parent1[:split]
                child[split:] = parent2[split:]
                child = repair(child)
                new_population.append(child)
            else:
                new_population.append(top_candidates[i])
        
        for i in range(len(new_population)):
            if np.random.rand() < mutation_rate:
                mutation = np.random.uniform(-0.05, 0.05, size=n_points)
                new_population[i] += mutation
                new_population[i] = repair(new_population[i])

        population = new_population
        population.sort(key=lambda x: evaluate(x))
        h = population[0]
        current_c5 = evaluate(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    if time() - start_time < budget_s:
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}, {'type': 'ineq', 'fun': constraint_bounds}]
        result = opt.minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6, options={'maxiter': 100})
        if result.success:
            best_h = result.x
            best_c5 = evaluate(best_h)

    return best_h, best_c5, n_points
