import numpy as np
import scipy.optimize as opt
import copy
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 45 if initial_h_values is None else len(initial_h_values)
    if n_points < 45:
        n_points = 45
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h = (h / np.sum(h)) * (n_points / 2)
        h = np.clip(h, 0, 1)
    else:
        h = np.random.rand(n_points)
        h = (h / np.sum(h)) * (n_points / 2)
        h = np.clip(h, 0, 1)

    def objective(h):
        h = np.clip(h, 0, 1)
        h = (h / np.sum(h)) * (n_points / 2)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1)] * n_points)
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = (best_h / np.sum(best_h)) * (n_points / 2)
    best_h = np.clip(best_h, 0, 1)
    final_c5 = objective(best_h)
    return (best_h, final_c5, n_points)
