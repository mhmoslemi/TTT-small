import numpy as np
import scipy.interpolate
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 450
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * n_points / 2.0
        h = np.clip(h, 0, n_points)
        h = h / (n_points / 2.0)
    
    h = np.clip(h, 0, 1)
    h = h * n_points / 2.0
    h = np.clip(h, 0, n_points)
    h = h / (n_points / 2.0)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    dt = 0.1
    max_iter = int((budget_s - (time.time() - start_time)) / dt)
    
    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * 2.0 / n_points
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Targeted mutation
        k_max = np.argmax(corr)
        indices = np.arange(max(0, k_max - 10), min(len(corr), k_max + 11))
        selected_indices = np.random.choice(indices, size=10, replace=False)
        for i in selected_indices:
            contribution = (2.0 / n_points) * corr[i]
            delta = -sign(contribution) * 0.1 / (1 + t / 500)
            h[i] += delta
            h[i] = np.clip(h[i], 0, 1)
        
        # Global mutation
        for _ in range(5):
            i = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.2 * np.exp(-t / 300))
            h[i] += delta
            h[i] = np.clip(h[i], 0, 1)
        
        # Rescale
        h = h * n_points / 2.0
        h = np.clip(h, 0, n_points)
        h = h / (n_points / 2.0)
        h = np.clip(h, 0, 1)
        
        if np.isnan(np.sum(h)) or np.any(h < 0) or np.any(h > 1) or abs(np.sum(h) - n_points / 2.0) > 1e-6:
            h = best_h.copy()
    
    h = best_h.copy()
    h = np.clip(h, 0, 1)
    h = h * n_points / 2.0
    h = np.clip(h, 0, n_points)
    h = h / (n_points / 2.0)
    h = np.clip(h, 0, 1)
    corr = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(corr) * 2.0 / n_points
    return (h, final_c5, n_points)

def sign(x):
    return np.sign(x)
