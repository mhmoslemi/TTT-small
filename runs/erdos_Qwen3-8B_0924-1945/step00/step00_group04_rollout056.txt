import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    target_sum = n_points / 2
    h = np.copy(initial_h_values)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    elapsed_time = 0
    learning_rate = 0.05
    decay_rate = 0.995
    max_iterations = 1000
    no_improvement = 0
    grad = np.zeros(n_points)
    dx = 2.0 / n_points

    for _ in range(max_iterations):
        start_time = time.time()
        cross_corr = np.correlate(h, 1 - h, mode="full") * dx
        idx = np.argmax(cross_corr)
        lag_k = idx - (n_points - 1)
        grad.fill(0.0)
        for j in range(n_points):
            if j + lag_k < n_points:
                grad[j] += (1 - h[j + lag_k]) * dx
            if j - lag_k >= 0:
                grad[j] -= h[j - lag_k] * dx
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            h = np.zeros(n_points)
        else:
            h = h * (target_sum / current_sum)
        new_c5 = evaluate_erdos_solution(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
        elapsed_time += time.time() - start_time
        if elapsed_time >= budget_s:
            break
        if no_improvement >= 50:
            break
        if learning_rate < 1e-5:
            learning_rate = 0.05
        else:
            learning_rate *= decay_rate
        if np.random.rand() < 0.1:
            perturb = np.random.uniform(-0.05, 0.05, size=n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
    return best_h, best_c5, n_points
