import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None  # Placeholder, replaced by actual initial values at runtime

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def objective(h):
        return compute_c5(h)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    constraints = {'type': 'eq', 'fun': constraint}
    
    for _ in range(500):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        worst_lag = np.argmax(correlate_vals)
        
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * dx
        
        learning_rate = 0.05 * (1 - _ / 500)
        h_new = h - learning_rate * gradients
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        noise_scale = 0.05
        noise = noise_scale * np.random.normal(0, 1, n_points)
        h_new += noise
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        current_c5 = compute_c5(h_new)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
        
        h = h_new
    
    return (best_h, best_c5, n_points)
