import numpy as np
import random

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)

    def compute_cross_corr(h, dx):
        n = len(h)
        corr = np.zeros(2*n - 1)
        for k in range(2*n - 1):
            if k < n:
                corr[k] = dx * np.sum(h[:n - k] * h[k:])
            else:
                corr[k] = dx * np.sum(h[:n - (k - n)] * h[k - n:])
        return corr

    def update_h(h, dx, max_k):
        n = len(h)
        grad = np.zeros(n)
        if max_k < n:
            for i in range(n - max_k):
                grad[i] += dx * h[i + max_k]
                grad[i + max_k] += dx * h[i]
        else:
            k_wrap = max_k - n
            for i in range(n - k_wrap):
                grad[i] += dx * h[i + k_wrap]
                grad[i + k_wrap] += dx * h[i]
        direction = -grad
        step_size = 0.1
        new_h = h + step_size * direction
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        scaling_factor = (n / 2) / new_sum if new_sum != 0 else 1.0
        new_h = new_h * scaling_factor
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        scaling_factor = (n / 2) / new_sum if new_sum != 0 else 1.0
        new_h = new_h * scaling_factor
        return new_h

    for _ in range(500):
        if _ * 0.001 > budget_s:
            break
        corr = compute_cross_corr(h, dx)
        max_k = np.argmax(corr)
        new_h = update_h(h, dx, max_k)
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            h = new_h
            best_c5 = new_c5
            best_h = h.copy()
        if np.abs(new_c5 - best_c5) < 1e-6:
            break
        if random.random() < 0.05:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            new_h = h + noise
            new_h = np.clip(new_h, 0.0, 1.0)
            new_sum = np.sum(new_h)
            scaling_factor = (n / 2) / new_sum if new_sum != 0 else 1.0
            new_h = new_h * scaling_factor
            new_h = np.clip(new_h, 0.0, 1.0)
            new_sum = np.sum(new_h)
            scaling_factor = (n / 2) / new_sum if new_sum != 0 else 1.0
            new_h = new_h * scaling_factor
            new_c5 = evaluate_erdos_solution(new_h, dx)
            if new_c5 < best_c5:
                h = new_h
                best_c5 = new_c5
                best_h = h.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    final_sum = np.sum(best_h)
    scaling_factor = (n_points / 2) / final_sum if final_sum != 0 else 1.0
    best_h = best_h * scaling_factor
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (best_h, final_c5, n_points)
