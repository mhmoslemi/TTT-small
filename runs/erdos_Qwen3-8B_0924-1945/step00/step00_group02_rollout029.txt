import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250  # Choose a smaller number of points to stay within budget
    dx = 2.0 / n_points
    initial_h_values = np.random.uniform(0, 1, n_points)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = np.inf
    
    # Precompute 1 - h for faster evaluation
    one_minus_h = 1.0 - h
    
    def evaluate_c5(h):
        corr = np.correlate(h, one_minus_h, mode="full")
        return np.max(corr) * dx
    
    # Cooling schedule parameters
    T_initial = 100.0
    T_final = 0.1
    alpha = 0.995
    sigma_initial = 0.05
    sigma_final = 0.01
    max_iterations = int(budget_s / 0.1)
    iteration = 0
    start_time = time.time()
    
    while iteration < max_iterations and time.time() - start_time < budget_s:
        # Generate Gaussian noise
        sigma = sigma_initial * (sigma_final / sigma_initial) ** (iteration / max_iterations)
        indices = np.random.choice(n_points, size=5, replace=False)
        noise = np.random.normal(0, sigma, size=n_points)
        new_h = h + noise
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(new_h)
        
        # Compute energy difference
        delta = new_c5 - best_c5
        
        # Acceptance probability
        if delta < 0 or np.random.rand() < np.exp(-delta / T_initial):
            h = new_h
            best_c5 = new_c5
            best_h = h.copy()
        
        # Cool down
        T_initial *= alpha
        
        iteration += 1
    
    # Final validation
    h = best_h
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    final_c5 = np.max(corr) * dx
    return h, final_c5, n_points
