import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):

    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_overlap = np.max(conv) * dx
        return max_overlap

        return np.sum(h_vec) - n_points / 2
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    start_time = time.time()
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-5,
        options={'maxiter': 100, 'ftol': 1e-5}
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s - 10:
        return (result.x, result.fun, n_points)
    
    if result.success:
        h_opt = np.clip(result.x, 0, 1)
        if np.isclose(np.sum(h_opt), n_points / 2, atol=1e-5):
            return (h_opt, result.fun, n_points)
    
    h_perturbed = h + np.random.normal(0, 0.1, n_points)
    h_perturbed = np.clip(h_perturbed, 0, 1)
    h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
    
    result = opt.minimize(
        objective,
        h_perturbed,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-5,
        options={'maxiter': 100, 'ftol': 1e-5}
    )
