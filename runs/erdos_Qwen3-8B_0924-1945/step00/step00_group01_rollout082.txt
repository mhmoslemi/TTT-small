import numpy as np
import random
import time
from scipy.signal import correlate
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    max_generations = 100
    population_size = 50
    mutation_rate = 0.05
    tournament_size = 5
    crossover_rate = 0.8
    stagnation_threshold = 10
    
    def initial_population(n_points, initial_h_values=None):
        if initial_h_values is not None:
            h_vals = (initial_h_values * (n_points / 2)) / np.sum(initial_h_values)
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
            return h_vals
        else:
            h_vals = np.random.rand(n_points)
            h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
            return h_vals
    
    def fitness(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def crossover(parent1, parent2):
        if np.random.rand() < crossover_rate:
            point = np.random.randint(1, len(parent1) - 1)
            child1 = np.concatenate([parent1[:point], parent2[point:]])
            child2 = np.concatenate([parent2[:point], parent1[point:]])
            return child1, child2
        else:
            return parent1.copy(), parent2.copy()
    
    def mutate(individual):
        mutation = np.random.normal(0, mutation_rate, len(individual))
        new_individual = individual + mutation
        new_individual = np.clip(new_individual, 0, 1)
        new_individual = new_individual / np.sum(new_individual) * (n_points / 2)
        return new_individual
    
    def tournament_selection(population, fitnesses):
        participants = np.random.choice(len(population), tournament_size, replace=False)
        best_idx = np.argmin(fitnesses[participants])
        return population[participants[best_idx]]
    
    initial_h = initial_population(n_points, initial_h_values)
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        new_h = initial_population(n_points)
        population.append(new_h)
    
    population = np.array([np.clip(h, 0, 1) for h in population])
    population = np.array([h / np.sum(h) * (n_points / 2) for h in population])
    fitnesses = np.array([fitness(h) for h in population])
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx]
    best_c5 = fitnesses[best_idx]
    
    generation = 0
    no_improvement = 0
    
    while generation < max_generations and time.time() - start_time < budget_s:
        if no_improvement >= stagnation_threshold:
            break
        new_population = []
        for _ in range(population_size):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.append(child1)
            new_population.append(child2)
        
        new_population = np.array(new_population)
        new_population = np.array([np.clip(h, 0, 1) for h in new_population])
        new_population = np.array([h / np.sum(h) * (n_points / 2) for h in new_population])
        new_fitnesses = np.array([fitness(h) for h in new_population])
        
        indices = np.argsort(new_fitnesses)
        new_population = new_population[indices[:population_size]]
        new_fitnesses = new_fitnesses[indices[:population_size]]
        
        population = new_population
        fitnesses = new_fitnesses
        
        if np.min(fitnesses) < best_c5:
            best_c5 = np.min(fitnesses)
            best_h = population[np.argmin(fitnesses)]
            no_improvement = 0
        else:
            no_improvement += 1
        
        generation += 1
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    best_c5 = fitness(best_h)
    
    return (best_h, best_c5, n_points)
