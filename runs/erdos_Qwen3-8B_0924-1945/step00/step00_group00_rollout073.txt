import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    
    # Default parameters
    n_points = 45  # Start with 45 points, can increase if needed
    dx = 2.0 / n_points
    total_integral = n_points / 2  # Required sum for the integral constraint
    
    # If initial_h_values provided, scale and use it as starting point
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0, 1)
        h = h * total_integral / np.sum(h)
    else:
        # Random initial vector
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * total_integral / np.sum(h)
    
    def compute_c5(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * total_integral / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(correlate_vals)
    
    # Optimizer: Use Particle Swarm Optimization (PSO)
    # Adapted from scipy.optimize implementation
    def pso(obj_func, bounds, initial_positions, n_particles=20, max_iter=100, seed=42):
        np.random.seed(seed)
        n_vars = len(initial_positions)
        positions = np.array([np.random.rand(n_vars) * (b[1] - b[0]) + b[0] for b in bounds])
        velocities = np.random.rand(n_particles, n_vars) * 0.1
        pbest_positions = positions.copy()
        pbest_values = np.array([obj_func(p) for p in positions])
        gbest_position = pbest_positions[np.argmin(pbest_values)]
        gbest_value = np.min(pbest_values)
        
        inertia = 0.7
        cognitive = 1.5
        social = 1.5
        for _ in range(max_iter):
            r1, r2 = np.random.rand(n_particles), np.random.rand(n_particles)
            velocities = inertia * velocities + cognitive * r1 * (pbest_positions - positions) + social * r2 * (gbest_position - positions)
            positions = positions + velocities
            positions = np.clip(positions, 0, 1)
            positions = positions * total_integral / np.sum(positions, axis=1, keepdims=True)
            
            current_values = np.array([obj_func(p) for p in positions])
            mask = current_values < pbest_values
            pbest_positions[mask] = positions[mask]
            pbest_values[mask] = current_values[mask]
            
            gbest_value = np.min(pbest_values)
            gbest_position = pbest_positions[np.argmin(pbest_values)]
        
        return gbest_position, gbest_value
    
    # Define bounds for h (must be between 0 and 1)
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run PSO
    start_time = time.time()
    best_h, best_c5 = pso(compute_c5, bounds, h, n_particles=20, max_iter=100)
    
    # Check time and output
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        # If time is exceeded, return the current best solution
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * total_integral / np.sum(best_h)
        best_c5 = compute_c5(best_h)
    else:
        # Ensure bounds and integral constraint
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * total_integral / np.sum(best_h)
        best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n_points)
