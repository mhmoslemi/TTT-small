import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 47 if initial_h_values is not None else 100
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h_vals = (initial_h_values * (n_points / 2)) / np.sum(initial_h_values)
    else:
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)

    best_h_vals = h_vals.copy()
    best_c5 = compute_c5(h_vals, dx)
    elapsed = 0

    def compute_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def optimize_for_lag(h_vals, k, dx):
        n = len(h_vals)
        indices = np.arange(n)
        def objective(x):
            x = np.clip(x, 0, 1)
            x_sum = np.sum(x)
            if np.isclose(x_sum, n / 2):
                return np.sum(x[indices] * (1 - x[(indices + k) % n]))
            else:
                return np.inf
        def constraint(x):
            return np.sum(x) - n / 2
        bounds = [(0, 1) for _ in range(n)]
        result = opt.minimize(objective, h_vals, bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}], method='SLSQP')
        if result.success:
            return result.x
        else:
            return np.clip(h_vals, 0, 1)

    for _ in range(100):
        time.sleep(0.01)
        elapsed += 0.01
        if elapsed > budget_s:
            break
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - n_points + 1
        k = np.clip(k, 0, n_points - 1)
        new_h = optimize_for_lag(h_vals, k, dx)
        new_h = np.clip(new_h, 0, 1)
        new_sum = np.sum(new_h)
        if np.isclose(new_sum, n / 2):
            h_vals = new_h
        else:
            scale = (n / 2) / new_sum
            h_vals = np.clip(new_h * scale, 0, 1)
        current_c5 = compute_c5(h_vals, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_vals = h_vals.copy()
        elif current_c5 > best_c5:
            break

    final_c5 = compute_c5(best_h_vals, dx)
    return (best_h_vals, final_c5, n_points)
