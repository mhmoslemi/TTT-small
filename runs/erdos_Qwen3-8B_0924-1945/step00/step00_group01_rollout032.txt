```python
import numpy as np
import scipy.optimize as optimize
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 96
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h_vals = initial_h_values.copy()
        h_vals = (h_vals * n_points / 2) / np.sum(h_vals)
    else:
        h_vals = np.zeros(n_points)
        h_vals[:n_points // 2] = 1
        h_vals = h_vals / (np.sum(h_vals) / (n_points / 2))
    
    max_iterations = 100
    no_improvement_threshold = 5
    best_c5 = np.inf
    best_h = h_vals.copy()
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 100:
            break
        
        # Compute all lags
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5_candidates = corr * dx
        current_max_c5 = np.max(c5_candidates)
        
        if current_max_c5 < best_c5:
            best_c5 = current_max_c5
            best_h = h_vals.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count >= no_improvement_threshold:
                break
        
        # Select the worst lag
        worst_k = np.argmax(corr)
        offset = worst_k - n_points + 1
        if offset < 0:
            offset += n_points
        worst_k = offset
        
        # Define objective for the worst lag
        def objective(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            h = h / (np.sum(h) / (n_points / 2))
            corr = np.correlate(h, 1 - h, mode='full')
            return corr[worst_k] * dx
        
        # Define constraints
        def equality_constraint(x):
            return np.sum(x) - n_points / 2
        
        # Optimization
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            objective, 
            h_vals, 
            bounds=bounds, 
            method='L-BFGS-B',
            constraints={'type': 'eq', 'fun': equality_constraint}
        )
        
        if result.success:
            h_vals = result.x
        else:
            h_vals = best_h.copy()
        
        # Clip and normalize
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals / (np.sum(h_vals) / (n_points / 2))
    
    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx
    return best_h, c5_bound, n_points
```
