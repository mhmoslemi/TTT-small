import numpy as np
import scipy.optimize as opt
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load("initial_h_values.npy")
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def repair_h(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        h = h * scale
        return h

    def evaluate_candidate(h):
        h = repair_h(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def objective_func(h):
        return -evaluate_candidate(h)

    def constraint_sum(h):
        return np.sum(h) - target_sum

    def bounds_func(h):
        return [(0, 1) for _ in range(len(h))]

    def initial_h():
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        h = h * scale
        return h

    best_h = initial_h()
    best_c5 = evaluate_candidate(best_h)
    elapsed = 0
    time_per_gen = 0.1

    for _ in range(100):
        if elapsed >= budget_s:
            break
        population = [best_h + np.random.uniform(-0.05, 0.05, n_points) for _ in range(100)]
        population = [repair_h(h) for h in population]
        fitness = [evaluate_candidate(h) for h in population]
        top_indices = np.argsort(fitness)[-50:]
        best_h = population[top_indices[0]]
        best_c5 = evaluate_candidate(best_h)
        elapsed += time_per_gen

    def local_refinement(h):
        h_opt = h.copy()
        cons = [{'type': 'eq', 'fun': constraint_sum, 'jac': lambda x: np.ones_like(x)}]
        bounds = bounds_func(h_opt)
        result = opt.minimize(objective_func, h_opt, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if result.success:
            h_opt = result.x
        else:
            h_opt = h.copy()
        return h_opt

    best_h = local_refinement(best_h)
    best_c5 = evaluate_candidate(best_h)

    return (best_h, best_c5, n_points)
