```python
import numpy as np
import scipy.signal
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 45
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.max(scipy.signal.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    
    v = np.zeros(n_points)
    momentum = 0.9
    learning_rate = 0.1
    noise_scale = 0.05
    noise_freq = 100
    max_iter = int(budget_s / 1)
    
    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        correlate_vals = scipy.signal.correlate(h, 1 - h, mode='full') * dx
        worst_lag = np.argmax(correlate_vals)
        
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * dx
        
        v = momentum * v + learning_rate * gradients
        h = h - v
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        if t % noise_freq == 0:
            noise = noise_scale * np.random.normal(0, 1, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        current_c5 = np.max(correlate_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        learning_rate *= 0.9995
    
    return (best_h, best_c5, n_points)
```
