import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    h = np.array(initial_h_values)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    learning_rate = 0.1
    momentum = 0.9
    noise_scale = 0.05
    decay_rate = 0.999
    iteration = 0
    max_iterations = int(budget_s / 1.5)
    
    while time.time() - start_time < budget_s:
        if iteration >= max_iterations:
            break
        
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        worst_lag = np.argmax(correlate_vals)
        
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * (2.0 / n_points)
        
        if iteration == 0:
            v = np.zeros(n_points)
        
        v = momentum * v + learning_rate * gradients
        h = h - v
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        if np.random.rand() < 0.05:
            h += noise_scale * np.random.normal(0, 1, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        current_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        learning_rate *= decay_rate
        iteration += 1
    
    return (best_h, best_c5, n_points)
