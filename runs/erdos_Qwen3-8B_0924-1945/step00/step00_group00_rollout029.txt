    max_iter = 500
    iter = 0

    while time.time() - start_time < budget_s and iter < max_iter:

            best_c5 = current_c5
            best_h = h.copy()
        if abs(current_c5 - best_c5) < 1e-5:
            break
        correlate_vals = np.correlate(h, 1 - h, mode='full') * (2.0 / n_points)
        worst_lag = np.argmax(correlate_vals)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            grad[i] = (1 - h[j]) * (2.0 / n_points)
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
