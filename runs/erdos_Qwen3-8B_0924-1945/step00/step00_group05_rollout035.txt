import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500  # Choose a moderate number of points
    dx = 2.0 / n_points
    initial_h_values = np.random.rand(n_points)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    sum_initial = np.sum(initial_h_values)
    initial_h_values = initial_h_values * (n_points / 2) / sum_initial
    
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = float('inf')
    max_iterations = 500
    learning_rate = 0.01
    momentum = 0.9
    perturb_prob = 0.05
    time_start = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - time_start > budget_s:
            break
        
        # Compute the overlap
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(overlap)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the worst shift
        worst_shift = np.argmax(overlap)
        
        # Compute gradient for worst shift
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + worst_shift) % n_points
            grad[i] += (1 - h[j])  # Contribution from h[i] in (1-h)
            grad[j] -= h[i]        # Contribution from h[j] in h
        
        # Update with momentum
        grad = grad * learning_rate
        h = h - grad
        
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        
        # Enforce sum constraint
        sum_h = np.sum(h)
        if sum_h == 0:
            h = initial_h_values.copy()
            sum_h = np.sum(h)
            h = h * (n_points / 2) / sum_h
        else:
            h = h * (n_points / 2) / sum_h
        
        # Add perturbation with probability
        if np.random.rand() < perturb_prob:
            noise = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + noise, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = initial_h_values.copy()
                sum_h = np.sum(h)
                h = h * (n_points / 2) / sum_h
            else:
                h = h * (n_points / 2) / sum_h
        
        # Reduce learning rate if no improvement
        if iteration % 50 == 0:
            learning_rate *= 0.995
    
    # Final check for best solution
    overlap = np.correlate(best_h, 1 - best_h, mode="full") * dx
    best_c5 = np.max(overlap)
    
    return (best_h, best_c5, n_points)
