import numpy as np
import scipy.signal as signal
import cvxpy as cp
import time

initial_h_values = None  # Placeholder for optional input

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Choose n_points <= 1000, start with 100 for efficiency
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h = np.array(initial_h_values[:n_points], dtype=np.float64)
    else:
        # Generate initial random h with sum h = n_points/2
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
    
    # Validate initial solution
    def evaluate(h, dx):
        corr = signal.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_c5 = evaluate(h, dx)
    best_h = h.copy()
    
    # Optimization parameters
    max_iterations = 50
    learning_rate = 0.01
    lambda_penalty = 1e-3
    early_stop_threshold = 0.3807
    time_limit = budget_s * 0.8  # Allocate 80% of budget for optimization
    
    start_time = time.time()
    
    for _ in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > time_limit:
            break
        
        # Convert h to variable
        h_var = cp.Variable(n_points, lb=0, ub=1)
        t = cp.Variable()
        
        # Objective: minimize t + lambda * sum((h - 0.5)^2)
        objective = t + lambda_penalty * cp.sum((h_var - 0.5)**2)
        
        # Constraints: t >= correlate(h, 1 - h) for all lags
        constraints = []
        for k in range(n_points):
            # Handle wrap-around for k > n_points
            indices = np.arange(n_points)
            if k < n_points:
                indices = (indices + k) % n_points
            else:
                indices = np.roll(np.arange(n_points), k % n_points)
            corr_val = cp.sum(h_var[indices] * (1 - h_var) * dx)
            constraints.append(t >= corr_val)
        
        # Constraint: sum(h) = n_points / 2
        constraints.append(cp.sum(h_var) == n_points / 2)
        
        # Solve the convex optimization problem
        try:
            problem = cp.Problem(cp.Minimize(objective), constraints)
            problem.solve(solver=cp.ECOS, eps=1e-6)
            if problem.status != 'OPTIMAL':
                break
            h_new = np.array(h_var.value)
            if np.isnan(h_new).any() or np.isinf(h_new).any():
                continue  # Skip if solution is invalid
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            current_c5 = evaluate(h_new, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_new.copy()
            h = h_new
        except Exception as e:
            print(f"Optimization error: {e}")
            break
        
        # Gradient descent refinement
        for _ in range(10):
            corr = signal.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            optimal_k = np.argmax(corr)
            if optimal_k < n_points:
                indices = (np.arange(n_points) + optimal_k) % n_points
            else:
                indices = np.roll(np.arange(n_points), optimal_k % n_points)
            gradient = (1 - h[indices]) - h[indices] * (1 - h)
            h -= learning_rate * gradient
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            current_c5 = evaluate(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Final check for better solution than initial
    if evaluate(best_h, dx) < best_c5:
        best_c5 = evaluate(best_h, dx)
    
    # Clip and normalize final solution
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Final validation
    assert (best_h >= 0).all() and (best_h <= 1).all()
    assert np.isclose(np.sum(best_h), n_points / 2, atol=1e-6)
    assert np.isfinite(best_c5)
    
    return best_h, best_c5, n_points
