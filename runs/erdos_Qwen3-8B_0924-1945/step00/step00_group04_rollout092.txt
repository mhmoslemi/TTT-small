    min_improvement = 1e-6
    
    start_time = time.time()
    
    while elapsed_time < budget_s:
        iterations += 1
        cross_corr = np.correlate(h, 1 - h, mode="full") * dx
        idx = np.argmax(cross_corr)
        lag_k = idx - (n_points - 1)
        
        # Compute gradient for the worst lag
        grad = np.zeros(n_points)
        for j in range(n_points):
            if j + lag_k < n_points:
                grad[j] += (1 - h[j + lag_k]) * dx
            if j - lag_k >= 0:
                grad[j] -= h[j - lag_k] * dx
        
        # Update h
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            h = np.zeros(n_points)
        else:
            h = h * (target_sum / current_sum)
        
        # Check for improvement
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5 - min_improvement:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        # Perturb every perturb_interval iterations
        if iterations % perturb_interval == 0:
            noise = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + noise, 0, 1)
            current_sum = np.sum(h)
            if current_sum == 0:
                h = np.zeros(n_points)
            else:
                h = h * (target_sum / current_sum)
        
        # Decay learning rate
        learning_rate *= decay_rate
        
        # Early stopping
        if no_improvement >= 50:
            break
        
        # Update elapsed time
        elapsed_time = time.time() - start_time
