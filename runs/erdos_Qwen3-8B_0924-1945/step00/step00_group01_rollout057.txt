import numpy as np
import random
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Predefined initial values if available (based on parent construction)
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_vals = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    else:
        n_points = 100  # Start with a manageable number of samples
        h_vals = np.random.rand(n_points)
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
    
    dx = 2.0 / n_points
    
    # Define the objective function
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Genetic Algorithm Parameters
    POP_SIZE = 50
    MUTATION_RATE = 0.05
    CROSSOVER_RATE = 0.8
    TOURNAMENT_SIZE = 5
    GEN_LIMIT = 100
    STAGNATION = 10
    
    def generate_population(pop_size):
        pop = []
        for _ in range(pop_size):
            candidate = np.random.rand(n_points)
            candidate = candidate * (n_points / 2) / np.sum(candidate)
            pop.append(candidate)
        return pop
    
    def tournament_selection(pop, fitnesses):
        indices = np.random.choice(len(pop), size=TOURNAMENT_SIZE, replace=False)
        best_idx = np.argmin(fitnesses[indices])
        return pop[indices[best_idx]]
    
    def crossover(parent1, parent2):
        if np.random.rand() < CROSSOVER_RATE:
            pt = np.random.randint(1, n_points - 1)
            child1 = np.concatenate([parent1[:pt], parent2[pt:]])
            child2 = np.concatenate([parent2[:pt], parent1[pt:]])
            return child1, child2
        return parent1, parent2
    
    def mutate(individual):
        individual = individual.copy()
        for i in range(n_points):
            if np.random.rand() < MUTATION_RATE:
                individual[i] += np.random.uniform(-0.1, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        return individual
    
    def evaluate_population(pop):
        fitnesses = []
        for h in pop:
            fitness = evaluate(h)
            fitnesses.append(fitness)
        return fitnesses
    
    best_solution = h_vals
    best_fitness = evaluate(best_solution)
    best_fitness_history = [best_fitness]
    
    pop = generate_population(POP_SIZE)
    pop_fitness = evaluate_population(pop)
    
    for gen in range(GEN_LIMIT):
        if time.time() - start_time > budget_s:
            break
        
        # Select parents
        parent1 = tournament_selection(pop, pop_fitness)
        parent2 = tournament_selection(pop, pop_fitness)
        
        # Crossover
        child1, child2 = crossover(parent1, parent2)
        
        # Mutate
        child1 = mutate(child1)
        child2 = mutate(child2)
        
        # Evaluate new offspring
        new_fitnesses = [evaluate(child1), evaluate(child2)]
        
        # Replace worst individuals
        combined_pop = pop + [child1, child2]
        combined_fitness = pop_fitness + new_fitnesses
        
        # Find the indices to replace
        replace_indices = np.argsort(combined_fitness)[-2:]
        
        # Update population
        pop = [combined_pop[i] for i in range(len(combined_pop)) if i not in replace_indices]
        pop_fitness = [combined_fitness[i] for i in range(len(combined_fitness)) if i not in replace_indices]
        
        # Update best solution
        current_best = min(pop_fitness)
        if current_best < best_fitness:
            best_fitness = current_best
            best_solution = pop[pop_fitness.index(current_best)]
            best_fitness_history.append(best_fitness)
        else:
            best_fitness_history.append(best_fitness)
        
        # Check for stagnation
        if len(best_fitness_history) > STAGNATION and np.std(best_fitness_history[-STAGNATION:]) < 1e-5:
            break
    
    # Final evaluation
    final_h = best_solution
    final_c5 = evaluate(final_h)
    
    # Final check to ensure constraints are satisfied
    assert np.all((final_h >= 0) & (final_h <= 1)), "Values out of bounds"
    assert np.isclose(np.sum(final_h), n_points / 2, atol=1e-6), "Sum constraint violated"
    
    # Return the result
    return (final_h, final_c5, n_points)
