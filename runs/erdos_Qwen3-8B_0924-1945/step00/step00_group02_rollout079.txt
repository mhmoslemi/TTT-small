```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 450
    if initial_h_values is not None and len(initial_h_values) == n_points:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = (h - 0.5) * 2.0
        h = np.clip(h, 0, 1)
        h = h * n_points / 2.0
        h = np.clip(h, 0, n_points)
        h = h / n_points
        h = np.clip(h, 0, 1)
    
    def clip_and_scale(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            h = np.full_like(h, 0.5)
        else:
            h = h * (n_points / 2.0) / total
        return h
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * 2.0 / n_points
    
    h = clip_and_scale(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    
    for t in range(1000):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        # Targeted mutation around max_k
        indices = np.arange(n_points)
        lag_indices = indices[max_k - 10 : max_k + 11]
        lag_indices = np.unique(np.clip(lag_indices, 0, n_points - 1))
        for i in lag_indices:
            contribution = corr[max_k] - corr[max_k - (i - (n_points - 1) % n_points)]
            delta = -np.sign(contribution) * (0.1 / (1 + t / 500)) * (max_val - 0.38)
            h[i] += delta
            h[i] = np.clip(h[i], 0, 1)
        
        # Global mutation
        indices = np.random.choice(n_points, size=5, replace=False)
        for i in indices:
            delta = np.random.normal(loc=0, scale=0.2 * np.exp(-t / 300))
            h[i] += delta
            h[i] = np.clip(h[i], 0, 1)
        
        h = clip_and_scale(h)
        c5 = compute_c5(h)
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        
        if t % 100 == 0:
            print(f"Iteration {t}, best C5: {best_c5}")
    
    return (best_h, best_c5, n_points)
```
