import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 450
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = 0.5 * (1 + np.random.uniform(-1, 1, n_points))
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * 2.0 / n_points
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    
    for t in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        
        # Targeted mutation
        indices = np.random.randint(max_k - 10, max_k + 11, size=10)
        indices = np.clip(indices, 0, n_points - 1)
        h_t = h.copy()
        for i in indices:
            contrib = (1 - h[i]) * h[i]
            delta = -np.sign(contrib) * 0.1 / (1 + t / 500)
            h_t[i] += delta
            h_t[i] = np.clip(h_t[i], 0, 1)
        h_t = h_t * (n_points / 2) / np.sum(h_t)
        
        # Global mutation
        indices = np.random.choice(n_points, size=5, replace=False)
        h_g = h.copy()
        for i in indices:
            h_g[i] += np.random.normal(0, 0.2 * np.exp(-t / 300))
            h_g[i] = np.clip(h_g[i], 0, 1)
        h_g = h_g * (n_points / 2) / np.sum(h_g)
        
        c5_t = compute_c5(h_t)
        c5_g = compute_c5(h_g)
        if c5_t < best_c5:
            best_h = h_t
            best_c5 = c5_t
        if c5_g < best_c5:
            best_h = h_g
            best_c5 = c5_g
        
        if t % 100 == 0 and best_c5 < 0.3807:
            break
    
    return best_h, best_c5, n_points
