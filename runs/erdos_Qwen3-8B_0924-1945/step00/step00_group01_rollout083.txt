import numpy as np
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 50
    max_iter = 200
    tol = 1e-6
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h_vals = np.array(initial_h_values, dtype=float)
    else:
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    
    best_h = h_vals.copy()
    best_c5 = float('inf')
    
    for _ in range(max_iter):
        corr = correlate(h_vals, 1 - h_vals, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_vals.copy()
        
        # Focus on the worst lag
        k_max = np.argmax(corr)
        overlap_k = corr[k_max]
        
        # Convex surrogate: minimize 0.5 * λ^2 + (overlap_k - λ)^2
        λ = cp.Variable()
        objective = 0.5 * λ**2 + (overlap_k - λ)**2
        prob = cp.Problem(cp.Minimize(objective))
        prob.solve()
        
        # Update h_vals using gradient descent
        grad = (overlap_k - λ.value) * (1 - 2 * λ.value)
        h_vals += grad * 0.1
        h_vals = np.clip(h_vals, 0, 1)
        mass = np.sum(h_vals)
        h_vals = h_vals * (n_points / 2) / mass
        
        # Early termination if no improvement
        if np.abs(current_c5 - best_c5) < tol and _ > 10:
            break
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr)
    
    return best_h, final_c5, n_points
