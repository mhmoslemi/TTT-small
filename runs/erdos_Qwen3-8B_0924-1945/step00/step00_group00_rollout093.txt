import numpy as np
import scipy.signal
import time
import random

# Pre-imported helpers
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45
    h = np.array(initial_h_values, dtype=np.float64)
    h = h[:n_points]
    h = np.clip(h, 0.0, 1.0)
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = float('inf')
    learning_rate = 0.1
    momentum = 0.9
    noise_scale = 0.05
    noise_freq = 100
    max_iter = int(budget_s / 1)
    iter_count = 0
    v = np.zeros(n_points)
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        correlate_vals = scipy.signal.correlate(h, 1 - h, mode='full') * (2.0 / n_points)
        worst_lag = np.argmax(correlate_vals)
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * (2.0 / n_points)
        v = momentum * v + learning_rate * gradients
        h = h - v
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2) / np.sum(h)
        if iter_count % noise_freq == 0:
            noise = noise_scale * np.random.normal(0, 1, n_points)
            h += noise
            h = np.clip(h, 0.0, 1.0)
            h = h * (n_points / 2) / np.sum(h)
        current_c5 = np.max(correlate_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if current_c5 < 0.3807:
            break
    return (best_h, best_c5, n_points)
