import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 256  # Optimal number of points for performance and accuracy
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h_vals = np.array(initial_h_values, dtype=np.float64)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    else:
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    
    best_h_vals = h_vals.copy()
    best_c5 = compute_c5(h_vals, dx)
    
    for _ in range(100):
        start_time = time.time()
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5_values = corr * dx
        max_lag = np.argmax(c5_values)
        
        # Define objective for current maximum lag
        def objective(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            h[-1] = (n_points / 2 - np.sum(h[:-1])) / dx
            corr = np.correlate(h, 1 - h, mode='full')
            return corr[max_lag] * dx
        
        # Constraints
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            result = opt.minimize(objective, h_vals, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            h_vals = result.x
            h_vals = np.clip(h_vals, 0, 1)
            h_vals[-1] = (n_points / 2 - np.sum(h_vals[:-1])) / dx
            current_c5 = compute_c5(h_vals, dx)
            if current_c5 < best_c5:
                best_h_vals = h_vals.copy()
                best_c5 = current_c5
        except:
            pass
        
        if time.time() - start_time > budget_s * 0.8:
            break
    
    # Final check
    h_vals = best_h_vals.copy()
    h_vals = np.clip(h_vals, 0, 1)
    h_vals[-1] = (n_points / 2 - np.sum(h_vals[:-1])) / dx
    final_c5 = compute_c5(h_vals, dx)
    
    return (h_vals, final_c5, n_points)

def compute_c5(h_vals, dx):
    corr = np.correlate(h_vals, 1 - h_vals, mode='full')
    return np.max(corr) * dx
