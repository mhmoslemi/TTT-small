def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    elapsed_time = 0
    alpha = 0.1
    decay_rate = 0.995
    max_iter = 1000
    improvement_threshold = 1e-4
    early_stopping = 30

    def evaluate_erdos_solution(h_vec):
        cross_corr = np.correlate(h_vec, 1 - h_vec, mode="full") * dx
        return cross_corr.max()

    def compute_gradient(h_vec, max_index):
        eps = 1e-8
        perturb = np.zeros_like(h_vec)
        perturb[max_index] = eps
        h_forward = np.clip(h_vec + perturb, 0, 1)
        h_backward = np.clip(h_vec - perturb, 0, 1)
        c5_forward = evaluate_erdos_solution(h_forward)
        c5_backward = evaluate_erdos_solution(h_backward)
        grad = (c5_forward - c5_backward) / (2 * eps)
        return grad

    start_time = time.time()
    for _ in range(max_iter):
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if elapsed_time >= budget_s:
            break
        cross_corr = np.correlate(h, 1 - h, mode="full") * dx
        max_index = np.argmax(cross_corr)
        grad = compute_gradient(h, max_index)
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
        elapsed_time = time.time() - start_time
        if np.abs(c5 - best_c5) < improvement_threshold:
            early_stopping -= 1
            if early_stopping <= 0:
                break
        else:
            early_stopping = 30
        alpha *= decay_rate

    return best_h, best_c5, n_points
