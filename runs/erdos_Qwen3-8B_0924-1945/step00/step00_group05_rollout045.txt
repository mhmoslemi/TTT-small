```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    h = np.array(initial_h_values, dtype=np.float64)
    
    if 'initial_h_values' in globals():
        h = np.clip(initial_h_values, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (n_points / 2) / sum_h
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    def objective(h):
        return compute_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h):
        return np.all((h >= 0) & (h <= 1))
    
    def update_h(h, learning_rate, momentum, perturb):
        corr = np.correlate(h, 1 - h, mode="full")
        c5_values = corr * dx
        max_c5_index = np.argmax(c5_values)
        max_c5_value = c5_values[max_c5_index]
        
        grad = np.zeros_like(h)
        for i in range(n_points):
            for k in range(n_points):
                j = (i + k) % n_points
                grad[i] += (1 - h[j]) * dx
                j = (i - k) % n_points
                grad[i] -= h[j] * dx
        
        grad *= (max_c5_value / (np.max(np.abs(grad)) + 1e-10))
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (n_points / 2) / sum_h
        if perturb and np.random.rand() < 0.05:
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                h = h * (n_points / 2) / sum_h
        
        return h
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    time_start = time.time()
    
    for iter in range(500):
        if time.time() - time_start > budget_s:
            break
        learning_rate = 0.01 * (0.999 ** iter)
        momentum = 0.9
        perturb = True
        h = update_h(h, learning_rate, momentum, perturb)
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    if compute_c5(best_h) < best_c5:
        best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_points
```
