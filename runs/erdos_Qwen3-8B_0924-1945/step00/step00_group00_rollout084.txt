import numpy as np
import scipy.optimize as opt
import time
from numba import njit
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 45
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h
    best_c5 = np.inf

    def evaluate_c5(h_values):
        h = h_values.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def update_h(h, velocity, inertia, cognitive, social):
        h_new = h + velocity
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new

    def pso():
        particles = [np.random.rand(n_points) for _ in range(20)]
        velocities = [np.random.rand(n_points) * 0.1 for _ in range(20)]
        personal_bests = particles.copy()
        global_best = np.min([evaluate_c5(p) for p in particles])
        best_idx = np.argmin([evaluate_c5(p) for p in particles])
        best_h = particles[best_idx]
        start_time = time.time()

        for it in range(100):
            if time.time() - start_time > budget_s:
                break
            for i in range(20):
                h = particles[i]
                vel = velocities[i]
                p_best = personal_bests[i]
                c1, c2 = 1.5, 1.5
                r1, r2 = np.random.rand(), np.random.rand()
                vel = inertia * vel + c1 * r1 * (p_best - h) + c2 * r2 * (best_h - h)
                h_new = update_h(h, vel, 0.7, c1, c2)
                particles[i] = h_new
                p_cost = evaluate_c5(particles[i])
                if p_cost < evaluate_c5(p_best):
                    personal_bests[i] = particles[i]
            current_costs = [evaluate_c5(p) for p in particles]
            best_idx = np.argmin(current_costs)
            if current_costs[best_idx] < global_best:
                global_best = current_costs[best_idx]
                best_h = particles[best_idx]
            if evaluate_c5(best_h) < best_c5:
                best_c5 = evaluate_c5(best_h)
                best_h = best_h.copy()

        return best_h, best_c5

    best_h, best_c5 = pso()
    return best_h, best_c5, n_points
