import numpy as np
import random
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 100
    max_generations = 100
    population_size = 50
    tournament_size = 5
    crossover_prob = 0.8
    mutation_amp = 0.01
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if initial_h_values is not None:
        h_vals = np.array(initial_h_values, dtype=float)
        sum_h = np.sum(h_vals)
        if np.abs(sum_h - target_sum) > 1e-6:
            h_vals *= target_sum / sum_h
    else:
        h_vals = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h_vals)
        h_vals *= target_sum / sum_h
    
    def evaluate(h):
        h_norm = h.copy()
        sum_h = np.sum(h_norm)
        if np.abs(sum_h - target_sum) > 1e-6:
            h_norm *= target_sum / sum_h
        diff = 1 - h_norm
        corr = correlate(h_norm, diff, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    population = [h_vals.copy()]
    for _ in range(population_size - 1):
        h = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h)
        h *= target_sum / sum_h
        population.append(h)
    
    best_h = h_vals
    best_c5 = evaluate(h_vals)
    start_time = time.time()
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        
        fitness = np.array([evaluate(h) for h in population])
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_best_c5 = fitness[best_idx]
        
        if current_best_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_best_c5
        
        # Selection
        parents = []
        for _ in range(population_size // 2):
            tournament = np.random.choice(range(population_size), tournament_size, replace=False)
            best_tournament = tournament[np.argmin(fitness[tournament])]
            parents.append(population[best_tournament].copy())
        
        # Crossover
        offspring = []
        for i in range(0, len(parents), 2):
            if np.random.rand() < crossover_prob:
                parent1 = parents[i]
                parent2 = parents[i+1]
                split = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:split], parent2[split:]])
                child2 = np.concatenate([parent2[:split], parent1[split:]])
                offspring.append(child1)
                offspring.append(child2)
            else:
                offspring.append(parents[i])
                offspring.append(parents[i+1])
        
        # Mutation
        for h in offspring:
            for i in range(n_points):
                if np.random.rand() < mutation_amp:
                    h[i] += np.random.uniform(-0.05, 0.05)
                    h[i] = np.clip(h[i], 0, 1)
            sum_h = np.sum(h)
            h *= target_sum / sum_h
        
        # Replacement
        for h in offspring:
            if np.random.rand() < 0.3:
                population.append(h)
        population = np.array(population)
        population = np.sort(population, axis=0)
        population = population[:population_size]
    
    h_vals = best_h
    sum_h = np.sum(h_vals)
    h_vals *= target_sum / sum_h
    diff = 1 - h_vals
    corr = correlate(h_vals, diff, mode='full')
    c5_bound = np.max(corr) * dx
    return (h_vals, c5_bound, n_points)
