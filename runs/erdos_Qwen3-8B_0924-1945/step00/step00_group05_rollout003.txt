import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Choose n_points based on budget
    n_points = 500  # 500 points offers good tradeoff between accuracy and speed
    
    dx = 2.0 / n_points
    
    # Get initial_h_values or construct a starting vector if not available
    if 'initial_h_values' in globals():
        h = np.clip(initial_h_values, 0, 1)
    else:
        h = np.zeros(n_points)
        # Start with a piecewise constant function that has equal areas
        # Constructing a starting pattern that is balanced
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        h = np.clip(h, 0, 1)
    
    # Ensure the sum constraint is met
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h if sum_h != 0 else h * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = np.inf
    
    max_iterations = 500
    learning_rate = 0.01
    momentum = 0.9
    perturbation_prob = 0.05
    decay_rate = 0.999
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Compute overlap integrals
        overlap = np.correlate(h, 1.0 - h, mode="full") * dx
        c5 = np.max(overlap)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # If we've reached the target, break
        if best_c5 <= 0.3807:
            break
        
        # Find the lag with maximum overlap
        max_idx = np.argmax(overlap)
        k = max_idx - (n_points - 1)  # Adjust for full mode
        
        # Compute gradient for the maximum overlap lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] += (1.0 - h[j])  # contribution from h[i] in 1 - h[j]
            j = i - k
            if 0 <= j < n_points:
                grad[i] -= h[j]  # contribution from h[j] in h[i]
        
        # Apply gradient descent with momentum
        grad = grad / np.linalg.norm(grad + 1e-12)  # Normalize to avoid explosion
        h = h - learning_rate * grad
        
        # Apply clipping to ensure h is in [0,1]
        h = np.clip(h, 0, 1)
        
        # Ensure the sum constraint is met
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (n_points / 2) / sum_h
        
        # Add random perturbation with small probability
        if np.random.rand() < perturbation_prob:
            noise = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + noise, 0, 1)
        
        # Rescale if needed
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (n_points / 2) / sum_h
        
        # Decay learning rate
        learning_rate *= decay_rate
    
    # Final evaluation of best solution
    final_overlap = np.correlate(best_h, 1.0 - best_h, mode="full") * dx
    final_c5 = np.max(final_overlap)
    
    return (best_h, final_c5, n_points)
