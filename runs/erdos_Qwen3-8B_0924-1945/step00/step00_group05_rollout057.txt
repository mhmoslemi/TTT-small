import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    initial_h_values = np.array([0.0]*n_points)
    
    # Default starting point if no parent construction provided
    if 'initial_h_values' not in globals():
        initial_h_values = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                initial_h_values[i] = 1.0
            else:
                initial_h_values[i] = 0.0
        initial_h_values = np.roll(initial_h_values, n_points // 4)
    
    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        h = h * (n_points / 2) / sum_h
    
    dx = 2.0 / n_points
    best_c5 = np.inf
    best_h = h.copy()
    best_n_points = n_points
    start_time = time.time()
    
    for _ in range(500):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Compute overlap integrals
        conv = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(conv)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the lag with maximum overlap
        k = np.argmax(conv)
        
        # Compute gradient for this lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] += (1 - h[j])  # contribution from h[i]
            grad[j] -= h[i]        # contribution from h[j] in the shifted term
        
        # Update h
        learning_rate = 0.01 * (0.999 ** _)
        h = h - learning_rate * grad
        
        # Clip and rescale
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (n_points / 2) / sum_h
        
        # Random perturbation with 5% probability
        if np.random.rand() < 0.05:
            noise = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + noise, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                h = h * (n_points / 2) / sum_h
    
    # Final evaluation
    conv = np.correlate(best_h, 1 - best_h, mode="full") * dx
    final_c5 = np.max(conv)
    
    return (best_h, final_c5, n_points)
