import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    learning_rate = 0.1
    decay_rate = 0.999
    noise_scale = 0.05
    noise_prob = 0.05
    max_iterations = 1000
    early_stop_patience = 100
    improvement_threshold = 1e-5
    
    for iter in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Compute correlate and find max_c5
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_c5 = np.max(correlate_vals)
        
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
        
        # Find the lag with maximum overlap
        max_index = np.argmax(correlate_vals)
        lag = max_index - (n_points - 1)
        
        # Compute gradient for this lag
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = i - lag
            if 0 <= j < n_points:
                gradients[i] = (1 - h[j]) * (2.0 / n_points)
        
        # Update h using gradient descent
        h = h - learning_rate * gradients
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Add random noise with probability noise_prob
        if np.random.rand() < noise_prob:
            noise = np.random.normal(0, noise_scale, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        learning_rate *= decay_rate
        
        # Early stopping
        if iter > 100 and np.abs(best_c5 - np.max(np.correlate(best_h, 1 - best_h, mode="full") * (2.0 / n_points))) < improvement_threshold:
            break
    
    return (best_h, best_c5, n_points)
