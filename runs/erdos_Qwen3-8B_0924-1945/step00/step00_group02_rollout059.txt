import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200  # choose a reasonable number of points
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5] * n_points)  # fallback if no initial provided

    # This function takes a h_values array and returns the max overlap integral
    def compute_c5(h):
        if h is None:
            return float('inf')
        h = np.clip(h, 0, 1)
        if np.sum(h) != n_points / 2:
            h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    # Initial step: adjust the initial h_values to satisfy the integral constraint
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.clip(h, 0, 1)
        if np.sum(h) != n_points / 2:
            h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
    else:
        h = np.array([0.5] * n_points, dtype=np.float64)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    # Perturbation parameters
    num_iterations = 300  # total iterations
    perturbations_per_iter = 5  # number of indices to perturb per iteration
    noise_scale = 0.1  # initial noise scale for perturbations

    for _ in range(num_iterations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Randomly select indices to perturb
        indices = np.random.choice(n_points, perturbations_per_iter, replace=False)
        h_perturbed = h.copy()

        # Add noise and clip to [0, 1]
        h_perturbed[indices] += np.random.normal(0, noise_scale, size=perturbations_per_iter)
        h_perturbed = np.clip(h_perturbed, 0, 1)

        # Rescale to meet the integral constraint
        if np.sum(h_perturbed) != n_points / 2:
            h_perturbed = h_perturbed / np.sum(h_perturbed) * (n_points / 2)
        h_perturbed = np.clip(h_perturbed, 0, 1)

        # Compute C5 and update best solution
        current_c5 = compute_c5(h_perturbed)
        if current_c5 < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = current_c5

        # Occasionally adjust noise scale for exploration
        if np.random.rand() < 0.1:
            noise_scale *= np.random.uniform(0.8, 1.2)

    elapsed = time.time() - start_time
    # Final check to ensure best_h meets constraints
    best_h = np.clip(best_h, 0, 1)
    if np.sum(best_h) != n_points / 2:
        best_h = best_h / np.sum(best_h) * (n_points / 2)
    best_h = np.clip(best_h, 0, 1)
    final_c5 = compute_c5(best_h)

    return (best_h, final_c5, n_points)
