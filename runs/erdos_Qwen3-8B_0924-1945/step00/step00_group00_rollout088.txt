import numpy as np
import scipy.optimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    max_iterations = 1000
    learning_rate = 0.1
    noise_scale = 0.05
    p_noise = 0.05
    decay_rate = 0.995
    early_stop = 100
    early_stop_threshold = 1e-5
    
    initial_h = initial_h_values if 'initial_h_values' in kwargs else np.random.rand(n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    for iteration in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        current_c5 = np.max(correlate_vals)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if abs(current_c5 - best_c5) < early_stop_threshold:
            if iteration > early_stop:
                break
        
        max_lag = np.argmax(correlate_vals)
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = i - max_lag
            if 0 <= j < n_points:
                gradients[i] = (1 - h[j]) * dx
        
        learning_rate *= decay_rate
        h = h - learning_rate * gradients
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        if random.random() < p_noise:
            noise = np.random.normal(0, noise_scale, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return (best_h, best_c5, n_points)
