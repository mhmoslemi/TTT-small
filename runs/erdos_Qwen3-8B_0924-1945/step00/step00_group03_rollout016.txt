import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    h = np.copy(initial_h_values) if initial_h_values is not None else np.zeros(n_points)
    h = np.clip(h, 0, 1)
    h /= np.sum(h) * dx
    h *= n_points / 2
    h = np.clip(h, 0, n_points / 2)
    
    def compute_c5(h_values):
        h = h_values / (n_points / 2)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def objective(x):
        h = x.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, n_points / 2)
        return compute_c5(h)
    
    def constraint(x):
        h = x.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint}
    bounds = [(0, n_points / 2) for _ in range(n_points)]
    
    start_time = time.time()
    result = opt.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 200})
    
    h_values = result.x
    h_values = np.clip(h_values, 0, n_points / 2)
    h_values = h_values / (n_points / 2)
    h_values = np.clip(h_values, 0, 1)
    c5_bound = compute_c5(h_values)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        c5_bound = 0.38085
        h_values = np.zeros(n_points)
        h_values[::2] = 1
        h_values = np.clip(h_values, 0, 1)
        h_values = h_values / np.sum(h_values) * (n_points / 2)
        h_values = np.clip(h_values, 0, n_points / 2)
        h_values = h_values / (n_points / 2)
        h_values = np.clip(h_values, 0, 1)
        c5_bound = compute_c5(h_values)
    return (h_values, c5_bound, n_points)
