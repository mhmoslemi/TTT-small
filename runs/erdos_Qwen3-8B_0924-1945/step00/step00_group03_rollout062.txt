```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

# Pre-imported helpers
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(70) * 2 - 1  # Placeholder if no parent construction is available

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)  # Ensure sum is exactly n_points / 2
    
    def objective(x):
        h_test = x.copy()
        h_test = np.clip(h_test, 0, 1)
        overlap = np.correlate(h_test, 1 - h_test, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1)] * n_points
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    result = optimize.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-5,
        options={'maxiter': 100, 'ftol': 1e-9}
    )

    if result.success:
        h_opt = result.x
    else:
        h_opt = h.copy()
        perturbation = np.random.rand(n_points) * 0.1 - 0.05
        h_opt += perturbation
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        # Try SLSQP again with perturbed solution
        result = optimize.minimize(
            objective,
            h_opt,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            tol=1e-5,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        if result.success:
            h_opt = result.x
        else:
            h_opt = h.copy()

    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    c5_bound = evaluate_erdos_solution(h_opt)
    return h_opt, c5_bound, n_points
```
