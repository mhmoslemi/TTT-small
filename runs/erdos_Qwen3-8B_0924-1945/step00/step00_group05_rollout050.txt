import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Assume this is provided
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        h = h * (n_points / 2) / sum_h
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    learning_rate = 0.01
    decay_rate = 0.999
    iteration = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        k_max = np.argmax(overlaps)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (n_points / 2) / sum_h
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            iteration = 0
        else:
            iteration += 1
        if iteration >= 100:
            learning_rate *= decay_rate
            iteration = 0
        if np.random.rand() < 0.01:
            noise = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + noise, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                h = h * (n_points / 2) / sum_h
        if np.isnan(best_c5):
            best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
