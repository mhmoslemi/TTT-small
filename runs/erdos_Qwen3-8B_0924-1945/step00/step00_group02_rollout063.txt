import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.clip(h, 0.0, 1.0)
        h = h * n_points / 2 / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = 0.5 * (1 + np.random.uniform(-1, 1, n_points))
        h = h * n_points / 2 / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    T = 100.0
    alpha = 0.995
    max_iterations = int(budget_s // 0.1)
    for t in range(max_iterations):
        start_time = time.time()
        indices = np.random.choice(n_points, size=5, replace=False)
        noise = np.random.normal(0, 0.05 * np.exp(-t / 200), size=n_points)
        new_h = h + noise
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = new_h * n_points / 2 / np.sum(new_h)
        corr = np.correlate(new_h, 1 - new_h, mode='full')
        c5_new = np.max(corr) * dx
        delta = c5_new - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = new_h.copy()
            best_c5 = c5_new
        T *= alpha
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    return best_h, best_c5, n_points
