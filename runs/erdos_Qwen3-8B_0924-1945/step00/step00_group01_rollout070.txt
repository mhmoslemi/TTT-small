import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 100  # Optimal number of samples for speed and accuracy
    dx = 2.0 / n_points

    # Feasibility check for initial solution
    if initial_h_values is not None:
        h_vals = initial_h_values.copy()
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    else:
        # Random initialization with constraint
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)

    # Objective function
    def evaluate(h):
        h = np.clip(h, 0, 1)
        _, c5 = evaluate_erdos_solution(h, n_points, dx)
        return c5

    # Genetic algorithm setup
    pop_size = 50
    generations = 100
    tournament_size = 5
    crossover_rate = 0.8
    mutation_rate = 0.01

    # Initialize population
    population = []
    for _ in range(pop_size):
        h = np.random.rand(n_points)
        h = (h * (n_points / 2)) / np.sum(h)
        population.append(h)

    best_h = h_vals.copy()
    best_c5 = evaluate(h_vals)
    start_time = time.time()

    for gen in range(generations):
        # Evaluate fitness
        fitness = [evaluate(h) for h in population]
        if np.min(fitness) < best_c5:
            best_c5 = np.min(fitness)
            best_h = population[np.argmin(fitness)]

        # Selection: tournament
        parents = []
        for _ in range(pop_size):
            contenders = np.random.choice(population, size=tournament_size, replace=False)
            winner = contenders[np.argmin([evaluate(h) for h in contenders])]
            parents.append(winner)

        # Crossover
        offspring = []
        for i in range(0, pop_size, 2):
            if np.random.rand() < crossover_rate:
                p1 = parents[i]
                p2 = parents[i+1]
                idx = np.random.randint(1, n_points - 1)
                c1 = np.concatenate([p1[:idx], p2[idx:]])
                c2 = np.concatenate([p2[:idx], p1[idx:]])
                offspring.append(c1)
                offspring.append(c2)
            else:
                offspring.append(parents[i])
                offspring.append(parents[i+1])

        # Mutation
        for i in range(pop_size):
            h = offspring[i].copy()
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    h[j] += np.random.uniform(-0.05, 0.05)
                    h[j] = np.clip(h[j], 0, 1)
            h = (h * (n_points / 2)) / np.sum(h)
            offspring[i] = h

        # Replacement
        population = offspring

        # Check time
        if time.time() - start_time > budget_s:
            break

    # Final evaluation
    final_c5 = evaluate(best_h)
    return (best_h, final_c5, n_points)
