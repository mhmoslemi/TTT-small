import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Define function to evaluate the overlap integral
    def compute_c5(h, dx):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode="full") * dx
        return np.max(corr)
    
    # Define function to compute gradient for a specific lag
    def compute_gradient(h, dx, lag):
        n = len(h)
        grad = np.zeros(n)
        for i in range(n):
            j = (i + lag) % n
            grad[i] += (1 - h[j])  # contribution from h[i] in (1 - h[j])
            j = (i - lag) % n
            grad[i] += -h[j]       # contribution from h[j] in h[i]
        return grad
    
    # Define function to clip and rescale h to maintain the sum constraint
    def clip_and_rescale(h):
        h_clip = np.clip(h, 0, 1)
        sum_clip = np.sum(h_clip)
        if sum_clip == 0:
            return initial_h_values.copy() * (n_points / 2) / np.sum(initial_h_values)
        return h_clip * (n_points / 2) / sum_clip
    
    # Use the initial_h_values as starting point
    initial_h_values = np.load('initial_h_values.npy')  # Preloaded from parent
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h if sum_h != 0 else initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    
    dx = 2.0 / n_points
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    start_time = time.time()
    
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        
        # Compute current overlap
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Find the lag with maximum overlap
        corr = np.correlate(h, 1 - h, mode="full") * dx
        max_idx = np.argmax(corr)
        lag = max_idx - (len(h) - 1)
        
        # Compute gradient for the lag with maximum overlap
        grad = compute_gradient(h, dx, lag)
        
        # Add a small random perturbation with 5% probability
        if np.random.rand() < 0.05:
            noise = np.random.normal(0, 0.05, size=n_points)
            grad += noise
        
        # Update h
        learning_rate = 0.01 * (0.999 ** _)
        h -= learning_rate * grad
        
        # Clip and rescale to maintain constraints
        h = clip_and_rescale(h)
        
        # Early stopping if no improvement for 50 iterations
        if _ > 50 and np.abs(current_c5 - best_c5) < 1e-6:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h, final_c5, n_points)
