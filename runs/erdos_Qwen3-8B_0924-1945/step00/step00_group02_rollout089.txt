import numpy as np
import scipy.optimize as optimize
import time
import random

initial_h_values = None  # Placeholder for parent construction if available

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.random.rand(n_points)
        h = 0.5 * (1 + h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def clamp_and_norm(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total) < 1e-8:
            return np.zeros_like(h)
        return h * (n_points / 2) / total
    
    best_h = h.copy()
    best_c5 = evaluate(best_h)
    T = 100.0
    alpha = 0.995
    start_time = time.time()
    
    for t in range(int(budget_s / 0.1)):
        if time.time() - start_time > budget_s:
            break
        indices = np.random.choice(n_points, 5, replace=False)
        noise = np.random.normal(0, 0.05 * np.exp(-t / 200), size=n_points)
        new_h = h.copy()
        new_h[indices] += noise[indices]
        new_h = clamp_and_norm(new_h)
        new_c5 = evaluate(new_h)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
        else:
            delta = new_c5 - best_c5
            if delta > 0 and np.random.rand() < np.exp(-delta / T):
                best_h = new_h.copy()
                best_c5 = new_c5
        T *= alpha
    
    return best_h, best_c5, n_points
