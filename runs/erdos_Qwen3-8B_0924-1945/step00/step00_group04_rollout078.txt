import numpy as np
import scipy.optimize as opt
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.array([0.314, 0.314, 0.314, 0.314, 0.314, 0.314, 0.314,
                                0.314, 0.314, 0.314, 0.314, 0.314, 0.314, 0.314,
                                0.314, 0.314, 0.314, 0.314, 0.314, 0.314, 0.314,
                                0.314, 0.314, 0.314, 0.314, 0.314, 0.314, 0.314,
                                0.314, 0.314, 0.314, 0.314, 0.314, 0.314, 0.314,
                                0.314, 0.314, 0.314, 0.314, 0.314, 0.314, 0.314,
                                0.314, 0.314, 0.314, 0.314, 0.314, 0.314, 0.314])
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def repair(h):
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h) if np.sum(h) > 0 else target_sum)
        return h

    def evaluate(h):
        h = repair(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    best_h = repair(initial_h_values)
    best_c5 = evaluate(best_h)
    elapsed = 0

    def genetic_search(pop_size=100, generations=50, mutation_rate=0.1):
        nonlocal best_h, best_c5, elapsed
        pop = [np.copy(best_h) + np.random.normal(0, 0.05, n_points) for _ in range(pop_size)]
        for gen in range(generations):
            if elapsed >= budget_s:
                return
            candidates = []
            for h in pop:
                h = repair(h)
                candidates.append((evaluate(h), h))
            candidates.sort()
            top_half = candidates[:int(len(candidates)/2)]
            new_pop = [h for _, h in top_half]
            for i in range(len(new_pop)):
                if np.random.rand() < mutation_rate:
                    new_pop[i] += np.random.normal(0, 0.05, n_points)
            pop = new_pop
            for _, h in top_half:
                if evaluate(h) < best_c5:
                    best_h = repair(h)
                    best_c5 = evaluate(best_h)
            elapsed += 1

    def local_refinement(maxiter=100, tol=1e-6):
        nonlocal best_h, best_c5, elapsed
        def objective(h):
            return -evaluate(h)
        result = opt.minimize(objective, best_h, method='SLSQP',
                              bounds=[(0, 1) for _ in range(n_points)],
                              constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}],
                              options={'maxiter': maxiter, 'ftol': tol})
        best_h = result.x
        best_c5 = evaluate(best_h)
        elapsed += maxiter

    try:
        genetic_search()
        local_refinement()
    except:
        pass

    return (best_h, best_c5, n_points)
