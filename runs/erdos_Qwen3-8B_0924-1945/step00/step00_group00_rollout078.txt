import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h_values
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Parameters
    learning_rate = 0.1
    noise_scale = 0.05
    p_noise = 0.05
    max_iter = 1000
    early_stop = 100
    best_c5 = np.inf
    best_h = h.copy()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Compute correlation
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_c5 = np.max(correlate_vals)
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
        
        # Find lag with max correlation
        max_lag = np.argmax(correlate_vals)
        if max_lag >= n_points:
            max_lag -= n_points
        
        # Compute gradient
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = i - max_lag
            if 0 <= j < n_points:
                gradients[i] = (1 - h[j]) * (2.0 / n_points)
        
        # Update h
        h -= learning_rate * gradients
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Add noise
        if np.random.rand() < p_noise:
            noise = np.random.normal(0, noise_scale, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early stopping
        if _ > 50 and np.abs(max_c5 - best_c5) < 1e-5:
            break
    
    # Final check
    h = best_h
    correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
    final_c5 = np.max(correlate_vals)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    return h, final_c5, n_points
