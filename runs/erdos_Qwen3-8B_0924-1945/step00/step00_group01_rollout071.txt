import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 50
    dx = 2.0 / n_points
    if hasattr(initial_h_values, '__iter__'):
        h_vals = np.array(initial_h_values, dtype=np.float64)
    else:
        h_vals = np.random.rand(n_points)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
    h_vals = np.clip(h_vals, 0, 1)
    best_h_vals = h_vals.copy()
    best_c5 = float('inf')
    best_n_points = n_points
    max_iter = 200
    tol = 1e-6
    start_time = time.time()
    
    for _ in range(max_iter):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        corr = corr * dx
        current_c5 = np.max(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_vals = h_vals.copy()
        
        if current_c5 < tol:
            break
        
        k_max = np.argmax(corr)
        overlap_max = corr[k_max]
        
        lam = cp.Variable()
        objective = cp.Minimize(0.5 * lam**2 + (overlap_max - lam)**2)
        prob = cp.Problem(objective, [lam >= 0])
        prob.solve()
        
        if prob.status != 'OPTIMAL':
            continue
        
        lam_val = lam.value
        grad = -lam_val * (1 - h_vals) + lam_val * h_vals
        grad = grad * dx
        h_vals += grad
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
    
    corr = np.correlate(best_h_vals, 1 - best_h_vals, mode='full')
    best_c5 = np.max(corr * dx)
    best_n_points = n_points
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        return (best_h_vals, best_c5, best_n_points)
    
    return (best_h_vals, best_c5, best_n_points)
