import numpy as np
import scipy.optimize as opt
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h_values = np.array([0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333,
                                 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333,
                                 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333,
                                 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333,
                                 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333])

    current_sum = np.sum(initial_h_values)
    target_h = initial_h_values * (target_sum / current_sum)
    target_h = np.clip(target_h, 0, 1)
    h = target_h * (target_sum / np.sum(target_h))

    def repair_h(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return h * scale

    def evaluate_fitness(h):
        h = repair_h(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def evaluate_fitness_negative(h):
        return -evaluate_fitness(h)

    def create_population(pop_size, current_h):
        pop = np.random.rand(pop_size, n_points)
        pop = pop * 0.1 + current_h
        for i in range(pop_size):
            pop[i] = repair_h(pop[i])

    elapsed = 0
    gen = 0

    while elapsed < budget_s:
        gen += 1
        pop = create_population(100, best_h)
        fitness = np.array([evaluate_fitness(h) for h in pop])
        top_indices = np.argsort(fitness)[:50]
        top_h = pop[top_indices]

        new_pop = []
        for i in range(0, 50, 2):
            if i + 1 < 50:
                parent1 = top_h[i]
                parent2 = top_h[i + 1]
                child = (parent1 + parent2) / 2
                child = repair_h(child)
                new_pop.append(child)
                child = parent1 + np.random.normal(0, 0.05, n_points)

                new_pop.append(child)
                child = parent2 + np.random.normal(0, 0.05, n_points)

            else:
                child = top_h[i] + np.random.normal(0, 0.05, n_points)
                child = repair_h(child)
                new_pop.append(child)

        pop = np.array(new_pop)
        fitness = np.array([evaluate_fitness(h) for h in pop])
        top_indices = np.argsort(fitness)
        best_h = pop[top_indices[0]]
        best_c5 = evaluate_fitness(best_h)

        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    # Local refinement with SLSQP
    def objective(h):
        return evaluate_fitness_negative(h)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    result = opt.minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
    best_h = result.x
    best_h = repair_h(best_h)
    best_c5 = evaluate_fitness(best_h)

    return (best_h, best_c5, n_points)
