import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with initial_h_values if available, else create a basic construction
    if 'initial_h_values' in globals():
        h = np.clip(initial_h_values, 0, 1)
    else:
        n_points = 100
        h = np.zeros(n_points)
        h[::2] = 1
        h = np.clip(h, 0, 1)
    
    # Ensure the sum constraint is met
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1
    h = h * scale
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Objective function to minimize the maximum overlap integral
    def objective(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1
        h = h * scale
        corr = np.correlate(h, 1 - h, mode="full")
        overlap = corr * dx
        return np.max(overlap)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint}
    
    # Initial guess
    initial_guess = h.copy()
    
    # Optimization settings
    max_iter = 100
    start_time = time.time()
    best_c5 = objective(h)
    best_h = h.copy()
    
    # Run optimization
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Perform optimization with bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds, constraints=cons)
        
        if result.success:
            h_candidate = result.x
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
            initial_guess = h_candidate
        else:
            break
    
    # Final check to ensure constraints are satisfied
    sum_h = np.sum(best_h)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1
    best_h = best_h * scale
    
    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode="full")
    overlap = corr * dx
    c5_final = np.max(overlap)
    
    return (best_h, c5_final, n_points)
