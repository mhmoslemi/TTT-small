import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    h = initial_h_values.copy() if 'initial_h_values' in globals() else np.random.rand(n_points)
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = np.inf
    
    start_time = time.time()
    
    learning_rate = 0.1
    decay_rate = 0.999
    noise_scale = 0.05
    noise_prob = 0.05
    max_iter = 1000
    early_stop_iter = 100
    early_stop_thres = 1e-5
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_c5 = np.max(correlate_vals)
        
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
        
        # Find lag with max value
        max_idx = np.argmax(correlate_vals)
        lag = max_idx - (n_points - 1)
        
        # Compute gradient for this lag
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = i - lag
            if 0 <= j < n_points:
                gradients[i] = (1 - h[j]) * (2.0 / n_points)
        
        # Update h
        h = h - learning_rate * gradients
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Add noise
        if random.random() < noise_prob:
            noise = np.random.normal(0, noise_scale, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Decay learning rate
        learning_rate *= decay_rate
        
        # Early stopping
        if iter > 0 and abs(best_c5 - np.max(np.correlate(h, 1 - h, mode="full") * (2.0 / n_points))) < early_stop_thres:
            break
    
    # Final check
    final_correlate_vals = np.correlate(best_h, 1 - best_h, mode="full") * (2.0 / n_points)
    final_c5 = np.max(final_correlate_vals)
    
    return (best_h, final_c5, n_points)
