import numpy as np
import scipy.optimize as opt
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    # Available initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 250
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    h_values = h.copy()
    n_points = len(h_values)
    best_c5 = np.inf
    best_h = h_values.copy()
    np.random.seed(seed)
    
    # Parameters
    T_initial = 100.0
    T_final = 0.1
    alpha = 0.995
    steps = int(budget_s / 0.1)
    step = 0
    T = T_initial
    max_lag = 2 * n_points - 1
    
    # Function to compute C5
    def compute_c5(h):
        val = correlate(h, 1 - h, mode="full")
        c5 = np.max(val) * 2.0 / n_points
        return c5
    
    # Optimization loop
    start_time = time.time()
    while step < steps and time.time() - start_time < budget_s:
        if np.random.rand() < 0.1:
            # Sparsification: perturb 10 indices
            indices = np.random.choice(n_points, 10, replace=False)
        else:
            # Standard perturbation: 5 indices
            indices = np.random.choice(n_points, 5, replace=False)
        
        # Generate Gaussian noise
        noise = np.random.normal(0, 0.05 * np.exp(-step / 200), n_points)
        new_h = h.copy() + noise
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Compute new c5
        new_c5 = compute_c5(new_h)
        
        # Acceptance
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
        else:
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                best_c5 = new_c5
                best_h = new_h.copy()
        
        # Update temperature
        T *= alpha
        step += 1
    
    h_values = best_h
    return h_values, best_c5, n_points
