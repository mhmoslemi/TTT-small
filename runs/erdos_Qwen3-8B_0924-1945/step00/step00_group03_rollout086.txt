import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    initial_h_values = np.random.rand(n_points)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values / np.sum(initial_h_values) * (n_points / 2)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    best_h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    best_time = 0
    n_iter = 0
    start_time = time.time()
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if np.abs(h_sum - n_points / 2) > 1e-6:
            h = h * (n_points / 2) / h_sum
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def optimize_convex_relaxation(h):
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        constraints = [t >= 0]
        for k in range(n_points):
            idx = (np.arange(n_points) + k) % n_points
            temp = h_var * (1 - h_var[idx])
            temp = temp * dx
            constraints.append(t >= cp.sum(temp))
        obj = t + 1e-3 * cp.sum_squares(h_var - 0.5)
        prob = cp.Problem(cp.Minimize(obj), constraints)
        try:
            prob.solve(solver='ECOS', eps=1e-6, max_iter=100)
            if prob.status == 'optimal':
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h_new = np.clip(h_new, 0, 1)
                return h_new
            else:
                return h
        except:
            return h
    
    def gradient_refinement(h, max_iter=100, learning_rate=1e-2):
        h = h.copy()
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        h = np.clip(h, 0, 1)
        for _ in range(max_iter):
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h[:] = h.copy()
                best_c5 = c5
            grad = np.zeros_like(h)
            for k in range(n_points):
                idx = (np.arange(n_points) + k) % n_points
                corr = h * (1 - h[idx])
                grad += np.gradient(corr)
            grad = grad * dx
            h = h - learning_rate * grad
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            h = np.clip(h, 0, 1)
        return h
    
    while time.time() - start_time < budget_s and n_iter < 50:
        h = optimize_convex_relaxation(best_h)
        h = gradient_refinement(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h[:] = h.copy()
            best_c5 = c5
        n_iter += 1
        best_time = time.time() - start_time
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n_points)
