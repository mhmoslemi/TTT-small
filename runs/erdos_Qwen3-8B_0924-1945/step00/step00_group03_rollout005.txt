```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in globals() else np.random.rand(70)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = n_points / 2
    h_values = initial_h_values.copy()
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def objective(x):
        x = np.clip(x, 0, 1)
        if np.abs(np.sum(x) - sum_h) > 1e-7:
            return np.inf
        return compute_c5(x)
    
    def grad_objective(x):
        x = np.clip(x, 0, 1)
        if np.abs(np.sum(x) - sum_h) > 1e-7:
            return np.inf * np.ones_like(x)
        return np.gradient(compute_c5(x), x)
    
    def update_h(h):
        h = h.copy()
        h = np.clip(h, 0, 1)
        if np.abs(np.sum(h) - sum_h) > 1e-7:
            h += (sum_h - np.sum(h)) / len(h)
        return h
    
    best_h = h_values.copy()
    best_c5 = compute_c5(best_h)
    time_used = 0
    
    for _ in range(500):
        if time_used >= budget_s:
            break
        start = time.time()
        result = opt.minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            jac=grad_objective,
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 1, 'disp': False}
        )
        time_used += time.time() - start
        if not result.success:
            continue
        current_h = update_h(result.x)
        current_c5 = compute_c5(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
    
    if np.abs(np.sum(best_h) - sum_h) > 1e-7:
        best_h += (sum_h - np.sum(best_h)) / len(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_points
```
