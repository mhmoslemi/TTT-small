import numpy as np
import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Load or create initial h_values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Create a simple initial guess: alternating 0s and 1s
        h = np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n_points)])
    
    # Normalize initial h_values
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    scale_factor = target_sum / current_sum
    h = h * scale_factor
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    
    start_time = time.time()
    
    # Genetic algorithm parameters
    pop_size = 100
    generations = 50
    mutation_rate = 0.1
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Generate population
        population = [best_h.copy() for _ in range(pop_size)]
        for i in range(pop_size):
            if np.random.rand() < mutation_rate:
                idx = np.random.randint(n_points)
                population[i][idx] += np.random.uniform(-0.05, 0.05)
        
        # Evaluate fitness
        fitness = []
        for h_candidate in population:
            h_candidate = np.clip(h_candidate, 0, 1)
            scaled_h = h_candidate * (target_sum / np.sum(h_candidate))
            scaled_h = np.clip(scaled_h, 0, 1)
            corr = np.correlate(scaled_h, 1 - scaled_h, mode="full")
            c5 = np.max(corr) * dx
            fitness.append((c5, h_candidate))
        
        # Select top 50 candidates
        fitness.sort()
        top_candidates = [f[1] for f in fitness[:50]]
        
        # Crossover
        new_population = []
        for i in range(0, len(top_candidates), 2):
            if i + 1 < len(top_candidates):
                parent1 = top_candidates[i]
                parent2 = top_candidates[i + 1]
                child = np.zeros_like(parent1)
                for j in range(n_points):
                    if np.random.rand() < 0.5:
                        child[j] = parent1[j]
                    else:
                        child[j] = parent2[j]
                new_population.append(child)
            else:
                new_population.append(top_candidates[i])
        
        # Mutate
        for i in range(len(new_population)):
            if np.random.rand() < mutation_rate:
                idx = np.random.randint(n_points)
                new_population[i][idx] += np.random.uniform(-0.05, 0.05)
        
        # Repair and evaluate
        for h_candidate in new_population:
            h_candidate = np.clip(h_candidate, 0, 1)
            scaled_h = h_candidate * (target_sum / np.sum(h_candidate))
            scaled_h = np.clip(scaled_h, 0, 1)
            corr = np.correlate(scaled_h, 1 - scaled_h, mode="full")
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = scaled_h.copy()
                best_c5 = c5
        
        # Local refinement with SLSQP
        if time.time() - start_time < budget_s:
            def objective(x):
                x = np.clip(x, 0, 1)
                scaled_x = x * (target_sum / np.sum(x))
                scaled_x = np.clip(scaled_x, 0, 1)
                corr = np.correlate(scaled_x, 1 - scaled_x, mode="full")
                return -np.max(corr) * dx
            
            def constraint(x):
                return np.sum(x) - target_sum
            
            bounds = [(0, 1) for _ in range(n_points)]
            res = opt.minimize(objective, best_h, method='SLSQP',
                               bounds=bounds, constraints={'type': 'eq', 'fun': constraint})
            if res.success:
                best_h = res.x
                best_c5 = -res.fun
        
    # Final evaluation
    h_final = np.clip(best_h, 0, 1)
    scaled_h = h_final * (target_sum / np.sum(h_final))
    scaled_h = np.clip(scaled_h, 0, 1)
    corr = np.correlate(scaled_h, 1 - scaled_h, mode="full")
    best_c5 = np.max(corr) * dx
    
    # Ensure constraints are met
    assert np.all(0 <= best_h <= 1)
    assert np.isclose(np.sum(best_h), target_sum, atol=1e-6)
    assert np.isfinite(best_c5)
    
    return (best_h, best_c5, n_points)
