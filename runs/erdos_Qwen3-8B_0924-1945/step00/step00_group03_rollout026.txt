import numpy as np
import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    initial_h = initial_h_values
    dx = 2.0 / n_points
    sum_h = n_points / 2

    # Convert initial_h to a pattern-based approach
    if n_points % 2 == 0:
        block_size = 2
    else:
        block_size = 1

    def block_pattern(n_blocks):
        pattern = np.zeros(n_blocks, dtype=int)
        pattern[::2] = 1
        pattern[1::2] = 0
        pattern = pattern.astype(float)
        if np.sum(pattern) != n_blocks / 2:
            pattern[-1] ^= 1
        return pattern

    def evaluate_c5(h_val):
        h = h_val.copy()
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def block_to_h(block_pattern, n_points):
        h = np.zeros(n_points)
        block_len = n_points // len(block_pattern)
        for i, val in enumerate(block_pattern):
            start = i * block_len
            end = start + block_len
            h[start:end] = val
        return h

    def perturb_pattern(pattern, perturb_strength=0.1):
        new_pattern = pattern.copy()
        for i in range(len(new_pattern)):
            new_pattern[i] += np.random.uniform(-perturb_strength, perturb_strength)
            new_pattern[i] = np.clip(new_pattern[i], 0, 1)
        if np.sum(new_pattern) < len(new_pattern) / 2:
            new_pattern[-1] = 1
        elif np.sum(new_pattern) > len(new_pattern) / 2:
            new_pattern[-1] = 0
        return new_pattern

    def refine_pattern(pattern, budget_s):
        start_time = time.time()
        best_pattern = pattern.copy()
        best_c5 = evaluate_c5(block_to_h(best_pattern, n_points))
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            for i in range(len(pattern) - 1):
                if np.random.random() > 0.5:
                    pattern[i], pattern[i + 1] = pattern[i + 1], pattern[i]
                    new_pattern = pattern.copy()
                    new_pattern[-1] = 1 if np.sum(new_pattern) < len(new_pattern) / 2 else 0
                    new_c5 = evaluate_c5(block_to_h(new_pattern, n_points))
                    if new_c5 < best_c5:
                        best_pattern = new_pattern
                        best_c5 = new_c5
            if np.random.random() < 0.1:
                best_pattern = perturb_pattern(best_pattern)
                best_pattern[-1] = 1 if np.sum(best_pattern) < len(best_pattern) / 2 else 0
                best_c5 = evaluate_c5(block_to_h(best_pattern, n_points))
        return best_pattern, best_c5

    def optimize_h_with_lbd(h_initial):
        h = h_initial.copy()
        h = np.clip(h, 0, 1)
        def obj_func(x):
            h[x] = np.clip(x, 0, 1)
            return evaluate_c5(h)
        result = opt.minimize(obj_func, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points)
        return result.x

    pattern = block_pattern(n_points // 2)
    pattern, best_c5 = refine_pattern(pattern, budget_s)
    h_solution = block_to_h(pattern, n_points)
    h_optimized = optimize_h_with_lbd(h_solution)
    h_optimized = np.clip(h_optimized, 0, 1)
    h_optimized = h_optimized / np.sum(h_optimized) * sum_h
    h_optimized = np.clip(h_optimized, 0, 1)
    final_c5 = evaluate_c5(h_optimized)
    return h_optimized, final_c5, n_points
