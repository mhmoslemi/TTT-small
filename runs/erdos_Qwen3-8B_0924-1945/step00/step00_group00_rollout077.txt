import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Default starting point
    n_points = 100
    if 'initial_h_values' in globals():
        h_values = np.array(initial_h_values)
        n_points = len(h_values)
        # Scale to ensure correct integral
        h_values = h_values * (n_points / 2) / np.sum(h_values)
    else:
        h_values = np.random.rand(n_points)
        h_values = h_values * (n_points / 2) / np.sum(h_values)
    
    best_c5 = float('inf')
    best_h = h_values.copy()
    start_time = time.time()
    
    # Optimization parameters
    max_iter = 1000
    learning_rate = 0.1
    decay_rate = 0.999
    noise_prob = 0.05
    noise_scale = 0.05
    early_stop_patience = 100
    early_stop_threshold = 1e-5
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Compute the correlate
        correlate_vals = np.correlate(h_values, 1 - h_values, mode="full") * (2 / n_points)
        max_c5 = np.max(correlate_vals)
        
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h_values.copy()
        
        # Find the lag with the maximum value
        max_idx = np.argmax(correlate_vals)
        lag = max_idx - (n_points - 1)
        
        # Compute gradient for this lag
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = i - lag
            if 0 <= j < n_points:
                gradients[i] = (1 - h_values[j]) * (2 / n_points)
        
        # Update h_values
        h_values = h_values - learning_rate * gradients
        h_values = np.clip(h_values, 0, 1)
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        
        # Add noise with probability noise_prob
        if random.random() < noise_prob:
            noise = np.random.normal(0, noise_scale, n_points)
            h_values = h_values + noise
            h_values = np.clip(h_values, 0, 1)
            h_values = h_values * (n_points / 2) / np.sum(h_values)
        
        # Decay learning rate
        learning_rate *= decay_rate
        
        # Early stopping
        if iter > early_stop_patience and np.abs(max_c5 - best_c5) < early_stop_threshold:
            break
    
    # Final check to ensure constraints
    h_values = np.clip(h_values, 0, 1)
    h_values = h_values * (n_points / 2) / np.sum(h_values)
    
    # Final objective evaluation
    correlate_vals = np.correlate(h_values, 1 - h_values, mode="full") * (2 / n_points)
    final_c5 = np.max(correlate_vals)
    
    return (h_values, final_c5, n_points)
