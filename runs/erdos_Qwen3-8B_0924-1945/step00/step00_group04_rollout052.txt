import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    target_sum = n_points / 2
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    elapsed_time = 0
    alpha = 0.1
    decay_rate = 0.95
    perturb = 1e-8
    max_iter = 500
    early_stop = 30
    early_stop_tol = 1e-5

    for i in range(max_iter):
        start_time = time.time()
        cross_corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_index = np.argmax(cross_corr)
        # Gradient computation
        h_perturb = h.copy()
        h_perturb[max_index] += perturb
        c5_perturb = evaluate_erdos_solution(h_perturb)
        forward_diff = (c5_perturb - best_c5) / perturb
        h_perturb = h.copy()
        h_perturb[max_index] -= perturb
        c5_perturb = evaluate_erdos_solution(h_perturb)
        backward_diff = (best_c5 - c5_perturb) / perturb
        grad = (forward_diff + backward_diff) / 2
        # Optimization step
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        # Evaluation
        c5_new = evaluate_erdos_solution(h_new)
        # Update best
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        # Check early stopping
        if i > early_stop and abs(c5_new - best_c5) < early_stop_tol:
            break
        # Update alpha and elapsed time
        alpha *= decay_rate
        elapsed_time += time.time() - start_time
        if elapsed_time >= budget_s:
            break
    return best_h, best_c5, n_points
