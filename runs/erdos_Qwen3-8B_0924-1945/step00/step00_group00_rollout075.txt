import numpy as np
import scipy.signal as signal
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with a reasonable number of samples
    dx = 2.0 / n_points
    max_iterations = 1000
    learning_rate = 0.01
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Ensure initial_h_values is used if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values[:n_points], dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = (h * n_points / 2.0) / np.sum(h)
    
    h = np.clip(h, 0, 1)
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Compute the overlap integrals
        correlate = signal.correlate(h, 1 - h, mode='full')
        overlap = correlate * dx
        max_overlap = np.max(overlap)
        
        # Update best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Find the lag with the maximum overlap
        max_idx = np.argmax(overlap)
        lag = max_idx - (n_points - 1)
        
        # Compute the gradient for this lag
        gradient = np.zeros_like(h)
        for i in range(n_points):
            j = i - lag
            if 0 <= j < n_points:
                gradient[i] = (1 - h[j]) * dx
        
        # Update h with gradient descent
        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h)
    
    # Final check and re-calculation
    if best_h is None:
        best_h = h
        best_c5 = np.max(signal.correlate(h, 1 - h, mode='full') * dx)
    
    # Ensure the best_h satisfies the constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2.0) / np.sum(best_h)
    
    # Final evaluation
    final_overlap = np.max(signal.correlate(best_h, 1 - best_h, mode='full') * dx)
    
    return best_h, final_overlap, n_points
