import numpy as np
import scipy.optimize as opt
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    initial_h_values = np.load('initial_h_values.npy')  # Assume initial construction is provided
    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h if sum_h != 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    learning_rate = 0.01
    decay_rate = 0.999
    iteration = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        overlaps = signal.correlate(h, 1 - h, mode='full') * dx
        k_max = np.argmax(overlaps)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h if sum_h != 0 else np.zeros(n_points)
        h = np.clip(h, 0, 1)
        new_c5 = evaluate_erdos_solution(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            iteration = 0
        else:
            iteration += 1
        if iteration >= 100:
            learning_rate *= decay_rate
            iteration = 0
        if np.random.rand() < 0.01:
            noise = np.random.normal(0, 0.05, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            h = h * (n_points / 2) / sum_h if sum_h != 0 else np.zeros(n_points)
            h = np.clip(h, 0, 1)
    
    return best_h, best_c5, n_points
