import numpy as np
import scipy.signal
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(scipy.signal.correlate(h, 1 - h, mode='full') * (2.0 / n_points))
    
    start_time = time.time()
    learning_rate = 0.1
    momentum = 0.9
    noise_scale = 0.05
    decay_rate = 0.999
    iterations = 0
    early_stop = 0
    
    while time.time() - start_time < budget_s:
        iterations += 1
        correlate_vals = scipy.signal.correlate(h, 1 - h, mode='full') * (2.0 / n_points)
        worst_lag = np.argmax(correlate_vals)
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * (2.0 / n_points)
        
        v = momentum * v + learning_rate * gradients
        h = h - v
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        if iterations % 100 == 0:
            h += noise_scale * np.random.normal(0, 1, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        current_c5 = np.max(correlate_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            early_stop = 0
        else:
            early_stop += 1
        
        if early_stop > 100:
            break
        
        learning_rate *= decay_rate
    
    return (best_h, best_c5, n_points)
