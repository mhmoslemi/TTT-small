import numpy as np
import random
import time
import scipy.signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Preprocess initial h_values if available
    initial_h_values = globals().get('initial_h_values', None)
    if initial_h_values is not None:
        n_points = len(initial_h_values)
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Fallback: Create a simple starting vector for testing
        n_points = 100
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        h = h * (n_points / 2) / total
        h = np.clip(h, 0, 1)
    
    # Function to evaluate C5
    def evaluate_c5(h):
        dx = 2.0 / n_points
        h_pad = np.concatenate([h, np.zeros(n_points)])
        one_minus_h = 1.0 - h
        corr = scipy.signal.correlate(h_pad, one_minus_h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Function to clip and rescale h to meet the integral constraint
    def clip_and_rescale(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return h
        scale = (n_points / 2) / total
        h = h * scale
        return h
    
    # Initial evaluation
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()
    
    # Optimization loop
    iterations = 0
    while time.time() - start_time < budget_s:
        iterations += 1
        
        # Randomly perturb 5 indices
        indices = np.random.choice(n_points, 5, replace=False)
        noise = np.random.normal(0, 0.1, 5)
        h_perturbed = h.copy()
        for i, idx in enumerate(indices):
            h_perturbed[idx] += noise[i]
        
        # Clip values and rescale
        h_perturbed = clip_and_rescale(h_perturbed)
        
        # Evaluate C5 for the perturbed h
        c5 = evaluate_c5(h_perturbed)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_perturbed.copy()
        
        # Occasionally adjust noise scale for exploration
        if iterations % 100 == 0:
            noise_scale = 0.1 * (1.0 - (iterations / 300))
    
    return (best_h, best_c5, n_points)
