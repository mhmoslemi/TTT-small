import numpy as np
import time
import scipy.optimize as opt
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with the initial construction if available
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        n_points = len(h_initial)
    else:
        # Default construction: evenly spaced points with alternating 0 and 1
        n_points = 70  # Preferred length for faster optimization
        h_initial = np.zeros(n_points)
        h_initial[::2] = 1  # Alternating 0 and 1
        # Adjust to correct the integral constraint
        h_initial = np.round(h_initial).astype(float)
        if np.sum(h_initial) != n_points // 2:
            h_initial[-1] = 1 - h_initial[-1]
    
    dx = 2.0 / n_points
    h_values = h_initial.copy()
    
    # Objective function for optimization
    def objective(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if np.abs(s - n_points / 2) > 1e-6:
            return np.inf
        c5 = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(c5)
    
    # Bounds and constraints
    bounds = Bounds(0, 1)
    
    # Optimization with L-BFGS-B
    result = opt.minimize(
        objective,
        h_values,
        method='L-BFGS-B',
        bounds=bounds,
        jac=None,
        tol=1e-6,
        options={'maxiter': 500, 'ftol': 1e-8}
    )
    
    h_values = result.x
    c5_bound = np.correlate(h_values, 1 - h_values, mode='full') * dx
    c5_bound = np.max(c5_bound)
    
    # Final validation
    assert (h_values >= 0).all() and (h_values <= 1).all()
    assert np.isclose(np.sum(h_values), n_points / 2, atol=1e-6)
    assert np.isfinite(c5_bound)
    
    return (h_values, c5_bound, n_points)
