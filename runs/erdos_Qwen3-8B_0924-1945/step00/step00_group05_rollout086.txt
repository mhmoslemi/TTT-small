```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h_values = np.random.rand(n_points) * 2 - 1
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values /= np.sum(initial_h_values) * dx
    initial_h_values *= n_points / 2

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        c = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    result = opt.minimize(objective, initial_h_values, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    h_values = result.x
    c5_bound = objective(h_values)
    return (h_values, c5_bound, n_points)
```
