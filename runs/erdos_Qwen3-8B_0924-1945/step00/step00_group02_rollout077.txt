import numpy as np
import numpy.random as npr
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    initial_h_values = kwargs.get('initial_h_values', None)
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.clip(h, 0, 1)
        h = (h * n_points / 2.0) / np.sum(h)
    else:
        h = 0.5 * (1 + npr.uniform(-1, 1, n_points))
        h = np.clip(h, 0, 1)
        h = (h * n_points / 2.0) / np.sum(h)
    h = np.clip(h, 0, 1)
    h = h * n_points / 2.0 / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * 2.0 / n_points
    start_time = time.time()
    iteration = 0
    threshold = 0.8
    for _ in range(1200):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        lag = np.argmax(corr)
        overlap = h * np.roll(1 - h, lag)
        overlap_x = overlap
        max_overlap = np.max(overlap_x)
        if max_overlap == 0:
            break
        target_indices = np.argsort(overlap_x)[::-1]
        target_indices = target_indices[:int(len(target_indices) * 0.1)]
        candidates = []
        for idx in target_indices:
            h_candidate = h.copy()
            h_candidate[idx] = 1 - h_candidate[idx]
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * n_points / 2.0 / np.sum(h_candidate)
            candidates.append(h_candidate)
            h_candidate = h.copy()
            h_candidate[idx] += 0.05
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * n_points / 2.0 / np.sum(h_candidate)
            candidates.append(h_candidate)
            h_candidate = h.copy()
            h_candidate[idx] -= 0.05
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * n_points / 2.0 / np.sum(h_candidate)
            candidates.append(h_candidate)
        for candidate in candidates:
            corr_candidate = np.correlate(candidate, 1 - candidate, mode="full")
            c5_candidate = np.max(corr_candidate) * 2.0 / n_points
            if c5_candidate < best_c5:
                best_h = candidate.copy()
                best_c5 = c5_candidate
        h = best_h.copy()
        iteration += 1
    return (best_h, best_c5, n_points)
