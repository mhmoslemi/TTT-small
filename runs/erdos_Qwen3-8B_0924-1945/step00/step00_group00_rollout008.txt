import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', None)
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * n_points / 2 / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    
    start_time = time.time()
    
    for i in range(500):
        if time.time() - start_time > budget_s:
            break
        
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        worst_lag = np.argmax(correlate_vals)
        
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * dx
        
        learning_rate = 0.05 * (1 - i / 500)
        h = h - learning_rate * gradients
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        if np.random.random() < 0.05:
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        current_c5 = np.max(np.correlate(h, 1 - h, mode="full") * dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if i % 100 == 0:
            if best_c5 < 0.3807:
                break
    
    return (best_h, best_c5, n_points)
