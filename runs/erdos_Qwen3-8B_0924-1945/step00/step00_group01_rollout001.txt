import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values) if initial_h_values is not None else 47
    dx = 2.0 / n_points
    h_vals = np.array(initial_h_values, dtype=float) if initial_h_values is not None else np.random.rand(n_points)
    h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    best_h_vals = h_vals.copy()
    best_c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * dx
    start_time = time.time()
    for _ in range(100):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 100:
            break
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5_vals = corr * dx
        max_c5_idx = np.argmax(c5_vals)
        lag = max_c5_idx - (n_points - 1)
        h = h_vals.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        def objective(x):
            x = np.clip(x, 0, 1)
            x = x / np.sum(x) * (n_points / 2)
            overlap = 0
            for i in range(n_points):
                j = (i + lag) % n_points
                overlap += x[i] * (1 - x[j])
            return overlap
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        result = opt.minimize(objective, h, bounds=bounds, constraints=cons, method='SLSQP')
        if result.success:
            h_vals = result.x
        else:
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
        current_c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_vals = h_vals.copy()
        if np.allclose(current_c5, best_c5):
            break
    h_vals = best_h_vals
    h_vals = np.clip(h_vals, 0, 1)
    h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
    final_c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * dx
    return h_vals, final_c5, n_points
