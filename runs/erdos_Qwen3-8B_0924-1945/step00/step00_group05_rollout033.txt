```python
import numpy as np
import scipy.optimize as opt
import time

# Pre-imported helpers
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.array(initial_h_values[:n_points], dtype=np.float64)
    target_sum = n_points / 2
    h_sum = np.sum(h)
    h = h * (target_sum / h_sum) if h_sum != 0 else h

    def evaluate(h_values):
        h_values = np.clip(h_values, 0.0, 1.0)
        h_values_sum = np.sum(h_values)
        if h_values_sum == 0:
            return float('inf')
        h_values = h_values * (target_sum / h_values_sum)
        h_corr = np.correlate(h_values, 1 - h_values, mode='full')
        c5 = np.max(h_corr * dx)
        return c5

    def objective(h_values):
        return evaluate(h_values)

    def constraint_sum(h_values):
        return np.sum(h_values) - target_sum

    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result = opt.minimize(objective, h, bounds=bounds, constraints=cons, method='L-BFGS-B', tol=1e-6)
    elapsed = time.time() - start_time

    if elapsed >= budget_s:
        return (h, evaluate(h), n_points)

    h_best = result.x
    c5_best = evaluate(h_best)
    return (h_best, c5_best, n_points)
```
