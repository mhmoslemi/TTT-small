import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points)
    initial_h = np.clip(initial_h, 0, 1)
    norm_factor = n_points / 2.0
    initial_h = initial_h * (norm_factor / np.sum(initial_h)) if np.sum(initial_h) != 0 else np.full(n_points, 1.0 / n_points) * norm_factor

    def objective(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (norm_factor / np.sum(h_vals)) if np.sum(h_vals) != 0 else h_vals
        one_minus_h = 1.0 - h_vals
        corr = np.correlate(h_vals, one_minus_h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def gradient(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (norm_factor / np.sum(h_vals)) if np.sum(h_vals) != 0 else h_vals
        one_minus_h = 1.0 - h_vals
        corr = np.correlate(h_vals, one_minus_h, mode="full")
        max_corr = np.max(corr)
        max_idx = np.argmax(corr)
        grad = np.zeros_like(h_vals)
        for i in range(n_points):
            h_vals_shifted = np.roll(h_vals, i)
            h_vals_shifted = np.clip(h_vals_shifted, 0, 1)
            h_vals_shifted = h_vals_shifted * (norm_factor / np.sum(h_vals_shifted)) if np.sum(h_vals_shifted) != 0 else h_vals_shifted
            one_minus_h_shifted = 1.0 - h_vals_shifted
            corr_shifted = np.correlate(h_vals_shifted, one_minus_h_shifted, mode="full")
            grad[i] = (corr_shifted[max_idx] - corr_shifted[max_idx - 1]) * dx
        return grad

    def constraint(h_vals):
        return np.sum(h_vals) - norm_factor

    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, jac=gradient, tol=1e-8, maxiter=100)

    best_h = result.x
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
