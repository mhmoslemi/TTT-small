import numpy as np
import scipy.optimize as optimize
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values) if initial_h_values is not None else 70
    dx = 2.0 / n_points
    h = np.copy(initial_h_values) if initial_h_values is not None else np.zeros(n_points)
    sum_h = np.sum(h)
    if not np.isclose(sum_h, n_points / 2, atol=1e-6):
        h[0] = n_points / 2 - sum_h
        h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    t_total = int(budget_s * 0.75)
    T0 = 0.7 * (1.0 + best_c5) / 3.0
    Tmax = int(0.1 * t_total)
    beta = 0.75
    gamma = 0.65
    lambda0 = 0.15 / dx
    start_time = time.time()

    for t in range(t_total):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        if t % 100 == 0 and t < t_total // 2:
            lambda0 *= 0.8
        T = T0 * (1 - t / Tmax) ** (1 / beta)
        noise = np.random.normal(0, lambda0, size=n_points)
        perturbed = h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        delta = (n_points / 2 - current_sum) / (n_points * 0.95)
        perturbed += delta * np.random.choice([-1, 1], size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        if np.sum(perturbed) != 0:
            perturbed *= (n_points / 2) / np.sum(perturbed)
        new_c5 = evaluate_erdos_solution(perturbed, dx)
        if new_c5 < best_c5 or np.random.rand() < np.exp(-(new_c5 - best_c5) / T):
            h = perturbed.copy()
            best_h = h.copy()
            best_c5 = new_c5
        if best_c5 < 0.3807:
            break

    if not np.isfinite(best_c5):
        best_c5 = evaluate_erdos_solution(best_h, dx)
    return (best_h, best_c5, n_points)
