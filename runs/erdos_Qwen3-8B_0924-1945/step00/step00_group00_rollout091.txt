import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    start_time = time.time()
    
    learning_rate = 0.05
    noise_scale = 0.05
    decay_rate = 0.999
    max_iter = 500
    early_stop_iter = 50
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        worst_lag = np.argmax(correlate_vals)
        
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * (2.0 / n_points)
        
        h -= learning_rate * gradients
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        if np.random.random() < 0.05:
            h += noise_scale * np.random.normal(0, 1, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        current_c5 = compute_c5(h, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if i % 100 == 0 and i > 0:
            learning_rate *= decay_rate
            noise_scale *= 0.995
        
        if i > 0 and abs(best_c5 - compute_c5(best_h, n_points)) < 1e-5:
            if i > early_stop_iter:
                break
    
    return best_h, best_c5, n_points

def compute_c5(h, n_points):
    correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
    return np.max(correlate_vals)
