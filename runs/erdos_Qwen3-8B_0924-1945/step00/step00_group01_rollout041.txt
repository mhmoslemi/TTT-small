```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h_vals = np.array(initial_h_values, dtype=np.float64)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    else:
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    
    n_iter = 100
    max_lag = 2 * n_points - 1
    best_h_vals = h_vals.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    for _ in range(n_iter):
        if time.time() - start_time > budget_s:
            break
        
        corr = np.correlate(h_vals, 1 - h_vals, mode='full') * dx
        max_k = np.argmax(corr)
        if max_k >= len(corr):
            max_k = len(corr) - 1
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        
        def objective(x):
            h = x.copy()
            corr = np.correlate(h, 1 - h, mode='full') * dx
            return corr[max_k]
        
        res = opt.minimize(objective, h_vals, bounds=bounds, constraints=cons, method='L-BFGS-B')
        if res.success:
            h_vals = res.x
        else:
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
        
        curr_corr = np.correlate(h_vals, 1 - h_vals, mode='full') * dx
        curr_c5 = np.max(curr_corr)
        
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h_vals = h_vals.copy()
        
        if np.max(np.abs(curr_corr - best_c5)) < 1e-5:
            break
    
    h_vals = best_h_vals
    h_vals = np.clip(h_vals, 0, 1)
    h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    corr = np.correlate(h_vals, 1 - h_vals, mode='full') * dx
    c5_bound = np.max(corr)
    
    return (h_vals, c5_bound, n_points)
```
