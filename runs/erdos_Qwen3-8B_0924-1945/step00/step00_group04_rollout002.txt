    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values.copy()

    elapsed_time = 0

    step_size = 0.1
    decay_rate = 0.995
    perturbation = 1e-8
    improvement_threshold = 1e-4
    
    for i in range(max_iter):
        start_time = time.time()
        
        # Compute cross correlation
        cross_corr = np.correlate(h, 1 - h, mode="full") * dx
        max_index = np.argmax(cross_corr)
        
        # Gradient approximation
        h_perturb = h.copy()
        h_perturb[max_index] += perturbation
        h_perturb = np.clip(h_perturb, 0, 1)
        c5_forward = evaluate_erdos_solution(h_perturb)
        h_perturb = h.copy()
        h_perturb[max_index] -= perturbation
        h_perturb = np.clip(h_perturb, 0, 1)
        c5_backward = evaluate_erdos_solution(h_perturb)
        grad = (c5_forward - c5_backward) / (2 * perturbation)
        
        # Update h
        h = h - step_size * grad
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h)) if np.sum(h) > 0 else h
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
        
        # Check for convergence
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
        
        # Decay step size
        step_size *= decay_rate
    
    return (best_h, best_c5, n_points)
