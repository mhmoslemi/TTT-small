import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h = initial_h[:n_points]
        h = np.clip(h, 0, 1)
        h = h / (np.sum(h) * dx) * (n_points / 2)
    else:
        h = np.zeros(n_points)
        h[::2] = 1
        h = h / (np.sum(h) * dx) * (n_points / 2)
    
    def compute_c5(h):
        h_pad = np.pad(h, (0, n_points), mode='constant')
        corr = np.correlate(h_pad, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def objective(x):
        h = np.clip(x, 0, 1)
        h = h / (np.sum(h) * dx) * (n_points / 2)
        return compute_c5(h)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint}
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    result = opt.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 500})
    h_opt = result.x
    c5_opt = compute_c5(h_opt)
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        h_opt = h
        c5_opt = compute_c5(h)
    
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt / (np.sum(h_opt) * dx) * (n_points / 2)
    final_c5 = compute_c5(h_opt)
    
    return (h_opt, final_c5, n_points)
