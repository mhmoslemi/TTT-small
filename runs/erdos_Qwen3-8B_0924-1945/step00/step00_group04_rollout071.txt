import numpy as np
import scipy.optimize as optimize
import time
import copy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default construction: evenly spaced blocks
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                h[i] = 1.0
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0, 1)

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        c5 = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(c5)

    def clip_and_rescale(h):
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        return h

    best_h = copy.deepcopy(h)
    best_c5 = evaluate(best_h)

    population_size = 50
    population = [copy.deepcopy(h) for _ in range(population_size)]
    for h in population:
        h = clip_and_rescale(h)

    def objective(h):
        return -evaluate(h)

    def mutate(h, mutation_factor=0.5, perturbation_range=0.05):
        h = copy.deepcopy(h)
        indices = np.random.choice(n_points, 3, replace=False)
        for i in indices:
            h[i] += np.random.uniform(-perturbation_range, perturbation_range) * h[i]
        return clip_and_rescale(h)

    def crossover(parent1, parent2, crossover_prob=0.7):
        child = copy.deepcopy(parent1)
        if np.random.random() < crossover_prob:
            for i in range(n_points):
                if np.random.random() < 0.5:
                    child[i] = parent2[i]
        return clip_and_rescale(child)

    start_time = time.time()
    iteration = 0
    no_improvement = 0
    population = [copy.deepcopy(best_h) for _ in range(population_size)]
    for h in population:
        h = clip_and_rescale(h)

    while time.time() - start_time < budget_s:
        new_population = []
        for _ in range(population_size // 2):
            parent1, parent2 = np.random.choice(population, 2, replace=False)
            child1 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = crossover(parent1, parent2)
            child2 = mutate(child2)
            new_population.append(child1)
            new_population.append(child2)
        population = new_population
        population.sort(key=objective)
        population = population[:population_size]
        current_c5 = -objective(population[0])
        if abs(current_c5 - best_c5) < 1e-6:
            no_improvement += 1
        else:
            no_improvement = 0
            best_c5 = current_c5
            best_h = copy.deepcopy(population[0])
        if no_improvement >= 50:
            break
        iteration += 1

    return best_h, best_c5, n_points
