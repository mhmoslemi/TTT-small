import numpy as np
import random
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 45
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.sort(h)
        h = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, n_points), h)
        h = np.floor(h * n_points / 2)
        h = np.clip(h, 0, n_points)
        h = h / n_points
        h = np.random.permutation(h)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    h[0] = h[0] + (n_points / 2 - np.sum(h)) / n_points
    h = np.clip(h, 0, 1)

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(correlate_vals)
        return c5

    def objective(h):
        return evaluate(h)

    def constraint(h):
        return np.sum(h) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0, 1) for _ in range(n_points)]

    result = optimize.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
    h_opt = result.x
    c5_opt = evaluate(h_opt)
    return h_opt, c5_opt, n_points
