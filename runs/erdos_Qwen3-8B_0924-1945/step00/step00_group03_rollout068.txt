import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Placeholder, should be provided by the environment
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Validate initial solution
    assert (initial_h_values >= 0).all() and (initial_h_values <= 1).all()
    assert np.isclose(np.sum(initial_h_values), n_points / 2, atol=1e-6)
    
    def evaluate_erdos_solution(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_solution = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(best_solution, dx)
    
    # Parameters
    lambda_penalty = 1e-3
    max_refinement_iterations = 100
    learning_rate = 1e-2
    total_time_used = 0
    
    # Convex relaxation setup
    h_var = cp.Variable(n_points, bounds=(0, 1))
    t = cp.Variable()
    
    # Compute all lags
    lags = np.arange(-n_points + 1, n_points)
    
    # Define constraints
    constraints = [t >= 0]
    for k in lags:
        offset = (np.arange(n_points) + k) % n_points
        expr = cp.sum(h_var * (1 - h_var[offset])) * dx
        constraints.append(t >= expr)
    
    # Objective
    objective = t + lambda_penalty * cp.sum((h_var - 0.5) ** 2)
    
    # Solve convex relaxation
    prob = cp.Problem(cp.Minimize(objective), constraints)
    try:
        prob.solve(solver=cp.ECOS, eps=1e-6, verbose=False)
        if prob.status == 'OPTIMAL':
            h_relaxed = np.array(h_var.value)
        else:
            h_relaxed = initial_h_values.copy()
    except:
        h_relaxed = initial_h_values.copy()
    
    # Refinement using gradient descent
    h_refined = h_relaxed.copy()
    for _ in range(max_refinement_iterations):
        if total_time_used >= budget_s:
            break
        start_time = time.time()
        
        # Compute C5
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        c5 = np.max(corr) * dx
        
        # Update
        if c5 < best_c5:
            best_c5 = c5
            best_solution = h_refined.copy()
        
        # Gradient descent step
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
        total_time_used += time.time() - start_time
    
    # Final validation
    assert (best_solution >= 0).all() and (best_solution <= 1).all()
    assert np.isclose(np.sum(best_solution), n_points / 2, atol=1e-6)
    final_c5 = evaluate_erdos_solution(best_solution, dx)
    
    return best_solution, final_c5, n_points
