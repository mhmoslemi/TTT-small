import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.ones(n_points) * 0.5
    h = h * (target_sum / np.sum(h))
    def objective(x):
        h = x.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr) * dx
        return max_corr
    def constraint(x):
        h = x.reshape(n_points)
        return np.sum(h) - target_sum
    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(lambda x: objective(x), h.flatten(), method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'ftol': 1e-8, 'maxiter': 100})
    best_h = result.x
    best_h = best_h.reshape(n_points)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h))
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
