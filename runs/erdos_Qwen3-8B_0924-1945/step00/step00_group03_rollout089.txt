import numpy as np
import cvxpy as cp
import time

initial_h_values = np.array([...])  # Assuming predefined initial values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    t0 = time.time()

    for _ in range(100):
        if time.time() - t0 > budget_s - 1:
            break

        # Convex relaxation
        h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
        t = cp.Variable()
        constraints = []
        for k in range(n_points):
            lag = (np.arange(n_points) + k) % n_points
            corr = cp.sum(h_var * (1 - h_var[lag])) * dx
            constraints.append(t >= corr)
        constraints.append(cp.sum(h_var) == n_points / 2)
        obj = t + 1e-3 * cp.sum_squares(h_var - 0.5)
        prob = cp.Problem(cp.Minimize(obj), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == 'optimal':
                h = h_var.value
                current_c5 = evaluate_erdos_solution(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        except:
            pass

        # Perturbation and refinement
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h /= np.sum(h) * dx
        h *= n_points / 2
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    # Final checks
    best_h = np.clip(best_h, 0, 1)
    best_h /= np.sum(best_h) * dx
    best_h *= n_points / 2
    final_c5 = evaluate_erdos_solution(best_h, dx)

    return best_h, final_c5, n_points
