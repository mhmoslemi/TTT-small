```python
import numpy as np
import numpy.random as npr
import scipy.optimize as sco
from scipy.optimize import Bounds
from scipy.optimize import minimize
from scipy.optimize import differential_evolution
import time

n_points = 45
dx = 2.0 / n_points

def evaluate_erdos_solution(h):
    correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(correlate_vals)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    if 'initial_h_values' in kwargs and kwargs['initial_h_values'] is not None:
        initial_h = np.array(kwargs['initial_h_values'], dtype=float)
    else:
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0, 1)
    
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=20,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        disp=False,
        polish=True,
        seed=seed,
        vectorized=True,
        x0=initial_h
    )
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, best_c5, n_points
```
