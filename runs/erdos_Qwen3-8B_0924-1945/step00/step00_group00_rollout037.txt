import numpy as np
import scipy.signal
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0, 1)
        h = h * n_points / 2.0
        h = h / np.sum(h) * n_points / 2.0
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * n_points / 2.0
        h = h / np.sum(h) * n_points / 2.0

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.max(scipy.signal.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()

    learning_rate = 0.01
    decay_rate = 0.999
    min_improvement = 1e-6
    max_iterations = int(budget_s / 0.1)

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        lag = np.argmax(corr)
        if lag < n_points:
            kernel = 1 - h[lag:]
        else:
            kernel = 1 - h[:n_points - lag]
        grad = np.zeros_like(h)
        for i in range(n_points):
            if lag < n_points:
                j = i - lag
                if 0 <= j < n_points:
                    grad[i] += kernel[j] * dx
            else:
                j = i + (lag - n_points)
                if 0 <= j < n_points:
                    grad[i] += kernel[j] * dx

        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * n_points / 2.0
        h = h / np.sum(h) * n_points / 2.0
        current_c5 = np.max(scipy.signal.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5 - min_improvement:
            best_h = h.copy()
            best_c5 = current_c5
        learning_rate *= decay_rate

    final_h = best_h
    final_c5 = best_c5
    return (final_h.tolist(), final_c5, n_points)
