import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h_vals = np.zeros(n_points)
    if 'initial_h_values' in globals():
        h_vals = np.array(initial_h_values)
        h_vals = h_vals[:n_points]
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
    else:
        h_vals = np.random.rand(n_points)
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)

    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def grad_objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        c5_max = np.max(corr)
        max_idx = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            h[i] += 1e-6
            corr = np.correlate(h, 1 - h, mode="full")
            grad_diff = (np.max(corr) - c5_max) * dx
            grad[i] = grad_diff
            h[i] -= 1e-6
        return grad

    result = optimize.minimize(
        objective,
        h_vals,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        jac=grad_objective,
        options={'ftol': 1e-8, 'maxiter': 100}
    )

    best_h = result.x.copy()
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    c5_bound = objective(best_h)

    return (best_h, c5_bound, n_points)
