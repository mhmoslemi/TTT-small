import numpy as np
import scipy.optimize as optimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = initial_h_values
    assert np.isclose(np.sum(initial_h), target_sum, atol=1e-6)
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    elapsed_time = 0.0
    max_iter = int(budget_s / 2)
    alpha = 0.1
    eps = 1e-8
    early_stop = 30
    improvement = np.inf

    for i in range(max_iter):
        if elapsed_time >= budget_s:
            break
        cross_corr = correlate(h, 1 - h, mode="full") * dx
        max_index = np.argmax(cross_corr)
        if max_index >= len(h) - 1:
            max_index = len(h) - 2
        # Compute gradient for max_index
        h_perturb = h.copy()
        h_perturb[max_index] += eps
        c5_perturb = evaluate_erdos_solution(h_perturb)
        forward_diff = (c5_perturb - best_c5) / eps
        h_perturb[max_index] -= 2 * eps
        c5_perturb = evaluate_erdos_solution(h_perturb)
        backward_diff = (best_c5 - c5_perturb) / eps
        grad = (forward_diff + backward_diff) / 2
        # Update h
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (target_sum / h_sum)
        # Evaluate
        new_c5 = evaluate_erdos_solution(h)
        elapsed_time += 0.01  # Approximate time per iteration
        if np.abs(new_c5 - best_c5) < improvement:
            improvement = np.abs(new_c5 - best_c5)
            best_c5 = new_c5
            best_h = h.copy()
        if i > 0 and np.abs(new_c5 - best_c5) < 1e-4 and i > early_stop:
            break
    return best_h, best_c5, n_points
