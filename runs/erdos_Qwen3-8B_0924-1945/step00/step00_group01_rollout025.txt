import numpy as np
import scipy.optimize as optimize
from scipy.interpolate import interp1d
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    if np.sum(initial_h_values) * dx != 1.0:
        initial_h_values = initial_h_values / np.sum(initial_h_values) * (n_points / 2)
    h_vals = initial_h_values.copy()
    n_points = len(h_vals)
    dx = 2.0 / n_points

    def compute_overlap(h_vals):
        h = np.concatenate([h_vals, np.zeros_like(h_vals)])
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        overlaps = corr[n_points - 1:2 * n_points - 1]
        return overlaps * dx

    def objective(h_vals):
        overlaps = compute_overlap(h_vals)
        return np.max(overlaps)

    def constraint_sum(h_vals):
        return np.sum(h_vals) - (n_points / 2)

    def constraint_bounds(h_vals):
        return np.all((h_vals >= 0) & (h_vals <= 1))

    def iterate_optimization(h_vals, iterations=100, ftol=1e-8):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = optimize.minimize(
            objective,
            h_vals,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[
                {'type': 'eq', 'fun': constraint_sum},
                {'type': 'ineq', 'fun': constraint_bounds}
            ],
            tol=ftol,
            options={'maxiter': iterations, 'ftol': ftol, 'gtol': ftol}
        )
        if result.success:
            h_vals = result.x
            if np.all((h_vals >= 0) & (h_vals <= 1)) and np.abs(np.sum(h_vals) - n_points / 2) < 1e-8:
                return h_vals
        return h_vals

    start_time = time.time()
    h_vals = iterate_optimization(h_vals, iterations=100)
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        h_vals = iterate_optimization(h_vals, iterations=200)
    c5 = compute_overlap(h_vals)
    return h_vals, c5, n_points
