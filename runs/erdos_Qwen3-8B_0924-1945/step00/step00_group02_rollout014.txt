import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    initial_h = kwargs.get('initial_h_values', None)
    
    if initial_h is not None:
        h = np.clip(initial_h, 0, 1)
        h = h * n_points / 2.0 / np.sum(h) if np.sum(h) > 0 else np.random.rand(n_points) * 0.5
    else:
        h = np.random.rand(n_points) * 0.5
        h = h * n_points / 2.0 / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
    
    T = 100.0
    alpha = 0.995
    max_iter = int(budget_s / 0.1)
    iter_count = 0
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        idxs = np.random.choice(n_points, size=5, replace=False)
        noise = np.random.normal(0, 0.05 * np.exp(-iter_count / 200), size=n_points)
        new_h = h + noise
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2.0) / np.sum(new_h) if np.sum(new_h) > 0 else new_h * 0.5
        
        corr = np.correlate(new_h, 1 - new_h, mode="full")
        c5_new = np.max(corr) * dx
        
        delta = c5_new - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = new_h.copy()
            best_c5 = c5_new
        
        if np.abs(best_c5 - np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx) > 1e-6:
            best_h = h.copy()
        
        T *= alpha
        iter_count += 1
    
    return best_h, best_c5, n_points
