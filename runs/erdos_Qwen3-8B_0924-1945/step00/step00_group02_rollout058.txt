import numpy as np
import numpy.random as npr
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = 450
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.copy(initial_h_values)
    else:
        h = 0.5 * (1 + npr.uniform(-1, 1, n_points))
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    h = h * (n_points / 2) / total if total != 0 else h * n_points / 2
    best_h = np.copy(h)
    best_c5 = compute_c5(h, n_points)
    start_time = time.time()

    for _ in range(int((budget_s - (time.time() - start_time)) // 0.1)):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        lag_indices = np.arange(-10, 11)
        target_lags = np.clip(max_k + lag_indices, 0, len(corr) - 1)
        target_lags = np.unique(target_lags)
        if len(target_lags) > 0:
            target_corr = corr[target_lags]
            max_corr = np.max(target_corr)
            if max_corr < best_c5:
                best_c5 = max_corr
                best_h = np.copy(h)
            for lag in target_lags:
                if lag < 0 or lag >= len(h):
                    continue
                sens = (1 - h[lag]) * (h[lag] - 0.5)  # Sensitivity to perturbation
                if sens != 0:
                    delta = -sign(sens) * 0.05
                    h[lag] += delta
                    h[lag] = np.clip(h[lag], 0, 1)
                    total = np.sum(h)
                    h = h * (n_points / 2) / total if total != 0 else h * n_points / 2
        if np.random.rand() < 0.1:
            indices = np.random.choice(n_points, 5, replace=False)
            noise = npr.normal(0, 0.1, 5)
            h[indices] += noise
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            h = h * (n_points / 2) / total if total != 0 else h * n_points / 2
        curr_c5 = compute_c5(h, n_points)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = np.copy(h)
        if _ % 100 == 0:
            h = np.copy(best_h)

    h = np.copy(best_h)
    total = np.sum(h)
    h = h * (n_points / 2) / total if total != 0 else h * n_points / 2
    final_c5 = compute_c5(h, n_points)
    return h, final_c5, n_points

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * 2.0 / n_points

def sign(x):
    return np.sign(x)
