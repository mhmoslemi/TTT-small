import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70  # Start with the initial 70-point construction
    initial_h = initial_h_values
    dx = 2.0 / n_points

    def objective(h):
        h = np.clip(h, 0, 1)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1)] * n_points
    constraint = {'type': 'eq', 'fun': constraint_sum}

    # Try initial optimization
    result = opt.minimize(
        objective,
        initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-6}
    )

    best_h = result.x
    best_c5 = objective(best_h)
    best_time = time.time()

    # Try a perturbed version
    perturbed_h = np.clip(initial_h + np.random.normal(0, 0.1, n_points), 0, 1)
    result_pert = opt.minimize(
        objective,
        perturbed_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-6}
    )
    pert_h = result_pert.x
    pert_c5 = objective(pert_h)

    # Compare and choose the better result
    if pert_c5 < best_c5:
        best_h = pert_h
        best_c5 = pert_c5

    elapsed = time.time() - best_time
    if elapsed < budget_s - 50:
        # Try Nelder-Mead as a fallback
        result_nm = opt.minimize(
            objective,
            best_h,
            method='Nelder-Mead',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 50, 'xatol': 1e-5}
        )
        nm_h = result_nm.x
        nm_c5 = objective(nm_h)
        if nm_c5 < best_c5:
            best_h = nm_h
            best_c5 = nm_c5

    # Final check
    best_h = np.clip(best_h, 0, 1)
    assert np.isclose(np.sum(best_h), n_points / 2), "Constraint violation"
    final_c5 = objective(best_h)

    return best_h, final_c5, n_points
