import numpy as np
import scipy.optimize as optimize
from scipy.signal import correlate

    initial_h_vals = initial_h_vals / np.sum(initial_h_vals) * (n_points_initial / 2)
    
    # Determine number of points based on time budget
    # Start with 500 points, and adjust if time runs out
    n_points = 500
    dx = 2.0 / n_points
    
    # Initial guess
    h = np.zeros(n_points)
    h[:n_points_initial] = initial_h_vals[:n_points_initial]
    h = h / np.sum(h) * (n_points / 2)
    
    # Objective function: max correlation scaled by dx
    def obj_func(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        corr = correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Gradient approximation via finite differences
    def grad_func(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        
        # Compute gradient approximation
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-6
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus / np.sum(h_plus) * (n_points / 2)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            grad[i] = (corr_plus[max_idx] - corr[max_idx]) * dx / 1e-6
        
        return grad.flatten()
    
    # Optimization settings
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
    
    # Optimization using L-BFGS-B
    result = optimize.minimize(
        fun=obj_func,
        x0=h.flatten(),
        method='L-BFGS-B',
        jac=grad_func,
        bounds=bounds,
        constraints=constraints,
        tol=1e-8,
        maxiter=100
    )
    
    # Extract the best solution
    best_h_flat = result.x
    best_h_vals = best_h_flat.reshape(n_points)

    best_h_vals = best_h_vals / np.sum(best_h_vals) * (n_points / 2)

    corr = correlate(best_h_vals, 1 - best_h_vals, mode="full")
    c5_bound = np.max(corr) * dx
