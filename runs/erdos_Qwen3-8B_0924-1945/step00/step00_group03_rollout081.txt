import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=np.float64)
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if np.abs(h_sum - n_points / 2) > 1e-5:
        h += (n_points / 2 - h_sum) / n_points
        h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    
    def objective(x):
        nonlocal best_c5
        x = np.clip(x, 0, 1)
        conv = np.correlate(x, 1 - x, mode='full')
        c5 = np.max(conv) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = x.copy()
        return c5
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        result = opt.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5, options={'maxiter': 100})
        h = result.x
    except:
        pass
    
    h = np.clip(h, 0, 1)
    conv = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(conv) * dx
    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        h = best_h
        c5 = best_c5
    
    h = np.clip(h, 0, 1)
    conv = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(conv) * dx
    return (h.tolist(), c5, n_points)
