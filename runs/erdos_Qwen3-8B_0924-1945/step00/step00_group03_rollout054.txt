import numpy as np
import scipy.optimize
import time
import copy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Preloaded initial solution
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def evaluate_erdos(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = np.copy(initial_h_values)
    best_h = h.copy()
    best_c5 = evaluate_erdos(h)
    t_total = int(budget_s * 0.75)
    T0 = 0.7 * (1.0 + best_c5) / 3.0
    Tmax = int(0.1 * t_total)
    beta = 0.75
    gamma = 0.65
    lambda0 = 0.15 / dx

    for t in range(t_total):
        if t % 100 == 0 and t < t_total // 2:
            lambda0 *= 0.8
        elif t % 100 == 0 and t >= t_total // 2:
            lambda0 *= 0.95

        noise = np.random.normal(0, lambda0, size=n_points)
        perturbed = h + noise
        perturbed = np.clip(perturbed, 0.0, 1.0)
        current_sum = np.sum(perturbed)
        delta = (n_points / 2 - current_sum) / (n_points * 0.95)
        perturbed += delta * np.random.choice([-1, 1], size=n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        perturbed = perturbed * (n_points / 2 / np.sum(perturbed)) if np.sum(perturbed) != 0 else perturbed

        new_c5 = evaluate_erdos(perturbed)
        if new_c5 < best_c5 or (np.random.rand() < np.exp(-(new_c5 - best_c5) / (T0 * (1 - t / Tmax) ** (1 / beta)))):
            h = perturbed.copy()
            best_h = h.copy()
            best_c5 = new_c5

        if best_c5 < 0.3807:
            break

    # Final validation
    assert (best_h >= 0).all() and (best_h <= 1).all()
    assert np.isclose(np.sum(best_h), n_points / 2, atol=1e-6)
    final_c5 = evaluate_erdos(best_h)
    return best_h, final_c5, n_points
