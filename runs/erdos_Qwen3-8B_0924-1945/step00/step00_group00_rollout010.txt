import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = kwargs.get('initial_h_values', None)
    
    n_points = 45  # start with 45 points for speed
    h = np.zeros(n_points)
    if initial_h is not None:
        h = np.array(initial_h, dtype=float)
    else:
        # Random initial guess with sum(h) = n_points / 2
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Normalize
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    start_time = time.time()
    
    learning_rate = 0.1
    momentum = 0.9
    noise_scale = 0.05
    noise_freq = 100
    max_iter = int(budget_s / 1.5)
    iter_count = 0
    
    v = np.zeros(n_points)
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        
        # Compute correlate values
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        current_max = np.max(correlate_vals)
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        
        # Find worst lag
        worst_lag = np.argmax(correlate_vals)
        # Compute gradient for worst lag
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * (2.0 / n_points)
        
        # Update momentum and h
        v = momentum * v + learning_rate * gradients
        h = h - v
        
        # Clip and normalize
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Add noise
        if iter_count % noise_freq == 0:
            noise = noise_scale * np.random.normal(0, 1, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early stopping if improvement is negligible
        if iter_count > 100 and current_max - best_c5 < 1e-5:
            break
    
    return (best_h, best_c5, n_points)

def compute_c5(h, n_points):
    correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
    return np.max(correlate_vals)
