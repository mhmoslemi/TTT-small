import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    target_sum = n_points / 2
    initial_h_values = np.array([0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333,
                                 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333,
                                 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333,
                                 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333,
                                 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333])
    def clip_and_scale(h, target_sum):
        h_clipped = np.clip(h, 0, 1)
        h_scaled = h_clipped * (target_sum / np.sum(h_clipped))
        return h_scaled

    def evaluate_c5(h):
        h = clip_and_scale(h, target_sum)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * (2.0 / n_points)

    def objective(h):
        return evaluate_c5(h)

    best_h = clip_and_scale(initial_h_values, target_sum)
    best_c5 = evaluate_c5(best_h)
    elapsed_time = 0
    start_time = time.time()
    population_size = 50
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [clip_and_scale(p, target_sum) for p in population]
    fitness = [objective(p) for p in population]
    best_idx = np.argmin(fitness)
    best_c5 = fitness[best_idx]
    best_h = population[best_idx]
    best_fitness = best_c5
    no_improvement_count = 0
    mutation_factor = 0.5
    crossover_prob = 0.7
    perturbation_range = 0.05
    while elapsed_time < budget_s:
        new_population = []
        for i in range(population_size):
            p1 = population[np.random.randint(population_size)]
            p2 = population[np.random.randint(population_size)]
            p3 = population[np.random.randint(population_size)]
            mutant = p1 + mutation_factor * (p2 - p3)
            mutant = np.clip(mutant, 0, 1)
            mutant = clip_and_scale(mutant, target_sum)
            new_population.append(mutant)
        new_fitness = [objective(m) for m in new_population]
        combined_population = population + new_population
        combined_fitness = fitness + new_fitness
        indices = np.argsort(combined_fitness)
        combined_population = np.array(combined_population)[indices]
        combined_fitness = np.array(combined_fitness)[indices]
        population = combined_population[:population_size]
        fitness = combined_fitness[:population_size]
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        if current_best_c5 < best_fitness - 1e-6:
            best_fitness = current_best_c5
            best_h = population[current_best_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        if no_improvement_count >= 50:
            break
        elapsed_time = time.time() - start_time
    return (best_h, best_fitness, n_points)
