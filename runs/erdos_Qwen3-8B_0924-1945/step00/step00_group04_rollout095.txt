import numpy as np
import scipy.optimize as opt
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Create a default starting vector
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 if i % 2 == 0 else 0.0
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def objective(h):
        return evaluate(h)

    def bounds_func(x):
        return np.clip(x, 0, 1)

    def rescale(x):
        x = np.clip(x, 0, 1)
        return x * (target_sum / np.sum(x))

    def mutation(x, bounds):
        mutation_factor = 0.5
        mutant = np.clip(x + mutation_factor * np.random.uniform(-1, 1, size=len(x)), 0, 1)
        return mutant

    def crossover(parent1, parent2):
        crossover_prob = 0.7
        child = np.copy(parent1)
        mask = np.random.rand(len(child)) < crossover_prob
        child[mask] = parent2[mask]
        return child

    def constraint(x):
        return np.sum(x) - target_sum

    def constraint_func(x):
        return constraint(x)

    # Optimization setup
    x0 = rescale(h)
    result = differential_evolution(
        objective,
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=[constraint_func],
        popsize=50,
        mutation=(0.5, 1),
        recombination=0.7,
        tol=1e-6,
        maxiter=int(budget_s / 1),
        seed=seed,
        polish=True,
        disp=False
    )

    best_h = rescale(result.x)
    best_c5 = evaluate(best_h)

    return best_h, best_c5, n_points
