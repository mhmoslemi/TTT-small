import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    if 'initial_h_values' in globals():
        h_vals = np.array(initial_h_values, dtype=np.float32)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    else:
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    
    dx = 2.0 / n_points
    max_iterations = 100
    improve_threshold = 1e-4
    no_improve_counter = 0
    best_c5 = np.inf
    best_h_vals = h_vals.copy()
    
    start_time = time.time()
    
    for _ in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 100:
            break
        
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5_values = corr * dx
        max_c5 = np.max(c5_values)
        
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h_vals = h_vals.copy()
            no_improve_counter = 0
        else:
            no_improve_counter += 1
        
        if no_improve_counter >= 5:
            break
        
        k = np.argmax(c5_values)
        corr_k = c5_values[k]
        
        def objective(x):
            h = x.reshape(-1)
            return corr_k - np.sum(h * (1 - h[(np.arange(n_points) + k) % n_points])) * dx
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        
        result = opt.minimize(objective, h_vals, method='L-BFGS-B', bounds=bounds, constraints=cons)
        if result.success:
            h_vals = result.x
        else:
            h_vals = best_h_vals.copy()
        
        h_vals = np.clip(h_vals, 0.0, 1.0)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    
    if np.sum(h_vals) != n_points / 2:
        h_vals = best_h_vals.copy()
    
    corr = np.correlate(h_vals, 1 - h_vals, mode='full')
    c5_bound = np.max(corr) * dx
    return (h_vals, c5_bound, n_points)
