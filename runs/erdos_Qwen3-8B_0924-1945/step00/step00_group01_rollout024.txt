import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 50
    if 'initial_h_values' in globals():
        h_vals = np.array(initial_h_values)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    else:
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)

    dx = 2.0 / n_points
    max_iterations = 200
    tolerance = 1e-6
    best_c5 = float('inf')
    best_h_vals = h_vals.copy()
    start_time = time.time()

    for _ in range(max_iterations):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        current_c5 = np.max(corr * dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_vals = h_vals.copy()

        if time.time() - start_time > budget_s - 10:
            break

        k_max = np.argmax(corr)
        overlap_k = corr[k_max] * dx
        lambda_var = cp.Variable()
        obj = 0.5 * lambda_var ** 2 + (overlap_k - lambda_var) ** 2
        prob = cp.Problem(cp.Minimize(obj))
        try:
            prob.solve()
        except:
            lambda_val = 0
        else:
            lambda_val = lambda_var.value

        gradient = np.zeros_like(h_vals)
        for i in range(n_points):
            for k in range(n_points):
                if 0 <= i - k < n_points:
                    gradient[i] += (1 - h_vals[k]) * (lambda_val - (1 - h_vals[k])) * dx
                if 0 <= i + k < n_points:
                    gradient[i] += h_vals[k] * (lambda_val - h_vals[k]) * dx
        h_vals -= gradient * 0.1
        h_vals = np.clip(h_vals, 0, 1)
        mass = np.sum(h_vals)
        h_vals = h_vals * (n_points / 2) / mass

    corr = np.correlate(best_h_vals, 1 - best_h_vals, mode='full')
    final_c5 = np.max(corr * dx)
    return (best_h_vals, final_c5, n_points)
