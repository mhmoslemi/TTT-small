import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 47  # Use initial_h_values if provided
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else None
    if initial_h_values is not None:
        h_vals = initial_h_values.copy()
        n_points = len(h_vals)
        sum_h = np.sum(h_vals)
        if np.isclose(sum_h, n_points / 2):
            pass
        else:
            h_vals = (h_vals * (n_points / 2)) / sum_h
    else:
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    
    dx = 2.0 / n_points
    max_iterations = 100
    improvement_threshold = 1e-4
    best_c5 = np.inf
    best_h = h_vals.copy()
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 100:
            break
        
        # Compute correlations
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5_values = corr * dx
        max_c5 = np.max(c5_values)
        
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h_vals.copy()
        
        if np.abs(max_c5 - best_c5) < improvement_threshold:
            break
        
        # Select the worst lag
        max_lag_index = np.argmax(c5_values)
        lag = max_lag_index - (n_points - 1)
        
        # Define the overlap function for this lag
        def objective(x):
            # Apply bounds and floor/ceiling
            x = np.clip(x, 0, 1)
            total = np.sum(x)
            # Adjust to satisfy the sum constraint using simple proportional adjustment
            if np.isclose(total, n_points / 2):
                return c5_values[max_lag_index]
            else:
                adjustment = (n_points / 2 - total) / n_points
                x = np.clip(x + adjustment, 0, 1)
                total = np.sum(x)
                if np.isclose(total, n_points / 2):
                    return c5_values[max_lag_index]
                else:
                    return np.inf
        
        # Optimize with bounds and equality constraint
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        result = opt.minimize(objective, h_vals, bounds=bounds, constraints=cons, method='SLSQP')
        
        if result.success:
            h_vals = result.x
        else:
            h_vals = np.clip(h_vals, 0, 1)
            total = np.sum(h_vals)
            if np.isclose(total, n_points / 2):
                pass
            else:
                adjustment = (n_points / 2 - total) / n_points
                h_vals = np.clip(h_vals + adjustment, 0, 1)
    
    # Final evaluation
    corr = np.correlate(h_vals, 1 - h_vals, mode='full')
    c5_values = corr * dx
    c5_bound = np.max(c5_values)
    
    # Validate constraints
    if not np.all((h_vals >= 0) & (h_vals <= 1)) or not np.isclose(np.sum(h_vals), n_points / 2):
        h_vals = best_h.copy()
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5_values = corr * dx
        c5_bound = np.max(c5_values)
    
    return (h_vals, c5_bound, n_points)
