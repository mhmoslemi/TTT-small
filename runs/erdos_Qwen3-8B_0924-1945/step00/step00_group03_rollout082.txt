import numpy as np
import time
import sys

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h.npy') if 'initial_h_values' in kwargs else np.random.rand(70)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    max_iter = 500
    perturb_prob = 0.05
    learning_rate = 0.02
    tolerance = 1e-4

    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        cross_corr = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            if k >= n_points:
                k_wrap = k - n_points
                cross_corr[k] = (dx * np.sum(h[:n_points - k_wrap] * h[k_wrap:]))
            else:
                cross_corr[k] = (dx * np.sum(h[:n_points - k] * h[k:]))
        current_max = np.max(cross_corr)
        max_k = np.argmax(cross_corr)

        grad = np.zeros(n_points)
        if max_k >= n_points:
            k_wrap = max_k - n_points
            for i in range(n_points - k_wrap):
                grad[i] += dx * h[i + k_wrap]
                grad[i + k_wrap] += dx * h[i]
        else:
            for i in range(n_points - max_k):
                grad[i] += dx * h[i + max_k]
                grad[i + max_k] += dx * h[i]
        direction = -grad
        new_h = h + learning_rate * direction
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        scaling = (n_points / 2) / new_sum if new_sum != 0 else 1.0
        new_h = new_h * scaling
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        scaling = (n_points / 2) / new_sum if new_sum != 0 else 1.0
        new_h = new_h * scaling

        if np.random.rand() < perturb_prob:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            new_h = h + noise
            new_h = np.clip(new_h, 0.0, 1.0)
            new_sum = np.sum(new_h)
            scaling = (n_points / 2) / new_sum if new_sum != 0 else 1.0
            new_h = new_h * scaling
            new_h = np.clip(new_h, 0.0, 1.0)

        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            h = new_h.copy()
            best_c5 = new_c5
            best_h = h.copy()
        if np.abs(new_c5 - best_c5) < tolerance:
            break

    if np.isnan(best_c5) or np.isinf(best_c5):
        best_c5 = evaluate_erdos_solution(best_h, dx)

    return (best_h, best_c5, n_points)
