import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250  # Reduce n_points to fit within time budget
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5] * n_points)  # Default if no parent construction is provided

    # Start with initial_h_values or a random guess if not provided
    h = np.array(initial_h_values, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    h = h * (n_points / 2.0) / np.sum(h)  # Normalize to meet the integral constraint
    best_h = h.copy()
    best_c5 = float('inf')
    T = 100.0  # Initial temperature
    alpha = 0.995  # Cooling rate
    total_iterations = int(budget_s / 0.1)
    start_time = time.time()

    for t in range(total_iterations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Gaussian perturbation
        indices = np.random.choice(n_points, 5, replace=False)
        noise = np.random.normal(0, 0.05 * np.exp(-t / 200), size=n_points)
        new_h = h + noise
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = new_h * (n_points / 2.0) / np.sum(new_h)  # Restore integral constraint

        # Compute C5
        try:
            corr = np.correlate(new_h, 1 - new_h, mode="full")
            c5_new = np.max(corr) * dx
        except:
            c5_new = best_c5  # In case of error, keep previous best

        # Update best solution if improved
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = new_h.copy()

        # Cool down
        T *= alpha

    return best_h, best_c5, n_points
