import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in kwargs:
        h = np.array(kwargs['initial_h_values'], dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h_norm = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h_norm, 1 - h_norm, mode="full") * dx
        return np.max(correlate_vals)
    
    learning_rate = 0.05
    noise_scale = 0.05
    max_iterations = 500
    early_stop = 50
    improvement_threshold = 1e-5
    
    for it in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        current_c5 = evaluate(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if it % 100 == 0 and it != 0:
            noise_scale *= 1.1
        
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        worst_lag = np.argmax(correlate_vals)
        
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * dx
        
        h = h - learning_rate * gradients
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h += noise_scale * np.random.normal(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    return (best_h, best_c5, n_points)
