import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_generations = 100
    population_size = 50
    tournament_size = 5
    crossover_prob = 0.8
    mutation_amp = 0.01
    n_points = 50
    dx = 2.0 / n_points

    # Initialize population
    if 'initial_h_values' in globals():
        initial = np.array(initial_h_values, dtype=np.float64)
        h_vals = (initial * (n_points / 2)) / np.sum(initial)
    else:
        h_vals = np.random.rand(n_points)
        h_vals -= h_vals.mean()
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    population = [h_vals.copy() for _ in range(population_size)]
    best_h = h_vals.copy()
    best_c5 = evaluate(best_h)

    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = [evaluate(ind) for ind in population]
        # Select parents using tournament selection
        parents = []
        for _ in range(population_size):
            tournament = random.sample(list(enumerate(fitness)), tournament_size)
            best_idx = min(tournament, key=lambda x: x[1])[0]
            parents.append(population[best_idx].copy())
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            if np.random.rand() < crossover_prob:
                p1, p2 = parents[i], parents[i+1]
                split = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([p1[:split], p2[split:]])
                child2 = np.concatenate([p2[:split], p1[split:]])
                offspring.append(child1)
                offspring.append(child2)
            else:
                offspring.append(parents[i].copy())
                offspring.append(parents[i+1].copy())
        # Mutation
        for i in range(population_size):
            if np.random.rand() < mutation_amp:
                offspring[i] += np.random.normal(0, 0.01, n_points)
                offspring[i] = np.clip(offspring[i], 0, 1)
                offspring[i] = offspring[i] / np.sum(offspring[i]) * (n_points / 2)
        # Replace population
        fitness_offspring = [evaluate(ind) for ind in offspring]
        combined = list(zip(population, fitness)) + list(zip(offspring, fitness_offspring))
        combined.sort(key=lambda x: x[1])
        population = [x[0] for x in combined[:population_size]]
        # Update best
        if evaluate(population[0]) < best_c5:
            best_h = population[0].copy()
            best_c5 = evaluate(best_h)
        print(f"Generation {gen + 1}, Best C5: {best_c5}")
    
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h / np.sum(final_h) * (n_points / 2)
    final_c5 = evaluate(final_h)
    return (final_h, final_c5, n_points)
