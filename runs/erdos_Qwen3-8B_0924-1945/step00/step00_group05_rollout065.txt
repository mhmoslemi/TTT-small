import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    n_points = 500
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.clip(initial_h_values, 0, 1)
    else:
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        h = (n_points / 2) * h / sum_h
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    learning_rate = 0.01
    momentum = 0.9
    for _ in range(500):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(corr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Find the maximum overlap
        max_index = np.argmax(corr)
        # Compute gradient for this max_index
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + max_index) % n_points
            grad[i] += (1 - h[j])
            j = (i - max_index) % n_points
            grad[i] -= h[j]
        # Update h with gradient descent
        h = h - learning_rate * grad
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        # Enforce sum constraint
        sum_h = np.sum(h)
        if sum_h == 0:
            h = best_h.copy()
        else:
            h = (n_points / 2) * h / sum_h
        # Perturbation with 5% probability
        if np.random.rand() < 0.05:
            noise = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + noise, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                h = (n_points / 2) * h / sum_h
        # Reduce learning rate every 100 iterations
        if _ % 100 == 0 and learning_rate > 1e-5:
            learning_rate *= 0.99
    # Final check
    corr = np.correlate(best_h, 1 - best_h, mode="full") * dx
    final_c5 = np.max(corr)
    return (best_h, final_c5, n_points)
