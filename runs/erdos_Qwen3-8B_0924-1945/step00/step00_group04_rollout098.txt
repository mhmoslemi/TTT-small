import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h_values = np.array([0.35, 0.25, 0.75, 0.3, 0.7, 0.5, 0.4, 0.6, 0.55, 0.35,
                                0.25, 0.65, 0.3, 0.5, 0.45, 0.6, 0.35, 0.55, 0.2, 0.5,
                                0.4, 0.65, 0.25, 0.5, 0.35, 0.55, 0.3, 0.6, 0.2, 0.5,
                                0.45, 0.5, 0.3, 0.6, 0.4, 0.55, 0.25, 0.5, 0.35, 0.65,
                                0.4, 0.5, 0.35, 0.6])
    h = np.clip(initial_h_values, 0, 1)
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
        
    best_h = h.copy()
    best_c5 = evaluate(best_h)
    elapsed = 0
    pop_size = 50
    pop = [h.copy() for _ in range(pop_size)]
    
    def mutate(h):
        h = h.copy()
        idxs = np.random.choice(n_points, 3, replace=False)
        diff = pop[np.random.randint(pop_size)] - pop[np.random.randint(pop_size)]
        h[idxs] += diff[idxs]
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0, 1)
        return h
    
    def crossover(a, b):
        child = a.copy()
        idxs = np.random.choice(n_points, int(n_points * 0.5), replace=False)
        child[idxs] = b[idxs]
        child = np.clip(child, 0, 1)
        child = child * (target_sum / np.sum(child))
        child = np.clip(child, 0, 1)
        return child
    
    def fitness(h):
        return -evaluate(h)
    
    for _ in range(10000):
        start = time.time()
        new_pop = []
        for i in range(pop_size):
            h_mut = mutate(pop[i])
            new_pop.append(h_mut)
        for i in range(pop_size):
            a, b = np.random.choice(pop, 2, replace=False)
            child = crossover(a, b)
            new_pop.append(child)
        fit = [fitness(h) for h in new_pop]
        sorted_indices = np.argsort(fit)
        new_pop = [new_pop[i] for i in sorted_indices[:pop_size]]
        new_pop = new_pop[:pop_size]
        for i in range(pop_size):
            if fitness(new_pop[i]) < fitness(pop[i]):
                pop[i] = new_pop[i]
        elapsed += time.time() - start
        if elapsed >= budget_s:
            break
    
    best_h = np.clip(pop[np.argmin([fitness(h) for h in pop])], 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h))
    best_h = np.clip(best_h, 0, 1)
    best_c5 = evaluate(best_h)
    
    return (best_h, best_c5, n_points)
