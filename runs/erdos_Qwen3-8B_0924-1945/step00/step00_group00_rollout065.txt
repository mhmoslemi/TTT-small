import numpy as np
import scipy.signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    h = np.array(initial_h_values, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    h /= np.sum(h) * 2.0
    h *= n_points / 2.0
    h = np.clip(h, 0.0, 1.0)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    dx = 2.0 / n_points
    learning_rate = 0.01
    for _ in range(10000):
        if time.time() - start_time > budget_s:
            break
        # Compute overlap integrals
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Find worst lag
        max_idx = np.argmax(corr)
        # Compute gradient direction
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = max_idx - i
            if 0 <= j < len(corr):
                grad[i] += (1 - h[j]) * dx
        # Update h
        h -= learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2.0) / np.sum(h)
        # Random perturbation
        if np.random.rand() < 0.05:
            h += np.random.normal(0, 0.01, size=n_points)
            h = np.clip(h, 0.0, 1.0)
            h = h * (n_points / 2.0) / np.sum(h)
    h = np.clip(best_h, 0.0, 1.0)
    h = h * (n_points / 2.0) / np.sum(h)
    best_c5 = compute_c5(h)
    return h.tolist(), best_c5, n_points

def compute_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    corr = scipy.signal.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
