```python
import numpy as np
import scipy.optimize as opt
import time
import copy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in globals() else np.random.rand(70)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def normalize_h(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return np.clip(h * scale, 0, 1)
    
    def evaluate_c5(h):
        h = normalize_h(h)
        h_1_minus = 1 - h
        corr = np.correlate(h, h_1_minus, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def objective_func(h):
        return evaluate_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    def constraint_bounds(h):
        return np.all((h >= 0) & (h <= 1))
    
    def optimize_h(h_initial):
        h = copy.deepcopy(h_initial)
        h = normalize_h(h)
        result = opt.minimize(objective_func, h, method='L-BFGS-B', bounds=[(0, 1)] * len(h))
        best_h = result.x
        best_c5 = evaluate_c5(best_h)
        return best_h, best_c5
    
    start_time = time.time()
    best_h = normalize_h(initial_h_values)
    best_c5 = evaluate_c5(best_h)
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        
        h_perturbed = best_h + np.random.normal(0, 0.05, len(best_h))
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = normalize_h(h_perturbed)
        current_c5 = evaluate_c5(h_perturbed)
        
        if current_c5 < best_c5:
            best_h = h_perturbed
            best_c5 = current_c5
    
    if time.time() - start_time > budget_s:
        pass
    else:
        h_optimized, c5_optimized = optimize_h(best_h)
        if evaluate_c5(h_optimized) < best_c5:
            best_h = h_optimized
            best_c5 = c5_optimized
    
    if time.time() - start_time > budget_s:
        pass
    else:
        h_final = best_h
        c5_final = evaluate_c5(h_final)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    
    return (best_h, best_c5, n_points)
```
