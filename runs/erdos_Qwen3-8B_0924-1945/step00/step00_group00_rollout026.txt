import numpy as np
import scipy.signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    h = initial_h_values.copy()
    h = (h - np.min(h)) / (np.max(h) - np.min(h))
    h = h / np.sum(h) * (n_points / 2)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.max(scipy.signal.correlate(h, 1 - h, mode="full")) * (2.0 / n_points)
    
    start_time = time.time()
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = scipy.signal.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        lag = np.argmax(corr)
        if max_corr < best_c5:
            best_c5 = max_corr
            best_h = h.copy()
        
        if lag > n_points - 1:
            lag -= n_points
        if lag < 0:
            lag += n_points
        
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = i - lag
            if 0 <= j < n_points:
                gradient[i] = (1 - h[j]) * (2.0 / n_points)
        
        h -= 0.01 * gradient
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        if np.max(np.abs(h - best_h)) < 1e-4 and np.abs(best_c5 - np.max(scipy.signal.correlate(h, 1 - h, mode="full")) * (2.0 / n_points)) < 1e-4:
            break
    
    evaluate_erdos_solution(best_h)
    c5_bound = np.max(scipy.signal.correlate(best_h, 1 - best_h, mode="full")) * (2.0 / n_points)
    return (best_h, c5_bound, n_points)
