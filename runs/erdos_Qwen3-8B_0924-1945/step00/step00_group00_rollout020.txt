import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup: choose n_points
    # We'll start with n_points = 45 (as it's the size of initial_h_values)
    # but try smaller sizes to speed up if needed
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h_values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Generate random initial values satisfying the constraint
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Clip and normalize
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Parameters
    max_iterations = 500
    learning_rate = 0.05
    noise_scale = 0.05
    early_stop_iterations = 50
    improvement_threshold = 1e-5
    
    start_time = time.time()
    
    for iter in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Compute the full correlate
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        current_c5 = np.max(correlate_vals)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Check early stopping condition
        if iter > 0 and abs(current_c5 - best_c5) < improvement_threshold:
            if iter - np.argmin([abs(c - best_c5) for c in [current_c5, best_c5]]) > early_stop_iterations:
                break
        
        # Find worst lag
        worst_lag = np.argmax(correlate_vals)
        
        # Compute gradient for worst lag
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * dx
        
        # Update h
        h = h - learning_rate * gradients
        
        # Clip values to [0,1]
        h = np.clip(h, 0, 1)
        
        # Normalize to maintain the constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Add structured noise
        noise = noise_scale * np.random.normal(0, 1, n_points)
        h += noise
        
        # Clip and normalize again
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final check for best solution
    final_correlate_vals = np.correlate(best_h, 1 - best_h, mode="full") * dx
    final_c5 = np.max(final_correlate_vals)
    
    # Check if initial_h_values is available and use it if current best is worse
    if 'initial_h_values' in globals():
        initial_correlate_vals = np.correlate(initial_h_values, 1 - initial_h_values, mode="full") * dx
        initial_c5 = np.max(initial_correlate_vals)
        if initial_c5 < final_c5:
            best_h = np.array(initial_h_values)
            best_c5 = initial_c5
    
    return (best_h, best_c5, n_points)
