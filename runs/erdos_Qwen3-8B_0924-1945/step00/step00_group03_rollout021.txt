import numpy as np
import scipy.signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Pre-imported initial_h_values provided by the system
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    max_iterations = 500
    learning_rate = 0.1
    perturbation_prob = 0.05
    tolerance = 1e-6
    iteration = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s and iteration < max_iterations:
        # Compute cross-correlation
        cross_corr = np.correlate(h, h, mode='full') * dx
        current_max = np.max(cross_corr)
        
        # Find the lag with maximum cross-correlation
        max_k = np.argmax(cross_corr)
        
        # Compute gradient for the max_k lag
        grad = np.zeros(n_points)
        if max_k >= n_points:
            k_wrap = max_k - n_points
            for i in range(n_points - k_wrap):
                grad[i] += dx * h[i + k_wrap]
                grad[i + k_wrap] += dx * h[i]
        else:
            for i in range(n_points - max_k):
                grad[i] += dx * h[i + max_k]
                grad[i + max_k] += dx * h[i]
        
        # Update h in the direction that reduces the maximum
        direction = -grad
        new_h = h + learning_rate * direction
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        scaling_factor = (n_points / 2) / new_sum if new_sum != 0 else 1.0
        new_h = new_h * scaling_factor
        new_h = np.clip(new_h, 0.0, 1.0)
        
        # Add small random perturbation with probability
        if np.random.rand() < perturbation_prob:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            new_h = h + noise
            new_h = np.clip(new_h, 0.0, 1.0)
            new_sum = np.sum(new_h)
            scaling_factor = (n_points / 2) / new_sum if new_sum != 0 else 1.0
            new_h = new_h * scaling_factor
            new_h = np.clip(new_h, 0.0, 1.0)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Update best solution
        if new_c5 < best_c5:
            h = new_h.copy()
            best_h = h.copy()
            best_c5 = new_c5
        
        # Check for convergence
        if np.abs(new_c5 - best_c5) < tolerance:
            break
        
        iteration += 1
    
    # Final validation
    assert (best_h >= 0).all() and (best_h <= 1).all()
    assert np.isclose(np.sum(best_h), n_points / 2, atol=1e-6)
    assert np.isfinite(best_c5)
    
    return best_h, best_c5, n_points
