import numpy as np
import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    if 'initial_h_values' in globals():
        initial_h_values = np.clip(initial_h_values, 0, 1)
        current_sum = np.sum(initial_h_values)
        if current_sum != 0:
            initial_h_values = initial_h_values * (n_points / 2) / current_sum
        initial_h_values = np.clip(initial_h_values, 0, 1)
    else:
        initial_h_values = np.zeros(n_points)
        # Construct a balanced step function with peaks and valleys
        initial_h_values[0: n_points//4] = 0
        initial_h_values[n_points//4 : 3*n_points//4] = 1
        initial_h_values[3*n_points//4 : ] = 0
        current_sum = np.sum(initial_h_values)
        if current_sum != 0:
            initial_h_values = initial_h_values * (n_points / 2) / current_sum
        initial_h_values = np.clip(initial_h_values, 0, 1)
    
    h = np.copy(initial_h_values)
    best_h = np.copy(h)
    best_c5 = np.inf
    time_start = time.time()
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum != 0:
            h = h * (n_points / 2) / current_sum
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)
    
    def optimize_h(h):
        if np.isclose(np.sum(h), n_points / 2) and np.all(h >= 0) and np.all(h <= 1):
            return evaluate(h)
        return np.inf
    
    def optimize(h, bounds, maxiter=100):
        res = opt.minimize(optimize_h, h, bounds=bounds, method='L-BFGS-B', tol=1e-8, maxiter=maxiter)
        h_opt = np.clip(res.x, 0, 1)
        current_sum = np.sum(h_opt)
        if current_sum != 0:
            h_opt = h_opt * (n_points / 2) / current_sum
        return np.clip(h_opt, 0, 1)
    
    def smart_gradient_descent(h):
        bounds = [(0, 1) for _ in range(n_points)]
        h_opt = optimize(h, bounds, maxiter=200)
        return h_opt
    
    def evolve_population(h, pop_size=50, mutation_rate=0.1, mutation_noise=0.05, generations=100):
        population = [np.copy(h) for _ in range(pop_size)]
        for gen in range(generations):
            if time.time() - time_start > budget_s:
                break
            new_population = []
            for _ in range(pop_size):
                parent = np.random.choice(population)
                if np.random.random() < 0.3:
                    child = np.copy(parent)
                    for i in range(n_points):
                        if np.random.random() < mutation_rate:
                            child[i] += np.random.normal(0, mutation_noise)
                            child[i] = np.clip(child[i], 0, 1)
                    current_sum = np.sum(child)
                    if current_sum != 0:
                        child = child * (n_points / 2) / current_sum
                    new_population.append(np.clip(child, 0, 1))
                else:
                    indices = np.random.choice(n_points, size=2, replace=False)
                    child = np.copy(parent)
                    child[indices[0]] = np.random.uniform(0, 1)
                    child[indices[1]] = np.random.uniform(0, 1)
                    current_sum = np.sum(child)
                    if current_sum != 0:
                        child = child * (n_points / 2) / current_sum
                    new_population.append(np.clip(child, 0, 1))
            population = new_population
            fitness = [evaluate(ind) for ind in population]
            best_idx = np.argmin(fitness)
            best_c5 = fitness[best_idx]
            if best_c5 < best_c5:
                best_c5 = best_c5
                best_h = np.copy(population[best_idx])
            if np.isclose(best_c5, best_c5):
                continue
            population = [population[i] for i in np.argsort(fitness)[:int(pop_size * 0.8)]]
            if np.random.random() < 0.2:
                h = smart_gradient_descent(np.copy(population[0]))
                population.append(h)
    
    evolve_population(h, pop_size=50, mutation_rate=0.1, mutation_noise=0.05, generations=100)
    
    # Final refinement with gradient descent
    h = smart_gradient_descent(best_h)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    h = np.clip(h, 0, 1)
    
    # Final evaluation
    overlaps = np.correlate(h, 1 - h, mode='full') * dx
    c5_bound = np.max(overlaps)
    
    if c5_bound < best_c5:
        best_h = h
        best_c5 = c5_bound
    
    return (best_h, best_c5, n_points)
