import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = np.interp(h, [0, len(h)-1], [0, 2])  # Rescale to [0, 2]
        h = np.clip(h, 0, 2)
        h = np.interp(h, [0, 2], [0, 1])  # Rescale to [0, 1]
        h = np.round(h, 5)
        h = np.clip(h, 0, 1)
        h = h * n_points / 2
        h = np.round(h, 5)
    
    h = np.clip(h, 0, n_points / 2)
    h = h / (n_points / 2)  # Normalize to sum to 1
    h = np.round(h, 5)
    
    # Objective function
    def objective(x):
        h_vals = x
        h_vals = np.clip(h_vals, 0, 1)
        sum_val = np.sum(h_vals)
        if abs(sum_val - n_points / 2) > 1e-6:
            return np.inf
        h_vals = h_vals / (n_points / 2)
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = np.round(h_vals, 5)
        one_minus_h = 1 - h_vals
        corr = np.correlate(h_vals, one_minus_h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Gradient approximation
    def gradient(x):
        h_vals = x
        h_vals = np.clip(h_vals, 0, 1)
        sum_val = np.sum(h_vals)
        if abs(sum_val - n_points / 2) > 1e-6:
            return np.inf * np.ones_like(h_vals)
        h_vals = h_vals / (n_points / 2)
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = np.round(h_vals, 5)
        one_minus_h = 1 - h_vals
        corr = np.correlate(h_vals, one_minus_h, mode='full')
        c5 = np.max(corr) * dx
        grad = np.zeros_like(h_vals)
        for i in range(n_points):
            h_temp = h_vals.copy()
            h_temp[i] += 1e-6
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp / (n_points / 2)
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = np.round(h_temp, 5)
            one_minus_h = 1 - h_temp
            corr_temp = np.correlate(h_temp, one_minus_h, mode='full')
            c5_temp = np.max(corr_temp) * dx
            grad[i] = (c5_temp - c5) / 1e-6
        return grad
    
    # Optimization
    result = opt.minimize(
        objective,
        h,
        method='L-BFGS-B',
        jac=gradient,
        bounds=[(0, 1) for _ in range(n_points)],
        options={'ftol': 1e-8, 'maxiter': 100},
        callback=lambda x: None
    )
    
    # Final evaluation and return
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / (n_points / 2)
    best_h = np.clip(best_h, 0, 1)
    best_h = np.round(best_h, 5)
    one_minus_h = 1 - best_h
    corr = np.correlate(best_h, one_minus_h, mode='full')
    best_c5 = np.max(corr) * dx
    
    return (best_h, best_c5, n_points)
