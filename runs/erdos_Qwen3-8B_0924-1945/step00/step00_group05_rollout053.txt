import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    initial_h_values = np.load('initial_h_values.npy')  # Assume initial construction is available
    h = np.clip(initial_h_values[:n_points], 0, 1)
    sum_h = np.sum(h)
    if sum_h == 0:
        h = np.random.rand(n_points)
        sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    dx = 2.0 / n_points
    learning_rate = 0.1
    iteration = 0
    start_time = time.time()
    while time.time() - start_time < budget_s:
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        k_max = np.argmax(overlaps)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.random.rand(n_points)
            sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h
        new_c5 = evaluate_erdos_solution(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            iteration = 0
        else:
            iteration += 1
            if iteration > 200:
                learning_rate *= 0.95
                iteration = 0
    h_values = best_h
    c5_bound = best_c5
    return h_values, c5_bound, n_points
