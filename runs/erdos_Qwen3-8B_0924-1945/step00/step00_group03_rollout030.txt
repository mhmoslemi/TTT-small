import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h.npy')  # Assume initial construction is provided
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_solution = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    start_time = time.time()
    
    for _ in range(500):
        if time.time() - start_time > budget_s - 10:
            break
        # Compute cross-correlation
        cross_corr = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            if k >= n_points:
                k_wrap = k - n_points
                cross_corr[k] = (dx * np.sum(h[:n_points - k_wrap] * h[k_wrap:]))
            else:
                cross_corr[k] = (dx * np.sum(h[:n_points - k] * h[k:]))
        current_max = np.max(cross_corr)
        
        # Determine lag with maximum cross-correlation
        max_k = np.argmax(cross_corr)
        
        # Compute gradient for the max_k
        grad = np.zeros(n_points)
        if max_k >= n_points:
            k_wrap = max_k - n_points
            for i in range(n_points - k_wrap):
                grad[i] += dx * h[i + k_wrap]
                grad[i + k_wrap] += dx * h[i]
        else:
            for i in range(n_points - max_k):
                grad[i] += dx * h[i + max_k]
                grad[i + max_k] += dx * h[i]
        
        # Update h in the direction that reduces the maximum
        direction = -grad
        step_size = 0.1
        new_h = h + step_size * direction
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        scaling_factor = (n_points / 2) / new_sum if new_sum != 0 else 1.0
        new_h = new_h * scaling_factor
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        scaling_factor = (n_points / 2) / new_sum if new_sum != 0 else 1.0
        new_h = new_h * scaling_factor
        
        # Random perturbation with 5% probability
        if np.random.rand() < 0.05:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            new_h = h + noise
            new_h = np.clip(new_h, 0.0, 1.0)
            new_sum = np.sum(new_h)
            scaling_factor = (n_points / 2) / new_sum if new_sum != 0 else 1.0
            new_h = new_h * scaling_factor
            new_h = np.clip(new_h, 0.0, 1.0)
        
        # Acceptance criteria
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            h = new_h.copy()
            best_solution = h.copy()
            best_c5 = new_c5
        
        # Check convergence
        if np.abs(new_c5 - best_c5) < 1e-6:
            break
    
    # Final validation
    assert (best_solution >= 0).all() and (best_solution <= 1).all()
    assert np.isclose(np.sum(best_solution), n_points / 2, atol=1e-6)
    assert np.isfinite(best_c5)
    return best_solution, best_c5, n_points
