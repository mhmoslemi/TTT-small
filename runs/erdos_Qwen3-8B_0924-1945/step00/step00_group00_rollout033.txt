import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 45
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
    start_time = time.time()
    v = np.zeros(n_points)
    momentum = 0.9
    learning_rate = 0.1
    noise_scale = 0.05
    max_iter = int(budget_s / 1)
    for t in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        worst_lag = np.argmax(correlate_vals)
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * dx
        v = momentum * v + learning_rate * gradients
        h = h - v
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        if np.random.random() < 0.05:
            h += noise_scale * np.random.normal(0, 1, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if t % 100 == 0:
            learning_rate *= 0.995
            if learning_rate < 1e-6:
                learning_rate = 1e-6
    return (best_h, best_c5, n_points)
