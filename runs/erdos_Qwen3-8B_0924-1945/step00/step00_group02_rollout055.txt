import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = 0.5 * (1 + np.random.uniform(-1, 1, n_points))
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if total == 0:
        h = np.zeros(n_points)
    elif total != n_points / 2:
        h = h * (n_points / 2) / total
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    start_time = time.time()
    for _ in range(int((budget_s - (time.time() - start_time)) // 0.1)):
        corr = np.correlate(h, 1 - h, mode="full")
        max_lag = np.argmax(corr)
        if max_lag < 0 or max_lag >= 2 * n_points - 1:
            break
        h_shifted = np.roll(h, max_lag)
        overlap = h * (1 - h_shifted)
        threshold = np.max(overlap) * 0.8
        target_indices = np.where(overlap > threshold)[0]
        target_indices = target_indices[:int(len(target_indices) * 0.1)]
        candidates = []
        for i in target_indices:
            h_candidate = h.copy()
            h_candidate[i] = 1 - h_candidate[i]
            h_candidate = np.clip(h_candidate, 0, 1)
            total_candidate = np.sum(h_candidate)
            if total_candidate == 0:
                continue
            h_candidate = h_candidate * (n_points / 2) / total_candidate
            candidates.append(h_candidate)
            h_candidate = h.copy()
            h_candidate[i] += np.random.uniform(-0.05, 0.05)
            h_candidate = np.clip(h_candidate, 0, 1)
            total_candidate = np.sum(h_candidate)
            if total_candidate == 0:
                continue
            h_candidate = h_candidate * (n_points / 2) / total_candidate
            candidates.append(h_candidate)
        for c in candidates:
            c5 = compute_c5(c, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = c.copy()
        if np.allclose(best_h, h, atol=1e-6):
            break
        h = best_h.copy()
    return best_h, best_c5, n_points

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * 2.0 / n_points
    return c5
