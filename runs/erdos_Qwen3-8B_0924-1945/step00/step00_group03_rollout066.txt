import numpy as np
import scipy.signal
import time
import random
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = evaluate_erdos_solution() if 'evaluate_erdos_solution' in globals() else None
    n_points = 70 if initial_h_values is None else len(initial_h_values)
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h_values = np.copy(initial_h_values)
        sum_h = np.sum(h_values)
        if abs(sum_h - n_points / 2) > 1e-6:
            h_values += (n_points / 2 - sum_h) / n_points
    else:
        h_values = np.zeros(n_points)
        for i in range(n_points):
            h_values[i] = 1 if i % 3 == 0 else 0
        sum_h = np.sum(h_values)
        h_values += (n_points / 2 - sum_h) / n_points
    h_values = np.clip(h_values, 0, 1)
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        conv = scipy.signal.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    def objective(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        x += (n_points / 2 - sum_x) / n_points
        x = np.clip(x, 0, 1)
        return compute_c5(x)
    
    def compute_c5_from_blocks(blocks, n_points, dx):
        block_len = 2 / n_points
        h_values = np.repeat(blocks, n_points // len(blocks) + 1)[:n_points]
        h_values = np.clip(h_values, 0, 1)
        sum_h = np.sum(h_values)
        h_values += (n_points / 2 - sum_h) / n_points
        h_values = np.clip(h_values, 0, 1)
        return compute_c5(h_values)
    
    def block_optimization():
        block_len = 10
        m = n_points // block_len
        if m * block_len < n_points:
            m += 1
        m = min(m, 20)
        blocks = np.zeros(m)
        for i in range(m):
            if i % 2 == 0:
                blocks[i] = 1
            else:
                blocks[i] = 0
        sum_blocks = np.sum(blocks)
        if abs(sum_blocks - m / 2) > 1e-6:
            idx = np.argsort(np.abs(blocks - 0.5))
            blocks[idx[0]] = 1 - blocks[idx[0]]
        
        def evaluate_block_pattern(blocks):
            return compute_c5_from_blocks(blocks, n_points, dx)
        
        best_blocks = blocks.copy()
        best_c5 = evaluate_block_pattern(best_blocks)
        for _ in range(500):
            i = np.random.randint(0, m)
            if i + 1 < m:
                blocks[i], blocks[i + 1] = blocks[i + 1], blocks[i]
                sum_blocks = np.sum(blocks)
                if abs(sum_blocks - m / 2) > 1e-6:
                    idx = np.argsort(np.abs(blocks - 0.5))
                    blocks[idx[0]] = 1 - blocks[idx[0]]
                c5 = evaluate_block_pattern(blocks)
                if c5 < best_c5:
                    best_blocks = blocks.copy()
                    best_c5 = c5
            if np.random.rand() < 0.1:
                idx = np.random.randint(0, m)
                blocks[idx] = 1 - blocks[idx]
                sum_blocks = np.sum(blocks)
                if abs(sum_blocks - m / 2) > 1e-6:
                    idx = np.argsort(np.abs(blocks - 0.5))
                    blocks[idx[0]] = 1 - blocks[idx[0]]
                c5 = evaluate_block_pattern(blocks)
                if c5 < best_c5:
                    best_blocks = blocks.copy()
                    best_c5 = c5
            if best_c5 < 0.3807:
                break
        return best_blocks
    
    start_time = time.time()
    best_blocks = block_optimization()
    best_h_values = np.repeat(best_blocks, n_points // len(best_blocks) + 1)[:n_points]
    best_h_values = np.clip(best_h_values, 0, 1)
    sum_h = np.sum(best_h_values)
    best_h_values += (n_points / 2 - sum_h) / n_points
    best_h_values = np.clip(best_h_values, 0, 1)
    best_c5 = compute_c5(best_h_values)
    
    if best_c5 < 0.3807:
        return (best_h_values, best_c5, n_points)
    
    def optimize_h_values():
        result = minimize(objective, best_h_values, method='L-BFGS-B', options={'maxiter': 500})
        if result.success:
            h_values = result.x
            sum_h = np.sum(h_values)
            h_values += (n_points / 2 - sum_h) / n_points
            h_values = np.clip(h_values, 0, 1)
            return compute_c5(h_values)
        return best_c5
    
    best_c5 = optimize_h_values()
    
    if best_c5 < 0.3807:
        return (best_h_values, best_c5, n_points)
    
    return (best_h_values, best_c5, n_points)
