```python
import numpy as np
import scipy.signal as signal
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 45
    if initial_h_values is not None and len(initial_h_values) == n_points:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    h = np.clip(h, 0, 1)
    best_c5 = float('inf')
    best_h = h.copy()
    dx = 2.0 / n_points
    max_iter = int(budget_s / 1.0)
    start_time = time.time()
    
    momentum_factor = 0.9
    learning_rate = 0.1
    noise_scale = 0.05
    iterations = 0
    v = np.zeros(n_points)
    improve_threshold = 1e-5
    no_improve_count = 0
    
    while time.time() - start_time < budget_s:
        if iterations > max_iter:
            break
        correlate_vals = signal.correlate(h, 1 - h, mode='full') * dx
        worst_lag = np.argmax(correlate_vals)
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * dx
        v = momentum_factor * v + learning_rate * gradients
        h = h - v
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        if random.random() < 0.05:
            h += noise_scale * np.random.normal(0, 1, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        current_c5 = np.max(correlate_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 50:
                break
        iterations += 1
    
    h = best_h
    correlate_vals = signal.correlate(h, 1 - h, mode='full') * dx
    best_c5 = np.max(correlate_vals)
    return (h.tolist(), best_c5, n_points)
```
