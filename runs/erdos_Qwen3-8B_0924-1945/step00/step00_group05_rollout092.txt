```python
import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                h[i] = 1.0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(x):
        h_val = np.clip(x, 0, 1)
        h_val = h_val * (n_points / 2) / np.sum(h_val)
        overlap = np.correlate(h_val, 1 - h_val, mode='full') * dx
        return np.max(overlap)
    
    def gradient(x):
        h_val = np.clip(x, 0, 1)
        h_val = h_val * (n_points / 2) / np.sum(h_val)
        overlap = np.correlate(h_val, 1 - h_val, mode='full') * dx
        k_max = np.argmax(overlap)
        grad = np.zeros_like(h_val)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                grad[i] += (1 - h_val[j]) * dx
                grad[j] -= h_val[i] * dx
        return -grad
    
    def constraint(x):
        return np.sum(np.clip(x, 0, 1)) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint, 'jac': lambda x: np.ones_like(x)}
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    for _ in range(100000):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        res = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, jac=gradient, constraints=cons, tol=1e-5)
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        if np.random.rand() < 0.01:
            noise = np.random.normal(0, 0.05, size=n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = objective(h)
    
    return h, final_c5, n_points
```
