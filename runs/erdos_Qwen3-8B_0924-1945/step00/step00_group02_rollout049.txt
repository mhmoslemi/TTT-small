import numpy as np
import random
import time
import scipy.optimize as opt
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 1000
    initial_h_values = np.array([0.5]*n_points)
    if 'initial_h_values' in kwargs:
        initial_h_values = kwargs['initial_h_values']
        n_points = len(initial_h_values)
    
    def rescale(h):
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total
    
    def clip_and_rescale(h):
        h = np.clip(h, 0, 1)
        h = rescale(h)
        return h
    
    def evaluate_c5(h):
        try:
            corr = correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            c5 = max_corr * dx
            return c5
        except:
            return float('inf')
    
    def mutate(h, mutation_rate=0.1, mutation_strength=0.1):
        indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
        h[indices] += np.random.uniform(-mutation_strength, mutation_strength, size=len(indices))
        return clip_and_rescale(h)
    
    def crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        mask = np.random.rand(n_points) < 0.5
        child[mask] = parent1[mask]
        child[~mask] = parent2[~mask]
        return clip_and_rescale(child)
    
    def tournament_selection(population, k=5):
        selected = np.random.choice(len(population), k, replace=False)
        best = selected[0]
        for i in selected[1:]:
            if evaluate_c5(population[i]) < evaluate_c5(population[best]):
                best = i
        return population[best]
    
    population = [initial_h_values.copy()]
    for _ in range(50):
        new_population = []
        for _ in range(len(population)):
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        population = new_population
        best_index = np.argmin([evaluate_c5(ind) for ind in population])
        best_h = population[best_index]
    
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, n_points)
