import numpy as np
import time
import random
import scipy.signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    h = initial_h_values.copy()
    h = h[:n_points] if len(h) > n_points else np.concatenate([h, np.zeros(n_points - len(h))])
    h = h[:n_points]
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    learning_rate = 0.1
    max_iter = 1000
    noise_scale = 0.05
    p_noise = 0.05
    decay_rate = 0.999
    early_stop = 100
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        correlate_vals = scipy.signal.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_c5 = np.max(correlate_vals)
        
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
        
        if max_c5 < 1e-6:
            break
        
        m = np.argmax(correlate_vals)
        
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = i - m
            if 0 <= j < n_points:
                gradients[i] = (1 - h[j]) * (2.0 / n_points)
        
        h = h - learning_rate * gradients
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        if random.random() < p_noise:
            noise = np.random.normal(0, noise_scale, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        learning_rate *= decay_rate
        
        if iter % early_stop == 0:
            if np.abs(max_c5 - best_c5) < 1e-5:
                break
    
    h = best_h
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    correlate_vals = scipy.signal.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
    c5_bound = np.max(correlate_vals)
    return (h, c5_bound, n_points)
