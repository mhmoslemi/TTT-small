import numpy as np
import scipy.optimize as opt
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial guess
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (target_sum / np.sum(initial_h))
        initial_h = np.clip(initial_h, 0, 1)
    
    def clip_and_scale(h):
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        return h
    
    def evaluate_c5(h):
        h = clip_and_scale(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def objective(h):
        return evaluate_c5(h)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = differential_evolution(
        objective,
        bounds,
        strategy='best2bin',
        popsize=50,
        mutation=(0.5, 1),
        recombination=0.7,
        tol=1e-6,
        maxiter=10000,
        seed=seed,
        disp=False,
        polish=True,
        polish_bounds=bounds,
        callback=None,
        vectorized=True
    )
    
    best_h = result.x
    best_h = clip_and_scale(best_h)
    best_c5 = evaluate_c5(best_h)
    
    return best_h, best_c5, n_points
