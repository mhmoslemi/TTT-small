import numpy as np
import random
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 450
    if initial_h_values is not None:
        h = np.clip(initial_h_values, 0, 1)
        h = (h * n_points / 2.0) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = (h * n_points / 2.0) / np.sum(h)
    
    def objective(x):
        h_flat = x.reshape(n_points)
        h_flat = np.clip(h_flat, 0, 1)
        h_flat = (h_flat * n_points / 2.0) / np.sum(h_flat)
        corr = np.correlate(h_flat, 1 - h_flat, mode="full")
        c5 = np.max(corr) * 2.0 / n_points
        return c5
    
    def constraint(x):
        h_flat = x.reshape(n_points)
        return np.sum(h_flat) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h
    
    for _ in range(600):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        # Global mutation
        noise = np.random.normal(0, 0.1, n_points)
        h_flat = h + noise
        h_flat = np.clip(h_flat, 0, 1)
        h_flat = (h_flat * n_points / 2.0) / np.sum(h_flat)
        c5 = objective(h_flat)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_flat
        # Targeted mutation
        corr = np.correlate(h_flat, 1 - h_flat, mode="full")
        max_idx = np.argmax(corr)
        noise = np.zeros(n_points)
        for i in range(max_idx - 10, max_idx + 11):
            if 0 <= i < n_points:
                noise[i] = np.random.normal(0, 0.05)
        h_flat = h_flat + noise
        h_flat = np.clip(h_flat, 0, 1)
        h_flat = (h_flat * n_points / 2.0) / np.sum(h_flat)
        c5 = objective(h_flat)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_flat
        h = best_h
    
    return best_h, best_c5, n_points
