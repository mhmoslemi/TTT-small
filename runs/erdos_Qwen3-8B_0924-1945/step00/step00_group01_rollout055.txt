import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with initial_h_values if available
    if 'initial_h_values' in globals():
        h_vals = np.array(initial_h_values)
    else:
        # Fallback: create a random binary sequence with sum n/2
        n_points = 47
        h_vals = np.random.choice([0, 1], size=n_points)
        h_vals = np.random.permutation(h_vals)
        sum_h = np.sum(h_vals)
        if sum_h != n_points // 2:
            h_vals[0] += (n_points // 2) - sum_h
        h_vals = h_vals.astype(float)
    
    n_points = len(h_vals)
    dx = 2.0 / n_points
    
    # Normalize to ensure sum is n/2
    if np.sum(h_vals) != n_points / 2:
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    
    best_h = h_vals.copy()
    best_c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode="full")) * dx
    iteration = 0
    max_iterations = 100
    no_improvement = 0
    
    start_time = time.time()
    
    while iteration < max_iterations and time.time() - start_time < budget_s:
        # Compute all lags
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        lag_values = corr * dx
        max_lag_idx = np.argmax(lag_values)
        max_lag_val = lag_values[max_lag_idx]
        
        # Define objective for the worst lag
        def objective(x):
            h_new = x.copy()
            # Clip values [0, 1]
            h_new = np.clip(h_new, 0, 1)
            # Renormalize to ensure sum is n/2
            h_new = h_new / np.sum(h_new) * (n_points / 2)
            # Compute overlap for the worst lag
            corr = np.correlate(h_new, 1 - h_new, mode="full")
            return corr[max_lag_idx] * dx
        
        # Define bounds and equality constraint
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        
        # Optimization
        try:
            result = opt.minimize(
                objective,
                h_vals,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                tol=1e-8,
                options={'maxiter': 100}
            )
        except:
            result = opt.minimize(
                objective,
                h_vals,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                tol=1e-8,
                options={'maxiter': 100}
            )
        
        # Update h_vals
        h_vals = result.x
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
        
        # Update best solution
        current_c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode="full")) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_vals.copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        if no_improvement >= 5:
            break
        
        iteration += 1
    
    # Final checks
    h_vals = best_h
    h_vals = np.clip(h_vals, 0, 1)
    h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
    c5_bound = np.max(np.correlate(h_vals, 1 - h_vals, mode="full")) * dx
    return (h_vals, c5_bound, n_points)
