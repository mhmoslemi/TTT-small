import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with initial_h_values if available, else use a basic construction
    if 'initial_h_values' in globals():
        h_vals = np.array(initial_h_values)
    else:
        n_points = 50
        h_vals = np.zeros(n_points)
        h_vals[:n_points//2] = 1.0
        h_vals = h_vals[:n_points//2 + 1]
        h_vals = np.pad(h_vals, (0, n_points - len(h_vals)), 'constant')
    
    n_points = len(h_vals)
    dx = 2.0 / n_points
    target_mass = n_points / 2
    
    # Normalize to meet the integral constraint
    h_vals = h_vals * target_mass / np.sum(h_vals)
    
    best_h_vals = h_vals.copy()
    best_c5_bound = np.inf
    
    # Precompute the correlation for quick evaluation
    def compute_corr(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr * dx
    
    def update_h_vals(h, dx, target_mass):
        h = np.clip(h, 0, 1)
        mass = np.sum(h)
        if mass == 0:
            return np.zeros_like(h)
        h = h * target_mass / mass
        return h
    
    # Main optimization loop
    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        
        corr = compute_corr(h_vals)
        current_c5 = np.max(corr)
        
        if current_c5 < best_c5_bound:
            best_c5_bound = current_c5
            best_h_vals = h_vals.copy()
        
        if current_c5 < 1e-6:
            break
        
        # Focus on the worst lag
        k_max = np.argmax(corr)
        overlap_k = corr[k_max]
        
        # Formulate the convex surrogate problem
        lambda_k = cp.Variable()
        objective = 0.5 * lambda_k**2 + (overlap_k - lambda_k)**2
        constraints = [lambda_k >= 0, lambda_k >= overlap_k]
        prob = cp.Problem(cp.Minimize(objective), constraints)
        prob.solve()
        
        # Update h_vals using gradient descent on the surrogate
        grad = np.zeros(n_points)
        for i in range(n_points):
            grad[i] = (1 - h_vals[i]) * (1 - 2 * lambda_k.value) * dx
        h_vals -= 0.01 * grad
        
        # Enforce bounds and normalize
        h_vals = update_h_vals(h_vals, dx, target_mass)
    
    # Final evaluation
    corr = compute_corr(best_h_vals)
    best_c5_bound = np.max(corr)
    
    return (best_h_vals, best_c5_bound, n_points)
