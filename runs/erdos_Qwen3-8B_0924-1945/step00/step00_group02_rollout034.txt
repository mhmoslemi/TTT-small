import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 450
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.clip(h, 0, 1)
        h = h * n_points / 2.0 / np.sum(h)
    else:
        h = 0.5 * (1 + np.random.uniform(-1, 1, n_points))
        h = np.clip(h, 0, 1)
        h = h * n_points / 2.0 / np.sum(h)
    h_values = h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    start_time = time.time()
    max_iterations = int(budget_s / 0.1)
    for t in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * 2.0 / n_points
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Targeted mutation
        max_k = np.argmax(corr)
        selected_lags = np.arange(max_k - 10, max_k + 11)
        selected_lags = np.clip(selected_lags, 0, len(corr) - 1)
        selected_indices = np.random.choice(selected_lags, size=10, replace=False)
        for idx in selected_indices:
            weight = corr[idx] / np.max(corr)
            delta = -np.sign(weight) * 0.1 / (1 + t / 500)
            h[idx] += delta
            h[idx] = np.clip(h[idx], 0, 1)
        h = h * n_points / 2.0 / np.sum(h)
        # Global mutation
        indices = np.random.choice(n_points, size=5, replace=False)
        for i in indices:
            h[i] += np.random.normal(0, 0.2 * np.exp(-t / 300))
            h[i] = np.clip(h[i], 0, 1)
        h = h * n_points / 2.0 / np.sum(h)
    h_values = best_h.copy()
    return h_values, best_c5, n_points
