import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Assuming it's preloaded

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        h = h * (n_points / 2) / sum_h
    else:
        h = np.zeros(n_points)
        h[0] = 1.0  # Fallback construction

    best_h = h.copy()
    best_c5 = float('inf')

    start_time = time.time()

    for iteration in range(500):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Compute the overlap integrals
        conv = np.correlate(h, 1 - h, mode="full")
        c5_values = conv * dx
        c5 = np.max(c5_values)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Find the lag with maximum overlap
        max_index = np.argmax(c5_values)

        # Compute the gradient for this lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            shift = max_index
            j = i + shift
            if 0 <= j < n_points:
                grad[i] += (1 - h[j])  # Contribution from h[i]
                grad[j] -= h[i]  # Contribution from h[j]

        # Update the function
        learning_rate = 0.01 * (0.999 ** iteration)
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)

        # Enforce the sum constraint
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (n_points / 2) / sum_h
        else:
            h = np.zeros(n_points)
            h[0] = 1.0

        # Random perturbation with 5% probability
        if np.random.rand() < 0.05:
            noise = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + noise, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                h = h * (n_points / 2) / sum_h
            else:
                h = np.zeros(n_points)
                h[0] = 1.0

    # Final evaluation
    conv = np.correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(conv * dx)
    return (best_h, final_c5, n_points)
