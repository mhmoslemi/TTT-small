import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Assume initial_h_values is available
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    def objective(x):
        h_flat = x.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(conv) * dx
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def evaluate_solution(h_flat):
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        c5 = np.max(conv) * dx
        return h_flat, c5
    
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    try:
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-5,
            options={'maxiter': 100}
        )
        if result.success:
            best_h = result.x
            best_c5 = objective(result.x)
    except:
        pass
    
    if np.isnan(best_c5) or best_c5 >= 0.38085:
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = np.clip(h + perturb, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        conv = np.correlate(perturbed_h, 1 - perturbed_h, mode='full')
        best_c5 = np.max(conv) * dx
        best_h = perturbed_h
    
    if np.isnan(best_c5):
        best_h = h
        conv = np.correlate(h, 1 - h, mode='full')
        best_c5 = np.max(conv) * dx
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return best_h, best_c5, n_points
