import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in globals() else np.random.rand(70)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Normalize initial_h_values
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (target_sum / np.sum(initial_h_values))

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    def constraint_sum(h):
        return np.sum(h) - target_sum

    # Initial guess
    h_initial = np.copy(initial_h_values)
    h_initial = h_initial * (target_sum / np.sum(h_initial))

    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]

    # Use SLSQP for constrained optimization
    result = optimize.minimize(
        objective,
        h_initial,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        method='SLSQP',
        options={'ftol': 1e-9, 'maxiter': 100}
    )

    # If optimization fails, fall back to initial values
    if result.success:
        h_opt = result.x
    else:
        h_opt = initial_h_values

    # Final normalization
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (target_sum / np.sum(h_opt))

    # Compute the final C5 value
    corr = np.correlate(h_opt, 1 - h_opt, mode='full') * dx
    c5_bound = np.max(corr)

    return (h_opt, c5_bound, n_points)
