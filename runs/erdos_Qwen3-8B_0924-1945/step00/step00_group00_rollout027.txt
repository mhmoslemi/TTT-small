```python
import numpy as np
import scipy.optimize
import time
import random

initial_h_values = None  # Assume provided by the environment

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 100  # Start with n_points=100 for better resolution
    h = np.zeros(n_points)
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    
    max_iter = 1000
    learning_rate = 0.1
    decay_rate = 0.999
    noise_prob = 0.05
    noise_scale = 0.05
    early_stop = 100
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Compute correlation
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        current_c5 = np.max(correlate_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Find max lag
        max_idx = np.argmax(correlate_vals)
        # Compute gradient for max lag
        m = max_idx - n_points + 1  # Lag value
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = i - m
            if 0 <= j < n_points:
                gradients[i] = (1 - h[j]) * dx
        # Update h
        h = h - learning_rate * gradients
        # Clip and normalize
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        # Add noise
        if random.random() < noise_prob:
            noise = np.random.normal(0, noise_scale, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        # Decay learning rate
        learning_rate *= decay_rate
        # Early stopping
        if iter > 0 and abs(current_c5 - best_c5) < 1e-5:
            early_stop -= 1
            if early_stop <= 0:
                break
    
    # Fallback to initial if no improvement
    if best_c5 == float('inf'):
        best_c5 = np.max(np.correlate(h, 1 - h, mode="full") * dx)
        best_h = h.copy()
    
    return (best_h, best_c5, n_points)
```
