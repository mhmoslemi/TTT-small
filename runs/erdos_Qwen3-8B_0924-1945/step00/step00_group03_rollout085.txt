import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    max_iterations = 500
    learning_rate = 1e-2
    perturbation_prob = 0.05
    tolerance = 1e-6
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
        
        cross_corr = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            if k >= n_points:
                k_wrap = k - n_points
                cross_corr[k] = dx * np.sum(h[:n_points - k_wrap] * h[k_wrap:])
            else:
                cross_corr[k] = dx * np.sum(h[:n_points - k] * h[k:])
        current_max = np.max(cross_corr)
        max_k = np.argmax(cross_corr)
        
        grad = np.zeros(n_points)
        if max_k >= n_points:
            k_wrap = max_k - n_points
            for i in range(n_points - k_wrap):
                grad[i] += dx * h[i + k_wrap]
                grad[i + k_wrap] += dx * h[i]
        else:
            for i in range(n_points - max_k):
                grad[i] += dx * h[i + max_k]
                grad[i + max_k] += dx * h[i]
        
        direction = -grad
        new_h = h + learning_rate * direction
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        if new_sum != 0:
            scaling_factor = (n_points / 2) / new_sum
            new_h = new_h * scaling_factor
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        if new_sum != 0:
            scaling_factor = (n_points / 2) / new_sum
            new_h = new_h * scaling_factor
        
        if np.random.rand() < perturbation_prob:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            new_h = h + noise
            new_h = np.clip(new_h, 0.0, 1.0)
            new_sum = np.sum(new_h)
            if new_sum != 0:
                scaling_factor = (n_points / 2) / new_sum
                new_h = new_h * scaling_factor
            new_h = np.clip(new_h, 0.0, 1.0)
        
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            h = new_h.copy()
            best_c5 = new_c5
            best_h = h.copy()
        
        if np.abs(new_c5 - best_c5) < tolerance:
            break
    
    h = best_h
    h = np.clip(h, 0.0, 1.0)
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (n_points / 2) / h_sum
    h = np.clip(h, 0.0, 1.0)
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (n_points / 2) / h_sum
    
    assert (h >= 0).all() and (h <= 1).all()
    assert np.isclose(np.sum(h), n_points / 2, atol=1e-6)
    final_c5 = evaluate_erdos_solution(h, dx)
    assert np.isfinite(final_c5)
    
    return h, final_c5, n_points
