import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 47
    if 'initial_h_values' in globals():
        h_vals = np.array(initial_h_values, dtype=float)
    else:
        h_vals = np.random.rand(n_points)
    n = n_points
    dx = 2.0 / n
    sum_h = n / 2
    h_vals = (h_vals * sum_h) / np.sum(h_vals)
    
    best_h = h_vals.copy()
    best_c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * dx
    
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time > budget_s - 100:
            break
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5_values = corr * dx
        max_c5_idx = np.argmax(c5_values)
        k = max_c5_idx - n + 1
        
        # Create a function to compute overlap for a specific lag
        def objective(x):
            x = np.clip(x, 0, 1)
            x_sum = np.sum(x)
            if np.isclose(x_sum, sum_h):
                return c5_values[max_c5_idx]
            else:
                return np.sum(x * (1 - np.roll(x, k)))
        
        def constraint_sum(x):
            return np.sum(x) - sum_h
        
        bounds = [(0, 1) for _ in range(n)]
        res = opt.minimize(
            objective, 
            h_vals, 
            method='SLSQP', 
            bounds=bounds, 
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-8, 'maxiter': 50}
        )
        
        if res.success:
            h_vals = res.x
        else:
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = (h_vals * sum_h) / np.sum(h_vals)
        
        current_c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_vals.copy()
        elif np.abs(current_c5 - best_c5) < 1e-6:
            break
    
    h_vals = best_h
    c5_bound = best_c5
    return h_vals, c5_bound, n_points
