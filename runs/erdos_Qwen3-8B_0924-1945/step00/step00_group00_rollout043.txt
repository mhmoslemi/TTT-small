import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in kwargs and kwargs['initial_h_values'] is not None:
        h = np.array(kwargs['initial_h_values'], dtype=float)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    for _ in range(500):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        worst_lag = np.argmax(correlate_vals)
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * dx
        
        learning_rate = max(0.001, 0.05 * (1 - _ / 500))
        h -= learning_rate * gradients
        
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        if np.random.rand() < 0.05:
            noise_scale = 0.05
            h += noise_scale * np.random.normal(0, 1, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        current_c5 = np.max(np.correlate(h, 1 - h, mode="full") * dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return (best_h, best_c5, n_points)
