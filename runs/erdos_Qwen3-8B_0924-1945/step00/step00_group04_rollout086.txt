        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(objective, h, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method='SLSQP', tol=1e-6, options={'maxiter': 100})

    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if np.abs(sum_h - target_sum) > 1e-6:
        best_h = best_h * target_sum / sum_h
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
