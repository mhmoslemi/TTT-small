import numpy as np
import scipy
import random
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else None
    n_points = 500
    if initial_h_values is not None:
        initial_h_values = initial_h_values[:n_points]
    else:
        # Create a baseline if no initial is given
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                h[i] = 1.0
            else:
                h[i] = 0.0
        initial_h_values = h
    initial_h_values = initial_h_values[:n_points]
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_sum = np.sum(initial_h_values)
    scale_factor = (n_points / 2) / initial_sum if initial_sum != 0 else 1.0
    initial_h_values *= scale_factor
    initial_h_values = np.clip(initial_h_values, 0, 1)
    
    population_size = 50
    population = [initial_h_values.copy() for _ in range(population_size)]
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h, n_points)
    start_time = time.time()
    
    def compute_c5(h, n_points):
        dx = 2.0 / n_points
        h_corr = correlate(h, 1 - h, mode="full")
        c5 = np.max(h_corr) * dx
        return c5
    
    def evaluate_individual(h):
        return compute_c5(h, n_points)
    
    def clip_and_rescale(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        scale_factor = (n_points / 2) / total
        h *= scale_factor
        return h
    
    for generation in range(500):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        # Evaluate all individuals
        fitness = [evaluate_individual(ind) for ind in population]
        # Select top 20% for crossover
        top_indices = np.argsort(fitness)[:int(population_size * 0.2)]
        top_population = [population[i] for i in top_indices]
        # Crossover: blend top individuals
        offspring = []
        for i in range(len(top_population) // 2):
            pa = top_population[i]
            pb = top_population[i + 1]
            child = (pa + pb) / 2
            child = clip_and_rescale(child)
            offspring.append(child)
        # Mutation: perturb 10% of all individuals
        for ind in population:
            if np.random.random() < 0.1:
                indices = np.random.choice(n_points, int(n_points * 0.1), replace=False)
                delta = np.random.uniform(-0.1, 0.1, size=len(indices))
                ind[indices] += delta
                ind = clip_and_rescale(ind)
        # Replace old population with new offspring and mutation results
        population = offspring + [ind for ind in population if np.random.random() < 0.5]
        # Update best
        for ind in population:
            current_c5 = evaluate_individual(ind)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = ind.copy()
    
    # Final check
    best_h = clip_and_rescale(best_h)
    return (best_h, best_c5, n_points)
