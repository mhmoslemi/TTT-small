import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 450
    initial_h_values = np.random.rand(n_points)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values /= np.sum(initial_h_values)
    initial_h_values *= n_points / 2
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    dx = 2.0 / n_points
    t_start = time.time()
    dt = 0.1
    max_iter = int((budget_s - (time.time() - t_start)) / dt)
    
    for t in range(max_iter):
        if time.time() - t_start > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        # Targeted mutation
        h_targeted = h.copy()
        lags = np.arange(max_k - 10, max_k + 11)
        lags = np.clip(lags, 0, 2 * n_points - 1)
        for lag in lags:
            contribution = get_contribution(h, lag, n_points)
            delta = -np.sign(contribution) * 0.1 / (1 + t / 500)
            h_targeted[lag % n_points] += delta
            h_targeted[lag % n_points] = np.clip(h_targeted[lag % n_points], 0, 1)
        h_targeted = rescale(h_targeted, n_points)
        c5_targeted = compute_c5(h_targeted, n_points)
        
        # Global mutation
        h_global = h.copy()
        indices = np.random.choice(n_points, 5, replace=False)
        for i in indices:
            noise = np.random.normal(0, 0.2 * np.exp(-t / 300), 1)
            h_global[i] += noise
            h_global[i] = np.clip(h_global[i], 0, 1)
        h_global = rescale(h_global, n_points)
        c5_global = compute_c5(h_global, n_points)
        
        # Select better candidate
        if c5_targeted < best_c5:
            h = h_targeted.copy()
            best_c5 = c5_targeted
        if c5_global < best_c5:
            h = h_global.copy()
            best_c5 = c5_global
        
        if best_c5 < 0.3807 and t % 100 == 0:
            best_h = h.copy()
    
    return best_h, best_c5, n_points

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    c5 = max_corr * 2.0 / n_points
    return c5

def get_contribution(h, lag, n_points):
    idx = lag % n_points
    if idx == 0:
        return 0
    return h[idx - 1] * (1 - h[idx])

def rescale(h, n_points):
    total = np.sum(h)
    if abs(total) < 1e-6:
        return np.array([0.5] * n_points)
    scale = (n_points / 2) / total
    return np.clip(h * scale, 0, 1)
