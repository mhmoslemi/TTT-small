import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5 if i % 4 == 0 else 0 if i % 3 == 0 else 1 for i in range(n_points)])
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h if sum_h > 0 else np.zeros(n_points)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    learning_rate = 0.1
    iteration = 0
    start_time = time.time()

    while time.time() - start_time < budget_s:
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        k_max = np.argmax(overlaps)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (n_points / 2) / sum_h
        else:
            h = np.zeros(n_points)
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            iteration = 0
        else:
            iteration += 1
            if iteration > 200:
                learning_rate *= 0.95
                iteration = 0
        if time.time() - start_time >= budget_s:
            break

    h = best_h
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h if sum_h > 0 else np.zeros(n_points)
    final_c5 = evaluate_erdos_solution(h)
    return (h, final_c5, n_points)
