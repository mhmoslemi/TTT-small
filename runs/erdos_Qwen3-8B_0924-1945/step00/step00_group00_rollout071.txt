import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial guess from provided or default
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = (h * n_points / 2) / np.sum(h)
    
    # Parameters
    momentum = 0.9
    learning_rate = 0.1
    decay_rate = 0.999
    noise_scale = 0.05
    noise_freq = 100
    max_iter = 500
    early_stop = 50
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    for t in range(max_iter):
        # Compute overlap integral
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        current_c5 = np.max(correlate_vals)
        
        # Early stopping
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        elif t > 0 and current_c5 >= best_c5 - 1e-5 and t > early_stop:
            break
        
        # Check time budget
        if time.time() - start_time > budget_s:
            break
        
        # Select worst lag
        worst_lag = np.argmax(correlate_vals)
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * (2.0 / n_points)
        
        # Update momentum
        if t == 0:
            v = np.zeros(n_points)
        else:
            v = momentum * v + learning_rate * gradients
        
        # Update h
        h = h - v
        
        # Clip and normalize
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Add noise
        if t % noise_freq == 0:
            noise = noise_scale * np.random.normal(0, 1, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Decay learning rate
        learning_rate *= decay_rate
    
    return (best_h, best_c5, n_points)
