import numpy as np
import scipy.optimize as opt
import time
import copy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    h = np.copy(initial_h_values)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    t_total = int(budget_s * 0.75)
    t_remaining = t_total
    
    T0 = 0.7 * (1.0 + best_c5) / 3.0
    Tmax = int(0.1 * t_total)
    beta = 0.75
    gamma = 0.65
    lambda0 = 0.15 / dx
    
    for t in range(t_total):
        if t_remaining <= 0:
            break
        if t % 100 == 0:
            lambda0 *= 0.8 if t < t_total // 2 else 0.95
        
        # Perturbation
        noise = np.random.normal(0, lambda0, size=n_points)
        perturbed = h + noise
        perturbed = np.clip(perturbed, 0.0, 1.0)
        
        # Constraint restoration
        current_sum = np.sum(perturbed)
        delta = (n_points / 2 - current_sum) / (n_points * 0.95)
        perturbed += delta * np.random.choice([-1, 1], size=n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        if np.sum(perturbed) != 0:
            perturbed *= (n_points / 2) / np.sum(perturbed)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(perturbed, dx)
        
        # Acceptance criteria
        if new_c5 < best_c5 or (np.random.rand() < np.exp(-(new_c5 - best_c5) / T0 * (1 - t / Tmax) ** (1 / beta))):
            h = perturbed.copy()
            best_h = h.copy()
            best_c5 = new_c5
            t_remaining -= 1
        
        t_remaining -= 1
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (best_h, final_c5, n_points)
