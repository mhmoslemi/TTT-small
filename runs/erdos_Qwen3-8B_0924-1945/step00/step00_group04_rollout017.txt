import numpy as np
import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scaling = target_sum / current_sum
        h = h * scaling
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        scaling = target_sum / current_sum
        return h * scaling

    best_h = repair(initial_h)
    best_c5 = evaluate_erdos_solution(best_h)

    start_time = time.time()

    def evaluate(h):
        h = repair(h)
        cross_corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(cross_corr * dx)
        return c5

    def fitness(h):
        return -evaluate(h)

    def local_refine(h):
        h = h.copy()
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.minimize(fitness, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        return result.x

    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        population = [np.clip(best_h + np.random.normal(0, 0.05, n_points), 0, 1) for _ in range(100)]
        population_fitness = [evaluate(h) for h in population]
        top_indices = np.argsort(population_fitness)[-50:]
        top_population = [population[i] for i in top_indices]
        new_population = []
        for i in range(0, len(top_population), 2):
            if i + 1 < len(top_population):
                parent1 = top_population[i]
                parent2 = top_population[i + 1]
                child = np.zeros(n_points)
                for j in range(n_points):
                    if np.random.rand() < 0.5:
                        child[j] = parent1[j]
                    else:
                        child[j] = parent2[j]
                new_population.append(child)
            new_population.append(top_population[i])
        population = new_population
        mutated_population = []
        for h in population:
            mutation = np.random.normal(0, 0.05, n_points)
            mutated = h + mutation
            mutated = np.clip(mutated, 0, 1)
            mutated = repair(mutated)
            mutated_population.append(mutated)
        population = mutated_population
        for h in population:
            current_c5 = evaluate(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5
        best_h = local_refine(best_h)

    best_h = repair(best_h)
    assert np.all(0 <= best_h <= 1)
    assert np.isclose(np.sum(best_h), target_sum, atol=1e-6)
    best_c5 = evaluate(best_h)
    return (best_h, best_c5, n_points)
