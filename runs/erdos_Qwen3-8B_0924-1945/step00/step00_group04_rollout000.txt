import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    # Determine number of points based on the initial data
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Random seed for reproducibility
    np.random.seed(seed)

    # Objective function to minimize the maximum overlap
    def obj_func(h_values):
        h = np.array(h_values, dtype=np.float64)
        cross_corr = np.correlate(h, 1 - h, mode="full")
        c5_value = np.max(cross_corr) * dx
        return c5_value

    # Constraint: sum of h_values must be target_sum
    def constraint_func(h_values):
        return np.sum(h_values) - target_sum

    # Bounds for each h value (0 <= h[i] <= 1)
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial guess
    initial_guess = initial_h_values.copy()

    # Optimization settings
    max_iter = 100
    tol = 1e-6
    options = {'maxiter': max_iter, 'ftol': tol, 'disp': False}

    # Optimization using SLSQP
    result = opt.minimize(
        obj_func,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_func},
        options=options
    )

    # Check if optimization succeeded
    if result.success:
        best_h_values = result.x
        best_c5 = result.fun
    else:
        # Fallback to initial guess if optimization failed
        best_h_values = initial_h_values
        best_c5 = obj_func(best_h_values)

    # Ensure the sum of h_values is correct
    assert np.isclose(np.sum(best_h_values), target_sum), "Sum constraint not satisfied"

    # Verify all values are within [0, 1]
    assert np.all((best_h_values >= 0) & (best_h_values <= 1)), "Values out of bounds"

    # Return the best solution found
    return (best_h_values, best_c5, n_points)
