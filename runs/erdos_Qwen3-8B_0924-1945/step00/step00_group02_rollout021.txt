import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 450
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = 0.5 * (1 + np.random.uniform(-1, 1, n_points))
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * 2.0 / n_points
        return c5
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_time = time.time()
    for t in range(500):
        elapsed = time.time() - best_time
        if elapsed > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        # Targeted mutation
        h_new = h.copy()
        for i in range(max_k - 10, max_k + 11):
            if 0 <= i < len(corr):
                contrib = corr[i] * h[i]
                delta = -np.sign(contrib) * 0.1 / (1 + t / 500)
                h_new[i] += delta
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Global mutation
        h_new = h.copy()
        indices = np.random.choice(len(h), 5, replace=False)
        for i in indices:
            delta = np.random.normal(0, 0.2 * np.exp(-t / 300))
            h_new[i] += delta
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = best_h.copy()
    
    return (best_h, best_c5, n_points)
