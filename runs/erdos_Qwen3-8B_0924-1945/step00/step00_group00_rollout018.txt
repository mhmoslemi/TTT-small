import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    
    # Default parameters
    n_points = 100
    max_iterations = 1000
    learning_rate = 0.1
    noise_scale = 0.05
    noise_prob = 0.05
    early_stop_patience = 100
    early_stop_threshold = 1e-5
    
    # Start with initial guess
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    for iteration in range(max_iterations):
        # Compute the correlation
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(correlate_vals)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Early stopping check
        if iteration > 0 and abs(best_c5 - previous_best_c5) < early_stop_threshold:
            if iteration - last_improvement > early_stop_patience:
                break
        previous_best_c5 = best_c5
        last_improvement = iteration
        
        # Find the lag with maximum overlap
        max_index = np.argmax(correlate_vals)
        lag = max_index - (n_points - 1)
        
        # Compute gradient for this lag
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = i - lag
            if 0 <= j < n_points:
                gradients[i] = (1 - h[j]) * (2.0 / n_points)
        
        # Update h
        h -= learning_rate * gradients
        
        # Clip and normalize
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Add noise with probability
        if np.random.rand() < noise_prob:
            noise = np.random.normal(0, noise_scale, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Final check for best_c5
    correlate_vals = np.correlate(best_h, 1 - best_h, mode="full") * (2.0 / n_points)
    final_c5 = np.max(correlate_vals)
    
    return (best_h, final_c5, n_points)
