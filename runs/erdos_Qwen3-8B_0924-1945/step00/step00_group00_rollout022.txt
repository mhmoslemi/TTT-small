```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 45
    if initial_h_values is None:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(correlate_vals)
    
    def update_h(h, momentum_factor, learning_rate, noise_scale):
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        worst_lag = np.argmax(correlate_vals)
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * (2.0 / n_points)
        v = momentum_factor * v + learning_rate * gradients
        h = h - v
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h += noise_scale * np.random.normal(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    v = np.zeros(n_points)
    momentum_factor = 0.9
    learning_rate = 0.1
    noise_scale = 0.05
    noise_frequency = 100
    iterations = 0
    while time.time() - start_time < budget_s:
        h = update_h(h, momentum_factor, learning_rate, noise_scale)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        iterations += 1
        if iterations % noise_frequency == 0:
            noise_scale = max(0.01, noise_scale * 0.9)
        if iterations > 500:
            break
    
    return (best_h, best_c5, n_points)
```
