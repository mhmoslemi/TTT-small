```python
import numpy as np
import scipy.optimize as optimize
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    max_iterations = 100
    improvement_threshold = 1e-4
    no_improvement_limit = 5
    dx = 2.0 / n_points

    if initial_h_values is not None:
        h_vals = np.array(initial_h_values, dtype=np.float64)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    else:
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)

    def compute_correlation(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr * dx

    def evaluate_c5(h):
        corr = compute_correlation(h)
        return np.max(corr)

    def optimize_lag(h, lag):
        def objective(x):
            x = np.clip(x, 0, 1)
            x = x / np.sum(x) * (n_points / 2)
            return compute_correlation(x)[lag]

        bounds = [(0, 1) for _ in range(n_points)]
        constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        result = optimize.minimize(objective, h, bounds=bounds, constraints=constraints, method='L-BFGS-B')
        if result.success:
            return result.x
        else:
            print("Optimization failed for lag", lag)
            return h

    start_time = time.time()
    best_h = h_vals.copy()
    best_c5 = evaluate_c5(best_h)
    no_improvement_count = 0

    for _ in range(max_iterations):
        corr = compute_correlation(best_h)
        max_lag = np.argmax(corr)
        new_h = optimize_lag(best_h, max_lag)
        new_c5 = evaluate_c5(new_h)
        if new_c5 < best_c5 - improvement_threshold:
            best_h = new_h
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count >= no_improvement_limit:
                break
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s - 100:
            break

    print(f"Final C5 value: {best_c5}")
    return (best_h, best_c5, n_points)
```
