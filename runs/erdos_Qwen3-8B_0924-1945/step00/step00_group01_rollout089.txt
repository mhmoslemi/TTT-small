import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = globals().get('initial_h_values', None)
    
    if initial_h_values is not None:
        h_vals = np.array(initial_h_values, dtype=float)
    else:
        n_points = 100
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * n_points / 2) / np.sum(h_vals)
    
    n_points = len(h_vals)
    dx = 2.0 / n_points
    max_iterations = 100
    improvement_threshold = 1e-5
    early_stop_counter = 0
    best_h_vals = h_vals.copy()
    best_c5 = np.inf
    
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 100:
            break
        
        # Compute all lags
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5_values = corr * dx
        max_c5 = np.max(c5_values)
        
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h_vals = h_vals.copy()
            early_stop_counter = 0
        else:
            early_stop_counter += 1
            if early_stop_counter >= 5:
                break
        
        # Find the lag with the maximum overlap
        max_idx = np.argmax(c5_values)
        if max_idx >= len(corr):
            max_idx = len(corr) - 1
        
        # Define the objective function for the selected lag
        def objective(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            h = h * (n_points / 2) / total
            corr = np.correlate(h, 1 - h, mode='full')
            return corr[max_idx] * dx
        
        # Constraints: sum = n_points / 2
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        
        # Bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimization
        result = opt.minimize(
            objective,
            h_vals,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'eps': 1e-6, 'disp': False}
        )
        
        if result.success:
            h_vals = result.x
        else:
            h_vals = best_h_vals.copy()
        
        # Enforce bounds and normalization
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
    
    # Final evaluation
    corr = np.correlate(best_h_vals, 1 - best_h_vals, mode='full')
    final_c5 = np.max(corr * dx)
    
    return (best_h_vals, final_c5, n_points)
