import numpy as np
import random
import time
from scipy.signal import correlate
from copy import deepcopy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 500  # balance between accuracy and speed
    dx = 2.0 / n_points
    initial_h = initial_h_values if 'initial_h_values' in kwargs else np.random.rand(n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    h_population = [deepcopy(initial_h) for _ in range(50)]
    best_h = deepcopy(initial_h)
    best_c5 = np.inf
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    for gen in range(1000):
        if time.time() - start_time > budget_s:
            break
        candidates = []
        for h in h_population:
            candidates.append(evaluate(h))
        selected = np.argsort(candidates)[:int(len(candidates) * 0.2)]
        top_h = [h_population[i] for i in selected]
        
        # Crossover
        if len(top_h) >= 2:
            for i in range(int(len(top_h) * 0.5)):
                h1, h2 = random.sample(top_h, 2)
                child = (h1 + h2) / 2
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                candidates.append(child)
        
        # Mutation
        for h in h_population:
            if random.random() < 0.1:
                indices = np.random.choice(n_points, 10, replace=False)
                h[indices] += np.random.uniform(-0.1, 0.1, size=10)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                candidates.append(h)
        
        # Replace population
        h_population = candidates[:50]
        current_best = min(h_population, key=evaluate)
        current_c5 = evaluate(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
    
    return (best_h, best_c5, n_points)
