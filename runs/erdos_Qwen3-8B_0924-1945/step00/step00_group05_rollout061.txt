    learning_rate = 0.01
    decay_rate = 0.999
    iteration = 0
    start_time = time.time()
    while time.time() - start_time < budget_s:
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        k_max = np.argmax(overlaps)

            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)

            h = h * (n_points / 2) / total
