import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n_points)
    start_time = time.time()
    for t in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        corr = corr * (2.0 / n_points)
        max_corr = np.max(corr)
        if max_corr < best_c5:
            best_c5 = max_corr
            best_h = h.copy()
        worst_lag = np.argmax(corr)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            grad[i] = (1 - h[j]) * (2.0 / n_points)
        h = h - 0.1 * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        if t % 100 == 0 and t > 0:
            h += 0.05 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    return best_h, best_c5, n_points
