import numpy as np
import scipy.optimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial construction
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    def evaluate(h):
        dx = 2.0 / n_points
        diff = 1 - h
        correlate_vals = np.correlate(h, diff, mode="full") * dx
        return np.max(correlate_vals)
    
    def compute_gradients(h, lag):
        gradients = np.zeros_like(h)
        for i in range(n_points):
            j = i - lag
            if 0 <= j < n_points:
                gradients[i] = (1 - h[j]) * (2.0 / n_points)
        return gradients
    
    best_h = h.copy()
    best_c5 = evaluate(best_h)
    best_time = time.time()
    
    max_iter = 1000
    learning_rate = 0.1
    decay_rate = 0.999
    noise_prob = 0.05
    noise_scale = 0.05
    
    for iter in range(max_iter):
        if time.time() - best_time > budget_s:
            break
        
        c5 = evaluate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_time = time.time()
        
        # Find lag with maximum correlation
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_lag = np.argmax(correlate_vals)
        
        # Compute gradient
        gradients = compute_gradients(h, max_lag)
        
        # Update with learning rate
        h = h - learning_rate * gradients
        
        # Clip and normalize
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Add noise
        if random.random() < noise_prob:
            noise = np.random.normal(0, noise_scale, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        learning_rate *= decay_rate
    
    h = best_h
    c5 = evaluate(h)
    return h, c5, n_points
