import numpy as np
import scipy.optimize as optimize
import time

initial_h_values = None  # This will be provided by the environment

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use 100 points for balance between speed and accuracy
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[0:35] = 1.0
        h[35:60] = 0.0
        h[60:] = 1.0
    # Normalize to meet the integral constraint
    h /= (np.sum(h) * dx)
    h *= n_points / 2.0 / dx
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    # Optimization loop
    for _ in range(300):  # 300 iterations
        if time.time() - start_time > budget_s:
            break
        # Perturb the current solution
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        new_h = h + perturbation
        # Clip to [0, 1]
        new_h = np.clip(new_h, 0.0, 1.0)
        # Scale to meet the integral constraint
        scale = (n_points / 2.0 / dx) / np.sum(new_h) * dx
        new_h *= scale
        # Compute C5
        current_c5 = compute_c5(new_h)
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = new_h.copy()
        h = new_h
    # Final evaluation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)

def compute_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h_array = np.array(h, dtype=np.float64)
    one_minus_h_array = 1.0 - h_array
    correlation = np.correlate(h_array, one_minus_h_array, mode="full")
    # The max overlap is the maximum value of the correlation, scaled by dx
    return np.max(correlation) * dx
