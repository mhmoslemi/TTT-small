import numpy as np
import cvxpy as cp
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    best_solution = np.array(initial_h_values, dtype=np.float64)
    best_c5 = evaluate_erdos_solution(best_solution, dx)
    best_c5 = best_c5[0]  # Take the first value as the max

    # Ensure constraints are met for initial solution
    if not (best_solution >= 0).all() or not (best_solution <= 1).all() or not np.isclose(np.sum(best_solution), n_points / 2, atol=1e-6):
        best_solution = np.full(n_points, 0.5)
        best_c5 = evaluate_erdos_solution(best_solution, dx)[0]

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = corr * dx
        return max(c5)

    def optimize_h(h_initial, lambda_penalty=1e-3, max_iter=50, tol=1e-5):
        h = cp.Variable(n_points)
        t = cp.Variable()
        constraints = [t >= 0]
        for k in range(1, n_points + 1):
            corr = np.zeros(n_points)
            for i in range(n_points):
                j = (i + k) % n_points
                corr[i] = h[i] * (1 - h[j])
            corr *= dx
            constraints.append(t >= cp.sum(corr))

        obj = t + lambda_penalty * cp.sum((h - 0.5) ** 2)
        prob = cp.Problem(cp.Minimize(obj), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False, eps=tol)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_opt = np.round(np.clip(h.value, 0, 1), 3)
                h_opt = h_opt / np.sum(h_opt) * n_points / 2
                return h_opt
        except:
            pass
        return h_initial

    # Main optimization loop
    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        h_opt = optimize_h(best_solution)
        current_c5 = evaluate(h_opt)
        if current_c5 < best_c5:
            best_solution = h_opt
            best_c5 = current_c5
        # Perturb solution with small random noise
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturb = np.clip(best_solution + perturb, 0, 1)
        h_perturb = h_perturb / np.sum(h_perturb) * n_points / 2
        current_c5_perturb = evaluate(h_perturb)
        if current_c5_perturb < best_c5:
            best_solution = h_perturb
            best_c5 = current_c5_perturb

    # Final check and cleanup
    best_solution = np.clip(best_solution, 0, 1)
    best_solution = best_solution / np.sum(best_solution) * n_points / 2
    final_c5 = evaluate(best_solution)
    return (best_solution, final_c5, n_points)
