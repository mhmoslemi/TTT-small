import numpy as np
import scipy.optimize as optimize
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = (h - np.mean(h)) * 2 + 0.5
        h = np.clip(h, 0, 1)
        h = h * n_points / 2.0
        h = h / np.sum(h) * n_points / 2.0

    h = np.clip(h, 0, 1)
    h = h * n_points / 2.0
    h = h / np.sum(h) * n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * n_points / 2.0

    best_c5 = np.inf
    best_h = h.copy()
    T = 100.0
    alpha = 0.995
    start_time = time.time()

    for t in range(int(budget_s / 0.1)):
        if time.time() - start_time > budget_s:
            break
        idxs = np.random.choice(n_points, 5, replace=False)
        noise = np.random.normal(0, 0.05 * np.exp(-t / 200), 5)
        new_h = h.copy()
        new_h[idxs] += noise
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * n_points / 2.0 / np.sum(new_h)
        new_h = np.clip(new_h, 0, 1)

        corr = np.correlate(new_h, 1 - new_h, mode="full")
        c5 = np.max(corr) * dx
        delta = c5 - best_c5

        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = new_h.copy()
            best_c5 = c5
            if best_c5 < 0.3807:
                best_h = h.copy()

        if T < 0.01:
            break
        T *= alpha

    return best_h, best_c5, n_points
