def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = np.inf
    elapsed = 0

    def objective(x):
        nonlocal best_c5
        h_vals = x
        conv = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(conv) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_vals.copy()
        return c5

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    result = opt.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5, options={'maxiter': 100})
    if result.success:
        h = result.x
    else:
        h = initial_h_values.copy()

    h = np.clip(h, 0, 1)
    if np.sum(h) < n_points / 2 - 1e-5:
        h[0] += (n_points / 2 - np.sum(h))
    conv = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(conv) * dx
    return (h, c5, n_points)
