import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.array([0.814, 0.661, 0.894, 0.552, 0.603, 0.503, 0.567, 0.840, 0.868, 0.905,
                                 0.609, 0.682, 0.364, 0.688, 0.486, 0.822, 0.452, 0.827, 0.658, 0.882,
                                 0.697, 0.314, 0.562, 0.497, 0.605, 0.645, 0.504, 0.786, 0.726, 0.784,
                                 0.671, 0.547, 0.651, 0.608, 0.727, 0.489, 0.546, 0.752, 0.333, 0.532,
                                 0.454, 0.695, 0.429, 0.560, 0.828, 0.617, 0.686, 0.713, 0.790, 0.735,
                                 0.675, 0.615, 0.418, 0.680, 0.392, 0.637, 0.653, 0.873, 0.719, 0.649,
                                 0.636, 0.572, 0.468, 0.486, 0.679, 0.525, 0.517, 0.693, 0.666, 0.666])
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def objective(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(objective, initial_h_values, method='SLSQP',
                          bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum},
                          options={'maxiter': 100, 'ftol': 1e-5, 'eps': 1e-5})
    
    if result.success:
        h = np.clip(result.x, 0, 1)
        c5_bound = objective(h)
    else:
        h = initial_h_values.copy()
        c5_bound = objective(h)

        # Perturb initial solution
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = np.clip(initial_h_values + perturb, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        c5_perturbed = objective(h_perturbed)
        if c5_perturbed < c5_bound:
            h = h_perturbed
            c5_bound = c5_perturbed

        # Try Nelder-Mead as a fallback
        result_nm = opt.minimize(objective, h, method='Nelder-Mead',
                                 bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum},
                                 options={'maxiter': 100, 'tol': 1e-5})
        if result_nm.success:
            h = np.clip(result_nm.x, 0, 1)
            c5_bound = objective(h)

    return h, c5_bound, n_points
