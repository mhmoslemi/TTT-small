import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

    h = initial_h_values.copy()
    h = h / np.sum(h) * n_points / 2
    best_result = h.copy()
    best_c5 = evaluate_erdos_solution(best_result, dx)
    
    time_start = time.time()
    
    for _ in range(50):
        if time.time() - time_start > budget_s - 1:
            break
        
        # Perturb the solution
        noise = np.random.normal(0, 0.05, n_points)
        h = np.clip(h + noise, 0, 1)
        h = h * n_points / 2 / np.sum(h)
        
        # Build the shifted h for all k
        shifted_h = np.zeros((n_points, n_points))
        for i in range(n_points):
            for k in range(n_points):
                idx = (i + k) % n_points
                shifted_h[i, k] = h[idx]
        
        # Set up the convex optimization problem
        h_var = cp.Variable(n_points, name='h')
        t = cp.Variable(name='t')
        
        objective = t + 1e-3 * cp.sum_squares(h_var - 0.5)
        constraints = [t >= cp.sum(h_var * (1 - shifted_h[:, k]) * dx) for k in range(n_points)]
        constraints += [0 <= h_var, h_var <= 1]
        constraints += [cp.sum(h_var) == n_points / 2]
        
        problem = cp.Problem(cp.Minimize(objective), constraints)
        try:
            problem.solve(solver=cp.ECOS, eps=1e-6)
            if problem.status == 'OPTIMAL':
                h = h_var.value
                h = np.clip(h, 0, 1)
                h = h * n_points / 2 / np.sum(h)
                c5 = evaluate_erdos_solution(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_result = h.copy()
        except:
            pass
        
        # Gradient descent refinement
        for _ in range(10):
            if time.time() - time_start > budget_s - 1:
                break
            c5 = evaluate_erdos_solution(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_result = h.copy()
            
            # Compute gradients
            grads = np.zeros(n_points)
            for k in range(n_points):
                for i in range(n_points):
                    j = (i + k) % n_points
                    grad = (1 - h[j]) * dx * np.heaviside(h[i], 0.5)
                    grads[i] += grad
            
            # Update h with gradient descent
            h = h - 0.1 * grads
            h = np.clip(h, 0, 1)
            h = h * n_points / 2 / np.sum(h)
    
    # Final check
    h = best_result
    h = np.clip(h, 0, 1)
    h = h * n_points / 2 / np.sum(h)
    final_c5 = evaluate_erdos_solution(h, dx)
    
    return (h, final_c5, n_points)
