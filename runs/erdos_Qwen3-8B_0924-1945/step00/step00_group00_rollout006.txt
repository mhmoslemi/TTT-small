```python
import numpy as np
import scipy.optimize
import time
import random

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 45
    h_values = np.array(initial_h_values, dtype=np.float64) if initial_h_values is not None else np.random.rand(n_points)
    
    # Ensure sum constraint and normalization
    h_values = np.clip(h_values, 0, 1)
    h_values = h_values * (n_points / 2) / np.sum(h_values)
    
    best_h = h_values.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    learning_rate = 0.05
    noise_scale = 0.05
    max_iterations = 500
    iteration = 0
    
    while time.time() - start_time < budget_s and iteration < max_iterations:
        # Compute correlate_vals
        correlate_vals = np.correlate(h_values, 1 - h_values, mode="full") * (2.0 / n_points)
        current_c5 = np.max(correlate_vals)
        
        # Track best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_values.copy()
        
        # Find worst lag
        worst_lag = np.argmax(correlate_vals)
        
        # Compute gradient for worst lag
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h_values[j]) * (2.0 / n_points)
        
        # Update h
        h_values = h_values - learning_rate * gradients
        
        # Clip and normalize
        h_values = np.clip(h_values, 0, 1)
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        
        # Add structured noise
        if iteration % 100 == 0:
            noise = noise_scale * np.random.normal(0, 1, n_points)
            h_values = h_values + noise
            h_values = np.clip(h_values, 0, 1)
            h_values = h_values * (n_points / 2) / np.sum(h_values)
        
        learning_rate *= 0.995
        iteration += 1
    
    # Final check
    correlate_vals = np.correlate(best_h, 1 - best_h, mode="full") * (2.0 / n_points)
    final_c5 = np.max(correlate_vals)
    
    # Ensure constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h, final_c5, n_points)
```
