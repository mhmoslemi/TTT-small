import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def rescale(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total) < 1e-10:
            return np.zeros_like(h)
        scale = (n_points / 2) / total
        return h * scale
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def evaluate_candidate(h_candidate):
        h_candidate = rescale(h_candidate)
        try:
            c5 = compute_c5(h_candidate)
        except:
            c5 = np.inf
        return c5
    
    def find_contributing_indices(h, lag):
        overlap = h * np.roll(1 - h, lag)
        max_val = np.max(overlap)
        threshold = max_val * 0.8
        indices = np.where(overlap > threshold)[0]
        return indices
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_iter = 0
    start_time = time.time()
    
    for t in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        lag = np.argmax(corr)
        indices = find_contributing_indices(h, lag)
        
        candidates = []
        for i in np.random.choice(indices, min(len(indices), 20), replace=False):
            h_candidate = h.copy()
            for delta in [-0.05, 0.05]:
                h_candidate[i] += delta
                h_candidate = rescale(h_candidate)
                candidates.append(h_candidate)
        
        scores = [evaluate_candidate(cand) for cand in candidates]
        best_idx = np.argmin(scores)
        if scores[best_idx] < best_c5:
            best_c5 = scores[best_idx]
            best_h = candidates[best_idx]
            best_iter = t
        else:
            if np.random.rand() < 0.1:
                h = rescale(np.random.uniform(0, 1, n_points))
    
    best_h = rescale(best_h)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n_points)
