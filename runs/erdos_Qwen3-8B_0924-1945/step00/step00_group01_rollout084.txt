import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Start with initial_h_values if available
    if 'initial_h_values' in globals():
        h_vals = np.copy(initial_h_values)
    else:
        # Default starting point: random values with integral constraint
        n_points = 50
        h_vals = np.random.rand(n_points)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
    
    n_points = len(h_vals)
    dx = 2.0 / n_points
    best_h_vals = h_vals.copy()
    best_c5_bound = 1.0  # Initialize with a high value

    start_time = time.time()
    max_iterations = 200
    for iteration in range(max_iterations):
        # Compute correlations
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        current_c5 = np.max(corr * dx)
        
        # Early termination if no improvement
        if current_c5 >= best_c5_bound:
            break
        
        best_c5_bound = current_c5
        best_h_vals = h_vals.copy()
        
        # Find worst lag
        k_max = np.argmax(corr)
        overlap_k = corr[k_max] * dx

        # Define variables for convex optimization
        lambda_k = cp.Variable(1)
        objective = cp.Minimize(0.5 * lambda_k**2 + (overlap_k - lambda_k)**2)
        constraints = [lambda_k >= 0, lambda_k >= overlap_k]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        
        # Update h_vals using gradient of the convex surrogate
        lambda_opt = lambda_k.value[0]
        grad = -overlap_k + lambda_opt
        h_vals += grad * 0.01
        
        # Clip h_vals to [0, 1]
        h_vals = np.clip(h_vals, 0, 1)
        
        # Enforce integral constraint
        mass = np.sum(h_vals)
        if mass != 0:
            h_vals = h_vals * (n_points / 2) / mass
        
        # Check time budget
        if time.time() - start_time > budget_s - 5:
            break

    # Final evaluation
    corr = np.correlate(best_h_vals, 1 - best_h_vals, mode='full')
    final_c5 = np.max(corr * dx)
    return (best_h_vals, final_c5, n_points)
