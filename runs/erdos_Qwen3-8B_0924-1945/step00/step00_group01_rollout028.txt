import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values) if 'initial_h_values' in kwargs else 47
    h_vals = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in kwargs else np.random.rand(n_points)
    
    # Normalize to meet the sum constraint
    h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
    
    dx = 2.0 / n_points
    max_iterations = 100
    no_improvement_threshold = 5
    best_c5 = np.inf
    best_h = h_vals.copy()
    iteration = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s and iteration < max_iterations:
        # Compute correlations
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = corr * dx
        current_max = np.max(c5)
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h_vals.copy()
        
        # Select the worst lag
        k = np.argmax(corr)
        
        # Define the objective function
        def objective(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h) if np.sum(h) != 0 else h
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            return overlap
        
        def constraint(x):
            return np.sum(x) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            result = opt.minimize(objective, h_vals, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint})
            h_vals = result.x
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals) if np.sum(h_vals) != 0 else h_vals
        except:
            h_vals = best_h.copy()
        
        # Check for improvement
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = corr * dx
        current_max = np.max(c5)
        
        if current_max >= best_c5:
            iteration += 1
            if iteration >= no_improvement_threshold:
                break
        else:
            best_c5 = current_max
            best_h = h_vals.copy()
            iteration = 0
    
    # Final check
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx
    return (best_h, c5_bound, n_points)
