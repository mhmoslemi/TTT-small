import numpy as np
import scipy
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 100
    if initial_h_values is not None:
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    learning_rate = 0.1
    max_iter = 1000
    decay_rate = 0.99
    noise_prob = 0.05
    noise_scale = 0.05
    iter_count = 0
    early_stop = 0
    while time.time() - start_time < budget_s and iter_count < max_iter:
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        current_c5 = np.max(correlate_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            early_stop = 0
        else:
            early_stop += 1
            if early_stop > 100:
                break
        if iter_count % 100 == 0 and iter_count > 0:
            noise_prob = max(noise_prob * 0.9, 0.01)
            noise_scale = max(noise_scale * 0.9, 0.01)
        m = np.argmax(correlate_vals)
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = i - m
            if 0 <= j < n_points:
                gradients[i] = (1 - h[j]) * (2.0 / n_points)
        h = h - learning_rate * gradients
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        if random.random() < noise_prob:
            noise = np.random.normal(0, noise_scale, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        learning_rate *= decay_rate
        iter_count += 1
    h = best_h
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
    c5_bound = np.max(correlate_vals)
    return (h.tolist(), c5_bound, n_points)
