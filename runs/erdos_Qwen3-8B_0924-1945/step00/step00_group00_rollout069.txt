import numpy as np
import scipy
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 45
    if initial_h_values is not None:
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()

    learning_rate = 0.05
    noise_scale = 0.05
    max_iterations = 500
    iterations = 0
    patience = 50
    improvement = float('inf')

    while time.time() - start_time < budget_s and iterations < max_iterations:
        dx = 2.0 / n_points
        correlate_vals = scipy.signal.correlate(h, 1 - h, mode='full') * dx
        worst_lag = np.argmax(correlate_vals)
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * dx
        h = h - learning_rate * gradients
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h += noise_scale * np.random.normal(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        current_c5 = np.max(correlate_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            improvement = 1
        else:
            improvement = improvement * 0.9
        iterations += 1
        if improvement < 1e-5:
            break

    dx = 2.0 / n_points
    correlate_vals = scipy.signal.correlate(best_h, 1 - best_h, mode='full') * dx
    best_c5 = np.max(correlate_vals)
    return (best_h.tolist(), best_c5, n_points)
