import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 450
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in globals() else None
    if initial_h_values is not None:
        h = np.clip(initial_h_values, 0, 1)
    else:
        h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    elapsed = 0
    for t in range(6000):
        start_time = time.time()
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        if max_k < 0 or max_k >= len(corr):
            continue
        # Targeted mutation
        candidates = []
        for lag in range(max_k - 10, max_k + 11):
            if 0 <= lag < len(corr):
                start_idx = max(0, lag)
                end_idx = min(len(h), len(h) + lag)
                for i in range(start_idx, end_idx):
                    contrib = h[i] * (1 - h[i + lag]) if i + lag < len(h) else 0
                    if contrib > 0:
                        h[i] -= 0.01
                    else:
                        h[i] += 0.01
                    h[i] = np.clip(h[i], 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    candidates.append(h.copy())
        # Global mutation
        for _ in range(5):
            idx = np.random.randint(0, n_points)
            h[idx] += np.random.normal(0, 0.05)
            h[idx] = np.clip(h[idx], 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            candidates.append(h.copy())
        # Evaluate
        for candidate in candidates:
            candidate = np.clip(candidate, 0, 1)
            candidate = candidate * (n_points / 2) / np.sum(candidate)
            c5 = compute_c5(candidate, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate.copy()
        h = best_h.copy()
        elapsed += time.time() - start_time
        if elapsed >= budget_s:
            break
    return best_h, best_c5, n_points

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * 2.0 / n_points
    return c5
