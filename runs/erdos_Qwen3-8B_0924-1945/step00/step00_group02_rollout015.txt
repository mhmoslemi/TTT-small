import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    if 'initial_h_values' in kwargs and kwargs['initial_h_values'] is not None:
        h = np.array(kwargs['initial_h_values'], dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * n_points / 2.0
        h = np.clip(h, 0, n_points)
        h = h / (n_points / 2.0)
    
    h = np.clip(h, 0, 1)
    h = h * n_points / 2.0
    h = np.clip(h, 0, n_points)
    h = h / (n_points / 2.0)
    
    def calc_c5(h):
        h = np.clip(h, 0, 1)
        h = h * n_points / 2.0
        h = np.clip(h, 0, n_points)
        h = h / (n_points / 2.0)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    best_h = h.copy()
    best_c5 = calc_c5(best_h)
    temp = 100.0
    alpha = 0.995
    max_iter = int(budget_s / 0.1)
    iter_count = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        if iter_count >= max_iter:
            break
        iter_count += 1
        indices = np.random.choice(n_points, 5, replace=False)
        noise = np.random.normal(0, 0.05, n_points)
        new_h = h.copy() + noise
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * n_points / 2.0
        new_h = np.clip(new_h, 0, n_points)
        new_h = new_h / (n_points / 2.0)
        
        c5_new = calc_c5(new_h)
        delta = c5_new - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h = new_h.copy()
            best_c5 = c5_new
        temp *= alpha
    
    h = np.clip(h, 0, 1)
    h = h * n_points / 2.0
    h = np.clip(h, 0, n_points)
    h = h / (n_points / 2.0)
    best_h = h.copy()
    best_c5 = calc_c5(best_h)
    
    return (best_h, best_c5, n_points)
