import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 250
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h = np.array(initial_h_values[:n_points], dtype=np.float64)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.inf
    T = 100.0
    alpha = 0.995
    dt = 0.1
    max_iter = int(budget_s / dt)
    
    for t in range(max_iter):
        start_time = time.time()
        indices = np.random.choice(n_points, size=5, replace=False)
        noise = np.random.normal(0, 0.05 * np.exp(-t / 200), size=n_points)
        new_h = h + noise
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        corr = np.correlate(new_h, 1 - new_h, mode="full")
        c5_new = np.max(corr) * dx
        
        if c5_new < best_c5:
            best_h = new_h.copy()
            best_c5 = c5_new
        
        delta = c5_new - best_c5
        if delta > 0:
            prob = np.exp(-delta / T)
            if np.random.rand() < prob:
                h = new_h.copy()
        
        T *= alpha
        elapsed = time.time() - start_time
        if elapsed > budget_s - 0.1:
            break
    
    corr_best = np.correlate(best_h, 1 - best_h, mode="full")
    best_c5 = np.max(corr_best) * dx
    return best_h, best_c5, n_points
