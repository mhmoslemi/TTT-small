import numpy as np
import scipy.signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points)
    initial_h = np.clip(initial_h, 0, 1)
    total_integral = np.sum(initial_h) * dx
    initial_h = initial_h * (1.0 / (total_integral * dx)) if total_integral != 0 else initial_h
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    for _ in range(2000):
        if time.time() - start_time > budget_s:
            break
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        max_lag = np.argmax(corr)
        if max_lag >= len(h):
            max_lag = len(h) - 1
        grad = np.zeros_like(h)
        for i in range(len(h)):
            j = i - max_lag
            if 0 <= j < len(h):
                grad[i] = (1 - h[j]) * dx
        h -= 0.01 * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    return (best_h, best_c5, n_points)
