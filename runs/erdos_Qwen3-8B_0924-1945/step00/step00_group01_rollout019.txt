```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    
    if initial_h_values is not None:
        h_vals = np.array(initial_h_values, dtype=np.float64)
    else:
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * n_points / 2) / np.sum(h_vals)
    
    best_h = h_vals.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    time_used = 0
    
    for _ in range(100):
        start_time = time.time()
        
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        current_c5 = corr * dx
        max_c5_idx = np.argmax(current_c5)
        
        obj_func = lambda x: corr[max_c5_idx] * dx
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = opt.minimize(obj_func, h_vals, bounds=bounds, constraints=cons, method='L-BFGS-B')
        if result.success:
            h_vals = np.clip(result.x, 0, 1)
            h_vals = (h_vals * n_points / 2) / np.sum(h_vals)
        else:
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = (h_vals * n_points / 2) / np.sum(h_vals)
        
        new_c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * dx
        time_used += time.time() - start_time
        
        if new_c5 < best_c5:
            best_h = h_vals.copy()
            best_c5 = new_c5
        elif time_used > budget_s - 100:
            break
    
    return (best_h, best_c5, n_points)
```
