import numpy as np
import scipy.signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    initial_sum = np.sum(h)
    target_sum = n_points / 2
    scaling_factor = target_sum / initial_sum
    h = h * scaling_factor
    h = np.clip(h, 0.0, 1.0)
    best_c5 = evaluate_erdos_solution(h, dx)
    
    max_iterations = 500
    perturbation_prob = 0.05
    learning_rate = 0.1
    tolerance = 1e-6
    time_start = time.time()
    
    for i in range(max_iterations):
        if time.time() - time_start > budget_s - 10:
            break
        cross_corr = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            if k >= n_points:
                k_wrap = k - n_points
                valid_indices = np.arange(n_points - k_wrap)
                cross_corr[k] = dx * np.sum(h[valid_indices] * h[k_wrap: k_wrap + len(valid_indices)])
            else:
                valid_indices = np.arange(n_points - k)
                cross_corr[k] = dx * np.sum(h[valid_indices] * h[k: k + len(valid_indices)])
        current_max = np.max(cross_corr)
        
        max_k = np.argmax(cross_corr)
        grad = np.zeros(n_points)
        if max_k >= n_points:
            k_wrap = max_k - n_points
            for i in range(n_points - k_wrap):
                grad[i] += dx * h[i + k_wrap]
                grad[i + k_wrap] += dx * h[i]
        else:
            for i in range(n_points - max_k):
                grad[i] += dx * h[i + max_k]
                grad[i + max_k] += dx * h[i]
        
        direction = -grad
        new_h = h + learning_rate * direction
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        if new_sum == 0:
            scaling_factor = 1.0
        else:
            scaling_factor = target_sum / new_sum
        new_h = new_h * scaling_factor
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        if new_sum == 0:
            scaling_factor = 1.0
        else:
            scaling_factor = target_sum / new_sum
        new_h = new_h * scaling_factor
        
        if np.random.rand() < perturbation_prob:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            new_h = h + noise
            new_h = np.clip(new_h, 0.0, 1.0)
            new_sum = np.sum(new_h)
            if new_sum == 0:
                scaling_factor = 1.0
            else:
                scaling_factor = target_sum / new_sum
            new_h = new_h * scaling_factor
            new_h = np.clip(new_h, 0.0, 1.0)
        
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
        
        if np.abs(new_c5 - best_c5) < tolerance:
            break
    
    h = best_h
    assert (h >= 0).all() and (h <= 1).all()
    assert np.isclose(np.sum(h), target_sum, atol=1e-6)
    assert np.isfinite(best_c5)
    
    return (h, best_c5, n_points)
