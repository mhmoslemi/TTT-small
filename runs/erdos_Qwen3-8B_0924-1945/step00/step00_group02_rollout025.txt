import numpy as np
import random
import time
import scipy.signal
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Default parameters
    n_points = 500
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else None
    
    if initial_h_values is None:
        # Construct initial h_values using a simple alternating pattern
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 if i % 2 == 0 else 0.0
        h = np.where(h == 1.0, 1.0, 0.0)
        # Rescale to meet the integral constraint
        total = np.sum(h)
        h = h * (n_points / 2.0) / total
        initial_h_values = h
    
    # Ensure initial_h_values is of correct length and within bounds
    if len(initial_h_values) != n_points:
        initial_h_values = np.zeros(n_points)
        for i in range(n_points):
            initial_h_values[i] = 1.0 if i % 2 == 0 else 0.0
        initial_h_values = np.where(initial_h_values == 1.0, 1.0, 0.0)
        total = np.sum(initial_h_values)
        initial_h_values = initial_h_values * (n_points / 2.0) / total
    
    best_h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    
    population_size = 50
    population = [initial_h_values.copy() for _ in range(population_size)]
    
    generation = 0
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    pop_fitness = [evaluate_erdos_solution(h, n_points) for h in population]
    
    while time.time() - start_time < budget_s and generation < 200:
        # Evaluate all individuals
        for i in range(population_size):
            population[i] = np.clip(population[i], 0.0, 1.0)
            total = np.sum(population[i])
            if total > 0:
                population[i] = population[i] * (n_points / 2.0) / total
            else:
                population[i] = np.full(n_points, 0.5)
        pop_fitness = [evaluate_erdos_solution(h, n_points) for h in population]
        
        # Select top 20% for crossover
        top_indices = np.argsort(pop_fitness)[:int(population_size * 0.2)]
        top_candidates = [population[i] for i in top_indices]
        
        # Crossover
        new_population = []
        for i in range(len(top_candidates) // 2):
            h1 = top_candidates[i]
            h2 = top_candidates[len(top_candidates) - 1 - i]
            child = (h1 + h2) / 2.0
            child = np.clip(child, 0.0, 1.0)
            total = np.sum(child)
            if total > 0:
                child = child * (n_points / 2.0) / total
            else:
                child = np.full(n_points, 0.5)
            new_population.append(child)
        
        # Mutation
        for i in range(population_size):
            if i in top_indices:
                continue
            h = population[i].copy()
            indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
            delta = np.random.uniform(-0.1, 0.1, size=len(indices))
            h[indices] += delta
            h = np.clip(h, 0.0, 1.0)
            total = np.sum(h)
            if total > 0:
                h = h * (n_points / 2.0) / total
            else:
                h = np.full(n_points, 0.5)
            new_population.append(h)
        
        # Replace population
        population = new_population
        pop_fitness = [evaluate_erdos_solution(h, n_points) for h in population]
        
        # Update best solution
        for i in range(len(population)):
            if pop_fitness[i] < best_c5:
                best_c5 = pop_fitness[i]
                best_h = population[i].copy()
        
        generation += 1
    
    # Final check on best_h
    best_h = np.clip(best_h, 0.0, 1.0)
    total = np.sum(best_h)
    if total > 0:
        best_h = best_h * (n_points / 2.0) / total
    else:
        best_h = np.full(n_points, 0.5)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    return (best_h, final_c5, n_points)

def evaluate_erdos_solution(h, n_points):
    h = np.array(h, dtype=np.float64)
    total = np.sum(h)
    if total > 0:
        h = h * (n_points / 2.0) / total
    else:
        h = np.full(n_points, 0.5)
    h = np.clip(h, 0.0, 1.0)
    one_minus_h = 1.0 - h
    corr = scipy.signal.correlate(h, one_minus_h, mode='full')
    dx = 2.0 / n_points
    max_c5 = np.max(corr) * dx
    return max_c5
