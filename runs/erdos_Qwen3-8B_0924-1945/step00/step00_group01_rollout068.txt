import numpy as np
import random
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess from parent construction
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in globals() else None
    
    # Determine n_points (number of samples)
    if initial_h_values is not None:
        n_points = len(initial_h_values)
        # Scale to meet the integral constraint
        h_vals = (initial_h_values * (n_points / 2)) / np.sum(initial_h_values)
        # Ensure sum is exactly n_points / 2
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
    else:
        # Use default n_points (minimum for reasonable performance)
        n_points = 100
        h_vals = np.random.rand(n_points)
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h_vals = h_vals.copy()
    best_c5 = evaluate_erdos_solution(h_vals, dx)
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 100
    mutation_rate = 0.05
    tournament_size = 5
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = np.zeros(population_size)
        for i in range(population_size):
            h = h_vals[i].copy()
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            fitness[i] = c5
        
        # Select parents using tournament selection
        parents = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size, replace=False)
            winner = tournament[np.argmax(fitness[tournament])]
            parents.append(h_vals[winner].copy())
        
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            if i + 1 < population_size:
                parent1 = parents[i]
                parent2 = parents[i+1]
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
                child2 = np.concatenate((parent2[:crossover_point], parent1[crossover_point:]))
                offspring.append(child1)
                offspring.append(child2)
            else:
                offspring.append(parents[i])
        
        # Mutation
        mutated_offspring = []
        for h in offspring:
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    h[i] += np.random.uniform(-0.05, 0.05)
                    h[i] = np.clip(h[i], 0, 1)
            # Ensure the sum constraint is met
            h = h / np.sum(h) * (n_points / 2)
            mutated_offspring.append(h)
        
        # Evaluate and select next population
        new_population = []
        for h in mutated_offspring:
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h_vals = h.copy()
                best_c5 = c5
            new_population.append((h, c5))
        
        # Select the best individuals for next generation
        new_population.sort(key=lambda x: x[1])
        new_population = [h for h, _ in new_population[:population_size]]
        h_vals = new_population
    
    # Final evaluation
    final_corr = np.correlate(best_h_vals, 1 - best_h_vals, mode="full")
    final_c5 = np.max(final_corr) * dx
    return (best_h_vals, final_c5, n_points)
