import numpy as np
import scipy
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = (h - 0.5) * 2
        h = np.clip(h, 0, 1)
        h = h * n_points / 2 / np.sum(h)
    h = np.clip(h, 0, 1)
    h = h * n_points / 2 / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
    T = 100.0
    alpha = 0.995
    start_time = time.time()
    for t in range(int((budget_s - (time.time() - start_time)) / 0.1)):
        if time.time() - start_time > budget_s:
            break
        indices = np.random.choice(n_points, 5, replace=False)
        delta = np.random.normal(0, 0.05 * np.exp(-t / 200), size=n_points)
        new_h = h.copy()
        new_h[indices] += delta[indices]
        new_h = np.clip(new_h, 0, 1)
        total = np.sum(new_h)
        if abs(total) < 1e-6:
            new_h = np.random.uniform(0, 1, n_points)
            new_h = (new_h - 0.5) * 2
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * n_points / 2 / np.sum(new_h)
        else:
            new_h = new_h * (n_points / 2) / total
        corr = np.correlate(new_h, 1 - new_h, mode="full")
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = new_h.copy()
        else:
            delta_c5 = c5 - best_c5
            if np.random.rand() < np.exp(-delta_c5 / T):
                best_c5 = c5
                best_h = new_h.copy()
        T *= alpha
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * n_points / 2 / np.sum(best_h)
    corr = np.correlate(best_h, 1 - best_h, mode="full")
    c5_bound = np.max(corr) * dx
    return (best_h, c5_bound, n_points)
