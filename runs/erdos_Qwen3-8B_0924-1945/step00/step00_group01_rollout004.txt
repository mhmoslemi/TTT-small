```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None  # Placeholder for parent construction

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 samples, adjust if needed
    dx = 2.0 / n_points
    h_vals = np.zeros(n_points)
    
    # Initialize h_vals using initial_h_values if provided
    if initial_h_values is not None:
        h_vals[:] = initial_h_values[:]
    else:
        # Random initial guess with sum(n_points / 2)
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * n_points / 2) / np.sum(h_vals)
    
    # Ensure sum of h_vals is exactly n_points / 2
    h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
    
    max_iterations = 100
    no_improvement_limit = 5
    best_c5 = np.inf
    best_h_vals = h_vals.copy()
    start_time = time.time()
    
    for _ in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 100:
            break
        
        # Compute all lags
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5_values = corr * dx
        
        # Find the current maximum C5
        current_max_c5 = np.max(c5_values)
        
        # Check for improvement
        if current_max_c5 < best_c5:
            best_c5 = current_max_c5
            best_h_vals = h_vals.copy()
        elif current_max_c5 >= best_c5 and _ > no_improvement_limit:
            break  # No improvement for too long, early exit
        
        # Select the worst lag
        worst_lag = np.argmax(c5_values)
        
        # Define objective function for the worst lag
        def objective(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            corr = np.correlate(h, 1 - h, mode='full')
            return corr[worst_lag] * dx
        
        # Optimization with bounds and equality constraint
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        
        try:
            result = opt.minimize(
                objective,
                h_vals,
                bounds=bounds,
                constraints=cons,
                method='L-BFGS-B'
            )
            h_vals = result.x
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
        except:
            # Fallback in case of solver failure
            h_vals = best_h_vals.copy()
        
        # Ensure values are in [0,1] and sum to n_points / 2
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
    
    # Final validation
    corr = np.correlate(best_h_vals, 1 - best_h_vals, mode='full')
    final_c5 = np.max(corr) * dx
    assert np.all((best_h_vals >= 0) & (best_h_vals <= 1)), "Values out of bounds"
    assert np.isclose(np.sum(best_h_vals), n_points / 2), "Sum constraint violated"
    
    return best_h_vals, final_c5, n_points
```
