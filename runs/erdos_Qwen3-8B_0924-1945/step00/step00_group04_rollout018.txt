    improvement_threshold = 1e-6

    for iteration in range(max_iterations):
        start_time = time.time()
        
        # Compute cross-correlation
        cross_corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        idx = np.argmax(cross_corr)
        lag_k = idx - (n_points - 1)
        
        # Compute gradient for the worst lag
        grad = np.zeros(n_points)
        for j in range(n_points):
            if j + lag_k < n_points:
                grad[j] += (1 - h[j + lag_k]) * (2.0 / n_points)
            if j - lag_k >= 0:
                grad[j] -= h[j - lag_k] * (2.0 / n_points)
        
        # Update h
        h = h - learning_rate * grad
        
        # Clip and rescale
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            h = np.zeros(n_points)
        else:
            h = h * (target_sum / current_sum)
        
        # Evaluate new C5
        new_c5 = evaluate_erdos_solution(h)
        
        # Check for improvement
        if new_c5 < best_c5 - improvement_threshold:
            best_c5 = new_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Check for early stopping
        if no_improvement_count >= 50 or time.time() - start_time > 1:
            break
        
        # Update elapsed time
        elapsed_time += time.time() - start_time
        
        # Decay learning rate
        if iteration % 50 == 0 and iteration > 0:
