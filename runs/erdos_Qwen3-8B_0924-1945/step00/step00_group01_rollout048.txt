        if current_c5 < best_c5:

            best_h_vals = h_vals.copy()
            early_stop_counter = 0
        else:
            early_stop_counter += 1

        if early_stop_counter >= 5:
            break

        k_max = np.argmax(corr)
        overlap_k_max = c5[k_max]

        lambda_k = cp.Variable()
        obj = 0.5 * lambda_k ** 2 + (overlap_k_max - lambda_k) ** 2

        prob = cp.Problem(cp.Minimize(obj))
        prob.solve()
