import numpy as np
import scipy.optimize as opt
import time

initial_h_values = np.array([0.75, 0.75, 0.75, 0.25, 0.25, 0.75, 0.75, 0.75, 0.75, 0.75,
                             0.75, 0.25, 0.75, 0.25, 0.25, 0.25, 0.25, 0.25, 0.75, 0.75,
                             0.25, 0.25, 0.25, 0.25, 0.25, 0.75, 0.75, 0.25, 0.75, 0.25,
                             0.75, 0.25, 0.75, 0.25, 0.75, 0.75, 0.25, 0.75, 0.75, 0.25,
                             0.75, 0.25])

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    elapsed_time = 0
    max_iter = 500
    alpha = 0.1
    decay = 0.995
    eps = 1e-8
    early_stop = 30
    improvement = np.inf
    start_time = time.time()

    for iter in range(max_iter):
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
        cross_corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_index = np.argmax(cross_corr)
        if max_index < 0 or max_index >= len(cross_corr):
            break
        # Perturb h to compute directional gradient
        h_plus = h.copy()
        h_plus[max_index] += eps
        h_plus = np.clip(h_plus, 0, 1)
        h_plus_sum = np.sum(h_plus)
        if abs(h_plus_sum - target_sum) > 1e-6:
            h_plus = h_plus - (h_plus_sum - target_sum) / n_points
        c5_plus = evaluate_erdos_solution(h_plus)
        h_minus = h.copy()
        h_minus[max_index] -= eps
        h_minus = np.clip(h_minus, 0, 1)
        h_minus_sum = np.sum(h_minus)
        if abs(h_minus_sum - target_sum) > 1e-6:
            h_minus = h_minus - (h_minus_sum - target_sum) / n_points
        c5_minus = evaluate_erdos_solution(h_minus)
        grad = (c5_plus - c5_minus) / (2 * eps)
        # Update h
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - target_sum) > 1e-6:
            h = h - (h_sum - target_sum) / n_points
        # Evaluate
        new_c5 = evaluate_erdos_solution(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            improvement = 0
        else:
            improvement += 1
        if improvement >= early_stop:
            break
        alpha *= decay

    return (best_h, best_c5, n_points)
