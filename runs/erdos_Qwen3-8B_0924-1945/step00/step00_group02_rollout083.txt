```python
import numpy as np
import random
import time
from scipy.signal import correlate

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 1000
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:n_points//2] = 1.0  # Start with a simple step function
    
    # Adjust initial h to meet the constraint sum(h) = n_points / 2
    sum_h = np.sum(h)
    if sum_h != n_points / 2:
        h = h * (n_points / 2) / sum_h
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        
        # Perturb 5 random indices
        indices = np.random.choice(n_points, 5, replace=False)
        noise = np.random.normal(0, 0.1, 5)
        h_perturbed = h.copy()
        for i, idx in enumerate(indices):
            h_perturbed[idx] += noise[i]
        
        # Clip values to [0, 1]
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Rescale to meet the sum constraint
        sum_perturbed = np.sum(h_perturbed)
        if sum_perturbed != 0:
            h_perturbed = h_perturbed * (n_points / 2) / sum_perturbed
        
        # Evaluate C5
        current_c5 = compute_c5(h_perturbed, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = current_c5
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    h_expanded = np.hstack([h, h[:1]])  # Pad to handle wrap-around
    one_minus_h = 1 - h
    correlation = correlate(h_expanded, one_minus_h, mode="full")
    max_overlap = np.max(correlation)
    return max_overlap * dx
```
