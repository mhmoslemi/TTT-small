import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = 0.5 * (1 + np.random.uniform(-1, 1, n_points))
    h = np.clip(h, 0, 1)
    h = h * n_points / 2.0 / np.sum(h) if np.sum(h) > 0 else h * n_points / 2.0

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        return evaluate_c5(h)

    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2.0

    def constraint_bounds(h_flat):
        h = h_flat.reshape(n_points)
        return np.all((h >= 0) & (h <= 1))

    bounds = Bounds(0, 1)
    n_iter = int(budget_s / 0.1)
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    for _ in range(n_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        lag = np.argmax(corr)
        overlap = h * (1 - np.roll(h, lag))
        indices = np.argsort(overlap)[::-1]
        target_indices = indices[:int(n_points * 0.1)]
        candidates = []
        for idx in target_indices:
            h_candidate = h.copy()
            h_candidate[idx] = 1 - h_candidate[idx]
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * n_points / 2.0 / np.sum(h_candidate) if np.sum(h_candidate) > 0 else h_candidate * n_points / 2.0
            candidates.append(h_candidate)
            h_candidate = h.copy()
            h_candidate[idx] += np.random.uniform(-0.05, 0.05)
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * n_points / 2.0 / np.sum(h_candidate) if np.sum(h_candidate) > 0 else h_candidate * n_points / 2.0
            candidates.append(h_candidate)
        for candidate in candidates:
            c5 = evaluate_c5(candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate.copy()
        h = best_h.copy()
        if np.abs(evaluate_c5(h) - best_c5) < 1e-6:
            break
    return best_h, best_c5, n_points
