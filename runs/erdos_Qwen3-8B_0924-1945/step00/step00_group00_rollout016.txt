import numpy as np
import scipy.signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    h = np.array(initial_h_values, dtype=np.float64)
    h = np.clip(h, 0, 1)
    h /= np.sum(h) * 2.0
    h *= n_points / 2.0

    def evaluate_c5(h):
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    best_c5 = evaluate_c5(h)
    best_h = h.copy()

    start_time = time.time()
    learning_rate = 0.01
    steps = 0
    max_steps = 1000
    while steps < max_steps and time.time() - start_time < budget_s:
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        max_corr_index = np.argmax(corr)
        max_corr_value = corr[max_corr_index]
        lag = max_corr_index - n_points + 1
        target_lag = lag

        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i - target_lag
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * (2.0 / n_points)
        grad *= learning_rate
        h -= grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h)

        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        steps += 1

    return best_h, best_c5, n_points
