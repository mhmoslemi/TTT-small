        k_max = np.argmax(corr_scaled)

        # Objective: 0.5 * lambda^2 + (overlap_k - lambda)^2 for the worst lag
        # We solve this using convex optimization
        
        lambda_var = cp.Variable()
        objective = 0.5 * lambda_var ** 2 + (corr_scaled[k_max] - lambda_var) ** 2

            step_size = 0.1
            h_vals += step_size * (1.0 - h_vals[k_max:] - (1.0 - h_vals[:-k_max]))

            # Re-normalize to meet the integral constraint
            total_mass = np.sum(h_vals)
