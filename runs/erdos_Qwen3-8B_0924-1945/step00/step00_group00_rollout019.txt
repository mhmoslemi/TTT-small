import numpy as np
import scipy.optimize as opt
import time
from functools import partial

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Default parameters
    n_points = 45  # Start with the provided initial_h_values
    initial_h_values = kwargs.get('initial_h_values', None)
    
    if initial_h_values is None:
        # Generate a random initial solution
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    else:
        h = initial_h_values
    
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    # Optimization parameters
    max_iter = 500
    learning_rate = 0.05
    noise_scale = 0.05
    early_stop_iter = 50
    early_stop_tol = 1e-5
    decay_rate = 0.995
    
    for iter in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Compute the full correlation
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        current_c5 = np.max(correlate_vals)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Check early stopping
        if iter > early_stop_iter and current_c5 - best_c5 < early_stop_tol:
            break
        
        # Find worst lag
        worst_lag = np.argmax(correlate_vals)
        
        # Compute gradients for the worst lag
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * (2.0 / n_points)
        
        # Update h
        h -= learning_rate * gradients
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Add noise
        if iter % 100 == 0:
            noise = noise_scale * np.random.normal(0, 1, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Decay learning rate
        learning_rate *= decay_rate
    
    return (best_h, best_c5, n_points)
