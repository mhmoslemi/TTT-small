import numpy as np
import numpy.random as npr
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = 500
    np.random.seed(seed)
    initial_h_values = np.array([0.5] * n_points)
    
    if 'initial_h_values' in kwargs:
        initial_h_values = kwargs['initial_h_values']
    
    # Normalize the initial solution
    h = np.clip(initial_h_values, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    h = np.clip(h, 0, 1)
    h = h.astype(float)
    
    def compute_c5(h):
        dx = 2.0 / n_points
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlaps)
    
    # Parameters
    POP_SIZE = 50
    MUTATION_NOISE = 0.05
    MUTATION_RATE = 0.1
    N_SELECT = 20
    MAX_STAGNATION = 50
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_gen = 0
    stagnation = 0
    
    start_time = time.time()
    
    for gen in range(1000):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Generate population
        population = []
        
        # Parent
        parent = h.copy()
        population.append(parent)
        
        # Random mutations
        for _ in range(POP_SIZE - 1):
            mutant = parent + MUTATION_NOISE * npr.randn(n_points)
            mutant = np.clip(mutant, 0, 1)
            current_sum_mutant = np.sum(mutant)
            if current_sum_mutant != 0:
                mutant = mutant * (n_points / 2) / current_sum_mutant
            mutant = np.clip(mutant, 0, 1)
            population.append(mutant)
        
        # Evaluate population
        fitness = []
        for candidate in population:
            c = compute_c5(candidate)
            fitness.append((c, candidate))
        
        # Sort by fitness
        fitness.sort()
        sorted_fitness = np.array(fitness)
        
        # Select top N_SELECT
        selected = sorted_fitness[:N_SELECT, 1]
        
        # Create new generation
        new_population = []
        for _ in range(POP_SIZE):
            # Crossover
            a = npr.choice(selected)
            b = npr.choice(selected)
            split = npr.randint(1, n_points - 1)
            child = np.concatenate([a[:split], b[split:]])
            child = np.clip(child, 0, 1)
            current_sum_child = np.sum(child)
            if current_sum_child != 0:
                child = child * (n_points / 2) / current_sum_child
            child = np.clip(child, 0, 1)
            new_population.append(child)
        
        # Apply mutations
        for i in range(len(new_population)):
            candidate = new_population[i]
            if npr.rand() < MUTATION_RATE:
                mutation = MUTATION_NOISE * npr.randn(n_points)
                candidate = candidate + mutation
            candidate = np.clip(candidate, 0, 1)
            current_sum_candidate = np.sum(candidate)
            if current_sum_candidate != 0:
                candidate = candidate * (n_points / 2) / current_sum_candidate
            candidate = np.clip(candidate, 0, 1)
            new_population[i] = candidate
        
        # Replace population
        population = new_population
        
        # Update best
        for candidate in population:
            c = compute_c5(candidate)
            if c < best_c5:
                best_h = candidate.copy()
                best_c5 = c
                best_gen = gen
                stagnation = 0
            else:
                stagnation += 1
        
        # Adaptive mutation
        if stagnation > MAX_STAGNATION:
            MUTATION_NOISE *= 0.9
            stagnation = 0
    
    # Final check
    h = best_h.copy()
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    h = np.clip(h, 0, 1)
    c5_bound = compute_c5(h)
    
    return (h, c5_bound, n_points)
