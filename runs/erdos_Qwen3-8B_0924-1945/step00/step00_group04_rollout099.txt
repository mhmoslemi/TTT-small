import scipy.optimize as opt

import time

    best_h = h

    def objective(x):
        nonlocal best_c5
        h_vals = x
        corr = correlate(h_vals, 1 - h_vals, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        if c5 < best_c5:
            best_c5 = c5

    def constraint(x):
        return np.sum(x) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    res = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        h_final = res.x
        corr = correlate(h_final, 1 - h_final, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        if c5 < best_c5:
            best_h = h_final
            best_c5 = c5
    else:
        h_final = best_h
    
    return (h_final, best_c5, n_points)
