import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h_values = np.load('initial_h_values.npy') if hasattr(globals(), 'initial_h_values') else np.random.rand(n_points)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * target_sum / np.sum(initial_h_values)
    
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    elapsed_time = 0
    alpha = 0.1
    decay = 0.9995
    eps = 1e-8
    max_iter = 500
    iter_count = 0
    
    while elapsed_time < budget_s and iter_count < max_iter:
        start_time = time.time()
        
        cross_corr = np.correlate(h, 1 - h, mode="full") * dx
        max_index = np.argmax(cross_corr)
        c5 = cross_corr[max_index]
        
        if np.isclose(c5, 0):
            break
        
        h_plus = h.copy()
        h_plus[max_index] += eps
        h_plus = np.clip(h_plus, 0, 1)
        h_plus = h_plus * target_sum / np.sum(h_plus)
        c5_plus = evaluate_erdos_solution(h_plus)
        
        h_minus = h.copy()
        h_minus[max_index] -= eps
        h_minus = np.clip(h_minus, 0, 1)
        h_minus = h_minus * target_sum / np.sum(h_minus)
        c5_minus = evaluate_erdos_solution(h_minus)
        
        grad = (c5_plus - c5_minus) / (2 * eps)
        
        h_update = -alpha * grad
        h = h + h_update
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        new_c5 = evaluate_erdos_solution(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
        
        elapsed_time = time.time() - start_time
        alpha *= decay
        iter_count += 1
    
    return best_h, best_c5, n_points
