import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 450
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        h = 0.5 * (1 + np.random.uniform(-1, 1, n_points))
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    t0 = time.time()
    dt = 0.1
    for t in range(int((budget_s - (time.time() - t0)) / dt)):
        if time.time() - t0 > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        lags = np.arange(len(corr))
        if max_idx < len(lags) - 1:
            max_k = lags[max_idx]
        else:
            max_k = lags[-1]

        # Targeted mutation
        lags_to_check = np.arange(max_k - 10, max_k + 11)
        lags_to_check = lags_to_check[(lags_to_check >= 0) & (lags_to_check < len(lags))]
        indices = np.random.choice(lags_to_check, size=10, replace=False)
        indices = np.unique(indices)

        delta = 0.1 / (1 + t / 500)
        for i in indices:
            contribution = corr[i] * 2 / n_points
            if contribution > 0:
                h[i] -= delta * contribution
            else:
                h[i] += delta * contribution
            h[i] = np.clip(h[i], 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)

        # Global mutation
        indices = np.random.choice(n_points, size=5, replace=False)
        noise = np.random.normal(0, 0.2 * np.exp(-t / 300), size=5)
        h[indices] += noise
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)

        c5 = compute_c5(h, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if t % 100 == 0:
            if best_c5 < 0.3807:
                break

    h = best_h.copy()
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum
    h = np.clip(h, 0, 1)
    c5 = compute_c5(h, n_points)
    return (h, c5, n_points)

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    dx = 2.0 / n_points
    c5 = max_corr * dx
    return c5
