import numpy as np
import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45  # Base size, can be adjusted for better results
    dx = 2.0 / n_points
    h_initial = initial_h_values if 'initial_h_values' in kwargs else np.random.rand(n_points)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    swarm_size = 20
    particles = np.random.rand(swarm_size, n_points)
    particles = np.clip(particles, 0, 1)
    particles = particles * (n_points / 2) / np.sum(particles, axis=1, keepdims=True)
    
    velocities = np.random.rand(swarm_size, n_points) * 0.1
    pbest = particles.copy()
    pbest_fitness = np.full(swarm_size, np.inf)
    gbest = particles[0].copy()
    gbest_fitness = np.inf
    best_fitness = np.inf
    
    for _ in range(100):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        for i in range(swarm_size):
            h = particles[i]
            correlate_vals = np.correlate(h, 1 - h, mode='full') * dx
            current_fitness = np.max(correlate_vals)
            
            if current_fitness < pbest_fitness[i]:
                pbest_fitness[i] = current_fitness
                pbest[i] = h.copy()
            
            if current_fitness < gbest_fitness:
                gbest_fitness = current_fitness
                gbest = h.copy()
        
        # Update velocities and positions
        w = 0.7 - 0.3 * (_ / 100)
        c1, c2 = 1.5, 1.5
        r1, r2 = np.random.rand(swarm_size), np.random.rand(swarm_size)
        
        velocities = w * velocities + c1 * r1 * (pbest - particles) + c2 * r2 * (gbest - particles)
        particles = particles + velocities
        
        # Clamp and normalize
        particles = np.clip(particles, 0, 1)
        particles = particles * (n_points / 2) / np.sum(particles, axis=1, keepdims=True)
        
        # Local search on top performers
        top_indices = np.argsort(pbest_fitness)[:int(swarm_size * 0.05)]
        for idx in top_indices:
            perturb = np.random.normal(0, 0.05, n_points)
            particles[idx] = np.clip(particles[idx] + perturb, 0, 1)
            particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
        
        if gbest_fitness < best_fitness:
            best_fitness = gbest_fitness
            best_h = gbest.copy()
    
    # Final evaluation
    correlate_vals = np.correlate(best_h, 1 - best_h, mode='full') * dx
    best_c5 = np.max(correlate_vals)
    
    # Ensure constraints
    assert np.all(0 <= best_h <= 1)
    assert np.isclose(np.sum(best_h), n_points / 2)
    
    return (best_h, best_c5, n_points)
