import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in globals() else None
    n_points = 47 if initial_h_values is not None else 100
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h_vals = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    else:
        h_vals = np.random.rand(n_points)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
    
    def compute_correlation(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return corr * dx

    def objective(h_vals, k):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return corr[k] * dx

    best_c5 = np.max(compute_correlation(h_vals))
    best_h_vals = h_vals.copy()
    prev_c5 = np.inf
    iter_count = 0
    max_iterations = 100
    early_stop = 0

    while iter_count < max_iterations and time.time() < time.time() + budget_s:
        corr = compute_correlation(h_vals)
        max_k = np.argmax(corr)
        if max_k >= len(corr) // 2:
            max_k = len(corr) - 1 - max_k
        if corr[max_k] < best_c5:
            best_c5 = corr[max_k]
            best_h_vals = h_vals.copy()

        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        result = opt.minimize(
            lambda x: objective(x, max_k),
            x0=h_vals,
            bounds=bounds,
            constraints=cons,
            method='SLSQP'
        )
        if result.success:
            h_vals = result.x
        else:
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals / np.sum(h_vals) * (n_points / 2)

        curr_c5 = np.max(compute_correlation(h_vals))
        if curr_c5 >= prev_c5:
            early_stop += 1
        else:
            early_stop = 0
        prev_c5 = curr_c5

        if early_stop >= 5:
            break

        iter_count += 1

    h_vals = np.clip(h_vals, 0, 1)
    h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
    final_c5 = np.max(compute_correlation(h_vals))
    if final_c5 > best_c5:
        h_vals = best_h_vals
        final_c5 = best_c5

    if final_c5 < 0.38085:
        return (h_vals, final_c5, n_points)
    else:
        return (best_h_vals, best_c5, n_points)
