import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    initial_h_values = np.load('initial_h_values.npy')[:n_points] if 'initial_h_values' in globals() else np.random.rand(n_points)
    h = np.clip(initial_h_values, 0, 1)
    h = h * (n_points / 2) / np.sum(h) if np.sum(h) > 1e-10 else np.random.rand(n_points)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    learning_rate = 0.1
    iteration = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        k_max = np.argmax(overlaps)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 1e-10:
            h = h * (n_points / 2) / sum_h
        else:
            h = np.random.rand(n_points)
        
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            iteration = 0
        else:
            iteration += 1
            if iteration > 200:
                learning_rate *= 0.95
                iteration = 0
        
        if np.abs(current_c5 - best_c5) < 1e-5:
            break
    
    h = np.clip(best_h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 1e-10:
        h = h * (n_points / 2) / sum_h
    else:
        h = np.random.rand(n_points)
    
    h_values = h
    c5_bound = evaluate_erdos_solution(h_values)
    n_points = len(h_values)
    return h_values, c5_bound, n_points
