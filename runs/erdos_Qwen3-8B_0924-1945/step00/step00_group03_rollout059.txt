    learning_rate = 1e-2
    perturbation_prob = 0.05
    tolerance = 1e-6
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        cross_corr = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            if k < n_points:
                valid_indices = np.arange(n_points - k)
                cross_corr[k] = (dx * np.sum(h[valid_indices] * h[valid_indices + k]))
            else:
                k_wrap = k - n_points
                valid_indices = np.arange(n_points - k_wrap)
                cross_corr[k] = (dx * np.sum(h[valid_indices] * h[valid_indices + k_wrap]))
        
        current_max = np.max(cross_corr)
        max_k = np.argmax(cross_corr)
        
        grad = np.zeros(n_points)
        if max_k < n_points:
            for i in range(n_points - max_k):
                grad[i] += dx * h[i + max_k]
                grad[i + max_k] += dx * h[i]
        else:
            k_wrap = max_k - n_points
            for i in range(n_points - k_wrap):
                grad[i] += dx * h[i + k_wrap]
                grad[i + k_wrap] += dx * h[i]
        
        direction = -grad
        step_size = 0.1
        new_h = h + step_size * direction
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        scaling_factor = (n_points / 2) / new_sum if new_sum != 0 else 1.0

        new_h = new_h * scaling_factor
        
        if np.random.rand() < perturbation_prob:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            new_h = h + noise
            new_h = np.clip(new_h, 0.0, 1.0)
            new_sum = np.sum(new_h)
            scaling_factor = (n_points / 2) / new_sum if new_sum != 0 else 1.0
            new_h = new_h * scaling_factor
            new_h = np.clip(new_h, 0.0, 1.0)

        if new_c5 < best_c5:

    assert (h >= 0).all() and (h <= 1).all()
    assert np.isclose(sum(h), n_points / 2, atol=1e-6)
    assert np.isfinite(final_c5)
    
    return (h, final_c5, n_points)
