import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.random.rand(70)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values / np.sum(initial_h_values) * (n_points / 2)
    best_h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    best_time = 0
    total_time = 0
    
    def perturb(h, scale=0.1):
        noise = np.random.normal(0, scale, len(h))
        h = h + noise
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    for _ in range(50):
        if total_time >= budget_s:
            break
        start_time = time.time()
        h = best_h.copy()
        h = perturb(h)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        
        x = np.arange(n_points)
        h_vec = cp.Variable(n_points)
        constraints = []
        for k in range(1, n_points):
            lag = (x + k) % n_points
            expr = h_vec * (1 - h_vec[lag])
            constraints.append(cp.sum(expr) * dx <= 1e6)
        
        obj = cp.Minimize(1e6)
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == "OPTIMAL":
                best_h = h_vec.value
                best_h = np.clip(best_h, 0, 1)
                best_h = best_h / np.sum(best_h) * (n_points / 2)
                best_c5 = evaluate_erdos_solution(best_h, dx)
                total_time += time.time() - start_time
                best_time = time.time()
        except:
            pass
        
        if total_time >= budget_s:
            break
    
    if best_c5 < 0.3807:
        pass
    
    return best_h, best_c5, n_points
