import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    if n_points == 0:
        n_points = 450
        h = 0.5 * (1 + np.random.uniform(-1, 1, n_points))
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = initial_h
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * 2.0 / n_points
        return c5
    
    def mutate_targeted(h, t, max_k, sigma):
        corr = np.correlate(h, 1 - h, mode="full")
        max_k_idx = np.argmax(corr)
        lags = np.arange(max_k_idx - 10, max_k_idx + 11)
        lags = np.clip(lags, 0, len(corr) - 1)
        lags = np.unique(lags)
        indices = np.random.choice(lags, min(10, len(lags)), replace=False)
        for lag in indices:
            h_shift = np.roll(h, lag)
            contribution = h_shift * (1 - h)
            weighted_contrib = contribution * h
            grad = np.gradient(weighted_contrib)
            delta = -sigma * grad
            h += delta
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def mutate_global(h, t, sigma_global):
        indices = np.random.choice(range(n_points), 5, replace=False)
        for i in indices:
            h[i] += np.random.normal(0, sigma_global)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def evaluate_fitness(h):
        return compute_c5(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_time = time.time()
    iteration = 0
    for _ in range(int(budget_s / 0.1)):
        if time.time() - best_time > 50:
            break
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_time = time.time()
        corr = np.correlate(h, 1 - h, mode="full")
        max_k_idx = np.argmax(corr)
        sigma = 0.1 / (1 + iteration / 500)
        h1 = mutate_targeted(h.copy(), iteration, max_k_idx, sigma)
        h2 = mutate_global(h.copy(), iteration, 0.2 * np.exp(-iteration / 300))
        c1 = compute_c5(h1)
        c2 = compute_c5(h2)
        if c1 < c2:
            h = h1
        else:
            h = h2
        if iteration % 200 == 0:
            if np.allclose(np.sum(h), n_points / 2, atol=1e-6):
                pass
            else:
                h = h * (n_points / 2) / np.sum(h)
        iteration += 1
    
    h = best_h
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    c5 = compute_c5(h)
    return h, c5, n_points
