```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    initial_h_values = np.load('initial_h.npy') if 'initial_h_values' in kwargs else None
    
    if initial_h_values is not None:
        h = np.clip(initial_h_values, 0.0, 1.0)
        h = h * n_points / 2.0 / np.sum(h)
    else:
        h = np.random.uniform(0.0, 1.0, n_points)
        h = h * n_points / 2.0 / np.sum(h)
    
    dx = 2.0 / n_points
    best_c5 = np.inf
    best_h = h.copy()
    
    T = 100.0
    alpha = 0.995
    max_iter = int(budget_s / 0.1)
    
    for t in range(max_iter):
        start_time = time.time()
        idxs = np.random.choice(n_points, 5, replace=False)
        noise = np.random.normal(0, 0.05 * np.exp(-t / 200), 5)
        new_h = h.copy()
        new_h[idxs] += noise
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = new_h * n_points / 2.0 / np.sum(new_h)
        
        corr = np.correlate(new_h, 1.0 - new_h, mode="full")
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = new_h.copy()
        
        delta = current_c5 - best_c5
        if delta > 0:
            prob = np.exp(-delta / T)
            if np.random.rand() < prob:
                h = new_h.copy()
        
        T *= alpha
        if time.time() - start_time > 0.1:
            break
    
    return best_h, best_c5, n_points
```
