import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use 100 points for faster optimization
    dx = 2.0 / n_points
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_vals = np.array(initial_h_values, dtype=float)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    else:
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    
    best_h_vals = h_vals.copy()
    best_c5 = compute_c5(h_vals, dx)
    start_time = time.time()
    
    # Optimization loop
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Compute all lags
        correlations = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5_values = correlations * dx
        max_index = np.argmax(c5_values)
        max_c5 = c5_values[max_index]
        
        # Optimization for worst lag
        def objective(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            h = (h - np.min(h)) / (np.max(h) - np.min(h)) * (n_points / 2)
            h = h / np.sum(h) * n_points / 2
            lag = max_index
            corr = np.correlate(h, 1 - h, mode='full')
            return corr[lag] * dx
        
        # Define bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        
        # Use L-BFGS-B solver
        result = opt.minimize(objective, h_vals, method='L-BFGS-B', bounds=bounds, constraints=cons)
        if result.success:
            h_vals = result.x
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = (h_vals - np.min(h_vals)) / (np.max(h_vals) - np.min(h_vals)) * (n_points / 2)
            h_vals = h_vals / np.sum(h_vals) * n_points / 2
            current_c5 = compute_c5(h_vals, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h_vals = h_vals.copy()
        else:
            # Try SLSQP as fallback
            result = opt.minimize(objective, h_vals, method='SLSQP', bounds=bounds, constraints=cons)
            if result.success:
                h_vals = result.x
                h_vals = np.clip(h_vals, 0, 1)
                h_vals = (h_vals - np.min(h_vals)) / (np.max(h_vals) - np.min(h_vals)) * (n_points / 2)
                h_vals = h_vals / np.sum(h_vals) * n_points / 2
                current_c5 = compute_c5(h_vals, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h_vals = h_vals.copy()
    
    # Final validation
    best_h_vals = np.clip(best_h_vals, 0, 1)
    best_h_vals = (best_h_vals - np.min(best_h_vals)) / (np.max(best_h_vals) - np.min(best_h_vals)) * (n_points / 2)
    best_h_vals = best_h_vals / np.sum(best_h_vals) * n_points / 2
    final_c5 = compute_c5(best_h_vals, dx)
    
    return (best_h_vals, final_c5, n_points)

def compute_c5(h_vals, dx):
    correlations = np.correlate(h_vals, 1 - h_vals, mode='full')
    return np.max(correlations) * dx
