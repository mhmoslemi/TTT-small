import numpy as np
import random
import time
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points)
    initial_h = (initial_h * (n_points / 2)) / np.sum(initial_h)
    h_vals = initial_h.copy()
    best_h = h_vals.copy()
    best_c5 = compute_c5(h_vals, n_points, dx)
    
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    mutation_amp = 0.05
    
    for gen in range(generations):
        start_time = time.time()
        population = []
        for _ in range(population_size):
            h = h_vals.copy()
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    h[i] += np.random.normal(0, mutation_amp)
                    h[i] = np.clip(h[i], 0, 1)
            sum_h = np.sum(h)
            h = h * (n_points / 2) / sum_h if sum_h != 0 else h
            population.append(h)
        
        fitness = []
        for h in population:
            c5 = compute_c5(h, n_points, dx)
            fitness.append(c5)
        
        population = [p for _, p in sorted(zip(fitness, population))]
        best_h = population[0]
        best_c5 = fitness[0]
        
        if time.time() - start_time > budget_s - 10:
            break
    
    final_h = best_h
    final_c5 = compute_c5(final_h, n_points, dx)
    
    return (final_h, final_c5, n_points)

def compute_c5(h_vals, n_points, dx):
    h = np.array(h_vals)
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
