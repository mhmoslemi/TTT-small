def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    improvement_threshold = 1e-4
    last_improvement = 0
    epsilon = 1e-8

    while elapsed_time < budget_s:
        start_time = time.time()
        
        # Compute cross-correlation
        cross_corr = np.correlate(h, 1 - h, mode="full") * dx
        max_index = np.argmax(cross_corr)
        max_c5 = cross_corr[max_index]

        # Compute gradient for dominant lag
        h_perturb = h.copy()
        h_perturb[max_index] += epsilon
        c5_forward = evaluate_erdos_solution(h_perturb)
        h_perturb[max_index] -= 2 * epsilon
        c5_backward = evaluate_erdos_solution(h_perturb)
        grad = (c5_forward - c5_backward) / (2 * epsilon)

        # Update h
        h -= alpha * grad
        h = np.clip(h, 0, 1)
        h /= np.sum(h) * dx
        h *= target_sum

        # Evaluate
        current_c5 = evaluate_erdos_solution(h)

        # Update best solution
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            last_improvement = 0
        else:
            last_improvement += 1

        # Manage step size and early stopping
        if last_improvement >= early_stop:
            alpha *= step_decay
            if alpha < 1e-6:
                break
        elif iteration >= max_iterations:
            break

        # Update time
        elapsed_time = time.time() - start_time
        iteration += 1
