import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 500  # Optimal number of points for balance between accuracy and speed
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Initialize with provided initial values or a default construction
    if initial_h_values is not None:
        h = np.clip(initial_h_values[:n_points], 0, 1)
    else:
        # Default construction: alternating 1 and 0 with appropriate scaling
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = np.clip(h, 0, 1)
    
    # Ensure the sum constraint is met
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h if sum_h != 0 else h
    
    # Define the objective function
    def objective(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        h = h * (n_points / 2) / sum_h
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Define the constraint: sum of h is n_points/2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    constraints = [{'type': 'eq', 'fun': constraint}]
    
    # Optimization using SLSQP
    result = opt.minimize(objective, h, method='SLSQP', constraints=constraints, bounds=[(0, 1)]*n_points, options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8})
    
    # Extract the best solution found
    h_best = result.x
    c5_bound = objective(h_best)
    
    # Ensure the solution remains within the bounds and meets the sum constraint
    h_best = np.clip(h_best, 0, 1)
    sum_h = np.sum(h_best)
    if sum_h != 0:
        h_best = h_best * (n_points / 2) / sum_h
    
    # Validate the result
    assert np.all(h_best >= 0) and np.all(h_best <= 1), "Values out of bounds"
    assert np.isclose(np.sum(h_best), n_points / 2), "Sum constraint not met"
    
    return (h_best, c5_bound, n_points)
