import numpy as np
import random
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Parameters
    n_points = 500
    population_size = 50
    max_generations = 500
    mutation_rate = 0.1
    tournament_size = 5
    crossover_rate = 0.2
    mutation_decay = 0.99
    survival_rate = 0.8
    early_stopping = 200
    
    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=np.float64)
        initial_h = (initial_h * n_points / 2) / np.sum(initial_h)
        population = [initial_h.copy() for _ in range(population_size)]
    else:
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    
    # Evaluate population
    def evaluate(h):
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if abs(sum_h - n_points / 2) > 1e-6:
            return float('inf')
        h_1 = 1 - h
        corr = correlate(h, h_1, mode='full')
        corr = np.abs(corr)
        return np.max(corr) / (2 / n_points)
    
    def rescale(h):
        h = h.copy()
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1
        h *= scale
        h = np.clip(h, 0, 1)
        return h
    
    def clip_and_rescale(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1
        h *= scale
        h = np.clip(h, 0, 1)
        return h
    
    def mutate(h, mutation_rate):
        indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
        h[indices] += np.random.uniform(-0.1, 0.1, len(indices))
        return clip_and_rescale(h)
    
    def crossover(parent1, parent2):
        child = (parent1 + parent2) / 2
        return clip_and_rescale(child)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate all individuals
        fitnesses = [evaluate(ind) for ind in population]
        
        # Find best individual
        best_idx = np.argmin(fitnesses)
        current_c5 = fitnesses[best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[best_idx].copy()
        
        # Select parents
        parents = []
        for _ in range(population_size // 2):
            tournament = np.random.choice(population_size, tournament_size, replace=False)
            winner = tournament[np.argmin([fitnesses[i] for i in tournament])]
            parents.append(population[winner])
        
        # Create offspring
        offspring = []
        for i in range(len(parents) // 2):
            parent1, parent2 = parents[2 * i], parents[2 * i + 1]
            if np.random.rand() < crossover_rate:
                child = crossover(parent1, parent2)
            else:
                child = np.random.uniform(0, 1, n_points)
            child = mutate(child, mutation_rate)
            offspring.append(child)
        
        # Replace population
        population = np.array(population)
        new_population = np.array(offspring)
        population = np.vstack([population, new_population])
        population = population[np.argsort(fitnesses)][:-population_size]
        
        # Early stopping
        if gen > early_stopping and np.abs(current_c5 - best_c5) < 1e-6:
            break
    
    # Ensure best_h is valid
    best_h = clip_and_rescale(best_h)
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    assert np.isclose(np.sum(best_h), n_points / 2, atol=1e-6)
    
    return (best_h.tolist(), best_c5, n_points)
