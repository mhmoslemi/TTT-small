import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with initial_h_values if available
    if 'initial_h_values' in globals():

    else:
        # If no initial vector, create a simple construction
        n_points = 400
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Clip h to [0, 1] and normalize
    h = np.clip(h, 0.0, 1.0)
    h = h * (n_points / 2) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    
    def evaluate_overlaps(h):
        # Compute the maximum overlap integral
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    best_h = h.copy()
    best_c5 = evaluate_overlaps(best_h)
    
    # Optimization parameters
    learning_rate = 0.01
    decay_rate = 0.999
    max_iterations = 6000
    iteration = 0
    time_start = time.time()
    
    # Main optimization loop
    for _ in range(max_iterations):
        elapsed = time.time() - time_start
        if elapsed > budget_s:
            break
        
        # Compute overlaps
        corr = np.correlate(h, 1 - h, mode="full") * dx
        k_max = np.argmax(corr)
        
        # Compute gradient for the dominant lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        
        # Update h
        h -= learning_rate * grad
        
        # Clip and normalize
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (n_points / 2) / sum_h
        
        # Evaluate and update best
        new_c5 = evaluate_overlaps(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            iteration = 0
        else:
            iteration += 1
        
        # Decay learning rate periodically
        if iteration >= 100:
            learning_rate *= decay_rate
            iteration = 0
        
        # Perturb occasionally to escape local minima
        if np.random.rand() < 0.01:
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            if sum_h > 0:
                h = h * (n_points / 2) / sum_h
    
    # Final check and return
    h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * (n_points / 2) / sum_h
    
    # Validate and return
    assert np.all(h >= 0) and np.all(h <= 1)
    assert np.isclose(np.sum(h), n_points / 2)
    
    return h, best_c5, n_points
