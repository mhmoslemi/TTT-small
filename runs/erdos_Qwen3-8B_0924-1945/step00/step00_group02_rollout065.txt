import numpy as np
import numpy.random as npr
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in kwargs else None
    
    if initial_h is not None:
        h = initial_h.copy()
        h = np.clip(h, 0, 1)
        h = h * n_points / 2.0 / np.sum(h)
    else:
        h = 0.5 * (1 + npr.uniform(-1, 1, n_points))
        h = np.clip(h, 0, 1)
        h = h * n_points / 2.0 / np.sum(h)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    c5_best = compute_c5(h)
    h_best = h.copy()
    
    start_time = time.time()
    
    for _ in range(1200):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag = np.argmax(corr)
        overlap = h * (1 - np.roll(h, max_lag))
        indices = np.argsort(overlap)[::-1]
        threshold = np.max(overlap) * 0.8
        target_idx = indices[overlap > threshold][:int(len(h) * 0.10)]
        
        candidates = []
        for idx in target_idx:
            h_candidate = h.copy()
            h_candidate[idx] = 1 - h_candidate[idx]
            h_candidate = np.clip(h_candidate, 0, 1)
            total = np.sum(h_candidate)
            if abs(total) < 1e-8:
                continue
            h_candidate = h_candidate * n_points / 2.0 / total
            candidates.append(h_candidate)
        
        for idx in target_idx:
            h_candidate = h.copy()
            h_candidate[idx] += np.random.uniform(-0.05, 0.05)
            h_candidate = np.clip(h_candidate, 0, 1)
            total = np.sum(h_candidate)
            if abs(total) < 1e-8:
                continue
            h_candidate = h_candidate * n_points / 2.0 / total
            candidates.append(h_candidate)
        
        for candidate in candidates:
            c5_candidate = compute_c5(candidate)
            if c5_candidate < c5_best:
                c5_best = c5_candidate
                h_best = candidate.copy()
        
        if np.allclose(h, h_best, atol=1e-6):
            break
    
    return h_best, c5_best, n_points
