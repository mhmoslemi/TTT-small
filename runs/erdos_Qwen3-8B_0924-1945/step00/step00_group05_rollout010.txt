        overlaps = np.correlate(h, 1 - h, mode="full") * dx

        grad = np.zeros(n_points)

        for i in range(n_points):

            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        
        # Update h
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)

        h = h * (n_points / 2) / sum_h if sum_h != 0 else h

        if new_c5 < best_c5:

            best_c5 = new_c5
            iteration = 0
        else:
            iteration += 1
        
        # Decay learning rate periodically
        if iteration % 100 == 0:
            learning_rate *= decay_rate

        if iteration % 500 == 0:
            noise = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + noise, 0, 1)
            sum_h = np.sum(h)
            h = h * (n_points / 2) / sum_h if sum_h != 0 else h
        
        # Early stopping if no improvement for 1000 iterations
        if iteration > 1000 and np.abs(new_c5 - best_c5) < 1e-5:
