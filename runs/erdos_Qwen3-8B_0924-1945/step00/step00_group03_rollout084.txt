import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Predefined initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Fallback: Construct an initial random valid solution
        n_points = 70
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Evaluate the initial C5
    def evaluate_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Parameters for optimization
    T0 = 1.0 / (1 + best_c5)
    Tmax = int(0.1 * budget_s)
    beta = 0.75
    gamma = 0.65
    lambda0 = 0.15 / dx
    
    start_time = time.time()
    for t in range(int(budget_s * 0.75)):
        if time.time() - start_time > budget_s:
            break
        if t % 100 == 0 and t > 0:
            lambda0 *= 0.8 if t < (budget_s * 0.75) // 2 else 0.95
        
        # Generate perturbation
        noise = np.random.normal(0, lambda0, n_points)
        perturbed = h + noise
        
        # Clip to [0, 1]
        perturbed = np.clip(perturbed, 0, 1)
        
        # Adjust sum constraint
        current_sum = np.sum(perturbed)
        delta = (n_points / 2 - current_sum) / (n_points * 0.95)
        perturbed += delta * np.random.choice([-1, 1], size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed) if np.sum(perturbed) != 0 else perturbed
        
        # Evaluate new solution
        new_c5 = evaluate_c5(perturbed, dx)
        
        # Acceptance criteria
        if new_c5 < best_c5 or (np.random.rand() < np.exp(-(new_c5 - best_c5) / (T0 * (1 - t / Tmax) ** (1 / beta)))):
            h = perturbed.copy()
            best_c5 = new_c5
            best_h = h.copy()
        
        # Cool temperature
        if t == Tmax:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    # Validate constraints
    assert np.all((best_h >= 0) & (best_h <= 1))
    assert np.isclose(np.sum(best_h), n_points / 2, atol=1e-6)
    
    return (best_h.tolist(), final_c5, n_points)
