    def evaluate(h):
        h = np.clip(h, 0, 1)

        correlation = correlate(h, 1 - h, mode='full')

        h = h.copy()
        idx = np.random.choice(n_points, int(mutation_rate * n_points), replace=False)

        return child

        for _ in range(49):
            h = mutate(best_h, mutation_rate=0.1)
            population.append(h)
        population = sorted(population, key=lambda x: evaluate(x))
        top = population[:20]
        new_population = []
        for i in range(50):
            if i < 20:
                parent1 = random.choice(top)
                parent2 = random.choice(top)
                child = crossover(parent1, parent2)
                new_population.append(child)

                new_population.append(h)
        population = new_population

            c5 = evaluate(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        h_values = best_h.copy()
    h_values = np.clip(h_values, 0, 1)
    sum_h = np.sum(h_values)
    if np.abs(sum_h - total_mass) > 1e-6:
        scale = total_mass / sum_h
        h_values *= scale
    return (h_values, best_c5, n_points)
