import numpy as np
import random
import time
import scipy.optimize as opt
import scipy.signal as signal

# Pre-imported helpers
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 500
    dx = 2.0 / n_points
    h = np.array(initial_h_values[:n_points], dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (n_points / 2) / h_sum
    h = np.clip(h, 0.0, 1.0)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    POP_SIZE = 50
    MUTATION_NOISE = 0.05
    MUTATION_RATE = 0.1
    N_SELECT = 20
    MAX_STAGNATION = 50
    STAGNATION_COUNT = 0
    MUTATION_NOISE_DECAY = 0.95
    MUTATION_NOISE_GROWTH = 1.1
    
    def evaluate_individual(h):
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (n_points / 2) / h_sum
        h = np.clip(h, 0.0, 1.0)
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(overlaps)
        return c5
    
    def mutate(h):
        noise = np.random.normal(0, MUTATION_NOISE, size=n_points)
        h = h + noise
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (n_points / 2) / h_sum
        h = np.clip(h, 0.0, 1.0)
        return h
    
    def crossover(h1, h2):
        split = np.random.randint(1, n_points - 1)
        child = np.concatenate((h1[:split], h2[split:]))
        child = np.clip(child, 0.0, 1.0)
        h_sum = np.sum(child)
        if h_sum != 0:
            child = child * (n_points / 2) / h_sum
        child = np.clip(child, 0.0, 1.0)
        return child
    
    def tournament_selection(population):
        selected = []
        for _ in range(N_SELECT):
            candidates = np.random.choice(population, size=5, replace=False)
            best = min(candidates, key=evaluate_individual)
            selected.append(best)
        return selected
    
    population = [mutate(h.copy()) for _ in range(POP_SIZE)]
    best_population = [h.copy() for _ in range(POP_SIZE)]
    population_fitness = [evaluate_individual(h) for h in population]
    population_fitness = np.array(population_fitness)
    for i in range(POP_SIZE):
        best_population[i] = population[i]
    
    while time.time() - start_time < budget_s:
        selected = tournament_selection(population)
        new_population = []
        for _ in range(POP_SIZE):
            parent1, parent2 = np.random.choice(selected, size=2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
        population_fitness = [evaluate_individual(h) for h in population]
        population_fitness = np.array(population_fitness)
        
        for i in range(POP_SIZE):
            if population_fitness[i] < best_c5:
                best_c5 = population_fitness[i]
                best_h = population[i].copy()
                STAGNATION_COUNT = 0
            else:
                STAGNATION_COUNT += 1
        
        if STAGNATION_COUNT >= MAX_STAGNATION:
            MUTATION_NOISE *= MUTATION_NOISE_DECAY
            STAGNATION_COUNT = 0
        
        if MUTATION_NOISE < 0.001:
            MUTATION_NOISE = 0.001
        
        if best_c5 < 0.38085:
            break
    
    # Final check
    best_h = np.clip(best_h, 0.0, 1.0)
    h_sum = np.sum(best_h)
    if h_sum != 0:
        best_h = best_h * (n_points / 2) / h_sum
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (best_h, final_c5, n_points)
