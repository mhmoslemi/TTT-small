import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Use initial_h_values if provided
    if 'initial_h_values' in kwargs:
        h = kwargs['initial_h_values'].copy()
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure h is within [0, 1] and satisfies the constraint
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    # Parameters
    learning_rate = 0.05
    learning_rate_decay = 0.9995
    noise_scale = 0.05
    noise_freq = 10
    max_iter = int(budget_s / 1.0)
    improvement_threshold = 1e-5
    early_stop_iter = 50
    
    for iter_num in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Compute full correlation
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        current_c5 = np.max(correlate_vals)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early stopping if improvement is too small
        if iter_num > 0 and (best_c5 - current_c5) < improvement_threshold:
            if iter_num - last_improvement > early_stop_iter:
                break
            last_improvement = iter_num
        
        # Target the worst lag
        worst_lag = np.argmax(correlate_vals)
        n_lags = len(correlate_vals)
        
        # Compute gradient for the worst lag
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * dx
        
        # Update h
        h = h - learning_rate * gradients
        learning_rate *= learning_rate_decay
        
        # Clip and normalize
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Add structured noise periodically
        if iter_num % noise_freq == 0:
            noise = noise_scale * np.random.normal(0, 1, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        last_improvement = iter_num
    
    # Final check
    correlate_vals = np.correlate(best_h, 1 - best_h, mode="full") * dx
    final_c5 = np.max(correlate_vals)
    
    return (best_h, final_c5, n_points)
