import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 47  # as provided by initial_h_values
    initial_h = initial_h_values

    else:
        h_vals = np.random.rand(n_points)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
    dx = 2.0 / n_points
    best_c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * dx
    best_h = h_vals.copy()
    max_iterations = 100
    no_improvement = 0
    start_time = time.time()
    
    for _ in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        # Compute correlations
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5_vals = corr * dx
        max_c5 = np.max(c5_vals)
        if np.isclose(max_c5, best_c5):
            no_improvement += 1
            if no_improvement >= 5:
                break
        else:
            no_improvement = 0
            best_c5 = max_c5
            best_h = h_vals.copy()
        
        # Choose lag with maximum overlap
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        if lag < 0:
            lag += n_points
        # Compute the objective for this lag
        def objective(x):
            h_new = x
            h_clamped = np.clip(h_new, 0, 1)
            total = np.sum(h_clamped)
            # Adjust to meet the sum constraint
            if np.isclose(total, n_points / 2):
                pass
            elif total < n_points / 2:
                diff = (n_points / 2) - total
                # Scale up the smallest values
                h_clamped = h_clamped / (np.sum(h_clamped) / (n_points / 2))
            else:
                diff = total - (n_points / 2)
                # Scale down the largest values
                h_clamped = h_clamped / (np.sum(h_clamped) / (n_points / 2))
            # Compute overlap for the selected lag
            k = lag
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h_clamped[i] * (1 - h_clamped[j])
            return overlap
        
        # Set up bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        
        # Run optimization
        try:
            result = optimize.minimize(objective, h_vals, bounds=bounds, constraints=cons, method='SLSQP')
            h_vals = result.x
            # Clamp values and normalize
            h_vals = np.clip(h_vals, 0, 1)
            total = np.sum(h_vals)
            if not np.isclose(total, n_points / 2):
                h_vals = h_vals / total * (n_points / 2)
            # Update best solution
            corr = np.correlate(h_vals, 1 - h_vals, mode='full')
            c5_vals = corr * dx
            current_c5 = np.max(c5_vals)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_vals.copy()
        except:
            # Fallback to previous solution
            h_vals = best_h.copy()
    
    # Final check
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (best_h, final_c5, n_points)
