import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    h = initial_h_values.copy()
    h = h / np.sum(h) * n_points / 2
    h = np.clip(h, 0, 1)
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    n_particles = 20
    particles = np.zeros((n_particles, n_points))
    for i in range(n_particles):
        particles[i] = np.random.rand(n_points)
        particles[i] = particles[i] / np.sum(particles[i]) * n_points / 2
        particles[i] = np.clip(particles[i], 0, 1)
    
    velocities = np.random.rand(n_particles, n_points) * 0.1
    pbest = particles.copy()
    pbest_c5 = np.full(n_particles, np.inf)
    gbest = particles[0]
    gbest_c5 = np.inf
    
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):
            h = particles[i]
            corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            c5 = np.max(corr)
            if c5 < pbest_c5[i]:
                pbest[i] = h.copy()
                pbest_c5[i] = c5
            if c5 < gbest_c5:
                gbest = h.copy()
                gbest_c5 = c5
        
        w = 0.7 + (0.3 * (_ / 100))
        c1, c2 = 1.5, 1.5
        
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = w * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] / np.sum(particles[i]) * n_points / 2
        
        if gbest_c5 < best_c5:
            best_c5 = gbest_c5
            best_h = gbest.copy()
    
    h = best_h
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2
    h = np.clip(h, 0, 1)
    best_c5 = evaluate_erdos_solution(h)
    return (h, best_c5, n_points)
