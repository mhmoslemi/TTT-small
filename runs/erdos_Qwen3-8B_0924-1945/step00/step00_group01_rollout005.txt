import numpy as np
import random
import time
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100  # use a manageable number of points for fast optimization
    dx = 2.0 / n_points
    
    # Function to evaluate C5
    def evaluate_c5(h):
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Function to ensure the integral constraint
    def enforce_integral(h):
        total = np.sum(h)
        if np.isclose(total, n_points / 2, atol=1e-6):
            return h
        return h * (n_points / 2) / total
    
    # Function to clip and normalize values to [0,1] and satisfy integral constraint
    def clip_and_normalize(h):
        clipped = np.clip(h, 0, 1)
        total = np.sum(clipped)
        if np.isclose(total, 0):
            return np.zeros_like(clipped)
        return clipped * (n_points / 2) / total
    
    # Function to generate initial population
    def generate_initial_population(pop_size):
        if 'initial_h_values' in globals():
            initial = np.array(initial_h_values, dtype=float)
            initial = initial * (n_points / 2) / np.sum(initial)
            initial = np.clip(initial, 0, 1)
            initial = clip_and_normalize(initial)
            population = np.array([initial + np.random.randn(n_points) * 0.05 for _ in range(pop_size)])
        else:
            population = np.array([np.random.rand(n_points) for _ in range(pop_size)])
            for i in range(pop_size):
                population[i] = clip_and_normalize(population[i])
        return population
    
    # Genetic algorithm parameters
    pop_size = 50
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initial population
    population = generate_initial_population(pop_size)
    best_h = population[0]
    best_c5 = evaluate_c5(best_h)
    
    # Track best solution
    best_solutions = []
    
    # Genetic algorithm loop
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = np.array([evaluate_c5(ind) for ind in population])
        
        # Track best solution
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        best_solutions.append((best_h, best_c5))
        
        # Selection: tournament selection
        tournament = np.random.choice(pop_size, size=tournament_size, replace=False)
        tournament_fitness = fitness[tournament]
        best_tournament_idx = tournament[np.argmin(tournament_fitness)]
        parent1 = population[best_tournament_idx].copy()
        
        # Second parent
        tournament2 = np.random.choice(pop_size, size=tournament_size, replace=False)
        tournament_fitness2 = fitness[tournament2]
        best_tournament_idx2 = tournament2[np.argmin(tournament_fitness2)]
        parent2 = population[best_tournament_idx2].copy()
        
        # Crossover
        if np.random.rand() < crossover_rate:
            crossover_point = np.random.randint(1, n_points - 1)
            offspring = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        else:
            offspring = parent1.copy()
        
        # Mutation
        if np.random.rand() < mutation_rate:
            mutation_strength = 0.05
            noise = np.random.randn(n_points) * mutation_strength
            offspring += noise
            offspring = clip_and_normalize(offspring)
        
        # Replace worst individual
        worst_idx = np.argmax(fitness)
        population[worst_idx] = offspring
        
        # Early stopping if best solution stagnates
        if len(best_solutions) > 10 and np.all(np.abs(np.array(best_solutions[-1][1]) - np.array(best_solutions[-10:][1])) < 1e-6):
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    
    # Ensure the solution is valid
    best_h = clip_and_normalize(best_h)
    final_c5 = evaluate_c5(best_h)
    
    # Return the result
    return (best_h, final_c5, n_points)
