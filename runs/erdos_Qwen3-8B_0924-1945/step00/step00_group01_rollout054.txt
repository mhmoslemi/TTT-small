import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 47 if 'initial_h_values' in globals() else 50
    if 'initial_h_values' in globals():
        h_vals = initial_h_values.copy()
        dx = 2.0 / n_points
        sum_h = np.sum(h_vals) * dx
        scale = (n_points / 2) / sum_h
        h_vals = np.clip(h_vals * scale, 0, 1)
    else:
        h_vals = np.random.rand(n_points)
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
    
    dx = 2.0 / n_points
    max_overlap = np.inf
    best_h_vals = h_vals.copy()
    best_c5 = np.inf
    
    def compute_correlation(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return corr * dx
    
    for _ in range(100):
        corr = compute_correlation(h_vals)
        max_corr = np.max(corr)
        if max_corr < max_overlap:
            max_overlap = max_corr
            best_h_vals = h_vals.copy()
            best_c5 = max_corr
            if max_corr < 0.3807:
                break
        
        k_index = np.argmax(corr)
        k = k_index - len(h_vals) + 1
        k = np.clip(k, -len(h_vals) + 1, len(h_vals) - 1)
        
        def objective(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if np.abs(h_sum - n_points / 2) > 1e-6:
                return np.inf
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            return overlap
        
        bounds = Bounds(0, 1)
        result = opt.minimize(
            objective,
            h_vals,
            bounds=bounds,
            method='SLSQP',
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        )
        
        if result.success and result.fun < max_overlap:
            h_vals = result.x.copy()
        else:
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        
        if np.sum(np.abs(h_vals - best_h_vals)) < 1e-6:
            break
    
    final_corr = compute_correlation(h_vals)
    final_c5 = np.max(final_corr)
    return (h_vals, final_c5, n_points)
