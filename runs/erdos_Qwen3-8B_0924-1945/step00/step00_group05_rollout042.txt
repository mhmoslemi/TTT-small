```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h):
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    initial_guess = np.array(initial_h_values) if initial_h_values is not None else np.random.rand(n_points)

    # Normalize initial guess
    initial_guess = initial_guess * (target_sum / np.sum(initial_guess)) if np.sum(initial_guess) != 0 else initial_guess

    result = opt.minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds, tol=1e-6, options={'maxiter': 100})
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, n_points
```
