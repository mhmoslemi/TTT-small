import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Assuming initial_h_values is accessible

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def clip_and_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.zeros_like(h)
        scaling_factor = target_sum / sum_clipped
        return h_clipped * scaling_factor

    def evaluate_erdos_solution(h):
        h_clipped = np.clip(h, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def objective(h):
        return evaluate_erdos_solution(h)

    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    h0 = clip_and_rescale(initial_h_values)
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    elapsed_time = 0

    if n_points < 100:
        result = optimize.minimize(objective, h0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-9, 'maxiter': 100})
        best_h = result.x
        best_c5 = evaluate_erdos_solution(best_h)
        return (best_h, best_c5, n_points)

    population_size = 50
    population = [clip_and_rescale(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]

    population = [clip_and_rescale(h) for h in population]
    population = [clip_and_rescale(h) for h in population]
    population = [clip_and_rescale(h) for h in population]
    population = [clip_and_rescale(h) for h in population]
    population = [clip_and_rescale(h) for h in population]
    population = [clip_and_rescale(h) for h in population]
    population = [clip_and_rescale(h) for h in population]

    for _ in range(int(budget_s / 0.1)):
        if elapsed_time >= budget_s:
            break
        new_population = []
        for i in range(population_size):
            idxs = np.random.choice(population_size, 3, replace=False)
            a, b, c = population[idxs]
            mut = a + np.random.uniform(-0.05, 0.05, n_points) * a
            mut = clip_and_rescale(mut)
            new_population.append(mut)
        population = new_population
        population.sort(key=lambda h: -evaluate_erdos_solution(h))
        best_h = population[0]
        best_c5 = evaluate_erdos_solution(best_h)
        elapsed_time += 0.1

    return (best_h, best_c5, n_points)
