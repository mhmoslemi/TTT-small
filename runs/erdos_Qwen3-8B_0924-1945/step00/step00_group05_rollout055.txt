import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in globals() else np.random.rand(70) * 2 - 1
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
    best_h = h.copy()
    best_c5 = np.inf
    time_start = time.time()

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr * dx)
        return c5

    def objective_with_constraints(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr * dx)
        return c5

    def evaluate_candidate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr * dx)
        return c5

    def constraint_sum(h_flat):
        return np.sum(h_flat.reshape(n_points)) - target_sum

    def run_optimization():
        nonlocal best_h, best_c5, time_start
        h_flat = h.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            objective,
            h_flat,
            method='L-BFGS-B',
            bounds=bounds,
            options={'ftol': 1e-7, 'maxiter': 100}
        )
        if result.success:
            h_candidate = result.x.copy()
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (target_sum / np.sum(h_candidate)) if np.sum(h_candidate) != 0 else h_candidate
            c5_candidate = evaluate_candidate(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
        return best_c5

    def run_evolution():
        nonlocal best_h, best_c5, time_start
        population_size = 20
        mutation_rate = 0.1
        mutation_scale = 0.1
        population = []
        for _ in range(population_size):
            h_flat = best_h.copy()
            h_flat += np.random.normal(0, mutation_scale, n_points)
            h_flat = np.clip(h_flat, 0, 1)
            h_flat = h_flat * (target_sum / np.sum(h_flat)) if np.sum(h_flat) != 0 else h_flat
            population.append(h_flat)
        for _ in range(100):
            if time.time() - time_start > budget_s - 10:
                break
            if len(population) < 2:
                break
            parent1 = np.random.choice(population)
            parent2 = np.random.choice(population)
            child = (parent1 + parent2) / 2
            child += np.random.normal(0, mutation_scale, n_points)
            child = np.clip(child, 0, 1)
            child = child * (target_sum / np.sum(child)) if np.sum(child) != 0 else child
            population.append(child)
            population = np.random.choice(population, size=population_size, replace=False)
            new_population = []
            for h_flat in population:
                c5 = evaluate_candidate(h_flat)
                new_population.append(h_flat)
                if c5 < best_c5:
                    best_h = h_flat.copy()
                    best_c5 = c5
            population = new_population

    for _ in range(3):
        if time.time() - time_start > budget_s - 10:
            break
        run_optimization()
        run_evolution()

    h_values = best_h
    c5_bound = best_c5
    return h_values, c5_bound, n_points
