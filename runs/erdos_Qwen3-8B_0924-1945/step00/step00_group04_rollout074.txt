import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 42
    initial_h = initial_h_values
    target_sum = n_points / 2
    
    # Objective function
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    # Constraint: sum(h) = target_sum
    def constraint_sum(h_flat):
        return np.sum(h_flat.reshape(n_points)) - target_sum
    
    # Bounds: 0 <= h[i] <= 1
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    h_initial = initial_h.copy()
    assert np.isclose(np.sum(h_initial), target_sum), "Initial sum not matching target"
    
    # Optimization parameters
    max_iter = 100
    tol = 1e-6
    
    start_time = time.time()
    
    # Run optimization
    result = optimize.minimize(
        fun=objective,
        x0=h_initial,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        method='SLSQP',
        options={'maxiter': max_iter, 'ftol': tol, 'disp': False}
    )
    
    # Check if optimization succeeded
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        # Fallback to initial if optimization failed
        best_h = h_initial
        best_c5 = objective(best_h)
    
    # Ensure constraints are satisfied
    best_h = np.clip(best_h, 0.0, 1.0)
    assert np.isclose(np.sum(best_h), target_sum), "Final sum not matching target"
    
    # Return the best result
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        print("Warning: Optimization exceeded budget time")
    
    return (best_h.tolist(), best_c5, n_points)
