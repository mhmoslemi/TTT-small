import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 450
    if initial_h_values is not None:
        h = np.copy(initial_h_values)
    else:
        h = 0.5 * (1 + np.random.uniform(-1, 1, n_points))
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    h = h * (n_points / 2) / total if total != 0 else h * (n_points / 2)
    best_h = np.copy(h)
    best_c5 = compute_c5(h, n_points)
    start_time = time.time()
    dt = 0.1
    for t in range(0, int((budget_s - (time.time() - start_time)) // dt)):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        candidates = [h.copy()]
        # Targeted mutation
        for i in range(max_k - 10, max_k + 11):
            if 0 <= i < len(corr):
                contribution = np.sum(h * (1 - h[i]))
                delta = -sign(contribution) * 0.1 / (1 + t / 500)
                h_candidate = h.copy()
                h_candidate[i] += delta
                h_candidate = np.clip(h_candidate, 0, 1)
                total_candidate = np.sum(h_candidate)
                h_candidate = h_candidate * (n_points / 2) / total_candidate if total_candidate != 0 else h_candidate * (n_points / 2)
                c5_candidate = compute_c5(h_candidate, n_points)
                candidates.append(h_candidate)
        # Global mutation
        for _ in range(5):
            idx = np.random.randint(0, len(h))
            delta = np.random.normal(0, 0.2 * np.exp(-t / 300))
            h_candidate = h.copy()
            h_candidate[idx] += delta
            h_candidate = np.clip(h_candidate, 0, 1)
            total_candidate = np.sum(h_candidate)
            h_candidate = h_candidate * (n_points / 2) / total_candidate if total_candidate != 0 else h_candidate * (n_points / 2)
            c5_candidate = compute_c5(h_candidate, n_points)
            candidates.append(h_candidate)
        # Select best candidate
        best_candidate = min(candidates, key=compute_c5)
        c5_candidate = compute_c5(best_candidate, n_points)
        if c5_candidate < best_c5:
            best_h = best_candidate
            best_c5 = c5_candidate
        h = best_h
    return (best_h, best_c5, n_points)

def sign(x):
    return np.sign(x)

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * 2.0 / n_points
    return c5
