import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    # Seed for reproducibility
    np.random.seed(seed)
    
    # Initial guess from parent, if available
    global initial_h_values
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 70
    h_values = initial_h_values.copy() if 'initial_h_values' in globals() else np.zeros(n_points)
    
    # Ensure the integral constraint is satisfied
    dx = 2.0 / n_points
    total_integral = np.sum(h_values) * dx
    if not np.isclose(total_integral, 1.0, atol=1e-6):
        h_values = np.random.rand(n_points)
        h_values = (h_values * n_points / 2).astype(np.float64)
    
    # Function to compute C5
    def compute_c5(h):
        nonlocal dx
        h = np.array(h, dtype=np.float64)
        h = np.clip(h, 0.0, 1.0)
        if np.sum(h) * dx != 1.0:
            raise ValueError("Integral constraint violated")
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Function to evaluate and return loss
    def objective(x):
        nonlocal h_values, dx, compute_c5
        h = x.copy()
        h = np.clip(h, 0.0, 1.0)
        if np.sum(h) * dx != 1.0:
            return np.inf
        c5 = compute_c5(h)
        return c5
    
    # Function to evaluate and return loss for gradient-based optimization
    def objective_with_grad(x):
        nonlocal h_values, dx, compute_c5
        h = x.copy()
        h = np.clip(h, 0.0, 1.0)
        if np.sum(h) * dx != 1.0:
            return np.inf, np.zeros_like(x)
        c5 = compute_c5(h)
        return c5, np.zeros_like(x)
    
    # Gradient descent with momentum
    def optimize_h(h_initial, max_iter=500, lr=0.01, momentum=0.9, tol=1e-6, seed=42):
        h = h_initial.copy()
        h = np.clip(h, 0.0, 1.0)
        if np.sum(h) * dx != 1.0:
            raise ValueError("Integral constraint violated")
        v = np.zeros_like(h)
        best_h = h.copy()
        best_c5 = compute_c5(h)
        start_time = time.time()
        for i in range(max_iter):
            # Compute gradient (approximated via finite differences)
            grad = np.zeros_like(h)
            for j in range(len(h)):
                h_plus = h.copy()
                h_plus[j] += 1e-4
                h_plus = np.clip(h_plus, 0.0, 1.0)
                if np.sum(h_plus) * dx != 1.0:
                    grad[j] = 0.0
                else:
                    c5_plus = compute_c5(h_plus)
                    grad[j] = (c5_plus - best_c5) / 1e-4
            # Update
            v = momentum * v - lr * grad
            h = h + v
            h = np.clip(h, 0.0, 1.0)
            # Ensure integral constraint
            if np.sum(h) * dx != 1.0:
                h = h_initial.copy()
                h = np.clip(h, 0.0, 1.0)
                if np.sum(h) * dx != 1.0:
                    raise ValueError("Integral constraint violated")
            # Evaluate
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            # Check for convergence
            if i % 10 == 0:
                print(f"Iteration {i}, C5: {best_c5:.6f}")
            if time.time() - start_time > budget_s:
                break
        return best_h, best_c5
    
    # Optimizing the initial guess
    h_opt, c5_opt = optimize_h(h_values, max_iter=500, lr=0.01, momentum=0.9, tol=1e-6, seed=42)
    
    # Final validation
    assert (h_opt >= 0).all() and (h_opt <= 1).all()
    assert np.isclose(np.sum(h_opt) * dx, 1.0, atol=1e-6)
    final_c5 = compute_c5(h_opt)
    
    return h_opt, final_c5, n_points
