import numpy as np
import scipy.optimize as optimize
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial construction
    if 'initial_h_values' in globals():
        initial_h = np.clip(initial_h_values, 0, 1)
        current_sum = np.sum(initial_h)
        if current_sum == 0:
            initial_h = np.zeros(n_points)
        else:
            initial_h = initial_h * (target_sum / current_sum)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (target_sum / np.sum(initial_h))
    else:
        # Default construction: alternate 1 and 0 values to spread the mass
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                initial_h[i] = 1.0
        initial_h = np.clip(initial_h, 0, 1)
        current_sum = np.sum(initial_h)
        initial_h = initial_h * (target_sum / current_sum)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (target_sum / np.sum(initial_h))
    
    def objective(h):
        h_clipped = np.clip(h, 0, 1)
        h_scaled = h_clipped * (target_sum / np.sum(h_clipped))
        h_final = np.clip(h_scaled, 0, 1)
        h_final = h_final * (target_sum / np.sum(h_final))
        h = h_final
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def evaluate_erdos_solution(h):
        h_clipped = np.clip(h, 0, 1)
        h_scaled = h_clipped * (target_sum / np.sum(h_clipped))
        h_final = np.clip(h_scaled, 0, 1)
        h_final = h_final * (target_sum / np.sum(h_final))
        corr = np.correlate(h_final, 1 - h_final, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Differential Evolution Setup
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective_with_bounds(x):
        h = x.copy()
        h_clipped = np.clip(h, 0, 1)
        h_scaled = h_clipped * (target_sum / np.sum(h_clipped))
        h_final = np.clip(h_scaled, 0, 1)
        h_final = h_final * (target_sum / np.sum(h_final))
        corr = np.correlate(h_final, 1 - h_final, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    result = differential_evolution(
        objective_with_bounds,
        bounds,
        strategy='best1bin',
        popsize=20,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.7,
        maxiter=1000,
        seed=seed,
        disp=False
    )
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
