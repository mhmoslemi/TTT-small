import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    best_solution = h.copy()
    
    def compute_cross_corr(h, dx):
        n = len(h)
        cross_corr = np.zeros(2*n - 1)
        for k in range(2*n - 1):
            if k < n:
                cross_corr[k] = (dx * np.sum(h[:n - k] * h[k:]))
            else:
                k_wrap = k - n
                cross_corr[k] = (dx * np.sum(h[:n - k_wrap] * h[k_wrap:]))
        return cross_corr
    
    def compute_gradient(h, max_k, n, dx):
        grad = np.zeros(n)
        if max_k >= n:
            k_wrap = max_k - n
            for i in range(n - k_wrap):
                grad[i] += dx * h[i + k_wrap]
                grad[i + k_wrap] += dx * h[i]
        else:
            for i in range(n - max_k):
                grad[i] += dx * h[i + max_k]
                grad[i + max_k] += dx * h[i]
        return grad
    
    max_iterations = 300
    learning_rate = 0.1
    perturbation_prob = 0.05
    tolerance = 1e-6
    time_start = time.time()
    
    for _ in range(max_iterations):
        if time.time() - time_start > budget_s - 10:
            break
        cross_corr = compute_cross_corr(h, dx)
        current_max = np.max(cross_corr)
        max_k = np.argmax(cross_corr)
        grad = compute_gradient(h, max_k, n_points, dx)
        direction = -grad
        new_h = h + learning_rate * direction
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        scaling_factor = (n_points / 2) / new_sum if new_sum != 0 else 1.0
        new_h = new_h * scaling_factor
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        scaling_factor = (n_points / 2) / new_sum if new_sum != 0 else 1.0
        new_h = new_h * scaling_factor
        if np.random.rand() < perturbation_prob:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            new_h = h + noise
            new_h = np.clip(new_h, 0.0, 1.0)
            new_sum = np.sum(new_h)
            scaling_factor = (n_points / 2) / new_sum if new_sum != 0 else 1.0
            new_h = new_h * scaling_factor
            new_h = np.clip(new_h, 0.0, 1.0)
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            h = new_h.copy()
            best_c5 = new_c5
            best_solution = h.copy()
        if np.abs(new_c5 - best_c5) < tolerance:
            break
    
    assert (best_solution >= 0).all() and (best_solution <= 1).all()
    assert np.isclose(np.sum(best_solution), n_points / 2, atol=1e-6)
    assert np.isfinite(best_c5)
    
    return (best_solution, best_c5, n_points)
