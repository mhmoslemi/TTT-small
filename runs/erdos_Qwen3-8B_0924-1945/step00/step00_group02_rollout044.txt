import numpy as np
import random
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in globals() else None
    n_points = 450
    if initial_h_values is not None:
        h = np.clip(initial_h_values, 0, 1)
        h = h * n_points / 2 / np.sum(h)
    else:
        h = 0.5 * (1 + np.random.uniform(-1, 1, n_points))
        h = np.clip(h, 0, 1)
        h = h * n_points / 2 / np.sum(h)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * 2.0 / n_points
        return c5
    
    c5_bound = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_bound
    start_time = time.time()
    
    for t in range(1000):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        
        # Targeted mutation
        perturb = np.zeros_like(h)
        lags = range(max_k - 10, max_k + 11)
        for lag in lags:
            if 0 <= lag < len(corr):
                contrib = corr[lag] / h.sum()
                perturb[lag % n_points] -= contrib * 0.05
        
        # Global mutation
        global_indices = np.random.choice(n_points, size=5, replace=False)
        perturb_global = np.random.normal(0, 0.1, size=n_points)
        perturb[global_indices] += perturb_global[global_indices]
        
        # Apply mutations
        h_mut = h + perturb
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * n_points / 2 / np.sum(h_mut)
        
        c5_mut = compute_c5(h_mut)
        
        if c5_mut < best_c5:
            best_c5 = c5_mut
            best_h = h_mut.copy()
        
        h = best_h.copy()
    
    return best_h, best_c5, n_points
