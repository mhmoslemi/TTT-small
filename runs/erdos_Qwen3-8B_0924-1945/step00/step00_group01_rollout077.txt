import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values) if initial_h_values is not None else 47
    dx = 2.0 / n_points
    h_vals = np.array(initial_h_values, dtype=float) if initial_h_values is not None else np.random.rand(n_points)
    h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    best_c5 = float('inf')
    best_h_vals = h_vals.copy()
    start_time = time.time()
    
    def compute_overlap(h_vals, lag):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
        overlap = 0.0
        for i in range(n_points):
            if i + lag < 0 or i + lag >= n_points:
                continue
            overlap += h_vals[i] * (1 - h_vals[i + lag])
        return overlap
    
    def objective(x, lag):
        return compute_overlap(x, lag)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    for _ in range(100):
        if time.time() - start_time > budget_s - 100:
            break
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = corr * dx
        max_c5 = np.max(c5)
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h_vals = h_vals.copy()
        
        lags = np.argsort(corr)[-10:]
        for lag in lags:
            if lag < 0:
                lag += n_points
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': constraint}
            result = opt.minimize(objective, h_vals, args=(lag,), bounds=bounds, constraints=cons, method='SLSQP')
            if result.success:
                h_vals = result.x
            else:
                h_vals = np.clip(h_vals, 0, 1)
                h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
        
        if np.allclose(h_vals, best_h_vals, atol=1e-6):
            break
    
    h_vals = np.clip(h_vals, 0, 1)
    h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
    corr = np.correlate(h_vals, 1 - h_vals, mode='full')
    c5_bound = np.max(corr) * dx
    return (h_vals, c5_bound, n_points)
