import numpy as np
import scipy.optimize as optimize
import time

initial_h_values = None  # Placeholder, will be provided by the environment

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 140  # We'll use 140 points for better resolution
    
    # Initialize with initial_h_values if available, otherwise create a baseline
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        # Construct a baseline: evenly spaced 1s and 0s to sum to n_points / 2
        h = np.zeros(n_points)
        h[::2] = 1.0  # Set every other point to 1.0
        h = np.minimum(np.maximum(h, 0.0), 1.0)
    h = np.random.permutation(h)  # Randomize for better exploration
    
    # Ensure sum is exactly n_points / 2
    sum_h = np.sum(h)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        h += (n_points / 2 - sum_h) / n_points
    
    # Precompute dx
    dx = 2.0 / n_points
    
    # Define objective function
    def objective(x):
        h_vals = x.copy()
        h_vals = np.minimum(np.maximum(h_vals, 0.0), 1.0)
        # Compute the convolution
        conv = np.correlate(h_vals, 1.0 - h_vals, mode='full')
        # Find the maximum convolution value
        max_conv = np.max(conv)
        # Compute C₅
        c5 = max_conv * dx
        return c5
    
    # Define constraint function
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    x0 = h.copy()
    
    # Optimization using SLSQP
    result = optimize.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-6}
    )
    
    # Check if optimization was successful
    if result.success:
        h_opt = result.x
    else:
        # If optimization failed, try a perturbed version of the initial guess
        h_perturb = h + np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        h_perturb = h_perturb - np.sum(h_perturb) / n_points + n_points / 2 / n_points
        result = optimize.minimize(
            objective,
            h_perturb,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-6}
        )
        h_opt = result.x
    
    # Final clamp and check
    h_final = np.clip(h_opt, 0.0, 1.0)
    sum_final = np.sum(h_final)
    if np.abs(sum_final - n_points / 2) > 1e-6:
        h_final += (n_points / 2 - sum_final) / n_points
    
    # Compute final C₅
    conv_final = np.correlate(h_final, 1.0 - h_final, mode='full')
    c5_bound = np.max(conv_final) * dx
    
    return (h_final.tolist(), c5_bound, n_points)
