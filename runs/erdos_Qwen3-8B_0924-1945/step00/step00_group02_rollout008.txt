import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = 0.5 * (1 + np.random.uniform(-1, 1, n_points))
    h = np.clip(h, 0, 1)
    h = h * n_points / 2.0 / np.sum(h)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * 2 / n_points
        return c5
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    start_time = time.time()
    for _ in range(1200):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode="full")
        lag = np.argmax(corr)
        overlap = h * (1 - np.roll(h, lag))
        threshold = np.max(overlap) * 0.8
        target_indices = np.where(overlap > threshold)[0]
        target_indices = np.random.choice(target_indices, size=max(1, len(target_indices)//10), replace=False)
        
        candidates = []
        for idx in target_indices:
            h_candidate = h.copy()
            h_candidate[idx] = 1 - h_candidate[idx]
            h_candidate[idx] += np.random.uniform(-0.05, 0.05)
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * n_points / 2.0 / np.sum(h_candidate)
            candidates.append(h_candidate)
        
        for _ in range(3):
            idx = np.random.randint(0, n_points)
            h_candidate = h.copy()
            h_candidate[idx] += np.random.uniform(-0.05, 0.05)
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * n_points / 2.0 / np.sum(h_candidate)
            candidates.append(h_candidate)
        
        for candidate in candidates:
            c5 = compute_c5(candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate.copy()
        
        h = best_h.copy()
    
    return (best_h, best_c5, n_points)
