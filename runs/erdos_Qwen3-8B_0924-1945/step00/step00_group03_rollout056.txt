import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    h_initial = initial_h_values
    h_values = h_initial.copy()
    sum_h = np.sum(h_initial)
    if not np.isclose(sum_h, n_points / 2, atol=1e-6):
        h_values = np.round(h_initial / sum_h * n_points / 2).astype(float)
    h_values /= np.sum(h_values) * dx
    h_values *= n_points / 2

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    def objective(x):
        h = x.reshape(n_points)
        return compute_c5(h)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = {'type': 'eq', 'fun': constraint_sum, 'jac': lambda x: np.ones_like(x)}
    bounds = [(0, 1) for _ in range(n_points)]

    def optimize_with_adam():
        x0 = h_values.copy()
        result = opt.minimize(objective, x0, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        return result

    start_time = time.time()
    result = optimize_with_adam()
    elapsed = time.time() - start_time

    if elapsed < budget_s:
        for _ in range(50):
            if elapsed >= budget_s:
                break
            perturbation = np.random.normal(0, 0.05, size=n_points)
            h_candidate = np.clip(result.x + perturbation, 0, 1)
            h_candidate /= np.sum(h_candidate) * dx
            h_candidate *= n_points / 2
            h_candidate = np.clip(h_candidate, 0, 1)
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < result.fun:
                result = opt.minimize(objective, h_candidate, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
                elapsed = time.time() - start_time

    h_values = np.clip(result.x, 0, 1)
    h_values /= np.sum(h_values) * dx
    h_values *= n_points / 2
    c5_bound = compute_c5(h_values)
    return h_values, c5_bound, n_points
