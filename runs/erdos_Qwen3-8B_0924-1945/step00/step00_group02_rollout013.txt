import numpy as np
import random
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values')
    n_points = 500
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-6:
            h = h * (n_points / 2) / h_sum
            h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if i % 3 == 0 else 0.5
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    start_time = time.time()
    
    def compute_c5(h, n_points):
        dx = 2.0 / n_points
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def clip_and_rescale(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            return np.zeros_like(h)
        h = h * (n_points / 2) / h_sum
        return h
    
    population_size = 50
    population = [h.copy() for _ in range(population_size)]
    
    for gen in range(1000):
        if time.time() - start_time > budget_s:
            break
        offspring = []
        for _ in range(population_size):
            parent1 = np.random.choice(population)
            parent2 = np.random.choice(population)
            child = (parent1 + parent2) / 2
            child = clip_and_rescale(child)
            offspring.append(child)
        population = offspring
        for h_candidate in population:
            current_c5 = compute_c5(h_candidate, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_candidate.copy()
    
    return (best_h, best_c5, n_points)
