import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else None
    n_points = 45 if initial_h_values is None else len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values if initial_h_values is not None else np.random.rand(n_points)
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = np.inf
    max_iter = 1000
    learning_rate = 0.1
    decay_rate = 0.999
    noise_prob = 0.05
    noise_scale = 0.05
    start_time = time.time()
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        current_c5 = np.max(correlate_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Find lag with max value
        max_idx = np.argmax(correlate_vals)
        lag = max_idx - (n_points - 1)
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = i - lag
            if 0 <= j < n_points:
                gradients[i] = (1 - h[j]) * dx
        # Update h
        h = h - learning_rate * gradients
        # Clip and normalize
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        # Add noise
        if np.random.rand() < noise_prob:
            noise = np.random.normal(0, noise_scale, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        learning_rate *= decay_rate
    # Final check
    correlate_vals = np.correlate(best_h, 1 - best_h, mode="full") * dx
    final_c5 = np.max(correlate_vals)
    return (best_h, final_c5, n_points)
