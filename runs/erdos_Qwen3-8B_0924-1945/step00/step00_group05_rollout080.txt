def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    best_h = initial_h_values.copy()
    best_c = evaluate_erdos_solution(best_h)
    best_n_points = len(best_h)
    start_time = time.time()
    
    n_points = 500
    n_gen = int((budget_s - 1) / 1.5)
    population_size = 50
    mutation_noise = 0.05
    mutation_rate = 0.1
    selection_size = 20
    stagnation_threshold = 50
    
    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total
    
    def evaluate(h):
        return evaluate_erdos_solution(h)
    
    def crossover(parent1, parent2):
        split = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:split], parent2[split:]])
        return clip_and_normalize(child)
    
    def mutate(h):
        noise = np.random.normal(0, mutation_noise, n_points)
        h += noise
        return clip_and_normalize(h)
    
    def tournament_selection(population, k=5):
        candidates = np.random.choice(population, size=k, replace=False)
        return candidates[np.argmin([evaluate(h) for h in candidates])]
    
    def create_initial_population(parent):
        population = [np.copy(parent)]
        for _ in range(population_size - 1):
            h = mutate(parent)
            population.append(h)
        return population
    
    def evolve_population(population):
        new_population = []
        for _ in range(population_size // 2):
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            child1 = crossover(parent1, parent2)
            child2 = mutate(child1)
            new_population.append(child1)
            new_population.append(child2)
        return new_population
    
    population = create_initial_population(best_h)
    
    for gen in range(n_gen):
        if time.time() - start_time > budget_s:
            break
        population.sort(key=evaluate)
        selected = population[:selection_size]
        new_population = evolve_population(selected)
        population = new_population
        current_best = population[0]
        current_c = evaluate(current_best)
        
        if current_c < best_c:
            best_c = current_c
            best_h = current_best
            best_n_points = n_points
            stagnation_counter = 0
        else:
            stagnation_counter += 1
        
        if stagnation_counter >= stagnation_threshold:
            mutation_noise *= 0.9
            stagnation_counter = 0
    
    if evaluate(best_h) < best_c:
        best_c = evaluate(best_h)
    
    if np.any(np.isnan(best_h)) or np.any(np.isinf(best_h)):
        best_h = clip_and_normalize(best_h)
    
    return (best_h, best_c, best_n_points)
