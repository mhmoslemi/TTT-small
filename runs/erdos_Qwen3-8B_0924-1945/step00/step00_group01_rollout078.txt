import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    if n_points == 0:
        n_points = 100
        h_vals = np.full(n_points, 0.5)
    else:
        h_vals = initial_h.copy()
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)

    dx = 2.0 / n_points
    best_h = h_vals.copy()
    best_c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * dx
    start_time = time.time()

    for _ in range(100):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 100:
            break

        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        overlaps = corr * dx
        max_k = np.argmax(overlaps)
        target_k = max_k

        def objective(x):
            h = x.reshape(-1)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * n_points / 2
            corr = np.correlate(h, 1 - h, mode='full')
            return corr[target_k] * dx

        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        result = optimize.minimize(objective, h_vals.flatten(), method='L-BFGS-B', bounds=bounds, constraints=constraints)
        if result.success:
            h_vals = result.x.reshape(-1)
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals / np.sum(h_vals) * n_points / 2
            current_c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_vals.copy()
        else:
            h_vals = best_h.copy()

    h_vals = np.clip(h_vals, 0, 1)
    h_vals = h_vals / np.sum(h_vals) * n_points / 2
    final_c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * dx
    return (h_vals, final_c5, n_points)
