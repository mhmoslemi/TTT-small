import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initial guess
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        initial_h_values[i] = 0.5 * (1 + np.sin(np.pi * i / n_points))
    initial_h_values = np.clip(initial_h_values, 0, 1)
    current_sum = np.sum(initial_h_values)
    initial_h_values = initial_h_values * (target_sum / current_sum)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (target_sum / np.sum(initial_h_values))
    best_h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    start_time = time.time()
    population_size = 100
    generations = 50
    mutation_rate = 0.1
    min_step = 0.05

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        population = []
        for _ in range(population_size):
            h = best_h.copy()
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    h[i] += np.random.uniform(-min_step, min_step)
                    h[i] = np.clip(h[i], 0, 1)
            h = h * (target_sum / np.sum(h))
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
            population.append(h)

        # Evaluate population
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            fitness.append(c5)

        # Select top 50
        idxs = np.argsort(fitness)[:50]
        population = [population[i] for i in idxs]
        fitness = [fitness[i] for i in idxs]

        # Crossover
        next_gen = []
        for i in range(0, len(population), 2):
            if i + 1 < len(population):
                p1, p2 = population[i], population[i + 1]
                child = np.zeros(n_points)
                for j in range(n_points):
                    if np.random.rand() < 0.5:
                        child[j] = p1[j]
                    else:
                        child[j] = p2[j]
                next_gen.append(child)
            else:
                next_gen.append(population[i])

        # Mutate
        for h in next_gen:
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    h[i] += np.random.uniform(-min_step, min_step)
                    h[i] = np.clip(h[i], 0, 1)
            h = h * (target_sum / np.sum(h))
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))

        # Update best
        for h in next_gen:
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Local refinement with SLSQP
    def objective(x):
        h = x.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    res = opt.minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}], tol=1e-6, options={'maxiter': 100})
    best_h = res.x
    best_c5 = evaluate_erdos_solution(best_h)

    # Final fitness check
    corr = np.correlate(best_h, 1 - best_h, mode="full")
    best_c5 = np.max(corr) * dx
    final_sum = np.sum(best_h)
    best_h = best_h * (target_sum / final_sum)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h))

    corr = np.correlate(best_h, 1 - best_h, mode="full")
    best_c5 = np.max(corr) * dx
    assert np.all(0 <= best_h <= 1)
    assert np.isclose(np.sum(best_h), target_sum, atol=1e-6)
    assert np.isfinite(best_c5)
    return (best_h, best_c5, n_points)
