import numpy as np
import scipy.optimize as opt
import time
import copy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.clip(initial_h_values, 0, 1).copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                h[i] = 1.0
            else:
                h[i] = 0.0
    h = h * (n_points / 2) / np.sum(h)

    max_iterations = 500
    learning_rate = 0.01
    momentum = 0.9
    perturbation_prob = 0.05
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()

    for _ in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Compute overlap integrals
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(overlap)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Find the lag with maximum overlap
        max_idx = np.argmax(overlap)
        k = max_idx - (n_points - 1)

        # Compute gradient for this lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] += (1 - h[j])
                grad[j] -= h[i]

        # Update h
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)

        # Enforce sum constraint
        sum_h = np.sum(h)
        if sum_h == 0:
            h = best_h.copy()
        else:
            h = h * (n_points / 2) / sum_h

        # Apply perturbation
        if np.random.rand() < perturbation_prob:
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = best_h.copy()
            else:
                h = h * (n_points / 2) / sum_h

        # Reduce learning rate if no improvement
        if np.abs(c5 - best_c5) < 1e-6:
            learning_rate *= 0.95

    return (best_h, best_c5, n_points)
