import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Initial setup
    if 'initial_h_values' in globals():
        h_vals = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h_vals)
    else:
        n_points = 50
        h_vals = np.zeros(n_points)
        h_vals[::2] = 1.0  # Initial guess: alternating 1s and 0s
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    
    dx = 2.0 / n_points
    best_h_vals = h_vals.copy()
    best_c5 = np.inf
    
    start_time = time.time()
    
    # Optimization parameters
    max_iter = 200
    tol = 1e-6
    lag_strategy = 'current'  # 'current' or 'randomized'
    rand_lags = np.random.choice(np.arange(n_points * 2 - 1), size=10, replace=False)
    
    for iter in range(max_iter):
        # Compute correlations
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        corr = corr * dx
        
        # Find worst lag
        if lag_strategy == 'current':
            k_max = np.argmax(corr)
        else:
            # Use a random selection of lags
            k_max = np.random.choice(rand_lags)
        
        current_c5 = corr[k_max]
        
        # Update best solution if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_vals = h_vals.copy()
        
        # Early termination if no improvement
        if iter > 0 and abs(best_c5 - current_c5) < tol:
            break
        
        # Convex optimization: set up surrogate objective
        # Minimize 0.5 * λ² + (overlap_k - λ_k)^2 for the worst lag
        overlap_k = corr[k_max]
        λ = cp.Variable()
        obj = 0.5 * λ**2 + (overlap_k - λ)**2
        prob = cp.Problem(cp.Minimize(obj))
        prob.solve()
        
        # Update h_vals using gradient of surrogate with respect to h_vals
        # The surrogate gradient is proportional to the overlap derivative
        # We assume a simpler update based on residual
        residual = overlap_k - λ.value
        h_vals = h_vals - residual * 0.1
        
        # Clip to [0, 1] and re-normalize
        h_vals = np.clip(h_vals, 0.0, 1.0)
        mass = np.sum(h_vals)
        h_vals = h_vals * (n_points / 2) / mass
        
        # Time check
        if time.time() - start_time > budget_s:
            break
    
    # Final evaluation
    corr = np.correlate(best_h_vals, 1 - best_h_vals, mode='full') * dx
    final_c5 = np.max(corr)
    
    return (best_h_vals, final_c5, n_points)
