import numpy as np
import scipy.optimize as opt
import time

initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 
                             0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    elapsed_time = 0
    eps = 1e-8
    alpha = 0.1
    decay_rate = 0.95
    max_iter = 500
    early_stop_steps = 30
    early_stop_threshold = 1e-4
    
    def compute_gradient(h, max_index):
        h_forward = h.copy()
        h_forward[max_index] += eps
        h_backward = h.copy()
        h_backward[max_index] -= eps
        c_forward = evaluate_erdos_solution(h_forward)
        c_backward = evaluate_erdos_solution(h_backward)
        grad = (c_forward - c_backward) / (2 * eps)
        return grad
    
    def evaluate_erdos_solution(h):
        dx = 2.0 / n_points
        cross_corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(cross_corr)
    
    for _ in range(max_iter):
        if elapsed_time >= budget_s:
            break
        start_time = time.time()
        cross_corr = np.correlate(h, 1 - h, mode="full") * 2.0 / n_points
        max_index = np.argmax(cross_corr)
        grad = compute_gradient(h, max_index)
        h -= alpha * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (target_sum / h_sum)
        new_c5 = evaluate_erdos_solution(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
        elapsed_time += time.time() - start_time
        if np.abs(new_c5 - best_c5) < early_stop_threshold:
            if _ > early_stop_steps:
                break
        alpha *= decay_rate
    
    return best_h, best_c5, n_points
