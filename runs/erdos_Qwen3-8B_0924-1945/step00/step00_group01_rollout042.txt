import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction or random binary sequence
    n_points = 100  # Tuned for performance and quality
    if 'initial_h_values' in globals():
        h_vals = np.array(initial_h_values)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    else:
        h_vals = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    
    dx = 2.0 / n_points
    max_iterations = 100
    early_stop = 5
    best_h_vals = h_vals.copy()
    best_c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * dx
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 100:
            break
        
        # Compute correlation values
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        corr = corr * dx
        max_corr = np.max(corr)
        
        # Early stopping if no progress
        if np.isclose(max_corr, best_c5, atol=1e-5):
            if iteration > 0 and iteration % early_stop == 0:
                break
        
        if max_corr < best_c5:
            best_c5 = max_corr
            best_h_vals = h_vals.copy()
        
        # Find the lag with maximum overlap
        lag_idx = np.argmax(corr)
        
        # Prepare for optimization
        def obj(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            h = (h * (n_points / 2)) / np.sum(h)
            corr = np.correlate(h, 1 - h, mode='full')[lag_idx] * dx
            return corr
        
        # Constraints: sum(h) = n/2
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        
        # Bounds for h_vals
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimization
        try:
            result = opt.minimize(
                fun=obj,
                x0=h_vals,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-8, 'maxiter': 100}
            )
            if result.success:
                h_vals = result.x
            else:
                h_vals = best_h_vals.copy()
        except:
            h_vals = best_h_vals.copy()
        
        # Clip and renormalize
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    
    # Final check
    corr = np.correlate(best_h_vals, 1 - best_h_vals, mode='full') * dx
    final_c5 = np.max(corr)
    
    # Return the best solution found
    return (best_h_vals, final_c5, n_points)
