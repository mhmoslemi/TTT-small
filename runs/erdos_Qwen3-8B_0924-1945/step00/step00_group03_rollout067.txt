import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    initial_h_values = np.load('initial_h_values.npy')  # Assume this is provided as a parent construction

    # Validate and convert initial_h_values into a block pattern
    h = initial_h_values.copy()
    block_size = n_points // 10
    block_pattern = np.zeros(n_points // block_size, dtype=int)
    for i in range(n_points // block_size):
        block_pattern[i] = 1 if i % 2 == 0 else 0
    block_pattern = np.repeat(block_pattern, block_size)
    block_pattern = np.clip(block_pattern, 0, 1)
    block_sum = np.sum(block_pattern)
    if abs(block_sum - n_points / 2) > 1e-6:
        delta = (n_points / 2 - block_sum) / (n_points // block_size)
        block_pattern += delta
        block_pattern = np.round(block_pattern).astype(int)

    def compute_c5(h_values):
        return np.max(np.correlate(h_values, 1 - h_values, mode='full')) * dx

    def evaluate_pattern(pattern):
        h_values = np.repeat(pattern, block_size)
        h_values = np.clip(h_values, 0, 1)
        h_values /= np.sum(h_values)
        h_values *= n_points / 2
        return compute_c5(h_values)

    best_pattern = block_pattern.copy()
    best_c5 = evaluate_pattern(best_pattern)

    def optimize_block_pattern(pattern, budget):
        for i in range(budget):
            if time.time() - start_time > budget_s:
                break
            if np.random.random() < 0.5:
                idx = np.random.randint(0, len(pattern))
                pattern[idx] = 1 - pattern[idx]
                pattern = np.clip(pattern, 0, 1)
                pattern = np.round(pattern).astype(int)
                pattern_sum = np.sum(pattern)
                delta = (n_points / 2 - pattern_sum) / len(pattern)
                pattern += delta
                pattern = np.clip(pattern, 0, 1)
                pattern = np.round(pattern).astype(int)
            else:
                idx1, idx2 = np.random.choice(len(pattern), 2, replace=False)
                pattern[idx1], pattern[idx2] = pattern[idx2], pattern[idx1]
            new_c5 = evaluate_pattern(pattern)
            if new_c5 < best_c5:
                best_pattern = pattern.copy()
                best_c5 = new_c5
        return best_pattern, best_c5

    start_time = time.time()
    best_pattern, best_c5 = optimize_block_pattern(block_pattern, 500)
    if best_c5 < 0.3807:
        return (np.repeat(best_pattern, block_size), best_c5, n_points)

    # Convert best_pattern to a continuous h_values array
    h_values = np.repeat(best_pattern, block_size)
    h_values = np.clip(h_values, 0, 1)
    h_values /= np.sum(h_values)
    h_values *= n_points / 2
    h_values = h_values.astype(float)

    # Local optimization using gradient descent
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        h /= np.sum(h)
        h *= n_points / 2
        return compute_c5(h)

    def gradient(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        h /= np.sum(h)
        h *= n_points / 2
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        grad = np.zeros_like(x)
        for i in range(len(x)):
            h[i] = 1.0
            corr = np.correlate(h, 1 - h, mode='full')
            c5_i = np.max(corr) * dx
            h[i] = 0.0
            corr = np.correlate(h, 1 - h, mode='full')
            c5_0 = np.max(corr) * dx
            grad[i] = (c5_i - c5_0) * dx
        return grad

    result = opt.minimize(objective, h_values, method='L-BFGS-B', jac=gradient, options={'maxiter': 500, 'disp': False})
    h_values = result.x
    h_values = np.clip(h_values, 0, 1)
    h_values /= np.sum(h_values)
    h_values *= n_points / 2

    final_c5 = compute_c5(h_values)
    if final_c5 < best_c5:
        best_c5 = final_c5

    assert np.all(h_values >= 0) and np.all(h_values <= 1)
    assert np.isclose(np.sum(h_values), n_points / 2, atol=1e-6)
    assert np.isfinite(final_c5)

    return (h_values, final_c5, n_points)
