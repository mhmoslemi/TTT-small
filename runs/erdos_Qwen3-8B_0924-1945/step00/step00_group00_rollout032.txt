import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial guess
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Parameters
    max_iterations = 500
    learning_rate = 0.1
    momentum_factor = 0.9
    noise_scale = 0.05
    early_stop_threshold = 1e-5
    early_stop_count = 50
    best_c5 = float('inf')
    best_h = h.copy()
    best_n_points = n_points
    start_time = time.time()
    
    for i in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Compute the full correlation
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        current_c5 = np.max(correlate_vals)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early stopping
        if i > 0 and abs(current_c5 - best_c5) < early_stop_threshold:
            early_stop_count -= 1
            if early_stop_count <= 0:
                break
        
        # Select worst lag
        worst_lag = np.argmax(correlate_vals)
        # Compute gradient for worst lag
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * (2.0 / n_points)
        
        # Momentum update
        if i == 0:
            v = np.zeros(n_points)
        else:
            v = momentum_factor * v + learning_rate * gradients
        
        # Update h
        h = h - v
        
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        
        # Normalize to maintain sum(h) = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
        # Add noise
        if i % 100 == 0:
            noise = noise_scale * np.random.normal(0, 1, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return best_h, best_c5, best_n_points
