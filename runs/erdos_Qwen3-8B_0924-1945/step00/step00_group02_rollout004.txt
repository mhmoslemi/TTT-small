import numpy as np
import random
import time
import scipy.signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 500
    if initial_h_values is not None and len(initial_h_values) == n_points:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = (h * 2 - 1) * 0.5 + 0.5  # Normalize to [0.5, 1.0]
        h = np.clip(h, 0, 1)
        h /= np.sum(h) * 2
        h *= n_points / 2
        h = np.clip(h, 0, 1)
    
    def evaluate_c5(h_vals):
        h = h_vals.copy()
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * 2
        h = np.clip(h, 0, 1)
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        corr = np.clip(corr, 0, 1)
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    population_size = 50
    population = [h.copy() for _ in range(population_size)]
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    start_time = time.time()
    generation = 0
    while time.time() - start_time < budget_s:
        generation += 1
        
        # Evaluate and sort
        fitness = [evaluate_c5(p) for p in population]
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fitness = np.sort(fitness)
        
        # Tournament selection
        selected = []
        for _ in range(population_size):
            indices = np.random.choice(range(population_size), 5, replace=False)
            best_idx = indices[np.argmin([sorted_fitness[i] for i in indices])]
            selected.append(sorted_pop[best_idx].copy())
        
        # Crossover
        crossover_size = int(population_size * 0.2)
        for i in range(crossover_size):
            parent1 = selected[i]
            parent2 = selected[np.random.randint(0, crossover_size)]
            child = (parent1 + parent2) / 2
            child = np.clip(child, 0, 1)
            child = child / np.sum(child) * 2
            child = np.clip(child, 0, 1)
            selected[i] = child
        
        # Mutation
        for i in range(population_size):
            if np.random.rand() < 0.1:
                indices = np.random.choice(n_points, int(n_points * 0.1), replace=False)
                noise = np.random.normal(0, 0.1, size=n_points)
                selected[i][indices] += noise[indices]
                selected[i] = np.clip(selected[i], 0, 1)
                selected[i] = selected[i] / np.sum(selected[i]) * 2
                selected[i] = np.clip(selected[i], 0, 1)
        
        # Replace population
        population = selected
        
        # Update best
        for p in population:
            current_c5 = evaluate_c5(p)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = p.copy()
        
        if generation % 10 == 0:
            print(f"Generation {generation}, Best C5: {best_c5}")
    
    return (best_h, best_c5, n_points)
