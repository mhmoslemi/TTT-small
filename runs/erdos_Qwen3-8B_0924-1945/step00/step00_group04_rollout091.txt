import numpy as np
import scipy.optimize as optimize
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Assuming initial_h_values is given as a file
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Validate and normalize initial_h_values
    def repair_h(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        scale = target_sum / current_sum
        return h * scale

    h = repair_h(initial_h_values.copy())
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        h = repair_h(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Genetic algorithm parameters
    pop_size = 100
    generations = 50
    mutation_rate = 0.1
    mutation_range = 0.05
    elite_size = 50

    for gen in range(generations):
        population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
        population = [repair_h(p) for p in population]
        fitness = [evaluate_c5(p) for p in population]
        population = [p for _, p in sorted(zip(fitness, population))]
        elite = population[:elite_size]
        population = elite
        for _ in range(pop_size - elite_size):
            parent1, parent2 = random.sample(elite, 2)
            child = np.zeros(n_points)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            if random.random() < mutation_rate:
                for i in range(n_points):
                    child[i] += np.random.uniform(-mutation_range, mutation_range)
                    child[i] = np.clip(child[i], 0, 1)
            child = repair_h(child)
            population.append(child)
        population = population[:pop_size]
        fitness = [evaluate_c5(p) for p in population]
        population = [p for _, p in sorted(zip(fitness, population))]
        current_h = population[0]
        current_c5 = fitness[0]
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
        if np.sum(np.abs(best_h - current_h)) < 1e-5:
            break

    # Local refinement
    bounds = [(0, 1) for _ in range(n_points)]
    def objective(x):
        x = np.clip(x, 0, 1)
        x = repair_h(x)
        corr = np.correlate(x, 1 - x, mode="full")
        return -np.max(corr) * dx

    result = optimize.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    best_h = result.x
    best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, n_points)
