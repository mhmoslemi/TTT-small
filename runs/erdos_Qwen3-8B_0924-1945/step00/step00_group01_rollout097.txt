import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Tune this for speed and accuracy
    dx = 2.0 / n_points
    h = np.full(n_points, 0.5)
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h) != n_points:

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def objective(x):
        h_vals = x
        ones_minus_h = 1 - h_vals
        corr = np.correlate(h_vals, ones_minus_h, mode="full")
        return np.max(corr * dx)
    
    def gradient(x):
        h_vals = x
        ones_minus_h = 1 - h_vals
        corr = np.correlate(h_vals, ones_minus_h, mode="full")
        max_corr = np.max(corr * dx)
        max_index = np.argmax(corr)
        # Gradient approximation via finite differences
        grad = np.zeros_like(x)
        for i in range(len(x)):
            x_i = x[i]
            x[i] = x_i + 1e-6
            corr_plus = np.correlate(x, ones_minus_h, mode="full")
            grad[i] = (np.max(corr_plus * dx) - max_corr) / 1e-6
            x[i] = x_i
        return grad
    
    bounds = Bounds(0, 1)
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
    
    result = opt.minimize(
        objective,
        h,
        method='L-BFGS-B',
        jac=gradient,
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    best_h = result.x
    c5_bound = objective(best_h)
    return (best_h, c5_bound, n_points)
