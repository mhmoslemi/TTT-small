```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 47  # Using the provided initial_h_values
    initial_h_values = kwargs.get('initial_h_values', np.random.rand(n_points))
    
    # Normalize initial h_values to meet the sum constraint
    h_vals = initial_h_values.copy()
    sum_h = np.sum(h_vals)
    if sum_h != 0:
        h_vals = (h_vals * (n_points / 2)) / sum_h
    
    dx = 2.0 / n_points
    
    def evaluate_c5(h_vals):
        shifted_h = np.zeros_like(h_vals)
        for i in range(n_points):
            shifted_h[(i + n_points) % n_points] = h_vals[i]
        overlap = np.correlate(h_vals, 1 - shifted_h, mode='full') * dx
        return np.max(overlap)
    
    def optimize_for_lag(h_vals, lag):
        def objective(x):
            x = np.clip(x, 0, 1)
            x_sum = np.sum(x)
            if np.isclose(x_sum, n_points / 2):
                shifted = np.zeros_like(x)
                for i in range(n_points):
                    shifted[(i + lag) % n_points] = x[i]
                return -np.sum(x * (1 - shifted))
            else:
                return np.inf
        
        def constraint(x):
            return np.sum(x) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.minimize(objective, h_vals, bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, method='SLSQP')
        if result.success:
            return result.x
        else:
            return np.clip(h_vals, 0, 1)
    
    start_time = time.time()
    best_h_vals = h_vals.copy()
    best_c5 = evaluate_c5(best_h_vals)
    no_improvement_counter = 0
    
    for _ in range(100):
        if time.time() - start_time > budget_s - 100:
            break
        
        # Compute current C5
        current_c5 = evaluate_c5(h_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_vals = h_vals.copy()
            no_improvement_counter = 0
        else:
            no_improvement_counter += 1
        
        if no_improvement_counter >= 5:
            break
        
        # Choose a worst-lag to optimize
        shifted_h = np.zeros_like(h_vals)
        for i in range(n_points):
            shifted_h[(i + n_points) % n_points] = h_vals[i]
        overlap = np.correlate(h_vals, 1 - shifted_h, mode='full') * dx
        worst_lag = np.argmax(overlap)
        
        # Optimize for worst lag
        h_vals = optimize_for_lag(h_vals, worst_lag)
    
    # Final evaluation
    final_c5 = evaluate_c5(h_vals)
    
    # Check if the best solution is better
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h_vals = h_vals.copy()
    
    # Ensure constraints are satisfied
    if not np.all((best_h_vals >= 0) & (best_h_vals <= 1)) or not np.isclose(np.sum(best_h_vals), n_points / 2):
        best_h_vals = np.clip(best_h_vals, 0, 1)
        best_h_vals = (best_h_vals * (n_points / 2)) / np.sum(best_h_vals)
    
    # Final evaluation after constraints
    final_c5 = evaluate_c5(best_h_vals)
    
    return (best_h_vals, final_c5, n_points)
```
