```python
import numpy as np
import scipy.optimize as optimize
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Selected for performance and sufficient resolution
    dx = 2.0 / n_points
    if initial_h_values is not None and len(initial_h_values) == n_points:
        h_vals = np.array(initial_h_values, dtype=np.float64)
    else:
        # Random binary initial guess, normalized to sum(n_points/2)
        h_vals = np.random.choice([0, 1], size=n_points)
        h_vals = (h_vals * n_points / 2) / np.sum(h_vals)
    
    # Feasibility checks
    if not np.all((h_vals >= 0) & (h_vals <= 1)):
        h_vals = np.clip(h_vals, 0, 1)
    if not np.isclose(np.sum(h_vals), n_points / 2):
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
        h_vals = np.clip(h_vals, 0, 1)
    
    best_h = h_vals.copy()
    best_c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * dx
    max_iter = 100
    no_improve = 0

    start_time = time.time()

    for _ in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 100:
            break
        
        # Compute all correlations
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5_values = corr * dx
        max_k = np.argmax(c5_values)
        current_c5 = c5_values[max_k]
        
        if current_c5 <= best_c5:
            no_improve += 1
            if no_improve >= 5:
                break
            continue
        
        best_c5 = current_c5
        best_h = h_vals.copy()
        
        # Define objective function for the worst lag
        def objective(x):
            h = x.reshape(n_points)
            h = np.clip(h, 0, 1)
            h = (h * (n_points / 2)) / np.sum(h)
            corr = np.correlate(h, 1 - h, mode='full')
            return corr[max_k] * dx
        
        # Optimization
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        
        try:
            result = optimize.minimize(
                objective,
                h_vals,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                tol=1e-6
            )
            h_vals = result.x
        except:
            h_vals = best_h.copy()
        
        # Ensure feasibility
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
        
        # Check for improvement
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5_values = corr * dx
        current_c5 = np.max(c5_values)
        if current_c5 <= best_c5:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            best_c5 = current_c5
            best_h = h_vals.copy()
    
    # Final validation
    h_vals = np.clip(best_h, 0, 1)
    h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    corr = np.correlate(h_vals, 1 - h_vals, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (h_vals, c5_bound, n_points)
```
