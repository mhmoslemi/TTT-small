import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 47
    if 'initial_h_values' in globals():
        h_vals = initial_h_values.copy()
    else:
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    dx = 2.0 / n_points
    best_h_vals = h_vals.copy()
    best_c5 = float('inf')
    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        current_c5 = np.max(corr * dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_vals = h_vals.copy()
        k_max = np.argmax(corr)
        overlap = corr[k_max] * dx
        # Solve convex optimization to adjust h_vals
        # Objective: minimize 0.5 * λ^2 + (overlap - λ)^2
        # This is for the worst lag
        λ = cp.Variable()
        objective = cp.Minimize(0.5 * λ**2 + (overlap - λ)**2)
        prob = cp.Problem(objective, [])
        prob.solve()
        λ_val = λ.value
        # Compute gradient and update h_vals
        # This is a crude approximation; could be more sophisticated
        h_vals += (overlap - λ_val) * 0.1
        # Clip to [0, 1] and re-normalize
        h_vals = np.clip(h_vals, 0, 1)
        mass = np.sum(h_vals)
        h_vals = h_vals * (n_points / 2) / mass
    # Final evaluation
    corr = np.correlate(best_h_vals, 1 - best_h_vals, mode='full')
    final_c5 = np.max(corr * dx)
    return (best_h_vals, final_c5, n_points)
