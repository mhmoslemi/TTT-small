import numpy as np
import scipy.optimize as optimize
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load("initial_h_values.npy")  # Assuming initial_h_values is available
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Clip and normalize to meet constraints
    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h if sum_h != 0 else np.zeros(n_points) + 0.5

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    learning_rate = 0.01
    decay_rate = 0.999
    iteration = 0
    start_time = time.time()

    while time.time() - start_time < budget_s:
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        k_max = np.argmax(overlaps)
        if k_max >= len(overlaps):
            k_max = len(overlaps) - 1
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (n_points / 2) / sum_h
        else:
            h = np.zeros(n_points) + 0.5

        new_c5 = evaluate_erdos_solution(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            iteration = 0
        else:
            iteration += 1

        if iteration >= 100:
            learning_rate *= decay_rate
            iteration = 0

        # Random perturbation every 500 iterations
        if iteration % 500 == 0:
            noise = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + noise, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                h = h * (n_points / 2) / sum_h

    # Final check to ensure correctness
    h = np.clip(best_h, 0, 1)
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h if sum_h != 0 else np.zeros(n_points) + 0.5

    return h, best_c5, n_points
