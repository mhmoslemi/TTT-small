```python
import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.random.rand(n_points)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n_points
        c5 = np.max(corr) * dx
        return c5

    def rescale(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return h
        scaling = (n_points / 2) / total
        return h * scaling

    def mutate(h):
        h = h.copy()
        indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
        delta = np.random.uniform(-0.1, 0.1, size=len(indices))
        h[indices] += delta
        h = rescale(h)
        return h

    def crossover(h1, h2):
        h = h1.copy()
        indices = np.random.choice(n_points, size=int(n_points * 0.3), replace=False)
        h[indices] = (h1[indices] + h2[indices]) / 2
        h = rescale(h)
        return h

    population = [mutate(initial_h_values) for _ in range(50)]
    best_h = population[0]
    best_c5 = evaluate(best_h)

    start_time = time.time()

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        if len(population) < 50:
            population += [mutate(best_h) for _ in range(50 - len(population))]
        candidates = population[:]
        random.shuffle(candidates)
        selected = []
        for _ in range(50):
            if not candidates:
                break
            a, b, c, d, e = random.sample(candidates, 5)
            selected.append(min([a, b, c, d, e], key=evaluate))
        population = selected
        for h in population:
            h_val = evaluate(h)
            if h_val < best_c5:
                best_h = h
                best_c5 = h_val
        if len(population) < 50:
            population += [mutate(best_h) for _ in range(50 - len(population))]
        population = sorted(population, key=evaluate)[:50]

    h_values = best_h
    c5_bound = best_c5
    return h_values, c5_bound, n_points
```
