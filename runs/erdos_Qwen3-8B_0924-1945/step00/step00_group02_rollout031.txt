import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = 0.5 * (1 + np.random.uniform(-1, 1, n_points))
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
    start_time = time.time()
    
    for _ in range(int((budget_s - (time.time() - start_time)) // 0.12)):
        corr = np.correlate(h, 1 - h, mode="full")
        lag = np.argmax(corr)
        overlap = h * (np.roll(1 - h, lag))
        threshold = np.max(overlap) * 0.8
        candidates = []
        idxs = np.where(overlap > threshold)[0]
        if len(idxs) > 0:
            idxs = np.random.choice(idxs, min(10, len(idxs)), replace=False)
            for i in idxs:
                h_candidate = h.copy()
                h_candidate[i] = 1 - h_candidate[i]
                h_candidate[i] += np.random.uniform(-0.05, 0.05)
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode="full")
                c5_candidate = np.max(corr_candidate) * dx
                candidates.append((c5_candidate, h_candidate))
        
        if len(candidates) > 0:
            c5_candidate, h_candidate = min(candidates)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
        
        if np.random.rand() < 0.05:
            h = 0.5 * (1 + np.random.uniform(-1, 1, n_points))
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return (best_h, best_c5, n_points)
