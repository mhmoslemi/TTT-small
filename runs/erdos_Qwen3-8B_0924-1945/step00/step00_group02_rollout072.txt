```python
import numpy as np
import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Default number of points
    n_points = 500
    # Maximum allowable C5 value
    target_c5 = 0.3807
    # Population size
    pop_size = 50
    # Tournament size
    tournament_size = 5
    # Mutation rate and strength
    mutation_rate = 0.1
    mutation_strength = 0.1
    # Crossover rate
    crossover_rate = 0.2
    # Initial population
    pop = []
    # Best solution so far
    best_h = None
    best_c5 = float('inf')
    
    # Generate initial population
    def generate_initial_population(n_points, initial_h_values=None):
        if initial_h_values is not None:
            pop = [np.copy(initial_h_values) for _ in range(pop_size)]
        else:
            # Create a simple initial population with uniform distribution
            pop = [np.random.rand(n_points) for _ in range(pop_size)]
        # Rescale all individuals to satisfy the integral constraint
        for i in range(len(pop)):
            pop[i] = pop[i] / np.sum(pop[i]) * (n_points / 2)
        return pop
    
    pop = generate_initial_population(n_points, initial_h_values)
    
    # Evaluate a candidate
    def evaluate_candidate(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Mutation function
    def mutate(h):
        indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
        h[indices] += np.random.uniform(-mutation_strength, mutation_strength, size=len(indices))
        h = np.clip(h, 0.0, 1.0)
        # Rescale to maintain integral constraint
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    # Crossover function
    def crossover(parent1, parent2):
        indices = np.random.choice(n_points, n_points // 2, replace=False)
        child = np.copy(parent1)
        child[indices] = parent2[indices]
        child = np.clip(child, 0.0, 1.0)
        child = child / np.sum(child) * (n_points / 2)
        return child
    
    # Tournament selection
    def tournament_selection(population):
        selected = np.random.choice(len(population), tournament_size, replace=False)
        best_idx = selected[np.argmin([evaluate_candidate(population[i]) for i in selected])]
        return population[best_idx]
    
    # Evolutionary algorithm
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate all candidates
        fitness = [evaluate_candidate(h) for h in pop]
        # Track the best candidate
        best_idx = np.argmin(fitness)
        current_c5 = fitness[best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = pop[best_idx]
        
        # Generate new population
        new_pop = []
        for _ in range(pop_size):
            # Tournament selection for parent1
            parent1 = tournament_selection(pop)
            # Tournament selection for parent2
            parent2 = tournament_selection(pop)
            # Crossover with some probability
            if np.random.rand() < crossover_rate:
                child = crossover(parent1, parent2)
            else:
                child = np.copy(parent1)
            # Mutation
            child = mutate(child)
            new_pop.append(child)
        
        # Replace old population
        pop = new_pop
    
    # Final evaluation
    final_c5 = evaluate_candidate(best_h)
    return best_h, final_c5, n_points
```
