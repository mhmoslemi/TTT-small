```python
import numpy as np
import scipy.optimize as opt
import random
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 200
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h = h[:n_points]
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def evaluate(h_values):
        h = h_values.copy()
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return max(correlate_vals)
    
    def objective(h_values):
        return evaluate(h_values)
    
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    constraints = [{'type': 'eq', 'fun': constraint}]
    
    result = opt.minimize(objective, h, method='SLSQP', constraints=constraints, options={'ftol': 1e-8, 'maxiter': 200})
    h = result.x
    c5 = evaluate(h)
    
    best_h = h.copy()
    best_c5 = c5
    
    if time.time() - start_time < budget_s:
        for _ in range(300):
            h = best_h.copy()
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0.0, 1.0)
            h = h * (n_points / 2) / np.sum(h)
            c5 = evaluate(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        for _ in range(100):
            h = best_h.copy()
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0.0, 1.0)
            h = h * (n_points / 2) / np.sum(h)
            c5 = evaluate(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    if time.time() - start_time < budget_s:
        for _ in range(100):
            h = best_h.copy()
            noise = np.random.normal(0, 0.1, n_points)
            h += noise
            h = np.clip(h, 0.0, 1.0)
            h = h * (n_points / 2) / np.sum(h)
            c5 = evaluate(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5, n_points
```
