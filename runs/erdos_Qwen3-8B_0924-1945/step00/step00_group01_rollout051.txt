from scipy.optimize import minimize
import time

    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h_shifted = np.roll(h, 1)
        corr = np.correlate(h, 1 - h_shifted, mode="full")
        max_corr = np.max(corr) * dx
        return max_corr
    
    def grad_objective(h_flat):
        h = h_flat.reshape(n_points)
        h_shifted = np.roll(h, 1)
        corr = np.correlate(h, 1 - h_shifted, mode="full")
        max_corr_idx = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            for j in range(n_points):
                if i <= j:
                    k = j - i
                    if k < n_points:
                        grad[i] += -corr[max_corr_idx] * (1 - h_shifted[k]) * dx
        return grad.flatten()
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - (n_points / 2)
    
    def constraint_bounds(h_flat):
        return np.all((h_flat >= 0) & (h_flat <= 1))
    
    # Optimization
    h_initial = initial_h.copy()
    h_initial = np.clip(h_initial, 0, 1)
    
    start_time = time.time()
    result = minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        jac=grad_objective,
        constraints=[
            {'type': 'eq', 'fun': constraint_sum},
            {'type': 'ineq', 'fun': constraint_bounds}
        ],
        options={
            'ftol': 1e-8,
            'maxiter': 100,
            'disp': False
        }
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (initial_h, 0.38085, n_points)

    best_h = np.clip(best_h, 0, 1)
    np.sum(best_h) = n_points / 2
    c5_bound = objective(best_h)
