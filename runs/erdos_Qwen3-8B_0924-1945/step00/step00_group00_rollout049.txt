```python
import numpy as np
import numpy.random as npr
import scipy.optimize as sco
import time
import functools

initial_h_values = None  # Provided by the harness

def run(seed=42, budget_s=600, **kwargs):
    n_points = 45
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = npr.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def evaluate(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def objective(h_vec):
        return evaluate(h_vec)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    result = sco.minimize(objective, h, method='SLSQP', constraints=cons, tol=1e-10, options={'ftol': 1e-10, 'maxiter': 100})
    
    best_h = result.x
    best_c5 = evaluate(best_h)
    
    return (best_h, best_c5, n_points)
```
