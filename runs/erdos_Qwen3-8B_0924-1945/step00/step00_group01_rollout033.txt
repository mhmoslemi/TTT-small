import numpy as np
import random
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction from parent, if available
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 50  # Preferred small n_points
    dx = 2.0 / n_points
    if initial_h_values is not None:
        initial_h = np.array(initial_h_values)
        initial_h = (initial_h * (n_points / 2)) / np.sum(initial_h)
        initial_h = np.clip(initial_h, 0, 1)
    else:
        initial_h = np.random.rand(n_points)
        initial_h = (initial_h * (n_points / 2)) / np.sum(initial_h)
        initial_h = np.clip(initial_h, 0, 1)
    
    # Objective function
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        return c5
    
    # Constraint: sum(h) = n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint}
    
    # Optimization using Sequential Least Squares Programming
    result = minimize(objective, initial_h, method='SLSQP', constraints=cons, options={'ftol': 1e-10, 'maxiter': 100})
    
    # Check if the optimization succeeded
    if result.success:
        h_vals = result.x
        c5_bound = result.fun
    else:
        # Fallback to initial_h if optimization failed
        h_vals = initial_h
        c5_bound = objective(h_vals)
    
    # Final checks
    h_vals = np.clip(h_vals, 0, 1)
    h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
    h_vals = np.clip(h_vals, 0, 1)
    c5_bound = np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * dx
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print("Warning: Budget exceeded")
    
    return (h_vals, c5_bound, n_points)
