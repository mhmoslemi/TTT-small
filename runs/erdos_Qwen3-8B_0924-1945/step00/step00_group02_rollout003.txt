import numpy as np
import scipy.optimize as opt
import copy
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 500
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h = h / (np.sum(h) * dx)
    else:
        h = np.zeros(n_points)
        h[250] = 1.0
        h[251] = 1.0
        h[252] = 1.0
        h[253] = 1.0
        h[254] = 1.0
        h[255] = 1.0
        h[256] = 1.0
        h[257] = 1.0
        h[258] = 1.0
        h[259] = 1.0
        h[::10] = 0.5
        h = h / (np.sum(h) * dx)

    population_size = 50
    tournament_size = 5
    mutation_rate = 0.1
    crossover_rate = 0.2
    best_h = copy.deepcopy(h)
    best_c5 = evaluate_erdos_solution(h, n_points, dx)
    start_time = time.time()

    def evaluate(h):
        try:
            corr = np.correlate(h, 1 - h, mode='full') * dx
            c5 = np.max(corr)
            return c5
        except:
            return float('inf')

    def mutate(h):
        indices = np.random.choice(n_points, int(mutation_rate * n_points), replace=False)
        delta = np.random.uniform(-0.1, 0.1, size=len(indices))
        h[indices] += delta
        h = np.clip(h, 0.0, 1.0)
        h = h / (np.sum(h) * dx)
        return h

    def crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        indices = np.random.choice(n_points, int(crossover_rate * n_points), replace=False)
        child[indices] = (parent1[indices] + parent2[indices]) / 2
        child = np.clip(child, 0.0, 1.0)
        child = child / (np.sum(child) * dx)
        return child

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        population = [copy.deepcopy(best_h) for _ in range(population_size)]
        for _ in range(population_size // 2):
            parent1 = np.random.choice(population)
            parent2 = np.random.choice(population)
            child = crossover(parent1, parent2)
            child = mutate(child)
            population.append(child)
        population = population[:population_size]
        selected = []
        for _ in range(population_size):
            candidates = np.random.choice(population, tournament_size, replace=False)
            winner = min(candidates, key=evaluate)
            selected.append(winner)
        population = selected
        for h_candidate in population:
            current_c5 = evaluate(h_candidate)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = copy.deepcopy(h_candidate)
        if best_c5 < 0.3807:
            break

    return best_h, best_c5, n_points
