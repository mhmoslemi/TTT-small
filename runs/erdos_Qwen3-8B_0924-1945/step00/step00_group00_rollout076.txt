import numpy as np
import scipy.signal
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=np.float64) if initial_h_values is not None else np.random.rand(n_points)
    h = np.clip(h, 0., 1.)
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    def calculate_c5(h):
        h_1 = 1.0 - h
        correlate = scipy.signal.correlate(h, h_1, mode='full')
        c5 = np.max(correlate) * dx
        return c5
    
    def update_h(h, c5, lag, learning_rate=0.01, noise_scale=0.01):
        h_1 = 1.0 - h
        correlate = scipy.signal.correlate(h, h_1, mode='full')
        max_correlate = correlate[lag]
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i - lag
            if 0 <= j < n_points:
                grad[i] = h_1[j] * dx
        grad = -grad
        h += grad * learning_rate
        h = np.clip(h, 0., 1.)
        h = h * (n_points / 2) / np.sum(h)
        h += np.random.normal(0, noise_scale, size=h.shape)
        h = np.clip(h, 0., 1.)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    while time.time() - start_time < budget_s:
        c5 = calculate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Find the lag with maximum overlap
        correlate = scipy.signal.correlate(h, 1.0 - h, mode='full')
        lag = np.argmax(correlate)
        h = update_h(h, c5, lag)
    
    return best_h, best_c5, n_points
