import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Default starting point: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 450
        h = 0.5 * (1 + np.random.uniform(-1, 1, n_points))
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points, dx)
    
    start_time = time.time()
    
    for _ in range(int((budget_s - (time.time() - start_time)) // 0.1)):
        # Evaluate current C5
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        
        # Get max overlap index
        max_k = np.argmax(corr)
        
        # Targeted mutation: adjust around max_k
        delta = 0.1 / (1 + _ / 500)
        indices = np.arange(max_k - 10, max_k + 11)
        indices = np.clip(indices, 0, n_points - 1)
        for i in np.random.choice(indices, size=10, replace=False):
            contrib = corr[max_k] - corr[max_k - i] - corr[max_k + i]
            h[i] += delta * contrib
        
        # Global mutation
        for _ in range(5):
            i = np.random.randint(0, n_points)
            noise = np.random.normal(0, 0.2 * np.exp(-_ / 300))
            h[i] += noise
            h[i] = np.clip(h[i], 0, 1)
        
        # Normalize to maintain integral constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate new candidate
        new_c5 = compute_c5(h, n_points, dx)
        if new_c5 < c5:
            h = h.copy()
            c5 = new_c5
        
        # Update best solution
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Final check and return
    h = best_h
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    c5 = compute_c5(h, n_points, dx)
    
    return h, c5, n_points

def compute_c5(h, n_points, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx
