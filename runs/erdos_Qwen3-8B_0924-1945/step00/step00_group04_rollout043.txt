import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in globals() else np.random.rand(42)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = np.clip(initial_h_values, 0, 1)
    h = h * (target_sum / np.sum(h))
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    elapsed_time = 0
    learning_rate = 0.05
    max_iterations = 1000
    no_improvement_count = 0

    def compute_gradient(h, lag_k):
        grad = np.zeros_like(h)
        for j in range(n_points):
            if j + lag_k < n_points:
                grad[j] += (1 - h[j + lag_k]) * dx
            if j - lag_k >= 0:
                grad[j] -= h[j - lag_k] * dx
        return grad

    def update_h(h, grad, learning_rate):
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if np.isclose(current_sum, 0):
            h = np.zeros_like(h)
        else:
            h = h * (target_sum / current_sum)
        return h

    start_time = time.time()
    for _ in range(max_iterations):
        if elapsed_time >= budget_s:
            break
        cross_corr = np.correlate(h, 1 - h, mode="full") * dx
        idx = np.argmax(cross_corr)
        lag_k = idx - (n_points - 1)
        grad = compute_gradient(h, lag_k)
        h = update_h(h, grad, learning_rate)
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        if no_improvement_count >= 50:
            break
        elapsed_time = time.time() - start_time
        learning_rate *= 0.995

    return best_h, best_c5, n_points
