import numpy as np
import scipy.optimize as optimize
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Initial guess
    n_points = 100
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values, dtype=float)
    else:
        h_initial = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                h_initial[i] = 1.0
            else:
                h_initial[i] = 0.0
    h_initial = h_initial / (n_points / 2) * (n_points / 2)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial / np.sum(h_initial) * (n_points / 2)

    dx = 2.0 / n_points
    h_values = h_initial.copy()

    def objective(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (n_points / 2) / h_sum
        c = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(c)

    def constraint(h):
        h = np.clip(h, 0, 1)
        return np.sum(h) - (n_points / 2)

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}

    result = optimize.minimize(objective, h_values, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-8, options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8})
    h_values = result.x
    h_values = np.clip(h_values, 0, 1)
    h_values = h_values * (n_points / 2) / np.sum(h_values)
    c5_bound = objective(h_values)

    return h_values, c5_bound, n_points
