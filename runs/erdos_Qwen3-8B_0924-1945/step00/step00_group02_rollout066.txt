```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 300
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * n_points / 2.0
        h = h / np.sum(h) * n_points / 2.0

    def clip_and_scale(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        scaling = (n_points / 2.0) / total
        return h * scaling

    h = clip_and_scale(h)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def evaluate_candidate(h_candidate):
        h_candidate = clip_and_scale(h_candidate)
        return compute_c5(h_candidate)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    dt = 0.1
    max_iter = int((budget_s - (time.time() - start_time)) / dt)
    if max_iter < 1:
        return (best_h, best_c5, n_points)

    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        lag = np.argmax(corr)
        overlap = h * np.roll(1 - h, lag)
        threshold = np.max(overlap) * 0.8
        contrib_indices = np.where(overlap > threshold)[0]
        target_indices = contrib_indices[:int(len(contrib_indices) * 0.1)]
        candidates = []

        for i in target_indices:
            h_candidate = h.copy()
            h_candidate[i] = 1 - h_candidate[i]
            candidates.append(h_candidate)
            h_candidate = h.copy()
            h_candidate[i] += np.random.uniform(-0.05, 0.05)
            h_candidate = clip_and_scale(h_candidate)
            candidates.append(h_candidate)

        if len(candidates) < 1:
            break

        c5_values = [evaluate_candidate(c) for c in candidates]
        best_idx = np.argmin(c5_values)
        new_h = candidates[best_idx]
        if c5_values[best_idx] < best_c5:
            best_c5 = c5_values[best_idx]
            best_h = new_h.copy()
        h = new_h

        if time.time() - start_time > budget_s - 1:
            break

    return (best_h, best_c5, n_points)
```
