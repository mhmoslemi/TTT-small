import numpy as np
import random
import time
from scipy.signal import correlate
from scipy.optimize import Bounds, LinearConstraint

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100  # Adjust for performance and quality
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5]*n_points)  # Fallback if no parent given
    if 'initial_h_values' in globals():
        initial_h_values = np.array(initial_h_values)
        initial_h_values = (initial_h_values * (n_points / 2)) / np.sum(initial_h_values)
        initial_h_values = np.clip(initial_h_values, 0, 1)
    
    # Ensure sum constraint
    def ensure_sum(h):
        total = np.sum(h)
        if np.isclose(total, n_points / 2, atol=1e-6):
            return h
        scale = (n_points / 2) / total
        return np.clip(h * scale, 0, 1)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = ensure_sum(h)
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Genetic Algorithm Parameters
    POP_SIZE = 50
    MUTATION_RATE = 0.15
    CROSS_RATE = 0.8
    TOURNAMENT_SIZE = 5
    GEN_LIMIT = 100
    STAGNATION = 10
    BEST = np.inf
    
    # Initial population
    def generate_initial_population():
        population = []
        for _ in range(POP_SIZE):
            h = np.random.rand(n_points)
            h = ensure_sum(h)
            population.append(h)
        return np.array(population)
    
    # Tournament selection
    def tournament_selection(pop, scores):
        tournament = np.random.choice(np.arange(POP_SIZE), size=TOURNAMENT_SIZE, replace=False)
        best_idx = tournament[np.argmin(scores[tournament])]
        return pop[best_idx]
    
    # Crossover
    def crossover(parent1, parent2):
        if np.random.rand() < CROSS_RATE:
            idx = np.random.randint(1, n_points - 1)
            child1 = np.concatenate((parent1[:idx], parent2[idx:]))
            child2 = np.concatenate((parent2[:idx], parent1[idx:]))
            return child1, child2
        return parent1, parent2
    
    # Mutation
    def mutate(individual):
        for i in range(n_points):
            if np.random.rand() < MUTATION_RATE:
                individual[i] += np.random.uniform(-0.1, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        return ensure_sum(individual)
    
    # Genetic Algorithm
    population = generate_initial_population()
    best_h = population[0]
    best_c5 = evaluate(best_h)
    stagnant_count = 0
    
    for gen in range(GEN_LIMIT):
        if time.time() - start_time > budget_s:
            break
        scores = np.array([evaluate(ind) for ind in population])
        best_idx = np.argmin(scores)
        best_c5 = scores[best_idx]
        if best_c5 < best_c5:
            best_c5 = best_c5
            best_h = population[best_idx]
            stagnant_count = 0
        else:
            stagnant_count += 1
            if stagnant_count >= STAGNATION:
                break
        
        # Selection and Reproduction
        new_population = []
        for _ in range(POP_SIZE // 2):
            parent1 = tournament_selection(population, scores)
            parent2 = tournament_selection(population, scores)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1.copy())
            child2 = mutate(child2.copy())
            new_population.append(child1)
            new_population.append(child2)
        
        # Replace worst with new
        scores_new = np.array([evaluate(ind) for ind in new_population])
        indices = np.argsort(scores_new)
        population = np.array([new_population[i] for i in indices[:POP_SIZE]])
    
    final_h = ensure_sum(best_h)
    final_c5 = evaluate(final_h)
    return (final_h, final_c5, n_points)
