import numpy as np
import scipy.signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h = initial_h_values
    h = np.array(initial_h, dtype=np.float64)
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if abs(total - n_points / 2) > 1e-8:
        h = h * (n_points / 2) / total
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = float('inf')
    dx = 2.0 / n_points
    time_start = time.time()
    
    for _ in range(500):
        if time.time() - time_start > budget_s:
            break
        correlate = scipy.signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlate) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        max_lag = np.argmax(correlate)
        lag = max_lag - n_points + 1
        if lag < 0:
            lag += n_points
        if lag >= n_points:
            lag -= n_points
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i - lag + n_points) % n_points
            grad[i] = (1 - h[j]) * dx
        alpha = 0.01
        h -= alpha * grad
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        h = h * (n_points / 2) / total
        h = np.clip(h, 0, 1)
    
    correlate = scipy.signal.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlate) * dx
    return best_h, final_c5, n_points
