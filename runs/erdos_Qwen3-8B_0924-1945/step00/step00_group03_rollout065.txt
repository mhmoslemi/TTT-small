import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.copy(initial_h_values)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    t_total = int(budget_s * 0.7)
    T0 = 0.7 * (1.0 + best_c5) / 3.0
    Tmax = int(t_total * 0.1)
    beta = 0.75
    gamma = 0.65
    lambda0 = 0.15 / dx
    t = 0
    start_time = time.time()
    
    while t < t_total and time.time() - start_time < budget_s:
        if t % 100 == 0:
            lambda0 *= 0.8 if t < t_total // 2 else 0.95
        noise = np.random.normal(0, lambda0, n_points)
        perturbed = h + noise
        perturbed = np.clip(perturbed, 0.0, 1.0)
        current_sum = np.sum(perturbed)
        delta = (n_points / 2 - current_sum) / (n_points * 0.95)
        perturbed += delta * np.random.choice([-1, 1], size=n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        if np.sum(perturbed) != 0:
            perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        new_c5 = evaluate_erdos_solution(perturbed, dx)
        if new_c5 < best_c5 or (np.random.rand() < np.exp(-(new_c5 - best_c5) / (T0 * (1 - t / Tmax) ** (1 / beta)))):
            h = perturbed.copy()
            best_h = h.copy()
            best_c5 = new_c5
            if best_c5 < 0.3807:
                break
        t += 1
    
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (best_h, final_c5, n_points)
