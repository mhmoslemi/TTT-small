import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values, dtype=np.float64)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full_like(h, 0.5)
        return h * (n_points / 2) / total
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    def optimize_with_cvxpy(h_initial):
        h = cp.Variable(n_points)
        t = cp.Variable()
        lam = 1e-2
        
        corr = np.correlate(h_initial, 1 - h_initial, mode='full')
        max_corr = np.max(corr) * dx
        constraints = []
        for k in range(n_points):
            idx = (np.arange(n_points) + k) % n_points
            dot = cp.sum(h[idx] * (1 - h_initial[idx]))
            constraints.append(t >= dot * dx)
        
        obj = t + lam * cp.sum_squares(h - 0.5)
        prob = cp.Problem(cp.Minimize(obj), constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        
        if prob.status == 'OPTIMAL':
            h_sol = np.array(h.value)
            h_sol = clip_and_normalize(h_sol)
            return h_sol
        return h_initial
    
    start_time = time.time()
    for _ in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        h_new = optimize_with_cvxpy(best_h)
        c5_new = evaluate(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    best_h = clip_and_normalize(best_h)
    exact_c5 = evaluate(best_h)
    
    return best_h, exact_c5, n_points
