import numpy as np
import scipy.signal
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h[mid:] = 0.0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    learning_rate = 0.01
    decay_rate = 0.999
    iteration = 0
    start_time = time.time()

    while time.time() - start_time < budget_s:
        overlaps = scipy.signal.correlate(h, 1 - h, mode="full") * dx
        k_max = np.argmax(overlaps)
        grad = np.zeros(n_points)

        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx

        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (n_points / 2) / sum_h
        else:
            h = np.zeros(n_points)
            h[:] = 0.5
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

        new_c5 = evaluate_erdos_solution(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            iteration = 0
        else:
            iteration += 1

        if iteration >= 100:
            learning_rate *= decay_rate
            iteration = 0

    return (best_h, best_c5, n_points)
