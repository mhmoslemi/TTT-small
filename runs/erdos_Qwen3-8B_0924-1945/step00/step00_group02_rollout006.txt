import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = 0.5 * (1 + np.random.uniform(-1, 1, n_points))
        h = h / np.sum(h) * (n_points / 2)
    
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
    
    T = 100.0
    alpha = 0.995
    start_time = time.time()
    
    for t in range(int(budget_s * 10)):
        if time.time() - start_time > budget_s:
            break
        indices = np.random.choice(n_points, 5, replace=False)
        delta = np.random.normal(0, 0.05 * np.exp(-t / 200), 5)
        new_h = h.copy()
        for i, idx in zip(delta, indices):
            new_h[idx] += i
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        corr = np.correlate(new_h, 1 - new_h, mode="full")
        c5_new = np.max(corr) * dx
        
        if c5_new < best_c5:
            best_h = new_h.copy()
            best_c5 = c5_new
        
        if np.random.rand() < np.exp((best_c5 - c5_new) / T):
            h = new_h.copy()
        
        T *= alpha
    
    return best_h, best_c5, n_points
