import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    if initial_h_values is None:
        n_points = 47
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    else:
        n_points = len(initial_h_values)
        h_vals = np.array(initial_h_values, dtype=np.float64)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)

    dx = 2.0 / n_points
    max_iterations = 100
    no_improvement_threshold = 5
    best_c5 = np.inf
    best_h_vals = h_vals.copy()

    for iter in range(max_iterations):
        start_time = time.time()
        corr = np.correlate(h_vals, 1 - h_vals, mode='full') * dx
        max_overlap = np.max(corr)
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h_vals = h_vals.copy()

        if iter >= no_improvement_threshold:
            break

        # Pick a random lag to optimize
        k = np.random.randint(1, len(corr) - 1)
        if k >= len(h_vals):
            k = len(h_vals) - 1

        # Define the overlap for the selected lag
        def objective(x):
            overlap = 0
            for i in range(len(x)):
                j = (i + k) % len(x)
                overlap += x[i] * (1 - x[j])
            return overlap

        # Optimization bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(len(h_vals))]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        res = opt.minimize(objective, h_vals, method='SLSQP', bounds=bounds, constraints=cons)
        if res.success:
            h_vals = res.x
        else:
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)

        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    # Final evaluation
    corr_final = np.correlate(h_vals, 1 - h_vals, mode='full') * dx
    final_c5 = np.max(corr_final)
    h_vals = np.clip(h_vals, 0, 1)
    h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)

    # Ensure sum is correct
    if not np.isclose(np.sum(h_vals), n_points / 2):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)

    # Return best solution if it's better
    corr_best = np.correlate(best_h_vals, 1 - best_h_vals, mode='full') * dx
    best_c5 = np.max(corr_best)
    if best_c5 < final_c5:
        h_vals = best_h_vals.copy()
        final_c5 = best_c5

    return (h_vals, final_c5, n_points)
