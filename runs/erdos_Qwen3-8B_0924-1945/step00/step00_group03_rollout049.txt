import numpy as np
import scipy
import numpy.random as npr
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    assert (h >= 0).all() and (h <= 1).all()
    assert np.isclose(np.sum(h), n_points / 2, atol=1e-6)
    best_solution = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    start_time = time.time()
    max_iterations = 500
    perturbation_prob = 0.05
    tolerance = 1e-6
    learning_rate = 1e-2
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
        
        cross_corr = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            if k >= n_points:
                k_wrap = k - n_points
                cross_corr[k] = (dx * np.sum(h[:n_points - k_wrap] * h[k_wrap:]))
            else:
                cross_corr[k] = (dx * np.sum(h[:n_points - k] * h[k:]))
        current_max = np.max(cross_corr)
        
        max_k = np.argmax(cross_corr)
        
        grad = np.zeros(n_points)
        if max_k >= n_points:
            k_wrap = max_k - n_points
            for i in range(n_points - k_wrap):
                grad[i] += dx * h[i + k_wrap]
                grad[i + k_wrap] += dx * h[i]
        else:
            for i in range(n_points - max_k):
                grad[i] += dx * h[i + max_k]
                grad[i + max_k] += dx * h[i]
        
        direction = -grad
        step_size = 0.1
        new_h = h + step_size * direction
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        scaling_factor = (n_points / 2) / new_sum if new_sum != 0 else 1.0
        new_h = new_h * scaling_factor
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        scaling_factor = (n_points / 2) / new_sum if new_sum != 0 else 1.0
        new_h = new_h * scaling_factor
        
        if np.random.rand() < perturbation_prob:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            new_h = h + noise
            new_h = np.clip(new_h, 0.0, 1.0)
            new_sum = np.sum(new_h)
            scaling_factor = (n_points / 2) / new_sum if new_sum != 0 else 1.0
            new_h = new_h * scaling_factor
            new_h = np.clip(new_h, 0.0, 1.0)
        
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            h = new_h.copy()
            best_c5 = new_c5
            best_solution = h.copy()
        
        if np.abs(new_c5 - best_c5) < tolerance:
            break
    
    # Final validation
    assert (best_solution >= 0).all() and (best_solution <= 1).all()
    assert np.isclose(np.sum(best_solution), n_points / 2, atol=1e-6)
    exact_c5 = max(np.correlate(best_solution, best_solution, mode='full')) * dx
    assert np.isclose(exact_c5, best_c5, atol=1e-8)
    
    return (best_solution, exact_c5, n_points)
