import numpy as np
import scipy.optimize as optimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 50
    if 'initial_h_values' in globals():
        h_vals = np.array(initial_h_values, dtype=np.float64)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    else:
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    
    dx = 2.0 / n_points
    max_iterations = 200
    tol = 1e-6
    best_c5 = np.inf
    best_h = h_vals.copy()
    
    for _ in range(max_iterations):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        corr = corr * dx
        current_c5 = np.max(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_vals.copy()
        
        if current_c5 < tol:
            break
        
        k_max = np.argmax(corr)
        overlap_k = corr[k_max]
        
        lamb = cp.Variable()
        obj = 0.5 * lamb**2 + (overlap_k - lamb)**2
        prob = cp.Problem(cp.Minimize(obj))
        result = prob.solve()
        if result is not None:
            gradient = (overlap_k - lamb.value) * 2
            h_vals -= gradient * 0.01
        
        h_vals = np.clip(h_vals, 0, 1)
        mass = np.sum(h_vals)
        h_vals = h_vals * (n_points / 2) / mass
    
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr * dx)
    return (best_h, c5_bound, n_points)
