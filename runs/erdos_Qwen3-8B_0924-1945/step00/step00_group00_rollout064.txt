import numpy as np
import scipy.signal as signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h_values = np.array([0.0] * n_points)
    
    if 'initial_h_values' in globals():
        initial_h_values = np.array(initial_h_values)
        initial_h_values = np.clip(initial_h_values, 0.0, 1.0)
        initial_h_values = initial_h_values / (np.sum(initial_h_values) * dx)
        initial_h_values = initial_h_values * (n_points / 2)
        initial_h_values = np.clip(initial_h_values, 0.0, 1.0)
        initial_h_values = initial_h_values / (np.sum(initial_h_values) * dx)
        initial_h_values = initial_h_values * (n_points / 2)
        initial_h_values = np.clip(initial_h_values, 0.0, 1.0)
        h = initial_h_values
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0.0, 1.0)
        h = h / (np.sum(h) * dx)
        h = h * (n_points / 2)
        h = np.clip(h, 0.0, 1.0)
        h = h / (np.sum(h) * dx)
        h = h * (n_points / 2)
        h = np.clip(h, 0.0, 1.0)
    
    best_h = h.copy()
    best_c5 = np.inf
    time_start = time.time()
    
    for _ in range(1000):
        if time.time() - time_start > budget_s:
            break
        
        # Compute correlate
        correlate = signal.correlate(h, 1 - h, mode='full')
        max_c5 = np.max(correlate) * dx
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
        
        # Find worst lag
        lag_index = np.argmax(correlate)
        lag = lag_index - (n_points - 1)
        
        # Compute gradient for worst lag
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i - lag + n_points) % n_points
            grad[i] = (1 - h[j]) * dx
        
        # Update h
        h = h - 0.01 * grad
        h = np.clip(h, 0.0, 1.0)
        h = h / (np.sum(h) * dx)
        h = h * (n_points / 2)
        h = np.clip(h, 0.0, 1.0)
        h = h / (np.sum(h) * dx)
        h = h * (n_points / 2)
    
    # Final check and fallback
    if np.isnan(best_c5) or best_c5 > 1.0:
        h = initial_h_values.copy()
        correlate = signal.correlate(h, 1 - h, mode='full')
        best_c5 = np.max(correlate) * dx
    
    h = np.clip(h, 0.0, 1.0)
    h = h / (np.sum(h) * dx)
    h = h * (n_points / 2)
    h = np.clip(h, 0.0, 1.0)
    h = h / (np.sum(h) * dx)
    h = h * (n_points / 2)
    
    return h, best_c5, n_points
