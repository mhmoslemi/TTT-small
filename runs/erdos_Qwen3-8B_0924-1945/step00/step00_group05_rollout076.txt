import scipy.signal
import time

    momentum = 0.9

        overlap = scipy.signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    for _ in range(max_iter):
        if time.time() - time_start > budget_s:
            break

            best_c5 = overlap
            best_h = h.copy()

        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad[i] += (1 - h[j])
            j = (i - k_max + n_points) % n_points
            grad[i] -= h[j]

        h = h - learning_rate * grad
