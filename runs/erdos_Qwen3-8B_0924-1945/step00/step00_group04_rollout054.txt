import numpy as np
import scipy.optimize as opt
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Load or construct initial h_values
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        # Construct a basic initial guess with evenly spaced 1s and 0s
        initial_h = np.zeros(n_points)
        initial_h[::3] = 1.0
        initial_h = np.clip(initial_h, 0, 1)

    # Normalize initial_h
    current_sum = np.sum(initial_h)
    initial_h = initial_h * (target_sum / current_sum)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (target_sum / np.sum(initial_h))
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            h[0] = 1.0
        else:
            h = h * (target_sum / h_sum)
        return h

    def evaluate(h):
        h = repair(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def objective(h):
        return -evaluate(h)

    def constraint(h):
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0, 1) for _ in range(n_points)]

    # Genetic algorithm parameters
    population_size = 100
    generations = 50
    mutation_rate = 0.1
    mutation_range = 0.05

    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        population = np.random.uniform(0, 1, (population_size, n_points))
        for i in range(population_size):
            perturbation = np.random.uniform(-mutation_range, mutation_range, n_points)
            population[i] = initial_h + perturbation
        population = np.array([repair(p) for p in population])
        fitness = np.array([evaluate(p) for p in population])
        top_indices = np.argsort(fitness)[:50]
        top_population = population[top_indices]

        # Crossover
        new_population = []
        for i in range(0, population_size, 2):
            if i + 1 < population_size:
                parent1 = top_population[np.random.randint(len(top_population))]
                parent2 = top_population[np.random.randint(len(top_population))]
                child = np.zeros(n_points)
                for j in range(n_points):
                    if np.random.rand() < 0.5:
                        child[j] = parent1[j]
                    else:
                        child[j] = parent2[j]
                new_population.append(child)
            else:
                new_population.append(top_population[np.random.randint(len(top_population))])
        new_population += [np.random.uniform(0, 1, n_points) for _ in range(population_size - len(new_population))]
        population = np.array([repair(p) for p in new_population])
        for p in population:
            if np.random.rand() < mutation_rate:
                perturbation = np.random.uniform(-mutation_range, mutation_range, n_points)
                p += perturbation
        population = np.array([repair(p) for p in population])
        fitness = np.array([evaluate(p) for p in population])
        top_indices = np.argsort(fitness)
        best_h = population[top_indices[0]]
        best_c5 = evaluate(best_h)

    # Local refinement with SLSQP
    if time.time() - start_time < budget_s:
        result = opt.minimize(objective, best_h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6, options={'maxiter': 100})
        best_h = result.x
        best_c5 = evaluate(best_h)

    # Final repair and check
    best_h = repair(best_h)
    best_c5 = evaluate(best_h)
    assert np.all(0 <= best_h <= 1)
    assert np.isclose(np.sum(best_h), target_sum, atol=1e-6)
    assert np.isfinite(best_c5)

    return (best_h, best_c5, n_points)
