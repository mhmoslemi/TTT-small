```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    initial_h_values = np.random.rand(n_points)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def constraint_bounds(h):
        return np.all((h >= 0) & (h <= 1))

    def eval_c5(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def optimize_h(initial_h):
        h = initial_h.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraint_sum, 'jac': lambda x: np.ones_like(x)}]
        res = optimize.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            tol=1e-8,
            options={'maxiter': 200}
        )
        return res.x

    def refine_pattern(h):
        blocks = np.round(np.mean(h.reshape(-1, n_points // 10), axis=1)).astype(int)
        blocks = blocks - np.mean(blocks)
        blocks = np.clip(blocks, 0, 1)
        blocks = blocks / np.sum(blocks) * (n_points // 10) / 2
        h_refined = np.repeat(blocks, n_points // 10)
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        return h_refined

    best_h = initial_h_values.copy()
    best_c5 = eval_c5(best_h)
    time_start = time.time()

    for _ in range(100):
        if time.time() - time_start > budget_s:
            break
        h = optimize_h(best_h)
        if constraint_bounds(h):
            c5 = eval_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h

    h_final = refine_pattern(best_h)
    c5_final = eval_c5(h_final)

    if time.time() - time_start > budget_s:
        return (best_h, best_c5, n_points)

    return (h_final, c5_final, n_points)
```
