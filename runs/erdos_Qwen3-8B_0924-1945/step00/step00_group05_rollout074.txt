```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
import time

# Pre-imported helpers
def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    # Compute 1 - h(x)
    one_minus_h = 1.0 - h_values
    # Compute correlation
    corr = np.correlate(h_values, one_minus_h, mode="full")
    # Multiply by dx
    corr *= dx
    # Find max overlap
    return np.max(corr)

def initial_h_values(n_points):
    # Construct initial h values
    h = np.zeros(n_points)
    # Position the first 1/3 in the first half
    num_first_half = int(n_points / 3)
    h[:num_first_half] = 1.0
    # Position the next 1/3 in the second half
    num_second_half = int(n_points / 3)
    h[-num_second_half:] = 1.0
    # Ensure the sum is correct
    if np.sum(h) != n_points / 2:
        # Normalize
        scale = (n_points / 2) / np.sum(h)
        h *= scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = initial_h_values(n_points)
    if np.sum(h) != n_points / 2:
        scale = (n_points / 2) / np.sum(h)
        h *= scale
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    time_remaining = budget_s - time.time()
    
    for _ in range(500):
        if time_remaining < 0:
            break
        # Generate a random perturbation
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        h_pert = h + perturbation
        h_pert = np.clip(h_pert, 0.0, 1.0)
        # Scale to meet the integral constraint
        sum_h = np.sum(h_pert)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h_pert *= scale
        # Compute C5
        c5 = evaluate_erdos_solution(h_pert, n_points)
        if c5 < best_c5:
            best_h = h_pert.copy()
            best_c5 = c5
        h = h_pert.copy()
        time_remaining = budget_s - time.time()
    
    return (best_h, best_c5, n_points)
```
