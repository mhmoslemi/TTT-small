import numpy as np
import scipy.signal
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45  # Start with the initial n_points
    h = initial_h_values.copy() if 'initial_h_values' in kwargs else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(scipy.signal.correlate(h, 1 - h, mode="full")) * (2 / n_points)
    
    start_time = time.time()
    learning_rate = 0.1
    momentum = 0.9
    noise_scale = 0.05
    decay_rate = 0.9995
    iteration = 0
    max_iterations = int(budget_s / 1)  # ~500 iterations at 1s/iter
    early_stop_counter = 0
    
    while time.time() - start_time < budget_s:
        correlate_vals = scipy.signal.correlate(h, 1 - h, mode="full") * (2 / n_points)
        worst_lag = np.argmax(correlate_vals)
        
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * (2 / n_points)
        
        if iteration == 0:
            velocity = np.zeros(n_points)
        else:
            velocity = momentum * velocity + learning_rate * gradients
        
        h = h - velocity
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        if np.random.rand() < 0.05:
            h += noise_scale * np.random.normal(0, 1, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        current_c5 = np.max(correlate_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            early_stop_counter = 0
        else:
            early_stop_counter += 1
        
        if early_stop_counter > 50:
            break
        
        learning_rate *= decay_rate
        iteration += 1
    
    h_values = best_h
    return h_values, best_c5, n_points
