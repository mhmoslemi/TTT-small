```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    max_iter = 100
    no_improve_thresh = 5
    dx = 2.0 / n_points
    initial_h_values = np.random.rand(n_points)
    # Normalize to meet sum constraint
    initial_h_values = (initial_h_values * (n_points / 2)) / np.sum(initial_h_values)
    h_vals = initial_h_values.copy()
    best_h_vals = h_vals.copy()
    best_c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * dx
    start_time = time.time()
    
    for iter in range(max_iter):
        # Compute all lags
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        # Only consider lags that make sense in the domain
        max_lag_idx = len(corr) // 2
        lag_values = corr[:max_lag_idx + 1]
        max_lag = np.argmax(lag_values)
        max_c5 = lag_values[max_lag] * dx
        
        # Update best values
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h_vals = h_vals.copy()
        
        # If no improvement for several iterations, break early
        if iter > 0 and max_c5 >= best_c5 - 1e-5:
            if iter > no_improve_thresh:
                break
        
        # Create bounds and constraint
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        
        # Objective function for the selected lag
        def objective(x):
            shifted = np.roll(x, max_lag)
            return np.sum(x * (1 - shifted)) * dx
        
        # Perform optimization
        try:
            res = opt.minimize(
                objective, h_vals, method='L-BFGS-B', bounds=bounds, constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
            )
            h_vals = np.clip(res.x, 0.0, 1.0)
            # Ensure sum constraint is met
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        except:
            # Fallback: use previous best
            h_vals = best_h_vals.copy()
    
    # Final evaluation
    final_corr = np.correlate(h_vals, 1 - h_vals, mode='full')
    final_c5 = np.max(final_corr) * dx
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 100:
        h_vals = best_h_vals
        final_c5 = best_c5
    
    return (h_vals, final_c5, n_points)
```
