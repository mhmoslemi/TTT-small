def run(seed=42, budget_s=600, **kwargs):

        h = np.clip(h, 0, 1)
    
    np.random.seed(seed)
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    learning_rate = 0.01
    decay_rate = 0.999
    iteration = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Compute overlaps
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        k_max = np.argmax(overlaps)
        
        # Compute gradient for the worst lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        
        # Update h
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 1e-10:
            h = h * (n_points / 2) / sum_h
        
        # Evaluate new h
        new_c5 = evaluate_erdos_solution(h)
        
        # Update best values
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            iteration = 0
        else:
            iteration += 1
        
        # Perturb every 500 iterations
        if iteration % 500 == 0:
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h > 1e-10:
                h = h * (n_points / 2) / sum_h
        
        # Decay learning rate
        if iteration % 100 == 0:
            learning_rate *= decay_rate
    
    # Final cleanup
    h_values = np.clip(best_h, 0, 1)
    sum_h = np.sum(h_values)
    if sum_h > 1e-10:
        h_values = h_values * (n_points / 2) / sum_h
    return h_values, best_c5, n_points
