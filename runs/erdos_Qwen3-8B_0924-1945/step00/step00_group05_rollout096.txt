import numpy as np
import random
import time
import scipy.optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h.npy') if 'initial_h_values' in locals() else np.zeros(70)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        h = h * (n_points / 2) / sum_h
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.inf
    time_start = time.time()
    
    # Optimizer settings
    POP_SIZE = 50
    MUTATION_RATE = 0.1
    MUTATION_NOISE = 0.05
    N_SELECT = 20
    MAX_STAGNATION = 50
    STAGNATION_COUNTER = 0
    
    def evaluate(h):
        nonlocal best_c5, best_h
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (n_points / 2) / sum_h
        h = np.clip(h, 0, 1)
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(overlaps)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    def tournament_selection(population, fitnesses):
        indices = np.random.randint(0, len(population), 5)
        best_idx = np.argmin(fitnesses[indices])
        return population[indices[best_idx]]
    
    def crossover(parent1, parent2):
        idx = np.random.randint(1, len(parent1) - 1)
        child = np.concatenate([parent1[:idx], parent2[idx:]])
        child = np.clip(child, 0, 1)
        sum_child = np.sum(child)
        if sum_child != 0:
            child = child * (n_points / 2) / sum_child
        child = np.clip(child, 0, 1)
        return child
    
    def mutate(h):
        h = h.copy()
        for i in range(len(h)):
            if np.random.random() < MUTATION_RATE:
                h[i] += np.random.normal(0, MUTATION_NOISE)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (n_points / 2) / sum_h
        h = np.clip(h, 0, 1)
        return h
    
    def evolve(population, fitnesses):
        new_population = []
        for _ in range(POP_SIZE):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        return new_population
    
    population = [h.copy() for _ in range(POP_SIZE)]
    fitnesses = np.array([evaluate(h) for h in population])
    
    while time.time() - time_start < budget_s:
        population = evolve(population, fitnesses)
        fitnesses = np.array([evaluate(h) for h in population])
        if np.min(fitnesses) < best_c5:
            best_c5 = np.min(fitnesses)
            best_h = population[np.argmin(fitnesses)].copy()
            STAGNATION_COUNTER = 0
        else:
            STAGNATION_COUNTER += 1
        if STAGNATION_COUNTER >= MAX_STAGNATION:
            MUTATION_NOISE *= 0.5
            STAGNATION_COUNTER = 0
        if np.min(fitnesses) < best_c5:
            best_c5 = np.min(fitnesses)
            best_h = population[np.argmin(fitnesses)].copy()
    
    h_values = best_h.copy()
    h_values = np.clip(h_values, 0, 1)
    sum_h = np.sum(h_values)
    if sum_h != 0:
        h_values = h_values * (n_points / 2) / sum_h
    h_values = np.clip(h_values, 0, 1)
    c5_bound = evaluate(h_values)
    return (h_values, c5_bound, n_points)
