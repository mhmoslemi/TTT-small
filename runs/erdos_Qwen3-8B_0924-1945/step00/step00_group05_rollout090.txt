import numpy as np
import random
import time
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Default starting point; use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        n_points = 40
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = np.concatenate([h, np.zeros(n_points//2)])
        h = np.random.permutation(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Function to normalize h to satisfy sum constraint
    def normalize(h):
        current_sum = np.sum(h)
        if current_sum == 0:
            return h
        scale = target_sum / current_sum
        return np.clip(h * scale, 0, 1)
    
    # Objective function to minimize the maximum overlap
    def objective(h_vec):
        h = h_vec.copy()
        h = normalize(h)
        h = np.clip(h, 0, 1)
        h = h * dx
        shifted = 1 - h
        corr = correlate(h, shifted, mode='full')
        c5 = np.max(corr)
        return c5
    
    # Constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization using differential evolution
    start_time = time.time()
    result = differential_evolution(objective, bounds, strategy='best1bin',
                                   popsize=15, maxiter=100, tol=1e-4,
                                   mutation=(0.5, 1), recombination=0.8,
                                   dispatch='multiprocessing', seed=seed)
    
    # Extract the best solution and compute C5
    best_h = result.x
    best_h = normalize(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * dx
    corr = correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr)
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        print("Time exceeded; returning best found so far.")
    
    return (best_h, c5_bound, n_points)
