import numpy as np
import random
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 45  # Use the same number of points as the initial construction
    dx = 2.0 / n_points
    
    # Initialize h_values
    h_values = np.copy(initial_h_values)
    # Scale initial h to satisfy the integral constraint
    target_sum = n_points / 2
    h_values = h_values.copy() / np.sum(h_values) * target_sum
    
    # Function to compute C5
    def compute_c5(h):
        h_reshaped = h.reshape(-1, 1)
        ones_minus_h = np.ones_like(h_reshaped) - h_reshaped
        corr = np.correlate(h_reshaped, ones_minus_h, mode='full')
        return np.max(corr) * dx
    
    best_h = h_values.copy()
    best_c5 = compute_c5(best_h)
    
    # Optimization parameters
    num_iterations = 300
    perturbation_size = 0.1
    for _ in range(num_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Perturb 5 random indices
        indices = np.random.choice(n_points, 5, replace=False)
        h = h_values.copy()
        for idx in indices:
            noise = np.random.normal(0, perturbation_size)
            h[idx] += noise
            h[idx] = np.clip(h[idx], 0, 1)
        
        # Rescale to meet the integral constraint
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (target_sum / h_sum)
        
        # Compute C5 for the new h
        current_c5 = compute_c5(h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    # Final sanity check and return
    # Ensure the best_h meets constraints
    best_h = np.clip(best_h, 0, 1)
    best_sum = np.sum(best_h)
    if best_sum != 0:
        best_h = best_h * (target_sum / best_sum)
    
    return (best_h, best_c5, n_points)
