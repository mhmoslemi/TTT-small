import numpy as np
import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 45
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = float('inf')
    time_start = time.time()
    learning_rate = 0.1
    decay_rate = 0.999
    noise_prob = 0.05
    noise_scale = 0.05
    max_iter = 1000
    early_stop_iter = 100
    early_stop_tol = 1e-5

    for iter in range(max_iter):
        elapsed = time.time() - time_start
        if elapsed > budget_s:
            break
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        current_c5 = np.max(correlate_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Select the lag with maximum value
        max_index = np.argmax(correlate_vals)
        m = max_index - n_points + 1
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = i - m
            if 0 <= j < n_points:
                gradients[i] = (1 - h[j]) * (2.0 / n_points)
        h -= learning_rate * gradients
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        # Add random noise
        if random.random() < noise_prob:
            noise = np.random.normal(0, noise_scale, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        # Decay learning rate
        learning_rate *= decay_rate
        # Early stopping
        if iter > 0 and abs(current_c5 - best_c5) < early_stop_tol:
            if iter - early_stop_iter > 0:
                break

    # Final check
    h = best_h
    correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
    final_c5 = np.max(correlate_vals)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return (h.tolist(), final_c5, n_points)
