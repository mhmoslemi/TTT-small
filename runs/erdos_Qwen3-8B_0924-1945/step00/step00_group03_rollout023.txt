import numpy as np
import scipy.optimize as optimize
import time

initial_h_values = None  # Placeholder for parent construction if available

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70  # Starting with the initial 70-point construction
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float) if initial_h_values is not None else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-6:
        h = (h * (n_points / 2) / sum_h) if sum_h != 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)
    
    def objective(x):
        nonlocal n_points, dx
        x = np.clip(x, 0, 1)
        conv = np.correlate(x, 1 - x, mode="full")
        max_overlap = np.max(conv) * dx
        return max_overlap
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1)] * n_points
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(h)
    
    def optimize_h(h_init):
        nonlocal best_h, best_c5
        result = optimize.minimize(
            objective,
            h_init,
            bounds=bounds,
            constraints=constraints,
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-5, 'eps': 1e-6}
        )
        if result.success:
            h_opt = np.clip(result.x, 0, 1)
            sum_h_opt = np.sum(h_opt)
            if abs(sum_h_opt - n_points / 2) > 1e-6:
                h_opt = (h_opt * (n_points / 2) / sum_h_opt) if sum_h_opt != 0 else np.zeros(n_points)
                h_opt = np.clip(h_opt, 0, 1)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
            return h_opt
        else:
            print("Optimization failed; trying perturbation")
            perturb = np.random.uniform(-0.1, 0.1, size=n_points)
            h_perturb = np.clip(h_init + perturb, 0, 1)
            sum_h_perturb = np.sum(h_perturb)
            if abs(sum_h_perturb - n_points / 2) > 1e-6:
                h_perturb = (h_perturb * (n_points / 2) / sum_h_perturb) if sum_h_perturb != 0 else np.zeros(n_points)
                h_perturb = np.clip(h_perturb, 0, 1)
            c5_perturb = objective(h_perturb)
            if c5_perturb < best_c5:
                best_h = h_perturb
                best_c5 = c5_perturb
            return h_perturb
    
    # Try initial optimization
    h_opt = optimize_h(h)
    elapsed = time.time() - start_time
    if elapsed < budget_s - 10:
        # Try one more optimized perturbation
        h_opt = optimize_h(h_opt)
    
    # Clamp and check sum
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if abs(sum_best - n_points / 2) > 1e-6:
        best_h = (best_h * (n_points / 2) / sum_best) if sum_best != 0 else np.zeros(n_points)
        best_h = np.clip(best_h, 0, 1)
    
    # Ensure we have a valid solution
    final_c5 = objective(best_h)
    return (best_h.tolist(), final_c5, n_points)
