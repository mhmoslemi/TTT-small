        cross_corr = np.correlate(h, 1 - h, mode="full") * dx
        max_index = np.argmax(cross_corr)
        max_c5 = cross_corr[max_index]

        # Compute directional gradient
        h_forward = h.copy()
        h_backward = h.copy()
        h_forward[max_index] += perturbation
        h_backward[max_index] -= perturbation

        # Clip and rescale
        h_forward = np.clip(h_forward, 0, 1)
        h_forward = h_forward * (target_sum / np.sum(h_forward))
        h_backward = np.clip(h_backward, 0, 1)
        h_backward = h_backward * (target_sum / np.sum(h_backward))

        # Evaluate forward and backward
        c5_forward = evaluate_erdos_solution(h_forward)
        c5_backward = evaluate_erdos_solution(h_backward)

        # Compute gradient
        grad = (c5_forward - max_c5) / perturbation
        grad = (grad + (max_c5 - c5_backward) / perturbation) / 2

        # Update h
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))

        # Evaluate new C5
        new_c5 = evaluate_erdos_solution(h)

        # Update best solution
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5

        # Early stopping
        if i > 0 and abs(new_c5 - best_c5) < improvement_threshold * best_c5:
            if i > early_stop:
                break

        # Decay step size
        alpha *= decay_rate

        elapsed_time += 1

    return best_h, best_c5, n_points
