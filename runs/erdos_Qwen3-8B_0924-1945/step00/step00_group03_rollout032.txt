    def objective_func(h_vec):
        h_vec = np.clip(h_vec, 0.0, 1.0)
        if np.sum(h_vec) != n_points / 2:
            return np.inf
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        c5_val = np.max(conv) * dx
        return c5_val

        return np.sum(h_vec) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_func}]
    
    result = optimize.minimize(
        objective_func,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-5,
        options={'maxiter': 100, 'ftol': 1e-5}
    )
    
    if result.success:
        best_h = np.clip(result.x, 0.0, 1.0)
        best_c5 = objective_func(best_h)
    else:
        # Perturb initial solution with noise and retry
        noise = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = np.clip(h + noise, 0.0, 1.0)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        result = optimize.minimize(
            objective_func,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            tol=1e-5,
            options={'maxiter': 100, 'ftol': 1e-5}
        )
        best_h = np.clip(result.x, 0.0, 1.0)
        best_c5 = objective_func(best_h)

        best_h = best_h * (n_points / 2) / np.sum(best_h)
