```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    max_iterations = 500
    initial_h_values = np.random.rand(n_points) * 2 - 1
    initial_h_values = np.clip(initial_h_values, 0, 1)
    sum_initial = np.sum(initial_h_values)
    h = initial_h_values * (n_points / 2) / sum_initial if sum_initial != 0 else initial_h_values * (n_points / 2)
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.inf
        h = h * (n_points / 2) / sum_h
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv * dx)
        return c5
    
    for _ in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        conv = np.correlate(h, 1 - h, mode="full")
        conv = conv * dx
        max_idx = np.argmax(conv)
        if max_idx >= len(conv) or max_idx < 0:
            continue
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + max_idx) % n_points
            if j < n_points:
                grad[i] += (1 - h[j])
            j = (i - max_idx) % n_points
            if j < n_points:
                grad[i] -= h[i]
        
        learning_rate = 0.01 * (0.999 ** _)
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = best_h.copy()
            continue
        h = h * (n_points / 2) / sum_h
        
        if np.random.rand() < 0.05:
            noise = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + noise, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = best_h.copy()
            else:
                h = h * (n_points / 2) / sum_h
    
    final_h = np.clip(best_h, 0, 1)
    sum_final = np.sum(final_h)
    if sum_final == 0:
        final_h = initial_h_values * (n_points / 2)
    else:
        final_h = final_h * (n_points / 2) / sum_final
    
    final_c5 = evaluate_c5(final_h)
    return (final_h.tolist(), final_c5, n_points)
```
