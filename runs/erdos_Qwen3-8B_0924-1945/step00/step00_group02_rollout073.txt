import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.loadtxt('initial_h_values.txt') if 'initial_h_values' in kwargs else None
    n_points = 250
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h = np.clip(initial_h_values, 0, 1)
        h = h * n_points / 2.0 / np.sum(h)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * n_points / 2.0 / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
    T = 100.0
    alpha = 0.995
    start_time = time.time()
    
    for t in range(int((budget_s - (time.time() - start_time)) / 0.1)):
        if time.time() - start_time > budget_s:
            break
        indices = np.random.choice(n_points, 5, replace=False)
        noise = np.random.normal(0, 0.05 * np.exp(-t / 200), 5)
        new_h = h.copy()
        for i, idx in enumerate(indices):
            new_h[idx] += noise[i]
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * n_points / 2.0 / np.sum(new_h)
        
        corr = np.correlate(new_h, 1 - new_h, mode="full")
        c5_new = np.max(corr) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = new_h.copy()
        
        if np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            h = new_h.copy()
        
        T *= alpha
    
    return (best_h, best_c5, n_points)
