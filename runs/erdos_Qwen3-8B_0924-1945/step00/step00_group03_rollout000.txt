import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in globals() else np.random.rand(70)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = n_points / 2
    h = np.copy(initial_h_values)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    elapsed = 0
    t_total = int(budget_s * 0.75)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return max(corr) * dx
    
    def perturb(h, temp, alpha, beta):
        noise = np.random.normal(0, alpha / (temp ** 0.5), size=n_points)
        perturbed = h + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        delta = (sum_h - current_sum) / (n_points * 0.95)
        perturbed += delta * np.random.choice([-1, 1], size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        if np.sum(perturbed) != 0:
            perturbed *= sum_h / np.sum(perturbed)
        return perturbed
    
    T0 = max(1.0, 1.0 / (1.0 + best_c5 * 2.0)) * 0.5
    Tmax = int(t_total * 0.1)
    beta = 0.75
    gamma = 0.65
    lambda0 = 0.15 / dx
    
    for t in range(t_total):
        if elapsed > budget_s:
            break
        temp = T0 * (1 - t / Tmax) ** (1 / beta)
        lambda_amp = lambda0 * (1 - t / Tmax) ** (1 / gamma)
        
        h_perturbed = perturb(h, temp, lambda_amp, beta)
        new_c5 = evaluate(h_perturbed)
        
        if new_c5 < best_c5 or (np.random.rand() < np.exp(-(new_c5 - best_c5) / temp)):
            h = h_perturbed.copy()
            best_c5 = new_c5
            best_h = h.copy()
        
        elapsed += 1
    
    final_c5 = evaluate(best_h)
    return best_h, final_c5, n_points
