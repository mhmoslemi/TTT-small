import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.random.uniform(0, 1, n_points)
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    scale = (n_points / 2) / total if total != 0 else 1.0
    h *= scale
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    start_time = time.time()
    
    dt = 0.1
    iterations = int((budget_s - (time.time() - start_time)) / dt)
    for _ in range(iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        lag = np.argmax(corr)
        overlap = h * (np.roll(1 - h, lag))
        overlap = np.clip(overlap, 0, 1)
        threshold = np.max(overlap) * 0.8
        target_indices = np.where(overlap > threshold)[0]
        if len(target_indices) < 1:
            break
        target_indices = target_indices[:int(len(target_indices) * 0.1)]
        candidates = []
        for idx in target_indices:
            h_candidate = h.copy()
            h_candidate[idx] = 1 - h_candidate[idx]
            h_candidate[idx] += np.random.uniform(-0.05, 0.05)
            h_candidate = np.clip(h_candidate, 0, 1)
            total_candidate = np.sum(h_candidate)
            scale_candidate = (n_points / 2) / total_candidate if total_candidate != 0 else 1.0
            h_candidate *= scale_candidate
            h_candidate = np.clip(h_candidate, 0, 1)
            candidates.append(h_candidate)
        for candidate in candidates:
            c5 = evaluate_erdos_solution(candidate, n_points)
            if c5 < best_c5:
                best_h = candidate.copy()
                best_c5 = c5
        h = best_h.copy()
    
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    scale = (n_points / 2) / total if total != 0 else 1.0
    h *= scale
    h = np.clip(h, 0, 1)
    best_c5 = evaluate_erdos_solution(h, n_points)
    return (h, best_c5, n_points)

def evaluate_erdos_solution(h, n_points):
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    c5 = max_corr * 2.0 / n_points
    return c5
