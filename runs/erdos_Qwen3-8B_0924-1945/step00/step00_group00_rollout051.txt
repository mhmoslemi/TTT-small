import numpy as np
import scipy.signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 90  # Number of points for the discretization
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in globals() else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    total_integral = np.sum(h) * dx
    if abs(total_integral - 1) > 1e-6:
        h = h * (1 / total_integral) * dx
    h = h / dx
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    h_vals = h.copy()
    best_c5 = np.inf
    best_h = h_vals.copy()
    time_start = time.time()
    learning_rate = 0.01
    for _ in range(1000):
        if time.time() - time_start > budget_s:
            break
        correlate = scipy.signal.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(correlate) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_vals.copy()
        # Find the worst lag
        lag = np.argmax(correlate)
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i - lag
            if 0 <= j < n_points:
                grad[i] = (1 - h_vals[j]) * dx
        # Update h_vals
        h_vals = h_vals - learning_rate * grad
        # Clip values to [0, 1]
        h_vals = np.clip(h_vals, 0, 1)
        # Normalize to satisfy integral constraint
        total_integral = np.sum(h_vals) * dx
        if abs(total_integral - 1) > 1e-6:
            h_vals = h_vals / total_integral * 1
        h_vals = h_vals / dx
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        h_vals = np.clip(h_vals, 0, 1)
    # Final check
    correlate = scipy.signal.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlate) * dx
    return (best_h, final_c5, n_points)
