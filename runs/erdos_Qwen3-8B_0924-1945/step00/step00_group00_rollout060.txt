import numpy as np
import random
import time
from scipy.optimize import minimize
from scipy.stats import uniform
from sklearn.metrics.pairwise import cosine_similarity

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed
    np.random.seed(seed)
    random.seed(seed)
    
    # Predefined constants
    default_n_points = 45
    n_points = default_n_points
    
    # Load initial solution or create random one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.uniform(0, 1, n_points)
        h /= np.sum(h)
    
    # Normalize to [0,1] and ensure sum constraint
    h = np.clip(h, 0, 1)
    h /= np.sum(h)
    
    # Function to compute C5
    def compute_c5(h, n_points):
        dx = 2.0 / n_points
        h_vec = np.array(h, dtype=np.float64)
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Objective function for optimization
    def objective(x, n_points):
        h = x.copy()
        h = np.clip(h, 0, 1)
        h /= np.sum(h)
        return compute_c5(h, n_points)
    
    # Constraint: sum(h) = n_points / 2
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    cons = ({'type': 'eq', 'fun': constraint})
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial optimization
    x0 = h.copy()
    result = minimize(objective, x0, args=(n_points,), method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    
    best_h = result.x
    best_c5 = compute_c5(best_h, n_points)
    
    # Local search with perturbations and restarts
    for _ in range(5):
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        h_perturb = best_h + perturb
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb /= np.sum(h_perturb)
        c5_perturb = compute_c5(h_perturb, n_points)
        if c5_perturb < best_c5:
            best_h = h_perturb
            best_c5 = c5_perturb
    
    # Final check and return
    best_h = np.clip(best_h, 0, 1)
    best_h /= np.sum(best_h)
    final_c5 = compute_c5(best_h, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
