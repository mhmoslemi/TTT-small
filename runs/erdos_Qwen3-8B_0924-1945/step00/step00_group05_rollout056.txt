    target_sum = n_points / 2

            if i < n_points // 2:
                h[i] = 1.0
            else:
                h[i] = 0.0
    
    # Adjust to meet the sum constraint
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-6:
        scale = target_sum / current_sum
        h *= scale
        h[h > 1.0] = 1.0
        h[h < 0.0] = 0.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iterations = 0
    
    # Optimization loop
    while time.time() - start_time < budget_s:
        iterations += 1
        
        # Introduce small random perturbations to avoid local minima
        perturbation = np.random.normal(0, 0.05, n_points)
        h_perturbed = h + perturbation
        
        # Clip values to [0,1]
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)

        current_sum_perturbed = np.sum(h_perturbed)

            h_perturbed *= scale
        
        # Evaluate overlap integral
        h_perturbed = h_perturbed.reshape(-1, 1)
        one_minus_h = 1.0 - h_perturbed
        correlate_result = np.correlate(h_perturbed, one_minus_h, mode='full')
