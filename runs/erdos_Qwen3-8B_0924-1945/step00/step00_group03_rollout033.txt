import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70  # Start with the given initial construction
    initial_h_values = np.array(kwargs.get('initial_h_values', np.random.rand(n_points)))
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    try:
        result = opt.minimize(objective, initial_h_values, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5, options={'maxiter': 100})
        h_opt = result.x
    except:
        h_opt = initial_h_values
    
    h_opt = np.clip(h_opt, 0, 1)
    if np.abs(np.sum(h_opt) - n_points / 2) > 1e-5:
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        h_opt = np.clip(h_opt, 0, 1)
    
    c5_bound = objective(h_opt)
    return (h_opt.tolist(), c5_bound, n_points)
