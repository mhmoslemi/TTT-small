import numpy as np
import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load("initial_h.npy")
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_sum = np.sum(initial_h_values)
    if abs(initial_sum - n_points / 2) > 1e-6:
        initial_h_values = (initial_h_values / initial_sum) * (n_points / 2)
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = np.inf

    def evaluate(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - n_points / 2) > 1e-6:
            h = (h / total) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def mutate(h, strength=0.1):
        h = h.copy()
        indices = np.random.choice(n_points, int(n_points * 0.1), replace=False)
        h[indices] += np.random.uniform(-strength, strength, len(indices))
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - n_points / 2) > 1e-6:
            h = (h / total) * (n_points / 2)
        return h

    def crossover(h1, h2):
        h = h1.copy()
        indices = np.random.choice(n_points, int(n_points * 0.5), replace=False)
        h[indices] = h2[indices]
        total = np.sum(h)
        if abs(total - n_points / 2) > 1e-6:
            h = (h / total) * (n_points / 2)
        return h

    def tournament_selection(population, size=5):
        candidates = np.random.choice(population, size, replace=False)
        return min(candidates, key=evaluate)

    population = [mutate(h, 0.15) for _ in range(50)]
    population = [mutate(mutate(h, 0.15), 0.1) for h in population]
    start_time = time.time()
    while time.time() - start_time < budget_s:
        population = [mutate(h, 0.1) for h in population]
        top = sorted(population, key=evaluate)[:20]
        new_population = [crossover(top[i], top[(i+1)%20]) for i in range(20)]
        new_population += [mutate(h, 0.15) for _ in range(30)]
        population = new_population
        current = evaluate(best_h)
        for h in population:
            c5 = evaluate(h)
            if c5 < current:
                best_h = h.copy()
                best_c5 = c5
    return best_h, best_c5, n_points
