```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None  # Provided externally, not used here

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Start with n_points = 250 for better performance within budget
    n_points = 250
    dx = 2.0 / n_points
    if initial_h_values is not None and len(initial_h_values) == n_points:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    T = 100.0
    alpha = 0.995
    max_iterations = int(budget_s / 0.1)
    start_time = time.time()
    
    for t in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Perturb 5 random samples
        indices = np.random.choice(n_points, 5, replace=False)
        noise = np.random.normal(0, 0.05 * np.exp(-t / 200), 5)
        new_h = h.copy()
        new_h[indices] += noise
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        new_h = np.clip(new_h, 0, 1)
        
        c5_new = evaluate_c5(new_h)
        if c5_new < best_c5:
            best_h = new_h.copy()
            best_c5 = c5_new
        else:
            # Accept with probability based on temperature
            delta = c5_new - best_c5
            if np.random.random() < np.exp(-delta / T):
                best_h = new_h.copy()
                best_c5 = c5_new
        
        T *= alpha
    
    return best_h, best_c5, n_points
```
