import numpy as np
import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 45 if initial_h_values is None else len(initial_h_values)
    dx = 2.0 / n_points
    n_particles = 20
    max_iterations = 100
    inertia_weight = 0.7
    c1, c2 = 1.5, 1.5
    best_h = np.copy(initial_h_values) if initial_h_values is not None else np.random.rand(n_points)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (n_points / 2.0) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    particles = [np.copy(best_h) for _ in range(n_particles)]
    velocities = [np.random.rand(n_points) * 0.1 for _ in range(n_particles)]
    p_best = [np.copy(p) for p in particles]
    g_best = np.copy(best_h)
    
    start_time = time.time()
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):
            h = particles[i]
            current_c5 = evaluate_erdos_solution(h, n_points)
            if current_c5 < evaluate_erdos_solution(p_best[i], n_points):
                p_best[i] = np.copy(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                g_best = np.copy(h)
        
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = inertia_weight * velocities[i] + c1 * r1 * (p_best[i] - particles[i]) + c2 * r2 * (g_best - particles[i])
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            particles[i] = particles[i] * (n_points / 2.0) / np.sum(particles[i])
        
        if iteration % 10 == 0:
            for i in range(n_particles):
                if np.random.rand() < 0.1:
                    perturb = np.random.normal(0, 0.05, n_points)
                    particles[i] = np.clip(particles[i] + perturb, 0.0, 1.0)
                    particles[i] = particles[i] * (n_points / 2.0) / np.sum(particles[i])
    
    best_h = np.copy(g_best)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (n_points / 2.0) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    
    return (best_h, best_c5, n_points)
