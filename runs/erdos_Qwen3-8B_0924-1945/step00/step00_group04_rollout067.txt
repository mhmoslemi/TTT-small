import numpy as np
import scipy.optimize as optimize
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    target_sum = n_points / 2
    dx = 2.0 / n_points

    if initial_h_values is not None:
        h = np.clip(initial_h_values, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.zeros(n_points)
        h[0::2] = 1
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    elapsed_time = 0
    learning_rate = 0.05
    max_iterations = 1000
    no_improvement_count = 0
    iteration = 0

    def compute_gradient(h, lag_k):
        n = len(h)
        grad = np.zeros(n)
        for j in range(n):
            if j + lag_k < n:
                grad[j] += (1 - h[j + lag_k]) * dx
            if j - lag_k >= 0:
                grad[j] -= h[j - lag_k] * dx
        return grad

    def update_h(h, grad, learning_rate):
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            h = np.zeros(n_points)
        else:
            h = h * (target_sum / current_sum)
        return h

    start_time = time.time()

    while elapsed_time < budget_s:
        iteration += 1
        cross_corr = np.correlate(h, 1 - h, mode="full") * dx
        idx = np.argmax(cross_corr)
        lag_k = idx - (n_points - 1)

        grad = compute_gradient(h, lag_k)
        h = update_h(h, grad, learning_rate)

        new_c5 = evaluate_erdos_solution(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        if no_improvement_count >= 50:
            break

        elapsed_time = time.time() - start_time

        if iteration % 100 == 0:
            learning_rate *= 0.995

    return (best_h, best_c5, n_points)
