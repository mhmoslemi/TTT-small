import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None  # Placeholder, defined by parent
dx = 2.0  # Will be adjusted based on n_points

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values, dx
    np.random.seed(seed)
    n_points = len(initial_h_values) if initial_h_values is not None else 100
    dx = 2.0 / n_points
    h_vals = np.array(initial_h_values, dtype=float) if initial_h_values is not None else np.random.rand(n_points)
    h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    h_vals = np.clip(h_vals, 0, 1)
    h_vals = h_vals * (n_points / 2) / np.sum(h_vals)

    max_iterations = 100
    improvement_threshold = 1e-5
    best_c5 = float('inf')
    best_h_vals = h_vals.copy()
    tol = 1e-5
    start_time = time.time()
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5_vals = corr * dx
        max_c5 = np.max(c5_vals)
        if np.abs(max_c5 - best_c5) < improvement_threshold:
            break
        best_c5 = max_c5
        k = np.argmax(corr)
        if k >= len(h_vals):
            k = len(h_vals) - 1
        if k < 0:
            k = 0
        # Define objective
        def objective(x):
            h = x.reshape(-1)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            return overlap
        # Constraints
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.minimize(objective, h_vals, method='SLSQP', bounds=bounds, constraints=cons, tol=tol)
        if result.success:
            h_vals = result.x
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
            new_c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h_vals = h_vals.copy()
        else:
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
    final_corr = np.correlate(best_h_vals, 1 - best_h_vals, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (best_h_vals, final_c5, n_points)
