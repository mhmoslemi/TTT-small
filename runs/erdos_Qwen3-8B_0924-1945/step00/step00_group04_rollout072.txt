import numpy as np
import scipy.optimize as opt
import sys

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
    h = initial_h_values.copy()
    target_sum = n_points / 2
    dx = 2.0 / n_points

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    bounds = [(0, 1)] * n_points
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100, 'ftol': 1e-6, 'disp': False}
    )

    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
