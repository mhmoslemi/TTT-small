import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Preloaded
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    def objective_func(h_values):
        conv = np.correlate(h_values, 1 - h_values, mode='full')
        max_overlap = np.max(conv) * dx
        return max_overlap
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial optimization with SLSQP
    result = optimize.minimize(
        objective_func,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-5, 'eps': 1e-6}
    )
    
    best_h = result.x.copy()

        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = np.clip(initial_h_values + perturbation, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        result = optimize.minimize(

            perturbed_h,
            method='SLSQP',
            bounds=bounds,

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = objective_func(best_h)
    
    if elapsed >= budget_s - 5:
        return (best_h.tolist(), final_c5, n_points)
    else:
        return (best_h.tolist(), final_c5, n_points)
