import scipy.optimize
import time
import random

    np.random.seed(seed)
    random.seed(seed)

    mutation_rate = 0.1
    generations = 50
    num_local_refinements = 3
    
    # Local Refinement Parameters
    local_maxiter = 100
    local_tolerance = 1e-6
    
    # Main Loop
    for _ in range(generations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Generate population
        population = [best_h.copy() for _ in range(population_size)]
        for i in range(population_size):
            perturb = np.random.uniform(-0.05, 0.05, size=n_points)
            population[i] += perturb
        
        # Evaluate and repair
        for i in range(population_size):
            h = population[i].copy()
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            if np.abs(current_sum - target_sum) > 1e-6:
                scaling = target_sum / current_sum
                h = h * scaling
            h = np.clip(h, 0, 1)
            population[i] = h
        
        # Evaluate fitness
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr * dx)
            fitness.append(c5)
        
        # Select top 50 candidates
        top_indices = np.argsort(fitness)[:50]
        top_candidates = [population[i] for i in top_indices]
        
        # Crossover and mutation
        new_population = []
        for i in range(0, len(top_candidates), 2):
            if i + 1 < len(top_candidates):
                parent1 = top_candidates[i]
                parent2 = top_candidates[i + 1]
                # Crossover: random segment swap
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:split], parent2[split:]])
                new_population.append(child)
            else:
                new_population.append(top_candidates[i])
        
        # Mutate
        for h in new_population:
            if np.random.rand() < mutation_rate:
                perturb = np.random.uniform(-0.05, 0.05, size=n_points)
                h += perturb
                h = np.clip(h, 0, 1)
                current_sum = np.sum(h)
                if np.abs(current_sum - target_sum) > 1e-6:
                    scaling = target_sum / current_sum
                    h = h * scaling
                h = np.clip(h, 0, 1)
        
        # Update population
        population = new_population
        for h in population:
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            if np.abs(current_sum - target_sum) > 1e-6:
                scaling = target_sum / current_sum
                h = h * scaling
            h = np.clip(h, 0, 1)

        for h in population:
            corr = np.correlate(h, 1 - h, mode="full")

            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5

        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Define objective function for SLSQP
        def objective(x):
            corr = np.correlate(x, 1 - x, mode="full")
            return -np.max(corr * dx)  # Minimize negative is maximize
        
        # Constraints
        def constraint_sum(x):
            return np.sum(x) - target_sum
        
        # Bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Initial guess
        x0 = best_h.copy()
        
        # Optimization
        result = scipy.optimize.minimize(
            objective,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            tol=local_tolerance,
            options={"maxiter": local_maxiter}
        )
        
        if result.success:
            best_h = result.x
            best_c5 = evaluate_erdos_solution(best_h)
    
    # Final feasibility check

    current_sum = np.sum(best_h)
    if np.abs(current_sum - target_sum) > 1e-6:
        scaling = target_sum / current_sum
        best_h = best_h * scaling
    best_h = np.clip(best_h, 0, 1)
    best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
