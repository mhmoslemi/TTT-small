import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * target_sum

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        scale = target_sum / np.sum(h)
        h = h * scale
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    def constraints(h_flat):
        h = h_flat.reshape(n_points)
        return [np.sum(h) - target_sum]

    bounds = [(0, 1) for _ in range(n_points)]

    result = opt.minimize(
        objective,
        h.flatten(),
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        options={'ftol': 1e-8, 'maxiter': 100}
    )

    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0, 1)
    scale = target_sum / np.sum(h_opt)
    h_opt = h_opt * scale

    corr = np.correlate(h_opt, 1 - h_opt, mode='full')
    c5_bound = np.max(corr) * dx

    return (h_opt, c5_bound, n_points)
