import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 50
    if 'initial_h_values' in globals():
        h_vals = np.array(initial_h_values, dtype=float)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    else:
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    dx = 2.0 / n_points
    max_iter = 200
    tol = 1e-6
    best_c5 = float('inf')
    best_h = h_vals.copy()
    start_time = time.time()
    
    for _ in range(max_iter):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        current_c5 = np.max(corr * dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_vals.copy()
        
        if time.time() - start_time > budget_s - 10:
            break
        
        k_max = np.argmax(corr)
        overlap = corr[k_max] * dx
        lambda_var = cp.Variable()
        objective = 0.5 * lambda_var**2 + (overlap - lambda_var)**2
        prob = cp.Problem(cp.Minimize(objective))
        try:
            prob.solve()
            lambda_val = lambda_var.value[0]
        except:
            lambda_val = 0.0
        
        grad = (1 - h_vals) * (lambda_val - 1) + h_vals * (lambda_val)
        h_vals += grad * 0.1
        h_vals = np.clip(h_vals, 0, 1)
        mass = np.sum(h_vals)
        h_vals = h_vals * (n_points / 2) / mass
    
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr * dx)
    return (best_h, final_c5, n_points)
