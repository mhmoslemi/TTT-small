import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    
    n_points = 1000
    dx = 2.0 / n_points
    max_iter = 500
    learning_rate = 0.01
    momentum = 0.9
    perturb_prob = 0.05
    min_improvement = 1e-6
    
    if initial_h_values is None:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 * (1 + np.sin(np.pi * i / n_points))
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h
    else:
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h

    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Compute overlap integrals
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = corr * dx
        c5_max = np.max(c5)
        
        if c5_max < best_c5:
            best_c5 = c5_max
            best_h = h.copy()
        
        # Find the lag with maximum overlap
        max_k = np.argmax(c5)
        
        # Compute gradient for the max overlap lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            # H[i] contributes to overlap with lag max_k
            j = (i + max_k) % n_points
            grad[i] += (1 - h[j])  # from h[i]
            grad[j] -= h[i]        # from (1-h)[j]
        
        # Update h with momentum and learning rate
        h = h - learning_rate * grad
        
        # Clip h to [0,1]
        h = np.clip(h, 0, 1)
        
        # Ensure sum constraint
        sum_h = np.sum(h)
        if sum_h == 0:
            h = best_h.copy()
        else:
            h = h * (n_points / 2) / sum_h
        
        # Perturbation
        if np.random.rand() < perturb_prob:
            noise = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + noise, 0, 1)
            sum_h = np.sum(h)
            h = h * (n_points / 2) / sum_h
        
        # Early stopping if no improvement
        if iter > 50 and np.abs(c5_max - best_c5) < min_improvement:
            break
    
    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode="full")
    c5_bound = corr * dx
    c5_bound = np.max(c5_bound)
    
    return (best_h, c5_bound, n_points)
