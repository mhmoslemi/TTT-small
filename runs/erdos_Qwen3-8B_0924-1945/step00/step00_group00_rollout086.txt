import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 100
    initial_h = initial_h_values if 'initial_h_values' in kwargs else np.random.rand(n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    def objective_func(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full') * (2.0 / n_points)
        return np.max(corr)
    
    def gradient_func(h, lag):
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i - lag
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * (2.0 / n_points)
        return grad
    
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = objective_func(h)
    iter_count = 0
    start_time = time.time()
    
    learning_rate = 0.1
    noise_scale = 0.05
    p_noise = 0.05
    max_iter = 1000
    early_stop = 100
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        iter_count += 1
        corr = np.correlate(h, 1 - h, mode='full') * (2.0 / n_points)
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        lag = np.argmax(corr)
        grad = gradient_func(h, lag)
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        if np.random.rand() < p_noise:
            noise = np.random.normal(0, noise_scale, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        if iter_count % 100 == 0 and iter_count > early_stop:
            if np.abs(current_c5 - best_c5) < 1e-5:
                break
    
    return (best_h, best_c5, n_points)
