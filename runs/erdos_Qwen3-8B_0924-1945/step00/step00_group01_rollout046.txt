import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h_vals = (initial_h_values * (n_points / 2)) / np.sum(initial_h_values)
    else:
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total
    
    def mutate(h, mutation_rate):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.uniform(-0.05, 0.05)
                h[i] = np.clip(h[i], 0, 1)
        return clip_and_normalize(h)
    
    def crossover(parent1, parent2):
        if np.random.rand() < 0.5:
            return parent1
        point = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:point], parent2[point:]])
        return clip_and_normalize(child)
    
    def tournament_selection(population, k=5):
        selected = np.random.choice(population, size=k)
        return min(selected, key=evaluate)
    
    population = [clip_and_normalize(np.random.rand(n_points)) for _ in range(50)]
    best_h = None
    best_c5 = float('inf')
    generation = 0
    
    while time.time() - start_time < budget_s - 100:
        generation += 1
        scores = [evaluate(ind) for ind in population]
        best_index = np.argmin(scores)
        best_c5 = scores[best_index]
        best_h = population[best_index]
        
        if best_c5 < 0.38085:
            break
        
        new_population = []
        for _ in range(50):
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            child = crossover(parent1, parent2)
            child = mutate(child, 0.1)
            new_population.append(child)
        
        population = new_population
    
    best_h = clip_and_normalize(best_h)
    best_c5 = evaluate(best_h)
    return (best_h, best_c5, n_points)
