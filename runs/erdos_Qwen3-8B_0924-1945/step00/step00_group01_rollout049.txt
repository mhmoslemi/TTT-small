def run(seed=42, budget_s=600, **kwargs):

    n_points = 47
    h_vals = initial_h_values.copy()
    h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
    dx = 2.0 / n_points
    best_h_vals = h_vals.copy()
    best_c5 = np.inf
    start_time = time.time()
    max_iter = 200
    eps = 1e-6
    improvement_threshold = 1e-6

    for iter in range(max_iter):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        current_c5 = np.max(corr * dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_vals = h_vals.copy()

        if current_c5 < improvement_threshold:
            break

        k_max = np.argmax(corr)
        overlap_k = corr[k_max] * dx

        # CVXPY optimization
        lambda_k = cp.Variable()
        objective = 0.5 * lambda_k**2 + (overlap_k - lambda_k)**2
        prob = cp.Problem(cp.Minimize(objective))
        prob.solve()
        lambda_k_val = lambda_k.value

        # Update h_vals with gradient descent
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)

        # Check time budget
        if time.time() - start_time > budget_s - 10:
            break

    # Final evaluation
    corr = np.correlate(best_h_vals, 1 - best_h_vals, mode='full')
    c5_bound = np.max(corr * dx)
    return (best_h_vals, c5_bound, n_points)
