import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h_vals = np.zeros(n_points)
    
    if 'initial_h_values' in globals():
        h_vals = np.array(initial_h_values)
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
    else:
        h_vals = np.zeros(n_points)
        h_vals[::3] = 1.0
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
    
    def objective(x):
        x = np.clip(x, 0, 1)
        x = x / np.sum(x) * (n_points / 2)
        one_minus_h = 1 - x
        corr = np.correlate(x, one_minus_h, mode="full")
        return np.max(corr) * dx
    
    def gradient(x):
        x = np.clip(x, 0, 1)
        x = x / np.sum(x) * (n_points / 2)
        one_minus_h = 1 - x
        corr = np.correlate(x, one_minus_h, mode="full")
        max_corr = np.max(corr)
        max_index = np.argmax(corr)
        corr_shifted = np.zeros_like(x)
        for i in range(len(corr)):
            corr_shifted[i] = corr[i]
        grad = np.zeros_like(x)
        for i in range(len(x)):
            tmp = x.copy()
            tmp[i] += 1e-6
            tmp = np.clip(tmp, 0, 1)
            tmp = tmp / np.sum(tmp) * (n_points / 2)
            tmp_one_minus_h = 1 - tmp
            tmp_corr = np.correlate(tmp, tmp_one_minus_h, mode="full")
            tmp_max_corr = np.max(tmp_corr)
            tmp_max_index = np.argmax(tmp_corr)
            grad[i] = (tmp_max_corr - max_corr) / 1e-6
        return grad
    
    result = optimize.minimize(
        objective,
        h_vals,
        method='L-BFGS-B',
        jac=gradient,
        bounds=[(0, 1) for _ in range(n_points)],
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
