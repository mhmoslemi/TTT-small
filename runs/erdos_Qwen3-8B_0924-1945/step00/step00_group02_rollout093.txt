import numpy as np
import time
import random
import scipy.signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(n_points)
        h[::2] = 1
        h = (h * n_points / 2).astype(float)
    
    # Ensure the initial h satisfies the integral constraint
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        # Randomly select indices to perturb
        indices = np.random.choice(n_points, size=5, replace=False)
        # Add noise
        noise = np.random.normal(0, 0.1, size=n_points)
        h_new = h + noise
        # Clip values to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        # Rescale to maintain the integral constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        # Compute the C5 value
        current_c5 = compute_c5(h_new, dx)
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5, n_points

def compute_c5(h, dx):
    shifted = 1 - h
    corr = scipy.signal.correlate(h, shifted, mode="full")
    return np.max(corr) * dx
