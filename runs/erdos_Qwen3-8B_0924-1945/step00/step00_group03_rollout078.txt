import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    initial_h_values = np.load('initial_h_values.npy')  # Assume this exists
    h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    # Function to evaluate and validate a solution
    def evaluate_and_validate(h, dx):
        h = np.clip(h, 0, 1)
        if not np.isclose(np.sum(h), n_points / 2, atol=1e-6):
            h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        c5 = evaluate_erdos_solution(h, dx)
        return h, c5
    
    # Function to compute all cross-correlation values
    def compute_all_correlations(h, dx):
        result = np.correlate(h, 1 - h, mode='full') * dx
        return result
    
    # Check if a solution is better
    def is_better(solution, current_best, dx):
        solution, c5 = evaluate_and_validate(solution, dx)
        if c5 < current_best:
            return solution, c5
        return current_best, current_best
    
    # Main optimization loop
    start_time = time.time()
    for i in range(50):
        # Perturb the solution
        perturbation = np.random.normal(0, 0.05, size=n_points)
        h_perturbed = h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Check if perturbed solution is better
        h_perturbed, c5_perturbed = evaluate_and_validate(h_perturbed, dx)
        if c5_perturbed < best_c5:
            best_h, best_c5 = h_perturbed, c5_perturbed
        
        # Perform a convex relaxation optimization
        try:
            # Define variables
            h_var = cp.Variable(n_points, lb=0, ub=1)
            t = cp.Variable()
            
            # Objective function
            objective = t + 1e-3 * cp.sum_squares(h_var - 0.5)
            
            # Constraints
            constraints = [t >= 0]
            for k in range(-n_points + 1, n_points):
                if k == 0:
                    continue
                lag = k % n_points
                corr = cp.sum(h_var * (1 - h_var[lag:lag + n_points]) * dx)
                constraints.append(t >= corr)
            
            # Solve the problem
            problem = cp.Problem(cp.Minimize(objective), constraints)
            problem.solve(solver=cp.ECOS, eps=1e-6)
            
            # Update the best solution
            if problem.status == 'OPTIMAL':
                h_new = h_var.value
                h_new, c5_new = evaluate_and_validate(h_new, dx)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new, c5_new
        except:
            pass
        
        # Early stopping if target is reached
        if best_c5 < 0.3807:
            break
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    # Final evaluation
    best_h, final_c5 = evaluate_and_validate(best_h, dx)
    return best_h, final_c5, n_points
