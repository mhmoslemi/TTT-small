import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    initial_h_values = np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        initial_h_values = kwargs['initial_h_values']
    
    def validate_solution(h, n_points):
        if np.any(h < 0) or np.any(h > 1):
            return False
        if abs(np.sum(h) - n_points / 2) > 1e-6:
            return False
        try:
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * 2.0 / n_points
        except:
            return False
        return True
    
    def clip_and_scale(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / total
        return h * scale
    
    h = clip_and_scale(initial_h_values)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * 2.0 / n_points
    
    dt = 0.1
    start_time = time.time()
    
    for t in range(int((budget_s - (time.time() - start_time)) // dt)):
        corr = np.correlate(h, 1 - h, mode="full")
        max_lag = np.argmax(corr)
        max_corr = corr[max_lag]
        overlap_x = h * (np.roll(1 - h, max_lag))
        overlap_x = np.clip(overlap_x, 0, 1)
        threshold = np.max(overlap_x) * 0.8
        target_indices = np.where(overlap_x > threshold)[0]
        target_indices = target_indices[:int(len(target_indices) * 0.1)]
        
        candidates = []
        for i in target_indices:
            h_candidate = h.copy()
            h_candidate[i] = 1 - h_candidate[i]
            h_candidate = clip_and_scale(h_candidate)
            candidates.append(h_candidate)
            h_candidate = h.copy()
            h_candidate[i] += np.random.uniform(-0.05, 0.05)
            h_candidate = clip_and_scale(h_candidate)
            candidates.append(h_candidate)
        
        for candidate in candidates:
            corr_candidate = np.correlate(candidate, 1 - candidate, mode="full")
            c5_candidate = np.max(corr_candidate) * 2.0 / n_points
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = candidate.copy()
        
        h = best_h.copy()
        if np.allclose(np.sum(h), n_points / 2):
            pass
        else:
            h = clip_and_scale(h)
        
        if t % 200 == 0:
            if not validate_solution(h, n_points):
                h = best_h.copy()
    
    return (best_h, best_c5, n_points)
