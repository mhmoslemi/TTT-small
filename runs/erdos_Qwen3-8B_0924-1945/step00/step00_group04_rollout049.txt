import numpy as np
import scipy.optimize as opt
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    initial_h_values = np.array([0.3333333333333333, 0.5, 0.5, 0.6666666666666666, 0.6666666666666666,
                                 0.6666666666666666, 0.6666666666666666, 0.6666666666666666,
                                 0.5, 0.3333333333333333, 0.3333333333333333, 0.3333333333333333,
                                 0.3333333333333333, 0.3333333333333333, 0.3333333333333333,
                                 0.3333333333333333, 0.3333333333333333, 0.3333333333333333,
                                 0.3333333333333333, 0.3333333333333333, 0.3333333333333333,
                                 0.3333333333333333, 0.3333333333333333, 0.3333333333333333,
                                 0.3333333333333333, 0.3333333333333333, 0.3333333333333333,
                                 0.3333333333333333, 0.3333333333333333, 0.3333333333333333,
                                 0.3333333333333333, 0.3333333333333333, 0.3333333333333333,
                                 0.3333333333333333, 0.3333333333333333, 0.3333333333333333,
                                 0.3333333333333333, 0.3333333333333333, 0.3333333333333333,
                                 0.3333333333333333, 0.3333333333333333, 0.5, 0.5, 0.5, 0.6666666666666666,
                                 0.6666666666666666, 0.6666666666666666])
    h = initial_h_values.copy()
    target_sum = n_points / 2
    
    def objective(x):
        h_vals = x.reshape((n_points, 1))
        one_minus_h = 1 - h_vals
        corr = correlate(h_vals, one_minus_h, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / n_points)
    
    def constraint_sum(x):
        return np.sum(x) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-6, 'disp': False}
    )
    
    h_opt = result.x
    c5_bound = objective(h_opt)
    
    # Validate constraints
    if np.any(h_opt < 0) or np.any(h_opt > 1):
        h_opt = np.clip(h_opt, 0, 1)
        np.sum(h_opt) = target_sum
        c5_bound = objective(h_opt)
    
    return h_opt, c5_bound, n_points
