```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 samples for balance between speed and precision
    dx = 2.0 / n_points
    initial_h_values = np.random.rand(n_points)
    initial_h_values = (initial_h_values / np.sum(initial_h_values)) * (n_points / 2)

    # Precompute 1 - h_vals for efficiency
    one_minus_h = 1.0 - initial_h_values

    def objective(h_vals):
        # Rescale to meet the integral constraint
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
        # Compute the correlation
        corr = np.correlate(h_vals, one_minus_h, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - n_points / 2

    def grad_objective(h_vals):
        # Gradient approximation using central differences
        grad = np.zeros_like(h_vals)
        eps = 1e-6
        for i in range(len(h_vals)):
            h_up = h_vals.copy()
            h_up[i] += eps
            h_down = h_vals.copy()
            h_down[i] -= eps
            grad[i] = (objective(h_up) - objective(h_down)) / (2 * eps)
        return grad

    # Set up optimization problem
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)}]

    # Initial guess
    h_vals = initial_h_values.copy()

    # Optimization
    start_time = time.time()
    result = optimize.minimize(
        objective,
        h_vals,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        jac=grad_objective,
        options={'ftol': 1e-8, 'maxiter': 100}
    )

    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (initial_h_values, objective(initial_h_values), n_points)

    # Ensure constraints are met after optimization
    h_vals = result.x
    h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
    h_vals = np.clip(h_vals, 0.0, 1.0)
    c5_bound = objective(h_vals)
    return (h_vals, c5_bound, n_points)
```
