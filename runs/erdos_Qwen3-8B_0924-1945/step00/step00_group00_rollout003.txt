import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    max_n_points = 100  # Moderate n_points for speed and effectiveness
    n_points = min(100, max_n_points)  # Choose a moderate number of points
    
    # Generate initial h_values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Normalize to ensure the integral is exactly 1
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.inf
    time_start = time.time()
    
    # Parameters for optimization
    learning_rate = 0.1
    learning_rate_decay = 0.999
    noise_scale = 0.05
    noise_prob = 0.05
    max_iterations = 1000
    early_stop = 100  # Early stopping if no improvement for 100 iterations
    
    for i in range(max_iterations):
        if time.time() - time_start > budget_s:
            break
        
        # Compute the full correlate
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_c5 = np.max(correlate_vals)
        
        # Update the best solution
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
        
        # Find the lag with the maximum value
        max_index = np.argmax(correlate_vals)
        lag = max_index - (n_points - 1)
        
        # Compute gradient for this lag
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = i - lag
            if 0 <= j < n_points:
                gradients[i] = (1 - h[j]) * (2.0 / n_points)
        
        # Update h with gradient descent
        h = h - learning_rate * gradients
        
        # Clip values within [0, 1]
        h = np.clip(h, 0, 1)
        
        # Normalize to maintain the integral constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Add random noise with probability noise_prob
        if random.random() < noise_prob:
            noise = np.random.normal(0, noise_scale, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Decay learning rate
        learning_rate *= learning_rate_decay
        
        # Early stopping if no improvement for early_stop iterations
        if i > 0 and abs(best_c5 - np.inf) < 1e-5 and i > early_stop:
            break
    
    # Final check
    h = best_h
    correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
    final_c5 = np.max(correlate_vals)
    
    # Ensure the integral constraint is satisfied
    h = h * (n_points / 2) / np.sum(h)
    correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
    final_c5 = np.max(correlate_vals)
    
    return (h, final_c5, n_points)
