import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    initial_h_values = np.array([0.5, 0.5, 1.0, 1.0, 0.5, 0.5, 0.0, 0.0, 0.5, 0.5, 
                                 1.0, 1.0, 0.5, 0.5, 0.0, 0.0, 0.5, 0.5, 1.0, 1.0, 
                                 0.5, 0.5, 0.0, 0.0, 0.5, 0.5, 1.0, 1.0, 0.5, 0.5, 
                                 0.0, 0.0, 0.5, 0.5, 1.0, 1.0, 0.5, 0.5, 0.0, 0.0, 
                                 0.5, 0.5, 1.0, 1.0, 0.5, 0.5, 0.0, 0.0, 0.5, 0.5])
    n_points = len(initial_h_values)
    
    # Check if initial_h_values is available
    if len(initial_h_values) > 0:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    # Function to update h with gradient descent and structured noise
    def update_h(h, learning_rate, noise_scale):
        h = np.clip(h, 0, 1)
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        worst_lag = np.argmax(correlate_vals)
        gradients = np.zeros_like(h)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * dx
        h = h - learning_rate * gradients
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        noise = noise_scale * np.random.normal(0, 1, n_points)
        h = np.clip(h + noise, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Main optimization loop
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        
        learning_rate = 0.05 * (1 - _ / 500)
        noise_scale = 0.05 if _ % 100 == 0 else 0.01
        
        h = update_h(h, learning_rate, noise_scale)
        
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return best_h, best_c5, n_points
