import numpy as np
import scipy.optimize as optimize
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    h = initial_h_values.copy()

    # Convert to block pattern
    m = n_points
    block_values = np.round(np.mean(h.reshape(-1, m), axis=1)).astype(int)
    block_sum = np.sum(block_values)
    if block_sum != m // 2:
        idx = np.where(block_values == 1)[0][0]
        block_values[idx] = 0
        block_values[np.where(block_values == 0)[0][0]] = 1
    # Convert block pattern to h array
    h = np.repeat(block_values, m)

    def compute_c5(h_values):
        h = h_values.reshape(-1, m)
        overlap = np.zeros((m, m), dtype=int)
        for i in range(m):
            for j in range(m):
                overlap[i, j] = int(np.any(np.abs(np.arange(m) - (i - j)) < m))
        return np.max((dx * np.dot(h.T, np.dot(overlap, 1 - h))).flatten())

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    time_remaining = budget_s

    for _ in range(1000):
        if time_remaining <= 0:
            break
        # Try swapping blocks
        for i in range(m - 1):
            for delta in [-1, 1]:
                j = i + delta
                if 0 <= j < m:
                    block_values[i], block_values[j] = block_values[j], block_values[i]
                    block_sum = np.sum(block_values)
                    if block_sum != m // 2:
                        idx = np.where(block_values == 1)[0][0]
                        block_values[idx] = 0
                        block_values[np.where(block_values == 0)[0][0]] = 1
                    h = np.repeat(block_values, m)
                    current_c5 = compute_c5(h)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h.copy()
                    # Revert swap
                    block_values[i], block_values[j] = block_values[j], block_values[i]
        time_remaining -= 0.1

    # Local optimization
    def objective(x):
        x = x.reshape(-1, m)
        x = np.clip(x, 0, 1)
        x = x * np.repeat(np.array([1 if i % 2 == 0 else 0 for i in range(m)]), m)
        x = x + np.repeat(np.array([0 if i % 2 == 0 else 1 for i in range(m)]), m) * (1 - x)
        x = x * np.repeat(np.array([1 if i % 3 == 0 else 0 for i in range(m)]), m)
        x = x + np.repeat(np.array([0 if i % 3 == 0 else 1 for i in range(m)]), m) * (1 - x)
        x = x * np.repeat(np.array([1 if i % 5 == 0 else 0 for i in range(m)]), m)
        x = x + np.repeat(np.array([0 if i % 5 == 0 else 1 for i in range(m)]), m) * (1 - x)
        x = np.clip(x, 0, 1)
        return compute_c5(x.flatten())

    def constraint(x):
        x = x.reshape(-1, m)
        x = x * np.repeat(np.array([1 if i % 2 == 0 else 0 for i in range(m)]), m)
        x = x + np.repeat(np.array([0 if i % 2 == 0 else 1 for i in range(m)]), m) * (1 - x)
        x = x * np.repeat(np.array([1 if i % 3 == 0 else 0 for i in range(m)]), m)
        x = x + np.repeat(np.array([0 if i % 3 == 0 else 1 for i in range(m)]), m) * (1 - x)
        x = x * np.repeat(np.array([1 if i % 5 == 0 else 0 for i in range(m)]), m)
        x = x + np.repeat(np.array([0 if i % 5 == 0 else 1 for i in range(m)]), m) * (1 - x)
        x = np.clip(x, 0, 1)
        return np.sum(x) - m / 2

    x0 = best_h.copy() / m
    result = optimize.minimize(objective, x0, method='L-BFGS-B', bounds=[(0, 1)] * n_points, constraints={'type': 'eq', 'fun': constraint})
    h_opt = result.x
    h_opt = h_opt.reshape(-1, m)
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * np.repeat(np.array([1 if i % 2 == 0 else 0 for i in range(m)]), m)
    h_opt = h_opt + np.repeat(np.array([0 if i % 2 == 0 else 1 for i in range(m)]), m) * (1 - h_opt)
    h_opt = h_opt * np.repeat(np.array([1 if i % 3 == 0 else 0 for i in range(m)]), m)
    h_opt = h_opt + np.repeat(np.array([0 if i % 3 == 0 else 1 for i in range(m)]), m) * (1 - h_opt)
    h_opt = h_opt * np.repeat(np.array([1 if i % 5 == 0 else 0 for i in range(m)]), m)
    h_opt = h_opt + np.repeat(np.array([0 if i % 5 == 0 else 1 for i in range(m)]), m) * (1 - h_opt)
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt.flatten()
    h_opt = h_opt * m
    h_opt = np.clip(h_opt, 0, m)
    h_opt = h_opt / m

    # Final validation
    assert (h_opt >= 0).all() and (h_opt <= 1).all()
    assert np.isclose(np.sum(h_opt), n_points / 2, atol=1e-6)
    h_values = h_opt

    # Final evaluation
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    c5 = np.max(corr) * dx

    return (h_values, c5, n_points)
