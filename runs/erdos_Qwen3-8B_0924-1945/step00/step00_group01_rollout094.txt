import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 50
    if 'initial_h_values' in globals():
        h_vals = np.array(initial_h_values, dtype=np.float64)
    else:
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = h_vals.copy()
    start_time = time.time()
    max_iter = 200
    tol = 1e-6
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s:
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        curr_c5 = np.max(corr * dx)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h_vals.copy()
        
        if curr_c5 < tol:
            break
        
        k_max = np.argmax(corr)
        overlap_k = corr[k_max] * dx
        t = cp.Variable()
        lambda_k = cp.Variable()
        
        obj = 0.5 * t**2 + (overlap_k - lambda_k)**2
        constraints = [lambda_k >= 0, lambda_k >= overlap_k]
        prob = cp.Problem(cp.Minimize(obj), constraints)
        prob.solve()
        
        if prob.status != 'OPTIMAL':
            break
        
        grad = (lambda_k.value - overlap_k) * (1 - 2 * h_vals)
        h_vals += grad * 0.1
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
        
        iter_count += 1
    
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr * dx)
    
    return (best_h, final_c5, n_points)
