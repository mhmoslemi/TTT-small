import numpy as np
import scipy.optimize as opt
import sys

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    initial_h_values = np.random.rand(n_points) * 2 - 1
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    h_vals = initial_h_values.copy()

    def objective(h_flat, k):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return corr[k] * (2.0 / n_points)

    def evaluate_c5(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def constraint_sum(h_flat):
        return np.sum(h_flat.reshape(n_points)) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]

    best_h_vals = h_vals.copy()
    best_c5 = evaluate_c5(best_h_vals)
    elapsed = 0
    max_iterations = 100
    improve_threshold = 1e-4
    no_improve_count = 0

    for _ in range(max_iterations):
        start_time = time.time()
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        max_idx = np.argmax(corr)
        max_c5 = corr[max_idx] * (2.0 / n_points)

        # Prepare optimization for the current worst lag
        def optimize_func(h_flat):
            return objective(h_flat, max_idx)

        result = opt.minimize(
            optimize_func,
            x0=h_vals.flatten(),
            bounds=bounds,
            constraints=constraints,
            method='L-BFGS-B',
            options={'ftol': 1e-6, 'maxiter': 50}
        )

        if result.success:
            h_vals = np.clip(result.x, 0, 1).reshape(n_points)
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        else:
            h_vals = np.clip(h_vals, 0, 1).reshape(n_points)
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)

        current_c5 = evaluate_c5(h_vals)
        elapsed += time.time() - start_time

        if current_c5 < best_c5 - improve_threshold:
            best_h_vals = h_vals.copy()
            best_c5 = current_c5
            no_improve_count = 0
        else:
            no_improve_count += 1

        if no_improve_count >= 5 or elapsed >= budget_s:
            break

    return (best_h_vals, best_c5, n_points)
