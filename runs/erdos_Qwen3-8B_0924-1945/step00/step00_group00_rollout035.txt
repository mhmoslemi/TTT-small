import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 45 if initial_h_values is None else len(initial_h_values)
    dx = 2.0 / n_points
    max_iter = 100
    swarm_size = 20
    particles = []
    velocities = []
    personal_bests = []
    global_best = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Initialize swarm
    if initial_h_values is not None:
        h = initial_h_values.copy()
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum != n_points / 2:
            h = h / h_sum * (n_points / 2)
        particles.append(h)
        personal_bests.append(h.copy())
        velocities.append(np.random.rand(n_points) * 0.1)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        h = h / h_sum * (n_points / 2)
        particles.append(h)
        personal_bests.append(h.copy())
        velocities.append(np.random.rand(n_points) * 0.1)
    
    # Function to compute the C5
    def compute_c5(h):
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum != n_points / 2:
            h = h / h_sum * (n_points / 2)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    # Initialize global best
    for p in particles:
        current_c5 = compute_c5(p)
        if current_c5 < best_c5:
            best_c5 = current_c5
            global_best = p.copy()
    
    # PSO parameters
    inertia_weight = 0.7
    cognitive_coeff = 1.5
    social_coeff = 1.5
    max_velocity = 0.1
    
    for iter_ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(swarm_size):
            h = particles[i].copy()
            vel = velocities[i].copy()
            # Compute C5
            current_c5 = compute_c5(h)
            # Update personal best
            if current_c5 < np.min(personal_bests[i]):
                personal_bests[i] = h.copy()
                np.min(personal_bests[i]) = current_c5
            # Update velocities and positions
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            vel = inertia_weight * vel + cognitive_coeff * r1 * (personal_bests[i] - h) + social_coeff * r2 * (global_best - h)
            # Clip velocities
            vel = np.clip(vel, -max_velocity, max_velocity)
            # Update position
            h = h + vel
            # Clip to [0, 1] and normalize
            h = np.clip(h, 0.0, 1.0)
            h_sum = np.sum(h)
            if h_sum != n_points / 2:
                h = h / h_sum * (n_points / 2)
            particles[i] = h
            # Update global best
            if current_c5 < best_c5:
                best_c5 = current_c5
                global_best = h.copy()
    
    # Final check and return
    final_c5 = compute_c5(global_best)
    return (global_best, final_c5, n_points)
