import numpy as np
import scipy.optimize as optimize
import sys
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h_values = np.random.rand(n_points)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    h_vals = initial_h_values.copy()
    best_h_vals = h_vals.copy()
    best_c5 = np.inf

    def objective(x):
        nonlocal best_c5
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        ones_minus_h = 1 - h_vals
        corr = np.correlate(h_vals, ones_minus_h, mode="full")
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h_vals = h_vals.copy()
        return c5

    def grad(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        ones_minus_h = 1 - h_vals
        corr = np.correlate(h_vals, ones_minus_h, mode="full")
        max_corr = np.max(corr)
        max_idx = np.argmax(corr)
        grad = np.zeros_like(h_vals)
        for i in range(n_points):
            h_vals[i] += 1e-6
            ones_minus_h = 1 - h_vals
            corr = np.correlate(h_vals, ones_minus_h, mode="full")
            c5 = np.max(corr) * dx
            h_vals[i] -= 1e-6
            grad[i] = (c5 - objective(h_vals)) / 1e-6
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        grad = np.clip(grad, -1e6, 1e6)
        return grad

    start_time = time.time()
    result = optimize.minimize(
        objective,
        h_vals,
        method='L-BFGS-B',
        jac=grad,
        bounds=[(0, 1) for _ in range(n_points)],
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        return (best_h_vals, best_c5, n_points)
    h_vals = result.x
    h_vals = np.clip(h_vals, 0, 1)
    h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
    ones_minus_h = 1 - h_vals
    corr = np.correlate(h_vals, ones_minus_h, mode="full")
    c5 = np.max(corr) * dx
    return (h_vals, c5, n_points)
