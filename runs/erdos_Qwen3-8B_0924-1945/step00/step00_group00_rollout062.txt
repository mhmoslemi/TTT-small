import numpy as np
import scipy.optimize as opt
import copy
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 45 if initial_h_values is None else len(initial_h_values)
    max_iter = 100
    n_particles = 20
    max_lag = n_points - 1
    dx = 2.0 / n_points
    best_h = np.zeros(n_points)
    best_c5 = float('inf')

    if initial_h_values is not None:
        initial_h = np.array(initial_h_values)
        initial_h = initial_h / np.sum(initial_h) * n_points / 2
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h / np.sum(initial_h) * n_points / 2
        best_h = initial_h
        best_c5 = evaluate_erdos_solution(best_h, dx)
    else:
        initial_h = np.random.rand(n_points)
        initial_h = initial_h / np.sum(initial_h) * n_points / 2
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h / np.sum(initial_h) * n_points / 2
        best_h = initial_h
        best_c5 = evaluate_erdos_solution(best_h, dx)

    particles = np.array([np.random.rand(n_points) for _ in range(n_particles)])
    particles = particles / np.sum(particles, axis=1, keepdims=True) * n_points / 2
    particles = np.clip(particles, 0, 1)
    particles = particles / np.sum(particles, axis=1, keepdims=True) * n_points / 2
    velocities = np.random.rand(n_particles, n_points) * 0.1

    pbest = copy.deepcopy(particles)
    pbest_c5 = np.array([evaluate_erdos_solution(p, dx) for p in pbest])
    gbest = copy.deepcopy(particles[np.argmin(pbest_c5)])
    gbest_c5 = np.min(pbest_c5)

    t0 = time.time()
    for iter in range(max_iter):
        if time.time() - t0 > budget_s:
            break
        if iter % 10 == 0:
            print(f"Iteration {iter}, Best C5: {gbest_c5}")
        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = 0.7 * velocities[i] + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] / np.sum(particles[i]) * n_points / 2
            c5 = evaluate_erdos_solution(particles[i], dx)
            if c5 < pbest_c5[i]:
                pbest[i] = copy.deepcopy(particles[i])
                pbest_c5[i] = c5
        gbest = copy.deepcopy(particles[np.argmin(pbest_c5)])
        gbest_c5 = np.min(pbest_c5)
        if gbest_c5 < best_c5:
            best_h = copy.deepcopy(gbest)
            best_c5 = gbest_c5
    best_c5 = evaluate_erdos_solution(best_h, dx)
    print(f"Final C5: {best_c5}")
    return (best_h, best_c5, n_points)
