import numpy as np
import scipy.signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Load initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Fallback to a basic uniform distribution
        n_points = 70
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = np.tile(h, 2)[:n_points]
        h = h / np.sum(h)
        h = np.clip(h, 0.0, 1.0)
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Validate initial solution
    assert np.all(h >= 0) and np.all(h <= 1)
    assert np.isclose(np.sum(h), n_points / 2)
    
    best_solution = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Optimization settings
    max_iterations = 500
    learning_rate = 1e-2
    perturbation_prob = 0.05
    tolerance = 1e-6
    early_exit = 0.3807
    
    start_time = time.time()
    
    for _ in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 10:
            break
        
        # Compute cross-correlation
        cross_corr = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            if k >= n_points:
                k_wrap = k - n_points
                cross_corr[k] = dx * np.sum(h[:n_points - k_wrap] * h[k_wrap:])
            else:
                cross_corr[k] = dx * np.sum(h[:n_points - k] * h[k:])
        
        current_max = np.max(cross_corr)
        if current_max < early_exit:
            break
        
        max_k = np.argmax(cross_corr)
        
        # Compute gradient for max_k
        grad = np.zeros(n_points)
        if max_k >= n_points:
            k_wrap = max_k - n_points
            for i in range(n_points - k_wrap):
                grad[i] += dx * h[i + k_wrap]
                grad[i + k_wrap] += dx * h[i]
        else:
            for i in range(n_points - max_k):
                grad[i] += dx * h[i + max_k]
                grad[i + max_k] += dx * h[i]
        
        # Update h
        direction = -grad
        step_size = 0.1 * learning_rate
        new_h = h + step_size * direction
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        if new_sum != 0:
            scaling_factor = (n_points / 2) / new_sum
            new_h *= scaling_factor
        new_h = np.clip(new_h, 0.0, 1.0)
        
        # Random perturbation
        if np.random.rand() < perturbation_prob:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            new_h = h + noise
            new_h = np.clip(new_h, 0.0, 1.0)
            new_sum = np.sum(new_h)
            if new_sum != 0:
                scaling_factor = (n_points / 2) / new_sum
                new_h *= scaling_factor
            new_h = np.clip(new_h, 0.0, 1.0)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            h = new_h.copy()
            best_c5 = new_c5
            best_solution = h.copy()
        
        # Check convergence
        if np.abs(new_c5 - best_c5) < tolerance:
            break
    
    # Final validation
    assert np.all(best_solution >= 0) and np.all(best_solution <= 1)
    assert np.isclose(np.sum(best_solution), n_points / 2, atol=1e-6)
    final_c5 = evaluate_erdos_solution(best_solution, dx)
    assert np.isfinite(final_c5)
    
    return best_solution, final_c5, n_points
