import numpy as np
import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = evaluate_erdos_solution() if 'initial_h_values' in locals() else None
    n_points = 500
    if initial_h_values is not None and len(initial_h_values) == n_points:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        dx = 2.0 / n_points
        return np.max(corr * dx)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    population_size = 50
    population = [h.copy() for _ in range(population_size)]
    
    for gen in range(500):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [compute_c5(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Select parents
        parents = np.argsort(fitness)[:int(population_size * 0.2)]
        offspring = []
        
        # Crossover
        for i in range(len(parents) // 2):
            p1 = population[parents[i]]
            p2 = population[parents[i + 1]]
            child = (p1 + p2) / 2
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            offspring.append(child)
        
        # Mutation
        for ind in population:
            mask = np.random.rand(n_points) < 0.1
            ind[mask] += np.random.uniform(-0.1, 0.1, size=np.sum(mask))
            ind = np.clip(ind, 0, 1)
            ind = ind * (n_points / 2) / np.sum(ind)
            offspring.append(ind)
        
        # Replace population
        population = offspring[:population_size]
    
    return best_h, best_c5, n_points
