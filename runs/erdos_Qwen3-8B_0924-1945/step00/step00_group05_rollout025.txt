import numpy as np
import scipy.optimize as opt
import random
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 500
    dx = 2.0 / n_points
    max_stagnation = 50
    pop_size = 50
    mutation_noise = 0.05
    mutation_rate = 0.1
    selection_size = 20

    if initial_h_values is not None:
        h = np.clip(initial_h_values, 0, 1)
        current_sum = np.sum(h)
        if current_sum != 0:
            h = h * (n_points / 2) / current_sum
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        h[:int(n_points / 2)] = 1
        h = np.clip(h, 0, 1)
    
    def evaluate_candidate(h):
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(overlaps)
        return c5

    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    stagnation = 0
    mutation_noise = 0.05

    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        return normalize(h)

    def crossover(parent1, parent2):
        split = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:split], parent2[split:]])
        return clip_and_normalize(child)

    def mutate(h):
        noise = np.random.normal(0, mutation_noise, size=n_points)
        h = h + noise
        return clip_and_normalize(h)

    for _ in range(1000):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

        population = []
        for _ in range(pop_size):
            candidate = mutate(best_h.copy())
            population.append(candidate)

        population.sort(key=lambda x: evaluate_candidate(x))
        population = population[:selection_size]

        new_population = []
        for _ in range(pop_size // 2):
            parent1 = random.choice(population)
            parent2 = random.choice(population)
            child = crossover(parent1, parent2)
            new_population.append(child)

        for _ in range(pop_size // 2):
            candidate = mutate(best_h.copy())
            new_population.append(candidate)

        new_population.sort(key=lambda x: evaluate_candidate(x))
        new_population = new_population[:pop_size]

        for candidate in new_population:
            c5 = evaluate_candidate(candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate.copy()
                stagnation = 0
            else:
                stagnation += 1

        if stagnation >= max_stagnation:
            mutation_noise *= 0.5
            stagnation = 0

    best_h = clip_and_normalize(best_h)
    final_c5 = evaluate_candidate(best_h)
    return (best_h, final_c5, n_points)
