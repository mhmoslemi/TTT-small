def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h = np.copy(initial_h_values)
    sum_h = np.sum(initial_h)
    if sum_h != n_points / 2:
        initial_h = initial_h * (n_points / 2) / sum_h
    h = initial_h
    
    def objective(x):
        h_new = x.reshape(n_points)
        h_new = np.clip(h_new, 0, 1)
        sum_new = np.sum(h_new)
        if sum_new != n_points / 2:
            h_new = h_new * (n_points / 2) / sum_new
        one_minus_h = 1 - h_new
        corr = np.correlate(h_new, one_minus_h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    bounds = [(0, 1) for _ in range(n_points)]
    result = optimize.minimize(objective, h, bounds=bounds, method='L-BFGS-B', jac=None, options={'maxiter': 500, 'ftol': 1e-8})
    h_best = result.x
    c5_best = objective(h_best)
    
    if c5_best > 0.38085:
        h_best = np.copy(initial_h)
        c5_best = objective(h_best)
    
    return h_best, c5_best, n_points
