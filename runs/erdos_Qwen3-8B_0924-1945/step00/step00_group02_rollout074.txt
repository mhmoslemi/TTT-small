import numpy as np
import scipy
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = (h - 0.5) * 2.0
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)

    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx

    T_initial = 100.0
    T_final = 0.01
    alpha = 0.995
    sigma_initial = 0.05
    sigma_decay = 0.99
    max_iterations = int(budget_s / 0.1)
    iteration = 0
    start_time = time.time()

    while time.time() - start_time < budget_s and iteration < max_iterations:
        iteration += 1
        T = T_initial * (1 - iteration / max_iterations) ** 0.9
        sigma = sigma_initial * (sigma_decay ** iteration)
        indices = np.random.choice(n_points, size=5, replace=False)
        delta = np.random.normal(0, sigma, size=n_points)
        new_h = h + delta
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        corr = np.correlate(new_h, 1 - new_h, mode="full")
        c5_new = np.max(corr) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = new_h.copy()
        elif np.random.rand() < np.exp((best_c5 - c5_new) / T):
            best_c5 = c5_new
            best_h = new_h.copy()

    h_values = best_h
    return h_values, best_c5, n_points
