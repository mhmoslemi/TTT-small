```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 450
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else None
    
    if initial_h_values is not None:
        h = np.clip(initial_h_values, 0, 1)
        h = h * n_points / 2 / np.sum(h) if np.sum(h) != 0 else np.full(n_points, 0.5)
    else:
        h = np.full(n_points, 0.5)
    
    def clip_and_rescale(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full(n_points, 0.5)
        h = h * n_points / 2 / total
        return h
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * 2.0 / n_points
        return c5
    
    def update_h(h, mutation_indices, mutation_strength):
        h = h.copy()
        for i in mutation_indices:
            delta = np.random.normal(0, mutation_strength)
            h[i] += delta
        h = clip_and_rescale(h)
        return h
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    
    for t in range(1000):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        target_k = max_k
        
        # Targeted mutation
        mutation_indices = np.random.choice(n_points, 10, replace=False)
        mutation_strength = 0.1 / (1 + t / 500)
        mutated_h = update_h(h, mutation_indices, mutation_strength)
        c5_targeted = compute_c5(mutated_h)
        
        # Global mutation
        mutation_indices = np.random.choice(n_points, 5, replace=False)
        mutation_strength = 0.2 * np.exp(-t / 300)
        mutated_h = update_h(h, mutation_indices, mutation_strength)
        c5_global = compute_c5(mutated_h)
        
        # Choose better candidate
        if c5_targeted < c5_global:
            h = mutated_h
            candidate_c5 = c5_targeted
        else:
            h = mutated_h
            candidate_c5 = c5_global
        
        if candidate_c5 < best_c5:
            best_h = h.copy()
            best_c5 = candidate_c5
        
        if t % 100 == 0:
            print(f"Iteration {t}, C5: {best_c5}")
    
    return best_h, best_c5, n_points
```
