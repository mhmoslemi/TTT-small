def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    n_points = len(initial_h_values)
    delta_x = 2.0 / n_points
    h_initial = initial_h_values.copy()
    h_initial = np.clip(h_initial, 0.0, 1.0)
    np.random.seed(seed)
    
    def evaluate_c5(h):
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = np.max(corr) * delta_x
        return c5
    
    def objective(x):
        return evaluate_c5(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    def constraint_bounds(x):
        return np.all((x >= 0.0) & (x <= 1.0))
    
    # Initial guess
    x0 = h_initial.copy()
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization
    result = opt.minimize(objective, x0, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 100})
    
    # Final solution
    h_values = result.x
    final_c5 = evaluate_c5(h_values)
    
    # Ensure constraints are satisfied
    assert np.allclose(np.sum(h_values), n_points / 2)
    assert np.all((h_values >= 0) & (h_values <= 1))
    
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        # Further local refinement
        for _ in range(100):
            h_values = np.clip(h_values, 0.0, 1.0)
            delta = np.random.uniform(-0.01, 0.01, size=n_points)
            h_values += delta
            h_values = np.clip(h_values, 0.0, 1.0)
            np.random.shuffle(h_values)
            h_values = (h_values * n_points / np.sum(h_values)).astype(float)
            current_c5 = evaluate_c5(h_values)
            if current_c5 < final_c5:
                final_c5 = current_c5
                h_values = h_values.copy()
    
    return h_values, final_c5, n_points
