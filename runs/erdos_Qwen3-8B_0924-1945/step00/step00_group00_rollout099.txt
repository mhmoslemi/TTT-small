import numpy as np
import scipy.signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    h = np.array(initial_h_values, dtype=np.float64)
    h /= np.sum(h) * 2.0
    h *= n_points / 2
    h = np.clip(h, 0, 1)
    start_time = time.time()
    best_h = h.copy()
    best_c5 = np.max(scipy.signal.correlate(h, 1 - h, mode='full')) * 2.0 / n_points
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i - (max_idx - n_points + 1)
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * 2.0 / n_points
        h -= grad * 0.01
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = np.max(corr) * 2.0 / n_points
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5, n_points
