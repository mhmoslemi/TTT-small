import numpy as np
import scipy.optimize as optimize
import random
import time

initial_h_values = None  # Placeholder; will be provided by the system

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45  # Start with the initial n_points
    if initial_h_values is not None:
        if len(initial_h_values) != n_points:
            raise ValueError("Initial h_values length does not match n_points")
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.clip(h, 0.0, 1.0)
        total = h.sum()
        if abs(total - n_points / 2) > 1e-6:
            h += (n_points / 2 - total) / n_points
        h = np.clip(h, 0.0, 1.0)
        h = h * n_points / (h.sum())
    else:
        h = np.random.rand(n_points)
        total = h.sum()
        h += (n_points / 2 - total) / n_points
        h = np.clip(h, 0.0, 1.0)
        h = h * n_points / (h.sum())

    def evaluate(h):
        dx = 2.0 / n_points
        h = np.clip(h, 0.0, 1.0)
        h = h * n_points / (h.sum())
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(correlate_vals)
        return c5

    def fitness(h):
        return evaluate(h)

    def constraint(h):
        return h.sum() - n_points / 2

    cons = {'type': 'eq', 'fun': constraint, 'jac': lambda x: np.ones_like(x)}

    def objective(h):
        return fitness(h)

    # Use differential evolution for optimization
    result = optimize.differential_evolution(
        objective,
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        constraints=cons,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        disp=False,
        seed=seed
    )

    best_h = result.x
    best_c5 = evaluate(best_h)

    # Ensure constraints are satisfied
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * n_points / (best_h.sum())
    best_c5 = evaluate(best_h)

    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 10:
        # Terminate early if we're close to the budget
        pass

    return (best_h.tolist(), best_c5, n_points)
