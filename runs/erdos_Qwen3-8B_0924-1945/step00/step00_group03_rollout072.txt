import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h.npy') if 'initial_h_values' in locals() else np.random.rand(70) * 2 - 1
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_sum = np.sum(initial_h_values)
    desired_sum = n_points / 2
    initial_h_values = np.clip(initial_h_values, 0.0, 1.0)
    initial_h_values = initial_h_values * (desired_sum / initial_sum) if initial_sum != 0 else initial_h_values
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    t_total = int(budget_s * 0.75)
    T0 = 0.7 * (1.0 + best_c5) / 3.0
    Tmax = int(0.1 * t_total)
    beta = 0.75
    gamma = 0.65
    lambda0 = 0.15 / dx
    
    start_time = time.time()
    
    for t in range(t_total):
        if time.time() - start_time > budget_s:
            break
        if t % 100 == 0:
            lambda0 *= 0.8 if t < t_total // 2 else 0.95
        
        noise = np.random.normal(0, lambda0, n_points)
        perturbed = h + noise
        perturbed = np.clip(perturbed, 0.0, 1.0)
        
        current_sum = np.sum(perturbed)
        delta = (desired_sum - current_sum) / (n_points * 0.95)
        perturbed += delta * np.random.choice([-1, 1], size=n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        perturbed = perturbed * (desired_sum / np.sum(perturbed)) if np.sum(perturbed) != 0 else perturbed
        
        new_c5 = evaluate_erdos_solution(perturbed, dx)
        
        if new_c5 < best_c5 or (np.random.rand() < np.exp(-(new_c5 - best_c5) / (T0 * (1 - t / Tmax) ** (1 / beta)))):
            h = perturbed.copy()
            best_c5 = new_c5
            best_h = h.copy()
        
        if best_c5 < 0.3807:
            break
    
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (best_h, final_c5, n_points)
