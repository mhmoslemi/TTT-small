import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    
    # Initial construction
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    learning_rate = 0.1
    iteration = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Compute overlaps
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        k_max = np.argmax(overlaps)
        
        # Compute gradient for the maximum lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        
        # Update h
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        
        # Normalize sum
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (n_points / 2) / sum_h
        
        # Evaluate new C5
        new_c5 = evaluate_erdos_solution(h)
        
        # Update best solution
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            iteration = 0
        else:
            iteration += 1
            if iteration > 200:
                learning_rate *= 0.95
                iteration = 0
        
        # Early termination if stuck
        if iteration > 500:
            break
    
    # Final checks
    h = best_h
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        h = h * (n_points / 2) / sum_h
    
    # Evaluate final C5
    c5_bound = evaluate_erdos_solution(h)
    
    # Validate constraints
    assert np.all(0 <= h <= 1), "h values out of bounds"
    assert np.isclose(np.sum(h), n_points / 2, atol=1e-5), "Sum constraint violated"
    
    return h, c5_bound, n_points
