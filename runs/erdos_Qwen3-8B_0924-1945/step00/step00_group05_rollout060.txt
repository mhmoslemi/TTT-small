import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 500
    dx = 2.0 / n_points
    pop_size = 50
    max_generations = 100
    mutation_noise = 0.05
    mutation_rate = 0.1
    selection_size = 20
    stagnation_threshold = 50
    
    # Start with initial h or random
    if initial_h_values is not None:
        h = np.clip(initial_h_values, 0, 1).copy()
    else:
        h = np.random.uniform(0, 1, n_points)

    h = h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        if np.isclose(np.sum(h), n_points / 2):
            overlaps = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(overlaps)
        else:
            return float('inf')
    
    # Best solution so far
    best_h = h.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        
        # Generate population
        population = []
        for _ in range(pop_size):
            candidate = h + np.random.normal(0, mutation_noise, n_points)
            candidate = np.clip(candidate, 0, 1)
            candidate = candidate * (n_points / 2) / np.sum(candidate)
            population.append(candidate)
        
        # Evaluate population
        fitness = [evaluate(ind) for ind in population]
        ranked = sorted(zip(population, fitness), key=lambda x: x[1])
        
        # Select top candidates
        selected = [ind for ind, fit in ranked[:selection_size]]

        for _ in range(pop_size // 2):

                break
            parent1, parent2 = np.random.choice(selected, size=2, replace=False)
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            new_population.append(child)
        
        # Mutation
        for ind in new_population:
            if np.random.rand() < mutation_rate:
                ind += np.random.normal(0, mutation_noise, n_points)
                ind = np.clip(ind, 0, 1)
                ind = ind * (n_points / 2) / np.sum(ind)
        
        # Replace population
        population = new_population
        
        # Update best
        for ind in population:
            current_c5 = evaluate(ind)

    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate(best_h)
