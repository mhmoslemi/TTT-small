import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    initial_h_values = np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        initial_h_values = kwargs['initial_h_values']
    else:
        initial_h_values = 0.5 * (1 + np.random.uniform(-1, 1, n_points))
    h = np.copy(initial_h_values)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h *= scale
    h = np.clip(h, 0, 1)
    h = np.round(h, 5)
    best_h = np.copy(h)
    best_c5 = np.inf
    max_iter = int(budget_s // 0.1)
    start_time = time.time()
    for t in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * 2.0 / n_points
        if c5 < best_c5:
            best_c5 = c5
            best_h = np.copy(h)
        if c5 < 0.38085:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        lag = np.argmax(corr)
        if lag >= n_points:
            lag = n_points - 1
        h_shifted = np.roll(h, lag)
        overlap = h * (1 - h_shifted)
        threshold = np.max(overlap) * 0.8
        indices = np.where(overlap > threshold)[0]
        indices = indices[:int(len(indices) * 0.1)]
        candidates = []
        for i in indices:
            h_candidate = np.copy(h)
            h_candidate[i] = 1 - h_candidate[i]
            h_candidate = np.clip(h_candidate, 0, 1)
            sum_c = np.sum(h_candidate)
            scale_c = (n_points / 2) / sum_c
            h_candidate *= scale_c
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = np.round(h_candidate, 5)
            candidates.append(h_candidate)
        for i in indices:
            h_candidate = np.copy(h)
            noise = np.random.normal(0, 0.02, n_points)
            h_candidate += noise
            h_candidate = np.clip(h_candidate, 0, 1)
            sum_c = np.sum(h_candidate)
            scale_c = (n_points / 2) / sum_c
            h_candidate *= scale_c
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = np.round(h_candidate, 5)
            candidates.append(h_candidate)
        if not candidates:
            continue
        best_c = np.inf
        best_cand = h
        for cand in candidates:
            corr_c = np.correlate(cand, 1 - cand, mode="full")
            c5_c = np.max(corr_c) * 2.0 / n_points
            if c5_c < best_c:
                best_c = c5_c
                best_cand = cand
        h = np.copy(best_cand)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h *= scale
    h = np.clip(h, 0, 1)
    h = np.round(h, 5)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * 2.0 / n_points
    return (h, c5, n_points)
