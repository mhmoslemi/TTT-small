```python
import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial_h_values
    n_points_initial = len(initial_h_values)
    h_vals = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points_initial
    
    # Normalize to satisfy the integral constraint
    h_vals = h_vals / np.sum(h_vals) * (n_points_initial / 2)
    
    # Function to compute the overlap integral and its gradient
    def objective_and_gradient(h_vals_flat):
        h = h_vals_flat.reshape(-1)
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += 1e-6
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
            grad[i] = (np.max(corr_plus) - c5) / 1e-6
        
        return c5, grad.flatten()
    
    # Define bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(len(h_vals))]
    constraints = [
        {'type': 'eq', 'fun': lambda h: np.sum(h) - len(h) / 2}
    ]
    
    # Initial optimization
    result = optimize.minimize(
        fun=lambda h: objective_and_gradient(h)[0],
        x0=h_vals,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        jac=lambda h: objective_and_gradient(h)[1],
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    # Extract the best result
    best_h = result.x
    best_c5 = objective_and_gradient(best_h)[0]
    
    # Final validation
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0.0, 1.0)
    final_h = final_h / np.sum(final_h) * (len(final_h) / 2)
    final_c5 = evaluate_erdos_solution(final_h)
    
    # Return the result
    n_points = len(final_h)
    return (final_h, final_c5, n_points)
```
