import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.clip(h, 0.0, 1.0)
        h = h * n_points / 2.0 / np.sum(h)
    else:
        h = np.random.uniform(0.0, 1.0, n_points)
        h = h * n_points / 2.0 / np.sum(h)
    best_h = h.copy()
    best_c5 = np.inf
    T = 100.0
    alpha = 0.995
    start_time = time.time()
    for t in range(6000):
        if time.time() - start_time > budget_s:
            break
        idxs = np.random.choice(n_points, 5, replace=False)
        delta = np.random.normal(0.0, 0.05 * np.exp(-t / 200), 5)
        new_h = h.copy()
        new_h[idxs] += delta
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = new_h * n_points / 2.0 / np.sum(new_h)
        corr = np.correlate(new_h, 1 - new_h, mode="full")
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = new_h.copy()
        if np.random.rand() < np.exp(-(c5 - best_c5) / T):
            h = new_h.copy()
        T *= alpha
    return best_h, best_c5, n_points
