import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

        corr = corr * dx
        current_c5 = np.max(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_vals = h_vals.copy()
        
        if current_c5 < tol:
            break
        
        k_max = np.argmax(corr)
        overlap_k = corr[k_max]
        
        # Solve the convex optimization problem
        λ = cp.Variable(1)
        objective = 0.5 * λ**2 + (overlap_k - λ)**2
        constraints = [λ >= 0]
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(max_iter=100)
        except:
            prob = cp.Problem(cp.Minimize(0.5 * cp.square(overlap_k)))
            prob.solve(max_iter=100)
        
        λ_opt = float(λ.value)
        
        # Update h_vals using gradient descent
        h_vals = h_vals + (overlap_k - λ_opt) * 0.1
        
        # Clip and normalize
        h_vals = np.clip(h_vals, 0, 1)
        mass = np.sum(h_vals)
        h_vals = h_vals * (n_points / 2) / mass
    
    # Final evaluation
    corr = correlate(best_h_vals, 1 - best_h_vals, mode='full')
    corr = corr * dx
    final_c5 = np.max(corr)
