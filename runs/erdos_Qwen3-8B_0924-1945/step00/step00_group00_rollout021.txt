import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction with 45 samples
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = (h * n_points / 2) / np.sum(h)
    
    # Parameters
    learning_rate = 0.1
    momentum_factor = 0.9
    noise_scale = 0.05
    max_iterations = int(budget_s // 1.5)
    early_stop = 50
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    v = np.zeros(n_points)
    
    for iteration in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Compute correlation
        correlate_vals = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        worst_lag = np.argmax(correlate_vals)
        
        # Compute gradient for worst lag
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * (2.0 / n_points)
        
        # Update momentum and h
        v = momentum_factor * v + learning_rate * gradients
        h -= v
        
        # Clip and normalize
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Add noise
        if iteration % 100 == 0:
            noise = noise_scale * np.random.normal(0, 1, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Track best solution
        current_c5 = np.max(correlate_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early stopping
        if iteration > 0 and abs(current_c5 - best_c5) < 1e-5:
            if iteration > early_stop:
                break
    
    return (best_h, best_c5, n_points)
