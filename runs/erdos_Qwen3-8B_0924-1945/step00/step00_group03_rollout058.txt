import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    initial_h_values = np.load('initial_h_values.npy')
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    if not np.isclose(sum_h, n_points / 2):
        h = h * (n_points / 2) / sum_h
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    start_time = time.time()
    
    def compute_correlate(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def perturb_h(h, scale=0.05):
        perturbation = np.random.normal(0, scale, h.shape)
        h_perturbed = h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = np.sum(h_perturbed)
        h_perturbed = h_perturbed * (n_points / 2) / sum_perturbed
        return h_perturbed
    
    for _ in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        h_perturbed = perturb_h(h)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = np.sum(h_perturbed)
        h_perturbed = h_perturbed * (n_points / 2) / sum_perturbed
        
        n = n_points
        h_var = cp.Variable(n)
        t = cp.Variable(n)
        lambda_val = 1e-3
        
        constraints = []
        for k in range(n):
            idx = (np.arange(n) + k) % n
            corr = cp.sum(h_var[idx] * (1 - h_var[np.arange(n)]))
            constraints.append(t[k] >= corr * dx)
        
        objective = cp.Minimize(t[0] + lambda_val * cp.sum_squares(h_var - 0.5))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver='ECOS', eps=1e-6, verbose=False)
            if prob.status == 'OPTIMAL':
                h_candidate = h_var.value
                h_candidate = np.clip(h_candidate, 0, 1)
                sum_candidate = np.sum(h_candidate)
                h_candidate = h_candidate * (n_points / 2) / sum_candidate
                c5_candidate = compute_correlate(h_candidate, dx)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
        
        h = best_h.copy()
    
    final_c5 = compute_correlate(best_h, dx)
    assert np.isclose(np.sum(best_h), n_points / 2, atol=1e-6)
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    assert np.isfinite(final_c5)
    
    return (best_h, final_c5, n_points)
