import numpy as np
import time
import scipy.optimize as opt
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    h = h * (n_points / 2) / total
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    n_iterations = 500
    learning_rate = 0.1
    momentum_factor = 0.9
    noise_scale = 0.05
    decay_rate = 0.999
    noise_freq = 100
    early_stop = 50
    
    v = np.zeros(n_points)
    for iter in range(n_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        correlate = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        current_c5 = np.max(correlate)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        worst_lag = np.argmax(correlate)
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * (2.0 / n_points)
        
        v = momentum_factor * v + learning_rate * gradients
        h += v
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        h = h * (n_points / 2) / total
        
        if iter % noise_freq == 0:
            noise = noise_scale * np.random.normal(0, 1, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            h = h * (n_points / 2) / total
        
        learning_rate *= decay_rate
    
    return (best_h, best_c5, n_points)
