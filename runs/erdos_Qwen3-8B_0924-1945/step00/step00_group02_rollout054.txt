import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if abs(total - n_points / 2) > 1e-6:
        h = np.zeros(n_points)
        h[:n_points // 2] = 1
        h = np.clip(h, 0, 1)
        total = np.sum(h)
    h = h * (n_points / 2) / total
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    start_time = time.time()
    for _ in range(int((budget_s - (time.time() - start_time)) // 0.1)):
        corr = np.correlate(h, 1 - h, mode="full")
        lag = np.argmax(corr)
        overlap = h * np.roll(1 - h, lag)
        top_indices = np.argsort(overlap)[::-1][:int(len(h) * 0.1)]
        candidates = []
        for i in top_indices:
            h_candidate = h.copy()
            h_candidate[i] = 1 - h_candidate[i]
            h_candidate = np.clip(h_candidate, 0, 1)
            total_candidate = np.sum(h_candidate)
            scale = (n_points / 2) / total_candidate
            h_candidate *= scale
            h_candidate = np.clip(h_candidate, 0, 1)
            candidates.append(h_candidate)
            h_candidate = h.copy()
            h_candidate[i] += np.random.uniform(-0.05, 0.05)
            h_candidate = np.clip(h_candidate, 0, 1)
            total_candidate = np.sum(h_candidate)
            scale = (n_points / 2) / total_candidate
            h_candidate *= scale
            h_candidate = np.clip(h_candidate, 0, 1)
            candidates.append(h_candidate)
        for candidate in candidates:
            c5_candidate = compute_c5(candidate, n_points)
            if c5_candidate < best_c5:
                best_h = candidate.copy()
                best_c5 = c5_candidate
        h = best_h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * 2.0 / n_points
    return c5
