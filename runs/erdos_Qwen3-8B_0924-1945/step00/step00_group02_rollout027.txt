import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = (h - 0.5) * 2.0
        h = np.clip(h, 0, 1)
        h = h * n_points / 2.0
        h = h / np.sum(h) * n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * n_points / 2.0
    h = h / np.sum(h) * n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * n_points / 2.0
    h = h / np.sum(h) * n_points / 2.0
    best_h = h.copy()
    best_c5 = np.inf
    T = 100.0
    alpha = 0.998
    start_time = time.time()
    for t in range(10000):
        if time.time() - start_time > budget_s:
            break
        indices = np.random.choice(n_points, 5, replace=False)
        delta = np.random.normal(0, 0.05, n_points)
        new_h = h.copy()
        new_h[indices] += delta[indices]
        new_h = np.clip(new_h, 0, 1)
        total = np.sum(new_h)
        if abs(total) < 1e-6:
            new_h = h.copy()
        else:
            new_h = new_h * (n_points / 2.0) / total
        new_h = np.clip(new_h, 0, 1)
        corr = np.correlate(new_h, 1.0 - new_h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = new_h.copy()
        delta_c5 = c5 - best_c5
        if delta_c5 < 0:
            best_h = new_h.copy()
            best_c5 = c5
        else:
            p = np.exp(-delta_c5 / T)
            if np.random.rand() < p:
                best_h = new_h.copy()
                best_c5 = c5
        T *= alpha
    return (best_h, best_c5, n_points)
