```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in globals() else np.random.rand(n_points)
    
    def clip_normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    def evaluate(h):
        h = clip_normalize(h)
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlaps)
    
    h = clip_normalize(initial_h_values)
    best_h = h.copy()
    best_c5 = evaluate(h)
    learning_rate = 0.1
    iteration = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        k_max = np.argmax(overlaps)
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        
        h = h - learning_rate * grad
        h = clip_normalize(h)
        current_c5 = evaluate(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            iteration = 0
        else:
            iteration += 1
            if iteration > 200:
                learning_rate *= 0.95
                iteration = 0
        
        if np.abs(current_c5 - best_c5) < 1e-5:
            break
    
    final_h = clip_normalize(h)
    final_c5 = evaluate(final_h)
    final_h = np.clip(final_h, 0, 1)
    assert np.all(0 <= final_h <= 1)
    assert np.isclose(np.sum(final_h), n_points / 2, atol=1e-5)
    
    return final_h, final_c5, n_points
```
