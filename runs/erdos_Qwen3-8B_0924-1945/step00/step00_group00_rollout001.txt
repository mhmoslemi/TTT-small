import numpy as np
import scipy.signal
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 45
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=np.float64) if initial_h_values is not None else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    
    for _ in range(10000):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        max_c5 = np.max(corr) * dx
        
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
        
        lag = np.argmax(corr)
        shift = lag - n_points + 1
        
        if shift < 0:
            shift += n_points
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i - shift) % n_points
            grad[i] = (1 - h[j]) * dx
        
        h = np.clip(h - 0.01 * grad, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    h = best_h
    c5 = compute_c5(h)
    return h, c5, n_points

def compute_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    corr = scipy.signal.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
