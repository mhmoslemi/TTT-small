def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_values = initial_h_values.copy()
    c5_bound = compute_c5(h_values, dx)
    start_time = time.time()
    
    def objective(h):
        h = h.copy()
        if not (0 <= h.min() and h.max() <= 1):
            h.clip(0, 1, out=h)
        h_norm = h * dx
        if not np.isclose(np.sum(h_norm), 1.0, atol=1e-8):
            h = h * (1.0 / np.sum(h_norm)) * n_points / 2
        c5 = compute_c5(h, dx)
        return c5
    
    def compute_c5(h, dx):
        h_1 = 1 - h
        corr = np.correlate(h, h_1, mode='full')
        return np.max(corr * dx)
    
    best_h = h_values.copy()
    best_c5 = c5_bound
    max_iter = 1500
    iter_count = 0
    perturb = 0.05
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        iter_count += 1
        perturbation = np.random.normal(0, perturb, size=n_points)
        h_candidate = h_values + perturbation
        h_candidate = np.clip(h_candidate, 0, 1)
        norm = np.sum(h_candidate) * dx
        if not np.isclose(norm, 1.0, atol=1e-8):
            h_candidate = h_candidate * (1.0 / norm) * n_points / 2
        c5_candidate = compute_c5(h_candidate, dx)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate
    
    h_values = best_h
    c5_bound = best_c5
    
    def objective_with_bounds(x):
        x = x.copy()
        x = np.clip(x, 0, 1)
        x_norm = x * dx
        if not np.isclose(np.sum(x_norm), 1.0, atol=1e-8):
            x = x * (1.0 / np.sum(x_norm)) * n_points / 2
        return compute_c5(x, dx)
    
    bounds = Bounds(0, 1)
    result = minimize(objective_with_bounds, h_values, bounds=bounds, method='L-BFGS-B', tol=1e-6)
    if result.success:
        h_values = result.x
        c5_bound = result.fun
    
    h_values = np.clip(h_values, 0, 1)
    norm = np.sum(h_values) * dx
    if not np.isclose(norm, 1.0, atol=1e-8):
        h_values = h_values * (1.0 / norm) * n_points / 2
    c5_bound = compute_c5(h_values, dx)
    
    if c5_bound < 0.3807:
        print("Target C₅ bound achieved.")
    
    return (h_values, c5_bound, n_points)
