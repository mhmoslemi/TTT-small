import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Construct a regular initial pattern
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 5 == 0:
                h[i] = 1.0
            elif i % 5 == 1:
                h[i] = 0.5
            elif i % 5 == 2:
                h[i] = 0.0
            elif i % 5 == 3:
                h[i] = 0.5
            else:
                h[i] = 1.0
        h = np.clip(h, 0, 1)
    
    # Normalize to meet the sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    def objective(x):
        h = x.reshape(n_points)
        return evaluate(h)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    def constraint_bounds(x):
        return np.all((0 <= x) & (x <= 1))
    
    # Optimization using gradient descent
    h_opt = h.copy()
    best_c5 = evaluate(h_opt)
    learning_rate = 0.01
    decay_rate = 0.999
    iteration = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Compute overlaps and choose the worst lag
        h = h_opt.copy()
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        k_max = np.argmax(overlap)
        k_max = np.min([k_max, n_points - 1])
        
        # Compute gradient for lag k_max
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        
        # Update h
        h = h - learning_rate * grad
        
        # Clip and normalize
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (n_points / 2) / sum_h
        
        # Evaluate and update best
        new_c5 = evaluate(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            h_opt = h.copy()
            iteration = 0
        else:
            iteration += 1
        
        # Decay learning rate
        if iteration >= 100:
            learning_rate *= decay_rate
            iteration = 0
        
        # Periodic perturbations
        if iteration % 500 == 0:
            noise = np.random.normal(0, 0.05, size=n_points)
            h_opt = np.clip(h_opt + noise, 0, 1)
            sum_h = np.sum(h_opt)
            if sum_h > 0:
                h_opt = h_opt * (n_points / 2) / sum_h
        
        # Early stopping if no improvement
        if best_c5 < 0.3807:
            break
    
    return h_opt, best_c5, n_points
