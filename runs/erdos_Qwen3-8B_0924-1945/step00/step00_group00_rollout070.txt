import numpy as np
import scipy.optimize as optimize
import time
import sys

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45  # Start with the initial sample size
    dx = 2.0 / n_points
    initial_h_values = kwargs.get('initial_h_values', None)
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    learning_rate = 0.05
    noise_scale = 0.05
    max_iter = 500
    iter_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        # Compute the full correlate
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        current_max = np.max(correlate_vals)
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        
        # Select the worst lag
        worst_lag = np.argmax(correlate_vals)
        # Compute the gradient for the worst lag
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * dx
        
        # Update h
        h = h - learning_rate * gradients
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Add structured noise
        if iter_count % 100 == 0 and iter_count > 0:
            noise = noise_scale * np.random.normal(0, 1, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        iter_count += 1
    
    # Ensure we have a valid best solution
    h = best_h
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
    final_c5 = np.max(correlate_vals)
    
    return (h.tolist(), final_c5, n_points)
