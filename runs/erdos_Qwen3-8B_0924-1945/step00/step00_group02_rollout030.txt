import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.clip(h, 0, 1)
        h = h * n_points / 2.0 / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * n_points / 2.0 / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * 2.0 / n_points
    
    start_time = time.time()
    for _ in range(int((budget_s - (time.time() - start_time)) / 0.1)):
        corr = np.correlate(h, 1 - h, mode="full")
        lag = np.argmax(corr)
        overlap = h * (np.roll(1 - h, lag))
        overlap = np.clip(overlap, 0, 1)
        candidates = []
        
        top_indices = np.argsort(overlap)[::-1][:int(len(overlap) * 0.1)]
        for i in top_indices:
            h_candidate = h.copy()
            h_candidate[i] = 1 - h_candidate[i]
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * n_points / 2.0 / np.sum(h_candidate)
            candidates.append(h_candidate)
        
        for i in np.random.choice(len(h), size=5, replace=False):
            noise = np.random.uniform(-0.05, 0.05, size=len(h))
            h_candidate = h.copy() + noise
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * n_points / 2.0 / np.sum(h_candidate)
            candidates.append(h_candidate)
        
        for candidate in candidates:
            try:
                c5 = np.max(np.correlate(candidate, 1 - candidate, mode="full")) * 2.0 / n_points
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = candidate.copy()
            except:
                pass
        
        if np.max(np.abs(h - best_h)) < 1e-5:
            break
        h = best_h
    
    h = np.clip(h, 0, 1)
    h = h * n_points / 2.0 / np.sum(h)
    return h, best_c5, n_points
