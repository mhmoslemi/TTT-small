                candidate = candidate * (n_points / 2) / candidate_sum
            population.append(candidate)

        for i in range(len(selected) // 2):
            p1 = selected[i]
            p2 = selected[len(selected) - 1 - i]
            crossover_point = np.random.randint(1, n_points - 1)

                child = child * (n_points / 2) / child_sum

            mutation = np.random.normal(0, mutation_noise, n_points)

    final_c5 = compute_c5(best_h, dx)
    return (best_h, final_c5, n_points)

def compute_c5(h, dx):
    overlaps = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlaps)
