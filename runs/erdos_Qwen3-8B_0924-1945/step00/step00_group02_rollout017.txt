```python
import numpy as np
import scipy.optimize

def run(seed=42, budget_s=600, **kwargs):
    import numpy as np
    import time
    
    np.random.seed(seed)
    
    n_points = 300
    initial_h_values = np.array([0.5 * (1 + np.random.uniform(-1, 1, n_points)) for _ in range(1)])
    h = initial_h_values[0]
    
    # Clip and scale to [0,1] and sum to n_points/2
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if abs(total - n_points / 2) > 1e-6:
        scale = (n_points / 2) / total
        h = h * scale
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = np.inf
    
    max_iterations = int(budget_s / 0.1)
    iteration = 0
    start_time = time.time()
    
    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        c5 = np.max(corr) * 2.0 / n_points
        return c5
    
    while iteration < max_iterations and time.time() - start_time < budget_s:
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        corr = np.correlate(h, 1 - h, mode="full")
        lag = np.argmax(corr)
        overlap = h * np.roll(1 - h, lag)
        overlap_vals = overlap * 2.0 / n_points
        max_overlap = np.max(overlap_vals)
        target_indices = np.where(overlap_vals > 0.8 * max_overlap)[0]
        
        candidates = []
        for i in np.random.choice(target_indices, min(len(target_indices), 10), replace=False):
            # Flip
            h_flip = h.copy()
            h_flip[i] = 1 - h_flip[i]
            total_flip = np.sum(h_flip)
            scale_flip = (n_points / 2) / total_flip
            h_flip = h_flip * scale_flip
            h_flip = np.clip(h_flip, 0, 1)
            candidates.append(h_flip)
            
            # Add noise
            h_noise = h.copy()
            h_noise[i] += np.random.uniform(-0.05, 0.05)
            h_noise = np.clip(h_noise, 0, 1)
            total_noise = np.sum(h_noise)
            scale_noise = (n_points / 2) / total_noise
            h_noise = h_noise * scale_noise
            h_noise = np.clip(h_noise, 0, 1)
            candidates.append(h_noise)
        
        for candidate in candidates:
            c5_candidate = compute_c5(candidate)
            if c5_candidate < c5:
                h = candidate.copy()
                break
        
        iteration += 1
    
    h = best_h
    c5 = compute_c5(h)
    return (h, c5, n_points)
```
