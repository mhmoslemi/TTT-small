import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    initial_h_values = np.load('initial_h_values.npy')  # Assume this is preloaded
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial check
    if not np.isclose(np.sum(initial_h_values), target_sum, atol=1e-6):
        initial_h_values = initial_h_values / np.sum(initial_h_values) * target_sum
    
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    elapsed_time = 0
    alpha = 0.1
    eps = 1e-8
    max_iter = 1000
    iter_count = 0
    improvement_threshold = 1e-6
    
    while elapsed_time < budget_s and iter_count < max_iter:
        start_time = time.time()
        
        # Compute cross-correlation
        cross_corr = np.correlate(h, 1 - h, mode="full") * dx
        max_index = np.argmax(cross_corr)
        max_val = cross_corr[max_index]
        
        # Compute gradient for the max lag
        h_forward = h.copy()
        h_forward[max_index] += eps
        h_forward = np.clip(h_forward, 0, 1)
        h_forward = h_forward * (target_sum / np.sum(h_forward))
        
        h_backward = h.copy()
        h_backward[max_index] -= eps
        h_backward = np.clip(h_backward, 0, 1)
        h_backward = h_backward * (target_sum / np.sum(h_backward))
        
        c5_forward = evaluate_erdos_solution(h_forward)
        c5_backward = evaluate_erdos_solution(h_backward)
        
        grad = (c5_forward - c5_backward) / (2 * eps)
        
        # Update h
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
        # Evaluate new C5
        new_c5 = evaluate_erdos_solution(h)
        
        # Update best solution
        if new_c5 < best_c5 - improvement_threshold:
            best_c5 = new_c5
            best_h = h.copy()
        
        # Logging
        if iter_count % 10 == 0:
            print(f"Iter {iter_count}, C5: {new_c5:.6f}")
        
        elapsed_time = time.time() - start_time
        iter_count += 1
        alpha *= 0.995
    
    return best_h, best_c5, n_points
