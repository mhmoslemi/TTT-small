import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = evaluate_erdos_solution.initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=np.float64)
    assert np.all(h >= 0) and np.all(h <= 1)
    assert np.isclose(np.sum(h), n_points / 2, atol=1e-6)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    max_iterations = 50
    time_limit = budget_s * 0.8
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > time_limit:
            break
        
        # Convex relaxation formulation
        h_var = cp.Variable(n_points, lb=0, ub=1)
        t = cp.Variable()
        lambda_penalty = 1e-3
        objective = t + lambda_penalty * cp.sum_squares(h_var - 0.5)
        
        constraints = []
        for k in range(1, n_points):
            # Wrap around for circular convolution
            indices = np.arange(n_points) + k
            indices[indices >= n_points] -= n_points
            expr = cp.sum(h_var * (1 - h_var[indices])) * dx
            constraints.append(t >= expr)
        
        constraints.append(cp.sum(h_var) == n_points / 2)
        
        problem = cp.Problem(cp.Minimize(objective), constraints)
        try:
            problem.solve(solver=cp.ECOS, eps=1e-6)
        except:
            pass
        
        if problem.status == 'optimal':
            h = h_var.value
            if h is not None:
                np.clip(h, 0, 1, out=h)
                np.sum(h, out=np.zeros(1))
                scale = (n_points / 2) / np.sum(h)
                h *= scale
                current_c5 = evaluate_erdos_solution(h, dx)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
        else:
            break
        
        # Gradient descent refinement
        h_refined = best_h.copy()
        lr = 1e-2
        for _ in range(100):
            if time.time() - start_time > time_limit:
                break
            c5 = evaluate_erdos_solution(h_refined, dx)
            if c5 < best_c5:
                best_h = h_refined.copy()
                best_c5 = c5
            # Perturb and clip
            noise = np.random.randn(n_points) * 0.05
            h_refined += noise
            np.clip(h_refined, 0, 1, out=h_refined)
            np.sum(h_refined, out=np.zeros(1))
            scale = (n_points / 2) / np.sum(h_refined)
            h_refined *= scale
        if evaluate_erdos_solution(h_refined, dx) < best_c5:
            best_h = h_refined.copy()
            best_c5 = evaluate_erdos_solution(h_refined, dx)
    
    # Final validation
    np.clip(best_h, 0, 1, out=best_h)
    scale = (n_points / 2) / np.sum(best_h)
    best_h *= scale
    final_c5 = evaluate_erdos_solution(best_h, dx)
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    assert np.isclose(np.sum(best_h), n_points / 2, atol=1e-6)
    assert np.isfinite(final_c5)
    
    return best_h, final_c5, n_points
