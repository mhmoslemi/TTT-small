```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    max_stagnation = 50
    pop_size = 50
    mutation_noise = 0.15
    mutation_rate = 0.1
    n_points = 500
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h = np.clip(initial_h_values, 0, 1)
    else:
        h = np.zeros(n_points)
        h[0:250] = 1.0
        h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (n_points / 2) / h_sum
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    assert np.isclose(h_sum, n_points / 2), "Initial h does not satisfy the sum constraint"
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    stagnation_counter = 0
    for _ in range(1000):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        population = []
        for _ in range(pop_size):
            candidate = h.copy()
            noise = np.random.randn(n_points) * mutation_noise
            candidate += noise
            candidate = np.clip(candidate, 0, 1)
            candidate_sum = np.sum(candidate)
            if candidate_sum != 0:
                candidate = candidate * (n_points / 2) / candidate_sum
            population.append(candidate)
        overlaps = []
        for candidate in population:
            corr = np.correlate(candidate, 1 - candidate, mode='full')
            corr *= dx
            c = np.max(corr)
            overlaps.append((c, candidate))
        overlaps.sort()
        new_population = []
        for _ in range(pop_size):
            if len(overlaps) >= 2:
                parent1, parent2 = np.random.choice(overlaps, size=2, replace=False)
                child = np.concatenate((parent1[1][:int(n_points/2)], parent2[1][int(n_points/2):]))
                child = np.clip(child, 0, 1)
                child_sum = np.sum(child)
                if child_sum != 0:
                    child = child * (n_points / 2) / child_sum
                new_population.append(child)
            else:
                new_population.append(np.random.rand(n_points))
        for candidate in new_population:
            if np.random.rand() < mutation_rate:
                noise = np.random.randn(n_points) * mutation_noise
                candidate += noise
                candidate = np.clip(candidate, 0, 1)
                candidate_sum = np.sum(candidate)
                if candidate_sum != 0:
                    candidate = candidate * (n_points / 2) / candidate_sum
            new_population[-1] = candidate
        overlaps = []
        for candidate in new_population:
            corr = np.correlate(candidate, 1 - candidate, mode='full')
            corr *= dx
            c = np.max(corr)
            overlaps.append((c, candidate))
        overlaps.sort()
        if overlaps[0][0] < best_c5:
            best_c5 = overlaps[0][0]
            best_h = overlaps[0][1].copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
        if stagnation_counter >= max_stagnation:
            mutation_noise *= 0.5
            stagnation_counter = 0
    h = best_h.copy()
    h_sum = np.sum(h)
    assert np.isclose(h_sum, n_points / 2), "Final h does not satisfy the sum constraint"
    corr = np.correlate(h, 1 - h, mode='full')
    corr *= dx
    c5_bound = np.max(corr)
    return (h, c5_bound, n_points)
```
