import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=np.float64)
    sum_h = np.sum(h)
    if not np.isclose(sum_h, n_points / 2):
        h = h * (n_points / 2) / sum_h
    best_h = h.copy()
    best_c5 = scipy.signal.correlation(h, h, mode='full')[1] * dx
    start_time = time.time()

    def compute_c5(h):
        corr = scipy.signal.correlation(h, h, mode='full')
        return np.max(corr) * dx

    for _ in range(500):
        if time.time() - start_time > budget_s - 10:
            break
        corr = scipy.signal.correlation(h, h, mode='full')
        max_c5 = np.max(corr)
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
        max_idx = np.argmax(corr)
        if max_idx >= n_points:
            k_wrap = max_idx - n_points
            grad = np.zeros(n_points)
            for i in range(n_points - k_wrap):
                grad[i] += dx * h[i + k_wrap]
                grad[i + k_wrap] += dx * h[i]
        else:
            grad = np.zeros(n_points)
            for i in range(n_points - max_idx):
                grad[i] += dx * h[i + max_idx]
                grad[i + max_idx] += dx * h[i]
        direction = -grad
        new_h = h + 0.1 * direction
        new_h = np.clip(new_h, 0.0, 1.0)
        sum_new = np.sum(new_h)
        if sum_new != 0:
            new_h = new_h * (n_points / 2) / sum_new
        new_h = np.clip(new_h, 0.0, 1.0)
        sum_new = np.sum(new_h)
        if sum_new != 0:
            new_h = new_h * (n_points / 2) / sum_new
        new_h = np.clip(new_h, 0.0, 1.0)
        if np.random.rand() < 0.05:
            noise = np.random.uniform(-0.05, 0.05, n_points)
            new_h = h + noise
            new_h = np.clip(new_h, 0.0, 1.0)
            sum_new = np.sum(new_h)
            if sum_new != 0:
                new_h = new_h * (n_points / 2) / sum_new
            new_h = np.clip(new_h, 0.0, 1.0)
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
        if np.isclose(best_c5, 0.3807, atol=1e-8):
            break

    h = best_h
    sum_h = np.sum(h)
    if not np.isclose(sum_h, n_points / 2):
        h = h * (n_points / 2) / sum_h
    h = np.clip(h, 0.0, 1.0)
    final_c5 = compute_c5(h)
    return h.tolist(), final_c5, n_points
