import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.array([0.0, 0.0, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5])
    h_values = initial_h_values.copy()
    h_values = h_values / np.sum(h_values) * (n_points / 2)
    h_values = np.clip(h_values, 0, 1)
    
    best_c5 = float('inf')
    best_h = h_values.copy()
    start_time = time.time()
    
    for iteration in range(500):
        if time.time() - start_time > budget_s:
            break
        # Compute full correlate
        correlate_vals = np.correlate(h_values, 1 - h_values, mode="full") * (2.0 / n_points)
        current_c5 = np.max(correlate_vals)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_values.copy()
        
        # Find worst lag
        worst_lag = np.argmax(correlate_vals)
        
        # Compute gradient for worst lag
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h_values[j]) * (2.0 / n_points)
        
        # Update h
        learning_rate = 0.05 * (0.99 ** iteration)
        h_values = h_values - learning_rate * gradients
        h_values = np.clip(h_values, 0, 1)
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        
        # Add structured noise
        if iteration % 100 == 0:
            noise_scale = 0.05
            h_values += noise_scale * np.random.normal(0, 1, n_points)
            h_values = np.clip(h_values, 0, 1)
            h_values = h_values * (n_points / 2) / np.sum(h_values)
    
    # Final check
    correlate_vals = np.correlate(best_h, 1 - best_h, mode="full") * (2.0 / n_points)
    final_c5 = np.max(correlate_vals)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h, final_c5, n_points)
