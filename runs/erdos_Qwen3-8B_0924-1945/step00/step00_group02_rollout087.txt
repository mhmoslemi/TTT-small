```python
import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 500
    max_gen = 500
    pop_size = 50
    tournament_size = 5
    crossover_rate = 0.2
    mutation_rate = 0.1
    mutation_scale = 0.1
    mutation_freq = 0.1
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    def evaluate(h):
        if h.shape[0] != n_points:
            h = np.resize(h, n_points)
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def rescale(h):
        total = np.sum(h)
        if abs(total - n_points / 2) < 1e-6:
            return h
        scale = (n_points / 2) / total
        return np.clip(h * scale, 0, 1)

    def mutate(h):
        indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
        h[indices] += np.random.uniform(-mutation_scale, mutation_scale, len(indices))
        h = np.clip(h, 0, 1)
        h = rescale(h)
        return h

    def crossover(p1, p2):
        mask = np.random.rand(n_points) > 0.5
        child = np.where(mask, p1, p2)
        child = rescale(child)
        return child

    def initialize_population():
        if initial_h_values is not None:
            pop = [np.copy(initial_h_values)]
        else:
            pop = [np.random.rand(n_points)]
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points)
            h = rescale(h)
            pop.append(h)
        return pop

    pop = initialize_population()
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        if gen % 50 == 0 and gen != 0:
            mutation_rate = 0.05
        if gen % 200 == 0:
            mutation_rate = 0.1
        if gen % 100 == 0:
            mutation_scale = 0.15
        elif gen % 150 == 0:
            mutation_scale = 0.05
        if gen % 10 == 0 and gen != 0:
            mutation_rate = 0.1
        if gen % 100 == 0:
            mutation_scale = 0.05
        if gen % 20 == 0:
            mutation_scale = 0.08
        if gen % 50 == 0:
            mutation_scale = 0.1
        if gen % 200 == 0:
            mutation_scale = 0.05
        if gen % 100 == 0:
            mutation_scale = 0.05

        scores = [evaluate(h) for h in pop]
        for i in range(pop_size):
            if scores[i] < best_c5:
                best_c5 = scores[i]
                best_h = np.copy(pop[i])

        new_pop = []
        for _ in range(pop_size // 2):
            tournament = np.random.choice(pop, tournament_size)
            winner = tournament[np.argmin([evaluate(h) for h in tournament])]
            new_pop.append(winner)
        for _ in range(pop_size - len(new_pop)):
            p1, p2 = np.random.choice(pop, 2, replace=False)
            if np.random.rand() < crossover_rate:
                child = crossover(p1, p2)
            else:
                child = np.copy(p1)
            child = mutate(child)
            new_pop.append(child)
        pop = new_pop

    best_h = rescale(best_h)
    c5_final = evaluate(best_h)
    return best_h, c5_final, n_points
```
