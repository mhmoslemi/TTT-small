import scipy.signal
import time
import random

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2 / n_points
    h = initial_h_values.copy()

        if time.time() - start_time > budget_s:

        corr = scipy.signal.correlate(h, 1 - h, mode='full')

        lag = np.argmax(corr)
        if max_corr < best_c5:
            best_c5 = max_corr
            best_h = h.copy()
        if lag >= n_points:
            lag -= n_points
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = i - lag
            if 0 <= j < n_points:
                gradient[i] = (1 - h[j]) * dx
        learning_rate = 0.01
        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if np.isclose(total, n_points / 2):
            pass
        else:
            h = h / total * (n_points / 2)
        h = np.clip(h, 0, 1)
    
    h = best_h.copy()
    total = np.sum(h)

    else:
        h = h / total * (n_points / 2)
    h = np.clip(h, 0, 1)
    corr = scipy.signal.correlate(h, 1 - h, mode='full')
