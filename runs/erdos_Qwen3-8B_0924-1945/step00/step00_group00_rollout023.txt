```python
import numpy as np
import scipy.optimize as opt
import time
import random

# Pre-imported helpers
def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    correlate_vals = np.correlate(h_values, 1 - h_values, mode="full") * dx
    return np.max(correlate_vals)

def initial_h_values(n_points):
    samples = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            samples[i] = 1.0
        else:
            samples[i] = 0.0
    return samples

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    h = initial_h_values(n_points)
    sum_h = np.sum(h)
    h = h / (sum_h / (n_points / 2)) if sum_h != n_points / 2 else h

    def objective(x):
        nonlocal n_points
        h = x.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        if np.abs(np.sum(h) - n_points / 2) > 1e-6:
            return np.inf
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    x0 = h.copy()

    start_time = time.time()
    result = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
    elapsed = time.time() - start_time

    h_values = result.x
    c5_bound = result.fun
    n_points_val = n_points

    if elapsed > budget_s:
        print("Time budget exceeded")
    else:
        print("Optimization completed within budget")

    return h_values, c5_bound, n_points_val
```
