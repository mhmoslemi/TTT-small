```python
import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Define parameters
    n_points = 100
    pop_size = 50
    generations = 100
    mutation_rate = 0.1
    tournament_size = 5
    
    # Initialize population
    if 'initial_h_values' in globals() and initial_h_values is not None:
        initial_h = initial_h_values.copy()
        norm_factor = n_points / 2.0 / np.sum(initial_h)
        h_vals = (initial_h * norm_factor).astype(np.float64)
    else:
        h_vals = np.random.rand(n_points)
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2.0)
    
    def evaluate_fitness(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    best_h = h_vals.copy()
    best_c5 = evaluate_fitness(h_vals)
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [evaluate_fitness(ind) for ind in [h_vals]]
        
        # Tournament selection
        parents = []
        for _ in range(pop_size):
            tournament = np.random.choice(range(pop_size), tournament_size, replace=False)
            best_idx = np.argmin([fitness[i] for i in tournament])
            parents.append(tournament[best_idx])
        
        # Crossover
        offspring = []
        for i in range(0, pop_size, 2):
            p1 = parents[i]
            p2 = parents[i+1]
            if np.random.rand() < 0.8:
                split = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([h_vals[p1][:split], h_vals[p2][split:]])
                child2 = np.concatenate([h_vals[p2][:split], h_vals[p1][split:]])
                offspring.extend([child1, child2])
            else:
                offspring.extend([h_vals[p1], h_vals[p2]])
        
        # Mutation
        for i in range(len(offspring)):
            h = offspring[i]
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    h[j] += np.random.uniform(-0.1, 0.1)
                    h[j] = np.clip(h[j], 0, 1)
            h = h * (n_points / 2.0) / np.sum(h)
            offspring[i] = h
        
        # Replace worst with offspring
        for i in range(len(offspring)):
            h = offspring[i]
            if np.random.rand() < 0.5:
                h_vals[i] = h
        
        # Update best
        current_fitness = evaluate_fitness(h_vals)
        if current_fitness < best_c5:
            best_c5 = current_fitness
            best_h = h_vals.copy()
    
    # Final check
    h_vals = best_h.copy()
    h_vals = np.clip(h_vals, 0, 1)
    h_vals = h_vals * (n_points / 2.0) / np.sum(h_vals)
    corr = np.correlate(h_vals, 1 - h_vals, mode='full')
    final_c5 = np.max(corr) * (2.0 / n_points)
    
    return h_vals, final_c5, n_points
```
