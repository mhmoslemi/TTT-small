        h_flat = x.reshape(n_points)
        conv = np.correlate(h_flat, 1 - h_flat, mode='full')
        c5 = np.max(conv) * dx
        return c5

        return np.sum(x) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial optimization with SLSQP
    result = optimize.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        tol=1e-5,
        options={'maxiter': 100}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # If optimization failed, try perturbing and optimizing
    if result.success:
        h_perturb = best_h + np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        h_perturb_sum = np.sum(h_perturb)
        h_perturb = h_perturb * (n_points / 2) / h_perturb_sum
        
        result_perturb = optimize.minimize(
            objective,
            h_perturb,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            tol=1e-5,
            options={'maxiter': 100}
        )

            best_h = result_perturb.x
