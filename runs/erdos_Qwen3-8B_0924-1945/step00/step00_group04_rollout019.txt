def run(seed=42, budget_s=600, **kwargs):

    else:

    learning_rate = 0.05
    learning_decay = 0.995
    max_iterations = 1000
    no_improvement_counter = 0
    start_time = time.time()
    
    def compute_c5(h):
        dx = 2.0 / n_points
        cross_corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(cross_corr)
    
    def update_h(h, gradient, learning_rate, target_sum):
        h_new = h - learning_rate * gradient
        h_new = np.clip(h_new, 0, 1)
        sum_h = np.sum(h_new)
        if sum_h == 0:
            return np.zeros(n_points)
        h_new = h_new * (target_sum / sum_h)
        return h_new
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        cross_corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        idx = np.argmax(cross_corr)
        lag_k = idx - (n_points - 1)
        grad = np.zeros(n_points)
        for j in range(n_points):
            if j + lag_k < n_points:
                grad[j] += (1 - h[j + lag_k]) * (2.0 / n_points)
            if j - lag_k >= 0:
                grad[j] -= h[j - lag_k] * (2.0 / n_points)
        h = update_h(h, grad, learning_rate, target_sum)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_counter = 0
        else:
            no_improvement_counter += 1
        if no_improvement_counter >= 50:
            break
        learning_rate *= learning_decay
    
    # Fallback to initial if no improvement
    if best_c5 > compute_c5(initial_h_values):
        best_h = np.array(initial_h_values, dtype=float)
        best_h = np.clip(best_h, 0, 1)
        best_h *= target_sum / np.sum(best_h)
        best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n_points)
