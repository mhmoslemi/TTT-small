        noise = np.random.normal(0, lambda0, size=n_points)

        # Constraint restoration
        current_sum = np.sum(perturbed)

        perturbed = np.clip(perturbed, 0.0, 1.0)
        if np.sum(perturbed) != 0:
            perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(perturbed, dx)

            best_c5 = new_c5

    # Final checks
    assert (best_solution >= 0).all() and (best_solution <= 1).all()
    assert np.isclose(np.sum(best_solution), n_points / 2, atol=1e-6)
    assert np.isfinite(best_c5)
    
    return (best_solution, best_c5, n_points)
