def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    max_iter = int(budget_s // 0.1)  # ~6,000 iterations for 600 seconds
    best_c5 = float('inf')
    best_h = np.copy(initial_h_values)
    
    # Preprocess initial solution
    best_h = np.clip(best_h, 0, 1)
    total_mass = np.sum(best_h)
    scaling = (n_points / 2) / total_mass if total_mass != 0 else 1
    best_h = best_h * scaling
    best_c5 = compute_c5(best_h, n_points)
    
    start_time = time.time()
    for t in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Compute correlations
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        max_corr = np.max(corr)
        lag = np.argmax(corr)
        
        # Identify contributing regions
        overlap = best_h * (1 - np.roll(best_h, lag))
        max_overlap = np.max(overlap)
        threshold = max_overlap * 0.8
        target_indices = np.where(overlap > threshold)[0]
        target_indices = np.random.choice(target_indices, size=min(len(target_indices), int(n_points * 0.1)), replace=False)
        
        # Generate candidates
        candidates = []
        for i in target_indices:
            h_candidate = np.copy(best_h)
            # Flip
            h_candidate[i] = 1 - h_candidate[i]
            h_candidate = np.clip(h_candidate, 0, 1)
            total_mass = np.sum(h_candidate)
            scaling = (n_points / 2) / total_mass if total_mass != 0 else 1
            h_candidate = h_candidate * scaling
            candidates.append(h_candidate)
            
            # Noise
            h_candidate = np.copy(best_h)
            h_candidate[i] += np.random.uniform(-0.05, 0.05)
            h_candidate = np.clip(h_candidate, 0, 1)
            total_mass = np.sum(h_candidate)
            scaling = (n_points / 2) / total_mass if total_mass != 0 else 1
            h_candidate = h_candidate * scaling
            candidates.append(h_candidate)
        
        # Evaluate candidates
        for h_candidate in candidates:
            c5 = compute_c5(h_candidate, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_candidate
        
        # Early termination check
        if t % 500 == 0 and t > 0:
            if np.abs(best_c5 - compute_c5(best_h, n_points)) < 1e-6:
                break
        
        # Fallback check
        if t % 200 == 0 and t > 0:
            if np.abs(best_c5 - compute_c5(best_h, n_points)) < 1e-6:
                best_h = np.copy(initial_h_values)
                best_h = np.clip(best_h, 0, 1)
                total_mass = np.sum(best_h)
                scaling = (n_points / 2) / total_mass if total_mass != 0 else 1
                best_h = best_h * scaling
                best_c5 = compute_c5(best_h, n_points)
    
    return best_h, best_c5, n_points

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    c5 = max_corr * 2.0 / n_points
    return c5
