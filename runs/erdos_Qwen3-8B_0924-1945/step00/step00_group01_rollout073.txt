import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Initial construction if available

        # Construct an initial guess based on a periodic pattern
        n_points = 47
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0.0
        h = h / np.sum(h) * (n_points / 2)
    
    n_points = len(h)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-8:
        h = h / sum_h * (n_points / 2)
    
    def objective(x):
        h_vals = x.copy()
        # Ensure values are in [0, 1]
        h_vals = np.clip(h_vals, 0.0, 1.0)
        # Normalize to satisfy the integral constraint
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
        # Compute the correlation
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx
    
    def gradient(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0.0, 1.0)
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
        # Compute the correlation
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        max_index = np.argmax(corr)
        # Compute derivative
        grad = np.zeros_like(h_vals)
        for i in range(n_points):
            h_vals[i] += 1e-6
            corr_new = np.correlate(h_vals, 1 - h_vals, mode="full")
            grad[i] = (np.max(corr_new) - np.max(corr)) / 1e-6
            h_vals[i] -= 1e-6
        return grad
    
    # Set up constraints
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
    
    # Minimize the objective
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        jac=gradient,
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    # Check if optimization was successful
    if result.success:
        best_h = result.x
    else:
        best_h = h
    
    # Ensure the final solution meets the constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    # Evaluate the final solution
    corr = np.correlate(best_h, 1 - best_h, mode="full")
    c5_bound = np.max(corr) * dx
    
    return (best_h, c5_bound, n_points)
