import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = np.roll(h, n_points//4)

    h = h / (np.sum(h) * dx) * (n_points / 2)

    def objective(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        norm = np.sum(h_vals) * dx
        if abs(norm - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def grad_objective(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        norm = np.sum(h_vals) * dx
        if abs(norm - n_points / 2) > 1e-6:
            return np.inf * np.ones_like(h_vals)
        h_vals = h_vals / (np.sum(h_vals) * dx) * (n_points / 2)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        c5 = np.max(corr) * dx
        max_corr = np.max(corr)
        max_idx = np.argmax(corr)
        l1, l2 = max_idx, max_idx
        if l1 < 0:
            l1 = 0
        if l2 >= len(h_vals):
            l2 = len(h_vals) - 1
        h_1 = h_vals
        h_2 = 1 - h_vals
        if l1 >= len(h_1):
            l1 = len(h_1) - 1
        if l2 >= len(h_2):
            l2 = len(h_2) - 1
        h1 = h_1[l1]
        h2 = h_2[l2]
        h1 = h1 if l1 >= 0 else 0
        h2 = h2 if l2 >= 0 else 0
        grad = np.zeros_like(h_vals)
        for i in range(len(h_vals)):
            h_vals[i] = 1.0
            corr = np.correlate(h_vals, 1 - h_vals, mode="full")
            c5 = np.max(corr) * dx
            grad[i] = (c5 - objective(h_vals.copy())) / 1e-6
            h_vals[i] = 0.0
        return grad

    result = opt.minimize(
        objective,
        h,
        method="L-BFGS-B",
        bounds=[(0, 1) for _ in range(n_points)],
        jac=grad_objective,
        options={
            'ftol': 1e-8,
            'gtol': 1e-8,
            'maxiter': 100,
            'disp': False
        }
    )
    h_best = result.x.copy()
    h_best = np.clip(h_best, 0, 1)
    norm = np.sum(h_best) * dx
    if abs(norm - n_points / 2) > 1e-6:
        h_best = h
    corr = np.correlate(h_best, 1 - h_best, mode="full")
    c5_bound = np.max(corr) * dx
    return (h_best, c5_bound, n_points)
