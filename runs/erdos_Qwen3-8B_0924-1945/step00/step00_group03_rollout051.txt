import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    initial_h_values = np.array([0.0] * n_points)
    initial_h_values = np.random.choice([0.0, 1.0], size=n_points, p=[0.6, 0.4])
    initial_h_values = initial_h_values + np.random.rand(n_points) * 0.1 - 0.05
    initial_h_values = np.clip(initial_h_values, 0.0, 1.0)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    initial_h_values = np.round(initial_h_values, 5)
    dx = 2.0 / n_points

    def objective(h):
        h = np.clip(h, 0.0, 1.0)
        conv = np.correlate(h, 1.0 - h, mode='full')
        max_overlap = np.max(conv) * dx
        return max_overlap

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = optimize.minimize(
        objective,
        initial_h_values,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'ftol': 1e-8, 'maxiter': 100}
    )

    if result.success:
        h_values = result.x
    else:
        h_values = initial_h_values.copy()
        for _ in range(3):
            perturb = np.random.uniform(-0.1, 0.1, size=n_points)
            h_values = np.clip(h_values + perturb, 0.0, 1.0)
            h_values = h_values * (n_points / 2) / np.sum(h_values)
            h_values = np.round(h_values, 5)
        h_values = np.clip(h_values, 0.0, 1.0)
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        h_values = np.round(h_values, 5)

    h_values = np.clip(h_values, 0.0, 1.0)
    h_values = h_values * (n_points / 2) / np.sum(h_values)
    h_values = np.round(h_values, 5)
    dx = 2.0 / n_points
    conv = np.correlate(h_values, 1.0 - h_values, mode='full')
    c5_bound = np.max(conv) * dx
    return (h_values, c5_bound, n_points)
