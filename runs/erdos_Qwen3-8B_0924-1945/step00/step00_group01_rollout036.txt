```python
import numpy as np
import cvxpy as cp
import time

initial_h_values = None  # Placeholder, will be provided by the environment

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 50
    if initial_h_values is not None and len(initial_h_values) == n_points:
        h_vals = initial_h_values.copy()
    else:
        h_vals = np.random.rand(n_points)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
    
    dx = 2.0 / n_points
    best_h_vals = h_vals.copy()
    best_c5 = float('inf')
    start_time = time.time()
    max_iter = 200
    early_stop = 0
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        current_c5 = np.max(corr * dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_vals = h_vals.copy()
            early_stop = 0
        else:
            early_stop += 1
        
        if early_stop >= 5:
            break
        
        k_max = np.argmax(corr)
        overlap_k = corr[k_max] * dx
        
        # Convex optimization surrogate: minimize 0.5*λ^2 + (overlap_k - λ)^2
        λ = cp.Variable()
        objective = 0.5 * λ**2 + (overlap_k - λ)**2
        prob = cp.Problem(cp.Minimize(objective))
        prob.solve()
        
        if prob.status == 'optimal':
            λ_val = λ.value
        else:
            λ_val = 1.0
        
        # Gradient of the objective w.r.t h_vals (simplified)
        # Here we use a simple update rule based on the gradient
        grad = np.zeros_like(h_vals)
        for i in range(n_points):
            # Approximate the gradient using finite differences
            h_plus = h_vals.copy()
            h_plus[i] += 1e-5
            h_minus = h_vals.copy()
            h_minus[i] -= 1e-5
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            diff = (corr_plus[k_max] - corr_minus[k_max]) * dx / (2 * 1e-5)
            grad[i] = diff
        
        # Update h_vals using the gradient with λ
        h_vals -= λ_val * grad
        h_vals = np.clip(h_vals, 0, 1)
        mass = np.sum(h_vals)
        h_vals = h_vals * (n_points / 2) / mass
    
    # Final evaluation
    corr = np.correlate(best_h_vals, 1 - best_h_vals, mode='full')
    final_c5 = np.max(corr * dx)
    return best_h_vals, final_c5, n_points
```
