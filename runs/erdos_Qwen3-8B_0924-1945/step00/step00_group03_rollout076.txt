import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in globals() else np.random.rand(70)
    n_points = initial_h_values.shape[0]
    dx = 2.0 / n_points
    sum_h = n_points / 2

    # Convert initial_h_values to block pattern
    block_size = 1
    m = n_points
    block_values = np.round(np.mean(np.reshape(initial_h_values, (m, block_size)), axis=1)).astype(int)
    block_values = np.where(block_values == 0, 0, 1)
    block_values = block_values - (np.sum(block_values) - m // 2) % 2
    block_values = np.where(block_values == 0, 0, 1)

    def evaluate_c5(h_values):
        h = h_values
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_c5(h_values):
        h = h_values
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def update_block_values(block_values, i, delta):
        if i + delta < 0 or i + delta >= len(block_values):
            return block_values
        block_values[i], block_values[i + delta] = block_values[i + delta], block_values[i]
        total = np.sum(block_values)
        if total > m // 2:
            block_values[-1] = 0
        elif total < m // 2:
            block_values[-1] = 1
        return block_values

    def block_refinement(block_values, max_iter=500, time_budget=600):
        best_block_values = block_values.copy()
        best_c5 = compute_c5(best_block_values)
        start_time = time.time()
        for _ in range(max_iter):
            if time.time() - start_time > time_budget:
                break
            for i in range(len(best_block_values)):
                for delta in [-1, 1]:
                    new_block = best_block_values.copy()
                    new_block = update_block_values(new_block, i, delta)
                    current_c5 = compute_c5(new_block)
                    if current_c5 < best_c5:
                        best_block_values = new_block
                        best_c5 = current_c5
            if best_c5 < 0.3807:
                break
        return best_block_values, best_c5

    best_block, best_c5 = block_refinement(block_values, max_iter=500, time_budget=budget_s)

    def local_optimization(h_values, max_iter=500, learning_rate=0.01):
        h = h_values.copy()
        for _ in range(max_iter):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            grad = np.zeros_like(h)
            for l in range(len(corr) - 1):
                grad[l] -= (corr[l] - max_corr) * (1 - h[l]) * (1 - h[l + 1]) * dx
                grad[l + 1] -= (corr[l + 1] - max_corr) * (1 - h[l]) * (1 - h[l + 1]) * dx
            grad = np.clip(grad, -0.1, 0.1)
            h += learning_rate * grad
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            if abs(total - sum_h) > 1e-6:
                h = h * (sum_h / total)
        return h

    h_values = np.zeros(n_points)
    for i in range(len(best_block)):
        for j in range(int(np.ceil(n_points / len(best_block)))):
            idx = i + j * len(best_block)
            if idx >= n_points:
                break
            h_values[idx] = best_block[i]

    h_values = local_optimization(h_values, max_iter=500, learning_rate=0.01)

    final_c5 = compute_c5(h_values)

    assert np.all(h_values >= 0) and np.all(h_values <= 1)
    assert np.isclose(np.sum(h_values), sum_h, atol=1e-6)
    assert np.isfinite(final_c5)

    return h_values, final_c5, n_points
