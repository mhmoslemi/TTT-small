            h = np.full(n_points, 0.5)

        h = h.copy()
        for i in range(n_points):
            h[i] += np.random.normal(0, mutation_noise * mutation_noise_factor)

            h = h * (n_points / 2) / total

                score = evaluate_c5(h)

        indices = np.argsort(scores)

        return [population[i] for i in top_indices]

        selected = []

            candidates = np.random.choice(range(len(population)), tournament_size, replace=False)
            best_idx = np.argmin([scores[i] for i in candidates])
            selected.append(population[candidates[best_idx]])
        return selected

    population = []
