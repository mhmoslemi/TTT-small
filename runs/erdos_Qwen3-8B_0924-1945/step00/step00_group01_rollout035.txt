import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 50
    if 'initial_h_values' in globals():
        h_vals = np.array(initial_h_values, dtype=float)
    else:
        h_vals = np.random.rand(n_points)
    h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    dx = 2.0 / n_points
    max_iter = 200
    tol = 1e-6
    best_c5 = np.inf
    best_h = h_vals.copy()
    best_n_points = n_points
    start_time = time.time()
    
    for _ in range(max_iter):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        current_c5 = corr * dx
        c5 = np.max(current_c5)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_vals.copy()
            best_n_points = n_points
        
        if time.time() - start_time > budget_s - 10:
            break
        
        if c5 < tol:
            break
        
        k_max = np.argmax(current_c5)
        overlap = current_c5[k_max]
        
        lambda_var = cp.Variable()
        objective = cp.Minimize(0.5 * lambda_var**2 + (overlap - lambda_var)**2)
        prob = cp.Problem(objective, [])
        prob.solve()
        if prob.status != 'OPTIMAL':
            continue
        
        grad = (overlap - lambda_var.value) * (1 - 2 * lambda_var.value)
        h_vals += grad * 0.01
        h_vals = np.clip(h_vals, 0, 1)
        mass = np.sum(h_vals)
        h_vals = h_vals * (n_points / 2) / mass
    
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr * dx)
    return (best_h, c5_bound, best_n_points)
