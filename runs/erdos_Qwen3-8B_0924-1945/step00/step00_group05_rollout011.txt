def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    h = initial_h_values.copy()
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    if sum_h == 0.0:
        h = np.zeros(n_points)
    else:
        h = h * (n_points / 2) / sum_h
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    learning_rate = 0.1
    iteration = 0
    start_time = time.time()

    while time.time() - start_time < budget_s:
        overlaps = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n_points)
        k_max = np.argmax(overlaps)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                grad[i] += (1.0 - h[j]) * (2.0 / n_points)
                grad[j] -= h[i] * (2.0 / n_points)
        h -= learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0.0:
            h = np.zeros(n_points)
        else:
            h = h * (n_points / 2) / sum_h
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            iteration = 0
        else:
            iteration += 1
            if iteration > 200:
                learning_rate *= 0.95
                iteration = 0

    # Re-evaluate to ensure consistency
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h, final_c5, n_points)
