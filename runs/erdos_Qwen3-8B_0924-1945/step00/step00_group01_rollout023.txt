import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = globals().get('initial_h_values', None)
    n_points = 50 if initial_h_values is None else len(initial_h_values)
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h_vals = np.array(initial_h_values, dtype=np.float64)
    else:
        h_vals = np.random.uniform(0, 1, n_points)
    h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    best_h_vals = h_vals.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = corr * dx
        current_c5 = np.max(c5)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_vals = h_vals.copy()
        
        # Focus on the worst lag
        k_max = np.argmax(corr)
        overlap_k = corr[k_max] * dx
        
        # Solve convex problem: minimize 0.5 * λ^2 + (overlap_k - λ)^2
        λ = cp.Variable()
        objective = 0.5 * λ**2 + (overlap_k - λ)**2
        prob = cp.Problem(cp.Minimize(objective))
        prob.solve()
        λ_val = λ.value
        
        # Update h_vals using gradient of the surrogate
        # For simplicity, we approximate the gradient
        # Use a small step to update
        step_size = 0.01
        h_vals = h_vals + step_size * (λ_val - (1 - h_vals))
        h_vals = np.clip(h_vals, 0, 1)
        mass = np.sum(h_vals)
        h_vals = h_vals * (n_points / 2) / mass
    
    # Final evaluation
    corr = np.correlate(best_h_vals, 1 - best_h_vals, mode='full')
    c5_bound = np.max(corr * dx)
    return (best_h_vals, c5_bound, n_points)
