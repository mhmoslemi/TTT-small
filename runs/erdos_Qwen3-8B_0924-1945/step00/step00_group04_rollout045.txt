import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize with provided initial_h_values or a random starting point
    if 'initial_h_values' in globals():
        initial_h_values = np.array(initial_h_values)
    else:
        initial_h_values = np.random.rand(n_points)
        initial_h_values = np.clip(initial_h_values, 0, 1)
        initial_h_values = initial_h_values * (target_sum / np.sum(initial_h_values))

    def clip_and_scale(h):
        h_clip = np.clip(h, 0, 1)
        h_scaled = h_clip * (target_sum / np.sum(h_clip))
        return h_scaled

    initial_h_values = clip_and_scale(initial_h_values)

    def evaluate_erdos(h):
        h = clip_and_scale(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr * dx)
        return c5

    def objective_function(h):
        return -evaluate_erdos(h)

    # Population-based differential evolution
    population_size = 50
    population = [np.random.rand(n_points) for _ in range(population_size)]
    population = [clip_and_scale(h) for h in population]
    population_fitness = [-evaluate_erdos(h) for h in population]
    best_h = population[np.argmax(population_fitness)]
    best_c5 = -np.max(population_fitness)
    best_iter = 0
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Mutation
        new_population = []
        for i in range(population_size):
            # Randomly select 3 indices for mutation
            indices = np.random.choice(n_points, 3, replace=False)
            mutant = population[i].copy()
            mutant[indices] += np.random.uniform(-0.05, 0.05, size=3)
            mutant = clip_and_scale(mutant)
            new_population.append(mutant)

        # Crossover and selection
        new_population_fitness = [objective_function(h) for h in new_population]
        combined_population = population + new_population
        combined_fitness = population_fitness + new_population_fitness
        combined_indices = np.argsort(combined_fitness)
        top_indices = combined_indices[:int(population_size * 0.7)]
        top_population = [combined_population[i] for i in top_indices]
        population = top_population
        population_fitness = [objective_function(h) for h in population]

        # Track best solution
        current_best_idx = np.argmax(population_fitness)
        current_best_h = population[current_best_idx]
        current_best_c5 = -population_fitness[current_best_idx]
        if abs(current_best_c5 - best_c5) < 1e-5:
            best_iter += 1
        else:
            best_c5 = current_best_c5
            best_h = current_best_h
            best_iter = 0
        if best_iter >= 50:
            break

    return (best_h, best_c5, n_points)
