import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 100
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = np.inf
    dx = 2.0 / n_points
    max_iter = 1000
    learning_rate = 0.1
    decay_rate = 0.995
    noise_scale = 0.05
    p_noise = 0.05
    early_stop = 100
    iter_count = 0
    start_time = time.time()
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return max(correlate_vals)
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if iter_count % 100 == 0:
            print(f"Iter {iter_count}, C5: {best_c5:.6f}")
        
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        max_c5_idx = np.argmax(correlate_vals)
        lag = max_c5_idx - n_points + 1
        gradients = np.zeros_like(h)
        for i in range(n_points):
            j = i - lag
            if 0 <= j < n_points:
                gradients[i] = (1 - h[j]) * dx
        
        h = h - learning_rate * gradients
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        if random.random() < p_noise:
            noise = np.random.normal(0, noise_scale, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        learning_rate *= decay_rate
        iter_count += 1
    
    h = best_h
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    c5 = compute_c5(h)
    return h, c5, n_points
