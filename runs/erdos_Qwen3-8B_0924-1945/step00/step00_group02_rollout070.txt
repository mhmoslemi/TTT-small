import numpy as np
import scipy.optimize as opt
import time
import random

# Pre-imported helpers
# evaluate_erdos_solution and initial_h_values are available

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Ensure initial values meet the integral constraint
    initial_sum = np.sum(initial_h_values)
    scale_factor = (n_points / 2) / initial_sum if initial_sum != 0 else 1.0
    h_values = initial_h_values * scale_factor
    
    # Clip values to [0, 1]
    h_values = np.clip(h_values, 0.0, 1.0)
    
    # Ensure sum constraint
    h_values /= np.sum(h_values) * 2.0 / n_points
    
    def evaluate_c5(h):
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2) / np.sum(h) if np.sum(h) != 0 else h
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def perturb(h):
        # Perturb 5 random indices
        indices = np.random.choice(n_points, size=5, replace=False)
        noise = np.random.normal(0, 0.1, 5)
        h = h.copy()
        for i, idx in enumerate(indices):
            h[idx] += noise[i]
            h[idx] = np.clip(h[idx], 0.0, 1.0)
        # Rescale to meet sum constraint
        sum_h = np.sum(h)
        if sum_h == 0:
            return h
        scale = (n_points / 2) / sum_h
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        return h
    
    best_h = h_values.copy()
    best_c5 = evaluate_c5(best_h)
    time_used = 0
    
    start_time = time.time()
    
    while time_used < budget_s:
        if time.time() - start_time >= budget_s:
            break
        h = perturb(best_h)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
        time_used = time.time() - start_time
    
    # Final check
    h_values = best_h
    sum_h = np.sum(h_values)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h_values = h_values * scale
    h_values = np.clip(h_values, 0.0, 1.0)
    h_values = h_values * (n_points / 2) / np.sum(h_values) if np.sum(h_values) != 0 else h_values
    h_values = np.clip(h_values, 0.0, 1.0)
    final_c5 = evaluate_c5(h_values)
    
    return h_values, final_c5, n_points
