import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    if initial_h_values is None:
        n_points = 100
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    else:
        n_points = len(initial_h_values)
        h_vals = np.array(initial_h_values, dtype=np.float64)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    
    dx = 2.0 / n_points
    max_iterations = 100
    no_improvement_threshold = 5
    best_c5 = np.inf
    best_h_vals = h_vals.copy()
    
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = corr * dx
        max_c5 = np.max(c5)
        
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h_vals = h_vals.copy()
        
        if max_c5 <= 0.3807:
            break
        
        max_idx = np.argmax(c5)
        k = max_idx
        
        def objective(x):
            h = x.copy()
            h[h < 0] = 0
            h[h > 1] = 1
            corr = np.correlate(h, 1 - h, mode='full')[k] * dx
            return corr
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            result = opt.minimize(
                objective,
                h_vals,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-6, 'maxiter': 20}
            )
            h_vals = result.x
            h_vals = np.clip(h_vals, 0, 1)
            norm_sum = np.sum(h_vals)
            h_vals = h_vals * (n_points / 2) / norm_sum
        except:
            h_vals = best_h_vals.copy()
    
    corr = np.correlate(best_h_vals, 1 - best_h_vals, mode='full')
    c5_bound = np.max(corr) * dx
    return (best_h_vals, c5_bound, n_points)
