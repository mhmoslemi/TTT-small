import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load("initial_h_values.npy") if 'initial_h_values' in globals() else None
    
    n_points = 500
    dx = 2.0 / n_points
    max_gen = 300
    pop_size = 50
    mutation_rate = 0.1
    mutation_noise = 0.05
    mutation_decay = 0.95
    stagnation_threshold = 50
    n_gen = 0
    best_c5 = float('inf')
    best_h = None
    
    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / total
        return h * scale
    
    def evaluate(h):
        nonlocal best_c5
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(overlaps)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    if initial_h_values is not None:
        initial_h_values = np.clip(initial_h_values, 0, 1)
        total = np.sum(initial_h_values)
        if total != 0:
            initial_h_values = initial_h_values * (n_points / 2) / total
        else:
            initial_h_values = np.zeros_like(initial_h_values)
    
    def evolve_population(population):
        new_population = []
        for _ in range(pop_size):
            parent1 = random.choice(population)
            parent2 = random.choice(population)
            child = np.copy(parent1)
            if np.random.random() < 0.5:
                crossover_point = np.random.randint(1, n_points)
                child[:crossover_point] = parent2[:crossover_point]
            child = clip_and_normalize(child + np.random.normal(0, mutation_noise, n_points))
            if np.random.random() < mutation_rate:
                i, j = np.random.choice(n_points, 2, replace=False)
                child[i], child[j] = child[j], child[i]
            new_population.append(child)
        return new_population
    
    population = [np.random.rand(n_points) for _ in range(pop_size)]
    population = [clip_and_normalize(h * (n_points / 2) / np.sum(h)) for h in population]
    population.append(clip_and_normalize(initial_h_values) if initial_h_values is not None else np.random.rand(n_points))
    
    start_time = time.time()
    
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        n_gen += 1
        current_population = []
        for h in population:
            current_population.append(h.copy())
        current_population = evolve_population(current_population)
        population = current_population
        current_c5 = evaluate(np.mean(population, axis=0))
        if n_gen % stagnation_threshold == 0:
            mutation_noise *= mutation_decay
        if best_c5 < 0.3807:
            break
    
    final_h = best_h
    final_c5 = best_c5
    return (final_h, final_c5, n_points)
