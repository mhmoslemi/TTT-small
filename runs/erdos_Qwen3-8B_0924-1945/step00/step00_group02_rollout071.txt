import numpy as np
import random
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Adjust initial_h_values to satisfy the integral constraint
    initial_h_values = np.clip(initial_h_values, 0.0, 1.0)
    sum_initial = np.sum(initial_h_values)
    if sum_initial != n_points / 2:
        scale = (n_points / 2) / sum_initial
        initial_h_values = np.clip(initial_h_values * scale, 0.0, 1.0)

    best_h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points, dx)
    time_start = time.time()

    for _ in range(300):
        elapsed = time.time() - time_start
        if elapsed > budget_s:
            break

        # Perturb random indices
        indices = np.random.choice(n_points, 5, replace=False)
        perturb = np.zeros(n_points)
        perturb[indices] = np.random.normal(0, 0.1, size=5)

        # Clip and rescale
        h = best_h + perturb
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h = h * scale
        else:
            h = np.zeros(n_points)
            h[np.random.choice(n_points, 1)] = 1.0

        # Evaluate C5
        try:
            c5 = evaluate_erdos_solution(h, n_points, dx)
        except:
            continue

        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    return (best_h, best_c5, n_points)

def evaluate_erdos_solution(h, n_points, dx):
    result = correlate(h, 1 - h, mode='full')
    result = result * dx
    return np.max(result)
