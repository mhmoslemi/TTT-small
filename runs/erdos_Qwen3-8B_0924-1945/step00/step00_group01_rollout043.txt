import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 100
    if initial_h_values is not None:
        n_points = len(initial_h_values)
    else:
        n_points = 100  # Default n_points
    h_vals = np.zeros(n_points)
    if initial_h_values is not None:
        h_vals = initial_h_values.copy()
    else:
        h_vals = np.random.rand(n_points)
    h_vals = np.minimum(np.maximum(h_vals, 0.0), 1.0)
    target_sum = n_points / 2
    h_vals = h_vals * target_sum / np.sum(h_vals)
    dx = 2.0 / n_points
    best_h = h_vals.copy()
    best_c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * dx
    start_time = time.time()
    for _ in range(100):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 100:
            break
        # Compute all lags
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        # Find the worst lag
        max_c = np.max(corr)
        max_lag = np.argmax(corr)
        # Define the objective function for the worst lag
        def objective(x):
            h = x.copy()
            h = np.minimum(np.maximum(h, 0.0), 1.0)
            h = h * target_sum / np.sum(h)
            overlap = corr[max_lag]  # Use the current value as initial guess
            # Recalculate the overlap for the worst lag
            h = h.reshape(-1)
            h_temp = h.copy()
            lag_index = max_lag
            if lag_index >= len(h_temp):
                lag_index -= len(h_temp)
            overlap = np.sum(h_temp * (1 - h_temp[(np.arange(len(h_temp)) + lag_index) % len(h_temp)])) * dx
            return overlap
        # Optimization with L-BFGS-B
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result = optimize.minimize(objective, h_vals, method='L-BFGS-B', bounds=bounds, constraints=constraints)
        if result.success:
            h_vals = result.x
        # Clip values and renormalize
        h_vals = np.clip(h_vals, 0.0, 1.0)
        h_vals = h_vals * target_sum / np.sum(h_vals)
        # Recompute C5
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_vals.copy()
        elif np.abs(current_c5 - best_c5) < 1e-6:
            # No improvement, break early
            break
    # Final validation
    h_vals = np.clip(best_h, 0.0, 1.0)
    h_vals = h_vals * target_sum / np.sum(h_vals)
    corr = np.correlate(h_vals, 1 - h_vals, mode='full')
    final_c5 = np.max(corr) * dx
    return (h_vals, final_c5, len(h_vals))
