import numpy as np
import scipy.optimize as opt
import sys

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    initial_h_values = np.array([0.343, 0.274, 0.274, 0.143, 0.286, 0.190, 0.224, 0.257, 0.224, 0.224, 0.257, 0.257, 0.224, 0.257, 0.224, 0.224, 0.257, 0.224, 0.257, 0.257, 0.257, 0.224, 0.257, 0.257, 0.224, 0.286, 0.224, 0.257, 0.224, 0.224, 0.257, 0.257, 0.224, 0.257, 0.257, 0.224, 0.257, 0.286, 0.286, 0.257, 0.286, 0.286, 0.257, 0.257, 0.286])
    target_sum = n_points / 2
    h = initial_h_values.copy()
    best_c5 = np.inf
    best_h = h.copy()
    
    def objective(x):
        nonlocal best_c5
        h = x.copy()
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - target_sum) > 1e-6:
            h = h * (target_sum / h_sum)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    def constraint(x):
        return np.sum(x) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial optimization
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        jac=None,
        tol=1e-6,
        options={'maxiter': 100}
    )
    
    if result.success:
        h = result.x
    else:
        h = initial_h_values.copy()
    
    # Final evaluation
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum - target_sum) > 1e-6:
        h = h * (target_sum / h_sum)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * (2.0 / n_points)
    
    return (h, c5, n_points)
