import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    h = initial_h.copy()
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        # Perturb 5 random indices
        indices = np.random.choice(n_points, 5, replace=False)
        noise = np.random.normal(0, 0.1, 5)
        h_perturbed = h.copy()
        for i in indices:
            h_perturbed[i] += noise[np.random.randint(5)]
            h_perturbed[i] = np.clip(h_perturbed[i], 0, 1)
        # Rescale to meet sum constraint
        sum_perturbed = np.sum(h_perturbed)
        if abs(sum_perturbed) < 1e-6:
            continue
        h_perturbed = h_perturbed * (n_points / 2) / sum_perturbed
        # Evaluate
        try:
            c5 = evaluate_erdos_solution(h_perturbed)
        except:
            c5 = np.inf
        if c5 < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = c5
    
    # Final check on best_h
    sum_best = np.sum(best_h)
    if abs(sum_best - n_points / 2) > 1e-6:
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    return (best_h, best_c5, n_points)
