import numpy as np
import scipy.signal
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        h = h * (n_points / 2.0) / total
    else:
        h = np.random.uniform(0.0, 1.0, n_points)
        total = np.sum(h)
        h = h * (n_points / 2.0) / total
    best_h = h.copy()
    best_c5 = float('inf')
    time_start = time.time()
    learning_rate = 0.01
    for _ in range(1000):
        if time.time() - time_start > budget_s:
            break
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        m = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i - m
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
        h -= learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        h = h * (n_points / 2.0) / total
    if 'initial_h_values' in globals():
        fallback_h = np.array(initial_h_values, dtype=np.float64)
        fallback_h = np.clip(fallback_h, 0.0, 1.0)
        total = np.sum(fallback_h)
        fallback_h = fallback_h * (n_points / 2.0) / total
        fallback_c5 = scipy.signal.correlate(fallback_h, 1 - fallback_h, mode='full')[np.argmax(scipy.signal.correlate(fallback_h, 1 - fallback_h, mode='full'))] * dx
    else:
        fallback_h = h.copy()
        fallback_c5 = scipy.signal.correlate(fallback_h, 1 - fallback_h, mode='full')[np.argmax(scipy.signal.correlate(fallback_h, 1 - fallback_h, mode='full'))] * dx
    h = best_h.copy()
    c5 = scipy.signal.correlate(h, 1 - h, mode='full')[np.argmax(scipy.signal.correlate(h, 1 - h, mode='full'))] * dx
    if c5 < fallback_c5:
        best_h = h.copy()
        best_c5 = c5
    h = best_h.copy()
    c5 = scipy.signal.correlate(h, 1 - h, mode='full')[np.argmax(scipy.signal.correlate(h, 1 - h, mode='full'))] * dx
    return (h.tolist(), c5, n_points)
