import numpy as np
import numpy.random as npr
import scipy.optimize as sco
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = 300
    dx = 2.0 / n_points
    initial_h_values = np.load('initial_h_values.npy')  # Assume this is loaded from parent construction
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    scale = (n_points / 2) / h_sum
    h *= scale
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    assert abs(h_sum - n_points / 2) < 1e-6

    def compute_c5(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx

    def evaluate_candidate(h_candidate):
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate_sum = np.sum(h_candidate)
        scale = (n_points / 2) / h_candidate_sum
        h_candidate *= scale
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate_sum = np.sum(h_candidate)
        assert abs(h_candidate_sum - n_points / 2) < 1e-6
        return compute_c5(h_candidate)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()

    for _ in range(1200):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        lag = np.argmax(corr)
        h_roll = np.roll(h, lag)
        overlap = h * (1 - h_roll)
        overlap_vals = overlap.flatten()
        top_indices = np.argsort(overlap_vals)[::-1][:int(len(overlap_vals)*0.1)]
        top_indices = np.unique(top_indices)

        candidates = []
        for i in top_indices:
            h_candidate = h.copy()
            h_candidate[i] = 1 - h_candidate[i]
            h_candidate_sum = np.sum(h_candidate)
            scale = (n_points / 2) / h_candidate_sum
            h_candidate *= scale
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate_sum = np.sum(h_candidate)
            assert abs(h_candidate_sum - n_points / 2) < 1e-6
            candidates.append(h_candidate)

        for i in top_indices:
            h_candidate = h.copy()
            h_candidate[i] += 0.05
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate_sum = np.sum(h_candidate)
            scale = (n_points / 2) / h_candidate_sum
            h_candidate *= scale
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate_sum = np.sum(h_candidate)
            assert abs(h_candidate_sum - n_points / 2) < 1e-6
            candidates.append(h_candidate)

        for i in top_indices:
            h_candidate = h.copy()
            h_candidate[i] -= 0.05
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate_sum = np.sum(h_candidate)
            scale = (n_points / 2) / h_candidate_sum
            h_candidate *= scale
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate_sum = np.sum(h_candidate)
            assert abs(h_candidate_sum - n_points / 2) < 1e-6
            candidates.append(h_candidate)

        for candidate in candidates:
            c5_candidate = evaluate_candidate(candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = candidate.copy()

        if best_c5 < compute_c5(h):
            h = best_h.copy()
        else:
            h = np.random.uniform(0, 1, n_points)
            h_sum = np.sum(h)
            scale = (n_points / 2) / h_sum
            h *= scale
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            assert abs(h_sum - n_points / 2) < 1e-6

    return (best_h, best_c5, n_points)
