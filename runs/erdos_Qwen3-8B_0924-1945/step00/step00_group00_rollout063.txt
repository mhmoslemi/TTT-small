import numpy as np
import scipy.signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        n_points = 45
        h = np.random.uniform(0, 1, n_points)
        h = (h * n_points / 2.0) / np.sum(h) * n_points / 2.0
        h = np.clip(h, 0, 1)
    
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    for _ in range(10000):
        if time.time() - start_time > budget_s:
            break
        
        # Compute correlate to find max C5
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        max_c5 = np.max(corr) * dx
        
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
        
        # Find worst lag
        lag_indices = np.argmax(corr)
        lag = lag_indices - (n_points - 1)
        
        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i - lag
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * dx
        
        # Update h
        h -= 0.01 * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h)
        
    # Final evaluation
    corr = scipy.signal.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    return best_h, final_c5, len(best_h)
