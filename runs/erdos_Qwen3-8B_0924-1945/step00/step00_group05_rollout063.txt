import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5 if i % 10 == 0 else 0.0 for i in range(n_points)])
    h = np.clip(initial_h_values, 0.0, 1.0)
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h if sum_h != 0 else h
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    learning_rate = 0.1
    iteration = 0
    start_time = time.time()

    def compute_overlap(h):
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        return overlaps

    while time.time() - start_time < budget_s:
        overlaps = compute_overlap(h)
        k_max = np.argmax(overlaps)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        h -= learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (n_points / 2) / sum_h
        else:
            h = h * (n_points / 2) / np.sum(np.clip(h, 1e-10, 1.0))
        new_c5 = evaluate_erdos_solution(h)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            iteration = 0
            learning_rate = 0.1
        else:
            iteration += 1
            if iteration > 200:
                learning_rate *= 0.95
                iteration = 0
        if np.abs(new_c5 - best_c5) < 1e-5:
            break

    h_values = best_h
    c5_bound = best_c5
    return h_values, c5_bound, n_points
