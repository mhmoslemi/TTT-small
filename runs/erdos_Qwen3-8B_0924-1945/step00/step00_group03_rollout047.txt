import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.sort(h)
        h = np.interp(h, np.linspace(0, 1, len(h)), np.linspace(0, 1, n_points))
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    start_time = time.time()
    
    def compute_c5(h):
        h_ = np.pad(h, (0, 1), 'constant')
        corr = np.correlate(h_, 1 - h, mode='full')
        return np.max(corr * dx)
    
    for _ in range(500):
        if time.time() - start_time > budget_s - 0.5:
            break
        # Perturb the solution
        perturb = np.random.rand(n_points) * 0.1 - 0.05
        perturb = np.clip(perturb, -0.1, 0.1)
        h_perturb = np.clip(h + perturb, 0, 1)
        h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
        
        # Solve convex relaxation
        h_var = cp.Variable(n_points, name='h')
        t = cp.Variable(name='t')
        constraints = [t >= 0]
        for k in range(n_points):
            idx = (np.arange(n_points) + k) % n_points
            term = h_var * (1 - h_var[idx])
            constraints.append(t >= cp.sum(term) * dx)
        constraints.append(h_var >= 0)
        constraints.append(h_var <= 1)
        constraints.append(cp.sum(h_var) == n_points / 2)
        obj = t + 1e-3 * cp.sum((h_var - 0.5) ** 2)
        prob = cp.Problem(cp.Minimize(obj), constraints)
        try:
            prob.solve(solver='ECOS', eps=1e-6)
        except:
            pass
        if prob.status == 'OPTIMAL':
            h = np.clip(h_var.value, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Gradient descent refinement
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h_refine = h.copy()
        for _ in range(10):
            c5 = compute_c5(h_refine)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_refine.copy()
            grad = np.zeros(n_points)
            for k in range(n_points):
                idx = (np.arange(n_points) + k) % n_points
                overlap = h_refine * (1 - h_refine[idx])
                grad += (overlap - np.mean(overlap)) * dx
            h_refine -= 0.01 * grad
            h_refine = np.clip(h_refine, 0, 1)
            h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
        
        if compute_c5(h_refine) < best_c5:
            best_c5 = compute_c5(h_refine)
            best_h = h_refine.copy()
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    exact_c5 = compute_c5(best_h)
    return (best_h, exact_c5, n_points)
