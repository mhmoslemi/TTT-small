import numpy as np
import scipy.signal as signal
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    max_lag = 2 * n_points - 1
    initial_h_values = np.random.rand(n_points)
    initial_h_values *= n_points / 2
    initial_h_values /= np.sum(initial_h_values)
    
    population_size = 50
    population = [initial_h_values.copy() for _ in range(population_size)]
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h, n_points, dx)
    best_time = 0
    time_remaining = budget_s
    
    for gen in range(500):
        start_time = time.time()
        # Evaluate all candidates
        fitness = [compute_c5(h, n_points, dx) for h in population]
        # Select top 20% for crossover
        top_indices = np.argsort(fitness)[:int(population_size * 0.2)]
        top_candidates = [population[i] for i in top_indices]
        # Crossover
        children = []
        for i in range(len(top_candidates) // 2):
            h1 = top_candidates[i]
            h2 = top_candidates[i + 1]
            child = (h1 + h2) / 2
            child = np.clip(child, 0, 1)
            child = rescale(child, n_points)
            children.append(child)
        # Mutation
        for h in population:
            indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
            h[indices] += np.random.uniform(-0.1, 0.1, size=len(indices))
            h = np.clip(h, 0, 1)
            h = rescale(h, n_points)
        # Replace population
        population = children + population
        # Update best
        for h in population:
            c5 = compute_c5(h, n_points, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        # Check time
        elapsed = time.time() - start_time
        time_remaining -= elapsed
        if time_remaining <= 0:
            break
    return (best_h, best_c5, n_points)

def compute_c5(h, n_points, dx):
    c = signal.correlate(h, 1 - h, mode='full')
    c = c * dx
    return np.max(c)

def rescale(h, n_points):
    total = np.sum(h)
    if abs(total - n_points / 2) < 1e-6:
        return h
    scale = (n_points / 2) / total
    return h * scale
