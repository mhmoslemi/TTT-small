import numpy as np
import random
import math
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 50
    dx = 2.0 / n_points

    # Initial setup
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        # Scale to satisfy integral constraint
        initial_sum = np.sum(initial_h)
        if initial_sum == 0:
            initial_h = np.random.rand(n_points)
            initial_sum = np.sum(initial_h)
        initial_h = initial_h * (n_points / 2) / initial_sum
    else:
        initial_h = np.random.rand(n_points)
        initial_sum = np.sum(initial_h)
        initial_h = initial_h * (n_points / 2) / initial_sum

    def fitness(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        norm = np.sum(h_vals)
        if not np.isclose(norm, n_points / 2, atol=1e-6):
            return float('inf')
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def evaluate_solution(h_vals):
        return fitness(h_vals)

    def crossover(parent1, parent2):
        point = np.random.randint(1, len(parent1) - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2

    def mutate(h_vals, mutation_rate=0.01):
        h_vals = h_vals.copy()
        for i in range(len(h_vals)):
            if np.random.rand() < mutation_rate:
                h_vals[i] += np.random.normal(0, 0.01)
                h_vals[i] = np.clip(h_vals[i], 0, 1)
        norm = np.sum(h_vals)
        if not np.isclose(norm, n_points / 2, atol=1e-6):
            h_vals = h_vals * (n_points / 2) / norm
        return h_vals

    def genetic_algorithm():
        population = [mutate(initial_h.copy()) for _ in range(50)]
        best_fitness = float('inf')
        best_solution = initial_h.copy()
        for gen in range(100):
            # Evaluate fitness
            fitnesses = [fitness(ind) for ind in population]
            # Track best
            if np.min(fitnesses) < best_fitness:
                best_fitness = np.min(fitnesses)
                best_solution = population[np.argmin(fitnesses)]
            if best_fitness <= 0.3807:
                break
            # Select parents
            parents = np.random.choice(population, size=20, replace=False)
            # Crossover
            offspring = []
            for i in range(0, 20, 2):
                p1, p2 = parents[i], parents[i+1]
                c1, c2 = crossover(p1, p2)
                offspring.append(c1)
                offspring.append(c2)
            # Mutate
            mutated = [mutate(ind) for ind in offspring]
            # Replace worst
            combined = population + mutated
            fitnesses = [fitness(ind) for ind in combined]
            sorted_indices = np.argsort(fitnesses)
            new_population = [combined[i] for i in sorted_indices[:50]]
            population = new_population
        return best_solution, best_fitness

    best_h, best_c5 = genetic_algorithm()
    return (best_h, best_c5, n_points)
