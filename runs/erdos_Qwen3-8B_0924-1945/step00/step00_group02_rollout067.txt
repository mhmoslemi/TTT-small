import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 450
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = 0.5 * (1 + np.random.uniform(-1, 1, n_points))
    h = np.clip(h, 0, 1)
    h = h * n_points / 2.0 / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    start_time = time.time()
    dt = 0.1
    
    for t in range(int(budget_s // dt)):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Targeted mutation: around max_idx
        idxs = np.arange(max_idx - 10, max_idx + 11)
        idxs = np.unique(np.clip(idxs, 0, len(h) - 1))
        selected = np.random.choice(idxs, size=10, replace=False)
        for i in selected:
            contrib = (1 - h[i]) * h[:len(h) - i] if i < len(h) else (1 - h[i]) * h
            contrib = contrib.sum() * 2 / n_points
            if contrib > 0:
                h[i] += -0.1 * (max_val - 0.3807) / (1 + t / 500)
            else:
                h[i] += 0.1 * (max_val - 0.3807) / (1 + t / 500)
        
        # Global mutation
        idxs = np.random.choice(range(len(h)), size=5, replace=False)
        for i in idxs:
            noise = np.random.normal(0, 0.2 * np.exp(-t / 300))
            h[i] += noise
        h = np.clip(h, 0, 1)
        h = h * n_points / 2.0 / np.sum(h)
        
        c5 = compute_c5(h, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Early termination if no improvement for 100 iterations
        if t % 100 == 0:
            if np.abs(best_c5 - compute_c5(h, n_points)) < 1e-6:
                break
    
    return (best_h, best_c5, n_points)

def compute_c5(h, n_points):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * 2.0 / n_points
