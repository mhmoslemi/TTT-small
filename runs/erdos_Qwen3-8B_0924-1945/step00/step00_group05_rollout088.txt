```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = 500
    dx = 2.0 / n_points
    initial_h_values = np.load('initial_h_values.npy')[:n_points]
    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    if sum_h == 0:
        h = np.ones(n_points) * 0.5
    else:
        h = h * (n_points / 2) / sum_h
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    learning_rate = 0.01
    momentum = 0.9
    decay_rate = 0.999
    for i in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(corr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        max_k = np.argmax(corr)
        grad = np.zeros(n_points)
        for j in range(n_points):
            k = max_k - j
            if 0 <= k < n_points:
                grad[j] += (1 - h[k]) * dx
            if 0 <= k + n_points < 2 * n_points:
                grad[j] -= h[k + n_points] * dx
        grad = grad / np.max(np.abs(grad)) if np.max(np.abs(grad)) > 1e-8 else grad
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.ones(n_points) * 0.5
        else:
            h = h * (n_points / 2) / sum_h
        learning_rate *= decay_rate
    if np.max(np.abs(grad)) < 1e-6:
        h = best_h
    corr = np.correlate(h, 1 - h, mode="full") * dx
    c5 = np.max(corr)
    return (h, c5, n_points)
```
