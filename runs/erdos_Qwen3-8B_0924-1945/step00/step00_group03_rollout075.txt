import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    initial_h_values = np.array([0.0] * n_points)
    
    # If initial_h_values is provided, use it directly
    if 'initial_h_values' in globals():
        initial_h_values = np.array(initial_h_values)
        initial_h_values = np.clip(initial_h_values, 0, 1)
        if not np.isclose(np.sum(initial_h_values) * dx, 1.0, atol=1e-6):
            raise ValueError("Initial h_values does not satisfy the integral constraint")
    
    # Convert initial h_values to a block-based representation
    def create_block_pattern(h_values, n_blocks):
        block_values = np.zeros(n_blocks, dtype=int)
        for i in range(n_blocks):
            start = i * n_points // n_blocks
            end = (i + 1) * n_points // n_blocks
            block_values[i] = np.mean(h_values[start:end])
            block_values[i] = 0 if block_values[i] < 0.5 else 1
        # Adjust sum to be n_blocks / 2
        while np.sum(block_values) != n_blocks // 2:
            for i in range(n_blocks):
                if block_values[i] == 0:
                    block_values[i] = 1
                    break
                else:
                    block_values[i] = 0
                    break
        return block_values
    
    def evaluate_c5(h_values):
        h_values = np.clip(h_values, 0, 1)
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(corr) * dx
    
    n_blocks = 15
    block_pattern = create_block_pattern(initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points), n_blocks)
    
    # Function to convert block pattern to h_values
    def block_to_h(block_pattern, n_points):
        block_len = n_points // n_blocks
        h_values = np.zeros(n_points)
        for i in range(n_blocks):
            start = i * block_len
            end = (i + 1) * block_len
            h_values[start:end] = block_pattern[i]
        return h_values
    
    # Local optimization using gradient descent
    def optimize_block_pattern(block_pattern, max_iter=500, lr=0.01):
        h_values = block_to_h(block_pattern, n_points)
        best_h = h_values.copy()
        best_c5 = evaluate_c5(best_h)
        best_pattern = block_pattern.copy()
        
        for _ in range(max_iter):
            # Perturb the block pattern
            for i in range(n_blocks):
                if np.random.rand() < 0.1:
                    if block_pattern[i] == 0:
                        block_pattern[i] = 1
                    else:
                        block_pattern[i] = 0
            # Normalize the pattern
            while np.sum(block_pattern) != n_blocks // 2:
                for i in range(n_blocks):
                    if block_pattern[i] == 0:
                        block_pattern[i] = 1
                        break
                    else:
                        block_pattern[i] = 0
                        break
            h_values = block_to_h(block_pattern, n_points)
            c5 = evaluate_c5(h_values)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_values.copy()
                best_pattern = block_pattern.copy()
            if best_c5 < 0.3807:
                break
        return best_h, best_c5, best_pattern
    
    start_time = time.time()
    best_h, best_c5, best_pattern = optimize_block_pattern(block_pattern, max_iter=500)
    
    # Apply local optimization with gradient descent
    def local_optimize(h_values, max_iter=500, lr=0.01):
        h_values = h_values.copy()
        for _ in range(max_iter):
            grad = np.zeros_like(h_values)
            corr = np.correlate(h_values, 1 - h_values, mode='full')
            for shift in range(len(corr) - 1):
                for i in range(n_points):
                    overlap = 0
                    for j in range(n_points):
                        if i + shift >= n_points or j + shift >= n_points:
                            continue
                        if h_values[i] * (1 - h_values[j]) > 0:
                            overlap += 1
                    grad[i] = -overlap
            h_values = np.clip(h_values - lr * grad, 0, 1)
            if np.sum(h_values) * dx < 0.99 or np.sum(h_values) * dx > 1.01:
                h_values = h_values * (1.0 / (np.sum(h_values) * dx))
            c5 = evaluate_c5(h_values)
            if c5 < best_c5:
                best_c5 = c5
            if best_c5 < 0.3807:
                break
        return h_values, best_c5
    
    best_h, best_c5 = local_optimize(best_h, max_iter=500)
    
    # Final checks
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    assert np.isclose(np.sum(best_h) * dx, 1.0, atol=1e-6)
    return best_h, best_c5, n_points
