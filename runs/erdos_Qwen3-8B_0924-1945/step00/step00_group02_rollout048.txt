```python
import numpy as np
import random
import time
import scipy.signal
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    n_points = 500
    initial_h_values = np.array([0.5 for _ in range(n_points)])
    
    if 'initial_h_values' in kwargs:
        initial_h_values = kwargs['initial_h_values']
    else:
        initial_h_values = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def normalize_h(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - target_sum) > 1e-7:
            scale = target_sum / total
            h = h * scale
        return h
    
    def compute_c5(h):
        try:
            corr = scipy.signal.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5
        except:
            return 1.0
    
    def evaluate(h):
        return compute_c5(h)
    
    def mutate(h, mutation_rate=0.1, mutation_strength=0.1):
        h = h.copy()
        indices = np.random.choice(len(h), int(len(h)*mutation_rate), replace=False)
        h[indices] += np.random.uniform(-mutation_strength, mutation_strength, size=len(indices))
        h = normalize_h(h)
        return h
    
    def crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        indices = np.random.choice(len(parent1), size=int(len(parent1)*0.5), replace=False)
        child[indices] = parent1[indices]
        child[~indices] = parent2[~indices]
        return normalize_h(child)
    
    pop_size = 50
    population = [mutate(initial_h_values, mutation_rate=0.15) for _ in range(pop_size)]
    best_h = population[0]
    best_c5 = evaluate(best_h)
    start_time = time.time()
    
    for generation in range(500):
        if time.time() - start_time > budget_s:
            break
        
        scores = [evaluate(ind) for ind in population]
        sorted_indices = np.argsort(scores)
        top_half = sorted_indices[:int(pop_size/2)]
        parents = [population[i] for i in top_half]
        
        new_population = []
        for _ in range(pop_size // 2):
            parent1, parent2 = random.choices(parents, k=2)
            child = crossover(parent1, parent2)
            new_population.append(mutate(child, mutation_rate=0.1))
        
        population = new_population
        current_scores = [evaluate(ind) for ind in population]
        best_idx = np.argmin(current_scores)
        current_best = population[best_idx]
        current_c5 = current_scores[best_idx]
        
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
        
        if generation % 10 == 0:
            print(f"Generation {generation}: Best C5 = {best_c5}")
    
    best_h = normalize_h(best_h)
    final_c5 = evaluate(best_h)
    return (best_h, final_c5, n_points)
```
