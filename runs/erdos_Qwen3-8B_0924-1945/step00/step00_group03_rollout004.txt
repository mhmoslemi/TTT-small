import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None  # Placeholder for parent construction

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70  # Start with the provided initial sample size
    dx = 2.0 / n_points
    h = np.copy(initial_h_values) if initial_h_values is not None else np.full(n_points, 1.0 / n_points)
    sum_h = np.sum(h)
    if not np.isclose(sum_h, n_points / 2, atol=1e-6):
        h += np.random.uniform(-0.1, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def evaluate_c5(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        return evaluate_c5(x)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    
    h_vals = result.x
    c5_bound = evaluate_c5(h_vals)
    return h_vals, c5_bound, n_points
