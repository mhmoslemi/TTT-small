import numpy as np
import scipy.optimize as opt
import time
import matplotlib.pyplot as plt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    h_values = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points)
    h_values = np.clip(h_values, 0, 1)
    h_values = h_values / np.sum(h_values) * (n_points / 2)
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    def objective(h):
        return compute_c5(h)
    
    def evaluate(h):
        return compute_c5(h)
    
    from scipy.optimize import Bounds
    bounds = Bounds(0, 1)
    
    def optimize():
        result = opt.minimize(objective, h_values, bounds=bounds, method='L-BFGS-B', options={'ftol': 1e-10})
        return result
    
    start_time = time.time()
    best_c5 = evaluate(h_values)
    best_h = h_values.copy()
    
    for _ in range(20):
        result = optimize()
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, n_points)
