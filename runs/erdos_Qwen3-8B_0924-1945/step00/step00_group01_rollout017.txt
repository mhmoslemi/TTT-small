import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = globals().get('initial_h_values', None)
    n_points = 500
    if initial_h_values is not None:
        h_vals = initial_h_values.copy()
    else:
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * n_points / 2) / np.sum(h_vals)
    dx = 2.0 / n_points
    max_iterations = 100
    improvement_threshold = 1e-5
    no_improvement_limit = 5
    best_h_vals = h_vals.copy()
    best_c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * dx
    start_time = time.time()
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5_values = corr * dx
        max_c5 = np.max(c5_values)
        if np.abs(max_c5 - best_c5) < improvement_threshold:
            no_improvement_count += 1
            if no_improvement_count >= no_improvement_limit:
                break
        else:
            no_improvement_count = 0
            best_c5 = max_c5
        max_index = np.argmax(c5_values)
        if max_index >= len(h_vals):
            max_index = len(h_vals) - 1
        def objective(x):
            h = x.copy()
            h[0] = h_vals[0]
            for i in range(1, n_points):
                h[i] = x[i]
            obj = corr[max_index] * dx
            return obj
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        result = opt.minimize(objective, h_vals, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if result.success:
            h_vals = result.x
            h_vals = np.clip(h_vals, 0.0, 1.0)
            sum_h = np.sum(h_vals)
            if np.abs(sum_h - n_points / 2) > 1e-6:
                h_vals = h_vals * (n_points / 2) / sum_h
        else:
            h_vals = best_h_vals.copy()
    corr_final = np.correlate(h_vals, 1 - h_vals, mode='full')
    c5_final = np.max(corr_final) * dx
    return (h_vals, c5_final, n_points)
