import numpy as np
import scipy.optimize as opt
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial_h_values if provided, otherwise construct a simple initial guess
    if 'initial_h_values' in globals():

    else:

        h[0::3] = 1.0
        h[1::3] = 0.5
        h[2::3] = 0.0
        h = np.clip(h, 0, 1)
        h /= np.sum(h) * dx
        h *= target_sum
        h = np.clip(h, 0, 1)
    
    # Compute initial C_5
    def evaluate_erdos_solution(h_flat):
        h = np.reshape(h_flat, (n_points,))
        cross_corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(cross_corr)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Optimization settings
    max_iter = 500
    step_size = 0.1
    epsilon = 1e-8
    early_stop_threshold = 1e-5
    early_stop_count = 0
    elapsed_time = 0
    start_time = opt.get_time()
    
    for i in range(max_iter):
        # Compute cross-correlation
        cross_corr = np.correlate(best_h, 1 - best_h, mode="full") * dx
        max_index = np.argmax(cross_corr)
        
        # Compute gradient for the max_index
        h_forward = best_h + epsilon
        h_backward = best_h - epsilon
        h_forward = np.clip(h_forward, 0, 1)
        h_forward = h_forward * (target_sum / np.sum(h_forward * dx)) if np.sum(h_forward * dx) != 0 else h_forward
        h_backward = np.clip(h_backward, 0, 1)
        h_backward = h_backward * (target_sum / np.sum(h_backward * dx)) if np.sum(h_backward * dx) != 0 else h_backward
        
        c5_forward = evaluate_erdos_solution(h_forward)
        c5_backward = evaluate_erdos_solution(h_backward)
        grad = (c5_forward - c5_backward) / (2 * epsilon)
        
        # Update h
        h_update = best_h - step_size * grad
        h_update = np.clip(h_update, 0, 1)
        h_update = h_update * (target_sum / np.sum(h_update * dx)) if np.sum(h_update * dx) != 0 else h_update
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_update)
        
        # Improve best solution
        if new_c5 < best_c5:
            best_h = h_update.copy()
            best_c5 = new_c5
            early_stop_count = 0
        else:
            early_stop_count += 1
        
        # Early stopping
        if early_stop_count >= 30:
            break
        
        # Update elapsed time
        elapsed_time = opt.get_time() - start_time
        if elapsed_time >= budget_s:
            break
        
        # Decay step size
        if i % 50 == 0 and i > 0:
            step_size *= 0.95
    
    return best_h, best_c5, n_points
