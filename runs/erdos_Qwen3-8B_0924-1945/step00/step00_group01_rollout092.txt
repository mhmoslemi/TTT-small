import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 50
    generations = 100
    mutation_rate = 0.05
    tournament_size = 5
    stagnation_threshold = 10
    
    if 'initial_h_values' in globals():
        initial = np.array(initial_h_values)
        initial_sum = np.sum(initial)
        h_vals = (initial * (n_points / 2)) / initial_sum
    else:
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if not np.isclose(total, n_points / 2, atol=1e-6):
            return float('inf')
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    best_h = h_vals.copy()
    best_c5 = evaluate(best_h)
    best_fitness = best_c5
    stagnation_count = 0
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        population = [h_vals.copy() for _ in range(pop_size)]
        
        for i in range(pop_size):
            h = population[i]
            if np.random.rand() < mutation_rate:
                indices = np.random.choice(n_points, size=1, replace=False)
                h[indices] += np.random.uniform(-0.1, 0.1)
                h = np.clip(h, 0, 1)
                h = (h * (n_points / 2)) / np.sum(h)
            population[i] = h
        
        population_fitness = [evaluate(h) for h in population]
        
        selected = []
        for _ in range(pop_size // 2):
            participants = np.random.choice(pop_size, tournament_size, replace=False)
            best_idx = np.argmin(population_fitness[participants])
            selected.append(population[participants[best_idx]])
        
        population = selected
        while len(population) < pop_size:
            h = np.random.rand(n_points)
            h = (h * (n_points / 2)) / np.sum(h)
            population.append(h)
        
        population_fitness = [evaluate(h) for h in population]
        population.sort(key=lambda x: x[1])
        population = [x[0] for x in population]
        
        current_best = population[0]
        current_fitness = evaluate(current_best)
        
        if current_fitness < best_fitness:
            best_fitness = current_fitness
            best_h = current_best
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        if stagnation_count >= stagnation_threshold:
            break
    
    final_h = best_h
    final_c5 = evaluate(final_h)
    
    return (final_h, final_c5, n_points)
