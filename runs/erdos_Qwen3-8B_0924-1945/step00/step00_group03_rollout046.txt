import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():

            if i % 2 == 0:
                h_values[i] = 1.0
        h_values = h_values[:n_points]

        h_values = h_values * (n_points / 2) / sum_h
    delta_x = 2 / n_points
    h = h_values.copy()
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * delta_x
    time_used = 0
    start_time = time.time()
    
    for _ in range(1000):
        time_used = time.time() - start_time
        if time_used >= budget_s:
            break
        x = np.arange(n_points) * delta_x
        h_correlate = np.correlate(h, 1 - h, mode='full')
        c5 = h_correlate * delta_x
        current_c5 = np.max(c5)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Local optimization
        def objective(x):
            h_new = x.reshape((n_points,))
            h_new = np.clip(h_new, 0, 1)
            sum_h_new = np.sum(h_new)
            if not np.isclose(sum_h_new, n_points / 2, atol=1e-6):
                h_new = h_new * (n_points / 2) / sum_h_new
            c5 = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * delta_x
            return c5
        # Gradient descent
        def gradient(x):
            h_new = x.reshape((n_points,))
            h_new = np.clip(h_new, 0, 1)
            sum_h_new = np.sum(h_new)
            if not np.isclose(sum_h_new, n_points / 2, atol=1e-6):
                h_new = h_new * (n_points / 2) / sum_h_new
            h_correlate = np.correlate(h_new, 1 - h_new, mode='full')
            c5 = h_correlate * delta_x
            max_index = np.argmax(c5)
            lag = (max_index - n_points + 1)
            if lag < 0:
                lag = 0
            if lag >= n_points:
                lag = n_points - 1
            corr = h_correlate[lag]
            grad = np.zeros(n_points)
            for i in range(n_points):
                for j in range(n_points):
                    if i + j >= n_points:
                        continue
                    overlap = 1 if (i + j) < n_points else 0
                    grad[i] += h_new[i] * (1 - h_new[j]) * overlap
            return grad * 100
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(objective, h, method='L-BFGS-B', jac=gradient, bounds=bounds, tol=1e-6, options={'maxiter': 50})

        if np.max(np.abs(h - best_h)) < 1e-6:
            break
        if best_c5 < 0.3807:

    sum_final_h = np.sum(final_h)
    if not np.isclose(sum_final_h, n_points / 2, atol=1e-6):
        final_h = final_h * (n_points / 2) / sum_final_h
    final_c5 = np.max(np.correlate(final_h, 1 - final_h, mode='full')) * delta_x
    return (final_h, final_c5, n_points)
