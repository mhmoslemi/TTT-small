import numpy as np
import random
import time
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    initial_h_values = kwargs.get('initial_h_values', None)
    
    # Define the number of points
    n_points = 500
    dx = 2.0 / n_points
    
    # Initialize h_vals based on initial_h_values if provided
    if initial_h_values is not None:
        h_vals = np.array(initial_h_values, dtype=np.float64)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
        h_vals = np.clip(h_vals, 0, 1)
    else:
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
        h_vals = np.clip(h_vals, 0, 1)
    
    def evaluate_c5(h_vals):
        h = h_vals
        ones_minus_h = 1 - h
        corr = np.correlate(h, ones_minus_h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def fitness(h_vals):
        return evaluate_c5(h_vals)
    
    def mutate(h_vals, mutation_rate=0.01, mutation_amp=0.05):
        h_vals = h_vals.copy()
        for i in range(len(h_vals)):
            if np.random.rand() < mutation_rate:
                h_vals[i] += np.random.normal(0, mutation_amp)
                h_vals[i] = np.clip(h_vals[i], 0, 1)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
        h_vals = np.clip(h_vals, 0, 1)
        return h_vals
    
    def crossover(parent1, parent2):
        point = np.random.randint(1, len(parent1) - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    population_size = 50
    population = [h_vals.copy() for _ in range(population_size)]
    best_fitness = float('inf')
    best_h = h_vals.copy()
    
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness_values = [fitness(ind) for ind in population]
        best_idx = np.argmin(fitness_values)
        current_best = fitness_values[best_idx]
        if current_best < best_fitness:
            best_fitness = current_best
            best_h = population[best_idx].copy()
        
        # Selection
        selected = np.random.choice(population_size, size=population_size, p=fitness_values / np.sum(fitness_values))
        
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            parent1 = population[selected[i]]
            parent2 = population[selected[i+1]]
            child1, child2 = crossover(parent1, parent2)
            offspring.append(child1)
            offspring.append(child2)
        
        # Mutation
        mutated_offspring = [mutate(ind) for ind in offspring]
        
        # Replacement
        population = mutated_offspring
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    return (best_h, final_c5, n_points)
