def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points_initial

    else:

        h = h * (n_points_initial / 2.0) / np.sum(h)
    
    h = np.clip(h, 0, 1)
    h = h * (n_points_initial / 2.0) / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
    t_start = time.time()
    
    for t in range(int(budget_s / 0.1)):
        if time.time() - t_start > budget_s:
            break
        indices = np.random.choice(n_points_initial, 5, replace=False)
        noise = np.random.normal(0, 0.05, n_points_initial)
        new_h = h + noise
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points_initial / 2.0) / np.sum(new_h)
        corr = np.correlate(new_h, 1 - new_h, mode="full")
        c5_new = np.max(corr) * dx
        if c5_new < best_c5:
            best_h = new_h.copy()
            best_c5 = c5_new
        elif np.random.rand() < np.exp((best_c5 - c5_new) / (100.0 * (1 - t / 600)**0.9)):
            best_h = new_h.copy()
            best_c5 = c5_new
