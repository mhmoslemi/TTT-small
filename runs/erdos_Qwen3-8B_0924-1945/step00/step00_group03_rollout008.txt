import numpy as np
import cvxpy as cp
import time

initial_h_values = np.array([...])  # Preloaded by the system

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    best_h_sum = np.sum(h)
    best_h = best_h / best_h_sum * (n_points / 2)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def evaluate_and_update(h):
        nonlocal best_c5, best_h
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    def optimize_with_convex_relaxation(h, dx):
        n = len(h)
        t = cp.Variable()
        h_vars = cp.Variable(n)
        objective = t + 1e-3 * cp.sum_squares(h_vars - 0.5)
        constraints = [t >= 0]
        for k in range(-n + 1, n):
            corr = 0
            for i in range(n):
                j = (i + k) % n
                corr += h_vars[i] * (1 - h_vars[j])
            constraints.append(t >= corr * dx)
        constraints += [0 <= h_vars, h_vars <= 1]
        constraints += [cp.sum(h_vars) == n / 2]
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver='ECOS', verbose=False, abstol=1e-6)
            if prob.status != 'OPTIMAL':
                return h
            return h_vars.value
        except:
            return h
    
    def refine_with_gradient_descent(h, dx, max_iter=100, lr=1e-2):
        h = h.copy()
        n = len(h)
        for _ in range(max_iter):
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n / 2)
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr)
            k = max_idx - n + 1
            if k < 0:
                k += n
            grad = np.zeros(n)
            for i in range(n):
                j = (i + k) % n
                grad[i] = (1 - h[j]) * dx
                grad[j] = -h[i] * dx
            h -= lr * grad
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n / 2)
        return h
    
    start_time = time.time()
    for _ in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        h = optimize_with_convex_relaxation(h, dx)
        h = refine_with_gradient_descent(h, dx)
        evaluate_and_update(h)
    
    h = best_h
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    final_c5 = compute_c5(h)
    return h, final_c5, n_points
