import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    h = initial_h_values
    dx = 2.0 / n_points
    h = np.array(h, dtype=np.float64)
    sum_h = np.sum(h)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        h = (h / sum_h) * (n_points / 2)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        conv = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(conv) * dx
        return c5

    def constraint_sum(h_flat):
        return np.sum(h_flat.reshape(n_points)) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(
        objective,
        h.flatten(),
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-5,
        options={'maxiter': 100}
    )

    if result.success and np.abs(result.fun - best_c5) < 1e-5:
        best_h = result.x.reshape(n_points)
        best_c5 = result.fun

    # Perturbation and re-optimization
    for _ in range(2):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = np.clip(h + perturbation, 0, 1)
        h_perturbed = (h_perturbed / np.sum(h_perturbed)) * (n_points / 2)
        result = opt.minimize(
            objective,
            h_perturbed.flatten(),
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            tol=1e-5,
            options={'maxiter': 100}
        )
        if result.success and result.fun < best_c5:
            best_h = result.x.reshape(n_points)
            best_c5 = result.fun

    # Final check and return
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        best_h = (best_h / sum_h) * (n_points / 2)
    best_c5 = compute_c5(best_h, dx)

    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv) * dx
