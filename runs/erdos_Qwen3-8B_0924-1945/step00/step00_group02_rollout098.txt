import numpy as np
import scipy.optimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    n_points = 500
    dx = 2.0 / n_points
    
    def eval_c5(h):
        h = np.array(h, dtype=np.float64)
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr * dx)
        return c5
    
    def perturb(h, mutation_rate=0.1, mutation_scale=0.1):
        h = h.copy()
        indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
        h[indices] += np.random.uniform(-mutation_scale, mutation_scale, size=len(indices))
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - n_points / 2) > 1e-6:
            h = h / total * (n_points / 2)
        return h
    
    def crossover(parent1, parent2):
        child = np.zeros(n_points)
        indices = np.random.choice(n_points, n_points // 2, replace=False)
        child[indices] = parent1[indices]
        child[~indices] = parent2[~indices]
        child = np.clip(child, 0, 1)
        total = np.sum(child)
        if abs(total - n_points / 2) > 1e-6:
            child = child / total * (n_points / 2)
        return child
    
    def generate_individual():
        h = np.random.uniform(0, 1, n_points)
        total = np.sum(h)
        h = h / total * (n_points / 2)
        return h
    
    def reproduce(population):
        next_gen = []
        for _ in range(len(population) // 2):
            p1, p2 = np.random.choice(population, 2, replace=False)
            child = crossover(p1, p2)
            next_gen.append(child)
        return next_gen
    
    def mutate(population, mutation_rate=0.1, mutation_scale=0.1):
        next_gen = []
        for h in population:
            next_gen.append(perturb(h, mutation_rate, mutation_scale))
        return next_gen
    
    def select(population, tournament_size=5):
        selected = []
        for _ in range(len(population)):
            candidates = np.random.choice(population, tournament_size, replace=False)
            best = min(candidates, key=eval_c5)
            selected.append(best)
        return selected
    
    def run_evolution(initial_population, max_iter=500, mutation_rate=0.1, mutation_scale=0.1):
        population = initial_population.copy()
        best_c5 = np.inf
        best_h = None
        start_time = time.time()
        
        for generation in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            population = select(population)
            population = reproduce(population)
            population = mutate(population, mutation_rate, mutation_scale)
            population.append(best_h)
            
            for h in population:
                current_c5 = eval_c5(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        
        return best_h, best_c5
    
    if 'initial_h_values' in globals():
        initial_population = [np.array(initial_h_values, dtype=np.float64)]
        for _ in range(49):
            initial_population.append(generate_individual())
    else:
        initial_population = [generate_individual() for _ in range(50)]
    
    best_h, best_c5 = run_evolution(initial_population, max_iter=500, mutation_rate=0.1, mutation_scale=0.1)
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h)
    if abs(total - n_points / 2) > 1e-6:
        best_h = best_h / total * (n_points / 2)
    
    return best_h.tolist(), best_c5, n_points
