import numpy as np
import math
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Assume this is available at runtime
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    start_time = time.time()
    
    max_iter = 500
    learning_rate = 0.1
    perturb_prob = 0.05
    tolerance = 1e-6
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
        
        cross_corr = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            if k < n_points:
                cross_corr[k] = dx * np.dot(h[:n_points - k], h[k:])
            else:
                k_wrap = k - n_points
                cross_corr[k] = dx * np.dot(h[:n_points - k_wrap], h[k_wrap:])
        
        max_k = np.argmax(cross_corr)
        grad = np.zeros(n_points)
        if max_k < n_points:
            for i in range(n_points - max_k):
                grad[i] += dx * h[i + max_k]
                grad[i + max_k] += dx * h[i]
        else:
            k_wrap = max_k - n_points
            for i in range(n_points - k_wrap):
                grad[i] += dx * h[i + k_wrap]
                grad[i + k_wrap] += dx * h[i]
        
        direction = -grad
        new_h = h + learning_rate * direction
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        scaling = (n_points / 2) / new_sum if new_sum != 0 else 1.0
        new_h = new_h * scaling
        new_h = np.clip(new_h, 0.0, 1.0)
        
        if np.random.rand() < perturb_prob:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            new_h = h + noise
            new_h = np.clip(new_h, 0.0, 1.0)
            new_sum = np.sum(new_h)
            scaling = (n_points / 2) / new_sum if new_sum != 0 else 1.0
            new_h = new_h * scaling
            new_h = np.clip(new_h, 0.0, 1.0)
        
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
        
        if np.abs(new_c5 - best_c5) < tolerance:
            break
    
    h = best_h
    assert (h >= 0).all() and (h <= 1).all()
    assert np.isclose(np.sum(h), n_points / 2, atol=1e-6)
    assert np.isfinite(best_c5)
    
    return h, best_c5, n_points
