import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(x):
        h_current = x
        h_shifted = np.roll(h_current, 1)
        corr = np.correlate(h_current, 1 - h_shifted, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = optimize.minimize(
        objective,
        h,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        method='SLSQP',
        tol=1e-6,
        options={'maxiter': 100}
    )

    h_best = result.x
    c5_best = objective(h_best)

    return h_best, c5_best, n_points
