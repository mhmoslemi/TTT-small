import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    initial_h_values = np.load('initial_h.npy') if 'initial_h_values' in kwargs else np.random.rand(n_points)
    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        h = (n_points / 2) * h / sum_h
    else:
        h = np.zeros(n_points)
        h[0] = 1
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    for _ in range(500):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        o = np.correlate(h, 1 - h, mode='full') * dx
        max_c5 = np.max(o)
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
        if max_c5 < 0.38085:
            break
        k = np.argmax(o)
        o_k = o[k]
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] += (1 - h[j]) * dx
            if i + k < n_points:
                grad[j] -= h[i] * dx
        h -= 0.01 * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.zeros(n_points)
            h[0] = 1
        else:
            h = (n_points / 2) * h / sum_h
    h = best_h
    sum_h = np.sum(h)
    if sum_h != 0:
        h = (n_points / 2) * h / sum_h
    else:
        h = np.zeros(n_points)
        h[0] = 1
    o = np.correlate(h, 1 - h, mode='full') * dx
    final_c5 = np.max(o)
    return (h.tolist(), final_c5, n_points)
