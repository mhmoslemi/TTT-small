```python
import numpy as np
import random
import time
import scipy.signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                h[i] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - n_points / 2) > 1e-6:
            return np.inf
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
        for i in indices:
            h[i] += np.random.uniform(-0.1, 0.1)
            h[i] = np.clip(h[i], 0, 1)
        total = np.sum(h)
        scale = (n_points / 2) / total
        h = h * scale
        return h
    
    def crossover(parent1, parent2):
        h = np.zeros(n_points)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                h[i] = parent1[i]
            else:
                h[i] = parent2[i]
        total = np.sum(h)
        scale = (n_points / 2) / total
        h = h * scale
        return h
    
    population_size = 50
    population = [h] * population_size
    best_h = h
    best_c5 = evaluate(h)
    start_time = time.time()
    
    for gen in range(1000):
        if time.time() - start_time > budget_s:
            break
        
        evaluations = [evaluate(ind) for ind in population]
        sorted_indices = np.argsort(evaluations)
        top_indices = sorted_indices[:int(population_size * 0.2)]
        
        new_population = []
        for i in range(population_size):
            if i < int(population_size * 0.2):
                parent1, parent2 = population[top_indices[0]], population[top_indices[1]]
                offspring = crossover(parent1, parent2)
                new_population.append(offspring)
            else:
                parent = population[sorted_indices[i]]
                offspring = mutate(parent)
                new_population.append(offspring)
        
        population = new_population
        current_best = min(evaluations)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[sorted_indices[0]]
    
    h_values = np.clip(best_h, 0, 1)
    return h_values, best_c5, n_points
```
