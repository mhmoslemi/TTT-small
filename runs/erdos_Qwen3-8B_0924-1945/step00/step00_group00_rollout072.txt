import numpy as np
import time
import scipy.optimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for balance between speed and accuracy
    h = np.zeros(n_points)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')

    start_time = time.time()

    for _ in range(500):  # Limit iterations to stay within budget
        if time.time() - start_time > budget_s:
            break

        # Compute the full correlation
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        current_c5 = np.max(correlate_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Find the worst lag
        worst_lag = np.argmax(correlate_vals)

        # Compute gradient for worst lag
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * dx

        # Update with momentum
        momentum = 0.9
        learning_rate = 0.1 * (1 - _ / 500)
        h = h - momentum * gradients * learning_rate

        # Clip and normalize
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

        # Add noise
        if _ % 100 == 0:
            noise_scale = 0.05
            h += noise_scale * np.random.normal(0, 1, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    # Final check
    dx = 2.0 / n_points
    correlate_vals = np.correlate(best_h, 1 - best_h, mode="full") * dx
    final_c5 = np.max(correlate_vals)
    return (best_h, final_c5, n_points)
