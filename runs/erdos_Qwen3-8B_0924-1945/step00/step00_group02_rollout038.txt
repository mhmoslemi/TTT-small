import numpy as np

import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    initial_h_values = kwargs.get('initial_h_values', None)
    
    # Initialize population
    population_size = 50
    population = []
    for _ in range(population_size):
        if initial_h_values is not None:
            h = initial_h_values.copy()
        else:
            h = np.random.rand(n_points)
        h = (h - np.min(h)) / (np.max(h) - np.min(h))  # Normalize to [0,1]
        h = h * (n_points / 2) / np.sum(h)  # Rescale to satisfy ∑h = n_points / 2
        population.append(h)
    
    best_h = population[0]
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    
    start_time = time.time()
    for gen in range(500):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate all candidates
        evaluations = []
        for h in population:
            try:
                corr = np.correlate(h, 1 - h, mode="full")
                c5 = np.max(corr) * (2.0 / n_points)
                evaluations.append((c5, h))
            except:
                evaluations.append((float('inf'), h))
        
        # Sort evaluations
        evaluations.sort()
        best_eval, best_h_candidate = evaluations[0]
        
        # Update best solution
        if best_eval < best_c5:
            best_c5 = best_eval
            best_h = best_h_candidate
        
        # Select top 20% for crossover
        top_indices = np.argsort(evaluations, axis=0)[:int(len(evaluations) * 0.2)]
        top_candidates = [evaluations[i][1] for i in top_indices]
        
        # Crossover
        new_population = []
        for i in range(len(top_candidates)):
            for j in range(i + 1, len(top_candidates)):
                h1 = top_candidates[i]
                h2 = top_candidates[j]
                h_offspring = (h1 + h2) / 2
                h_offspring = np.clip(h_offspring, 0, 1)
                h_offspring = h_offspring * (n_points / 2) / np.sum(h_offspring)
                new_population.append(h_offspring)
        
        # Mutation
        for h in population:
            if np.random.rand() < 0.1:
                indices = np.random.choice(n_points, 10, replace=False)
                h[indices] += np.random.uniform(-0.1, 0.1, size=10)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                new_population.append(h)
        
        # Replace old population
        population = new_population[:population_size]
    
    return (best_h, best_c5, n_points)
