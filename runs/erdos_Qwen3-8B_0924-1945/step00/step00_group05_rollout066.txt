import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    initial_h_values = np.load('initial_h.npy') if 'initial_h_values' in locals() else np.random.rand(n_points)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(initial_h_values)
    if sum_h != 0:
        initial_h_values = (n_points / 2) * initial_h_values / sum_h
    h = initial_h_values.copy()
    best_c5 = np.inf
    best_h = h.copy()
    max_iterations = 500
    decay_rate = 0.9995
    learning_rate = 0.05
    momentum = 0.9
    perturbation_rate = 0.05
    start_time = time.time()
    for iter in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(corr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Compute gradient for the maximum lag
        max_lag = np.argmax(corr)
        lag = max_lag - n_points + 1
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                grad[i] += 1 - h[j]
            j = i - lag
            if 0 <= j < n_points:
                grad[i] -= h[j]
        grad = grad * (1 / dx)
        # Update with momentum and learning rate
        h = h - learning_rate * grad
        # Clip values
        h = np.clip(h, 0, 1)
        # Scale to meet constraint
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (n_points / 2) / sum_h
        # Add random perturbation with some probability
        if np.random.rand() < perturbation_rate:
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                h = h * (n_points / 2) / sum_h
        learning_rate *= decay_rate
    # Final check
    corr = np.correlate(best_h, 1 - best_h, mode="full") * dx
    final_c5 = np.max(corr)
    return (best_h, final_c5, n_points)
