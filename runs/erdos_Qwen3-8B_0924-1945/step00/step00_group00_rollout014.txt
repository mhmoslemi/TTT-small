import numpy as np
import scipy.signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h = initial_h_values
    h = np.array(initial_h, dtype=np.float64)
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    h = h * (n_points / 2) / total if total != 0 else h

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)

    start_time = time.time()
    learning_rate = 0.01
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        correlate = scipy.signal.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(correlate)
        max_val = correlate[max_idx]
        if max_val <= best_c5:
            break
        lag = max_idx - len(h) + 1
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i - lag
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * 2  # scaled by 2 to match dx
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        h = h * (n_points / 2) / total if total != 0 else h
        current_c5 = evaluate_erdos_solution(h, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return best_h, best_c5, n_points
