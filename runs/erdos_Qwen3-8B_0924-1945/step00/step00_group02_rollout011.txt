import numpy as np
import random
import time
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 450
    dt = 0.1
    max_iter = int(budget_s / dt)
    initial_h_values = np.load("initial_h_values.npy")
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1
    h = h * scale
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = 1.0
    start_time = time.time()
    for t in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * 2.0 / n_points
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Targeted mutation
        max_k = np.argmax(corr)
        for i in range(max(0, max_k - 10), min(len(corr), max_k + 11)):
            if i >= len(h):
                continue
            contribution = h[i] * (1 - h[(i + k) % len(h)]) if (i + k) % len(h) < len(h) else 0
            grad = contribution
            delta = -grad * 0.1 / (1 + t / 500)
            h[i] += delta
            h[i] = np.clip(h[i], 0, 1)
        # Global mutation
        for _ in range(5):
            idx = np.random.randint(0, len(h))
            noise = np.random.normal(0, 0.2 * np.exp(-t / 300))
            h[idx] += noise
            h[idx] = np.clip(h[idx], 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1
        h = h * scale
        h = np.clip(h, 0, 1)
    return best_h, best_c5, n_points
