import numpy as np
import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    # Set seed
    random.seed(seed)
    np.random.seed(seed)
    
    # Load initial solution if available
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 45  # Start with 45 points for faster evaluation
    
    # Define parameters for PSO
    swarm_size = 50
    max_iter = 100
    inertia_weight = 0.7
    cognitive_coeff = 1.5
    social_coeff = 1.5
    vel_limit = 0.1
    
    # Initialize swarm
    if initial_h_values is not None:
        h_initial = initial_h_values.copy()
        h_initial = (h_initial - h_initial.min()) / (h_initial.max() - h_initial.min())
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        h_swarm = np.tile(h_initial[np.newaxis, :], (swarm_size, 1))
    else:
        h_swarm = np.random.uniform(0, 1, (swarm_size, n_points))
        h_swarm = h_swarm * (n_points / 2) / np.mean(np.sum(h_swarm, axis=1), axis=0)[:, np.newaxis]
    
    # Initialize velocities
    vel_swarm = np.random.uniform(-vel_limit, vel_limit, (swarm_size, n_points))
    
    # Personal best and global best
    pbest_swarm = h_swarm.copy()
    pbest_fitness = np.full(swarm_size, np.inf)
    gbest = h_swarm[0].copy()
    gbest_fitness = np.inf
    
    # Evaluate initial fitness
    def compute_c5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    for i in range(swarm_size):
        pbest_fitness[i] = compute_c5(h_swarm[i])
    
    gbest_fitness = np.min(pbest_fitness)
    gbest = h_swarm[np.argmin(pbest_fitness)].copy()
    
    # Main PSO loop
    start_time = time.time()
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Update velocities and positions
        for j in range(swarm_size):
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            vel_swarm[j] = (inertia_weight * vel_swarm[j] +
                            cognitive_coeff * r1 * (pbest_swarm[j] - h_swarm[j]) +
                            social_coeff * r2 * (gbest - h_swarm[j]))
            vel_swarm[j] = np.clip(vel_swarm[j], -vel_limit, vel_limit)
            
            # Update position
            h_swarm[j] += vel_swarm[j]
            h_swarm[j] = np.clip(h_swarm[j], 0, 1)
            h_swarm[j] = h_swarm[j] * (n_points / 2) / np.sum(h_swarm[j])
        
        # Evaluate fitness
        for j in range(swarm_size):
            current_c5 = compute_c5(h_swarm[j])
            if current_c5 < pbest_fitness[j]:
                pbest_swarm[j] = h_swarm[j].copy()
                pbest_fitness[j] = current_c5
        
        # Update global best
        current_best_idx = np.argmin(pbest_fitness)
        if pbest_fitness[current_best_idx] < gbest_fitness:
            gbest = pbest_swarm[current_best_idx].copy()
            gbest_fitness = pbest_fitness[current_best_idx]
        
        # Introduce local search on top performers
        top_indices = np.argsort(pbest_fitness)[:5]
        for j in top_indices:
            perturbation = np.random.normal(0, 0.05, n_points)
            h_swarm[j] += perturbation
            h_swarm[j] = np.clip(h_swarm[j], 0, 1)
            h_swarm[j] = h_swarm[j] * (n_points / 2) / np.sum(h_swarm[j])
    
    # Final evaluation
    final_c5 = compute_c5(gbest)
    
    # Ensure constraints are satisfied
    if np.any(gbest < 0) or np.any(gbest > 1):
        gbest = np.clip(gbest, 0, 1)
        gbest = gbest * (n_points / 2) / np.sum(gbest)
    
    # Final check
    if np.abs(np.sum(gbest) - n_points / 2) > 1e-6:
        gbest = gbest * (n_points / 2) / np.sum(gbest)
    
    return (gbest.tolist(), final_c5, n_points)
