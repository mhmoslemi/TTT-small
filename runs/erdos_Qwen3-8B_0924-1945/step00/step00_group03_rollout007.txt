import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    sum_h = np.sum(initial_h)
    if not np.isclose(sum_h, n_points / 2, atol=1e-6):
        initial_h = initial_h * (n_points / 2) / sum_h
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    elapsed = 0

    def compute_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        scale = (n_points / 2) / np.sum(h)
        h = h * scale
        return h

    def optimize_with_convex_relaxation(h0, lambda_penalty=1e-3, max_iter=10, tol=1e-6):
        h = h0.copy()
        h = clip_and_normalize(h)
        for _ in range(max_iter):
            t = cp.Variable()
            h_var = cp.Variable(h.shape, bounds=(0, 1))
            constraints = []
            for k in range(1, n_points):
                if k >= n_points:
                    k_mod = k - n_points
                else:
                    k_mod = k
                corr = np.zeros(n_points + k)
                corr[k:] = h_var[:n_points - k] * (1 - h_var[k:])
                corr = np.sum(corr[:n_points]) * dx
                constraints.append(t >= corr)
            objective = t + lambda_penalty * cp.sum_squares(h_var - 0.5)
            prob = cp.Problem(cp.Minimize(objective), constraints)
            try:
                prob.solve(solver='ECOS', eps=tol)
                if prob.status != 'OPTIMAL':
                    break
                h = np.array(h_var.value).flatten()
                h = clip_and_normalize(h)
                c5 = compute_c5(h)
                if c5 < best_c5:
                    nonlocal best_h, best_c5
                    best_h = h.copy()
                    best_c5 = c5
            except:
                break
        return h

    best_h = optimize_with_convex_relaxation(best_h)
    best_c5 = compute_c5(best_h)

    def gradient_refinement(h, max_steps=100, lr=1e-2):
        h = h.copy()
        for _ in range(max_steps):
            c5 = compute_c5(h)
            if c5 < best_c5:
                nonlocal best_h, best_c5
                best_h = h.copy()
                best_c5 = c5
            grad = np.zeros_like(h)
            for k in range(1, n_points):
                if k >= n_points:
                    k_mod = k - n_points
                else:
                    k_mod = k
                corr = np.zeros(n_points + k)
                corr[k:] = h[:n_points - k] * (1 - h[k:])
                corr = np.sum(corr[:n_points]) * dx
                grad += 2 * (np.correlate(h, 1 - h, mode='full')[k] * dx) * (h - 0.5)
            h = h - lr * grad
            h = clip_and_normalize(h)
            if np.isnan(h).any():
                h = np.random.rand(n_points) * 0.5
                h = clip_and_normalize(h)
        return h

    best_h = gradient_refinement(best_h)

    def perturb_and_refine(h, num_perturb=5, perturb_scale=0.1, max_iter=50):
        for _ in range(num_perturb):
            perturbed = h + perturb_scale * np.random.randn(n_points)
            perturbed = clip_and_normalize(perturbed)
            h = gradient_refinement(perturbed, max_steps=max_iter)
            c5 = compute_c5(h)
            if c5 < best_c5:
                nonlocal best_h, best_c5
                best_h = h.copy()
                best_c5 = c5
            if best_c5 < 0.3807:
                break
        return h

    best_h = perturb_and_refine(best_h)

    best_h = clip_and_normalize(best_h)
    final_c5 = compute_c5(best_h)

    if elapsed > budget_s - 5:
        return (best_h, final_c5, n_points)
    return (best_h, final_c5, n_points)
