import numpy as np
import cvxpy as cp
import time

initial_h_values = np.array([0.1839, 0.6678, 0.5229, 0.6078, 0.9291, 0.7091, 0.5728, 0.5897, 0.6986, 0.8461,
                             0.8042, 0.7866, 0.6444, 0.5486, 0.8356, 0.8723, 0.6124, 0.5863, 0.8648, 0.8424,
                             0.5401, 0.7027, 0.5542, 0.6219, 0.5842, 0.7667, 0.6319, 0.6709, 0.6968, 0.6268,
                             0.7681, 0.6370, 0.5855, 0.5205, 0.5629, 0.5333, 0.6009, 0.6956, 0.6578, 0.5967,
                             0.6183, 0.6139, 0.6453, 0.7468, 0.5613, 0.6735, 0.6916, 0.7220, 0.6346, 0.6946,
                             0.6436, 0.6593, 0.5874, 0.6811])
n_points = len(initial_h_values)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    h_vals = initial_h_values.copy()
    n_points = len(h_vals)
    dx = 2.0 / n_points
    total_mass = n_points / 2
    h_vals = (h_vals * total_mass) / np.sum(h_vals)
    
    max_iterations = 200
    improvement_threshold = 1e-6
    best_h_vals = h_vals.copy()
    best_c5_bound = np.inf
    
    start_time = time.time()
    
    for _ in range(max_iterations):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        corr_scaled = corr * dx
        current_c5 = np.max(corr_scaled)
        
        if current_c5 < best_c5_bound:
            best_c5_bound = current_c5
            best_h_vals = h_vals.copy()
        
        if time.time() - start_time > budget_s - 10:
            break
        
        k_max = np.argmax(corr_scaled)
        overlap_k_max = corr_scaled[k_max]
        
        lambda_var = cp.Variable()
        objective = 0.5 * lambda_var**2 + (overlap_k_max - lambda_var)**2
        constraints = [lambda_var >= 0]
        prob = cp.Problem(cp.Minimize(objective), constraints)
        prob.solve()
        
        if prob.status != 'OPTIMAL':
            continue
        
        lambda_opt = lambda_var.value
        grad = (1 - 2 * lambda_opt) * (1 - h_vals) * h_vals
        h_vals += grad * 0.1
        
        h_vals = np.clip(h_vals, 0, 1)
        mass = np.sum(h_vals)
        h_vals = h_vals * (total_mass) / mass
        
        if np.allclose(np.sum(h_vals), total_mass, atol=1e-6):
            pass
        
        if np.max(np.abs(h_vals - best_h_vals)) < improvement_threshold:
            break
    
    corr = np.correlate(best_h_vals, 1 - best_h_vals, mode='full')
    corr_scaled = corr * dx
    final_c5_bound = np.max(corr_scaled)
    
    return best_h_vals, final_c5_bound, n_points
