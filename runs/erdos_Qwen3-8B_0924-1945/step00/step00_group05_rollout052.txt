        # Compute overlap for all lags
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        k_max = np.argmax(overlaps)

        grad = np.zeros_like(h)

            if 0 <= j < n_points:

        # Update h
        h -= learning_rate * grad
        
        # Clip and normalize
        h = np.clip(h, 0, 1)

        if iteration > 200:
