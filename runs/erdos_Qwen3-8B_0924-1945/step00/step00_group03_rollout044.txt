import numpy as np
import random
import time
import scipy.signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in globals() else np.random.rand(70)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.copy(initial_h_values)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    t_total = int(budget_s * 0.75)
    T0 = max(1.0, 1.0 / (1.0 + best_c5 * 2.0)) * 0.5
    Tmax = int(0.1 * t_total)
    beta = 0.75
    gamma = 0.65
    lambda0 = 0.15 / dx
    start_time = time.time()
    
    for t in range(t_total):
        if time.time() - start_time > budget_s:
            break
        if t % 100 == 0:
            lambda0 *= 0.8 if t < t_total // 2 else 0.95
        
        noise = np.random.normal(0, lambda0, size=n_points)
        perturbed = h + noise
        perturbed = np.clip(perturbed, 0.0, 1.0)
        
        current_sum = np.sum(perturbed)
        delta = (n_points / 2 - current_sum) / (n_points * 0.95)
        perturbed += delta * np.random.choice([-1, 1], size=n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        perturbed = perturbed * (n_points / 2 / np.sum(perturbed)) if np.sum(perturbed) != 0 else perturbed
        
        new_c5 = evaluate_erdos_solution(perturbed, dx)
        
        if new_c5 < best_c5 or (np.random.rand() < np.exp(-(new_c5 - best_c5) / (T0 * (1 - t / Tmax) ** (1 / beta)))):
            h = perturbed.copy()
            best_h = h.copy()
            best_c5 = new_c5
            
        if best_c5 < 0.3807:
            break
    
    final_h = np.clip(h, 0.0, 1.0)
    final_sum = np.sum(final_h)
    final_h = final_h * (n_points / 2) / final_sum if final_sum != 0 else final_h
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    assert np.all(final_h >= 0) and np.all(final_h <= 1)
    assert np.isclose(np.sum(final_h), n_points / 2, atol=1e-6)
    assert np.isfinite(final_c5)
    
    return (final_h, final_c5, n_points)
