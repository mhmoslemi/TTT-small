import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', None)
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    def compute_correlation(h):
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return corr
    
    def update_h(h, worst_lag, learning_rate, noise_scale):
        gradients = np.zeros(n_points)
        for i in range(n_points):
            j = (i - worst_lag) % n_points
            gradients[i] = (1 - h[j]) * dx
        h -= learning_rate * gradients
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h += noise_scale * np.random.normal(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = compute_correlation(h)
        worst_lag = np.argmax(corr)
        if corr[worst_lag] < best_c5:
            best_c5 = corr[worst_lag]
            best_h = h.copy()
        learning_rate = 0.05 * (1 - _ / 500)
        noise_scale = 0.05
        h = update_h(h, worst_lag, learning_rate, noise_scale)
    
    final_corr = compute_correlation(best_h)
    best_c5 = np.max(final_corr)
    return (best_h, best_c5, n_points)
