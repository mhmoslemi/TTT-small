import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 47 if 'initial_h_values' in globals() else 53
    if 'initial_h_values' in globals():
        h_vals = np.array(initial_h_values, dtype=np.float64)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    else:
        h_vals = np.random.rand(n_points)
        h_vals = (h_vals * (n_points / 2)) / np.sum(h_vals)
    
    dx = 2.0 / n_points
    max_iterations = 100
    improvement_threshold = 0.001
    best_c5 = np.inf
    best_h_vals = h_vals.copy()
    
    start_time = time.time()
    for i in range(max_iterations):
        if time.time() - start_time > budget_s - 100:
            break
        
        # Compute all correlations
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5_values = corr * dx
        max_c5 = np.max(c5_values)
        
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h_vals = h_vals.copy()
            if abs(best_c5 - 0.3807) < 0.0001:
                break
        
        # Pick a random lag to optimize
        if np.any(c5_values > 0):
            lag_idx = np.random.choice(np.where(c5_values > 0)[0])
            lag = lag_idx - (n_points - 1)
        else:
            lag = 0
        
        # Define the objective function for this lag
        def objective(x):
            h = x.copy()
            h[0] = h_vals[0]
            h[-1] = h_vals[-1]
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            if not np.isclose(total, n_points / 2):
                h = (h / total) * (n_points / 2)
            val = 0
            for i in range(n_points):
                j = (i + lag) % n_points
                val += h[i] * (1 - h[j])
            return val
        
        # Define constraints
        def constraint(x):
            return np.sum(x) - n_points / 2
        
        # Bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Initial guess
        x0 = h_vals.copy()
        
        # Optimize
        try:
            result = opt.minimize(
                objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint}]
            )
            h_vals = result.x
        except:
            h_vals = np.clip(h_vals, 0, 1)
            total = np.sum(h_vals)
            if not np.isclose(total, n_points / 2):
                h_vals = (h_vals / total) * (n_points / 2)
    
    # Final evaluation
    corr = np.correlate(best_h_vals, 1 - best_h_vals, mode='full')
    final_c5 = np.max(corr * dx)
    return (best_h_vals, final_c5, n_points)
