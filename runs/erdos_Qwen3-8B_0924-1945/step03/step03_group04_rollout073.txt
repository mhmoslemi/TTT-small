```python
import numpy as np
import time
import scipy.optimize as opt

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    T_initial = 1.0
    T_final = 1e-6
    T_max = 200
    T_current = T_initial
    T_schedule = T_initial * (1 - np.linspace(0, 1, T_max))

    for t in range(T_max):
        if time.time() - start_time > budget_s - 5:
            break
        if t % 10 == 0:
            if np.random.rand() < 0.5:
                T_initial *= 0.5
        indices = np.random.choice(n_points, size=int(n_points * 0.2), replace=False)
        noise = np.random.normal(0, 0.05, size=len(indices))
        h_perturb = h.copy()
        h_perturb[indices] += noise
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        sum_perturb = np.sum(h_perturb)
        scale_perturb = target_sum / sum_perturb if sum_perturb != 0 else 1.0
        h_perturb = h_perturb * scale_perturb
        c5_perturb = objective(h_perturb.flatten())
        if c5_perturb < best_c5:
            best_c5 = c5_perturb
            best_h = h_perturb.copy()
        else:
            delta = c5_perturb - best_c5
            if np.random.rand() < np.exp(-delta / T_current):
                best_c5 = c5_perturb
                best_h = h_perturb.copy()
        T_current = T_schedule[t]
        if t % 100 == 0:
            if np.random.rand() < 0.1:
                best_h = np.random.uniform(0, 1, n_points)
                sum_h = np.sum(best_h)
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                best_h = best_h * scale
                c5 = objective(best_h.flatten())
                best_c5 = c5

    h_best = best_h.copy()
    sum_h_best = np.sum(h_best)
    scale = target_sum / sum_h_best if sum_h_best != 0 else 1.0
    h_best = h_best * scale
    result = opt.minimize(
        objective, h_best.flatten(), 
        method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 50}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt.flatten())
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()
    return best_h.tolist(), best_c5, n_points
```
