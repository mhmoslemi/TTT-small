import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        h[:n_points // 2] = 1.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()

    def global_search():
        nonlocal best_h, best_c5
        pop_size = 50
        pop = np.random.uniform(0.0, 1.0, (pop_size, n_points))
        for i in range(100):
            if time.time() - start_time > budget_s - 5:
                break
            for j in range(pop_size):
                a, b, c = np.random.choice(pop_size, 3, replace=False)
                mutant = pop[a] + 0.5 * (pop[b] - pop[c])
                mutant = np.clip(mutant, 0.0, 1.0)
                sum_mutant = np.sum(mutant)
                scale = target_sum / sum_mutant if sum_mutant != 0 else 1.0
                mutant = mutant * scale
                corr = np.correlate(mutant, 1.0 - mutant, mode="full")
                c5 = np.max(corr * dx)
                if c5 < objective(pop[j].flatten()):
                    pop[j] = mutant
            for j in range(pop_size):
                h_opt = pop[j].copy()
                sum_h_opt = np.sum(h_opt)
                scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
                h_opt = h_opt * scale
                c5_opt = objective(h_opt.flatten())
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()

    global_search()

    def local_refinement():
        nonlocal best_h, best_c5
        result = opt.minimize(
            objective, best_h.flatten(), method='L-BFGS-B',
            bounds=bounds, constraints=cons, tol=1e-8,
            options={'maxiter': 100}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_opt = objective(h_opt.flatten())
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    local_refinement()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), method='L-BFGS-B',
            bounds=bounds, constraints=cons, tol=1e-8,
            options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_opt = objective(h_opt.flatten())
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
