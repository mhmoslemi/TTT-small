import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()

    def differential_evolution(pop_size, max_gen):
        pop = np.random.uniform(0, 1, (pop_size, n_points))
        pop = (pop * target_sum).astype(np.float64)
        pop /= np.sum(pop, axis=1, keepdims=True)
        best_pop = pop.copy()
        best_c5 = float('inf')

        for gen in range(max_gen):
            for i in range(pop_size):
                a, b, c = np.random.choice(pop_size, 3, replace=False)
                mutant = pop[a] + 0.85 * (pop[b] - pop[c])
                mutant = np.clip(mutant, 0, 1)
                mutant = (mutant * target_sum) / np.sum(mutant)
                mutant = mutant.astype(np.float64)
                c5 = objective(mutant.flatten())
                if c5 < best_c5:
                    best_c5 = c5
                    best_pop = mutant.copy()
            if time.time() - start_time > budget_s - 10:
                break
        return best_pop, best_c5

    # Initial global search
    pop_size = 50
    max_gen = 250
    h_opt, best_c5 = differential_evolution(pop_size, max_gen)

    # Local refinement
    result = opt.minimize(
        objective, h_opt.flatten(), 
        method='SLSQP', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 150}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt.flatten())
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        # Additional local refinement with restarts
        for _ in range(3):
            result = opt.minimize(
                objective, best_h.flatten(), 
                method='L-BFGS-B', bounds=bounds, constraints=cons,
                tol=1e-8, options={'maxiter': 50}
            )
            h_opt = result.x.reshape(n_points)
            h_opt = np.clip(h_opt, 0.0, 1.0)
            sum_h_opt = np.sum(h_opt)
            scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
            h_opt = h_opt * scale
            c5_bound = objective(h_opt.flatten())
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h = h_opt.copy()
            if time.time() - start_time > budget_s - 10:
                break

    return best_h.tolist(), best_c5, n_points
