import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iterations = 200
    improvement_threshold = 1e-5
    noise_scale = 0.05

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 5 == 0:
                h[i] = 1.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    current_h = h.copy()
    best_k = 0
    last_improvement = 0

    start_time = time.time()

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
        corr = np.correlate(current_h, 1.0 - current_h, mode="full")
        k_star = np.argmax(corr)
        k_star = min(k_star, len(corr) - 1)
        c5_current = corr[k_star] * dx
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = current_h.copy()
            best_k = k_star
            last_improvement = 0
        else:
            last_improvement += 1
            if last_improvement >= 5:
                break

        def sub_objective(h_flat):
            h = h_flat.reshape(n_points)
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            corr = np.correlate(h, 1.0 - h, mode="full")
            return corr[k_star] * dx

        result = opt.minimize(
            sub_objective, current_h.flatten(), 
            method='SLSQP', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 100}
        )
        current_h = result.x.reshape(n_points)
        current_h = np.clip(current_h, 0.0, 1.0)
        sum_h_opt = np.sum(current_h)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        current_h = current_h * scale

        if np.random.rand() < 0.1:
            noise = np.random.uniform(-noise_scale, noise_scale, size=n_points)
            current_h += noise
            current_h = np.clip(current_h, 0.0, 1.0)
            sum_h = np.sum(current_h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            current_h = current_h * scale

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
