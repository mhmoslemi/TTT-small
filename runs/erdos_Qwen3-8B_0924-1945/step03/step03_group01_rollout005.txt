import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # Genetic algorithm setup
    POP_SIZE = 50
    MAX_GEN = 50
    MUT_STD = 0.05
    MUT_RATE = 0.2

    def generate_population(base_h):
        pop = []
        for _ in range(POP_SIZE):
            h = base_h.copy()
            noise = np.random.normal(0, MUT_STD, size=n_points)
            h += noise
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            pop.append(h)
        return np.array(pop)

    pop = generate_population(h)

    for gen in range(MAX_GEN):
        # Tournament selection
        tournament = np.random.choice(POP_SIZE, size=5, replace=False)
        parents = pop[tournament]
        parent1, parent2 = parents[np.random.choice(5, size=2, replace=False)]

        # Crossover
        cross_point = np.random.choice(n_points, size=2, replace=False)
        child = np.copy(parent1)
        child[cross_point[0]:cross_point[1]] = parent2[cross_point[0]:cross_point[1]]

        # Mutation
        if np.random.rand() < MUT_RATE:
            noise = np.random.normal(0, MUT_STD, size=n_points)
            child += noise
            child = np.clip(child, 0.0, 1.0)
            sum_h = np.sum(child)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            child = child * scale

        # Clip and scale
        child = np.clip(child, 0.0, 1.0)
        sum_h = np.sum(child)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        child = child * scale

        # Evaluate and replace
        child_c5 = objective(child.flatten())
        if child_c5 < best_c5:
            best_c5 = child_c5
            best_h = child.copy()
        # Replace worst 10%
        indices = np.argsort([objective(p.flatten()) for p in pop])
        pop[indices[-int(POP_SIZE * 0.1):]] = [child] * int(POP_SIZE * 0.1)

        if time.time() - start_time > budget_s - 5:
            break

    # Final local optimization
    if time.time() - start_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
