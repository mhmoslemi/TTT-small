import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    h = np.zeros(n_points, dtype=np.float64)
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()
    pop_size = 50
    mutation_factor = 0.5
    crossover_rate = 0.9
    mutation_scale = 0.5
    pop = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
    for i in range(pop_size):
        pop[i] = pop[i] * (target_sum / np.sum(pop[i]))

    for _ in range(100):
        if time.time() - start_time > budget_s - 5:
            break
        new_pop = []
        for idx in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + mutation_scale * (pop[b] - pop[c])
            trial = np.clip(mutant, 0.0, 1.0)
            trial = trial * (target_sum / np.sum(trial))
            trial_c5 = objective(trial)
            if trial_c5 < objective(pop[idx]):
                new_pop.append(trial)
            else:
                new_pop.append(pop[idx])
        pop = new_pop
        for p in pop:
            if objective(p) < best_c5:
                best_c5 = objective(p)
                best_h = p.copy()
        if _ % 10 == 0:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            best_h += noise
            best_h = np.clip(best_h, 0.0, 1.0)
            best_h = best_h * (target_sum / np.sum(best_h))

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + mutation_scale * (pop[b] - pop[c])
            trial = np.clip(mutant, 0.0, 1.0)
            trial = trial * (target_sum / np.sum(trial))
            trial_c5 = objective(trial)
            if trial_c5 < objective(pop[idx]):
                pop[idx] = trial
            for p in pop:
                if objective(p) < best_c5:
                    best_c5 = objective(p)
                    best_h = p.copy()

    return best_h.tolist(), best_c5, n_points
