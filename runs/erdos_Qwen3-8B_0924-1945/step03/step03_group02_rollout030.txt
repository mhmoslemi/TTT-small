import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        h[100:200] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    gen = 0
    pop_size = 50
    mut = 0.85
    crossp = 0.85
    elite = []
    best_idx = 0
    n_gen = 250

    while gen < n_gen and time.time() - start_time < budget_s - 10:
        if not elite:
            h_pop = np.random.uniform(0, 1, (pop_size, n_points))
            h_pop = h_pop * (target_sum / np.sum(h_pop, axis=1, keepdims=True))
        else:
            h_pop = np.random.uniform(0, 1, (pop_size, n_points))
            h_pop = h_pop * (target_sum / np.sum(h_pop, axis=1, keepdims=True))
            h_pop[0] = elite[best_idx].copy()

        for i in range(pop_size):
            if np.random.rand() < 0.1:
                h_pop[i] = np.random.uniform(0, 1, n_points)
                h_pop[i] = h_pop[i] * (target_sum / np.sum(h_pop[i]))

        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            vec = h_pop[a] + mut * (h_pop[b] - h_pop[c])
            vec = np.clip(vec, 0.0, 1.0)
            vec = vec * (target_sum / np.sum(vec))
            vec = np.random.permutation(vec)
            for j in range(n_points):
                if np.random.rand() < crossp:
                    vec[j] = h_pop[i][j]
            vec = np.clip(vec, 0.0, 1.0)
            vec = vec * (target_sum / np.sum(vec))
            h_pop[i] = vec

        obj_vals = np.array([objective(h.pop) for h.pop in h_pop])
        elite_idx = np.argmin(obj_vals)
        elite.append(h_pop[elite_idx].copy())
        best_idx = elite_idx
        best_h = elite[best_idx].copy()
        best_c5 = obj_vals[best_idx]
        gen += 1

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
