import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        h[0:n_points//2] = 0.2
        h[n_points//2:] = 0.8
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))

    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0.0, 1.0)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    n_iter = 0
    momentum = 0.9
    learning_rate = 0.05
    noise_factor = 0.05
    decay_rate = 0.99
    convergence_threshold = 1e-6
    max_iter = 400

    while n_iter < max_iter and time.time() - start_time < budget_s - 5:
        corr = np.correlate(h, 1.0 - h, mode="full")
        current_c5 = np.max(corr * dx)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        grad = np.zeros(n_points)
        for k in range(1, len(corr)//2):
            val = corr[k] * dx
            for i in range(n_points):
                grad[i] += (val - corr[k+1] * dx) * (1.0 - h[i]) * dx
                grad[i] += (val - corr[k-1] * dx) * h[i] * dx
        grad -= (np.sum(h) - target_sum) * 1000
        grad = np.clip(grad, -1e6, 1e6)

        m = momentum * m + grad
        h = np.clip(h - learning_rate * m, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0.0, 1.0)

        noise = np.random.uniform(-noise_factor, noise_factor, size=n_points)
        h += noise
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))

        if np.abs(current_c5 - best_c5) < convergence_threshold:
            break

        learning_rate *= decay_rate
        noise_factor *= decay_rate
        n_iter += 1

    if time.time() - start_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
