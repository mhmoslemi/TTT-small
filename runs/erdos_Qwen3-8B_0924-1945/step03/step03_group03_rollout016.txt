import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        h[np.argsort(np.random.rand(n_points))] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()

    # Global phase: Differential Evolution for global search
    def de_objective(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def de_constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    de_cons = [{'type': 'eq', 'fun': de_constraint}]
    de_bounds = [(0.0, 1.0) for _ in range(n_points)]

    de_result = opt.differential_evolution(
        de_objective,
        de_bounds,
        constraints=de_cons,
        popsize=50,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        maxiter=100
    )
    h_opt = de_result["x"]
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt)
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    # Local phase: L-BFGS-B for fine-tuning
    def lbfgs_objective(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    lbfgs_result = opt.minimize(
        lbfgs_objective,
        best_h,
        method='L-BFGS-B',
        bounds=de_bounds,
        constraints=cons,
        tol=1e-8,
        options={'maxiter': 50, 'ftol': 1e-9}
    )
    h_opt = lbfgs_result["x"]
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt)
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 10:
        # Final refinement
        final_result = opt.minimize(
            lbfgs_objective,
            best_h,
            method='L-BFGS-B',
            bounds=de_bounds,
            constraints=cons,
            tol=1e-10,
            options={'maxiter': 10, 'ftol': 1e-10}
        )
        h_opt = final_result["x"]
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
