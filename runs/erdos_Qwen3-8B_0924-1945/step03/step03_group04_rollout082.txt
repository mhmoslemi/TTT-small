import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        indices = np.arange(n_points)
        indices = np.sort(np.random.choice(indices, size=n_points // 2, replace=False))
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    pop_size = 50
    max_gen = 100
    mutation_rate = 0.1
    elite_ratio = 0.2
    local_refine_interval = 5

    population = [np.copy(h) for _ in range(pop_size)]
    population = [h * (target_sum / np.sum(h)) for h in population]
    for _ in range(max_gen):
        fitness = np.array([objective(h) for h in population])
        elite_idx = np.argsort(fitness)[:int(elite_ratio * pop_size)]
        elite = [population[i] for i in elite_idx]
        new_population = elite.copy()
        for _ in range(pop_size - len(elite)):
            parent1, parent2 = np.random.choice(elite, size=2, replace=False)
            child = (parent1 + parent2) / 2
            child = np.clip(child, 0.0, 1.0)
            sum_child = np.sum(child)
            child = child * (target_sum / sum_child)
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, size=n_points)
                child += noise
                child = np.clip(child, 0.0, 1.0)
                sum_child = np.sum(child)
                child = child * (target_sum / sum_child)
            new_population.append(child)
        population = new_population
        if _ % local_refine_interval == 0:
            for h in elite:
                h = h * (target_sum / np.sum(h))
                result = opt.minimize(
                    objective, h.flatten(), 
                    method='SLSQP', bounds=bounds, constraints=cons,
                    tol=1e-8, options={'maxiter': 50}
                )
                h_opt = result.x.reshape(n_points)
                h_opt = np.clip(h_opt, 0.0, 1.0)
                sum_h_opt = np.sum(h_opt)
                scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
                h_opt = h_opt * scale
                c5_bound = objective(h_opt.flatten())
                if c5_bound < best_c5:
                    best_c5 = c5_bound
                    best_h = h_opt.copy()
        if time.time() - start_time > budget_s - 5:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
