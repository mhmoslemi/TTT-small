import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()
    population_size = 20
    mutation_factor = 0.9
    crossover_prob = 0.8
    max_iterations = 200
    population = np.random.uniform(0.0, 1.0, (population_size, n_points))
    for i in range(population_size):
        population[i] = np.clip(population[i] * (target_sum / np.sum(population[i])), 0.0, 1.0)

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
        trial_population = []
        for j in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant_sum = np.sum(mutant)
            mutant = mutant * (target_sum / mutant_sum if mutant_sum != 0 else 1.0)
            trial_population.append(mutant)
        trial_population = np.array(trial_population)
        for j in range(population_size):
            if np.random.rand() < crossover_prob:
                trial_population[j] = population[j]
        population = np.array([trial_population[j] if np.random.rand() < crossover_prob else population[j] for j in range(population_size)])
        for j in range(population_size):
            population[j] = np.clip(population[j], 0.0, 1.0)
            population[j] = population[j] * (target_sum / np.sum(population[j]) if np.sum(population[j]) != 0 else 1.0)
        for j in range(population_size):
            c5 = objective(population[j])
            if c5 < best_c5:
                best_c5 = c5
                best_h = population[j].copy()
        if np.random.rand() < 0.05:
            noise = np.random.normal(0, 0.05, n_points)
            best_h += noise
            best_h = np.clip(best_h, 0.0, 1.0)
            best_h_sum = np.sum(best_h)
            best_h = best_h * (target_sum / best_h_sum if best_h_sum != 0 else 1.0)

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
