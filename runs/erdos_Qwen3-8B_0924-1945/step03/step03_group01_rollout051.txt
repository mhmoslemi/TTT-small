```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points // 2):
            x = np.random.uniform(0, 2)
            idx = np.abs(np.linspace(0, 2, n_points) - x).argmin()
            h[idx] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    result = opt.differential_evolution(
        objective, bounds, constraints=cons, 
        strategy='best1bin', popsize=50, tol=1e-8, 
        mutation=(0.5, 1), recombination=0.9, 
        maxiter=100, polish=True, 
        disp=False, seed=seed
    )
    best_h = result.x
    best_c5 = objective(best_h)

    result = opt.minimize(
        objective, best_h, 
        method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 50}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 5:
        result = opt.minimize(
            objective, best_h, 
            method='SLSQP', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        best_h = result.x
        best_c5 = objective(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    best_h = best_h * scale

    return best_h.tolist(), best_c5, n_points
```
