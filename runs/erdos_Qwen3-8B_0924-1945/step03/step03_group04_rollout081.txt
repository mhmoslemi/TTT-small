import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    pop_size = 50
    population = [np.copy(h) for _ in range(pop_size)]
    for i in range(pop_size):
        population[i] = np.clip(population[i], 0.0, 1.0)
        sum_pop = np.sum(population[i])
        population[i] *= target_sum / sum_pop if sum_pop != 0 else 1.0

    for gen in range(100):
        fitness = np.array([objective(ind) for ind in population])
        elite_idx = np.argsort(fitness)[:int(pop_size * 0.2)]
        elites = [population[i] for i in elite_idx]

        new_population = []
        for _ in range(pop_size - len(elites)):
            parent1 = np.random.choice(elites)
            parent2 = np.random.choice(elites)
            offspring = (parent1 + parent2) / 2
            offspring += np.random.normal(0, 0.05, size=n_points)
            offspring = np.clip(offspring, 0.0, 1.0)
            sum_offspring = np.sum(offspring)
            offspring *= target_sum / sum_offspring if sum_offspring != 0 else 1.0
            new_population.append(offspring)
        population = elites + new_population

        for ind in population:
            ind = np.clip(ind, 0.0, 1.0)
            sum_ind = np.sum(ind)
            ind *= target_sum / sum_ind if sum_ind != 0 else 1.0

        current_best = population[np.argmin(fitness)]
        current_best = np.clip(current_best, 0.0, 1.0)
        sum_best = np.sum(current_best)
        current_best *= target_sum / sum_best if sum_best != 0 else 1.0
        if objective(current_best) < best_c5:
            best_c5 = objective(current_best)
            best_h = current_best.copy()
        if time.time() - start_time > budget_s - 5:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='SLSQP', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
