```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    pop_size = 50
    gen_count = 50
    mutation_std = 0.05
    tournament_size = 5

    population = [np.clip(h + np.random.normal(0, mutation_std, n_points), 0, 1) * (target_sum / np.sum(np.clip(h + np.random.normal(0, mutation_std, n_points), 0, 1))) for _ in range(pop_size)]

    for gen in range(gen_count):
        tournament = np.random.choice(population, size=tournament_size)
        parents = tournament[np.argsort([objective(ind) for ind in tournament])[:2]]
        offspring = np.zeros(n_points)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                offspring[i] = parents[0][i]
            else:
                offspring[i] = parents[1][i]
        offspring = np.clip(offspring + np.random.normal(0, mutation_std, n_points), 0, 1)
        offspring = offspring * (target_sum / np.sum(offspring))
        population.append(offspring)
        population.pop(np.random.randint(0, len(population)))
        for ind in population:
            if objective(ind) < best_c5:
                best_c5 = objective(ind)
                best_h = ind.copy()
        if time.time() - start_time > budget_s - 5:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
