import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = np.array(initial_h_values, dtype=np.float64) if initial_h_values is not None else np.zeros(n_points)
    h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
    h = np.clip(h, 0.0, 1.0)
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    T_initial = 1.0
    T_final = 1e-6
    T_max = 200
    T_current = T_initial
    improve_counter = 0
    for t in range(T_max):
        if time.time() - start_time > budget_s - 5:
            break
        if t % 10 == 0:
            if improve_counter >= 100:
                T_initial *= 0.5
                improve_counter = 0
        indices = np.random.choice(n_points, size=int(n_points * 0.2), replace=False)
        noise = np.random.normal(loc=0, scale=0.05, size=n_points)
        h_perturbed = h.copy()
        h_perturbed[indices] += noise[indices]
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = h_perturbed * (target_sum / np.sum(h_perturbed)) if np.sum(h_perturbed) != 0 else h_perturbed
        corr = np.correlate(h_perturbed, 1.0 - h_perturbed, mode="full")
        c5_new = np.max(corr * dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_perturbed.copy()
            improve_counter = 0
        else:
            delta = c5_new - best_c5
            if np.random.rand() < np.exp(-delta / T_current):
                best_h = h_perturbed.copy()
                improve_counter = 0
            else:
                improve_counter += 1
        T_current = T_initial * (1.0 - t / T_max)
    h_opt = best_h.copy()
    h_opt = np.clip(h_opt, 0.0, 1.0)
    h_opt = h_opt * (target_sum / np.sum(h_opt)) if np.sum(h_opt) != 0 else h_opt
    corr = np.correlate(h_opt, 1.0 - h_opt, mode="full")
    c5_bound = np.max(corr * dx)
    return best_h.tolist(), c5_bound, n_points
