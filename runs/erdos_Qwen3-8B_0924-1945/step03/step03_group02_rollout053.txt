```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = np.zeros(n_points)
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.zeros(n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum
    
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    
    def de_search(pop_size=50, gen=250, mut=0.85, crossp=0.85):
        nonlocal best_c5, best_h
        pop = np.random.uniform(0, 1, (pop_size, n_points))
        pop = pop * (target_sum / np.sum(pop, axis=1, keepdims=True))
        for _ in range(gen):
            for i in range(pop_size):
                a, b, c = np.random.choice(pop_size, 3, replace=False)
                mutant = pop[a] + mut * (pop[b] - pop[c])
                mutant = np.clip(mutant, 0, 1)
                mutant = mutant * (target_sum / np.sum(mutant))
                corr = np.correlate(mutant, 1.0 - mutant, mode="full")
                c5 = np.max(corr * dx)
                if c5 < objective(pop[i]):
                    pop[i] = mutant
            best_candidate = pop[np.argmin([objective(ind) for ind in pop])]
            c5 = objective(best_candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = best_candidate.copy()
            if time.time() - start_time > budget_s - 10:
                break
    
    de_search(pop_size=50, gen=250, mut=0.85, crossp=0.85)
    
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 100}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
    
    return best_h.tolist(), best_c5, n_points
```
