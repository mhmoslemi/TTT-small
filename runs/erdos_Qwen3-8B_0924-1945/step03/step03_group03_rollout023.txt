```python
import numpy as np
import scipy.optimize as opt
import time
import cma

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    population_size = 50
    sigma = 0.5
    es = cma.CMAEvolutionStrategy(h.flatten(), sigma, {'popsize': population_size})
    es.set_seed(seed)

    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        solutions = es.ask()
        for sol in solutions:
            sol = np.clip(sol, 0.0, 1.0)
            sum_h = np.sum(sol)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            sol = sol * scale
            c5 = objective(sol)
            if c5 < best_c5:
                best_c5 = c5
                best_h = sol.copy()
        es.tell(solutions, [objective(sol) for sol in solutions])
        es.optimize()
        if _ % 5 == 0:
            es.result()
            best_h = es.result()[0]
            best_c5 = objective(best_h)
            if best_c5 < float('inf'):
                h = best_h.copy()

    if time.time() - start_time < budget_s:
        es.optimize()
        best_h = es.result()[0]
        best_c5 = objective(best_h)

    return best_h.tolist(), best_c5, n_points
```
