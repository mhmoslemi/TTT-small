import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    pop_size = 50
    mutation_factor = 0.5
    crossover_rate = 0.9
    mutation_scale = 0.5
    max_iter = 100
    early_stop = 50
    start_time = time.time()

    def evolve(population):
        nonlocal best_c5
        new_pop = []
        for i in range(pop_size):
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[idxs]
            mutant = a + mutation_factor * (b - c)
            trial = np.clip(mutant, 0.0, 1.0) * (target_sum / np.sum(np.clip(mutant, 0.0, 1.0)))
            if np.random.rand() < crossover_rate:
                trial = np.random.choice([a, trial], size=n_points, p=[crossover_rate, 1 - crossover_rate])
            trial = np.clip(trial, 0.0, 1.0)
            trial = trial * (target_sum / np.sum(trial))
            c5 = objective(trial)
            if c5 < best_c5:
                best_c5 = c5
                best_h = trial.copy()
            new_pop.append(trial)
        return new_pop

    population = [np.clip(h + np.random.uniform(-0.05, 0.05, size=n_points), 0.0, 1.0) * (target_sum / np.sum(np.clip(h + np.random.uniform(-0.05, 0.05, size=n_points), 0.0, 1.0))) for _ in range(pop_size)]

    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        population = evolve(population)
        if _ % 10 == 0:
            population[0] = np.clip(best_h + np.random.uniform(-0.05, 0.05, size=n_points), 0.0, 1.0) * (target_sum / np.sum(np.clip(best_h + np.random.uniform(-0.05, 0.05, size=n_points), 0.0, 1.0)))
        if len(population) > 0 and np.max([objective(ind) for ind in population]) < best_c5:
            best_c5 = np.max([objective(ind) for ind in population])
        if _ > 0 and np.abs(best_c5 - np.min([objective(ind) for ind in population])) < 1e-8:
            break

    if time.time() - start_time < budget_s:
        result = opt.minimize(objective, best_h.flatten(), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 50})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
