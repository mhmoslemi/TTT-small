import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    pop_size = 50
    pop = [h.copy() for _ in range(pop_size)]
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s - 5:
            break
        # Evaluate fitness
        fitness = []
        for ind in pop:
            ind_clipped = np.clip(ind, 0.0, 1.0)
            sum_h = np.sum(ind_clipped)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            ind_scaled = ind_clipped * scale
            corr = np.correlate(ind_scaled, 1.0 - ind_scaled, mode="full")
            c5 = np.max(corr * dx)
            fitness.append(c5)
        # Select elites
        elite_indices = np.argsort(fitness)[:int(pop_size * 0.2)]
        elites = [pop[i] for i in elite_indices]
        # Crossover
        for i in range(pop_size // 2):
            p1 = np.random.choice(elites)
            p2 = np.random.choice(elites)
            child = (p1 + p2) / 2
            # Mutate
            mutation = np.random.normal(0, 0.05, n_points)
            child = np.clip(child + mutation, 0.0, 1.0)
            sum_child = np.sum(child)
            scale = target_sum / sum_child if sum_child != 0 else 1.0
            child = child * scale
            # Replace worst
            worst_idx = np.argmax(fitness)
            pop[worst_idx] = child
            fitness[worst_idx] = objective(child.flatten())
        # Local refine
        if gen % 5 == 0:
            for i in range(min(5, len(elites))):
                ind = elites[i].copy()
                ind_clipped = np.clip(ind, 0.0, 1.0)
                sum_h = np.sum(ind_clipped)
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                ind_scaled = ind_clipped * scale
                result = opt.minimize(
                    lambda x: objective(x),
                    ind_scaled.flatten(),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    tol=1e-8,
                    options={'maxiter': 50}
                )
                ind_opt = result.x.reshape(n_points)
                ind_opt = np.clip(ind_opt, 0.0, 1.0)
                sum_h_opt = np.sum(ind_opt)
                scale_opt = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
                ind_opt = ind_opt * scale_opt
                c5_opt = objective(ind_opt.flatten())
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = ind_opt.copy()
    
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
    
    return best_h.tolist(), best_c5, n_points
