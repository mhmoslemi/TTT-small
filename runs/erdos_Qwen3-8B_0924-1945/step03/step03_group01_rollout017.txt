```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points // 2):
            h[i * 2] = 1
        h = h * (target_sum / np.sum(h))
    
    def obj_func(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def cons_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': cons_func}
    
    start_time = time.time()
    
    population_size = 50
    generations = 50
    mutation_std = 0.05
    elite_ratio = 0.2
    best_h = h.copy()
    best_c5 = obj_func(h)
    
    for _ in range(generations):
        population = [h.copy() for _ in range(population_size)]
        for i in range(population_size):
            noise = np.random.normal(0, mutation_std, n_points)
            population[i] += noise
            population[i] = np.clip(population[i], 0, 1)
            population[i] *= (target_sum / np.sum(population[i])) if np.sum(population[i]) != 0 else 1.0
        
        population_scores = [obj_func(ind) for ind in population]
        sorted_indices = np.argsort(population_scores)
        population = [population[i] for i in sorted_indices]
        population_scores = [population_scores[i] for i in sorted_indices]
        
        elite_count = int(population_size * elite_ratio)
        new_population = population[:elite_count]
        for _ in range(population_size - elite_count):
            parent1, parent2 = np.random.choice(population_size, 2, replace=False)
            child = (population[parent1] + population[parent2]) / 2
            noise = np.random.normal(0, mutation_std, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            child *= (target_sum / np.sum(child)) if np.sum(child) != 0 else 1.0
            new_population.append(child)
        
        population = new_population
        population_scores = [obj_func(ind) for ind in population]
        best_idx = np.argmin(population_scores)
        best_h = population[best_idx]
        best_c5 = population_scores[best_idx]
        
        if time.time() - start_time > budget_s - 5:
            break
    
    if time.time() - start_time < budget_s:
        result = opt.minimize(
            obj_func, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-6, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0, 1)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = obj_func(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt
    
    return best_h.tolist(), best_c5, n_points
```
