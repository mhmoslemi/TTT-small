import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    
    def de_optimize(h0, budget_s):
        pop_size = 20
        mut = 0.9
        cross = 0.8
        max_iter = int(budget_s * 0.6)
        pop = np.random.uniform(0.0, 1.0, (pop_size, n_points))
        pop = np.array([h0] + [np.random.uniform(0.0, 1.0, n_points) for _ in range(pop_size - 1)])
        best = h0.copy()
        best_c5 = float('inf')
        for _ in range(max_iter):
            for i in range(pop_size):
                a, b, c = np.random.choice(pop_size, 3, replace=False)
                mutant = pop[a] + mut * (pop[b] - pop[c])
                mutant = np.clip(mutant, 0.0, 1.0)
                mutant = mutant * (target_sum / np.sum(mutant) if np.sum(mutant) != 0 else 1.0)
                corr = np.correlate(mutant, 1.0 - mutant, mode="full")
                c5 = np.max(corr * dx)
                trial = mutant.copy()
                for j in range(n_points):
                    if np.random.rand() < cross:
                        trial[j] = mutant[j]
                trial = np.clip(trial, 0.0, 1.0)
                trial = trial * (target_sum / np.sum(trial) if np.sum(trial) != 0 else 1.0)
                corr = np.correlate(trial, 1.0 - trial, mode="full")
                trial_c5 = np.max(corr * dx)
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best = trial.copy()
                if trial_c5 < objective(pop[i]):
                    pop[i] = trial.copy()
            if best_c5 < objective(best):
                best = best.copy()
            if time.time() - time.time() > budget_s - 10:
                break
        return best, best_c5

    h_opt, c5_bound = de_optimize(h.copy(), budget_s)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    
    result = opt.minimize(
        objective, h_opt.flatten(), 
        method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 50}
    )
    h_final = result.x.reshape(n_points)
    h_final = np.clip(h_final, 0.0, 1.0)
    sum_h_final = np.sum(h_final)
    scale = target_sum / sum_h_final if sum_h_final != 0 else 1.0
    h_final = h_final * scale
    c5_final = objective(h_final.flatten())
    
    return h_final.tolist(), c5_final, n_points
