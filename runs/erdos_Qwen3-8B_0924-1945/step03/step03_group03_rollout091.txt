import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    pop_size = 50
    pop = [np.clip(np.random.rand(n_points), 0.0, 1.0) * (target_sum / np.sum(np.random.rand(n_points))) for _ in range(pop_size)]
    pop = [np.clip(p, 0.0, 1.0) for p in pop]
    pop = [p * (target_sum / np.sum(p)) for p in pop]
    pop = [p for p in pop if np.all(p >= 0) and np.all(p <= 1)]
    pop = [p for p in pop if np.isclose(np.sum(p), target_sum, atol=1e-6)]

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s - 5:
            break
        mutants = []
        for i in range(pop_size):
            a, b, c = np.random.choice(range(pop_size), 3, replace=False)
            mutant = pop[a] + 0.5 * (pop[b] - pop[c])
            mutants.append(mutant)
        trials = []
        for i in range(pop_size):
            mutant = mutants[i % pop_size]
            trial = pop[i] + np.random.rand(n_points) * (mutant - pop[i]) * 0.9
            trial = np.clip(trial, 0.0, 1.0)
            trial = trial * (target_sum / np.sum(trial))
            trials.append(trial)
        for i in range(pop_size):
            trial = trials[i]
            c5 = objective(trial.flatten())
            if c5 < objective(pop[i].flatten()):
                pop[i] = trial
        for p in pop:
            if np.all(p >= 0) and np.all(p <= 1) and np.isclose(np.sum(p), target_sum, atol=1e-6):
                c5 = objective(p.flatten())
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = p.copy()
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
            mutants = []
            for i in range(pop_size):
                a, b, c = np.random.choice(range(pop_size), 3, replace=False)
                mutant = pop[a] + 0.5 * (pop[b] - pop[c])
                mutants.append(mutant)
            trials = []
            for i in range(pop_size):
                mutant = mutants[i % pop_size]
                trial = pop[i] + np.random.rand(n_points) * (mutant - pop[i]) * 0.9
                trial = np.clip(trial, 0.0, 1.0)
                trial = trial * (target_sum / np.sum(trial))
                trials.append(trial)
            for i in range(pop_size):
                trial = trials[i]
                c5 = objective(trial.flatten())
                if c5 < objective(pop[i].flatten()):
                    pop[i] = trial
            for p in pop:
                if np.all(p >= 0) and np.all(p <= 1) and np.isclose(np.sum(p), target_sum, atol=1e-6):
                    c5 = objective(p.flatten())
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = p.copy()
    return best_h.tolist(), best_c5, n_points
