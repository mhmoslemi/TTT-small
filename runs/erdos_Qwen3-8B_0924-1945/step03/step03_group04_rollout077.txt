import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # Differential evolution setup
    def de_objective(x):
        return objective(x)

    # Start with initial population
    pop_size = 10
    pop = np.random.uniform(0.0, 1.0, (pop_size, n_points))
    pop = [p * (target_sum / np.sum(p)) for p in pop]

    # Optimization with differential evolution
    for gen in range(50):
        if time.time() - start_time > budget_s - 5:
            break
        new_pop = []
        for i in range(pop_size):
            # Random selection of three individuals
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            a, b, c = pop[a], pop[b], pop[c]
            # Mutation
            mutant = a + np.random.uniform(-0.5, 0.5, n_points) * (b - c)
            # Crossover
            trial = np.where(np.random.rand(n_points) < 0.5, mutant, pop[i])
            # Evaluation with constraint handling
            trial = np.clip(trial, 0.0, 1.0)
            sum_t = np.sum(trial)
            scale = target_sum / sum_t if sum_t != 0 else 1.0
            trial = trial * scale
            trial_c5 = objective(trial.flatten())
            # Selection
            if trial_c5 < objective(pop[i].flatten()):
                new_pop.append(trial)
            else:
                new_pop.append(pop[i])
        pop = new_pop
        # Update best solution
        for candidate in pop:
            c5 = objective(candidate.flatten())
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate.copy()

    # Refinement with L-BFGS-B
    if time.time() - start_time < budget_s - 10:
        result = opt.minimize(
            objective, best_h.flatten(),
            method='L-BFGS-B', bounds=bounds,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(),
            method='L-BFGS-B', bounds=bounds,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
