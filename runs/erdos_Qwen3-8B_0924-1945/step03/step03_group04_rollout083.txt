import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    n_particles = 40
    particles = [h.copy() for _ in range(n_particles)]
    pbest = [p.copy() for p in particles]
    gbest = h.copy()
    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        v = np.random.rand(n_particles, n_points) * 0.05
        for i in range(n_particles):
            h_i = particles[i]
            pbest_i = pbest[i]
            gbest_i = gbest
            r1, r2 = np.random.rand(2)
            v[i] = 0.8 * v[i] + 1.5 * r1 * (pbest_i - h_i) + 1.5 * r2 * (gbest_i - h_i)
            h_prop = h_i + v[i]
            h_prop = np.clip(h_prop, 0.0, 1.0)
            sum_h_prop = np.sum(h_prop)
            scale = target_sum / sum_h_prop if sum_h_prop != 0 else 1.0
            h_prop = h_prop * scale
            c5 = objective(h_prop)
            if c5 < objective(pbest[i]):
                pbest[i] = h_prop.copy()
            if c5 < objective(gbest):
                gbest = h_prop.copy()
        if objective(gbest) < best_c5:
            best_c5 = objective(gbest)
            best_h = gbest.copy()
        particles = [p.copy() for p in pbest]
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
    return best_h.tolist(), best_c5, n_points
