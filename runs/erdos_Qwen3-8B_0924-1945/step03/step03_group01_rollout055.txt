import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    def ga_run():
        POP_SIZE = 50
        MUT_STDEV = 0.05
        CROSSOVER_RATE = 0.8
        MAX_GENERATIONS = 50
        ELITISM = 10
        MUTATION_RATE = 0.1

        def mutate(h):
            noise = np.random.normal(0, MUT_STDEV, size=n_points)
            h += noise
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            return h * scale

        def crossover(h1, h2):
            if np.random.rand() < CROSSOVER_RATE:
                p1, p2 = np.random.permutation(n_points)
                h = h1.copy()
                h[p1:p2] = h2[p1:p2]
                h = np.clip(h, 0.0, 1.0)
                sum_h = np.sum(h)
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                return h * scale
            return h1.copy()

        def evaluate(h):
            return objective(h)

        pop = [mutate(h.copy()) for _ in range(POP_SIZE)]
        for _ in range(MAX_GENERATIONS):
            fitness = np.array([evaluate(ind) for ind in pop])
            order = np.argsort(fitness)
            new_pop = pop[order[:ELITISM]]
            while len(new_pop) < POP_SIZE:
                p1, p2 = np.random.choice(POP_SIZE, 2, replace=False)
                h1, h2 = pop[p1], pop[p2]
                child = crossover(h1, h2)
                if np.random.rand() < MUTATION_RATE:
                    child = mutate(child)
                new_pop.append(child)
            pop = new_pop

        best_idx = np.argmin([evaluate(ind) for ind in pop])
        best_h = pop[best_idx].copy()
        return best_h

    h = ga_run()
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h = h * scale

    result = opt.minimize(
        objective, h.flatten(), 
        method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 50}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt.flatten())

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 5:
        result = opt.minimize(
            objective, h_opt.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())

    return h_opt.tolist(), c5_bound, n_points
