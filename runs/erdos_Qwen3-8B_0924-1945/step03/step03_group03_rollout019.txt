import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    def feasible(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return np.abs(sum_h - target_sum) < 1e-8
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    population_size = 50
    mutation_factor = 0.5
    crossover_rate = 0.9
    mutation_scale = 0.5
    max_iterations = 100
    early_stop = 50
    
    population = np.random.uniform(0.0, 1.0, (population_size, n_points))
    population = [h * (target_sum / np.sum(h)) for h in population]
    
    best_h = np.copy(population[0])
    best_c5 = objective(best_h.flatten())
    
    start_time = time.time()
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
        
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = mutant * (target_sum / np.sum(mutant))
            
            if np.random.rand() < crossover_rate:
                trial = mutant
            else:
                trial = population[i].copy()
                mask = np.random.rand(n_points) < crossover_rate
                trial[mask] = mutant[mask]
            
            trial = np.clip(trial, 0.0, 1.0)
            trial = trial * (target_sum / np.sum(trial))
            
            if feasible(trial.flatten()):
                c5 = objective(trial.flatten())
                if c5 < best_c5:
                    best_h = np.copy(trial)
                    best_c5 = c5
        
        if iteration % 10 == 0:
            population = [np.copy(best_h)] + [np.copy(p) for p in population[1:]]
            population = np.array([np.random.uniform(0.0, 1.0, n_points) * (target_sum / np.sum(np.random.uniform(0.0, 1.0, n_points))) for _ in range(population_size - 1)])
        
        if np.abs(best_c5 - objective(best_h.flatten())) < 1e-8 and iteration > early_stop:
            break
    
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = mutant * (target_sum / np.sum(mutant))
            
            if np.random.rand() < crossover_rate:
                trial = mutant
            else:
                trial = population[i].copy()
                mask = np.random.rand(n_points) < crossover_rate
                trial[mask] = mutant[mask]
            
            trial = np.clip(trial, 0.0, 1.0)
            trial = trial * (target_sum / np.sum(trial))
            
            if feasible(trial.flatten()):
                c5 = objective(trial.flatten())
                if c5 < best_c5:
                    best_h = np.copy(trial)
                    best_c5 = c5
    
    return best_h.tolist(), best_c5, n_points
