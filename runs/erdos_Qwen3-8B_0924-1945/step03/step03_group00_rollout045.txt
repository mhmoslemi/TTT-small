import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            level = int(i // (n_points // 4))
            h[i] = 0.5 + 0.5 * (-1) ** level
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    learning_rate = 0.01
    momentum = 0.9
    l2_reg = 0.001
    h_opt = h.copy()
    m = np.zeros(n_points)
    iteration = 0
    early_stop = 10
    prev_c5 = np.inf

    while time.time() - start_time < budget_s - 5 and iteration < 500:
        corr = np.correlate(h_opt, 1.0 - h_opt, mode="full")
        current_c5 = np.max(corr * dx)

        grad = np.zeros(n_points)
        for k in range(1, len(corr) // 2):
            val = corr[k] * dx
            for i in range(n_points):
                grad[i] += (val - corr[k + 1] * dx) * (1.0 - h_opt[i]) * dx
                grad[i] += (val - corr[k - 1] * dx) * h_opt[i] * dx

        grad -= l2_reg * (np.sum(h_opt) - target_sum)
        m = momentum * m + grad
        h_opt = np.clip(h_opt - learning_rate * m, 0.0, 1.0)
        sum_h = np.sum(h_opt)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_opt = h_opt * scale

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()

        if abs(current_c5 - prev_c5) < 1e-6 and iteration > 10:
            break
        prev_c5 = current_c5
        iteration += 1

        if np.random.rand() < 0.2:
            noise = np.random.uniform(-0.02, 0.02, size=n_points)
            h_opt += noise
            h_opt = np.clip(h_opt, 0.0, 1.0)
            sum_h = np.sum(h_opt)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_opt = h_opt * scale

    if time.time() - start_time < budget_s - 5:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
