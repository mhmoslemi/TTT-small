import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = np.zeros(n_points)
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[::5] = 1
        h = h * (target_sum / np.sum(h))
    
    def objective(h_flat, k):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return corr[k] * dx
    
    def compute_max_overlap(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum
    
    def update_h(h_flat, target_k):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        best_k = np.argmax(corr) - n_points + 1
        best_k = np.clip(best_k, 0, n_points - 1)
        print("Target k:", target_k, "Actual k:", best_k)
        def objective2(h_flat):
            h = h_flat.reshape(n_points)
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            return objective(h_flat, best_k)
        cons = [{'type': 'eq', 'fun': constraint}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.minimize(objective2, h_flat, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})
        return result.x
    
    best_c5 = compute_max_overlap(h)
    best_h = h.copy()
    best_k = np.argmax(np.correlate(h, 1.0 - h, mode="full")) - n_points + 1
    best_k = np.clip(best_k, 0, n_points - 1)
    start_time = time.time()
    h_flat = h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        h_flat = update_h(h_flat, best_k)
        current_c5 = compute_max_overlap(h_flat)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_flat.copy()
            best_k = np.argmax(np.correlate(h_flat, 1.0 - h_flat, mode="full")) - n_points + 1
            best_k = np.clip(best_k, 0, n_points - 1)
        else:
            h_flat = np.random.uniform(0, 1, n_points)
            sum_h = np.sum(h_flat)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_flat = h_flat * scale
    return best_h.tolist(), best_c5, n_points
