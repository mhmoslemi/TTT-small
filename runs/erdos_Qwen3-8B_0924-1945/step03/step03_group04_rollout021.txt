import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    def perturb(h, sigma, fraction):
        idxs = np.random.choice(n_points, size=int(fraction * n_points), replace=False)
        noise = np.random.normal(0, sigma, size=n_points)
        h = h + noise
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        return h

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    T_initial = 1.0
    T_max = 200
    T_current = T_initial
    improved_count = 0
    sigma = 0.05
    decay_rate = 0.999

    for t in range(T_max):
        if time.time() - start_time > budget_s - 5:
            break
        if t % 10 == 0:
            if improved_count < 10:
                sigma = 0.05
            else:
                sigma = 0.02
            improved_count = 0
        h_perturbed = perturb(h.copy(), sigma, 0.2)
        c5_new = objective(h_perturbed.flatten())
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_perturbed.copy()
            improved_count = 0
        else:
            delta = c5_new - best_c5
            if np.random.random() < np.exp(-delta / T_current):
                best_h = h_perturbed.copy()
                improved_count = 0
            else:
                improved_count += 1
        T_current *= decay_rate
        h = best_h.copy()

    if time.time() - start_time < budget_s:
        result = np.minimize(objective, best_h.flatten(), bounds=[(0, 1)] * n_points, method='L-BFGS-B', tol=1e-8, options={'maxiter': 50})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
