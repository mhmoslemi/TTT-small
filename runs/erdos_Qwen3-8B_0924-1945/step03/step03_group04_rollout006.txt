import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # PSO parameters
    n_particles = 40
    max_iter = 200
    omega = 0.8
    alpha = 1.5
    beta = 1.5
    inertia_weight_decay = 0.95
    social_coefficient_factor = 1.1
    noise_amplitude = 0.05
    rescale_threshold = 1e-5

    particles = [h + np.random.uniform(-noise_amplitude, noise_amplitude, size=n_points) for _ in range(n_particles)]
    particles = [np.clip(p, 0.0, 1.0) * (target_sum / np.sum(np.clip(p, 0.0, 1.0))) for p in particles]
    pbest = particles.copy()
    gbest = min(pbest, key=lambda p: objective(p.flatten()))
    best_c5 = objective(gbest.flatten())

    for iter in range(max_iter):
        for i, p in enumerate(particles):
            # Update velocity and position
            v = np.random.uniform(-0.1, 0.1, size=n_points)
            v = omega * v + alpha * np.random.rand() * (pbest[i] - p) + beta * np.random.rand() * (gbest - p)
            new_p = p + v
            new_p = np.clip(new_p, 0.0, 1.0)
            new_p = new_p * (target_sum / np.sum(new_p)) if np.sum(new_p) != 0 else new_p

            # Evaluate objective
            current_c5 = objective(new_p.flatten())
            if current_c5 < objective(pbest[i].flatten()):
                pbest[i] = new_p
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    gbest = pbest[i]
            particles[i] = new_p

        # Adapt parameters
        omega *= inertia_weight_decay
        beta *= social_coefficient_factor
        if iter % 10 == 0 and iter > 0:
            if np.max(np.abs(np.sum(particles, axis=1) - target_sum)) > rescale_threshold:
                particles = [p * (target_sum / np.sum(p)) for p in particles]
                pbest = [p * (target_sum / np.sum(p)) for p in pbest]
                gbest = gbest * (target_sum / np.sum(gbest))
                best_c5 = objective(gbest.flatten())

        if time.time() - start_time > budget_s - 5:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, gbest.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
