```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # PSO parameters
    n_particles = 40
    inertia = 0.8
    cognitive = 1.5
    social = 1.5
    max_iter = 200
    noise_amp = 0.05
    particles = [h.copy() for _ in range(n_particles)]
    velocities = np.zeros((n_particles, n_points))
    pbest = [p.copy() for p in particles]
    pbest_obj = [objective(p.flatten()) for p in particles]
    gbest = particles[np.argmin(pbest_obj)]
    gbest_obj = min(pbest_obj)
    stagnation_threshold = 50
    stagnation_count = 0
    iter_count = 0

    while iter_count < max_iter and time.time() - start_time < budget_s - 5:
        for i in range(n_particles):
            # Update velocity and position
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (pbest[i] - particles[i]) + social * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            # Ensure bounds
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            # Rescale to meet sum constraint
            sum_p = np.sum(particles[i])
            scale = target_sum / sum_p if sum_p != 0 else 1.0
            particles[i] *= scale
            # Evaluate objective
            obj = objective(particles[i].flatten())
            # Update personal best
            if obj < pbest_obj[i]:
                pbest[i] = particles[i].copy()
                pbest_obj[i] = obj
        # Update global best
        current_gbest_obj = min(pbest_obj)
        if current_gbest_obj < gbest_obj:
            gbest = pbest[np.argmin(pbest_obj)].copy()
            gbest_obj = current_gbest_obj
        # Check for stagnation
        if pbest_obj == pbest_obj[-1:] * len(pbest_obj):
            stagnation_count += 1
        else:
            stagnation_count = 0
        if stagnation_count > stagnation_threshold:
            inertia = 0.4
            cognitive = 2.5
            stagnation_count = 0
        iter_count += 1
        # Track best solution
        if gbest_obj < best_c5:
            best_c5 = gbest_obj
            best_h = gbest.copy()

    # Refinement with L-BFGS-B
    if time.time() - start_time < budget_s - 5:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
