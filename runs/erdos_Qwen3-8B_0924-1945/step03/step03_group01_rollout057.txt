import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.linspace(0, n_points - 1, n_points // 2, dtype=int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    POP_SIZE = 50
    MAX_GENERATIONS = 50
    MUTATION_STD = 0.05
    TOURNAMENT_SIZE = 5

    population = [np.clip(h + np.random.normal(0, MUTATION_STD, n_points), 0.0, 1.0).copy() for _ in range(POP_SIZE)]
    for p in population:
        sum_p = np.sum(p)
        scale_p = target_sum / sum_p if sum_p != 0 else 1.0
        p = p * scale_p
    population = np.array(population)

    for _ in range(MAX_GENERATIONS):
        tournament = np.random.choice(population, size=TOURNAMENT_SIZE, replace=False)
        parents = tournament[np.argmin([objective(p.flatten()) for p in tournament])]
        offspring = []
        for _ in range(POP_SIZE // 2):
            p1 = np.random.choice(population)
            p2 = np.random.choice(population)
            child = np.zeros(n_points)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = p1[i]
                else:
                    child[i] = p2[i]
            child = np.clip(child, 0.0, 1.0)
            sum_child = np.sum(child)
            scale_child = target_sum / sum_child if sum_child != 0 else 1.0
            child = child * scale_child
            mutated = child + np.random.normal(0, MUTATION_STD, n_points)
            mutated = np.clip(mutated, 0.0, 1.0)
            sum_mut = np.sum(mutated)
            scale_mut = target_sum / sum_mut if sum_mut != 0 else 1.0
            mutated = mutated * scale_mut
            offspring.append(mutated)
        offspring = np.array(offspring)
        population = np.vstack([population, offspring])
        population = np.random.permutation(population)[:POP_SIZE]
        fitness = [objective(p.flatten()) for p in population]
        best_idx = np.argmin(fitness)
        best_c5_candidate = fitness[best_idx]
        if best_c5_candidate < best_c5:
            best_c5 = best_c5_candidate
            best_h = population[best_idx].copy()
        if time.time() - start_time > budget_s - 5:
            break

    if time.time() - start_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
