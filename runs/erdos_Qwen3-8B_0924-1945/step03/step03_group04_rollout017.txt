import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    pop_size = 50
    mutation_rate = 0.1
    elite_ratio = 0.2
    generations = 100
    local_refine_freq = 5
    local_refine_iters = 50
    max_iter = 100

    population = [h.copy() for _ in range(pop_size)]
    population = [np.clip(p, 0.0, 1.0) * (target_sum / np.sum(p)) for p in population]
    best_individual = min(population, key=objective)

    for gen in range(generations):
        fitness = [objective(p) for p in population]
        elite_idx = np.argsort(fitness)[:int(elite_ratio * pop_size)]
        elites = [population[i] for i in elite_idx]
        offspring = []

        for _ in range(pop_size - len(elite_idx)):
            parent1 = np.random.choice(elites)
            parent2 = np.random.choice(elites)
            child = (parent1 + parent2) / 2.0
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, size=n_points)
                child += noise
            child = np.clip(child, 0.0, 1.0) * (target_sum / np.sum(child))
            offspring.append(child)

        population = elites + offspring
        population = [np.clip(p, 0.0, 1.0) * (target_sum / np.sum(p)) for p in population]
        population = [p for p in population if np.sum(p) > 1e-6]
        population = [p for p in population if np.max(p) < 1.0001 and np.min(p) > -1e-6]
        population = population[:pop_size]

        current_best = min(population, key=objective)
        if objective(current_best) < objective(best_individual):
            best_individual = current_best

        if gen % local_refine_freq == 0:
            local_pop = [best_individual for _ in range(5)]
            for p in local_pop:
                result = opt.minimize(
                    objective, p.flatten(), 
                    method='SLSQP', bounds=bounds, constraints=cons,
                    tol=1e-8, options={'maxiter': local_refine_iters}
                )
                p_opt = result.x.reshape(n_points)
                p_opt = np.clip(p_opt, 0.0, 1.0)
                sum_p = np.sum(p_opt)
                scale = target_sum / sum_p if sum_p != 0 else 1.0
                p_opt = p_opt * scale
                if objective(p_opt) < objective(best_individual):
                    best_individual = p_opt.copy()

        if time.time() - start_time > budget_s - 5:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_individual.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < objective(best_individual):
            best_individual = h_opt.copy()

    return best_individual.tolist(), objective(best_individual), n_points
