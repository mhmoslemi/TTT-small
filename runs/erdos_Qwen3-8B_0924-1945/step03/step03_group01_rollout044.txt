import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    def ga_run():
        pop_size = 50
        gen = 50
        mut_std = 0.05
        tournament_size = 5
        elite_ratio = 0.2
        pop = []
        for _ in range(pop_size):
            candidate = h.copy()
            noise = np.random.normal(0, mut_std, n_points)
            candidate += noise
            candidate = np.clip(candidate, 0, 1)
            sum_h = np.sum(candidate)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            candidate *= scale
            pop.append(candidate)
        for _ in range(gen):
            pop = np.array(pop)
            fitness = np.array([objective(p) for p in pop])
            elite_num = int(pop_size * elite_ratio)
            elite_idx = np.argpartition(fitness, elite_num)[:elite_num]
            elite_pop = pop[elite_idx]
            selected = []
            for _ in range(pop_size - elite_num):
                candidates = np.random.choice(pop_size, tournament_size, replace=False)
                best_idx = candidates[np.argmin(fitness[candidates])]
                selected.append(pop[best_idx])
            selected = np.array(selected)
            offspring = []
            for i in range(0, len(selected), 2):
                if i + 1 >= len(selected):
                    break
                parent1 = selected[i]
                parent2 = selected[i + 1]
                crossover = np.random.rand(n_points) < 0.5
                child = np.where(crossover, parent1, parent2)
                noise = np.random.normal(0, mut_std, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                sum_h = np.sum(child)
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                child *= scale
                offspring.append(child)
            pop = np.concatenate([elite_pop, offspring])
            pop = pop[:pop_size]
            pop = np.array([np.clip(p, 0, 1) for p in pop])
            fitness = np.array([objective(p) for p in pop])
            sorted_idx = np.argsort(fitness)
            pop = pop[sorted_idx]
            pop = pop[:pop_size]
            best_idx = np.argmin(fitness)
            best_h_candidate = pop[best_idx]
            if objective(best_h_candidate) < best_c5:
                best_c5 = objective(best_h_candidate)
                best_h = best_h_candidate.copy()

    for _ in range(2):
        ga_run()
        if time.time() - start_time > budget_s - 5:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
