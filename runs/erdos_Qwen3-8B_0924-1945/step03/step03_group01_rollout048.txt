```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        m = n_points // 2
        x = np.linspace(0, 2, m)
        x = x + 0.25 * (1 + 0.5 * np.sin(2 * np.pi * np.arange(m) / m))
        h = np.zeros(n_points)
        h[0] = 1
        h[-1] = 1
        for i in range(1, m - 1):
            idx = int(np.round((x[i] / 2) * n_points))
            if 0 <= idx < n_points:
                h[idx] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    def optimize_with_de(h0):
        result = opt.differential_evolution(
            objective,
            bounds,
            args=(),
            strategy='best1bin',
            mutation=(0.8, 1.0),
            recombination=0.9,
            population_size=50,
            generations=50,
            tol=1e-5,
            polish=True,
            disp=False,
            max_evals=100000
        )
        best_de = result.x
        best_de = np.clip(best_de, 0.0, 1.0)
        sum_de = np.sum(best_de)
        scale = target_sum / sum_de if sum_de != 0 else 1.0
        best_de = best_de * scale
        c5 = objective(best_de)
        return best_de, c5

    for _ in range(3):
        h_opt, c5 = optimize_with_de(h.copy())
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
        if time.time() - start_time > budget_s - 5:
            break
        h = h_opt.copy()
        if np.random.rand() < 0.2:
            noise = np.random.uniform(-0.1, 0.1, size=n_points)
            h = np.clip(h + noise, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 100}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
