import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()
    pop_size = 20
    mutation = 0.9
    crossover = 0.8
    max_iter = 200

    def de_optimize():
        nonlocal best_c5, best_h
        h_pop = np.random.uniform(0, 1, (pop_size, n_points))
        h_pop = h_pop * (target_sum / np.sum(h_pop, axis=1, keepdims=True))
        for _ in range(max_iter):
            for i in range(pop_size):
                j, k, l = np.random.choice(pop_size, 3, replace=False)
                mutant = h_pop[j] + mutation * (h_pop[k] - h_pop[l])
                mutant = np.clip(mutant, 0, 1)
                mutant = mutant * (target_sum / np.sum(mutant))
                trial = np.zeros(n_points)
                for m in range(n_points):
                    if np.random.rand() < crossover:
                        trial[m] = mutant[m]
                    else:
                        trial[m] = h_pop[i][m]
                trial = np.clip(trial, 0, 1)
                trial = trial * (target_sum / np.sum(trial))
                obj_trial = objective(trial.flatten())
                obj_current = objective(h_pop[i].flatten())
                if obj_trial < obj_current:
                    h_pop[i] = trial
            current_obj = [objective(h.flatten()) for h in h_pop]
            best_idx = np.argmin(current_obj)
            best_c5_candidate = current_obj[best_idx]
            if best_c5_candidate < best_c5:
                best_c5 = best_c5_candidate
                best_h = h_pop[best_idx].copy()
        return best_h

    h_opt = de_optimize()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 10:
        result = opt.minimize(
            objective, h_opt.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
