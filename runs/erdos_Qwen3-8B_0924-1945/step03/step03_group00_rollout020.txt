```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            level = int(i / (n_points // 4))
            h[i] = 0.5 + 0.5 * (-1)**level
        h = h * (target_sum / np.sum(h))
    
    h = h.copy()
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    learning_rate = 0.01
    momentum = 0.9
    l2_reg = 0.001
    max_iter = 500
    iter_count = 0
    m = np.zeros(n_points)
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 5:
        corr = np.correlate(h, 1.0 - h, mode="full")
        current_c5 = np.max(corr * dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        grad = np.zeros(n_points)
        for k in range(1, len(corr)//2):
            val = corr[k] * dx
            for i in range(n_points):
                grad[i] += (val - corr[k+1] * dx) * (1.0 - h[i]) * dx
                grad[i] += (val - corr[k-1] * dx) * h[i] * dx
        
        grad -= l2_reg * (np.sum(h) - target_sum)
        m = momentum * m + grad
        h = np.clip(h - learning_rate * m, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))
        
        if np.random.rand() < 0.1:
            noise = np.random.uniform(-0.02, 0.02, size=n_points)
            h += noise
            h = np.clip(h, 0.0, 1.0)
            h = h * (target_sum / np.sum(h))
        
        iter_count += 1
    
    if time.time() - start_time < budget_s - 5:
        result = opt.minimize(
            lambda x: np.max(np.correlate(x, 1.0 - x, mode="full") * dx),
            best_h,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            tol=1e-8,
            options={'maxiter': 50}
        )
        h_opt = result.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        h_opt = h_opt * (target_sum / np.sum(h_opt))
        c5_bound = np.max(np.correlate(h_opt, 1.0 - h_opt, mode="full") * dx)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
    
    return best_h.tolist(), best_c5, n_points
```
