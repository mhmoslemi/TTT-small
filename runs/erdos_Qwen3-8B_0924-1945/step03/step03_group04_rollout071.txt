import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = np.array(initial_h_values, dtype=np.float64) if initial_h_values is not None else np.zeros(n_points)
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h))
    
    def compute_c5(h):
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    best_c5 = compute_c5(h)
    start_time = time.time()
    T_initial = 1.0
    T_max = 200
    T_current = T_initial
    noisy = False
    
    for t in range(T_max):

            break
        if t % 10 == 0 and not noisy:
            noisy = True
            T_initial *= 0.5
            T_max = int(T_max * 0.9)
        indices = np.random.choice(n_points, size=int(n_points * 0.2), replace=False)
        noise = np.random.normal(0, 0.05, size=n_points)
        h_new = h.copy()
        h_new[indices] += noise[indices]
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = h_new * (target_sum / np.sum(h_new))
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        else:

            if np.random.rand() < np.exp(-delta / T_current):
