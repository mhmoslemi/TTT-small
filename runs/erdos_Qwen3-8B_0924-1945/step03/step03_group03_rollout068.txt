```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def normalize(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        return h * scale

    population_size = 50
    mutation_factor = 0.5
    crossover_rate = 0.9
    mutation_scale = 0.5
    max_iterations = 100
    early_stop = 50

    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [normalize(p) for p in population]
    population[0] = h.copy()

    best_h = h.copy()
    best_c5 = evaluate(h)

    start_time = time.time()

    for iter in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
        new_population = []
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            trial = np.clip(population[i] + mutation_scale * (mutant - population[i]), 0, 1)
            trial = normalize(trial)
            trial_c5 = evaluate(trial)
            if trial_c5 < evaluate(population[i]):
                new_population.append(trial)
            else:
                new_population.append(population[i])
        population = new_population
        if np.min([evaluate(p) for p in population]) < best_c5:
            best_c5 = np.min([evaluate(p) for p in population])
            best_h = population[np.argmin([evaluate(p) for p in population])]

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            trial = np.clip(population[np.random.randint(population_size)] + mutation_scale * (mutant - population[np.random.randint(population_size)]), 0, 1)
            trial = normalize(trial)
            trial_c5 = evaluate(trial)
            if trial_c5 < evaluate(population[np.random.randint(population_size)]):
                population[np.random.randint(population_size)] = trial
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial

    return best_h.tolist(), best_c5, n_points
```
