import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    h = h.copy()
    for _ in range(300):
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if time.time() - start_time > budget_s - 5:
            break
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_minus = h.copy()
            h_minus[i] -= 1e-5
            corr_plus = np.correlate(h_plus, 1.0 - h_plus, mode="full")
            corr_minus = np.correlate(h_minus, 1.0 - h_minus, mode="full")
            grad[i] = (np.max(corr_plus * dx) - np.max(corr_minus * dx)) / (2e-5)
        h = np.clip(h - 0.1 * grad, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (target_sum / sum_h)
        if time.time() - start_time > budget_s - 5:
            break
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            lambda x: np.max(np.correlate(x, 1.0 - x, mode="full") * dx),
            best_h,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        )
        h_opt = result.x
        c5_bound = np.max(np.correlate(h_opt, 1.0 - h_opt, mode="full") * dx)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt
    return best_h.tolist(), best_c5, n_points
