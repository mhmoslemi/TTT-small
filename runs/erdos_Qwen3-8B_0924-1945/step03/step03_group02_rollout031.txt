```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 5 == 0:
                h[i] = 1.0
            else:
                h[i] = 0.5
        h = h * (target_sum / np.sum(h))

    def objective(h_flat, k_star):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = corr[k_star] * dx
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = h.copy()
    best_k = 0
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s - 10:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = np.argmax(corr)
        def obj_func(h_flat):
            return objective(h_flat, k_star)
        result = opt.minimize(obj_func, h.flatten(), method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt, k_star)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
            best_k = k_star
        h = h_opt.copy()
        if np.random.rand() < 0.1:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            h += noise
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        k_star = np.argmax(corr)
        def obj_func(h_flat):
            return objective(h_flat, k_star)
        result = opt.minimize(obj_func, best_h.flatten(), method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt, k_star)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
