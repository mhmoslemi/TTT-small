import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    def de_minimize(initial_guess):
        population_size = 20
        mutation_factor = 0.9
        crossover_prob = 0.8
        max_iterations = 200
        population = np.random.rand(population_size, n_points)
        population = (population - 0.5) * 2 * (target_sum / n_points) + 0.5
        population = population * (target_sum / np.mean(np.sum(population, axis=1)))
        best_fitness = float('inf')
        best_solution = initial_guess.copy()

        for _ in range(max_iterations):
            for i in range(population_size):
                a, b, c = np.random.choice(population_size, 3, replace=False)
                mutant = population[a] + mutation_factor * (population[b] - population[c])
                mutant = np.clip(mutant, 0.0, 1.0)
                mutant = mutant * (target_sum / np.sum(mutant))
                trial = population[i].copy()
                for j in range(n_points):
                    if np.random.rand() < crossover_prob:
                        trial[j] = mutant[j]
                trial = np.clip(trial, 0.0, 1.0)
                trial = trial * (target_sum / np.sum(trial))
                fitness = objective(trial.flatten())
                if fitness < objective(population[i].flatten()):
                    population[i] = trial
            current_best = np.min([objective(p.flatten()) for p in population])
            if current_best < best_fitness:
                best_fitness = current_best
                best_solution = population[np.argmin([objective(p.flatten()) for p in population])].copy()
        return best_solution, best_fitness

    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    h_de, c5_de = de_minimize(h)
    best_c5 = c5_de
    best_h = h_de.copy()
    if time.time() - start_time < budget_s - 50:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
    return best_h.tolist(), best_c5, n_points
