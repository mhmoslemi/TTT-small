```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    pop_size = 50
    max_gen = 150
    mutation_rate = 0.1
    elite_ratio = 0.2
    local_refine_freq = 5
    local_refine_iter = 50

    population = [np.copy(h) for _ in range(pop_size)]
    for idx in range(pop_size):
        population[idx] = np.clip(population[idx], 0.0, 1.0)
        sum_pop = np.sum(population[idx])
        population[idx] *= target_sum / sum_pop

    for gen in range(max_gen):
        fitness = []
        for indiv in population:
            indiv = np.clip(indiv, 0.0, 1.0)
            sum_indiv = np.sum(indiv)
            indiv *= target_sum / sum_indiv
            corr = np.correlate(indiv, 1.0 - indiv, mode="full")
            c5 = np.max(corr * dx)
            fitness.append(c5)

        fitness = np.array(fitness)
        elite_idx = np.argsort(fitness)[:int(elite_ratio * pop_size)]
        elite = [population[i] for i in elite_idx]

        new_population = []
        for _ in range(pop_size - len(elite)):
            parent1, parent2 = np.random.choice(elite, size=2, replace=False)
            child = (parent1 + parent2) / 2.0
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, size=n_points)
                child += noise
            child = np.clip(child, 0.0, 1.0)
            sum_child = np.sum(child)
            child *= target_sum / sum_child
            new_population.append(child)

        population = elite + new_population
        for indiv in population:
            indiv = np.clip(indiv, 0.0, 1.0)
            sum_indiv = np.sum(indiv)
            indiv *= target_sum / sum_indiv

        if (gen + 1) % local_refine_freq == 0:
            top_indices = np.argsort(fitness)[:5]
            for idx in top_indices:
                h_opt = population[idx].copy()
                h_opt = np.clip(h_opt, 0.0, 1.0)
                sum_opt = np.sum(h_opt)
                h_opt *= target_sum / sum_opt
                result = opt.minimize(
                    objective, h_opt.flatten(), 
                    method='SLSQP', bounds=bounds, constraints=cons,
                    tol=1e-8, options={'maxiter': local_refine_iter}
                )
                h_opt = result.x.reshape(n_points)
                h_opt = np.clip(h_opt, 0.0, 1.0)
                sum_opt = np.sum(h_opt)
                h_opt *= target_sum / sum_opt
                c5_bound = objective(h_opt.flatten())
                if c5_bound < best_c5:
                    best_c5 = c5_bound
                    best_h = h_opt.copy()

        if gen % 10 == 0:
            current_fitness = np.array([objective(p.flatten()) for p in population])
            best_fit_idx = np.argmin(current_fitness)
            best_fit = population[best_fit_idx]
            if objective(best_fit.flatten()) < best_c5:
                best_c5 = objective(best_fit.flatten())
                best_h = best_fit.copy()

        if time.time() - start_time > budget_s - 5:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_opt = np.sum(h_opt)
        h_opt *= target_sum / sum_opt
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
