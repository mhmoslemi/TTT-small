import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # Initialize momentum and learning rate
    momentum = 0.9
    learning_rate = 0.1
    decay_rate = 0.99
    max_iter = 200
    iter_count = 0

    while iter_count < max_iter and time.time() - start_time < budget_s - 5:
        # Evaluate current C5
        h_opt = h.copy()
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        corr = np.correlate(h_opt, 1.0 - h_opt, mode="full")
        current_c5 = np.max(corr * dx)

        # Gradient estimation
        epsilon = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h_opt.copy()
            h_plus[i] += epsilon
            sum_h_plus = np.sum(h_plus)
            scale_plus = target_sum / sum_h_plus if sum_h_plus != 0 else 1.0
            h_plus = h_plus * scale_plus
            corr_plus = np.correlate(h_plus, 1.0 - h_plus, mode="full")
            c5_plus = np.max(corr_plus * dx)

            h_minus = h_opt.copy()
            h_minus[i] -= epsilon
            sum_h_minus = np.sum(h_minus)
            scale_minus = target_sum / sum_h_minus if sum_h_minus != 0 else 1.0
            h_minus = h_minus * scale_minus
            corr_minus = np.correlate(h_minus, 1.0 - h_minus, mode="full")
            c5_minus = np.max(corr_minus * dx)

            grad[i] = (c5_plus - c5_minus) / (2 * epsilon)

        # Momentum update
        momentum = max(0.0, min(1.0, momentum))
        gradient_current = grad
        gradient_momentum = momentum * gradient_momentum + (1 - momentum) * gradient_current
        h_update = h_opt - learning_rate * gradient_momentum

        # Clip and rescale
        h_update = np.clip(h_update, 0.0, 1.0)
        sum_h_update = np.sum(h_update)
        scale_update = target_sum / sum_h_update if sum_h_update != 0 else 1.0
        h_update = h_update * scale_update

        # Update and check if improvement
        h = h_update.copy()
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5_bound = np.max(corr * dx)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h.copy()

        # Decay learning rate
        learning_rate *= decay_rate

        # Introduce random noise periodically
        if iter_count % 10 == 0:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            h = h + noise
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale

        iter_count += 1

    # Final refinement with L-BFGS-B
    if time.time() - start_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
