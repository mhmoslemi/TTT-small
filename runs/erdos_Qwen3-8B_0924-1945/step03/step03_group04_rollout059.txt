import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # PSO setup
    num_particles = 40
    particles = [np.copy(h) + np.random.uniform(-0.05, 0.05, size=n_points) for _ in range(num_particles)]
    velocities = np.zeros((num_particles, n_points))
    pbest = np.copy(particles)
    pbest_values = np.array([objective(p) for p in particles])
    gbest = np.copy(particles[np.argmin(pbest_values)])
    gbest_value = np.min(pbest_values)

    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(num_particles):
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            velocities[i] = 0.8 * velocities[i] + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            # Clip and rescale
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_p = np.sum(particles[i])
            particles[i] *= target_sum / sum_p if sum_p != 0 else 1.0
            # Evaluate
            curr_val = objective(particles[i])
            if curr_val < pbest_values[i]:
                pbest[i] = np.copy(particles[i])
                pbest_values[i] = curr_val
        # Update global best
        if np.min(pbest_values) < gbest_value:
            gbest_value = np.min(pbest_values)
            gbest = np.copy(pbest[np.argmin(pbest_values)])
        # Update best solution
        if gbest_value < best_c5:
            best_c5 = gbest_value
            best_h = np.copy(gbest)

    # Final refinement with L-BFGS-B
    if time.time() - start_time < budget_s - 5:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    elapsed_time = time.time() - start_time
    return best_h.tolist(), best_c5, n_points
