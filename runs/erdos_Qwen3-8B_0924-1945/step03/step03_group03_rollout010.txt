```python
import numpy as np
import scipy.optimize as opt
import time
import cma

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    class CMAESOptimizer:
        def __init__(self, initial_guess, n_points, target_sum, dx, budget_s):
            self.n_points = n_points
            self.target_sum = target_sum
            self.dx = dx
            self.budget_s = budget_s
            self.best_h = initial_guess.copy()
            self.best_c5 = objective(initial_guess)
            self.start_time = time.time()

        def clip_and_scale(self, h_flat):
            h = h_flat.copy()
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = self.target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            return h

        def evaluate(self, h_flat):
            h = self.clip_and_scale(h_flat)
            corr = np.correlate(h, 1.0 - h, mode="full")
            c5 = np.max(corr * self.dx)
            return c5

        def optimize(self):
            pop_size = 50
            sigma = 0.5
            options = {
                'popsize': pop_size,
                'sigma': sigma,
                'ftol': 1e-8,
                'bounds': [(0.0, 1.0) for _ in range(self.n_points)],
                'verb_disp': False,
                'maxiter': int(self.budget_s / 10)
            }
            result = cma.fmin(self.evaluate, self.best_h, sigma, options=options)
            self.best_h = result[0]
            self.best_c5 = self.evaluate(self.best_h)
            return self.best_h, self.best_c5

    optimizer = CMAESOptimizer(h.copy(), n_points, target_sum, dx, budget_s)
    optimizer.optimize()

    elapsed_time = time.time() - optimizer.start_time
    if elapsed_time < budget_s:
        optimizer.best_h, optimizer.best_c5 = optimizer.optimize()

    return optimizer.best_h.tolist(), optimizer.best_c5, n_points
```
