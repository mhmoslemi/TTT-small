import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    population_size = 50
    pop = np.random.uniform(0, 1, (population_size, n_points))
    pop = [p * (target_sum / np.sum(p)) for p in pop]
    
    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        mutant = []
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant_vec = pop[a] + 0.5 * (pop[b] - pop[c])
            mutant_vec = np.clip(mutant_vec, 0, 1)
            mutant_vec = mutant_vec * (target_sum / np.sum(mutant_vec))
            mutant.append(mutant_vec)
        mutant = np.array(mutant)
        trial = []
        for i in range(population_size):
            j = np.random.randint(population_size)
            if np.random.random() < 0.9:
                trial_vec = np.where(np.random.rand(n_points) < 0.9, mutant[i], pop[i])
            else:
                trial_vec = mutant[i]
            trial_vec = np.clip(trial_vec, 0, 1)
            trial_vec = trial_vec * (target_sum / np.sum(trial_vec))
            trial.append(trial_vec)
        trial = np.array(trial)
        obj_vals = np.array([objective(t) for t in trial])
        indices = np.argsort(obj_vals)
        pop = np.array([pop[i] for i in indices[:population_size]])
        if np.min(obj_vals) < best_c5:
            best_c5 = np.min(obj_vals)
            best_h = pop[np.argmin(obj_vals)]
    
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
            mutant = []
            for i in range(population_size):
                a, b, c = np.random.choice(population_size, 3, replace=False)
                mutant_vec = pop[a] + 0.5 * (pop[b] - pop[c])
                mutant_vec = np.clip(mutant_vec, 0, 1)
                mutant_vec = mutant_vec * (target_sum / np.sum(mutant_vec))
                mutant.append(mutant_vec)
            mutant = np.array(mutant)
            trial = []
            for i in range(population_size):
                j = np.random.randint(population_size)
                if np.random.random() < 0.9:
                    trial_vec = np.where(np.random.rand(n_points) < 0.9, mutant[i], pop[i])
                else:
                    trial_vec = mutant[i]
                trial_vec = np.clip(trial_vec, 0, 1)
                trial_vec = trial_vec * (target_sum / np.sum(trial_vec))
                trial.append(trial_vec)
            trial = np.array(trial)
            obj_vals = np.array([objective(t) for t in trial])
            indices = np.argsort(obj_vals)
            pop = np.array([pop[i] for i in indices[:population_size]])
            if np.min(obj_vals) < best_c5:
                best_c5 = np.min(obj_vals)
                best_h = pop[np.argmin(obj_vals)]
    
    return best_h.tolist(), best_c5, n_points
