import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    pop_size = 20
    mut_scale = 0.9
    cross_rate = 0.8
    max_iter = int(budget_s * 0.7 / (n_points * 10))
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def de_step(h_pop):
        h_pop = np.clip(h_pop, 0.0, 1.0)
        h_pop = h_pop * (target_sum / np.sum(h_pop, axis=1, keepdims=True))
        corr = np.array([np.max(np.correlate(h, 1.0 - h, mode="full") * dx) for h in h_pop])
        return np.min(corr)
    
    h_pop = np.random.uniform(0.0, 1.0, (pop_size, n_points))
    h_pop = h_pop * (target_sum / np.sum(h_pop, axis=1, keepdims=True))
    best_pop = h_pop.copy()
    
    for _ in range(max_iter):
        for i in range(pop_size):
            idxs = np.random.choice(pop_size, 3, replace=False)
            mutant = h_pop[idxs[0]] + mut_scale * (h_pop[idxs[1]] - h_pop[idxs[2]])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = mutant * (target_sum / np.sum(mutant, axis=1, keepdims=True))
            trial = mutant.copy()
            for j in range(n_points):
                if np.random.rand() < cross_rate:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0.0, 1.0)
            trial = trial * (target_sum / np.sum(trial, axis=1, keepdims=True))
            corr = np.max(np.correlate(trial[0], 1.0 - trial[0], mode="full") * dx)
            if corr < de_step(h_pop):
                h_pop[i] = trial
        best_pop = h_pop[np.argmin([np.max(np.correlate(h, 1.0 - h, mode="full") * dx) for h in h_pop])]
        best_c5 = np.max(np.correlate(best_pop[0], 1.0 - best_pop[0], mode="full") * dx)
        if time.time() - start_time > budget_s - 10:
            break
    
    h = best_pop[0]
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h = h * scale
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5_bound = np.max(corr * dx)
    
    return h.tolist(), c5_bound, n_points
