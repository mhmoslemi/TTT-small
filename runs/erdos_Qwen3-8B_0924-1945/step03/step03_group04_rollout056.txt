import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = h * (target_sum / np.sum(h))

    def compute_c5(h):
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr * dx)

    def constraint(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()

    pop_size = 50
    pop = np.random.uniform(0.0, 1.0, (pop_size, n_points))
    pop = pop * (target_sum / np.sum(pop, axis=1)[:, None])

    def evaluate(h):
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr * dx)

    for gen in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        fitness = np.array([evaluate(ind) for ind in pop])
        elites = pop[np.argsort(fitness)[:int(pop_size * 0.2)]]
        new_pop = elites.copy()

        for _ in range(int(pop_size * 0.8)):
            p1, p2 = np.random.choice(elites, size=2, replace=False)
            child = (p1 + p2) / 2
            sigma = 0.05
            noise = np.random.normal(0, sigma, n_points)
            child += noise
            child = np.clip(child, 0.0, 1.0)
            child = child * (target_sum / np.sum(child))
            new_pop = np.vstack((new_pop, child))
        pop = new_pop[:pop_size]

        for _ in range(5):
            if time.time() - start_time > budget_s - 10:
                break
            idx = np.random.choice(range(pop_size), size=5)
            candidates = pop[idx]
            result = opt.minimize(
                lambda x: evaluate(x),
                candidates[0],
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint},
                tol=1e-8,
                options={'maxiter': 50}
            )
            for c in candidates:
                if np.allclose(result.x, c, atol=1e-5):
                    break
            else:
                result = opt.minimize(
                    lambda x: evaluate(x),
                    candidates[0],
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint},
                    tol=1e-8,
                    options={'maxiter': 50}
                )
            for c in candidates:
                if np.allclose(result.x, c, atol=1e-5):
                    break
            for c in candidates:
                if np.max(np.abs(result.x - c)) < 1e-5:
                    break
            else:
                result = opt.minimize(
                    lambda x: evaluate(x),
                    candidates[0],
                    method='COBYLA',
                    bounds=bounds,
                    constraints={'type': 'eq', 'fun': constraint},
                    tol=1e-8,
                    options={'maxiter': 50}
                )
            best_candidate = result.x
            if evaluate(best_candidate) < best_c5:
                best_c5 = evaluate(best_candidate)
                best_h = best_candidate.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            lambda x: evaluate(x),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            tol=1e-8,
            options={'maxiter': 50}
        )
        best_h = result.x
        best_c5 = evaluate(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (target_sum / np.sum(best_h))
    return best_h.tolist(), best_c5, n_points
