import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        h[np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()
    pop_size = 50
    particles = [h.copy() for _ in range(pop_size)]
    velocities = np.random.uniform(-0.1, 0.1, (pop_size, n_points))
    pbest = [p.copy() for p in particles]
    pbest_values = [objective(p) for p in pbest]
    gbest = best_h.copy()
    gbest_value = best_c5

    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        for i in range(pop_size):
            h = particles[i].copy()
            v = velocities[i].copy()
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            fit = objective(h)
            if fit < pbest_values[i]:
                pbest[i] = h.copy()
                pbest_values[i] = fit
            if fit < gbest_value:
                gbest = h.copy()
                gbest_value = fit

            scale_new = target_sum / sum_h_new if sum_h_new != 0 else 1.0
            h_new = h_new * scale_new
            particles[i] = h_new.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(objective, gbest.flatten(), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 50})

        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt)
        if c5_bound < gbest_value:
            gbest = h_opt.copy()
            gbest_value = c5_bound

    return gbest.tolist(), gbest_value, n_points
