import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.sort(np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int))
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    def cma_es_run():
        nonlocal best_c5, best_h
        pop_size = 50
        mu = pop_size // 2
        sigma = 0.5
        mean = h.copy()
        cov = np.eye(n_points) * sigma
        for _ in range(200):
            if time.time() - start_time > budget_s - 5:
                break
            theta = np.random.multivariate_normal(mean, cov, pop_size)
            theta = np.clip(theta, 0.0, 1.0)
            sum_theta = np.sum(theta, axis=1)
            scale = target_sum / sum_theta[:, np.newaxis] if np.any(sum_theta != 0) else 1.0
            theta = theta * scale[:, np.newaxis]
            fitness = np.array([objective(theta[i]) for i in range(pop_size)])
            idxs = np.argsort(fitness)
            best_idx = idxs[0]
            best_theta = theta[best_idx]
            if fitness[best_idx] < best_c5:
                best_c5 = fitness[best_idx]
                best_h = best_theta.copy()
            mean = theta[idxs[:mu], :].mean(axis=0)
            diff = theta[idxs[:mu], :] - mean[:, np.newaxis]
            cov = np.cov(diff, rowvar=False)
            sigma = np.sqrt(np.trace(cov))
            if _ % 10 == 0:
                theta = np.random.multivariate_normal(mean, cov, pop_size)
                theta = np.clip(theta, 0.0, 1.0)
                sum_theta = np.sum(theta, axis=1)
                scale = target_sum / sum_theta[:, np.newaxis] if np.any(sum_theta != 0) else 1.0
                theta = theta * scale[:, np.newaxis]
                fitness = np.array([objective(theta[i]) for i in range(pop_size)])
                best_theta = theta[np.argmin(fitness)]
                if fitness[np.argmin(fitness)] < best_c5:
                    best_c5 = fitness[np.argmin(fitness)]
                    best_h = best_theta.copy()
    start_time = time.time()
    cma_es_run()
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(objective, best_h.flatten(), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 50})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
    return best_h.tolist(), best_c5, n_points
