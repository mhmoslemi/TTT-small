import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_gen = 100
    pop_size = 50
    elite_ratio = 0.2
    mutation_rate = 0.1
    local_refine_freq = 5
    local_max_iter = 50
    local_tol = 1e-8

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    population = [h.copy() for _ in range(pop_size)]
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()

    for gen in range(max_gen):
        fitness = []
        for h in population:
            h_flat = h.flatten()
            h_flat = np.clip(h_flat, 0.0, 1.0)
            sum_h = np.sum(h_flat)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_flat = h_flat * scale
            corr = np.correlate(h_flat, 1.0 - h_flat, mode="full")
            c5 = np.max(corr * dx)
            fitness.append(c5)
        
        elite_idx = np.argsort(fitness)[:int(elite_ratio * pop_size)]
        elites = [population[i] for i in elite_idx]
        population = elites.copy()

        for _ in range(pop_size - len(elites)):
            parent1, parent2 = np.random.choice(elites, size=2, replace=False)
            child = (parent1 + parent2) / 2
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0.0, 1.0)
                sum_child = np.sum(child)
                scale = target_sum / sum_child if sum_child != 0 else 1.0
                child = child * scale
            population.append(child)

        population = [np.clip(p, 0.0, 1.0) for p in population]
        population = [p * (target_sum / np.sum(p)) for p in population]
        fitness = []
        for h in population:
            h_flat = h.flatten()
            corr = np.correlate(h_flat, 1.0 - h_flat, mode="full")
            c5 = np.max(corr * dx)
            fitness.append(c5)
        population = [p for _, p in sorted(zip(fitness, population))]
        current_best = population[0]
        current_c5 = fitness[0]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
        if gen % local_refine_freq == 0 and gen > 0:
            result = opt.minimize(
                objective, best_h.flatten(), 
                method='SLSQP', bounds=bounds, constraints=cons,
                tol=local_tol, options={'maxiter': local_max_iter}
            )
            h_opt = result.x.reshape(n_points)
            h_opt = np.clip(h_opt, 0.0, 1.0)
            sum_h_opt = np.sum(h_opt)
            scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
            h_opt = h_opt * scale
            c5_bound = objective(h_opt.flatten())
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h = h_opt.copy()
        if time.time() - start_time > budget_s - 5:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=local_tol, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
