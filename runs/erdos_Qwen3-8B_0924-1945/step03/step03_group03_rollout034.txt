import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    population_size = 50
    mutation_factor = 0.5
    crossover_rate = 0.9
    mutation_scale = 0.5
    max_iterations = 100
    early_stop = 50
    start_time = time.time()

    def evolve():
        nonlocal best_c5, best_h
        population = [h.copy() for _ in range(population_size)]
        for p in population:
            p = np.clip(p, 0.0, 1.0)
            sum_p = np.sum(p)
            p = p * (target_sum / sum_p if sum_p != 0 else 1.0)
        population = np.array(population)
        for _ in range(max_iterations):
            if time.time() - start_time > budget_s - 5:
                break
            mutant = np.zeros_like(population[0])
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            sum_mutant = np.sum(mutant)
            mutant = mutant * (target_sum / sum_mutant if sum_mutant != 0 else 1.0)
            trial = population[np.random.randint(population_size)]
            trial = trial * (1 - crossover_rate) + mutant * crossover_rate
            trial = np.clip(trial, 0.0, 1.0)
            sum_trial = np.sum(trial)
            trial = trial * (target_sum / sum_trial if sum_trial != 0 else 1.0)
            c5 = objective(trial.flatten())
            if c5 < objective(population[np.random.randint(population_size)].flatten()):
                population[np.random.randint(population_size)] = trial
            for i in range(population_size):
                if objective(population[i].flatten()) < best_c5:
                    best_c5 = objective(population[i].flatten())
                    best_h = population[i].copy()
            if _ % early_stop == 0:
                current = objective(best_h.flatten())
                if current == best_c5:
                    break
        return best_h

    best_h = evolve()
    return best_h.tolist(), best_c5, n_points
