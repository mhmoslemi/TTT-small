import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()
    pop_size = 50
    population = [h.copy() for _ in range(pop_size)]
    for p in population:
        p = np.clip(p, 0.0, 1.0)
        p = p * (target_sum / np.sum(p))
    population = np.array(population)

    for _ in range(100):
        for i in range(pop_size):
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[idxs]
            mutant = a + np.random.rand(n_points) * (b - c)
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = mutant * (target_sum / np.sum(mutant))
            j = np.random.randint(n_points)
            trial = np.copy(population[i])
            trial[j] = mutant[j]
            trial = np.clip(trial, 0.0, 1.0)
            trial = trial * (target_sum / np.sum(trial))
            c5_trial = objective(trial.flatten())
            if c5_trial < objective(population[i].flatten()):
                population[i] = trial
        best_h = population[np.argmin([objective(p.flatten()) for p in population])]
        best_c5 = objective(best_h.flatten())
        if time.time() - start_time > budget_s - 5:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(50):
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[idxs]
            mutant = a + np.random.rand(n_points) * (b - c)
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = mutant * (target_sum / np.sum(mutant))
            j = np.random.randint(n_points)
            trial = np.copy(population[np.random.randint(pop_size)])
            trial[j] = mutant[j]
            trial = np.clip(trial, 0.0, 1.0)
            trial = trial * (target_sum / np.sum(trial))
            c5_trial = objective(trial.flatten())
            if c5_trial < objective(population[np.random.randint(pop_size)].flatten()):
                population[np.random.randint(pop_size)] = trial
        best_h = population[np.argmin([objective(p.flatten()) for p in population])]
        best_c5 = objective(best_h.flatten())

    return best_h.tolist(), best_c5, n_points
