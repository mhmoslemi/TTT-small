import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    population_size = 50
    pop = np.random.uniform(0, 1, (population_size, n_points))
    pop = [h * (target_sum / np.sum(h)) for h in pop]

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s - 5:
            break
        mutants = []
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = pop[a] + 0.5 * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant_sum = np.sum(mutant)
            mutant = mutant * (target_sum / mutant_sum) if mutant_sum != 0 else mutant
            mutants.append(mutant)
        mutants = np.array(mutants)
        trials = []
        for i in range(population_size):
            trial = pop[i] * 0.5 + mutants[i] * 0.5
            trial = np.clip(trial, 0.0, 1.0)
            trial_sum = np.sum(trial)
            trial = trial * (target_sum / trial_sum) if trial_sum != 0 else trial
            trials.append(trial)
        trials = np.array(trials)
        for i in range(population_size):
            c5 = objective(trials[i])
            if c5 < objective(pop[i]):
                pop[i] = trials[i]
            if c5 < best_c5:
                best_c5 = c5
                best_h = trials[i].copy()
        if _ % 5 == 0:
            noise = np.random.normal(0, 0.05, n_points)
            pop += noise
            for i in range(population_size):
                pop[i] = np.clip(pop[i], 0.0, 1.0)
                pop[i] = pop[i] * (target_sum / np.sum(pop[i])) if np.sum(pop[i]) != 0 else pop[i]

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
            mutants = []
            for i in range(population_size):
                a, b, c = np.random.choice(population_size, 3, replace=False)
                mutant = pop[a] + 0.5 * (pop[b] - pop[c])
                mutant = np.clip(mutant, 0.0, 1.0)
                mutant_sum = np.sum(mutant)
                mutant = mutant * (target_sum / mutant_sum) if mutant_sum != 0 else mutant
                mutants.append(mutant)
            mutants = np.array(mutants)
            trials = []
            for i in range(population_size):
                trial = pop[i] * 0.5 + mutants[i] * 0.5
                trial = np.clip(trial, 0.0, 1.0)
                trial_sum = np.sum(trial)
                trial = trial * (target_sum / trial_sum) if trial_sum != 0 else trial
                trials.append(trial)
            trials = np.array(trials)
            for i in range(population_size):
                c5 = objective(trials[i])
                if c5 < objective(pop[i]):
                    pop[i] = trials[i]
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = trials[i].copy()
            if _ % 5 == 0:
                noise = np.random.normal(0, 0.05, n_points)
                pop += noise
                for i in range(population_size):
                    pop[i] = np.clip(pop[i], 0.0, 1.0)
                    pop[i] = pop[i] * (target_sum / np.sum(pop[i])) if np.sum(pop[i]) != 0 else pop[i]

    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    best_h = best_h * (target_sum / sum_h) if sum_h != 0 else best_h
    return best_h.tolist(), best_c5, n_points
