import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # Parameters for the genetic algorithm
    pop_size = 50
    max_gen = 200
    elite_ratio = 0.2
    mutation_rate = 0.1
    alpha = 0.5  # for BLX-α crossover
    local_refine_interval = 5

    # Initial population
    population = [np.copy(h) for _ in range(pop_size)]
    for ind in population:
        ind = np.clip(ind, 0.0, 1.0)
        sum_h = np.sum(ind)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        ind *= scale

    def evaluate(ind):
        ind = np.clip(ind, 0.0, 1.0)
        sum_h = np.sum(ind)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        ind *= scale
        corr = np.correlate(ind, 1.0 - ind, mode="full")
        return np.max(corr * dx)

    for gen in range(max_gen):
        # Evaluate fitness
        fitness = [evaluate(ind) for ind in population]
        # Select elites
        elite_indices = np.argsort(fitness)[:int(elite_ratio * pop_size)]
        elites = [population[i] for i in elite_indices]
        # Crossover
        new_population = []
        for _ in range(pop_size - len(elites)):
            parent1, parent2 = np.random.choice(elites, size=2, replace=False)
            if np.random.rand() < 0.5:
                # Arithmetic crossover
                offspring = (parent1 + parent2) / 2
            else:
                # BLX-α crossover
                lower = np.min([parent1, parent2])
                upper = np.max([parent1, parent2])
                offspring = np.random.uniform(lower * alpha, upper * alpha, size=n_points)
            new_population.append(offspring)
        # Mutate
        for i in range(len(new_population)):
            offspring = new_population[i]
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    offspring[j] += np.random.normal(0, 0.05)
            new_population[i] = np.clip(offspring, 0.0, 1.0)
            sum_h = np.sum(new_population[i])
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            new_population[i] *= scale
        # Replace population
        population = elites + new_population
        # Local refinement
        if gen % local_refine_interval == 0:
            for ind in population[:5]:
                ind = np.clip(ind, 0.0, 1.0)
                sum_h = np.sum(ind)
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                ind *= scale
                result = opt.minimize(
                    objective, ind.flatten(),
                    method='SLSQP', bounds=bounds, constraints=cons,
                    tol=1e-8, options={'maxiter': 50}
                )
                ind = result.x.reshape(n_points)
                ind = np.clip(ind, 0.0, 1.0)
                sum_h = np.sum(ind)
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                ind *= scale
                c5 = objective(ind.flatten())
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = ind.copy()
        # Check for time limit
        if time.time() - start_time > budget_s - 5:
            break
        # Track the best solution
        for ind in population:
            c5 = evaluate(ind)
            if c5 < best_c5:
                best_c5 = c5
                best_h = ind.copy()

    # Final refinement
    result = opt.minimize(
        objective, best_h.flatten(),
        method='SLSQP', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 50}
    )
    best_h = result.x.reshape(n_points)
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    best_h *= scale
    c5_bound = objective(best_h.flatten())

    return best_h.tolist(), c5_bound, n_points
