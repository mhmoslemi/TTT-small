import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    particles = [np.copy(h) for _ in range(40)]
    velocities = np.random.uniform(-0.05, 0.05, size=(40, n_points))
    pbest_positions = [np.copy(p) for p in particles]
    pbest_values = [objective(p) for p in particles]
    gbest_position = np.copy(particles[0])
    gbest_value = pbest_values[0]

    for iter in range(200):
        for i in range(40):
            r1, r2 = np.random.rand(2)
            velocities[i] = 0.8 * velocities[i] + 1.5 * r1 * (pbest_positions[i] - particles[i]) + 1.5 * r2 * (gbest_position - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_h = np.sum(particles[i])
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            particles[i] = particles[i] * scale
            current_val = objective(particles[i])
            if current_val < pbest_values[i]:
                pbest_positions[i] = np.copy(particles[i])
                pbest_values[i] = current_val
        current_gbest = np.min(pbest_values)
        if current_gbest < gbest_value:
            gbest_value = current_gbest
            gbest_position = np.copy(pbest_positions[np.argmin(pbest_values)])
        if time.time() - start_time > budget_s - 5:
            break

    h = gbest_position
    best_h = h.copy()
    best_c5 = gbest_value

    if time.time() - start_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
