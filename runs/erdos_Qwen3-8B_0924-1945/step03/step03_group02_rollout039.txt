import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        scale = target_sum / sum_h
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()

    def differential_evolution():
        nonlocal best_c5, best_h
        pop_size = 50
        mut = 0.85
        cross = 0.85
        gen = 250
        pop = np.random.uniform(0, 1, (pop_size, n_points))
        pop = np.clip(pop, 0, 1)
        pop = pop * (target_sum / np.sum(pop, axis=1, keepdims=True))
        for _ in range(gen):
            if time.time() - start_time > budget_s - 10:
                break
            for i in range(pop_size):
                j, k, l = np.random.choice(pop_size, 3, replace=False)
                mutant = pop[j] + mut * (pop[k] - pop[l])
                mutant = np.clip(mutant, 0, 1)
                mutant = mutant * (target_sum / np.sum(mutant))
                trial = pop[i].copy()
                idx = np.random.choice(n_points, int(cross * n_points), replace=False)
                trial[idx] = mutant[idx]
                trial = np.clip(trial, 0, 1)
                trial = trial * (target_sum / np.sum(trial))
                if objective(trial.flatten()) < objective(pop[i].flatten()):
                    pop[i] = trial
            current_pop = pop
            for candidate in current_pop:
                c5 = objective(candidate.flatten())
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = candidate.copy()
        return best_h

    h = differential_evolution()

    def local_refinement():
        nonlocal best_h, best_c5
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint},
            tol=1e-8, options={'maxiter': 150}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        if sum_h_opt == 0:
            return
        scale = target_sum / sum_h_opt
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    local_refinement()

    if time.time() - start_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint},
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        if sum_h_opt == 0:
            return best_h.tolist(), best_c5, n_points
        scale = target_sum / sum_h_opt
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
