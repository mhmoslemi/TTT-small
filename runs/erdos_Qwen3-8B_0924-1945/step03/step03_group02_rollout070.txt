import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()
    for _ in range(200):
        corr = np.correlate(h, 1.0 - h, mode="full")
        if np.max(corr) == 0:
            break
        k_max = np.argmax(corr)
        lag = k_max - (n_points - 1)
        if lag < 0:
            lag = lag + n_points
        if lag > n_points - 1:
            lag = lag - n_points
        if lag < 0:
            lag = lag + n_points
        if lag > n_points - 1:
            lag = lag - n_points
        lag = np.clip(lag, 0, n_points - 1)
        overlap = corr[lag] * dx
        if overlap < 1e-6:
            break

        def objective_sub(h_flat_sub):
            h_sub = h_flat_sub.reshape(n_points)
            h_sub = np.clip(h_sub, 0.0, 1.0)
            sum_h_sub = np.sum(h_sub)
            scale_sub = target_sum / sum_h_sub if sum_h_sub != 0 else 1.0
            h_sub = h_sub * scale_sub
            corr_sub = np.correlate(h_sub, 1.0 - h_sub, mode="full")
            c5_sub = corr_sub[lag] * dx
            return c5_sub

        def constraint_sub(h_flat_sub):
            h_sub = h_flat_sub.reshape(n_points)
            sum_h_sub = np.sum(h_sub)
            return sum_h_sub - target_sum

        cons_sub = [{'type': 'eq', 'fun': constraint_sub}]
        result_sub = opt.minimize(
            objective_sub, h.flatten(), 
            method='SLSQP', bounds=bounds, constraints=cons_sub,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result_sub.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale_opt = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale_opt
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
        if time.time() - start_time > budget_s - 10:
            break
        h = h_opt.copy()
        if np.random.rand() < 0.2:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            h += noise
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale_opt = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale_opt
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
