import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        k = np.round(np.linspace(0, n_points - 1, n_points // 5)).astype(int)
        h[k] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    best_k = 0

    start_time = time.time()

    for _ in range(100):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = np.argmax(corr) - n_points + 1
        if k_star < 0:
            k_star = 0
        elif k_star >= n_points:
            k_star = n_points - 1

        def sub_objective(h_flat):
            h = h_flat.reshape(n_points)
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            corr = np.correlate(h, 1.0 - h, mode="full")
            c5 = corr[k_star] * dx
            return c5

        result = opt.minimize(
            sub_objective, h.flatten(), 
            method='SLSQP', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
            best_k = k_star
        if time.time() - start_time > budget_s - 10:
            break
        h = h_opt.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        k_star = np.argmax(corr) - n_points + 1
        if k_star < 0:
            k_star = 0
        elif k_star >= n_points:
            k_star = n_points - 1

        def sub_objective(h_flat):
            h = h_flat.reshape(n_points)
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            corr = np.correlate(h, 1.0 - h, mode="full")
            c5 = corr[k_star] * dx
            return c5

        result = opt.minimize(
            sub_objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
