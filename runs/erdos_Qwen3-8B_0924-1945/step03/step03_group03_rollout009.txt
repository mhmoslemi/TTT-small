```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    population_size = 50
    max_iterations = 100
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    if initial_h_values is not None:
        h_initial = np.array(initial_h_values, dtype=np.float64)
        h_initial = h_initial * (target_sum / np.sum(h_initial))
        h_initial = np.clip(h_initial, 0.0, 1.0)
        best_h = h_initial.copy()
        best_c5 = objective(best_h.flatten())
    else:
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))
        best_h = h.copy()
        best_c5 = objective(best_h.flatten())

    particles = np.zeros((population_size, n_points))
    particles[0] = best_h
    velocities = np.zeros((population_size, n_points))
    pbest = np.zeros((population_size, n_points))
    pbest[0] = best_h
    gbest = best_h.copy()
    gbest_c5 = best_c5

    for _ in range(max_iterations):
        for i in range(population_size):
            h = particles[i]
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            corr = np.correlate(h, 1.0 - h, mode="full")
            c5 = np.max(corr * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if c5 < gbest_c5:
                gbest = h.copy()
                gbest_c5 = c5
            pbest[i] = h.copy()

        for i in range(population_size):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (pbest[i] - particles[i]) + social * r2 * (gbest - particles[i])
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_h = np.sum(particles[i])
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            particles[i] = particles[i] * scale

        if time.time() - time.time() > budget_s - 5:
            break

    return best_h.tolist(), best_c5, n_points
```
