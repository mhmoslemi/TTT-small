import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()
    pop_size = 50
    population = [h.copy() for _ in range(pop_size)]
    for p in population:
        p = np.clip(p, 0.0, 1.0)
        sum_p = np.sum(p)
        p = p * (target_sum / sum_p if sum_p != 0 else 1.0)

    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        new_population = []
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[c] + 0.5 * (population[a] - population[b])
            trial = np.clip(mutant, 0.0, 1.0)
            sum_trial = np.sum(trial)
            trial = trial * (target_sum / sum_trial if sum_trial != 0 else 1.0)
            if np.random.rand() < 0.9:
                trial = population[i] * (1 - 0.9) + trial * 0.9
            trial = np.clip(trial, 0.0, 1.0)
            sum_trial = np.sum(trial)
            trial = trial * (target_sum / sum_trial if sum_trial != 0 else 1.0)
            new_population.append(trial)
        population = new_population
        for trial in population:
            c5 = objective(trial.flatten())
            if c5 < best_c5:
                best_c5 = c5
                best_h = trial.copy()
        if _ % 10 == 0:
            for i in range(pop_size):
                noise = np.random.uniform(-0.05, 0.05, size=n_points)
                population[i] += noise
                population[i] = np.clip(population[i], 0.0, 1.0)
                sum_p = np.sum(population[i])
                population[i] = population[i] * (target_sum / sum_p if sum_p != 0 else 1.0)

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[c] + 0.5 * (population[a] - population[b])
            trial = np.clip(mutant, 0.0, 1.0)
            sum_trial = np.sum(trial)
            trial = trial * (target_sum / sum_trial if sum_trial != 0 else 1.0)
            if np.random.rand() < 0.9:
                trial = population[np.random.choice(pop_size)] * (1 - 0.9) + trial * 0.9
            trial = np.clip(trial, 0.0, 1.0)
            sum_trial = np.sum(trial)
            trial = trial * (target_sum / sum_trial if sum_trial != 0 else 1.0)
            c5 = objective(trial.flatten())
            if c5 < best_c5:
                best_c5 = c5
                best_h = trial.copy()

    return best_h.tolist(), best_c5, n_points
