import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    
    def de_augmented(pop, args):
        left = args[0]
        right = args[1]
        n = len(pop)
        pop_size = len(pop[0])
        mutant_pop = []
        for i in range(n):
            idxs = [j for j in range(n) if j != i]
            a, b, c = pop[np.random.choice(idxs, 3, replace=False)]
            mutant = a + np.random.uniform(-1, 1, pop_size) * (b - c)
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = mutant * (target_sum / np.sum(mutant) if np.sum(mutant) != 0 else 1.0)
            mutant_pop.append(mutant)
        mutant_pop = np.array(mutant_pop)
        cross_pop = []
        for i in range(n):
            a, b = pop[i], mutant_pop[i]
            mask = np.random.rand(pop_size) < 0.8
            cross = np.where(mask, a, b)
            cross = np.clip(cross, 0.0, 1.0)
            cross = cross * (target_sum / np.sum(cross) if np.sum(cross) != 0 else 1.0)
            cross_pop.append(cross)
        cross_pop = np.array(cross_pop)
        return cross_pop
    
    def de_objective(pop):
        obj_vals = []
        for h in pop:
            obj_vals.append(objective(h))
        return np.array(obj_vals)
    
    def de_fitness(x):
        return de_objective(x)
    
    def de_bound(x):
        return np.array([np.clip(x[i], 0.0, 1.0) for i in range(len(x))])
    
    def de_pop_init(n, size):
        return np.random.uniform(0.0, 1.0, (n, size))
    
    def de_step(pop):
        mutant_pop = de_augmented(pop, (0.0, 1.0))
        return mutant_pop
    
    def de_stop(pop, fitness, best_fitness):
        return np.std(fitness) < 1e-6 or np.max(fitness) - np.min(fitness) < 1e-6
    
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    
    pop_size = 20
    pop = de_pop_init(pop_size, n_points)
    pop = np.array([np.clip(p, 0.0, 1.0) for p in pop])
    pop = np.array([p * (target_sum / np.sum(p) if np.sum(p) != 0 else 1.0) for p in pop])
    
    for _ in range(200):
        if time.time() - start_time > budget_s - 10:
            break
        mutant_pop = de_augmented(pop, (0.0, 1.0))
        mutant_pop = np.array([np.clip(p, 0.0, 1.0) for p in mutant_pop])
        mutant_pop = np.array([p * (target_sum / np.sum(p) if np.sum(p) != 0 else 1.0) for p in mutant_pop])
        fitness = de_objective(mutant_pop)
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = mutant_pop[best_idx].copy()
        pop = mutant_pop
    
    if time.time() - start_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
    
    return best_h.tolist(), best_c5, n_points
