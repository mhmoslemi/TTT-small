import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # Genetic algorithm setup
    POP_SIZE = 50
    MAX_GEN = 50
    MUT_STDEV = 0.05
    TOURNAMENT_SIZE = 5

    population = [np.copy(h) for _ in range(POP_SIZE)]
    for _ in range(POP_SIZE):
        noise = np.random.normal(0, MUT_STDEV, n_points)
        h_temp = np.clip(h + noise, 0.0, 1.0)
        sum_h_temp = np.sum(h_temp)
        scale = target_sum / sum_h_temp if sum_h_temp != 0 else 1.0
        h_temp = h_temp * scale
        population[_] = h_temp

    for _ in range(MAX_GEN):
        # Evaluate fitness
        fitness = np.zeros(POP_SIZE)
        for i in range(POP_SIZE):
            h_flat = population[i].flatten()
            fitness[i] = objective(h_flat)
        
        # Tournament selection
        parents = []
        for _ in range(POP_SIZE):
            contestants = np.random.choice(POP_SIZE, TOURNAMENT_SIZE, replace=False)
            winner = contestants[np.argmin(fitness[contestants])]
            parents.append(population[winner])
        
        # Crossover
        offspring = []
        for i in range(0, POP_SIZE, 2):
            if i + 1 < POP_SIZE:
                parent1 = parents[i]
                parent2 = parents[i + 1]
                # Two-point crossover
                point1 = np.random.randint(1, n_points - 1)
                point2 = np.random.randint(point1, n_points)
                child1 = np.concatenate((parent1[:point1], parent2[point1:point2], parent1[point2:]))
                child2 = np.concatenate((parent2[:point1], parent1[point1:point2], parent2[point2:]))
                offspring.append(child1)
                offspring.append(child2)
        
        # Mutation
        for i in range(len(offspring)):
            offspring[i] = np.clip(offspring[i] + np.random.normal(0, MUT_STDEV, n_points), 0.0, 1.0)
            sum_h = np.sum(offspring[i])
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            offspring[i] = offspring[i] * scale
        
        # Replace population
        for i in range(POP_SIZE):
            population[i] = np.clip(population[i] + np.random.normal(0, MUT_STDEV, n_points), 0.0, 1.0)
            sum_h = np.sum(population[i])
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            population[i] = population[i] * scale
        
        # Selection based on fitness
        fitness = np.zeros(POP_SIZE)
        for i in range(POP_SIZE):
            h_flat = population[i].flatten()
            fitness[i] = objective(h_flat)
        
        indices = np.argsort(fitness)
        population = [population[i] for i in indices[:POP_SIZE]]
        
        # Update best solution
        for i in range(POP_SIZE):
            h_flat = population[i].flatten()
            c5 = objective(h_flat)
            if c5 < best_c5:
                best_c5 = c5
                best_h = population[i].copy()
        
        if time.time() - start_time > budget_s - 5:
            break

    # Final refinement with local optimization
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
