import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    def differential_evolution(population, generations=50, size=10, strategy='best1bin'):
        pop_size = len(population)
        for _ in range(generations):
            for i in range(pop_size):
                a, b, c = np.random.choice(pop_size, 3, replace=False)
                mutant = population[a] + np.random.rand() * (population[b] - population[c])
                mutant = np.clip(mutant, 0.0, 1.0)
                mutant = mutant * (target_sum / np.sum(mutant)) if np.sum(mutant) != 0 else mutant
                mutant = np.clip(mutant, 0.0, 1.0)
                mutant = mutant * (target_sum / np.sum(mutant)) if np.sum(mutant) != 0 else mutant
                target = population[i]
                if objective(mutant) < objective(target):
                    population[i] = mutant
            # Track best solution
            best = np.argmin([objective(h) for h in population])
            best_c5_candidate = objective(population[best])
            if best_c5_candidate < best_c5:
                best_c5 = best_c5_candidate
                best_h = population[best].copy()
            if time.time() - start_time > budget_s - 5:
                break
        return population

    population = np.random.uniform(0.0, 1.0, (n_points, 10))
    population = population * (target_sum / np.sum(population, axis=0)) if np.sum(population, axis=0) != 0 else population
    population = np.clip(population, 0.0, 1.0)
    population = population * (target_sum / np.sum(population, axis=0)) if np.sum(population, axis=0) != 0 else population

    population = differential_evolution(population, generations=50, size=10, strategy='best1bin')

    best_h = population[np.argmin([objective(h) for h in population])]
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (target_sum / np.sum(best_h)) if np.sum(best_h) != 0 else best_h
    best_c5 = objective(best_h)

    if time.time() - start_time < budget_s - 5:
        result = opt.minimize(
            objective, best_h, 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    elapsed_time = time.time() - start_time
    return best_h.tolist(), best_c5, n_points
