import numpy as np
import time
import scipy.optimize as opt

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()

    pop_size = 50
    pop = np.random.uniform(0.0, 1.0, (pop_size, n_points))
    pop = (pop * target_sum / np.sum(pop, axis=1, keepdims=True)).clip(0.0, 1.0)

    for _ in range(300):
        if time.time() - start_time > budget_s - 5:
            break
        mutant = np.copy(pop[np.random.randint(pop_size, size=pop_size)])
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant[i] = pop[a] + 0.5 * (pop[b] - pop[c])
        mutant = np.clip(mutant, 0.0, 1.0)
        mutant = (mutant * target_sum / np.sum(mutant, axis=1, keepdims=True)).clip(0.0, 1.0)
        trial = np.copy(pop)
        for i in range(pop_size):
            if np.random.rand() < 0.9:
                trial[i] = mutant[i]
        trial = np.clip(trial, 0.0, 1.0)
        trial = (trial * target_sum / np.sum(trial, axis=1, keepdims=True)).clip(0.0, 1.0)
        obj = np.array([objective(trial[i]) for i in range(pop_size)])
        if np.min(obj) < best_c5:
            best_c5 = np.min(obj)
            best_h = trial[np.argmin(obj)]
        pop = trial

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
            mutant = np.copy(pop[np.random.randint(pop_size, size=pop_size)])
            for i in range(pop_size):
                a, b, c = np.random.choice(pop_size, 3, replace=False)
                mutant[i] = pop[a] + 0.5 * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = (mutant * target_sum / np.sum(mutant, axis=1, keepdims=True)).clip(0.0, 1.0)
            trial = np.copy(pop)
            for i in range(pop_size):
                if np.random.rand() < 0.9:
                    trial[i] = mutant[i]
            trial = np.clip(trial, 0.0, 1.0)
            trial = (trial * target_sum / np.sum(trial, axis=1, keepdims=True)).clip(0.0, 1.0)
            obj = np.array([objective(trial[i]) for i in range(pop_size)])
            if np.min(obj) < best_c5:
                best_c5 = np.min(obj)
                best_h = trial[np.argmin(obj)]
            pop = trial

    return best_h.tolist(), best_c5, n_points
