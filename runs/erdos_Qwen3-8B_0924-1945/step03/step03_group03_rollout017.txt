import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    pop_size = 50
    pop = np.random.uniform(0.0, 1.0, (pop_size, n_points))
    pop = [p * (target_sum / np.sum(p)) for p in pop]
    best_h = h.copy()
    best_c5 = float('inf')

    start_time = time.time()
    for _ in range(100):
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + 0.5 * (pop[b] - pop[c])
            trial = np.clip(mutant, 0.0, 1.0)
            trial = trial * (target_sum / np.sum(trial))
            c5 = objective(trial.flatten())
            if c5 < objective(pop[i].flatten()):
                pop[i] = trial
        current_pop = [objective(p.flatten()) for p in pop]
        if np.min(current_pop) < best_c5:
            best_c5 = np.min(current_pop)
            best_h = pop[np.argmin(current_pop)].copy()
        if time.time() - start_time > budget_s - 5:
            break
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(50):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + 0.5 * (pop[b] - pop[c])
            trial = np.clip(mutant, 0.0, 1.0)
            trial = trial * (target_sum / np.sum(trial))
            c5 = objective(trial.flatten())
            if c5 < objective(pop[np.random.randint(pop_size)].flatten()):
                pop[np.random.randint(pop_size)] = trial
        current_pop = [objective(p.flatten()) for p in pop]
        best_c5 = np.min(current_pop)
        best_h = pop[np.argmin(current_pop)].copy()

    return best_h.tolist(), best_c5, n_points
