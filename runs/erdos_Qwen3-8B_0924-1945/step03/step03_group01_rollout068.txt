import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    n_pop = 50
    n_gen = 50
    mut_std = 0.05
    pop = [h.copy() for _ in range(n_pop)]
    for _ in range(n_gen):
        fitness = [objective(p) for p in pop]
        idxs = np.argsort(fitness)
        parents = [pop[i] for i in idxs[:n_pop // 2]]
        offspring = []
        for _ in range(n_pop // 2):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = (p1 + p2) / 2
            noise = np.random.normal(0, mut_std, child.shape)
            child += noise
            child = np.clip(child, 0.0, 1.0)
            sum_child = np.sum(child)
            scale = target_sum / sum_child if sum_child != 0 else 1.0
            child = child * scale
            offspring.append(child)
        pop = [p for p in pop if np.random.rand() < 0.9] + offspring
        pop = [np.clip(p, 0.0, 1.0) for p in pop]
        for p in pop:
            sum_p = np.sum(p)
            scale = target_sum / sum_p if sum_p != 0 else 1.0
            p = p * scale
        fitness = [objective(p) for p in pop]
        idxs = np.argsort(fitness)
        pop = [pop[i] for i in idxs[:n_pop]]
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = pop[0].copy()
        if time.time() - start_time > budget_s - 5:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
