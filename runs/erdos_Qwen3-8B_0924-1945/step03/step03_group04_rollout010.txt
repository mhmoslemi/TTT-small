import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))
    h = h.copy()

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    T_initial = 1.0
    T_max = 200
    T_final = 1e-6
    perturb_rate = 0.2
    noise_std = 0.05
    improvement_threshold = 1e-4
    sigma_decay = 0.5
    sigma = noise_std
    stagnation_counter = 0
    best_c5 = float('inf')

    for t in range(T_max):
        if time.time() - start_time > budget_s - 5:
            break
        if t % 10 == 0:
            if stagnation_counter > 100:
                sigma *= sigma_decay
                stagnation_counter = 0
        indices = np.random.choice(n_points, int(n_points * perturb_rate), replace=False)
        noise = np.random.normal(0, sigma, size=n_points)
        h_perturb = h.copy()
        h_perturb[indices] += noise[indices]
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        sum_h_perturb = np.sum(h_perturb)
        scale = target_sum / sum_h_perturb if sum_h_perturb != 0 else 1.0
        h_perturb = h_perturb * scale
        c5_perturb = objective(h_perturb)
        if c5_perturb < best_c5:
            best_h = h_perturb.copy()
            best_c5 = c5_perturb
            stagnation_counter = 0
        else:
            delta_c5 = c5_perturb - best_c5
            if np.random.rand() < np.exp(-delta_c5 / T_initial * (1 - t / T_max)):
                best_h = h_perturb.copy()
                best_c5 = c5_perturb
                stagnation_counter = 0
            else:
                stagnation_counter += 1
        T_current = T_initial * (1 - t / T_max)
        h = best_h.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = np.linalg.lstsq(np.eye(n_points), best_h, rcond=None)[0]
        result = result * (target_sum / np.sum(result))
        result = np.clip(result, 0.0, 1.0)
        c5_refined = objective(result)
        if c5_refined < best_c5:
            best_h = result.copy()
            best_c5 = c5_refined

    return best_h.tolist(), best_c5, n_points
