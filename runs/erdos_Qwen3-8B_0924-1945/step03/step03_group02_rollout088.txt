```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()

    # Evolutionary optimization with Differential Evolution
    def evolve():
        nonlocal best_c5, best_h
        pop_size = 20
        mutation = 0.9
        crossover = 0.8
        max_iter = 200
        pop = np.random.uniform(0, 1, (pop_size, n_points))
        for _ in range(max_iter):
            for i in range(pop_size):
                idxs = np.random.choice(pop_size, 3, replace=False)
                mutant = pop[idxs[0]] + mutation * (pop[idxs[1]] - pop[idxs[2]])
                mutant = np.clip(mutant, 0, 1)
                mutant = mutant * (target_sum / np.sum(mutant))
                trial = np.copy(pop[i])
                for j in range(n_points):
                    if np.random.rand() < crossover:
                        trial[j] = mutant[j]
                trial = np.clip(trial, 0, 1)
                trial = trial * (target_sum / np.sum(trial))
                if objective(trial.flatten()) < objective(pop[i].flatten()):
                    pop[i] = trial
            # Update best
            for i in range(pop_size):
                if objective(pop[i].flatten()) < best_c5:
                    best_c5 = objective(pop[i].flatten())
                    best_h = pop[i].copy()
            if time.time() - start_time > budget_s - 10:
                break

    evolve()

    # Local refinement with L-BFGS-B
    if time.time() - start_time < budget_s - 10:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
