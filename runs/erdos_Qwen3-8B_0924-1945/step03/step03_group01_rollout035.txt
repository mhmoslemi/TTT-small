import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        m = n_points // 2
        x = np.linspace(0, 1, m + 1)[1:]
        x = 0.5 * (1 + 0.5 * np.sin(2 * np.pi * np.arange(m) / m))
        x = 0.5 * x + 0.5
        h = np.zeros(n_points)
        h[np.floor(x * n_points).astype(int)] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    def differential_evolution():
        nonlocal best_c5, best_h
        pop_size = 50
        mutation = 0.8
        crossover = 0.9
        pop = np.random.uniform(0, 1, (pop_size, n_points))
        for _ in range(50):
            for i in range(pop_size):
                a, b, c = np.random.choice(pop_size, 3, replace=False)
                mutant = pop[a] + mutation * (pop[b] - pop[c])
                mutant = np.clip(mutant, 0, 1)
                trial = pop[i].copy()
                for j in range(n_points):
                    if np.random.rand() < crossover:
                        trial[j] = mutant[j]
                trial = trial * (target_sum / np.sum(trial)) if np.sum(trial) != 0 else trial
                trial = np.clip(trial, 0, 1)
                obj_pop = objective(pop[i].flatten())
                obj_trial = objective(trial.flatten())
                if obj_trial < obj_pop:
                    pop[i] = trial
            best_pop = pop[np.argmin([objective(p.flatten()) for p in pop])]
            best_pop = best_pop * (target_sum / np.sum(best_pop)) if np.sum(best_pop) != 0 else best_pop
            best_pop = np.clip(best_pop, 0, 1)
            c5 = objective(best_pop.flatten())
            if c5 < best_c5:
                best_c5 = c5
                best_h = best_pop.copy()
            if time.time() - start_time > budget_s - 5:
                break

    differential_evolution()

    def local_refinement():
        nonlocal best_c5, best_h
        result = opt.minimize(
            objective, best_h.flatten(),
            method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint},
            tol=1e-8, options={'maxiter': 100}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    local_refinement()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(),
            method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint},
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
