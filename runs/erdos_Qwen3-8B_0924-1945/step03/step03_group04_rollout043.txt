```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    
    def update_particle(h, pbest, gbest, omega, alpha, beta):
        v = np.random.rand(n_points) * (pbest - h) * alpha + np.random.rand(n_points) * (gbest - h) * beta
        h_new = h + v * omega
        h_new = np.clip(h_new, 0.0, 1.0)
        sum_h_new = np.sum(h_new)
        scale = target_sum / sum_h_new if sum_h_new != 0 else 1.0
        h_new = h_new * scale
        return h_new
    
    particles = [h.copy()]
    pbest = [h.copy()]
    gbest = h.copy()
    best_c5 = objective(h)
    
    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(len(particles)):
            h = particles[i]
            pbest_i = pbest[i]
            h_new = update_particle(h, pbest_i, gbest, 0.8, 1.5, 1.5)
            c5 = objective(h_new)
            if c5 < objective(pbest_i):
                pbest[i] = h_new.copy()
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
            if c5 < objective(gbest):
                gbest = h_new.copy()
        particles = pbest
        if _ % 10 == 0:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            particles = [h + noise for h in particles]
            particles = [np.clip(h, 0.0, 1.0) for h in particles]
            particles = [h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h for h in particles]
    
    if time.time() - start_time < budget_s - 10:
        result = opt.minimize(objective, best_h.flatten(), 
                              method='L-BFGS-B', bounds=bounds, 
                              tol=1e-8, options={'maxiter': 50})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
    
    return best_h.tolist(), best_c5, n_points
```
