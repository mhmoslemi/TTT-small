import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # Global optimization: Differential Evolution
    def global_optimize():
        pop_size = 50
        max_iter = 100
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        pop = np.random.uniform(0.0, 1.0, (pop_size, n_points))
        pop = np.array([h * (target_sum / np.sum(h)) for h in pop], dtype=np.float64)
        for _ in range(max_iter):
            for i in range(pop_size):
                j, k, l = np.random.choice(pop_size, 3, replace=False)
                mutant = pop[j] + 0.5 * (pop[k] - pop[l])
                mutant = np.clip(mutant, 0.0, 1.0)
                mutant = mutant * (target_sum / np.sum(mutant))
                if np.random.rand() < 0.9:
                    trial = mutant
                else:
                    trial = pop[i]
                trial = np.clip(trial, 0.0, 1.0)
                trial = trial * (target_sum / np.sum(trial))
                obj_trial = objective(trial.flatten())
                obj_current = objective(pop[i].flatten())
                if obj_trial < obj_current:
                    pop[i] = trial
            current_best = pop[np.argmin([objective(h.flatten()) for h in pop])]
            if objective(current_best.flatten()) < best_c5:
                best_c5 = objective(current_best.flatten())
                best_h = current_best.copy()
            if time.time() - start_time > budget_s - 5:
                break

    # Local optimization: L-BFGS-B
    def local_optimize(h_initial):
        result = opt.minimize(objective, h_initial.flatten(), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        return h_opt, c5_bound

    global_optimize()
    if time.time() - start_time < budget_s:
        best_h, best_c5 = local_optimize(best_h)

    return best_h.tolist(), best_c5, n_points
