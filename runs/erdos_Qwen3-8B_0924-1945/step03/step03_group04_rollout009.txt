import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[:n_points // 2] = 1
        h = h * (target_sum / np.sum(h))

    def calc_c5(h_vals):
        h = h_vals.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    best_c5 = calc_c5(h)
    best_h = h.copy()
    start_time = time.time()
    temp = 1.0
    temp_decay = 0.995
    max_iter = 200
    best_no_improve = 0
    perturb_size = 20
    noise_std = 0.05

    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        if best_no_improve > 100:
            noise_std = 0.02
        else:
            noise_std = 0.05
        indices = np.random.choice(n_points, size=perturb_size, replace=False)
        h_pert = h.copy()
        h_pert[indices] += np.random.normal(0, noise_std, size=perturb_size)
        h_pert = np.clip(h_pert, 0.0, 1.0)
        sum_h = np.sum(h_pert)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_pert = h_pert * scale
        c5_new = calc_c5(h_pert)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_pert.copy()
            best_no_improve = 0
        else:
            delta = c5_new - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                best_h = h_pert.copy()
            best_no_improve += 1
        temp *= temp_decay

    elapsed = time.time() - start_time
    if elapsed < budget_s:
        result = np.zeros(n_points)
        result[:] = best_h[:]
        result = np.clip(result, 0.0, 1.0)
        sum_h = np.sum(result)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        result = result * scale
        c5_refined = calc_c5(result)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = result.copy()

    return best_h.tolist(), best_c5, n_points
