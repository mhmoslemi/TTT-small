import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[::2] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()
    pop_size = 50
    particles = np.random.rand(pop_size, n_points)
    particles = particles * 2.0 - 1.0
    particles = np.clip(particles, 0.0, 1.0)
    particles = particles * (target_sum / np.sum(particles, axis=1, keepdims=True))
    pbest = particles.copy()
    gbest = particles[0].copy()
    best_c5 = float('inf')

    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        velocities = np.random.rand(pop_size, n_points) * 0.1
        for i in range(pop_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = 0.7 * velocities[i] + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
            current_c5 = objective(particles[i].flatten())
            if current_c5 < objective(pbest[i].flatten()):
                pbest[i] = particles[i].copy()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    gbest = particles[i].copy()
        if _ % 10 == 0:
            best_h = gbest.copy()
            if np.isnan(best_c5):
                best_c5 = objective(best_h.flatten())
        if _ % 20 == 0:
            h = gbest.copy()
            h = h * (target_sum / np.sum(h))
            c5 = objective(h.flatten())
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    if time.time() - start_time < budget_s:
        result = opt.minimize(objective, best_h.flatten(), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 50})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
