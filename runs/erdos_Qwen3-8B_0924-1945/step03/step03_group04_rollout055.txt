import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        mask = np.zeros(n_points, dtype=bool)
        mask[::5] = True
        h = np.zeros(n_points)
        h[mask] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    pop_size = 50
    pop = [h.copy() for _ in range(pop_size)]
    for _ in range(pop_size):
        pop[_] = np.clip(pop[_], 0.0, 1.0)
        sum_h = np.sum(pop[_])
        pop[_] *= (target_sum / sum_h if sum_h != 0 else 1.0)

    for gen in range(100):
        scores = [objective(p.copy().flatten()) for p in pop]
        elite_idx = np.argsort(scores)[:int(pop_size * 0.2)]
        elites = [pop[i].copy() for i in elite_idx]
        offspring = []

        for _ in range(pop_size // 2):
            parent1 = np.random.choice(elites)
            parent2 = np.random.choice(elites)
            child = (parent1 + parent2) / 2
            child = np.clip(child, 0.0, 1.0)
            sum_h = np.sum(child)
            child *= (target_sum / sum_h if sum_h != 0 else 1.0)
            offspring.append(child)
            child = np.random.choice(elites)
            child = np.clip(child, 0.0, 1.0)
            sum_h = np.sum(child)
            child *= (target_sum / sum_h if sum_h != 0 else 1.0)
            offspring.append(child)

        for _ in range(pop_size - len(offspring)):
            child = np.random.uniform(0.0, 1.0, n_points)
            sum_h = np.sum(child)
            child *= (target_sum / sum_h if sum_h != 0 else 1.0)
            offspring.append(child)

        for i in range(pop_size):
            if np.random.rand() < 0.1:
                noise = np.random.normal(0.0, 0.05, n_points)
                pop[i] = np.clip(pop[i] + noise, 0.0, 1.0)
                sum_h = np.sum(pop[i])
                pop[i] *= (target_sum / sum_h if sum_h != 0 else 1.0)

        for i in range(pop_size):
            if np.random.rand() < 0.5:
                pop[i] = offspring[i].copy()

        for i in range(pop_size):
            if scores[i] < best_c5:
                best_c5 = scores[i]
                best_h = pop[i].copy()

        if time.time() - start_time > budget_s - 5:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='SLSQP', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
