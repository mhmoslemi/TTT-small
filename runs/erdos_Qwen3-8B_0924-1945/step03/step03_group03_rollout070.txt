import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()
    population_size = 50
    population = [np.clip(np.random.rand(n_points), 0.0, 1.0) * (target_sum / np.sum(np.clip(np.random.rand(n_points), 0.0, 1.0))) for _ in range(population_size)]
    population = [np.clip(p, 0.0, 1.0) for p in population]

    for _ in range(100):
        if time.time() - start_time > budget_s - 5:
            break
        mutants = []
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = population[a] + 0.5 * (population[b] - population[c])
            mutants.append(np.clip(mutant, 0.0, 1.0))
        trials = []
        for i in range(population_size):
            mutant = mutants[i % len(mutants)]
            trial = np.clip(population[i] + 0.9 * (mutant - population[i]), 0.0, 1.0)
            trials.append(trial)
        trials = [np.clip(t, 0.0, 1.0) for t in trials]
        for i in range(population_size):
            trials[i] = np.clip(trials[i], 0.0, 1.0)
            sum_h = np.sum(trials[i])
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            trials[i] = trials[i] * scale
        for i in range(population_size):
            trial_c5 = objective(trials[i].flatten())
            if trial_c5 < objective(population[i].flatten()):
                population[i] = trials[i]
        current_c5 = min(objective(p.flatten()) for p in population)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = np.array(population[np.argmin([objective(p.flatten()) for p in population])])
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = population[a] + 0.5 * (population[b] - population[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            sum_h = np.sum(mutant)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            mutant = mutant * scale
            trial_c5 = objective(mutant.flatten())
            if trial_c5 < objective(population[np.random.randint(population_size)].flatten()):
                population[np.random.randint(population_size)] = mutant
        best_c5 = min(objective(p.flatten()) for p in population)
        best_h = np.array(population[np.argmin([objective(p.flatten()) for p in population])])
    return best_h.tolist(), best_c5, n_points
