import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()

    def pso_update(pop, velocities, pbest, gbest, omega, c1, c2, n_iter, budget):
        for _ in range(n_iter):
            if time.time() - start_time > budget - 5:
                break
            for i in range(len(pop)):
                r1, r2 = np.random.rand(2)
                velocities[i] = omega * velocities[i] + c1 * r1 * (pbest[i] - pop[i]) + c2 * r2 * (gbest - pop[i])
                pop[i] = pop[i] + velocities[i]
                pop[i] = np.clip(pop[i], 0.0, 1.0)
                sum_h = np.sum(pop[i])
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                pop[i] = pop[i] * scale
                curr_c5 = objective(pop[i].flatten())
                if curr_c5 < objective(pbest[i].flatten()):
                    pbest[i] = pop[i].copy()
            new_c5 = objective(gbest.flatten())
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = gbest.copy()
            gbest = np.mean(pbest, axis=0)
        return best_h, best_c5

    pop_size = 50
    pop = np.array([h.copy() for _ in range(pop_size)])
    velocities = np.random.uniform(-0.1, 0.1, (pop_size, n_points))
    pbest = pop.copy()
    gbest = pop[0].copy()

    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        pop = pso_update(pop, velocities, pbest, gbest, 0.7, 1.5, 1.5, 10, budget_s - time.time())

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(objective, best_h.flatten(), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 50})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
