```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # Differential Evolution Setup
    def de_objective(x):
        x = np.clip(x, 0.0, 1.0)
        sum_x = np.sum(x)
        scale = target_sum / sum_x if sum_x != 0 else 1.0
        x = x * scale
        corr = np.correlate(x, 1.0 - x, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def de_constraint(x):
        return np.sum(x) - target_sum

    # DE optimization
    from scipy.optimize import differential_evolution
    result = differential_evolution(
        de_objective,
        bounds,
        strategy='best1bin',
        popsize=10,
        maxiter=50,
        tol=1e-8,
        constraints=[{'type': 'eq', 'fun': de_constraint}],
        disp=False
    )
    best_h = result.x
    best_c5 = de_objective(best_h)

    # Final refinement with L-BFGS-B
    result = opt.minimize(
        objective, best_h,
        method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 50}
    )
    best_h = result.x
    best_c5 = objective(best_h)

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h,
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        best_h = result.x
        best_c5 = objective(best_h)

    return best_h.tolist(), best_c5, n_points
```
