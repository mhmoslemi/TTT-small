```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    def _clip_and_scale(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        return h.flatten()

    # Initial differential evolution phase
    from scipy.optimize import differential_evolution
    def objective_de(h_flat):
        h = _clip_and_scale(h_flat)
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    bounds_de = [(0.0, 1.0) for _ in range(n_points)]
    result = differential_evolution(objective_de, bounds_de, strategy='best1bin', popsize=50, tol=1e-6, maxiter=100, mutation=(0.5, 1), recombination=0.8, disp=False)
    best_h = _clip_and_scale(result.x)
    best_c5 = objective(best_h)

    # Local refinement with L-BFGS-B
    def objective_lbb(h_flat):
        h = _clip_and_scale(h_flat)
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    result_lbb = opt.minimize(objective_lbb, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})
    best_h = _clip_and_scale(result_lbb.x)
    best_c5 = objective(best_h)

    # Final refinement
    result_final = opt.minimize(objective_lbb, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 50})
    best_h = _clip_and_scale(result_final.x)
    best_c5 = objective(best_h)

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 5:
        result = opt.minimize(objective_lbb, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 50})
        best_h = _clip_and_scale(result.x)
        best_c5 = objective(best_h)

    return best_h.tolist(), best_c5, n_points
```
