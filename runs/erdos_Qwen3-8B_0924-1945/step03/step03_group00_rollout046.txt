import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[:n_points // 2] = 0.2
        h[n_points // 2::2] = 0.8
        h = np.concatenate([h, h[::-1][1:-1]])
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    learning_rate = 0.05
    momentum = 0.9
    noise_factor = 0.05
    iter_count = 0
    max_iter = 400

    for _ in range(max_iter):
        h_opt = best_h.copy()
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h = np.sum(h_opt)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_opt = h_opt * scale

        corr = np.correlate(h_opt, 1.0 - h_opt, mode="full")
        current_c5 = np.max(corr * dx)

        grad = np.zeros(n_points)
        for k in range(1, len(corr) // 2):
            val = corr[k] * dx
            for i in range(n_points):
                grad[i] += (val - corr[k + 1] * dx) * (1.0 - h_opt[i]) * dx
                grad[i] += (val - corr[k - 1] * dx) * h_opt[i] * dx

        grad -= (np.sum(h_opt) - target_sum) * 1000
        m = momentum * np.zeros(n_points) + grad
        h_opt = np.clip(h_opt - learning_rate * m, 0.0, 1.0)
        sum_h = np.sum(h_opt)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_opt = h_opt * scale

        noise = np.random.uniform(-noise_factor, noise_factor, size=n_points)
        h_opt += noise
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h = np.sum(h_opt)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_opt = h_opt * scale

        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

        if time.time() - start_time > budget_s - 5:
            break

        iter_count += 1
        if iter_count % 50 == 0:
            learning_rate *= 0.98
            noise_factor *= 0.99

    if time.time() - start_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, 
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h = np.sum(h_opt)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
