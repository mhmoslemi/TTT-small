import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    pop_size = 50
    max_gen = 100
    mut_rate = 0.1
    elite_pct = 0.2

    for gen in range(max_gen):
        if time.time() - start_time > budget_s - 5:
            break
        pop = [h.copy() for _ in range(pop_size)]
        for i in range(pop_size):
            if np.random.rand() < mut_rate:
                noise = np.random.normal(0, 0.05, n_points)
                pop[i] += noise
                pop[i] = np.clip(pop[i], 0.0, 1.0)
                pop[i] = pop[i] * (target_sum / np.sum(pop[i]))
        for i in range(pop_size):
            if np.random.rand() < 0.5:
                j = np.random.randint(0, pop_size)
                pop[i] = (pop[i] + pop[j]) / 2
                pop[i] = np.clip(pop[i], 0.0, 1.0)
                pop[i] = pop[i] * (target_sum / np.sum(pop[i]))
        fitness = [objective(p) for p in pop]
        elite_idx = np.argsort(fitness)[:int(elite_pct * pop_size)]
        best_h = pop[elite_idx[0]]
        best_c5 = fitness[elite_idx[0]]
        for i in elite_idx:
            result = opt.minimize(
                objective, pop[i].flatten(),
                method='SLSQP', bounds=bounds, constraints=cons,
                tol=1e-8, options={'maxiter': 50}
            )
            if result.success:
                h_opt = result.x.reshape(n_points)
                h_opt = np.clip(h_opt, 0.0, 1.0)
                sum_h_opt = np.sum(h_opt)
                scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
                h_opt = h_opt * scale
                c5_opt = objective(h_opt.flatten())
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        if gen % 5 == 0:
            result = opt.minimize(
                objective, best_h.flatten(),
                method='L-BFGS-B', bounds=bounds, constraints=cons,
                tol=1e-8, options={'maxiter': 50}
            )
            if result.success:
                h_opt = result.x.reshape(n_points)
                h_opt = np.clip(h_opt, 0.0, 1.0)
                sum_h_opt = np.sum(h_opt)
                scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
                h_opt = h_opt * scale
                c5_opt = objective(h_opt.flatten())
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(),
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        if result.success:
            h_opt = result.x.reshape(n_points)
            h_opt = np.clip(h_opt, 0.0, 1.0)
            sum_h_opt = np.sum(h_opt)
            scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
            h_opt = h_opt * scale
            c5_opt = objective(h_opt.flatten())
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
