```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            return 1.0
        h = h * (target_sum / sum_h)
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    for _ in range(250):
        if time.time() - start_time > budget_s - 10:
            break
        population = np.random.uniform(0, 1, (50, n_points))
        population = np.clip(population, 0.0, 1.0)
        for i in range(population.shape[0]):
            if np.sum(population[i]) == 0:
                population[i] = h.copy()
            else:
                population[i] = population[i] * (target_sum / np.sum(population[i]))
        best_candidate = population[0]
        for i in range(1, population.shape[0]):
            candidate = population[i]
            if np.sum(candidate) == 0:
                c5 = 1.0
            else:
                candidate = candidate * (target_sum / np.sum(candidate))
                corr = np.correlate(candidate, 1.0 - candidate, mode="full")
                c5 = np.max(corr * dx)
            if c5 < objective(best_candidate):
                best_candidate = candidate
        h = best_candidate.copy()
        if np.random.rand() < 0.1:
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                h = h * (target_sum / sum_h)

    if time.time() - start_time < budget_s:
        result = opt.minimize(
            objective, h.flatten(), 
            method='L-BFGS-B', bounds=bounds, 
            tol=1e-8, options={'maxiter': 100}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        if sum_h_opt != 0:
            h_opt = h_opt * (target_sum / sum_h_opt)
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, 
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        if sum_h_opt != 0:
            h_opt = h_opt * (target_sum / sum_h_opt)
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
