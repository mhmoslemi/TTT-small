import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = time.time()
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=n_points // 2, replace=False)
        h[indices] = 1.0
        h = h * (target_sum / np.sum(h))
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    
    def surrogate_objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = corr * dx
        return (1 / 5000) * np.log(np.sum(np.exp(5000 * overlaps)))
    
    def constraint(h_flat):
        return np.sum(h_flat) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    
    def optimize():
        nonlocal best_c5, best_h
        h_flat = h.copy()
        result = opt.minimize(
            surrogate_objective, h_flat,
            bounds=bounds, constraints=cons, method='SLSQP',
            tol=1e-8, options={'maxiter': 100}
        )
        h_opt = result.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
        return best_c5
    
    for _ in range(5):
        h = best_h.copy()
        result = opt.minimize(
            objective, h,
            bounds=bounds, constraints=cons,
            method='L-BFGS-B', tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
        if time.time() - start_time > budget_s - 5:
            break
        h = best_h.copy()
        noise = np.random.normal(0, 0.02, size=n_points)
        h = h + noise
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
    
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(3):
            h = best_h.copy()
            result = opt.minimize(
                objective, h,
                bounds=bounds, constraints=cons,
                method='L-BFGS-B', tol=1e-8, options={'maxiter': 30}
            )
            h_opt = result.x
            h_opt = np.clip(h_opt, 0.0, 1.0)
            sum_h_opt = np.sum(h_opt)
            scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
            h_opt = h_opt * scale
            c5_bound = objective(h_opt)
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h = h_opt.copy()
    
    return best_h.tolist(), best_c5, n_points
