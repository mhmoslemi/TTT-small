import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h_values if initial_h_values is not None else np.zeros(n_points)
    
    # Generate Halton sequence for structured initial guess
    def halton_index(i, base):
        index = 0
        f = 1.0 / base
        while i > 0:
            i, remainder = divmod(i, base)
            index += remainder * f
            f *= 1.0 / base
        return index
    
    h = np.zeros(n_points)
    idx = np.argsort([halton_index(i, 2) for i in range(n_points)])
    h[idx[:int(n_points/2)]] = 1
    h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def cmaes_objective(x):
        return evaluate(x)

    def cmaes_constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': cmaes_constraint}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate(best_h)

    # CMA-ES parameters
    mu = 50
    sigma = 0.5
    pop_size = 50
    max_iterations = 150
    early_stop = 50
    best_c5_history = [best_c5]

    from cma import CMAEvolutionStrategy

    # Adapting to the optimization problem
    try:
        es = CMAEvolutionStrategy(h.copy(), sigma, {'popsize': pop_size})
        for iteration in range(max_iterations):
            solutions = es.ask()
            for i in range(len(solutions)):
                solutions[i] = np.clip(solutions[i], 0.0, 1.0)
                sum_h = np.sum(solutions[i])
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                solutions[i] = solutions[i] * scale
            values = [evaluate(solution) for solution in solutions]
            es.tell(solutions, values)
            es.result()
            if iteration % 10 == 0:
                current_h = es.best_solution()[0]
                current_c5 = evaluate(current_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
                best_c5_history.append(current_c5)
            if len(best_c5_history) > early_stop:
                if np.max(np.abs(np.array(best_c5_history[-early_stop:]) - best_c5)) < 1e-5:
                    break
            if time.time() - start_time > budget_s - 5:
                break
        best_h = es.best_solution()[0]
        best_c5 = evaluate(best_h)
    except Exception as e:
        pass

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(evaluate, best_h, bounds=bounds, constraints=cons, method='L-BFGS-B')
        best_h = result.x
        best_c5 = evaluate(best_h)

    return best_h.tolist(), best_c5, n_points
