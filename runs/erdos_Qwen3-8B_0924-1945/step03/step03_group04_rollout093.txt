import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    n_particles = 40
    particles = [np.random.uniform(0.0, 1.0, n_points) for _ in range(n_particles)]
    particles = [p * (target_sum / np.sum(p)) for p in particles]
    pbest = particles.copy()
    gbest = min(pbest, key=objective)
    best_c5 = objective(gbest)

    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(n_particles):
            h = particles[i]
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            corr = np.correlate(h, 1.0 - h, mode="full")
            c5 = np.max(corr * dx)
            if c5 < objective(pbest[i]):
                pbest[i] = h.copy()
        gbest = min(pbest, key=objective)
        best_c5 = objective(gbest)
        if _ % 10 == 0:
            for i in range(n_particles):
                h = particles[i]
                v = np.random.uniform(-0.05, 0.05, n_points)
                h = h + v
                h = np.clip(h, 0.0, 1.0)
                sum_h = np.sum(h)
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                h = h * scale
                particles[i] = h.copy()

    if time.time() - start_time < budget_s - 5:
        from scipy.optimize import minimize
        result = minimize(
            objective, gbest, method='L-BFGS-B', bounds=bounds,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h = np.sum(h_opt)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
