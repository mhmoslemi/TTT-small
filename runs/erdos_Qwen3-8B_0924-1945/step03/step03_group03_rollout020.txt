import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    population_size = 50

    start_time = time.time()
    particles = [h.copy() for _ in range(population_size)]
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(population_size)]
    pbest = [p.copy() for p in particles]
    gbest = min(pbest, key=lambda x: objective(x.flatten()))

    for _ in range(100):
        for i in range(population_size):
            h = particles[i]
            v = velocities[i]
            h_flat = h.flatten()
            pbest_val = objective(pbest[i].flatten())
            gbest_val = objective(gbest.flatten())
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            v = 0.7 * v + 1.5 * r1 * (pbest[i] - h) + 1.5 * r2 * (gbest - h)
            velocities[i] = v
            h = h + v
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            particles[i] = h
            current_val = objective(h.flatten())
            if current_val < pbest_val:
                pbest[i] = h.copy()
            if current_val < objective(gbest.flatten()):
                gbest = h.copy()
        if time.time() - start_time > budget_s - 5:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(objective, gbest.flatten(), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 50})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < objective(gbest.flatten()):
            best_h = h_opt.copy()
            best_c5 = c5_bound
    else:
        best_h = gbest.copy()
        best_c5 = objective(best_h.flatten())

    return best_h.tolist(), best_c5, n_points
