import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    population_size = 50
    population = [h.copy() for _ in range(population_size)]
    for p in population:
        p = np.clip(p, 0.0, 1.0)
        sum_p = np.sum(p)
        p = p * (target_sum / sum_p)
    population = np.array(population)

    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s - 5:
            break
        mutant = np.zeros_like(population[0])
        a, b, c = np.random.choice(population_size, 3, replace=False)
        mutant[:] = population[a] + 0.5 * (population[b] - population[c])
        mutant = np.clip(mutant, 0.0, 1.0)
        sum_mutant = np.sum(mutant)
        mutant = mutant * (target_sum / sum_mutant)
        trial = np.copy(population[np.random.randint(population_size)])
        crossover_mask = np.random.rand(n_points) < 0.9
        trial[crossover_mask] = mutant[crossover_mask]
        trial = np.clip(trial, 0.0, 1.0)
        sum_trial = np.sum(trial)
        trial = trial * (target_sum / sum_trial)
        c5_trial = objective(trial.flatten())
        if c5_trial < objective(population[np.random.randint(population_size)].flatten()):
            population[np.random.randint(population_size)] = trial
        for p in population:
            if objective(p.flatten()) < best_c5:
                best_c5 = objective(p.flatten())
                best_h = p.copy()
        if gen % 10 == 0:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            best_h += noise
            best_h = np.clip(best_h, 0.0, 1.0)
            sum_h = np.sum(best_h)
            best_h = best_h * (target_sum / sum_h)
            c5_best = objective(best_h.flatten())
            if c5_best < best_c5:
                best_c5 = c5_best
                best_h = best_h.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(objective, best_h.flatten(), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 50})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
