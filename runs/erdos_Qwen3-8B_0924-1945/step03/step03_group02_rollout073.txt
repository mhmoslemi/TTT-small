```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        x_centers = np.linspace(0.5, 1.5, n_points // 2)
        h = np.zeros(n_points)
        h[np.searchsorted(x_grid, x_centers)] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s - 10:
            break
        pop_size = 20
        pop = np.random.uniform(0, 1, (pop_size, n_points))
        pop = np.array([h * (target_sum / np.sum(h)) for h in pop], dtype=np.float64)
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + 0.9 * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * (target_sum / np.sum(mutant))
            trial = mutant + 0.8 * (np.random.rand(n_points) - 0.5)
            trial = np.clip(trial, 0, 1)
            trial = trial * (target_sum / np.sum(trial))
            if np.random.rand() < 0.8:
                pop[i] = trial
        for h_opt in pop:
            c5_candidate = objective(h_opt)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_opt.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
