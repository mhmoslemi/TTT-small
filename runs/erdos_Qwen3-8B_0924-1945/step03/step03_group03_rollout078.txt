import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Generate a structured initial guess using Halton sequence
        h = np.zeros(n_points)
        indices = np.argsort(np.random.uniform(0, 1, n_points))
        h[indices[:n_points // 2]] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # Differential Evolution for global search
    def de_objective(h_flat):
        return objective(h_flat)

    de_bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_de = opt.differential_evolution(de_objective, de_bounds, constraints=cons, strategy='best1bin', popsize=50, tol=1e-6, maxiter=100, mutation=(0.5, 1), recombination=0.9, seed=seed)
    best_h = result_de.x
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    best_h = best_h * scale
    best_c5 = objective(best_h)

    # Local refinement with L-BFGS-B
    result_lbgfs = opt.minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 50, 'disp': False})
    best_h = result_lbgfs.x
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    best_h = best_h * scale
    best_c5 = objective(best_h)

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 5:
        # Additional refinement if time allows
        result_lbgfs = opt.minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10, options={'maxiter': 20, 'disp': False})
        best_h = result_lbgfs.x
        best_h = np.clip(best_h, 0.0, 1.0)
        sum_h = np.sum(best_h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        best_h = best_h * scale
        best_c5 = objective(best_h)

    return best_h.tolist(), best_c5, n_points
