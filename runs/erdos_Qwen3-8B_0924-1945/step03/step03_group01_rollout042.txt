import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        scale = target_sum / sum_h
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    population_size = 50
    generations = 50
    mutation_std = 0.05
    crossover_rate = 0.9

    class GA:
        def __init__(self, initial_population):
            self.population = initial_population
            self.fitnesses = np.array([objective(ind) for ind in self.population])

        def select_parents(self):
            tournament = np.random.choice(np.arange(len(self.population)), size=5, replace=False)
            best_idx = tournament[np.argmin(self.fitnesses[tournament])]
            return self.population[best_idx].copy()

        def crossover(self, parent1, parent2):
            if np.random.rand() < crossover_rate:
                idx1, idx2 = np.random.choice(n_points, size=2, replace=False)
                child = np.copy(parent1)
                child[idx1:idx2] = parent2[idx1:idx2]
                return child
            else:
                return np.copy(parent1)

        def mutate(self, individual):
            noise = np.random.normal(0, mutation_std, size=n_points)
            individual += noise
            individual = np.clip(individual, 0.0, 1.0)
            sum_h = np.sum(individual)
            if sum_h == 0:
                return np.zeros(n_points)
            scale = target_sum / sum_h
            return individual * scale

        def evolve(self):
            new_population = []
            for _ in range(population_size):
                parent1 = self.select_parents()
                parent2 = self.select_parents()
                child = self.crossover(parent1, parent2)
                child = self.mutate(child)
                new_population.append(child)
            self.population = new_population
            self.fitnesses = np.array([objective(ind) for ind in self.population])

    ga = GA([h.copy() for _ in range(population_size)])
    for _ in range(generations):
        ga.evolve()
        best_idx = np.argmin(ga.fitnesses)
        if ga.fitnesses[best_idx] < best_c5:
            best_c5 = ga.fitnesses[best_idx]
            best_h = ga.population[best_idx].copy()
        if time.time() - start_time > budget_s - 30:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        if sum_h_opt == 0:
            h_opt = np.zeros(n_points)
        else:
            scale = target_sum / sum_h_opt
            h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
