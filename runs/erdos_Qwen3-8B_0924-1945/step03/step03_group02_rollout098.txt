import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    perturbation = 0.1
    decay_rate = 0.95
    max_iterations = 200
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_max = np.argmax(corr * dx)
        if k_max >= len(corr) - 1:
            k_max = len(corr) - 2
        k_max = np.clip(k_max, 0, len(corr) - 1)
        h_sub = h[:n_points - k_max]
        h_shift = h[k_max:]
        def sub_objective(h_flat_sub):
            h_sub = h_flat_sub.reshape(len(h_sub))
            h_sub = np.clip(h_sub, 0.0, 1.0)
            sum_sub = np.sum(h_sub)
            scale = target_sum / sum_sub if sum_sub != 0 else 1.0
            h_sub = h_sub * scale
            corr_sub = np.correlate(h_sub, 1.0 - h_shift, mode="full")
            return -corr_sub[0]  # minimize the overlap
        result = opt.minimize(
            sub_objective, h[:n_points - k_max], 
            method='SLSQP', bounds=[(0.0, 1.0) for _ in range(len(h_sub))], constraints=cons
        )
        h_opt = result.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        h[:n_points - k_max] = h_opt
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        c5_bound = objective(h.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h.copy()
        if np.random.rand() < 0.1:
            noise = np.random.uniform(-perturbation, perturbation, size=n_points)
            h += noise
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
        perturbation *= decay_rate

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
