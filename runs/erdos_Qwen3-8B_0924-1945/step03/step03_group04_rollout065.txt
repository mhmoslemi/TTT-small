import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # Differential Evolution setup
    def de_objective(x):
        x = np.clip(x, 0.0, 1.0)
        sum_x = np.sum(x)
        scale = target_sum / sum_x if sum_x != 0 else 1.0
        x = x * scale
        corr = np.correlate(x, 1.0 - x, mode="full")
        return np.max(corr * dx)

    def de_constraint(x):
        return np.sum(x) - target_sum

    de_bounds = [(0.0, 1.0) for _ in range(n_points)]
    de_cons = [{'type': 'eq', 'fun': de_constraint}]

    # Run differential evolution
    de_result = opt.differential_evolution(
        de_objective, de_bounds, constraints=de_cons,
        strategy='best1bin', popsize=10, maxiter=50, tol=1e-8,
        polish=True, mutation=(0.5, 1), recombination=0.8,
        disp=False, x0=h.flatten()
    )
    best_de_h = de_result.x
    best_de_h = np.clip(best_de_h, 0.0, 1.0)
    sum_de_h = np.sum(best_de_h)
    scale = target_sum / sum_de_h if sum_de_h != 0 else 1.0
    best_de_h = best_de_h * scale
    best_c5 = objective(best_de_h)

    # Refine with L-BFGS-B
    result = opt.minimize(
        objective, best_de_h.flatten(), 
        method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 50}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt.flatten())
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 5:
        # Add random noise to maintain diversity
        noise = np.random.uniform(-0.05, 0.05, size=n_points)
        best_h = best_h + noise
        best_h = np.clip(best_h, 0.0, 1.0)
        sum_h = np.sum(best_h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        best_h = best_h * scale
        c5_bound = objective(best_h.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound

    return best_h.tolist(), best_c5, n_points
