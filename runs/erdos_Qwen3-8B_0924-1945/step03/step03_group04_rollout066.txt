import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    pop_size = 50
    pop = [h.copy() for _ in range(pop_size)]
    best_h = h.copy()
    best_c5 = objective(best_h)
    gen = 0
    gen_limit = 100
    time_limit = budget_s - 5

    while gen < gen_limit and time.time() < time_limit:
        fitness = [objective(ind) for ind in pop]
        elite_indices = np.argsort(fitness)[:int(pop_size * 0.2)]
        elites = [pop[i] for i in elite_indices]

        offspring = []
        for _ in range(pop_size - len(elites)):
            parent1, parent2 = np.random.choice(elites, size=2, replace=False)
            child = (parent1 + parent2) / 2
            noise = np.random.normal(0, 0.05, size=n_points)
            child += noise
            child = np.clip(child, 0.0, 1.0)
            sum_child = np.sum(child)
            scale = target_sum / sum_child if sum_child != 0 else 1.0
            child = child * scale
            offspring.append(child)

        pop = elites + offspring
        pop = [np.clip(ind, 0.0, 1.0) for ind in pop]
        for i in range(len(pop)):
            sum_pop = np.sum(pop[i])
            scale = target_sum / sum_pop if sum_pop != 0 else 1.0
            pop[i] = pop[i] * scale

        for ind in pop:
            if objective(ind) < best_c5:
                best_c5 = objective(ind)
                best_h = ind.copy()

        if gen % 5 == 0:
            for ind in pop[:5]:
                result = opt.minimize(
                    objective, ind,
                    method='SLSQP', bounds=bounds, constraints=cons,
                    tol=1e-8, options={'maxiter': 50}
                )
                h_opt = result.x
                h_opt = np.clip(h_opt, 0.0, 1.0)
                sum_h = np.sum(h_opt)
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                h_opt = h_opt * scale
                if objective(h_opt) < best_c5:
                    best_c5 = objective(h_opt)
                    best_h = h_opt.copy()

        gen += 1

    return best_h.tolist(), best_c5, n_points
