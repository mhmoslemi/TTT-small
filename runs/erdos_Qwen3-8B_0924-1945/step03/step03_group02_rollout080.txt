import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # Stage 1: Global optimization with differential evolution
    def de_objective(x):
        return objective(x)

    def de_constraint(x):
        return constraint(x)

    de_bounds = [(0.0, 1.0) for _ in range(n_points)]
    de_result = opt.differential_evolution(
        de_objective, de_bounds, constraints=[de_constraint],
        strategy='best1bin', popsize=50, tol=1e-6, mutation=(0.85, 1),
        recombination=0.85, maxiter=250, disp=False, polish=True,
        seed=seed
    )
    best_h = de_result.x
    best_c5 = objective(best_h)

    # Stage 2: Local refinement with SLSQP
    result = opt.minimize(
        objective, best_h, 
        method='SLSQP', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 150}
    )
    h_opt = result.x
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt)
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    # Stage 3: Escape local optima with perturbation
    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        noise = np.random.normal(0, 0.05, size=n_points)
        h_perturbed = best_h + noise
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        sum_perturbed = np.sum(h_perturbed)
        scale = target_sum / sum_perturbed if sum_perturbed != 0 else 1.0
        h_perturbed = h_perturbed * scale
        c5_perturbed = objective(h_perturbed)
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h, 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
