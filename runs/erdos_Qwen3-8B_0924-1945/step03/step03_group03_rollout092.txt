import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()
    population_size = 50
    population = [h.copy() for _ in range(population_size)]
    for _ in range(population_size):
        candidate = h.copy()
        candidate += np.random.uniform(-0.05, 0.05, size=n_points)
        candidate = np.clip(candidate, 0.0, 1.0)
        sum_candidate = np.sum(candidate)
        candidate = candidate * (target_sum / sum_candidate)
        population.append(candidate)
    population = np.array(population)
    population = np.clip(population, 0.0, 1.0)
    population = population * (target_sum / np.sum(population, axis=1, keepdims=True))
    best_c5 = float('inf')
    best_h = h.copy()
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = population[a] + 0.5 * (population[b] - population[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = mutant * (target_sum / np.sum(mutant))
            trial = population[i] + 0.9 * (mutant - population[i])
            trial = np.clip(trial, 0.0, 1.0)
            trial = trial * (target_sum / np.sum(trial))
            c5 = objective(trial.flatten())
            if c5 < best_c5:
                best_c5 = c5
                best_h = trial.copy()
            if np.random.rand() < 0.5:
                population[i] = trial.copy()
        if _ % 10 == 0:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            best_h += noise
            best_h = np.clip(best_h, 0.0, 1.0)
            sum_best = np.sum(best_h)
            best_h = best_h * (target_sum / sum_best)
    return best_h.tolist(), best_c5, n_points
