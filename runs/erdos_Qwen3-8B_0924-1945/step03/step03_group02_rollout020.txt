```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[::5] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        scale = target_sum / sum_h
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    k_star = np.argmax(np.correlate(h, 1.0 - h, mode="full")) - n_points + 1
    for _ in range(200):
        if time.time() - start_time > budget_s - 10:
            break
        result = opt.minimize(
            lambda x: objective(x),
            best_h.flatten(),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            tol=1e-8,
            options={'maxiter': 100}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        if sum_h_opt == 0:
            continue
        h_opt = h_opt * (target_sum / sum_h_opt)
        corr = np.correlate(h_opt, 1.0 - h_opt, mode="full")
        current_c5 = np.max(corr * dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            k_star = np.argmax(corr) - n_points + 1
        if np.random.rand() < 0.1:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            h = best_h + noise
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                h = h * (target_sum / sum_h)
            corr = np.correlate(h, 1.0 - h, mode="full")
            k_star_candidate = np.argmax(corr) - n_points + 1
            if np.abs(k_star_candidate - k_star) > 5:
                k_star = k_star_candidate

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            lambda x: objective(x),
            best_h.flatten(),
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            tol=1e-8,
            options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        if sum_h_opt != 0:
            h_opt = h_opt * (target_sum / sum_h_opt)
        corr = np.correlate(h_opt, 1.0 - h_opt, mode="full")
        final_c5 = np.max(corr * dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_opt.copy()
    
    return best_h.tolist(), best_c5, n_points
```
