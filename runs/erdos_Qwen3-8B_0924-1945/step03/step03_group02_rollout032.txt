import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 5 == 0:
                h[i] = 1.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    current_k = 0

    for _ in range(200):
        if time.time() - start_time > budget_s - 10:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr_idx = np.argmax(corr)
        k_candidate = max_corr_idx - n_points + 1
        if k_candidate < 0:
            k_candidate = 0
        elif k_candidate >= n_points:
            k_candidate = n_points - 1
        if k_candidate != current_k:
            current_k = k_candidate
            def sub_objective(h_flat):
                h = h_flat.reshape(n_points)
                h = np.clip(h, 0.0, 1.0)
                sum_h = np.sum(h)
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                h = h * scale
                corr_k = np.correlate(h, 1.0 - h, mode="full")[k_candidate]
                return corr_k * dx

            result = opt.minimize(sub_objective, h.flatten(), bounds=bounds, constraints=cons, method='SLSQP', tol=1e-8, options={'maxiter': 100})
            h = result.x.reshape(n_points)
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            c5 = objective(h.flatten())
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(objective, best_h.flatten(), bounds=bounds, constraints=cons, method='L-BFGS-B', tol=1e-8, options={'maxiter': 100})
        h = result.x.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        c5 = objective(h.flatten())
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    return best_h.tolist(), best_c5, n_points
