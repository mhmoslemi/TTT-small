import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = np.zeros(n_points)
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0.0, 1.0)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    pop_size = 50
    elite_percent = 0.2
    generations = 100
    mutation_rate = 0.1
    max_mutations = int(mutation_rate * n_points)
    max_gen_stagnant = 10
    stagnant_gen = 0

    for gen in range(generations):
        population = [np.copy(h) for _ in range(pop_size)]
        for i in range(pop_size):
            population[i] = np.clip(population[i], 0.0, 1.0)
            sum_pop = np.sum(population[i])
            population[i] = population[i] * (target_sum / sum_pop)
        fitness = np.array([objective(p) for p in population])
        elite_indices = np.argsort(fitness)[:int(elite_percent * pop_size)]
        elites = [population[i] for i in elite_indices]
        population = elites
        for i in range(len(population) - len(elites)):
            population.append(np.random.rand(n_points))
        for i in range(len(population) - len(elites)):
            population[i] = np.clip(population[i], 0.0, 1.0)
            sum_pop = np.sum(population[i])
            population[i] = population[i] * (target_sum / sum_pop)

        for _ in range(len(population) // 2):
            idx1, idx2 = np.random.choice(len(population), size=2, replace=False)
            parent1 = population[idx1]
            parent2 = population[idx2]
            child = (parent1 + parent2) / 2
            if np.random.rand() < mutation_rate:
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[j] += np.random.normal(0, 0.05)
                        child[j] = np.clip(child[j], 0.0, 1.0)
            sum_child = np.sum(child)
            child = child * (target_sum / sum_child)
            population.append(child)
        population = [np.clip(p, 0.0, 1.0) for p in population]
        fitness = np.array([objective(p) for p in population])
        population = [p for _, p in sorted(zip(fitness, population))]
        population = population[:pop_size]
        for p in population:
            sum_p = np.sum(p)
            p = p * (target_sum / sum_p)
            p = np.clip(p, 0.0, 1.0)
            p = p * (target_sum / np.sum(p))
            population = [p for _ in population]
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0]
            stagnant_gen = 0
        else:
            stagnant_gen += 1
        if stagnant_gen >= max_gen_stagnant:
            mutation_rate = 0.15
            pop_size = 20
        if time.time() - start_time > budget_s - 5:
            break

    if time.time() - start_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='SLSQP', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
