import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()
    population_size = 50
    mutation_factor = 0.5
    crossover_rate = 0.9
    mutation_scale = 0.5
    population = [h.copy() for _ in range(population_size)]
    for p in population:
        p = np.clip(p, 0.0, 1.0)
        p = p * (target_sum / np.sum(p))

    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        new_population = []
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = population[a] + mutation_scale * (population[b] - population[c])
            trial = np.clip(mutant, 0.0, 1.0)
            trial = trial * (target_sum / np.sum(trial))
            trial_c5 = objective(trial.flatten())
            if trial_c5 < objective(population[i].flatten()):
                new_population.append(trial)
            else:
                new_population.append(population[i])
        population = new_population
        for p in population:
            p = np.clip(p, 0.0, 1.0)
            p = p * (target_sum / np.sum(p))
        for p in population:
            p_c5 = objective(p.flatten())
            if p_c5 < best_c5:
                best_c5 = p_c5
                best_h = p.copy()
        if _ % 10 == 0:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            best_h += noise
            best_h = np.clip(best_h, 0.0, 1.0)
            best_h = best_h * (target_sum / np.sum(best_h))
    
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        population = [best_h.copy() for _ in range(population_size)]
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
            new_population = []
            for i in range(population_size):
                a, b, c = np.random.choice(population_size, 3, replace=False)
                mutant = population[a] + mutation_scale * (population[b] - population[c])
                trial = np.clip(mutant, 0.0, 1.0)
                trial = trial * (target_sum / np.sum(trial))
                trial_c5 = objective(trial.flatten())
                if trial_c5 < objective(population[i].flatten()):
                    new_population.append(trial)
                else:
                    new_population.append(population[i])
            population = new_population
            for p in population:
                p = np.clip(p, 0.0, 1.0)
                p = p * (target_sum / np.sum(p))
            for p in population:
                p_c5 = objective(p.flatten())
                if p_c5 < best_c5:
                    best_c5 = p_c5
                    best_h = p.copy()
    
    return best_h.tolist(), best_c5, n_points
