import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.round(h).astype(int)
        h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1).astype(float)
    h = h * (target_sum / np.sum(h))
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    
    def compute_overlap(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = corr * dx
        return np.max(overlaps)
    
    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        overlaps = compute_overlap(h)
        if overlaps < best_c5:
            best_c5 = overlaps
            best_h = h.copy()
        k = np.argmin(np.correlate(h, 1.0 - h, mode="full") * dx)
        if k < n_points:
            for i in range(n_points):
                if h[i] == 0 and h[(i + k) % n_points] == 1:
                    h[i] += 1
                    break
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
    
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
            overlaps = compute_overlap(h)
            if overlaps < best_c5:
                best_c5 = overlaps
                best_h = h.copy()
            k = np.argmin(np.correlate(h, 1.0 - h, mode="full") * dx)
            if k < n_points:
                for i in range(n_points):
                    if h[i] == 0 and h[(i + k) % n_points] == 1:
                        h[i] += 1
                        break
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
    
    h = np.clip(best_h, 0, 1)
    sum_h = np.sum(h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h = h * scale
    c5_bound = compute_overlap(h)
    return h.tolist(), c5_bound, n_points
