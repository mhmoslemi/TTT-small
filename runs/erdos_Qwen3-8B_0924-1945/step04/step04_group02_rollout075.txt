```python
import numpy as np
import scipy.optimize as opt
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, WhiteKernel

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0.0, 1.0, n_points)
        h = h * target_sum / np.sum(h)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    samples = []
    values = []

    kernel = RBF(length_scale=1.0, length_scale_bounds=(1e-5, 1e5)) + WhiteKernel(noise_level=1e-5)
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-10, n_restarts_optimizer=25)

    n_initial_samples = 20
    for _ in range(n_initial_samples):
        h_candidate = np.random.uniform(0.0, 1.0, n_points)
        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        samples.append(h_candidate)
        values.append(objective(h_candidate))
        samples[-1] = samples[-1].copy()
        values[-1] = values[-1]

    samples = np.array(samples)
    values = np.array(values)

    for iter in range(500):
        if len(samples) < 2:
            h_candidate = np.random.uniform(0.0, 1.0, n_points)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            samples = np.append(samples, [h_candidate], axis=0)
            values = np.append(values, [objective(h_candidate)], axis=0)
            continue

        gp.fit(samples, values)
        mu, sigma = gp.predict(samples, return_std=True)
        mu = np.reshape(mu, (len(samples), 1))
        sigma = np.reshape(sigma, (len(samples), 1))
        mu = np.hstack((mu, sigma))

        def ei(x):
            x = np.atleast_2d(x)
            mu_x, sigma_x = gp.predict(x, return_std=True)
            mu_x = mu_x.flatten()
            sigma_x = sigma_x.flatten()
            z = (mu_x - best_c5) / (sigma_x + 1e-8)
            ei = (mu_x - best_c5) * np.exp(-z**2 / 2) / np.sqrt(2 * np.pi) + sigma_x * np.exp(-z**2 / 2) / np.sqrt(2 * np.pi) * np.sqrt(1 + (np.sqrt(2 * np.pi) * (mu_x - best_c5) / sigma_x)**2)
            return ei

        def evaluate(x):
            x = np.atleast_2d(x)
            return ei(x).flatten()

        def pf(x):
            x = np.atleast_2d(x)
            return np.reshape(evaluate(x), (len(x), 1))

        def logpdf(x):
            x = np.atleast_2d(x)
            return np.reshape(gp.log_marginal_likelihood(x), (len(x), 1))

        def acq_func(x):
            return np.reshape(evaluate(x), (len(x), 1))

        xnew = opt.minimize(acq_func, h, bounds=bounds, method='L-BFGS-B', tol=1e-8, options={'maxiter': 50})
        xnew = xnew.x

        h_candidate = xnew.copy()
        h_candidate = np.clip(h_candidate, 0.0, 1.0)
        sum_h = np.sum(h_candidate)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_candidate = h_candidate * scale

        samples = np.append(samples, [h_candidate], axis=0)
        values = np.append(values, [objective(h_candidate)], axis=0)

        if objective(h_candidate) < best_c5:
            best_c5 = objective(h_candidate)
            best_h = h_candidate.copy()

        if time.time() - start_time > budget_s - 5:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
