```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.cos(np.pi * x))
        h = h * (target_sum / np.sum(h))
    def exact_objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    def normalize(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        return h * (target_sum / sum_h)
    n_particles = 30
    n_iterations = 200
    omega = 0.7
    phi_p = 1.5
    phi_g = 1.5
    particles = np.random.rand(n_particles, n_points)
    particles = normalize(particles)
    velocities = np.random.rand(n_particles, n_points) * 0.1
    pbest = particles.copy()
    pbest_fitness = np.full(n_particles, np.inf)
    gbest = particles[0].copy()
    gbest_fitness = np.inf
    start_time = time.time()
    for _ in range(n_iterations):
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(n_particles):
            h = particles[i]
            r1, r2 = np.random.rand(2)
            velocities[i] = omega * velocities[i] + phi_p * r1 * (pbest[i] - h) + phi_g * r2 * (gbest - h)
            h_new = h + velocities[i]
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = normalize(h_new)
            fitness = exact_objective(h_new.flatten())
            if fitness < pbest_fitness[i]:
                pbest[i] = h_new.copy()
                pbest_fitness[i] = fitness
            if fitness < gbest_fitness:
                gbest = h_new.copy()
                gbest_fitness = fitness
        particles = pbest.copy()
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            exact_objective, gbest.flatten(), 
            method='L-BFGS-B', bounds=[(0.0, 1.0) for _ in range(n_points)], 
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = exact_objective(h_opt.flatten())
    else:
        h_opt = gbest.copy()
        c5_bound = gbest_fitness
    return h_opt.tolist(), c5_bound, n_points
```
