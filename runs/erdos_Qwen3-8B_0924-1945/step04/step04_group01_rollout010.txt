```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[::4] = 1.0
        h = h * (target_sum / np.sum(h))
    
    def objective(x_flat, alpha=1000, tau=10):
        x = x_flat.reshape(n_points)
        h = 1 / (1 + np.exp(-tau * x))
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = corr * dx
        weights = np.exp(alpha * overlaps)
        return -np.log(np.sum(weights) / np.sum(np.exp(alpha * overlaps)))

    def exact_objective(x_flat):
        x = x_flat.reshape(n_points)
        h = 1 / (1 + np.exp(-10 * x))
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def constraint(x_flat):
        x = x_flat.reshape(n_points)
        h = 1 / (1 + np.exp(-10 * x))
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        return sum_h - target_sum

    bounds = [(0.0, np.log(1e6)) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    for _ in range(150):
        result = opt.minimize(objective, h.flatten(), 
                             method='SLSQP', bounds=bounds, constraints=cons,
                             tol=1e-8, options={'maxiter': 100})
        x_opt = result.x
        h_opt = 1 / (1 + np.exp(-10 * x_opt))
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = exact_objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
        if time.time() - start_time > budget_s - 5:
            break
        h = h_opt.copy()
        if _ % 10 == 0:
            noise = np.random.normal(0, 0.05, size=n_points)
            h = h + noise
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(exact_objective, best_h.flatten(), 
                             method='L-BFGS-B', bounds=bounds, constraints=cons,
                             tol=1e-8, options={'maxiter': 50})
        h_opt = result.x
        h_opt = 1 / (1 + np.exp(-10 * h_opt))
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = exact_objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
