```python
import numpy as np
import scipy.optimize as opt
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, WhiteKernel

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * target_sum

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    kernel = RBF(length_scale=1.0, length_scale_bounds=(1e-5, 1e5)) + WhiteKernel(noise_level=1e-5)
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-10, n_restarts_optimizer=25)

    samples = []
    values = []

    def acquisition(x, gp, y_max):
        if len(samples) == 0:
            return np.random.rand(n_points)
        mu, sigma = gp.predict(x.reshape(1, -1), return_std=True)
        imp = mu - y_max
        res = imp / (sigma + 1e-8)
        return res

    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        if len(samples) == 0:
            x_new = np.random.rand(n_points)
            samples.append(x_new)
            val_new = objective(x_new)
            values.append(val_new)
            gp.fit(samples, values)
            continue
        x_new = np.random.rand(n_points)
        samples.append(x_new)
        val_new = objective(x_new)
        values.append(val_new)
        gp.fit(samples, values)
        y_max = np.min(values)
        x_new = np.random.rand(n_points)
        samples.append(x_new)
        val_new = objective(x_new)
        values.append(val_new)
        gp.fit(samples, values)

    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        x_new = np.random.rand(n_points)
        samples.append(x_new)
        val_new = objective(x_new)
        values.append(val_new)
        gp.fit(samples, values)
        y_max = np.min(values)
        x_new = np.random.rand(n_points)
        samples.append(x_new)
        val_new = objective(x_new)
        values.append(val_new)
        gp.fit(samples, values)

    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        mu, sigma = gp.predict(np.random.rand(n_points).reshape(1, -1), return_std=True)
        acquisition_values = acquisition(np.random.rand(n_points), gp, np.min(values))
        x_new = np.random.rand(n_points)
        samples.append(x_new)
        val_new = objective(x_new)
        values.append(val_new)
        gp.fit(samples, values)

    idx = np.argmin(values)
    best_c5 = values[idx]
    best_h = samples[idx].copy()

    h_opt = best_h.copy()
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale

    result = opt.minimize(
        objective, h_opt.flatten(), 
        method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 50}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt.flatten())
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
