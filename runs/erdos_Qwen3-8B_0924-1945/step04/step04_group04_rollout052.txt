import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 600
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        angles = np.linspace(0, 2 * np.pi, n_points, endpoint=False)
        sawtooth = np.abs(np.sin(angles) - 1.0)
        h = np.clip(1.0 - sawtooth, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # Phase 1: Differential Evolution for global search
    def de_objective(x):
        x = np.clip(x, 0.0, 1.0)
        sum_x = np.sum(x)
        scale = target_sum / sum_x if sum_x != 0 else 1.0
        x_scaled = x * scale
        corr = np.correlate(x_scaled, 1.0 - x_scaled, mode="full")
        return np.max(corr * dx)

    pop_size = 30
    initial_population = np.random.rand(pop_size, n_points)
    initial_population = initial_population * (target_sum / np.sum(initial_population, axis=1, keepdims=True))
    initial_population = np.clip(initial_population, 0.0, 1.0)

    def de_run():
        nonlocal best_c5, best_h
        pop = initial_population.copy()
        for _ in range(100):
            mut = np.random.rand(pop_size, n_points) * 0.5
            cross = np.random.rand(pop_size, n_points) < 0.9
            trial = pop + mut * (pop[np.random.randint(0, pop_size, pop_size), :] - pop)
            trial = np.clip(trial, 0.0, 1.0)
            trial = trial * (target_sum / np.sum(trial, axis=1, keepdims=True))
            fit_pop = np.array([de_objective(ind) for ind in pop])
            fit_trial = np.array([de_objective(ind) for ind in trial])
            for i in range(pop_size):
                if fit_trial[i] < fit_pop[i]:
                    pop[i] = trial[i]
            fit_pop = np.array([de_objective(ind) for ind in pop])
            best_idx = np.argmin(fit_pop)
            best_c5 = fit_pop[best_idx]
            best_h = pop[best_idx].copy()
            if time.time() - start_time > budget_s - 5:
                break

    de_run()

    # Phase 2: Local refinement with L-BFGS-B
    result = opt.minimize(
        objective, best_h.flatten(), 
        method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 500}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt.flatten())
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 500}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
