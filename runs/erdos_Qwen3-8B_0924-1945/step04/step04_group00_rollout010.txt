```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = np.array(initial_h_values, dtype=np.float64) if initial_h_values is not None else np.random.rand(n_points)
    h = np.round(h).astype(int)
    h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        overlaps = corr * dx
        max_overlap = np.max(overlaps)
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        k = np.argmin(overlaps)
        if k >= len(h):
            break
        if k < len(h):
            for i in range(len(h)):
                if h[i] == 0 and h[(i + k) % len(h)] == 1:
                    h[i] = 1
                    break
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (target_sum / sum_h)
        else:
            h = np.zeros_like(h)
    
    for _ in range(100):
        if time.time() - start_time > budget_s - 5:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        overlaps = corr * dx
        max_overlap = np.max(overlaps)
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        k = np.argmin(overlaps)
        if k >= len(h):
            break
        if k < len(h):
            for i in range(len(h)):
                if h[i] == 0 and h[(i + k) % len(h)] == 1:
                    h[i] = 1
                    break
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (target_sum / sum_h)
        else:
            h = np.zeros_like(h)
    
    return best_h.tolist(), best_c5, n_points
```
