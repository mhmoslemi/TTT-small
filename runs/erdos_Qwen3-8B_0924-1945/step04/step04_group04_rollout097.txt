import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 600
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = None

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # Differential evolution for global exploration
    def de_objective(x):
        return objective(x)

    pop_size = 30
    pop = np.random.uniform(0, 1, (pop_size, n_points))
    pop = np.clip(pop, 0, 1)
    pop = pop * (target_sum / np.sum(pop, axis=1, keepdims=True))
    best_pop = pop.copy()

    for _ in range(100):
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + 0.5 * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * (target_sum / np.sum(mutant))
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * (target_sum / np.sum(mutant))
            for j in range(n_points):
                if np.random.rand() < 0.9:
                    pop[i, j] = mutant[j]
            pop[i] = np.clip(pop[i], 0, 1)
            pop[i] = pop[i] * (target_sum / np.sum(pop[i]))
            current_c5 = objective(pop[i])
            if current_c5 < objective(best_pop[i]):
                best_pop[i] = pop[i].copy()

        best_h = best_pop[np.argmin([objective(h) for h in best_pop])]
        if time.time() - start_time > budget_s - 5:
            break

    # Local refinement with L-BFGS-B
    result = opt.minimize(
        objective, best_h.flatten(), 
        method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 500}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt.flatten())

    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 500}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
