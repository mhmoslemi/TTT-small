import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_lag = int(2 / dx)
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        intervals = np.array([0.2, 0.6, 1.0, 1.4, 1.8])
        for i in range(len(intervals)):
            pos = intervals[i]
            idx = np.searchsorted(np.linspace(0, 2, n_points), pos)
            h[idx - 1:idx + 5] = 1
        h = h * (target_sum / np.sum(h))
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)
    
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    
    particles = [h.copy() for _ in range(20)]
    velocities = [np.random.uniform(-0.05, 0.05, n_points) for _ in range(20)]
    personal_best = [p.copy() for p in particles]
    global_best = h.copy()
    best_c5 = objective(h)
    
    for _ in range(200):
        for i in range(20):
            h = particles[i].copy()
            v = velocities[i]
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_scaled = h * scale
            corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
            current_c5 = np.max(corr * dx)
            
            if current_c5 < objective(personal_best[i]):
                personal_best[i] = h.copy()
            
            r1, r2 = np.random.rand(2)
            cognitive = personal_best[i] - particles[i]
            social = global_best - particles[i]
            v = v * 0.5 + r1 * cognitive + r2 * social
            particles[i] += v
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_h_p = np.sum(particles[i])
            particles[i] = particles[i] * (target_sum / sum_h_p if sum_h_p != 0 else 1.0)
            
            h_eval = particles[i].copy()
            sum_h_e = np.sum(h_eval)
            scale_e = target_sum / sum_h_e if sum_h_e != 0 else 1.0
            h_scaled_e = h_eval * scale_e
            corr_e = np.correlate(h_scaled_e, 1.0 - h_scaled_e, mode="full")
            c5_e = np.max(corr_e * dx)
            
            if c5_e < best_c5:
                best_c5 = c5_e
                global_best = h_eval.copy()
        
        if time.time() - start_time > budget_s - 5:
            break
    
    if time.time() - start_time < budget_s:
        result = opt.minimize(
            objective, global_best.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
    
    return best_h.tolist(), best_c5, n_points
