import numpy as np
import numpy.random as npr
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.cos(np.pi * x))
        h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h))
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    
    n_pop = 20
    pop = np.random.rand(n_pop, n_points)
    pop = pop * (target_sum / np.sum(pop, axis=1)[:, None])
    pop = np.clip(pop, 0.0, 1.0)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    for _ in range(200):
        fitness = np.zeros(n_pop)
        for i in range(n_pop):
            h_flat = pop[i]
            fitness[i] = objective(h_flat)
        
        idxs = np.argsort(fitness)
        selected = pop[idxs[:n_pop//2]]
        
        new_pop = []
        for _ in range(n_pop//2):
            p1, p2 = selected[np.random.choice(n_pop//2, 2, replace=False)]
            child = p1 * np.random.rand(n_points) + p2 * (1 - np.random.rand(n_points))
            child = np.clip(child, 0.0, 1.0)
            child = child * (target_sum / np.sum(child))
            new_pop.append(child)
        
        for i in range(n_pop//2):
            h_flat = new_pop[i]
            noise = np.random.normal(0, 0.05, size=n_points)
            h_flat += noise
            h_flat = np.clip(h_flat, 0.0, 1.0)
            sum_h = np.sum(h_flat)
            h_flat = h_flat * (target_sum / sum_h) if sum_h != 0 else h_flat
        
        pop = np.vstack((selected, new_pop))
        pop = pop[np.random.choice(n_pop, n_pop, replace=False)]
        
        for i in range(n_pop):
            h_flat = pop[i]
            if np.random.rand() < 0.25:
                idxs = np.random.choice(n_points, 3, replace=False)
                start, end = np.sort(idxs)
                h_flat[start:end] = 1.0 - h_flat[start:end]
            if np.random.rand() < 0.25:
                idxs = np.random.choice(n_points, 3, replace=False)
                start, end = np.sort(idxs)
                h_flat[start:end] = np.random.rand(end - start)
            h_flat = np.clip(h_flat, 0.0, 1.0)
            sum_h = np.sum(h_flat)
            h_flat = h_flat * (target_sum / sum_h) if sum_h != 0 else h_flat
            pop[i] = h_flat
        
        for i in range(n_pop):
            h_flat = pop[i]
            if fitness[i] < best_c5:
                best_c5 = fitness[i]
                best_h = h_flat.copy()
        
        if time.time() - start_time > budget_s - 10:
            break
    
    if time.time() - start_time < budget_s:
        pop = [best_h]
        for _ in range(5):
            h_flat = pop[0].copy()
            noise = np.random.normal(0, 0.05, size=n_points)
            h_flat += noise
            h_flat = np.clip(h_flat, 0.0, 1.0)
            sum_h = np.sum(h_flat)
            h_flat = h_flat * (target_sum / sum_h) if sum_h != 0 else h_flat
            pop.append(h_flat)
            h_flat = pop[0]
            for i in range(len(pop)):
                h_flat = pop[i]
                if objective(h_flat) < best_c5:
                    best_c5 = objective(h_flat)
                    best_h = h_flat.copy()
    
    return best_h.tolist(), best_c5, n_points
