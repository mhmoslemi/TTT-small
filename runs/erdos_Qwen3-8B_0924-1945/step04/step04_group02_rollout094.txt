import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    class PSOParticle:
        def __init__(self, pos):
            self.pos = pos.copy()
            self.pbest = pos.copy()
            self.pbest_val = objective(self.pos)
            self.vel = np.zeros(n_points)

    def update_particle(particle, gbest, omega, c1, c2):
        r1 = np.random.rand(n_points)
        r2 = np.random.rand(n_points)
        particle.vel = omega * particle.vel + c1 * r1 * (particle.pbest - particle.pos) + c2 * r2 * (gbest - particle.pos)
        particle.pos += particle.vel
        particle.pos = np.clip(particle.pos, 0.0, 1.0)
        sum_h = np.sum(particle.pos)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        particle.pos *= scale
        current_val = objective(particle.pos)
        if current_val < particle.pbest_val:
            particle.pbest = particle.pos.copy()
            particle.pbest_val = current_val

    start_time = time.time()
    n_particles = 40
    particles = [PSOParticle(np.random.rand(n_points)) for _ in range(n_particles)]
    for p in particles:
        sum_h = np.sum(p.pos)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        p.pos *= scale

    best_h = h.copy()
    best_c5 = float('inf')
    omega_start, omega_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    mutation_amp_start, mutation_amp_end = 0.05, 0.02
    max_generations = 200
    generation = 0

    while generation < max_generations and time.time() - start_time < budget_s - 5:
        omega = omega_start - (omega_start - omega_end) * generation / max_generations
        mutation_amp = mutation_amp_start - (mutation_amp_start - mutation_amp_end) * generation / max_generations
        gbest = particles[0].pos.copy()
        gbest_val = particles[0].pbest_val
        for p in particles:
            if p.pbest_val < gbest_val:
                gbest = p.pbest.copy()
                gbest_val = p.pbest_val
        for p in particles:
            update_particle(p, gbest, omega, c1, c2)
            if np.random.rand() < 0.1:
                noise = np.random.normal(0, mutation_amp, n_points)
                p.pos += noise
                p.pos = np.clip(p.pos, 0.0, 1.0)
                sum_h = np.sum(p.pos)
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                p.pos *= scale
        for p in particles:
            if p.pbest_val < best_c5:
                best_c5 = p.pbest_val
                best_h = p.pbest.copy()
        generation += 1

    if time.time() - start_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, 
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
