import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.round(h).astype(int)
    else:
        h = np.zeros(n_points, dtype=int)
        h[:int(n_points/2)] = 1
        h = h.astype(float)
    h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
    
    def autocorr(h):
        h_expanded = np.concatenate([h, h])
        corr = np.correlate(h_expanded, h_expanded, mode='full')
        return corr[n_points-1:2*n_points-1] * dx
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.round(h).astype(int)
        h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
        return np.max(autocorr(h))
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        h = best_h.copy()
        h = np.round(h).astype(int)
        h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
        corr = autocorr(h)
        min_corr = np.min(corr)
        lag = np.argmin(corr)
        indices = np.where((h == 0) & (h[lag:] == 1))[0]
        if len(indices) == 0:
            continue
        i = np.random.choice(indices)
        h[i] = 1
        h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    h = best_h.copy()
    h = np.round(h).astype(int)
    h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
    c5_bound = objective(h)
    return h.tolist(), c5_bound, n_points
