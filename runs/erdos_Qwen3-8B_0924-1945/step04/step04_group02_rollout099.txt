import numpy as np
import scipy.optimize as opt
import time
import GPy

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def rescale_h(h):
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        return h * scale
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = rescale_h(h)
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = rescale_h(h)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def get_c5(h):
        h = rescale_h(h)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    
    def ei(gp, h_test):
        mean, var = gp.predict(h_test.reshape(1, -1))
        mean = mean[0]
        var = var[0]
        if var < 1e-6:
            return -np.inf

        imp = u - 0.01
        ei = (imp * prob) - (mean * stats.norm.cdf((u - mean) / np.sqrt(var)))
        return ei
    
    def get_acquisition(gp, h_test):
        h_test_clipped = np.clip(h_test, 0.0, 1.0)
        h_test_scaled = rescale_h(h_test_clipped)
        h_test_flat = h_test_scaled.flatten()
        return ei(gp, h_test_flat)
    
    def generate_initial_samples():
        samples = []
        for _ in range(20):
            h = np.random.rand(n_points)
            h = rescale_h(h)
            samples.append(h)
        return np.array(samples)
    
    def build_gp(samples):
        X = np.array([h.flatten() for h in samples])
        Y = np.array([get_c5(h) for h in samples])
        kernel = GPy.kern.RBF(input_dim=n_points, variance=1.0, lengthscale=1.0) + GPy.kern.WhiteNoise(input_dim=n_points, variance=1e-5)
        model = GPy.models.GPR((X, Y), kernel=kernel, noise_var=1e-8)
        model.optimize(messages=False)
        return model
    
    def select_next_point(gp, samples, n_samples=10):
        candidates = np.random.rand(n_samples, n_points)
        candidates = np.clip(candidates, 0.0, 1.0)
        candidates = rescale_h(candidates)
        scores = np.array([get_c5(c) for c in candidates])
        idx = np.argmin(scores)
        return candidates[idx]
    
    def optimize():
        h = rescale_h(h)
        samples = []
        for _ in range(20):
            samples.append(h.copy())
            h = select_next_point(build_gp(samples), samples)
        gp = build_gp(samples)
        best_h = samples[np.argmin([get_c5(s) for s in samples])]
        best_c5 = get_c5(best_h)
        for _ in range(500):
            if time.time() - start_time > budget_s - 10:
                break
            h_candidate = select_next_point(gp, samples)
            samples.append(h_candidate)
            gp = build_gp(samples)
            best_h = samples[np.argmin([get_c5(s) for s in samples])]
            best_c5 = get_c5(best_h)
        h_opt = best_h.copy()
        result = opt.minimize(
            objective, h_opt.flatten(), 
            method='L-BFGS-B', bounds=bounds, 
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = rescale_h(h_opt)
        c5_bound = get_c5(h_opt)
        return h_opt, c5_bound
    
    start_time = time.time()
    h, c5_bound = optimize()
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, h.flatten(), 
            method='L-BFGS-B', bounds=bounds, 
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = rescale_h(h_opt)
        c5_bound = get_c5(h_opt)
    return h.tolist(), c5_bound, n_points
