```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def repair(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        return h * scale

    def objective(h_flat):
        return evaluate(h_flat)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    n_particles = 30
    particles = np.random.rand(n_particles, n_points)
    particles = np.array([p * (target_sum / np.sum(p)) for p in particles])
    velocities = np.random.rand(n_particles, n_points) * 0.1
    pbest = particles.copy()
    pbest_fitness = np.array([evaluate(p) for p in pbest])
    gbest = pbest[np.argmin(pbest_fitness)]
    gbest_fitness = np.min(pbest_fitness)

    best_c5 = gbest_fitness
    best_h = gbest.copy()
    start_time = time.time()
    iteration = 0
    no_improvement = 0

    while time.time() - start_time < budget_s - 10 and no_improvement < 100:
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = 0.7 * velocities[i] + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            particles[i] = particles[i] + velocities[i]
            particles[i] = repair(particles[i])
            fitness = evaluate(particles[i])
            if fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fitness
                if fitness < gbest_fitness:
                    gbest = pbest[i].copy()
                    gbest_fitness = fitness
                    best_c5 = gbest_fitness
                    best_h = gbest.copy()
                    no_improvement = 0
            else:
                no_improvement += 1
        iteration += 1

    return best_h.tolist(), best_c5, n_points
```
