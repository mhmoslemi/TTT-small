import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.cos(np.pi * x))
        h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h))
    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    population_size = 20
    population = [h.copy() for _ in range(population_size)]
    for _ in range(500):
        if time.time() - start_time > budget_s - 20:
            break
        fitness = [evaluate(p) for p in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        if best_c5 < 1e-10:
            break
        if _ % 10 == 0:
            new_population = []
            for i in range(population_size):
                parent1, parent2 = np.random.choice(population_size, 2, replace=False)
                child = population[parent1].copy()
                if np.random.rand() < 0.8:
                    split = np.random.randint(1, n_points - 1)
                    child[:split] = population[parent2][:split]
                if np.random.rand() < 0.3:
                    idx = np.random.choice(n_points, 5, replace=False)
                    child[idx] += np.random.uniform(-0.1, 0.1, size=5)
                    child = np.clip(child, 0.0, 1.0)
                    sum_h = np.sum(child)
                    scale = target_sum / sum_h if sum_h != 0 else 1.0
                    child = child * scale
                if np.random.rand() < 0.1:
                    idx = np.random.choice(n_points, 10, replace=False)
                    child[idx] = np.random.uniform(0.0, 1.0, size=10)
                sum_h = np.sum(child)
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                child = child * scale
                new_population.append(child)
            population = new_population
    if time.time() - start_time < budget_s - 10:
        for _ in range(5):
            if time.time() - start_time > budget_s - 10:
                break
            idx = np.argmin(fitness)
            h_opt = population[idx].copy()
            h_opt = np.clip(h_opt, 0.0, 1.0)
            sum_h = np.sum(h_opt)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_opt = h_opt * scale
            def objective(h_flat):
                h = h_flat.reshape(n_points)
                h = np.clip(h, 0.0, 1.0)
                sum_h = np.sum(h)
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                h = h * scale
                corr = np.correlate(h, 1.0 - h, mode="full")
                return np.max(corr * dx)
            result = opt.minimize(objective, h_opt.flatten(), method='L-BFGS-B', bounds=[(0.0, 1.0)]*n_points, tol=1e-8)
            h_opt = result.x.reshape(n_points)
            h_opt = np.clip(h_opt, 0.0, 1.0)
            sum_h = np.sum(h_opt)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_opt = h_opt * scale
            c5_bound = evaluate(h_opt)
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h = h_opt.copy()
    return best_h.tolist(), best_c5, n_points
