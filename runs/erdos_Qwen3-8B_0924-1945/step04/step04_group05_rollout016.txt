import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def init_h():
        if initial_h_values is not None:
            h = np.array(initial_h_values, dtype=np.float64)
        else:
            x_grid = np.linspace(0, 2, n_points)
            idxs = np.argsort(np.abs(x_grid - 1.0))
            h = np.zeros(n_points)
            h[idxs[:n_points//2]] = 1.0
            h = np.clip(h, 0.0, 1.0)
            h = h * (target_sum / np.sum(h))
        return h

    h = init_h()

    def objective(h_flat):
        h_opt = h_flat.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h = np.sum(h_opt)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_opt = h_opt * scale
        corr = np.correlate(h_opt, 1.0 - h_opt, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    def optimize_with_solvers(h_initial, max_iter=200, solver='L-BFGS-B'):
        result = opt.minimize(
            objective, h_initial.flatten(), 
            method=solver, bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': max_iter}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5 = objective(h_opt.flatten())
        return h_opt, c5

    h, c5 = optimize_with_solvers(h, max_iter=200, solver='L-BFGS-B')
    best_h, best_c5 = h, c5

    for _ in range(2):
        if time.time() - start_time > budget_s - 10:
            break
        h = np.roll(h, np.random.randint(0, n_points))
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        h_opt, c5 = optimize_with_solvers(h, max_iter=100, solver='L-BFGS-B')
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    for _ in range(3):
        if time.time() - start_time > budget_s - 10:
            break
        h_perturb = h + np.random.normal(0, 0.05, size=n_points)
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        sum_h_perturb = np.sum(h_perturb)
        scale = target_sum / sum_h_perturb if sum_h_perturb != 0 else 1.0
        h_perturb = h_perturb * scale
        h_opt, c5 = optimize_with_solvers(h_perturb, max_iter=50, solver='L-BFGS-B')
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    if time.time() - start_time < budget_s - 50:
        h = np.roll(best_h, np.random.randint(0, n_points))
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        h_opt, c5 = optimize_with_solvers(h, max_iter=50, solver='L-BFGS-B')
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    return best_h.tolist(), best_c5, n_points
