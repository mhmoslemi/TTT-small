```python
import numpy as np
import cvxpy as cp
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[::2] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    for _ in range(200):
        h_init = h.copy()
        h_init = np.clip(h_init, 0.0, 1.0)
        sum_h = np.sum(h_init)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_init = h_init * scale

        variables = cp.Variable(n_points, name='h')
        C = cp.Variable(name='C')
        constraints = [variables >= 0, variables <= 1, cp.sum(variables) == target_sum]

        for k in range(1, n_points):
            corr = cp.sum(variables[:n_points - k] * (1 - variables[k:]))
            constraints.append(corr <= C)

        prob = cp.Problem(cp.Minimize(C), constraints)
        try:
            result = prob.solve(solver=cp.ECOS, eps=1e-8, verbose=False)
            if result is not None and C.value < best_c5:
                best_c5 = C.value
                best_h = variables.value.copy()
        except:
            pass

        h = best_h.copy()
        if time.time() - start_time > budget_s - 5:
            break
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=[(0.0, 1.0)]*n_points, 
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
