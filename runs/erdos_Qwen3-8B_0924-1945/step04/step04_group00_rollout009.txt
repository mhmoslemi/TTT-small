```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.round(h)
        h = h * (target_sum / np.sum(h))
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = corr * dx
        return np.max(overlaps)
    def update_h(h):
        h = h.astype(np.float64)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        return h
    def find_bad_lag(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = corr * dx
        min_overlap_idx = np.argmin(overlaps)
        if min_overlap_idx >= len(overlaps) - 1:
            return None
        return min_overlap_idx
    def flip_bit(h, lag):
        idxs = np.where((h == 0) & (h[(np.arange(n_points) + lag) % n_points] == 1))[0]
        if len(idxs) > 0:
            i = np.random.choice(idxs)
            h[i] = 1.0
            return h
        return h
    best_h = h.copy()
    best_c5 = objective(h)
    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        h = update_h(h)
        lag = find_bad_lag(h)
        if lag is not None:
            h = flip_bit(h, lag)
        c5 = objective(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
            h = update_h(best_h)
            lag = find_bad_lag(h)
            if lag is not None:
                h = flip_bit(h, lag)
            c5 = objective(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h.tolist(), best_c5, n_points
```
