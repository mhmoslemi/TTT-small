import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.round(h).astype(float)
        h = h * (target_sum / np.sum(h))
    def compute_correlations(h):
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))
        corr = np.correlate(h, 1.0 - h, mode="full")
        return corr * dx
    def exact_objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = compute_correlations(h)
        return np.max(corr)
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s - 5:
            break
        corr = compute_correlations(h)
        max_corr = np.max(corr)
        lag = np.argmin(corr)
        if max_corr < 1e-10:
            break
        indices = np.where((h == 0) & (np.roll(h, lag) == 1))[0]
        if len(indices) > 0:
            i = np.random.choice(indices)
            h[i] = 1.0
            h = h * (target_sum / np.sum(h))
        c5_bound = exact_objective(h.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h.copy()
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(100):
            if time.time() - start_time > budget_s - 5:
                break
            corr = compute_correlations(best_h)
            max_corr = np.max(corr)
            lag = np.argmin(corr)
            if max_corr < 1e-10:
                break
            indices = np.where((best_h == 0) & (np.roll(best_h, lag) == 1))[0]
            if len(indices) > 0:
                i = np.random.choice(indices)
                best_h[i] = 1.0
                best_h = best_h * (target_sum / np.sum(best_h))
            c5_bound = exact_objective(best_h.flatten())
            if c5_bound < best_c5:
                best_c5 = c5_bound
    return best_h.tolist(), best_c5, n_points
