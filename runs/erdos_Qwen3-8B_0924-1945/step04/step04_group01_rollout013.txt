```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.cos(np.pi * x_grid))
        h = h * (target_sum / np.sum(h))

    def exact_objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    def generate_binary():
        h = np.random.rand(n_points)
        h = (h < 0.5).astype(float)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        return h * scale

    def binary_to_soft(h, temp):
        h = h.copy()
        h = np.log(h / (1 - h)) / temp
        return h

    def soft_to_binary(x, temp):
        h = 1 / (1 + np.exp(-x * temp))
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        return h * scale

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def mutation(h_flat, mutation_rate):
        h = h_flat.copy()
        h = h.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        mask = np.random.rand(n_points) < mutation_rate
        h[mask] = np.random.rand(np.sum(mask))
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        return h * scale

    def compute_overlap(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return corr * dx

    def update_weights(weights, h_flat, top_k=5):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = corr * dx
        sorted_indices = np.argsort(overlaps[-n_points + 1:])
        top_indices = sorted_indices[-top_k:]
        weights[top_indices] += 1.0
        return weights

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    for iteration in range(200):
        if iteration == 0:
            h = generate_binary()
        else:
            h = mutation(h.flatten(), 0.1)
        h = h.flatten()
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        c5 = exact_objective(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if time.time() - start_time > budget_s - 30:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        weights = np.zeros(n_points)
        for _ in range(50):
            h = mutation(best_h, 0.1)
            h = h.flatten()
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            c5 = exact_objective(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            weights = update_weights(weights, h)
            h = soft_to_binary(binary_to_soft(best_h, 5), 5)
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            c5 = exact_objective(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    return best_h.tolist(), best_c5, n_points
```
