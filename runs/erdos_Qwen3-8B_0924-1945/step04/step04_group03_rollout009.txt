```python
import numpy as np
import time
import scipy.optimize as opt
import random

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for _ in range(n_points // 2):
            idx = np.random.randint(n_points)
            h[idx] = 1.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h > 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def normalize(h):
        sum_h = np.sum(h)
        return h * (target_sum / sum_h) if sum_h > 0 else h

    def mutate(parent):
        a = np.random.randint(n_points)
        b = np.random.randint(n_points)
        c = np.random.randint(n_points)
        mutant = np.copy(parent)
        mutant[a] = np.clip(parent[b] + 0.5 * (parent[c] - parent[a]), 0.0, 1.0)
        mutant = normalize(mutant)
        return mutant

    def crossover(parent1, parent2):
        child = np.copy(parent1)
        idx = np.random.choice(n_points)
        child[idx] = parent2[idx]
        return normalize(child)

    best_h = h.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()
    generation = 0
    while time.time() - start_time < budget_s:
        population = [normalize(np.random.rand(n_points)) for _ in range(10)]
        population.append(best_h)
        population = [normalize(h) for h in population]
        evaluations = [evaluate(h) for h in population]
        indices = np.argsort(evaluations)
        top_parents = [population[i] for i in indices[:10]]
        new_population = []
        for _ in range(10):
            parent1 = random.choice(top_parents)
            parent2 = random.choice(top_parents)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        population = top_parents + new_population
        evaluations = [evaluate(h) for h in population]
        indices = np.argsort(evaluations)
        population = [population[i] for i in indices[:10]]
        current_c5 = evaluate(population[0])
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[0]
        generation += 1
        if generation % 100 == 0:
            print(f"Generation {generation}, best C5: {best_c5}")
        if generation > 1000:
            break

    return best_h.tolist(), best_c5, n_points
```
