```python
import numpy as np
import scipy.optimize as opt
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel, WhiteKernel

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, size=n_points)
        h = h * target_sum / np.sum(h)
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    X = np.array([h.tolist()])
    y = np.array([objective(h.flatten())])

    kernel = ConstantKernel(1.0, (1e-2, 1e2)) * RBF(1.0, (1e-2, 1e2)) + WhiteKernel(1e-3, (1e-5, 1e-1))
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-5, n_restarts_optimizer=5)

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()

    for i in range(500):
        gp.fit(X, y)
        if i > 0 and np.max(y) < best_c5:
            best_c5 = np.max(y)
            best_h = X[np.argmax(y), :]

        mu, sigma = gp.predict(X, return_std=True)
        ei = (mu - best_c5) * np.exp(-0.5 * (mu - best_c5) ** 2 / sigma ** 2)
        ei /= sigma
        ei = np.where(sigma > 1e-8, ei, 0)
        next_idx = np.argmax(ei)
        next_x = X[next_idx, :]

        noise = np.random.normal(0, 0.01, size=n_points)
        next_h = next_x + noise
        next_h = np.clip(next_h, 0.0, 1.0)
        sum_h = np.sum(next_h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        next_h_scaled = next_h * scale
        next_c5 = objective(next_h.flatten())
        X = np.vstack((X, next_h_scaled))
        y = np.append(y, next_c5)

        if time.time() - start_time > budget_s - 5:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
