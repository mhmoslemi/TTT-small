import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    n_particles = 30
    particles = np.random.rand(n_particles, n_points)
    particles = particles * (target_sum / np.sum(particles, axis=1, keepdims=True))
    pbest = particles.copy()
    pbest_obj = np.array([objective(p) for p in pbest])
    gbest = particles[np.argmin(pbest_obj)]
    gbest_obj = np.min(pbest_obj)

    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        velocities = np.random.rand(n_particles, n_points) * 0.1
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = 0.7 * velocities[i] + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            particles[i] = np.clip(particles[i] + velocities[i], 0.0, 1.0)
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
            obj = objective(particles[i])
            if obj < pbest_obj[i]:
                pbest[i] = particles[i].copy()
                pbest_obj[i] = obj
        if np.min(pbest_obj) < gbest_obj:
            gbest = pbest[np.argmin(pbest_obj)]
            gbest_obj = np.min(pbest_obj)
        if np.random.rand() < 0.1:
            noise = np.random.normal(0, 0.05, n_points)
            particles += noise
            particles = np.clip(particles, 0.0, 1.0)
            particles = particles * (target_sum / np.sum(particles, axis=1, keepdims=True))
        if time.time() - start_time > budget_s - 5:
            break

    h_opt = gbest.copy()
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt)

    return h_opt.tolist(), c5_bound, n_points
