```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[::2] = 1
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    for _ in range(200):
        h_flat = h.copy()
        h_flat = np.clip(h_flat, 0.0, 1.0)
        sum_h = np.sum(h_flat)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_flat = h_flat * scale

        h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
        C = cp.Variable()

        constraints = [
            cp.sum(h_var) == target_sum,
            C >= 0
        ]

        for k in range(1, n_points):
            corr_k = cp.sum(h_var[:n_points - k] * (1 - h_var[k:]))
            constraints.append(C >= corr_k)

        prob = cp.Problem(cp.Minimize(C), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-8)
            if prob.status == 'optimal':
                h_opt = h_var.value
                h_opt = np.clip(h_opt, 0.0, 1.0)
                sum_h_opt = np.sum(h_opt)
                scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
                h_opt = h_opt * scale
                c5_bound = objective(h_opt)
                if c5_bound < best_c5:
                    best_c5 = c5_bound
                    best_h = h_opt.copy()
        except:
            pass

        if time.time() - start_time > budget_s - 5:
            break
        h = h_opt.copy()
        noise = np.random.normal(0, 0.02, size=n_points)
        h = h + noise
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        h_flat = best_h.copy()
        h_flat = np.clip(h_flat, 0.0, 1.0)
        sum_h = np.sum(h_flat)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_flat = h_flat * scale
        result = opt.minimize(
            objective, h_flat, 
            method='L-BFGS-B', bounds=[(0.0, 1.0)]*n_points, 
            constraints=constraint,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
