import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.round(h).astype(np.int64)
        h = h * (target_sum / np.sum(h))
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h_bin = (h > 0.5).astype(np.int64)
        corr = np.correlate(h_bin, 1 - h_bin, mode="full")
        return np.max(corr * dx)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    
    for _ in range(300):
        if time.time() - start_time > budget_s - 5:
            break
        h = np.clip(h, 0, 1)
        h_bin = (h > 0.5).astype(np.int64)
        corr = np.correlate(h_bin, 1 - h_bin, mode="full")
        min_idx = np.argmin(corr[1:-1])
        k = min_idx + 1
        i = np.random.randint(0, n_points)
        j = (i + k) % n_points
        if h_bin[i] == 0 and h_bin[j] == 1:
            h_bin[i] = 1
        elif h_bin[i] == 1 and h_bin[j] == 0:
            h_bin[j] = 1
        h = h_bin.astype(np.float64)
        h = h * (target_sum / np.sum(h))
    
    h = np.clip(h, 0, 1)
    h_bin = (h > 0.5).astype(np.int64)
    c5 = compute_c5(h)
    return h.tolist(), c5, n_points
