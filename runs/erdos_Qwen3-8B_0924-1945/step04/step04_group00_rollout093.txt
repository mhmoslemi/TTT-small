import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.cos(np.pi * x))
        h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h))
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    
    def fitness(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return -np.max(corr * dx)  # Negative for minimization
    
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    pop_size = 20
    population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
    for i in range(pop_size):
        population[i] = population[i] * (target_sum / np.sum(population[i]))
    
    best_h = np.copy(population[0])
    best_fitness = fitness(best_h)
    
    for gen in range(500):
        fitnesses = [fitness(h) for h in population]
        indices = np.argsort(fitnesses)
        top_parents = [population[i] for i in indices[:5]]
        
        new_population = []
        for _ in range(pop_size // 2):
            parent1, parent2 = np.random.choice(top_parents, 2, replace=False)
            child = np.copy(parent1)
            for i in np.random.choice(range(n_points), size=10, replace=False):
                child[i] = parent2[i]
            child = np.clip(child, 0.0, 1.0)
            sum_child = np.sum(child)
            scale = target_sum / sum_child if sum_child != 0 else 1.0
            child = child * scale
            new_population.append(child)
        
        for h in population:
            if np.random.rand() < 0.75:
                mut = np.random.normal(0, 0.1, n_points)
                h = h + mut
                h = np.clip(h, 0.0, 1.0)
                sum_h = np.sum(h)
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                h = h * scale
                new_population.append(h)
        
        population = new_population
        for h in population:
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            h = np.clip(h, 0.0, 1.0)
        
        for h in population:
            fitness_val = fitness(h)
            if fitness_val < best_fitness:
                best_fitness = fitness_val
                best_h = np.copy(h)
        
        if time.time() - time.time() > budget_s - 5:
            break
    
    result = opt.minimize(
        objective, best_h.flatten(), 
        method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 50}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt.flatten())
    
    return h_opt.tolist(), c5_bound, n_points
