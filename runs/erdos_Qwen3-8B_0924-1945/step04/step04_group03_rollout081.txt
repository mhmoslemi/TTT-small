```python
import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points // 2):
            h[i] = 1.0
        h = h * (target_sum / np.sum(h))

    def compute_c5(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def normalize_h(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        return h * scale

    def mutate(h, F=0.5):
        idxs = np.random.choice(n_points, size=3, replace=False)
        a, b, c = h[idxs[0]], h[idxs[1]], h[idxs[2]]
        mutant = a + F * (b - c)
        return normalize_h(mutant)

    def crossover(parent1, parent2):
        child = parent1.copy()
        mask = np.random.rand(n_points) < 0.5
        child[mask] = parent2[mask]
        return normalize_h(child)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    best_fitness = best_c5

    start_time = time.time()
    n_population = 30
    population = [normalize_h(np.random.rand(n_points)) for _ in range(n_population)]
    population = [h for h in population if np.all(h >= 0) and np.all(h <= 1)]

    for _ in range(2000):
        if time.time() - start_time > budget_s - 5:
            break
        fitnesses = [compute_c5(h) for h in population]
        ranked = sorted(zip(fitnesses, population), key=lambda x: x[0])
        top_half = ranked[:len(ranked) // 2]
        top_half = [h for _, h in top_half]

        new_population = []
        for _ in range(len(top_half) // 2):
            a, b = np.random.choice(top_half, size=2, replace=False)
            mutant = mutate(a)
            child = crossover(mutant, b)
            new_population.append(child)
        population = top_half + new_population
        population = sorted(population, key=compute_c5)[:n_population]

        current_fitness = compute_c5(population[0])
        if current_fitness < best_fitness:
            best_fitness = current_fitness
            best_h = population[0].copy()

    return best_h.tolist(), best_fitness, n_points
```
