import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        m = 4
        l = 1.0 / m
        g = 0.5 * (1.0 - m * l)
        cluster_positions = np.arange(m) * (l + g)
        cluster_positions = np.clip(cluster_positions, 0, 2)
        h = np.zeros(n_points)
        for pos in cluster_positions:
            idx = np.searchsorted(np.linspace(0, 2, n_points), pos)
            h[idx - 1:idx + int(l / dx)] = 1
        h = h * (target_sum / np.sum(h))

    def evaluate(h):
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def normalize(h):
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        return h * scale

    population_size = 30
    pop = [np.random.permutation(n_points) for _ in range(population_size)]
    pop = [np.zeros(n_points) for _ in range(population_size)]
    for i in range(population_size):
        indices = np.random.choice(n_points, size=int(target_sum), replace=False)
        pop[i][indices] = 1.0
    pop = [normalize(p) for p in pop]

    best_h = pop[0]
    best_c5 = evaluate(best_h)

    start_time = time.time()
    generation = 0
    improve_count = 0

    while time.time() - start_time < budget_s - 5:
        generation += 1
        scores = [evaluate(p) for p in pop]
        inds = np.argsort(scores)
        elite_size = int(population_size * 0.2)
        elites = pop[inds[:elite_size]]

        new_pop = []
        for _ in range(population_size - elite_size):
            a, b, c = np.random.choice(elites, 3, replace=False)
            mutant = a + 0.5 * (b - c)
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = normalize(mutant)
            new_pop.append(mutant)
        pop = elites + new_pop

        current_scores = [evaluate(p) for p in pop]
        best_idx = np.argmin(current_scores)
        current_best = pop[best_idx]
        current_c5 = current_scores[best_idx]

        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
            improve_count = 0
        else:
            improve_count += 1

        if improve_count >= 50:
            break

    return best_h.tolist(), best_c5, n_points
