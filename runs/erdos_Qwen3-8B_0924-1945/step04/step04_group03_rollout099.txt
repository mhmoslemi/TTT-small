```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    n_particles = 30
    particles = np.random.rand(n_particles, n_points)
    particles = particles * (target_sum / np.sum(particles, axis=1, keepdims=True))
    particles = np.clip(particles, 0.0, 1.0)
    velocities = np.random.rand(n_particles, n_points) * 0.01
    pbest = particles.copy()
    pbest_fitness = np.full(n_particles, float('inf'))
    gbest = particles[0].copy()
    gbest_fitness = float('inf')

    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(n_particles):
            h = particles[i]
            fitness = objective(h.flatten())
            if fitness < pbest_fitness[i]:
                pbest[i] = h.copy()
                pbest_fitness[i] = fitness
            if fitness < gbest_fitness:
                gbest = h.copy()
                gbest_fitness = fitness
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = 0.7 * velocities[i] + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_h = np.sum(particles[i])
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            particles[i] = particles[i] * scale
        if np.min(np.abs(pbest_fitness - gbest_fitness)) < 1e-6:
            break

    best_h = gbest.copy()
    best_c5 = gbest_fitness
    return best_h.tolist(), best_c5, n_points
```
