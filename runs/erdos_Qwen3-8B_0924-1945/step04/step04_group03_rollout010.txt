import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_c5 = float('inf')
    best_h = h.copy()
    best_fitness = objective(best_h.flatten())
    
    n_particles = 30
    particles = np.random.rand(n_particles, n_points)
    particles = particles * target_sum / np.sum(particles, axis=1, keepdims=True)
    particles = particles.clip(0, 1)
    pbest = particles.copy()
    pbest_fitness = np.array([objective(p.reshape(n_points)).flatten()[0] for p in particles])
    gbest = particles[np.argmin(pbest_fitness)].copy()
    gbest_fitness = np.min(pbest_fitness)
    
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    max_iter = int(budget_s * 0.95 / 0.01)
    
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        
        velocities = np.random.rand(n_particles, n_points) * 0.1
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (pbest[i] - particles[i]) + social * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
        
        for i in range(n_particles):
            current_fitness = objective(particles[i].flatten())
            if current_fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = current_fitness
        
        if np.min(pbest_fitness) < gbest_fitness:
            gbest = pbest[np.argmin(pbest_fitness)].copy()
            gbest_fitness = np.min(pbest_fitness)
        
        if gbest_fitness < best_c5:
            best_c5 = gbest_fitness
            best_h = gbest.copy()
    
    return best_h.tolist(), best_c5, n_points
