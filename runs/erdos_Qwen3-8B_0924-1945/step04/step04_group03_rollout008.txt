import numpy as np
import time
import scipy.optimize as opt

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.sort(h)
        h = np.interp(h, np.linspace(0, 1, n_points), np.linspace(0, 1, n_points))
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale

        h_new = h.copy()
        h_new = np.clip(h_new, 0.0, 1.0)
        sum_h_new = np.sum(h_new)
        scale_new = target_sum / sum_h_new if sum_h_new != 0 else 1.0
        h_new = h_new * scale_new

        h = h_new.copy()

        c5_bound = objective(h.copy())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h.copy()
        if time.time() - start_time > budget_s - 5:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(100):
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale

            h_new = h.copy()
            h_new = np.clip(h_new, 0.0, 1.0)
            sum_h_new = np.sum(h_new)
            scale_new = target_sum / sum_h_new if sum_h_new != 0 else 1.0
            h_new = h_new * scale_new

            h = h_new.copy()

            c5_bound = objective(h.copy())
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h = h.copy()
            if time.time() - start_time > budget_s - 5:
                break

    return best_h.tolist(), best_c5, n_points
