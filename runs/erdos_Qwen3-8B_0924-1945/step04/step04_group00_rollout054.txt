import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.cos(np.pi * x))
        h = h * (target_sum / np.sum(h))
    def exact_objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    population_size = 20
    population = [h.copy() for _ in range(population_size)]
    for _ in range(500):
        fitness = [exact_objective(p) for p in population]
        ranked = sorted(zip(fitness, population), key=lambda x: x[0])
        top_solutions = ranked[:5]
        if _ % 10 == 0:
            noise = np.random.normal(0, 0.05, size=n_points)
            h = h + noise
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            population[0] = h.copy()
        for i in range(population_size):
            if i % 2 == 0:
                parent1, parent2 = np.random.choice(top_solutions, size=2, replace=False)
                child = (parent1[1] + parent2[1]) / 2
            else:
                parent = np.random.choice(top_solutions)
                child = parent[1].copy()
                seg = np.random.choice(range(n_points), size=2)
                start, end = sorted(seg)
                child[start:end+1] = 1.0 - child[start:end+1]
            child = np.clip(child, 0.0, 1.0)
            sum_h = np.sum(child)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            child = child * scale
            population[i] = child.copy()
        for i, p in enumerate(population):
            if np.random.random() < 0.2:
                seg = np.random.choice(range(n_points), size=2)
                start, end = sorted(seg)
                p[start:end+1] += np.random.normal(0, 0.1, end - start + 1)
                p = np.clip(p, 0.0, 1.0)
                sum_h = np.sum(p)
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                p = p * scale
                population[i] = p.copy()
        for i, p in enumerate(population):
            if i % 5 == 0:
                res = opt.minimize(
                    lambda x: exact_objective(x),
                    p,
                    bounds=bounds,
                    constraints=cons,
                    method='L-BFGS-B',
                    tol=1e-8,
                    options={'maxiter': 10}
                )
                population[i] = res.x.copy()
        current_best = min(ranked, key=lambda x: x[0])
        if current_best[0] < best_c5:
            best_c5 = current_best[0]
            best_h = current_best[1].copy()
        if time.time() - start_time > budget_s - 5:
            break
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        res = opt.minimize(
            lambda x: exact_objective(x),
            best_h,
            bounds=bounds,
            constraints=cons,
            method='L-BFGS-B',
            tol=1e-8,
            options={'maxiter': 50}
        )
        best_h = res.x.copy()
        best_c5 = exact_objective(best_h)
    return best_h.tolist(), best_c5, n_points
