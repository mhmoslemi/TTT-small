import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        pos = np.random.uniform(0, 2, 10)
        pos = np.sort(pos)
        for i in range(10):
            idx = np.searchsorted(np.linspace(0, 2, n_points), pos[i])
            h[idx - 1:idx + 1] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    n_particles = 20
    swarm = np.random.uniform(0.0, 1.0, (n_particles, n_points))
    swarm = swarm * (target_sum / np.sum(swarm, axis=1)[:, np.newaxis])
    swarm = np.clip(swarm, 0.0, 1.0)
    weights = np.random.uniform(0.5, 0.9, n_particles)

    for gen in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        pbest = swarm.copy()
        gbest = swarm[0].copy()
        gbest_c5 = objective(gbest.flatten())
        for i in range(n_particles):
            h = swarm[i]
            r1, r2 = np.random.rand(2)
            vel = (weights[i] * (pbest[i] - h) + 
                   r1 * np.random.uniform(0, 1, n_points) + 
                   r2 * (gbest - h))
            h_new = h + vel
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = h_new * (target_sum / np.sum(h_new))
            c5 = objective(h_new.flatten())
            if c5 < objective(pbest[i].flatten()):
                pbest[i] = h_new
                if c5 < gbest_c5:
                    gbest = h_new
                    gbest_c5 = c5
        swarm = pbest.copy()
        weights = np.clip(weights * 0.99 + np.random.uniform(0.4, 0.6, n_particles), 0.5, 0.9)
        if gbest_c5 < best_c5:
            best_c5 = gbest_c5
            best_h = gbest.copy()

    if time.time() - start_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
