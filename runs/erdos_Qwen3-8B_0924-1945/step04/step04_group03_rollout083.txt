```python
import numpy as np
import time
import scipy.optimize as opt

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.sort(h)
        h = np.interp(h, np.linspace(0, 1, n_points), np.linspace(0, 1, n_points))
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    n_population = 30
    mutation_factor = 0.5
    crossover_rate = 0.9
    generations = 200
    improvement_threshold = 50

    population = [np.random.rand(n_points) for _ in range(n_population)]
    population = [h * target_sum / np.sum(h) for h in population]

    best_fitness = float('inf')
    no_improvement = 0

    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
        fitness = [objective(p) for p in population]
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_fitness = fitness[best_idx]
        if current_fitness < best_fitness:
            best_fitness = current_fitness
            best_h = current_best.copy()
            no_improvement = 0
        else:
            no_improvement += 1
        if no_improvement > improvement_threshold:
            break
        new_population = []
        for _ in range(n_population // 2):
            a, b, c = np.random.choice(n_population, 3, replace=False)
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = mutant * target_sum / np.sum(mutant)
            new_population.append(mutant)
        population = new_population + population[:n_population // 2]

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_fitness:
            best_fitness = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_fitness, n_points
```
