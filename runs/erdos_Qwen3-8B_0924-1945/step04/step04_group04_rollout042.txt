import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 600
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        angles = np.linspace(0, 2 * np.pi, n_points, endpoint=False)
        sawtooth = np.abs(np.sin(angles) - 1.0)
        h = np.clip(1.0 - sawtooth, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        scale = target_sum / sum_h
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # Global phase-space exploration
    for _ in range(150):
        if np.random.rand() < 0.2:
            h_perturb = h.copy()
            h_perturb += np.sin(angles * 10) * 0.05
            h_perturb = np.clip(h_perturb, 0.0, 1.0)
            sum_perturb = np.sum(h_perturb)
            if sum_perturb != 0:
                h_perturb *= target_sum / sum_perturb
            h = h_perturb
        if np.random.rand() < 0.1:
            h_phase = h.copy()
            h_phase = np.roll(h_phase, int(np.random.uniform(-n_points//4, n_points//4)))
            h_phase = np.clip(h_phase, 0.0, 1.0)
            sum_phase = np.sum(h_phase)
            if sum_phase != 0:
                h_phase *= target_sum / sum_phase
            h = h_phase
        if np.random.rand() < 0.05:
            h_noise = h.copy()
            noise = np.random.normal(0, 0.03, n_points)
            h_noise += noise
            h_noise = np.clip(h_noise, 0.0, 1.0)
            sum_noise = np.sum(h_noise)
            if sum_noise != 0:
                h_noise *= target_sum / sum_noise
            h = h_noise

        result = opt.minimize(
            objective, h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        if sum_h_opt != 0:
            h_opt *= target_sum / sum_h_opt
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
        if time.time() - start_time > budget_s - 100:
            break

    # Final refinement
    result = opt.minimize(
        objective, best_h.flatten(), 
        method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 200}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    if sum_h_opt != 0:
        h_opt *= target_sum / sum_h_opt
    c5_bound = objective(h_opt.flatten())
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 100}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        if sum_h_opt != 0:
            h_opt *= target_sum / sum_h_opt
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
