import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 0.0, target_sum)
        h = h / np.sum(h) * target_sum

    def obj(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0.0, target_sum)
        h /= np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    n_particles = 30
    particles = np.random.rand(n_particles, n_points)
    particles = particles / np.sum(particles, axis=1, keepdims=True) * target_sum
    particles = np.clip(particles, 0.0, target_sum)
    particles = particles / np.sum(particles, axis=1, keepdims=True) * target_sum

    pbest = particles.copy()
    pbest_obj = np.array([obj(p) for p in pbest])
    gbest = pbest[np.argmin(pbest_obj)]
    gbest_obj = np.min(pbest_obj)

    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            vel = 0.7 * np.random.rand(n_points)
            vel += 1.5 * r1 * (pbest[i] - particles[i])
            vel += 1.5 * r2 * (gbest - particles[i])
            particles[i] += vel
            particles[i] = np.clip(particles[i], 0.0, target_sum)
            particles[i] = particles[i] / np.sum(particles[i]) * target_sum
            obj_i = obj(particles[i])
            if obj_i < pbest_obj[i]:
                pbest[i] = particles[i].copy()
                pbest_obj[i] = obj_i
        current_gbest = pbest[np.argmin(pbest_obj)]
        if np.min(pbest_obj) < gbest_obj:
            gbest = current_gbest.copy()
            gbest_obj = np.min(pbest_obj)
        if np.random.rand() < 0.1:
            perturbation = np.random.uniform(-0.01, 0.01, n_points)
            particles += perturbation
            particles = np.clip(particles, 0.0, target_sum)
            particles = particles / np.sum(particles, axis=1, keepdims=True) * target_sum
        if np.min(pbest_obj) < best_c5:
            best_h = pbest[np.argmin(pbest_obj)].copy()
            best_c5 = np.min(pbest_obj)

    return best_h.tolist(), best_c5, n_points
