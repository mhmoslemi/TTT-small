import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))
    def exact_objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    def repair(h):
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        return h
    def mutate(h):
        h = h.copy()
        indices = np.random.choice(n_points, 5, replace=False)
        h[indices] = 1.0 - h[indices]
        return h
    def crossover(parent1, parent2):
        split = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:split], parent2[split:]])
        return child
    def fitness(h):
        return evaluate(h)
    population = [repair(np.random.rand(n_points) * (target_sum / np.sum(np.random.rand(n_points)))) for _ in range(20)]
    best_h = np.array(population[0])
    best_fitness = fitness(best_h)
    start_time = time.time()
    for gen in range(500):
        fitnesses = [fitness(ind) for ind in population]
        parents = np.argsort(fitnesses)[:5]
        parent1, parent2 = population[parents[0]], population[parents[1]]
        child = crossover(parent1, parent2)
        child = mutate(child)
        child = repair(child)
        if fitness(child) < best_fitness:
            best_fitness = fitness(child)
            best_h = child.copy()
        population.append(child)
        if len(population) > 20:
            population.pop(np.argmax(fitnesses))
        if time.time() - start_time > budget_s - 5:
            break
    h_opt = best_h
    h_opt = repair(h_opt)
    c5_bound = exact_objective(h_opt)
    return h_opt.tolist(), c5_bound, n_points
