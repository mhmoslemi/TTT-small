```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))
    def exact_objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    n_particles = 30
    n_iterations = 150
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    velocities = np.zeros_like(h)
    personal_bests = h.copy()
    global_best = h.copy()
    for _ in range(n_iterations):
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = w * velocities[i] + c1 * r1 * (personal_bests[i] - h[i]) + c2 * r2 * (global_best - h[i])
            h[i] = h[i] + velocities[i]
            h[i] = np.clip(h[i], 0.0, 1.0)
            sum_h = np.sum(h[i])
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h[i] = h[i] * scale
            c5 = exact_objective(h[i])
            if c5 < exact_objective(personal_bests[i]):
                personal_bests[i] = h[i].copy()
            if c5 < exact_objective(global_best):
                global_best = h[i].copy()
        if time.time() - start_time > budget_s - 5:
            break
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            exact_objective, global_best.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = exact_objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
    return best_h.tolist(), best_c5, n_points
```
