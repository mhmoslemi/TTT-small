import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.cos(np.pi * x))
        h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    h = h * (target_sum / sum_h)
    best_c5 = float('inf')
    best_h = h.copy()
    population_size = 20
    population = [h.copy() for _ in range(population_size)]
    start_time = time.time()
    for gen in range(500):
        fitness = []
        for h in population:
            h = np.clip(h, 0.0, 1.0)
            h = h * (target_sum / np.sum(h))
            corr = np.correlate(h, 1.0 - h, mode="full")
            overlaps = corr * dx
            max_overlap = np.max(overlaps)
            fitness.append(max_overlap)
        parents = np.argsort(fitness)[:5]
        selected = np.random.choice(parents, size=population_size, replace=True)
        new_population = []
        for i in range(population_size):
            parent_idx = selected[i]
            parent = population[parent_idx].copy()
            if np.random.rand() < 0.75:
                segment_len = int(np.random.rand() * n_points) + 1
                start_idx = np.random.randint(0, n_points - segment_len)
                end_idx = start_idx + segment_len
                parent[start_idx:end_idx] = 1.0 - parent[start_idx:end_idx]
            if np.random.rand() < 0.25:
                noise = np.random.normal(0, 0.1, n_points)
                parent += noise
                parent = np.clip(parent, 0.0, 1.0)
            parent = parent * (target_sum / np.sum(parent))
            new_population.append(parent)
        population = new_population
        for h in population:
            h = np.clip(h, 0.0, 1.0)
            h = h * (target_sum / np.sum(h))
            corr = np.correlate(h, 1.0 - h, mode="full")
            overlaps = corr * dx
            max_overlap = np.max(overlaps)
            if max_overlap < best_c5:
                best_c5 = max_overlap
                best_h = h.copy()
        if time.time() - start_time > budget_s - 5:
            break
    if time.time() - start_time < budget_s:
        h = best_h.copy()
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))
        def exact_objective(x):
            h = x.reshape(n_points)
            corr = np.correlate(h, 1.0 - h, mode="full")
            overlaps = corr * dx
            return np.max(overlaps)
        result = opt.minimize(exact_objective, h.flatten(), method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-8)
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        h_opt = h_opt * (target_sum / np.sum(h_opt))
        c5_bound = exact_objective(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
    return best_h.tolist(), best_c5, n_points
