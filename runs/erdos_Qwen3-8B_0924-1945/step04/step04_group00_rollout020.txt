```python
import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = (h < 0.5).astype(float)
        h = (h * target_sum) / np.sum(h)
    def compute_c5(h):
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        if np.random.rand() < 0.8:
            k = np.random.randint(1, n_points)
            h_k = h[k % n_points]
            h_k_next = h[(k + 1) % n_points]
            if h_k == 0 and h_k_next == 1:
                h[k % n_points] = 1
                h[(k + 1) % n_points] = 0
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
            k = np.random.randint(1, n_points)
            h_k = h[k % n_points]
            h_k_next = h[(k + 1) % n_points]
            if h_k == 0 and h_k_next == 1:
                h[k % n_points] = 1
                h[(k + 1) % n_points] = 0
            h = np.clip(h, 0.0, 1.0)
            h = h * (target_sum / np.sum(h))
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h.tolist(), best_c5, n_points
```
