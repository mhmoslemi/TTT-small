```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 600
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.sort(h)
        h = h - np.min(h)
        h = h / np.max(h)
        h = h * 0.5 + 0.5
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        scale = target_sum / sum_h
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    def de_objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        scale = target_sum / sum_h
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    # Differential Evolution
    pop_size = 30
    npop = pop_size
    population = np.random.rand(npop, n_points)
    population = np.clip(population, 0.0, 1.0)
    population = population * (target_sum / np.sum(population, axis=1, keepdims=True))
    best_de = population[0].copy()
    best_de_val = de_objective(best_de.flatten())

    for _ in range(100):
        for i in range(npop):
            a, b, c = np.random.choice(npop, 3, replace=False)
            mutant = population[a] + 0.5 * (population[b] - population[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = mutant * (target_sum / np.sum(mutant))
            trial = mutant + 0.1 * np.random.rand(n_points)
            trial = np.clip(trial, 0.0, 1.0)
            trial = trial * (target_sum / np.sum(trial))
            trial_val = de_objective(trial.flatten())
            if trial_val < best_de_val:
                best_de_val = trial_val
                best_de = trial.copy()
            if trial_val < de_objective(population[i].flatten()):
                population[i] = trial.copy()

        if time.time() - start_time > budget_s - 200:
            break

    h_opt = best_de.copy()
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt.flatten())
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    # Local refinement
    result = opt.minimize(
        objective, best_h.flatten(), 
        method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 500}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt.flatten())
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 200}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
