```python
import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h /= np.sum(h)
        h *= target_sum

    def evaluate(h):
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def normalize(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return h * (target_sum / s)

    def mutate(parent):
        a, b, c = np.random.choice(n_points, 3, replace=False)
        h = np.copy(parent)
        h[a] += np.random.uniform(-0.05, 0.05)
        h[b] += np.random.uniform(-0.05, 0.05)
        h[c] += np.random.uniform(-0.05, 0.05)
        return normalize(np.clip(h, 0.0, 1.0))

    def crossover(parent1, parent2):
        child = np.copy(parent1)
        mask = np.random.rand(n_points) < 0.5
        child[mask] = parent2[mask]
        return normalize(np.clip(child, 0.0, 1.0))

    population = [normalize(np.random.rand(n_points)) for _ in range(30)]
    population = [h for h in population if np.abs(np.sum(h) - target_sum) < 1e-6]
    best_h = population[0]
    best_c5 = evaluate(best_h)
    start_time = time.time()
    last_improvement = 0

    while time.time() - start_time < budget_s:
        if last_improvement > 100:
            break
        new_population = []
        for _ in range(len(population) // 2):
            parent1, parent2 = np.random.choice(population, 2, replace=False)
            child = crossover(parent1, parent2)
            new_population.append(child)
            child = mutate(child)
            new_population.append(child)
        population = new_population + population[:len(population) // 2]
        population = sorted(population, key=lambda x: evaluate(x))
        population = population[:20]
        current_c5 = evaluate(population[0])
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[0]
            last_improvement = 0
        else:
            last_improvement += 1

    return best_h.tolist(), best_c5, n_points
```
