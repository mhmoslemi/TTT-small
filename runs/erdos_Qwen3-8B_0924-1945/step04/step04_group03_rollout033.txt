```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 0.0, 1.0)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    n_particles = 30
    particles = np.random.rand(n_particles, n_points)
    particles = particles / np.sum(particles, axis=1, keepdims=True) * target_sum
    particles = np.clip(particles, 0.0, 1.0)
    pbest = particles.copy()
    pbest_values = np.array([objective(p) for p in pbest])
    gbest = pbest[np.argmin(pbest_values)]
    gbest_value = np.min(pbest_values)

    vel = np.random.rand(n_particles, n_points) * 0.01
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    best_c5 = gbest_value
    best_h = gbest.copy()
    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            vel[i] = inertia * vel[i] + cognitive * r1 * (pbest[i] - particles[i]) + social * r2 * (gbest - particles[i])
            particles[i] += vel[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_h = np.sum(particles[i])
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            particles[i] = particles[i] * scale
            current_val = objective(particles[i])
            if current_val < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_val
                if current_val < gbest_value:
                    gbest = pbest[i].copy()
                    gbest_value = current_val
                    best_c5 = gbest_value
                    best_h = gbest.copy()
        if np.random.rand() < 0.1:
            for i in range(n_particles):
                noise = np.random.uniform(-0.05, 0.05, size=n_points)
                particles[i] += noise
                particles[i] = np.clip(particles[i], 0.0, 1.0)
                sum_h = np.sum(particles[i])
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                particles[i] = particles[i] * scale

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
