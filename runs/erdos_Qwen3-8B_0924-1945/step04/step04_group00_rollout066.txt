import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.cos(np.pi * x))
        h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h))
    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    def repair(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        return h.flatten()
    def mutate(h_flat):
        h = h_flat.reshape(n_points)
        h = repair(h)
        mutation_rate = 0.75
        if np.random.rand() < mutation_rate:
            idx = np.random.choice(n_points, size=5, replace=False)
            h[idx] = 1.0 - h[idx]
        if np.random.rand() < mutation_rate:
            idx = np.random.choice(n_points, size=10, replace=False)
            h[idx] += np.random.normal(0, 0.1, size=10)
            h = np.clip(h, 0.0, 1.0)
        if np.random.rand() < mutation_rate:
            shift = np.random.uniform(-0.05, 0.05)
            idx = np.random.choice(n_points, size=1, replace=False)
            h[idx] += shift
            h = np.clip(h, 0.0, 1.0)
        return h.flatten()
    def crossover(p1, p2):
        p1 = p1.reshape(n_points)
        p2 = p2.reshape(n_points)
        p1 = repair(p1)
        p2 = repair(p2)
        idx = np.random.randint(0, n_points)
        child = np.concatenate([p1[:idx], p2[idx:]])
        return child.flatten()
    population = [mutate(h.copy()) for _ in range(20)]
    best_h = population[0]
    best_c5 = evaluate(best_h)
    start_time = time.time()
    for _ in range(500):
        fitness = [evaluate(ind) for ind in population]
        selected = np.argsort(fitness)[:5]
        parents = [population[i] for i in selected]
        new_population = []
        for _ in range(len(population) // 2):
            p1, p2 = np.random.choice(parents, size=2, replace=False)
            child = crossover(p1, p2)
            child = mutate(child)
            new_population.append(child)
        population = new_population + [np.random.uniform(0, 1, n_points) for _ in range(len(population) - len(new_population))]
        population = [repair(ind) for ind in population]
        for ind in population:
            current_c5 = evaluate(ind)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = ind
        if time.time() - start_time > budget_s - 5:
            break
    if time.time() - start_time < budget_s:
        for _ in range(10):
            x = np.random.uniform(0, 1, n_points)
            x = repair(x)
            result = opt.minimize(
                lambda h_flat: evaluate(h_flat),
                x,
                method='L-BFGS-B',
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                tol=1e-8
            )
            if result.success:
                h = result.x
                c5 = evaluate(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
    return best_h.tolist(), best_c5, n_points
