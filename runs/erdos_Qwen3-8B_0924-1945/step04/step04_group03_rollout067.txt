import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if np.random.rand() < 0.5 else 0
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def normalize(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        return h * scale

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()

    pop_size = 30
    elite_ratio = 0.2
    mutation_factor = 0.5
    crossover_prob = 1.0
    gen_limit = 2000
    improvement_patience = 50

    population = [np.random.rand(n_points) for _ in range(pop_size)]
    population = [normalize(p) for p in population]

    best_fitness = [evaluate(p) for p in population]
    best_index = np.argmin(best_fitness)
    best_h = population[best_index].copy()
    best_c5 = best_fitness[best_index]

    no_improvement = 0
    for gen in range(gen_limit):
        if time.time() - start_time > budget_s - 5:
            break
        elites = np.argsort(best_fitness)[:int(elite_ratio * pop_size)]
        new_pop = []
        for _ in range(pop_size):
            a, b, c = np.random.choice(elites, 3, replace=False)
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = normalize(mutant)
            new_pop.append(mutant)
        population = new_pop
        fitness = [evaluate(p) for p in population]
        for i in range(pop_size):
            if fitness[i] < best_fitness[i]:
                best_fitness[i] = fitness[i]
        current_best = np.min(best_fitness)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[np.argmin(best_fitness)].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        if no_improvement >= improvement_patience:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(100):
            if time.time() - start_time > budget_s - 5:
                break
            a, b, c = np.random.choice(range(pop_size), 3, replace=False)
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = normalize(mutant)
            fitness = evaluate(mutant)
            if fitness < best_c5:
                best_c5 = fitness
                best_h = mutant.copy()

    return best_h.tolist(), best_c5, n_points
