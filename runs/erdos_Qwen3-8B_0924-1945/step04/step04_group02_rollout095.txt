import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0.0, 1.0, size=n_points)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    n_particles = 40
    particles = np.random.uniform(0.0, 1.0, (n_particles, n_points))
    particles = particles * (target_sum / np.sum(particles, axis=1, keepdims=True))
    p_bests = particles.copy()
    velocities = np.random.uniform(-0.05, 0.05, (n_particles, n_points))
    inertia = 0.9
    c1, c2 = 2.0, 2.0
    mutation_amp = 0.05
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    for gen in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(n_particles):
            h_flat = particles[i, :]
            current_c5 = objective(h_flat)
            if current_c5 < objective(p_bests[i, :]):
                p_bests[i, :] = h_flat.copy()
        g_best = p_bests[np.argmin([objective(p) for p in p_bests])]
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i, :] = (inertia * velocities[i, :] +
                               c1 * r1 * (p_bests[i, :] - particles[i, :]) +
                               c2 * r2 * (g_best - particles[i, :]))
            particles[i, :] += velocities[i, :]
            particles[i, :] = np.clip(particles[i, :], 0.0, 1.0)
            sum_h = np.sum(particles[i, :])
            particles[i, :] *= target_sum / sum_h if sum_h != 0 else 1.0
        if gen % 5 == 0:
            for i in range(n_particles):
                if np.random.rand() < 0.1:
                    particles[i, :] += np.random.normal(0.0, mutation_amp, size=n_points)
                    particles[i, :] = np.clip(particles[i, :], 0.0, 1.0)
                    sum_h = np.sum(particles[i, :])
                    particles[i, :] *= target_sum / sum_h if sum_h != 0 else 1.0
        for i in range(n_particles):
            h_flat = p_bests[i, :]
            c5 = objective(h_flat)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_flat.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
