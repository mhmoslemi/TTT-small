import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.choice([0, 1], size=n_points)
        h = h.astype(np.float64)
        h = h * target_sum / np.sum(h)
    def compute_overlap(h):
        h = np.clip(h, 0.0, 1.0)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    best_h = h.copy()
    best_c5 = compute_overlap(h)
    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time > budget_s - 10:
            break
        h = np.clip(h, 0.0, 1.0)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1.0 - h, mode="full")
        min_overlap = np.min(corr * dx)
        min_idx = np.argmin(corr * dx)
        k = min_idx + 1
        if k >= n_points:
            k = n_points - 1
        mask = (np.roll(h, k) == 1) & (h == 0)
        if np.any(mask):
            idx = np.random.choice(np.where(mask)[0])
            h[idx] = 1
        h = np.clip(h, 0.0, 1.0)
        h = h * target_sum / np.sum(h)
        current_c5 = compute_overlap(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(100):
            if time.time() - start_time > budget_s - 10:
                break
            h = np.clip(best_h, 0.0, 1.0)
            h = h * target_sum / np.sum(h)
            corr = np.correlate(h, 1.0 - h, mode="full")
            min_overlap = np.min(corr * dx)
            min_idx = np.argmin(corr * dx)
            k = min_idx + 1
            if k >= n_points:
                k = n_points - 1
            mask = (np.roll(h, k) == 1) & (h == 0)
            if np.any(mask):
                idx = np.random.choice(np.where(mask)[0])
                h[idx] = 1
            h = np.clip(h, 0.0, 1.0)
            h = h * target_sum / np.sum(h)
            current_c5 = compute_overlap(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    return best_h.tolist(), best_c5, n_points
