```python
import numpy as np
import scipy.optimize as opt
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, WhiteKernel

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 500
    fallback_factor = 1.01

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        m = 4
        l = 1.0 / m
        g = (1.0 - m * l) / 2
        cluster_positions = np.arange(m) * (l + g) + g
        h = np.zeros(n_points)
        for pos in cluster_positions:
            idx = np.searchsorted(np.linspace(0, 2, n_points), pos)
            h[idx - 1:idx + int(round(l / dx))] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    samples = []
    values = []

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def ei(X, model, y_max):
        mu, sigma = model.predict(X, return_std=True)
        sigma = np.maximum(sigma, 1e-8)
        imp = y_max - mu
        Z = imp / sigma
        cdf = np.vectorize(lambda x: np.sqrt(2 * np.pi) ** -1 * np.integrate(lambda t: np.exp(-0.5 * t**2), -np.inf, x, limit=1000, reltol=1e-7))(Z)
        pdf = np.vectorize(lambda x: np.exp(-0.5 * x**2) / np.sqrt(2 * np.pi))(Z)
        ei = imp * (cdf + (1 - np.exp(-Z**2 / 2)) / Z)
        ei[Z < 1e-5] = 0
        return ei

    X0 = h.copy()
    for _ in range(50):
        X0 = np.clip(X0, 0.0, 1.0)
        sum_h = np.sum(X0)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        X0 = X0 * scale
        samples.append(X0)
        values.append(evaluate(X0))
        X0 += np.random.uniform(-0.05, 0.05, n_points)
        X0 = np.clip(X0, 0.0, 1.0)
        sum_h = np.sum(X0)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        X0 = X0 * scale

    X = np.array(samples)
    y = np.array(values)
    kernel = RBF(length_scale=1.0, length_scale_bounds=(1e-5, 1e5)) + WhiteKernel(noise_level=1e-5, noise_level_bounds=(1e-10, 1e-3))
    model = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=25, alpha=1e-10)
    model.fit(X, y)

    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 50:
            break
        if len(samples) < 20:
            X_new = np.random.uniform(0, 1, (1, n_points))
            X_new = np.clip(X_new, 0.0, 1.0)
            sum_h = np.sum(X_new)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            X_new = X_new * scale
            samples.append(X_new)
            values.append(evaluate(X_new))
        else:
            X_opt = np.random.rand(n_points)
            X_opt = np.clip(X_opt, 0.0, 1.0)
            sum_h = np.sum(X_opt)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            X_opt = X_opt * scale
            y_opt = evaluate(X_opt)
            if y_opt < np.min(values):
                samples.append(X_opt)
                values.append(y_opt)
            else:
                X_opt = np.random.rand(n_points)
                X_opt = np.clip(X_opt, 0.0, 1.0)
                sum_h = np.sum(X_opt)
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                X_opt = X_opt * scale
                samples.append(X_opt)
                values.append(evaluate(X_opt))
        X = np.array(samples)
        y = np.array(values)
        model = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=25, alpha=1e-10)
        model.fit(X, y)

    X_eval = np.random.rand(n_points)
    X_eval = np.clip(X_eval, 0.0, 1.0)
    sum_h = np.sum(X_eval)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    X_eval = X_eval * scale
    best_c5 = evaluate(X_eval)
    best_h = X_eval.copy()

    for _ in range(100):
        if time.time() - start_time > budget_s - 50:
            break
        X_new = np.random.rand(n_points)
        X_new = np.clip(X_new, 0.0, 1.0)
        sum_h = np.sum(X_new)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        X_new = X_new * scale
        y_new = evaluate(X_new)
        if y_new < best_c5:
            best_c5 = y_new
            best_h = X_new.copy()

    if time.time() - start_time < budget_s - 5:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
