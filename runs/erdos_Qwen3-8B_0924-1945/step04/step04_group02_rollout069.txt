```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        positions = np.random.uniform(0, 2, size=20)
        widths = np.random.uniform(0.1, 0.3, size=20)
        for pos, w in zip(positions, widths):
            idx = np.searchsorted(np.linspace(0, 2, n_points), pos)
            h[idx - 2:idx + int(round(w / dx))] = 1
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    n_particles = 20
    swarm = np.random.uniform(0.0, 1.0, (n_particles, n_points))
    swarm = swarm * (target_sum / np.sum(swarm, axis=1, keepdims=True))
    swarm = np.clip(swarm, 0.0, 1.0)
    personal_best = swarm.copy()
    global_best = swarm[0].copy()
    best_c5 = evaluate(global_best)

    for _ in range(200):
        inertia = 0.5 + 0.4 * (_ / 200)
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            vel = inertia * np.random.rand(n_points) + r1 * (personal_best[i] - swarm[i]) + r2 * (global_best - swarm[i])
            swarm[i] += vel
            swarm[i] = np.clip(swarm[i], 0.0, 1.0)
            sum_h = np.sum(swarm[i])
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            swarm[i] = swarm[i] * scale
            current_c5 = evaluate(swarm[i])
            if current_c5 < evaluate(personal_best[i]):
                personal_best[i] = swarm[i].copy()
            if current_c5 < best_c5:
                best_c5 = current_c5
                global_best = swarm[i].copy()
        if time.time() - start_time > budget_s - 5:
            break

    result = opt.minimize(
        lambda h_flat: evaluate(h_flat), best_h.flatten(),
        method='L-BFGS-B', bounds=bounds,
        tol=1e-8, options={'maxiter': 50}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = evaluate(h_opt)
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
