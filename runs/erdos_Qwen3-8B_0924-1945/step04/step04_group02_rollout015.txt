import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        l = 0.5
        g = 0.1
        cluster_positions = np.arange(4) * (l + g) + g
        cluster_positions = np.clip(cluster_positions, 0, 2)
        for pos in cluster_positions:
            idx = np.searchsorted(np.linspace(0, 2, n_points), pos)
            h[idx - 1:idx + int(round(l / dx))] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    n_particles = 20
    particles = [np.random.uniform(0, 1, n_points) for _ in range(n_particles)]
    particles = [p * (target_sum / np.sum(p)) for p in particles]
    personal_best = particles[:]
    global_best = min(particles, key=objective)
    best_c5 = objective(global_best)

    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(n_particles):
            h = particles[i]
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_scaled = h * scale
            corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
            c5 = np.max(corr * dx)
            if c5 < objective(personal_best[i]):
                personal_best[i] = h.copy()
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                global_best = h.copy()
        for i in range(n_particles):
            h = particles[i]
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_scaled = h * scale
            corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
            c5 = np.max(corr * dx)
            best_p = personal_best[i]
            best_g = global_best
            r1 = np.random.rand()
            r2 = np.random.rand()
            cog = r1 * (best_p - h)
            soc = r2 * (best_g - h)
            vel = 0.5 * cog + 0.5 * soc
            h_new = h + vel
            h_new = np.clip(h_new, 0.0, 1.0)
            sum_h_new = np.sum(h_new)
            scale_new = target_sum / sum_h_new if sum_h_new != 0 else 1.0
            h_new = h_new * scale_new
            particles[i] = h_new

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
