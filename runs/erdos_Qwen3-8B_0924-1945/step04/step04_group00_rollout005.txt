import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))
    def exact_objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    def repair(h):
        h = h.copy()
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = target_sum / sum_h
            h = h * scale
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = target_sum / sum_h
            h = h * scale
        return h
    def mutate(h, mutation_rate=0.1, sigma=0.1):
        h = h.copy()
        h = repair(h)
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.normal(0, sigma)
                h[i] = np.clip(h[i], 0.0, 1.0)
        return repair(h)
    def crossover(parent1, parent2):
        h = np.random.rand(n_points)
        h = h * parent1 + (1 - h) * parent2
        return repair(h)
    best_c5 = float('inf')
    best_h = h.copy()
    population = [repair(np.random.rand(n_points)) for _ in range(20)]
    start_time = time.time()
    for gen in range(500):
        fitness = [exact_objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
        if time.time() - start_time > budget_s - 5:
            break
        new_population = []
        for _ in range(20):
            parent1, parent2 = np.random.choice(population, 2, replace=False)
            child = crossover(parent1, parent2)
            if np.random.rand() < 0.7:
                child = mutate(child)
            new_population.append(child)
        population = new_population
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(10):
            h_opt = best_h.copy()
            h_opt = repair(h_opt)
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            result = opt.minimize(lambda x: exact_objective(x), h_opt, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            h_opt = result.x
            h_opt = repair(h_opt)
            c5_bound = exact_objective(h_opt)
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h = h_opt.copy()
    return best_h.tolist(), best_c5, n_points
