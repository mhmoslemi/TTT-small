import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))
    def exact_objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    def normalize(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        return h * (target_sum / sum_h)
    n_particles = 30
    n_iterations = 100
    omega = 0.7
    phi_p = 1.5
    phi_g = 1.5
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    particles = np.random.uniform(0, 1, (n_particles, n_points))
    particles = np.array([normalize(p) for p in particles])
    personal_best = particles.copy()
    personal_best_vals = np.array([exact_objective(p.flatten()) for p in personal_best])
    global_best = particles[0].copy()
    global_best_val = personal_best_vals[0]
    start_time = time.time()
    for _ in range(n_iterations):
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = omega * velocities[i] + phi_p * r1 * (personal_best[i] - particles[i]) + phi_g * r2 * (global_best - particles[i])
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            particles[i] = normalize(particles[i])
            current_val = exact_objective(particles[i].flatten())
            if current_val < personal_best_vals[i]:
                personal_best[i] = particles[i].copy()
                personal_best_vals[i] = current_val
                if current_val < global_best_val:
                    global_best = particles[i].copy()
                    global_best_val = current_val
        if _ % 10 == 0:
            noise = np.random.normal(0, 0.05, size=n_points)
            particles += noise
            particles = np.clip(particles, 0.0, 1.0)
            particles = np.array([normalize(p) for p in particles])
    if time.time() - start_time < budget_s:
        result = opt.minimize(
            exact_objective, global_best.flatten(), 
            method='L-BFGS-B', bounds=[(0.0, 1.0) for _ in range(n_points)], 
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = exact_objective(h_opt.flatten())
        if c5_bound < global_best_val:
            global_best = h_opt.copy()
            global_best_val = c5_bound
    return global_best.tolist(), global_best_val, n_points
