import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = (h > 0.5).astype(np.float64)
        h = h * (target_sum / np.sum(h))
    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_overlap = np.max(corr * dx)
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        if time.time() - start_time > budget_s - 5:
            break
        k = np.argmin(corr * dx)
        if k < len(h) - 1:
            indices = np.where((h[:len(h)-k] == 0) & (h[k:] == 1))
            if len(indices[0]) > 0:
                i = np.random.choice(indices[0])
                h[i] = 1.0
                h[i + k] = 0.0
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(10):
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            corr = np.correlate(h, 1.0 - h, mode="full")
            max_overlap = np.max(corr * dx)
            if max_overlap < best_c5:
                best_c5 = max_overlap
                best_h = h.copy()
            if time.time() - start_time > budget_s - 5:
                break
            k = np.argmin(corr * dx)
            if k < len(h) - 1:
                indices = np.where((h[:len(h)-k] == 0) & (h[k:] == 1))
                if len(indices[0]) > 0:
                    i = np.random.choice(indices[0])
                    h[i] = 1.0
                    h[i + k] = 0.0
    return best_h.tolist(), best_c5, n_points
