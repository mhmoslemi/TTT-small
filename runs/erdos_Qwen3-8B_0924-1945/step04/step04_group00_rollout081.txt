```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.cos(np.pi * x))
        h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h))
    best_c5 = float('inf')
    best_h = h.copy()
    pop_size = 20
    max_gen = 500
    elite_size = 5
    mut_rate = 0.75
    cross_rate = 0.8
    tournament_size = 5
    gen = 0
    population = [h.copy() for _ in range(pop_size)]
    start_time = time.time()
    while gen < max_gen and time.time() - start_time < budget_s - 5:
        fitness = []
        for h in population:
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            corr = np.correlate(h, 1.0 - h, mode="full")
            overlaps = corr * dx
            max_overlap = np.max(overlaps)
            if max_overlap < 1e-10:
                fitness.append(1e-10)
            else:
                fitness.append(max_overlap)
        indices = np.argsort(fitness)
        selected = [population[i] for i in indices[:elite_size]]
        new_population = []
        for _ in range(pop_size - elite_size):
            parent1, parent2 = np.random.choice(selected, size=2, replace=False)
            if np.random.rand() < cross_rate:
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:split], parent2[split:]])
            else:
                child = np.random.rand(n_points)
            if np.random.rand() < mut_rate:
                seg_len = int(n_points * 0.05)
                seg_start = np.random.randint(0, n_points - seg_len)
                child[seg_start:seg_start + seg_len] *= np.random.uniform(0.8, 1.2)
                child = np.clip(child, 0.0, 1.0)
                sum_child = np.sum(child)
                scale = target_sum / sum_child if sum_child != 0 else 1.0
                child = child * scale
            new_population.append(child)
        population = selected + new_population
        for h in population:
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            corr = np.correlate(h, 1.0 - h, mode="full")
            overlaps = corr * dx
            max_overlap = np.max(overlaps)
            if max_overlap < best_c5:
                best_c5 = max_overlap
                best_h = h.copy()
        gen += 1
    h = best_h
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h = h * scale
    corr = np.correlate(h, 1.0 - h, mode="full")
    overlaps = corr * dx
    c5_bound = np.max(overlaps)
    return h.tolist(), c5_bound, n_points
```
