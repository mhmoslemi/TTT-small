import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        indices = np.round(np.linspace(0, n_points - 1, n_points // 2)).astype(int)
        h = np.zeros(n_points)
        h[indices] = 1
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    variants = []
    # Base variant
    variants.append(h.copy())
    # Symmetric variant
    h_sym = h.copy()
    h_sym[n_points//2:] = h[:n_points//2][::-1]
    variants.append(h_sym)
    # Uniform variant
    h_uni = np.zeros(n_points)
    spacing = n_points // 2
    for i in range(n_points // 2):
        h_uni[i * spacing] = 1
    h_uni = np.clip(h_uni, 0, 1)
    h_uni = h_uni * (target_sum / np.sum(h_uni))
    variants.append(h_uni)
    # Staggered variant
    h_stag = np.zeros(n_points)
    for i in range(n_points // 2):
        h_stag[i * spacing + i % 3] = 1
    h_stag = np.clip(h_stag, 0, 1)
    h_stag = h_stag * (target_sum / np.sum(h_stag))
    variants.append(h_stag)
    # Random variant
    h_ran = np.random.rand(n_points)
    h_ran = np.clip(h_ran, 0, 1)
    h_ran = h_ran * (target_sum / np.sum(h_ran))
    variants.append(h_ran)

    for variant in variants:
        h_opt = variant.copy()
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    for _ in range(20):
        h = best_h.copy()
        noise = np.random.normal(0, 0.05, size=n_points)
        h += noise
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        c5_bound = objective(h.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h.copy()

    if time.time() - start_time > budget_s - 5:
        return best_h.tolist(), best_c5, n_points

    result = opt.minimize(
        objective, best_h.flatten(), 
        method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 50}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt.flatten())
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
