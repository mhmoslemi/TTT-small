```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 if (i % 4 == 0) else 0.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    particles = [np.random.uniform(0, 1, n_points) for _ in range(20)]
    particles = [p * (target_sum / np.sum(p)) for p in particles]
    best_particles = particles[:]

    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(len(particles)):
            p = particles[i].copy()
            p = np.clip(p, 0.0, 1.0)
            sum_p = np.sum(p)
            scale = target_sum / sum_p if sum_p != 0 else 1.0
            p = p * scale
            c5 = objective(p)
            if c5 < objective(best_particles[0]):
                best_particles[0] = p.copy()
            if np.random.rand() < 0.1:
                noise = np.random.uniform(-0.05, 0.05, size=n_points)
                p += noise
                p = np.clip(p, 0.0, 1.0)
                sum_p = np.sum(p)
                scale = target_sum / sum_p if sum_p != 0 else 1.0
                p = p * scale
            particles[i] = p.copy()

    for i in range(len(best_particles)):
        p = best_particles[i].copy()
        p = np.clip(p, 0.0, 1.0)
        sum_p = np.sum(p)
        scale = target_sum / sum_p if sum_p != 0 else 1.0
        p = p * scale
        c5 = objective(p)
        if c5 < best_c5:
            best_c5 = c5
            best_h = p.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
