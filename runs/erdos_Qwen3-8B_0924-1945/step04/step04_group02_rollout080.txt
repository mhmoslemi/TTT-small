import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 if i % 20 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    particles = [h.copy() for _ in range(20)]
    velocities = [np.random.uniform(-0.05, 0.05, n_points) for _ in range(20)]
    personal_best = [p.copy() for p in particles]
    global_best = best_h.copy()
    for _ in range(200):
        for i in range(20):
            h = particles[i].copy()
            v = velocities[i]
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_scaled = h * scale
            corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
            c5 = np.max(corr * dx)
            if c5 < objective(personal_best[i]):
                personal_best[i] = h.copy()
            if c5 < objective(global_best):
                global_best = h.copy()
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            cognitive = np.random.uniform(0.0, 0.5, n_points)
            social = np.random.uniform(0.0, 0.5, n_points)
            inertia = 0.8 + 0.2 * (_ / 200)
            v = inertia * v + cognitive * r1 * (personal_best[i] - h) + social * r2 * (global_best - h)
            velocities[i] = v
            h = h + v
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_scaled = h * scale
            particles[i] = h.copy()
        if time.time() - start_time > budget_s - 5:
            break
        h = np.mean(particles, axis=0)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        c5_bound = objective(h)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
