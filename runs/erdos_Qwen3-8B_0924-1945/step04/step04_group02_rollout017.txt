import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    n_particles = 40
    particles = [np.random.uniform(0, 1, n_points) for _ in range(n_particles)]
    particles = [p * (target_sum / np.sum(p)) for p in particles]
    pbest = particles.copy()
    gbest = min(pbest, key=lambda x: objective(x.flatten()))
    best_fitness = objective(gbest.flatten())

    for gen in range(200):
        if time.time() - start_time > budget_s - 10:
            break
        velocities = [np.random.uniform(-0.05, 0.05, n_points) for _ in range(n_particles)]
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            v = velocities[i]
            p = particles[i]
            best_p = pbest[i]
            best_g = gbest
            new_v = 0.9 * v + 2.0 * r1 * (best_p - p) + 2.0 * r2 * (best_g - p)
            new_p = p + new_v
            new_p = np.clip(new_p, 0.0, 1.0)
            sum_new_p = np.sum(new_p)
            scale_new_p = target_sum / sum_new_p if sum_new_p != 0 else 1.0
            new_p = new_p * scale_new_p
            new_fitness = objective(new_p.flatten())
            if new_fitness < objective(pbest[i].flatten()):
                pbest[i] = new_p
                if new_fitness < best_fitness:
                    best_fitness = new_fitness
                    gbest = new_p
            particles[i] = new_p
        if gen % 5 == 0:
            elites = sorted(pbest, key=lambda x: objective(x.flatten()))[:int(n_particles * 0.05)]
            for e in elites:
                result = opt.minimize(
                    objective, e.flatten(), 
                    method='L-BFGS-B', bounds=bounds, constraints=cons,
                    tol=1e-8, options={'maxiter': 50}
                )
                h_opt = result.x.reshape(n_points)
                h_opt = np.clip(h_opt, 0.0, 1.0)
                sum_h_opt = np.sum(h_opt)
                scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
                h_opt = h_opt * scale
                c5_bound = objective(h_opt.flatten())
                if c5_bound < best_fitness:
                    best_fitness = c5_bound
                    gbest = h_opt
        if best_fitness < best_c5:
            best_c5 = best_fitness
            best_h = gbest.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
