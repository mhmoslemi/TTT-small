import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        base_pattern = np.zeros(n_points, dtype=np.float64)
        block_size = n_points // 4
        for i in range(n_points):
            if i % block_size < block_size // 3:
                base_pattern[i] = 1.0
        h = base_pattern.copy()
        h = h * (target_sum / np.sum(h))

    def exact_objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break

        # Genetic algorithm phase
        population = [np.random.rand(n_points) for _ in range(100)]
        for p in population:
            p = p * (target_sum / np.sum(p)) if np.sum(p) != 0 else p
            p = np.clip(p, 0.0, 1.0)

        evaluated = []
        for p in population:
            p_reshaped = p.reshape(n_points)
            p_reshaped = np.clip(p_reshaped, 0.0, 1.0)
            sum_p = np.sum(p_reshaped)
            scale = target_sum / sum_p if sum_p != 0 else 1.0
            p_reshaped = p_reshaped * scale
            corr = np.correlate(p_reshaped, 1.0 - p_reshaped, mode="full")
            c5 = np.max(corr * dx)
            evaluated.append((c5, p_reshaped))
        evaluated.sort()
        best_candidate = evaluated[0][1]

        # Local refinement with SLSQP
        result = opt.minimize(
            exact_objective, best_candidate.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = exact_objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
        h = h_opt.copy()

        if _ % 10 == 0:
            noise = np.random.normal(0, 0.05, size=n_points)
            h = h + noise
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            exact_objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = exact_objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
