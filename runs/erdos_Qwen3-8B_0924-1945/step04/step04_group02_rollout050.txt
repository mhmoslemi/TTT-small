```python
import numpy as np
import scipy.optimize as opt
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, WhiteKernel

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 0.0, 1.0)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    X = [h.tolist()]
    y = [objective(h)]
    gp_kernel = RBF(length_scale=1.0) + WhiteKernel(noise_level=1e-5)
    gp = GaussianProcessRegressor(kernel=gp_kernel, alpha=1e-10)

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    iterations = 0

    while time.time() - start_time < budget_s - 100 and iterations < 500:
        gp.fit(X, y)
        mu, sigma = gp.predict(X, return_std=True)
        mu = np.array(mu)
        sigma = np.array(sigma)

        def ei(x):
            mu_x = gp.predict([x.tolist()])
            sigma_x = np.sqrt(gp.kernel_(x, x) + 1e-8)
            imp = mu_x - best_c5
            imp[imp < 0] = 0
            return imp * np.exp(-0.5 * (mu_x - best_c5)**2 / sigma_x**2) / sigma_x

        def acquisition(x):
            return -np.min(ei(x))

        def bounds_transform(x):
            return np.clip(x, 0.0, 1.0)

        def bounds_transform_inv(x):
            return x

        def objective_with_transform(x):
            x = bounds_transform(x)
            return objective(x)

        def constraint(x):
            x = bounds_transform(x)
            return np.sum(x) - target_sum

        def constraint_with_transform(x):
            return constraint(x)

        def bounds_with_transform(x):
            return [0.0, 1.0] * len(x)

        def bounds_transform_func(x):
            return bounds_transform(x)

        def bounds_transform_inv_func(x):
            return bounds_transform_inv(x)

        def constraint_with_transform_func(x):
            return constraint_with_transform(x)

        def bounds_with_transform_func(x):
            return bounds_with_transform(x)

        def objective_with_transform_func(x):
            return objective_with_transform(x)

        def constraint_with_transform_func_func(x):
            return constraint_with_transform_func(x)

        def bounds_with_transform_func_func(x):
            return bounds_with_transform_func(x)

        def objective_with_transform_func_func(x):
            return objective_with_transform_func(x)

        result = opt.minimize(
            objective_with_transform_func,
            h.flatten(),
            method='L-BFGS-B',
            bounds=bounds_with_transform_func,
            constraints={'type': 'eq', 'fun': constraint_with_transform_func},
            tol=1e-8,
            options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = bounds_transform(h_opt)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
            X.append(h_opt.tolist())
            y.append(c5_bound)
            if len(X) > 20:
                X.pop(0)
                y.pop(0)
        h = h_opt.copy()
        iterations += 1

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
