```python
import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.cos(np.pi * x))
        h = h * (target_sum / np.sum(h))

    def compute_c5(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def normalize(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        return h.flatten()

    n_particles = 30
    n_iterations = 300
    w = 0.7
    c1, c2 = 1.5, 1.5
    max_vel = 0.1
    best_h = h.copy()
    best_c5 = compute_c5(h)

    particles = np.random.uniform(0, 1, (n_particles, n_points))
    particles = normalize(particles)
    velocities = np.zeros((n_particles, n_points))
    pbest = particles.copy()
    pbest_c5 = np.array([compute_c5(p) for p in pbest])
    gbest = pbest[np.argmin(pbest_c5)]
    gbest_c5 = np.min(pbest_c5)

    start_time = time.time()
    for _ in range(n_iterations):
        if time.time() - start_time > budget_s - 10:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = w * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            particles[i] = particles[i] + velocities[i]
            particles[i] = normalize(particles[i])
            current_c5 = compute_c5(particles[i])
            if current_c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = current_c5
        if np.min(pbest_c5) < gbest_c5:
            gbest = pbest[np.argmin(pbest_c5)]
            gbest_c5 = np.min(pbest_c5)
        if gbest_c5 < best_c5:
            best_h = gbest.copy()
            best_c5 = gbest_c5

    return best_h.tolist(), best_c5, n_points
```
