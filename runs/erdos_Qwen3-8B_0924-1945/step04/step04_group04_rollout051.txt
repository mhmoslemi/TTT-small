import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 600
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        if np.abs(np.sum(h) - target_sum) > 1e-5:
            h = h * target_sum / np.sum(h)
    else:
        x_grid = np.linspace(0, 2, n_points)
        angles = np.linspace(0, 2 * np.pi, n_points, endpoint=False)
        h = np.random.rand(n_points)
        h = np.clip(h, 0.0, 1.0)
        h = h * target_sum / np.sum(h)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # Phase 1: Differential Evolution for global search
    def de_objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    population = np.random.rand(30, n_points)
    population = np.clip(population, 0.0, 1.0)
    population = population * target_sum / np.sum(population, axis=1, keepdims=True)

    for _ in range(100):
        for i in range(population.shape[0]):
            a, b, c = np.random.choice(population.shape[0], 3, replace=False)
            mutant = population[a] + 0.5 * (population[b] - population[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = mutant * target_sum / np.sum(mutant)
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = mutant * target_sum / np.sum(mutant)
            trial = population[i] + 0.9 * (mutant - population[i])
            trial = np.clip(trial, 0.0, 1.0)
            trial = trial * target_sum / np.sum(trial)
            trial = np.clip(trial, 0.0, 1.0)
            if objective(trial.flatten()) < objective(population[i].flatten()):
                population[i] = trial

        current_best = np.min([objective(p.flatten()) for p in population])
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[np.argmin([objective(p.flatten()) for p in population])].copy()

        if time.time() - start_time > budget_s - 300:
            break

    # Phase 2: Local refinement with L-BFGS-B
    result = opt.minimize(
        objective, best_h.flatten(), 
        method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 500}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt.flatten())
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 200}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
