```python
import numpy as np
import scipy.optimize as opt
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, WhiteKernel

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        h_scaled = h * (target_sum / sum_h)
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)
    
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    
    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf'), h_flat
        h_scaled = h * (target_sum / sum_h)
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx), h_flat
    
    def expected_improvement(X, X_candidate, y, gp):
        mean, std = gp.predict(X_candidate, return_std=True)
        best = np.min(y)
        imp = mean - best
        theta = 1e-3
        with np.errstate(divide='ignore', invalid='ignore'):
            Z = np.maximum(0, (imp - theta) / std)
        ei = imp * np.exp(-0.5 * Z**2) * Z
        return np.sum(ei)
    
    X_initial = np.array([h]).reshape(1, -1)
    y_initial = [objective(h)]
    
    gp_kernel = RBF(length_scale=1.0, length_scale_bounds=(1e-5, 1e5)) + WhiteKernel(noise_level=1e-5)
    gp = GaussianProcessRegressor(kernel=gp_kernel, alpha=1e-10, n_restarts_optimizer=10)
    gp.fit(X_initial, y_initial)
    
    X_candidates = np.random.uniform(0, 1, (20, n_points))
    X_candidates = X_candidates * (target_sum / np.sum(X_candidates, axis=1, keepdims=True))
    X_candidates = X_candidates.reshape(-1, n_points)
    y_candidates = np.array([objective(x) for x in X_candidates])
    
    X_train = X_candidates
    y_train = y_candidates
    gp.fit(X_train, y_train)
    
    start_time = time.time()
    
    for _ in range(500):
        X_candidate = np.random.rand(1, n_points)
        X_candidate = X_candidate * (target_sum / np.sum(X_candidate, axis=1, keepdims=True))
        ei = expected_improvement(X_train, X_candidate, y_train, gp)
        if ei < 1e-3:
            break
        X_candidate = X_candidate.reshape(1, -1)
        y_candidate, _ = evaluate(X_candidate.reshape(-1))
        X_train = np.vstack((X_train, X_candidate))
        y_train = np.append(y_train, y_candidate)
        gp.fit(X_train, y_train)
    
    y_train = np.array([objective(x) for x in X_train])
    best_idx = np.argmin(y_train)
    best_h = X_train[best_idx].reshape(n_points)
    best_c5 = y_train[best_idx]
    
    for _ in range(50):
        result = opt.minimize(
            objective, best_h,
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 10}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0.0, 1.0)
        sum_h = np.sum(best_h)
        best_h = best_h * (target_sum / sum_h)
        best_c5 = objective(best_h)
        if best_c5 < 0.38085:
            break
    
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 5:
        result = opt.minimize(
            objective, best_h,
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0.0, 1.0)
        sum_h = np.sum(best_h)
        best_h = best_h * (target_sum / sum_h)
        best_c5 = objective(best_h)
    
    return best_h.tolist(), best_c5, n_points
```
