```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 600
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        angles = np.linspace(0, 2 * np.pi, n_points, endpoint=False)
        h = np.random.rand(n_points)
        h = (h < 0.5).astype(float)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # Differential evolution for global search
    def de_objective(x):
        x = np.clip(x, 0.0, 1.0)
        sum_x = np.sum(x)
        scale = target_sum / sum_x if sum_x != 0 else 1.0
        scaled = x * scale
        corr = np.correlate(scaled, 1.0 - scaled, mode="full")
        return np.max(corr * dx)

    # DE parameters
    pop_size = 30
    mutation = 0.5
    crossover = 0.9
    max_iter = 100
    elite_size = 5
    pop = np.random.rand(pop_size, n_points)
    pop = (pop < 0.5).astype(float)
    pop = pop * (target_sum / np.sum(pop, axis=1, keepdims=True))
    pop = np.clip(pop, 0.0, 1.0)

    for _ in range(max_iter):
        offspring = []
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + mutation * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant_sum = np.sum(mutant)
            scale = target_sum / mutant_sum if mutant_sum != 0 else 1.0
            mutant_scaled = mutant * scale
            corr = np.correlate(mutant_scaled, 1.0 - mutant_scaled, mode="full")
            mutant_val = np.max(corr * dx)
            trial = mutant.copy()
            for j in range(n_points):
                if np.random.rand() < crossover:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0.0, 1.0)
            trial_sum = np.sum(trial)
            trial_scale = target_sum / trial_sum if trial_sum != 0 else 1.0
            trial_scaled = trial * trial_scale
            corr_trial = np.correlate(trial_scaled, 1.0 - trial_scaled, mode="full")
            trial_val = np.max(corr_trial * dx)
            if trial_val < de_objective(pop[i]):
                offspring.append(trial)
            else:
                offspring.append(pop[i])
        pop = np.array(offspring)
        pop = np.clip(pop, 0.0, 1.0)
        pop = pop * (target_sum / np.sum(pop, axis=1, keepdims=True))
        pop = np.clip(pop, 0.0, 1.0)
        elite = np.sort([de_objective(x) for x in pop])[:elite_size]
        if np.max(elite) < best_c5:
            best_c5 = np.max(elite)
            best_h = pop[np.argmin(elite)]

        if time.time() - start_time > budget_s - 5:
            break

    # Local refinement with L-BFGS-B
    h_opt = best_h.copy()
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale

    result = opt.minimize(
        objective, h_opt.flatten(), 
        method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 500}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt.flatten())
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 500}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
