```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.cos(np.pi * x))
        h = h * (target_sum / np.sum(h))
    def exact_objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    population_size = 20
    population = [h.copy() for _ in range(population_size)]
    for gen in range(300):
        fitness = np.array([exact_objective(p) for p in population])
        parents = np.argsort(fitness)[:5]
        parents = population[parents]
        new_population = []
        for _ in range(population_size // 2):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = p1.copy()
            crossover_idx = np.random.randint(1, n_points - 1)
            child[crossover_idx:] = p2[crossover_idx:]
            child = np.clip(child, 0.0, 1.0)
            sum_h = np.sum(child)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            child = child * scale
            new_population.append(child)
        for _ in range(population_size // 2):
            h_mut = np.random.choice(population, 1)[0].copy()
            mutation_mask = np.random.rand(n_points) < 0.1
            h_mut[mutation_mask] += np.random.normal(0, 0.05, np.sum(mutation_mask))
            h_mut = np.clip(h_mut, 0.0, 1.0)
            sum_h = np.sum(h_mut)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_mut = h_mut * scale
            new_population.append(h_mut)
        population = new_population
        current_fitness = np.min(fitness)
        if current_fitness < best_c5:
            best_c5 = current_fitness
            best_h = population[np.argmin(fitness)].copy()
        if time.time() - start_time > budget_s - 5:
            break
    for _ in range(10):
        result = opt.minimize(
            exact_objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = exact_objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
    return best_h.tolist(), best_c5, n_points
```
