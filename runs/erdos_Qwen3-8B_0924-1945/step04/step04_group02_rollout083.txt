```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_particles = 40
    generations = 200
    elite_ratio = 0.05
    c1 = 2.0
    c2 = 2.0
    omega_start = 0.9
    omega_end = 0.4
    mutation_start = 0.05
    mutation_end = 0.02
    max_refinement_iters = 50
    tolerance = 1e-8

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0.0, 1.0, n_points)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    def evaluate(solution):
        h = solution.copy()
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    # Initialize swarm
    particles = np.random.uniform(0.0, 1.0, (n_particles, n_points))
    particles = particles * (target_sum / np.sum(particles, axis=1, keepdims=True))
    velocities = np.random.uniform(-0.05, 0.05, (n_particles, n_points))
    pbest_positions = particles.copy()
    pbest_values = np.array([evaluate(p) for p in particles])
    gbest_position = particles[np.argmin(pbest_values)]
    gbest_value = np.min(pbest_values)

    best_c5 = gbest_value
    best_h = gbest_position.copy()
    start_time = time.time()

    for gen in range(generations):
        omega = omega_start - (omega_start - omega_end) * gen / generations
        mutation_amp = mutation_start - (mutation_start - mutation_end) * gen / generations

        for i in range(n_particles):
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = omega * velocities[i] + c1 * r1 * (pbest_positions[i] - particles[i]) + c2 * r2 * (gbest_position - particles[i])

            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_p = np.sum(particles[i])
            particles[i] *= target_sum / sum_p if sum_p != 0 else 1.0

            # Evaluate
            current_value = evaluate(particles[i])
            if current_value < pbest_values[i]:
                pbest_positions[i] = particles[i].copy()
                pbest_values[i] = current_value

        # Update global best
        current_best_index = np.argmin(pbest_values)
        current_best_value = pbest_values[current_best_index]
        if current_best_value < gbest_value:
            gbest_position = pbest_positions[current_best_index].copy()
            gbest_value = current_best_value
            best_h = gbest_position.copy()
            best_c5 = gbest_value

        # Apply mutation to all particles
        for i in range(n_particles):
            mutation = np.random.normal(0, 1, n_points) * mutation_amp
            particles[i] += mutation
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_p = np.sum(particles[i])
            particles[i] *= target_sum / sum_p if sum_p != 0 else 1.0

        # Check for early termination
        if time.time() - start_time > budget_s - 10:
            break

        # Evaluate elites
        elite_indices = np.argsort(pbest_values)[:int(n_particles * elite_ratio)]
        elite_solutions = pbest_positions[elite_indices]
        elite_values = pbest_values[elite_indices]
        if np.min(elite_values) < best_c5:
            best_c5 = np.min(elite_values)
            best_h = elite_solutions[np.argmin(elite_values)]

    # Final refinement with L-BFGS-B
    if time.time() - start_time < budget_s - 20:
        best_h_flat = best_h.flatten()
        result = opt.minimize(
            objective, best_h_flat, 
            method='L-BFGS-B', bounds=[(0.0, 1.0) for _ in range(n_points)], constraints=({'type': 'eq', 'fun': constraint}),
            tol=tolerance, options={'maxiter': max_refinement_iters}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        # Evaluate best solution again
        best_h = np.clip(best_h, 0.0, 1.0)
        sum_h = np.sum(best_h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        best_h = best_h * scale
        best_c5 = evaluate(best_h)

    return best_h.tolist(), best_c5, n_points
```
