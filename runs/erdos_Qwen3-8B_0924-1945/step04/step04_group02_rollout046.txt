import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        m = 4
        l = 1.0 / m
        g = (1.0 - m * l) / 2
        cluster_positions = np.arange(m) * (l + g) + g
        cluster_positions = np.clip(cluster_positions, 0, 2)
        h = np.zeros(n_points)
        for pos in cluster_positions:
            idx = np.searchsorted(np.linspace(0, 2, n_points), pos)
            h[idx - 1:idx + int(round(l / dx))] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # PSO parameters
    n_particles = 40
    n_generations = 200
    inertia_decay = (0.9 - 0.4) / n_generations
    mutation_amp_decay = (0.05 - 0.02) / n_generations
    c1, c2 = 2.0, 2.0
    elite_fraction = 0.05

    # Initialize particles
    particles = np.random.rand(n_particles, n_points)
    particles = particles * (target_sum / np.sum(particles, axis=1, keepdims=True))
    particles = np.clip(particles, 0.0, 1.0)
    velocities = np.random.rand(n_particles, n_points) * 0.01

    pbest_positions = particles.copy()
    pbest_scores = np.array([objective(p) for p in particles])
    gbest_position = particles[np.argmin(pbest_scores)].copy()
    gbest_score = np.min(pbest_scores)

    for gen in range(n_generations):
        # Update velocities and positions
        inertia = 0.9 - gen * inertia_decay
        mutation_amp = 0.05 - gen * mutation_amp_decay
        for i in range(n_particles):
            r1, r2 = np.random.rand(2, n_points)
            velocities[i] = inertia * velocities[i] + c1 * r1 * (pbest_positions[i] - particles[i]) + c2 * r2 * (gbest_position - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))

        # Evaluate and update pbest
        scores = np.array([objective(p) for p in particles])
        for i in range(n_particles):
            if scores[i] < pbest_scores[i]:
                pbest_scores[i] = scores[i]
                pbest_positions[i] = particles[i].copy()

        # Update gbest
        best_idx = np.argmin(pbest_scores)
        if pbest_scores[best_idx] < gbest_score:
            gbest_score = pbest_scores[best_idx]
            gbest_position = pbest_positions[best_idx].copy()

        # Track elites
        elite_idx = np.argsort(pbest_scores)[:int(n_particles * elite_fraction)]
        elites = pbest_positions[elite_idx].copy()

        # Early termination check
        if time.time() - start_time > budget_s - 5:
            break

    # Final refinement with L-BFGS-B
    elite_candidates = elites.copy()
    for i in range(len(elite_candidates)):
        h_opt = elite_candidates[i].copy()
        result = opt.minimize(
            objective, h_opt.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
