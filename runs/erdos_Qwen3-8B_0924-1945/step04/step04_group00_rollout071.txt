```python
import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = (h > 0.5).astype(np.float64)
        h = h * (target_sum / np.sum(h))

    def compute_c5(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = corr * dx
        max_overlap = np.max(overlaps)
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        k = np.argmin(overlaps[1: n_points])
        for _ in range(2):
            i = np.random.randint(0, n_points)
            j = (i + k) % n_points
            if h[i] == 0 and h[j] == 1:
                h[i] = 1
                break
        h = h * (target_sum / np.sum(h))

    if time.time() - start_time < budget_s:
        h = np.clip(best_h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        final_c5 = compute_c5(h)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h.copy()

    return best_h.tolist(), best_c5, n_points
```
