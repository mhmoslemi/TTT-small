```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.full(n_points, 0.5)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        h_flat = h.copy()
        h_flat = h_flat.reshape(-1)
        h_flat = np.clip(h_flat, 0.0, 1.0)
        sum_h = np.sum(h_flat)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_flat = h_flat * scale

        m = n_points
        c = cp.Variable()
        h = cp.Variable(m, bounds=(0.0, 1.0))
        constraints = [cp.sum(h) == target_sum]
        for k in range(1, m):
            corr = cp.sum(h[:(m - k)] * h[k:])
            constraints.append(corr <= c)

        prob = cp.Problem(cp.Minimize(c), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == 'optimal':
                h_opt = h.value
                c_opt = c.value
                h_opt = np.clip(h_opt, 0.0, 1.0)
                sum_h_opt = np.sum(h_opt)
                scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
                h_opt = h_opt * scale
                c5 = objective(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                h = h_opt.copy()
        except:
            pass

        noise = np.random.normal(0, 0.05, size=n_points)
        h = h + noise
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        h_flat = best_h.copy()
        h_flat = h_flat.reshape(-1)
        h_flat = np.clip(h_flat, 0.0, 1.0)
        sum_h = np.sum(h_flat)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_flat = h_flat * scale
        c5 = objective(h_flat)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_flat.copy()

    return best_h.tolist(), best_c5, n_points
```
