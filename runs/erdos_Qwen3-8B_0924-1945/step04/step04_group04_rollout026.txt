```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 600
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.sort(h)
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    def differential_evolution(pop_size=30, max_iter=100):
        population = np.random.rand(pop_size, n_points)
        population = np.clip(population, 0, 1)
        population = population * (target_sum / np.mean(np.sum(population, axis=1)))
        pop_obj = np.array([objective(p) for p in population])
        best_idx = np.argmin(pop_obj)
        best_pop = population[best_idx]
        for _ in range(max_iter):
            for i in range(pop_size):
                a, b, c = np.random.choice(pop_size, 3, replace=False)
                mutant = population[a] + 0.5 * (population[b] - population[c])
                mutant = np.clip(mutant, 0, 1)
                mutant = mutant * (target_sum / np.sum(mutant))
                trial = np.clip(population[i] + 0.9 * (mutant - population[i]), 0, 1)
                trial = trial * (target_sum / np.sum(trial))
                trial_obj = objective(trial)
                if trial_obj < pop_obj[i]:
                    population[i] = trial
                    pop_obj[i] = trial_obj
                    if trial_obj < best_c5:
                        best_c5 = trial_obj
                        best_h = trial.copy()
            if time.time() - start_time > budget_s - 100:
                break
        return best_h

    h = differential_evolution()

    def l_bfgs_refinement(max_iter=500):
        result = opt.minimize(objective, h.flatten(), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': max_iter})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
        return best_h

    h = l_bfgs_refinement()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        h = l_bfgs_refinement(max_iter=100)

    return best_h.tolist(), best_c5, n_points
```
