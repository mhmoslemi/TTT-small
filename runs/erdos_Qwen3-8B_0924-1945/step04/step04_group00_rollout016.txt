```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    population_size = 20
    population = [h.copy() for _ in range(population_size)]
    
    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    
    def repair(h):
        h = h.copy()
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        h = np.clip(h, 0, 1)
        return h
    
    def mutate(h):
        h = h.copy()
        h_noise = np.random.normal(0, 0.1, n_points)
        h += h_noise
        h = repair(h)
        return h
    
    def crossover(parent1, parent2):
        child = parent1.copy()
        indices = np.random.choice(n_points, size=n_points // 2, replace=False)
        child[indices] = parent2[indices]
        child = repair(child)
        return child
    
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        fitness = [evaluate(h) for h in population]
        idxs = np.argsort(fitness)
        top_parents = [population[idxs[0]], population[idxs[1]]]
        new_population = []
        for _ in range(population_size):
            p1, p2 = np.random.choice(top_parents, size=2, replace=False)
            child = crossover(p1, p2)
            if np.random.rand() < 0.75:
                child = mutate(child)
            new_population.append(child)
        population = new_population
    
    for h in population:
        c5 = evaluate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            lambda x: evaluate(x),
            best_h.flatten(),
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        )
        best_h = result.x.reshape(n_points)
        best_h = np.clip(best_h, 0.0, 1.0)
        sum_h = np.sum(best_h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        best_h = best_h * scale
        best_c5 = evaluate(best_h)
    
    return best_h.tolist(), best_c5, n_points
```
