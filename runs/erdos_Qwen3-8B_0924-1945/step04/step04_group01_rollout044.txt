import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = np.array(initial_h_values, dtype=np.float64) if initial_h_values is not None else np.zeros(n_points)
    h = np.random.binomial(1, 0.5, n_points)
    h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h

    def compute_c5(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def objective(h_flat, tau=0.1):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = corr * dx
        return np.max(overlaps)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    iter_count = 0

    while time.time() - start_time < budget_s - 10 and iter_count < 100:
        h_flat = h.copy()
        h_flat = np.clip(h_flat, 0.0, 1.0)
        h_flat = h_flat * (target_sum / np.sum(h_flat)) if np.sum(h_flat) != 0 else h_flat

        # Apply softmax for continuous refinement
        h_cont = np.exp(h_flat / tau) / (np.sum(np.exp(h_flat / tau)) + 1e-12)
        h_cont = np.clip(h_cont, 0.0, 1.0)
        h_cont = h_cont * (target_sum / np.sum(h_cont)) if np.sum(h_cont) != 0 else h_cont
        result = opt.minimize(
            objective, h_cont.flatten(), 
            method='SLSQP', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale

        c5_bound = compute_c5(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

        # Add small random noise for exploration
        noise = np.random.normal(0, 0.05, size=n_points)
        h = h_opt + noise
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        iter_count += 1

    # Final refinement
    result = opt.minimize(
        compute_c5, best_h.flatten(), 
        method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 50}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = compute_c5(h_opt)
    best_h = h_opt.copy()

    return best_h.tolist(), c5_bound, n_points
