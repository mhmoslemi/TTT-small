import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_generations = 200
    n_particles = 40
    elite_fraction = 0.05
    mutation_amp_start = 0.05
    mutation_amp_end = 0.02
    inertia_start = 0.9
    inertia_end = 0.4
    c1 = 2.0
    c2 = 2.0
    elite_count = int(n_particles * elite_fraction)
    mutation_amp = mutation_amp_start
    inertia = inertia_start
    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0.0, 1.0, n_points)
        h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0.0, 1.0)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    particles = np.array([np.random.uniform(0.0, 1.0, n_points) for _ in range(n_particles)])
    particles = particles * (target_sum / np.sum(particles, axis=1, keepdims=True))
    particles = np.clip(particles, 0.0, 1.0)
    pbest = particles.copy()
    pbest_fitness = np.array([objective(p) for p in particles])
    gbest = particles[np.argmin(pbest_fitness)]
    gbest_fitness = np.min(pbest_fitness)
    best_h = gbest.copy()
    best_c5 = gbest_fitness

    start_time = time.time()
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
        velocities = np.random.uniform(-0.1, 0.1, size=(n_particles, n_points))
        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (inertia * velocities[i] + 
                             c1 * r1 * (pbest[i] - particles[i]) + 
                             c2 * r2 * (gbest - particles[i]))
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_h = np.sum(particles[i])
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            particles[i] = particles[i] * scale
            fitness = objective(particles[i])
            if fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fitness
        gbest = particles[np.argmin(pbest_fitness)]
        gbest_fitness = np.min(pbest_fitness)
        if gbest_fitness < best_c5:
            best_c5 = gbest_fitness
            best_h = gbest.copy()
        if mutation_amp > mutation_amp_end:
            mutation_amp -= (mutation_amp_start - mutation_amp_end) / max_generations
        inertia -= (inertia_start - inertia_end) / max_generations

    if best_c5 > 0.38085:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
