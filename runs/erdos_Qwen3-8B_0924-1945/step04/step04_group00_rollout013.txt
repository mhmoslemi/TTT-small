import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_particles = 30
    if initial_h_values is not None:
        initial_h = np.array(initial_h_values, dtype=np.float64)
        initial_h = initial_h * (target_sum / np.sum(initial_h))
    else:
        initial_h = np.random.uniform(0, 1, n_points)
        initial_h = initial_h * (target_sum / np.sum(initial_h))
    particles = np.tile(initial_h[np.newaxis, :], (n_particles, 1))
    velocities = np.random.uniform(-0.1, 0.1, (n_points, n_particles))
    pbest = particles.copy()
    pbest_fitness = np.full(n_particles, np.inf)
    gbest = particles[0].copy()
    gbest_fitness = np.inf
    best_h = initial_h.copy()
    best_c5 = np.inf
    omega = 0.7
    phi_p = 1.5
    phi_g = 1.5
    max_iter = 200
    elapsed_time = 0
    start_time = time.time()
    
    def compute_c5(h):
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    
    for _ in range(max_iter):
        if elapsed_time > budget_s - 10:
            break
        for i in range(n_particles):
            h = particles[i]
            fitness = compute_c5(h)
            if fitness < pbest_fitness[i]:
                pbest[i] = h.copy()
                pbest_fitness[i] = fitness
            if fitness < gbest_fitness:
                gbest = h.copy()
                gbest_fitness = fitness
        for i in range(n_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[:, i] = omega * velocities[:, i] + phi_p * r1 * (pbest[i] - particles[i]) + phi_g * r2 * (gbest - particles[i])
            particles[:, i] = particles[:, i] + velocities[:, i]
            particles[:, i] = np.clip(particles[:, i], 0.0, 1.0)
            particles[:, i] = particles[:, i] * (target_sum / np.sum(particles[:, i]))
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s - 10:
            break
    
    best_h = gbest.copy()
    best_c5 = compute_c5(best_h)
    
    if elapsed_time < budget_s:
        result = opt.minimize(
            compute_c5, best_h, 
            method='L-BFGS-B', bounds=[(0.0, 1.0) for _ in range(n_points)], 
            tol=1e-8, options={'maxiter': 50}
        )
        best_h = result.x
        best_c5 = compute_c5(best_h)
    
    return best_h.tolist(), best_c5, n_points
