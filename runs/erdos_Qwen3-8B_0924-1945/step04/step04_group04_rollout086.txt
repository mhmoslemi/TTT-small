```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 600
    dx = 2.0 / n_points
    target_sum = n_points / 2
    x_grid = np.linspace(0, 2, n_points)

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        idx = np.argsort(h)
        h[idx[:n_points//2]] = 1.0
        h[idx[n_points//2:]] = 0.0
        h = h * (target_sum / np.sum(h))

    def objective_flat(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint_flat(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_flat}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    def de_optimize():
        pop_size = 30
        pop = np.random.rand(pop_size, n_points)
        pop = pop * (target_sum / np.sum(pop, axis=1)[:, None])
        pop = np.clip(pop, 0.0, 1.0)

        for _ in range(100):
            if time.time() - start_time > budget_s - 100:
                break
            mutant = np.zeros((pop_size, n_points))
            for i in range(pop_size):
                idx = np.random.choice(pop_size, 3, replace=False)
                mutant[i] = pop[idx[0]] + 0.5 * (pop[idx[1]] - pop[idx[2]])
                mutant[i] = np.clip(mutant[i], 0.0, 1.0)
                mutant[i] = mutant[i] * (target_sum / np.sum(mutant[i]))
            trial = np.zeros((pop_size, n_points))
            for i in range(pop_size):
                mask = np.random.rand(n_points) < 0.9
                trial[i] = np.where(mask, mutant[i], pop[i])
                trial[i] = np.clip(trial[i], 0.0, 1.0)
                trial[i] = trial[i] * (target_sum / np.sum(trial[i]))
            fitness = np.array([objective_flat(trial[i]) for i in range(pop_size)])
            elite_idx = np.argmin(fitness)
            pop[elite_idx] = trial[elite_idx]
        return pop[np.argmin([objective_flat(p) for p in pop])]

    best_h = de_optimize()
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (target_sum / np.sum(best_h))

    result = opt.minimize(
        objective_flat, best_h.flatten(), 
        method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 500}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective_flat(h_opt)
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    if time.time() - start_time < budget_s - 50:
        noise = np.random.uniform(-0.01, 0.01, size=n_points)
        temp_h = best_h + noise
        temp_h = np.clip(temp_h, 0.0, 1.0)
        temp_h = temp_h * (target_sum / np.sum(temp_h))
        c5_temp = objective_flat(temp_h)
        if c5_temp < best_c5:
            best_c5 = c5_temp
            best_h = temp_h.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective_flat, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 100}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective_flat(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
