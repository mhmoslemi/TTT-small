```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        m = 4
        l = 1.0 / m
        g = (1.0 - m * l) / 2
        cluster_positions = np.arange(m) * (l + g) + g
        cluster_positions = np.clip(cluster_positions, 0, 2)
        h = np.zeros(n_points)
        for pos in cluster_positions:
            idx = np.searchsorted(np.linspace(0, 2, n_points), pos)
            h[idx - 1:idx + int(round(l / dx))] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # PSO parameters
    n_particles = 40
    n_generations = 200
    inertia = 0.9
    inertia_decay = (0.9 - 0.4) / n_generations
    c1 = c2 = 2.0
    mutation_amp = 0.05
    mutation_amp_decay = (0.05 - 0.02) / n_generations
    elite_fraction = 0.05
    n_elite = int(n_particles * elite_fraction)
    elite_index = np.random.choice(n_particles, size=n_elite, replace=False)

    # Initialize swarm
    particles = np.random.rand(n_particles, n_points)
    particles = particles * (target_sum / np.sum(particles, axis=1, keepdims=True))
    velocities = np.random.rand(n_particles, n_points) * 0.1

    # Best positions
    pbest_pos = particles.copy()
    pbest_obj = np.full(n_particles, np.inf)
    gbest_pos = particles[elite_index[0]].copy()
    gbest_obj = np.inf

    for gen in range(n_generations):
        # Evaluate fitness
        for i in range(n_particles):
            h_flat = particles[i]
            obj = objective(h_flat)
            pbest_obj[i] = obj
            if obj < pbest_obj[i]:
                pbest_pos[i] = h_flat.copy()
                pbest_obj[i] = obj

        # Update global best
        for i in range(n_particles):
            if pbest_obj[i] < gbest_obj:
                gbest_obj = pbest_obj[i]
                gbest_pos = pbest_pos[i].copy()

        # Update velocities and positions
        r1 = np.random.rand(n_particles, n_points)
        r2 = np.random.rand(n_particles, n_points)
        velocities = (inertia * velocities +
                      c1 * r1 * (pbest_pos - particles) +
                      c2 * r2 * (gbest_pos - particles))
        inertia -= inertia_decay
        mutation_amp -= mutation_amp_decay

        for i in range(n_particles):
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_h = np.sum(particles[i])
            if sum_h != 0:
                particles[i] *= (target_sum / sum_h)

        # Apply mutation
        for i in range(n_particles):
            for j in range(n_points):
                particles[i][j] += mutation_amp * np.random.normal(0, 1)
                particles[i][j] = np.clip(particles[i][j], 0.0, 1.0)
                sum_h = np.sum(particles[i])
                if sum_h != 0:
                    particles[i][j] *= (target_sum / sum_h)

        # Select elites
        elite_indices = np.argsort(pbest_obj)[:n_elite]
        elite_h = pbest_pos[elite_indices]
        if gen % 5 == 0:
            for h in elite_h:
                res = opt.minimize(
                    objective, h.flatten(), 
                    method='L-BFGS-B', bounds=bounds, constraints=cons,
                    tol=1e-8, options={'maxiter': 50}
                )
                h_opt = res.x.reshape(n_points)
                h_opt = np.clip(h_opt, 0.0, 1.0)
                sum_h_opt = np.sum(h_opt)
                scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
                h_opt = h_opt * scale
                c5_bound = objective(h_opt.flatten())
                if c5_bound < best_c5:
                    best_c5 = c5_bound
                    best_h = h_opt.copy()

        if time.time() - start_time > budget_s - 5:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        res = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = res.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
