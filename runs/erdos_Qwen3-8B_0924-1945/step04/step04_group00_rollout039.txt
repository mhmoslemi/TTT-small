```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(low=0.0, high=1.0, size=n_points)
        h = h * (target_sum / np.sum(h))
    def evaluate_c5(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    n_particles = 30
    particles = np.random.uniform(low=0.0, high=1.0, size=(n_particles, n_points))
    particles = particles * (target_sum / np.sum(particles, axis=1, keepdims=True))
    velocities = np.random.uniform(low=-0.1, high=0.1, size=(n_particles, n_points))
    pbest_positions = particles.copy()
    pbest_scores = np.array([evaluate_c5(p) for p in particles])
    gbest_position = particles[np.argmin(pbest_scores)]
    gbest_score = np.min(pbest_scores)
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = 0.7 * velocities[i] + 1.5 * r1 * (pbest_positions[i] - particles[i]) + 1.5 * r2 * (gbest_position - particles[i])
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_h = np.sum(particles[i])
            if sum_h != 0:
                particles[i] = particles[i] * (target_sum / sum_h)
            score = evaluate_c5(particles[i])
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest_positions[i] = particles[i].copy()
        current_best_idx = np.argmin(pbest_scores)
        current_best_score = pbest_scores[current_best_idx]
        if current_best_score < gbest_score:
            gbest_score = current_best_score
            gbest_position = pbest_positions[current_best_idx].copy()
        if current_best_score < best_c5:
            best_c5 = current_best_score
            best_h = gbest_position.copy()
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            lambda h_flat: evaluate_c5(h_flat),
            best_h.flatten(),
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            tol=1e-8,
            options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = evaluate_c5(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
    return best_h.tolist(), best_c5, n_points
```
