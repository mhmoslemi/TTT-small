```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    population_size = 20
    population = [np.clip(np.random.rand(n_points) * (target_sum / np.sum(np.random.rand(n_points))), 0.0, 1.0) for _ in range(population_size)]
    population = [h * (target_sum / np.sum(h)) for h in population]
    
    best_c5 = float('inf')
    best_h = np.copy(population[0])
    
    start_time = time.time()
    for gen in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        
        fitness = [objective(h_flat) for h_flat in population]
        idxs = np.argsort(fitness)
        parents = [population[i] for i in idxs[:5]]
        
        next_gen = []
        for _ in range(population_size // 2):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = np.copy(p1)
            crossover_point = np.random.randint(1, n_points - 1)
            child[crossover_point:] = p2[crossover_point:]
            next_gen.append(child)
        
        for _ in range(population_size // 2):
            h = np.copy(population[np.random.randint(population_size)])
            mutation_mask = np.random.rand(n_points) < 0.1
            h[mutation_mask] += np.random.normal(0, 0.05, size=np.sum(mutation_mask))
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            next_gen.append(h)
        
        population = next_gen
        for h in population:
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
        
        for h in population:
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
        
        for h in population:
            fitness = objective(h)
            if fitness < best_c5:
                best_c5 = fitness
                best_h = np.copy(h)
    
    if time.time() - start_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt
    
    return best_h.tolist(), best_c5, n_points
```
