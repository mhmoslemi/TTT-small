import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 600
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        angles = np.linspace(0, 2 * np.pi, n_points, endpoint=False)
        h = np.random.rand(n_points)
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            scale = 1.0
        else:
            scale = target_sum / sum_h
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    def de_objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            scale = 1.0
        else:
            scale = target_sum / sum_h
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def de_constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    de_bounds = [(0.0, 1.0) for _ in range(n_points)]
    de_cons = [{'type': 'eq', 'fun': de_constraint}]

    pop_size = 30
    pop = np.random.rand(pop_size, n_points)
    pop = np.clip(pop, 0.0, 1.0)
    pop = pop * (target_sum / np.sum(pop, axis=1, keepdims=True))

    for _ in range(100):
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + 0.5 * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = mutant * (target_sum / np.sum(mutant))
            mutant = np.random.rand(n_points) * mutant + (1 - np.random.rand(n_points)) * pop[i]
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = mutant * (target_sum / np.sum(mutant))
            corr = np.correlate(mutant, 1.0 - mutant, mode="full")
            mutant_val = np.max(corr * dx)
            pop[i] = mutant
            if mutant_val < objective(pop[i].flatten()):
                pop[i] = mutant

        current_vals = [objective(p.flatten()) for p in pop]
        best_idx = np.argmin(current_vals)
        best_h = pop[best_idx].copy()
        best_c5 = current_vals[best_idx]

        if time.time() - start_time > budget_s - 300:
            break

    h_opt = best_h.copy()
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    if sum_h_opt == 0:
        scale = 1.0
    else:
        scale = target_sum / sum_h_opt
    h_opt = h_opt * scale

    result = opt.minimize(
        objective, h_opt.flatten(), 
        method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 500}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    if sum_h_opt == 0:
        scale = 1.0
    else:
        scale = target_sum / sum_h_opt
    h_opt = h_opt * scale
    c5_bound = objective(h_opt.flatten())
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    if time.time() - start_time < budget_s - 50:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 500}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        if sum_h_opt == 0:
            scale = 1.0
        else:
            scale = target_sum / sum_h_opt
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
