import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 600
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        angles = np.linspace(0, 2 * np.pi, n_points, endpoint=False)
        sawtooth = np.abs(np.sin(angles) - 1.0)
        h = np.clip(1.0 - sawtooth, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    population_size = 30
    pop = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    pop = [h * (target_sum / np.sum(h)) for h in pop]

    def de_step(pop, F, CR):
        new_pop = []
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * (target_sum / np.sum(mutant))
            trial = np.zeros(n_points)
            j_rand = np.random.randint(n_points)
            for j in range(n_points):
                if np.random.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = pop[i][j]
            trial = np.clip(trial, 0, 1)
            trial = trial * (target_sum / np.sum(trial))
            if objective(trial.flatten()) < objective(pop[i].flatten()):
                new_pop.append(trial)
            else:
                new_pop.append(pop[i])
        return new_pop

    for _ in range(100):
        pop = de_step(pop, 0.5, 0.9)
        if time.time() - start_time > budget_s - 5:
            break

    h_opt = np.array(pop[0])
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt.flatten())
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    result = opt.minimize(
        objective, best_h.flatten(), 
        method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 500}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt.flatten())
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
