```python
import numpy as np
import time
import scipy.optimize as opt

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    best_c5 = float('inf')
    best_h = None

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        scale = target_sum / sum_h
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    def initialize_population(n_pop):
        pop = []
        for _ in range(n_pop):
            h = np.zeros(n_points)
            indices = np.random.choice(n_points, size=int(n_points/2), replace=False)
            h[indices] = 1
            sum_h = np.sum(h)
            if sum_h != 0:
                h *= target_sum / sum_h
            pop.append(h)
        return np.array(pop)

    if initial_h_values is not None:
        h_population = np.array([np.array(initial_h_values, dtype=np.float64)])
    else:
        h_population = initialize_population(30)

    start_time = time.time()
    last_improvement = 0
    max_generations = 600 * 10
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
        fitness = np.array([objective(h) for h in h_population])
        best_idx = np.argmin(fitness)
        current_c5 = fitness[best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_population[best_idx].copy()
            last_improvement = 0
        else:
            last_improvement += 1
        if last_improvement >= 50:
            break
        top_indices = np.argsort(fitness)[:int(len(h_population)*0.2)]
        top_h = h_population[top_indices]
        new_population = []
        for _ in range(len(h_population) // 2):
            a, b, c = np.random.choice(top_indices, 3, replace=False)
            h_a, h_b, h_c = top_h[a], top_h[b], top_h[c]
            mutant = h_a + 0.5 * (h_b - h_c)
            mutant = np.clip(mutant, 0.0, 1.0)
            if np.sum(mutant) != 0:
                mutant *= target_sum / np.sum(mutant)
            new_population.append(mutant)
        h_population = np.vstack([h_population, new_population])
        h_population = np.random.choice(h_population, size=len(h_population), replace=False)
        h_population = h_population[:len(h_population)//2]

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
