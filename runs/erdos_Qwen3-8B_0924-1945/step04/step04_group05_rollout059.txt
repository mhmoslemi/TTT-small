```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 + 0.5 * np.sin(np.pi * i / (n_points / 2))
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    for _ in range(50):
        for init in [h.copy(), np.zeros(n_points), np.ones(n_points)]:
            init = init * (target_sum / np.sum(init))
            init = np.clip(init, 0.0, 1.0)
            h_flat = init.copy()
            h_flat = h_flat.reshape(n_points)

            x = cp.Variable(n_points, bounds=(0.0, 1.0))
            C = cp.Variable()

            constraints = []
            constraints.append(cp.sum(x) == target_sum)
            for k in range(1, n_points):
                constraints.append(cp.sum(x[:n_points - k] * x[k:]) <= C)

            prob = cp.Problem(cp.Minimize(C), constraints)
            try:
                prob.solve(solver=cp.ECOS)
                if prob.status == 'OPTIMAL':
                    h_opt = x.value
                    h_opt = np.clip(h_opt, 0.0, 1.0)
                    h_opt = h_opt * (target_sum / np.sum(h_opt))
                    c5 = evaluate(h_opt)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_opt.copy()
            except:
                pass

        noise = np.random.normal(0, 0.05, size=n_points)
        h = best_h + noise
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))

        if time.time() - start_time > budget_s - 5:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            lambda h_flat: evaluate(h_flat), best_h.flatten(), 
            method='L-BFGS-B', bounds=[(0.0, 1.0)] * n_points,
            constraints={'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat) - target_sum}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        h_opt = h_opt * (target_sum / np.sum(h_opt))
        c5_bound = evaluate(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
