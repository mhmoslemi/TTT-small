```python
import numpy as np
import scipy.optimize as opt
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, WhiteKernel

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    kernel = RBF(length_scale=1.0, length_scale_bounds=(1e-5, 1000.0)) + WhiteKernel(noise_level=1e-5)
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-5, n_restarts_optimizer=25, normalize_y=True)

    samples = []
    values = []

    for _ in range(500):
        if len(samples) == 0:
            h_candidate = h.copy()
        else:
            if len(samples) < 20:
                h_candidate = np.random.uniform(0, 1, n_points)
                h_candidate = h_candidate * (target_sum / np.sum(h_candidate))
            else:
                mean, std = gp.predict(samples, return_std=True)
                sigma = std * 1.5
                h_candidate = samples[np.argmax(mean - 1.5 * sigma)]
                h_candidate = h_candidate + np.random.normal(0, 0.05, n_points)
                h_candidate = np.clip(h_candidate, 0.0, 1.0)
                h_candidate = h_candidate * (target_sum / np.sum(h_candidate))

        h_candidate = h_candidate.copy()
        h_candidate = np.clip(h_candidate, 0.0, 1.0)
        sum_h_candidate = np.sum(h_candidate)
        scale = target_sum / sum_h_candidate if sum_h_candidate != 0 else 1.0
        h_scaled = h_candidate * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        c5 = np.max(corr * dx)
        samples.append(h_candidate)
        values.append(c5)

        if len(samples) > 100:
            samples.pop(0)
            values.pop(0)

        gp.fit(samples, values)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()

        if time.time() - start_time > budget_s - 5:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
