import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h = (h > 0.5).astype(np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        h[:int(n_points/2)] = 1.0
    h = h * (target_sum / np.sum(h))
    max_lag = int(np.ceil(2 * n_points))
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        corr = np.zeros(max_lag, dtype=np.float64)
        for k in range(1, max_lag):
            idx = np.arange(n_points)
            idx_k = np.mod(idx + k, n_points)
            corr[k] = np.sum(h[idx] * h[idx_k]) * dx
        min_corr = np.min(corr)
        if min_corr < 1e-10:
            min_corr = 1e-10
        if min_corr < best_c5:
            best_c5 = min_corr
            best_h = h.copy()
        k = np.argmin(corr)
        idx = np.arange(n_points)
        idx_k = np.mod(idx + k, n_points)
        same = np.where(np.logical_and(h[idx] == 0, h[idx_k] == 1))[0]
        if len(same) > 0:
            i = np.random.choice(same)
            h[i] = 1.0
        h = h * (target_sum / np.sum(h))
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        h = best_h.copy()
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
            corr = np.zeros(max_lag, dtype=np.float64)
            for k in range(1, max_lag):
                idx = np.arange(n_points)
                idx_k = np.mod(idx + k, n_points)
                corr[k] = np.sum(h[idx] * h[idx_k]) * dx
            min_corr = np.min(corr)
            if min_corr < 1e-10:
                min_corr = 1e-10
            if min_corr < best_c5:
                best_c5 = min_corr
                best_h = h.copy()
            k = np.argmin(corr)
            idx = np.arange(n_points)
            idx_k = np.mod(idx + k, n_points)
            same = np.where(np.logical_and(h[idx] == 0, h[idx_k] == 1))[0]
            if len(same) > 0:
                i = np.random.choice(same)
                h[i] = 1.0
            h = h * (target_sum / np.sum(h))
    return best_h.tolist(), best_c5, n_points
