import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        m = 4
        l = 1.0 / m
        g = (1.0 - m * l) / 2
        cluster_positions = np.arange(m) * (l + g) + g
        cluster_positions = np.clip(cluster_positions, 0, 2)
        h = np.zeros(n_points)
        for pos in cluster_positions:
            idx = np.searchsorted(np.linspace(0, 2, n_points), pos)
            h[idx - 1:idx + int(round(l / dx))] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    for _ in range(200):
        particles = np.random.rand(20, n_points)
        particles = (particles - 0.5) * 0.4 + h.copy()
        particles = np.clip(particles, 0.0, 1.0)
        for i in range(particles.shape[0]):
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
        personal_best = particles.copy()
        global_best = particles[0].copy()
        for i in range(particles.shape[0]):
            pos = particles[i]
            r1, r2 = np.random.rand(2)
            vel = np.random.rand(n_points) * 0.5
            cog = np.random.rand(n_points) * 0.5
            soc = np.random.rand(n_points) * 0.5
            vel = vel * 0.9 + cog * personal_best[i] + soc * global_best
            pos = pos + vel
            pos = np.clip(pos, 0.0, 1.0)
            pos = pos * (target_sum / np.sum(pos))
            obj = objective(pos.flatten())
            if obj < objective(personal_best[i].flatten()):
                personal_best[i] = pos.copy()
            if obj < objective(global_best.flatten()):
                global_best = pos.copy()
        h = global_best.copy()
        c5_bound = objective(h.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h.copy()
        if time.time() - start_time > budget_s - 5:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
