import numpy as np
import scipy.optimize as opt
import time
import GPy

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    if initial_h_values is not None:
        samples = [h]
    else:
        samples = [np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points)) for _ in range(20)]

    X = np.array([s for s in samples])
    Y = np.array([objective(s.flatten()) for s in samples])

    kernel = GPy.kern.RBF(1) + GPy.kern.WhiteNoise(1e-5)
    m = GPy.models.GPR((X, Y), kernel=kernel, normalised=True)

    for _ in range(500):
        if time.time() - start_time > budget_s - 100:
            break
        mu, var = m.predict(X)
        std = np.sqrt(var)
        ei = (Y - np.min(Y)) * np.exp(-0.5 * (mu - np.min(Y))**2 / var) * np.sqrt(2 * np.pi * var)
        ei = (ei - np.min(ei)) / (np.max(ei) - np.min(ei)) + 1e-6
        next_idx = np.argmax(ei)
        next_x = X[next_idx]

        h_new = np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))
        h_new = h_new * (target_sum / np.sum(h_new))
        h_new = h_new.reshape(n_points)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = h_new * (target_sum / np.sum(h_new))

        h_new_flat = h_new.flatten()
        result = opt.minimize(
            objective, h_new_flat, 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
        X = np.vstack([X, h_opt])
        Y = np.append(Y, c5_bound)
        m = GPy.models.GPR((X, Y), kernel=kernel, normalised=True)

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
