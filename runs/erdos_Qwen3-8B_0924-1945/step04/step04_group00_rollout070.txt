import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.round(h).astype(int)
    else:
        h = np.random.randint(0, 2, n_points)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.round(h).astype(int)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def update_h(h_flat):
        h = h_flat.reshape(n_points)
        h = np.round(h).astype(int)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1 - h, mode="full")
        min_corr_idx = np.argmin(corr[1:])
        lag = min_corr_idx + 1
        pos = np.where((h == 0) & ((h[np.roll(range(n_points), lag)] == 1)))
        if len(pos[0]) > 0:
            i = np.random.choice(pos[0])
            h[i] = 1
        h = np.round(h).astype(int)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        return h.flatten()

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        h = update_h(h)
        c5_bound = objective(h)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h.copy()

    return best_h.tolist(), best_c5, n_points
