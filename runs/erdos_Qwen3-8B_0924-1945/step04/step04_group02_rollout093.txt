import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        m = 4
        l = 1.0 / m
        g = (1.0 - m * l) / 2
        cluster_positions = np.arange(m) * (l + g) + g
        cluster_positions = np.clip(cluster_positions, 0, 2)
        h = np.zeros(n_points)
        for pos in cluster_positions:
            idx = np.searchsorted(np.linspace(0, 2, n_points), pos)
            h[idx - 1:idx + int(round(l / dx))] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    particles = [h.copy() for _ in range(20)]
    particle_fitness = [objective(p) for p in particles]
    best_idx = np.argmin(particle_fitness)
    best_h = particles[best_idx].copy()
    best_c5 = particle_fitness[best_idx]

    n_iter = 0
    max_iter = 200
    while n_iter < max_iter and time.time() - start_time < budget_s - 5:
        for i in range(len(particles)):
            h = particles[i].copy()
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_scaled = h * scale
            corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
            fitness = np.max(corr * dx)

            if fitness < particle_fitness[i]:
                particle_fitness[i] = fitness
                if fitness < best_c5:
                    best_c5 = fitness
                    best_h = h.copy()

            rand1 = np.random.rand(n_points)
            rand2 = np.random.rand(n_points)
            h_new = h + (np.random.rand(n_points) - 0.5) * 0.05
            h_new = np.clip(h_new, 0.0, 1.0)
            sum_h_new = np.sum(h_new)
            scale_new = target_sum / sum_h_new if sum_h_new != 0 else 1.0
            h_new_scaled = h_new * scale_new
            corr_new = np.correlate(h_new_scaled, 1.0 - h_new_scaled, mode="full")
            fitness_new = np.max(corr_new * dx)

            if fitness_new < fitness:
                particles[i] = h_new.copy()
                particle_fitness[i] = fitness_new

        n_iter += 1

    if time.time() - start_time < budget_s - 5:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
