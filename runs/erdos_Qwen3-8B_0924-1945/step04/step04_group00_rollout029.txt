```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.cos(np.pi * x))
        h = h * (target_sum / np.sum(h))
    def exact_objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    population = [h.copy() for _ in range(20)]
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        fitness = [exact_objective(ind) for ind in population]
        parents = np.argsort(fitness)[:5]
        parent1, parent2 = population[parents[0]], population[parents[1]]
        child = np.copy(parent1)
        if np.random.rand() < 0.8:
            split = np.random.randint(1, n_points - 1)
            child[:split] = parent2[:split]
        if np.random.rand() < 0.75:
            indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
            noise = np.random.normal(0, 0.1, size=n_points)
            child[indices] += noise[indices]
            child = np.clip(child, 0.0, 1.0)
            sum_h = np.sum(child)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            child = child * scale
        population.append(child)
        population.pop(np.argmax(fitness))
        if fitness[np.argmax(fitness)] < best_c5:
            best_c5 = fitness[np.argmax(fitness)]
            best_h = population[np.argmax(fitness)].copy()
        if _ % 10 == 0:
            if time.time() - start_time > budget_s - 5:
                break
            for ind in population:
                indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
                noise = np.random.normal(0, 0.1, size=n_points)
                ind[indices] += noise[indices]
                ind = np.clip(ind, 0.0, 1.0)
                sum_h = np.sum(ind)
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                ind = ind * scale
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(10):
            if time.time() - start_time > budget_s - 5:
                break
            result = opt.minimize(
                exact_objective, best_h.flatten(), 
                method='L-BFGS-B', bounds=bounds, constraints=cons,
                tol=1e-8, options={'maxiter': 50}
            )
            h_opt = result.x.reshape(n_points)
            h_opt = np.clip(h_opt, 0.0, 1.0)
            sum_h_opt = np.sum(h_opt)
            scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
            h_opt = h_opt * scale
            c5_bound = exact_objective(h_opt.flatten())
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h = h_opt.copy()
    return best_h.tolist(), best_c5, n_points
```
