```python
import numpy as np
import scipy.optimize as opt
import time
import GPy

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iterations = 500
    refinement_steps = 50

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    X_initial = np.array([h.copy() for _ in range(20)])
    Y_initial = np.array([objective(x) for x in X_initial])

    kernel = GPy.kern.RBF(input_dim=n_points, lengthscale=1.0) + GPy.kern.WhiteNoise(1e-5)
    model = GPy.models.GPR((X_initial, Y_initial.T), kernel=kernel, noise_var=1e-5)

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
        mean, var = model.predict(X_initial)
        EI = (mean - best_c5) * np.exp(-0.5 * var / (mean - best_c5)**2)
        next_index = np.argmax(EI)
        next_x = X_initial[next_index].copy()
        np.random.uniform(-0.1, 0.1, size=n_points, out=next_x)
        next_x = np.clip(next_x, 0.0, 1.0)
        sum_next = np.sum(next_x)
        if sum_next != 0:
            next_x *= target_sum / sum_next
        next_y = objective(next_x)
        new_X = np.vstack((X_initial, next_x))
        new_Y = np.vstack((Y_initial, [next_y]))
        model.set_XY((new_X, new_Y.T))
        X_initial = new_X
        Y_initial = new_Y
        if next_y < best_c5:
            best_c5 = next_y
            best_h = next_x.copy()

    if time.time() - start_time < budget_s - 100:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-6, options={'maxiter': refinement_steps}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(50):
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            h_candidate = best_h + noise
            h_candidate = np.clip(h_candidate, 0.0, 1.0)
            sum_h = np.sum(h_candidate)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()

    return best_h.tolist(), best_c5, n_points
```
