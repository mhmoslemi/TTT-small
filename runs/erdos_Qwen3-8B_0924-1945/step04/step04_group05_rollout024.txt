```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        max_idx = n_points // 2
        h = np.zeros(n_points)
        h[0:max_idx] = 1
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    def create_variant():
        if np.random.rand() < 0.3:
            return np.random.rand(n_points) * 2 - 1
        elif np.random.rand() < 0.3:
            return np.zeros(n_points)
        else:
            h = np.zeros(n_points)
            pos = np.linspace(0, 2, n_points)
            indices = np.argsort(np.abs(pos - np.random.uniform(0, 2)))
            h[indices[:n_points//2]] = 1
            h = np.clip(h, 0.0, 1.0)
            h = h * (target_sum / np.sum(h))
            return h

    for _ in range(10):
        h = create_variant()
        for _ in range(3):
            result = opt.minimize(
                objective, h.flatten(), 
                method='L-BFGS-B', bounds=bounds, constraints=cons,
                tol=1e-6, options={'maxiter': 100}
            )
            h_opt = result.x.reshape(n_points)
            h_opt = np.clip(h_opt, 0.0, 1.0)
            sum_h_opt = np.sum(h_opt)
            scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
            h_opt = h_opt * scale
            c5_bound = objective(h_opt.flatten())
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h = h_opt.copy()
            if time.time() - start_time > budget_s - 5:
                break
            h = h_opt.copy()

    if time.time() - start_time < budget_s - 5:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 100}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
