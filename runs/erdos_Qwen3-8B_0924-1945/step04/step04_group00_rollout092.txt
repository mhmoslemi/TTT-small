```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
    h = np.clip(h, 0.0, 1.0)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    for gen in range(300):
        if time.time() - start_time > budget_s - 5:
            break
        population = []
        for _ in range(20):
            if np.random.rand() < 0.5:
                h_new = h.copy()
            else:
                h_new = np.random.rand(n_points)
                h_new = h_new / np.sum(h_new) * target_sum
                h_new = np.clip(h_new, 0.0, 1.0)
            population.append(h_new)
        population = [np.clip(p, 0.0, 1.0) for p in population]
        population = [p * (target_sum / np.sum(p)) for p in population]

        fitness = [objective(p) for p in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]

        if gen % 10 == 0:
            for i in range(len(population)):
                if np.random.rand() < 0.3:
                    idx = np.random.choice(n_points, 5, replace=False)
                    population[i][idx] = np.random.rand(5)
                    population[i] = np.clip(population[i], 0.0, 1.0)
                    population[i] = population[i] * (target_sum / np.sum(population[i]))
                if np.random.rand() < 0.2:
                    scale = np.random.uniform(0.9, 1.1)
                    population[i] = population[i] * scale
                    population[i] = np.clip(population[i], 0.0, 1.0)
                    population[i] = population[i] * (target_sum / np.sum(population[i]))

        h = best_h.copy()

    elapsed = time.time() - start_time
    if elapsed < budget_s - 5:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
