```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = np.array(initial_h_values, dtype=np.float64) if initial_h_values is not None else np.random.rand(n_points)
    h = np.round(h).astype(float)
    h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h

    def obj(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def constr(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    constraints = [{'type': 'eq', 'fun': constr}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = obj(best_h.flatten())
    start_time = time.time()

    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        result = opt.minimize(obj, h.flatten(), bounds=bounds, constraints=constraints, method='L-BFGS-B', tol=1e-8, options={'maxiter': 50})
        h = result.x.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        c5 = obj(h.flatten())
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if time.time() - start_time > budget_s - 5:
            break
        if _ % 5 == 0:
            noise = np.random.normal(0, 0.05, size=n_points)
            h = h + noise
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
    return best_h.tolist(), best_c5, n_points
```
