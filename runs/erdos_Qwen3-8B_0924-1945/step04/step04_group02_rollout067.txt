import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points) * 2
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 0, target_sum)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 0, 1)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    n_particles = 20
    particles = np.random.rand(n_particles, n_points) * 2
    particles = particles / np.sum(particles, axis=1, keepdims=True) * target_sum
    particles = np.clip(particles, 0, 1)
    particles = particles / np.sum(particles, axis=1, keepdims=True) * target_sum
    particles = np.clip(particles, 0, 1)

    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(n_particles):
            h_flat = particles[i]
            h = h_flat.reshape(n_points)
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_scaled = h * scale
            corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
            c5 = np.max(corr * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_scaled.copy()
            particles[i] = h_flat
            particles[i] += np.random.uniform(-0.025, 0.025, size=n_points)
            particles[i] = np.clip(particles[i], 0, 1)
            sum_h = np.sum(particles[i])
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            particles[i] = particles[i] * scale
            particles[i] = np.clip(particles[i], 0, 1)
        if np.random.rand() < 0.1:
            rand_idx = np.random.choice(n_points, size=10, replace=False)
            particles[:, rand_idx] += np.random.uniform(-0.05, 0.05, size=(n_particles, 10))
            particles = np.clip(particles, 0, 1)
            particles = particles / np.sum(particles, axis=1, keepdims=True) * target_sum
            particles = np.clip(particles, 0, 1)

    result = opt.minimize(
        objective, best_h.flatten(), 
        method='L-BFGS-B', bounds=[(0.0, 1.0) for _ in range(n_points)], 
        tol=1e-8, options={'maxiter': 50}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt.flatten())
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=[(0.0, 1.0) for _ in range(n_points)], 
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
