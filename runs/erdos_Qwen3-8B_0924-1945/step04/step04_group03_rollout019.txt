import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h /= np.sum(h)
        h *= target_sum

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    n_population = 30
    F = 0.5
    n_gen = 2000
    no_improvement = 0

    for gen in range(n_gen):
        if time.time() - start_time > budget_s - 5:
            break
        pop = [np.clip(h + np.random.rand(n_points) * 0.1, 0.0, 1.0) for _ in range(n_population)]
        for i in range(n_population):
            pop[i] = pop[i] * (target_sum / np.sum(pop[i])) if np.sum(pop[i]) != 0 else pop[i]
        fitness = [objective(p) for p in pop]
        sorted_indices = np.argsort(fitness)
        best_pop = [pop[i] for i in sorted_indices[:int(n_population * 0.2)]]
        new_pop = []
        while len(new_pop) < n_population:
            a, b, c = np.random.choice(best_pop, 3, replace=False)
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = mutant * (target_sum / np.sum(mutant)) if np.sum(mutant) != 0 else mutant
            new_pop.append(mutant)
        pop = best_pop + new_pop
        pop = pop[:n_population]
        fitness = [objective(p) for p in pop]
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = pop[best_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        if no_improvement >= 50:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
            a, b, c = np.random.choice(best_pop, 3, replace=False)
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = mutant * (target_sum / np.sum(mutant)) if np.sum(mutant) != 0 else mutant
            fitness = objective(mutant)
            if fitness < best_c5:
                best_c5 = fitness
                best_h = mutant.copy()

    return best_h.tolist(), best_c5, n_points
