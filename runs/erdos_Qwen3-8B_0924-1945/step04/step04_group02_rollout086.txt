```python
import numpy as np
import scipy.optimize as opt
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel, WhiteKernel

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = np.zeros(n_points)
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        m = 4
        l = 1.0 / m
        g = (1.0 - m * l) / 2
        cluster_positions = np.arange(m) * (l + g) + g
        cluster_positions = np.clip(cluster_positions, 0, 2)
        h = np.zeros(n_points)
        for pos in cluster_positions:
            idx = np.searchsorted(np.linspace(0, 2, n_points), pos)
            h[idx - 1:idx + int(round(l / dx))] = 1
        h = h * (target_sum / np.sum(h))
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)
    
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    
    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)
    
    def acquisition(x, gp, y_opt):
        mu, sigma = gp.predict(x, return_std=True)
        mu = np.where(sigma > 0, mu, np.inf)
        improvement = mu - y_opt
        return np.where(improvement > 0, improvement * np.exp(-sigma ** 2 / 2), 0)
    
    def generate_initial_samples():
        samples = []
        for _ in range(20):
            h_sample = np.random.rand(n_points)
            sum_h = np.sum(h_sample)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_sample = h_sample * scale
            samples.append(h_sample)
        return np.array(samples)
    
    initial_samples = generate_initial_samples()
    initial_scores = np.array([evaluate(sample) for sample in initial_samples])
    X = initial_samples
    y = initial_scores
    
    kernel = ConstantKernel(1.0, (1e-3, 1e3)) * RBF(1.0, (1e-3, 1e3)) + WhiteKernel(1e-5, (1e-10, 1e-3))
    gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=10, alpha=1e-5, normalize_y=True)
    gp.fit(X, y)
    
    best_h = h.copy()
    best_c5 = float('inf')
    X_train = X
    y_train = y
    start_time = time.time()
    max_iterations = 500
    iterations = 0
    while iterations < max_iterations and time.time() - start_time < budget_s - 100:
        X_eval = np.random.rand(50, n_points)
        X_eval = np.clip(X_eval, 0.0, 1.0)
        X_eval = X_eval * (target_sum / np.sum(X_eval, axis=1, keepdims=True)) if np.sum(X_eval, axis=1) != 0 else X_eval
        scores = np.array([evaluate(x) for x in X_eval])
        acquisition_values = acquisition(X_eval, gp, np.min(y_train))
        best_idx = np.argmax(acquisition_values)
        best_candidate = X_eval[best_idx]
        best_score = scores[best_idx]
        if best_score < best_c5:
            best_c5 = best_score
            best_h = best_candidate.copy()
        X_train = np.vstack((X_train, best_candidate))
        y_train = np.append(y_train, best_score)
        gp.fit(X_train, y_train)
        iterations += 1
    
    X_final = np.vstack((X_train, np.random.rand(50, n_points)))
    X_final = np.clip(X_final, 0.0, 1.0)
    X_final = X_final * (target_sum / np.sum(X_final, axis=1, keepdims=True)) if np.sum(X_final, axis=1) != 0 else X_final
    scores_final = np.array([evaluate(x) for x in X_final])
    best_final_idx = np.argmin(scores_final)
    best_h = X_final[best_final_idx].copy()
    best_c5 = scores_final[best_final_idx]
    
    result = opt.minimize(
        objective, best_h, 
        method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 50}
    )
    h_opt = result.x
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = evaluate(h_opt)
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()
    
    return best_h.tolist(), best_c5, n_points
```
