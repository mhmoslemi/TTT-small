```python
import numpy as np
import scipy.optimize as opt
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, WhiteKernel

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 500
    n_initial = 20
    kernel = RBF(1.0) + WhiteKernel(1e-5)
    acq_func = 'ei'
    refinement_steps = 50
    tolerance = 1e-6

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 0, target_sum)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    X = np.zeros((n_initial, n_points))
    y = np.zeros(n_initial)
    for i in range(n_initial):
        if initial_h_values is None:
            h_tmp = np.random.rand(n_points)
            h_tmp = h_tmp / np.sum(h_tmp) * target_sum
            h_tmp = np.clip(h_tmp, 0, target_sum)
        else:
            h_tmp = np.array(initial_h_values, dtype=np.float64)
            h_tmp = h_tmp / np.sum(h_tmp) * target_sum
            h_tmp = np.clip(h_tmp, 0, target_sum)
        X[i] = h_tmp
        y[i] = objective(h_tmp)

    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-5)
    gp.fit(X, y)

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
        if acq_func == 'ei':
            mean, std = gp.predict(X, return_std=True)
            ei = (mean - best_c5) * std * np.sqrt(2 * np.pi)
            ei[std < 1e-5] = -1
            next_idx = np.argmax(ei)
        else:
            next_idx = np.random.randint(0, n_initial)
        next_h = X[next_idx].copy()
        next_h += np.random.randn(n_points) * 0.05
        next_h = np.clip(next_h, 0, 1)
        next_h = next_h * (target_sum / np.sum(next_h) if np.sum(next_h) != 0 else 1.0)
        next_c5 = objective(next_h)
        X = np.vstack((X, next_h))
        y = np.append(y, next_c5)
        gp.fit(X, y)
        if next_c5 < best_c5:
            best_c5 = next_c5
            best_h = next_h.copy()
        if np.random.rand() < 0.1:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            best_h += noise
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (target_sum / np.sum(best_h) if np.sum(best_h) != 0 else 1.0)
            new_c5 = objective(best_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = best_h.copy()
        if i % 10 == 0:
            print(f"Iteration {i} | Best C5: {best_c5:.6f}")

    result = opt.minimize(
        objective, best_h, 
        method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=tolerance, options={'maxiter': refinement_steps}
    )
    h_opt = result.x
    h_opt = np.clip(h_opt, 0, 1)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt)
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
