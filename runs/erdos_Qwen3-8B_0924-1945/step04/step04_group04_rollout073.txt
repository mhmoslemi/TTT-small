```python
import numpy as np
import scipy.optimize as opt
import time
import numpy.random as npr

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    npr.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            scale = 1.0
        else:
            scale = target_sum / sum_h
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)
    
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    elapsed = 0
    
    for _ in range(100):
        direction = np.random.uniform(-1, 1, size=n_points)
        direction = direction / np.sum(np.abs(direction))
        h_update = h + 0.2 * direction
        h_update = np.clip(h_update, 0.0, 1.0)
        sum_h_update = np.sum(h_update)
        if sum_h_update == 0:
            scale = 1.0
        else:
            scale = target_sum / sum_h_update
        h_update = h_update * scale
        c5 = objective(h_update)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_update.copy()
        h = h_update.copy()
        elapsed = time.time() - start_time
        if elapsed > budget_s - 5:
            break
    
    if elapsed < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 400}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        if sum_h_opt == 0:
            scale = 1.0
        else:
            scale = target_sum / sum_h_opt
        h_opt = h_opt * scale
        c5_bound = objective(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
    
    return best_h.tolist(), best_c5, n_points
```
