import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    particles = np.random.uniform(0.0, 1.0, (20, n_points))
    particles = particles * (target_sum / np.sum(particles, axis=1, keepdims=True))
    personal_best = particles.copy()
    global_best = particles[0].copy()

    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(20):
            h_opt = particles[i].copy()
            h_opt = np.clip(h_opt, 0.0, 1.0)
            sum_h_opt = np.sum(h_opt)
            scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
            h_opt = h_opt * scale
            c5 = objective(h_opt)
            if c5 < objective(personal_best[i]):
                personal_best[i] = h_opt.copy()
            if c5 < objective(global_best):
                global_best = h_opt.copy()
        for i in range(20):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            cognitive = personal_best[i] + np.random.rand() * (global_best - personal_best[i])
            social = personal_best[i] + np.random.rand() * (global_best - personal_best[i])
            particles[i] = cognitive * r1 + social * r2
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_p = np.sum(particles[i])
            particles[i] = particles[i] * (target_sum / sum_p if sum_p != 0 else 1.0)
        if np.random.rand() < 0.1:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            particles += noise
            particles = np.clip(particles, 0.0, 1.0)
            for i in range(20):
                sum_p = np.sum(particles[i])
                particles[i] = particles[i] * (target_sum / sum_p if sum_p != 0 else 1.0)

    h_opt = global_best.copy()
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt)

    result = opt.minimize(
        objective, h_opt.flatten(), 
        method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 50}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt.flatten())

    return h_opt.tolist(), c5_bound, n_points
