import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for _ in range(200):
            idx = np.random.randint(0, n_points)
            h[idx] = 1
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def normalize(h):
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        return h * scale

    best_h = h.copy()
    best_c5 = evaluate(h)
    start_time = time.time()

    pop_size = 30
    pop = [np.random.choice([0, 1], size=n_points, p=[0.5, 0.5]) for _ in range(pop_size)]
    pop = [normalize(p) for p in pop]
    pop = [p * (target_sum / np.sum(p)) for p in pop]

    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        fitness = [evaluate(p) for p in pop]
        best_idx = np.argmin(fitness)
        best_fitness = fitness[best_idx]
        if best_fitness < best_c5:
            best_c5 = best_fitness
            best_h = pop[best_idx].copy()

        new_pop = []
        for _ in range(pop_size // 2):
            a, b, c = np.random.choice(pop, 3, replace=False)
            mutant = a + 0.5 * (b - c)
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = normalize(mutant)
            new_pop.append(mutant)
        pop = new_pop + pop[:pop_size // 2]

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = optimize_with_l_bfgs(best_h, n_points, target_sum, dx)
        if result is not None:
            h_opt, c5 = result
            if c5 < best_c5:
                best_h = h_opt
                best_c5 = c5

    return best_h.tolist(), best_c5, n_points

        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

        h = h_flat.reshape(n_points)
