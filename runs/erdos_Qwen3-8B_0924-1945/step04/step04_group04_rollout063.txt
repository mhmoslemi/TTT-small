import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 600
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        angles = np.linspace(0, 2 * np.pi, n_points, endpoint=False)
        sawtooth = np.abs(np.sin(angles) - 1.0)
        h = np.clip(1.0 - sawtooth, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # Stage 1: Differential Evolution for global search
    def de_objective(x):
        x_clipped = np.clip(x, 0.0, 1.0)
        sum_x = np.sum(x_clipped)
        scale = target_sum / sum_x if sum_x != 0 else 1.0
        x_scaled = x_clipped * scale
        corr = np.correlate(x_scaled, 1.0 - x_scaled, mode="full")
        return np.max(corr * dx)

    from scipy.optimize import differential_evolution
    result_de = differential_evolution(
        de_objective, bounds, strategy='best1bin', popsize=30,
        mutation=(0.5, 1.0), recombination=0.9, tol=1e-8,
        maxiter=100, disp=False, polish=True
    )
    h_de = result_de.x

    # Stage 2: Local refinement with L-BFGS-B
    def lbfgs_objective(x_flat):
        x = x_flat.reshape(n_points)
        x = np.clip(x, 0.0, 1.0)
        sum_x = np.sum(x)
        scale = target_sum / sum_x if sum_x != 0 else 1.0
        x_scaled = x * scale
        corr = np.correlate(x_scaled, 1.0 - x_scaled, mode="full")
        return -np.max(corr * dx)  # Negative for minimization

    result_lbfgs = opt.minimize(
        lbfgs_objective, h_de, method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 500}
    )
    h_opt = result_lbfgs.x
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt)

    # Final evaluation
    best_h = h_opt
    best_c5 = c5_bound

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 50:
        result_final = opt.minimize(
            lbfgs_objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 100}
        )
        h_final = result_final.x
        h_final = np.clip(h_final, 0.0, 1.0)
        sum_h_final = np.sum(h_final)
        scale = target_sum / sum_h_final if sum_h_final != 0 else 1.0
        h_final = h_final * scale
        c5_bound = objective(h_final)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_final

    return best_h.tolist(), best_c5, n_points
