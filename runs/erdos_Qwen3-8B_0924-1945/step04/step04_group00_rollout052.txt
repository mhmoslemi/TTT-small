```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))
    def exact_objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    n_particles = 30
    particles = np.array([np.random.rand(n_points) * (target_sum / np.sum(np.random.rand(n_points))) for _ in range(n_particles)])
    velocities = np.random.rand(n_particles, n_points) * 0.1
    pbest = particles.copy()
    pbest_fitness = np.array([exact_objective(p.reshape(-1)) for p in particles])
    gbest = particles[np.argmin(pbest_fitness)].copy()
    gbest_fitness = np.min(pbest_fitness)
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = 0.7 * velocities[i] + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_h = np.sum(particles[i])
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            particles[i] = particles[i] * scale
            current_fitness = exact_objective(particles[i].copy())
            if current_fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = current_fitness
        current_gbest_fitness = np.min(pbest_fitness)
        if current_gbest_fitness < gbest_fitness:
            gbest = pbest[np.argmin(pbest_fitness)].copy()
            gbest_fitness = current_gbest_fitness
        if _ % 10 == 0:
            noise = np.random.normal(0, 0.05, size=n_points)
            particles += noise
            particles = np.clip(particles, 0.0, 1.0)
            for i in range(n_particles):
                sum_h = np.sum(particles[i])
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                particles[i] = particles[i] * scale
        if gbest_fitness < best_c5:
            best_c5 = gbest_fitness
            best_h = gbest.copy()
    if time.time() - start_time < budget_s:
        result = opt.minimize(
            exact_objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=[(0.0, 1.0) for _ in range(n_points)], 
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}],
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = exact_objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
    return best_h.tolist(), best_c5, n_points
```
