```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    n_particles = 30
    particles = []
    velocities = []
    pbest = []
    pbest_obj = []
    gbest = h.copy()
    gbest_obj = objective(h)
    for _ in range(n_particles):
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(h.copy())
        pbest_obj.append(objective(h))
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(n_particles):
            h = particles[i].copy()
            v = velocities[i].copy()
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            obj = objective(h)
            if obj < pbest_obj[i]:
                pbest[i] = h.copy()
                pbest_obj[i] = obj
            if obj < gbest_obj:
                gbest = h.copy()
                gbest_obj = obj
            r1, r2 = np.random.rand(2)
            v = 0.7 * v + 1.5 * r1 * (pbest[i] - h) + 1.5 * r2 * (gbest - h)
            velocities[i] = v
            h_new = h + v
            h_new = np.clip(h_new, 0.0, 1.0)
            sum_h_new = np.sum(h_new)
            scale_new = target_sum / sum_h_new if sum_h_new != 0 else 1.0
            particles[i] = h_new * scale_new
        if _ % 10 == 0:
            noise = np.random.normal(0, 0.02, n_points)
            particles[0] = particles[0] + noise
            particles[0] = np.clip(particles[0], 0.0, 1.0)
            sum_h = np.sum(particles[0])
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            particles[0] = particles[0] * scale
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, gbest.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
    return best_h.tolist(), best_c5, n_points
```
