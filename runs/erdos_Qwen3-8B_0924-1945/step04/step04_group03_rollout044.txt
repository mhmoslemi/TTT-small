import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0.0, 1.0, n_points)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    n_particles = 30
    particles = np.random.uniform(0.0, 1.0, (n_particles, n_points))
    particles = particles * (target_sum / np.sum(particles, axis=1, keepdims=True))
    pbest = particles.copy()
    pbest_values = np.array([objective(p) for p in pbest])
    gbest = pbest[np.argmin(pbest_values)]
    gbest_value = np.min(pbest_values)
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    iterations = 0

    while time.time() - start_time < budget_s - 5:
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (pbest[i] - particles[i]) + social * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_h = np.sum(particles[i])
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            particles[i] *= scale
            current_value = objective(particles[i])
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
        if np.min(pbest_values) < gbest_value:
            gbest = pbest[np.argmin(pbest_values)]
            gbest_value = np.min(pbest_values)
        iterations += 1
        if iterations % 100 == 0:
            particles = np.random.uniform(0.0, 1.0, (n_particles, n_points))
            particles = particles * (target_sum / np.sum(particles, axis=1, keepdims=True))

    best_h = gbest
    best_c5 = gbest_value

    return best_h.tolist(), best_c5, n_points
