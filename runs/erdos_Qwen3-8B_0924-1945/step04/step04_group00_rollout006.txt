```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.cos(np.pi * x))
        h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h))
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    
    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    
    def repair(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        return h * scale
    
    def mutate(h, mutation_rate):
        h = h.copy()
        if np.random.rand() < mutation_rate:
            idx = np.random.choice(n_points, size=10, replace=False)
            h[idx] = np.random.uniform(0.0, 1.0, size=10)
        return repair(h)
    
    def crossover(parent1, parent2):
        point = np.random.randint(1, n_points - 1)
        child = np.concatenate((parent1[:point], parent2[point:]))
        return repair(child)
    
    pop_size = 20
    population = [repair(np.random.uniform(0.0, 1.0, n_points)) for _ in range(pop_size)]
    for gen in range(500):
        fitness = [evaluate(ind) for ind in population]
        parents = np.argsort(fitness)[:int(pop_size * 0.8)]
        new_population = []
        for _ in range(pop_size):
            parent1, parent2 = np.random.choice(parents, size=2, replace=False)
            child = crossover(population[parent1], population[parent2])
            child = mutate(child, 0.75)
            new_population.append(child)
        population = new_population
        if gen % 10 == 0:
            current_c5 = evaluate(population[0])
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = population[0].copy()
            if time.time() - start_time > budget_s - 10:
                break
    if time.time() - start_time < budget_s:
        result = opt.minimize(
            evaluate, best_h.flatten(),
            method='L-BFGS-B', bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        )
        best_h = result.x.reshape(n_points)
        best_h = np.clip(best_h, 0.0, 1.0)
        sum_h = np.sum(best_h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        best_h = best_h * scale
        best_c5 = evaluate(best_h)
    return best_h.tolist(), best_c5, n_points
```
