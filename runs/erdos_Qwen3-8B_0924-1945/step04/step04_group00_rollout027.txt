```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.cos(np.pi * x))
        h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0.0, 1.0)
    h = h * (target_sum / np.sum(h))
    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = corr * dx
        return np.max(overlaps)
    def repair(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        return h.flatten()
    def mutate(h_flat, mutation_rate=0.1):
        h = h_flat.reshape(n_points)
        if np.random.rand() < mutation_rate:
            indices = np.random.choice(n_points, size=5, replace=False)
            h[indices] = np.random.uniform(0.0, 1.0, size=5)
        if np.random.rand() < mutation_rate:
            indices = np.random.choice(n_points, size=10, replace=False)
            h[indices] += np.random.normal(0.0, 0.05, size=10)
            h = np.clip(h, 0.0, 1.0)
        if np.random.rand() < mutation_rate:
            inv_mask = np.random.choice(n_points, size=10, replace=False)
            h[inv_mask] = 1.0 - h[inv_mask]
        return repair(h.flatten())
    def crossover(parent1, parent2):
        idx = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:idx], parent2[idx:]])
        return repair(child)
    def select_parents(population):
        indices = np.random.choice(len(population), size=5, replace=False)
        return [population[i] for i in indices]
    population = [np.random.uniform(0.0, 1.0, n_points) for _ in range(20)]
    population = [repair(h) for h in population]
    population = [h for h in population if np.all(0.0 <= h <= 1.0)]
    population = [h for h in population if abs(np.sum(h) - target_sum) < 1e-5]
    best_h = population[0]
    best_c5 = evaluate(best_h)
    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        parents = select_parents(population)
        children = []
        for p1, p2 in zip(parents[::2], parents[1::2]):
            child = crossover(p1, p2)
            children.append(child)
        for child in children:
            child = mutate(child)
            children.append(child)
        population += children
        population = sorted(population, key=lambda x: evaluate(x))
        population = population[:20]
        for h in population:
            if abs(np.sum(h) - target_sum) > 1e-5:
                h = repair(h)
        population = [h for h in population if np.all(0.0 <= h <= 1.0)]
        population = [h for h in population if abs(np.sum(h) - target_sum) < 1e-5]
        for h in population:
            c5 = evaluate(h)
            if c5 < best_c5:
                best_h = h
                best_c5 = c5
    if time.time() - start_time < budget_s:
        for _ in range(10):
            h = best_h.copy()
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            def obj_func(x):
                x = x.reshape(n_points)
                x = np.clip(x, 0.0, 1.0)
                sum_x = np.sum(x)
                scale = target_sum / sum_x if sum_x != 0 else 1.0
                x = x * scale
                corr = np.correlate(x, 1.0 - x, mode="full")
                return np.max(corr * dx)
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            result = opt.minimize(obj_func, h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-8, options={'maxiter': 100})
            h = result.x
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            c5 = evaluate(h)
            if c5 < best_c5:
                best_h = h
                best_c5 = c5
    return best_h.tolist(), best_c5, n_points
```
