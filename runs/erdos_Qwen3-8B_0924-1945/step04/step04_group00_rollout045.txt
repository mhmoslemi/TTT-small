```python
import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.round(h)
        h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = corr * dx
        min_overlap = np.min(overlaps)
        if min_overlap < 1e-10:
            break
        k = np.argmin(overlaps)
        if k >= len(h):
            k = len(h) - 1
        if k < len(h):
            for i in range(len(h)):
                if h[i] == 0 and h[(i + k) % len(h)] == 1:
                    h[i] = 1
                    break
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    for _ in range(50):
        if time.time() - start_time > budget_s - 5:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        overlaps = corr * dx
        min_overlap = np.min(overlaps)
        if min_overlap < 1e-10:
            break
        k = np.argmin(overlaps)
        if k >= len(best_h):
            k = len(best_h) - 1
        if k < len(best_h):
            for i in range(len(best_h)):
                if best_h[i] == 0 and best_h[(i + k) % len(best_h)] == 1:
                    best_h[i] = 1
                    break
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (target_sum / np.sum(best_h))
        c5 = compute_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5
    
    return best_h.tolist(), best_c5, n_points
```
