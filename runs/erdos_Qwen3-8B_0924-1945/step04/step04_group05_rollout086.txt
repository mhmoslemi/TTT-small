import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, LinearConstraint

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = h * (target_sum / np.sum(h))
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    def lag_constraints(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return corr * dx

    def solve_convex_relaxation(h_init):
        h = h_init.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        h = h.reshape(n_points)
        h = h.astype(float)

        max_lag = n_points - 1
        c = opt.Variable(n_points)
        h_var = opt.Variable(n_points)
        constraints = [
            opt.constraint(sum(h_var) == target_sum),
            opt.constraint(0 <= h_var <= 1)
        ]
        
        for k in range(1, max_lag + 1):
            idx = np.arange(n_points - k)
            constraints.append(opt.constraint(opt.sum(h_var[idx] * h_var[idx + k]) <= c[k - 1]))

        prob = opt.Problem(opt.Minimize(c[0]), constraints)
        try:
            result = prob.solve(solver='ECOS', eps=1e-8)
            if result is not None:
                return h_var.value
        except:
            pass
        return h

    def refine_solution(h):
        h = h.copy()
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        return h

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()

    for _ in range(3):
        h = solve_convex_relaxation(h)
        h = refine_solution(h)
        c5_bound = objective(h.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h.copy()
        if time.time() - start_time > budget_s - 5:
            break

    if time.time() - start_time < budget_s:
        for _ in range(5):
            noise = np.random.normal(0, 0.02, size=n_points)
            h = best_h + noise
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            c5_bound = objective(h.flatten())
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h = h.copy()

    return best_h.tolist(), best_c5, n_points
