import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[::2] = 1
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    best_h_orig = h.copy()
    start_time = time.time()

    def solve_with_bounds(h_init, noise):
        h = h_init.copy()
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale

        def obj(x):
            h_flat = x
            h_flat = np.clip(h_flat, 0.0, 1.0)
            sum_h_flat = np.sum(h_flat)
            scale_flat = target_sum / sum_h_flat if sum_h_flat != 0 else 1.0
            h_flat = h_flat * scale_flat
            corr = np.correlate(h_flat, 1.0 - h_flat, mode="full")
            return np.max(corr * dx)

        result = opt.minimize(
            obj, h, bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 100}
        )
        h_opt = result.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale_opt = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale_opt
        c5_opt = objective(h_opt)
        return h_opt, c5_opt

    for _ in range(10):
        if time.time() - start_time > budget_s - 60:
            break
        h_noise = np.random.normal(0, 0.05, size=n_points)
        h = best_h + h_noise
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        h_opt, c5_opt = solve_with_bounds(h, 0.05)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        if np.random.rand() < 0.5:
            h_noise = np.random.normal(0, 0.1, size=n_points)
            h = best_h + h_noise
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            h_opt, c5_opt = solve_with_bounds(h, 0.1)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    if time.time() - start_time < budget_s - 60:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0.0, 1.0)
        sum_h = np.sum(h_final)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_final = h_final * scale
        corr = np.correlate(h_final, 1.0 - h_final, mode="full")
        final_c5 = np.max(corr * dx)
        return h_final.tolist(), final_c5, n_points

    h_final = best_h.copy()
    h_final = np.clip(h_final, 0.0, 1.0)
    sum_h = np.sum(h_final)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h_final = h_final * scale
    corr = np.correlate(h_final, 1.0 - h_final, mode="full")
    final_c5 = np.max(corr * dx)
    return h_final.tolist(), final_c5, n_points
