import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))
    def exact_objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    def normalize(h):
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        return h * scale
    def clip_and_normalize(h):
        h = np.clip(h, 0.0, 1.0)
        return normalize(h)
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    n_particles = 30
    n_iterations = 200
    omega = 0.7
    phi_p = 1.5
    phi_g = 1.5
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    positions = np.random.uniform(0, 1, (n_particles, n_points))
    personal_best = positions.copy()
    personal_best_values = np.array([exact_objective(p.reshape(n_points)) for p in positions])
    global_best = positions[0].copy()
    global_best_value = personal_best_values[0]
    for _ in range(n_iterations):
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = omega * velocities[i] + phi_p * r1 * (personal_best[i] - positions[i]) + phi_g * r2 * (global_best - positions[i])
            positions[i] = positions[i] + velocities[i]
            positions[i] = clip_and_normalize(positions[i])
            current_value = exact_objective(positions[i].copy())
            if current_value < personal_best_values[i]:
                personal_best[i] = positions[i].copy()
                personal_best_values[i] = current_value
            if current_value < global_best_value:
                global_best = positions[i].copy()
                global_best_value = current_value
        if time.time() - start_time > budget_s - 5:
            break
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            exact_objective, global_best.reshape(n_points), 
            method='L-BFGS-B', bounds=[(0.0, 1.0) for _ in range(n_points)], 
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x
        h_opt = clip_and_normalize(h_opt)
        c5_bound = exact_objective(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
    return best_h.tolist(), best_c5, n_points
