```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = h * (target_sum / np.sum(h))
    
    def evaluate_correlation(h):
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    
    for _ in range(200):
        h_flat = h.copy()
        h_flat = np.clip(h_flat, 0.0, 1.0)
        sum_h = np.sum(h_flat)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_flat = h_flat * scale
        
        max_lag = n_points - 1
        variables = cp.Variable(n_points)
        objective = cp.Minimize(cp.Variable(1))
        constraints = [cp.sum(variables) == target_sum]
        
        for k in range(1, max_lag + 1):
            if k >= n_points:
                break
            inner = cp.sum(cp.multiply(variables[:n_points - k], variables[k:]))
            constraints.append(inner <= objective)
        
        prob = cp.Problem(objective, constraints)
        try:
            result = prob.solve(solver=cp.ECOS, eps=1e-6, verbose=False)
            if result is not None and result < best_c5:
                best_c5 = result
                best_h = variables.value
        except:
            pass
        
        h = best_h.copy()
        if time.time() - start_time > budget_s - 5:
            break
        noise = np.random.normal(0, 0.02, size=n_points)
        h += noise
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
    
    h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h = h * scale
    final_c5 = evaluate_correlation(h)
    
    return h.tolist(), final_c5, n_points
```
