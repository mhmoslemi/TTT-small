```python
import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import minimize
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, WhiteKernel

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    X = [h.tolist()]
    y = [objective(h)]
    kernel = RBF(length_scale=1.0, length_scale_bounds=(1e-5, 1e5)) + WhiteKernel(noise_level=1e-5, noise_level_bounds=(1e-10, 1e-3))

    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-10, n_restarts_optimizer=25)
        gp.fit(np.array(X), np.array(y))
        if _ > 5 and len(X) > 10:
            mu, sigma = gp.predict(np.array([h.tolist()]), return_std=True)
            def ei(x):
                x_vec = np.array(x).reshape(1, -1)
                mu_pred, sigma_pred = gp.predict(x_vec, return_std=True)
                z = (mu_pred - best_c5) / (sigma_pred + 1e-8)
                ei = (mu_pred - best_c5) * np.sqrt(1 - np.exp(-2 * z ** 2)) + sigma * np.sqrt(1 - np.exp(-2 * z ** 2))
                return ei[0]
            def acq_func(x):
                return -ei(x)
            res = minimize(acq_func, x0=h.tolist(), bounds=[(0, 1) for _ in range(n_points)], method='L-BFGS-B')
            next_h = res.x
        else:
            next_h = np.random.uniform(0, 1, n_points)
            next_h = next_h * (target_sum / np.sum(next_h))
        next_h = np.clip(next_h, 0.0, 1.0)
        next_h = next_h * (target_sum / np.sum(next_h))
        next_c5 = objective(next_h)
        if next_c5 < best_c5:
            best_c5 = next_c5
            best_h = next_h.copy()
        X.append(next_h.tolist())
        y.append(next_c5)

    h = best_h.copy()
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h = h * scale
    c5_bound = objective(h)
    return h.tolist(), c5_bound, n_points
```
