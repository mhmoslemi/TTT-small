import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.cos(np.pi * x))
        h = h * (target_sum / np.sum(h))
    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = corr * dx
        return np.max(overlaps)
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    n_particles = 30
    inertia = 0.7
    phi_p = 1.5
    phi_g = 1.5
    particles = np.random.uniform(0, 1, (n_particles, n_points))
    particles = (particles * target_sum / np.sum(particles, axis=1, keepdims=True)).astype(np.float64)
    pbest = particles.copy()
    pbest_fitness = np.array([evaluate(p) for p in pbest])
    gbest = pbest[np.argmin(pbest_fitness)]
    gbest_fitness = np.min(pbest_fitness)
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = inertia * velocities[i] + phi_p * r1 * (pbest[i] - particles[i]) + phi_g * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            sum_h = np.sum(particles[i])
            if sum_h != 0:
                particles[i] *= target_sum / sum_h
            fitness = evaluate(particles[i])
            if fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fitness
        if np.min(pbest_fitness) < gbest_fitness:
            gbest = pbest[np.argmin(pbest_fitness)]
            gbest_fitness = np.min(pbest_fitness)
        if np.random.rand() < 0.05:
            noise = np.random.normal(0, 0.05, n_points)
            gbest += noise
            gbest = np.clip(gbest, 0, 1)
            sum_h = np.sum(gbest)
            if sum_h != 0:
                gbest *= target_sum / sum_h
            gbest_fitness = evaluate(gbest)
        if _ % 10 == 0:
            h = gbest.copy()
            c5_bound = gbest_fitness
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h = h.copy()
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 10:
        result = opt.minimize(
            evaluate, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = evaluate(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
    return best_h.tolist(), best_c5, n_points
