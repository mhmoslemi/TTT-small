```python
import numpy as np
import scipy.optimize as opt
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel, WhiteKernel

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 500
    n_initial = 20

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * target_sum / np.sum(h)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    X = []
    y = []
    if initial_h_values is not None:
        h_initial = np.array(initial_h_values, dtype=np.float64)
        h_initial = h_initial * target_sum / np.sum(h_initial)
        X.append(h_initial)
        y.append(objective(h_initial))
    else:
        for _ in range(n_initial):
            h = np.random.uniform(0, 1, n_points)
            h = h * target_sum / np.sum(h)
            X.append(h)
            y.append(objective(h))

    X = np.array(X)
    y = np.array(y)

    kernel = ConstantKernel(1.0, (1e-3, 1e3)) * RBF(1.0, (1e-3, 1e3)) + WhiteKernel(1e-5, (1e-10, 1e-3))
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-5, n_restarts_optimizer=10)

    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        gp.fit(X, y)
        mu, sigma = gp.predict(X, return_std=True)
        mu = np.clip(mu, 0.0, 1.0)

        def ei(x):
            if sigma.shape[0] == 0:
                return np.inf
            mu_pred = gp.predict(x.reshape(1, -1))[0]
            sigma_pred = sigma
            imp = mu_pred - best_c5
            prob = np.exp(-0.5 * (imp / sigma_pred)**2) * np.sqrt(2 * np.pi) * sigma_pred
            ei_val = imp * prob
            return ei_val

        def acquisition(x):
            x = x.reshape(1, -1)
            mu, sigma = gp.predict(x, return_std=True)
            mu = np.clip(mu, 0.0, 1.0)
            imp = mu - best_c5
            prob = np.exp(-0.5 * (imp / sigma)**2) * np.sqrt(2 * np.pi) * sigma
            ei_val = imp * prob
            return ei_val

        def bounds_transform(x):
            x_clipped = np.clip(x, 0.0, 1.0)
            x_norm = (x_clipped - 0.0) / (1.0 - 0.0)
            return x_norm

        def bounds_inverse(x_norm):
            x = x_norm * (1.0 - 0.0) + 0.0
            return x

        def sample_new_candidate():
            z = np.random.normal(size=n_points)
            x = bounds_inverse(np.clip(z, 0, 1))
            x = np.clip(x, 0.0, 1.0)
            sum_h = np.sum(x)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            x = x * scale
            return x

        new_candidate = sample_new_candidate()
        X_new = np.array([new_candidate])
        y_new = objective(new_candidate)
        X = np.vstack((X, X_new))
        y = np.append(y, y_new)
        best_c5 = min(best_c5, y_new)
        best_h = new_candidate.copy()

        if len(X) > 30:
            X = X[-30:]
            y = y[-30:]

    X = np.array(X)
    y = np.array(y)

    if len(X) > 0:
        gp.fit(X, y)
        best_h_candidate = X[np.argmin(y)]
        best_h = best_h_candidate.copy()
        best_c5 = np.min(y)

    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    best_h = best_h * scale

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
