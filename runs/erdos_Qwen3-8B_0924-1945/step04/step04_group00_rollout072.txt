import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points) > 0.5
        h = h.astype(np.float64)
        h = (h * target_sum) / np.sum(h)
    def exact_objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_overlap = np.max(corr * dx)
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        lag = np.argmax(corr * dx)
        if lag >= len(h) - 1:
            continue
        indices = np.where((h[:len(h) - lag] == 1) & (h[lag:] == 0))[0]
        if len(indices) == 0:
            continue
        i = np.random.choice(indices)
        h[i] = 1 - h[i]
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
            h = np.clip(h, 0.0, 1.0)

            h = h * scale
            corr = np.correlate(h, 1.0 - h, mode="full")
