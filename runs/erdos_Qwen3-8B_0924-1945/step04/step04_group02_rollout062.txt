import numpy as np
import scipy.optimize as opt
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, WhiteKernel

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    X = []
    y = []
    for _ in range(20):
        h_candidate = np.random.rand(n_points)
        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        X.append(h_candidate)
        y.append(objective(h_candidate))
    X = np.array(X)
    y = np.array(y)

    kernel = RBF(length_scale=1.0, length_scale_bounds=(1e-5, 1e5)) + WhiteKernel(noise_level=1e-5)
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-5, n_restarts_optimizer=25, random_state=seed)

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()

            break

        mu, sigma = gp.predict(X, return_std=True)
        sigma[sigma < 1e-5] = 1e-5

            sum_x = np.sum(x_flat)

            x_scaled = x_flat * scale
            corr = np.correlate(x_scaled, 1.0 - x_scaled, mode="full")
            return np.max(corr * dx)

            return objective_for_ei(x) + 100 * bounds_for_ei(x)

        h_new = x_new.x
        h_new = np.clip(h_new, 0.0, 1.0)
        sum_h = np.sum(h_new)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_new = h_new * scale

        new_c5 = objective(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            X = np.vstack([X, h_new])

        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
