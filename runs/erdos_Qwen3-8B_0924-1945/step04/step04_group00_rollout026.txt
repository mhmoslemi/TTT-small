import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.cos(np.pi * x))
        h = h * (target_sum / np.sum(h))
    def exact_objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    class Particle:
        def __init__(self):
            self.h = np.random.uniform(0, 1, n_points)
            self.h = self.h * (target_sum / np.sum(self.h))
            self.v = np.random.uniform(-0.1, 0.1, n_points)
            self.pbest = self.h.copy()
            self.pbest_val = exact_objective(self.h)
    n_particles = 30
    particles = [Particle() for _ in range(n_particles)]
    gbest = particles[0].h.copy()
    gbest_val = particles[0].pbest_val
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        for p in particles:
            p.h = p.h + p.v
            p.h = np.clip(p.h, 0.0, 1.0)
            sum_h = np.sum(p.h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            p.h = p.h * scale
            current_val = exact_objective(p.h)
            if current_val < p.pbest_val:
                p.pbest_val = current_val
                p.pbest = p.h.copy()
            if current_val < gbest_val:
                gbest_val = current_val
                gbest = p.h.copy()
        for p in particles:
            r1, r2 = np.random.rand(2)
            p.v = 0.7 * p.v + 1.5 * r1 * (p.pbest - p.h) + 1.5 * r2 * (gbest - p.h)
            p.v = np.clip(p.v, -0.1, 0.1)
        if _ % 50 == 0:
            if np.random.rand() < 0.05:
                for p in particles:
                    noise = np.random.normal(0, 0.05, n_points)
                    p.h = p.h + noise
                    p.h = np.clip(p.h, 0.0, 1.0)
                    sum_h = np.sum(p.h)
                    scale = target_sum / sum_h if sum_h != 0 else 1.0
                    p.h = p.h * scale
        if _ % 10 == 0:
            if np.random.rand() < 0.1:
                for p in particles:
                    p.h = np.random.uniform(0, 1, n_points)
                    p.h = p.h * (target_sum / np.sum(p.h))
                    p.v = np.random.uniform(-0.1, 0.1, n_points)
                    p.pbest = p.h.copy()
                    p.pbest_val = exact_objective(p.h)
    best_h = gbest.copy()
    best_c5 = gbest_val
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            exact_objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = exact_objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
    return best_h.tolist(), best_c5, n_points
