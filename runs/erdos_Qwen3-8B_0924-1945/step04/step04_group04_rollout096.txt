```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 if i % 3 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    # Step 1: Global search with structured perturbations
    for _ in range(100):
        direction = np.random.rand(n_points) * 2 - 1
        direction = direction / np.linalg.norm(direction, 1)
        h_perturbed = h + 0.05 * direction
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        sum_h_perturbed = np.sum(h_perturbed)
        scale = target_sum / sum_h_perturbed if sum_h_perturbed != 0 else 1.0
        h_perturbed = h_perturbed * scale
        c5 = objective(h_perturbed.flatten())
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_perturbed.copy()
        if time.time() - start_time > budget_s - 150:
            break

    # Step 2: Local refinement with L-BFGS-B
    result = opt.minimize(
        objective, best_h.flatten(), 
        method='L-BFGS-B', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 300}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt.flatten())
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    # Step 3: Final refinement and evaluation
    for _ in range(100):
        noise = np.random.normal(0, 0.02, size=n_points)
        h_perturbed = h_opt + noise
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        sum_h_perturbed = np.sum(h_perturbed)
        scale = target_sum / sum_h_perturbed if sum_h_perturbed != 0 else 1.0
        h_perturbed = h_perturbed * scale
        c5 = objective(h_perturbed.flatten())
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_perturbed.copy()
        if time.time() - start_time > budget_s - 10:
            break

    return best_h.tolist(), best_c5, n_points
```
