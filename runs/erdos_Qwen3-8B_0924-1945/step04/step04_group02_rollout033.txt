```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0.0, 1.0, n_points)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    n_particles = 40
    particles = np.random.uniform(0.0, 1.0, (n_particles, n_points))
    particles = particles * (target_sum / np.sum(particles, axis=1, keepdims=True))
    pbest = particles.copy()
    pbest_fitness = np.array([objective(p) for p in pbest])
    gbest = pbest[np.argmin(pbest_fitness)]
    gbest_fitness = np.min(pbest_fitness)
    inertia = 0.9
    inertia_decay = (0.9 - 0.4) / 200
    c1, c2 = 2.0, 2.0
    mutation_amp = 0.05
    mutation_decay = (0.05 - 0.02) / 200
    elite_fraction = 0.05
    max_generations = 200
    generations = 0

    while generations < max_generations and time.time() - start_time < budget_s - 20:
        velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = inertia * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
            fitness = objective(particles[i])
            if fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fitness
        gbest = pbest[np.argmin(pbest_fitness)]
        gbest_fitness = np.min(pbest_fitness)
        if gbest_fitness < best_c5:
            best_c5 = gbest_fitness
            best_h = gbest.copy()
        inertia -= inertia_decay
        mutation_amp -= mutation_decay
        generations += 1

    if time.time() - start_time < budget_s - 20:
        elites = pbest[np.argsort(pbest_fitness)[:int(n_particles * elite_fraction)]]
        for i in range(len(elites)):
            result = opt.minimize(
                objective, elites[i].flatten(), 
                method='L-BFGS-B', bounds=bounds, constraints=cons,
                tol=1e-8, options={'maxiter': 50}
            )
            h_opt = result.x.reshape(n_points)
            h_opt = np.clip(h_opt, 0.0, 1.0)
            sum_h_opt = np.sum(h_opt)
            scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
            h_opt = h_opt * scale
            c5_bound = objective(h_opt.flatten())
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
