import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 if np.random.rand() < 0.5 else 0.0
        h = h * (target_sum / np.sum(h))

    def compute_c5(h):
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def normalize(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return h * (target_sum / s)

    def mutate(parent):
        a, b, c = np.random.choice(n_points, 3, replace=False)
        mutant = np.copy(parent)
        mutant[a] += np.random.uniform(-0.2, 0.2)
        mutant[b] += np.random.uniform(-0.2, 0.2)
        mutant[c] += np.random.uniform(-0.2, 0.2)
        mutant = np.clip(mutant, 0.0, 1.0)
        mutant = normalize(mutant)
        return mutant

    population = [normalize(np.random.rand(n_points)) for _ in range(30)]
    best_h = np.copy(population[0])
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    gen = 0
    no_improvement = 0

    while time.time() - start_time < budget_s - 5 and no_improvement < 50:
        fitness = [compute_c5(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5_current = fitness[best_idx]
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = np.copy(population[best_idx])
            no_improvement = 0
        else:
            no_improvement += 1

        new_population = []
        for _ in range(15):
            a, b, c = np.random.choice(n_points, 3, replace=False)
            parent1 = population[np.random.choice(len(population))]
            parent2 = population[np.random.choice(len(population))]
            parent3 = population[np.random.choice(len(population))]
            mutant = np.copy(parent1)
            mutant[a] += np.random.uniform(-0.2, 0.2)
            mutant[b] += np.random.uniform(-0.2, 0.2)
            mutant[c] += np.random.uniform(-0.2, 0.2)
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = normalize(mutant)
            new_population.append(mutant)

        population += new_population
        population.sort(key=lambda x: compute_c5(x))
        population = population[:30]
        gen += 1

    return best_h.tolist(), best_c5, n_points
