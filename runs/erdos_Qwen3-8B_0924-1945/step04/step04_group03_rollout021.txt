import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    n_particles = 30
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    particles = np.random.rand(n_particles, n_points)
    particles = particles * (target_sum / np.mean(np.sum(particles, axis=1)))
    pbest = particles.copy()
    pbest_values = np.array([objective(p) for p in pbest])
    gbest = pbest[np.argmin(pbest_values)]
    gbest_value = np.min(pbest_values)

    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        velocities = np.random.rand(n_particles, n_points) * 0.01
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (pbest[i] - particles[i]) + social * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            sum_p = np.sum(particles[i])
            scale = target_sum / sum_p if sum_p != 0 else 1.0
            particles[i] *= scale
            curr_val = objective(particles[i])
            if curr_val < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = curr_val
        curr_gbest = np.argmin(pbest_values)
        if pbest_values[curr_gbest] < gbest_value:
            gbest = pbest[curr_gbest].copy()
            gbest_value = pbest_values[curr_gbest]
        if np.random.rand() < 0.05:
            perturbation = np.random.normal(0, 0.05, n_points)
            particles += perturbation
            particles = np.clip(particles, 0, 1)
            for i in range(n_particles):
                sum_p = np.sum(particles[i])
                scale = target_sum / sum_p if sum_p != 0 else 1.0
                particles[i] *= scale

    h_best = gbest
    sum_h_best = np.sum(h_best)
    scale = target_sum / sum_h_best if sum_h_best != 0 else 1.0
    h_best *= scale
    c5_bound = objective(h_best)

    return h_best.tolist(), c5_bound, n_points
