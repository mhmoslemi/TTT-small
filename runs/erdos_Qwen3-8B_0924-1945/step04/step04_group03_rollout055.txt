import numpy as np
import time
import scipy.optimize as opt

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.sort(h)
        h = np.interp(h, np.linspace(0, 1, n_points), np.linspace(0, 1, n_points))
        h = np.round(h * 100).astype(int)
        h = np.where(h > 0, h / 100, 0)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    n_population = 30
    gen = 0
    pop = [np.random.rand(n_points) for _ in range(n_population)]
    pop = [h * target_sum / np.sum(h) for h in pop]
    pop = [np.clip(h, 0.0, 1.0) for h in pop]

    while time.time() - start_time < budget_s:
        gen += 1
        fitness = [objective(h) for h in pop]
        idxs = np.argsort(fitness)
        top = idxs[:int(n_population * 0.2)]
        best_idx = idxs[0]

        new_pop = []
        for _ in range(n_population):
            a, b, c = np.random.choice(top, 3, replace=False)
            h_a, h_b, h_c = pop[a], pop[b], pop[c]
            mutant = h_a + 0.5 * (h_b - h_c)
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = mutant * target_sum / np.sum(mutant)
            new_pop.append(mutant)

        pop = new_pop
        fitness = [objective(h) for h in pop]
        idxs = np.argsort(fitness)
        pop = [pop[i] for i in idxs[:n_population]]
        current_c5 = fitness[0]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = pop[0].copy()

        if gen % 50 == 0 and time.time() - start_time > budget_s - 10:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
