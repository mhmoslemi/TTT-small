import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    generation = 0
    n_particles = 40
    particles = np.random.rand(n_particles, n_points)
    particles = particles * target_sum / np.sum(particles, axis=1, keepdims=True)
    pbest = particles.copy()
    pbest_fitness = np.full(n_particles, np.inf)
    gbest = particles[0].copy()
    gbest_fitness = np.inf
    inertia_weight = 0.9
    omega_decay = (0.9 - 0.4) / 200
    c1, c2 = 2.0, 2.0
    mutation_amp = 0.05
    mutation_decay = (0.05 - 0.02) / 200
    elite_fraction = 0.05
    elite_size = int(n_particles * elite_fraction)
    max_generations = 200

    while generation < max_generations and time.time() - start_time < budget_s - 10:
        velocities = np.random.rand(n_particles, n_points) * 0.01
        for i in range(n_particles):
            pos = particles[i]
            vel = velocities[i]
            r1, r2 = np.random.rand(2)
            vel = (inertia_weight * vel +
                   c1 * r1 * (pbest[i] - pos) +
                   c2 * r2 * (gbest - pos))
            velocities[i] = vel
            pos = pos + vel
            pos = np.clip(pos, 0.0, 1.0)
            sum_pos = np.sum(pos)
            scale = target_sum / sum_pos if sum_pos != 0 else 1.0
            pos = pos * scale
            particles[i] = pos
            fitness = objective(pos.flatten())
            if fitness < pbest_fitness[i]:
                pbest[i] = pos.copy()
                pbest_fitness[i] = fitness
            if fitness < gbest_fitness:
                gbest = pos.copy()
                gbest_fitness = fitness
        inertia_weight -= omega_decay
        mutation_amp -= mutation_decay
        generation += 1

        if generation % 5 == 0:
            elites = particles[np.argsort(pbest_fitness)[:elite_size]]
            for i in range(elite_size):
                result = opt.minimize(
                    objective, elites[i].flatten(),
                    method='L-BFGS-B', bounds=bounds,
                    tol=1e-8, options={'maxiter': 50}
                )
                h_opt = result.x.reshape(n_points)
                h_opt = np.clip(h_opt, 0.0, 1.0)
                sum_h_opt = np.sum(h_opt)
                scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
                h_opt = h_opt * scale
                c5_bound = objective(h_opt.flatten())
                if c5_bound < best_c5:
                    best_c5 = c5_bound
                    best_h = h_opt.copy()

        if time.time() - start_time > budget_s - 10:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(),
            method='L-BFGS-B', bounds=bounds,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
