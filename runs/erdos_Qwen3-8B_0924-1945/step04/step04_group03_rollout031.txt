import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                h[i] = 1.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def normalize(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        scaling = target_sum / np.sum(h) if np.sum(h) != 0 else 1.0
        return (h * scaling).flatten()

    pop_size = 30
    pop = [normalize(np.random.rand(n_points)) for _ in range(pop_size)]
    best_h = h.copy()
    best_c5 = evaluate(h)
    start_time = time.time()
    gen = 0
    no_improvement = 0

    while time.time() - start_time < budget_s:
        gen += 1
        fitness = [evaluate(ind) for ind in pop]
        fit_pop = list(zip(fitness, pop))
        fit_pop.sort()
        elite = fit_pop[:int(0.2 * pop_size)]
        elite_h = [ind[1] for ind in elite]
        new_pop = elite_h[:int(0.5 * pop_size)]
        while len(new_pop) < pop_size:
            a, b, c = np.random.choice(elite_h, 3, replace=False)
            mutant = a + 0.5 * (b - c)
            mutant = normalize(mutant)
            new_pop.append(mutant)
        pop = new_pop
        current_c5 = evaluate(pop[0])
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = pop[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        if no_improvement > 50:
            break

    return best_h.tolist(), best_c5, n_points
