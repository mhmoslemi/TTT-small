import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    n_particles = 40
    n_generations = 200
    elite_fraction = 0.05
    mutation_amp = 0.05
    inertia_weight = 0.9
    c1 = 2.0
    c2 = 2.0

    positions = np.random.rand(n_particles, n_points)
    positions = positions * target_sum / np.sum(positions, axis=1, keepdims=True)
    velocities = np.random.rand(n_particles, n_points) * 0.01
    pbest_positions = positions.copy()
    pbest_fitness = np.full(n_particles, np.inf)
    gbest_position = positions[0].copy()
    gbest_fitness = np.inf

    best_c5 = np.inf
    best_h = h.copy()

    start_time = time.time()

    for gen in range(n_generations):
        if time.time() - start_time > budget_s - 10:
            break
        for i in range(n_particles):
            h_flat = positions[i]
            h = h_flat.reshape(n_points)
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_scaled = h * scale
            corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
            fitness = -np.max(corr * dx)
            if fitness < pbest_fitness[i]:
                pbest_fitness[i] = fitness
                pbest_positions[i] = h_flat.copy()
            if fitness < gbest_fitness:
                gbest_fitness = fitness
                gbest_position = h_flat.copy()
        
        for i in range(n_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = (inertia_weight * velocities[i] + 
                            c1 * r1 * (pbest_positions[i] - positions[i]) + 
                            c2 * r2 * (gbest_position - positions[i]))
            positions[i] = positions[i] + velocities[i]
            positions[i] = np.clip(positions[i], 0.0, 1.0)
            sum_h = np.sum(positions[i])
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            positions[i] = positions[i] * scale

        mutation_amp = 0.05 * (1 - gen / n_generations)
        for i in range(n_particles):
            for j in range(n_points):
                if np.random.rand() < 0.1:
                    positions[i, j] += mutation_amp * np.random.normal(0, 1)
                    positions[i, j] = np.clip(positions[i, j], 0.0, 1.0)
                    sum_h = np.sum(positions[i])
                    scale = target_sum / sum_h if sum_h != 0 else 1.0
                    positions[i, j] = positions[i, j] * scale

        elites = np.argsort(pbest_fitness)[:int(n_particles * elite_fraction)]
        for i in elites:
            h_flat = pbest_positions[i]
            h = h_flat.reshape(n_points)
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_scaled = h * scale
            corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
            c5 = np.max(corr * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_scaled.copy()

    result = opt.minimize(
        objective, best_h.flatten(), 
        method='L-BFGS-B', bounds=bounds,
        tol=1e-8, options={'maxiter': 50}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt.flatten())
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
