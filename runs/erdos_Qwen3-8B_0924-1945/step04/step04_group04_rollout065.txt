import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 600
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        angles = np.linspace(0, 2 * np.pi, n_points, endpoint=False)
        h = np.random.rand(n_points)
        h = np.clip(1.0 - np.abs(np.sin(angles) - 1.0), 0.0, 1.0)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    def de_search(population, max_iter=100):
        for _ in range(max_iter):
            mutant = np.random.normal(size=n_points)
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = mutant * (target_sum / np.sum(mutant))
            trial = population[np.random.randint(len(population))] + 0.5 * (mutant - population[np.random.randint(len(population))])
            trial = np.clip(trial, 0.0, 1.0)
            trial = trial * (target_sum / np.sum(trial))
            trial_c5 = objective(trial.flatten())
            population_c5 = [objective(p.flatten()) for p in population]
            if trial_c5 < np.min(population_c5):
                population[population_c5.index(np.min(population_c5))] = trial
            if time.time() - start_time > budget_s - 100:
                break
        return population

    def lbfgs_refine(h):
        result = opt.minimize(
            objective, h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 500}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        return h_opt

    population = [np.random.rand(n_points) for _ in range(30)]
    for i in range(len(population)):
        population[i] = np.clip(population[i], 0.0, 1.0)
        population[i] = population[i] * (target_sum / np.sum(population[i]))
    population = de_search(population, max_iter=100)

    best_pop = population[0]
    best_pop_c5 = objective(best_pop.flatten())
    for p in population:
        p_c5 = objective(p.flatten())
        if p_c5 < best_pop_c5:
            best_pop = p
            best_pop_c5 = p_c5

    best_h = lbfgs_refine(best_pop)

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 100:
        for _ in range(5):
            noise = np.random.normal(0, 0.01, n_points)
            best_h += noise
            best_h = np.clip(best_h, 0.0, 1.0)
            sum_h = np.sum(best_h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            best_h = best_h * scale
            c5 = objective(best_h.flatten())
            if c5 < best_pop_c5:
                best_pop_c5 = c5

    final_h = best_h
    final_c5 = objective(final_h.flatten())
    return final_h.tolist(), final_c5, n_points
