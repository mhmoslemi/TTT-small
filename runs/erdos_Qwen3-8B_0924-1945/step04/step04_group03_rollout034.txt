```python
import numpy as np
import time
import scipy.optimize as opt

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0.0, 1.0, n_points)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def repair(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        return h * scale

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    n_particles = 30
    particles = np.random.uniform(0.0, 1.0, (n_particles, n_points))
    particles = np.array([repair(p) for p in particles])
    pbest = particles.copy()
    pbest_fitness = np.array([objective(p) for p in pbest])
    gbest = particles[np.argmin(pbest_fitness)]
    gbest_fitness = np.min(pbest_fitness)
    best_c5 = gbest_fitness
    best_h = gbest.copy()

    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    max_iter = 1000

    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (pbest[i] - particles[i]) + social * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = repair(particles[i])
            curr_fitness = objective(particles[i])
            if curr_fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = curr_fitness
        if np.min(pbest_fitness) < best_c5:
            best_c5 = np.min(pbest_fitness)
            gbest = pbest[np.argmin(pbest_fitness)]
            best_h = gbest.copy()
        if iter % 100 == 0:
            inertia -= 0.001
            if inertia < 0.4:
                inertia = 0.4
        if np.abs(best_c5 - gbest_fitness) < 1e-6:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=[(0.0, 1.0)]*n_points, 
            tol=1e-8, options={'maxiter': 100}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
    
    return best_h.tolist(), best_c5, n_points
```
