```python
import numpy as np
import scipy.optimize as opt
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, WhiteKernel

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)
    
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    
    def evaluate(h_vec):
        h = h_vec.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)
    
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    
    kernel = RBF(length_scale=1.0) + WhiteKernel(noise_level=1e-5)
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-6, n_restarts_optimizer=10)
    
    X = np.array([h])
    y = np.array([evaluate(h)])
    
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        
        mu, sigma = gp.predict(X, return_std=True)
        acq = mu - 1.96 * sigma
        idx = np.argmax(acq)
        x_new = np.random.rand(n_points)
        x_new = x_new * (target_sum / np.sum(x_new))
        
        y_new = evaluate(x_new)
        X = np.vstack((X, x_new))
        y = np.append(y, y_new)
        
        gp.fit(X, y)
        
        if y_new < best_c5:
            best_c5 = y_new
            best_h = x_new.copy()
    
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 100:
        result = opt.minimize(
            objective, best_h.flatten(),
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-6, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = evaluate(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
    
    return best_h.tolist(), best_c5, n_points
```
