```python
import numpy as np
import scipy.optimize as opt
import time
from scipy.stats import norm
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, WhiteKernel

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        m = 4
        l = 1.0 / m
        g = (1.0 - m * l) / 2
        cluster_positions = np.arange(m) * (l + g) + g
        cluster_positions = np.clip(cluster_positions, 0, 2)
        h = np.zeros(n_points)
        for pos in cluster_positions:
            idx = np.searchsorted(np.linspace(0, 2, n_points), pos)
            h[idx - 1:idx + int(round(l / dx))] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    class BayesianOptimization:
        def __init__(self, n_points, target_sum, dx, objective, bounds, initial_points=20):
            self.n_points = n_points
            self.target_sum = target_sum
            self.dx = dx
            self.objective = objective
            self.bounds = bounds
            self.X = []
            self.Y = []
            self.gp = GaussianProcessRegressor(
                kernel=RBF(length_scale=1.0) + WhiteKernel(noise_level=1e-5),
                alpha=1e-10,
                n_restarts_optimizer=10,
                random_state=seed
            )
            self.acquisition_function = self.expected_improvement
            self.initial_points = initial_points

        def expected_improvement(self, x):
            mu, sigma = self.gp.predict(x, return_std=True)
            mu_star = np.max(self.Y)
            imp = mu - mu_star
            imp[imp < 0] = 0
            return imp - 1.96 * sigma

        def sample_uniform(self):
            sample = np.random.uniform(low=[b[0] for b in self.bounds],
                                       high=[b[1] for b in self.bounds],
                                       size=self.n_points)
            sample = np.clip(sample, 0.0, 1.0)
            sum_sample = np.sum(sample)
            scale = self.target_sum / sum_sample if sum_sample != 0 else 1.0
            sample = sample * scale
            return sample

        def get_next_point(self):
            if len(self.X) < self.initial_points:
                return self.sample_uniform()
            else:
                acq = self.acquisition_function
                x = np.random.uniform(low=[b[0] for b in self.bounds],
                                     high=[b[1] for b in self.bounds],
                                     size=(1, self.n_points))
                x = np.clip(x, 0.0, 1.0)
                x = x.reshape(1, self.n_points)
                acq_vals = acq(x)
                idx = np.argmax(acq_vals)
                next_point = x[idx, :]
                return next_point

        def update(self, x, y):
            self.X.append(x)
            self.Y.append(y)
            self.gp.fit(self.X, self.Y)

    bo = BayesianOptimization(n_points, target_sum, dx, objective, bounds)
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    iterations = 0

    while iterations < 500 and time.time() - start_time < budget_s - 100:
        h_candidate = bo.get_next_point()
        h_candidate = np.clip(h_candidate, 0.0, 1.0)
        sum_h = np.sum(h_candidate)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h_candidate * scale
        c5 = objective(h_scaled.flatten())
        bo.update([h_scaled], [c5])
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_scaled.copy()
        iterations += 1

    def refine_solution(h_candidate, budget):
        h_flat = h_candidate.flatten()
        result = opt.minimize(
            lambda x: objective(x),
            h_flat,
            method='L-BFGS-B',
            bounds=bounds,
            tol=1e-8,
            options={'maxiter': int(budget)}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        return h_opt

    if time.time() - start_time < budget_s - 100:
        best_h = refine_solution(best_h, 50)

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, 
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
