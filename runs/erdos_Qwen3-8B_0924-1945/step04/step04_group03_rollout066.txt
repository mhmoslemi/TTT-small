import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_lags = int(2.0 / dx) - 1

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[:int(n_points / 2)] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    def evolve_population(population, F=0.5):
        new_population = []
        for _ in range(len(population)):
            a, b, c = np.random.choice(population, 3, replace=False)
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0.0, 1.0)
            sum_mutant = np.sum(mutant)
            scale = target_sum / sum_mutant if sum_mutant != 0 else 1.0
            mutant = mutant * scale
            new_population.append(mutant)
        return new_population

    def evaluate_population(population):
        scores = []
        for h_flat in population:
            h = h_flat.copy()
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_scaled = h * scale
            corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
            scores.append(np.max(corr * dx))
        return scores

    population_size = 30
    population = [np.clip(np.random.rand(n_points), 0.0, 1.0) * (target_sum / np.sum(np.clip(np.random.rand(n_points), 0.0, 1.0))) for _ in range(population_size)]

    for gen in range(int(budget_s / 10)):
        scores = evaluate_population(population)
        ranked = sorted(zip(scores, population), key=lambda x: x[0])
        elite = ranked[:int(population_size * 0.2)]
        new_population = [e[1] for e in elite]
        while len(new_population) < population_size:
            a, b, c = np.random.choice(population, 3, replace=False)
            mutant = a + 0.5 * (b - c)
            mutant = np.clip(mutant, 0.0, 1.0)
            sum_mutant = np.sum(mutant)
            scale = target_sum / sum_mutant if sum_mutant != 0 else 1.0
            mutant = mutant * scale
            new_population.append(mutant)
        population = new_population
        current_best = ranked[0][0]
        if current_best < best_c5:
            best_c5 = current_best
            best_h = ranked[0][1].copy()
        if time.time() - start_time > budget_s - 5:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(50):
            a, b, c = np.random.choice(population, 3, replace=False)
            mutant = a + 0.5 * (b - c)
            mutant = np.clip(mutant, 0.0, 1.0)
            sum_mutant = np.sum(mutant)
            scale = target_sum / sum_mutant if sum_mutant != 0 else 1.0
            mutant = mutant * scale
            scores = evaluate_population([mutant])
            if scores[0] < best_c5:
                best_c5 = scores[0]
                best_h = mutant.copy()

    return best_h.tolist(), best_c5, n_points
