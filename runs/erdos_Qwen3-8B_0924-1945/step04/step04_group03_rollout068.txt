import numpy as np
import time
from scipy.optimize import differential_evolution

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h_population = [np.array(initial_h_values, dtype=np.float64)]
    else:
        h_population = []
        for _ in range(30):  # initial population
            h = np.zeros(n_points)
            indices = np.random.choice(n_points, size=int(target_sum), replace=False)
            h[indices] = 1.0
            h = h * (target_sum / np.sum(h))
            h_population.append(h)

    def fitness(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    start_time = time.time()

    for _ in range(50):  # generations
        if time.time() - start_time > budget_s - 5:
            break
        population = np.array(h_population)
        fitness_scores = np.array([fitness(h) for h in population])
        best_idx = np.argmin(fitness_scores)
        best_c5 = fitness_scores[best_idx]
        best_h = population[best_idx].copy()
        
        # Select top 20% for breeding
        top_idx = np.argsort(fitness_scores)[:int(len(population) * 0.2)]
        top_population = population[top_idx]
        
        new_population = []
        for _ in range(len(population) // 2):
            a, b, c = np.random.choice(top_idx, 3, replace=False)
            h_a = top_population[a]
            h_b = top_population[b]
            h_c = top_population[c]
            h_mutant = h_a + 0.5 * (h_b - h_c)
            h_mutant = np.clip(h_mutant, 0.0, 1.0)
            sum_h = np.sum(h_mutant)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_mutant = h_mutant * scale
            new_population.append(h_mutant)
        
        # Replace worst half
        population = np.concatenate([top_population, new_population])
        population = np.array([np.clip(h, 0.0, 1.0) for h in population])
        for h in population:
            sum_h = np.sum(h)
            if sum_h != 0:
                scale = target_sum / sum_h
                h = h * scale
            else:
                h = np.zeros(n_points)
        h_population = population

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = differential_evolution(
            fitness,
            bounds,
            strategy='best1bin',
            popsize=15,
            tol=1e-8,
            maxiter=100,
            mutation=(0.5, 1.0),
            recombination=0.9,
            seed=seed
        )
        best_h = result.x
        best_c5 = fitness(best_h)

    return best_h.tolist(), best_c5, n_points
