```python
import numpy as np
import time
import scipy.optimize as opt

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.cos(np.pi * x))
        h = h * (target_sum / np.sum(h))
    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    def normalize(h):
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        return h * scale
    def clip_and_normalize(h):
        h = np.clip(h, 0.0, 1.0)
        return normalize(h)
    def objective(h_flat):
        return evaluate(h_flat)
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    n_particles = 30
    n_iter = 1000
    particles = np.random.rand(n_particles, n_points)
    particles = np.array([normalize(p) for p in particles])
    if initial_h_values is not None:
        particles[0] = normalize(np.array(initial_h_values, dtype=np.float64))
    pbest = particles.copy()
    gbest = particles[0].copy()
    best_c5 = evaluate(pbest[0])
    velocities = np.random.rand(n_particles, n_points) * 0.1
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    for _ in range(n_iter):
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (pbest[i] - particles[i]) + social * r2 * (gbest - particles[i])
            particles[i] = particles[i] + velocities[i]
            particles[i] = clip_and_normalize(particles[i])
            current_c5 = evaluate(particles[i])
            if current_c5 < evaluate(pbest[i]):
                pbest[i] = particles[i].copy()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    gbest = particles[i].copy()
        if best_c5 < evaluate(gbest):
            best_c5 = evaluate(gbest)
            best_h = gbest.copy()
    if time.time() - start_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = evaluate(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
    return best_h.tolist(), best_c5, n_points
```
