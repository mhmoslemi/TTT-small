import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        m = 4
        l = 1.0 / m
        g = 0.5 * (1.0 - m * l)
        cluster_positions = np.arange(m) * (l + g)
        cluster_positions = np.clip(cluster_positions, 0, 2)
        for pos in cluster_positions:
            idx = np.searchsorted(np.linspace(0, 2, n_points), pos)
            h[idx - 1:idx + int(l / dx)] = 1
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    n_population = 30
    for _ in range(1000):
        pop = [np.copy(h) for _ in range(n_population)]
        for i in range(n_population):
            p = np.random.permutation(n_points)
            h_pop = np.zeros(n_points)
            h_pop[p[:int(n_points / 2)]] = 1
            h_pop = h_pop * (target_sum / np.sum(h_pop))
            pop[i] = h_pop
        fit = []
        for h_pop in pop:
            c = objective(h_pop)
            fit.append((c, h_pop))
        fit.sort()
        best_fit, best_pop = fit[0]
        if best_fit < best_c5:
            best_c5 = best_fit
            best_h = best_pop
        if time.time() - start_time > budget_s - 5:
            break
        new_pop = []
        for _ in range(n_population // 2):
            a, b, c = np.random.choice(range(n_population), 3, replace=False)
            h_a, h_b, h_c = pop[a], pop[b], pop[c]
            h_mut = h_a + 0.5 * (h_b - h_c)
            h_mut = np.clip(h_mut, 0.0, 1.0)
            sum_h_mut = np.sum(h_mut)
            scale = target_sum / sum_h_mut if sum_h_mut != 0 else 1.0
            h_mut = h_mut * scale
            new_pop.append(h_mut)
        for h_pop in new_pop:
            c = objective(h_pop)
            fit.append((c, h_pop))
        fit.sort()
        fit = fit[:n_population // 2]
        pop = [x[1] for x in fit]
        pop.append(best_pop)
        pop = pop[:n_population]
        h = pop[0]

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(100):
            a, b, c = np.random.choice(range(n_population), 3, replace=False)
            h_a, h_b, h_c = pop[a], pop[b], pop[c]
            h_mut = h_a + 0.5 * (h_b - h_c)
            h_mut = np.clip(h_mut, 0.0, 1.0)
            sum_h_mut = np.sum(h_mut)
            scale = target_sum / sum_h_mut if sum_h_mut != 0 else 1.0
            h_mut = h_mut * scale
            c = objective(h_mut)
            if c < best_c5:
                best_c5 = c
                best_h = h_mut

    return best_h.tolist(), best_c5, n_points
