```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    n_particles = 40
    n_generations = 200
    elite_fraction = 0.05
    mutation_amp_start = 0.05
    mutation_amp_end = 0.02
    inertia_start = 0.9
    inertia_end = 0.4
    c1 = c2 = 2.0

    particles = np.random.uniform(0, 1, (n_particles, n_points))
    particles = particles * (target_sum / np.sum(particles, axis=1, keepdims=True))
    velocities = np.random.uniform(-0.05, 0.05, (n_particles, n_points))
    pbest = particles.copy()
    pbest_fitness = np.full(n_particles, float('inf'))
    gbest = particles[0].copy()
    gbest_fitness = float('inf')

    for gen in range(n_generations):
        for i in range(n_particles):
            h_val = particles[i]
            sum_h = np.sum(h_val)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_scaled = h_val * scale
            corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
            fitness = -np.max(corr * dx)

            if fitness < pbest_fitness[i]:
                pbest[i] = h_val.copy()
                pbest_fitness[i] = fitness

        gbest_fitness = np.min(pbest_fitness)
        gbest_index = np.argmin(pbest_fitness)
        gbest = pbest[gbest_index].copy()

        inertia = inertia_start - (inertia_start - inertia_end) * gen / n_generations
        mutation_amp = mutation_amp_start - (mutation_amp_start - mutation_amp_end) * gen / n_generations

        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_h = np.sum(particles[i])
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            particles[i] *= scale

            if np.random.rand() < 0.1:
                noise = np.random.normal(0, mutation_amp, n_points)
                particles[i] += noise
                particles[i] = np.clip(particles[i], 0.0, 1.0)
                sum_h = np.sum(particles[i])
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                particles[i] *= scale

        if np.min(pbest_fitness) < gbest_fitness:
            gbest_fitness = np.min(pbest_fitness)
            gbest = pbest[np.argmin(pbest_fitness)].copy()

        if gen % 5 == 0:
            elites = pbest[np.argsort(pbest_fitness)][:int(n_particles * elite_fraction)]
            for h_candidate in elites:
                result = opt.minimize(
                    objective, h_candidate,
                    method='L-BFGS-B', bounds=bounds, constraints=cons,
                    tol=1e-8, options={'maxiter': 50}
                )
                h_opt = result.x
                h_opt = np.clip(h_opt, 0.0, 1.0)
                sum_h_opt = np.sum(h_opt)
                scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
                h_opt = h_opt * scale
                c5_bound = objective(h_opt)
                if c5_bound < best_c5:
                    best_c5 = c5_bound
                    best_h = h_opt.copy()

        if time.time() - start_time > budget_s - 5:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h,
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
