```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    n_particles = 40
    n_generations = 200
    elite_fraction = 0.05
    inertia_weight = 0.9
    c1, c2 = 2.0, 2.0
    mutation_amp = 0.05
    decay_rate = (0.4 - inertia_weight) / n_generations
    mutation_decay = (0.02 - mutation_amp) / n_generations

    particles = np.random.rand(n_particles, n_points)
    particles = particles / np.sum(particles, axis=1, keepdims=True) * target_sum
    particles = np.clip(particles, 0.0, 1.0)
    pbest = particles.copy()
    pbest_fitness = np.full(n_particles, float('inf'))
    gbest = particles[0].copy()
    gbest_fitness = float('inf')

    for gen in range(n_generations):
        for i in range(n_particles):
            h = particles[i]
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_scaled = h * scale
            corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
            fitness = -np.max(corr * dx)
            if fitness < pbest_fitness[i]:
                pbest[i] = h.copy()
                pbest_fitness[i] = fitness
        gbest_idx = np.argmin(pbest_fitness)
        gbest = pbest[gbest_idx].copy()
        gbest_fitness = pbest_fitness[gbest_idx]

        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            v = np.random.rand(n_points)
            v = inertia_weight * v + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            particles[i] += v
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_h = np.sum(particles[i])
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            particles[i] *= scale

        inertia_weight -= decay_rate
        mutation_amp -= mutation_decay

        if gen % 5 == 0:
            elite_count = int(n_particles * elite_fraction)
            elites = particles[np.argsort(pbest_fitness)[:elite_count]]
            for i in range(elite_count):
                h = elites[i]
                h = np.clip(h, 0.0, 1.0)
                sum_h = np.sum(h)
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                h_scaled = h * scale
                corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
                fitness = -np.max(corr * dx)
                if fitness < pbest_fitness[i]:
                    pbest[i] = h.copy()
                    pbest_fitness[i] = fitness

        if time.time() - start_time > budget_s - 5:
            break

    elite_count = int(n_particles * elite_fraction)
    elites = particles[np.argsort(pbest_fitness)[:elite_count]]
    best_candidate = elites[0].copy()
    best_candidate = np.clip(best_candidate, 0.0, 1.0)
    sum_h = np.sum(best_candidate)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    best_candidate *= scale

    def final_objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    result = opt.minimize(
        final_objective, best_candidate.flatten(), 
        method='L-BFGS-B', bounds=bounds,
        tol=1e-8, options={'maxiter': 50}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = final_objective(h_opt.flatten())

    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(5):
            result = opt.minimize(
                final_objective, best_h.flatten(), 
                method='L-BFGS-B', bounds=bounds,
                tol=1e-8, options={'maxiter': 50}
            )
            h_opt = result.x.reshape(n_points)
            h_opt = np.clip(h_opt, 0.0, 1.0)
            sum_h_opt = np.sum(h_opt)
            scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
            h_opt = h_opt * scale
            c5_bound = final_objective(h_opt.flatten())
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
