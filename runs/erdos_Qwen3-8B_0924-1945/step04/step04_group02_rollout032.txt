import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    n_particles = 40
    particles = np.random.rand(n_particles, n_points)
    particles = particles * target_sum / np.sum(particles, axis=1, keepdims=True)
    pbest = particles.copy()
    pbest_fitness = np.full(n_particles, np.inf)
    gbest = particles[0].copy()
    gbest_fitness = np.inf
    velocities = np.random.rand(n_particles, n_points) * 0.01

    inertia_weight = 0.9
    c1, c2 = 2.0, 2.0
    mutation_amp = 0.05

    best_c5 = np.inf
    best_h = h.copy()
    start_time = time.time()

    for gen in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(n_particles):
            h_flat = particles[i]
            h_flat = np.clip(h_flat, 0.0, 1.0)
            sum_h = np.sum(h_flat)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_scaled = h_flat * scale
            corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
            fitness = np.max(corr * dx)
            if fitness < pbest_fitness[i]:
                pbest[i] = h_flat.copy()
                pbest_fitness[i] = fitness
        gbest = pbest[np.argmin(pbest_fitness)]
        gbest_fitness = np.min(pbest_fitness)

        if gbest_fitness < best_c5:
            best_c5 = gbest_fitness
            best_h = gbest.copy()

        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (inertia_weight * velocities[i] +
                             c1 * r1 * (pbest[i] - particles[i]) +
                             c2 * r2 * (gbest - particles[i]))
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_h = np.sum(particles[i])
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            particles[i] = particles[i] * scale

        velocities *= 0.99
        mutation_amp = max(0.02, mutation_amp - 0.0001)
        for i in range(n_particles):
            noise = np.random.normal(0, mutation_amp, n_points)
            particles[i] += noise
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_h = np.sum(particles[i])
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            particles[i] = particles[i] * scale

    if time.time() - start_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, 
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
