```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    max_gen = 200
    n_particles = 40
    elite_frac = 0.05
    n_elite = int(n_particles * elite_frac)
    pbest = [np.full(n_points, np.nan) for _ in range(n_particles)]
    gbest = np.full(n_points, np.nan)
    velocity = np.zeros((n_particles, n_points))
    inertia = 0.9
    inertia_decay = (inertia - 0.4) / max_gen
    c1 = 2.0
    c2 = 2.0
    mutation_amp = 0.05
    mutation_decay = (mutation_amp - 0.02) / max_gen

    for gen in range(max_gen):
        for i in range(n_particles):
            h_i = h[i * n_points:(i + 1) * n_points]
            h_i = np.clip(h_i, 0.0, 1.0)
            sum_h_i = np.sum(h_i)
            scale_i = target_sum / sum_h_i if sum_h_i != 0 else 1.0
            h_scaled_i = h_i * scale_i
            corr_i = np.correlate(h_scaled_i, 1.0 - h_scaled_i, mode="full")
            c5_i = np.max(corr_i * dx)
            if np.isnan(c5_i):
                c5_i = best_c5
            if c5_i < pbest[i]:
                pbest[i] = c5_i
                h_pbest = h_i.copy()
                h_pbest = h_pbest * (target_sum / np.sum(h_pbest))
                pbest[i] = h_pbest
        gbest = np.min(pbest, axis=0)
        for i in range(n_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocity[i] = (inertia * velocity[i] + 
                          c1 * r1 * (pbest[i] - h[i]) + 
                          c2 * r2 * (gbest - h[i]))
            h_i = h[i] + velocity[i]
            h_i = np.clip(h_i, 0.0, 1.0)
            sum_h_i = np.sum(h_i)
            scale_i = target_sum / sum_h_i if sum_h_i != 0 else 1.0
            h_i = h_i * scale_i
            h[i] = h_i
        inertia -= inertia_decay
        mutation_amp -= mutation_decay
        for i in range(n_particles):
            if np.random.rand() < 0.1:
                noise = np.random.normal(0, mutation_amp, size=n_points)
                h_i = h[i] + noise
                h_i = np.clip(h_i, 0.0, 1.0)
                sum_h_i = np.sum(h_i)
                scale_i = target_sum / sum_h_i if sum_h_i != 0 else 1.0
                h[i] = h_i * scale_i
        if gen % 5 == 0:
            for i in range(n_elite):
                h_candidate = pbest[i]
                result = opt.minimize(
                    objective, h_candidate,
                    method='L-BFGS-B', bounds=bounds, constraints=cons,
                    tol=1e-8, option={'maxiter': 50}
                )
                h_candidate = result.x
                h_candidate = np.clip(h_candidate, 0.0, 1.0)
                sum_h_candidate = np.sum(h_candidate)
                scale_candidate = target_sum / sum_h_candidate if sum_h_candidate != 0 else 1.0
                h_candidate = h_candidate * scale_candidate
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()
        if time.time() - start_time > budget_s - 5:
            break

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h,
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, option={'maxiter': 50}
        )
        h_opt = result.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale_opt = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale_opt
        c5_bound = objective(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()
    return best_h.tolist(), best_c5, n_points
```
