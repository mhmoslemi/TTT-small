```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    n_particles = 30
    particles = np.random.rand(n_particles, n_points)
    particles = particles * (target_sum / np.sum(particles, axis=1, keepdims=True))
    particles = np.clip(particles, 0.0, 1.0)
    pbest = particles.copy()
    pbest_vals = np.array([objective(p) for p in pbest])
    gbest = pbest[np.argmin(pbest_vals)]
    gbest_val = np.min(pbest_vals)

    omega = 0.7
    c1 = 1.5
    c2 = 1.5
    inertia_decay = 0.001
    max_iter = int(budget_s * 0.95 / 0.01)
    iter_count = 0
    no_improvement = 0

    while iter_count < max_iter and time.time() - start_time < budget_s - 5:
        velocities = np.random.rand(n_particles, n_points) * 0.01
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] +
                             c1 * r1 * (pbest[i] - particles[i]) +
                             c2 * r2 * (gbest - particles[i]))
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_h = np.sum(particles[i])
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            particles[i] = particles[i] * scale

        current_vals = np.array([objective(p) for p in particles])
        for i in range(n_particles):
            if current_vals[i] < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_vals[i]
                if current_vals[i] < gbest_val:
                    gbest = pbest[i].copy()
                    gbest_val = current_vals[i]
                    best_h = gbest.copy()
                    best_c5 = gbest_val
                    no_improvement = 0
            else:
                no_improvement += 1

        if no_improvement >= 50:
            break

        iter_count += 1

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, 
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
