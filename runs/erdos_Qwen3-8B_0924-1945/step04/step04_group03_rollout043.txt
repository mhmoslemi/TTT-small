```python
import numpy as np
import time
import scipy.optimize as opt

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_population = 30
    F = 0.5
    max_generations = int(budget_s / (n_population * 10))
    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    if initial_h_values is not None:
        h_init = initial_h_values
        h_init = np.array(h_init, dtype=np.float64)
        h_init = h_init * (target_sum / np.sum(h_init))
        best_h = h_init.copy()
    else:
        h_init = np.zeros(n_points)
        indices = np.random.choice(n_points, size=n_points // 2, replace=False)
        h_init[indices] = 1.0
        h_init = h_init * (target_sum / np.sum(h_init))
        best_h = h_init.copy()

    def evaluate(h):
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (target_sum / total)

    pop = [normalize(np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])) for _ in range(n_population)]
    pop = [normalize(p) for p in pop]
    pop = [p for p in pop if np.sum(p) > 0]

    start_time = time.time()
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 10:
            break
        fitness = [evaluate(p) for p in pop]
        best_idx = np.argmin(fitness)
        best_fitness = fitness[best_idx]
        if best_fitness < best_c5:
            best_c5 = best_fitness
            best_h = pop[best_idx].copy()

        indices = np.argsort(fitness)
        top20 = indices[:int(n_population * 0.2)]
        top20 = [pop[i] for i in top20]

        new_pop = []
        for _ in range(n_population // 2):
            a, b, c = np.random.choice(top20, 3, replace=False)
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = normalize(mutant)
            new_pop.append(mutant)

        pop = new_pop + pop[:n_population // 2]
        pop = [p for p in pop if np.sum(p) > 0]
        if len(pop) < n_population:
            pop += [normalize(np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])) for _ in range(n_population - len(pop))]

    best_h = normalize(best_h)
    final_c5 = evaluate(best_h)
    return best_h.tolist(), final_c5, n_points
```
