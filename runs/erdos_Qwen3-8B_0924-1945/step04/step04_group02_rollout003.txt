```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < 100:
                h[i] = 1.0
            elif 100 <= i < 200:
                h[i] = 0.5
            elif 200 <= i < 300:
                h[i] = 0.0
            else:
                h[i] = 0.5
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    n_particles = 20
    particles = np.array([np.random.uniform(0.0, 1.0, n_points) for _ in range(n_particles)])
    particles = particles * (target_sum / np.sum(particles, axis=1, keepdims=True))
    particles = np.clip(particles, 0.0, 1.0)
    personal_best = particles.copy()
    global_best = np.min(particles, axis=0)
    velocities = np.random.uniform(-0.1, 0.1, size=(n_particles, n_points))
    inertia = 0.5
    cognitive = 1.5
    social = 1.5

    for gen in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (personal_best[i] - particles[i]) + social * r2 * (global_best - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
            c5_val = objective(particles[i].flatten())
            if c5_val < objective(personal_best[i].flatten()):
                personal_best[i] = particles[i].copy()
            if c5_val < objective(global_best.flatten()):
                global_best = particles[i].copy()
        if np.min([objective(p.flatten()) for p in particles]) < best_c5:
            best_c5 = np.min([objective(p.flatten()) for p in particles])
            best_h = global_best.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
