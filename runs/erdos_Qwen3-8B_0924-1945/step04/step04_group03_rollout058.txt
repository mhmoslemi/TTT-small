import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    n_particles = 30
    particles = np.random.uniform(0, 1, (n_particles, n_points))
    particles = particles * (target_sum / np.sum(particles, axis=1)[:, None])
    p_best = particles.copy()
    g_best = particles[0].copy()
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))

    omega = 0.7
    c1 = 1.5
    c2 = 1.5

    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = omega * velocities[i] + c1 * r1 * (p_best[i] - particles[i]) + c2 * r2 * (g_best - particles[i])
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            sum_h = np.sum(particles[i])
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            particles[i] = particles[i] * scale

            current_c5 = objective(particles[i].flatten())
            if current_c5 < objective(p_best[i].flatten()):
                p_best[i] = particles[i].copy()
                if current_c5 < objective(g_best.flatten()):
                    g_best = particles[i].copy()

        if time.time() - start_time > budget_s - 5:
            break

    best_h = g_best.copy()
    best_c5 = objective(best_h.flatten())

    return best_h.tolist(), best_c5, n_points
