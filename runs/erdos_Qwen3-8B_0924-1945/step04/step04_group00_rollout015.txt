import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.cos(np.pi * x))
        h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    def evaluate(solution):
        h = solution.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = corr * dx
        return np.max(overlaps)
    
    def local_refinement(h, max_iter=10):
        for _ in range(max_iter):
            h = h.copy()
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            corr = np.correlate(h, 1.0 - h, mode="full")
            overlaps = corr * dx
            max_overlap = np.max(overlaps)
            def obj_func(x_flat):
                x = x_flat.reshape(n_points)
                x = np.clip(x, 0, 1)
                sum_x = np.sum(x)
                scale_x = target_sum / sum_x if sum_x != 0 else 1.0
                x = x * scale_x
                corr_x = np.correlate(x, 1.0 - x, mode="full")
                overlaps_x = corr_x * dx
                return np.max(overlaps_x)
            res = opt.minimize(obj_func, h.flatten(), method='L-BFGS-B', bounds=[(0,1)]*n_points, tol=1e-8)
            h = res.x.reshape(n_points)
        return h
    
    def mutate(solution, mutation_rate=0.3):
        solution = solution.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                solution[i] += np.random.normal(0, 0.1)
                solution[i] = np.clip(solution[i], 0, 1)
        sum_h = np.sum(solution)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        solution *= scale
        return solution
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        child[indices] = parent2[indices]
        sum_h = np.sum(child)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        child *= scale
        return child
    
    def tournament_selection(population, k=5):
        competitors = np.random.choice(population, size=k)
        return min(competitors, key=evaluate)
    
    population_size = 20
    population = [h.copy() for _ in range(population_size)]
    best_h = h.copy()
    best_c5 = evaluate(h)
    start_time = time.time()
    
    for gen in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        new_population = []
        for _ in range(population_size // 2):
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            child1 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = crossover(parent1, parent2)
            child2 = mutate(child2)
            new_population.append(child1)
            new_population.append(child2)
        population = new_population
        for candidate in population:
            c5 = evaluate(candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate.copy()
    
    best_h = local_refinement(best_h)
    best_c5 = evaluate(best_h)
    
    return best_h.tolist(), best_c5, n_points
