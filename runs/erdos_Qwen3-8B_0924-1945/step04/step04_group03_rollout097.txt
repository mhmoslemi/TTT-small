import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    class Particle:
        def __init__(self, pos):
            self.pos = pos.copy()
            self.pbest = pos.copy()
            self.pbest_val = objective(pos)
            self.vel = np.random.uniform(-0.05, 0.05, size=n_points)

    particles = [Particle(h.copy()) for _ in range(30)]
    gbest = particles[0].pos.copy()
    gbest_val = particles[0].pbest_val

    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        for p in particles:
            p.vel = 0.7 * p.vel + 1.5 * np.random.rand(n_points) * (p.pbest - p.pos) + 1.5 * np.random.rand(n_points) * (gbest - p.pos)
            p.pos = p.pos + p.vel
            p.pos = np.clip(p.pos, 0.0, 1.0)
            sum_h = np.sum(p.pos)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            p.pos = p.pos * scale
            curr_val = objective(p.pos)
            if curr_val < p.pbest_val:
                p.pbest = p.pos.copy()
                p.pbest_val = curr_val
        curr_gbest_val = float('inf')
        for p in particles:
            if p.pbest_val < curr_gbest_val:
                curr_gbest_val = p.pbest_val
                gbest = p.pbest.copy()
        if curr_gbest_val < gbest_val:
            gbest_val = curr_gbest_val
            best_h = gbest.copy()
        if np.random.rand() < 0.1:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            best_h += noise
            best_h = np.clip(best_h, 0.0, 1.0)
            sum_h = np.sum(best_h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            best_h = best_h * scale
            curr_val = objective(best_h)
            if curr_val < best_c5:
                best_c5 = curr_val

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
