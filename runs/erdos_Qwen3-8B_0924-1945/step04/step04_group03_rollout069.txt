import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    n_particles = 30
    particles = np.random.rand(n_particles, n_points)
    particles = particles * (target_sum / np.sum(particles, axis=1, keepdims=True))
    particles = np.clip(particles, 0.0, 1.0)
    pbest = particles.copy()
    pbest_vals = np.full(n_particles, np.inf)
    gbest = particles[0].copy()
    gbest_val = float('inf')

    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(n_particles):
            h = particles[i]
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_scaled = h * scale
            corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
            val = np.max(corr * dx)
            if val < pbest_vals[i]:
                pbest_vals[i] = val
                pbest[i] = h.copy()
        best_val = np.min(pbest_vals)
        if best_val < gbest_val:
            gbest_val = best_val
            gbest = pbest[np.argmin(pbest_vals)].copy()

        for i in range(n_particles):
            h = particles[i]
            r1, r2 = np.random.rand(2)
            vel = 0.7 * (pbest[i] - h) * r1 + 1.5 * (gbest - h) * r2
            h = h + vel
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_scaled = h * scale
            particles[i] = h_scaled.copy()

        if np.random.rand() < 0.1:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            particles += noise
            particles = np.clip(particles, 0.0, 1.0)
            particles = particles * (target_sum / np.sum(particles, axis=1, keepdims=True))

        if best_val < best_c5:
            best_c5 = best_val
            best_h = gbest.copy()

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = objective(h_opt.flatten())
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
