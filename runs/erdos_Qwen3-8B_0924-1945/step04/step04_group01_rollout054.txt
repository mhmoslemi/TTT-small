```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.sort(h)
        h = np.searchsorted(h, np.linspace(0, 1, n_points // 2 + 1)[1:])
        h = np.array([1.0 if i < n_points // 2 else 0.0 for i in range(n_points)])

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def objective(h_flat, alpha=1000):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        overlaps = corr * dx
        return (1.0 / alpha) * np.log(np.sum(np.exp(alpha * overlaps)))

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    for _ in range(100):
        res = opt.minimize(objective, h.flatten(), method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})
        h_opt = res.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5 = evaluate(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
        if time.time() - start_time > budget_s - 5:
            break
        h = h_opt.copy()
        noise = np.random.normal(0, 0.05, size=n_points)
        h = h + noise
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        res = opt.minimize(evaluate, best_h.flatten(), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 50})
        h_opt = res.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5 = evaluate(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
```
