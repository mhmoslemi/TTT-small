import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_generations = 200
    n_particles = 40
    elite_fraction = 0.05
    mutation_amp_start = 0.05
    mutation_amp_end = 0.02
    inertia_start = 0.9
    inertia_end = 0.4
    c1 = 2.0
    c2 = 2.0
    elite_count = int(n_particles * elite_fraction)

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        h = np.clip(h, 0.0, target_sum)
        h = h / np.sum(h) * target_sum

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    class Particle:
        def __init__(self, pos):
            self.pos = pos.copy()
            self.pbest = pos.copy()
            self.pbest_fitness = objective(pos)

    particles = [Particle(h.copy()) for _ in range(n_particles)]
    best_h = h.copy()
    best_c5 = objective(h)
    start_time = time.time()

    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
        inertia = inertia_start - (inertia_start - inertia_end) * gen / max_generations
        mutation_amp = mutation_amp_start - (mutation_amp_start - mutation_amp_end) * gen / max_generations

        for p in particles:
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            vel = inertia * p.vel + c1 * r1 * (p.pbest - p.pos) + c2 * r2 * (best_h - p.pos)
            p.pos += vel
            p.pos = np.clip(p.pos, 0.0, 1.0)
            sum_p = np.sum(p.pos)
            scale = target_sum / sum_p if sum_p != 0 else 1.0
            p.pos = p.pos * scale
            fitness = objective(p.pos)
            if fitness < p.pbest_fitness:
                p.pbest = p.pos.copy()
                p.pbest_fitness = fitness
        if gen % 10 == 0:
            for p in particles:
                p.pos += mutation_amp * np.random.randn(n_points)
                p.pos = np.clip(p.pos, 0.0, 1.0)
                sum_p = np.sum(p.pos)
                scale = target_sum / sum_p if sum_p != 0 else 1.0
                p.pos = p.pos * scale
                fitness = objective(p.pos)
                if fitness < p.pbest_fitness:
                    p.pbest = p.pos.copy()
                    p.pbest_fitness = fitness
        best_h = np.array([p.pbest for p in particles]).min(axis=0)
        best_c5 = objective(best_h)

    h_opt = best_h.copy()
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    result = opt.minimize(
        lambda x: objective(x),
        h_opt,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    h_opt = result.x
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt)

    return h_opt.tolist(), c5_bound, n_points
