```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.cos(np.pi * x))
        h = h * (target_sum / np.sum(h))
    def exact_objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    population = [h.copy()]
    for _ in range(100):
        if len(population) < 20:
            for _ in range(20 - len(population)):
                h_new = np.random.rand(n_points)
                h_new = h_new * (target_sum / np.sum(h_new))
                population.append(h_new)
        fit = [exact_objective(h) for h in population]
        idxs = np.argsort(fit)
        population = [population[i] for i in idxs[:15]]
        new_population = []
        for _ in range(15):
            p1, p2 = np.random.choice(range(15), 2, replace=False)
            child = population[p1].copy()
            if np.random.rand() < 0.8:
                split = np.random.randint(1, n_points - 1)
                child[:split] = population[p2][:split]
            if np.random.rand() < 0.3:
                seg = np.random.choice(n_points, 5, replace=False)
                child[seg] += np.random.normal(0, 0.05, size=5)
                child[seg] = np.clip(child[seg], 0, 1)
            if np.random.rand() < 0.2:
                seg = np.random.choice(n_points, 10, replace=False)
                child[seg] *= np.random.uniform(0.8, 1.2, size=10)
                child[seg] = np.clip(child[seg], 0, 1)
            child = np.clip(child, 0, 1)
            sum_h = np.sum(child)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            child = child * scale
            new_population.append(child)
        population += new_population
        population = np.array(population)
        fit = [exact_objective(h) for h in population]
        idxs = np.argsort(fit)
        population = [population[i] for i in idxs[:20]]
        if time.time() - start_time > budget_s - 5:
            break
    population = np.array(population)
    fit = [exact_objective(h) for h in population]
    idxs = np.argsort(fit)
    best_h = population[idxs[0]]
    if time.time() - start_time < budget_s - 5:
        for _ in range(5):
            h_opt = best_h.copy()
            h_opt = np.clip(h_opt, 0, 1)
            sum_h = np.sum(h_opt)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_opt = h_opt * scale
            result = opt.minimize(
                exact_objective, h_opt.flatten(), 
                method='L-BFGS-B', bounds=bounds, constraints=cons,
                tol=1e-8, options={'maxiter': 50}
            )
            h_opt = result.x.reshape(n_points)
            h_opt = np.clip(h_opt, 0, 1)
            sum_h_opt = np.sum(h_opt)
            scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
            h_opt = h_opt * scale
            c5_bound = exact_objective(h_opt.flatten())
            if c5_bound < exact_objective(best_h.flatten()):
                best_h = h_opt.copy()
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        h_opt = best_h.copy()
        h_opt = np.clip(h_opt, 0, 1)
        sum_h = np.sum(h_opt)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_opt = h_opt * scale
        result = opt.minimize(
            exact_objective, h_opt.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0, 1)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = exact_objective(h_opt.flatten())
        if c5_bound < exact_objective(best_h.flatten()):
            best_h = h_opt.copy()
    return best_h.tolist(), exact_objective(best_h), n_points
```
