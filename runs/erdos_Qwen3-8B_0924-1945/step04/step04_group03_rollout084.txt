```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0.0, 1.0, size=n_points)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    n_particles = 30
    particles = np.random.uniform(0.0, 1.0, size=(n_particles, n_points))
    particles = particles * (target_sum / np.sum(particles, axis=1, keepdims=True))
    pbest = particles.copy()
    pbest_fitness = np.array([objective(p) for p in pbest])
    gbest = pbest[np.argmin(pbest_fitness)]
    gbest_fitness = np.min(pbest_fitness)

    omega = 0.7
    c1 = 1.5
    c2 = 1.5
    max_iter = 500
    iter_count = 0

    while time.time() - start_time < budget_s - 5 and iter_count < max_iter:
        velocities = np.random.uniform(-0.1, 0.1, size=(n_particles, n_points))
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = omega * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
            fitness = objective(particles[i])
            if fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fitness
        if np.min(pbest_fitness) < gbest_fitness:
            gbest = pbest[np.argmin(pbest_fitness)]
            gbest_fitness = np.min(pbest_fitness)
        iter_count += 1

    best_h = gbest.copy()
    best_c5 = gbest_fitness
    return best_h.tolist(), best_c5, n_points
```
