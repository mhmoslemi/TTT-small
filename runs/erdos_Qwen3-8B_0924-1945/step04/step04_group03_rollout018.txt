import numpy as np
import time
import scipy.optimize as opt

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.sort(h)
        h = np.repeat(h, 2)
        h = np.sort(h)
        h = np.clip(h, 0, 2)
        h = h[:n_points]
        h = np.interp(h, np.linspace(0, 2, n_points), np.linspace(0, 1, n_points))
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    population_size = 30
    elite_percentage = 0.2
    mutation_factor = 0.5
    generations = 0
    no_improvement = 0
    best_fitness = float('inf')
    population = [np.random.rand(n_points) for _ in range(population_size)]

    for i in range(population_size):
        population[i] = np.clip(population[i], 0.0, 1.0)
        sum_p = np.sum(population[i])
        population[i] = population[i] * (target_sum / sum_p if sum_p != 0 else 1.0)

    while time.time() - start_time < budget_s - 5 and no_improvement < 50:
        fitness = [evaluate(ind) for ind in population]
        elite_size = int(population_size * elite_percentage)
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = [population[i] for i in elite_indices]
        new_population = []
        for _ in range(population_size - elite_size):
            a, b, c = np.random.choice(elite_indices, 3, replace=False)
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            sum_mut = np.sum(mutant)
            mutant = mutant * (target_sum / sum_mut if sum_mut != 0 else 1.0)
            new_population.append(mutant)
        population = elite + new_population
        for i in range(len(population)):
            population[i] = np.clip(population[i], 0.0, 1.0)
            sum_p = np.sum(population[i])
            population[i] = population[i] * (target_sum / sum_p if sum_p != 0 else 1.0)
        fitness = [evaluate(ind) for ind in population]
        current_best = min(fitness)
        if current_best < best_fitness:
            best_fitness = current_best
            no_improvement = 0
        else:
            no_improvement += 1
        generations += 1

    best_h = population[np.argmin(fitness)]
    best_c5 = best_fitness
    return best_h.tolist(), best_c5, n_points
