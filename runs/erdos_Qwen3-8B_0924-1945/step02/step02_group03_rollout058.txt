import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2 if np.sum(h) > 0 else np.random.uniform(0, 1, n_points)

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    bounds = [(0, 1) for _ in range(n_points)]

    def objective(x):
        return evaluate_candidate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]

    def tournament_selection(population, fitnesses, size=5):
        indices = np.random.choice(len(population), size=size, replace=False)
        best_index = np.argmin(fitnesses[indices])
        return population[indices[best_index]]

    def crossover(parent1, parent2):
        alpha = np.random.uniform(-0.5, 0.5)
        return np.clip(alpha * parent1 + (1 - alpha) * parent2, 0, 1)

    def mutate(h):
        h = h.copy()
        for i in range(len(h)):
            if np.random.rand() < 0.2:
                h[i] += np.random.uniform(-0.1, 0.1)
                h[i] = np.clip(h[i], 0, 1)
        h = h / np.sum(h) * n_points / 2
        return h

    population = [np.random.uniform(0, 1, n_points) for _ in range(50)]
    for i in range(len(population)):
        population[i] = np.clip(population[i], 0, 1)
        population[i] = population[i] / np.sum(population[i]) * n_points / 2

    fitnesses = [evaluate_candidate(h) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx]
    best_c5 = fitnesses[best_idx]

    start_time = time()
    for _ in range(500):
        if time() - start_time > budget_s:
            break
        parent1 = tournament_selection(population, fitnesses)
        parent2 = tournament_selection(population, fitnesses)
        child = crossover(parent1, parent2)
        child = mutate(child)
        child_fitness = evaluate_candidate(child)
        if child_fitness < best_c5:
            best_c5 = child_fitness
            best_h = child
        if child_fitness < np.min(fitnesses):
            idx = np.argmin(fitnesses)
            population[idx] = child
            fitnesses[idx] = child_fitness
        else:
            idx = np.argmax(fitnesses)
            population[idx] = child
            fitnesses[idx] = child_fitness

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
