import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize import basinhopping
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in kwargs:
        initial_h_values = kwargs['initial_h_values']
    else:
        initial_h_values = np.random.rand(n_points)
        initial_h_values = initial_h_values / np.sum(initial_h_values) * n_points / 2
    initial_h_values = np.clip(initial_h_values, 0, 1)
    
    def normalize(h):
        h = np.clip(h, 0, 1)
        return h * (n_points / 2) / np.sum(h)
    
    def objective(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        return objective(normalize(h))
    
    def perturb(h, step_size=0.2):
        perturbation = np.random.uniform(-step_size, step_size, size=n_points)
        new_h = h + perturbation
        return normalize(new_h)
    
    def target_func(x):
        return evaluate_candidate(x)
    
    def accept_func(x, target_x, target_score, score):
        return True
    
    start_time = time.time()
    current_h = normalize(initial_h_values)
    best_h = current_h.copy()
    best_c5 = evaluate_candidate(current_h)
    
    # Basin-hopping with adaptive cooling
    temperature = 1.0
    perturbation = 0.2
    max_iter = 200
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Perturb and evaluate
        new_h = perturb(current_h, perturbation)
        current_c5 = evaluate_candidate(new_h)
        if current_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = current_c5
        # Local optimization
        res = minimize(objective, new_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-5)
        if res.success:
            current_h = res.x
            current_c5 = evaluate_candidate(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        # Cooling
        temperature *= 0.99
        perturbation *= 0.99
    
    best_h = normalize(best_h)
    return (best_h, best_c5, n_points)
