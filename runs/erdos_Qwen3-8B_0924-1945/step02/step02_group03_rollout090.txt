import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else None

    def validate_h(h):
        return np.sum(np.clip(h, 0, 1)) == n_points / 2

    def construct_initial_h():
        if initial_h_values is not None:
            h = initial_h_values.copy()
            if not validate_h(h):
                h = np.random.uniform(0, 1, n_points)
                h = h / np.sum(h) * n_points / 2
            return h
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
        return h

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        c5 = compute_c5(h)
        return c5

    best_h = construct_initial_h()
    best_c5 = evaluate_candidate(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time()

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    result = differential_evolution(objective, bounds, strategy='best1bin', popsize=15, tol=1e-5, mutation=(0.5, 1), recombination=0.8, maxiter=500, seed=seed, polish=True, disp=False, workers=-1)

    if time() - start_time < budget_s:
        result = differential_evolution(objective, bounds, strategy='best1bin', popsize=15, tol=1e-5, mutation=(0.5, 1), recombination=0.8, maxiter=500, seed=seed, polish=True, disp=False, workers=-1)

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    best_c5 = evaluate_candidate(best_h)

    return (best_h, best_c5, n_points)
