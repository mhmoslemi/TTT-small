import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else None
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * n_points / 2
    h = np.clip(h, 0, 1)

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for i in range(100):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            current_h = res.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            break
    if time.time() - start_time < budget_s:
        for _ in range(50):
            if time.time() - start_time > budget_s:
                break
            perturbation = np.random.uniform(-0.1, 0.1, n_points)
            h_perturbed = np.clip(best_h + perturbation, 0, 1)
            h_perturbed = h_perturbed / np.sum(h_perturbed) * n_points / 2
            current_c5 = compute_c5(h_perturbed)
            if current_c5 < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = current_c5
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    return (best_h, best_c5, n_points)
