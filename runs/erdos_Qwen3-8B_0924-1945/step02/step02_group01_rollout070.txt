```python
import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr * dx)
        return max_corr

    def clip_and_rescale(h):
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        return h * scale

    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.normal(0, 0.1)
        return clip_and_rescale(h)

    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = parent2[i]
        return clip_and_rescale(child)

    def tournament_selection(population, size=5):
        contenders = np.random.choice(population, size=size)
        return min(contenders, key=evaluate)

    population_size = 50
    population = [clip_and_rescale(np.random.choice([0, 1], n_points, p=[0.5, 0.5])) for _ in range(population_size)]
    best_h = clip_and_rescale(h)
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        parents = [tournament_selection(population) for _ in range(2)]
        offspring = crossover(parents[0], parents[1])
        offspring = mutate(offspring)
        population.append(offspring)
        if len(population) > population_size:
            population.pop(np.argmax([evaluate(h) for h in population]))
        current_h = population[0]
        current_c5 = evaluate(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    best_h = clip_and_rescale(best_h)
    best_c5 = evaluate(best_h)
    return best_h.tolist(), best_c5, n_points
```
