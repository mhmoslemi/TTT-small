import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if np.random.rand() < 0.5 else 0
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()

    def run_optimization():
        best_h = h.copy()
        best_c5 = evaluate_candidate(best_h)
        population = [best_h.copy()]
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            if len(population) < 6:
                if np.random.rand() < 0.5:
                    h_pattern = np.zeros(n_points)
                    for i in range(n_points):
                        if np.random.rand() < 0.5:
                            h_pattern[i] = 1
                    h_pattern = h_pattern / np.sum(h_pattern) * n_points / 2
                    population.append(h_pattern)
                else:
                    h_pattern = np.zeros(n_points)
                    m = np.random.randint(3, 8)
                    interval_length = 2 / m
                    selected_intervals = np.random.choice(m, size=np.random.randint(2, m-1), replace=False)
                    for i in range(n_points):
                        x_i = i * 2 / n_points
                        if np.floor(x_i / interval_length) in selected_intervals:
                            h_pattern[i] = 1
                    h_pattern = h_pattern / np.sum(h_pattern) * n_points / 2
                    population.append(h_pattern)
            candidates = population[:]
            for candidate in candidates:
                x0 = candidate.copy()
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
                if res.success:
                    current_h = res.x
                    current_c5 = evaluate_candidate(current_h)
                    if current_c5 < best_c5:
                        best_h = current_h.copy()
                        best_c5 = current_c5
                    if current_c5 < evaluate_candidate(candidate):
                        population.append(current_h.copy())
                        population = population[:6]
            population = list(set(population))
        return best_h, best_c5

    best_h, best_c5 = run_optimization()

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
