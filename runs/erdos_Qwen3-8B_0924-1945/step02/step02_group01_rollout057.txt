```python
import numpy as np
import time
import random

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))
    
    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    def mutate(h):
        h = h.copy()
        indices = np.random.choice(n_points, int(n_points * 0.1), replace=False)
        h[indices] += np.random.normal(0, 0.1, size=len(indices))
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        return h
    
    def crossover(h1, h2):
        h = h1.copy()
        indices = np.random.choice(n_points, n_points // 2, replace=False)
        h[indices] = h2[indices]
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        return h
    
    def select_parents(population):
        candidates = np.random.choice(len(population), 5, replace=False)
        fitness = np.array([evaluate(ind) for ind in population[candidates]])
        best_idx = candidates[np.argmin(fitness)]
        return population[best_idx].copy()
    
    population = [np.random.rand(n_points) for _ in range(50)]
    for i in range(len(population)):
        population[i] = np.clip(population[i], 0.0, 1.0)
        sum_h = np.sum(population[i])
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        population[i] = population[i] * scale
    
    best_h = population[0].copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        new_population = []
        for _ in range(50):
            parent1 = select_parents(population)
            parent2 = select_parents(population)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
        for ind in population:
            sum_h = np.sum(ind)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            ind[:] = ind * scale
        
        for ind in population:
            c5 = evaluate(ind)
            if c5 < best_c5:
                best_c5 = c5
                best_h = ind.copy()
    
    return best_h.tolist(), best_c5, n_points
```
