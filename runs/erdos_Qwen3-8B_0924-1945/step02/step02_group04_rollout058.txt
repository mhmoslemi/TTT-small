import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize import basinhopping
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    h = np.clip(initial_h_values, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)

    def compute_c5(h, n_points):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return compute_c5(h, n_points)

    def normalize(h):
        h = np.clip(h, 0, 1)
        return h * (n_points / 2) / np.sum(h)

    def perturb(h, stepsize=0.1):
        perturbation = np.random.uniform(-stepsize, stepsize, size=len(h))
        new_h = h + perturbation
        return normalize(new_h)

    def basinhopping_callback(x, fval, convergence):
        nonlocal best_h, best_c5
        if fval < best_c5:
            best_h = x.copy()
            best_c5 = fval

    start_time = time.time()
    if time.time() - start_time < budget_s:
        result = basinhopping(objective, best_h, niter=100, T=1.0, stepsize=0.1, minimizer_kwargs={"method": "L-BFGS-B", "bounds": [(0, 1)]*n_points, "tol": 1e-5}, callback=basinhopping_callback)
        best_h = normalize(result.x)
        best_c5 = compute_c5(best_h, n_points)
    if time.time() - start_time < budget_s:
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = normalize(h)
            current_c5 = compute_c5(h, n_points)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    best_h = normalize(best_h)
    return (best_h, best_c5, n_points)
