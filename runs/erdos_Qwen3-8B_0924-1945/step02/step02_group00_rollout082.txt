```python
import numpy as np
import time
import scipy.optimize as opt
import random

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    h_flat = h.flatten()

    def fitness(x):
        return objective(x)

    def evolve(pop):
        new_pop = []
        for _ in range(len(pop)):
            a, b, c = random.sample(pop, 3)
            p = a + np.random.uniform(-0.1, 0.1, len(a))
            p = np.clip(p, 0, 1)
            p = p * (target_sum / np.sum(p)) if np.sum(p) != 0 else p
            new_pop.append(p)
        return new_pop

    pop = [h_flat + np.random.uniform(-0.1, 0.1, n_points) for _ in range(50)]
    pop = [np.clip(p, 0, 1) * (target_sum / np.sum(np.clip(p, 1e-6, 1))) if np.sum(np.clip(p, 1e-6, 1)) != 0 else p for p in pop]

    best = min(pop, key=objective)
    best_fitness = objective(best)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        new_pop = evolve(pop)
        for p in new_pop:
            p = np.clip(p, 0, 1)
            p = p * (target_sum / np.sum(p)) if np.sum(p) != 0 else p
            fitness_p = objective(p)
            if fitness_p < best_fitness:
                best = p
                best_fitness = fitness_p
        pop = new_pop
        if time.time() - start_time > budget_s - 10:
            break

    result = opt.minimize(objective, best, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 200})
    h_opt = result.x
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = objective(h_opt)
    return h_opt.tolist(), c5_bound, n_points
```
