import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum
    
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    
    # Stage 1: Optimization with L-BFGS-B
    result = opt.minimize(objective, h.flatten(), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 300})
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    
    # Stage 2: SLSQP refinement
    result = opt.minimize(objective, h_opt.flatten(), method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 200})
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    
    # Stage 3: Differential Evolution for global search
    def de_objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.differential_evolution(de_objective, bounds, strategy='best1bin', maxiter=100, popsize=50, tol=1e-8, mutation=(0.5, 1), recombination=0.8, disp=False)
    h_de = result.x
    h_de = np.clip(h_de, 0.0, 1.0)
    sum_h_de = np.sum(h_de)
    scale = target_sum / sum_h_de if sum_h_de != 0 else 1.0
    h_de = h_de * scale
    
    # Final evaluation
    h_best = h_opt if objective(h_opt.flatten()) < objective(h_de) else h_de
    h_best = np.clip(h_best, 0.0, 1.0)
    sum_h_best = np.sum(h_best)
    scale = target_sum / sum_h_best if sum_h_best != 0 else 1.0
    h_best = h_best * scale
    
    c5_bound = objective(h_best.flatten())
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        # Final refinement with SLSQP
        result = opt.minimize(objective, h_best.flatten(), method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})
        h_best = result.x.reshape(n_points)
        h_best = np.clip(h_best, 0.0, 1.0)
        sum_h_best = np.sum(h_best)
        scale = target_sum / sum_h_best if sum_h_best != 0 else 1.0
        h_best = h_best * scale
        c5_bound = objective(h_best.flatten())
    
    return h_best.tolist(), c5_bound, n_points
