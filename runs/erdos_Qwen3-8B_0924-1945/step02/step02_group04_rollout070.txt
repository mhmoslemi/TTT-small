import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize import basinhopping
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    h = np.clip(initial_h_values, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)

    def compute_c5(h, n_points):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return compute_c5(h, n_points)

    def normalize(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def perturb(h, magnitude=0.2):
        return np.clip(h + np.random.uniform(-magnitude, magnitude, size=len(h)), 0, 1)

    def evaluate(h):
        h = normalize(h)
        return objective(h)

    start_time = time.time()
    best_h = normalize(h)
    best_c5 = compute_c5(best_h, n_points)

    def minimize_with_check(x, *args):
        res = minimize(evaluate, x, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-5)
        if res.success:
            return res.x
        return x

    def basinhopping_callback(x, fval, convergence):
        nonlocal best_h, best_c5
        current_h = normalize(x)
        current_c5 = compute_c5(current_h, n_points)
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5

    max_iter = 100
    step_size = 0.1
    temperature = 1.0

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        perturbed = perturb(best_h, step_size)
        res = minimize_with_check(perturbed)
        if np.allclose(res, perturbed, atol=1e-5):
            perturbed = perturb(best_h, step_size * 2)
            res = minimize_with_check(perturbed)
        if res is not None:
            current_h = normalize(res)
            current_c5 = compute_c5(current_h, n_points)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        temperature *= 0.99

    best_h = normalize(best_h)

    return (best_h, best_c5, n_points)
