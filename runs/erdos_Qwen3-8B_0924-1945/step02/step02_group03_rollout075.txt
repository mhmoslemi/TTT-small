import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    def differential_evolution(population_size, max_iter, mutation_factor, crossover_prob):
        nonlocal best_h, best_c5
        population = [np.clip(np.random.rand(n_points), 0, 1) for _ in range(population_size)]
        population = [h / np.sum(h) * n_points / 2 for h in population]
        population = [np.clip(h, 0, 1) for h in population]
        population = [h for h in population if np.abs(np.sum(h) - n_points / 2) < 1e-6]
        population = population[:population_size]
        for _ in range(max_iter):
            if time() - start_time > budget_s:
                break
            mutants = []
            for i in range(population_size):
                a, b, c = np.random.choice(population_size, 3, replace=False)
                mutant = population[a] + mutation_factor * (population[b] - population[c])
                mutant = np.clip(mutant, 0, 1)
                mutant = mutant / np.sum(mutant) * n_points / 2
                mutants.append(mutant)
            trial_population = []
            for i in range(population_size):
                trial = population[i].copy()
                mutant = mutants[i].copy()
                for j in range(n_points):
                    if np.random.rand() < crossover_prob:
                        trial[j] = mutant[j]
                trial = np.clip(trial, 0, 1)
                trial = trial / np.sum(trial) * n_points / 2
                trial_population.append(trial)
            for trial, current in zip(trial_population, population):
                trial_c5 = compute_c5(trial)
                current_c5 = compute_c5(current)
                if trial_c5 < current_c5:
                    population[i] = trial
            for candidate in population:
                candidate_c5 = compute_c5(candidate)
                if candidate_c5 < best_c5:
                    best_h = candidate.copy()
                    best_c5 = candidate_c5
        return best_h, best_c5

    population_size = 30
    max_iter = 300
    mutation_factor = 0.5
    crossover_prob = 0.8
    h, best_c5 = differential_evolution(population_size, max_iter, mutation_factor, crossover_prob)

    for _ in range(100):
        if time() - start_time > budget_s:
            break
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        current_c5 = evaluate_candidate(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
