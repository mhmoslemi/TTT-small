import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if np.random.rand() < 0.5 else 0
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    for _ in range(500):
        if time() - start_time > budget_s:
            break
        idx = np.random.randint(n_points)
        h_new = best_h.copy()
        h_new[idx] += np.random.uniform(-0.2, 0.2)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * n_points / 2
        c5_new = evaluate_candidate(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    for _ in range(50):
        if time() - start_time > budget_s:
            break
        h_new = np.random.uniform(0, 1, n_points)
        h_new = h_new / np.sum(h_new) * n_points / 2
        c5_new = evaluate_candidate(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
