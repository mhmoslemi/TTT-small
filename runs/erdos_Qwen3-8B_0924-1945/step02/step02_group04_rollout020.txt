import numpy as np
from scipy.optimize import basinhopping, minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def objective(h):
        return evaluate_candidate(h)
    
    def normalize(h):
        h = np.clip(h, 0, 1)
        return h * (n_points / 2) / np.sum(h)
    
    def perturb(h, step_size):
        perturb = np.random.uniform(-step_size, step_size, size=n_points)
        return np.clip(h + perturb, 0, 1)
    
    def target_func(x):
        return objective(x)
    
    def minimize_func(x):
        res = minimize(objective, x, method='L-BFGS-B', bounds=[(0,1)]*n_points, tol=1e-5)
        return res.x
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturbed = perturb(best_h, 0.15)
        perturbed = normalize(perturbed)
        res = minimize(objective, perturbed, method='L-BFGS-B', bounds=[(0,1)]*n_points, tol=1e-5)
        if res.success:
            current_h = res.x
            current_c5 = evaluate_candidate(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
    
    best_h = normalize(best_h)
    return (best_h, best_c5, n_points)
