```python
import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    target_sum = n_points / 2
    pop_size = 50
    elite_ratio = 0.2
    mutation_rate = 0.1
    crossover_prob = 0.8
    max_generations = 400
    early_stop = 20

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        n_ones = n_points // 2
        indices = np.random.choice(n_points, n_ones, replace=False)
        h[indices] = 1.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr * dx)
        return max_corr

    def rescale(h):
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        scale = target_sum / sum_h
        return h * scale

    population = [np.zeros(n_points) for _ in range(pop_size)]
    for i in range(pop_size):
        n_ones = n_points // 2
        indices = np.random.choice(n_points, n_ones, replace=False)
        population[i][indices] = 1.0
        population[i] = rescale(population[i])
    
    best_h = rescale(h)
    best_c5 = evaluate(best_h)

    start_time = time.time()
    generation = 0
    no_improvement = 0
    while generation < max_generations and time.time() - start_time < budget_s:
        generation += 1
        new_population = []

        for _ in range(pop_size):
            parent1, parent2 = np.random.choice(population, size=2, replace=False)
            if np.random.rand() < crossover_prob:
                child = np.concatenate((parent1[:n_points//2], parent2[n_points//2:]))
            else:
                child = np.copy(parent1)
            
            if np.random.rand() < mutation_rate:
                mutate_idx = np.random.choice(n_points, size=int(n_points * mutation_rate), replace=False)
                child[mutate_idx] += np.random.normal(0, 0.1, size=len(mutate_idx))
                child = rescale(child)
            
            new_population.append(rescale(child))
        
        population = new_population
        population.sort(key=evaluate)
        population = population[:int(pop_size * elite_ratio)]

        current_c5 = evaluate(population[0])
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = rescale(population[0])
            no_improvement = 0
        else:
            no_improvement += 1
        
        if no_improvement >= early_stop:
            break

    best_h = rescale(best_h)
    final_c5 = evaluate(best_h)
    return best_h.tolist(), final_c5, n_points
```
