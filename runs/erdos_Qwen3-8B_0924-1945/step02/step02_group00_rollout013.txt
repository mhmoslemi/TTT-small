import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        h[::5] = 1.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    start_time = time.time()
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    # Initial population using differential evolution
    pop_size = 100
    population = np.random.rand(pop_size, n_points)
    population = np.clip(population, 0.0, 1.0)
    population = population * (target_sum / np.mean(np.sum(population, axis=1)))

    def de_eval(individual):
        return objective(individual)

    best_h = h.copy()
    best_c5 = objective(h)

    # Differential evolution
    for _ in range(500):
        if time.time() - start_time > budget_s - 100:
            break

        mutant = population[np.random.randint(pop_size)]
        mutant += np.random.normal(0, 0.05, n_points)
        mutant = np.clip(mutant, 0.0, 1.0)
        mutant = mutant * (target_sum / np.sum(mutant))

        cross = np.random.rand(n_points) < 0.9
        trial = np.where(cross, mutant, population[np.random.randint(pop_size)])
        trial = np.clip(trial, 0.0, 1.0)
        trial = trial * (target_sum / np.sum(trial))

        trial_c5 = objective(trial)
        if trial_c5 < best_c5:
            best_c5 = trial_c5
            best_h = trial.copy()
        population = np.vstack([population, trial])

    # Refine with SLSQP
    result = opt.minimize(objective, best_h.flatten(), method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 300})
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale

    c5_bound = objective(h_opt.flatten())
    return h_opt.tolist(), c5_bound, n_points
