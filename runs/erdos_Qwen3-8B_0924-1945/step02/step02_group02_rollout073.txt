import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        h = 0.5 + 0.25 * np.sin(np.pi * x_grid) + 0.25 * np.cos(2 * np.pi * x_grid)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        h = h * (target_sum / sum_h)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()
    result = opt.differential_evolution(
        objective, bounds, constraints=cons, 
        strategy='best1bin', popsize=20, 
        maxiter=100, tol=1e-5, 
        mutation=(0.5, 1.0), recombination=0.9, 
        seed=seed, disp=False
    )
    h_opt = result.x
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    if sum_h_opt == 0:
        h_opt = np.full(n_points, 0.5)
    else:
        h_opt = h_opt * (target_sum / sum_h_opt)
    c5_bound = objective(h_opt)
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt.copy()

    if time.time() - start_time < budget_s - 100:
        result = opt.minimize(
            objective, best_h, 
            method='SLSQP', bounds=bounds, 
            constraints=cons, tol=1e-8, 
            options={'maxiter': 100}
        )
        h_opt = result.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        if sum_h_opt == 0:
            h_opt = np.full(n_points, 0.5)
        else:
            h_opt = h_opt * (target_sum / sum_h_opt)
        c5_bound = objective(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return best_h.tolist(), best_c5, n_points
