import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def rescale(h):
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        return h * scale

    population_size = 50
    population = np.zeros((population_size, n_points), dtype=np.float64)
    for i in range(population_size):
        indices = np.random.choice(n_points, size=int(n_points // 2), replace=False)
        population[i, indices] = 1.0
        population[i] = rescale(population[i])

    best_h = population[0].copy()
    best_c5 = evaluate(best_h.flatten())

    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s - 100:
            break
        parents = []
        for _ in range(2):
            candidates = np.random.choice(population_size, size=5, replace=False)
            best = candidates[0]
            for idx in candidates[1:]:
                if evaluate(population[idx].flatten()) < evaluate(population[best].flatten()):
                    best = idx
            parents.append(population[best].copy())
        offspring = np.zeros((population_size, n_points), dtype=np.float64)
        for i in range(population_size):
            if np.random.rand() < 0.8:
                p1, p2 = parents[0], parents[1]
                crossover = np.random.rand(n_points) < 0.5
                offspring[i] = np.where(crossover, p1, p2)
            else:
                offspring[i] = population[np.random.choice(population_size)].copy()
            mutation_mask = np.random.rand(n_points) < 0.1
            offspring[i][mutation_mask] += np.random.normal(0, 0.1, np.sum(mutation_mask))
            offspring[i] = rescale(offspring[i])
        population = np.vstack([population, offspring])
        population = population[np.argsort([evaluate(p.flatten()) for p in population])[:population_size]]
        current_c5 = evaluate(population[0].flatten())
        if current_c5 < best_c5:
            best_h = population[0].copy()
            best_c5 = current_c5
    best_h = rescale(best_h)
    return best_h.tolist(), best_c5, n_points
