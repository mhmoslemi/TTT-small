import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    
    def normalize(h):
        h = np.clip(h, 0, 1)
        return h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        h = normalize(h)
        return compute_c5(h)
    
    def objective(h):
        return evaluate_candidate(h)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def structured_initializations():
        if 'initial_h_values' in kwargs:
            yield normalize(initial_h_values)
        # Uniform intervals
        for k in range(3, 8):
            h = np.zeros(n_points)
            for i in range(k):
                h[np.random.choice(n_points, size=1, replace=False)] = 1
            h = normalize(h)
            yield h
        # Symmetric bilinear
        for slope in [-0.5, 0.5]:
            h = 0.5 * (1 + np.tanh(slope * (np.linspace(0, 2, n_points) - 1)))
            h = normalize(h)
            yield h
        # Random binary
        h = np.random.choice([0, 1], size=n_points)
        h = normalize(h)
        yield h
        # Lanczos
        h = np.zeros(n_points)
        h[5:-5] = 1
        h = normalize(h)
        yield h
    
    population = list(structured_initializations())
    population = [np.random.permutation(p) for p in population]
    
    start_time = time()
    
    def de_optimization():
        nonlocal population
        best_h = min(population, key=evaluate_candidate)
        best_c5 = evaluate_candidate(best_h)
        for gen in range(300):
            if time() - start_time > budget_s:
                break
            candidates = np.random.choice(population, size=5, replace=False)
            for h1, h2, h3 in zip(*[np.random.permutation(candidates) for _ in range(3)]):
                trial = h1 + 0.8 * (h2 - h3)
                trial = normalize(trial)
                trial_c5 = evaluate_candidate(trial)
                if trial_c5 < evaluate_candidate(population[-1]):
                    population[-1] = trial
            current_best = min(population, key=evaluate_candidate)
            if evaluate_candidate(current_best) < best_c5:
                best_h = current_best
                best_c5 = evaluate_candidate(best_h)
        return best_h, best_c5
    
    best_h, best_c5 = de_optimization()
    
    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = normalize(h)
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5
    
    if time() - start_time < budget_s:
        for _ in range(50):
            if time() - start_time > budget_s:
                break
            # Refinement phase
            h = best_h.copy()
            for _ in range(50):
                if time() - start_time > budget_s:
                    break
                h = normalize(h + np.random.normal(0, 0.05, n_points))
                h = np.clip(h, 0, 1)
                h = normalize(h)
                current_c5 = evaluate_candidate(h)
                if current_c5 < best_c5:
                    best_h = h
                    best_c5 = current_c5
    
    best_h = normalize(best_h)
    final_c5 = evaluate_candidate(best_h)
    return (best_h, final_c5, n_points)
