import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize import basinhopping
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def objective(h):
        return compute_c5(h)

    def evaluate_candidate(h):
        h = normalize(h)
        return compute_c5(h)

    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    start_time = time.time()

    def callback(x):
        nonlocal best_h, best_c5
        current_c5 = evaluate_candidate(x)
        if current_c5 < best_c5:
            best_h = x.copy()
            best_c5 = current_c5

    # Basin-hopping optimization
    def perturbation_func(x):
        return np.random.uniform(-0.2, 0.2, size=len(x))

    def minimize_with_normalization(x):
        x = normalize(x)
        result = minimize(objective, x, method='L-BFGS-B', bounds=[(0, 1)] * n_points, tol=1e-5)
        if result.success:
            return result.x
        return x

    # Run basinhopping
    bh_result = basinhopping(
        minimize_with_normalization,
        best_h,
        niter=150,
        T=1.0,
        stepsize=0.1,
        minimization_method='L-BFGS-B',
        callback=callback,
        disp=False,
        interval=10
    )

    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Fallback random search
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            h = np.random.rand(n_points)
            h = normalize(h)
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = normalize(best_h)
    return (best_h, best_c5, n_points)
