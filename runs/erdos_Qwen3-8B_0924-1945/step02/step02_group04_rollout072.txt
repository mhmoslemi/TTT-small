import numpy as np
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', None)
    
    if initial_h_values is not None:
        h = np.clip(initial_h_values, 0, 1)
    else:
        h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return compute_c5(h)
    
    bounds = [(0, 1) for _ in range(n_points)]
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    start_time = time.time()
    
    swarm_size = 30
    particles = [np.random.uniform(0, 1, n_points) for _ in range(swarm_size)]
    particles = [p * (n_points / 2) / np.sum(p) for p in particles]
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    pbest = particles.copy()
    gbest = min(particles, key=lambda x: evaluate_candidate(x))
    best_c5 = evaluate_candidate(gbest)
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = 0.8 * velocities[i] + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            current_c5 = evaluate_candidate(particles[i])
            if current_c5 < evaluate_candidate(pbest[i]):
                pbest[i] = particles[i].copy()
            if current_c5 < best_c5:
                gbest = particles[i].copy()
                best_c5 = current_c5
    
    best_h = gbest.copy()
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    return (best_h, best_c5, n_points)
