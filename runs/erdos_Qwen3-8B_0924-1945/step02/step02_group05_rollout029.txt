import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    for _ in range(1000):
        if time() - start_time > budget_s:
            break
        corr_vals = np.correlate(best_h, 1 - best_h, mode="full") * 2.0 / n_points
        max_idx = np.argmax(corr_vals)
        k = max_idx
        x_range = np.arange(max(0, k - 10), min(n_points, k + 10))
        grad = np.zeros(n_points)
        for x in x_range:
            if x + k < n_points:
                grad[x] += (1 - best_h[x + k])
            if x < n_points - k:
                grad[x + k] -= best_h[x]
        step_size = 0.01
        h_update = np.clip(best_h - step_size * grad, 0, 1)
        h_update = h_update / np.sum(h_update) * n_points / 2
        current_c5 = evaluate_candidate(h_update)
        if current_c5 < best_c5:
            best_h = h_update
            best_c5 = current_c5
        else:
            break

    for _ in range(100):
        if time() - start_time > budget_s:
            break
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
        current_c5 = evaluate_candidate(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
