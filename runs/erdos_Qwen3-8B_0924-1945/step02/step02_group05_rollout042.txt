import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    def gradient_descent(h, max_iter=1000, tol=1e-5):
        for _ in range(max_iter):
            corr_vals = np.correlate(h, 1 - h, mode="full") * 2.0 / n_points
            max_corr = np.max(corr_vals)
            if max_corr < tol:
                break
            k = np.argmax(corr_vals)
            x_range = np.arange(n_points - k)
            g = np.zeros(n_points)
            for x in x_range:
                if h[x] > 0:
                    g[x] += (1 - h[x + k]) if x + k < n_points else 0
                if h[x + k] > 0 and x + k < n_points:
                    g[x + k] += -h[x]
            step = 0.01
            h -= g * step
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * n_points / 2
        return h

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    for _ in range(100):
        if time() - start_time > budget_s:
            break
        best_h = gradient_descent(best_h, max_iter=500)
        current_c5 = evaluate_candidate(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
        else:
            break

    if time() - start_time < budget_s:
        for _ in range(50):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
