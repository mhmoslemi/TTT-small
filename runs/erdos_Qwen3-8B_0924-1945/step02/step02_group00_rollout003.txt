```python
import numpy as np
import time
import random
from deap import base, creator, tools, algorithms

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 5 == 0:
                h[i] = 1.0
        h = h * (target_sum / np.sum(h))

    def compute_c5(h_vec):
        h = np.clip(h_vec, 0.0, 1.0)
        h_sum = np.sum(h)
        scale = target_sum / h_sum if h_sum != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def evaluate(individual):
        return compute_c5(individual), 

    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)

    def mut_gaussian(individual, indpb=0.1):
        for i in range(len(individual)):
            if random.random() < indpb:
                individual[i] += np.random.normal(0, 0.1)
                individual[i] = np.clip(individual[i], 0.0, 1.0)
        return individual,

    def cx_blend(ind1, ind2):
        alpha = 0.5
        for i in range(len(ind1)):
            ind1[i], ind2[i] = (
                ind1[i] + alpha * (ind2[i] - ind1[i]),
                ind2[i] + alpha * (ind1[i] - ind2[i])
            )
            ind1[i] = np.clip(ind1[i], 0.0, 1.0)
            ind2[i] = np.clip(ind2[i], 0.0, 1.0)
        return ind1, ind2

    def create_initial_population():
        pop = []
        for _ in range(50):
            ind = np.random.rand(n_points)
            ind = ind * (target_sum / np.sum(ind))
            ind = np.clip(ind, 0.0, 1.0)
            pop.append(ind)
        return pop

    def refine_individual(ind):
        ind = np.clip(ind, 0.0, 1.0)
        h_sum = np.sum(ind)
        scale = target_sum / h_sum if h_sum != 0 else 1.0
        ind = ind * scale
        return ind

    start_time = time.time()

    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.rand, n_points)
    toolbox.register("individual", lambda: refine_individual(toolbox.attr_float()))
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    toolbox.register("mate", cx_blend)
    toolbox.register("mutate", mut_gaussian)
    toolbox.register("select", tools.selTournament, tournsize=5)
    toolbox.register("evaluate", evaluate)

    population = create_initial_population()
    for _ in range(100):
        offspring = algorithms.varAnd(population, toolbox, cxpb=0.5, mutpb=0.1)
        fits = toolbox.map(toolbox.evaluate, offspring)
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = fit
        population = tools.selBest(population + offspring, len(population))

    best_individual = min(population, key=lambda x: x.fitness.values[0])
    best_individual = refine_individual(best_individual)

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(150):
            offspring = algorithms.varAnd(population, toolbox, cxpb=0.5, mutpb=0.1)
            fits = toolbox.map(toolbox.evaluate, offspring)
            for fit, ind in zip(fits, offspring):
                ind.fitness.values = fit
            population = tools.selBest(population + offspring, len(population))
            best_individual = min(population, key=lambda x: x.fitness.values[0])
            best_individual = refine_individual(best_individual)

    best_c5 = compute_c5(best_individual)
    return best_individual.tolist(), best_c5, n_points
```
