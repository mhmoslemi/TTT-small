```python
import numpy as np
import scipy.optimize as opt
import time
import random
from deap import base, creator, tools, algorithms

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    def eval_func(individual):
        h = np.array(individual, dtype=np.float64)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx),

    def mutate(individual, indpb):
        for i in range(len(individual)):
            if random.random() < indpb:
                individual[i] = 1 - individual[i]
        return individual,

    def cx_two_point(ind1, ind2):
        size = len(ind1)
        cx1, cx2 = list(ind1), list(ind2)
        p1, p2 = sorted(random.sample(range(size), 2))
        cx1[p1:p2], cx2[p1:p2] = cx2[p1:p2], cx1[p1:p2]
        return cx1, cx2

    def generate_initial_population(pop_size):
        pop = []
        for _ in range(pop_size):
            ind = np.random.choice([0, 1], size=n_points)
            total = np.sum(ind)
            if total < 0.5 * n_points:
                diff = 0.5 * n_points - total
                ind[random.sample(range(n_points), int(diff))] = 1
            elif total > 0.5 * n_points:
                diff = total - 0.5 * n_points
                ind[random.sample(range(n_points), int(diff))] = 0
            pop.append(ind)
        return pop

    start_time = time.time()
    result = opt.minimize(objective, h.flatten(), method='L-BFGS-B', bounds=[(0.0, 1.0)]*n_points, constraints=[{'type': 'eq', 'fun': constraint}], tol=1e-8, options={'maxiter': 500})
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
        creator.create("Individual", list, fitness=creator.FitnessMin)

        toolbox = base.Toolbox()
        toolbox.register("evaluate", eval_func)
        toolbox.register("mate", cx_two_point)
        toolbox.register("mutate", mutate, indpb=0.1)
        toolbox.register("select", tools.selTournament, tournsize=3)
        toolbox.register("population", generate_initial_population)
        toolbox.register("individual", lambda: np.random.choice([0, 1], size=n_points))
        population = toolbox.population(n=50)
        stats = tools.Statistics(lambda ind: ind.fitness.values)
        stats.register("avg", np.mean)
        stats.register("std", np.std)
        stats.register("min", np.min)
        stats.register("max", np.max)

        algorithms.eaSimple(population, toolbox, cxpb=0.5, mutpb=0.1, ngen=100, stats=stats, halloffame=None, verbose=False)
        h_opt = np.array([ind for ind in population if ind.fitness.values[0] == min([ind.fitness.values[0] for ind in population])][0])
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale

        result = opt.minimize(objective, h_opt.flatten(), method='SLSQP', bounds=[(0.0, 1.0)]*n_points, constraints=[{'type': 'eq', 'fun': constraint}], tol=1e-8, options={'maxiter': 300})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale

    c5_bound = objective(h_opt.flatten())
    return h_opt.tolist(), c5_bound, n_points
```
