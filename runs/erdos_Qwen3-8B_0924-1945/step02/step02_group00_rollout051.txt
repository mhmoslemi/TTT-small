import numpy as np
import scipy.optimize as opt
import time
import random

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))
    
    def compute_c5(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.inf
        scale = target_sum / sum_h
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum
    
    def generate_initial_pattern():
        h = np.zeros(n_points, dtype=np.float64)
        idxs = np.random.choice(n_points, size=int(n_points / 2), replace=False)
        h[idxs] = 1.0
        return h
    
    def evolve_pattern(h, max_iter=500, pop_size=50, elite_size=5):
        population = [h.copy() for _ in range(pop_size)]
        for _ in range(max_iter):
            if np.random.rand() < 0.2:
                population = [np.random.choice([0, 1], size=n_points, p=[0.5, 0.5]) for _ in range(pop_size)]
            else:
                offspring = []
                for _ in range(pop_size):
                    parent1, parent2 = np.random.choice(population, size=2, replace=False)
                    child = parent1.copy()
                    for i in range(n_points):
                        if np.random.rand() < 0.5:
                            child[i] = parent2[i]
                    offspring.append(child)
                population += offspring
                population = np.array(population)
                population = population[np.argsort([compute_c5(p) for p in population])[:pop_size]]
            best = population[0]
            if compute_c5(best) < compute_c5(h):
                h = best
        return h
    
    def refine_pattern(h):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result = opt.minimize(lambda x: compute_c5(x), h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 300})
        return result.x
    
    h = evolve_pattern(h)
    h = refine_pattern(h)
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    if sum_h != 0:
        h = h * (target_sum / sum_h)
    h = np.clip(h, 0.0, 1.0)
    c5_bound = compute_c5(h)
    return h.tolist(), c5_bound, n_points
