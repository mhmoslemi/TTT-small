import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def constraint(h):
        return np.sum(h) - target_sum

    def rescale(h):
        return h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h

    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        mask = np.random.rand(n_points) < mutation_rate
        h[mask] += np.random.normal(0, 0.1, np.sum(mask))
        h = np.clip(h, 0.0, 1.0)
        return rescale(h)

    def crossover(h1, h2):
        h = h1.copy()
        mask = np.random.rand(n_points) < 0.5
        h[mask] = h2[mask]
        return rescale(h)

    population_size = 50
    population = [rescale(np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])) for _ in range(population_size)]
    best_h = rescale(h)
    best_c5 = evaluate(best_h)

    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        parents = []
        for _ in range(2):
            candidates = np.random.choice(population, size=5, replace=False)
            best = min(candidates, key=evaluate)
            parents.append(best)
        offspring = []
        for _ in range(population_size // 2):
            h1, h2 = np.random.choice(parents, size=2, replace=False)
            child = crossover(h1, h2)
            child = mutate(child)
            offspring.append(child)
        population = population + offspring
        population = sorted(population, key=evaluate)[:population_size]
        current_c5 = evaluate(population[0])
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[0]
        if np.abs(best_c5 - evaluate(best_h)) < 1e-6:
            break

    return best_h.tolist(), best_c5, n_points
