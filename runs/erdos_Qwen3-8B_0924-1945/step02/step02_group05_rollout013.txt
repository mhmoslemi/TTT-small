import numpy as np
from scipy.optimize import minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy() if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    start_time = time()

    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    perturbations = [0.1, 0.05, 0.02, 0.01, 0.005]
    for _ in range(200):
        if time() - start_time > budget_s:
            break
        for p in perturbations:
            h_perturbed = best_h.copy()
            indices = np.random.choice(n_points, size=5, replace=False)
            for i in indices:
                h_perturbed[i] += np.random.uniform(-p, p)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed / np.sum(h_perturbed) * n_points / 2
            current_c5 = evaluate_candidate(h_perturbed)
            if current_c5 < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = current_c5
        if best_c5 < 0.3808:
            break

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h_rand = np.random.uniform(0, 1, n_points)
            h_rand = h_rand / np.sum(h_rand) * n_points / 2
            current_c5 = evaluate_candidate(h_rand)
            if current_c5 < best_c5:
                best_h = h_rand.copy()
                best_c5 = current_c5

    h_final = np.clip(best_h, 0, 1)
    h_final = h_final / np.sum(h_final) * n_points / 2
    c5_final = evaluate_candidate(h_final)
    return (h_final, c5_final, n_points)
