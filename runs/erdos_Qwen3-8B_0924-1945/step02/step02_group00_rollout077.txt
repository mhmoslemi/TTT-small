```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_correlation = np.inf
    best_h = np.zeros(n_points)

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    def generate_pattern(pattern_type):
        if pattern_type == 0:
            h = np.zeros(n_points)
            for i in range(n_points):
                h[i] = 1.0 if i % 5 == 0 else 0.0
            h = h * (target_sum / np.sum(h))
        elif pattern_type == 1:
            h = np.zeros(n_points)
            for i in range(n_points):
                h[i] = 0.5 + 0.25 * np.sin(np.pi * i / n_points)
            h = h * (target_sum / np.sum(h))
        elif pattern_type == 2:
            h = np.random.rand(n_points)
            h = (h < 0.5).astype(float)
            h = h * (target_sum / np.sum(h))
        elif pattern_type == 3:
            h = np.zeros(n_points)
            for i in range(n_points):
                if np.random.rand() < 0.05 or i % 3 == 0:
                    h[i] = 1.0
            h = h * (target_sum / np.sum(h))
        return h

    def optimize_pattern(h_initial):
        h_flat = h_initial.copy()
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraint}]
        result = opt.minimize(objective, h_flat, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 300})
        h_opt = result.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5 = objective(h_opt)
        return h_opt, c5

    def differential_evolution(population, max_iter=200):
        for _ in range(max_iter):
            for i in range(len(population)):
                a, b, c = np.random.choice(len(population), 3, replace=False)
                mutant = population[a] + 0.8 * (population[b] - population[c])
                mutant = np.clip(mutant, 0.0, 1.0)
                sum_mutant = np.sum(mutant)
                scale = target_sum / sum_mutant if sum_mutant != 0 else 1.0
                mutant = mutant * scale
                candidate = mutant
                candidate_c5 = objective(candidate)
                if candidate_c5 < objective(population[i]):
                    population[i] = candidate
        return population

    def evaluate_pattern(h):
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    start_time = time.time()
    patterns = [generate_pattern(pt) for pt in range(4)]
    patterns = [optimize_pattern(p) for p in patterns]
    patterns = [p[0] for p in patterns]
    patterns = np.array(patterns)

    for i in range(len(patterns)):
        h = patterns[i]
        c5 = evaluate_pattern(h)
        if c5 < max_correlation:
            max_correlation = c5
            best_h = h

    for i in range(10):
        if time.time() - start_time > budget_s - 20:
            break
        pop_size = 100
        population = np.array([np.random.rand(n_points) for _ in range(pop_size)])
        for j in range(pop_size):
            population[j] = optimize_pattern(population[j])[0]
        population = differential_evolution(population)
        for j in range(pop_size):
            h = population[j]
            c5 = evaluate_pattern(h)
            if c5 < max_correlation:
                max_correlation = c5
                best_h = h

    best_h = np.clip(best_h, 0.0, 1.0)
    sum_best = np.sum(best_h)
    scale = target_sum / sum_best if sum_best != 0 else 1.0
    best_h = best_h * scale

    c5_bound = evaluate_pattern(best_h)
    return best_h.tolist(), c5_bound, n_points
```
