import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return compute_c5(h)
    
    def objective(x):
        return evaluate_candidate(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    def objective_grad(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        max_index = np.argmax(correlate_vals)
        grad = np.zeros_like(h)
        for k in range(1, n_points):
            lag_vals = correlate_vals[n_points - k]
            dh = np.zeros_like(h)
            for i in range(n_points):
                if i + k < n_points:
                    dh[i] += (1 - h[i + k]) * dx
                if i - k >= 0:
                    dh[i] += h[i - k] * dx
            if k == max_index:
                grad += dh
        return grad
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        x0 = best_h.copy()
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, jac=objective_grad, tol=1e-5)
        if res.success:
            current_h = res.x
            current_c5 = evaluate_candidate(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            break
    
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        perturbation = np.zeros_like(best_h)
        indices = np.argsort(best_h)
        for i in range(len(indices)):
            if i < 0.2 * len(indices):
                perturbation[indices[i]] = np.random.uniform(-0.05, 0.05)
        new_h = best_h + perturbation
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        new_c5 = evaluate_candidate(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    return (best_h, best_c5, n_points)
