```python
import numpy as np
import time
import random

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr * dx)
        return max_corr

    def rescale(h):
        s = np.sum(h)
        if s != 0:
            h = h * (target_sum / s)
        return h

    def clip_and_rescale(h):
        h = np.clip(h, 0.0, 1.0)
        return rescale(h)

    def mutate(h):
        h = h.copy()
        for i in range(n_points):
            if random.random() < 0.1:
                h[i] += np.random.normal(0, 0.1)
                h[i] = np.clip(h[i], 0.0, 1.0)
        return h

    def crossover(parent1, parent2):
        child = np.zeros(n_points)
        for i in range(n_points):
            if random.random() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child

    def tournament_selection(population, k=5):
        selected = random.sample(population, k)
        return min(selected, key=evaluate)

    population_size = 50
    population = [rescale(np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])) for _ in range(population_size)]
    best_h = population[0]
    best_c5 = evaluate(best_h)

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for _ in range(population_size // 2):
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            child = crossover(parent1, parent2)
            child = mutate(child)
            child = clip_and_rescale(child)
            new_population.append(child)
        population = new_population + [tournament_selection(population) for _ in range(population_size // 2)]
        current_best = min(population, key=evaluate)
        current_c5 = evaluate(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
    best_h = clip_and_rescale(best_h)
    return best_h.tolist(), best_c5, n_points
```
