import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    # Differential evolution phase
    from scipy.optimize import differential_evolution
    def de_objective(x):
        return evaluate_candidate(x)

    result = differential_evolution(
        de_objective,
        bounds,
        strategy='best1bin',
        popsize=30,
        tol=1e-5,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        maxiter=300,
        disp=False
    )
    best_h = result.x
    best_c5 = evaluate_candidate(best_h)

    # Local refinement phase
    def de_fitness(x):
        return evaluate_candidate(x)

    def de_constraint(x):
        return np.sum(x) - n_points / 2

    cons_local = [{'type': 'eq', 'fun': de_constraint}]
    res = minimize(
        de_fitness,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons_local,
        tol=1e-7
    )
    if res.success:
        best_h = res.x
        best_c5 = evaluate_candidate(best_h)
    else:
        best_c5 = evaluate_candidate(best_h)

    # Final validation and return
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
