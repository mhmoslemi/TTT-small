import numpy as np
import scipy.optimize as opt
import time
import random

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    max_n_points = 200
    n_points = max_n_points
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    def evaluate(h):
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def generate_initial_pattern():
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))
        h = np.random.permutation(h)
        return h

    def evolve_population(population, fitness_func, n_gen=100):
        population = np.array(population)
        population = np.clip(population, 0.0, 1.0)
        fitness = np.array([fitness_func(ind) for ind in population])
        for _ in range(n_gen):
            indices = np.random.choice(len(population), size=50, replace=False)
            selected = population[indices]
            parents = np.mean(selected, axis=0)
            children = parents + np.random.normal(0, 0.1, size=parents.shape)
            children = np.clip(children, 0.0, 1.0)
            fitness_children = np.array([fitness_func(ind) for ind in children])
            population = np.vstack([population, children])
            fitness = np.concatenate([fitness, fitness_children])
            indices = np.argsort(fitness)[:len(population)//2]
            population = population[indices]
            fitness = fitness[indices]
        return population

    def refine_solution(h):
        def refined_objective(h_flat):
            h = h_flat.reshape(n_points)
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            corr = np.correlate(h, 1.0 - h, mode="full")
            c5 = np.max(corr * dx)
            return c5

        def refined_constraint(h_flat):
            h = h_flat.reshape(n_points)
            sum_h = np.sum(h)
            return sum_h - target_sum

        cons = [{'type': 'eq', 'fun': refined_constraint}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        result = opt.minimize(refined_objective, h.flatten(), method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})
        h = result.x.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        return h

    start_time = time.time()
    population = [h.copy() for _ in range(5)]
    for _ in range(3):
        population = evolve_population(population, evaluate)
        population = [refine_solution(ind) for ind in population]
    
    best_solution = min(population, key=evaluate)
    best_c5 = evaluate(best_solution)

    if time.time() - start_time < budget_s:
        for _ in range(2):
            population = evolve_population(population, evaluate)
            population = [refine_solution(ind) for ind in population]
            best_solution = min(population, key=evaluate)
            best_c5 = evaluate(best_solution)

    return best_solution.tolist(), best_c5, n_points
