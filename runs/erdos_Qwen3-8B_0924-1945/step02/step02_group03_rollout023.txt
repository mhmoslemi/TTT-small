import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time()

    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    # Differential Evolution
    def de_optimize():
        pop_size = 30
        pop = np.random.uniform(0, 1, (pop_size, n_points))
        pop = pop / np.sum(pop, axis=1, keepdims=True) * n_points / 2

        for _ in range(300):
            if time() - start_time > budget_s:
                break
            mutant = np.zeros_like(pop)
            for i in range(pop_size):
                a, b, c = np.random.choice(pop_size, 3, replace=False)
                mutant[i] = pop[a] + 0.5 * (pop[b] - pop[c])
                mutant[i] = np.clip(mutant[i], 0, 1)
                mutant[i] = mutant[i] / np.sum(mutant[i]) * n_points / 2
            trial = np.zeros((pop_size, n_points))
            for i in range(pop_size):
                for j in range(n_points):
                    if np.random.rand() < 0.8:
                        trial[i, j] = mutant[i, j]
                    else:
                        trial[i, j] = pop[i, j]
                trial[i] = trial[i] / np.sum(trial[i]) * n_points / 2
            fitness_pop = np.array([compute_c5(p) for p in pop])
            fitness_trial = np.array([compute_c5(t) for t in trial])
            for i in range(pop_size):
                if fitness_trial[i] < fitness_pop[i]:
                    pop[i] = trial[i]
            best_h_de = np.min(pop, axis=0)
            best_c5_de = compute_c5(best_h_de)
            if best_c5_de < best_c5:
                best_h = best_h_de.copy()
                best_c5 = best_c5_de

    de_optimize()

    # Local refinement with SLSQP
    def slsqp_refine():
        x0 = best_h.copy()
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            current_h = res.x
            current_c5 = evaluate_candidate(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5

    slsqp_refine()

    # Fallback random search if needed
    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
