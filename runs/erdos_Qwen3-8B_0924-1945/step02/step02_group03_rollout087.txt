import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    # DE optimization
    pop_size = 30
    mut_factor = 0.5
    cross_over = 0.8
    max_iter = 300
    best_de = best_h.copy()

    for _ in range(max_iter):
        if time() - start_time > budget_s:
            break
        mutant = np.random.uniform(0, 1, (pop_size, n_points))
        mutant = np.clip(mutant, 0, 1)
        mutant = mutant / np.sum(mutant, axis=1, keepdims=True) * n_points / 2
        for i in range(pop_size):
            indices = np.random.choice(n_points, 3, replace=False)
            mutant[i] = np.random.choice([best_h[indices[0]], best_h[indices[1]], best_h[indices[2]]], size=n_points)
        for i in range(pop_size):
            trial = np.copy(best_de)
            trial = np.random.choice(mutant[i], size=n_points)
            trial = np.clip(trial, 0, 1)
            trial = trial / np.sum(trial) * n_points / 2
            current_c5 = evaluate_candidate(trial)
            if current_c5 < evaluate_candidate(best_de):
                best_de = trial.copy()

    best_h = best_de.copy()
    best_c5 = evaluate_candidate(best_h)

    # Local refinement
    for _ in range(200):
        if time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            current_h = res.x
            current_c5 = evaluate_candidate(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            break

    # Final evaluation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    final_c5 = evaluate_candidate(best_h)
    return (best_h, final_c5, n_points)
