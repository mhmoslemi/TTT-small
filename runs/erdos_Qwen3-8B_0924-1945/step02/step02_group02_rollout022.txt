import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 1000
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        h = 0.5 + 0.25 * np.sin(np.pi * x_grid) + 0.25 * np.cos(2 * np.pi * x_grid)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * (target_sum / sum_h) if sum_h != 0 else h
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_h = h.copy()
    best_c5 = objective(best_h.flatten())

    start_time = time.time()

    # Differential Evolution for global search
    def de_objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * (target_sum / sum_h) if sum_h != 0 else h
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    result = opt.differential_evolution(
        de_objective, bounds, constraints=cons, popsize=20, 
        maxiter=100, tol=1e-5, recombination=0.9, mutation=0.5,
        polish=False, workers=-1, disp=False
    )
    h_opt = result.x
    h_opt = np.clip(h_opt, 0, 1)
    sum_h_opt = np.sum(h_opt)
    h_opt = h_opt * (target_sum / sum_h_opt) if sum_h_opt != 0 else h_opt
    c5_bound = objective(h_opt)
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt

    # Local refinement with SLSQP
    result = opt.minimize(
        objective, best_h.flatten(), 
        method='SLSQP', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 100}
    )
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0, 1)
    sum_h_opt = np.sum(h_opt)
    h_opt = h_opt * (target_sum / sum_h_opt) if sum_h_opt != 0 else h_opt
    c5_bound = objective(h_opt)
    if c5_bound < best_c5:
        best_c5 = c5_bound
        best_h = h_opt

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 10:
        result = opt.minimize(
            objective, best_h.flatten(), 
            method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0, 1)
        sum_h_opt = np.sum(h_opt)
        h_opt = h_opt * (target_sum / sum_h_opt) if sum_h_opt != 0 else h_opt
        c5_bound = objective(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt

    return best_h.tolist(), best_c5, n_points
