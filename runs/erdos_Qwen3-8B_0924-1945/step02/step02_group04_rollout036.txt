import numpy as np
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', None)
    if initial_h_values is not None:
        h = np.clip(initial_h_values, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return compute_c5(h)

    swarm_size = 30
    particles = [h.copy() for _ in range(swarm_size)]
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    pbest = [p.copy() for p in particles]
    gbest = h.copy()
    best_c5 = evaluate_candidate(gbest)

    omega = 0.8
    c1 = 1.5
    c2 = 1.5
    cooling_rate = 0.995
    iteration = 0
    max_iterations = 1000
    start_time = time.time()

    while iteration < max_iterations and time.time() - start_time < budget_s:
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = omega * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            current_c5 = evaluate_candidate(particles[i])
            if current_c5 < evaluate_candidate(pbest[i]):
                pbest[i] = particles[i].copy()
            if current_c5 < best_c5:
                gbest = particles[i].copy()
                best_c5 = current_c5
        omega *= cooling_rate
        iteration += 1

    best_h = np.clip(gbest, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    return (best_h, best_c5, n_points)
