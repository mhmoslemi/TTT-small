import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def mutate(h):
        h = h.copy()
        mask = np.random.rand(n_points) < 0.1
        h[mask] += np.random.randn(np.sum(mask)) * 0.1
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        return h

    def crossover(a, b):
        mask = np.random.rand(n_points) < 0.5
        return np.where(mask, a, b)

    def select_parents(population, fitnesses):
        parents = []
        for _ in range(2):
            idxs = np.random.choice(len(population), size=5, replace=False)
            best_idx = np.argmin(fitnesses[idxs])
            parents.append(population[idxs[best_idx]])
        return parents

    def evolve(population, fitnesses):
        new_pop = []
        for _ in range(len(population) // 2):
            parent1, parent2 = select_parents(population, fitnesses)
            child1 = crossover(parent1, parent2)
            child1 = mutate(child1)
            new_pop.append(child1)
            child2 = crossover(parent1, parent2)
            child2 = mutate(child2)
            new_pop.append(child2)
        new_pop += [population[i] for i in np.argsort(fitnesses)[:len(population)//5]]
        return new_pop

    population = [np.random.rand(n_points) * 2 - 1 for _ in range(50)]
    population = [h * (target_sum / np.sum(h)) for h in population]
    best_h = h
    best_c5 = evaluate(best_h)

    start_time = time.time()
    for gen in range(200):
        fitnesses = np.array([evaluate(h) for h in population])
        population = evolve(population, fitnesses)
        current_c5 = np.min(fitnesses)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[np.argmin(fitnesses)]
        if time.time() - start_time > budget_s - 100:
            break

    h_opt = best_h
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale

    c5_bound = evaluate(h_opt)
    return h_opt.tolist(), c5_bound, n_points
