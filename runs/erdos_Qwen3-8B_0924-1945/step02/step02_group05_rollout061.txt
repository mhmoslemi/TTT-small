import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    
    def normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        return compute_c5(normalize(h))
    
    def objective(h):
        return evaluate_candidate(h)
    
    def generate_patterns():
        patterns = []
        intervals = np.linspace(0, 2, n_points + 1)
        for _ in range(3):
            idxs = np.random.choice(n_points, size=5, replace=False)
            mask = np.zeros(n_points)
            mask[idxs] = 1
            mask = normalize(mask)
            patterns.append(mask)
        for _ in range(2):
            slope = np.random.uniform(-0.5, 0.5)
            x = np.linspace(0, 2, n_points)
            mask = 0.5 * (1 + np.tanh(slope * (x - 1)))
            mask = normalize(mask)
            patterns.append(mask)
        if 'initial_h_values' in kwargs:
            patterns.append(normalize(initial_h_values))
        return patterns
    
    def run_de():
        patterns = generate_patterns()
        bounds = [(0, 1) for _ in range(n_points)]
        result = differential_evolution(objective, bounds, polish=True, strategy='best1bin', maxiter=200, popsize=5, tol=1e-5, mutation=(0.8, 1), recombination=0.9, seed=seed, disp=False)
        best_h = result.x
        best_c5 = evaluate_candidate(best_h)
        return best_h, best_c5
    
    start_time = time()
    best_h, best_c5 = run_de()
    
    if time() - start_time < budget_s:
        for _ in range(10):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = normalize(h)
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    best_h = normalize(best_h)
    final_c5 = evaluate_candidate(best_h)
    
    return (best_h, final_c5, n_points)
