import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy() if 'initial_h_values' in kwargs else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    dx = 2.0 / n_points

    def compute_c5(h):
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time()

    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    def perturb(h, delta=0.1):
        h = h.copy()
        idx = np.random.choice(len(h), int(len(h) * delta), replace=False)
        h[idx] += np.random.uniform(-0.2, 0.2, len(idx))
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return h

    def sa_phase():
        temp = 1.0
        cooling = 0.995
        for _ in range(1000):
            if time() - start_time > budget_s:
                break
            if temp < 1e-5:
                break
            h_new = perturb(best_h)
            c5_new = evaluate_candidate(h_new)
            delta = c5_new - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                best_h = h_new
                best_c5 = c5_new
            temp *= cooling

    def ga_phase():
        population = [best_h.copy() for _ in range(30)]
        for _ in range(500):
            if time() - start_time > budget_s:
                break
            selected = np.random.choice(len(population), 5, replace=False)
            parents = [population[i] for i in selected]
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = (p1 + p2) / 2
            child = perturb(child, delta=0.05)
            c5_child = evaluate_candidate(child)
            worst_idx = np.argmin([evaluate_candidate(ind) for ind in population])
            population[worst_idx] = child

    sa_phase()
    ga_phase()

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
