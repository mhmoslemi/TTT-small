import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    def differential_evolution(pop_size=30, max_iter=300):
        population = np.random.uniform(0, 1, (pop_size, n_points))
        population = np.array([p / np.sum(p) * n_points / 2 for p in population])
        for _ in range(max_iter):
            if time() - start_time > budget_s:
                break
            mutant = np.zeros_like(population)
            for i in range(pop_size):
                a, b, c = np.random.choice(pop_size, 3, replace=False)
                mutant[i] = population[a] + 0.5 * (population[b] - population[c])
                mutant[i] = np.clip(mutant[i], 0, 1)
                mutant[i] = mutant[i] / np.sum(mutant[i]) * n_points / 2
            trial = np.zeros((pop_size, n_points))
            for i in range(pop_size):
                for j in range(n_points):
                    if np.random.rand() < 0.8:
                        trial[i, j] = mutant[i, j]
                    else:
                        trial[i, j] = population[i, j]
            trial = np.array([t / np.sum(t) * n_points / 2 for t in trial])
            fitness_pop = np.array([compute_c5(p) for p in population])
            fitness_trial = np.array([compute_c5(t) for t in trial])
            for i in range(pop_size):
                if fitness_trial[i] < fitness_pop[i]:
                    population[i] = trial[i]
            best_pop = np.argmin(fitness_pop)
            best_h = population[best_pop]
            best_c5 = fitness_pop[best_pop]
        return best_h, best_c5

    best_h, best_c5 = differential_evolution()
    if time() - start_time < budget_s:
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            best_h = res.x
            best_c5 = evaluate_candidate(best_h)

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
