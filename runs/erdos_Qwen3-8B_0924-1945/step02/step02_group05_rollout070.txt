import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy() if 'initial_h_values' in kwargs else np.zeros(n_points)
    for i in range(n_points):
        if np.random.rand() < 0.5:
            h[i] = 1
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()

    # Generate a few structured patterns
    def generate_uniform_pattern(m):
        intervals = np.linspace(0, 2, m + 1)
        selected = np.random.choice(m, size=int(m / 2), replace=False)
        selected_intervals = intervals[selected]
        selected_intervals = np.concatenate((selected_intervals, intervals[selected + 1]))
        h_pattern = np.zeros(n_points)
        for i in range(n_points):
            x = i * 2.0 / n_points
            for interval in selected_intervals:
                if interval - 0.1 <= x <= interval + 0.1:
                    h_pattern[i] = 1
                    break
        return h_pattern

    def generate_bilinear_pattern(slope):
        x = np.linspace(0, 2, n_points)
        h_pattern = 0.5 * (1 + np.tanh(slope * (x - 1)))
        return h_pattern

    # Add some structured patterns to the initial population
    initial_patterns = [h.copy()]
    for m in [4, 5, 6]:
        initial_patterns.append(generate_uniform_pattern(m))
    for slope in [0.1, 0.5, 1.0, 2.0]:
        initial_patterns.append(generate_bilinear_pattern(slope))

    # Normalize all patterns to satisfy the integral constraint
    for i in range(len(initial_patterns)):
        initial_patterns[i] = initial_patterns[i] / np.sum(initial_patterns[i]) * n_points / 2

    # Evaluate initial patterns and keep the best one
    best_h = initial_patterns[0]
    best_c5 = evaluate_candidate(best_h)
    for i in range(1, len(initial_patterns)):
        candidate = initial_patterns[i]
        candidate_c5 = evaluate_candidate(candidate)
        if candidate_c5 < best_c5:
            best_h = candidate
            best_c5 = candidate_c5

    # Define an optimization function
    def optimize_candidate(start_h):
        best_h = start_h.copy()
        best_c5 = evaluate_candidate(best_h)
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            x0 = best_h.copy()
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
            if res.success:
                current_h = res.x
                current_c5 = evaluate_candidate(current_h)
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
        return best_h, best_c5

    # Optimize the best initial candidate
    best_h, best_c5 = optimize_candidate(best_h)

    # Local search with random restarts
    for _ in range(5):
        if time() - start_time > budget_s:
            break
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
        current_c5 = evaluate_candidate(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)

    return (best_h, current_c5, n_points)
