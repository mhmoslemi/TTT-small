import numpy as np
from scipy.optimize import differential_evolution
from numpy.random import choice, uniform, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    
    def validate_h(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if np.abs(total - n_points / 2) > 1e-5:
            h = h * (n_points / 2) / total
        return h
    
    def compute_correlation(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        h = validate_h(h)
        return compute_correlation(h)
    
    def objective(h):
        return evaluate_candidate(h)
    
    def generate_structured_patterns():
        patterns = []
        # Uniform interval patterns
        for intervals in [3, 5, 7]:
            h = np.zeros(n_points)
            for i in range(intervals):
                start = int(i * n_points / intervals)
                end = int((i + 1) * n_points / intervals)
                h[start:end] = 1
            h = validate_h(h)
            patterns.append(h)
        # Symmetric bilinear patterns
        for slope in [-0.5, 0.5]:
            h = np.zeros(n_points)
            for i in range(n_points):
                x = i / n_points * 2
                h[i] = 0.5 * (1 + np.tanh(slope * (x - 1)))
            h = validate_h(h)
            patterns.append(h)
        # Random binary pattern
        h = np.random.choice([0, 1], size=n_points)
        h = validate_h(h)
        patterns.append(h)
        # Modified initial_h_values
        if 'initial_h_values' in kwargs:
            h = initial_h_values.copy()
            h = validate_h(h)
            patterns.append(h)
        return patterns
    
    # Initialize population
    initial_population = generate_structured_patterns()
    h_pop = np.array(initial_population)
    best_h = h_pop[0]
    best_c5 = evaluate_candidate(best_h)
    for h in h_pop:
        c5 = evaluate_candidate(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    # Parameters for DE
    pop_size = 5
    bounds = [(0, 1) for _ in range(n_points)]
    result = differential_evolution(objective, bounds, popsize=pop_size, strategy='best1bin', tol=1e-5, maxiter=300, recombination=0.9, mutation=0.8, seed=seed)
    result_h = result.x
    result_c5 = evaluate_candidate(result_h)
    
    if result_c5 < best_c5:
        best_h = result_h
        best_c5 = result_c5
    
    # Final validation
    best_h = validate_h(best_h)
    final_c5 = evaluate_candidate(best_h)
    
    return (best_h, final_c5, n_points)
