import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()

    def run_optimization():
        best_h = h.copy()
        best_c5 = evaluate_candidate(best_h)
        for _ in range(1000):
            if time() - start_time > budget_s:
                break
            corr_vals = np.correlate(best_h, 1 - best_h, mode="full") * (2.0 / n_points)
            max_corr = np.max(corr_vals)
            max_idx = np.argmax(corr_vals)
            if max_corr <= best_c5:
                break
            k = max_idx
            active_range = np.arange(max(0, k - 10), min(n_points, k + 10))
            grad = np.zeros(n_points)
            for x in active_range:
                if x + k < n_points:
                    grad[x] += 1 - best_h[x + k]
                if x < n_points - k:
                    grad[x + k] -= best_h[x]
            step = 0.05
            new_h = best_h - grad * step
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h / np.sum(new_h) * n_points / 2
            current_c5 = compute_c5(new_h)
            if current_c5 < best_c5:
                best_h = new_h
                best_c5 = current_c5
        return best_h, best_c5

    best_h, best_c5 = run_optimization()

    if time() - start_time < budget_s:
        for _ in range(200):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
