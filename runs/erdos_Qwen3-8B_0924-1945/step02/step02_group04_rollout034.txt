import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize import basinhopping
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    h = np.clip(initial_h_values, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)

    def compute_c5(h, n_points):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def normalize(h):
        h = np.clip(h, 0, 1)
        return h * (n_points / 2) / np.sum(h)

    def objective(h):
        h = normalize(h)
        return compute_c5(h, n_points)

    def perturb(h, stepsize=0.2):
        perturbation = np.random.uniform(-stepsize, stepsize, size=h.shape)
        new_h = h + perturbation
        return normalize(new_h)

    def local_minimize(h):
        result = minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-5)
        if result.success:
            return result.x
        return h

    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        h_perturbed = perturb(best_h)
        h_local = local_minimize(h_perturbed)
        current_c5 = compute_c5(h_local, n_points)
        if current_c5 < best_c5:
            best_h = h_local
            best_c5 = current_c5

    best_h = normalize(best_h)
    return (best_h, best_c5, n_points)
