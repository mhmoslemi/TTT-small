import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time()

    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    def mutate(x):
        x = x.copy()
        for i in range(len(x)):
            if np.random.rand() < 0.2:
                x[i] += np.random.uniform(-0.1, 0.1)
                x[i] = np.clip(x[i], 0, 1)
        return x

    def crossover(x1, x2):
        x = np.zeros_like(x1)
        for i in range(len(x)):
            if np.random.rand() < 0.5:
                x[i] = x1[i]
            else:
                x[i] = x2[i]
        return x

    population = [mutate(h) for _ in range(50)]
    population = [np.clip(p, 0, 1) for p in population]
    population = [p / np.sum(p) * n_points / 2 for p in population]
    population = [p for p in population if np.sum(p) > 1e-6]

    for _ in range(500):
        if time() - start_time > budget_s:
            break
        if len(population) < 1:
            break
        parents = np.random.choice(population, size=5, replace=False)
        offspring = []
        for i in range(5):
            if i % 2 == 0:
                p1 = parents[i // 2]
                p2 = parents[(i // 2) + 1]
                child = crossover(p1, p2)
                child = mutate(child)
                offspring.append(child)
            else:
                p = parents[i // 2]
                child = mutate(p)
                offspring.append(child)
        offspring = [np.clip(o, 0, 1) for o in offspring]
        offspring = [o / np.sum(o) * n_points / 2 for o in offspring]
        offspring = [o for o in offspring if np.sum(o) > 1e-6]
        population += offspring
        population = np.array(population)
        fitnesses = [fitness(p) for p in population]
        indices = np.argsort(fitnesses)
        population = population[indices[:50]]
        best_h = population[0]
        best_c5 = fitness(best_h)

    for _ in range(50):
        if time() - start_time > budget_s:
            break
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
        current_c5 = evaluate_candidate(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
