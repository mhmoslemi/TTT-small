```python
import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    population_size = 50
    tournament_size = 5
    mutation_rate = 0.1
    crossover_prob = 0.8
    elitism_rate = 0.2
    generations = int(budget_s * 0.5 / (population_size * 5))
    best_h = h.copy()

    def rescale(h):
        sum_h = np.sum(h)
        scale = target_sum / sum_h
        return np.clip(h * scale, 0.0, 1.0)

    def mutate(h):
        h = h.copy()
        mask = np.random.rand(n_points) < mutation_rate
        h[mask] += np.random.normal(0, 0.05, np.sum(mask))
        return rescale(h)

    def crossover(h1, h2):
        h = h1.copy()
        mask = np.random.rand(n_points) < crossover_prob
        h[mask] = h2[mask]
        return rescale(h)

    def tournament_selection(population):
        indices = np.random.choice(len(population), tournament_size, replace=False)
        return population[indices[np.argmin([evaluate(population[i]) for i in indices])]]

    population = [rescale(np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])) for _ in range(population_size)]
    for _ in range(generations):
        if evaluate(best_h) < evaluate(population[np.argmin([evaluate(p) for p in population])]):
            best_h = population[np.argmin([evaluate(p) for p in population])]
        new_population = []
        for _ in range(population_size // 2):
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            child1 = crossover(parent1, parent2)
            child2 = mutate(child1)
            new_population.extend([child1, child2])
        population = new_population
        if not population:
            break
        if evaluate(best_h) < evaluate(population[np.argmin([evaluate(p) for p in population])]):
            best_h = population[np.argmin([evaluate(p) for p in population])]
        if len(population) < population_size:
            population += [rescale(np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])) for _ in range(population_size - len(population))]

    best_h = rescale(best_h)
    c5_bound = evaluate(best_h)
    return best_h.tolist(), c5_bound, n_points
```
