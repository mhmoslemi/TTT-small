import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy() if 'initial_h_values' in kwargs else np.random.uniform(0, 1, n_points)
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    def de_step(population, best_h):
        mutant = np.random.normal(0, 0.5, n_points)
        trial = population[np.random.choice(len(population))] + mutant
        trial = np.clip(trial, 0, 1)
        trial = trial / np.sum(trial) * n_points / 2
        trial_c5 = evaluate_candidate(trial)
        if trial_c5 < best_c5:
            return trial, trial_c5
        return best_h, best_c5

    population = [np.random.uniform(0, 1, n_points) for _ in range(30)]
    population = [np.clip(p, 0, 1) for p in population]
    population = [p / np.sum(p) * n_points / 2 for p in population]
    population_best = min(population, key=evaluate_candidate)
    population_best_c5 = evaluate_candidate(population_best)

    for _ in range(300):
        if time() - start_time > budget_s:
            break
        new_population = []
        for p in population:
            mutant = np.random.normal(0, 0.5, n_points)
            trial = p + mutant
            trial = np.clip(trial, 0, 1)
            trial = trial / np.sum(trial) * n_points / 2
            trial_c5 = evaluate_candidate(trial)
            if trial_c5 < population_best_c5:
                new_population.append(trial)
            else:
                new_population.append(p)
        population = new_population
        population_best = min(population, key=evaluate_candidate)
        population_best_c5 = evaluate_candidate(population_best)

    best_h = population_best
    best_c5 = population_best_c5

    for _ in range(100):
        if time() - start_time > budget_s:
            break
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        current_c5 = evaluate_candidate(h)
        if current_c5 < best_c5:
            best_h = h
            best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    return (best_h, best_c5, n_points)
