import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    if np.any(initial_h_values < 0) or np.any(initial_h_values > 1):
        initial_h_values = np.clip(initial_h_values, 0, 1)
        initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    def compute_c5(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        correlate_vals = np.correlate(h_vals, 1 - h_vals, mode="full") * dx
        return np.max(correlate_vals)
    def objective(h_vals):
        return compute_c5(h_vals)
    def generate_structured_patterns():
        patterns = []
        def uniform_pattern(k_intervals):
            intervals = np.linspace(0, 2, k_intervals + 1)
            h = np.zeros(n_points)
            for i in range(k_intervals):
                start = np.searchsorted(intervals, 0) + i
                end = np.searchsorted(intervals, 2) + i + 1
                h[start:end] = 1
            h = h * (n_points / 2) / np.sum(h)
            return h
        def symmetric_bilinear_pattern(slope):
            x = np.linspace(0, 2, n_points)
            h = 0.5 * (1 + np.tanh(slope * (x - 1)))
            h = h * (n_points / 2) / np.sum(h)
            return h
        def random_binary_pattern():
            h = np.random.choice([0, 1], size=n_points)
            h = h * (n_points / 2) / np.sum(h)
            return h
        def lanczos_pattern():
            x = np.linspace(0, 2, n_points)
            h = np.sinc(x - 1)
            h = h * (n_points / 2) / np.sum(h)
            return h
        patterns.append(uniform_pattern(3))
        patterns.append(uniform_pattern(5))
        patterns.append(uniform_pattern(7))
        patterns.append(symmetric_bilinear_pattern(0.5))
        patterns.append(symmetric_bilinear_pattern(-0.5))
        patterns.append(random_binary_pattern())
        patterns.append(lanczos_pattern())
        return patterns
    def bounds_function(h_vals):
        return np.clip(h_vals, 0, 1)
    def constraint(h_vals):
        return np.sum(h_vals) - n_points / 2
    bounds = [(0, 1) for _ in range(n_points)]
    start_time = time()
    population = generate_structured_patterns()
    population.append(h.copy())
    population = np.array(population)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for gen in range(300):
        if time() - start_time > budget_s:
            break
        for i in range(len(population)):
            if np.random.rand() < 0.1:
                h_new = np.random.uniform(0, 1, n_points)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
            else:
                indices = np.random.choice(len(population), 3, replace=False)
                h1, h2, h3 = population[indices]
                h_new = h1 + 0.8 * (h2 - h3)
            h_new = bounds_function(h_new)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            if np.random.rand() < 0.5:
                population[i] = h_new.copy()
    for _ in range(200):
        if time() - start_time > budget_s:
            break
        h_new = np.random.uniform(0, 1, n_points)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    best_h = bounds_function(best_h)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
