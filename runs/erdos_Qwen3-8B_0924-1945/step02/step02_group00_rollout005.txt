```python
import numpy as np
import scipy.optimize as opt
import time
import random
from deap import creator, base, tools, algorithms

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def evaluate(individual):
        h = np.array(individual, dtype=np.float64)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5,

    def mutate(individual, indpb):
        for i in range(len(individual)):
            if random.random() < indpb:
                individual[i] = 1 - individual[i]
        return individual,

    def crossover(ind1, ind2):
        size = len(ind1)
        ind1[0:size//2] = ind2[0:size//2]
        ind2[0:size//2] = ind1[0:size//2]
        return ind1, ind2

    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)

    toolbox = base.Toolbox()
    toolbox.register("individual", lambda: [random.randint(0, 1) for _ in range(n_points)])
    toolbox.register("population", tools.initRepeat, creator.Individual, toolbox.individual)
    toolbox.register("mate", crossover)
    toolbox.register("mutate", mutate, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=5)
    toolbox.register("evaluate", evaluate)

    pop = toolbox.population(n=50)
    tools.initIndsWithFitness(pop, toolbox.evaluate)

    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        offspring = algorithms.varAnd(pop, toolbox, cxpb=0.5, mutpb=0.1)
        fits = toolbox.map(toolbox.evaluate, offspring)
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = fit
        pop = toolbox.select(pop + offspring, len(pop))
        if gen % 10 == 0:
            best = min(pop, key=lambda ind: ind.fitness.values[0])
            h_opt = np.array(best)
            h_opt = np.clip(h_opt, 0.0, 1.0)
            sum_h_opt = np.sum(h_opt)
            scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
            h_opt = h_opt * scale

    h_opt = np.array([ind for ind in pop if ind.fitness.values[0] == min([ind.fitness.values[0] for ind in pop])][0])
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = evaluate(h_opt)[0]
    return h_opt.tolist(), c5_bound, n_points
```
