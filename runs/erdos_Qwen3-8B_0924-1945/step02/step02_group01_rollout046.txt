```python
import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    target_sum = n_points / 2
    population_size = 50
    generations = 200
    mutation_rate = 0.1
    crossover_rate = 0.8
    elitism_rate = 0.2
    tournament_size = 5
    mutation_std = 0.05
    max_iterations = 1000

    def generate_initial_population():
        pop = []
        for _ in range(population_size):
            indices = np.random.choice(n_points, size=int(n_points / 2), replace=False)
            h = np.zeros(n_points)
            h[indices] = 1.0
            h = h * (target_sum / np.sum(h))
            pop.append(h)
        return pop

    def evaluate(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def tournament_selection(pop, scores):
        selected = []
        for _ in range(len(pop)):
            tournament = np.random.choice(scores, size=tournament_size)
            winner_idx = np.argmin(tournament)
            selected.append(pop[winner_idx])
        return selected

    def crossover(parent1, parent2):
        if np.random.rand() < crossover_rate:
            child = np.copy(parent1)
            mask = np.random.rand(n_points) < 0.5
            child[mask] = parent2[mask]
            return child
        return parent1

    def mutate(h):
        h = np.copy(h)
        mutation_indices = np.random.choice(n_points, size=int(n_points * mutation_rate), replace=False)
        h[mutation_indices] += np.random.normal(0, mutation_std, size=len(mutation_indices))
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))
        return h

    def rescale(h):
        h = np.copy(h)
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))
        return h

    def select_survivors(pop, scores, new_pop):
        sorted_indices = np.argsort(scores)
        top_indices = sorted_indices[:int(elitism_rate * len(pop))]
        survivors = [pop[i] for i in top_indices]
        while len(survivors) < len(pop):
            survivors.append(new_pop.pop())
        return survivors

    if initial_h_values is not None:
        initial_pop = [np.array(initial_h_values, dtype=np.float64) * (target_sum / np.sum(initial_h_values))]
    else:
        initial_pop = generate_initial_population()

    population = initial_pop
    best_h = np.copy(population[0])
    best_c5 = evaluate(best_h)

    start_time = time.time()
    iteration = 0
    while iteration < generations and time.time() - start_time < budget_s:
        scores = [evaluate(h) for h in population]
        new_pop = []

        for _ in range(len(population)):
            parent1 = tournament_selection(population, scores)[0]
            parent2 = tournament_selection(population, scores)[0]
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_pop.append(child)

        population = select_survivors(population, scores, new_pop)
        current_c5 = evaluate(population[0])
        if current_c5 < best_c5:
            best_h = np.copy(population[0])
            best_c5 = current_c5
        iteration += 1

    best_h = rescale(best_h)
    best_c5 = evaluate(best_h)
    return best_h.tolist(), best_c5, n_points
```
