import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', np.random.rand(n_points))
    h = np.clip(initial_h_values, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return compute_c5(h)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    start_time = time.time()
    max_iterations = 1000
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        h_new = h + np.random.uniform(-0.1, 0.1, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        current_c5 = evaluate_candidate(h_new)
        if current_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = current_c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    return (best_h, best_c5, n_points)
