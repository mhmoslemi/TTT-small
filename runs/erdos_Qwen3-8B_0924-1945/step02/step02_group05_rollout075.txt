import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time
import copy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    
    def normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        h = h * (n_points / 2) / total
        return h
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        h = normalize(h)
        return compute_c5(h)
    
    def fitness(h):
        return evaluate_candidate(h)
    
    def objective(x):
        return fitness(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    start_time = time()
    
    def generate_uniform_pattern(m, active_frac):
        intervals = np.linspace(0, 2, m + 1)
        active_intervals = np.random.choice(m, int(m * active_frac), replace=False)
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * 2.0 / n_points
            for j in range(m):
                if intervals[j] <= x < intervals[j + 1]:
                    if j in active_intervals:
                        h[i] = 1
                    break
        return normalize(h)
    
    def generate_bilinear_pattern(slope):
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * 2.0 / n_points
            h[i] = 0.5 * (1 + np.tanh(slope * (x - 1)))
        return normalize(h)
    
    def generate_random_pattern(density):
        h = np.random.binomial(1, density, size=n_points)
        return normalize(h)
    
    def run_mcmc(initial_h, budget_s):
        current_h = initial_h.copy()
        best_h = current_h.copy()
        best_c5 = compute_c5(current_h)
        budget_remaining = budget_s
        temp = 1.0
        cooling_rate = 0.995
        iter_count = 0
        max_iter = 1000
        while iter_count < max_iter and budget_remaining > 0.1:
            if time() - start_time > budget_s:
                break
            budget_remaining = budget_s - (time() - start_time)
            iter_count += 1
            
            if np.random.rand() < 0.3:
                h_candidate = generate_uniform_pattern(np.random.choice([3,4,5,6,7]), np.random.uniform(0.4, 0.6))
            elif np.random.rand() < 0.3:
                h_candidate = generate_bilinear_pattern(np.random.choice([-0.5, -0.25, 0, 0.25, 0.5]))
            else:
                h_candidate = generate_random_pattern(np.random.uniform(0.5, 0.8))
            
            current_c5 = compute_c5(current_h)
            candidate_c5 = compute_c5(h_candidate)
            delta = candidate_c5 - current_c5
            if delta < 0 or np.random.rand() < np.exp(-temp * delta):
                current_h = h_candidate.copy()
                if candidate_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = candidate_c5
            temp *= cooling_rate
        
        return best_h, best_c5
    
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if np.random.rand() < 0.5 else 0
        h = normalize(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_h, best_c5 = run_mcmc(best_h, budget_s)
    
    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = normalize(h)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    best_h = normalize(best_h)
    current_c5 = compute_c5(best_h)
    return (best_h, current_c5, n_points)
