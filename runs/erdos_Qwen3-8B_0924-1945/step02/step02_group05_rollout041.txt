import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()

    def run_optimization():
        best_h = h.copy()
        best_c5 = evaluate_candidate(best_h)
        for _ in range(1000):
            if time() - start_time > budget_s:
                break
            corr_vals = np.correlate(best_h, 1 - best_h, mode="full") * 2.0 / n_points
            max_k = np.argmax(corr_vals)
            k = max_k
            x_start = max(0, k - 10)
            x_end = min(n_points - 1, k + 10)
            grads = np.zeros(n_points)
            for x in range(x_start, x_end + 1):
                if x + k < n_points:
                    grads[x] += (1 - best_h[x + k])
                if x < n_points - k:
                    grads[x + k] -= best_h[x]
            step_size = 0.01
            new_h = best_h - step_size * grads
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h / np.sum(new_h) * n_points / 2
            new_c5 = evaluate_candidate(new_h)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
        return best_h, best_c5

    best_h, best_c5 = run_optimization()

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
