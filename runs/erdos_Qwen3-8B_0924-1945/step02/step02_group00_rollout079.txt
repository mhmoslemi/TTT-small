```python
import numpy as np
import scipy.optimize as opt
import time
import random

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    best_c5 = float('inf')
    best_h = None

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    def generate_base_pattern():
        if initial_h_values is not None:
            h = np.array(initial_h_values, dtype=np.float64)
            h = h * (target_sum / np.sum(h))
            h = np.round(h).astype(float)
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
            h = np.round(h).astype(float)
            h = np.clip(h, 0, 1)
            return h
        else:
            h = np.zeros(n_points, dtype=np.float64)
            for i in range(n_points):
                h[i] = 1.0 if i % 5 == 0 else 0.0
            h = h * (target_sum / np.sum(h))
            return h

    def perturb_pattern(h, perturb_rate=0.03):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < perturb_rate:
                h[i] = 1 - h[i]
        h = h * (target_sum / np.sum(h))
        return h

    def create_patterns():
        patterns = []
        base = generate_base_pattern()
        patterns.append(base)
        patterns.append(np.random.choice([0, 1], size=n_points, p=[0.5, 0.5]))
        patterns.append(np.sin(np.linspace(0, np.pi, n_points)) > 0)
        patterns.append(np.zeros(n_points))
        for i in range(n_points // 10):
            patterns.append(np.zeros(n_points))
            patterns[-1][i * 10] = 1
        patterns.append(np.zeros(n_points))
        for i in range(n_points // 2):
            patterns[-1][i * 2] = 1
        return [np.clip(p * (target_sum / np.sum(p)), 0, 1) for p in patterns]

    patterns = create_patterns()

    def objective(x):
        return evaluate(x)

    def optimize_pattern(p):
        x0 = p.copy()
        x0 = x0.flatten()
        cons = [{'type': 'eq', 'fun': constraint}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        try:
            res = opt.minimize(objective, x0, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 300})
            res = opt.minimize(objective, res.x, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
            res_x = res.x
            res_x = np.clip(res_x, 0.0, 1.0)
            sum_h = np.sum(res_x)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            res_x = res_x * scale
            res_x = np.clip(res_x, 0.0, 1.0)
            return res_x
        except:
            return p

    start_time = time.time()
    for p in patterns:
        h = optimize_pattern(p)
        c5 = evaluate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    elapsed = time.time() - start_time
    if elapsed < budget_s:
        def de_objective(x):
            return evaluate(x)

        def de_constraint(x):
            return constraint(x)

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': de_constraint}]

        def de_func(x):
            return de_objective(x)

        from deap import base, creator, tools, algorithms
        creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
        creator.create("Individual", list, fitness=creator.FitnessMin)

       toolbox = base.Toolbox()
        toolbox.register("attr_bool", lambda: 0 if np.random.rand() < 0.5 else 1)
        toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_bool, n=n_points)
        toolbox.register("population", tools.initRepeat, list, toolbox.individual)

        toolbox.register("evaluate", de_func)
        toolbox.register("mate", tools.cxTwoPoint)
        toolbox.register("mutate", tools.mutFlipBit, indpb=0.05)
        toolbox.register("select", tools.selTournament, tournsize=3)

        pop = toolbox.population(n=50)
        stats = tools.Statistics(lambda ind: ind.fitness.values)
        stats.register("avg", np.mean)
        stats.register("std", np.std)
        stats.register("min", np.min)
        stats.register("max", np.max)

        algorithms.eaSimple(pop, toolbox, cxpb=0.5, mutpb=0.1, ngen=200, stats=stats, halting_criteria=None, verbose=False)

        for ind in pop:
            ind.fitness.values = de_func(ind)
        for ind in pop:
            if ind.fitness < best_c5:
                best_h = ind
                best_c5 = ind.fitness

    final_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(final_h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    final_h = final_h * scale
    final_h = np.clip(final_h, 0.0, 1.0)

    return final_h.tolist(), best_c5, n_points
```
