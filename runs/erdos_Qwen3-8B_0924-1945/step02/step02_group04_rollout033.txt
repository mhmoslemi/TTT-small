import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize import basinhopping
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
    h = np.clip(h, 0, 1)

    def normalize(h):
        h_clip = np.clip(h, 0, 1)
        return h_clip * (n_points / 2) / np.sum(h_clip)

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def objective(h):
        return compute_c5(h)

    start_time = time.time()
    best_h = h
    best_c5 = compute_c5(h)

    def callback(x):
        nonlocal best_h, best_c5
        current_c5 = compute_c5(x)
        if current_c5 < best_c5:
            best_h = x.copy()
            best_c5 = current_c5

    def perturb(x, stepsize):
        perturbation = np.random.uniform(-stepsize, stepsize, size=len(x))
        new_h = x + perturbation
        new_h = np.clip(new_h, 0, 1)
        new_h = normalize(new_h)
        return new_h

    def minimize_wrapper(x):
        result = minimize(objective, x, method='L-BFGS-B', bounds=[(0,1)]*n_points, tol=1e-6)
        if result.success:
            return result.x
        else:
            return x

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        current_h = normalize(best_h)
        perturbed_h = perturb(current_h, 0.1)
        local_min = minimize_wrapper(perturbed_h)
        if compute_c5(local_min) < best_c5:
            best_h = local_min
            best_c5 = compute_c5(best_h)

    best_h = normalize(best_h)
    return (best_h, best_c5, n_points)
