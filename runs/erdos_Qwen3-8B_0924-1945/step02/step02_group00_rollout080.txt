import numpy as np
import scipy.optimize as opt
import time
import random
from deap import base, creator, tools, algorithms

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def valid_pattern(h):
        return np.all((h >= 0) & (h <= 1)) and np.isclose(np.sum(h), target_sum, atol=1e-6)

    if initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=np.float64)
        if not valid_pattern(h_init):
            h_init = np.zeros(n_points)
            for i in range(n_points):
                h_init[i] = 1.0 if i % 5 == 0 else 0.0
            h_init = h_init * (target_sum / np.sum(h_init))
    else:
        h_init = np.zeros(n_points)
        for i in range(n_points):
            h_init[i] = 1.0 if i % 5 == 0 else 0.0
        h_init = h_init * (target_sum / np.sum(h_init))

    def compute_c5(h):
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    if not valid_pattern(h_init):
        h_init = np.zeros(n_points)
        for i in range(n_points):
            h_init[i] = 1.0 if i % 5 == 0 else 0.0
        h_init = h_init * (target_sum / np.sum(h_init))

    def objective(individual):
        h = np.array(individual, dtype=np.float64)
        return compute_c5(h)

    def evaluate(individual):
        return objective(individual),

    def mutate(individual, indpb):
        for i in range(len(individual)):
            if random.random() < indpb:
                individual[i] = 1 - individual[i]
        return individual,

    def crossover(ind1, ind2, cxpb):
        if random.random() < cxpb:
            idx = random.randint(1, len(ind1) - 1)
            ind1[idx:], ind2[idx:] = ind2[idx:].copy(), ind1[idx:].copy()
        return ind1, ind2

    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)

   toolbox = base.Toolbox()
    toolbox.register("attr_bool", random.choice, [0, 1])
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_bool, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    toolbox.register("evaluate", evaluate)
    toolbox.register("mate", crossover)
    toolbox.register("mutate", mutate)
    toolbox.register("select", tools.selTournament, tournsize=3)

    pop = toolbox.population(n=50)
    for ind in pop:
        ind.fitness.values = toolbox.evaluate(ind)

    start_time = time.time()
    best = min(pop, key=lambda x: x.fitness.values[0])
    best_fitness = best.fitness.values[0]

    for gen in range(500):
        offspring = algorithms.varAnd(pop, toolbox, cxpb=0.9, mutpb=0.1)
        fits = map(toolbox.evaluate, offspring)
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = fit
        pop = toolbox.select(pop + offspring, len(pop))
        best = min(pop, key=lambda x: x.fitness.values[0])
        best_fitness = best.fitness.values[0]
        if time.time() - start_time > budget_s - 20:
            break

    best_solution = best
    h_opt = np.array(best_solution, dtype=np.float64)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h = np.sum(h_opt)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h_opt = h_opt * scale
    c5_bound = compute_c5(h_opt)

    if time.time() - start_time < budget_s:
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.minimize(objective, h_opt, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 300})
        h_opt = result.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h = np.sum(h_opt)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_opt = h_opt * scale
        c5_bound = compute_c5(h_opt)

    return h_opt.tolist(), c5_bound, n_points
