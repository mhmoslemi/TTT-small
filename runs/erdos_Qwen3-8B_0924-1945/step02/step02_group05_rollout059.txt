import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        shift = np.correlate(h, 1 - h, mode="full")
        c5_vals = shift * dx
        return np.max(c5_vals)

    def objective(h):
        return compute_c5(h)

    bounds = [(0, 1) for _ in range(n_points)]

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def generate_patterns(n_points):
        patterns = []
        for i in range(3):
            intervals = np.random.randint(3, 7)
            indices = np.random.choice(n_points, intervals, replace=False)
            h = np.zeros(n_points)
            h[indices] = 1
            h = h / np.sum(h) * n_points / 2
            patterns.append(h)
        for i in range(2):
            slope = np.random.uniform(-0.5, 0.5)
            x = np.linspace(0, 2, n_points)
            h = 0.5 * (1 + np.tanh(slope * (x - 1)))
            h = h / np.sum(h) * n_points / 2
            patterns.append(h)
        return patterns

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    patterns = generate_patterns(n_points)
    population = [np.clip(p, 0, 1) for p in patterns]
    population = [p / np.sum(p) * n_points / 2 for p in population]
    population = [p for p in population if np.all(0 <= p <= 1)]
    population = [p for p in population if np.isclose(np.sum(p), n_points / 2, atol=1e-5)]

    for _ in range(300):
        if time() - start_time > budget_s:
            break
        idxs = np.random.choice(len(population), 5, replace=False)
        candidates = [population[i] for i in idxs]
        h1, h2, h3 = candidates
        F = 0.8
        trial = h1 + F * (h2 - h3)
        trial = np.clip(trial, 0, 1)
        trial = trial / np.sum(trial) * n_points / 2
        c5_trial = evaluate_candidate(trial)
        if c5_trial < best_c5:
            best_h = trial.copy()
            best_c5 = c5_trial
        if np.random.rand() < 0.9:
            population.append(trial)
            population.pop(np.random.randint(len(population)))

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
