```python
import numpy as np
import scipy.optimize as opt
import time
from deap import base, creator, tools, algorithms

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def generate_initial_pattern():
        if initial_h_values is not None:
            h = np.array(initial_h_values, dtype=np.float64)
        else:
            h = np.zeros(n_points, dtype=np.float64)
            for i in range(n_points):
                h[i] = 1.0 if i % 5 == 0 else 0.0
            h = h * (target_sum / np.sum(h))
        return h

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    h_initial = generate_initial_pattern()
    h_initial = h_initial * (target_sum / np.sum(h_initial))

    def fitness(individual):
        h = individual.copy()
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return (c5,)

    creator.create("FitnessMin", base.Fitness, weights=(1.0,))
    creator.create("Individual", np.ndarray, fitness=creator.FitnessMin)

    def mut_gaussian(individual, indpb):
        for i in range(len(individual)):
            if np.random.random() < indpb:
                individual[i] += np.random.normal(0, 0.1)
                individual[i] = np.clip(individual[i], 0.0, 1.0)
        return individual,

    def mut_swap(individual, indpb):
        for i in range(len(individual)):
            if np.random.random() < indpb:
                j = np.random.randint(0, len(individual))
                individual[i], individual[j] = individual[j], individual[i]
        return individual,

    def mut_blend(individual, indpb):
        for i in range(len(individual)):
            if np.random.random() < indpb:
                a = np.random.uniform(0.5, 1.5)
                individual[i] *= a
                individual[i] = np.clip(individual[i], 0.0, 1.0)
        return individual,

    def cx_two_point(ind1, ind2):
        size = len(ind1)
        cx1, cx2 = np.copy(ind1), np.copy(ind2)
        p1, p2 = np.random.randint(1, size - 1, 2)
        cx1[p1:p2], cx2[p1:p2] = cx2[p1:p2], cx1[p1:p2]
        return cx1, cx2

    def cx_uniform(ind1, ind2):
        size = len(ind1)
        ind1, ind2 = np.copy(ind1), np.copy(ind2)
        for i in range(size):
            if np.random.random() < 0.5:
                ind1[i], ind2[i] = ind2[i], ind1[i]
        return ind1, ind2

    def cx_blend(ind1, ind2):
        alpha = np.random.uniform(0.5, 1.5)
        for i in range(len(ind1)):
            ind1[i] = (ind1[i] + alpha * ind2[i]) / (1 + alpha)
            ind1[i] = np.clip(ind1[i], 0.0, 1.0)
        return ind1, ind2

    def evaluate(individual):
        return objective(individual)

    def select_top(individuals, k):
        sorted_inds = sorted(individuals, key=lambda ind: ind.fitness.values[0])
        return sorted_inds[:k]

    def mutate(individual, indpb):
        if np.random.random() < 0.3:
            return mut_gaussian(individual, indpb)
        elif np.random.random() < 0.3:
            return mut_swap(individual, indpb)
        else:
            return mut_blend(individual, indpb)

    def run_genetic_algorithm():
        population = [creator.Individual(h_initial.copy()) for _ in range(50)]
        for ind in population:
            ind.fitness.values = fitness(ind)

        toolbox = base.Toolbox()
        toolbox.register("select", tools.selTournament, tournsize=5)
        toolbox.register("mate", cx_blend)
        toolbox.register("mutate", mutate)
        toolbox.register("evaluate", evaluate)
        toolbox.register("individual", creator.Individual, creator.Individual)
        toolbox.register("population", tools.initRepeat, list, toolbox.individual)

        stats = tools.Statistics(lambda ind: ind.fitness.values)
        stats.register("avg", np.mean)
        stats.register("std", np.std)
        stats.register("min", np.min)
        stats.register("max", np.max)

        pop = toolbox.population(n=50)
        for ind in pop:
            ind.fitness.values = fitness(ind)

        for gen in range(500):
            offspring = algorithms.varAnd(pop, toolbox, cxpb=0.5, mutpb=0.1)
            fits = map(toolbox.evaluate, offspring)
            for fit, ind in zip(fits, offspring):
                ind.fitness.values = fit
            pop = toolbox.select(pop + offspring, len(pop))
            record = stats.compile(pop)
            if gen % 10 == 0:
                print(f"Generation {gen}: {record}")
            if gen >= 450:
                break

        best = select_top(pop, 1)[0]
        return best

    def refine_best(best, budget_s):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraint}]
        start_time = time.time()
        result = opt.minimize(objective, best.flatten(), method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 300})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        elapsed_time = time.time() - start_time
        if elapsed_time < budget_s:
            result = opt.minimize(objective, h_opt.flatten(), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 500})
            h_opt = result.x.reshape(n_points)
            h_opt = np.clip(h_opt, 0.0, 1.0)
            sum_h_opt = np.sum(h_opt)
            scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
            h_opt = h_opt * scale
        return h_opt

    start_time = time.time()
    best = run_genetic_algorithm()
    h_opt = refine_best(best, budget_s)
    c5_bound = objective(h_opt)
    elapsed_time = time.time() - start_time
    h_opt = h_opt.tolist()
    return h_opt, c5_bound, n_points
```
