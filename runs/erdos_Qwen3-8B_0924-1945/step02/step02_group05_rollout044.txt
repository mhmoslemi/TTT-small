import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if np.random.rand() < 0.5 else 0
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    population = [best_h.copy()]
    if 'initial_h_values' in kwargs:
        population.append(initial_h_values.copy())
    for _ in range(5):
        pattern = np.zeros(n_points)
        m = np.random.randint(3, 8)
        len_pattern = 2.0 / m
        for i in range(n_points):
            if np.random.rand() < 0.5:
                pattern[i] = 1
        pattern = pattern / np.sum(pattern) * n_points / 2
        population.append(pattern)
    for cand in population:
        cand = np.clip(cand, 0, 1)
        cand = cand / np.sum(cand) * n_points / 2
        if evaluate_candidate(cand) < best_c5:
            best_h = cand.copy()
            best_c5 = evaluate_candidate(cand)
    for _ in range(100):
        if time() - start_time > budget_s:
            break
        x0 = best_h.copy()
        res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5, options={'maxiter': 100})
        if res.success:
            current_h = res.x
            current_c5 = evaluate_candidate(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            break
    if time() - start_time < budget_s:
        for _ in range(200):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
