import numpy as np
from scipy.optimize import differential_evolution, Bounds
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if np.random.rand() < 0.5 else 0
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(h):
        return evaluate_candidate(h)

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]

    start_time = time()

    def generate_structured_patterns():
        patterns = []
        for _ in range(5):
            if 'initial_h_values' in kwargs:
                h = initial_h_values.copy()
            else:
                h = np.zeros(n_points)
                for i in range(n_points):
                    h[i] = 1 if np.random.rand() < 0.5 else 0
                h = np.clip(h, 0, 1)
                h = h / np.sum(h) * n_points / 2
            patterns.append(h)
        return patterns

    structured_patterns = generate_structured_patterns()
    population = structured_patterns

    for _ in range(300):
        if time() - start_time > budget_s:
            break
        new_population = []
        for i in range(len(population)):
            idxs = np.random.choice(len(population), 3, replace=False)
            h1, h2, h3 = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            h = h1 + 0.8 * (h2 - h3)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * n_points / 2
            new_population.append(h)
        population = new_population

    best_h = min(population, key=evaluate_candidate)
    best_c5 = evaluate_candidate(best_h)

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
