import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy() if 'initial_h_values' in kwargs else np.random.uniform(0, 1, n_points).clip(0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    bounds = [(0, 1) for _ in range(n_points)]

    def objective(x):
        h = np.clip(x, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    start_time = time()
    result = differential_evolution(objective, bounds, strategy='best1bin', 
                                    popsize=15, mutation=0.5, recombination=0.8, 
                                    tol=1e-5, maxiter=500, disp=False, polish=False)

    best_h = result.x
    best_c5 = result.fun
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = compute_c5(best_h)

    return (best_h, current_c5, n_points)
