import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = kwargs.get('initial_h_values', None)
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time()

    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    def differential_evolution():
        pop_size = 30
        mut_factor = 0.5
        cross_prob = 0.8
        max_iter = 300
        pop = np.random.uniform(0, 1, (pop_size, n_points))
        pop = pop / np.sum(pop, axis=1, keepdims=True) * n_points / 2
        best_pop = pop.copy()
        best_fitness = np.inf
        for i in range(max_iter):
            if time() - start_time > budget_s:
                break
            mutant = np.zeros((pop_size, n_points))
            for j in range(pop_size):
                a, b, c = np.random.choice(pop_size, 3, replace=False)
                mutant[j] = pop[a] + mut_factor * (pop[b] - pop[c])
                mutant[j] = np.clip(mutant[j], 0, 1)
                mutant[j] = mutant[j] / np.sum(mutant[j]) * n_points / 2
            trial = np.zeros((pop_size, n_points))
            for j in range(pop_size):
                for k in range(n_points):
                    if np.random.rand() < cross_prob:
                        trial[j][k] = mutant[j][k]
                    else:
                        trial[j][k] = pop[j][k]
                trial[j] = trial[j] / np.sum(trial[j]) * n_points / 2
            fitness_pop = np.array([evaluate_candidate(ind) for ind in pop])
            fitness_trial = np.array([evaluate_candidate(ind) for ind in trial])
            for j in range(pop_size):
                if fitness_trial[j] < fitness_pop[j]:
                    pop[j] = trial[j].copy()
                    if fitness_trial[j] < best_fitness:
                        best_fitness = fitness_trial[j]
                        best_pop = pop[j].copy()
        return best_pop, best_fitness

    best_h, best_c5 = differential_evolution()

    def local_refinement(x0):
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            return res.x
        else:
            return x0

    best_h = local_refinement(best_h)
    best_c5 = evaluate_candidate(best_h)

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    final_c5 = evaluate_candidate(best_h)
    return (best_h, final_c5, n_points)
