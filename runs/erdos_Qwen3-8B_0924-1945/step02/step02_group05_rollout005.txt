import numpy as np
from scipy.optimize import minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def update_h(h, max_k, step_size, active_range):
        dx = 2.0 / n_points
        corr_vals = np.correlate(h, 1 - h, mode="full")
        g = np.zeros_like(h)
        for x in range(*active_range):
            k = x + max_k
            if k < len(corr_vals):
                g[x] += (1 - h[k])
                if x + max_k < len(h):
                    g[x + max_k] -= h[x]
        h -= step_size * g
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return h

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    max_iter = 1000
    iter_count = 0
    step_size = 0.01

    while iter_count < max_iter and time() - start_time < budget_s:
        corr_vals = np.correlate(best_h, 1 - best_h, mode="full") * 2.0 / n_points
        max_k = np.argmax(corr_vals)
        active_start = max(0, max_k - 10)
        active_end = min(n_points - 1, max_k + 10)
        best_h = update_h(best_h, max_k, step_size, (active_start, active_end))
        current_c5 = evaluate_candidate(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
        iter_count += 1

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
