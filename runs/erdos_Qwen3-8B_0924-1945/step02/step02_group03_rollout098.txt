import numpy as np
from scipy.optimize import minimize, Bounds
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else None

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def create_initial_h():
        if initial_h_values is not None:
            h = initial_h_values.copy()
        else:
            h = np.random.uniform(0, 1, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * n_points / 2
        return h

    def mutate(h, mutation_rate=0.5):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        h = h / np.sum(h) * n_points / 2
        return h

    def crossover(parent1, parent2):
        child = parent1.copy()
        for i in range(n_points):
            if np.random.rand() < 0.8:
                child[i] = parent2[i]
        return child

    def optimize():
        pop_size = 30
        population = [create_initial_h() for _ in range(pop_size)]
        best_h = population[0]
        best_c5 = evaluate_candidate(best_h)
        for _ in range(300):
            if time() - start_time > budget_s:
                break
            mutants = []
            for i in range(pop_size):
                a, b, c = np.random.choice(population, 3, replace=False)
                mutant = a + mutation_rate * (b - c)
                mutant = np.clip(mutant, 0, 1)
                mutant = mutant / np.sum(mutant) * n_points / 2
                mutants.append(mutant)
            population += mutants
            population = np.array(population)
            population = np.unique(population, axis=0)
            if len(population) > pop_size:
                population = population[:pop_size]
            for i in range(pop_size):
                if np.random.rand() < 0.5:
                    population[i] = crossover(population[i], population[np.random.randint(pop_size)])
            for h in population:
                current_c5 = evaluate_candidate(h)
                if current_c5 < best_c5:
                    best_h = h
                    best_c5 = current_c5
        return best_h, best_c5

    def refine_h(h):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        res = minimize(fitness, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            return res.x
        else:
            return h

    start_time = time()
    best_h, best_c5 = optimize()
    best_h = refine_h(best_h)
    final_c5 = evaluate_candidate(best_h)
    return (best_h, final_c5, n_points)
