import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2 if np.sum(h) > 0 else np.random.uniform(0, 1, n_points)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h_vals):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h_vals, 1 - h_vals, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals / np.sum(h_vals) * n_points / 2 if np.sum(h_vals) > 0 else np.random.uniform(0, 1, n_points)
        h_vals = h_vals / np.sum(h_vals) * n_points / 2
        return compute_c5(h_vals)

    bounds = [(0, 1) for _ in range(n_points)]

    def obj_func(x):
        return evaluate_candidate(x)

    start_time = time()
    result = differential_evolution(obj_func, bounds, strategy='best1bin', popsize=15, mutation=(0.8, 1.0), recombination=0.9, tol=1e-5, maxiter=300, callback=lambda x: None, disp=False)
    best_h = result.x
    best_c5 = result.fun
    elapsed = time() - start_time

    if elapsed < budget_s - 50:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h_rand = np.random.uniform(0, 1, n_points)
            h_rand = h_rand / np.sum(h_rand) * n_points / 2
            current_c5 = evaluate_candidate(h_rand)
            if current_c5 < best_c5:
                best_h = h_rand
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    best_c5 = evaluate_candidate(best_h)
    return (best_h, best_c5, n_points)
