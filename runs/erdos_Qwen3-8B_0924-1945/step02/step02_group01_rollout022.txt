import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points_options = [200, 250, 300]
    n_points = n_points_options[0]  # Default
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr * dx)
        return max_corr

    def rescale(h):
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
        return h

    def generate_initial_population(pop_size, n_points):
        population = []
        for _ in range(pop_size):
            h = np.zeros(n_points, dtype=np.float64)
            ones = np.random.choice(n_points, size=int(n_points / 2), replace=False)
            h[ones] = 1.0
            h = rescale(h)
            population.append(h)
        return np.array(population)

    def tournament_selection(population, fitnesses, tournament_size=5):
        participants = np.random.choice(len(population), tournament_size, replace=False)
        best_idx = participants[np.argmin(fitnesses[participants])]
        return population[best_idx]

    def crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        for i in range(len(parent1)):
            if np.random.rand() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child

    def mutate(individual, mutation_rate=0.1):
        for i in range(len(individual)):
            if np.random.rand() < mutation_rate:
                individual[i] += np.random.normal(0, 0.1)
        return rescale(individual)

    def evolve_population(population, fitnesses, pop_size, n_points):
        new_population = []
        for _ in range(pop_size // 2):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1 = crossover(parent1, parent2)
            child1 = mutate(child1)
            new_population.append(child1)
            child2 = crossover(parent1, parent2)
            child2 = mutate(child2)
            new_population.append(child2)
        return np.array(new_population)

    def run_evolution(n_points, budget_s):
        pop_size = 50
        population = generate_initial_population(pop_size, n_points)
        fitnesses = np.array([evaluate(h) for h in population])
        best_fitness = np.inf
        best_h = population[0]
        start_time = time.time()
        
        for gen in range(200):
            if time.time() - start_time > budget_s - 100:
                break
            population = evolve_population(population, fitnesses, pop_size, n_points)
            fitnesses = np.array([evaluate(h) for h in population])
            current_best = np.min(fitnesses)
            if current_best < best_fitness:
                best_fitness = current_best
                best_h = population[np.argmin(fitnesses)]
        
        return rescale(best_h), best_fitness

    h_opt, c5_bound = run_evolution(n_points, budget_s)
    return h_opt.tolist(), c5_bound, n_points
