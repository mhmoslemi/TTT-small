```python
import numpy as np
import scipy.optimize as opt
import time
import random
from deap import base, creator, tools, algorithms

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def generate_initial_pattern():
        if initial_h_values is not None:
            h = np.array(initial_h_values, dtype=np.float64)
            h = np.random.permutation(h)
            h = h * (target_sum / np.sum(h))
            return h
        else:
            h = np.zeros(n_points, dtype=np.float64)
            for i in range(n_points):
                h[i] = 1.0 if i % 5 == 0 else 0.0
            h = h * (target_sum / np.sum(h))
            return h

    def evaluate_individual(individual):
        h = np.array(individual, dtype=np.float64)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5,

    def mutate(individual, indpb):
        for i in range(len(individual)):
            if random.random() < indpb:
                individual[i] = 1.0 - individual[i]
        return individual,

    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(len(parent1)):
            if random.random() < 0.5:
                child[i] = parent2[i]
        return child,

    def repair(individual):
        h = np.array(individual, dtype=np.float64)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        return h.tolist(),

    def create_individual():
        return [random.choice([0.0, 1.0]) for _ in range(n_points)]

    def run_evolutionary_search():
        h_base = generate_initial_pattern()
        initial_population = [h_base]
        for _ in range(4):
            new_pop = []
            for _ in range(50):
                new_pop.append(create_individual())
            initial_population += new_pop
        initial_population = [repair(ind) for ind in initial_population]
        initial_population = list(set(initial_population))
        initial_population = initial_population[:50]
        initial_population = [repair(ind) for ind in initial_population]

        creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
        creator.create("Individual", list, fitness=creator.FitnessMin)

        toolbox = base.Toolbox()
        toolbox.register("individual", tools.initIterate, creator.Individual, lambda: [random.choice([0.0, 1.0]) for _ in range(n_points)])
        toolbox.register("population", tools.initRepeat, list, toolbox.individual)
        toolbox.register("mate", crossover)
        toolbox.register("mutate", mutate, indpb=0.1)
        toolbox.register("select", tools.selTournament, tournsize=3)
        toolbox.register("evaluate", evaluate_individual)

        pop = toolbox.population(n=50)
        for ind in pop:
            ind.fitness.values = toolbox.evaluate(ind)
        for ind in initial_population:
            ind.fitness.values = toolbox.evaluate(ind)
        pop = initial_population + pop

        stats = tools.Statistics(lambda ind: ind.fitness.values)
        stats.register("avg", np.mean)
        stats.register("std", np.std)
        stats.register("min", np.min)
        stats.register("max", np.max)

        algorithms.eaSimple(pop, toolbox, cxpb=0.5, mutpb=0.1, ngen=200, stats=stats, halloffame=tools.HallOfFame(1), verbose=False)

        best = tools.HallOfFame(1)[0]
        best = repair(best)
        return best

    def run_local_refinement(h):
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            corr = np.correlate(h, 1.0 - h, mode="full")
            c5 = np.max(corr * dx)
            return c5

        def constraint(h_flat):
            h = h_flat.reshape(n_points)
            sum_h = np.sum(h)
            return sum_h - target_sum

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraint}]
        result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 500})
        h_opt = result.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        return h_opt

    start_time = time.time()
    h = run_evolutionary_search()
    if time.time() - start_time < budget_s:
        h = run_local_refinement(h)
    c5_bound = evaluate_individual(h)
    return h.tolist(), c5_bound[0], n_points
```
