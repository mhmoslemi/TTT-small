import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    
    def generate_patterns():
        patterns = []
        for m in [3, 4, 5, 6, 7]:
            intervals = [np.linspace(0, 2, m + 1)]
            for i in range(1, m + 1):
                intervals.append(np.linspace(2 - i * 2 / m, 2 - (i - 1) * 2 / m, m + 1))
            for start, end in intervals:
                mask = (np.array([start, end]) <= np.linspace(0, 2, n_points))
                h = np.zeros(n_points)
                h[np.logical_and(mask[0], ~mask[1])] = 1
                h = h / np.sum(h) * n_points / 2
                patterns.append(h)
        return patterns
    
    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def objective(x):
        return evaluate_candidate(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    population = [initial_h_values.copy()]
    for p in generate_patterns():
        population.append(p)
    
    best_h = initial_h_values.copy()
    best_c5 = evaluate_candidate(best_h)
    
    start_time = time()
    
    def optimize_candidate(h):
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            return res.x
        return h
    
    for _ in range(100):
        if time() - start_time > budget_s:
            break
        h = population[np.random.randint(len(population))]
        h = optimize_candidate(h)
        c5 = evaluate_candidate(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    for _ in range(50):
        if time() - start_time > budget_s:
            break
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
        c5 = evaluate_candidate(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    final_c5 = evaluate_candidate(best_h)
    return (best_h, final_c5, n_points)
