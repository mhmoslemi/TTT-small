```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        h = 0.5 + 0.25 * np.sin(np.pi * x_grid) + 0.25 * np.cos(2 * np.pi * x_grid)
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        h = h * (target_sum / sum_h if sum_h != 0 else 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()

    try:
        # Differential evolution for global exploration
        result = opt.differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=20,
            mutation=(0.5, 1.0),
            recombination=0.8,
            tol=1e-5,
            maxiter=100,
            disp=False,
            polish=True,
            constraints=cons,
            seed=seed
        )
        best_h = result.x
        best_c5 = objective(best_h)
        if time.time() - start_time > budget_s - 60:
            pass
        else:
            # Local refinement with SLSQP
            result = opt.minimize(
                objective,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                tol=1e-8,
                options={'maxiter': 50}
            )
            best_h = result.x
            best_c5 = objective(best_h)
    except:
        pass

    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    best_h = best_h * (target_sum / sum_h if sum_h != 0 else 1.0)
    best_c5 = objective(best_h)

    return best_h.tolist(), best_c5, n_points
```
