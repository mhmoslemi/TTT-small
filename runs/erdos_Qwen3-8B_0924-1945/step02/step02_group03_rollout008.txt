import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 60
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    # Genetic Algorithm Phase
    population_size = 50
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h / np.sum(h) * n_points / 2 for h in population]
    population = [np.clip(p, 0, 1) for p in population]
    population = [evaluate_candidate(p) for p in population]
    for _ in range(500):
        if time() - start_time > budget_s:
            break
        # Tournament Selection
        indices = np.random.choice(range(population_size), size=5, replace=False)
        best_idx = indices[np.argmin([population[i] for i in indices])]
        parent = population[best_idx]
        # Mutation
        mutation_rate = 0.2
        if np.random.rand() < mutation_rate:
            h_mut = parent + np.random.uniform(-0.1, 0.1, n_points)
        else:
            h_mut = parent
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut / np.sum(h_mut) * n_points / 2
        h_mut_c5 = compute_c5(h_mut)
        # Replace worst
        worst_idx = np.argmax(population)
        population[worst_idx] = h_mut_c5
        population[worst_idx] = h_mut
        # Track best
        if h_mut_c5 < best_c5:
            best_h = h_mut.copy()
            best_c5 = h_mut_c5

    # Local Refinement Phase
    for _ in range(100):
        if time() - start_time > budget_s:
            break
        x0 = best_h.copy()
        res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7)
        if res.success:
            current_h = res.x
            current_c5 = evaluate_candidate(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            break

    # Final Validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
