```python
import numpy as np
import time
import random

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def rescale(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        return h * scale

    pop_size = 50
    tournament_size = 5
    mutation_rate = 0.1
    crossover_prob = 0.8
    elitism_rate = 0.2
    generations = 200
    max_time = budget_s

    population = [rescale(np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])) for _ in range(pop_size)]
    best_h = population[0]
    best_c5 = evaluate(best_h)

    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > max_time:
            break

        # Evaluate fitness
        fitness = [evaluate(ind) for ind in population]
        fitness = np.array(fitness)

        # Tournament selection
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(range(pop_size), size=tournament_size)
            winner = tournament[np.argmin(fitness[tournament])]
            selected.append(population[winner])

        # Crossover
        offspring = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < pop_size else selected[i]
            if np.random.rand() < crossover_prob:
                child1, child2 = np.random.choice([0, 1], size=(2, n_points))
                child1 = np.where(child1 == 1, parent1, child2)
                child2 = np.where(child2 == 1, parent2, child1)
            else:
                child1 = parent1
                child2 = parent2
            offspring.extend([child1, child2])

        # Mutation
        for i in range(len(offspring)):
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    offspring[i][j] = 1 - offspring[i][j]

        # Rescale offspring
        offspring = [rescale(ind) for ind in offspring]

        # Elitism
        sorted_indices = np.argsort(fitness)
        new_population = [population[i] for i in sorted_indices[:int(pop_size * elitism_rate)]]
        new_population += offspring[:pop_size - len(new_population)]

        population = new_population
        current_c5 = evaluate(population[0])
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[0]

    best_h = rescale(best_h)
    best_c5 = evaluate(best_h)
    return best_h.tolist(), best_c5, n_points
```
