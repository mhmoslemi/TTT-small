import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.clip(h, 0.0, 1.0)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        h[::5] = 1.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()

    def optimize_with_method(method, maxiter=500):
        result = opt.minimize(objective, h.flatten(), method=method, bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': maxiter})
        if result.success:
            return result.x, result.fun
        else:
            return h.flatten(), np.inf

    result1 = optimize_with_method('L-BFGS-B', maxiter=300)
    if result1[1] < np.inf:
        h_opt = result1[0]
    else:
        h_opt = h.flatten()

    elapsed = time.time() - start_time
    if elapsed < budget_s - 200:
        result2 = optimize_with_method('SLSQP', maxiter=200)
        if result2[1] < np.inf:
            h_opt = result2[0]
        else:
            h_opt = h_opt

    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h = np.sum(h_opt)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h_opt = h_opt * scale

    if elapsed < budget_s - 30:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        pop_size = 100
        tol = 1e-5
        maxiter = 200
        def de_objective(x):
            return objective(x)
        def de_constraint(x):
            return constraint(x)
        cons_de = [{'type': 'eq', 'fun': de_constraint}]
        result = opt.differential_evolution(de_objective, bounds, constraints=cons_de, strategy='best1bin', maxiter=maxiter, popsize=pop_size, tol=tol, mutation=(0.8, 1.0), recombination=0.9, polish=True, disp=False)
        h_opt = result.x

    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h = np.sum(h_opt)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h_opt = h_opt * scale

    c5_bound = objective(h_opt)
    return h_opt.tolist(), c5_bound, n_points
