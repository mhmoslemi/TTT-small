import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def perturb(h, scale):
        return np.clip(h + np.random.normal(0, scale, size=h.shape), 0.0, 1.0)

    def objective_wrapper(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    start_time = time.time()

    h_perturbed = perturb(h, 0.03)
    h_perturbed = h_perturbed * (target_sum / np.sum(h_perturbed))
    h_opt = h_perturbed.copy()

    def get_candidates(h, num_candidates=100):
        candidates = [h.copy()]
        for _ in range(num_candidates - 1):
            new_h = perturb(h, 0.03)
            new_h = new_h * (target_sum / np.sum(new_h))
            candidates.append(new_h)
        return candidates

    candidates = get_candidates(h_opt)

    def evaluate_candidate(h):
        h = h.copy()
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5, h

    best_c5 = float('inf')
    best_h = h_opt.copy()

    for candidate in candidates:
        c5, h = evaluate_candidate(candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    h_opt = best_h

    if time.time() - start_time < budget_s * 0.5:
        result = opt.minimize(objective_wrapper, h_opt.flatten(), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 300})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale

    if time.time() - start_time < budget_s * 0.8:
        result = opt.minimize(objective_wrapper, h_opt.flatten(), method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 300})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale

    c5_bound = objective(h_opt.flatten())
    return h_opt.tolist(), c5_bound, n_points
