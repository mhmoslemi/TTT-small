import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    h = np.clip(initial_h_values, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def objective(h):
        return compute_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        x0 = best_h.copy()
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            current_h = res.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        else:
            break
    
    if time.time() - start_time < budget_s:
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    return (best_h, best_c5, n_points)
