import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    if 'initial_h_values' in kwargs:
        initial_h = kwargs['initial_h_values']
    else:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if np.random.rand() < 0.5 else 0
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h / np.sum(initial_h) * n_points / 2
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h, n_points)
    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        def objective(x):
            return compute_c5(x, n_points)
        def constraint(x):
            return np.sum(x) - n_points / 2
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraint}]
        x0 = best_h.copy()
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            current_h = res.x
            current_c5 = compute_c5(current_h, n_points)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        perturb = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
        current_c5 = compute_c5(perturbed_h, n_points)
        if current_c5 < best_c5:
            best_h = perturbed_h.copy()
            best_c5 = current_c5
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    return (best_h, best_c5, n_points)

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(correlate_vals)
