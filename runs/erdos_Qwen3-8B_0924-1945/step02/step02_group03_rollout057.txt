import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time()

    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    population_size = 50
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h / np.sum(h) * n_points / 2 for h in population]
    population_fitness = [evaluate_candidate(h) for h in population]
    best_idx = np.argmin(population_fitness)

    for _ in range(500):
        if time() - start_time > budget_s:
            break
        parents = np.random.choice(population_size, size=2, replace=False)
        parent1, parent2 = population[parents[0]], population[parents[1]]
        child = (parent1 + np.random.rand(n_points) * (parent2 - parent1)) / 2
        child = np.clip(child, 0, 1)
        child = child / np.sum(child) * n_points / 2
        current_c5 = evaluate_candidate(child)
        if current_c5 < population_fitness[best_idx]:
            population[best_idx] = child
            population_fitness[best_idx] = current_c5
            best_idx = np.argmin(population_fitness)
        if np.min(population_fitness) < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = population_fitness[best_idx]

    top_candidates = np.argsort(population_fitness)[:5]
    for idx in top_candidates:
        h = population[idx].copy()
        res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            current_h = res.x
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h / np.sum(current_h) * n_points / 2
            current_c5 = evaluate_candidate(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
