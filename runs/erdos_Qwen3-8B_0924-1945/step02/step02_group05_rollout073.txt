import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = np.clip(initial_h_values, 0, 1)
    h = h / np.sum(h) * n_points / 2 if np.sum(h) > 0 else np.random.uniform(0, 1, n_points)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)
    
    def objective(h):
        return evaluate_candidate(h)

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()

    def structured_initializations():
        patterns = []
        # Uniform interval patterns
        for length in range(3, 8):
            intervals = np.zeros(n_points)
            for i in range(length):
                idx = np.random.randint(0, n_points - length + 1)
                intervals[idx:idx+length] = 1
            patterns.append(intervals)
        # Symmetric bilinear
        slopes = np.random.uniform(-0.5, 0.5, 5)
        for slope in slopes:
            x = np.linspace(0, 2, n_points)
            h = 0.5 * (1 + np.tanh(slope * (x - 1)))
            h = h / np.sum(h) * n_points / 2
            patterns.append(h)
        # Random binary
        for _ in range(5):
            h = np.random.choice([0, 1], n_points)
            h = h / np.sum(h) * n_points / 2
            patterns.append(h)
        # Lanczos interpolation
        def lanczos(x, n):
            x = np.array(x)
            x = (x * n - 1) / (n - 1)
            x = np.clip(x, -1, 1)
            return 0.5 * (1 + np.sinc(x) * np.sinc(x / 2))
        x = np.linspace(0, 2, n_points)
        h = lanczos(x, 2)
        h = h / np.sum(h) * n_points / 2
        patterns.append(h)
        return patterns

    structured_patterns = structured_initializations()
    population = [np.clip(p, 0, 1) for p in structured_patterns]
    population = [p / np.sum(p) * n_points / 2 for p in population]
    best_h = population[0]
    best_c5 = evaluate_candidate(best_h)

    def mutation(h, F=0.8):
        idxs = np.random.choice(len(h), 3, replace=False)
        h1, h2, h3 = h[idxs[0]], h[idxs[1]], h[idxs[2]]
        trial = h1 + F * (h2 - h3)
        trial = np.clip(trial, 0, 1)
        trial = trial / np.sum(trial) * n_points / 2
        return trial

    def evaluate_and_compare(h):
        nonlocal best_c5, best_h
        c5 = evaluate_candidate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    for gen in range(300):
        if time() - start_time > budget_s:
            break
        for i in range(len(population)):
            h = population[i]
            trial = mutation(h)
            evaluate_and_compare(trial)
            if np.random.rand() < 0.9:
                population[i] = trial
            else:
                population[i] = h

    for _ in range(100):
        if time() - start_time > budget_s:
            break
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
        c5 = evaluate_candidate(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    final_c5 = evaluate_candidate(best_h)
    return (best_h, final_c5, n_points)
