```python
import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    if 'initial_h_values' in kwargs:
        h = kwargs['initial_h_values'].copy()
    else:
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def objective(x):
        x = np.clip(x, 0, 1)
        x = x / np.sum(x) * n_points / 2
        return compute_c5(x)

    bounds = [(0, 1) for _ in range(n_points)]
    start_time = time()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-5,
        maxiter=500,
        seed=seed,
        disp=False
    )

    best_h = result.x
    best_c5 = compute_c5(best_h)

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * n_points / 2
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    return (best_h, best_c5, n_points)
```
