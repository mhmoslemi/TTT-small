import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if np.random.rand() < 0.5 else 0
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    def run_optimization():
        nonlocal best_h, best_c5
        for _ in range(1000):
            if time() - start_time > budget_s:
                break
            corr_vals = np.correlate(best_h, 1 - best_h, mode="full")
            max_corr_idx = np.argmax(corr_vals)
            max_corr_val = corr_vals[max_corr_idx]
            if max_corr_val <= best_c5:
                break
            if max_corr_idx < 0 or max_corr_idx >= len(corr_vals):
                break
            dx = 2.0 / n_points
            max_corr = corr_vals[max_corr_idx] * dx
            if max_corr <= best_c5:
                break
            lags = np.arange(len(corr_vals))
            top_lags = lags[np.argsort(corr_vals)[::-1][:5]]
            top_lags = np.unique(top_lags[np.isfinite(corr_vals[top_lags])])
            for lag in top_lags:
                if lag < 0 or lag >= len(corr_vals):
                    continue
                corr = corr_vals[lag]
                if corr <= best_c5:
                    continue
                start_idx = max(0, lag - 10)
                end_idx = min(n_points - 1, lag + 10)
                grad = np.zeros_like(best_h)
                for i in range(start_idx, end_idx + 1):
                    j = i - lag
                    if 0 <= j < n_points:
                        grad[i] += (1 - best_h[j])
                        grad[j] -= best_h[i]
                grad_norm = np.linalg.norm(grad)
                if grad_norm < 1e-5:
                    continue
                step_size = 0.05 * (1 - (max_corr - best_c5) / (max_corr - 0.3))
                step_size = np.clip(step_size, 0.001, 0.1)
                h_adjust = best_h - step_size * grad
                h_adjust = np.clip(h_adjust, 0, 1)
                h_adjust = h_adjust / np.sum(h_adjust) * n_points / 2
                current_c5 = compute_c5(h_adjust)
                if current_c5 < best_c5:
                    best_h = h_adjust.copy()
                    best_c5 = current_c5
        return best_h, best_c5

    best_h, best_c5 = run_optimization()

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
