import numpy as np
from scipy.optimize import minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy() if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2 if np.sum(h) > 0 else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_h(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_h(x)

    bounds = [(0, 1) for _ in range(n_points)]
    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h)

    for _ in range(200):
        if time() - start_time >= budget_s:
            break
        perturbation = np.random.normal(0, 0.1, n_points)
        candidate = h + perturbation
        candidate = np.clip(candidate, 0, 1)
        candidate = candidate / np.sum(candidate) * n_points / 2
        current_c5 = evaluate_h(candidate)
        if current_c5 < best_c5:
            best_h = candidate.copy()
            best_c5 = current_c5

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time >= budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_h(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    return (best_h, best_c5, n_points)
