```python
import numpy as np
import scipy.optimize as opt
import time
import random
from deap import base, creator, tools, algorithms

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    initial_patterns = []
    initial_patterns.append(h.copy())
    np.random.shuffle(h)
    initial_patterns.append(h.copy())
    h = np.zeros(n_points)
    h[:int(n_points / 2)] = 1.0
    initial_patterns.append(h.copy())
    h = np.zeros(n_points)
    h[::5] = 1.0
    initial_patterns.append(h.copy())
    h = np.zeros(n_points)
    h[::10] = 1.0
    initial_patterns.append(h.copy())
    h = np.zeros(n_points)
    h[::20] = 1.0
    initial_patterns.append(h.copy())

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    def evaluate(individual):
        nonlocal best_c5, best_h
        individual = np.clip(individual, 0.0, 1.0)
        sum_h = np.sum(individual)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = individual * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        c5 = np.max(corr * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_scaled.copy()
        return c5,

    for pattern in initial_patterns:
        pattern = np.clip(pattern, 0.0, 1.0)
        sum_h = np.sum(pattern)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_scaled = pattern * scale
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode="full")
        c5 = np.max(corr * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_scaled.copy()

    def fitness(individual):
        return evaluate(individual)

    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", np.ndarray, fitness=creator.FitnessMin)

    toolbox = base.Toolbox()
    toolbox.register("attr_float", lambda: np.random.rand())
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)

    toolbox.register("evaluate", fitness)
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutPolynomialBounded, eta=0.5, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=5)

    population = toolbox.population(n=50)
    algorithms.eaMuPlusLambda(population, toolbox, μ=50, λ=50, cxpb=0.9, mutpb=0.1, ngen=500, stats=None, halting_condition=None, verbose=False)

    final_h = best_h.copy()
    final_h = np.clip(final_h, 0.0, 1.0)
    sum_h = np.sum(final_h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    final_h = final_h * scale
    c5_bound = objective(final_h.flatten())

    return final_h.tolist(), c5_bound, n_points
```
