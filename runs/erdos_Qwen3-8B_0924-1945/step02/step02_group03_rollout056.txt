import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = kwargs.get('initial_h_values', None)
    if initial_h_values is not None:
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
    else:
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    population_size = 50
    population = [h.copy() for _ in range(population_size)]
    for _ in range(500):
        if time() - start_time > budget_s:
            break
        parents = np.random.choice(population_size, size=5, replace=False)
        parent = population[np.random.choice(parents)]
        child = parent.copy()
        for i in range(n_points):
            if np.random.rand() < 0.2:
                child[i] += np.random.uniform(-0.1, 0.1)
                child[i] = np.clip(child[i], 0, 1)
        child = np.clip(child, 0, 1)
        child = child / np.sum(child) * n_points / 2
        child_c5 = evaluate_candidate(child)
        if child_c5 < best_c5:
            best_h = child.copy()
            best_c5 = child_c5
        population.append(child)
        population.pop(np.random.choice(range(population_size)))

    for _ in range(50):
        if time() - start_time > budget_s:
            break
        x0 = best_h.copy()
        res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            current_h = res.x
            current_c5 = evaluate_candidate(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            break

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
