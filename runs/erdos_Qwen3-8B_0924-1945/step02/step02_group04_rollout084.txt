import numpy as np
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = np.clip(initial_h_values, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return compute_c5(h)
    
    # Particle Swarm Optimization parameters
    swarm_size = 30
    swarm = [h.copy() for _ in range(swarm_size)]
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    pbest = [s.copy() for s in swarm]
    gbest = h.copy()
    best_c5 = compute_c5(gbest)
    
    for _ in range(1000):
        if time.time() > time.time() + budget_s:
            break
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = 0.8 * velocities[i] + 1.5 * r1 * (pbest[i] - swarm[i]) + 1.5 * r2 * (gbest - swarm[i])
            swarm[i] = swarm[i] + velocities[i]
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            current_c5 = compute_c5(swarm[i])
            if current_c5 < compute_c5(pbest[i]):
                pbest[i] = swarm[i].copy()
            if current_c5 < best_c5:
                gbest = swarm[i].copy()
                best_c5 = current_c5
    
    # Fallback to initial if no improvement in last 50 iterations
    if best_c5 >= compute_c5(initial_h_values):
        gbest = initial_h_values.copy()
        gbest = np.clip(gbest, 0, 1)
        gbest = gbest * (n_points / 2) / np.sum(gbest)
        best_c5 = compute_c5(gbest)
    
    best_h = gbest
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    return (best_h, best_c5, n_points)
