import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    if time() - start_time < budget_s:
        def de_fitness(x):
            return evaluate_candidate(x)

        result = differential_evolution(
            de_fitness, bounds, strategy='best1bin', popsize=30, maxiter=300,
            tol=1e-6, mutation=(0.5, 1), recombination=0.8, disp=False
        )
        best_h = result.x
        best_c5 = evaluate_candidate(best_h)

    if time() - start_time < budget_s:
        result = minimize(
            objective, best_h, method='SLSQP', bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum}, tol=1e-5
        )
        if result.success:
            best_h = result.x
            best_c5 = evaluate_candidate(best_h)

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
