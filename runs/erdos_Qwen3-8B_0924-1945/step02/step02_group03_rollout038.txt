import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2 if np.sum(h) > 0 else np.random.uniform(0, 1, n_points)

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2 if np.sum(h) > 0 else np.random.uniform(0, 1, n_points)
        return compute_c5(h)

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    result = differential_evolution(
        lambda x: evaluate_candidate(x),
        bounds,
        strategy='best1bin',
        popsize=15,
        mutation=0.5,
        recombination=0.8,
        tol=1e-8,
        maxiter=500,
        polish=True,
        disp=False
    )

    if time() - start_time < budget_s:
        best_h = result.x
        best_c5 = evaluate_candidate(best_h)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
