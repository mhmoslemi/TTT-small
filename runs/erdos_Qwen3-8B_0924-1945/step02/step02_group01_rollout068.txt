import numpy as np
import random
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    target_sum = n_points / 2
    pop_size = 50
    tournament_size = 5
    mutation_rate = 0.1
    crossover_prob = 0.8
    elitism_rate = 0.2
    max_generations = 200
    early_stop_gen = 20
    mutation_std = 0.1
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h = h * (target_sum / np.sum(h))
        best_h = h.copy()

    def evaluate(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr * dx)
        return max_corr

    def create_individual():
        indices = np.random.choice(n_points, size=int(n_points / 2), replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        h = h * (target_sum / np.sum(h))
        return h

    population = [create_individual() for _ in range(pop_size)]

    for gen in range(max_generations):
        if elapsed_time >= budget_s:
            break
        start_time = time.time()

        # Evaluate fitness
        fitness = np.array([evaluate(h) for h in population])
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]

        if best_c5 < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = best_c5

        # Selection
        parents = []
        for _ in range(pop_size // 2):
            tournament = np.random.choice(pop_size, tournament_size)
            winner = tournament[np.argmin(fitness[tournament])]
            parents.append(population[winner])

        # Crossover
        offspring = []
        for i in range(0, len(parents), 2):
            if np.random.rand() < crossover_prob:
                p1, p2 = parents[i], parents[i + 1]
                child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
            else:
                child = parents[i].copy()
            offspring.append(child)

        # Mutation
        for child in offspring:
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[j] += np.random.normal(0, mutation_std)
                    child[j] = np.clip(child[j], 0.0, 1.0)
            child = child * (target_sum / np.sum(child))
            offspring.append(child)

        # Combine and select
        population = parents + offspring
        fitness = np.array([evaluate(h) for h in population])
        sorted_indices = np.argsort(fitness)
        population = np.array(population)[sorted_indices]
        population = population[:pop_size]

        # Elitism
        num_elite = int(pop_size * elitism_rate)
        population = population[:num_elite] + np.random.choice(population, size=pop_size - num_elite)
