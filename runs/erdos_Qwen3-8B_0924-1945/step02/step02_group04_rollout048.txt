import numpy as np
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    h = np.clip(initial_h_values, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return compute_c5(h)
    
    def update_swarm(h, pbest, gbest, v, omega, c1, c2):
        r1, r2 = np.random.rand(2)
        v = omega * v + c1 * r1 * (pbest - h) + c2 * r2 * (gbest - h)
        h = h + v
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h, v
    
    bounds = [(0, 1) for _ in range(n_points)]
    swarm_size = 30
    particles = [np.random.uniform(0, 1, n_points) * (n_points / 2) / np.sum(np.random.uniform(0, 1, n_points)) for _ in range(swarm_size)]
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    pbest = particles.copy()
    gbest = particles[0]
    best_c5 = evaluate_candidate(pbest[0])
    
    start_time = time.time()
    omega = 0.8
    c1 = 1.5
    c2 = 1.5
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        for i in range(swarm_size):
            h = particles[i]
            v = velocities[i]
            h, v = update_swarm(h, pbest[i], gbest, v, omega, c1, c2)
            current_c5 = evaluate_candidate(h)
            if current_c5 < evaluate_candidate(pbest[i]):
                pbest[i] = h.copy()
            if current_c5 < best_c5:
                best_c5 = current_c5
                gbest = h.copy()
        omega *= 0.995
    
    best_h = np.clip(gbest, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    return (best_h, best_c5, n_points)
