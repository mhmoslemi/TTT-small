```python
import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            if i % 5 == 0:
                h[i] = 1.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(h)
    for _ in range(100):
        h = best_h.copy()
        for i in range(n_points):
            h[i] = np.clip(h[i], 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr * dx)
        k = max_idx
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] += h[j]
        step_size = 0.05 / (np.max(np.abs(grad)) + 1e-10)
        h -= grad * step_size
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        c5 = objective(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if time.time() - start_time > budget_s - 10:
            break
    h = best_h.copy()
    sum_h = np.sum(h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    h = h * scale
    c5_bound = objective(h)
    return h.tolist(), c5_bound, n_points
```
