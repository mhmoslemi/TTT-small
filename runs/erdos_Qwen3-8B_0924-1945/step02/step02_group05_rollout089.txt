import numpy as np
from scipy.optimize import minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in kwargs:
        h = kwargs['initial_h_values'].copy()
    else:
        h = np.zeros(n_points)
        h[:10] = 1
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def compute_gradients(h, k):
        dx = 2.0 / n_points
        max_x = min(n_points - 1, int((2 - k) / dx))
        grad = np.zeros(n_points)
        for x in range(max_x + 1):
            grad[x] += (1 - h[x + k]) * dx
            grad[x + k] -= h[x] * dx
        return grad

    def update_h(h, grad, step_size):
        h += grad * step_size
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return h

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    iterations = 0
    prev_c5 = np.inf

    while time() - start_time < budget_s - 30 and iterations < 1000:
        iterations += 1
        corr_vals = np.correlate(best_h, 1 - best_h, mode="full") * 2.0 / n_points
        max_k = np.argmax(corr_vals)
        grad = compute_gradients(best_h, max_k)
        step_size = 0.01
        best_h = update_h(best_h, grad, step_size)
        current_c5 = evaluate_candidate(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            if abs(current_c5 - prev_c5) < 1e-5:
                break
            prev_c5 = current_c5
        else:
            best_h = update_h(best_h, -grad, step_size)
            current_c5 = evaluate_candidate(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
