import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    target_sum = n_points // 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constrain(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    population_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    elitism_rate = 0.2
    tournament_size = 5
    generations = 200
    max_no_improvement = 20

    pop = np.random.rand(population_size, n_points)
    for i in range(population_size):
        pop[i] = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
        pop[i] = pop[i] * (target_sum / np.sum(pop[i]))

    best_h = pop[0].copy()
    best_c5 = evaluate(best_h.flatten())

    start_time = time.time()
    no_improvement = 0

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break

        parents = []
        for _ in range(2):
            tournament = np.random.choice(pop, size=tournament_size)
            winner = tournament[np.argmin([evaluate(ind.flatten()) for ind in tournament])]
            parents.append(winner)

        offspring = []
        for i in range(population_size // 2):
            p1, p2 = parents[0], parents[1]
            if np.random.rand() < crossover_rate:
                child = np.zeros(n_points)
                for j in range(n_points):
                    if np.random.rand() < 0.5:
                        child[j] = p1[j]
                    else:
                        child[j] = p2[j]
            else:
                child = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
            child = child * (target_sum / np.sum(child))
            offspring.append(child)

        for i in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                for j in range(n_points):
                    if np.random.rand() < 0.1:
                        offspring[i][j] += np.random.normal(0, 0.1)
                        offspring[i][j] = np.clip(offspring[i][j], 0.0, 1.0)
                sum_h = np.sum(offspring[i])
                scale = target_sum / sum_h if sum_h != 0 else 1.0
                offspring[i] = offspring[i] * scale

        pop = np.array(offspring)
        pop = np.vstack((pop, pop))
        pop = np.unique(pop, axis=0)
        pop = pop[:population_size]

        fitness = [evaluate(ind.flatten()) for ind in pop]
        pop = pop[np.argsort(fitness)]
        pop = pop[:int(population_size * elitism_rate)]

        current_c5 = evaluate(pop[0].flatten())
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = pop[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > max_no_improvement:
                break

    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    best_h = best_h * scale
    c5_bound = evaluate(best_h.flatten())
    return best_h.tolist(), c5_bound, n_points
