import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if np.random.rand() < 0.5 else 0
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()

    def generate_structured_patterns():
        patterns = []
        # Uniform intervals
        for m in [3, 4, 5, 6, 7]:
            intervals = np.linspace(0, 2, m + 1)
            selected = np.random.choice(np.arange(m), size=3, replace=False)
            selected_intervals = intervals[selected]
            selected_intervals = np.sort(np.concatenate((selected_intervals, selected_intervals + (2 / m))))
            h = np.zeros(n_points)
            for i in range(n_points):
                x = i * 2.0 / n_points
                for s in selected_intervals:
                    if s <= x < s + (2 / m):
                        h[i] = 1
                        break
            h = h / np.sum(h) * n_points / 2
            patterns.append(h)
        # Symmetric bilinear
        for slope in np.linspace(0.5, 2, 5):
            h = np.zeros(n_points)
            for i in range(n_points):
                x = i * 2.0 / n_points
                h[i] = 0.5 * (1 + np.tanh(slope * (x - 1)))
            h = h / np.sum(h) * n_points / 2
            patterns.append(h)
        # Random binary
        h = np.random.choice([0, 1], size=n_points)
        h = h / np.sum(h) * n_points / 2
        patterns.append(h)
        return patterns

    population = []
    if 'initial_h_values' in kwargs:
        population.append(initial_h_values)
    else:
        population.extend(generate_structured_patterns())

    best_h = np.copy(population[0])
    best_c5 = compute_c5(best_h)

    def optimize_candidate(h):
        x0 = h.copy()
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            current_h = res.x
            current_c5 = evaluate_candidate(current_h)
            return current_h, current_c5
        return h, best_c5

    for _ in range(100):
        if time() - start_time > budget_s:
            break
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
        current_c5 = evaluate_candidate(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    for i in range(len(population)):
        h = population[i].copy()
        h, current_c5 = optimize_candidate(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    for _ in range(50):
        if time() - start_time > budget_s:
            break
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
        current_c5 = evaluate_candidate(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    final_c5 = evaluate_candidate(best_h)

    return (best_h, final_c5, n_points)
