```python
import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            if i % 5 == 0:
                h[i] = 1.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def gradient(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr * dx)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + max_idx) % n_points
            grad[i] += h[j]
        return -grad

    h_flat = h.flatten()
    best_h = h_flat.copy()
    best_c5 = objective(best_h)
    best_time = 0

    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        grad = gradient(h_flat)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-6:
            break
        alpha = 0.1
        h_new = h_flat - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        sum_h_new = np.sum(h_new)
        scale = target_sum / sum_h_new if sum_h_new != 0 else 1.0
        h_new = h_new * scale
        h_new_flat = h_new.flatten()
        c5_new = objective(h_new_flat)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new_flat.copy()
        h_flat = h_new_flat

    return best_h.tolist(), best_c5, n_points
```
