```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    iterations = 0
    best_h = h.copy()
    best_c5 = objective(h)

    for _ in range(3):
        if iterations >= 300:
            break
        result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})
        h = result.x
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        c5 = objective(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        iterations += 100

        if iterations >= 200:
            result = opt.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})
            h = result.x
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            c5 = objective(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            iterations += 100

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})
        best_h = result.x
        best_h = np.clip(best_h, 0.0, 1.0)
        sum_h = np.sum(best_h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        best_h = best_h * scale
        best_c5 = objective(best_h)

    return best_h.tolist(), best_c5, n_points
```
