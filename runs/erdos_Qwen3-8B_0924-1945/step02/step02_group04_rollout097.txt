import numpy as np
from scipy.optimize import basinhopping, minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    current_h = np.clip(initial_h_values, 0, 1)
    current_h = current_h * (n_points / 2) / np.sum(current_h)
    best_h = current_h.copy()
    best_c5 = compute_c5(current_h, n_points)
    start_time = time.time()

    def objective(h):
        return compute_c5(h, n_points)

    def compute_c5(h, n_points):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def normalize(h):
        h = np.clip(h, 0, 1)
        return h * (n_points / 2) / np.sum(h)

    def perturb(h, magnitude):
        perturbation = np.random.uniform(-magnitude, magnitude, size=n_points)
        return normalize(h + perturbation)

    def callback(x):
        nonlocal best_h, best_c5
        c5 = compute_c5(x, n_points)
        if c5 < best_c5:
            best_h = x.copy()
            best_c5 = c5

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        current_h = perturb(current_h, 0.2)
        result = minimize(objective, current_h, method='L-BFGS-B', bounds=[(0, 1)] * n_points, tol=1e-5)
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h, n_points)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            break

    best_h = normalize(best_h)
    return (best_h, best_c5, n_points)
