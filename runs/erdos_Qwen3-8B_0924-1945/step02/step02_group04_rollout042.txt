import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in kwargs:
        h = kwargs['initial_h_values'].copy()
    else:
        h = np.random.uniform(0, 1, n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_vec):
        h = h_vec.copy()
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def objective_grad(h_vec):
        h = h_vec.copy()
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        max_val = np.max(correlate_vals)
        max_index = np.argmax(correlate_vals)
        
        grad = np.zeros_like(h)
        for k in range(1, n_points):
            lag_vals = correlate_vals[n_points - k]
            dh = np.zeros_like(h)
            for i in range(n_points):
                if i + k < n_points:
                    dh[i] += (1 - h[i + k]) * dx
                if i - k >= 0:
                    dh[i] += h[i - k] * dx
            if k == max_index:
                grad += dh
        return grad
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, jac=objective_grad, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        else:
            break
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturbation = np.zeros_like(best_h)
        indices = np.argsort(best_h)
        for i in range(len(indices)):
            if i < 0.2 * len(indices):
                perturbation[indices[i]] = np.random.uniform(-0.05, 0.05)
        new_h = best_h + perturbation
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        new_c5 = objective(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    return (best_h, best_c5, n_points)
