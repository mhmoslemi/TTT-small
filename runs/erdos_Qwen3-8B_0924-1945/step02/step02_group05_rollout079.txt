import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    h = h / np.sum(h) * n_points / 2
    h = np.clip(h, 0, 1)

    def compute_c5(h):
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def update_h(h, lag, step_size):
        active_range = np.arange(max(0, lag - 10), min(n_points, lag + 10))
        grad = np.zeros(n_points)
        for x in active_range:
            if x + lag < n_points:
                grad[x] += 1 - h[x + lag]
            if x >= lag:
                grad[x - lag] -= h[x]
        h[active_range] -= step_size * grad[active_range]
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return h

    def run_optimization():
        best_h = h.copy()
        best_c5 = evaluate_candidate(best_h)
        for _ in range(1000):
            if time() - start_time > budget_s:
                break
            corr_vals = np.correlate(best_h, 1 - best_h, mode="full") * dx
            max_corr = np.max(corr_vals)
            if max_corr <= 0.3807:
                break
            best_lag = np.argmax(corr_vals)
            best_h = update_h(best_h, best_lag, 0.01)
            current_c5 = evaluate_candidate(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = best_h.copy()
            else:
                best_h = best_h.copy()
        return best_h, best_c5

    start_time = time()
    best_h, best_c5 = run_optimization()

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
