import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def update_h(h, k):
        dx = 2.0 / n_points
        active_start = max(0, k - 10)
        active_end = min(n_points - 1, k + 10)
        active_indices = np.arange(active_start, active_end + 1)
        gradient = np.zeros(n_points)
        for i in active_indices:
            gradient[i] += (1 - h[i + k]) if i + k < n_points else 0
            if i + k < n_points:
                gradient[i + k] -= h[i]
        step_size = 0.01
        h[active_indices] -= step_size * gradient[active_indices]
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return h

    def run_optimization():
        best_h = h.copy()
        best_c5 = evaluate_candidate(best_h)
        for _ in range(1000):
            if time() - start_time > budget_s:
                break
            corr_vals = np.correlate(best_h, 1 - best_h, mode="full") * 2.0 / n_points
            max_k = np.argmax(corr_vals)
            if corr_vals[max_k] < 0.38:
                break
            best_h = update_h(best_h, max_k)
            current_c5 = evaluate_candidate(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = best_h.copy()
        return best_h, best_c5

    start_time = time()
    best_h, best_c5 = run_optimization()
    if time() - start_time < budget_s:
        for _ in range(50):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    return (best_h, best_c5, n_points)
