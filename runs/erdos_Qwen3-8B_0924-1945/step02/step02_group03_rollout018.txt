import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    if 'initial_h_values' in kwargs:
        h = kwargs['initial_h_values'].copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if np.random.rand() < 0.5 else 0
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    # Simulated Annealing Phase
    T = 1.0
    cool_rate = 0.995
    for _ in range(2000):
        if time() - start_time > budget_s:
            break
        x = best_h.copy()
        perturb = np.random.normal(0, 0.05, n_points)
        x += perturb
        x = np.clip(x, 0, 1)
        x = x / np.sum(x) * n_points / 2
        current_c5 = evaluate_candidate(x)
        delta = current_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h = x.copy()
            best_c5 = current_c5
        T *= cool_rate
        if T < 1e-5:
            break

    # Genetic Algorithm Phase
    population_size = 30
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    for p in population:
        p = p / np.sum(p) * n_points / 2
    best_pop = population[0]
    best_pop_c5 = evaluate_candidate(best_pop)
    for _ in range(500):
        if time() - start_time > budget_s:
            break
        # Tournament selection
        indices = np.random.choice(range(population_size), 5, replace=False)
        candidates = [population[i] for i in indices]
        for c in candidates:
            c = c / np.sum(c) * n_points / 2
        best_cand = min(candidates, key=lambda x: evaluate_candidate(x))
        # Crossover
        parent1 = best_cand
        parent2 = np.random.choice(population)
        child = (parent1 + parent2) / 2
        # Mutation
        mutation_mask = np.random.rand(n_points) < 0.1
        child[mutation_mask] += np.random.normal(0, 0.05, sum(mutation_mask))
        child = np.clip(child, 0, 1)
        child = child / np.sum(child) * n_points / 2
        # Evaluate and replace
        child_c5 = evaluate_candidate(child)
        if child_c5 < best_pop_c5:
            best_pop = child.copy()
            best_pop_c5 = child_c5
        # Replace worst
        population.remove(best_pop)
        population.append(child)
        population.sort(key=lambda x: evaluate_candidate(x))

    best_h = best_pop.copy()
    best_c5 = best_pop_c5
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    return (best_h, best_c5, n_points)
