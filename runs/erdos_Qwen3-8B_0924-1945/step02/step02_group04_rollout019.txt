import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    h = np.clip(initial_h_values, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n_points)
    
    def compute_c5(h, n_points):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    
    for _ in range(100):
        if time.time() - time.time() > budget_s:
            break
        res = minimize(compute_c5, best_h, args=(n_points,), method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5, options={'maxiter': 50})
        if res.success:
            current_h = res.x
            current_c5 = compute_c5(current_h, n_points)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            break
    
    for _ in range(50):
        if time.time() - time.time() > budget_s:
            break
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        h = best_h + perturb
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = compute_c5(h, n_points)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    return (best_h, best_c5, n_points)
