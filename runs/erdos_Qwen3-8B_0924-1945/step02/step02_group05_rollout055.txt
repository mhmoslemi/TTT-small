import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time
import copy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy() if 'initial_h_values' in kwargs else np.zeros(n_points)
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)
    
    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()

    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    def run_optimization():
        nonlocal best_h, best_c5
        for _ in range(200):
            if time() - start_time > budget_s:
                break
            corr_vals = np.correlate(best_h, 1 - best_h, mode="full") * (2.0 / n_points)
            max_k = np.argmax(corr_vals)
            if corr_vals[max_k] < 0.38:
                break
            h_k = best_h[:n_points - max_k]
            h_k_plus = best_h[max_k:]
            grad = np.zeros(n_points)
            for i in range(len(h_k)):
                grad[i] += 1 - h_k_plus[i]
                grad[i + max_k] -= h_k[i]
            step = 0.05
            new_h = best_h.copy()
            new_h[:n_points - max_k] -= step * h_k
            new_h[max_k:] -= step * h_k_plus
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h / np.sum(new_h) * n_points / 2
            current_c5 = evaluate_candidate(new_h)
            if current_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = current_c5

    run_optimization()

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
