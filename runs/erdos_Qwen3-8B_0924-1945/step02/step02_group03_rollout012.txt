import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    def de_step(population, best_h):
        mutant = np.random.uniform(0, 1, n_points)
        mutant = np.clip(mutant, 0, 1)
        mutant = mutant / np.sum(mutant) * n_points / 2
        trial = np.random.rand(n_points)
        for i in range(n_points):
            if np.random.rand() < 0.8:
                trial[i] = mutant[i]
        trial = np.clip(trial, 0, 1)
        trial = trial / np.sum(trial) * n_points / 2
        trial_c5 = evaluate_candidate(trial)
        if trial_c5 < best_c5:
            return trial, trial_c5
        return best_h, best_c5

    for _ in range(300):
        if time() - start_time > budget_s:
            break
        new_h, new_c5 = de_step(best_h, best_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    res = minimize(lambda x: evaluate_candidate(x), best_h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-5)
    if res.success:
        best_h = res.x
        best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
