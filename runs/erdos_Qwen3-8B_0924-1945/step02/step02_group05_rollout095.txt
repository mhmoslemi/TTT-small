import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 90
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    
    def validate_h(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if np.isclose(total, n_points / 2, atol=1e-5):
            return h
        else:
            return h * (n_points / 2) / total

    def compute_c5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def objective(h):
        return compute_c5(validate_h(h))

    bounds = [(0, 1) for _ in range(n_points)]
    
    def generate_initial_population(size=5):
        population = []
        for _ in range(size):
            if 'initial_h_values' in kwargs:
                h = initial_h_values.copy()
                h = validate_h(h)
                population.append(h)
            else:
                h = np.random.uniform(0, 1, n_points)
                h = validate_h(h)
                population.append(h)
        return population

    def differential_evolution_runner():
        population = generate_initial_population()
        best_h = min(population, key=objective)
        best_c5 = objective(best_h)
        start_time = time()
        for _ in range(300):
            if time() - start_time > budget_s:
                break
            candidates = np.random.choice(population, size=5, replace=False)
            a, b, c = candidates
            mutant = a + 0.8 * (b - c)
            mutant = validate_h(mutant)
            mutant_c5 = objective(mutant)
            if mutant_c5 < objective(best_h):
                best_h = mutant
                best_c5 = mutant_c5
            population.remove(best_h)
            population.append(mutant)
        return best_h, best_c5

    best_h, best_c5 = differential_evolution_runner()
    
    if time() - start_time < budget_s:
        for _ in range(50):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = validate_h(h)
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5

    best_h = validate_h(best_h)
    current_c5 = objective(best_h)
    return (best_h, current_c5, n_points)
