```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 300

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    start_time = time.time()

    def optimize(h0):
        result = opt.minimize(objective, h0, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': max_iter})
        h_opt = result.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h = np.sum(h_opt)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h_opt = h_opt * scale
        return h_opt

    h_opt = optimize(h)
    h_opt = optimize(h_opt)

    if time.time() - start_time < budget_s:
        def slsqp_optimize(h0):
            result = opt.minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': max_iter})
            h_opt = result.x
            h_opt = np.clip(h_opt, 0.0, 1.0)
            sum_h = np.sum(h_opt)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h_opt = h_opt * scale
            return h_opt

        h_opt = slsqp_optimize(h_opt)
        h_opt = slsqp_optimize(h_opt)

    c5_bound = objective(h_opt)
    return h_opt.tolist(), c5_bound, n_points
```
