import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if np.random.rand() < 0.5 else 0
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()

    def generate_pattern():
        x = np.linspace(0, 2, n_points)
        m = np.random.choice([3, 4, 5, 6, 7])
        intervals = np.linspace(0, 2, m + 1)
        selected = np.random.choice(np.arange(m), size=m // 2, replace=False)
        selected_intervals = intervals[selected]
        selected_intervals = np.sort(selected_intervals)
        selected_intervals = np.append(selected_intervals, intervals[1:-1][~np.isin(intervals[1:-1], selected_intervals)])
        selected_intervals = np.sort(selected_intervals)
        h = np.zeros(n_points)
        for i in range(n_points):
            for interval in selected_intervals:
                if interval <= x[i] < interval + 2 / m:
                    h[i] = 1
                    break
        h = h / np.sum(h) * n_points / 2
        return h

    def optimize_pattern(h):
        res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        return res.x

    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    initial_patterns = [generate_pattern() for _ in range(5)]
    initial_patterns.append(h)
    population = [optimize_pattern(p) for p in initial_patterns]
    population = [np.clip(p, 0, 1) for p in population]
    population = [p / np.sum(p) * n_points / 2 for p in population]
    population = [p for p in population if np.all(0 <= p <= 1)]

    for _ in range(200):
        if time() - start_time > budget_s:
            break
        current_h = np.random.choice(population)
        current_h = current_h.copy()
        for _ in range(10):
            idx = np.random.randint(n_points)
            perturbation = np.random.uniform(-0.2, 0.2)
            current_h[idx] = np.clip(current_h[idx] + perturbation, 0, 1)
            current_h = current_h / np.sum(current_h) * n_points / 2
            c5 = evaluate_candidate(current_h)
            if c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = c5
        if abs(best_c5 - evaluate_candidate(best_h)) < 1e-6:
            break

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
