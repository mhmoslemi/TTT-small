import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    max_points = 400
    n_points = max_points
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    def calc_c5(h):
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    def genetic_search():
        pop_size = 50
        generations = 500
        elite_size = 5
        mutation_rate = 0.1
        crossover_rate = 0.8
        penalty = 100.0

        def fitness(x):
            c5 = calc_c5(x)
            sum_h = np.sum(x)
            penalty_sum = penalty * np.abs(sum_h - target_sum)
            return c5 + penalty_sum

        pop = np.random.rand(pop_size, n_points)
        pop = pop * 2 - 1
        pop = np.clip(pop, 0.0, 1.0)
        pop = pop * (target_sum / np.sum(pop, axis=1)[:, None]) if np.sum(pop, axis=1).any() != 0 else pop

        for gen in range(generations):
            fitness_values = np.array([fitness(ind) for ind in pop])
            elite_indices = np.argsort(fitness_values)[:elite_size]
            elites = pop[elite_indices]

            new_pop = []
            for _ in range(pop_size - elite_size):
                parent1, parent2 = np.random.choice(elite_indices, 2, replace=False)
                child = np.copy(pop[parent1])
                for i in range(n_points):
                    if np.random.rand() < crossover_rate:
                        child[i] = pop[parent2][i]
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] = np.clip(np.random.rand(), 0, 1)
                child = child * (target_sum / np.sum(child)) if np.sum(child) != 0 else child
                new_pop.append(child)
            pop = np.vstack([elites, new_pop])
            pop = np.clip(pop, 0.0, 1.0)

            if gen % 50 == 0:
                best = pop[np.argmin(fitness_values)]
                c5_val = calc_c5(best)
                print(f"Gen {gen}: Best C5 = {c5_val}")

        best = pop[np.argmin(fitness_values)]
        return best

    start_time = time.time()
    best_h = genetic_search()

    if time.time() - start_time < budget_s:
        result = opt.minimize(
            objective,
            best_h.flatten(),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            tol=1e-8,
            options={'maxiter': 300}
        )
        best_h = result.x.reshape(n_points)
        best_h = np.clip(best_h, 0.0, 1.0)
        sum_h = np.sum(best_h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        best_h = best_h * scale

    c5_bound = calc_c5(best_h)
    return best_h.tolist(), c5_bound, n_points
