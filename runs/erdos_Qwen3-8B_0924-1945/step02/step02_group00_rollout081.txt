```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def generate_base_pattern():
        if initial_h_values is not None:
            h = np.array(initial_h_values, dtype=np.float64)
            noise = np.random.normal(0, 0.03, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
            return h
        else:
            h = np.zeros(n_points)
            for i in range(n_points):
                h[i] = 1 if i % 5 == 0 else 0
            h = h * (target_sum / np.sum(h))
            return h

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    def refine_candidate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        result = opt.minimize(evaluate, h.flatten(), method='SLSQP', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': constraint}, tol=1e-8, options={'maxiter': 100})
        return result.x.reshape(n_points)

    def fitness(h_flat):
        return evaluate(h_flat)

    start_time = time.time()
    candidates = []
    for _ in range(5):
        h = generate_base_pattern()
        candidates.append(h.flatten())

    pop_size = 50
    best = min(candidates, key=fitness)
    best_fit = fitness(best)

    class Individual:
        def __init__(self, genes):
            self.genes = genes

    def mutate(ind, mutation_rate=0.1):
        genes = ind.genes.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                genes[i] += np.random.uniform(-0.1, 0.1)
                genes[i] = np.clip(genes[i], 0, 1)
        return Individual(genes)

    def crossover(ind1, ind2):
        genes = ind1.genes.copy()
        for i in range(n_points):
            if np.random.rand() < 0.5:
                genes[i] = ind2.genes[i]
        return Individual(genes)

    def select(population, k=5):
        return sorted(population, key=fitness)[:k]

    def evolve(population, generations=100):
        for _ in range(generations):
            new_pop = []
            selected = select(population, k=5)
            for _ in range(pop_size):
                parent1 = np.random.choice(selected)
                parent2 = np.random.choice(selected)
                child = crossover(parent1, parent2)
                child = mutate(child)
                new_pop.append(child)
            population = new_pop
            current_best = min(population, key=fitness)
            if fitness(current_best.genes) < best_fit:
                best = current_best.genes
                best_fit = fitness(best)
                if best_fit < 0.381:
                    break
        return best

    population = [Individual(h) for h in candidates]
    population = evolve(population, generations=300)

    best = population[0].genes
    best_fit = fitness(best)

    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 50:
        refined = refine_candidate(best)
        best = refined
        best_fit = fitness(best)

    return best.tolist(), best_fit, n_points
```
