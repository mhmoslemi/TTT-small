import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_generations = 200
    pop_size = 50
    elite_ratio = 0.2
    mutation_rate = 0.1
    crossover_prob = 0.8
    tournament_size = 5
    
    if initial_h_values is not None:
        h_pop = np.array([np.zeros(n_points) for _ in range(pop_size)])
        for i in range(pop_size):
            h = np.zeros(n_points)
            indices = np.random.choice(n_points, size=int(target_sum), replace=False)
            h[indices] = 1.0
            h = h * (target_sum / np.sum(h))
            h_pop[i] = h
    else:
        h_pop = np.array([np.zeros(n_points) for _ in range(pop_size)])
        for i in range(pop_size):
            h = np.zeros(n_points)
            indices = np.random.choice(n_points, size=int(target_sum), replace=False)
            h[indices] = 1.0
            h = h * (target_sum / np.sum(h))
            h_pop[i] = h
    
    def evaluate(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    
    best_h = h_pop[0]
    best_c5 = evaluate(best_h)
    time_start = time.time()
    for gen in range(max_generations):
        if time.time() - time_start > budget_s:
            break
        
        # Select parents via tournament selection
        parents = []
        for _ in range(2):
            candidates = np.random.choice(pop_size, tournament_size, replace=False)
            winner = candidates[np.argmin([evaluate(h_pop[i]) for i in candidates])]
            parents.append(h_pop[winner])
        
        # Crossover
        offspring = []
        for _ in range(pop_size // 2):
            if np.random.rand() < crossover_prob:
                p1, p2 = parents[0], parents[1]
                child = np.zeros(n_points)
                for i in range(n_points):
                    if np.random.rand() < 0.5:
                        child[i] = p1[i]
                    else:
                        child[i] = p2[i]
            else:
                child = np.random.choice(parents, size=1)[0]
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            h = offspring[i]
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    h[j] += np.random.normal(0, 0.1)
                    h[j] = np.clip(h[j], 0.0, 1.0)
            h = h * (target_sum / np.sum(h))
            offspring[i] = h
        
        # Combine population
        new_pop = np.array(offspring)
        for i in range(pop_size):
            new_pop[i] = np.clip(new_pop[i], 0.0, 1.0)
            new_pop[i] = new_pop[i] * (target_sum / np.sum(new_pop[i]))
        
        # Elitism
        sorted_indices = np.argsort([evaluate(h) for h in new_pop])
        new_pop = new_pop[sorted_indices[:int(pop_size * elite_ratio)]]
        
        # Update best
        for h in new_pop:
            c5 = evaluate(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
    
    # Final evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (target_sum / np.sum(best_h))
    final_c5 = evaluate(best_h)
    return best_h.tolist(), final_c5, n_points
