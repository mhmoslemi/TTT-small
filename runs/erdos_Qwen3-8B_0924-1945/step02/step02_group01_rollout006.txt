import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        n_ones = n_points // 2
        indices = np.random.choice(n_points, n_ones, replace=False)
        h[indices] = 1.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    population_size = 50
    tournament_size = 5
    mutation_rate = 0.1
    crossover_prob = 0.8
    elitism_rate = 0.2
    generations = 200
    max_time = budget_s * 0.8

    h_pop = np.zeros((population_size, n_points), dtype=np.float64)
    h_pop[0] = h
    fitness = np.zeros(population_size)

    for i in range(population_size):
        n_ones = n_points // 2
        indices = np.random.choice(n_points, n_ones, replace=False)
        h_pop[i, indices] = 1.0
        h_pop[i] = h_pop[i] * (target_sum / np.sum(h_pop[i]))

    fitness = np.array([evaluate(h_pop[i]) for i in range(population_size)])

    start_time = time.time()
    best_idx = np.argmin(fitness)
    best_h = h_pop[best_idx].copy()

    for gen in range(generations):
        if time.time() - start_time > max_time:
            break
        new_pop = np.zeros((population_size, n_points), dtype=np.float64)
        for i in range(population_size):
            parent1, parent2 = np.random.choice(population_size, 2, replace=False)
            if np.random.rand() < crossover_prob:
                child = np.zeros(n_points)
                for j in range(n_points):
                    if np.random.rand() < 0.5:
                        child[j] = h_pop[parent1, j]
                    else:
                        child[j] = h_pop[parent2, j]
            else:
                child = h_pop[parent1].copy()

            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[j] += np.random.normal(0, 0.1)
                    child[j] = np.clip(child[j], 0.0, 1.0)

            sum_child = np.sum(child)
            if sum_child != 0:
                child = child * (target_sum / sum_child)
            else:
                child = np.zeros(n_points)
                n_ones = n_points // 2
                indices = np.random.choice(n_points, n_ones, replace=False)
                child[indices] = 1.0
                child = child * (target_sum / np.sum(child))

            new_pop[i] = child

        fitness_new = np.array([evaluate(new_pop[i]) for i in range(population_size)])
        combined_pop = np.vstack((h_pop, new_pop))
        combined_fitness = np.concatenate((fitness, fitness_new))
        indices = np.argsort(combined_fitness)[:(int(population_size * (1 - elitism_rate)))]
        h_pop = combined_pop[indices]
        fitness = combined_fitness[indices]

        if np.min(fitness) < fitness[best_idx]:
            best_idx = np.argmin(fitness)
            best_h = h_pop[best_idx].copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    if sum_h != 0:
        best_h = best_h * (target_sum / sum_h)
    else:
        best_h = np.zeros(n_points)
        n_ones = n_points // 2
        indices = np.random.choice(n_points, n_ones, replace=False)
        best_h[indices] = 1.0
        best_h = best_h * (target_sum / np.sum(best_h))

    c5_bound = evaluate(best_h)
    return best_h.tolist(), c5_bound, n_points
