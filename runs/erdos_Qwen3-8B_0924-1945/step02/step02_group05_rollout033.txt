```python
import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 120
    if 'initial_h_values' in kwargs:
        h = kwargs['initial_h_values'].copy()
    else:
        h = np.zeros(n_points)
        h = np.random.choice([0, 1], size=n_points, p=[0.25, 0.75])
        h = h / np.sum(h) * n_points / 2
    h = np.clip(h, 0, 1)

    def compute_c5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        norm = np.sum(h)
        if norm == 0:
            return np.inf
        h = h / norm * n_points / 2
        return compute_c5(h)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(x):
        return evaluate_candidate(x)

    start_time = time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=150,
        popsize=10,
        mutation=(0.5, 1),
        recombination=0.9,
        tol=1e-5,
        polish=True,
        disp=False,
        mutation_func=None,
        seed=seed
    )

    if time() - start_time < budget_s:
        for _ in range(50):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < result.fun:
                result = differential_evolution(
                    objective,
                    bounds,
                    strategy='best1bin',
                    maxiter=5,
                    popsize=5,
                    mutation=(0.5, 1),
                    recombination=0.9,
                    tol=1e-5,
                    polish=True,
                    disp=False,
                    mutation_func=None,
                    seed=seed
                )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    c5 = evaluate_candidate(best_h)
    return (best_h, c5, n_points)
```
