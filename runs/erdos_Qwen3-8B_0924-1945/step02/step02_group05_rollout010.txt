import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    
    def validate_h(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h) * 2.0 / n_points
        h = h * (n_points / 2) / total
        return h

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def objective(h):
        h = validate_h(h)
        return compute_c5(h)

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()

    def generate_initial_population(size=5):
        population = []
        for _ in range(size):
            if 'initial_h_values' in kwargs:
                init = initial_h_values.copy()
            else:
                init = np.random.uniform(0, 1, n_points)
            init = validate_h(init)
            population.append(init)
        return population

    population = generate_initial_population()

    def optimize():
        best_h = validate_h(population[0])
        best_c5 = compute_c5(best_h)
        gen = 0
        while time() - start_time < budget_s and gen < 300:
            gen += 1
            new_population = []
            for _ in range(len(population)):
                idxs = np.random.choice(len(population), 3, replace=False)
                h1, h2, h3 = population[idxs[0]], population[idxs[1]], population[idxs[2]]
                mutant = h1 + 0.8 * (h2 - h3)
                mutant = validate_h(mutant)
                if compute_c5(mutant) < compute_c5(best_h):
                    new_population.append(mutant)
                else:
                    new_population.append(population[np.random.choice(len(population))])
            population = new_population
            current_h = validate_h(population[0])
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        return best_h, best_c5

    best_h, best_c5 = optimize()

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = validate_h(h)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5

    return (best_h, best_c5, n_points)
