```python
import numpy as np
import scipy.optimize as opt
import time
import random
from deap import base, creator, tools, algorithms

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))
    
    def evaluate(individual):
        h = np.array(individual, dtype=np.float64)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx),
    
    def constraint(individual):
        h = np.array(individual, dtype=np.float64)
        sum_h = np.sum(h)
        return sum_h - target_sum
    
    def generate_initial_population(size):
        pop = []
        for _ in range(size):
            h = np.zeros(n_points)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    h[i] = 1.0
            h = h * (target_sum / np.sum(h))
            pop.append(h.tolist())
        return pop
    
    def evolve_population(population):
        creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
        creator.create("Individual", list, fitness=creator.FitnessMin)
        
        toolbox = base.Toolbox()
        toolbox.register("evaluate", evaluate)
        toolbox.register("select", tools.selTournament, tournsize=5)
        toolbox.register("mate", tools.cxBlend, alpha=0.5)
        toolbox.register("mutate", tools.mutPolynomialBounded, eta=0.5, indpb=0.1)
        toolbox.register("population", generate_initial_population)
        toolbox.register("individual", tools.initIterate, creator.Individual, toolbox.population)
        
        pop = algorithms.varAnd(population, toolbox, cxpb=0.5, mutpb=0.5)
        fits = toolbox.map(toolbox.evaluate, pop)
        for fit, ind in zip(fits, pop):
            ind.fitness.values = fit
        pop = tools.selBest(pop, len(pop))
        return pop
    
    def refine_solution(h_values):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        def objective(x):
            x = np.clip(x, 0.0, 1.0)
            return evaluate(x)[0]
        
        result = opt.minimize(objective, h_values, method='SLSQP',
                              bounds=bounds, constraints=constraints,
                              tol=1e-8, options={'maxiter': 300})
        return result.x
    
    start_time = time.time()
    best_h = h.tolist()
    best_c5 = evaluate(best_h)[0]
    
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        if initial_h_values is not None:
            base_pop = [h.tolist()]
        else:
            base_pop = generate_initial_population(5)
        
        pop = evolve_population(base_pop)
        for ind in pop:
            ind = refine_solution(ind)
            c5 = evaluate(ind)[0]
            if c5 < best_c5:
                best_h = ind
                best_c5 = c5
    
    if time.time() - start_time < budget_s:
        best_h = refine_solution(best_h)
    
    best_c5 = evaluate(best_h)[0]
    return best_h, best_c5, n_points
```
