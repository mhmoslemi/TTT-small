```python
import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr * dx)
        return max_corr

    def rescale(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return h
        return h * (target_sum / sum_h)

    def clip_and_rescale(h):
        h = np.clip(h, 0.0, 1.0)
        return rescale(h)

    population_size = 50
    tournament_size = 5
    mutation_rate = 0.1
    crossover_rate = 0.8
    elitism_rate = 0.2
    generations = 300
    max_no_improvement = 20

    population = [rescale(np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])) for _ in range(population_size)]
    best_h = population[0]
    best_c5 = evaluate(best_h)
    best_fitness = best_c5
    no_improvement = 0

    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break

        parents = []
        for _ in range(population_size // 2):
            candidates = np.random.choice(population, tournament_size, replace=False)
            winner = min(candidates, key=evaluate)
            parents.append(winner)

        offspring = []
        for i in range(0, len(parents), 2):
            parent1 = parents[i]
            parent2 = parents[i+1] if i+1 < len(parents) else parents[i]
            child = np.zeros(n_points)
            for j in range(n_points):
                if np.random.rand() < crossover_rate:
                    child[j] = parent1[j]
                else:
                    child[j] = parent2[j]
            child = clip_and_rescale(child)
            offspring.append(child)

        for i in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                indices = np.random.choice(n_points, size=int(n_points * mutation_rate), replace=False)
                offspring[i][indices] += np.random.normal(0, 0.1, size=len(indices))
                offspring[i] = clip_and_rescale(offspring[i])

        population = offspring
        population.sort(key=evaluate)
        population = population[:int(population_size * elitism_rate)]

        current_fitness = evaluate(population[0])
        if current_fitness < best_fitness:
            best_fitness = current_fitness
            best_h = population[0]
            no_improvement = 0
        else:
            no_improvement += 1

        if no_improvement >= max_no_improvement:
            break

    best_h = clip_and_rescale(best_h)
    c5_bound = evaluate(best_h)
    return best_h.tolist(), c5_bound, n_points
```
