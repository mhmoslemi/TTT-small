import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    if np.shape(initial_h_values) != (n_points,):
        initial_h_values = np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2 if np.sum(h) != 0 else np.random.uniform(0, 1, n_points)
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    population_size = 50
    population = [np.random.uniform(0, 1, n_points).copy() for _ in range(population_size)]
    for i in range(population_size):
        population[i] = np.clip(population[i], 0, 1)
        population[i] = population[i] / np.sum(population[i]) * n_points / 2
    population = [evaluate_candidate(ind) for ind in population]
    population = [np.random.uniform(0, 1, n_points).copy() for _ in range(population_size)]
    for i in range(population_size):
        population[i] = np.clip(population[i], 0, 1)
        population[i] = population[i] / np.sum(population[i]) * n_points / 2
    population = [evaluate_candidate(ind) for ind in population]

    for _ in range(100):
        if time() - start_time > budget_s:
            break
        parents = np.random.choice(range(population_size), size=5, replace=False)
        parent1 = population[parents[0]]
        parent2 = population[parents[1]]
        offspring = (parent1 + parent2) / 2 + np.random.uniform(-0.1, 0.1, n_points)
        offspring = np.clip(offspring, 0, 1)
        offspring = offspring / np.sum(offspring) * n_points / 2
        offspring_c5 = evaluate_candidate(offspring)
        if offspring_c5 < population[parents[2]]:
            population[parents[2]] = offspring_c5
        for i in range(population_size):
            if evaluate_candidate(population[i]) < evaluate_candidate(population[parents[3]]):
                population[parents[3]] = population[i]
        if evaluate_candidate(population[0]) < best_c5:
            best_h = population[0]
            best_c5 = evaluate_candidate(population[0])

    for _ in range(50):
        if time() - start_time > budget_s:
            break
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        current_c5 = evaluate_candidate(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    h = np.clip(best_h, 0, 1)
    h = h / np.sum(h) * n_points / 2
    current_c5 = evaluate_candidate(h)
    return (h, current_c5, n_points)
