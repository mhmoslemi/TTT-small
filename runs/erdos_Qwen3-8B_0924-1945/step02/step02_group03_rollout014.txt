import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    bounds = [(0, 1) for _ in range(n_points)]
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)

    start_time = time()
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        mutation=(0.5, 1),
        recombination=0.8,
        popsize=15,
        tol=1e-5,
        maxiter=500,
        disp=False,
        polish=True,
        callback=lambda x: None
    )

    if time() - start_time < budget_s:
        h_candidate = result.x
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate / np.sum(h_candidate) * n_points / 2
        best_c5 = compute_c5(h_candidate)
        best_h = h_candidate

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = compute_c5(best_h)
    return (best_h, current_c5, n_points)
