import numpy as np
from scipy.optimize import minimize, Bounds
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        for i in range(n_points):
            h[i] = 1 if np.random.rand() < 0.5 else 0
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    def mutate(x):
        idxs = np.random.choice(n_points, size=3, replace=False)
        a, b, c = x[idxs]
        factor = 0.5
        mutant = x.copy()
        mutant[idxs[0]] = np.clip(a + factor * (b - c), 0, 1)
        mutant[idxs[1]] = np.clip(b + factor * (c - a), 0, 1)
        mutant[idxs[2]] = np.clip(c + factor * (a - b), 0, 1)
        mutant = mutant / np.sum(mutant) * n_points / 2
        return mutant

    def crossover(parent, mutant):
        child = parent.copy()
        idxs = np.random.choice(n_points, size=n_points // 2, replace=False)
        child[idxs] = mutant[idxs]
        child = child / np.sum(child) * n_points / 2
        return child

    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    start_time = time()

    for _ in range(300):
        if time() - start_time > budget_s:
            break
        pop = [mutate(best_h) for _ in range(30)]
        for p in pop:
            if compute_c5(p) < best_c5:
                best_h = p.copy()
                best_c5 = compute_c5(p)
        best_h = crossover(best_h, pop[np.argmin([compute_c5(p) for p in pop])])
        best_c5 = compute_c5(best_h)

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
