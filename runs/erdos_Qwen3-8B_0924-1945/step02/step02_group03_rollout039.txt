```python
import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    
    def validate_h(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - n_points / 2) > 1e-6:
            h = h / total * n_points / 2
        return h
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        h = validate_h(h)
        return compute_c5(h)
    
    def objective(h):
        return evaluate_candidate(h)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = initial_h_values.copy()
    best_c5 = evaluate_candidate(best_h)
    
    start_time = time()
    
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        tol=1e-5,
        mutation=(0.5, 1),
        recombination=0.8,
        maxiter=500,
        disp=False,
        polish=False,
        seed=seed
    )
    
    best_h = result.x
    best_c5 = evaluate_candidate(best_h)
    
    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = validate_h(h)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    best_h = validate_h(best_h)
    return (best_h, best_c5, n_points)
```
