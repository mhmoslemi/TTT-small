```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[::5] = 1.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    def lb(x):
        return np.clip(x, 0.0, 1.0)

    def refine(h_flat):
        h = h_flat.reshape(n_points)
        h = lb(h)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        h = lb(h)
        return h.flatten()

    start_time = time.time()
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=[(0.0, 1.0)]*n_points, constraints={'type': 'eq', 'fun': constraint}, tol=1e-8, options={'maxiter': 100})
    h_opt = refine(result.x)
    best_c5 = objective(h_opt)

    if time.time() - start_time < budget_s:
        def custom_loss(h_flat):
            h = refine(h_flat)
            return objective(h)

        def custom_constr(h_flat):
            h = refine(h_flat)
            return constraint(h)

        def custom_bounds(h_flat):
            h = h_flat.reshape(n_points)
            return np.clip(h, 0.0, 1.0).flatten()

        result = opt.minimize(custom_loss, h_opt, method='SLSQP', bounds=[(0.0, 1.0)]*n_points, constraints={'type': 'eq', 'fun': custom_constr}, tol=1e-8, options={'maxiter': 200})
        h_opt = refine(result.x)

    c5_bound = objective(h_opt)
    return h_opt.tolist(), c5_bound, n_points
```
