import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        h[::5] = 1.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()

    def optimize_with_method(h_init, method, max_iter=300):
        result = opt.minimize(objective, h_init, method=method, bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': max_iter})
        h_opt = result.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        return h_opt

    h_opt = optimize_with_method(h, 'L-BFGS-B', 500)
    h_opt = optimize_with_method(h_opt, 'SLSQP', 300)

    if time.time() - start_time < budget_s:
        def de_objective(h_flat):
            h = h_flat.reshape(n_points)
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            corr = np.correlate(h, 1.0 - h, mode="full")
            c5 = np.max(corr * dx)
            return c5

        def de_constraint(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) - target_sum

        cons_de = [{'type': 'eq', 'fun': de_constraint}]
        bounds_de = [(0.0, 1.0) for _ in range(n_points)]

        result_de = opt.differential_evolution(de_objective, bounds_de, constraints=cons_de, popsize=100, strategy='best1bin', tol=1e-6, maxiter=150, mutation=(0.8, 1.0), recombination=0.9, disp=False)
        h_opt = result_de.x
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale

    c5_bound = objective(h_opt)
    return h_opt.tolist(), c5_bound, n_points
