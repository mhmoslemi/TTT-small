import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def fitness(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def evaluate_candidate(h):
        return fitness(h)

    def objective(x):
        return evaluate_candidate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    iterations = 0

    def perturb(h, strength=0.1):
        return np.clip(h + strength * np.random.randn(n_points), 0, 1)

    def evolve():
        nonlocal best_h, best_c5, iterations
        for _ in range(1000):
            if time() - start_time > budget_s:
                break
            iterations += 1
            h = perturb(best_h)
            h = h / np.sum(h) * n_points / 2
            c5 = evaluate_candidate(h)
            if c5 < best_c5:
                best_h, best_c5 = h.copy(), c5

    def random_search():
        nonlocal best_h, best_c5, iterations
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            c5 = evaluate_candidate(h)
            if c5 < best_c5:
                best_h, best_c5 = h.copy(), c5

    evolve()
    random_search()

    if time() - start_time < budget_s:
        for _ in range(50):
            if time() - start_time > budget_s:
                break
            h = perturb(best_h)
            h = h / np.sum(h) * n_points / 2
            c5 = evaluate_candidate(h)
            if c5 < best_c5:
                best_h, best_c5 = h.copy(), c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
