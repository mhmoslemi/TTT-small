```python
import numpy as np
import scipy.optimize as opt
import time
import random

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def obj_func(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    def clamp_array(arr):
        return np.clip(arr, 0.0, 1.0)

    def normalize_array(arr):
        s = np.sum(arr)
        if s == 0:
            return arr
        return arr * (target_sum / s)

    def evolve_population(pop, n_gen, n_pop, mutation_rate, crossover_rate, fitness_func):
        for _ in range(n_gen):
            fitness = np.array([fitness_func(ind) for ind in pop])
            sorted_indices = np.argsort(fitness)
            top = pop[sorted_indices[:n_pop//2]]
            new_pop = []
            for _ in range(n_pop//2):
                parent1, parent2 = random.sample(top, 2)
                child = np.copy(parent1)
                for i in range(n_points):
                    if np.random.rand() < crossover_rate:
                        child[i] = parent2[i]
                if np.random.rand() < mutation_rate:
                    idx = np.random.randint(n_points)
                    child[idx] += np.random.uniform(-0.1, 0.1)
                    child = clamp_array(child)
                new_pop.append(child)
            pop = new_pop
            for _ in range(n_pop//2):
                rand_idx = np.random.randint(n_pop//2)
                pop.append(pop[rand_idx])
            pop = [normalize_array(ind) for ind in pop]
        return pop

    def refine_candidate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        def refined_obj(h_flat):
            h = h_flat.reshape(n_points)
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            corr = np.correlate(h, 1.0 - h, mode="full")
            c5 = np.max(corr * dx)
            return c5
        res = opt.minimize(refined_obj, h.flatten(), method='SLSQP', bounds=[(0.0, 1.0)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-8, options={'maxiter': 200})
        return res.x

    def evaluate_candidate(h_flat):
        h = h_flat.reshape(n_points)
        h = normalize_array(h)
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    start_time = time.time()
    h_flat_init = h.flatten()
    h_flat_init = refine_candidate(h_flat_init)
    best_h = h_flat_init
    best_c5 = evaluate_candidate(best_h)

    if budget_s > time.time() - start_time:
        population = [h_flat_init]
        for _ in range(10):
            population.append(np.random.uniform(0.0, 1.0, n_points))
        population.append(np.random.choice([0.0, 1.0], size=n_points, p=[0.5, 0.5]))
        population.append(np.random.choice([0.0, 1.0], size=n_points, p=[0.4, 0.6]))
        population.append(np.random.choice([0.0, 1.0], size=n_points, p=[0.3, 0.7]))
        population.append(np.random.choice([0.0, 1.0], size=n_points, p=[0.25, 0.75]))
        population = [normalize_array(p) for p in population]
        population = [refine_candidate(p) for p in population]
        population = [p.flatten() for p in population]
        population = [np.clip(p, 0.0, 1.0) for p in population]

        population = evolve_population(population, 50, 50, 0.2, 0.8, evaluate_candidate)
        for candidate in population:
            current_c5 = evaluate_candidate(candidate)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = candidate

    h_opt = best_h
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(3):
            h_flat = np.random.uniform(0.0, 1.0, n_points)
            h_flat = normalize_array(h_flat)
            h_flat = refine_candidate(h_flat)
            current_c5 = evaluate_candidate(h_flat)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_flat

    c5_bound = best_c5
    return h_opt.tolist(), c5_bound, n_points
```
