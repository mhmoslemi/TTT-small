import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', None)
    
    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    if initial_h_values is not None:
        h = np.clip(initial_h_values, 0, 1).copy()
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    
    start_time = time.time()
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        
        # Perturb the current solution
        perturbation = np.random.normal(0, 0.05, n_points)
        h = h + perturbation
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        current_c5 = evaluate_candidate(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h, best_c5, n_points)
