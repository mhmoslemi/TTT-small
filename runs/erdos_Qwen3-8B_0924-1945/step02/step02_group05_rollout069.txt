import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)
    
    def fitness(h):
        return evaluate_candidate(h)
    
    def gradient_update(h, k):
        dx = 2.0 / n_points
        max_corr = np.max(np.correlate(h, 1 - h, mode="full") * dx)
        if max_corr < 0.3805:
            return h
        corr_vals = np.correlate(h, 1 - h, mode="full") * dx
        target_k = np.argmax(corr_vals)
        if target_k < 0 or target_k >= len(corr_vals):
            return h
        x_start = max(0, target_k - 10)
        x_end = min(n_points - 1, target_k + 10)
        grad = np.zeros(n_points)
        for x in range(x_start, x_end + 1):
            if x + target_k < n_points:
                grad[x] += (1 - h[x + target_k])
            if x - target_k >= 0:
                grad[x - target_k] -= h[x]
        step_size = 0.01
        h = h - step_size * grad
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return h
    
    def objective(x):
        return evaluate_candidate(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    
    for _ in range(1000):
        if time() - start_time > budget_s - 30:
            break
        best_h = gradient_update(best_h, 0)
        current_c5 = evaluate_candidate(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
        else:
            break
    
    for _ in range(50):
        if time() - start_time > budget_s - 30:
            break
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
        current_c5 = evaluate_candidate(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    final_c5 = evaluate_candidate(best_h)
    
    return (best_h, final_c5, n_points)
