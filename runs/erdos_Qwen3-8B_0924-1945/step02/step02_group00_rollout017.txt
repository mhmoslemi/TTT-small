```python
import numpy as np
import scipy.optimize as opt
import time
import random
from deap import base, creator, tools, algorithms

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)
    
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum
    
    def generate_initial_population():
        pop = []
        for _ in range(5):
            if initial_h_values is not None:
                base = np.array(initial_h_values, dtype=np.float64)
            else:
                base = np.zeros(n_points, dtype=np.float64)
                for i in range(n_points):
                    base[i] = 1.0 if i % 5 == 0 else 0.0
                base = base * (target_sum / np.sum(base))
            base = np.clip(base, 0.0, 1.0)
            noise = np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(base + noise, 0.0, 1.0)
            scaled = perturbed * (target_sum / np.sum(perturbed))
            pop.append(scaled)
        return np.array(pop)
    
    def evolve_population(pop, n_gen, cxpb, mutpb):
        creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
        creator.create("Individual", np.ndarray, fitness=creator.FitnessMin)
        
        stats = tools.Statistics(lambda ind: ind.fitness.values)
        stats.register("avg", np.mean)
        stats.register("std", np.std)
        stats.register("min", np.min)
        stats.register("max", np.max)
        
        toolbox = base.Toolbox()
        toolbox.register("evaluate", objective)
        toolbox.register("mate", tools.cxBlend, alpha=0.5)
        toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
        toolbox.register("select", tools.selTournament, tournsize=5)
        toolbox.register("population", tools初始化, n=n_points)
        toolbox.register("individual", tools.initRepeat, creator.Individual, np.random.uniform, low=0, high=1, k=n_points)
        
        pop = algorithms.varAnd(pop, toolbox, cxpb=cxpb, mutpb=mutpb)
        return pop
    
    def refine_solution(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        res = opt.minimize(lambda x: objective(x), h.flatten(), 
                          method='SLSQP', bounds=[(0, 1)]*n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          tol=1e-8, options={'maxiter': 100})
        return res.x
    
    def evaluate_candidate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx), h
    
    def create_candidate():
        while True:
            h = np.random.rand(n_points)
            h = h * (target_sum / np.sum(h))
            h = np.clip(h, 0.0, 1.0)
            if np.abs(np.sum(h) - target_sum) < 1e-6:
                return h
    
    def search():
        best_c5 = float('inf')
        best_h = np.zeros(n_points)
        
        pop = generate_initial_population()
        for _ in range(20):
            for i in range(len(pop)):
                h = refine_solution(pop[i])
                c5, _ = evaluate_candidate(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
                pop[i] = h
        
        for _ in range(10):
            new_pop = []
            for _ in range(len(pop)):
                h = create_candidate()
                new_pop.append(h)
            pop = new_pop
        
        for _ in range(10):
            new_pop = []
            for _ in range(len(pop)):
                h = refine_solution(pop[_])
                c5, _ = evaluate_candidate(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
                new_pop.append(h)
            pop = new_pop
        
        return best_h, best_c5
    
    start_time = time.time()
    h_opt, c5_bound = search()
    elapsed_time = time.time() - start_time
    
    if elapsed_time < budget_s:
        h_opt = refine_solution(h_opt)
        c5_bound = objective(h_opt)
    
    return h_opt.tolist(), c5_bound, n_points
```
