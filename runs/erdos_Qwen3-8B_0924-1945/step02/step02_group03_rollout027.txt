import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy() if 'initial_h_values' in kwargs else np.random.uniform(0, 1, n_points)
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def compute_c5(h):
        return evaluate_candidate(h)

    def objective(x):
        return compute_c5(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    def simulated_annealing(start, budget, temp_start=1.0, temp_end=1e-5, cooling=0.995):
        x = start.copy()
        best_x = x.copy()
        best_val = compute_c5(x)
        t = temp_start
        for _ in range(budget):
            if time() - start_time > budget_s:
                break
            perturbation = np.random.uniform(-0.1, 0.1, n_points)
            trial = np.clip(x + perturbation, 0, 1)
            trial = trial / np.sum(trial) * n_points / 2
            trial_val = compute_c5(trial)
            if trial_val < best_val:
                best_val = trial_val
                best_x = trial.copy()
                x = trial.copy()
            else:
                delta = trial_val - best_val
                if np.random.rand() < np.exp(-delta / t):
                    x = trial.copy()
            t *= cooling
        return best_x, best_val

    def genetic_algorithm(population, budget, mutation_rate=0.1, tournament_size=5):
        population = [np.clip(x, 0, 1) for x in population]
        population = [x / np.sum(x) * n_points / 2 for x in population]
        best_x = min(population, key=compute_c5)
        best_val = compute_c5(best_x)
        for _ in range(budget):
            if time() - start_time > budget_s:
                break
            parents = []
            for _ in range(tournament_size):
                candidates = np.random.choice(population, size=5)
                winner = min(candidates, key=compute_c5)
                parents.append(winner)
            parent1, parent2 = np.random.choice(parents, size=2, replace=False)
            child = (parent1 + parent2) / 2
            mutation_mask = np.random.rand(n_points) < mutation_rate
            child[mutation_mask] += np.random.uniform(-0.1, 0.1, np.sum(mutation_mask))
            child = np.clip(child, 0, 1)
            child = child / np.sum(child) * n_points / 2
            child_val = compute_c5(child)
            if child_val < best_val:
                best_x = child.copy()
                best_val = child_val
            population.append(child)
            if len(population) > 30:
                population.pop(np.argmax([compute_c5(x) for x in population]))
        return best_x, best_val

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    h_sa = best_h.copy()
    h_sa, _ = simulated_annealing(h_sa, int(budget_s * 0.3))
    best_h = h_sa.copy()
    best_c5 = evaluate_candidate(best_h)

    h_ga = best_h.copy()
    h_ga, _ = genetic_algorithm([h_ga], int(budget_s * 0.7))
    best_h = h_ga.copy()
    best_c5 = evaluate_candidate(best_h)

    return (best_h, best_c5, n_points)
