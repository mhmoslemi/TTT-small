import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    if 'initial_h_values' in kwargs:
        h = np.array(kwargs['initial_h_values'], dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)
    
    def objective(x):
        return evaluate_candidate(x)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    
    def de_search():
        pop_size = 30
        mut_factor = 0.5
        cross_prob = 0.8
        max_iter = 300
        pop = np.random.uniform(0, 1, (pop_size, n_points))
        pop = np.clip(pop, 0, 1)
        pop = pop / np.sum(pop, axis=1, keepdims=True) * n_points / 2
        best_pop = pop.copy()
        best_c5_pop = np.full(pop_size, np.inf)
        
        for _ in range(max_iter):
            if time() - start_time > budget_s:
                break
            for i in range(pop_size):
                idxs = np.random.choice(pop_size, 3, replace=False)
                mutant = pop[idxs[0]] + mut_factor * (pop[idxs[1]] - pop[idxs[2]])
                mutant = np.clip(mutant, 0, 1)
                mutant = mutant / np.sum(mutant) * n_points / 2
                trial = np.copy(pop[i])
                for j in range(n_points):
                    if np.random.rand() < cross_prob:
                        trial[j] = mutant[j]
                trial = np.clip(trial, 0, 1)
                trial = trial / np.sum(trial) * n_points / 2
                c5_trial = compute_c5(trial)
                if c5_trial < best_c5_pop[i]:
                    best_c5_pop[i] = c5_trial
                    pop[i] = trial
            best_idx = np.argmin(best_c5_pop)
            best_h = pop[best_idx].copy()
            best_c5 = best_c5_pop[best_idx]
        
        return best_h, best_c5
    
    best_h, best_c5 = de_search()
    
    def sqp_refinement(x):
        res = minimize(objective, x, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5, options={'max_iter': 100})
        if res.success:
            return res.x
        else:
            return x
    
    best_h = sqp_refinement(best_h)
    best_c5 = evaluate_candidate(best_h)
    
    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    final_c5 = evaluate_candidate(best_h)
    return (best_h, final_c5, n_points)
