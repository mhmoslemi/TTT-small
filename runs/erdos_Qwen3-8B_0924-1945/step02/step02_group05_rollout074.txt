import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if np.random.rand() < 0.5 else 0
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    dx = 2.0 / n_points

    def evaluate_candidate(h_vec):
        h = normalize(h_vec)
        return compute_c5(h)

    def random_binary_pattern():
        h = np.random.rand(n_points)
        h = (h < 0.6).astype(float)
        return normalize(h)

    def uniform_pattern(m, frac):
        step = 2.0 / m
        intervals = np.arange(0, 2, step)
        active = np.random.choice(np.arange(m), size=int(m * frac), replace=False)
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            for j in range(m):
                if intervals[j] <= x < intervals[j + 1]:
                    if j in active:
                        h[i] = 1
                    break
        return normalize(h)

    def bilinear_pattern(slope):
        h = 0.5 * (1 + np.tanh(slope * (np.arange(n_points) * dx - 1)))
        return normalize(h)

    def run_mcmc():
        current_h = normalize(h.copy())
        best_h = current_h.copy()
        best_c5 = compute_c5(best_h)
        temp = 10.0
        temp_decay = 0.995
        iter_count = 0
        max_iter = 500
        while iter_count < max_iter and time() < budget_s:
            iter_count += 1
            h_candidate = current_h.copy()
            perturb = np.random.normal(0, 0.1, n_points)
            h_candidate += perturb
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = normalize(h_candidate)
            c5_candidate = compute_c5(h_candidate)
            delta_c5 = c5_candidate - best_c5
            if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temp):
                current_h = h_candidate.copy()
                if c5_candidate < best_c5:
                    best_h = current_h.copy()
                    best_c5 = c5_candidate
            temp *= temp_decay
        return best_h, best_c5

    best_h, best_c5 = run_mcmc()

    if time() < budget_s:
        for _ in range(10):
            if time() >= budget_s:
                break
            h = random_binary_pattern()
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5

    best_h = normalize(best_h)
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n_points)
