import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    
    def validate_h(h):
        h = np.clip(h, 0, 1)
        norm = np.sum(h)
        if np.isclose(norm, n_points / 2):
            return h
        return h * (n_points / 2) / norm
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def objective(h):
        return compute_c5(validate_h(h))
    
    def generate_structured_patterns():
        patterns = []
        # Uniform intervals with 3-7 intervals
        for n_intervals in range(3, 8):
            widths = np.repeat(2.0 / n_intervals, n_intervals)
            intervals = np.cumsum(widths)
            h = np.zeros(n_points)
            for i in range(n_intervals):
                start = int(intervals[i] * n_points / 2)
                end = int(intervals[i+1] * n_points / 2)
                h[start:end] = 1
            h = validate_h(h)
            patterns.append(h)
        return patterns
    
    def initial_population():
        base_patterns = generate_structured_patterns()
        if 'initial_h_values' in kwargs:
            base_patterns.append(validate_h(initial_h_values))
        # Add a random binary pattern
        h = np.random.choice([0, 1], size=n_points)
        h = validate_h(h)
        base_patterns.append(h)
        # Add a smooth interpolation pattern
        h = np.linspace(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = validate_h(h)
        base_patterns.append(h)
        return np.array(base_patterns)
    
    initial_population = initial_population()
    best_h = initial_population[np.argmin([compute_c5(p) for p in initial_population])]
    best_c5 = compute_c5(best_h)
    
    start_time = time()
    
    def optimize():
        nonlocal best_h, best_c5
        pop_size = 5
        bounds = [(0, 1) for _ in range(n_points)]
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=pop_size,
            tol=1e-5,
            maxiter=300,
            mutation=(0.8, 1),
            recombination=0.9,
            seed=seed,
            callback=lambda x: None
        )
        
        if result.success:
            candidate = result.x
            current_c5 = compute_c5(validate_h(candidate))
            if current_c5 < best_c5:
                best_h = validate_h(candidate)
                best_c5 = current_c5
        
        # Final population evaluation
        for h in initial_population:
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5
    
    optimize()
    
    # Final refinement
    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = validate_h(h)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5
    
    best_h = validate_h(best_h)
    current_c5 = compute_c5(best_h)
    return (best_h, current_c5, n_points)
