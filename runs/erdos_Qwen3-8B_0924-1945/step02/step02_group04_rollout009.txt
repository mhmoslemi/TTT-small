import numpy as np
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = np.clip(initial_h_values, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def compute_c5(h):
        return evaluate_candidate(h)
    
    swarm_size = 30
    particles = [np.random.uniform(0, 1, n_points) for _ in range(swarm_size)]
    particles = [p * (n_points / 2) / np.sum(p) for p in particles]
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    pbest = particles[:]
    gbest = min(particles, key=compute_c5)
    best_c5 = compute_c5(gbest)
    best_h = gbest
    
    for _ in range(1000):
        if time.time() - time.time() > budget_s:
            break
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = 0.8 * velocities[i] + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            current_c5 = compute_c5(particles[i])
            if current_c5 < compute_c5(pbest[i]):
                pbest[i] = particles[i].copy()
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = particles[i].copy()
                gbest = best_h
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    return (best_h, best_c5, n_points)
