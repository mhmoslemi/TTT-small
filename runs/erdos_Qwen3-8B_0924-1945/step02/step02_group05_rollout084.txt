import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 50
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    
    # Validate and normalize initial_h_values
    h = np.clip(initial_h_values, 0, 1)
    h = h * (n_points / 2) / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        return compute_c5(h)
    
    def objective(x):
        return evaluate_candidate(x)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def structured_initializations(n_points):
        patterns = []
        # Uniform interval patterns
        for num_intervals in range(3, 8):
            intervals = np.linspace(0, 2, num_intervals + 1)
            h = np.zeros(n_points)
            for i in range(num_intervals):
                lower = np.searchsorted(intervals, 0.0 + i * 2.0 / num_intervals)
                upper = np.searchsorted(intervals, 0.0 + (i + 1) * 2.0 / num_intervals)
                h[lower:upper] = 1.0
            h = h * (n_points / 2) / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
            patterns.append(h)
        # Random binary pattern
        h = np.random.choice([0, 1], size=n_points)
        h = h * (n_points / 2) / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        patterns.append(h)
        # Symmetric bilinear
        slope = np.random.uniform(-0.5, 0.5)
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.tanh(slope * (x - 1)))
        h = h * (n_points / 2) / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        patterns.append(h)
        return patterns
    
    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    
    # Generate structured initializations
    initial_population = structured_initializations(n_points)
    initial_population.append(h)
    initial_population = np.array(initial_population)
    
    # Optimization with differential evolution
    result = differential_evolution(
        objective,
        bounds,
        popsize=10,
        strategy='best1bin',
        mutation=(0.5, 1),
        recombination=0.9,
        tol=1e-5,
        polish=True,
        maxiter=300,
        disp=False,
        seed=seed
    )
    
    best_h = result.x
    best_c5 = evaluate_candidate(best_h)
    
    # Final refinement with randomization
    if time() - start_time < budget_s:
        for _ in range(50):
            if time() - start_time > budget_s:
                break
            h_rand = np.random.uniform(0, 1, n_points)
            h_rand = h_rand * (n_points / 2) / np.sum(h_rand) if np.sum(h_rand) > 0 else np.zeros(n_points)
            current_c5 = evaluate_candidate(h_rand)
            if current_c5 < best_c5:
                best_h = h_rand.copy()
                best_c5 = current_c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h) if np.sum(best_h) > 0 else np.zeros(n_points)
    final_c5 = evaluate_candidate(best_h)
    return (best_h, final_c5, n_points)
