import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    bounds = [(0, 1) for _ in range(n_points)]

    def objective(h):
        h_clipped = np.clip(h, 0, 1)
        h_scaled = h_clipped / np.sum(h_clipped) * n_points / 2
        return compute_c5(h_scaled)

    def constraint(h):
        h_clipped = np.clip(h, 0, 1)
        return np.sum(h_clipped) - n_points / 2

    start_time = time()
    result = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=15, tol=1e-5, mutation=0.5, recombination=0.8,
        maxiter=500, disp=False, polish=False, seed=seed
    )
    best_h = result.x
    best_c5 = compute_c5(np.clip(best_h, 0, 1) / np.sum(np.clip(best_h, 0, 1)) * n_points / 2)

    if time() - start_time < budget_s:
        best_h = np.random.uniform(0, 1, n_points)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h / np.sum(best_h) * n_points / 2
        best_c5 = compute_c5(best_h)

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = compute_c5(best_h)
    return (best_h, current_c5, n_points)
