import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    # Genetic algorithm
    population = [np.random.uniform(0, 1, n_points) for _ in range(50)]
    population = [h / np.sum(h) * n_points / 2 for h in population]
    population = [np.clip(p, 0, 1) for p in population]
    population = [p for p in population if np.isclose(np.sum(p), n_points / 2, atol=1e-6)]
    population = population[:50]

    for _ in range(500):
        if time() - start_time > budget_s:
            break
        # Tournament selection
        indices = np.random.choice(len(population), 5, replace=False)
        candidates = [population[i] for i in indices]
        best_candidate = min(candidates, key=evaluate_candidate)
        # Crossover
        parent1 = best_candidate
        parent2 = np.random.choice(population, 1)[0]
        offspring = (parent1 + parent2) / 2
        # Mutation
        mutation_rate = 0.2
        if np.random.rand() < mutation_rate:
            mutation_strength = 0.1
            idx = np.random.randint(n_points)
            offspring[idx] += np.random.uniform(-mutation_strength, mutation_strength)
            offspring = np.clip(offspring, 0, 1)
            offspring = offspring / np.sum(offspring) * n_points / 2
        # Evaluate
        current_c5 = evaluate_candidate(offspring)
        if current_c5 < best_c5:
            best_h = offspring.copy()
            best_c5 = current_c5
        # Replace worst
        if np.random.rand() < 0.2:
            population.remove(best_candidate)
            population.append(offspring)
            population = [p for p in population if np.isclose(np.sum(p), n_points / 2, atol=1e-6)]
            population = population[:50]

    if time() - start_time < budget_s:
        # Local refinement
        for _ in range(5):
            if time() - start_time > budget_s:
                break
            x0 = best_h.copy()
            res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
            if res.success:
                current_h = res.x
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h / np.sum(current_h) * n_points / 2
                current_c5 = evaluate_candidate(current_h)
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
            else:
                break
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
