import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    # Differential evolution setup
    def objective(x):
        return evaluate_candidate(x)

    def _evaluate(x):
        return objective(x)

    # DE parameters
    pop_size = 15
    mutation_factor = 0.5
    crossover_prob = 0.8
    max_iter = 500

    # Run DE
    result = differential_evolution(
        func=objective,
        bounds=bounds,
        strategy='best1bin',
        popsize=pop_size,
        mutation=(mutation_factor, mutation_factor),
        recombination=crossover_prob,
        tol=1e-6,
        polish=True,
        disp=False,
        maxiter=max_iter,
        seed=seed
    )

    best_h = result.x
    best_c5 = evaluate_candidate(best_h)

    # Ensure we don't exceed budget
    if time() - start_time < budget_s:
        # Fallback to random sampling if DE is not sufficient
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    # Final checks
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    final_c5 = evaluate_candidate(best_h)

    return (best_h, final_c5, n_points)
