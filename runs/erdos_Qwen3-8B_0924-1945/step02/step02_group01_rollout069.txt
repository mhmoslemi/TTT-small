```python
import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 4 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    population_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    elitism_rate = 0.2
    tournament_size = 5
    generations = 200
    max_time = budget_s - 100

    def create_individual():
        indices = np.random.choice(n_points, size=int(n_points/2), replace=False)
        h = np.zeros(n_points, dtype=np.float64)
        h[indices] = 1.0
        h = h * (target_sum / np.sum(h))
        return h

    population = [create_individual() for _ in range(population_size)]
    best_h = population[0]
    best_c5 = evaluate(best_h)

    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > max_time:
            break
        parents = []
        for _ in range(2):
            competitors = np.random.choice(population, size=tournament_size, replace=False)
            winner = min(competitors, key=evaluate)
            parents.append(winner)
        offspring = []
        for p1, p2 in zip(parents[::2], parents[1::2]):
            if np.random.rand() < crossover_rate:
                child = np.random.rand(n_points)
                child[child < 0.5] = p1[child < 0.5]
                child[child >= 0.5] = p2[child >= 0.5]
            else:
                child = p1.copy()
            mutation_mask = np.random.rand(n_points) < mutation_rate
            child[mutation_mask] += np.random.normal(0, 0.1, size=np.sum(mutation_mask))
            child = np.clip(child, 0.0, 1.0)
            sum_h = np.sum(child)
            child = child * (target_sum / sum_h if sum_h != 0 else 1.0)
            offspring.append(child)
        population += offspring
        population.sort(key=evaluate)
        population = population[:int(population_size * (1 - elitism_rate))]
        current_c5 = evaluate(population[0])
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[0]

    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    best_h = best_h * (target_sum / sum_h if sum_h != 0 else 1.0)
    c5_bound = evaluate(best_h)
    return best_h.tolist(), c5_bound, n_points
```
