import numpy as np
import scipy.optimize as opt
import time
import random
import DEAP

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))
    
    def objective(h_flat):
        h = h_flat.reshape(n_points) * (target_sum / np.sum(h_flat))
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - target_sum
    
    def constraint_bounds(h_flat):
        return np.clip(h_flat, 0, 1)
    
    def validate(h_flat):
        h = h_flat.reshape(n_points)
        if np.any(h < 0) or np.any(h > 1):
            return np.inf
        if np.abs(np.sum(h) - target_sum) > 1e-6:
            return np.inf
        return 0
    
    def evolve(population):
        offspring = []
        for ind in population:
            ind[:] = np.clip(ind, 0, 1)
            if np.abs(np.sum(ind) - target_sum) > 1e-4:
                ind = ind * (target_sum / np.sum(ind))
        return population
    
    def build_initial_pattern():
        np.random.seed(seed)
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 if np.random.rand() < 0.5 else 0.0
        h = h * (target_sum / np.sum(h))
        return h
    
    def generate_candidate_pattern():
        if initial_h_values is not None:
            h = np.array(initial_h_values, dtype=np.float64)
        else:
            h = build_initial_pattern()
        return h
    
    def evaluate_pattern(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    def evaluate_population(population):
        scores = []
        for ind in population:
            score = evaluate_pattern(ind)
            scores.append(score)
        return np.array(scores)
    
    def select_parents(scores, population):
        tournament = np.random.choice(len(population), size=5)
        best = tournament[np.argmin(scores[tournament])]
        return population[best].copy()
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = parent2[i]
        return child
    
    def mutate(ind):
        for i in range(n_points):
            if np.random.rand() < 0.1:
                ind[i] = 1.0 - ind[i]
        ind = ind * (target_sum / np.sum(ind))
        return ind
    
    def get_best(population, scores):
        best_idx = np.argmin(scores)
        return population[best_idx].copy()
    
    def evolve_population(population, scores):
        new_population = []
        for _ in range(len(population)):
            parent = select_parents(scores, population)
            child = crossover(parent, parent)
            child = mutate(child)
            new_population.append(child)
        return new_population
    
    def run_evolution():
        population = [generate_candidate_pattern() for _ in range(50)]
        for _ in range(500):
            scores = evaluate_population(population)
            population = evolve_population(population, scores)
            best = get_best(population, scores)
            if np.abs(np.sum(best) - target_sum) <= 1e-6:
                break
        return best
    
    def optimize_with_slsqp(h):
        def objective_func(x):
            h_flat = x.reshape(n_points)
            h_flat = np.clip(h_flat, 0, 1)
            h_flat = h_flat * (target_sum / np.sum(h_flat))
            corr = np.correlate(h_flat, 1.0 - h_flat, mode="full")
            c5 = np.max(corr * dx)
            return c5
        
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.minimize(objective_func, h.flatten(), method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 300})
        return result.x
    
    start_time = time.time()
    best_h = generate_candidate_pattern()
    best_c5 = evaluate_pattern(best_h)
    
    for _ in range(3):
        candidate_h = run_evolution()
        candidate_c5 = evaluate_pattern(candidate_h)
        if candidate_c5 < best_c5:
            best_h = candidate_h
            best_c5 = candidate_c5
    
    best_h = optimize_with_slsqp(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h))
    
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 50:
        for _ in range(5):
            candidate_h = run_evolution()
            candidate_c5 = evaluate_pattern(candidate_h)
            if candidate_c5 < best_c5:
                best_h = candidate_h
                best_c5 = candidate_c5
    
    best_h = optimize_with_slsqp(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h))
    
    return best_h.tolist(), best_c5, n_points
