import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy() if 'initial_h_values' in kwargs else np.zeros(n_points)
    for i in range(n_points):
        h[i] = 1 if np.random.rand() < 0.5 else 0
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)
    
    def fitness(h):
        return evaluate_candidate(h)
    
    def objective(x):
        return fitness(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time()
    
    def generate_pattern(pattern_type):
        if pattern_type == 'uniform':
            m = np.random.randint(3, 8)
            intervals = np.random.choice(np.arange(1, m), size=np.random.randint(1, m//2+1), replace=False)
            h = np.zeros(n_points)
            for i in range(n_points):
                x = i * 2.0 / n_points
                for interval in intervals:
                    if (x >= (interval - 0.5)/m and x < (interval + 0.5)/m):
                        h[i] = 1
                        break
            h = h / np.sum(h) * n_points / 2
            return h
        elif pattern_type == 'bilinear':
            slope = np.random.uniform(-2, 2)
            h = 0.5 * (1 + np.tanh(slope * (np.linspace(0, 2, n_points) - 1)))
            h = h / np.sum(h) * n_points / 2
            return h
        elif pattern_type == 'random':
            h = np.random.rand(n_points)
            h = h / np.sum(h) * n_points / 2
            return h
        else:
            return np.zeros(n_points)
    
    population = [generate_pattern(np.random.choice(['uniform', 'bilinear', 'random'])) for _ in range(5)]
    
    def optimize_candidate(h):
        best_h = h.copy()
        best_c5 = evaluate_candidate(best_h)
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
            if res.success:
                current_h = res.x
                current_c5 = evaluate_candidate(current_h)
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
            else:
                break
        return best_h, best_c5
    
    best_h, best_c5 = optimize_candidate(h)
    
    if time() - start_time < budget_s:
        for _ in range(50):
            if time() - start_time > budget_s:
                break
            h = np.random.rand(n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
