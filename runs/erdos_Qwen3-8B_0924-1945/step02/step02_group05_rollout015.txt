import numpy as np
from scipy.optimize import minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if np.random.rand() < 0.8 else 0
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def evaluate_and_clip(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def gradient_step(h, k, step_size=0.01):
        dx = 2.0 / n_points
        start = max(0, k - 10)
        end = min(n_points - 1, k + 10)
        for i in range(start, end + 1):
            if i + k < n_points:
                h[i] -= step_size * (1 - h[i + k])
            if i - k >= 0:
                h[i - k] += step_size * h[i]
        return h

    def run_optimization():
        best_h = h.copy()
        best_c5 = evaluate_candidate(best_h)
        start_time = time()
        for _ in range(1000):
            if time() - start_time > budget_s:
                break
            corr_vals = np.correlate(best_h, 1 - best_h, mode="full") * 2.0 / n_points
            max_k = np.argmax(corr_vals)
            h = gradient_step(best_h.copy(), max_k)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_and_clip(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            else:
                break
        return best_h, best_c5

    best_h, best_c5 = run_optimization()

    if time() - start_time < budget_s:
        for _ in range(200):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_and_clip(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_and_clip(best_h)
    return (best_h, current_c5, n_points)
