```python
import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        x_grid = np.linspace(0, 2, n_points)
        h = 0.5 + 0.25 * np.sin(np.pi * x_grid) + 0.25 * np.cos(2 * np.pi * x_grid)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        h = h * (target_sum / sum_h)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()

    # Differential Evolution for Global Search
    def de_objective(h_flat):
        return objective(h_flat)

    result_de = opt.differential_evolution(
        de_objective, bounds, strategy='best1bin', popsize=20,
        mutation=(0.5, 1.0), recombination=0.9, tol=1e-5,
        maxiter=100, disp=False, callback=lambda x: None
    )
    h_de = result_de.x
    c5_de = objective(h_de)
    if c5_de < best_c5:
        best_c5 = c5_de
        best_h = h_de.copy()

    # Local Refinement with SLSQP
    result_slsqp = opt.minimize(
        objective, best_h, method='SLSQP', bounds=bounds, constraints=cons,
        tol=1e-8, options={'maxiter': 100}
    )
    h_slsqp = result_slsqp.x
    c5_slsqp = objective(h_slsqp)
    if c5_slsqp < best_c5:
        best_c5 = c5_slsqp
        best_h = h_slsqp.copy()

    if time.time() - start_time < budget_s - 5:
        # Final refinement
        result_final = opt.minimize(
            objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons,
            tol=1e-8, options={'maxiter': 50}
        )
        h_final = result_final.x
        c5_final = objective(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    return best_h.tolist(), best_c5, n_points
```
