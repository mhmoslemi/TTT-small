```python
import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    target_sum = n_points / 2
    pop_size = 50
    generations = 200
    mutation_rate = 0.1
    crossover_rate = 0.8
    elitism_rate = 0.2
    tournament_size = 5

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        ones = np.random.choice(n_points, size=int(n_points / 2), replace=False)
        h[ones] = 1.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr * dx)
        return max_corr

    def mutate(h):
        h = h.copy()
        mask = np.random.rand(n_points) < mutation_rate
        h[mask] += np.random.normal(0, 0.1, sum(mask))
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        return h

    def crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        mask = np.random.rand(n_points) < crossover_rate
        child[mask] = parent1[mask]
        child[~mask] = parent2[~mask]
        sum_h = np.sum(child)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        child = child * scale
        return child

    def tournament_selection(population):
        indices = np.random.choice(len(population), tournament_size, replace=False)
        best = population[indices[0]]
        for idx in indices[1:]:
            if evaluate(population[idx]) < evaluate(best):
                best = population[idx]
        return best

    population = []
    for _ in range(pop_size):
        h = np.zeros(n_points, dtype=np.float64)
        ones = np.random.choice(n_points, size=int(n_points / 2), replace=False)
        h[ones] = 1.0
        h = h * (target_sum / np.sum(h))
        population.append(h)

    best_h = population[0]
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for _ in range(pop_size // 2):
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        population = new_population
        population += [best_h] * int(elitism_rate * pop_size)
        for h in population:
            if evaluate(h) < best_c5:
                best_h = h
                best_c5 = evaluate(h)
        if gen % 10 == 0 and time.time() - start_time > budget_s:
            break

    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    best_h = best_h * scale
    return best_h.tolist(), best_c5, n_points
```
