```python
import numpy as np
import scipy.optimize as opt
import time
import random

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def generate_initial_pattern():
        if initial_h_values is not None:
            h = np.array(initial_h_values, dtype=np.float64)
            h = h * (target_sum / np.sum(h))
            h = np.clip(h, 0.0, 1.0)
            return h
        else:
            h = np.zeros(n_points, dtype=np.float64)
            for i in range(n_points):
                h[i] = 1.0 if i % 5 == 0 else 0.0
            h = h * (target_sum / np.sum(h))
            return h

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    h = generate_initial_pattern()

    def inject_random_perturbation(h):
        perturb = np.random.normal(0, 0.05, h.shape)
        h_perturbed = np.clip(h + perturb, 0.0, 1.0)
        sum_perturbed = np.sum(h_perturbed)
        scale = target_sum / sum_perturbed if sum_perturbed != 0 else 1.0
        return h_perturbed * scale

    h = inject_random_perturbation(h)

    def evaluate_c5(h):
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    def evolve_population(population, mutation_rate=0.1, crossover_rate=0.7):
        new_population = []
        for i in range(len(population)):
            if random.random() < crossover_rate:
                j = (i + random.randint(1, len(population)-1)) % len(population)
                parent1, parent2 = population[i], population[j]
                child = np.copy(parent1)
                mask = np.random.rand(n_points) < 0.5
                child[mask] = parent2[mask]
                child = np.clip(child, 0.0, 1.0)
                sum_child = np.sum(child)
                scale = target_sum / sum_child if sum_child != 0 else 1.0
                child = child * scale
            else:
                child = np.copy(population[i])
            if random.random() < mutation_rate:
                mutation = np.random.normal(0, 0.05, n_points)
                child += mutation
                child = np.clip(child, 0.0, 1.0)
                sum_child = np.sum(child)
                scale = target_sum / sum_child if sum_child != 0 else 1.0
                child = child * scale
            new_population.append(child)
        return new_population

    population = [np.random.uniform(0, 1, n_points) for _ in range(50)]
    for p in population:
        sum_p = np.sum(p)
        scale = target_sum / sum_p if sum_p != 0 else 1.0
        p = p * scale

    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        population = evolve_population(population)
        for p in population:
            sum_p = np.sum(p)
            scale = target_sum / sum_p if sum_p != 0 else 1.0
            p = p * scale
        current_c5 = [evaluate_c5(p) for p in population]
        best_idx = np.argmin(current_c5)
        best_c5 = current_c5[best_idx]
        best_h = population[best_idx]
        if best_c5 < best_c5:
            best_c5 = best_c5
            best_h = population[best_idx]

    def refine_solution(h):
        def objective_refine(h_flat):
            h = h_flat.reshape(n_points)
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            corr = np.correlate(h, 1.0 - h, mode="full")
            c5 = np.max(corr * dx)
            return c5
        cons = [{'type': 'eq', 'fun': constraint}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.minimize(objective_refine, h.flatten(), method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 300})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        c5_refined = evaluate_c5(h_opt)
        return h_opt, c5_refined

    best_h, best_c5 = refine_solution(best_h)

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        result = opt.minimize(objective, best_h.flatten(), method='SLSQP', bounds=[(0.0, 1.0)]*n_points, constraints=cons, tol=1e-8, options={'maxiter': 300})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        best_c5 = evaluate_c5(h_opt)

    return best_h.tolist(), best_c5, n_points
```
