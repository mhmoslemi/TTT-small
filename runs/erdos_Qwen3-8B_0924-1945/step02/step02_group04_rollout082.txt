import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize import basinhopping
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    def objective_func(h):
        return compute_c5(h)

    def perturbation_func(x):
        perturb = np.random.uniform(-0.2, 0.2, size=len(x))
        return x + perturb

    def run_basinhopping():
        best_h = h.copy()
        best_c5 = compute_c5(best_h)
        start_time = time.time()
        # Basin-hopping parameters
        niter = 100
        T = 1.0
        stepsize = 0.1
        for _ in range(niter):
            if time.time() - start_time > budget_s:
                break
            # Perturbation
            perturbed_h = perturbation_func(best_h)
            perturbed_h = normalize(perturbed_h)
            # Local optimization
            res = minimize(objective_func, perturbed_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, tol=1e-5)
            if res.success:
                current_h = res.x
                current_c5 = compute_c5(current_h)
                if current_c5 < best_c5:
                    best_h = current_h
                    best_c5 = current_c5
            T *= 0.99
        return best_h, best_c5

    # Fallback if basinhopping fails
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    try:
        best_h, best_c5 = run_basinhopping()
    except:
        best_h = h.copy()
        best_c5 = compute_c5(best_h)
    best_h = normalize(best_h)
    return (best_h, best_c5, n_points)
