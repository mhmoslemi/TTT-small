import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy() if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def generate_uniform_pattern(num_intervals):
        positions = np.linspace(0, 2, n_points)
        selected = np.random.choice(np.arange(n_points), size=num_intervals, replace=False)
        h_pattern = np.zeros(n_points)
        h_pattern[selected] = 1
        h_pattern = h_pattern / np.sum(h_pattern) * n_points / 2
        return h_pattern

    def generate_symmetric_pattern(slope):
        positions = np.linspace(0, 2, n_points)
        h_pattern = 0.5 * (1 + np.tanh(slope * (positions - 1)))
        h_pattern = h_pattern / np.sum(h_pattern) * n_points / 2
        return h_pattern

    def generate_random_binary():
        h_pattern = np.random.rand(n_points)
        h_pattern = h_pattern / np.sum(h_pattern) * n_points / 2
        return h_pattern

    def objective(x):
        return evaluate_candidate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()
    population = []

    def add_pattern(pattern):
        if len(population) < 5:
            population.append(pattern)

    add_pattern(generate_uniform_pattern(5))
    add_pattern(generate_uniform_pattern(7))
    add_pattern(generate_symmetric_pattern(2))
    add_pattern(generate_symmetric_pattern(4))
    add_pattern(generate_random_binary())

    def optimize_individual(x0):
        if time() - start_time > budget_s:
            return x0
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            return res.x
        return x0

    for i in range(len(population)):
        population[i] = optimize_individual(population[i])

    best_c5 = np.inf
    best_h = h.copy()

    for indiv in population:
        c5 = evaluate_candidate(indiv)
        if c5 < best_c5:
            best_c5 = c5
            best_h = indiv.copy()

    for _ in range(100):
        if time() - start_time > budget_s:
            break
        h = np.random.rand(n_points)
        h = h / np.sum(h) * n_points / 2
        c5 = evaluate_candidate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    final_c5 = evaluate_candidate(best_h)
    return (best_h, final_c5, n_points)
