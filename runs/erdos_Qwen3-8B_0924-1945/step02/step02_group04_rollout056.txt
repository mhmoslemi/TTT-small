import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    best_h = initial_h_values.copy()
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = compute_c5(best_h, n_points)
    
    dx = 2.0 / n_points
    bounds = [(0, 1) for _ in range(n_points)]
    
    def compute_c5(h, n_points):
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def objective(h):
        return compute_c5(h, n_points)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def objective_grad(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        max_index = np.argmax(correlate_vals)
        grad = np.zeros_like(h)
        for k in range(1, n_points):
            lag_vals = correlate_vals[n_points - k]
            dh = np.zeros_like(h)
            for i in range(n_points):
                if i + k < n_points:
                    dh[i] += (1 - h[i + k]) * dx
                if i - k >= 0:
                    dh[i] += h[i - k] * dx
            if k == max_index:
                grad += dh
        return grad
    
    start_time = time.time()
    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, jac=objective_grad, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = compute_c5(current_h, n_points)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        else:
            break
        if np.random.rand() < 0.3:
            perturbation = np.random.uniform(-0.05, 0.05, n_points)
            new_h = best_h + perturbation
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            current_c5 = compute_c5(new_h, n_points)
            if current_c5 < best_c5:
                best_h = new_h
                best_c5 = current_c5
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    return (best_h, best_c5, n_points)
