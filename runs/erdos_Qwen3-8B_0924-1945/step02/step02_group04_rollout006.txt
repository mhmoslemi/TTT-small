import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize import basinhopping
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    h = np.clip(initial_h_values, 0, 1)
    h = h * (n_points / 2) / np.sum(np.clip(h, 0, 1))
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def normalize(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(np.clip(h, 0, 1))
        return h
    
    def objective(h):
        return compute_c5(h)
    
    def evaluate_candidate(h):
        return objective(normalize(h))
    
    def perturb(h, stepsize=0.1):
        perturbation = np.random.uniform(-stepsize, stepsize, size=len(h))
        return normalize(h + perturbation)
    
    def run_basinhopping():
        best_h = h.copy()
        best_c5 = objective(best_h)
        T = 1.0
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            new_h = perturb(best_h)
            res = minimize(objective, new_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-5)
            if res.success:
                current_h = res.x
                current_c5 = objective(current_h)
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
            T *= 0.99
        return best_h, best_c5
    
    start_time = time.time()
    best_h, best_c5 = run_basinhopping()
    if time.time() - start_time < budget_s:
        for _ in range(50):
            if time.time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = normalize(h)
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    best_h = normalize(best_h)
    return (best_h, best_c5, n_points)
