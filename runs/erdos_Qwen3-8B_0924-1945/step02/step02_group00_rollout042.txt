import numpy as np
import time
import random
from deap import base, creator, tools, algorithms

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    def objective(individual):
        h = np.array(individual, dtype=np.float64)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)

        h = np.array(individual, dtype=np.float64)
        return np.sum(h) - target_sum

    def generate_initial_pattern():
        if initial_h_values is not None:
            h = np.array(initial_h_values, dtype=np.float64)
            h += np.random.rand(n_points) * 0.03
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            return h
        else:
            # Greedy placement with random density
            h = np.zeros(n_points)
            placed = 0
            while placed < n_points / 2:
                idx = np.random.randint(n_points)
                if h[idx] == 0:
                    h[idx] = 1.0
                    placed += 1

            scale = target_sum / sum_h if sum_h != 0 else 1.0
            return h * scale

    creator.create("Individual", list, fitness=creator.FitnessMin)

    def eval_func(individual):
        return objective(individual)

    def mut_func(individual, indpb):
        for i in range(len(individual)):
            if random.random() < indpb:
                individual[i] = 1 - individual[i]
        return individual,

    def cx_func(ind1, ind2):
        size = len(ind1)
        child1, child2 = np.copy(ind1), np.copy(ind2)

            if random.random() < 0.5:
                child1[i], child2[i] = child2[i], child1[i]
        return child1, child2

    def validate_h(individual):
        h = np.array(individual, dtype=np.float64)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        return h

    def optimize_pattern(pattern):
        def custom_evaluate(individual):
            h = validate_h(individual)
            corr = np.correlate(h, 1.0 - h, mode="full")
            c5 = np.max(corr * dx)
            return c5,

        toolbox = base.Toolbox()
        toolbox.register("evaluate", custom_evaluate)
        toolbox.register("mate", cx_func)
        toolbox.register("mutate", mut_func, indpb=0.1)
        toolbox.register("select", tools.selTournament, tournsize=5)

        pop = tools.population(indpb=0.1, ind_size=len(pattern), creator=creator.Individual)
        for i in range(len(pop)):
            pop[i] = validate_h(pattern)
        pop = tools.selTournament(pop, len(pop))
        stats = tools.Statistics(lambda ind: ind.fitness.values)
        stats.register("avg", np.mean)
        stats.register("std", np.std)
        stats.register("min", np.min)
        stats.register("max", np.max)

        # Run DE
        algorithm = algorithms.varOr
        for gen in range(500):
            offspring = algorithm(pop, len(pop), cxpb=0.9, mutpb=0.1, offspring_size=len(pop))
            offspring = list(map(validate_h, offspring))
            fits = [eval_func(ind) for ind in offspring]
            for ind, fit in zip(offspring, fits):
                ind.fitness.values = fit
            pop = tools.selBest(pop + offspring, len(pop))
            if gen % 50 == 0:
                print(f"Generation {gen}: Best C5 = {np.min([ind.fitness.values[0] for ind in pop])}")
            if np.min([ind.fitness.values[0] for ind in pop]) < best_c5:
                best_c5 = np.min([ind.fitness.values[0] for ind in pop])
                best_h = np.array([ind for ind in pop if ind.fitness.values[0] == best_c5][0])

        # Final fine-tuning with SLSQP
        def final_objective(h_flat):
            h = np.array(h_flat, dtype=np.float64)
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            corr = np.correlate(h, 1.0 - h, mode="full")
            c5 = np.max(corr * dx)
            return c5

        def final_constraint(h_flat):
            h = np.array(h_flat, dtype=np.float64)

        best_h = best_h * scale

        c5_candidate = objective(h_candidate)
