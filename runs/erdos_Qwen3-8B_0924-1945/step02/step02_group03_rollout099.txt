import numpy as np
from scipy.optimize import differential_evolution, Bounds
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    def de_objective(x):
        return objective(x)

    def de_constraint(x):
        return constraint_sum(x)

    def de_bounds(x):
        return bounds

    def de_start():
        return np.random.uniform(0, 1, n_points)

    def de_clamp_and_rescale(x):
        x = np.clip(x, 0, 1)
        x = x / np.sum(x) * n_points / 2
        return x

    def de_fitness(x):
        return de_objective(x)

    def de_evaluate(x):
        return de_fitness(x)

    def de_run():
        nonlocal best_h, best_c5
        result = differential_evolution(
            de_evaluate,
            bounds=bounds,
            strategy='best1bin',
            popsize=30,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-5,
            maxiter=300,
            constraints=de_constraint,
            seed=seed,
            disp=False
        )
        if result.success:
            candidate = de_clamp_and_rescale(result.x)
            candidate_c5 = evaluate_candidate(candidate)
            if candidate_c5 < best_c5:
                best_h = candidate.copy()
                best_c5 = candidate_c5
        else:
            pass

    def sqp_run(x0):
        nonlocal best_h, best_c5
        result = minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            tol=1e-5,
            options={'maxiter': 100}
        )

            candidate = de_clamp_and_rescale(result.x)
            candidate_c5 = evaluate_candidate(candidate)
            if candidate_c5 < best_c5:
                best_h = candidate.copy()
                best_c5 = candidate_c5
        else:
            pass

    for _ in range(5):
        if time() - start_time > budget_s:
            break
        de_run()

    if time() - start_time < budget_s:
        for _ in range(5):
            if time() - start_time > budget_s:
                break
            x0 = de_start()
            sqp_run(x0)

    best_h = de_clamp_and_rescale(best_h)
    best_c5 = evaluate_candidate(best_h)
    return (best_h, best_c5, n_points)
