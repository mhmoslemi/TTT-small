```python
import numpy as np
import scipy.optimize as opt
import time
from deap import base, creator, tools, algorithms

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def evaluate(individual):
        h = np.array(individual, dtype=np.float64)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return (c5,)

    def mutUniformInd(individual, indpb):
        for i in range(len(individual)):
            if np.random.random() < indpb:
                individual[i] = np.random.uniform(0.0, 1.0)
        return individual,

    def cxBlend(ind1, ind2):
        alpha = 0.5
        size = len(ind1)
        for i in range(size):
            ind1[i], ind2[i] = (ind1[i] + alpha * (ind2[i] - ind1[i])),
                               (ind2[i] + alpha * (ind1[i] - ind2[i]))
        return ind1, ind2

    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)

    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0.0, 1.0)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    toolbox.register("evaluate", evaluate)
    toolbox.register("mate", cxBlend)
    toolbox.register("mutate", mutUniformInd, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=5)

    start_time = time.time()

    pop = toolbox.population(n=50)
    for ind in pop:
        ind.fitness.values = toolbox.evaluate(ind)

    for gen in range(500):
        offspring = algorithms.varAnd(pop, toolbox, cxpb=0.9, mutpb=0.1)
        fits = [toolbox.evaluate(ind) for ind in offspring]
        for ind, fit in zip(offspring, fits):
            ind.fitness.values = fit

        pop = toolbox.select(offspring, k=len(pop))

        if time.time() - start_time > budget_s - 10:
            break

    best_ind = min(pop, key=lambda ind: ind.fitness.values[0])
    best_h = np.array(best_ind, dtype=np.float64)
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    best_h = best_h * scale

    result = opt.minimize(objective, best_h.flatten(), method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 300})
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)
    sum_h_opt = np.sum(h_opt)
    scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
    h_opt = h_opt * scale

    c5_bound = objective(h_opt.flatten())
    return h_opt.tolist(), c5_bound, n_points
```
