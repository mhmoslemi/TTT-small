import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    h = np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
    else:
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    bounds = [(0, 1) for _ in range(n_points)]

    def get_best(h_list):
        best = None
        best_c5 = float('inf')
        for h in h_list:
            c5 = evaluate_candidate(h)
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
        return best, best_c5

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)
    population = [h.copy() for _ in range(50)]
    for _ in range(500):
        if time() - start_time > budget_s:
            break
        selected = np.random.choice(len(population), size=5, replace=False)
        parents = [population[i] for i in selected]
        parent1, parent2 = np.random.choice(parents, size=2, replace=False)
        child = (parent1 + np.random.rand(n_points) * (parent2 - parent1)) * 0.5
        child = np.clip(child, 0, 1)
        child = child / np.sum(child) * n_points / 2
        population.append(child)
        population.pop(np.random.choice(len(population)))
        current_h, current_c5 = get_best(population)
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
