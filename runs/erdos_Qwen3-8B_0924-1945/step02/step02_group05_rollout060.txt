import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 80
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if np.random.rand() < 0.5 else 0
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()

    def objective(x):
        return evaluate_candidate(x)

    def constraint(x):
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint}]

    def run_de():
        best_h = h.copy()
        best_c5 = evaluate_candidate(best_h)
        result = differential_evolution(
            objective,
            bounds,
            constraints=cons,
            strategy='best1bin',
            mutation=(0.5, 1),
            recombination=0.9,
            popsize=15,
            tol=1e-5,
            maxiter=300,
            disp=False,
            polish=True,
            vectorized=True
        )
        if result.success:
            best_h = result.x
            best_c5 = evaluate_candidate(best_h)
        return best_h, best_c5

    best_h, best_c5 = run_de()

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
