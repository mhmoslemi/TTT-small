import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in kwargs:
        h = kwargs['initial_h_values'].copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if np.random.rand() < 0.5 else 0
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    dx = 2.0 / n_points

    def compute_c5(h):
        corr_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def gradient_update(h, k, step_size=0.01):
        active_start = max(0, k - 10)
        active_end = min(n_points - 1, k + 10)
        grad = np.zeros_like(h)
        for x in range(active_start, active_end + 1):
            if x + k < n_points:
                grad[x] += (1 - h[x + k])
            if x >= k:
                grad[x - k] -= h[x]
        h -= grad * step_size
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return h

    start_time = time()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    max_iter = 1000
    early_stop = 0

    for _ in range(max_iter):
        if time() - start_time > budget_s:
            break
        corr_vals = np.correlate(best_h, 1 - best_h, mode="full") * dx
        max_k = np.argmax(corr_vals)
        if corr_vals[max_k] < 0.38085:
            break
        best_h = gradient_update(best_h, max_k)
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            early_stop = 0
        else:
            early_stop += 1
        if early_stop >= 5:
            break

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = compute_c5(best_h)
    return (best_h, current_c5, n_points)
