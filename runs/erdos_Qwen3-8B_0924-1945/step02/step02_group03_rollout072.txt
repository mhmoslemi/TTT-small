import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import uniform
from time import time
from copy import deepcopy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = kwargs.get('initial_h_values', np.zeros(n_points))
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    def ga_run(pop_size=50, max_iter=500, mutation_rate=0.2, crossover_rate=0.8):
        population = np.random.uniform(0, 1, (pop_size, n_points))
        population = np.array([p / np.sum(p) * n_points / 2 for p in population])
        population = np.clip(population, 0, 1)

        for _ in range(max_iter):
            if time() - start_time > budget_s:
                break
            fitness_vals = np.array([evaluate_candidate(p) for p in population])
            elite_idx = np.argsort(fitness_vals)[0]
            elite = population[elite_idx]

            if fitness_vals[elite_idx] < best_c5:
                best_h = elite.copy()
                best_c5 = fitness_vals[elite_idx]

            parents = np.random.choice(population.shape[0], size=pop_size, replace=False)
            offspring = []
            for i in range(pop_size):
                p1, p2 = population[parents[i % pop_size]], population[parents[(i + 1) % pop_size]]
                if np.random.rand() < crossover_rate:
                    alpha = np.random.uniform(-0.5, 0.5)
                    child = p1 + alpha * (p1 - p2)
                else:
                    child = p1.copy()
                if np.random.rand() < mutation_rate:
                    child += np.random.uniform(-0.1, 0.1, size=n_points)
                child = np.clip(child, 0, 1)
                child = child / np.sum(child) * n_points / 2
                offspring.append(child)
            population = np.array(offspring)
            if np.any(np.isnan(population)):
                population = np.random.uniform(0, 1, (pop_size, n_points))
                population = np.array([p / np.sum(p) * n_points / 2 for p in population])
                population = np.clip(population, 0, 1)

    ga_run(pop_size=50, max_iter=500, mutation_rate=0.2, crossover_rate=0.8)

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
