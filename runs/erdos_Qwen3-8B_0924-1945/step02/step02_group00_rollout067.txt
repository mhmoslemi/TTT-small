```python
import numpy as np
import scipy.optimize as opt
import time
import random

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def random_pattern():
        h = np.random.rand(n_points)
        h -= h.min()
        h /= h.max()
        h = np.round(h).astype(float)
        h = h * (target_sum / np.sum(h))
        return h

    def spaced_pattern():
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 5 == 0:
                h[i] = 1.0
        h = h * (target_sum / np.sum(h))
        return h

    def smooth_pattern():
        h = np.sin(np.linspace(0, np.pi, n_points))
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        return h

    def refine_pattern(h_initial):
        h = h_initial.copy()
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        result = opt.minimize(objective, h.flatten(), method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        sum_h_opt = np.sum(h_opt)
        scale = target_sum / sum_h_opt if sum_h_opt != 0 else 1.0
        h_opt = h_opt * scale
        return h_opt

    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(best_h.flatten())

    for _ in range(5):
        h = np.random.rand(n_points)
        h -= h.min()
        h /= h.max()
        h = np.round(h).astype(float)
        h = h * (target_sum / np.sum(h))
        h = refine_pattern(h)
        c5 = objective(h.flatten())
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    for _ in range(3):
        h = spaced_pattern()
        h = refine_pattern(h)
        c5 = objective(h.flatten())
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    for _ in range(3):
        h = smooth_pattern()
        h = refine_pattern(h)
        c5 = objective(h.flatten())
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        for _ in range(3):
            h = random_pattern()
            h = refine_pattern(h)
            c5 = objective(h.flatten())
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5

    return best_h.tolist(), best_c5, n_points
```
