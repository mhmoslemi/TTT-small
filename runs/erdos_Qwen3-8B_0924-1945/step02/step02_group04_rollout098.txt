import numpy as np
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = kwargs.get('initial_h_values', None)
    
    # Initialize h
    if initial_h_values is not None:
        h = np.clip(initial_h_values, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def evaluate_candidate(h):
        h_clipped = np.clip(h, 0, 1)
        h_normalized = h_clipped * (n_points / 2) / np.sum(h_clipped)
        return compute_c5(h_normalized)
    
    # PSO Parameters
    swarm_size = 30
    inertia = 0.8
    cognitive = 1.5
    social = 1.5
    max_iter = 1000
    cooling_rate = 0.995
    max_no_improve = 50
    perturbation_rate = 0.4
    
    # Initialize swarm
    particles = np.zeros((swarm_size, n_points))
    velocities = np.zeros((swarm_size, n_points))
    pbest = np.zeros((swarm_size, n_points))
    gbest = np.zeros((swarm_size, n_points))
    best_c5 = np.inf
    
    # Initialize particles
    for i in range(swarm_size):
        if initial_h_values is not None:
            particles[i] = np.clip(initial_h_values, 0, 1)
        else:
            particles[i] = np.random.uniform(0, 1, n_points)
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        pbest[i] = particles[i].copy()
    
    gbest = particles[0].copy()
    best_c5 = evaluate_candidate(gbest)
    
    start_time = time.time()
    no_improve_count = 0
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (pbest[i] - particles[i]) + social * r2 * (gbest - particles[i])
            
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            current_c5 = evaluate_candidate(particles[i])
            if current_c5 < evaluate_candidate(pbest[i]):
                pbest[i] = particles[i].copy()
        
        # Update global best
        for i in range(swarm_size):
            if evaluate_candidate(particles[i]) < best_c5:
                best_c5 = evaluate_candidate(particles[i])
                gbest = particles[i].copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        
        # Early termination if no improvement
        if no_improve_count > max_no_improve:
            break
        
        # Adaptive inertia weight
        inertia = max(inertia * cooling_rate, 0.5)
    
    # Fallback to initial_h_values if no improvement
    if best_c5 >= evaluate_candidate(initial_h_values) if initial_h_values is not None else True:
        h = np.clip(initial_h_values, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        best_c5 = evaluate_candidate(h)
    
    # Final clipping and normalization
    best_h = np.clip(gbest, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h, best_c5, n_points)
