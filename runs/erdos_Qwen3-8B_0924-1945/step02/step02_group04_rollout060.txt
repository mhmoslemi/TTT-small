import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    
    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    def run_pso():
        swarm_size = 30
        particles = np.random.uniform(0, 1, (swarm_size, n_points))
        particles = particles * (n_points / 2) / np.sum(particles, axis=1, keepdims=True)
        velocities = np.zeros((swarm_size, n_points))
        pbest = particles.copy()
        gbest = particles.copy()
        best_c5 = np.inf
        
        for _ in range(1000):
            if time.time() > time.time() + budget_s:
                break
            r1, r2 = np.random.rand(2)
            velocities = 0.8 * velocities + 1.5 * r1 * (pbest - particles) + 1.5 * r2 * (gbest - particles)
            particles += velocities
            particles = np.clip(particles, 0, 1)
            particles = particles * (n_points / 2) / np.sum(particles, axis=1, keepdims=True)
            
            current_c5 = np.array([evaluate_candidate(p) for p in particles])
            mask = current_c5 < evaluate_candidate(pbest)
            pbest[mask] = particles[mask].copy()
            best_c5 = np.min([best_c5, np.min(current_c5)])
            gbest = particles[np.argmin(current_c5)].copy()
        
        best_h = gbest
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        return best_h, best_c5
    
    if 'initial_h_values' in kwargs:
        h = initial_h.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        best_h = h
        best_c5 = evaluate_candidate(h)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        best_h = h
        best_c5 = evaluate_candidate(h)
    
    best_h, best_c5 = run_pso()
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
