import numpy as np
import scipy.optimize as opt
import time
import random
from deap import base, creator, tools, algorithms

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    def create_initial_pattern():
        if initial_h_values is not None:
            h = np.array(initial_h_values, dtype=np.float64)
            h = np.clip(h, 0.0, 1.0)
            h = h * (n_points / 2.0) / np.sum(h)
            h = np.round(h).astype(int)
            h = h * (n_points / 2.0) / np.sum(h)
            return h
        else:
            h = np.zeros(n_points, dtype=np.float64)
            for i in range(n_points):
                if i % 5 == 0:
                    h[i] = 1.0
            h = h * (n_points / 2.0) / np.sum(h)
            return h

    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = create_initial_pattern()
    h = np.clip(h, 0.0, 1.0)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        h = h * (target_sum / sum_h) if sum_h != 0 else h
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def evaluate_individual(ind):
        h = np.array(ind, dtype=np.float64)
        sum_h = np.sum(h)
        h = h * (target_sum / sum_h) if sum_h != 0 else h
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr * dx)

    def generate_initial_population(pop_size):
        pop = []
        for _ in range(pop_size):
            ind = np.random.rand(n_points)
            ind = np.clip(ind, 0.0, 1.0)
            sum_ind = np.sum(ind)
            ind = ind * (target_sum / sum_ind) if sum_ind != 0 else ind
            pop.append(ind)
        return pop

    start_time = time.time()
    best_c5 = float('inf')
    best_h = h

    def evolve_population(pop, pop_size):
        nonlocal best_c5
        creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
        creator.create("Individual", list, fitness=creator.FitnessMin)

        toolbox = base.Toolbox()
        toolbox.register("attr_float", np.random.rand)
        toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
        toolbox.register("population", tools.initRepeat, list, toolbox.individual)

        toolbox.register("evaluate", evaluate_individual)
        toolbox.register("mate", tools.cxBlend, alpha=0.5)
        toolbox.register("mutate", tools.mutPolynomialBounded, eta=0.5, indpb=0.1)
        toolbox.register("select", tools.selTournament, tournsize=5)

        pop = toolbox.population(n=pop_size)
        for _ in range(100):
            offspring = algorithms.varOr(pop, toolbox, lambda ind: ind, lambda ind: ind, lambda ind: ind, n=pop_size)
            fits = [toolbox.evaluate(ind) for ind in offspring]
            for ind, fit in zip(offspring, fits):
                ind.fitness.values = (fit,)
            pop = toolbox.select(offspring, k=len(pop))
            if np.min([ind.fitness.values[0] for ind in pop]) < best_c5:
                best_c5 = np.min([ind.fitness.values[0] for ind in pop])
                best_h = np.array([ind for ind in pop if ind.fitness.values[0] == best_c5][0])
        return pop

    if time.time() - start_time < budget_s:
        pop = generate_initial_population(50)
        pop = evolve_population(pop, 150)

    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    best_h = best_h * (target_sum / sum_h) if sum_h != 0 else best_h
    best_c5 = evaluate_individual(best_h)

    return best_h.tolist(), best_c5, n_points
