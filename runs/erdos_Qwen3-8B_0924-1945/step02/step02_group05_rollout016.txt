import numpy as np
from scipy.optimize import minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    
    # Initialize h
    h = initial_h_values.copy() if 'initial_h_values' in kwargs else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def gradient_descent_step(h, max_corr, k, step_size):
        active_x = np.arange(max(0, k - 10), min(n_points, k + n_points))
        gradient = np.zeros_like(h)
        for x in active_x:
            if x + k < n_points:
                gradient[x] += (1 - h[x + k])
            if x < n_points:
                gradient[x] -= h[x]
        h[active_x] -= gradient[active_x] * step_size
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return h

    def run_optimization():
        best_h = h.copy()
        best_c5 = evaluate_candidate(best_h)
        start_time = time()
        for _ in range(1000):
            if time() - start_time > budget_s:
                break
            corr_vals = np.correlate(best_h, 1 - best_h, mode="full") * 2.0 / n_points
            max_corr = np.max(corr_vals)
            k = np.argmax(corr_vals)
            if max_corr < 0.38085:
                break
            best_h = gradient_descent_step(best_h, max_corr, k, 0.05)
            current_c5 = evaluate_candidate(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = best_h.copy()
        return best_h, best_c5

    best_h, best_c5 = run_optimization()

    if time() - time() < budget_s:
        for _ in range(100):
            if time() - time() > budget_s:
                break
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
