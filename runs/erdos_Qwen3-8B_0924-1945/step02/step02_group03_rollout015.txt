import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    
    def clip_and_rescale(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / total
        return h * scale

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = clip_and_rescale(h)
        return compute_c5(h)

    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_candidate(h)

    start_time = time()
    best_h = initial_h_values.copy()
    best_c5 = evaluate_candidate(best_h)
    
    if 'initial_h_values' in kwargs:
        h_initial = clip_and_rescale(initial_h_values)
        best_h = h_initial
        best_c5 = compute_c5(h_initial)
    
    if time() - start_time < budget_s:
        result = differential_evolution(objective, bounds, strategy='best1bin',
                                       popsize=15, mutation=0.5, recombination=0.8,
                                       tol=1e-5, maxiter=500, seed=seed, disp=False)
        best_h = clip_and_rescale(result.x)
        best_c5 = evaluate_candidate(best_h)
    
    best_h = clip_and_rescale(best_h)
    best_c5 = evaluate_candidate(best_h)
    return (best_h, best_c5, n_points)
