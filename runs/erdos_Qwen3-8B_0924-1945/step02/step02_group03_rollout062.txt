import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = kwargs.get('initial_h_values', None)
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    def de_objective(x):
        nonlocal best_c5
        if time() - start_time > budget_s:
            return float('inf')
        x = np.clip(x, 0, 1)
        x = x / np.sum(x) * n_points / 2
        current_c5 = compute_c5(x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = x.copy()
        return current_c5

    result = differential_evolution(de_objective, bounds, strategy='best1bin', 
                                   popsize=15, mutation=0.5, recombination=0.8, 
                                   tol=1e-5, maxiter=500, disp=False,
                                   workers=1)

    best_h = np.clip(result.x, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
