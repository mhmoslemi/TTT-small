```python
import numpy as np
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 250
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    pop_size = 50
    tournament_size = 5
    mutation_rate = 0.1
    crossover_prob = 0.8
    elitism_rate = 0.2
    max_generations = 200
    best_h = h.copy()
    best_c5 = evaluate(h)

    def initialize_population():
        population = []
        for _ in range(pop_size):
            h = np.zeros(n_points, dtype=np.float64)
            indices = np.random.choice(n_points, size=int(n_points // 2), replace=False)
            h[indices] = 1.0
            sum_h = np.sum(h)
            scale = target_sum / sum_h if sum_h != 0 else 1.0
            h = h * scale
            population.append(h)
        return population

    population = initialize_population()

    def tournament_selection(population):
        selected = []
        for _ in range(2):
            candidates = np.random.choice(population, size=tournament_size)
            winner = min(candidates, key=lambda x: evaluate(x))
            selected.append(winner)
        return selected

    def crossover(parent1, parent2):
        child = np.copy(parent1)
        if np.random.rand() < crossover_prob:
            mask = np.random.rand(n_points) < 0.5
            child[mask] = parent2[mask]
        return child

    def mutate(h):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.normal(0, 0.1)
                h[i] = np.clip(h[i], 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        return h

    def fitness(h):
        return evaluate(h)

    start_time = time.time()
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        next_population = []
        for _ in range(pop_size // 2):
            parent1, parent2 = tournament_selection(population)
            child1 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = crossover(parent1, parent2)
            child2 = mutate(child2)
            next_population.append(child1)
            next_population.append(child2)
        population = next_population
        population.sort(key=fitness)
        new_population = population[:int(pop_size * elitism_rate)]
        while len(new_population) < pop_size:
            new_population.append(mutate(np.random.rand(n_points) * 1.0))
        population = new_population
        current_best = min(population, key=fitness)
        if fitness(current_best) < best_c5:
            best_h = current_best
            best_c5 = fitness(current_best)

    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    scale = target_sum / sum_h if sum_h != 0 else 1.0
    best_h = best_h * scale
    c5_bound = evaluate(best_h)
    return best_h.tolist(), c5_bound, n_points
```
