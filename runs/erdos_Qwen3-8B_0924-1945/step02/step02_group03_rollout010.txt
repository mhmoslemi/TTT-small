import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.stats import uniform
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
    h = np.clip(h, 0, 1)

    def compute_c5(h):
        dx = 2.0 / n_points
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def fitness(h):
        return evaluate_candidate(h)

    def objective(x):
        return fitness(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    start_time = time()

    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    population = [best_h.copy()]
    for _ in range(49):
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
        population.append(h.copy())

    for gen in range(500):
        if time() - start_time > budget_s:
            break
        if len(population) < 2:
            break
        parent1 = population[np.random.choice(len(population))]
        parent2 = population[np.random.choice(len(population))]
        child = (parent1 + parent2) / 2 + np.random.uniform(-0.1, 0.1, n_points)
        child = np.clip(child, 0, 1)
        child = child / np.sum(child) * n_points / 2
        child_c5 = evaluate_candidate(child)
        population.append(child)
        population.pop(np.random.choice(len(population)))
        current_c5 = min([evaluate_candidate(p) for p in population])
        if current_c5 < best_c5:
            best_h = np.array([p for p in population if evaluate_candidate(p) == current_c5][0])
            best_c5 = current_c5

    def refine(h):
        res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            return h / np.sum(h) * n_points / 2
        return h

    best_h = refine(best_h)
    best_c5 = evaluate_candidate(best_h)

    if time() - start_time < budget_s:
        for _ in range(100):
            if time() - start_time > budget_s:
                break
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * n_points / 2
            current_c5 = evaluate_candidate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
