```python
import numpy as np
import scipy.optimize as opt
import time
import random

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            h[i] = 1.0 if i % 5 == 0 else 0.0
        h = h * (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        sum_h = np.sum(h)
        return sum_h - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()

    def evolution_search(start_pop, max_gen=500, pop_size=50, elite_size=5, mutation_rate=0.1, crossover_rate=0.8):
        population = np.random.rand(pop_size, n_points)
        population = np.clip(population, 0, 1) * target_sum / np.mean(np.sum(population, axis=1))
        best = np.inf
        best_pop = start_pop
        for gen in range(max_gen):
            fitness = np.array([objective(ind) for ind in population])
            elite_indices = np.argsort(fitness)[:elite_size]
            elite = population[elite_indices]
            if np.min(fitness) < best:
                best = np.min(fitness)
                best_pop = population[np.argmin(fitness)]
            if gen % 10 == 0:
                print(f"Generation {gen}, Best C5: {best:.6f}")
            if time.time() - start_time > budget_s - 10:
                break
            new_pop = []
            for i in range(pop_size):
                parent1 = random.choice(elite)
                parent2 = random.choice(elite)
                if np.random.rand() < crossover_rate:
                    child = (parent1 + parent2) / 2
                else:
                    child = parent1
                if np.random.rand() < mutation_rate:
                    child += np.random.normal(0, 0.05, n_points)
                child = np.clip(child, 0, 1) * target_sum / np.mean(np.sum(child, axis=1))
                new_pop.append(child)
            population = np.array(new_pop)
        return best_pop

    h_opt = evolution_search(h)
    h_opt = np.clip(h_opt, 0, 1) * target_sum / np.mean(np.sum(h_opt, axis=1))

    if time.time() - start_time < budget_s:
        result = opt.minimize(objective, h_opt.flatten(), method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 300})
        h_opt = result.x.reshape(n_points)
        h_opt = np.clip(h_opt, 0, 1) * target_sum / np.mean(np.sum(h_opt, axis=1))

    c5_bound = objective(h_opt.flatten())
    return h_opt.tolist(), c5_bound, n_points
```
