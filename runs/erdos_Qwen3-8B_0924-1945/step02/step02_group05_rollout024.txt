import numpy as np
from scipy.optimize import minimize
from scipy.stats import uniform
from numpy.random import choice, random, seed
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    if 'initial_h_values' in kwargs:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if np.random.rand() < 0.5 else 0
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2

    dx = 2.0 / n_points
    def compute_c5(h):
        correlate_vals = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(correlate_vals)

    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        return compute_c5(h)

    def objective(x):
        return evaluate_candidate(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_candidate(best_h)

    def generate_uniform_pattern(m, fraction):
        intervals = np.linspace(0, 2, m + 1)
        selected = np.random.choice(m, int(fraction * m), replace=False)
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            for j in range(m):
                if intervals[j] <= x < intervals[j + 1]:
                    if j in selected:
                        h[i] = 1
                    break
        h = h / np.sum(h) * n_points / 2
        return h

    def generate_bilinear_pattern(slope):
        h = 0.5 * (1 + np.tanh(slope * (np.linspace(0, 2, n_points) - 1)))
        h = h / np.sum(h) * n_points / 2
        return h

    def generate_random_pattern(p):
        h = np.random.binomial(1, p, n_points)
        h = h / np.sum(h) * n_points / 2
        return h

    patterns = [generate_uniform_pattern(5, 0.5), generate_bilinear_pattern(0.5), generate_random_pattern(0.7)]
    h_candidates = [np.clip(p, 0, 1) for p in patterns]

    for h_candidate in h_candidates:
        c5 = evaluate_candidate(h_candidate)
        if c5 < best_c5:
            best_h = h_candidate
            best_c5 = c5

    for _ in range(1000):
        if time() - start_time > budget_s:
            break
        idx = np.random.randint(0, len(h_candidates))
        h_candidate = h_candidates[idx]
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturbed = h_candidate + perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed / np.sum(h_perturbed) * n_points / 2
        c5 = evaluate_candidate(h_perturbed)
        if c5 < best_c5:
            best_h = h_perturbed
            best_c5 = c5
        h_candidates.append(h_perturbed)
        if len(h_candidates) > 20:
            h_candidates.pop(0)

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    current_c5 = evaluate_candidate(best_h)
    return (best_h, current_c5, n_points)
