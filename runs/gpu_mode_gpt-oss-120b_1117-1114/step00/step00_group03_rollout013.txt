@triton.jit
def batched_matmul_kernel(
        A_ptr, B_ptr, C_ptr,
        M, N, K,
        stride_Am, stride_Ak,
        stride_Bk, stride_Bn,
        stride_Cm, stride_Cn,
        stride_A_batch, stride_B_batch, stride_C_batch,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(0)
    pid_batch = tl.program_id(2)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N

    # load pointers base for this batch
    a_ptrs = A_ptr + pid_batch * stride_A_batch
    b_ptrs = B_ptr + pid_batch * stride_B_batch
    c_ptrs = C_ptr + pid_batch * stride_C_batch

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute the K offset
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask for K dimension
        mask_k = offs_k < K

        # load A and B tiles
        a = tl.load(a_ptrs + (offs_m[:, None] * stride_Am) + (offs_k[None, :] * stride_Ak),
                    mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs + (offs_k[:, None] * stride_Bk) + (offs_n[None, :] * stride_Bn),
                    mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # compute dot product accumulate
        acc += tl.dot(a, b)

    # write back
    c = acc.to(tl.float32)
    tl.store(c_ptrs + (offs_m[:, None] * stride_Cm) + (offs_n[None, :] * stride_Cn),
             c, mask=mask_m[:, None] & mask_n[None, :])

import torch
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          M, N, K,
                          stride_Am, stride_Ak,
                          stride_Bk, stride_Bn,
                          stride_Cm, stride_Cn,
                          stride_A_batch, stride_B_batch, stride_C_batch,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(0)
    pid_batch = tl.program_id(2)
    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    # base pointers for this batch
    a_ptr = A_ptr + pid_batch * stride_A_batch
    b_ptr = B_ptr + pid_batch * stride_B_batch
    c_ptr = C_ptr + pid_batch * stride_C_batch
    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K
        # load A tile
        a = tl.load(a_ptr + (offs_m[:, None] * stride_Am) + (offs_k[None, :] * stride_Ak),
                    mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        # load B tile
        b = tl.load(b_ptr + (offs_k[:, None] * stride_Bk) + (offs_n[None, :] * stride_Bn),
                    mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        # dot product
        acc += tl.dot(a, b)
    # store result
    c = acc
    tl.store(c_ptr + (offs_m[:, None] * stride_Cm) + (offs_n[None, :] * stride_Cn),
             c, mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    """
    Triton-accelerated implementation of the outgoing Triangular Multiplicative Update.
    The expensive contraction out_{i,j,d} = sum_k left_{i,k,d} * right_{j,k,d}
    is computed by a fused batched GEMM kernel in Triton.
    """
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    dtype = torch.float32
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, _ = input_tensor.shape

    # LayerNorm weight/bias
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Linear weights
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Ensure tensors are float32 on same device
    norm_weight = norm_weight.to(dtype=device)
    norm_bias = norm_bias.to(dtype=device)
    left_proj_weight = left_proj_weight.to(dtype=device)
    right_proj_weight = right_proj_weight.to(dtype=device)
    left_gate_weight = left_gate_weight.to(dtype=device)
    right_gate_weight = right_gate_weight.to(dtype=device)
    out_gate_weight = out_gate_weight.to(dtype=device)
    to_out_norm_weight = to_out_norm_weight.to(dtype=device)
    to_out_norm_bias = to_out_norm_bias.to(dtype=device)
    to_out_weight = to_out_weight.to(dtype=device)

    # Input to float32
    x = input_tensor.to(dtype)

    # LayerNorm over last dim
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [bs,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Mask handling
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)   # [bs,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gating
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Prepare tensors for Triton GEMM
    # Permute to [bs, hidden, N, N] and reshape batch*hidden.
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # transposed on last two dims

    # Combine leading dimensions
    batch_hidden = bs * hidden_dim
    A = left_perm.view(batch_hidden, N, N)
    B = right_perm.view(batch_hidden, N, N)
    C = torch.empty_like(A)

    # Strides (in elements)
    stride_A_batch = A.stride(0)
    stride_Am = A.stride(1)
    stride_Ak = A.stride(2)

    stride_B_batch = B.stride(0)
    stride_Bk = B.stride(1)
    stride_Bn = B.stride(2)

    stride_C_batch = C.stride(0)
    stride_Cm = C.stride(1)
    stride_Cn = C.stride(2)

    # Block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_N), triton.cdiv(N, BLOCK_M), batch_hidden)
    batched_matmul_kernel[grid](
        A.data_ptr(),
        B.data_ptr(),
        C.data_ptr(),
        N, N, N,
        stride_Am, stride_Ak,
        stride_Bk, stride_Bn,
        stride_Cm, stride_Cn,
        stride_A_batch, stride_B_batch, stride_C_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back
    out = C.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1)   # [bs,N,N,hidden]

    # Apply output layernorm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear
    out = torch.nn.functional.linear(out, to_out_weight)   # shape [bs,N,N,dim]

    return out

"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The heavy part of the forward pass is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication for each hidden dimension d.
A fused Triton kernel performs this multiplication for all (batch * hidden)
matrices in one launch. All other operations (layer‑norms, linear
projections, gating) are done with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: batched GEMM A (M×K) * B (K×N) -> C (M×N)
# A, B, C are stored as (batch, M, K), (batch, K, N), (batch, M, N)
# -------------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Am, stride_Ak,
    stride_Bk, stride_Bn,
    stride_Cm, stride_Cn,
    stride_A_batch, stride_B_batch, stride_C_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(0)          # block column
    pid_b = tl.program_id(2)          # which matrix in the batch

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch element
    a_ptr = A_ptr + pid_b * stride_A_batch
    b_ptr = B_ptr + pid_b * stride_B_batch
    c_ptr = C_ptr + pid_b * stride_C_batch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a = tl.load(
            a_ptr + (offs_m[:, None] * stride_Am) + (offs_k[None, :] * stride_Ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptr + (offs_k[:, None] * stride_Bk) + (offs_n[None, :] * stride_Bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)          # fp32 accumulation

    # write back
    tl.store(
        c_ptr + (offs_m[:, None] * stride_Cm) + (offs_n[None, :] * stride_Cn),
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# -------------------------------------------------------------------------
# Entry point required by the evaluation harness
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module with a Triton kernel for the
    O(N³) contraction. Returns a tensor of shape [bs, seq_len, seq_len, dim].
    """
    input_tensor, mask, weights, config = data

    # -----------------------------------------------------------------
    # Extract configuration
    # -----------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, seq_len, _, _ = input_tensor.shape
    device = input_tensor.device
    dtype = torch.float32

    # -----------------------------------------------------------------
    # Load weights onto the device (they may be on CPU)
    # -----------------------------------------------------------------
    def to_dev(t):
        return t.to(device=device, dtype=dtype)

    norm_weight = to_dev(weights["norm.weight"])
    norm_bias = to_dev(weights["norm.bias"])
    left_proj_weight = to_dev(weights["left_proj.weight"])
    right_proj_weight = to_dev(weights["right_proj.weight"])
    left_gate_weight = to_dev(weights["left_gate.weight"])
    right_gate_weight = to_dev(weights["right_gate.weight"])
    out_gate_weight = to_dev(weights["out_gate.weight"])
    to_out_norm_weight = to_dev(weights["to_out_norm.weight"])
    to_out_norm_bias = to_dev(weights["to_out_norm.bias"])
    to_out_weight = to_dev(weights["to_out.weight"])

    # -----------------------------------------------------------------
    # Core PyTorch part of the forward pass
    # -----------------------------------------------------------------
    x = input_tensor.to(dtype)

    # Input LayerNorm over the feature dimension
    x = torch.nn.functional.layer_norm(
        x, (dim,), weight=norm_weight, bias=norm_bias
    )

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, left_proj_weight)   # [bs, N, N, hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply optional mask (mask may be all‑ones when nomask=True)
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)   # [bs, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # Gating values
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Prepare tensors for the Triton kernel
    #   left  : [bs, N, N, hidden] -> [bs, hidden, N, N]
    #   right : [bs, N, N, hidden] -> transpose k/j -> [bs, hidden, N, N]
    # -----------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (bs, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # (bs, hidden, N, N) with k as first dim

    # Merge batch and hidden dimensions -> (B*H, N, N)
    bh = bs * hidden_dim
    A = left_perm.view(bh, seq_len, seq_len)
    B = right_perm.view(bh, seq_len, seq_len)
    C = torch.empty_like(A)

    # Strides in elements (not bytes)
    stride_A_batch = A.stride(0)
    stride_Am = A.stride(1)
    stride_Ak = A.stride(2)

    stride_B_batch = B.stride(0)
    stride_Bk = B.stride(1)
    stride_Bn = B.stride(2)

    stride_C_batch = C.stride(0)
    stride_Cm = C.stride(1)
    stride_Cn = C.stride(2)

    # -----------------------------------------------------------------
    # Launch Triton GEMM kernel
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(seq_len, BLOCK_N),   # columns
        triton.cdiv(seq_len, BLOCK_M),   # rows
        bh,                              # batch * hidden
    )

    batched_matmul_kernel[grid](
        A.data_ptr(),
        B.data_ptr(),
        C.data_ptr(),
        seq_len, seq_len, seq_len,
        stride_Am, stride_Ak,
        stride_Bk, stride_Bn,
        stride_Cm, stride_Cn,
        stride_A_batch, stride_B_batch, stride_C_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [bs, N, N, hidden]
    out = C.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1)

    # -----------------------------------------------------------------
    # Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # shape [bs, N, N, dim]

    return out
