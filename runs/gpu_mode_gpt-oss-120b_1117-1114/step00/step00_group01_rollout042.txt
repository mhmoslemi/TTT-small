@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_k, stride_r_j, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid_b = tl.program_id(0)  # batch
    pid_m = tl.program_id(1)  # i tile
    pid_n = tl.program_id(2)  # j tile

    # base indices
    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N

    # determine bounds masks for i and j (used later)
    i_idx = i_base + tl.arange(0, BLOCK_M)
    j_idx = j_base + tl.arange(0, BLOCK_N)
    mask_i = i_idx < N
    mask_j = j_idx < N

    # loop over hidden dimension
    for h_start in range(0, H, BLOCK_H):
        # compute actual h block size (tail)
        cur_h = tl.minimum(BLOCK_H, H - h_start)
        # create h indices
        h_idx = h_start + tl.arange(0, BLOCK_H)
        mask_h = h_idx < H

        # accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

        # loop over k dimension tiles
        for k_start in range(0, N, BLOCK_K):
            k_idx = k_start + tl.arange(0, BLOCK_K)
            mask_k = k_idx < N

            # pointers for left [b,i,k,h]
            left_ptrs = left_ptr + \
                pid_b * stride_l_b + \
                i_idx[:, None, None] * stride_l_i + \
                k_idx[None, :, None] * stride_l_k + \
                h_idx[None, None, :] * stride_l_h

            # pointers for right [b,k,j,h] (right already permuted)
            right_ptrs = right_ptr + \
                pid_b * stride_r_b + \
                k_idx[:, None, None] * stride_r_k + \
                j_idx[None, :, None] * stride_r_j + \
                h_idx[None, None, :] * stride_r_h

            # load with masks
            left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :] & mask_h[None, None, :], other=0.0)
            right = tl.load(right_ptrs, mask=mask_k[:, None, None] & mask_j[None, :] & mask_h[None, None, :], other=0.0)

            # compute per hidden slice dot
            # left: (M, K, H), right: (K, N, H)
            # loop over hidden within BLOCK_H
            for hb in range(0, BLOCK_H):
                # early exit for tail
                if hb >= cur_h:
                    break
                left_slice = left[:, :, hb]   # (M, K)
                right_slice = right[:, :, hb] # (K, N)
                # dot product
                acc[:, :, hb] += tl.dot(left_slice, right_slice)

        # after k loops, store acc to out
        # compute output pointer for block
        out_ptrs = out_ptr + \
            pid_b * stride_o_b + \
            i_idx[:, None, None] * stride_o_i + \
            j_idx[None, :, None] * stride_o_j + \
            h_idx[None, None, :] * stride_o_h

        # store with mask: valid i,j,h
        mask_store = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
        tl.store(out_ptrs, acc, mask=mask_store)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on input (dim)
    x = input_tensor
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # compute mean and var over dim
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    inv_std = torch.rsqrt(var + eps)
    x_norm = (x - mean) * inv_std
    x_norm = x_norm * norm_weight + norm_bias

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # If mask not None: else ones
    if mask is None:
        mask = torch.ones(bs, N, N, device=device, dtype=x.dtype)
    mask_unsq = mask.unsqueeze(-1).type_as(x_norm)  # [B,N,N,1]

    # apply mask to left and right
    left = left * mask_unsq
    right = right * mask_unsq

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Need right_t permuted
    right_t = right.permute(0, 2, 1, 3).contiguous()

    # Allocate output tensor
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Triton launch config

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_k, stride_r_j, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    B,
    N: tl.constexpr,
    H: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N

    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # loop over hidden dim blocks
    for h_start in range(0, H, BLOCK_H):
        h_end = tl.minimum(h_start + BLOCK_H, H)
        cur_h = h_end - h_start
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

        # loop over k dimension blocks
        num_k_blocks = tl.cdiv(N, BLOCK_K)
        for kk in range(num_k_blocks):
            k_base = kk * BLOCK_K
            offs_k = k_base + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # compute pointers
            left_ptrs = left_ptr + \
                pid_b * stride_l_b + \
                offs_i[:, None, None] * stride_l_i + \
                offs_k[None, :, None] * stride_l_k + \
                offs_h[None, None, :] * stride_l_h

            right_ptrs = right_ptr + \
                pid_b * stride_r_b + \
                offs_k[:, None, None] * stride_r_k + \
                offs_j[None, :, None] * stride_r_j + \
                offs_h[None, None, :] * stride_r_h

            left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :] & mask_h[None, None, :], other=0.0, dtype=tl.float16)
            right = tl.load(right_ptrs, mask=mask_k[:, None, None] & mask_j[None, :] & mask_h[None, None, :], other=0.0, dtype=tl.float16)

            # Compute per hidden slice dot product
            for hb in range(0, BLOCK_H):
                # skip tail hidden if out of bounds
                if hb >= cur_h:
                    break

"""
TriMul (outgoing) implementation with a fused Triton kernel.

The kernel computes the core N³ reduction:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

where left/right are linearly projected, masked and gated versions of the
input. The surrounding LayerNorm, gating, and final linear projection are
performed in PyTorch.  The Triton kernel processes tiles of the (i, j)
output plane for a block of hidden channels, accumulating the sum over k
with high arithmetic intensity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,                     # pointers
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,  # left strides
    stride_r_b, stride_r_k, stride_r_j, stride_r_h,  # right strides (permute k,j)
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,  # out strides
    B,                                               # batch size (runtime)
    N: tl.constexpr,                                 # sequence length (compile‑time)
    H: tl.constexpr,                                 # hidden dim (compile‑time)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)          # batch dimension
    pid_m = tl.program_id(1)          # i tile
    pid_n = tl.program_id(2)          # j tile

    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N

    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # iterate over hidden‑dim blocks
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # accumulator for this (i,j,hidden) tile
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

        # number of k‑tiles (known at compile time)
        num_k_tiles = tl.cdiv(N, BLOCK_K)
        for kk in range(num_k_tiles):
            k_base = kk * BLOCK_K
            offs_k = k_base + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # pointers into left [b,i,k,h] and right [b,k,j,h]
            left_ptrs = (
                left_ptr
                + pid_b * stride_l_b
                + offs_i[:, None, None] * stride_l_i
                + offs_k[None, :, None] * stride_l_k
                + offs_h[None, None, :] * stride_l_h
            )
            right_ptrs = (
                right_ptr
                + pid_b * stride_r_b
                + offs_k[:, None, None] * stride_r_k
                + offs_j[None, :, None] * stride_r_j
                + offs_h[None, None, :] * stride_r_h
            )

            # load tiles (masked for out‑of‑bounds)
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :] & mask_h[None, None, :],
                other=0.0,
                dtype=tl.float16,
            )
            right = tl.load(
                right_ptrs,
                mask=mask_k[:, None, None] & mask_j[None, :] & mask_h[None, None, :],
                other=0.0,
                dtype=tl.float16,
            )

            # per‑hidden‑channel dot product and accumulation
            for hb in range(0, BLOCK_H):
                # all loops are compile‑time; H is a multiple of BLOCK_H in our configs
                l_slice = left[:, :, hb]    # (M, K)
                r_slice = right[:, :, hb]   # (K, N)
                acc[:, :, hb] += tl.dot(l_slice, r_slice)  # (M, N)

        # store the accumulated tile
        out_ptrs = (
            out_ptr
            + pid_b * stride_o_b
            + offs_i[:, None, None] * stride_o_i
            + offs_j[None, :, None] * stride_o_j
            + offs_h[None, None, :] * stride_o_h
        )
        mask_store = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
        tl.store(out_ptrs, acc, mask=mask_store)


def _layer_norm(x, weight, bias, eps=1e-5):
    """LayerNorm over the last dimension (float32)."""
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    inv_std = torch.rsqrt(var + eps)
    return (x - mean) * inv_std * weight + bias


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: Tuple (input, mask, weights, config)
            - input: Tensor [B, N, N, dim]
            - mask: Tensor [B, N, N] (or None)
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm on input ----------
    x_norm = _layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
    )

    # ---------- Linear projections ----------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ---------- Mask ----------
    if mask is None:
        mask = torch.ones(B, N, N, device=device, dtype=x_norm.dtype)
    mask = mask.unsqueeze(-1).type_as(x_norm)  # [B,N,N,1]

    left = left * mask
    right = right * mask

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to half for the Triton kernel ----------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Right tensor needs to be indexed as (b, k, j, h)
    right_t = right.permute(0, 2, 1, 3).contiguous()

    # ---------- Allocate output for the core reduction ----------
    out = torch.empty(B, N, N, hidden_dim, dtype=torch.float32, device=device)

    # ---------- Triton launch parameters ----------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_H = 8               # 128 hidden_dim is divisible by 8

    # Strides (row‑major contiguous tensors)
    stride_l = left.stride()
    stride_r = right_t.stride()
    stride_o = out.stride()

    stride_l_b, stride_l_i, stride_l_k, stride_l_h = stride_l
    stride_r_b, stride_r_k, stride_r_j, stride_r_h = stride_r
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = stride_o

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Launch the kernel (N and hidden_dim are compile‑time arguments)
    tri_mul_kernel[grid](
        left,
        right_t,
        out,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_k,
        stride_r_j,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        B,
        N=N,
        H=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ---------- Post‑processing ----------
    # LayerNorm over the hidden dimension
    out_norm = _layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
    )
    # Apply output gate
    out_norm = out_norm * out_gate

    # Final linear projection back to dim
    output = F.linear(out_norm, weights["to_out.weight"])

    return output
