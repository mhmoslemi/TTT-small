def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    eps = 1e-6? Using default epsilon 1e-5 for layer norm.

    # LayerNorm on last dim
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # shape [B,N,N,dim]

    # Linear projections: left_proj, right_proj (weight shape hidden_dim x dim)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])  # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if not None? Actually mask given always. If config["nomask"] is True then mask may be all ones? Let's just apply regardless:
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)  # shape [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # Gate linear layers
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Compute triangular multiplication using triton kernel:
    # L shape [B, N, N, hidden_dim] -> permute to [B, hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # For matmul, we need right transposed on last two dims: K dimension is N.
    right_T = right_perm.transpose(-2, -1).contiguous()

    B, H, N, _ = left_perm.shape
    # flatten batch and hidden_dim:
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_T.view(B * H, N, N)
    # allocate output
    out_flat = torch.empty_like(left_flat)

    # Get strides (in elements)
    a_ptr = left_flat
    b_ptr = right_flat
    c_ptr = out_flat

    # Determine strides for A (batch, M, K)
    a_stride_b, a_stride_m, a_stride_k = a_ptr.stride()
    b_stride_b, b_stride_k, b_stride_n = b_ptr.stride()
    c_stride_b, c_stride_m, c_stride_n = c_ptr.stride()

    # Define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_N),
        triton.cdiv(N, BLOCK_M),
        B * H,
    )
    batched_matmul_kernel[grid](
        a_ptr, b_ptr, c_ptr,
        N, N, N,
        a_stride_b, a_stride_m, a_stride_k,
        b_stride_b, b_stride_k, b_stride_n,
        c_stride_b, c_stride_m, c_stride_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # Reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape [B,N,N,hidden_dim]

    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # Apply out_gate
    out = out * out_gate

    # final linear to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

@triton.jit
def batched_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Determine batch id, row block, col block
    pid_b = tl.program_id(2)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(0)

    # Compute offsets
    # row offset
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    # column offset
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # K offset (loop variable)
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute offset for k dimension
        k_offset = k * BLOCK_K
        # Gather A tile
        a_ptrs = A + pid_b * stride_Ab + offs_m[:, None] * stride_Am + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_Ak
        # Gather B tile
        b_ptrs = B + pid_b * stride_Bb + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

        # apply masks for valid indices (when M/N/K not divisible)
        a_mask = (offs_m[:, None] < M) & ((k_offset + tl.arange(0, BLOCK_K))[None, :] < K)
        b_mask = ((k_offset + tl.arange(0, BLOCK_K))[:, None] < K) & (offs_n[None, :] < N)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # accumulate
        acc += tl.dot(a, b)

    # store results
    c_ptrs = C + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)

"""
Optimized TriMul (outgoing) implementation.

The kernel computes the core O(N³) update:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We treat each hidden dimension independently and batch the
matrix multiplications across (batch * hidden_dim) using a
Triton tiled GEMM kernel. All other operations (LayerNorm,
Linear projections, gating) remain in PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ------------------------------------------------------------------
# Triton kernel: batched matmul (A @ Bᵀ) where A,B have shape
# (batch, M, K) and (batch, K, N) respectively.
# ------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A, B, C,               # pointers
    M, N, K,               # matrix dimensions
    stride_Ab, stride_Am, stride_Ak,   # A strides: batch, row, col
    stride_Bb, stride_Bk, stride_Bn,   # B strides: batch, row(K), col(N)
    stride_Cb, stride_Cm, stride_Cn,   # C strides: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(2)          # batch * hidden dim
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(0)          # block col

    # offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # load A tile (M x K)
        a_ptrs = A + pid_b * stride_Ab \
                 + offs_m[:, None] * stride_Am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_Ak

        # load B tile (K x N)
        b_ptrs = B + pid_b * stride_Bb \
                 + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_Bk \
                 + offs_n[None, :] * stride_Bn

        # mask out-of-bounds elements
        a_mask = (offs_m[:, None] < M) & ((k_offset + tl.arange(0, BLOCK_K))[None, :] < K)
        b_mask = ((k_offset + tl.arange(0, BLOCK_K))[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # write back
    c_ptrs = C + pid_b * stride_Cb \
             + offs_m[:, None] * stride_Cm \
             + offs_n[None, :] * stride_Cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


# ------------------------------------------------------------------
# Entry point
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Performs the forward pass of the outgoing TriMul module using a
    Triton kernel for the O(N³) multiplicative update.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if present)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.to(dtype).unsqueeze(-1)       # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and apply gates
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul via Triton (batched GEMM)
    #    out = left @ rightᵀ over the sequence dimension.
    # ------------------------------------------------------------------
    # Permute to bring hidden_dim as a batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()      # [B, hidden_dim, N, N]

    # Transpose the last two dimensions of right for multiplication
    right_T = right_perm.transpose(-2, -1).contiguous()      # [B, hidden_dim, N, N]

    B, H, N, _ = left_perm.shape
    batch_hidden = B * H

    # Collapse batch & hidden into a single leading dimension
    left_flat = left_perm.view(batch_hidden, N, N)
    right_flat = right_T.view(batch_hidden, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)

    # Strides (in elements)
    stride_Ab, stride_Am, stride_Ak = left_flat.stride()
    stride_Bb, stride_Bk, stride_Bn = right_flat.stride()
    stride_Cb, stride_Cm, stride_Cn = out_flat.stride()

    # Tiling parameters (tuned for typical N ≤ 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_N),   # blocks along N (cols)
        triton.cdiv(N, BLOCK_M),   # blocks along M (rows)
        batch_hidden,              # batch * hidden
    )

    batched_matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
