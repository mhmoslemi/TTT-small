"""
TriMul forward kernel (outgoing version) with Triton acceleration.

The heavy O(N³) reduction
    out[b, i, j, d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed by a custom Triton kernel that executes a batched
matrix multiplication L · Rᵀ for each hidden channel d.  All
lighter operations (LayerNorm, linear projections, gating) are
handled in PyTorch.  The final output is returned in float32.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched matmul of shape (B, hidden_dim, N, N)
# ----------------------------------------------------------------------
@triton.jit
def triton_trimul_matmul(
    left_ptr,                # *fp16   [B, N, N, hidden_dim]
    right_ptr,               # *fp16   [B, N, N, hidden_dim]
    out_ptr,                 # *fp16   [B, N, N, hidden_dim]
    N,                       # int32   sequence length
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_j, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr,   # tile size for i dimension
    BLOCK_N: tl.constexpr,   # tile size for j dimension
    BLOCK_K: tl.constexpr,   # tile size for k reduction
    HIDDEN_DIM: tl.constexpr # number of hidden channels
):
    pid_m = tl.program_id(0)          # i‑tile
    pid_n = tl.program_id(1)          # j‑tile
    pid_z = tl.program_id(2)          # combined batch * hidden_dim

    # --------------------------------------------------------------
    # Decode batch index and hidden‑channel index
    # --------------------------------------------------------------
    b = pid_z // HIDDEN_DIM
    d = pid_z - b * HIDDEN_DIM

    # --------------------------------------------------------------
    # Indices of the current tile
    # --------------------------------------------------------------
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # Base pointer offsets for batch and channel
    offs_b = b * stride_l_b
    offs_d_l = d * stride_l_d
    offs_d_r = d * stride_r_d
    offs_d_o = d * stride_o_d

    # --------------------------------------------------------------
    # Accumulator
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for kk in range(num_k):
        k_base = kk * BLOCK_K
        k = k_base + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ------------ left tile (i, k) ---------------------------
        # left[b, i, k, d]  -> shape (BLOCK_M, BLOCK_K)
        offs_i = i[:, None] * stride_l_i                # (M,1)
        offs_k_l = k[None, :] * stride_l_k               # (1,K)
        left_ptrs = left_ptr + offs_b + offs_d_l + offs_i + offs_k_l
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ------------ right tile (j, k) transposed ---------------
        # we need right_T[k, j] = right[b, j, k, d]
        # shape (BLOCK_K, BLOCK_N)
        offs_j = j[None, :] * stride_r_j                # (1,N)
        offs_k_r = k[:, None] * stride_r_k               # (K,1)
        right_ptrs = right_ptr + offs_b + offs_d_r + offs_k_r + offs_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # ------------ dot product --------------------------------
        acc += tl.dot(left, right)   # (M,K)·(K,N) -> (M,N)

    # --------------------------------------------------------------
    # Store result
    # --------------------------------------------------------------
    offs_i_o = i[:, None] * stride_o_i
    offs_j_o = j[None, :] * stride_o_j
    out_ptrs = out_ptr + b * stride_o_b + offs_d_o + offs_i_o + offs_j_o
    tl.store(out_ptrs, acc, mask=mask_ij)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim]
            - mask:        torch.Tensor, shape [B, N, N] (float, 0/1)
            - weights: dict of torch.Tensors with keys:
                'norm.weight', 'norm.bias',
                'left_proj.weight', 'right_proj.weight',
                'left_gate.weight', 'right_gate.weight',
                'out_gate.weight',
                'to_out_norm.weight', 'to_out_norm.bias',
                'to_out.weight'
            - config: dict with keys 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (dim dimension)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim] float32

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) and gates
    # ------------------------------------------------------------------
    # Linear projections
    left_proj = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right_proj = F.linear(x, weights["right_proj.weight"])

    # Gates (sigmoid)
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 3) Apply mask (if provided) and gate the projections
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)        # [B,N,N,1]
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    left = left_proj * left_gate   # [B,N,N,hidden_dim]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4) Cast to fp16 for the heavy reduction
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Allocate output tensor for the reduction (still fp16)
    out_fp16 = torch.empty_like(left_fp16)  # shape [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 5) Triton kernel launch (batched matmul per hidden channel)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides (int64)
    s_l_b, s_l_i, s_l_k, s_l_d = left_fp16.stride()
    s_r_b, s_r_j, s_r_k, s_r_d = right_fp16.stride()
    s_o_b, s_o_i, s_o_j, s_o_d = out_fp16.stride()

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    triton_trimul_matmul[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        N,
        s_l_b, s_l_i, s_l_k, s_l_d,
        s_r_b, s_r_j, s_r_k, s_r_d,
        s_o_b, s_o_i, s_o_j, s_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN_DIM=hidden_dim,
    )

    # ------------------------------------------------------------------
    # 6) Convert back to float32 and apply post‑processing
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)           # [B,N,N,hidden_dim]

    # LayerNorm on hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Multiply by out‑gate
    out = out * out_gate

    # Final linear projection back to dim
    output = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return output
