```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Set dtype for performance
    # Compute x_norm
    # Layernorm: mean across last dim
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask handling
    if mask is not None:
        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Allocate output tensor
    out_fp16 = torch.empty_like(left_fp16)  # same shape

    # Prepare strides
    B, N, _, _ = left_fp16.shape

    # get strides in elements (not bytes)
    stride_left = left_fp16.stride()
    # define each attr:
    stride_left_b, stride_left_i, stride_left_j, stride_left_d = stride_left
    # right strides
    stride_right = right_fp16.stride()
    stride_right_b, stride_right_i, stride_right_j, stride_right_d = stride_right
    # out strides:
    stride_out = out_fp16.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = stride_out

    # define block sizes (tuned)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64   # try 64

    # compute grid
    num_blocks_i = triton.cdiv(N, BLOCK_M)
    num_blocks_j = triton.cdiv(N, BLOCK_N)
    # combine batch and hidden_dim groups
    total_groups = B * hidden_dim

    grid = (total_groups, num_blocks_i, num_blocks_j)

    trimul_outgoing_kernel[grid](
        left_fp16, right_fp16, out_fp16,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_j, stride_left_d,
        stride_right_b, stride_right_i, stride_right_j, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Cast out back to float32
    out = out_fp16.to(torch.float32)

    # Apply to_out_norm (layernorm over hidden_dim)
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # Multiply by out_gate
    out = out * out_gate

    # Final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

```python
    B, N, hidden_dim,
    stride_left_b, stride_left_i, stride_left_j, stride_left_d,
    stride_right_b, stride_right_i, stride_right_j, stride_right_d,

    pid_n = tl.program_id(2)
```

```python
import torch
import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_left_b, stride_left_i, stride_left_j, stride_left_d,
    stride_right_b, stride_right_i, stride_right_j, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)  # combined batch and hidden dim
    pid_m = tl.program_id(1)  # i block
    pid_n = tl.program_id(2)  # j block

    # Decode batch and hidden index
    b = pid0 // hidden_dim
    d = pid0 % hidden_dim

    # Compute the absolute indices for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile [M, K] : left[b, i, k, d]
        left_ptrs = left_ptr + b * stride_left_b + d * stride_left_d + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_j
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        left = left.to(tl.float32)

        # Load right tile [K, N] : right[b, j, k, d] -> treat rows=k, cols=j
        right_ptrs = right_ptr + b * stride_right_b + d * stride_right_d + offs_k[:, None] * stride_right_i + offs_n[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        right = right.to(tl.float32)

        acc += tl.dot(left, right)

    # Write out accumulator
    out_ptrs = out_ptr + b * stride_out_b + d * stride_out_d + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the outgoing triangle multiplicative update (TriMul).

    Steps:
      1. Layer‑norm the input.
      2. Linear projections (left/right) and gating.
      3. Apply mask if provided.
      4. Compute the O(N³) bilinear update using a tiled Triton kernel.
      5. Layer‑norm the result, apply the output gate and final linear projection.
    """
    import torch.nn.functional as F

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --- 1. Input layernorm -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --- 2. Linear projections ----------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --- 3. Mask (optional) ---------------------------------------------------
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # --- 4. Gates ------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --- 5. Triton kernel for the cubic term ---------------------------------
    # Cast to half precision for bandwidth‑limited kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Output buffer (half precision; will be cast back later)
    out_fp16 = torch.empty_like(left_fp16)

    B, N, _, _ = left_fp16.shape

    # Strides in elements (not bytes)
    sL_b, sL_i, sL_j, sL_d = left_fp16.stride()
    sR_b, sR_i, sR_j, sR_d = right_fp16.stride()
    sO_b, sO_i, sO_j, sO_d = out_fp16.stride()

    # Tile sizes – tuned for the B200 architecture
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    num_blocks_i = triton.cdiv(N, BLOCK_M)
    num_blocks_j = triton.cdiv(N, BLOCK_N)
    total_groups = B * hidden_dim

    grid = (total_groups, num_blocks_i, num_blocks_j)

    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        sL_b,
        sL_i,
        sL_j,
        sL_d,
        sR_b,
        sR_i,
        sR_j,
        sR_d,
        sO_b,
        sO_i,
        sO_j,
        sO_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    torch.cuda.synchronize()

    # Convert back to float32 for the remaining layers
    out = out_fp16.to(torch.float32)

    # --- 6. Output layernorm -------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --- 7. Output gate ------------------------------------------------------
    out = out * out_gate

    # --- 8. Final linear projection -----------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update (TriMul).

The heavy O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched GEMM for each hidden dimension d using a
tiled Triton kernel. All other operations (LayerNorms, Linear projections,
gates, final output projection) are performed with PyTorch for simplicity.
The kernel works on half‑precision tensors to reduce memory traffic and
accumulates in FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: computes out = Σ_k left[...,i,k,d] * right[...,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_left_b, stride_left_i, stride_left_j, stride_left_d,
    stride_right_b, stride_right_i, stride_right_j, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid0 = tl.program_id(0)          # combined batch * hidden dim
    pid_m = tl.program_id(1)         # i‑block
    pid_n = tl.program_id(2)         # j‑block

    # decode batch index b and hidden channel d
    b = pid0 // hidden_dim
    d = pid0 % hidden_dim

    # offsets for the current tiles
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: left[b, i, k, d]  -> shape (M, K)
        left_ptrs = (left_ptr
                     + b * stride_left_b
                     + d * stride_left_d
                     + offs_m[:, None] * stride_left_i
                     + offs_k[None, :] * stride_left_j)
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile: right[b, j, k, d] (treated as rows=k, cols=j) -> shape (K, N)
        right_ptrs = (right_ptr
                      + b * stride_right_b
                      + d * stride_right_d
                      + offs_k[:, None] * stride_right_i
                      + offs_n[None, :] * stride_right_j)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # matrix multiply for this K‑block
        acc += tl.dot(left, right)

    # write back the result (store as FP16 to save bandwidth)
    out_ptrs = (out_ptr
                + b * stride_out_b
                + d * stride_out_d
                + offs_m[:, None] * stride_out_i
                + offs_n[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module with a Triton kernel for the
    cubic update.  Returns a tensor of shape [bs, seq_len, seq_len, dim]
    (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # 1) Input LayerNorm
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # 2) Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # 3) Optional mask (mask may be None for the “nomask” case)
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # 4) Gating (sigmoid)
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton kernel for the O(N³) contraction
    # ------------------------------------------------------------------
    # Cast to half precision for the kernel to cut memory traffic
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Output buffer (half precision)
    out_fp16 = torch.empty_like(left_fp16)

    B, N, _, _ = left_fp16.shape

    # Strides in element‑units (not bytes)
    sL_b, sL_i, sL_j, sL_d = left_fp16.stride()
    sR_b, sR_i, sR_j, sR_d = right_fp16.stride()
    sO_b, sO_i, sO_j, sO_d = out_fp16.stride()

    # Tiling parameters – tuned for the B200 GPU
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    num_blocks_i = triton.cdiv(N, BLOCK_M)
    num_blocks_j = triton.cdiv(N, BLOCK_N)
    total_groups = B * hidden_dim                     # each (b,d) pair gets its own GEMM

    grid = (total_groups, num_blocks_i, num_blocks_j)

    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        sL_b,
        sL_i,
        sL_j,
        sL_d,
        sR_b,
        sR_i,
        sR_j,
        sR_d,
        sO_b,
        sO_i,
        sO_j,
        sO_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,           # reasonable occupancy on B200
    )
    torch.cuda.synchronize()

    # Back to FP32 for the remaining layers
    out = out_fp16.to(torch.float32)

    # 6) Output LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # 7) Apply output gate
    out = out * out_gate

    # 8) Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
