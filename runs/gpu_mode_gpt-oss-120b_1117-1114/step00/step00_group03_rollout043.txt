    stride_bo, stride_io, stride_jo,

"""
TriMul "outgoing" implementation.

The forward pass consists of:
1. LayerNorm over the last dimension.
2. Linear projections (left/right) and gating.
3. Core cubic operation:
     out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix‑multiply for each hidden channel d.
   Implemented with a Triton kernel that tiles the (i, j) axes and
   iterates over k in blocks.
4. LayerNorm + gating on the hidden dimension.
5. Final linear projection back to the input dimension.

Only the N³ reduction (step 3) is performed in Triton; the rest
uses regular PyTorch ops.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _trimul_kernel(
    left_ptr,          # *float32
    right_ptr,         # *float32
    out_ptr,           # *float32
    N,                 # i32   sequence length
    stride_bl, stride_il, stride_kl,   # i32 strides for left: batch, i, k
    stride_br, stride_jr, stride_kr,   # i32 strides for right: batch, j, k
    stride_bo, stride_io, stride_jo,   # i32 strides for out : batch, i, j
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    K_BLOCKS: tl.constexpr,
):
    pid_m = tl.program_id(0)               # tile over i
    pid_n = tl.program_id(1)               # tile over j
    pid_b = tl.program_id(2)               # batch dimension

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    b  = pid_b

    # ranges inside the tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # accumulator for a (BLOCK_M x BLOCK_N) tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k in blocks
    for kb in range(K_BLOCKS):
        k0 = kb * BLOCK_K
        cur_k = k0 + offs_k

        # masks for the current block
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = cur_k < N

        # ---------------------------------------------------------
        # left tile: shape (BLOCK_M, BLOCK_K)
        # address: batch + i*stride_il + k*stride_kl
        left_ptrs = left_ptr + \
            b * stride_bl + \
            offs_i[:, None] * stride_il + \
            cur_k[None, :] * stride_kl
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile: shape (BLOCK_K, BLOCK_N)
        # address: batch + k*stride_kr + j*stride_jr   (k is the first dim)
        right_ptrs = right_ptr + \
            b * stride_br + \
            cur_k[:, None] * stride_kr + \
            offs_j[None, :] * stride_jr
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # accumulate matrix product for this k‑block
        acc += tl.dot(left, right)
        # ---------------------------------------------------------

    # write back the tile
    out_ptrs = out_ptr + \
        b * stride_bo + \
        offs_i[:, None] * stride_io + \
        offs_j[None, :] * stride_jo
    mask_out = (offs_i < N)[:, None] & (offs_j < N)[None, :]
    tl.store(out_ptrs, acc, mask=mask_out)


def custom_kernel(data):
    """
    Args:
        data: Tuple (input, mask, weights, config)
            - input: torch.Tensor of shape [bs, N, N, dim]
            - mask: torch.Tensor of shape [bs, N, N] (may be all‑ones)
            - weights: dict with all model parameters
            - config: dict with "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [bs, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, _ = input_tensor.shape

    # ----- LayerNorm on the last dimension -----
    x = torch.nn.functional.layer_norm(
        input_tensor.float(),
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ----- Linear projections -----
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ----- Optional mask -----
    if mask is None:
        mask = torch.ones(bs, N, N, device=x.device, dtype=x.dtype)
    mask = mask.unsqueeze(-1).type_as(x)
    left = left * mask
    right = right * mask

    # ----- Gating -----
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----- Bring hidden dimension forward for the kernel -----
    # shape becomes [bs, hidden_dim, N, N] and stays contiguous
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    out = torch.empty_like(left)

    # ----- Triton kernel configuration -----
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    K_BLOCKS = (N + BLOCK_K - 1) // BLOCK_K  # compile‑time constant for the kernel

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), bs)

    # ----- Run the kernel for each hidden channel -----
    for h in range(hidden_dim):
        left_h  = left[:, h, :, :]   # [bs, N, N]
        right_h = right[:, h, :, :]  # [bs, N, N]
        out_h   = out[:, h, :, :]    # [bs, N, N]

        # strides (in elements) for each view
        stride_bl, stride_il, stride_kl = left_h.stride()
        stride_br, stride_jr, stride_kr = right_h.stride()
        stride_bo, stride_io, stride_jo = out_h.stride()

        _trimul_kernel[grid](
            left_h,
            right_h,
            out_h,
            N,
            stride_bl, stride_il, stride_kl,
            stride_br, stride_jr, stride_kr,
            stride_bo, stride_io, stride_jo,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            K_BLOCKS=K_BLOCKS,
        )

    # ----- Restore original layout -----
    out = out.permute(0, 2, 3, 1).contiguous()   # [bs, N, N, hidden_dim]

    # ----- Final LayerNorm, gating and linear -----
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
