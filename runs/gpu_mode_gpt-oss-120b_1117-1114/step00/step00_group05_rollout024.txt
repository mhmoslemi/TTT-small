    # Use torch.nn.functional.layer_norm for speed:
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    right = torch.nn.functional.linear(x, right_proj_weight)

@triton.jit
def batched_gemm_kernel(A_ptr, B_ptr, C_ptr,
                        M, N, K,
                        stride_Ab, stride_Ai, stride_Ak,
                        stride_Bb, stride_Bk, stride_Bj,
                        stride_Cb, stride_Ci, stride_Cj,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # create accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # load A and B
        a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
        b_ptrs = B_ptr + pid_b * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bj

        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # compute dot
        acc += tl.dot(a, b)

    # store result
    c_ptrs = C_ptr + pid_b * stride_Cb + offs_m[:, None] * stride_Ci + offs_n[None, :] * stride_Cj

"""
TriMul (outgoing) implementation using Triton for the cubic
pairwise multiplication.
The algorithm follows the reference PyTorch module:

1. Layer‑norm the input tensor.
2. Linear projections to a hidden dimension (left/right).
3. Apply an optional mask and sigmoid gates (left/right).
4. Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   using a custom batched GEMM Triton kernel (the expensive N³ step).
5. Layer‑norm the result, multiply by an output gate, and apply the final
   linear projection back to the original dimension.

All heavy arithmetic (the triple sum) is performed in the Triton kernel;
the remaining operations use standard PyTorch for simplicity.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Ai, stride_Ak,
    stride_Bb, stride_Bk, stride_Bj,
    stride_Cb, stride_Ci, stride_Cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_b = tl.program_id(0)  # batch * hidden
    pid_m = tl.program_id(1)  # row block
    pid_n = tl.program_id(2)  # col block

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # pointers for A and B
        a_ptrs = (
            A_ptr
            + pid_b * stride_Ab
            + offs_m[:, None] * stride_Ai
            + offs_k[None, :] * stride_Ak
        )
        b_ptrs = (
            B_ptr
            + pid_b * stride_Bb
            + offs_k[:, None] * stride_Bk
            + offs_n[None, :] * stride_Bj
        )

        # masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # accumulate
        acc += tl.dot(a, b)

    # write out result
    c_ptrs = (
        C_ptr
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Ci
        + offs_n[None, :] * stride_Cj
    )
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple of (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (optional, broadcastable)
            - weights_dict: mapping of weight names to torch Tensors
            - config_dict: contains "dim" (C) and "hidden_dim" (H)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. LayerNorm over the feature dimension C
    # -----------------------------------------------------------------
    eps = 1e-5
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast to hidden dim)
    # -----------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(dtype=dtype, device=device).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Sigmoid gates derived from the normalized input
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Prepare tensors for the batched GEMM:
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    This is a batch of H independent matrix multiplications per batch.
    # -----------------------------------------------------------------
    # reshape to (B, H, N, N) and make contiguous for simple view
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 2, 1).contiguous()   # (B, H, N, N) where dims are (B,H,K,J)

    BH = B * H
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_t.view(BH, N, N)

    # allocate output tensor for the GEMM
    out_flat = torch.empty_like(left_flat)

    # -----------------------------------------------------------------
    # 6. Triton kernel launch
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        BH,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,          # M
        N,          # N
        N,          # K
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # -----------------------------------------------------------------
    # 7. Reshape back to [B, N, N, H]
    # -----------------------------------------------------------------
    out_hidden = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)  # (B, N, N, H)

    # -----------------------------------------------------------------
    # 8. Output LayerNorm, output gate, and final linear projection
    # -----------------------------------------------------------------
    out_norm = F.layer_norm(
        out_hidden,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate
    output = F.linear(out_norm, weights["to_out.weight"])

    return output
