```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, n_elements, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    Triton-accelerated forward pass of TriMul (outgoing version).

    The kernel accelerates the elementwise multiplication of projected tensors
    with their respective gates. The O(N³) pairwise product is performed by
    reshaping the tensors and using a batched GEMM (torch.bmm), which is
    highly optimized on CUDA. The final steps apply LayerNorm, gating and the
    output linear projection.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on input
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # Linear projections (no bias)
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # Gating values
    left_gate = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # Apply mask if present
    if not config.get("nomask", False):
        # mask shape (B, N, N) -> (B, N, N, 1) broadcast
        mask_exp = mask.unsqueeze(-1).type_as(left)  # cast to same dtype
        left = left * mask_exp
        right = right * mask_exp

    # Ensure contiguous for Triton
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # Allocate output tensors for gated left/right
    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    # Triton elementwise multiplication kernel (left * left_gate, right * right_gate)
    total_elems = left.numel()
    BLOCK = 1024  # tunable, must be multiple of 32
    grid = (triton.cdiv(total_elems, BLOCK),)

    # Launch kernels
    elemwise_mul_kernel[grid](
        left.data_ptr(),
        left_gate.data_ptr(),
        left_out.data_ptr(),
        total_elems,
        BLOCK=BLOCK,
    )
    elemwise_mul_kernel[grid](
        right.data_ptr(),
        right_gate.data_ptr(),
        right_out.data_ptr(),
        total_elems,
        BLOCK=BLOCK,
    )

    # At this point left_out/right_out are gated (and masked) tensors
    # Compute the triangular multiplicative update via batched GEMM
    B, N, _, D = left_out.shape  # D == hidden_dim
    # reshape to (B*D, N, N) for batched matmul
    left_flat = left_out.permute(0, 3, 1, 2).reshape(B * D, N, N)
    right_flat = right_out.permute(0, 3, 1, 2).reshape(B * D, N, N)

    # Perform batched matrix multiplication: left * right^T over k dimension
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # restore shape (B, N, N, hidden_dim)
    out = out_flat.view(B, D, N, N).permute(0, 2, 3, 1)

    # Apply output LayerNorm, gating and final linear projection
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # Ensure float32 output
    out = out.to(torch.float32)

    return out
```

"""TriMul (outgoing) forward pass with Triton acceleration.

The kernel accelerates the element‑wise multiplication of the projected tensors
with their respective gates.  The O(N³) “tri‑multiplicative” update is
implemented by reshaping the gated tensors and using a batched GEMM
(`torch.bmm`), which is already highly optimized on CUDA.  The rest of the
operations (LayerNorm, linear projections, gating, final projection) are
performed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, n_elements, BLOCK_SIZE: tl.constexpr):
    """Element‑wise multiplication a * b → out.

    All tensors are 1‑D flattened views (contiguous) of the same shape.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be ignored if nomask)
            - weights: dict of torch.Tensors containing model parameters
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm on the input ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ---------- Linear projections (no bias) ----------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Gating values ----------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ---------- Apply mask if required ----------
    if not config.get("nomask", False):
        # mask: (B, N, N) → (B, N, N, 1) for broadcasting
        mask_exp = mask.unsqueeze(-1).type_as(left)
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Triton kernel: gate the projected tensors ----------
    # Ensure contiguity for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    total_elems = left.numel()
    BLOCK = 1024  # power‑of‑2, multiple of 32
    grid = (triton.cdiv(total_elems, BLOCK),)

    # left_out = left * left_gate
    elemwise_mul_kernel[grid](
        left,
        left_gate,
        left_out,
        total_elems,
        BLOCK=BLOCK,
    )
    # right_out = right * right_gate
    elemwise_mul_kernel[grid](
        right,
        right_gate,
        right_out,
        total_elems,
        BLOCK=BLOCK,
    )

    # ---------- Tri‑multiplicative update via batched GEMM ----------
    B, N, _, D = left_out.shape  # D == hidden_dim
    # reshape to (B*D, N, N)
    left_flat = left_out.permute(0, 3, 1, 2).reshape(B * D, N, N)
    right_flat = right_out.permute(0, 3, 1, 2).reshape(B * D, N, N)

    # out_flat[i] = left_flat[i] @ right_flat[i].T   (sum over k)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, D, N, N).permute(0, 2, 3, 1)

    # ---------- Output LayerNorm, gating, final linear ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # Ensure final dtype is float32
    out = out.to(torch.float32)

    return out
