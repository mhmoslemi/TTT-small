     projection back to the original channel dimension.

Step 5 is the expensive O(N³·H) part; we accelerate it with a batched GEMM via
torch.bmm (cuBLAS) while the element‑wise operations are fused in a small
Triton kernel (`elementwise_mul_kernel`).  The final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# -------------------------------------------------------------------------
# Triton kernel: element‑wise multiplication (a * b)
# -------------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr,               # pointer to tensor a (contiguous)
    b_ptr,               # pointer to tensor b (contiguous)
    out_ptr,             # pointer to output tensor
    size,                # total number of elements (int)
    BLOCK: tl.constexpr,  # tiles per program
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK
    offs = block_start + tl.arange(0, BLOCK)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


def elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Element‑wise multiplication a * b using Triton.
    Both tensors must be contiguous, same shape, and on the same CUDA device.
    """
    assert a.shape == b.shape, "Shapes must match for elementwise_mul"
    out = torch.empty_like(a)
    total_elems = a.numel()
    BLOCK = 1024  # reasonable tile size
    grid = (triton.cdiv(total_elems, BLOCK),)
    elementwise_mul_kernel[grid](a, b, out, total_elems, BLOCK=BLOCK)
    return out


# -------------------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor:  [B, N, N, C]   (float32, CUDA)
            - mask:         [B, N, N]      (float32 or bool, CUDA)
            - weights: dict with all required weight tensors
            - config:  dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # -----------------------------------------------------------------
    # unpack
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # -----------------------------------------------------------------
    # pull weights onto the correct device / dtype
    # -----------------------------------------------------------------
    def w(name):
        return weights[name].to(device=device, dtype=dtype)

    norm_weight = w("norm.weight")
    norm_bias = w("norm.bias")
    left_proj_weight = w("left_proj.weight")
    right_proj_weight = w("right_proj.weight")
    left_gate_weight = w("left_gate.weight")
    right_gate_weight = w("right_gate.weight")
    out_gate_weight = w("out_gate.weight")
    to_out_norm_weight = w("to_out_norm.weight")
    to_out_norm_bias = w("to_out_norm.bias")
    to_out_weight = w("to_out.weight")       # shape [dim, hidden_dim]

    # -----------------------------------------------------------------
    # 1) Layer‑norm on input
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # 3) Apply pair‑wise mask (if any)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_broadcast = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask_broadcast
        right = right * mask_broadcast

    # -----------------------------------------------------------------
    # 4) Gating weights (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # -----------------------------------------------------------------
    # 5) Element‑wise gated multiplication (Triton)
    # -----------------------------------------------------------------
    left = elementwise_mul(left, left_gate)
    right = elementwise_mul(right, right_gate)

    # -----------------------------------------------------------------
    # 6) Core triangular update: batched GEMM over the hidden dimension
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as B*hidden_dim independent matrix multiplications.
    # -----------------------------------------------------------------
    # reshape tensors to (B*H, N, N) where axis 1 = i (or j) and axis 2 = k
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # matrix product: (i,k) @ (k,j)ᵀ  -> (i,j)
    prod = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim]
    out = prod.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 7) Final hidden‑dim layer‑norm
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 8) Gated scaling using the out_gate (Triton)
    # -----------------------------------------------------------------
    out = elementwise_mul(out, out_gate)

    # -----------------------------------------------------------------
    # 9) Linear projection back to original channel dimension
    # -----------------------------------------------------------------
    output = F.linear(out, to_out_weight)   # shape [B, N, N, dim]

    return output
```
