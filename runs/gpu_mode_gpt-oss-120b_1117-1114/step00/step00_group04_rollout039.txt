class TriMul(nn.Module):
    __init__(dim, hidden_dim)
        self.norm = LayerNorm(dim)
        self.left_proj = Linear(dim, hidden_dim, bias=False)
        self.right_proj = Linear(dim, hidden_dim, bias=False)
        self.left_gate = Linear(dim, hidden_dim, bias=False)
        self.right_gate = Linear(dim, hidden_dim, bias=False)
        self.out_gate = Linear(dim, hidden_dim, bias=False)
        self.to_out_norm = LayerNorm(hidden_dim)
        self.to_out = Linear(hidden_dim, dim, bias=False)

forward(x, mask)
    bs, N, _, dim = x.shape
    x = self.norm(x)
    left = self.left_proj(x)
    right = self.right_proj(x)
    mask = mask.unsqueeze(-1)
    left = left * mask
    right = right * mask
    left_gate = self.left_gate(x).sigmoid()
    right_gate = self.right_gate(x).sigmoid()
    out_gate = self.out_gate(x).sigmoid()
    left = left * left_gate
    right = right * right_gate
    out = einsum('... i k d, ... j k d -> ... i j d', left, right)
    out = self.to_out_norm(out)
    out = out * out_gate
    return self.to_out(out)

@triton.jit
def final_proj_kernel(
    out_ptr, out_gate_ptr,
    norm_weight_ptr, norm_bias_ptr,
    to_out_weight_ptr,
    out_final_ptr,
    B, N, hidden_dim, dim_out,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_norm_w,
    stride_norm_b,
    stride_to_out_w_d, stride_to_out_w_h,
    stride_out_final_b, stride_out_final_i, stride_out_final_j, stride_out_final_d,
    EPS: tl.constexpr = 1e-5,
    BLOCK_D: tl.constexpr = 32,
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return
    # decode indices
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # compute base offsets for out and gate
    out_base = out_ptr + b * stride_out_b + i * stride_out_i + j * stride_out_j
    gate_base = out_gate_ptr + b * stride_gate_b + i * stride_gate_i + j * stride_gate_j

    # first reduction for mean/var
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)
    for h in range(0, hidden_dim):
        val = tl.load(out_base + h * stride_out_h)
        sum_val += val
        sum_sq += val * val
    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # second pass: compute gated values and accumulate output per d block
    # iterate over d blocks
    for d_off in range(0, dim_out, BLOCK_D):
        out_acc = tl.zeros([BLOCK_D], dtype=tl.float32)
        # iterate over hidden
        for h in range(0, hidden_dim):
            # load out, gate
            out_val = tl.load(out_base + h * stride_out_h)
            gate_val = tl.load(gate_base + h * stride_gate_h)
            # apply LN
            norm_w = tl.load(norm_weight_ptr + h * stride_norm_w)  # stride_norm_w typically 1
            norm_b = tl.load(norm_bias_ptr + h * stride_norm_b)    # stride_norm_b typically 1
            normed = (out_val - mean) * inv_std
            normed = normed * norm_w + norm_b
            # apply gating
            gated = normed * gate_val
            # load weight for this d block: shape (BLOCK_D, hidden_dim)
            # we need to load from weight pointer: to_out_weight_ptr + d_off * stride_to_out_w_d + h * stride_to_out_w_h

            d_range = tl.arange(0, BLOCK_D)
            w = tl.load(w_ptr + d_range * stride_to_out_w_d, mask=d_range < dim_out - d_off, other=0.0)
            # accumulate:
            out_acc += w * gated
        # store output vector block
        out_final_base = out_final_ptr + b * stride_out_final_b + i * stride_out_final_i + j * stride_out_final_j + d_off * stride_out_final_d

TriMul (outgoing) forward pass implemented with Triton.

The heavy N³ pairwise multiplication is performed with a batched
matrix‑multiply (torch.matmul) which leverages cuBLAS.  The final
LayerNorm, gating and projection back to the original feature dimension
are fused into a Triton kernel that operates over each (batch, i, j)
position independently.

All tensors are assumed to reside on the CUDA device.
"""

import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm (to_out_norm), out_gate multiplication,
# and final linear projection (to_out).  Operates on a single (b,i,j)
# position per program instance.
# ----------------------------------------------------------------------
@triton.jit
def _final_proj_kernel(
    out_ptr,               # hidden tensor after einsum, shape (B,N,N,H)
    out_gate_ptr,          # gating tensor, same shape
    norm_w_ptr,            # to_out_norm.weight, shape (H,)
    norm_b_ptr,            # to_out_norm.bias,   shape (H,)
    proj_w_ptr,            # to_out.weight, shape (D, H)
    out_final_ptr,         # output tensor, shape (B,N,N,D)
    B, N,                  # batch size, sequence length (runtime)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_norm_w, stride_norm_b,
    stride_proj_w_d, stride_proj_w_h,
    stride_out_final_b, stride_out_final_i, stride_out_final_j, stride_out_final_d,
    hidden_dim: tl.constexpr,   # H
    dim_out: tl.constexpr,      # D
    EPS: tl.constexpr = 1e-5,
    BLOCK_D: tl.constexpr = 32,
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # Decode linear index -> (b,i,j)
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # Base pointers for this position
    out_base = out_ptr + b * stride_out_b + i * stride_out_i + j * stride_out_j
    gate_base = out_gate_ptr + b * stride_gate_b + i * stride_gate_i + j * stride_gate_j

    # --------------------------------------------------------------
    # Compute mean and variance over hidden dimension (layernorm)
    # --------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)
    for h in range(0, hidden_dim):
        x = tl.load(out_base + h * stride_out_h)
        sum_val += x
        sum_sq += x * x
    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # --------------------------------------------------------------
    # For each block of output dimensions, accumulate the dot product
    # --------------------------------------------------------------
    d_range = tl.arange(0, BLOCK_D)
    for d_off in range(0, dim_out, BLOCK_D):
        # accumulator for a block of D values
        acc = tl.zeros([BLOCK_D], dtype=tl.float32)

        # loop over hidden dimension
        for h in range(0, hidden_dim):
            # load tensors
            out_val = tl.load(out_base + h * stride_out_h)
            gate_val = tl.load(gate_base + h * stride_gate_h)

            # LayerNorm on hidden value
            norm_w = tl.load(norm_w_ptr + h * stride_norm_w)
            norm_b = tl.load(norm_b_ptr + h * stride_norm_b)
            normed = (out_val - mean) * inv_std
            normed = normed * norm_w + norm_b

            # Apply out_gate
            gated = normed * gate_val

            # Load projection weight for this hidden index and a block of D
            w_ptr = proj_w_ptr + d_off * stride_proj_w_d + h * stride_proj_w_h
            w = tl.load(
                w_ptr + d_range * stride_proj_w_d,
                mask=d_range < (dim_out - d_off),
                other=0.0,
            )
            # Accumulate
            acc += w * gated

        # Store result block
        out_final_base = (
            out_final_ptr
            + b * stride_out_final_b
            + i * stride_out_final_i
            + j * stride_out_final_j
            + d_off * stride_out_final_d
        )
        mask_store = d_range < (dim_out - d_off)
        tl.store(out_final_base + d_range * stride_out_final_d, acc, mask=mask_store)


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] (bool / 0‑1)
            - weights: dict of model weight tensors
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load weights (all assumed to be on the correct device)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)          # [B, N, N, H]
    right = F.linear(x, right_proj_w)        # [B, N, N, H]

    # ------------------------------------------------------------------
    # Mask handling
    # ------------------------------------------------------------------
    if nomask:
        mask_f = torch.ones_like(mask, dtype=dtype, device=device)
    else:
        mask_f = mask.to(dtype=dtype, device=device)
    mask_f = mask_f.unsqueeze(-1)            # [B, N, N, 1]
    left = left * mask_f
    right = right * mask_f

    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core multiplicative update: einsum -> batched matmul
    # Equivalent to: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # batched matmul over the hidden dimension
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))
    out = out_perm.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]

    # ------------------------------------------------------------------
    # Final projection: fused LayerNorm(to_out_norm), out_gate, and
    # Linear(to_out). Implemented in Triton.
    # ------------------------------------------------------------------
    B, N, _, _ = out.shape
    out_gate = out_gate.contiguous()

    # Allocate output tensor
    out_final = torch.empty((B, N, N, dim), dtype=torch.float32, device=device)

    # Strides (convert to Python ints for Triton)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h = out_gate.stride()
    stride_norm_w = to_out_norm_w.stride(0)
    stride_norm_b = to_out_norm_b.stride(0)
    stride_proj_w_d = to_out_w.stride(0)   # distance between rows (dim * hidden)
    stride_proj_w_h = to_out_w.stride(1)   # distance between columns (should be 1)
    stride_out_final_b, stride_out_final_i, stride_out_final_j, stride_out_final_d = out_final.stride()

    total = B * N * N
    grid = (triton.cdiv(total, 1),)

    _final_proj_kernel[grid](
        out,
        out_gate,
        to_out_norm_w,
        to_out_norm_b,
        to_out_w,
        out_final,
        B,
        N,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        stride_gate_b,
        stride_gate_i,
        stride_gate_j,
        stride_gate_h,
        stride_norm_w,
        stride_norm_b,
        stride_proj_w_d,
        stride_proj_w_h,
        stride_out_final_b,
        stride_out_final_i,
        stride_out_final_j,
        stride_out_final_d,
        hidden_dim=hidden_dim,
        dim_out=dim,
        BLOCK_D=32,
        EPS=1e-5,
    )
    return out_final
