@triton.jit
def elementwise_fused_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    B, N, hidden_dim,
    stride_left_b, stride_left_i, stride_left_j, stride_left_d,
    stride_right_b, stride_right_i, stride_right_j, stride_right_d,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_d,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_d,
    BLOCK_SIZE: tl.constexpr):

    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    # total elements
    num_elements = B * N * N * hidden_dim
    mask = offs < num_elements
    # compute linear index to multi-dim:
    # idx = offs
    # We can compute b = idx // (N*N*hidden_dim)
    # rest = idx % (N*N*hidden_dim)
    # i = rest // (N*hidden_dim)
    # rest2 = rest % (N*hidden_dim)
    # j = rest2 // hidden_dim
    # d = rest2 % hidden_dim
    idx = offs

    # compute indices
    b = idx // (N * N * hidden_dim)
    idx_rem = idx - b * (N * N * hidden_dim)
    i = idx_rem // (N * hidden_dim)
    idx_rem2 = idx_rem - i * (N * hidden_dim)
    j = idx_rem2 // hidden_dim
    d = idx_rem2 - j * hidden_dim

    # compute pointers using strides
    left_offset = b * stride_left_b + i * stride_left_i + j * stride_left_j + d * stride_left_d
    right_offset = b * stride_right_b + i * stride_right_i + j * stride_right_j + d * stride_right_d
    left_gate_offset = b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j + d * stride_lgate_d
    right_gate_offset = b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j + d * stride_rgate_d

    # mask index for b,i,j (no d)
    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # load values
    left_val = tl.load(left_ptr + left_offset, mask=mask, other=0.0)
    right_val = tl.load(right_ptr + right_offset, mask=mask, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + left_gate_offset, mask=mask, other=0.0)
    right_gate_val = tl.load(right_gate_ptr + right_gate_offset, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_offset, mask=mask, other=0.0)

    # fuse operations:
    out_left = left_val * mask_val * left_gate_val
    out_right = right_val * mask_val * right_gate_val

    tl.store(out_left_ptr + left_offset, out_left, mask=mask)

    device = input_tensor.device

        mask = mask.to(torch.float32)

    # load weights to proper device/dtype float32
    norm_weight = weights['norm.weight'].to(device)
    norm_bias = weights['norm.bias'].to(device)
    left_proj_weight = weights['left_proj.weight'].to(device)
    right_proj_weight = weights['right_proj.weight'].to(device)
    left_gate_weight = weights['left_gate.weight'].to(device)
    right_gate_weight = weights['right_gate.weight'].to(device)
    out_gate_weight = weights['out_gate.weight'].to(device)
    to_out_norm_weight = weights['to_out_norm.weight'].to(device)
    to_out_norm_bias = weights['to_out_norm.bias'].to(device)
    to_out_weight = weights['to_out.weight'].to(device)

    # Normalize
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)
    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # Apply mask and gates via Triton kernel
    out_left = torch.empty_like(left)
    out_right = torch.empty_like(right)

    B, N, _, _ = left.shape  # B,N,N,hidden_dim
    # Launch kernel
    BLOCK_SIZE = 1024
    total_elements = B * N * N * hidden_dim
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)
    apply_mask_gate_kernel[grid](
        left, right, mask,
        left_gate, right_gate,
        out_left, out_right,
        B, N, hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE
    )

    # Core multiplication
    # For speed use bfloat16
    left_f16 = out_left.to(torch.bfloat16)
    right_f16 = out_right.to(torch.bfloat16)
    out = torch.einsum('b i k d, b j k d -> b i j d', left_f16, right_f16)
    out = out.to(torch.float32)

    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)
    out = out * out_gate

    out = torch.nn.functional.linear(out, to_out_weight)  # bias false

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor using the provided `norm.weight`/`bias`.
2. Linear projections (`left_proj`, `right_proj`) and gating linear layers
   (`left_gate`, `right_gate`, `out_gate`) are computed with PyTorch.
3. A Triton kernel fuses three element‑wise operations:
       * mask broadcasting,
       * multiplication by the left/right gates,
       * writing the masked‑gated tensors to new buffers.
   This kernel processes the flattened `[B,N,N,hidden]` tensors in a 1‑D
   launch, avoiding Python loops and keeping the heavy element‑wise work on
   the GPU.
4. The core multiplicative update
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is performed with a batched einsum (equivalent to a set of
   `N×N` matrix multiplications per hidden channel). The tensors are cast
   to `bfloat16` for speed and the result cast back to `float32`.
5. The result is layer‑normed (`to_out_norm`), multiplied by the
   `out_gate`, and finally projected back to the original dimension with
   `to_out`.

Only the element‑wise mask‑gate fusion is implemented in Triton; all other
operations use PyTorch for simplicity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: apply mask and left/right gates in a single pass
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_gate_kernel(
    left_ptr,          # float32*  [B,N,N,hidden]
    right_ptr,         # float32*  [B,N,N,hidden]
    mask_ptr,          # float32*  [B,N,N]
    left_gate_ptr,     # float32*  [B,N,N,hidden]
    right_gate_ptr,    # float32*  [B,N,N,hidden]
    out_left_ptr,      # float32*  [B,N,N,hidden]  (output)
    out_right_ptr,     # float32*  [B,N,N,hidden]  (output)
    B, N, hidden_dim,  # int32 scalars
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * hidden_dim
    mask = offs < total

    # ----- decode flat index to (b,i,j,d) -----
    idx = offs
    stride_b = N * N * hidden_dim
    stride_i = N * hidden_dim
    stride_j = hidden_dim

    b = idx // stride_b
    rem = idx - b * stride_b
    i = rem // stride_i
    rem2 = rem - i * stride_i
    j = rem2 // stride_j
    d = rem2 - j * stride_j

    # flattened mask index (no hidden dimension)
    mask_idx = b * (N * N) + i * N + j

    # load tensors
    left_val      = tl.load(left_ptr      + idx, mask=mask, other=0.0)
    right_val     = tl.load(right_ptr     + idx, mask=mask, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + idx, mask=mask, other=0.0)
    right_gate_val= tl.load(right_gate_ptr+ idx, mask=mask, other=0.0)
    mask_val      = tl.load(mask_ptr      + mask_idx, mask=mask, other=0.0)

    out_left_val  = left_val  * mask_val * left_gate_val
    out_right_val = right_val * mask_val * right_gate_val

    tl.store(out_left_ptr  + idx, out_left_val,  mask=mask)
    tl.store(out_right_ptr + idx, out_right_val, mask=mask)


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N]       (any dtype, 0/1)
            - weights      : dict of torch.Tensors containing all model
                            parameters (`norm.weight`, `norm.bias`, etc.)
            - config       : dict with at least `dim` and `hidden_dim`.

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32).
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Move all weight tensors to the correct device / dtype
    # ------------------------------------------------------------------
    def to_param(name):
        return weights[name].to(device, dtype=torch.float32)

    norm_weight        = to_param("norm.weight")
    norm_bias          = to_param("norm.bias")
    left_proj_weight   = to_param("left_proj.weight")
    right_proj_weight  = to_param("right_proj.weight")
    left_gate_weight   = to_param("left_gate.weight")
    right_gate_weight  = to_param("right_gate.weight")
    out_gate_weight    = to_param("out_gate.weight")
    to_out_norm_weight = to_param("to_out_norm.weight")
    to_out_norm_bias   = to_param("to_out_norm.bias")
    to_out_weight      = to_param("to_out.weight")

    # ------------------------------------------------------------------
    # Prepare mask (float32, broadcastable)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(input_tensor.shape[:3], device=device, dtype=torch.float32)
    else:
        mask = mask.to(dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)      # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)    # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3) Gating linear layers + sigmoid
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate   = F.linear(x, out_gate_weight).sigmoid()

    # ------------------------------------------------------------------
    # 4) Fuse mask and gates using Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B, N, N, hidden_dim
    out_left  = torch.empty_like(left)
    out_right = torch.empty_like(right)

    BLOCK_SIZE = 1024
    total_elements = B * N * N * hidden_dim
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    apply_mask_gate_kernel[grid](
        left, right, mask,
        left_gate, right_gate,
        out_left, out_right,
        B, N, hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # After the kernel: out_left = left * mask * left_gate
    #                    out_right = right * mask * right_gate

    # ------------------------------------------------------------------
    # 5) Core triangular multiplicative update (batched einsum)
    #    Cast to bfloat16 for speed, then back to float32.
    # ------------------------------------------------------------------
    left_bf16  = out_left.to(torch.bfloat16)
    right_bf16 = out_right.to(torch.bfloat16)
    out = torch.einsum('b i k d, b j k d -> b i j d', left_bf16, right_bf16)
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Output layer‑norm, gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    out = out * out_gate            # apply the final gate
    out = F.linear(out, to_out_weight)   # final projection (no bias)

    return out
