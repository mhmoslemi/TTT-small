def batch_matmul_kernel(A_ptr, B_ptr, C_ptr,
                        M, N, K,
                        stride_A_batch, stride_A_m, stride_A_k,
                        stride_B_batch, stride_B_k, stride_B_n,
                        stride_C_batch, stride_C_m, stride_C_n,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
                        BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # create mask for m and n within bounds
    m_mask = offs_m < M
    n_mask = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # pointer to A block for given batch
    # A_ptr offset: batch*stride_A_batch + offs_m*stride_A_m
    a_ptrs = A_ptr + pid_batch * stride_A_batch + offs_m[:, None] * stride_A_m + tl.arange(0, BLOCK_K)[None, :] * stride_A_k

    # pointer to B block
    b_ptrs = B_ptr + pid_batch * stride_B_batch + tl.arange(0, BLOCK_K)[:, None] * stride_B_k + offs_n[None, :] * stride_B_n

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load tiles
        a = tl.load(a_ptrs, mask=m_mask[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k * BLOCK_K + tl.arange(0, BLOCK_K)[:, None] < K) & n_mask[None, :], other=0.0)
        # compute dot product
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_A_k
        b_ptrs += BLOCK_K * stride_B_k

    # write out
    c_ptrs = C_ptr + pid_batch * stride_C_batch + offs_m[:, None] * stride_C_m + offs_n[None, :] * stride_C_n
    tl.store(c_ptrs, acc.to(tl.float32), mask=m_mask[:, None] & n_mask[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    # bring all to correct dtype (float32)
    # everything on same device
    device = input_tensor.device
    dtype = input_tensor.dtype # could be float16? But assume float32.
    # At runtime as we use triton, need tensors memory contiguous and appropriate dtype.

    # Compute normalized x
    x_norm = torch.nn.functional.layer_norm(input_tensor,
                                            normalized_shape=(dim,),
                                            weight=weights['norm.weight'],
                                            bias=weights['norm.bias'])
    # Additionally convert to float32, if needed.
    # We'll compute everything in float32.
    # Compute projections
    left_proj = torch.nn.functional.linear(x_norm, weights['left_proj.weight'])
    right_proj = torch.nn.functional.linear(x_norm, weights['right_proj.weight'])
    # mask handling
    if mask is not None:

        mask_unsq = mask.unsqueeze(-1).to(dtype=left_proj.dtype)
        left = left_proj * mask_unsq
        right = right_proj * mask_unsq
    else:
        left = left_proj
        right = right_proj

    left_gate = torch.nn.functional.linear(x_norm, weights['left_gate.weight']).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights['right_gate.weight']).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights['out_gate.weight']).sigmoid()

    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape

@triton.jit
def batch_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute pointer base for A and B
    # For A: base pointer for batch, plus row offset
    a_ptrs = A_ptr + pid_batch * stride_Ab + offs_m[:, None] * stride_Am + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    # For B: base pointer for batch, plus col offset
    b_ptrs = B_ptr + pid_batch * stride_Bb + tl.arange(0, BLOCK_K)[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dim
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute mask for K dimension
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # load A and B tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

        # increment pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # write result
    c_ptrs = C_ptr + pid_batch * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) implementation with a fused Triton batched GEMM kernel.

Algorithm:
1. Layer‑norm the input `x` over the channel dimension.
2. Linear projections → left/right (shape B,N,N,hd) and three gating tensors.
3. Apply optional mask, then gating (sigmoid) to left/right.
4. Compute the core contraction:
       out[b,i,j,hd] = Σ_k left[b,i,k,hd] * right[b,j,k,hd]
   This is a batched matrix multiplication of shape (B*hd, N, N) @ (B*hd, N, N)^T,
   implemented in Triton for high throughput (float16 inputs, float32 accumulation).
5. Layer‑norm + output gate on the hidden dimension, then linear projection back to `dim`.

The kernel only covers the N³ contraction; all other operations are performed
with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batch_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to A and B tiles for this batch
    a_ptrs = A_ptr + pid_batch * stride_Ab + offs_m[:, None] * stride_Am + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    b_ptrs = B_ptr + pid_batch * stride_Bb + tl.arange(0, BLOCK_K)[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    c_ptrs = C_ptr + pid_batch * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: [B, N, N, dim]        (float32/float16 on CUDA)
            mask_tensor : [B, N, N]            (bool/float, optional)
            weights_dict : mapping of parameter names to tensors
            config_dict  : contains "dim", "hidden_dim", "nomask" (bool)

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----- LayerNorm on input -----
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ----- Linear projections -----
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hd]
    right = F.linear(x, weights["right_proj.weight"])        # [B,N,N,hd]

    # ----- Optional mask -----
    if not nomask:
        mask_f = mask.to(dtype=left.dtype).unsqueeze(-1)    # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ----- Gating tensors -----
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()   # [B,N,N,hd]
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----- Cast to half for the heavy N³ contraction -----
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Prepare tensors for the batched GEMM: (B,hd,N,N) -> (B*hd,N,N)
    # left: row=i, col=k ; right: row=k, col=j (transposed for GEMM)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B,hd,N,N] (i,k)
    right_perm = right_h.permute(0, 3, 2, 1).contiguous() # [B,hd,N,N] (k,j)

    B, N, _, _ = left.shape
    batch = B * hidden_dim

    left_flat = left_perm.view(batch, N, N)
    right_flat = right_perm.view(batch, N, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # ----- Triton kernel launch -----
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    batch_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- Reshape back to [B,N,N,hd] (float32) -----
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ----- Output LayerNorm, gating, and final projection -----
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
