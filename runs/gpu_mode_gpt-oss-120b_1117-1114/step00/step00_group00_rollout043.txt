def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])

    # linear projections
    left = F.linear(x, weights['left_proj.weight'])
    right = F.linear(x, weights['right_proj.weight'])

    if not nomask:
        mask_ = mask.unsqueeze(-1).type_as(x)
        left = left * mask_
        right = right * mask_

    left_gate = torch.sigmoid(F.linear(x, weights['left_gate.weight']))
    right_gate = torch.sigmoid(F.linear(x, weights['right_gate.weight']))
    out_gate = torch.sigmoid(F.linear(x, weights['out_gate.weight']))

    left = left * left_gate
    right = right * right_gate

    # cast to half for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output tensor
    out = torch.empty((input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2], hidden_dim), dtype=torch.float16, device=input_tensor.device)

    # get strides
    stride_left = left_fp16.stride()
    stride_right = right_fp16.stride()
    stride_out = out.stride()
    B, N = input_tensor.shape[0], input_tensor.shape[1]
    C = hidden_dim

    # kernel launch params
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 32
    BLOCK_D = 16

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B)

    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B, N, C,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D
    )

    # convert out to float32 for subsequent layers
    out_f32 = out.to(torch.float32)

    # to_out_norm
    out_norm = F.layer_norm(out_f32, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # apply out_gate
    out_norm = out_norm * out_gate

    # final linear projection
    output = F.linear(out_norm, weights["to_out.weight"])

    return output

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, C,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Compute out[b,i,j,c] = sum_k left[b,i,k,c] * right[b,j,k,c]"""
    pid_i = tl.program_id(0)  # tile idx for i
    pid_j = tl.program_id(1)  # tile idx for j
    pid_b = tl.program_id(2)  # batch index

    # Base indices for this program
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    b = pid_b

    # thread offsets within the block
    offs_i = tl.arange(0, BLOCK_M)[:, None, None]  # (M,1,1)
    offs_j = tl.arange(0, BLOCK_N)[None, :, None]  # (1,N,1)

    i_idx = i_base + offs_i  # (M,1,1)
    j_idx = j_base + offs_j  # (1,N,1)

    # bounds masks for i and j (broadcast later)
    mask_i = i_idx < N
    mask_j = j_idx < N

    for d_off in range(0, C, BLOCK_D):
        # offsets for hidden dimension within this block
        offs_d = tl.arange(0, BLOCK_D)[None, None, :]  # (1,1,D)
        d_idx = d_off + offs_d                     # (1,1,D)
        mask_d = d_idx < C

        # accumulator for this (i,j) tile and d chunk, in float32
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        for k_off in range(0, N, BLOCK_K):
            offs_k = tl.arange(0, BLOCK_K)[None, :, None]  # (1,K,1)
            k_idx = k_off + offs_k                         # (1,K,1)
            mask_k = k_idx < N

            # ----- load left tile -----
            # left shape: (B, N_i, N_k, C)
            # pointer computes b*stride_l_b + i*stride_l_i + k*stride_l_k + d*stride_l_h
            left_ptrs = (left_ptr
                         + b * stride_l_b
                         + i_idx * stride_l_i
                         + k_idx * stride_l_k
                         + d_idx * stride_l_h)
            mask_left = mask_i & mask_k & mask_d
            left_tile = tl.load(left_ptrs, mask=mask_left, other=0.0).to(tl.float32)

            # ----- load right tile -----
            # right shape: (B, N_j, N_k, C)
            # we need right[b, j, k, d] -> use k as first axis, j as second
            right_ptrs = (right_ptr
                         + b * stride_r_b
                         + k_idx * stride_r_k
                         + j_idx * stride_r_i
                         + d_idx * stride_r_h)
            mask_right = mask_k & mask_j & mask_d
            right_tile = tl.load(right_ptrs, mask=mask_right, other=0.0).to(tl.float32)

            # ----- compute dot products for each d sub‑channel -----
            for d_sub in range(0, BLOCK_D):
                # slice the tile for the current d_sub (shape (M,K) and (K,N))
                left_sub = left_tile[:, :, d_sub]    # (M,K)
                right_sub = right_tile[:, :, d_sub]  # (K,N)
                # matrix multiply MxK * KxN -> MxN
                acc[:, :, d_sub] += tl.dot(left_sub, right_sub)

        # ----- write the accumulated result -----
        out_ptrs = (out_ptr
                    + b * stride_o_b
                    + i_idx * stride_o_i
                    + j_idx * stride_o_j
                    + d_idx * stride_o_h)
        mask_out = mask_i & mask_j & mask_d
        # cast accumulator to the dtype of the output tensor (float16)
        tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.
    Arguments:
        data: tuple (input, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------- LayerNorm on the input ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -------- Optional mask ----------
    if not nomask:
        # mask shape [B,N,N] -> broadcast to hidden dim
        mask_ = mask.unsqueeze(-1).type_as(x)
        left = left * mask_
        right = right * mask_

    # -------- Gated multiplicative factors ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------- Triton kernel for the einsum (pairwise contraction) ----------
    # Cast to float16 to reduce memory bandwidth
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape
    C = hidden_dim

    # Allocate output of the contraction (still hidden_dim channels)
    out_tmp = torch.empty((B, N, N, C), dtype=torch.float16, device=device)

    # Strides (in elements) for each tensor
    stride_left = left_fp16.stride()   # (B, i, k, d)
    stride_right = right_fp16.stride() # (B, i, k, d) – we will treat i as the j‑axis
    stride_out = out_tmp.stride()      # (B, i, j, d)

    # Kernel launch configuration (tune as needed)
    BLOCK_M = 16   # rows of i
    BLOCK_N = 16   # columns of j
    BLOCK_K = 32   # reduction tile size
    BLOCK_D = 16   # hidden‑channel tile size

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B)

    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_tmp,
        B, N, C,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # Convert back to float32 for the remaining layers
    out = out_tmp.to(torch.float32)

    # -------- Output normalisation and final projection ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output

"""
Implementation of the “outgoing” TriMul operation using Triton for the costly
einsum contraction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

The surrounding LayerNorms and linear projections are kept in PyTorch for
readability.  The contraction is tiled over the i, j, k and hidden dimension
axes and computed in float32 accumulation while the inputs/outputs use
float16 to reduce memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const float16 *
    right_ptr,         # const float16 *
    out_ptr,           # float16 *
    B, N, C,           # batch, seq_len, hidden_dim
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Compute the TriMul contraction with a tiling scheme."""
    pid_i = tl.program_id(0)          # i‑tile index
    pid_j = tl.program_id(1)          # j‑tile index
    pid_b = tl.program_id(2)          # batch index

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    b = pid_b

    # thread‑local offsets (broadcasted later)
    offs_i = tl.arange(0, BLOCK_M)[:, None, None]   # (M,1,1)
    offs_j = tl.arange(0, BLOCK_N)[None, :, None]   # (1,N,1)

    i_idx = i_base + offs_i          # (M,1,1)
    j_idx = j_base + offs_j          # (1,N,1)

    # masks for i and j (used for loads / stores)
    mask_i = i_idx < N
    mask_j = j_idx < N

    # -----------------------------------------------------------------
    # Loop over hidden‑dimension blocks
    # -----------------------------------------------------------------
    for d_off in range(0, C, BLOCK_D):
        offs_d = tl.arange(0, BLOCK_D)[None, None, :]   # (1,1,D)
        d_idx = d_off + offs_d                          # (1,1,D)
        mask_d = d_idx < C

        # Accumulator for this (i,j) tile & d‑block, float32
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # -----------------------------------------------------------------
        # Reduction over the k dimension
        # -----------------------------------------------------------------
        for k_off in range(0, N, BLOCK_K):
            offs_k = tl.arange(0, BLOCK_K)[None, :, None]   # (1,K,1)
            k_idx = k_off + offs_k                           # (1,K,1)
            mask_k = k_idx < N

            # clamp indices to stay inside allocated memory
            i_clamped = tl.minimum(i_idx, N - 1)
            j_clamped = tl.minimum(j_idx, N - 1)
            k_clamped = tl.minimum(k_idx, N - 1)
            d_clamped = tl.minimum(d_idx, C - 1)

            # -----------------------------------------------------------------
            # Load left[b, i, k, d]   (shape: M x K x D)
            # -----------------------------------------------------------------
            left_ptrs = (
                left_ptr
                + b * stride_l_b
                + i_clamped * stride_l_i
                + k_clamped * stride_l_k
                + d_clamped * stride_l_h
            )
            mask_left = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
            left_tile = tl.load(left_ptrs, mask=mask_left, other=0.0).to(tl.float32)

            # -----------------------------------------------------------------
            # Load right[b, j, k, d]   (shape: K x N x D)
            # -----------------------------------------------------------------
            right_ptrs = (
                right_ptr
                + b * stride_r_b
                + k_clamped * stride_r_k
                + j_clamped * stride_r_i
                + d_clamped * stride_r_h
            )
            mask_right = mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
            right_tile = tl.load(right_ptrs, mask=mask_right, other=0.0).to(tl.float32)

            # -----------------------------------------------------------------
            # For each hidden‑channel slice compute a small matrix multiply
            # -----------------------------------------------------------------
            for d_sub in range(0, BLOCK_D):
                left_sub = left_tile[:, :, d_sub]      # (M, K)
                right_sub = right_tile[:, :, d_sub]    # (K, N)
                acc[:, :, d_sub] += tl.dot(left_sub, right_sub)

        # -----------------------------------------------------------------
        # Store the accumulated result for this hidden‑channel block
        # -----------------------------------------------------------------
        out_ptrs = (
            out_ptr
            + b * stride_o_b
            + i_clamped * stride_o_i
            + j_clamped * stride_o_j
            + d_clamped * stride_o_h
        )
        mask_out = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
        tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.

    Args:
        data: Tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim] (float32)
            - mask  : torch.Tensor [B, N, N] (bool/float)
            - weights: dict of model parameters (float32)
            - config : dict with keys
                * dim         – input channel dimension
                * hidden_dim – projection dimension
                * nomask      – bool, True ⇒ ignore provided mask
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Linear projections and optional mask
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    if not nomask:
        m = mask.unsqueeze(-1).type_as(x)            # broadcast to hidden_dim
        left = left * m
        right = right * m

    # -----------------------------------------------------------------
    # 3) Gating
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4) Triton kernel for the contracted einsum
    # -----------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape
    C = hidden_dim

    out_tmp = torch.empty((B, N, N, C), dtype=torch.float16, device=device)

    # strides for left/right/out (in elements)
    stride_l = left_fp16.stride()   # (B, i, k, d)
    stride_r = right_fp16.stride()
    stride_o = out_tmp.stride()     # (B, i, j, d)

    # -----------------------------------------------------------------
    # 5) Kernel launch parameters (tuned for typical B200 characteristics)
    # -----------------------------------------------------------------
    BLOCK_M = 16   # rows of i
    BLOCK_N = 16   # cols of j
    BLOCK_K = 32   # reduction tile
    BLOCK_D = 16   # hidden‑channel tile

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B)

    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_tmp,
        B, N, C,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 6) Remaining post‑processing in PyTorch
    # -----------------------------------------------------------------
    out = out_tmp.to(torch.float32)   # back to FP32 for norms/linear

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
