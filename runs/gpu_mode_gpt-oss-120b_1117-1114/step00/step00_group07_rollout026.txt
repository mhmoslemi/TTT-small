```python
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, M, N, K,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    # offsets for rows i
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # K loop
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load left slice: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + batch_id * stride_l_batch + offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load right slice: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + batch_id * stride_r_batch + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_n
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(left, right)  # left (M,K), right (K,N) -> (M,N)

    # Write back
    out_ptrs = out_ptr + batch_id * stride_o_batch + offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_n
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH, M, N, K,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets within M,N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        mask_k = offs_k < K

        # load left tile (M x K)
        left_ptrs = left_ptr + batch_id * stride_l_batch + offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load right tile (K x N)
        right_ptrs = right_ptr + batch_id * stride_r_batch + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_n
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product and accumulate
        # tl.dot returns same dtype as its inputs (here half). Cast to fp32 before accumulation.
        prod = tl.dot(left, right)
        acc += tl.cast(prod, tl.float32)

    # write out
    out_tile = acc.to(tl.float16)  # store as half
    out_ptrs = out_ptr + batch_id * stride_o_batch + offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_n
    tl.store(out_ptrs, out_tile, mask=mask_m[:, None] & mask_n[None, :])
```

```
def custom_kernel(data):
    """
    Triton-accelerated outgoing triangular multiplicative update (TriMul).
    The heavy O(N^3) pairwise reduction is performed by a custom
    batched matrix‑multiply kernel written in Triton. All other
    operations (LayerNorm, linear projections, gates) are executed
    in PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # layer norm
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # linear projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # mask handling
    if mask is None:
        mask = torch.ones_like(input_tensor[...,0], dtype=x.dtype, device=device)
    mask = mask.unsqueeze(-1)  # shape (B,N,N,1)
    left = left * mask
    right = right * mask

    # gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # cast to half for kernel
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # reshape for batched matmul: (B, H, N, N) -> (B*H, N, N)
    B, N, _, _ = left_f16.shape
    left_perm = left_f16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_f16.permute(0, 3, 2, 1).contiguous()
    batch = B * hidden_dim

    left_mat = left_perm.view(batch, N, N)
    right_mat = right_perm.view(batch, N, N)

    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # stride calculations (int64)
    stride_l_batch = left_mat.stride(0)
    stride_l_m = left_mat.stride(1)
    stride_l_k = left_mat.stride(2)
    stride_r_batch = right_mat.stride(0)
    stride_r_k = right_mat.stride(1)
    stride_r_n = right_mat.stride(2)
    stride_o_batch = out_mat.stride(0)
    stride_o_m = out_mat.stride(1)
    stride_o_n = out_mat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (batch,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    tri_mul_kernel[grid](
        left_mat, right_mat, out_mat,
        batch, N, N, N,
        stride_l_batch, stride_l_m, stride_l_k,
        stride_r_batch, stride_r_k, stride_r_n,
        stride_o_batch, stride_o_m, stride_o_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # final norm and linear
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```

"""
TriMul (outgoing) – Triton‑accelerated implementation.

The computational bottleneck of the outgoing triangle multiplicative update is the
3‑D reduction

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

for each batch * hidden dimension (h).  By permuting the tensors we can view it
as a batched matrix multiplication:

    left  : (B, H, N, N)   → (B*H, N, N)   with indices (i, k)
    right : (B, H, N, N)   → (B*H, N, N)   with indices (k, j)

Then  out = left @ right  (standard GEMM) for every (B*H) matrix pair.
The kernel below performs this batched GEMM in Triton, using 16‑bit storage
and 32‑bit accumulation, while all surrounding LayerNorms, linear layers and
gates are executed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,          # float16*  [batch, M, K]
    right_ptr,         # float16*  [batch, K, N]
    out_ptr,           # float16*  [batch, M, N]
    BATCH: tl.constexpr,
    M: tl.constexpr,
    N: tl.constexpr,
    K: tl.constexpr,
    # strides for left
    stride_l_batch, stride_l_m, stride_l_k,
    # strides for right
    stride_r_batch, stride_r_k, stride_r_n,
    # strides for out
    stride_o_batch, stride_o_m, stride_o_n,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)          # which (B*H) matrix we are processing
    pid_m = tl.program_id(1)             # tile row
    pid_n = tl.program_id(2)             # tile column

    # -------------------------------------------------------------
    # Offsets inside the M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Loop over K dimension in blocks of size BLOCK_K
    num_blocks_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_blocks_k):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + batch_id * stride_l_batch \
                    + offs_m[:, None] * stride_l_m \
                    + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + batch_id * stride_r_batch \
                     + offs_k[:, None] * stride_r_k \
                     + offs_n[None, :] * stride_r_n
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)

        # dot product (half precision) -> cast to fp32 before accumulation
        prod = tl.dot(left, right)                 # fp16 result
        acc += tl.cast(prod, tl.float32)

    # -------------------------------------------------------------
    # Store result back as float16
    out_tile = acc.to(tl.float16)
    out_ptrs = out_ptr + batch_id * stride_o_batch \
               + offs_m[:, None] * stride_o_m \
               + offs_n[None, :] * stride_o_n
    tl.store(out_ptrs,
             out_tile,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm over the last dimension ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])          # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if mask is None:
        mask = torch.ones_like(input_tensor[..., 0], device=device, dtype=x.dtype)
    mask = mask.unsqueeze(-1)                               # (B,N,N,1)
    left = left * mask
    right = right * mask

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Convert to half‑precision and reshape for batched GEMM ----------
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    B, N, _, _ = left_f16.shape  # B = batch size, N = seq_len

    # (B, N, N, H) -> (B, H, N, N) and make contiguous
    left_perm = left_f16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_f16.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J) with K==J==N

    batch = B * hidden_dim                       # total number of independent GEMMs
    left_mat = left_perm.view(batch, N, N)       # (batch, M=N, K=N)
    right_mat = right_perm.view(batch, N, N)     # (batch, K=N, N)

    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # ---------- Strides (in element units) ----------
    stride_l_batch = left_mat.stride(0)
    stride_l_m = left_mat.stride(1)
    stride_l_k = left_mat.stride(2)

    stride_r_batch = right_mat.stride(0)
    stride_r_k = right_mat.stride(1)
    stride_r_n = right_mat.stride(2)

    stride_o_batch = out_mat.stride(0)
    stride_o_m = out_mat.stride(1)
    stride_o_n = out_mat.stride(2)

    # ---------- Kernel launch ----------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch, N, N, N,
        stride_l_batch, stride_l_m, stride_l_k,
        stride_r_batch, stride_r_k, stride_r_n,
        stride_o_batch, stride_o_m, stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ---------- Reshape back to (B, N, N, H) ----------
    out = out_mat.view(B, hidden_dim, N, N)            # (B, H, N, N)
    out = out.permute(0, 2, 3, 1).contiguous()        # (B, N, N, H)
    out = out.to(torch.float32)

    # ---------- Final LayerNorm, gating and projection ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
