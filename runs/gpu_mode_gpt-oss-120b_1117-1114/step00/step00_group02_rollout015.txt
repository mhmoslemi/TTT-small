@triton.jit
def mask_gate_kernel(left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
                    mask_ptr,
                    B, N, H, stride_left_b, stride_left_i, stride_left_j, stride_left_h,
                    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
                    stride_mask_b, stride_mask_i, stride_mask_j,
                    USE_MASK: tl.constexpr, BLOCK_H: tl.constexpr):

    # program IDs
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    # compute offset base
    off_left = b * stride_left_b + i * stride_left_i + j * stride_left_j
    off_gate = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j
    if USE_MASK:
        mask_val = tl.load(mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j)
        mask_f = tl.where(mask_val > 0, 1.0, 0.0)  # assuming mask in float or int
    else:
        mask_f = 1.0

    # loop over hidden dimension blocks
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    for hb in range(num_h_blocks):
        d = tl.arange(0, BLOCK_H) + hb * BLOCK_H
        mask_h = d < H

        # left and right loads
        left_ptrs = left_ptr + off_left + d * stride_left_h
        left_vals = tl.load(left_ptrs, mask=mask_h, other=0.0)

        left_gate_ptrs = left_gate_ptr + off_gate + d * stride_gate_h
        left_gate_vals = tl.load(left_gate_ptrs, mask=mask_h, other=0.0)

        # apply mask and gating
        left_new = left_vals * left_gate_vals * mask_f
        tl.store(left_ptrs, left_new, mask=mask_h)

        # right side
        right_ptrs = right_ptr + off_left + d * stride_left_h  # right has same strides as left

        right_new = right_vals * right_gate_vals * mask_f

"""
TriMul outgoing implementation with a fused Triton kernel.

The kernel fuses masking and gating for the left/right projected tensors.
After the projection and sigmoid gating are computed in PyTorch, the kernel
applies:
    left  = left_proj  * left_gate  * mask
    right = right_proj * right_gate * mask
where mask has shape (B, N, N) and is broadcasted over the hidden dimension.
The core triangular multiplication is then performed as a batched GEMM:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
which is equivalent to a batch of matrix multiplications per hidden channel
and is efficiently computed with torch.bmm (cuBLAS).  Finally a LayerNorm,
output gating, and a linear projection produce the result of shape
(B, N, N, dim).

Only the mask/gate fusion is realized in Triton, satisfying the requirement
that part of the computation runs in a custom kernel while keeping the heavy
O(N³) part on cuBLAS for speed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def mask_gate_kernel(
    left_ptr,            # *float32  left tensor (B,N,N,H)
    right_ptr,           # *float32  right tensor (B,N,N,H)
    left_gate_ptr,       # *float32  left gate tensor (B,N,N,H)
    right_gate_ptr,      # *float32  right gate tensor (B,N,N,H)
    mask_ptr,            # *float32  mask tensor (B,N,N)  (may be unused)
    B, N, H,             # int32     batch, seq_len, hidden_dim
    # strides for left
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    # strides for right
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    # strides for left_gate
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    # strides for right_gate
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    # strides for mask
    stride_mask_b, stride_mask_i, stride_mask_j,
    # compile‑time constants
    USE_MASK: tl.constexpr,         # 0 or 1
    BLOCK_H: tl.constexpr,          # block size in hidden dimension
):
    # program IDs identify a single (b, i, j) element
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    # Base offsets for the (b,i,j) coordinate in each tensor
    left_base   = b * stride_left_b   + i * stride_left_i   + j * stride_left_j
    right_base  = b * stride_right_b  + i * stride_right_i  + j * stride_right_j
    lgate_base  = b * stride_lgate_b  + i * stride_lgate_i  + j * stride_lgate_j
    rgate_base  = b * stride_rgate_b  + i * stride_rgate_i  + j * stride_rgate_j

    # Load mask value (scalar) if required
    if USE_MASK:
        mask_val = tl.load(mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j)
        mask_f = tl.where(mask_val != 0, 1.0, 0.0)   # broadcast scalar
    else:
        mask_f = 1.0

    # Loop over hidden dimension in blocks of size BLOCK_H
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    for hb in range(num_h_blocks):
        d = tl.arange(0, BLOCK_H) + hb * BLOCK_H
        d_mask = d < H

        # ----- left side -----
        left_ptrs   = left_ptr   + left_base   + d * stride_left_h
        left_val    = tl.load(left_ptrs, mask=d_mask, other=0.0)

        lgate_ptrs  = left_gate_ptr + lgate_base + d * stride_lgate_h
        lgate_val   = tl.load(lgate_ptrs, mask=d_mask, other=0.0)

        left_new = left_val * lgate_val * mask_f
        tl.store(left_ptrs, left_new, mask=d_mask)

        # ----- right side -----
        right_ptrs  = right_ptr  + right_base  + d * stride_right_h
        right_val   = tl.load(right_ptrs, mask=d_mask, other=0.0)

        rgate_ptrs  = right_gate_ptr + rgate_base + d * stride_rgate_h
        rgate_val   = tl.load(rgate_ptrs, mask=d_mask, other=0.0)

        right_new = right_val * rgate_val * mask_f
        tl.store(right_ptrs, right_new, mask=d_mask)


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated TriMul (outgoing) forward pass.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim] float32 CUDA tensor
            - mask_tensor : [B, N, N]    float32 (or bool) CUDA tensor
            - weights_dict: contains all model weights as tensors
            - config_dict : contains 'dim', 'hidden_dim', and 'nomask' flag
    Returns:
        output tensor of shape [B, N, N, dim] (float32, on the same device)
    """
    # Unpack
    input_tensor, mask_tensor, weights, config = data

    # -------------------- config & dimensions --------------------
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------- fetch weights --------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # Ensure dtype consistency (use float32 for all ops)
    dtype = input_tensor.dtype
    device = input_tensor.device

    # -------------------- LayerNorm on input --------------------
    x = F.layer_norm(input_tensor, normalized_shape=(dim,),
                     weight=norm_weight.to(dtype), bias=norm_bias.to(dtype))

    # -------------------- Linear projections (no bias) --------------------
    # Shape: (B, N, N, hidden_dim)
    left_proj = F.linear(x, left_proj_weight.to(dtype))
    right_proj = F.linear(x, right_proj_weight.to(dtype))

    # -------------------- Gating linear layers (sigmoid) --------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight.to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight.to(dtype)))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight.to(dtype)))  # used later

    # -------------------- Prepare mask --------------------
    # Convert mask to float32 (1.0 for true, 0.0 for false)
    if nomask:
        mask = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask = mask_tensor.to(dtype)
    mask = mask.contiguous()

    # Make all tensors contiguous for Triton pointer arithmetic
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # -------------------- Triton kernel: apply mask & gating --------------------
    BLOCK_H = 64  # fits well in a warp; hidden_dim ≤ 768
    grid = (B, N, N)   # program_id(0)=batch, (1)=i, (2)=j

    mask_gate_kernel[grid](
        left_ptr=left_proj.data_ptr(),
        right_ptr=right_proj.data_ptr(),
        left_gate_ptr=left_gate.data_ptr(),
        right_gate_ptr=right_gate.data_ptr(),
        mask_ptr=mask.data_ptr(),
        B=B,
        N=N,
        H=hidden_dim,
        # strides for left_proj
        stride_left_b=left_proj.stride(0),
        stride_left_i=left_proj.stride(1),
        stride_left_j=left_proj.stride(2),
        stride_left_h=left_proj.stride(3),
        # strides for right_proj
        stride_right_b=right_proj.stride(0),
        stride_right_i=right_proj.stride(1),
        stride_right_j=right_proj.stride(2),
        stride_right_h=right_proj.stride(3),
        # strides for left_gate
        stride_lgate_b=left_gate.stride(0),
        stride_lgate_i=left_gate.stride(1),
        stride_lgate_j=left_gate.stride(2),
        stride_lgate_h=left_gate.stride(3),
        # strides for right_gate
        stride_rgate_b=right_gate.stride(0),
        stride_rgate_i=right_gate.stride(1),
        stride_rgate_j=right_gate.stride(2),
        stride_rgate_h=right_gate.stride(3),
        # strides for mask
        stride_mask_b=mask.stride(0),
        stride_mask_i=mask.stride(1),
        stride_mask_j=mask.stride(2),
        USE_MASK=0 if nomask else 1,
        BLOCK_H=BLOCK_H,
    )
    # At this point left_proj and right_proj contain the masked & gated tensors.

    # -------------------- Triangular multiplication via batched GEMM --------------------
    # Rearrange to (B, H, N, N) for batched matmul
    left_perm = left_proj.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_proj.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B_, H_, N_, _ = left_perm.shape
    left_flat = left_perm.view(B_ * H_, N_, N_)          # (B*H, N, N)
    right_flat = right_perm.view(B_ * H_, N_, N_)        # (B*H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)
    out = out_flat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # -------------------- Output layer norm, gating, and final projection --------------------
    out = F.layer_norm(out, normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight.to(dtype),
                       bias=to_out_norm_bias.to(dtype))

    out = out * out_gate   # elementwise gating

    # Final linear projection back to original dimension
    out = F.linear(out, to_out_weight.to(dtype))  # (B, N, N, dim)

    return out.to(torch.float32)  # ensure float32 output
