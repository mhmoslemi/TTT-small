@triton.jit
def batch_matmul_kernel(A_ptr, B_ptr, C_ptr,
                        N, H_total,
                        stride_Abatch, stride_Ai, stride_Ak,
                        stride_Bbatch, stride_Bk, stride_Bj,
                        stride_Cbatch, stride_Ci, stride_Cj,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets for M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out-of-bounds
    m_mask = offs_m < N
    n_mask = offs_n < N

    # compute pointers for this batch
    a_batch_ptr = A_ptr + batch_idx * stride_Abatch
    b_batch_ptr = B_ptr + batch_idx * stride_Bbatch
    c_batch_ptr = C_ptr + batch_idx * stride_Cbatch

    # load A block
    a_ptrs = a_batch_ptr + offs_m[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
    a = tl.load(a_ptrs, mask=m_mask[:, None] & (offs_k < N), other=0.0)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute real offset for current block
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = offs_k < N

        # compute sub-block pointers
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bj

        a = tl.load(a_ptrs, mask=m_mask[:, None] & k_mask[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=k_mask[:, None] & n_mask[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_Ci + offs_n[None, :] * stride_Cj
    tl.store(c_ptrs, acc, mask=m_mask[:, None] & n_mask[None, :])

"""Triton-accelerated forward pass for the TriMul operator (outgoing version).

The kernel computes, for every batch and hidden channel h,
    out_{i,j} = Σ_k left_{i,k} * right_{j,k}
where `right` is transposed so that the computation reduces to a standard
matrix multiplication left @ right^T.  The surrounding LayerNorm, linear
projections, gating and final projection are performed with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batch_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    N,
    stride_Ab, stride_Ai, stride_Ak,
    stride_Bb, stride_Bi, stride_Bk,
    stride_Cb, stride_Ci, stride_Cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ B for many independent batches.
    A, B, C are 3‑D tensors of shape (batch, N, N) stored in row‑major order.
    The multiplication is standard matrix multiply (i.e. over the shared K dimension).
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets within the N×N matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = A_ptr + batch_id * stride_Ab
    b_base = B_ptr + batch_id * stride_Bb
    c_base = C_ptr + batch_id * stride_Cb

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A: (M, K)
        a_ptrs = a_base + offs_m[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B: (K, N)
        b_ptrs = b_base + offs_k[:, None] * stride_Bi + offs_n[None, :] * stride_Bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_base + offs_m[:, None] * stride_Ci + offs_n[None, :] * stride_Cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """Forward pass of the outgoing TriMul operator with a Triton kernel for the
    O(N³) pairwise multiplication.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (will be broadcast)
            - weights: dict of torch.Tensors with the model parameters
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    # configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, C = input_tensor.shape
    assert C == dim, "input last dim must match config dim"

    # ------------------------------------------------------------------
    # 1) LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias   = weights["norm.bias"].to(dtype)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(dtype)
    right_proj_w = weights["right_proj.weight"].to(dtype)

    left  = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3) Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).type_as(left)  # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(dtype)
    right_gate_w = weights["right_gate.weight"].to(dtype)
    out_gate_w   = weights["out_gate.weight"].to(dtype)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Batched matrix multiplication via Triton:
    #    out_{b,h,i,j} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    #    (equivalently: for each (b,h) compute L @ R^T)
    # ------------------------------------------------------------------
    # Rearrange tensors to (B, hidden_dim, N, N) and make them contiguous.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()       # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()      # [B, H, N, N] (transposed)

    # Collapse the first two dimensions so that each (b, h) pair is an
    # independent batch for the kernel.
    left_flat  = left_perm.view(-1, N, N)
    right_flat = right_perm.view(-1, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)

    # Strides (in elements, not bytes)
    stride_Ab = left_flat.stride(0)
    stride_Ai = left_flat.stride(1)
    stride_Ak = left_flat.stride(2)

    stride_Bb = right_flat.stride(0)
    stride_Bi = right_flat.stride(1)   # row stride (k)
    stride_Bk = right_flat.stride(2)   # column stride (j)

    stride_Cb = out_flat.stride(0)
    stride_Ci = out_flat.stride(1)
    stride_Cj = out_flat.stride(2)

    # Tiling parameters – chosen to fit B200's shared‑memory budget
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    batch_total = left_flat.shape[0]  # B * hidden_dim
    grid = (batch_total,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    batch_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_Ab, stride_Ai, stride_Ak,
        stride_Bb, stride_Bi, stride_Bk,
        stride_Cb, stride_Ci, stride_Cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output layernorm, gating, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype)
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(dtype)
    out = F.linear(out, to_out_w)  # [B, N, N, dim]

    return out

"""Triton‑accelerated forward pass of the outgoing TriMul operator.

The heavy O(N³) part
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
is implemented as a batched matrix multiplication:
    left(b, h) @ right(b, h).T   ->   (B·H, N, N) @ (B·H, N, N)ᵀ.
All surrounding LayerNorms, linear projections and gating are kept in
PyTorch for simplicity.  The final tensor has shape [B, N, N, dim] and
dtype float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batch_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    N,
    stride_Ab, stride_Ai, stride_Ak,
    stride_Bb, stride_Bi, stride_Bj,
    stride_Cb, stride_Ci, stride_Cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C = A @ B  (A,B,C : (batch, N, N)).

    A and B are stored in row‑major order.
    The kernel processes a single (batch, block_i, block_j) tile.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)   # tile row index
    pid_n = tl.program_id(2)   # tile col index

    # offsets inside the N×N matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for the current batch
    a_base = A_ptr + batch_id * stride_Ab
    b_base = B_ptr + batch_id * stride_Bb
    c_base = C_ptr + batch_id * stride_Cb

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile: (M, K)
        a_ptrs = a_base + offs_m[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile: (K, N)
        b_ptrs = b_base + offs_k[:, None] * stride_Bi + offs_n[None, :] * stride_Bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (M,N)

    # write back
    c_ptrs = c_base + offs_m[:, None] * stride_Ci + offs_n[None, :] * stride_Cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    """Forward pass of the TriMul (outgoing) operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C]
        - mask         : torch.Tensor, shape [B, N, N] (may be all‑ones)
        - weights      : dict of torch.Tensor containing model parameters
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """

    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, C = input_tensor.shape
    assert C == dim, "input channel dimension must equal config['dim']"

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(dtype)
    norm_b = weights["norm.bias"].to(dtype)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(dtype)
    right_proj_w = weights["right_proj.weight"].to(dtype)

    left  = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3) Apply mask (element‑wise broadcast)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).type_as(left)   # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(dtype)
    right_gate_w = weights["right_gate.weight"].to(dtype)
    out_gate_w   = weights["out_gate.weight"].to(dtype)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Batched matmul via Triton (left @ right^T)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) and make contiguous so that we can view
    # each (b, h) pair as an independent batch.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (transposed)

    left_flat  = left_perm.view(-1, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(-1, N, N)  # (B*H, N, N)

    out_flat = torch.empty_like(left_flat)

    # Strides expressed in element counts (not bytes)
    stride_Ab = left_flat.stride(0)
    stride_Ai = left_flat.stride(1)
    stride_Ak = left_flat.stride(2)

    stride_Bb = right_flat.stride(0)
    stride_Bi = right_flat.stride(1)   # row (k) stride
    stride_Bj = right_flat.stride(2)   # col (j) stride

    stride_Cb = out_flat.stride(0)
    stride_Ci = out_flat.stride(1)
    stride_Cj = out_flat.stride(2)

    # tiling parameters – empirically good for the B200
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    batch_total = left_flat.shape[0]          # B * hidden_dim
    grid = (
        batch_total,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batch_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_Ab, stride_Ai, stride_Ak,
        stride_Bb, stride_Bi, stride_Bj,
        stride_Cb, stride_Ci, stride_Cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore the original layout: [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(dtype)   # shape (dim, hidden_dim)
    out = F.linear(out, to_out_w)                   # [B, N, N, dim]

    return out
