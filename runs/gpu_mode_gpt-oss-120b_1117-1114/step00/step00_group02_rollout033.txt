def tri_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_a_batch, stride_b_batch, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(2)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(0)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    a_batch_ptr = a_ptr + pid_b * stride_a_batch
    b_batch_ptr = b_ptr + pid_b * stride_b_batch
    c_batch_ptr = c_ptr + pid_b * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        a_mask = (offs_m[:, None] < M) & (k_offset[None, :] < K)

        b = tl.load(b_batch_ptr + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn,

    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape (bs,N,N,hidden_dim)

    else:
        mask_f = mask.to(x.dtype)
    mask_f = mask_f.unsqueeze(-1)  # (bs,N,N,1)
    left = left * mask_f
    right = right * mask_f

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))

    left_h = left.to(torch.float16)

    left_h = left_h.permute(0,3,1,2).contiguous().view(-1, N, N) # (B*H, N, N)

    right_h_T = right_h.transpose(1,2).contiguous()  # (B*H, N, N) as (K,N)

    out = torch.empty((total, N, N), dtype=torch.float32, device=device)

    stride_ak = left_h.stride(2)

"""
TriMul (outgoing) implementation using Triton for the O(N³) matrix multiplication.
The forward pass consists of:
1. LayerNorm over the channel dimension.
2. Linear projections and gating (all in PyTorch).
3. Masking applied before gating.
4. Batched matrix multiplication per hidden dimension:
   out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
   This is computed as a set of GEMMs (left @ rightᵀ) with a custom Triton kernel.
5. LayerNorm on the hidden dimension, gating, and final linear projection.
All tensors are kept in float32 (except optional half‑precision parts) to match the
reference PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) where each batch item corresponds
# to one (hidden_dim) slice of the TriMul tensor.
# ----------------------------------------------------------------------
@triton.jit
def tri_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions
    stride_am, stride_ak,              # A strides (batch, row, col)
    stride_bk, stride_bn,              # B strides (batch, row, col) ; B is already transposed (K x N)
    stride_cm, stride_cn,              # C strides (batch, row, col)
    stride_a_batch, stride_b_batch, stride_c_batch,  # stride between batch matrices
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)           # batch index
    pid_m = tl.program_id(1)           # block row index
    pid_n = tl.program_id(0)           # block column index

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the beginning of the current batch matrix
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offs = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile {M x K}
        a_mask = (offs_m[:, None] < M) & (k_offs[None, :] < K)
        a = tl.load(
            a_batch + offs_m[:, None] * stride_am + k_offs[None, :] * stride_ak,
            mask=a_mask,
            other=0.0,
        )

        # Load B tile {K x N}
        b_mask = (k_offs[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(
            b_batch + k_offs[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=b_mask,
            other=0.0,
        )

        # Ensure FP32 computation
        a_f32 = a.to(tl.float32)
        b_f32 = b.to(tl.float32)

        acc += tl.dot(a_f32, b_f32)

    # Write back the result tile
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(
        c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn,
        acc,
        mask=c_mask,
    )

# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.
    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 2. Linear projections
    # --------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 3. Mask (if present)
    # --------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        mask_f = torch.ones((bs, N, N), dtype=dtype, device=device)
    else:
        mask_f = mask.to(dtype)
    mask_f = mask_f.unsqueeze(-1)          # [B, N, N, 1]
    left = left * mask_f
    right = right * mask_f

    # --------------------------------------------------------------
    # 4. Gates
    # --------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Batched GEMM (left @ rightᵀ) using Triton
    # --------------------------------------------------------------
    # Reshape to (B*H, N, N) and transpose right for Bᵀ
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)          # [B*H, N, N]
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N)        # [B*H, N, N]
    right_mat_T = right_mat.transpose(1, 2).contiguous()                    # [B*H, N, N] (K x N)

    total = left_mat.shape[0]  # B * hidden_dim

    # Output buffer
    out_mat = torch.empty((total, N, N), dtype=torch.float32, device=device)

    # Strides
    stride_a_batch = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_b_batch = right_mat_T.stride(0)
    stride_bk = right_mat_T.stride(1)   # K stride
    stride_bn = right_mat_T.stride(2)   # N stride

    stride_c_batch = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    # Block sizes (tuned for fp32 GEMM)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_N),   # columns
        triton.cdiv(N, BLOCK_M),   # rows
        total,                     # batch dimension
    )

    tri_matmul_kernel[grid](
        left_mat,
        right_mat_T,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_a_batch=stride_a_batch,
        stride_b_batch=stride_b_batch,
        stride_c_batch=stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear projection
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    out = out * out_gate                     # element‑wise gating

    to_out_weight = weights["to_out.weight"]
    output = F.linear(out, to_out_weight)    # final projection to dim

    return output
