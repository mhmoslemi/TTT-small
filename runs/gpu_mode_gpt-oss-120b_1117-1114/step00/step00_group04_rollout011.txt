import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,i,j,h] = sum_{k=0}^{N-1} left[b,i,k,h] * right[b,j,k,h]
    using a tiled GEMM where the reduction dimension k is tiled by BLOCK_K.
    Batch (B) and hidden (H) are fused into the third program id.
    """
    pid_z = tl.program_id(2)  # combines batch and hidden dimension
    pid_i = tl.program_id(0)  # i tile
    pid_j = tl.program_id(1)  # j tile

    # decode batch and hidden index from pid_z
    b = pid_z // H
    h = pid_z % H

    # starting indices for this tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # range vectors
    offs_i = i_start + tl.arange(0, BLOCK_M)[:, None]       # (BLOCK_M, 1)
    offs_j = j_start + tl.arange(0, BLOCK_N)[None, :]       # (1, BLOCK_N)

    # boundary masks for i and j dimensions
    mask_i = i_start + tl.arange(0, BLOCK_M) < N
    mask_j = j_start + tl.arange(0, BLOCK_N) < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        k_start = kb * BLOCK_K
        # masks for k
        mask_k = k_start + tl.arange(0, BLOCK_K) < N

        # offsets for left tile (M x K)
        offs_k_left = k_start + tl.arange(0, BLOCK_K)[None, :]   # (1, BLOCK_K)
        left_ptrs = left_ptr + \
                    b * stride_left_b + \
                    offs_i * stride_left_i + \
                    offs_k_left * stride_left_k + \
                    h * stride_left_h
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # offsets for right tile transposed (K x N)
        # we want a tile shaped (BLOCK_K, BLOCK_N) where first dim = k, second = j
        offs_k_right = k_start + tl.arange(0, BLOCK_K)[:, None]   # (BLOCK_K, 1)
        right_ptrs = right_ptr + \
                     b * stride_right_b + \
                     offs_k_right * stride_right_k + \
                     offs_j * stride_right_i + \
                     h * stride_right_h
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # GEMM: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        prod = tl.dot(left, right)          # dtype same as left/right (fp16)
        acc += tl.cast(prod, tl.float32)

    # write result
    out_ptrs = out_ptr + \
               b * stride_out_b + \
               offs_i * stride_out_i + \
               offs_j * stride_out_j + \
               h * stride_out_h
    store_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, tl.cast(acc, tl.float16), mask=store_mask)

    """

    norm_weight = weights["norm.weight"]

    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)

    if not nomask:

        left = left * mask_
        right = right * mask_

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # stay fp32 for final gating

    left = left * left_gate
    right = right * right_gate

    # ---- Cast to half for the reduction -----------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape  # H == hidden_dim

    # Prepare output tensor for the kernel
    out_h = torch.empty_like(left_h, dtype=torch.float16, device=left_h.device)

    # Compute strides (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_h.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_h.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_h.stride()

    # ----------------------------------------------------------------------
    # Triton launch parameters
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # i dimension
        triton.cdiv(N, BLOCK_N),   # j dimension
        B * H                      # fused batch+hidden dimension
    )

    trimul_einsum_kernel[grid](
        left_h, right_h, out_h,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to FP32 for the remaining layers
    out = out_h.to(torch.float32)

    # ---- Output layer norm ------------------------------------------------
    out_norm = F.layer_norm(out, (hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias)

    # ---- Apply final gating ------------------------------------------------
    out_norm = out_norm * out_gate

    # ---- Final linear projection back to original dim --------------------
    out_final = F.linear(out_norm, to_out_weight)

    return out_final

 Triton‑accelerated forward pass of the "outgoing" TriMul operator.

 The heavy 4‑D einsum
     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
 is implemented as a tiled GEMM in the trio of dimensions (i, j, h)
 while the batch and hidden dimensions are fused into a single program‑id.
 All surrounding LayerNorms, linear projections and gating are performed
 in PyTorch for simplicity.

import triton
import triton.language as tl

def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using
    a tiled matrix multiplication (M×K)·(K×N) → (M×N).
    The batch (B) and hidden (H) dimensions are fused into the third
    program‑id (pid_z = b*H + h).
    """
    pid_z = tl.program_id(2)          # fused batch & hidden
    pid_i = tl.program_id(0)          # tile along i (rows)
    pid_j = tl.program_id(1)          # tile along j (cols)

    # decode batch index b and hidden channel h
    b = pid_z // H
    h = pid_z % H

    # start offsets for this tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # range vectors
    offs_i = i_start + tl.arange(0, BLOCK_M)[:, None]   # (M,1)
    offs_j = j_start + tl.arange(0, BLOCK_N)[None, :]   # (1,N)

    # boundary masks for i and j
    mask_i = i_start + tl.arange(0, BLOCK_M) < N
    mask_j = j_start + tl.arange(0, BLOCK_N) < N

    # accumulator in FP32 for numerical safety
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction dimension k
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        k_start = kb * BLOCK_K
        mask_k = k_start + tl.arange(0, BLOCK_K) < N

        # ---- load left tile (M x K) ------------------------------------
        offs_k_left = k_start + tl.arange(0, BLOCK_K)[None, :]   # (1,K)
        left_ptrs = left_ptr + \
                    b * stride_left_b + \
                    offs_i * stride_left_i + \
                    offs_k_left * stride_left_k + \
                    h * stride_left_h
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # ---- load right tile transposed (K x N) -----------------------
        # tile layout: rows = k, cols = j
        offs_k_right = k_start + tl.arange(0, BLOCK_K)[:, None]   # (K,1)
        right_ptrs = right_ptr + \
                     b * stride_right_b + \
                     offs_k_right * stride_right_k + \
                     offs_j * stride_right_i + \
                     h * stride_right_h
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # ---- GEMM accumulation ----------------------------------------
        prod = tl.dot(left, right)                     # fp16 result
        acc += tl.cast(prod, tl.float32)

    # ---- write the result ---------------------------------------------
    out_ptrs = out_ptr + \
               b * stride_out_b + \
               offs_i * stride_out_i + \
               offs_j * stride_out_j + \
               h * stride_out_h
    store_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, tl.cast(acc, tl.float16), mask=store_mask)


def custom_kernel(data):
    """
    Forward pass of the TriMul module.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] (float32)
            - mask:         [B, N, N] (float32 or bool)
            - weights: dict with all linear / layer‑norm parameters
            - config:  dict containing "dim", "hidden_dim", "nomask" (optional)

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------------
    # unpack weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections to hidden dimension
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)

    # Optional mask (broadcasted on the hidden dimension)
    if not nomask:
        mask_ = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # Gates (sigmoid), cast to half‑precision for speed
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # keep fp32 for final gating

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------
    # Cast to half‑precision for the massive reduction
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape          # H == hidden_dim

    # Allocate output buffer for the kernel
    out_h = torch.empty_like(left_h, dtype=torch.float16, device=left_h.device)

    # Strides in element counts
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_h.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_h.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_h.stride()

    # Kernel launch configuration
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        B * H                      # fused batch × hidden
    )

    trimul_einsum_kernel[grid](
        left_h, right_h, out_h,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------------
    # Continue in FP32 for the remaining lightweight operations
    out = out_h.to(torch.float32)

    # Output LayerNorm over hidden dimension
    out_norm = F.layer_norm(out, (hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias)

    # Apply final gating
    out_norm = out_norm * out_gate

    # Final linear projection back to original dimension
    out_final = F.linear(out_norm, to_out_weight)

    return out_final
