"""
Optimized TriMul (outgoing) forward pass.

The kernel fuses the optional pairwise mask with the elementwise gating
operations.  After LayerNorm and linear projections are performed with PyTorch
(leveraging cuBLAS), the Triton kernel computes

    left  = (left_proj * left_gate) * mask
    right = (right_proj * right_gate) * mask

where the mask is broadcast over the hidden dimension.  The core bilinear
operation is then carried out as a batched GEMM:

    out_{b,i,j,:} = sum_k left_{b,i,k,:} * right_{b,j,k,:}
                 = (left_{b,:, :, :}) @ (right_{b,:, :, :})^T

implemented via `torch.bmm`.  Finally a second LayerNorm, output gating and a
linear projection produce the result tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,      # pointers
    total_elements, hidden_dim,                 # runtime ints
    BLOCK: tl.constexpr,                        # compile‑time const
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    # load projection and gate values
    proj = tl.load(proj_ptr + offs, mask=m, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=m, other=0.0)

    # mask is flattened [B, N, N]; each hidden entry repeats the same mask
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=0.0)

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out, mask=m)

def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]
            - mask         : torch.Tensor [B, N, N] (float, 0/1)
            - weights      : dict of torch.Tensor with the model parameters
            - config       : dict containing at least
                {"dim": int, "hidden_dim": int, "nomask": bool}
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    use_mask = not config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load weights (all assumed to be float32; cast if necessary)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(dtype=torch.float32, device=device)
    norm_bias   = weights["norm.bias"].to(dtype=torch.float32, device=device)

    left_proj_weight  = weights["left_proj.weight"].to(dtype=torch.float32, device=device)
    right_proj_weight = weights["right_proj.weight"].to(dtype=torch.float32, device=device)

    left_gate_weight  = weights["left_gate.weight"].to(dtype=torch.float32, device=device)
    right_gate_weight = weights["right_gate.weight"].to(dtype=torch.float32, device=device)
    out_gate_weight   = weights["out_gate.weight"].to(dtype=torch.float32, device=device)

    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype=torch.float32, device=device)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(dtype=torch.float32, device=device)

    to_out_weight = weights["to_out.weight"].to(dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # LayerNorm on input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor.to(torch.float32),
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections + gating (still on GPU, cuBLAS)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_weight)      # [B, N, N, hidden]
    right_proj = F.linear(x, right_proj_weight)

    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    B, N, _, _ = left_proj.shape

    # ------------------------------------------------------------------
    # Optional mask + gate fusion via Triton
    # ------------------------------------------------------------------
    if use_mask:
        mask_f = mask.to(dtype=torch.float32, device=device)
        # flatten tensors for the kernel (contiguous layout: ... hidden is innermost)
        total = left_proj.numel()
        BLOCK = 32768  # multiple of 32, reasonable size for this elementwise kernel
        grid = (triton.cdiv(total, BLOCK),)

        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # flatten pointers
        lp_flat = left_proj.view(-1)
        lg_flat = left_gate.view(-1)
        rp_flat = right_proj.view(-1)
        rg_flat = right_gate.view(-1)
        mask_flat = mask_f.view(-1)
        left_out_flat = left.view(-1)
        right_out_flat = right.view(-1)

        _mask_gate_kernel[grid](
            lp_flat,
            lg_flat,
            mask_flat,
            left_out_flat,
            total,
            hidden_dim,
            BLOCK=BLOCK,
        )
        _mask_gate_kernel[grid](
            rp_flat,
            rg_flat,
            mask_flat,
            right_out_flat,
            total,
            hidden_dim,
            BLOCK=BLOCK,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # Batched matrix multiplication: out_{i,j} = sum_k left_{i,k} * right_{j,k}
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N) for torch.bmm
    left_bh = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_bh = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # compute out = left @ right^T   (B*hidden) independent GEMMs
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))

    # reshape back to [B, N, N, hidden]
    out = out_bh.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # Second LayerNorm, output gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # back to dim

    return out
