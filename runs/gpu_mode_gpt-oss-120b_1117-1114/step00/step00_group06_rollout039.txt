    pid_j = tl.program_id(1)
    pid_b = tl.program_id(2)

    # compute block start indices:
    i_offset = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = i_offset < N
    mask_j = j_offset < N

    # loop over k blocks

        mask_k = k_offset < N

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # config contains dim, hidden_dim, maybe eps, nomask.
    device = input_tensor.device
    dtype = input_tensor.dtype  # assume float32
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask
    nomask = config.get("nomask", False) # maybe not present but we can ignore if not present
    if not nomask:
        mask_exp = mask.to(dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp
    else:
        # ensure mask is all ones to maintain consistency
        pass

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Allocate output buffer
    out = torch.empty((B, N, N, hidden_dim), dtype=x.dtype, device=device)

    # Load strides
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # Kernel parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * hidden_dim)

    # Launch kernel
    tri_mul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Perform to_out_norm and gating

"""
TriMul (outgoing) forward pass accelerated with Triton.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension.
2. Project the normalized tensor to a hidden dimension with two linear layers
   (left_proj, right_proj) and compute three sigmoid gates.
3. Apply the mask (if present) and the per‑position gates to obtain `left`
   and `right` tensors of shape [B, N, N, H] where H = hidden_dim.
4. Compute the cubic contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   using a Triton kernel that treats the hidden dimension as an extra
   program‑id dimension.  Each thread therefore computes a matrix product
   for a single hidden channel.
5. Layer‑norm the intermediate result, apply the output gate and project
   back to the original dimension.

The heavy O(N³) contraction is carried out entirely on the GPU with a
tiled mat‑mul kernel, while the remaining linear / layer‑norm work is
performed with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute for a single hidden channel:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Grid dimensions:
        program_id(0) -> i‑tile
        program_id(1) -> j‑tile
        program_id(2) -> batch * hidden (combined)
    """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    # decode batch index and hidden index from pid_z
    batch = pid_z // H
    h = pid_z % H

    # start offsets for the tiles
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # thread‑local ranges
    i = i_start + tl.arange(0, BLOCK_M)
    j = j_start + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    # bounds masks
    mask_i = i < N
    mask_j = j < N

    # base pointers for this batch / hidden channel
    left_base  = left_ptr  + batch * stride_l_batch  + h * stride_l_h
    right_base = right_ptr + batch * stride_r_batch + h * stride_r_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for tile_k in range(num_k_tiles):
        k_start = tile_k * BLOCK_K
        cur_k = k_start + k
        mask_k = cur_k < N

        # left block: (M, K)
        a_ptrs = left_base + i[:, None] * stride_l_i + cur_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right block: (K, N)
        b_ptrs = right_base + cur_k[:, None] * stride_r_k + j[None, :] * stride_r_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate matmul
        acc += tl.dot(a, b)

    # store the result
    out_ptrs = out_ptr + batch * stride_o_batch + h * stride_o_h \
               + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated TriMul forward.

    Args:
        data: Tuple (input_tensor, mask_tensor, weight_dict, config_dict)
            - input_tensor: [B, N, N, dim]  (float32, on CUDA)
            - mask_tensor:  [B, N, N]       (bool/float, on CUDA)
            - weight_dict: mapping from string names to torch tensors
            - config_dict: contains "dim", "hidden_dim" and optional "nomask"

    Returns:
        output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------
    # 1. Layer‑norm over the feature dimension (dim)
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -------------------------------------------------
    # 2. Linear projections
    # -------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------
    # 3. Optional mask
    # -------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------
    # 4. Gates (sigmoid)
    # -------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5. Cubic contraction via Triton
    # -------------------------------------------------
    out = torch.empty((B, N, N, H), dtype=dtype, device=device)

    # strides for the three tensors
    s_l = left.stride()
    s_r = right.stride()
    s_o = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,                     # combine batch & hidden dim
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_o[0], s_o[1], s_o[2], s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------
    # 6. Post‑processing: layer‑norm, output gate, final linear
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
