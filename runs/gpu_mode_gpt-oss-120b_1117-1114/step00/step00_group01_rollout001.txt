def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # LayerNorm on dim dimension
    # Use torch.nn.functional.layer_norm: input shape [B,N,N,dim]
    # Provide normalized_shape=dim and weight,bias.
    # Normalizing over last dimension.
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Compute linear projections
    # Combine weight matrix shape [hidden_dim, dim]
    left = torch.nn.functional.linear(x, weights['left_proj.weight'])
    right = torch.nn.functional.linear(x, weights['right_proj.weight'])

    # Apply mask if present (determined by config or mask shape)
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.nn.functional.linear(x, weights['left_gate.weight']).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights['right_gate.weight']).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights['out_gate.weight']).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute out via Triton matmul per hidden_dim channel
    out = triton_matmul_per_channel(left, right)  # shape [B,N,N,hidden_dim]

    # to_out_norm: layernorm over hidden_dim
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, weights['to_out.weight'])
    return out
```

@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_dl,
    stride_br, stride_ir, stride_kr, stride_dr,
    stride_bo, stride_io, stride_jo, stride_do,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    pid_n = tl.program_id(1)
    pid_z = tl.program_id(2)  # batch*hidden channel index
    # compute batch & hidden index
    total_bc = B * H
    b = pid_z // H
    h = pid_z % H

    # offset base pointers for this batch and channel
    left_base = left_ptr + b * stride_bl + h * stride_dl
    right_base = right_ptr + b * stride_br + h * stride_dr
    out_base = out_ptr + b * stride_bo + h * stride_do

    # compute offsets for each program block
    # i (row) offset: pid * BLOCK_M
    # j (col) offset: pid_n * BLOCK_N

    i_offset = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for bounds
    i_mask = i_offset < N
    j_mask = j_offset < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # Load left block (M x K)
        left_ptrs = left_base + i_offset[:, None] * stride_il + k_offset[None, :] * stride_kl
        left_block = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        # Load right block (N x K) transposed -> we want right[j,k] so treat as (K x N) matrix for matmul.
        # right matrix shape for matmul: (K,N) with stride_k = stride_kr and stride_n = stride_ir
        right_ptrs = right_base + k_offset[:, None] * stride_kr + j_offset[None, :] * stride_ir
        right_block = tl.load(right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        # Multiply accumulate
        acc += tl.dot(left_block, right_block)

        # Advance? We already compute ptr each iteration.
    # Write back result
    out_ptrs = out_base + i_offset[:, None] * stride_io + j_offset[None, :] * stride_jo
    tl.store(out_ptrs, acc.to(tl.float16) if out_ptr.dtype==tl.float16 else acc, mask=i_mask[:, None] & j_mask[None, :])

"""
Optimized outgoing TriMul implementation.

The heavy N^3 contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with a custom Triton kernel that treats the hidden
dimension `d` as an independent batch.  The remaining operations
(layernorm, linear projections, gating) are done in PyTorch.
The kernel computes a per‑channel matrix multiplication L_d @ R_dᵀ
using a tiled dot‑product accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ------------------------------------------------------------------
# Triton kernel for the per‑channel batched matmul (left @ rightᵀ)
# ------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_dl,
    stride_br, stride_ir, stride_kr, stride_dr,
    stride_bo, stride_io, stride_jo, stride_do,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)               # i‑block
    pid_n = tl.program_id(1)               # j‑block
    pid_z = tl.program_id(2)               # batch * hidden index

    # decode batch and hidden index
    total_h = H
    b = pid_z // total_h
    h = pid_z % total_h

    # base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_bl + h * stride_dl
    right_base = right_ptr + b * stride_br + h * stride_dr
    out_base   = out_ptr   + b * stride_bo + h * stride_do

    # offsets within the N×N matrix
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    i_mask = i < N
    j_mask = j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + i[:, None] * stride_il + k_offset[None, :] * stride_kl
        left = tl.load(left_ptrs,
                       mask=i_mask[:, None] & k_mask[None, :],
                       other=0.0)

        # right block transposed: we need right[j, k] => treat as (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + k_offset[:, None] * stride_kr + j[None, :] * stride_ir
        right = tl.load(right_ptrs,
                        mask=k_mask[:, None] & j_mask[None, :],
                        other=0.0)

        # accumulate dot product
        acc += tl.dot(left, right)

    # write back
    out_ptrs = out_base + i[:, None] * stride_io + j[None, :] * stride_jo
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


def _triton_tri_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched per‑channel matrix multiply using Triton.
    left, right: [B, N, N, H] (float32, contiguous)
    returns out: same shape, dtype float32
    """
    assert left.shape == right.shape, "left and right must have identical shapes"
    B, N, _, H = left.shape

    # ensure contiguous and on CUDA
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # strides in element units
    stride_bl, stride_il, stride_kl, stride_dl = left.stride()
    stride_br, stride_ir, stride_kr, stride_dr = right.stride()
    stride_bo, stride_io, stride_jo, stride_do = out.stride()

    # tile sizes – chosen to give good occupancy for the B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,                     # flattened batch‑hidden dimension
    )
    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_bl, stride_il, stride_kl, stride_dl,
        stride_br, stride_ir, stride_kr, stride_dr,
        stride_bo, stride_io, stride_jo, stride_do,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ------------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator with a Triton‑accelerated
    contraction.

    Args:
        data: (input, mask, weights, config)
            - input: Tensor [B, N, N, dim] (float32)
            - mask:  Tensor [B, N, N] (bool/float) or None
            - weights: dict with all linear/ln parameters
            - config: dict with 'dim', 'hidden_dim', 'nomask' entries

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ----- LayerNorm over input dim -----
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ----- Projections -----
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ----- Optional mask (skip if nomask) -----
    if not config.get("nomask", False) and mask is not None:
        mask_f = mask.unsqueeze(-1).to(dtype)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ----- Gating -----
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----- Core N^3 contraction using Triton -----
    # left/right shape: [B, N, N, hidden_dim]
    out = _triton_tri_mul(left, right)   # same dtype (float32)

    # ----- Output normalization and gating -----
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ----- Final linear -----
    out = F.linear(out, weights["to_out.weight"])

    return out
