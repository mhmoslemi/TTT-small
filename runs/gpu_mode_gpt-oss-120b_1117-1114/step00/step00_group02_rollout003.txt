@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds in M,N
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to start of the batch slice
    a_batch_ptr = A_ptr + pid_batch * stride_ab
    b_batch_ptr = B_ptr + pid_batch * stride_bb
    c_batch_ptr = C_ptr + pid_batch * stride_cb

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute offsets for K block
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile [M,K]
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B tile [K,N]
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot
        acc += tl.dot(a, b)   # shape (BLOCK_M, BLOCK_N)

    # apply mask to store
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])
```

    stride_cm, stride_cn,

"""
TriMul forward pass with a fused batched GEMM implemented in Triton.
The heavy part of the computation is the "outgoing" update:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched matrix multiplication
    left_d @ right_d^T for each hidden dimension d.
A custom Triton kernel performs the batch‑wise GEMM, while the rest
(LayerNorm, linear layers, gates) are done in PyTorch.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ B) where A,B are
# (batch, M, K) and (batch, K, N) respectively.
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    M, N, K,                       # matrix sizes
    stride_am, stride_ak,          # A: row stride, col stride
    stride_bk, stride_bn,          # B: row stride (K), col stride (N)
    stride_cm, stride_cn,          # C: row stride, col stride
    stride_ab, stride_bb, stride_cb,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B  (C, A, B of shape [batch, M, N/K])"""
    pid_batch = tl.program_id(0)          # batch index
    pid_m = tl.program_id(1)              # block row
    pid_n = tl.program_id(2)              # block column

    # ---- block offsets ----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---- move to correct batch slice ----
    a_batch = A_ptr + pid_batch * stride_ab
    b_batch = B_ptr + pid_batch * stride_bb
    c_batch = C_ptr + pid_batch * stride_cb

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- K-loop ----
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am
                            + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile: (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk
                            + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Matrix multiply tile
        acc += tl.dot(a, b)

    # ---- write back ----
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm
                        + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom TriMul forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim] torch.Tensor (CUDA)
            - mask_tensor:  [B, N, N]   torch.Tensor (or None)
            - weights_dict: mapping from weight name to torch.Tensor
            - config_dict: contains "dim", "hidden_dim", and optional "nomask"
    Returns:
        output_tensor: [B, N, N, dim] torch.Tensor (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device
    dtype = torch.float32  # final output type

    # ------------------------------------------------------------------
    # load weights and cast them to the compute dtype
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=dtype)
    norm_bias = weights["norm.bias"].to(device=device, dtype=dtype)

    left_proj_weight = weights["left_proj.weight"].to(device=device, dtype=dtype)
    right_proj_weight = weights["right_proj.weight"].to(device=device, dtype=dtype)

    left_gate_weight = weights["left_gate.weight"].to(device=device, dtype=dtype)
    right_gate_weight = weights["right_gate.weight"].to(device=device, dtype=dtype)
    out_gate_weight = weights["out_gate.weight"].to(device=device, dtype=dtype)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device=device, dtype=dtype)

    to_out_weight = weights["to_out.weight"].to(device=device, dtype=dtype)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = input_tensor.to(dtype)  # promote to float32
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x = (x - mean) / torch.sqrt(var + eps)
    x = x * norm_weight + norm_bias               # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask application
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_ = mask.unsqueeze(-1).to(dtype)       # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Batched GEMM (Triangle multiplicative update)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    → for each d: out_d = left_d @ right_d^T
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # move hidden dimension to second axis and make tensors contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # flatten batch & hidden into a single batch dimension
    batch_h = B * hidden_dim
    left_mat = left_perm.view(batch_h, N, N)          # [B*H, N, N]
    right_mat = right_perm.view(batch_h, N, N)        # [B*H, N, N]

    # transpose right for GEMM
    right_T = right_mat.transpose(-2, -1).contiguous()  # [B*H, N, N]

    # allocate output buffer
    out_mat = torch.empty_like(left_mat)

    # ------------------------------------------------------------------
    # Triton launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_h,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[
        grid
    ](
        left_mat,
        right_T,
        out_mat,
        N,
        N,
        N,
        left_mat.stride(1),          # stride_am (row stride)
        left_mat.stride(2),          # stride_ak (col stride)
        right_T.stride(1),           # stride_bk (row stride in K)
        right_T.stride(2),           # stride_bn (col stride in N)
        out_mat.stride(1),           # stride_cm
        out_mat.stride(2),           # stride_cn
        left_mat.stride(0),          # batch stride A
        right_T.stride(0),           # batch stride B
        out_mat.stride(0),           # batch stride C
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout: [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # back to original dim

    return out
