```
@triton.jit
def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb, # batch strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    # program id for i and j tiling: compute pid_m = pid // num_pid_n, pid_n = pid % num_pid_n
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n
    
    # offsets within block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        # mask for load
        a_ptrs = a_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        # load blocks
        a = tl.load(a_ptrs, mask = (offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask = (offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)
        acc += tl.dot(a, b) # dot does matrix multiplication? tl.dot expects shape (M,K) and (K,N) returns (M,N)
    # store result
    c_ptrs = c_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask = (offs_m[:, None] < M) & (offs_n[None, :] < N))
```

```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    # extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # layernorm x
    x_norm = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)
    
    # linear projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # (B,N,N,hidden)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)
    
    # mask
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp
    
    # gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()
    
    left = left * left_gate
    right = right * right_gate
    
    # Cast to half for kernel
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    
    # Prepare output tensor
    out = torch.empty_like(left_h, dtype=torch.float32)  # shape B,N,N,hidden
    
    # Triton kernel launch
    B, N, _, H = left_h.shape
    # strides
    stride_ab, stride_am, stride_ak, stride_ah = left_h.stride()
    stride_bb, stride_bj, stride_bk, stride_bh = right_h.stride()
    stride_cb, stride_cm, stride_cn, stride_ch = out.stride()
    
    # block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    
    # grid
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)

        left_h, right_h, out,

        stride_bb, stride_bj, stride_bk,
        stride_cb, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # final linear
    out_final = torch.nn.functional.linear(out_norm, to_out_weight)
    return out_final
```

```
    a_ptr, b_ptr, c_ptr,
```
    B, M, N, K, H,
    stride_ab, stride_am, stride_ak, stride_ah,
    stride_bb, stride_bj, stride_bk, stride_bh,

    h = pid_bh % H

    b_ptr = b_ptr + b * stride_bb + h * stride_bh
    c_ptr = c_ptr + b * stride_cb + h * stride_ch

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections generate left/right tensors of shape [B, N, N, H].
3. Apply the optional mask and compute three sigmoid gates.
4. Fuse gates into the projected tensors.
5. Compute the contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   via a Triton matmul kernel which treats each hidden slice as an
   independent batch‑matrix‑multiply (left @ rightᵀ).  The kernel
   distributes the i/j dimensions over program ids and the (batch, hidden)
   pair over a third axis.
6. Apply a second LayerNorm, multiply by the output gate and a final
   linear projection back to the original feature dimension.

Only the O(N³·H) contraction is performed on Triton; all other ops use
PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, M, N, K, H,
    stride_ab, stride_am, stride_ak, stride_ah,
    stride_bb, stride_bj, stride_bk, stride_bh,
    stride_cb, stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid_m = tl.program_id(0)               # block over i
    pid_n = tl.program_id(1)               # block over j
    pid_bh = tl.program_id(2)              # combined batch and hidden index

    # decode batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # base pointers for this (b, h) slice
    a_base = a_ptr + b * stride_ab + h * stride_ah
    b_base = b_ptr + b * stride_bb + h * stride_bh
    c_base = c_ptr + b * stride_cb + h * stride_ch

    # block offsets in the i and j dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # load A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a_mask = mask_m[:, None] & (offs_k[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # load B block transposed: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj
        b_mask = (offs_k[:, None] < K) & mask_n[None, :]
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # matrix multiply for the current k‑tile
        acc += tl.dot(a, b)

    # write out the result tile
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be all‑ones)
            - weights: dict of model parameters
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ----- unpack weights -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    eps = 1e-5

    # ----- LayerNorm on the input ----------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ----- Linear projections ---------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ----- Mask (if provided) --------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)       # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ----- Gates -----------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----- Ensure contiguous memory for Triton ----------------------------
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, H = left.shape

    # ----- Allocate output tensor (float32) --------------------------------
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # ----- Strides (in element units) ------------------------------------
    stride_ab, stride_am, stride_ak, stride_ah = left.stride()
    stride_bb, stride_bj, stride_bk, stride_bh = right.stride()
    stride_cb, stride_cm, stride_cn, stride_ch = out.stride()

    # ----- Triton kernel launch -------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    tri_mul_matmul_kernel[grid](
        left, right, out,
        B, N, N, N, H,
        stride_ab, stride_am, stride_ak, stride_ah,
        stride_bb, stride_bj, stride_bk, stride_bh,
        stride_cb, stride_cm, stride_cn, stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- Post‑processing -------------------------------------------------
    # LayerNorm over the hidden dimension
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    # Apply the output gate
    out_norm = out_norm * out_gate

    # Final linear projection back to `dim`
    output = torch.nn.functional.linear(out_norm, to_out_weight)

    return output
