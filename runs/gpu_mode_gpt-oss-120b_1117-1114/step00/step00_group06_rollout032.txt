def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_b_left, stride_i_left, stride_k_left, stride_d_left,
    stride_b_right, stride_i_right, stride_k_right, stride_d_right,
    stride_b_out, stride_i_out, stride_j_out, stride_d_out,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_dc = tl.program_id(2)

    b = pid_dc // hidden_dim
    d = pid_dc % hidden_dim

    if b >= B:
        return

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Compute base pointers for this batch and channel
    left_base = left_ptr + b * stride_b_left + d * stride_d_left
    right_base = right_ptr + b * stride_b_right + d * stride_d_right

    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0, dtype=tl.bfloat16)

"""
TriMul (outgoing) forward pass with Triton kernel for the O(N³) reduction.

Algorithm
---------
1. Layer‑norm the input on the last dimension.
2. Linear projections and gating (all in PyTorch, float32).
3. Apply mask (if provided) and gating, then cast the tensors to bfloat16.
4. Triton kernel computes, for each batch b and hidden channel d,
   the matrix product:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   i.e. a batched GEMM where the right operand is transposed.
   The kernel tiles the i‑, j‑ and k‑dimensions (BLOCK_I×BLOCK_J×BLOCK_K)
   and accumulates in float32.
5. Apply the output LayerNorm, the output gate and the final linear projection
   back to the original feature dimension (all in PyTorch, float32).

The kernel processes one hidden channel per program (grid dimension 2 = B*hidden_dim)
so that per‑program register pressure stays modest.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_gemm_kernel(
    left_ptr,            # *bf16   [B, N, N, H]
    right_ptr,           # *bf16   [B, N, N, H]
    out_ptr,             # *f32    [B, N, N, H]
    B, N, hidden_dim,    # runtime ints
    stride_b_left, stride_i_left, stride_k_left, stride_d_left,
    stride_b_right, stride_i_right, stride_k_right, stride_d_right,
    stride_b_out, stride_i_out, stride_j_out, stride_d_out,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM for the TriMul core:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
    The right operand is accessed transposed (k,j) to obtain the needed sum.
    """

    pid_i = tl.program_id(0)          # tile over i dimension
    pid_j = tl.program_id(1)          # tile over j dimension
    pid_dc = tl.program_id(2)         # combined batch * hidden‑channel index

    # decode batch and hidden‑channel from pid_dc
    b = pid_dc // hidden_dim
    d = pid_dc % hidden_dim
    if b >= B:
        return

    # i and j coordinates for this tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in float32
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # base pointers for this (b, d) slice
    left_base = left_ptr + b * stride_b_left + d * stride_d_left
    right_base = right_ptr + b * stride_b_right + d * stride_d_right

    # iterate over the reduction axis k
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_I, BLOCK_K)
        left_ptrs = left_base + \
                    offs_i[:, None] * stride_i_left + \
                    offs_k[None, :] * stride_k_left
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.bfloat16)
        left = left.to(tl.float32)

        # right tile transposed: (k, j) -> shape (BLOCK_K, BLOCK_J)
        right_ptrs = right_base + \
                     offs_k[:, None] * stride_i_right + \
                     offs_j[None, :] * stride_k_right
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0,
                        dtype=tl.bfloat16)
        right = right.to(tl.float32)

        # GEMM contribution
        acc += tl.dot(left, right)   # (BLOCK_I, BLOCK_K) @ (BLOCK_K, BLOCK_J)

    # store the result for this (b, d) slice
    out_ptrs = out_ptr + b * stride_b_out + \
               offs_i[:, None] * stride_i_out + \
               offs_j[None, :] * stride_j_out + \
               d * stride_d_out
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    out = out * out_gate                     # gating after norm
    to_out_weight = weights["to_out.weight"].to(device)   # (dim, hidden_dim)
    out = F.linear(out, to_out_weight)       # final projection to [B, N, N, dim]

    return out
```
