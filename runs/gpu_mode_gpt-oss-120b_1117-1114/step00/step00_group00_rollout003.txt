@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    total, M, N, K,
    stride_Ab, stride_Ak, stride_Abatch,
    stride_Bk, stride_Bn, stride_Bbatch,
    stride_Cm, stride_Cn, stride_Cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)  # linear program id across total * (ceil(M/BLOCK_M)*ceil(N/BLOCK_N))
    total_tiles_m = tl.cdiv(M, BLOCK_M)
    total_tiles_n = tl.cdiv(N, BLOCK_N)
    total_tiles = total_tiles_m * total_tiles_n
    batch = pid // total_tiles
    tile_id = pid % total_tiles
    pid_m = tile_id // total_tiles_n
    pid_n = tile_id % total_tiles_n

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers
    a_ptrs = A_ptr + batch * stride_Abatch + offs_m[:, None] * stride_Ab + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + batch * stride_Bbatch + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk
    # Store
    c_ptrs = C_ptr + batch * stride_Cbatch + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc.to(tl.float16) if needed? but we'll store float32 directly)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # layernorm
    eps = 1e-5
    w = weights["norm.weight"]
    b = weights["norm.bias"]
    # compute mean,var across dim -1
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * w + b

    # linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape (bs,N,N,hidden_dim)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # mask multiply if not nomask (config may have "nomask" flag)
    if not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp
    else:
        # mask may still be provided but ignore
        pass

    # gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # contraction using Triton GEMM
    # transpose right on dims 1 and 2
    right_T = right.transpose(1,2)  # shape (bs,N,N,hidden_dim)

    # permute for batch*hidden
    left_perm = left.permute(0,3,1,2).contiguous()   # (bs, hidden_dim, i, k)
    right_perm = right_T.permute(0,3,1,2).contiguous()  # (bs, hidden_dim, k, i)
    # flatten
    total = bs * hidden_dim
    left_flat = left_perm.view(total, N, N)
    right_flat = right_perm.view(total, N, N)
    # allocate output buffer
    out_flat = torch.empty_like(left_flat)  # same dtype float32, shape (total,N,N)

    # kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_tiles = grid_m * grid_n
    grid = (total * total_tiles, )
    batched_gemm_kernel[grid](
        left_flat, right_flat, out_flat,
        total, N, N, N,
        # strides
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4
    )
    # reshape back
    out_hidden = out_flat.view(bs, hidden_dim, N, N).permute(0,2,3,1)  # shape (bs,N,N,hidden_dim)

    # out norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    eps2 = 1e-5
    mean_out = out_hidden.mean(-1, keepdim=True)
    var_out = out_hidden.var(-1, unbiased=False, keepdim=True)
    out_normed = (out_hidden - mean_out) / torch.sqrt(var_out + eps2)
    out_normed = out_normed * to_out_norm_weight + to_out_norm_bias

    # apply out_gate
    out_normed = out_normed * out_gate

    # final linear
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out_normed, to_out_weight)   # shape (bs,N,N, dim)

    return out

@triton.jit
def batched_gemm_kernel(A_ptr, B_ptr, C_ptr,
                        total, M, N, K,
                        stride_a_batch, stride_a_m, stride_a_k,
                        stride_b_batch, stride_b_k, stride_b_n,
                        stride_c_batch, stride_c_m, stride_c_n,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    # compute tile indices
    total_tiles_m = tl.cdiv(M, BLOCK_M)
    total_tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = total_tiles_m * total_tiles_n
    batch_idx = pid // tiles_per_batch
    tile_idx = pid % tiles_per_batch
    pid_m = tile_idx // total_tiles_n
    pid_n = tile_idx % total_tiles_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute masks for boundary
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to current tile in A and B
    a_ptrs = A_ptr + batch_idx * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = B_ptr + batch_idx * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load A and B
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # store to C
    c_ptrs = C_ptr + batch_idx * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul outgoing implementation (AlphaFold3).

The forward pass consists of:
1. LayerNorm on the input (dim).
2. Linear projections to hidden_dim (left_proj, right_proj).
3. Optional mask multiplication.
4. Gated multiplicative update:
     left = left * sigmoid(linear(x, left_gate))
     right = right * sigmoid(linear(x, right_gate))
   and the core contraction:
     out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication left @ right^T per (batch, hidden) channel.
   This contraction is performed by a custom Triton GEMM kernel.
5. LayerNorm on the hidden dimension, gating with out_gate and a final linear
   projection back to the original dim.

All heavy computation (the O(N³) contraction) is executed in Triton; the
remaining linear / norm / sigmoid steps are performed with PyTorch for
conciseness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM kernel (outgoing TriMul contraction)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    total, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)

    # number of tiles per batch in each dim
    tiles_m = tl.cdiv(M, BLOCK_M)
    tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = tiles_m * tiles_n

    batch_idx = pid // tiles_per_batch
    tile_idx = pid % tiles_per_batch
    pid_m = tile_idx // tiles_n
    pid_n = tile_idx % tiles_n

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current tile
    a_ptrs = (
        A_ptr
        + batch_idx * stride_a_batch
        + offs_m[:, None] * stride_a_m
        + offs_k[None, :] * stride_a_k
    )
    b_ptrs = (
        B_ptr
        + batch_idx * stride_b_batch
        + offs_k[:, None] * stride_b_k
        + offs_n[None, :] * stride_b_n
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_blocks):
        cur_k = k * BLOCK_K
        # masks for the current k‑block
        mask_k = (cur_k + offs_k) < K
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)
        # advance to next k‑block
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # Store results
    c_ptrs = (
        C_ptr
        + batch_idx * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (float, 0/1)
        - weights      : dict mapping weight names to torch.Tensor
        - config       : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Apply mask if present
    # --------------------------------------------------------------
    if not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(dtype)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4) Gates (sigmoids of linear projections)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Core contraction via Triton (batched GEMM)
    # --------------------------------------------------------------
    # transpose right on the two sequence axes to obtain B^T needed for GEMM
    right_t = right.transpose(1, 2)  # shape [B,N,N,hidden] -> [B,N,N,hidden] with i/k swapped

    # reshape to (B*hidden, N, N) for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, i, k)
    right_perm = right_t.permute(0, 3, 1, 2).contiguous()  # (B, hidden, k, i)

    total = B * hidden_dim
    left_flat = left_perm.view(total, N, N)
    right_flat = right_perm.view(total, N, N)
    out_flat = torch.empty_like(left_flat)  # will hold (B*hidden, N, N)

    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    tiles_m = triton.cdiv(N, BLOCK_M)
    tiles_n = triton.cdiv(N, BLOCK_N)
    grid = (total * tiles_m * tiles_n,)

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        total,
        N,
        N,
        N,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to [B, N, N, hidden]
    out_hidden = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6) Output LayerNorm, gating, final linear
    # --------------------------------------------------------------
    out_norm = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate
    out = F.linear(out_norm, weights["to_out.weight"])   # [B,N,N,dim]

    return out
