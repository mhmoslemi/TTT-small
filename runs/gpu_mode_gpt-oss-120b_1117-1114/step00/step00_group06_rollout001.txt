"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

The heavy part of the TriMul operator is:
    out[b, i, j, d] = sum_k left[b, i, k, d] * right[b, j, k, d]

We rewrite it as a batched matrix multiplication:
    left_reshape = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_reshape = right.permute(0, 3, 1, 2) # (B, H, N, N)
    out_reshape = batched_matmul(left_reshape, right_reshape.transpose(-2, -1))

The batched GEMM is implemented in Triton (half precision, FP32 accumulation)
and the surrounding LayerNorm / Linear / gating operations are done with
PyTorch.  This satisfies the requirement of having a custom kernel while
keeping the implementation simple and correct.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ B) where
#   A: (batch, M, K)   B: (batch, K, N)   C: (batch, M, N)
# All tensors are contiguous.  Half precision is used for the inputs,
# accumulator is FP32.
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    BATCH, M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(2)          # batch dimension (BATCH)
    pid_m = tl.program_id(0)              # block row
    pid_n = tl.program_id(1)              # block column

    # offsets within M and N
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for A (M x K)
        a_ptrs = (
            a_ptr
            + pid_batch * batch_stride_a
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # pointers for B (K x N)
        b_ptrs = (
            b_ptr
            + pid_batch * batch_stride_b
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # dot product accumulation
        acc += tl.dot(a, b)

    # write results
    c_ptrs = (
        c_ptr
        + pid_batch * batch_stride_c
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )
# ----------------------------------------------------------------------


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a custom Triton GEMM.
    Args:
        data: tuple (input_tensor, mask_tensor, weight_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # LayerNorm on the last dim
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(device, dtype),
        bias=weights["norm.bias"].to(device, dtype),
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # Projections
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(device, dtype), bias=None)
    right = F.linear(x, weights["right_proj.weight"].to(device, dtype), bias=None)

    # ------------------------------------------------------------------
    # Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(device, dtype), bias=None)
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(device, dtype), bias=None)
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(device, dtype), bias=None)
    )

    # ------------------------------------------------------------------
    # Apply mask (if present) and gates
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # (bs, N, N, 1)
        left = left * mask
        right = right * mask
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the batched GEMM
    # ------------------------------------------------------------------
    # shape after projection: (bs, N, N, hidden_dim)
    # permute to (bs, hidden_dim, N, N) and flatten batch & heads
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    batch_total = bs * hidden_dim

    left_flat = left_perm.view(batch_total, N, N).to(torch.float16)
    right_flat = right_perm.view(batch_total, N, N).to(torch.float16)

    # transpose the second tensor to get (batch, K, N) = (batch, N, N)
    right_t = right_flat.transpose(-2, -1).contiguous()

    # output placeholder
    out_flat = torch.empty_like(right_t, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # Triton launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_total,
    )

    # strides (in elements)
    a_batch_stride = left_flat.stride(0)
    a_stride_m = left_flat.stride(1)  # row stride (M)
    a_stride_k = left_flat.stride(2)  # col stride (K)

    b_batch_stride = right_t.stride(0)
    b_stride_k = right_t.stride(1)    # K stride
    b_stride_n = right_t.stride(2)    # N stride

    c_batch_stride = out_flat.stride(0)
    c_stride_m = out_flat.stride(1)
    c_stride_n = out_flat.stride(2)

    # launch kernel
    batched_matmul_kernel[grid](
        left_flat,
        right_t,
        out_flat,
        a_batch_stride,
        a_stride_m,
        a_stride_k,
        b_batch_stride,
        b_stride_k,
        b_stride_n,
        c_batch_stride,
        c_stride_m,
        c_stride_n,
        batch_total,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to (bs, N, N, hidden_dim) and FP32
    # ------------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).to(dtype)

    # ------------------------------------------------------------------
    # Output LayerNorm
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device, dtype),
        bias=weights["to_out_norm.bias"].to(device, dtype),
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # Apply final gating and linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(device, dtype), bias=None)

    return out
