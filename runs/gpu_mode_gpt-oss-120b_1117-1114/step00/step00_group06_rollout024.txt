@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_i, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)
    b = pid_z // H
    h = pid_z % H
    pid_m = tl.program_id(0)  # row tile (i)
    pid_n = tl.program_id(1)  # col tile (j)

    # Compute global offsets for i and j
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for i/j within bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Compute pointers for left: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + b * stride_l_b + h * stride_l_h + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Compute pointers for right transposed: shape (BLOCK_K, BLOCK_N)
        # We need element (b, h, k, j)
        b_ptrs = right_ptr + b * stride_r_b + h * stride_r_h + offs_k[:, None] * stride_r_i + offs_n[None, :] * stride_r_k
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K) dot (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
    # End for

    # Write out
    out_ptrs = out_ptr + b * stride_o_b + h * stride_o_h + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # Load weights
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)
    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    to_out_weight = weights["to_out.weight"].to(device)

    # Step1: LayerNorm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Step2: Projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Step3: Mask
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)  # shape [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # Step4: Gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Step5: Compute triangular multiplication using Triton
    # Permute to [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape

    out_perm = torch.empty_like(left_perm)

    # Compute strides in elements
    stride_l_b, stride_l_h, stride_l_i, stride_l_k = left_perm.stride()
    stride_r_b, stride_r_h, stride_r_i, stride_r_k = right_perm.stride()
    stride_o_b, stride_o_h, stride_o_i, stride_o_j = out_perm.stride()

    # Define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    tri_mul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B, H, N,
        stride_l_b, stride_l_h, stride_l_i, stride_l_k,
        stride_r_b, stride_r_h, stride_r_i, stride_r_k,
        stride_o_b, stride_o_h, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Permute back to original shape
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # Step6: LayerNorm on hidden_dim
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Step7: Multiply out_gate
    out = out * out_gate

    # Step8: Final linear projection
    output = torch.nn.functional.linear(out, to_out_weight)

    return output
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_i, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out = left @ right^T for each (batch, head) pair.
    left, right, out are shape [B, H, N, N] in row-major order.
    """
    pid_z = tl.program_id(2)
    b = pid_z // H
    h = pid_z % H

    pid_m = tl.program_id(0)   # tile row index i
    pid_n = tl.program_id(1)   # tile col index j

    # global offsets for i and j
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile [i,k]
        a_ptrs = left_ptr + \
            b * stride_l_b + \
            h * stride_l_h + \
            offs_i[:, None] * stride_l_i + \
            offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right transposed tile [k,j] (i.e., right[j,k])
        b_ptrs = right_ptr + \
            b * stride_r_b + \
            h * stride_r_h + \
            offs_k[:, None] * stride_r_i + \
            offs_j[None, :] * stride_r_k
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Store result
    out_ptrs = out_ptr + \
        b * stride_o_b + \
        h * stride_o_h + \
        offs_i[:, None] * stride_o_i + \
        offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton-accelerated forward pass for the outgoing Triangle Multiplicative Update.
    Input tuple:
        (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Load weight tensors onto the same device / dtype
    norm_weight = weights["norm.weight"].to(device=device, dtype=dtype)
    norm_bias = weights["norm.bias"].to(device=device, dtype=dtype)

    left_proj_weight = weights["left_proj.weight"].to(device=device, dtype=dtype)
    right_proj_weight = weights["right_proj.weight"].to(device=device, dtype=dtype)

    left_gate_weight = weights["left_gate.weight"].to(device=device, dtype=dtype)
    right_gate_weight = weights["right_gate.weight"].to(device=device, dtype=dtype)
    out_gate_weight = weights["out_gate.weight"].to(device=device, dtype=dtype)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device=device, dtype=dtype)

    to_out_weight = weights["to_out.weight"].to(device=device, dtype=dtype)

    # 1) LayerNorm on input
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # 2) Linear projections
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # 3) Mask (if present and not disabled)
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(dtype)
        left = left * mask_
        right = right * mask_

    # 4) Gating
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # 5) Triangular multiplication via Triton
    # Rearrange to [B, hidden_dim, N, N] for efficient matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    out_perm = torch.empty_like(left_perm)

    # strides in elements
    stride_l_b, stride_l_h, stride_l_i, stride_l_k = left_perm.stride()
    stride_r_b, stride_r_h, stride_r_i, stride_r_k = right_perm.stride()
    stride_o_b, stride_o_h, stride_o_i, stride_o_j = out_perm.stride()

    # Tile sizes (tuned for typical B200 GPU)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    tri_mul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B, H, N,
        stride_l_b, stride_l_h, stride_l_i, stride_l_k,
        stride_r_b, stride_r_h, stride_r_i, stride_r_k,
        stride_o_b, stride_o_h, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # 6) Restore original layout
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # 7) LayerNorm on hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # 8) Apply out gate
    out = out * out_gate

    # 9) Final linear projection back to model dimension
    output = F.linear(out, to_out_weight)

    return output

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_i, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out = left @ rightᵀ for each (batch, head) pair.
    Shapes:
        left, right, out : [B, H, N, N] (row‑major)
    The kernel tiles the i (row) and j (col) dimensions with BLOCK_M/BLOCK_N
    and iterates over k with BLOCK_K.
    """
    pid_z = tl.program_id(2)
    b = pid_z // H
    h = pid_z % H

    pid_m = tl.program_id(0)  # tile row (i)
    pid_n = tl.program_id(1)  # tile col (j)

    # Global offsets for the current tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator for the output tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the k dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)  -> left[b, h, i, k]
        a_ptrs = left_ptr + \
            b * stride_l_b + \
            h * stride_l_h + \
            offs_i[:, None] * stride_l_i + \
            offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # rightᵀ tile: shape (BLOCK_K, BLOCK_N)  -> right[b, h, k, j]

            offs_j[None, :] * stride_r_k
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Accumulate the dot product
        acc += tl.dot(a, b)

    # Store the result tile
    out_ptrs = out_ptr + \
        b * stride_o_b + \
        h * stride_o_h + \
        offs_i[:, None] * stride_o_i + \
        offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor, shape [B, N, N] (may be all‑ones)
        - weights      : dict of torch.Tensors with the layer parameters
        - config       : dict containing ``dim``, ``hidden_dim`` and optional ``nomask`` flag

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load model parameters onto the correct device / dtype
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=dtype)
    norm_bias   = weights["norm.bias"].to(device=device, dtype=dtype)

    left_proj_weight  = weights["left_proj.weight"].to(device=device, dtype=dtype)
    right_proj_weight = weights["right_proj.weight"].to(device=device, dtype=dtype)

    left_gate_weight  = weights["left_gate.weight"].to(device=device, dtype=dtype)
    right_gate_weight = weights["right_gate.weight"].to(device=device, dtype=dtype)
    out_gate_weight   = weights["out_gate.weight"].to(device=device, dtype=dtype)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device=device, dtype=dtype)

    to_out_weight = weights["to_out.weight"].to(device=device, dtype=dtype)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask application
    # ------------------------------------------------------------------
    if not nomask:
        mask_ = mask.unsqueeze(-1).to(dtype)          # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate   = F.linear(x, out_gate_weight).sigmoid()

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triangular multiplicative update via Triton
    #    out = left @ rightᵀ  (per head & batch)
    # ------------------------------------------------------------------
    # Bring heads to the second dimension for a batched matmul
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    out_perm = torch.empty_like(left_perm)

    # Strides in elements (not bytes)
    stride_l_b, stride_l_h, stride_l_i, stride_l_k = left_perm.stride()
    stride_r_b, stride_r_h, stride_r_i, stride_r_k = right_perm.stride()
    stride_o_b, stride_o_h, stride_o_i, stride_o_j = out_perm.stride()

    # Tile sizes – chosen to fit typical B200 resources
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        B * H,                     # (batch, head) pairs
    )

    tri_mul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B, H, N,
        stride_l_b, stride_l_h, stride_l_i, stride_l_k,
        stride_r_b, stride_r_h, stride_r_i, stride_r_k,
        stride_o_b, stride_o_h, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Restore original layout
    # ------------------------------------------------------------------
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 7) Hidden‑dim LayerNorm
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final projection back to original dimensionality
    # ------------------------------------------------------------------
    output = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return output
