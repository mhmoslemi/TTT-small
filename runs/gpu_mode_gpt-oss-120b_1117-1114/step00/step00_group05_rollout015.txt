def batched_matmul_kernel(A, B, C,
                          stride_ah0, stride_ah1, stride_ah2,
                          stride_bh0, stride_bh1, stride_bh2,
                          stride_ch0, stride_ch1, stride_ch2,
                          M, N, K,
                          BLOCK_M: tl.constexpr,
                          BLOCK_N: tl.constexpr,
                          BLOCK_K: tl.constexpr):
    # Batch index
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers start for A and B
    a_ptrs = A + batch * stride_ah0 + offs_m[:, None] * stride_ah1
    b_ptrs = B + batch * stride_bh0 + offs_n[None, :] * stride_bh2

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Current K offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask for K dimension
        mask_k = offs_k < K

        # Load A block: shape (BLOCK_M, BLOCK_K)

        acc += tl.dot(a, b)  # yields (BLOCK_M, BLOCK_N) in float32

    # Write back C
    c_ptrs = C + batch * stride_ch0 + offs_m[:, None] * stride_ch1 + offs_n[None, :] * stride_ch2
    mask = mask_m[:, None] & mask_n[None, :]

"""
TriMul custom kernel (outgoing version)

The kernel accelerates the core O(N³) operation:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

All other operations (LayerNorm, linear projections, gating, final
normalization and linear projection) are performed in PyTorch.  The
left/right tensors are reshaped to (B*H, N, N) and a batched GEMM is
computed with a Triton kernel that iterates over the K‑dimension in
blocks.  Computation is performed in float32 for numerical fidelity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication A @ B where
# A : (batch, M, K)   (M = N, K = N)
# B : (batch, K, N)
# C : (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A, B, C,
    stride_ah0, stride_ah1, stride_ah2,
    stride_bh0, stride_bh1, stride_bh2,
    stride_ch0, stride_ch1, stride_ch2,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)          # combined batch * hidden_dim
    pid_m    = tl.program_id(1)          # tile row
    pid_n    = tl.program_id(2)          # tile col

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current batch
    a_ptrs = A + batch_id * stride_ah0 + offs_m[:, None] * stride_ah1
    b_ptrs = B + batch_id * stride_bh0 + offs_n[None, :] * stride_bh2

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_blocks):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile of A (M x K)
        a = tl.load(
            a_ptrs + offs_k[None, :] * stride_ah2,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load a tile of B (K x N)
        b = tl.load(
            b_ptrs + offs_k[:, None] * stride_bh1,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        # Multiply‑accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N) in fp32

    # Write result
    c_ptrs = C + batch_id * stride_ch0 + offs_m[:, None] * stride_ch1 + offs_n[None, :] * stride_ch2
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask)


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Executes the outgoing TriMul operation using a Triton kernel for the
    intensive O(N³) matrix multiplication.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. First LayerNorm (over the last dimension)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, C]  float32

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask
    # --------------------------------------------------------------
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1).to(left.dtype)       # broadcast over hidden_dim
        left = left * mask_unsq
        right = right * mask_unsq

    # --------------------------------------------------------------
    # 4. Gating values
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------
    # 5. Apply gates
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6. Core TriMul (batched GEMM) via Triton
    # --------------------------------------------------------------
    # reshape to (B, hidden_dim, N, N) and then merge batch+hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)   i,k
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)   k,j

    batch_dim = B * hidden_dim
    left_flat = left_perm.view(batch_dim, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(batch_dim, N, N) # (B*H, N, N)

    out_flat = torch.empty_like(left_flat)        # allocate result (float32)

    # Triton launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, hidden_dim, N, N) then to (B, N, N, hidden_dim)
    out_perm = out_flat.view(B, hidden_dim, N, N)
    out = out_perm.permute(0, 2, 3, 1)  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 7. Post‑process: LayerNorm, out‑gate and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # -> [B, N, N, dim]

    # Ensure final dtype is float32
    return out.to(torch.float32)
