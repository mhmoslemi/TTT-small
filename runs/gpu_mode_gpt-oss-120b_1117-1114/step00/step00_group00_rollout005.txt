@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute coordinates of the start of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Masks for out-of-bounds rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the batch
    a_ptrs = A_ptr + pid_batch * stride_Ab
    b_ptrs = B_ptr + pid_batch * stride_Bb
    c_ptrs = C_ptr + pid_batch * stride_Cb

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute k offsets
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        # Masks for k range
        mask_k = offs_k < K

        # Load A block (M x K)
        a = tl.load(
            a_ptrs + (offs_m[:, None] * stride_Am) + (offs_k[None, :] * stride_Ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B block (K x N)
        b = tl.load(
            b_ptrs + (offs_k[:, None] * stride_Bk) + (offs_n[None, :] * stride_Bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Compute dot-product
        acc += tl.dot(a, b)
        # Advance pointers to next k tile
        # Actually we use offsets each iteration rather than pointer increment, so pointer unchanged.
    # Store result
    c = acc.to(tl.float16)
    tl.store(
        c_ptrs + (offs_m[:, None] * stride_Cm) + (offs_n[None, :] * stride_Cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )
```

"""
TriMul forward implementation with a Triton batched matrix‑multiply kernel.

The heavy operation is
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matmul of shape (B*hidden_dim, N, N) × (B*hidden_dim, N, N)^T.
The kernel computes this reduction in FP16 with FP32 accumulation.
All other linear layers, LayerNorms and gates are performed in PyTorch.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched matmul (A @ B) where A: (batch, M, K), B: (batch, K, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr,                     # *float16
    B_ptr,                     # *float16
    C_ptr,                     # *float16
    BATCH, M, N, K,            # dimensions (int)
    stride_Ab, stride_Am, stride_Ak,   # strides for A
    stride_Bb, stride_Bk, stride_Bn,   # strides for B
    stride_Cb, stride_Cm, stride_Cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Compute per‑batch base pointers
    a_batch_ptr = A_ptr + pid_batch * stride_Ab
    b_batch_ptr = B_ptr + pid_batch * stride_Bb
    c_batch_ptr = C_ptr + pid_batch * stride_Cb

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k_tile * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a = tl.load(
            a_batch_ptr
            + (offs_m[:, None] * stride_Am)
            + (offs_k[None, :] * stride_Ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (K x N)
        b = tl.load(
            b_batch_ptr
            + (offs_k[:, None] * stride_Bk)
            + (offs_n[None, :] * stride_Bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate
        acc += tl.dot(a, b)

    # Write back result in FP16
    c = acc.to(tl.float16)
    tl.store(
        c_batch_ptr
        + (offs_m[:, None] * stride_Cm)
        + (offs_n[None, :] * stride_Cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )

# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the Triangle Multiplicative Update (TriMul) using Triton.

    Args:
        data: Tuple containing (input, mask, weights, config)
            - input: Tensor [B, N, N, C] (float32)
            - mask: Tensor [B, N, N] (bool or float, may be None)
            - weights: Mapping from weight name to tensor.
            - config: Mapping with keys "dim" and "hidden_dim".

    Returns:
        Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # LayerNorm on the input (dim = C)
    # --------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, C]

    # --------------------------------------------------------------
    # Linear projections (no bias) + optional mask
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    if mask is None:
        mask_tensor = torch.ones(input_tensor.shape[:3], dtype=input_tensor.dtype, device=device)
    else:
        mask_tensor = mask.to(input_tensor.dtype)
    mask_tensor = mask_tensor.unsqueeze(-1)  # [B, N, N, 1]

    left = left * mask_tensor
    right = right * mask_tensor

    # --------------------------------------------------------------
    # Gates (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Prepare tensors for the Triton batched matmul
    # --------------------------------------------------------------
    # Convert to half precision for the kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Rearrange to [B, hidden_dim, N, N] and make contiguous
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_h.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Transpose the second matrix on the last two dims (K ↔ N)
    right_perm_T = right_perm.transpose(-2, -1).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    batch_mat = B * H

    # Collapse batch and hidden-dim into a single batch dimension for the kernel
    left_mat = left_perm.view(batch_mat, N, N)
    right_mat = right_perm_T.view(batch_mat, N, N)

    # Allocate output
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=device)

    # Strides (in element units)
    stride_Ab = left_mat.stride(0)
    stride_Am = left_mat.stride(1)
    stride_Ak = left_mat.stride(2)

    stride_Bb = right_mat.stride(0)
    stride_Bk = right_mat.stride(1)
    stride_Bn = right_mat.stride(2)

    stride_Cb = out_mat.stride(0)
    stride_Cm = out_mat.stride(1)
    stride_Cn = out_mat.stride(2)

    # --------------------------------------------------------------
    # Launch Triton kernel
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (batch_mat, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch_mat,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # --------------------------------------------------------------
    # Post‑processing (LayerNorm, out‑gate, final linear)
    # --------------------------------------------------------------
    out = out.to(torch.float32)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # broadcast multiplication

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(torch.float32)
