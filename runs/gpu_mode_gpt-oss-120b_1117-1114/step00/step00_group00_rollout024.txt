def matmul_tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_a_hidden, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_hidden, stride_b_k, stride_b_j,
    stride_c_batch, stride_c_hidden, stride_c_i, stride_c_j,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,

    num_blocks_m: tl.constexpr,
    num_blocks_n: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,

    # Determine which batch*hidden we are processing
    batch_hidden = pid // (num_blocks_m * num_blocks_n)
    block_id = pid % (num_blocks_m * num_blocks_n)

    pid_n = block_id % num_blocks_n

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

    batch_idx = batch_hidden // HIDDEN_DIM
    hidden_idx = batch_hidden % HIDDEN_DIM

    a_base = a_ptr + batch_idx * stride_a_batch + hidden_idx * stride_a_hidden
    b_base = b_ptr + batch_idx * stride_b_batch + hidden_idx * stride_b_hidden

def custom_kernel(data):
    """
    Triton implementation of the TriMul "outgoing" update.
    Performs LayerNorm, linear projections, gating, the core
    pairwise contraction using a fused batched GEMM kernel, followed
    by a second LayerNorm, output gating and a final linear projection.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # LayerNorm over last dim
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask if provided (mask may be None)
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Prepare tensors for Triton contraction
    # permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right_T is transposed on the last two dims (k,j)
    right_T = right.permute(0, 3, 2, 1).contiguous()

    # allocate output tensor
    out_perm = torch.empty_like(left_perm, device=device)

    # Strides
    stride_a_batch, stride_a_hidden, stride_a_i, stride_a_k = left_perm.stride()
    stride_b_batch, stride_b_hidden, stride_b_k, stride_b_j = right_T.stride()
    stride_c_batch, stride_c_hidden, stride_c_i, stride_c_j = out_perm.stride()

    M = seq_len
    N_dim = seq_len
    K = seq_len

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_batch_hidden = bs * hidden_dim
    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N_dim + BLOCK_N - 1) // BLOCK_N
    grid = (total_batch_hidden * num_blocks_m * num_blocks_n,)

    # Launch Triton kernel
    matmul_tri_mul_kernel[grid](
        left_perm, right_T, out_perm,
        stride_a_batch, stride_a_hidden, stride_a_i, stride_a_k,
        stride_b_batch, stride_b_hidden, stride_b_k, stride_b_j,
        stride_c_batch, stride_c_hidden, stride_c_i, stride_c_j,
        M, N_dim, K,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        total_batch_hidden=total_batch_hidden,
        num_blocks_m=num_blocks_m,
        num_blocks_n=num_blocks_n,
        HIDDEN_DIM=hidden_dim,
    )

    # Reshape back to original layout
    out = out_perm.permute(0, 2, 3, 1)  # shape [B,N,N,hidden_dim]

    # Second LayerNorm over hidden_dim
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply output gate
    out_gated = out_norm * out_gate

    # Final linear projection to original dim
    final_out = F.linear(out_gated, weights["to_out.weight"])

    return final_out

"""
TriMul (outgoing) implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension (dim).
2. Linear projections (left/right) + linear gates (left_gate, right_gate, out_gate).
   Apply the mask (if any) and the sigmoid gates.
3. Core contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication per hidden dimension.
   Implemented as a custom Triton kernel that
   * processes each (batch, hidden) pair as an independent GEMM,
   * uses tiling (BLOCK_M×BLOCK_N) and loops over K in BLOCK_K,
   * accumulates in float32.
4. Second LayerNorm over the hidden dimension, multiply by out_gate,
   and a final linear projection back to the original dimension.
All steps except the heavy contraction are performed with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_a_hidden, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_hidden, stride_b_k, stride_b_j,
    stride_c_batch, stride_c_hidden, stride_c_i, stride_c_j,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    total_batch_hidden: tl.constexpr,
    num_blocks_m: tl.constexpr, num_blocks_n: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    """Batched GEMM for TriMul.
    Computes C = A @ B where A is (M,K) and B is (K,N) for every
    (batch, hidden) pair.
    """
    pid = tl.program_id(0)

    # split pid into (batch_hidden, block_i, block_j)
    batch_hidden = pid // (num_blocks_m * num_blocks_n)
    block_rem   = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_rem // num_blocks_n
    pid_n = block_rem % num_blocks_n

    # offsets for the output block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # decompose batch_hidden into batch and hidden indices
    batch_idx  = batch_hidden // HIDDEN_DIM
    hidden_idx = batch_hidden % HIDDEN_DIM

    # base pointers for this batch & hidden slice
    a_base = a_ptr + batch_idx * stride_a_batch + hidden_idx * stride_a_hidden
    b_base = b_ptr + batch_idx * stride_b_batch + hidden_idx * stride_b_hidden
    c_base = c_ptr + batch_idx * stride_c_batch + hidden_idx * stride_c_hidden

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A block: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_base + (offs_m[:, None] * stride_a_i) + (offs_k[None, :] * stride_a_k),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B block: (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_base + (offs_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_j),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # write back the result
    tl.store(
        c_base + (offs_m[:, None] * stride_c_i) + (offs_n[None, :] * stride_c_j),
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (float, 0/1) – may be all‑ones.
            - weights: dict of model parameters.
            - config: dict containing "dim" and "hidden_dim".

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ----- 1. Input LayerNorm -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]

    # ----- 2. Linear projections & gating ------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,HD]
    right = F.linear(x, weights["right_proj.weight"])        # [B,N,N,HD]

    # apply mask (broadcast over hidden dim)
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,HD]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B,N,N,HD]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))      # [B,N,N,HD]

    left = left * left_gate
    right = right * right_gate

    # ----- 3. Core contraction (Triton) --------------------------------------
    # Bring hidden dimension to the second axis for easier GEMM tiling
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B,HD,N,N] (i,k)
    # Right needs to be transposed on the last two axes: (k,j)
    right_T = right.permute(0, 3, 2, 1).contiguous()           # [B,HD,N,N] (k,j)

    out_perm = torch.empty_like(left_perm, device=device)      # [B,HD,N,N]

    # strides (in elements)
    stride_a_batch, stride_a_hidden, stride_a_i, stride_a_k = left_perm.stride()
    stride_b_batch, stride_b_hidden, stride_b_k, stride_b_j = right_T.stride()
    stride_c_batch, stride_c_hidden, stride_c_i, stride_c_j = out_perm.stride()

    M = seq_len
    N_dim = seq_len
    K = seq_len

    # tiling parameters (tuned for the B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_batch_hidden = bs * hidden_dim
    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N_dim + BLOCK_N - 1) // BLOCK_N
    grid = (total_batch_hidden * num_blocks_m * num_blocks_n,)

    _tri_mul_matmul_kernel[grid](
        left_perm, right_T, out_perm,
        stride_a_batch, stride_a_hidden, stride_a_i, stride_a_k,
        stride_b_batch, stride_b_hidden, stride_b_k, stride_b_j,
        stride_c_batch, stride_c_hidden, stride_c_i, stride_c_j,
        M, N_dim, K,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        total_batch_hidden=total_batch_hidden,
        num_blocks_m=num_blocks_m,
        num_blocks_n=num_blocks_n,
        HIDDEN_DIM=hidden_dim,
    )

    # reshape back to [B,N,N,HD]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ----- 4. Second LayerNorm + output gate ---------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_gated = out_norm * out_gate

    # ----- 5. Final linear projection ----------------------------------------
    final_out = F.linear(out_gated, weights["to_out.weight"])

    return final_out
