    pid_j = tl.program_id(1)

import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_j, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_d = tl.program_id(2)

    # offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < H

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # number of k-tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k in range(num_k):
        # k offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block (i, k, d)
        left_ptrs = left_ptr + (offs_i[:, None, None] * stride_left_i
                                 + offs_k[None, :, None] * stride_left_k
                                 + offs_d[None, None, :] * stride_left_h)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                       other=0.0,
                       dtype=tl.float16)

        # right block transposed (k, j, d)
        right_ptrs = right_ptr + (offs_k[:, None, None] * stride_right_k
                                  + offs_j[None, :, None] * stride_right_j
                                  + offs_d[None, None, :] * stride_right_h)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
                        other=0.0,
                        dtype=tl.float16)

        # compute dot product for each d
        for d in range(BLOCK_D):
            # left slice (M, K)
            left_slice = left[:, :, d]
            # right slice (K, N)
            right_slice = right[:, :, d]
            # dot returns FP16, cast to FP32
            acc[:, :, d] += tl.dot(left_slice, right_slice).to(tl.float32)

    # Store results
    out_ptrs = out_ptr + (offs_i[:, None, None] * stride_out_i
                          + offs_j[None, :, None] * stride_out_j
                          + offs_d[None, None, :] * stride_out_h)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])
```

"""
TriMul (outgoing) forward pass with Triton‑accelerated cubic multiplication.
The routine follows the reference PyTorch implementation but offloads the
expensive einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
to a custom Triton kernel.  The kernel tiles the (i,j) output plane and a
small block of hidden channels (d) while iterating over the k dimension in
chunks (BLOCK_K).  Inputs are cast to fp16 for bandwidth savings; the
accumulator is fp32 to preserve precision.  All other operations (layer‑norm,
linear projections and gates) are performed with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,               # pointer to left tensor   [i, k, d]
    right_ptr,              # pointer to right tensor  [j, k, d] (stored as [j,k,d])
    out_ptr,                # pointer to output tensor  [i, j, d]
    N,                      # sequence length
    H,                      # hidden dimension
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_j, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_i = tl.program_id(0)   # i tile
    pid_j = tl.program_id(1)   # j tile
    pid_d = tl.program_id(2)   # hidden‑channel tile

    # Offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < H

    # Accumulator for the tile (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)

    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block   shape (M, K, D)
        left_ptrs = left_ptr + (
            offs_i[:, None, None] * stride_left_i
            + offs_k[None, :, None] * stride_left_k
            + offs_d[None, None, :] * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # Load right block transposed (K, N, D) so we can use tl.dot
        right_ptrs = right_ptr + (
            offs_k[:, None, None] * stride_right_k
            + offs_j[None, :, None] * stride_right_j
            + offs_d[None, None, :] * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # Accumulate dot products for each hidden slice
        for d in range(BLOCK_D):
            left_slice = left[:, :, d]      # (M, K)
            right_slice = right[:, :, d]    # (K, N)
            # tl.dot returns fp16; cast to fp32 before accumulation
            acc[:, :, d] += tl.dot(left_slice, right_slice).to(tl.float32)

    # Store the result tile
    out_ptrs = out_ptr + (
        offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + offs_d[None, None, :] * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: Tuple (input, mask, weights, config)
            - input: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim" (C) and "hidden_dim" (H)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1) LayerNorm over the channel dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2) Linear projections
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])    # [B,N,N,H]

    # -----------------------------------------------------------------
    # 3) Apply mask (if provided)
    # -----------------------------------------------------------------
    if mask.dtype != torch.float32:
        mask = mask.to(torch.float32)
    mask_exp = mask.unsqueeze(-1)          # [B,N,N,1]
    left = left * mask_exp
    right = right * mask_exp

    # -----------------------------------------------------------------
    # 4) Gates
    # -----------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Cubic multiplication via Triton
    # -----------------------------------------------------------------
    # Cast to fp16 for the kernel (reduces memory traffic)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape
    out_fp16 = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Tiling parameters (tuned to stay within register limits)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 8

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        triton.cdiv(hidden_dim, BLOCK_D),
    )

    for b in range(B):
        # Sub‑tensors for the current batch element
        l_ptr = left_fp16[b]
        r_ptr = right_fp16[b]
        o_ptr = out_fp16[b]

        # Strides in element units
        stride_l_i, stride_l_k, stride_l_h = l_ptr.stride()
        stride_r_j, stride_r_k, stride_r_h = r_ptr.stride()
        stride_o_i, stride_o_j, stride_o_h = o_ptr.stride()

        tri_mul_kernel[grid](
            l_ptr,
            r_ptr,
            o_ptr,
            N,
            hidden_dim,
            stride_l_i,
            stride_l_k,
            stride_l_h,
            stride_r_j,
            stride_r_k,
            stride_r_h,
            stride_o_i,
            stride_o_j,
            stride_o_h,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )

    # Convert back to float32 for the remaining layers
    out = out_fp16.to(torch.float32)

    # -----------------------------------------------------------------
    # 6) Output layer‑norm and gating
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # -----------------------------------------------------------------
    # 7) Final linear projection back to original dimension
    # -----------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
