@triton.jit
def apply_gate_mask_kernel(
    left_proj_ptr, left_gate_ptr, right_proj_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N, H,
    stride_lproj_b, stride_lproj_i, stride_lproj_j, stride_lproj_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rproj_b, stride_rproj_i, stride_rproj_j, stride_rproj_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lout_b, stride_lout_i, stride_lout_j, stride_lout_h,
    stride_rout_b, stride_rout_i, stride_rout_j, stride_rout_h,
    BLOCK: tl.constexpr
):
   pid = tl.program_id(0)
   total_elements = B * N * N * H
   # compute flat index
   offs = pid * BLOCK + tl.arange(0, BLOCK)
   mask = offs < total_elements
   # compute indices
   # flatten: i = ((flat // H) % N) ... careful
   # we can compute tuple:
   b = offs // (N * N * H)
   rem = offs % (N * N * H)
   i = rem // (N * H)
   rem2 = rem % (N * H)
   j = rem2 // H
   d = rem2 % H

   # compute pointer offsets
   left_proj_ptrs = left_proj_ptr + b*stride_lproj_b + i*stride_lproj_i + j*stride_lproj_j + d*stride_lproj_h
   left_gate_ptrs = left_gate_ptr + b*stride_lgate_b + i*stride_lgate_i + j*stride_lgate_j + d*stride_lgate_h
   right_proj_ptrs = right_proj_ptr + b*stride_rproj_b + i*stride_rproj_i + j*stride_rproj_j + d*stride_rproj_h
   right_gate_ptrs = right_gate_ptr + b*stride_rgate_b + i*stride_rgate_i + j*stride_rgate_j + d*stride_rgate_h
   mask_ptrs = mask_ptr + b*stride_mask_b + i*stride_mask_i + j*stride_mask_j
   # mask scalar broadcast to hidden dim
   mask_val = tl.load(mask_ptrs, mask=mask, other=0.0)  # shape broadcast across d (but each d uses same mask)
   # Actually we need to load mask and broadcast: we can load mask without d dimension. Since each d uses same mask scalar.
   # Use tl.load(...) per element (different d) but same mask index.
   left_proj = tl.load(left_proj_ptrs, mask=mask)
   left_gate = tl.load(left_gate_ptrs, mask=mask)
   right_proj = tl.load(right_proj_ptrs, mask=mask)
   right_gate = tl.load(right_gate_ptrs, mask=mask)

   # apply sigmoid to gates
   left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate))
   right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))

   # apply mask and gates
   left_out = left_proj * mask_val * left_gate_sig
   right_out = right_proj * mask_val * right_gate_sig

   # store
   tl.store(left_out_ptr + b*stride_lout_b + i*stride_lout_i + j*stride_lout_j + d*stride_lout_h, left_out, mask=mask)
   tl.store(right_out_ptr + b*stride_rout_b + i*stride_rout_i + j*stride_rout_j + d*stride_rout_h, right_out, mask=mask)
```

"""
Optimized TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension.
2. Apply four linear projections (left/right and their gates) to the
   normalized tensor.
3. Fuse mask, sigmoid gating and elementwise multiplication in a
   Triton kernel (one thread per output element of shape
   [B, N, N, hidden_dim]).
4. Compute the O(N³) contraction using a batched matmul:
   out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d].
   This is equivalent to a batch of hidden_dim independent matrix
   multiplications.
5. Layer‑norm the result over the hidden dimension, multiply by the
   output gate and apply the final linear projection back to the
   original dimension.
All heavy arithmetic is performed in float32 and the final output
has shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: fuse mask, sigmoid(gate) and elementwise multiplication.
# --------------------------------------------------------------------------- #
@triton.jit
def apply_gate_mask_kernel(
    left_proj_ptr,   # float*  shape [B,N,N, H]
    left_gate_ptr,   # float*  shape [B,N,N, H]
    right_proj_ptr,  # float*  shape [B,N,N, H]
    right_gate_ptr,  # float*  shape [B,N,N, H]
    mask_ptr,        # float*  shape [B,N,N]
    left_out_ptr,    # float*  shape [B,N,N, H] (output)
    right_out_ptr,   # float*  shape [B,N,N, H] (output)
    B, N, H,         # int32 sizes
    # strides for left_proj
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,
    # strides for left_gate
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    # strides for right_proj
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_h,
    # strides for right_gate
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    # strides for mask (no hidden dim)
    stride_m_b, stride_m_i, stride_m_j,
    # strides for left_out
    stride_lo_b, stride_lo_i, stride_lo_j, stride_lo_h,
    # strides for right_out
    stride_ro_b, stride_ro_i, stride_ro_j, stride_ro_h,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # ------- decode flat index into (b,i,j,d) -------
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    d = rem2 % H

    # ------- compute raw pointers -------
    lp_ptr = left_proj_ptr + b * stride_lp_b + i * stride_lp_i + j * stride_lp_j + d * stride_lp_h
    lg_ptr = left_gate_ptr + b * stride_lg_b + i * stride_lg_i + j * stride_lg_j + d * stride_lg_h
    rp_ptr = right_proj_ptr + b * stride_rp_b + i * stride_rp_i + j * stride_rp_j + d * stride_rp_h
    rg_ptr = right_gate_ptr + b * stride_rg_b + i * stride_rg_i + j * stride_rg_j + d * stride_rg_h
    m_ptr  = mask_ptr + b * stride_m_b + i * stride_m_i + j * stride_m_j   # no hidden dim offset

    # ------- load values -------
    lp = tl.load(lp_ptr, mask=mask, other=0.0)
    lg = tl.load(lg_ptr, mask=mask, other=0.0)
    rp = tl.load(rp_ptr, mask=mask, other=0.0)
    rg = tl.load(rg_ptr, mask=mask, other=0.0)
    m  = tl.load(m_ptr,  mask=mask, other=0.0)

    # ------- sigmoid(gate) -------
    lg_sig = 1.0 / (1.0 + tl.exp(-lg))
    rg_sig = 1.0 / (1.0 + tl.exp(-rg))

    # ------- apply mask and gated scaling -------
    left_out  = lp * lg_sig * m
    right_out = rp * rg_sig * m

    # ------- store results -------
    lo_ptr = left_out_ptr + b * stride_lo_b + i * stride_lo_i + j * stride_lo_j + d * stride_lo_h
    ro_ptr = right_out_ptr + b * stride_ro_b + i * stride_ro_i + j * stride_ro_j + d * stride_ro_h
    tl.store(lo_ptr, left_out,  mask=mask)
    tl.store(ro_ptr, right_out, mask=mask)


# --------------------------------------------------------------------------- #
# Entry point called by the evaluation harness.
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton
    kernel for the masked gated projection step.
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------- #
    # 1. LayerNorm on input
    # ------------------------------------------------------------------- #
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left_proj   = F.linear(x_norm, weights["left_proj.weight"])
    right_proj  = F.linear(x_norm, weights["right_proj.weight"])
    left_gate   = F.linear(x_norm, weights["left_gate.weight"])
    right_gate  = F.linear(x_norm, weights["right_gate.weight"])
    out_gate    = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # ------------------------------------------------------------------- #
    # 3. Prepare mask (float) – if no mask, use all‑ones
    # ------------------------------------------------------------------- #
    if mask is None:
        mask = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask = mask.to(dtype=dtype, device=device)

    # ------------------------------------------------------------------- #
    # 4. Fuse mask + sigmoid(gate) + elementwise multiplication via Triton
    # ------------------------------------------------------------------- #
    left_out  = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    total_elements = B * N * N * hidden_dim
    BLOCK = 128
    grid = (triton.cdiv(total_elements, BLOCK),)

    apply_gate_mask_kernel[grid](
        left_proj, left_gate,
        right_proj, right_gate,
        mask,
        left_out, right_out,
        B, N, hidden_dim,
        # strides for left_proj
        left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
        # strides for left_gate
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        # strides for right_proj
        right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
        # strides for right_gate
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        # strides for mask
        mask.stride(0), mask.stride(1), mask.stride(2),
        # strides for left_out
        left_out.stride(0), left_out.stride(1), left_out.stride(2), left_out.stride(3),
        # strides for right_out
        right_out.stride(0), right_out.stride(1), right_out.stride(2), right_out.stride(3),
        BLOCK=BLOCK,
    )

    # free unused tensors to save memory (optional)
    del left_proj, right_proj, left_gate, right_gate

    # ------------------------------------------------------------------- #
    # 5. O(N³) contraction via batched matrix multiplication
    #    left_out[b,i,k,d]  -> [B, H, N, N]
    #    right_out[b,j,k,d] -> [B, H, N, N]
    #    out = left @ rightᵀ over the k dimension.
    # ------------------------------------------------------------------- #
    left_perm  = left_out.permute(0, 3, 1, 2)   # [B, H, N, N]
    right_perm = right_out.permute(0, 3, 1, 2)  # [B, H, N, N]

    # batched matmul: (B*H, N, N) @ (B*H, N, N).T  -> (B*H, N, N)
    out_perm = torch.matmul(left_perm, right_perm.transpose(-1, -2))
    out = out_perm.permute(0, 2, 3, 1)  # back to [B, N, N, H]

    # ------------------------------------------------------------------- #
    # 6. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------- #
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate
    output = F.linear(out_norm, weights["to_out.weight"])

    # ensure float32 output
    return output.to(torch.float32)
