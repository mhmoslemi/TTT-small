"""
TriMul "outgoing" implementation with a custom Triton kernel for the
heavy O(N³·hidden_dim) reduction.

The kernel computes, for each (batch, hidden) pair:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

All other operations (LayerNorms, Linear projections, gates) are
performed in PyTorch. The kernel works on a flattened batch×hidden_dim
dimension to reuse the classic tiled GEMM pattern.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (left @ right.T)
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,                 # *float32   – left matrix (batch_h, M, K)
    right_ptr,                # *float32   – right matrix transposed (batch_h, K, N)
    out_ptr,                  # *float32   – output matrix (batch_h, M, N)

    batch_h, M, N, K,         # int32   – dimensions
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_i, stride_out_j,

    BLOCK_M: tl.constexpr,    # compile‑time block sizes
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)                 # batch * hidden dimension
    pid_i = tl.program_id(0)                 # block row
    pid_j = tl.program_id(1)                 # block col

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # predicates for boundary checks
    mask_i = offs_i < M
    mask_j = offs_j < N

    # accumulator in FP32 (more accurate)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k = tl.cdiv(K, BLOCK_K)

    for k in range(num_k):
        # global offset for this K‑tile
        k_offset = k * BLOCK_K
        k_idx = k_offset + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        # pointers for the current tile
        a_ptr = (left_ptr
                 + pid_b * stride_left_batch
                 + offs_i[:, None] * stride_left_i
                 + k_idx[None, :] * stride_left_k)

        b_ptr = (right_ptr
                 + pid_b * stride_right_batch
                 + k_idx[:, None] * stride_right_k
                 + offs_j[None, :] * stride_right_n)

        # load tiles with masks
        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # accumulate dot product
        acc += tl.dot(a, b)

    # store result
    c_ptr = (out_ptr
             + pid_b * stride_out_batch
             + offs_i[:, None] * stride_out_i
             + offs_j[None, :] * stride_out_j)

    tl.store(c_ptr, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module (outgoing version).
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------
    # Extract configuration
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Helper: load weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias) and gating
    # ------------------------------------------------------------------
    # shape: [B, N, N, hidden_dim]
    left = F.linear(x, left_proj_weight)      # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)

    # mask handling – mask is [B, N, N]; broadcast to the channel dim
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Heavy reduction performed by Triton: out = Σ_k left[...,k,:] * right[...,k,:]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # N is seq_len
    batch_hidden = B * hidden_dim

    # reshape to (batch_hidden, N, N) and make them contiguous for Triton
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(batch_hidden, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(batch_hidden, N, N)

    # right matrix transposed on the last two dims -> (batch_hidden, K, N)
    right_mat_T = right_mat.transpose(1, 2).contiguous()

    out_mat = torch.empty_like(left_mat)

    # strides in element units (not bytes)
    stride_left_batch, stride_left_i, stride_left_k = left_mat.stride()
    stride_right_batch, stride_right_k, stride_right_n = right_mat_T.stride()
    stride_out_batch, stride_out_i, stride_out_j = out_mat.stride()

    # block sizes – tuned for typical B200 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),      # blocks over i
        triton.cdiv(N, BLOCK_N),      # blocks over j
        batch_hidden,                 # one program per (batch, hidden) pair
    )

    trimul_kernel[grid](
        left_mat,
        right_mat_T,
        out_mat,
        batch_hidden,
        N, N, N,                     # M, N, K are all equal to seq_len
        stride_left_batch, stride_left_i, stride_left_k,
        stride_right_batch, stride_right_k, stride_right_n,
        stride_out_batch, stride_out_i, stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # Final LayerNorm, gating and output projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # (B,N,N,dim)

    return out
