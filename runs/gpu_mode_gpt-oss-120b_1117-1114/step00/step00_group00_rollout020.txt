```
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bn,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0) # rows block
    pid_n = tl.program_id(1) # cols block
    pid_z = tl.program_id(2) # batch block

    # Compute offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Determine valid masks for i, j
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < K  # but K==N

    # Compute base pointers for the batch
    a_batch_ptr = a_ptr + pid_z * stride_am
    b_batch_ptr = b_ptr + pid_z * stride_bm
    c_batch_ptr = c_ptr + pid_z * stride_cm

    # Initialize accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k_tile * BLOCK_K + offs_k

        # Load A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_i[:, None] * stride_ai) + (k_offset[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & (k_offset[None, :] < K), other=0.0)

        # Load B: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch_ptr + (k_offset[:, None] * stride_bk) + (offs_j[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=(k_offset[:, None] < K) & mask_j[None, :], other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_batch_ptr + (offs_i[:, None] * stride_ci) + (offs_j[None, :] * stride_cj)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

    right_mat = right_perm.view(M, N, N)
    out_mat = torch.empty_like(left_mat)

    # Strides
    stride_am = left_mat.stride(0)
    stride_ai = left_mat.stride(1)

    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), M)

        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8
    )

    # Reshape back
    out = out_mat.view(B, H, N, N).permute(0,2,3,1)  # shape (B,N,N,H)

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (H,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply out_gate
    out = out * out_gate

    # Final linear
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul (outgoing) forward implementation with a Triton kernel for the
core N³ contraction. The heavy work ``out[b,i,j,d] = sum_k left[b,i,k,d] *
right[b,j,k,d]`` is performed as a batched GEMM (B*hidden_dim batches,
each of size N×N) inside Triton. All other operations (LayerNorm,
Linear projections, gating) are handled with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bn,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C[m,i,j] = sum_k A[m,i,k] * B[m,k,j]"""
    pid_i = tl.program_id(0)          # block row
    pid_j = tl.program_id(1)          # block col
    pid_b = tl.program_id(2)          # batch (M = B*hidden_dim)

    # Offsets for this block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for boundary conditions
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers to the start of the current batch
    a_batch = a_ptr + pid_b * stride_am
    b_batch = b_ptr + pid_b * stride_bm
    c_batch = c_ptr + pid_b * stride_cm

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for tk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = tk * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile [BLOCK_M, BLOCK_K]
        a_ptrs = a_batch + (offs_i[:, None] * stride_ai) + (offs_k[None, :] * stride_ak)
        a_mask = mask_i[:, None] & (offs_k[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # Load B tile [BLOCK_K, BLOCK_N]
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk) + (offs_j[None, :] * stride_bn)
        b_mask = (offs_k[:, None] < K) & mask_j[None, :]
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Store results
    c_ptrs = c_batch + (offs_i[:, None] * stride_ci) + (offs_j[None, :] * stride_cj)
    c_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) operation.
    Arguments:
        data = (input_tensor, mask, weights, config)
            - input_tensor: Tensor[bs, N, N, dim] (cuda)
            - mask        : Tensor[bs, N, N] (cuda) – may be all‑ones
            - weights     : dict of weight tensors (cuda)
            - config      : dict with keys "dim" and "hidden_dim"
    Returns:
        Tensor[bs, N, N, dim] (cuda, float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, _ = input_tensor.shape

    # ---------- Load weights ----------
    norm_weight = weights["norm.weight"].to(device=device, dtype=dtype)
    norm_bias = weights["norm.bias"].to(device=device, dtype=dtype)
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=dtype)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=dtype)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=dtype)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=dtype)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype)

    # ---------- LayerNorm on input ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ---------- Linear projections ----------
    left = F.linear(x, left_proj_w)          # (bs, N, N, hidden_dim)
    right = F.linear(x, right_proj_w)       # (bs, N, N, hidden_dim)

    # ---------- Apply mask ----------
    if mask is not None:
        mask = mask.to(dtype=dtype).unsqueeze(-1)  # (bs, N, N, 1)
        left = left * mask
        right = right * mask

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ---------- Core contraction (batched GEMM) ----------
    # left: (bs, N, N, hidden_dim)  -> (bs, hidden_dim, N, N)
    # right: (bs, N, N, hidden_dim) -> (bs, hidden_dim, N, N) with k<->j swapped
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (bs, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (bs, hidden, k, j)

    M = bs * hidden_dim                     # total batched GEMMs
    left_mat = left_perm.view(M, N, N)      # (M, i, k)
    right_mat = right_perm.view(M, N, N)    # (M, k, j)
    out_mat = torch.empty_like(left_mat)    # (M, i, j)

    # Strides (in elements)
    stride_am = left_mat.stride(0)
    stride_ai = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_bm = right_mat.stride(0)
    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)

    stride_cm = out_mat.stride(0)
    stride_ci = out_mat.stride(1)
    stride_cj = out_mat.stride(2)

    # Block sizes – tuned for the B200 architecture
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        M,
    )
    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to (bs, N, N, hidden_dim)
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1)  # (bs,N,N,hidden)

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )

    # ---------- Gating and final linear ----------
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (bs, N, N, dim)

    return out
