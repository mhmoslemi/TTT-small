import torch
import triton
import triton.language as tl

@triton.jit
def apply_mask_gate_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr, out_left_ptr, out_right_ptr,
    B, N, hidden_dim,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    BLOCK_H: tl.constexpr
):
    """
    Fuse mask multiplication and gating for the left and right projected tensors.
    For each (b,i,k) position we apply:
        left_out  = left  * mask[b,i,k] * left_gate
        right_out = right * mask[b,i,k] * right_gate
    The hidden dimension is processed in blocks of size BLOCK_H.
    """
    pid = tl.program_id(0)       # flattened index over (b,i,k)
    hid_block = tl.program_id(1) # block index over hidden dimension

    # -------------------------------------------------
    # Decode linear indices
    # -------------------------------------------------
    # total positions per batch = N * N
    total_n2 = N * N
    b = pid // total_n2
    rem = pid % total_n2
    i = rem // N
    k = rem % N

    # -------------------------------------------------
    # Load mask scalar (float)
    # -------------------------------------------------
    mask_offset = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k
    m = tl.load(mask_ptr + mask_offset)

    # -------------------------------------------------
    # Hidden‑dim block indices
    # -------------------------------------------------
    offs_h = hid_block * BLOCK_H + tl.arange(0, BLOCK_H)
    mask_h = offs_h < hidden_dim

    # Base pointers for the four tensors (without hidden offset)
    base_left = left_ptr + b * stride_left_b + i * stride_left_i + k * stride_left_k
    base_right = right_ptr + b * stride_right_b + i * stride_right_i + k * stride_right_k
    base_lgate = left_gate_ptr + b * stride_lg_b + i * stride_lg_i + k * stride_lg_k
    base_rgate = right_gate_ptr + b * stride_rg_b + i * stride_rg_i + k * stride_rg_k

    # -------------------------------------------------------------------------
    # Load vectors for this hidden block
    # -------------------------------------------------------------------------
    left_val = tl.load(base_left + offs_h * stride_left_h, mask=mask_h, other=0.0)
    right_val = tl.load(base_right + offs_h * stride_right_h, mask=mask_h, other=0.0)
    left_gate_val = tl.load(base_lgate + offs_h * stride_lg_h, mask=mask_h, other=0.0)
    right_gate_val = tl.load(base_rgate + offs_h * stride_rg_h, mask=mask_h, other=0.0)

    # -------------------------------------------------------------------------
    # Apply mask and gate
    # -------------------------------------------------------------------------
    left_out = left_val * m * left_gate_val
    right_out = right_val * m * right_gate_val

    # -------------------------------------------------------------------------
    # Store results
    # -------------------------------------------------------------------------
    tl.store(out_left_ptr + b * stride_left_b + i * stride_left_i + k * stride_left_k + offs_h * stride_left_h,
             left_out, mask=mask_h)
    tl.store(out_right_ptr + b * stride_right_b + i * stride_right_i + k * stride_right_k + offs_h * stride_right_h,
             right_out, mask=mask_h)

    # -----------------------------------------------------------------

    norm_bias = weights["norm.bias"].to(dtype).to(device)

    right_proj_weight = weights["right_proj.weight"].to(dtype).to(device)

    left_gate_weight = weights["left_gate.weight"].to(dtype).to(device)
    right_gate_weight = weights["right_gate.weight"].to(dtype).to(device)
    out_gate_weight = weights["out_gate.weight"].to(dtype).to(device)

    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype).to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(dtype).to(device)

    to_out_weight = weights["to_out.weight"].to(dtype).to(device)

    # -----------------------------------------------------------------
    # LayerNorm over input dim
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # Linear projections and gates (no bias)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight).contiguous()
    right = torch.nn.functional.linear(x, right_proj_weight).contiguous()

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight)).contiguous()
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight)).contiguous()
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight)).contiguous()

    # -----------------------------------------------------------------
    # Prepare mask (broadcast to [B,N,N,1])
    # -----------------------------------------------------------------
    if nomask:
        mask_tensor = torch.ones_like(mask, dtype=dtype, device=device)
    else:
        mask_tensor = mask.to(dtype).to(device)

    # -----------------------------------------------------------------
    # Triton kernel: apply mask and gates in one fused pass
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape

    # Allocate output buffers
    left_fused = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    # Strides (in elements)
    l_str = left.stride()
    r_str = right.stride()
    lg_str = left_gate.stride()
    rg_str = right_gate.stride()
    m_str = mask_tensor.stride()

    # Kernel launch configuration
    BLOCK_H = 32
    total_positions = B * N * N
    num_h_blocks = (hidden_dim + BLOCK_H - 1) // BLOCK_H
    grid = (total_positions, num_h_blocks)

    apply_mask_gate_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_tensor,
        out_left_ptr=left_fused,
        out_right_ptr=right_fused,
        B=int(B),
        N=int(N),
        hidden_dim=int(hidden_dim),
        stride_left_b=int(l_str[0]),
        stride_left_i=int(l_str[1]),
        stride_left_k=int(l_str[2]),
        stride_left_h=int(l_str[3]),
        stride_right_b=int(r_str[0]),
        stride_right_i=int(r_str[1]),
        stride_right_k=int(r_str[2]),
        stride_right_h=int(r_str[3]),
        stride_lg_b=int(lg_str[0]),
        stride_lg_i=int(lg_str[1]),
        stride_lg_k=int(lg_str[2]),
        stride_lg_h=int(lg_str[3]),
        stride_rg_b=int(rg_str[0]),
        stride_rg_i=int(rg_str[1]),
        stride_rg_k=int(rg_str[2]),
        stride_rg_h=int(rg_str[3]),
        stride_mask_b=int(m_str[0]),
        stride_mask_i=int(m_str[1]),
        stride_mask_k=int(m_str[2]),
        BLOCK_H=BLOCK_H,
    )

    # -----------------------------------------------------------------
    # Triangular update via batched matrix multiplication
    # -----------------------------------------------------------------
    # reshape to (B*hidden_dim, N, N) for bmm
    left_mat = left_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # out_mat[b*H + h] = left_mat @ right_mat.T   (shape B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, hidden_dim)
    out = out_mat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # -----------------------------------------------------------------
    # Final layer‑norm, out‑gate, and projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) kernel.

The forward pass consists of:
1. Layer‑norm over the last channel of the input.
2. Two linear projections (left/right) and three gating projections.
3. A Triton kernel that fuses:
       left  = left  * mask * left_gate
       right = right * mask * right_gate
   The hidden dimension is processed in BLOCK_H‑sized tiles.
4. A batched matrix multiplication implements the core Einstein sum:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is realized as a batch of (N×N) @ (N×N)ᵀ products for every
   hidden channel.
5. Second Layer‑norm, out‑gate multiplication, and final linear projection
   back to the original channel dimension.
All operations are performed in float32; the heavy mask‑gate fusion is
accelerated with Triton.
"""

    Hidden dimension is processed in blocks of size BLOCK_H.

        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_tensor,
        out_left_ptr=left_fused,
        out_right_ptr=right_fused,
        B=int(B),
        N=int(N),
        hidden_dim=int(hidden_dim),
        stride_left_b=int(l_str[0]),
        stride_left_i=int(l_str[1]),
        stride_left_k=int(l_str[2]),
        stride_left_h=int(l_str[3]),
        stride_right_b=int(r_str[0]),
        stride_right_i=int(r_str[1]),
        stride_right_k=int(r_str[2]),
        stride_right_h=int(r_str[3]),
        stride_lg_b=int(lg_str[0]),
        stride_lg_i=int(lg_str[1]),
        stride_lg_k=int(lg_str[2]),
        stride_lg_h=int(lg_str[3]),
        stride_rg_b=int(rg_str[0]),
        stride_rg_i=int(rg_str[1]),
        stride_rg_k=int(rg_str[2]),
        stride_rg_h=int(rg_str[3]),
        stride_mask_b=int(m_str[0]),
        stride_mask_i=int(m_str[1]),
        stride_mask_k=int(m_str[2]),
        BLOCK_H=BLOCK_H,
    )

    # -------------------------------------------------
    # 5) Triangular update via batched matrix multiplication
    # -------------------------------------------------
    # reshape to (B*hidden_dim, N, N)
    left_mat = left_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # out_mat[b*H + h] = left_mat @ right_mat.T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to (B, N, N, hidden_dim)
    out = out_mat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # -------------------------------------------------
    # 6) Second LayerNorm, out‑gate, final projection
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)

    return out
