import torch
import triton
import triton.language as tl

def layer_norm(x, weight, bias, eps=1e-5):
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias
```

@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # compute per program id (global block)
    pid = tl.program_id(0)
    # total blocks per batch
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    total_blocks_per_batch = num_blocks_m * num_blocks_n
    batch_idx = pid // total_blocks_per_batch
    block_idx = pid % total_blocks_per_batch
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute mask for boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask_k for K dim
        mask_k = offs_k < K

        # compute pointer offsets
        a_ptrs = A_ptr + batch_idx * stride_az + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = B_ptr + batch_idx * stride_bz + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # convert to fp32 for accumulation if needed
        a_fp = a.to(tl.float32)
        b_fp = b.to(tl.float32)

        acc += tl.dot(a_fp, b_fp)

    # write back
    c_ptrs = C_ptr + batch_idx * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Ensure weights are on same device and dtype
    def to_device(t):
        return t.to(device)

    norm_weight = to_device(norm_weight)
    norm_bias = to_device(norm_bias)
    left_proj_weight = to_device(left_proj_weight)
    right_proj_weight = to_device(right_proj_weight)
    left_gate_weight = to_device(left_gate_weight)
    right_gate_weight = to_device(right_gate_weight)
    out_gate_weight = to_device(out_gate_weight)
    to_out_norm_weight = to_device(to_out_norm_weight)
    to_out_norm_bias = to_device(to_out_norm_bias)
    to_out_weight = to_device(to_out_weight)

    # LayerNorm on input
    # x shape [B, N, N, dim]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # Linear projections (no bias)
    left = F.linear(x_norm, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_weight)

    # Apply mask
    mask_unsq = mask_tensor.unsqueeze(-1).to(left.dtype)
    left = left * mask_unsq
    right = right * mask_unsq

    # Gates
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to bfloat16 for GEMM
    left_bf16 = left.to(torch.bfloat16)
    right_bf16 = right.to(torch.bfloat16)

    # Prepare transposed right for GEMM
    # right_t shape (B, hidden_dim, N, N) where K dimension is row index (k) and N dimension is j
    right_t = right_bf16.permute(0, 3, 2, 1).contiguous()

    # Permute left to (B, hidden_dim, N, N)
    left_perm = left_bf16.permute(0, 3, 1, 2).contiguous()

    B, N, _, H = left.shape
    # flatten batch and hidden_dim
    batch_h = B * H
    left_reshape = left_perm.view(batch_h, N, N)
    right_reshape = right_t.view(batch_h, N, N)

    # output buffer
    out_reshape = torch.empty_like(left_reshape, dtype=torch.bfloat16)

    # strides in elements
    stride_az = left_reshape.stride(0)
    stride_am = left_reshape.stride(1)
    stride_ak = left_reshape.stride(2)

    stride_bz = right_reshape.stride(0)
    stride_bk = right_reshape.stride(1)
    stride_bn = right_reshape.stride(2)

    stride_cz = out_reshape.stride(0)
    stride_cm = out_reshape.stride(1)
    stride_cn = out_reshape.stride(2)

    # block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # compute grid
    num_blocks_m = triton.cdiv(N, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)
    total_blocks = batch_h * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    batched_gemm_kernel[grid](
        left_reshape, right_reshape, out_reshape,
        stride_az, stride_am, stride_ak,
        stride_bz, stride_bk, stride_bn,
        stride_cz, stride_cm, stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape out back to (B,N,N,H)
    out = out_reshape.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    # convert to float32 for subsequent operations
    out = out.to(torch.float32)

    # to_out_norm (layernorm on hidden dim)
    mean_out = out.mean(dim=-1, keepdim=True)
    var_out = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_out) * torch.rsqrt(var_out + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # Apply out_gate
    out_norm = out_norm * out_gate

    # Final linear projection
    output = F.linear(out_norm, to_out_weight)  # shape [B,N,N,dim]

    return output

"""
custom_kernel: Triton‑accelerated implementation of the "outgoing" TriMul operator.

Algorithm
---------
1. Layer‑normalize the 4‑D input tensor (dim = hidden feature size).
2. Compute left/right projections and three gating vectors with linear layers.
   Apply the (optional) mask, then multiply by the gating vectors.
3. Cast the gated tensors to float16 and evaluate the core O(N³) contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is equivalent to a batched matrix‑multiply:
        left_perm  = left.permute(0,3,1,2)   # (B,H,N,N)  -> (i,k)
        right_t    = right.permute(0,3,2,1)  # (B,H,N,N)  -> (k,j)
        out = batch_matmul(left_perm, right_t)   # (B,H,N,N)
   The matmul is performed by a custom Triton kernel that tiles the
   multiplication (BLOCK_M×BLOCK_N×BLOCK_K) and accumulates in fp32.
4. Convert the result back to fp32, apply a second LayerNorm, multiply by
   the output gate, and finally project back to the original dimension.
All heavy compute (the N³ contraction) runs on the GPU via Triton, while
the surrounding linear & norm operations stay in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C[b] = A[b] @ B[b]  (A: M×K, B: K×N) per batch."""
    pid = tl.program_id(0)

    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    batch_idx = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to the current tile
        a_ptrs = A_ptr + batch_idx * stride_az + \
                 offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = B_ptr + batch_idx * stride_bz + \
                 offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load with bounds checks
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0, dtype=tl.float16)

        # Compute block‑dot and accumulate in fp32
        acc += tl.dot(a, b).to(tl.float32)

    # Write result
    c_ptrs = C_ptr + batch_idx * stride_cz + \
             offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul module, using Triton for the O(N³) core.
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (broadcastable)
            - weights_dict: dict of model weights (tensors already on CUDA)
            - config_dict: contains "dim" and "hidden_dim"
    Returns:
        torch.Tensor: output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------ #
    # Unpack inputs and move everything to the correct device / dtype
    # ------------------------------------------------------------------ #
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Helper to ensure tensors live on the same device
    def to_dev(t):
        return t.to(device)

    # ------------------------------------------------------------------ #
    # Load model parameters
    # ------------------------------------------------------------------ #
    norm_weight = to_dev(weights["norm.weight"])
    norm_bias = to_dev(weights["norm.bias"])

    left_proj_weight = to_dev(weights["left_proj.weight"])
    right_proj_weight = to_dev(weights["right_proj.weight"])

    left_gate_weight = to_dev(weights["left_gate.weight"])
    right_gate_weight = to_dev(weights["right_gate.weight"])
    out_gate_weight = to_dev(weights["out_gate.weight"])

    to_out_norm_weight = to_dev(weights["to_out_norm.weight"])
    to_out_norm_bias = to_dev(weights["to_out_norm.bias"])
    to_out_weight = to_dev(weights["to_out.weight"])

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------ #
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_weight + norm_bias                                 # [B,N,N,dim]

    # ------------------------------------------------------------------ #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------ #
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------ #
    # 3. Masking (mask may be all‑ones)
    # ------------------------------------------------------------------ #
    mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # [B,N,N,1]
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------ #
    # 4. Gating
    # ------------------------------------------------------------------ #
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5. Core O(N³) contraction via Triton
    # ------------------------------------------------------------------ #
    # Cast to float16 for the kernel (half‑precision arithmetic)
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # Re‑arrange tensors so that we can treat each (batch, hidden) pair as a
    # separate GEMM: left_perm [B,H,N,N] (i,k) , right_t [B,H,N,N] (k,j)
    left_perm = left_f16.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_t   = right_f16.permute(0, 3, 2, 1).contiguous()  # (B,H,N,N)

    B, N, _, H = left.shape
    batch_h = B * H

    # Flatten batch & hidden dimensions
    left_mat = left_perm.view(batch_h, N, N)
    right_mat = right_t.view(batch_h, N, N)

    # Output buffer
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # Strides in number of elements (not bytes)
    stride_az = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_bz = right_mat.stride(0)
    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)

    stride_cz = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    # Tunable tile sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: one program per (batch_h, tile_m, tile_n)
    num_blocks_m = triton.cdiv(N, BLOCK_M)
    num_blocks_n = triton.cdiv(N, BLOCK_N)
    total_blocks = batch_h * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    _batched_gemm_kernel[grid](
        left_mat, right_mat, out_mat,
        stride_az, stride_am, stride_ak,
        stride_bz, stride_bk, stride_bn,
        stride_cz, stride_cm, stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim] and promote to fp32
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------ #
    # 6. Output LayerNorm, output gate and final projection
    # ------------------------------------------------------------------ #
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) * torch.rsqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    out_norm = out_norm * out_gate
    output = F.linear(out_norm, to_out_weight)   # [B,N,N,dim]

    return output
