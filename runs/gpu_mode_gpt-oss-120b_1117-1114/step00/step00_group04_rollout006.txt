    # x, gate: [B,N,N,H] ; mask: [B,N,N]
    assert x.is_cuda and gate.is_cuda and mask.is_cuda

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul module (outgoing version).
    
    Arguments:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
        
    Returns:
        output tensor of shape [batch, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = torch.float32
    # Cast inputs to float32
    x = input_tensor.to(dtype)
    mask = mask_tensor.to(dtype)
    
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    
    if nomask:
        # ignore provided mask, use all ones
        mask = torch.ones_like(mask, dtype=dtype, device=device)
    
    # LayerNorm on the last dimension
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)
    
    # Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)
    
    # Gating weights
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))
    
    # Apply mask and gating using Triton kernel
    # mask has shape [B,N,N]; broadcast in kernel
    left = apply_gate_mask(left, left_gate, mask)
    right = apply_gate_mask(right, right_gate, mask)
    
    # Core multiplicative update: batched matmul across hidden dimension
    # Permute to [B, hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B = left_perm.shape[0]
    H = left_perm.shape[1]
    N = left_perm.shape[2]
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)
    # Matrix multiplication: (i,k) * (j,k) -> out[i,j]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out_perm = out_mat.view(B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # [B,N,N,hidden_dim]
    
    # Output layer norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)
    
    # Apply output gate
    out = out * out_gate
    
    # Final linear projection back to dim
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)
    return out

"""
TriMul (outgoing) forward pass with partial Triton acceleration.

The module processes an input tensor x ∈ ℝ^{B×N×N×C} with
 - LayerNorm over C,
 - Linear projections to a hidden dimension H,
 - Masking and sigmoid gating,
 - Core update: out_{b,i,j,d} = Σ_k left_{b,i,k,d} · right_{b,j,k,d},
   implemented as a batched matrix multiplication (leveraging cuBLAS),
 - LayerNorm over the hidden dimension, output gating,
 - Final linear projection back to C.

The element‑wise mask‑and‑gate multiplication is performed by a custom
Triton kernel (apply_gate_mask_kernel), which avoids materialising a
broadcasted mask of shape [B,N,N, H] and works directly on the flattened
tensor. All other operations use standard PyTorch (torch.nn.functional).

The implementation respects the required signature `custom_kernel(data)` and
runs entirely on the GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication by gate and mask (broadcasted).
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    x_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elems,                     # total number of elements in x/gate (B*N*N*H)
    hidden_dim: tl.constexpr,        # H
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets
    mask = offs < total_elems                         # boundary mask

    # Load inputs
    x = tl.load(x_ptr + offs, mask=mask)
    g = tl.load(gate_ptr + offs, mask=mask)

    # Compute the index for the mask (shape B,N,N) by integer division
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply mask and gate
    out = x * g * m

    # Store
    tl.store(out_ptr + offs, out, mask=mask)


def apply_gate_mask(x: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Element‑wise: out = x * gate * mask, where mask has shape [B, N, N] and is
    broadcasted over the hidden dimension.
    """
    assert x.is_cuda and gate.is_cuda and mask.is_cuda
    total = x.numel()
    out = torch.empty_like(x)
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    apply_gate_mask_kernel[grid](
        x,                # x_ptr
        gate,             # gate_ptr
        mask,             # mask_ptr (no extra dim)
        out,              # out_ptr
        total,            # total_elems
        hidden_dim=x.shape[-1],
        BLOCK=BLOCK
    )
    return out


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module (outgoing version) using Triton for the
    mask‑gate element‑wise multiplication.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask_tensor:  torch.Tensor of shape [B, N, N] (bool or float)
            - weights_dict: mapping str → torch.Tensor containing all model weights
            - config_dict:  contains at least "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # Unpack
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    # Config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Cast inputs
    x = input_tensor.to(dtype)
    mask = mask_tensor.to(dtype)

    if nomask:
        mask = torch.ones_like(mask, dtype=dtype, device=device)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the channel dimension C
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4) Apply mask and gates (Triton kernel)
    # ------------------------------------------------------------------
    # Ensure contiguity for Triton
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask = mask.contiguous()                     # shape [B,N,N]

    left = apply_gate_mask(left, left_gate, mask)   # left = left * left_gate * mask
    right = apply_gate_mask(right, right_gate, mask)  # right = right * right_gate * mask

    # ------------------------------------------------------------------
    # 5) Core multiplicative update: batched matrix multiplication
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # Rearrange to [B, H, N, N] for efficient batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B,H,N,N]

    B = left_perm.shape[0]
    H = left_perm.shape[1]
    N = left_perm.shape[2]

    # Fold B and H into a single batch dimension for torch.bmm
    left_mat = left_perm.view(B * H, N, N)               # [B*H, N, N]
    right_mat = right_perm.view(B * H, N, N)             # [B*H, N, N]

    # Matrix multiplication over the K dimension
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N]

    # Reshape back to [B, N, N, H]
    out_perm = out_mat.view(B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()      # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6) Output LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 7) Output gating
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to original dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
