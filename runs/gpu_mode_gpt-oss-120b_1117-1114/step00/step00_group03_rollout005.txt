@triton.jit
def batched_gemm_kernel(A_ptr, B_ptr, C_ptr,
                        stride_ah, stride_am, stride_ak,
                        stride_bh, stride_bk, stride_bn,
                        stride_ch, stride_cm, stride_cn,
                        M, N, K,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_outer = tl.program_id(0)  # outer index
    pid_tile = tl.program_id(1)   # tile index over output matrix
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid_tile // num_pid_n
    pid_n = pid_tile % num_pid_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for this outer batch
    A_batch_ptr = A_ptr + pid_outer * stride_ah
    B_batch_ptr = B_ptr + pid_outer * stride_bh
    C_batch_ptr = C_ptr + pid_outer * stride_ch

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = A_batch_ptr + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        b_ptrs = B_batch_ptr + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

        # load fragments

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # compute dot product
        acc += tl.dot(a, b)

    # write back
    c_ptrs = C_batch_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # extract dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Ensure dtype float32
    x = input_tensor
    # Activation: Normalization 1
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps=1e-5
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections:
    left_proj_w = weights["left_proj.weight"] # shape hidden_dim, dim
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    # Compute via torch.nn.functional.linear
    left = torch.nn.functional.linear(x_norm, left_proj_w)  # (B,N,N,hidden)
    right = torch.nn.functional.linear(x_norm, right_proj_w)

    # Mask (if has mask). Optionally ignore if mask is None. Provided.
    if mask is not None:
        # ensure same dtype as left
        mask_ = mask.to(left.dtype).unsqueeze(-1)  # shape B,N,N,1
        left = left * mask_
        right = right * mask_
    # Gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Compute batched GEMM using Triton
    # Convert to float16 for kernel
    left_fp16 = left.permute(0,3,1,2).contiguous().to(torch.float16)  # (B, hidden, N, N)
    right_fp16 = right.permute(0,3,1,2).contiguous().to(torch.float16)
    # transpose right along last two dims
    right_T_fp16 = right_fp16.transpose(-1, -2).contiguous()  # (B, hidden, N, N)
    B_val, H, N, _ = left_fp16.shape
    outer = B_val * H

    # flatten
    A = left_fp16.view(outer, N, N)
    B_ = right_T_fp16.view(outer, N, N)
    C = torch.empty_like(A, dtype=torch.float16)

    # Get strides
    stride_ah = A.stride(0)
    stride_am = A.stride(1)

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(A_ptr, B_ptr, C_ptr,
                        stride_ah, stride_am, stride_ak,
                        stride_bh, stride_bk, stride_bn,
                        stride_ch, stride_cm, stride_cn,
                        M, N, K,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """
    Batched GEMM kernel: computes C = A @ B   (A,B,C shaped [batch, M, K] and [batch, K, N]).
    The batch dimension corresponds to outer = batch (i.e., B*hidden).
    Supports float16 inputs with float32 accumulation.
    """
    pid_outer = tl.program_id(0)          # batch index
    pid_tile = tl.program_id(1)           # tile index over output matrix

    # Determine tile location (m, n)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid_tile // num_pid_n
    pid_n = pid_tile % num_pid_n

    # Compute offsets for each tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds checks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the beginning of the current batch's matrices
    A_batch = A_ptr + pid_outer * stride_ah
    B_batch = B_ptr + pid_outer * stride_bh
    C_batch = C_ptr + pid_outer * stride_ch

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = A_batch + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        b_ptrs = B_batch + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

        # Load sub‑tiles; convert to float32 for dot product
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write result back as float16
    c = acc.to(tl.float16)
    c_ptrs = C_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) operator with a fused Triton GEMM.
    Arguments:
        data: tuple (input, mask, weights, config)
            - input: Tensor[B, N, N, dim] (float32)
            - mask: Tensor[B, N, N] (float32 or bool)
            - weights: dict of model parameters
            - config: dict containing "dim", "hidden_dim", and optional "nomask"
    Returns:
        Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -------------------------------------------------------------------------
    # 1. First LayerNorm over the last dimension (dim)
    # -------------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # -------------------------------------------------------------------------
    # 2. Linear projections and gating
    # -------------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask if it exists and the config does not request no‑masking
    if not config.get("nomask", False):
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # 3. Batched GEMM (the N³ core) using Triton
    #    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Reshape to (batch*hidden, N, N) and perform a standard matrix multiply
    #    with the right operand transposed.
    # -------------------------------------------------------------------------
    B, N_seq, _, _ = left.shape
    # Move hidden dimension to second axis and cast to float16 for the kernel
    left_fp16 = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)     # (B, H, N, N)
    right_fp16 = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B, H, N, N)

    # Transpose the inner two dimensions of right (so we can do A @ B_T)
    right_T_fp16 = right_fp16.transpose(-1, -2).contiguous()                # (B, H, N, N)

    # Flatten batch & hidden dimension into a single outer dimension
    B_val, H, N, _ = left_fp16.shape
    outer = B_val * H
    A = left_fp16.view(outer, N, N)
    B_mat = right_T_fp16.view(outer, N, N)
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides (in elements, not bytes)
    stride_ah = A.stride(0)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)

    stride_bh = B_mat.stride(0)
    stride_bk = B_mat.stride(1)
    stride_bn = B_mat.stride(2)

    stride_ch = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (outer, triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N))
    batched_gemm_kernel[grid](
        A, B_mat, C,
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bk, stride_bn,
        stride_ch, stride_cm, stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # -------------------------------------------------------------------------
    # 4. Restore original layout and continue with the remaining layers
    # -------------------------------------------------------------------------
    out = C.view(B_val, H, N, N).permute(0, 2, 3, 1).to(torch.float32)   # (B, N, N, H)

    # Second LayerNorm over hidden dimension
    out_norm = F.layer_norm(out, (hidden_dim,),
                            weight=weights["to_out_norm.weight"],
                            bias=weights["to_out_norm.bias"],
                            eps=eps)

    # Apply output gate
    out_norm = out_norm * out_gate

    # Final linear projection back to original dimension
    final = F.linear(out_norm, weights["to_out.weight"])

    return final

        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to original layout and convert to float32 for the rest
    out = (
        C.view(B_val, H, N, N)
        .permute(0, 2, 3, 1)
        .to(torch.float32)
    )  # shape [B, N, N, hidden]

    # ---------------------------------------------------------------------
    # 4. Second LayerNorm, output gate, final linear projection
    # ---------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```
