@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # compute batch index and hidden dim
    b = pid_b // H
    h = pid_b % H

    # compute base offsets
    off_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    off_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for boundary
    mask_i = off_i < N
    mask_j = off_j < N

    # initialize accumulator with zeros of float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        off_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = off_k < N

        # pointers for left A block
        left_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + off_i[:, None] * stride_left_i + off_k[None, :] * stride_left_k
        # load A block with mask
        a = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)  # shape (BLOCK_M,BLOCK_K)

        # pointers for right B_T block
        right_ptrs = right_ptr + b*stride_right_b + h*stride_right_h + off_j[None, :] * stride_right_i + off_k[:, None] * stride_right_k

        # dot product accumulate
        acc += tl.dot(a, b_tile)  # shape (BLOCK_M,BLOCK_N)
    # store result to out
    out_ptrs = out_ptr + b*stride_out_b + h*stride_out_h + off_i[:, None] * stride_out_i + off_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16) if ??? else acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul custom kernel (outgoing version)

The kernel computes the core O(N³) operation of the TriMul module:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

where `left` and `right` are the projected and gated pairwise representations.
The computation is a batched matrix multiplication with the second
operand transposed, equivalent to:
    out_h = left_h @ right_h.T   for each hidden channel h.

We implement this heavy inner loop in Triton, tiling the i/j dimensions
and iterating over k.  Hidden channels are fused into the batch dimension
(`batch_id = b * hidden_dim + h`) so a single Triton kernel handles the
full [B, N, N, H] tensor.  The rest of the TriMul forward pass (LayerNorm,
projections, gating, final LayerNorm and output linear) is performed in
PyTorch for simplicity.

The kernel works on fp16 tensors (the reference implementation uses bf16);
the accumulation is performed in fp32 for numerical stability.
"""
import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def trimul_matmul_kernel(
    left_ptr,               # *fp16   left  [B, N, N, H]
    right_ptr,              # *fp16   right [B, N, N, H] (will be accessed as transposed)
    out_ptr,                # *fp16   output [B, N, N, H]
    B, N, H,                # int32   dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,  # tile size in i dimension
    BLOCK_N: tl.constexpr,  # tile size in j dimension
    BLOCK_K: tl.constexpr,  # tile size in k dimension
):
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile
    pid_bn = tl.program_id(2)         # fused batch * hidden channel

    # decode batch and hidden index
    b = pid_bn // H
    h = pid_bn % H

    # Range of i/j indices for this program
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: left[b, i, k, h]
        left_offs = (b * stride_left_b
                     + h * stride_left_h
                     + offs_i[:, None] * stride_left_i
                     + offs_k[None, :] * stride_left_k)
        a = tl.load(left_ptr + left_offs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right tile (transposed view): right[b, j, k, h] -> we need B_T[k, j]
        right_offs = (b * stride_right_b
                      + h * stride_right_h
                      + offs_j[None, :] * stride_right_i   # j dimension uses stride_i
                      + offs_k[:, None] * stride_right_k)  # k dimension uses stride_k
        b_tile = tl.load(right_ptr + right_offs,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0).to(tl.float32)

        # accumulate partial dot product
        acc += tl.dot(a, b_tile)

    # write result
    out_offs = (b * stride_out_b
                + h * stride_out_h
                + offs_i[:, None] * stride_out_i
                + offs_j[None, :] * stride_out_j)
    tl.store(out_ptr + out_offs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with the heavy O(N³) part
    executed in a Triton kernel.
    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (float32 or bool)
            - weights: dict of torch.Tensors containing all model weights
            - config: dict with keys "dim", "hidden_dim", "nomask" (optional)
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        # mask broadcast to hidden_dim
        mask_exp = mask.unsqueeze(-1).to(dtype=left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating values (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel for the O(N³) update
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel (reference uses bf16; fp16 is similar)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16 = torch.empty_like(left_fp16)  # same shape

    B, N, _, _ = left_fp16.shape

    # strides in element units
    s_left = left_fp16.stride()
    s_right = right_fp16.stride()
    s_out = out_fp16.stride()

    stride_left_b, stride_left_i, stride_left_k, stride_left_h = s_left
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = s_right
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = s_out

    # Tiling parameters (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    trimul_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to fp32 for the remaining layers
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=1e-5)

    # ------------------------------------------------------------------
    # 7. Multiply by out_gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
