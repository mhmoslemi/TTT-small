import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_idx = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute start offsets for this block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for valid i, j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile
        a_ptrs = left_ptr + batch_idx * stride_left_batch + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # Convert to float32 for accumulation
        a = a.to(tl.float32)

        # Load B tile (right transposed)
        b_ptrs = right_ptr + batch_idx * stride_right_batch + offs_k[:, None] * stride_right_i + offs_j[None, :] * stride_right_k
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        b = b.to(tl.float32)

        # Compute block matrix multiplication
        acc += tl.dot(a, b)  # returns (BLOCK_M, BLOCK_N)

    # Store result
    c_ptrs = out_ptr + batch_idx * stride_out_batch + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    # Convert acc to float16 for storage
    acc_fp16 = acc.to(tl.float16)
    tl.store(c_ptrs, acc_fp16, mask=mask_i[:, None] & mask_j[None, :])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   N: tl.constexpr,
                   stride_left_batch, stride_left_i, stride_left_k,
                   stride_right_batch, stride_right_i, stride_right_k,
                   stride_out_batch, stride_out_i, stride_out_j,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """Batched matrix multiplication for TriMul.

    Computes out = left @ right.T for each batch (flattened over hidden dim).
    left, right, out are stored as [batch, M, K], [batch, K, N], [batch, M, N] respectively.
    """
    batch_idx = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Offsets for this tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (shape BLOCK_M x BLOCK_K)
        a_ptrs = left_ptr + batch_idx * stride_left_batch \
                 + offs_i[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load B tile (shape BLOCK_K x BLOCK_N) from right (already transposed)
        b_ptrs = right_ptr + batch_idx * stride_right_batch \
                 + offs_k[:, None] * stride_right_i \
                 + offs_j[None, :] * stride_right_k
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # Store result
    c_ptrs = out_ptr + batch_idx * stride_out_batch \
             + offs_i[:, None] * stride_out_i \
             + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul (outgoing) module.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # LayerNorm on input
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    # Linear projections (float32)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])
    # Apply mask if present
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_
    # Gating
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))
    left = left * left_gate
    right = right * right_gate
    # Cast to half for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    # Rearrange for batched GEMM: [B, H, N, N] -> flatten B*H
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]
    # Transpose right on the last two axes to get K dimension first
    right_trans = right_perm.transpose(2, 3).contiguous()    # [B, H, N, N] where dim2=N (K), dim3=N (i)
    # Flatten batch*hidden dimension
    batch_h = bs * hidden_dim
    left_flat = left_perm.view(batch_h, N, N)
    right_flat = right_trans.view(batch_h, N, N)
    # Allocate output (float16)
    out_fp16 = torch.empty_like(left_flat, dtype=torch.float16, device=device)
    # Strides
    stride_left_batch, stride_left_i, stride_left_k = left_flat.stride()
    stride_right_batch, stride_right_i, stride_right_k = right_flat.stride()
    stride_out_batch, stride_out_i, stride_out_j = out_fp16.stride()
    # Block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        batch_h,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_fp16,
        N,
        stride_left_batch,
        stride_left_i,
        stride_left_k,
        stride_right_batch,
        stride_right_i,
        stride_right_k,
        stride_out_batch,
        stride_out_i,
        stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Reshape back to [B, N, N, H]
    out = out_fp16.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)
    # Output layernorm
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out_norm * out_gate
    # Final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"])
    return out

"""
TriMul (outgoing) forward pass with a Triton kernel for the
core N³ multiplication.

The kernel implements a batched matrix multiplication:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
by reshaping the tensors to [B*D, N, N] and computing
out = left @ rightᵀ for each flattened batch using a
standard tiled GEMM (BLOCK_M × BLOCK_N output tiles,
BLOCK_K inner reduction). The surrounding LayerNorms,
Linear projections, and gating are performed in PyTorch.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    N: tl.constexpr,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM used inside TriMul.

    left  : [batch, M, K]  (M=N, K=N)
    right : [batch, K, N]  (K=N, N=N) – already transposed
    out   : [batch, M, N]
    """
    batch_idx = tl.program_id(0)          # flattened (B * hidden_dim)
    pid_i = tl.program_id(1)              # output row tile
    pid_j = tl.program_id(2)              # output col tile

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over K
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile: [BLOCK_M, BLOCK_K]
        a_ptrs = (
            left_ptr
            + batch_idx * stride_left_batch
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # B tile: [BLOCK_K, BLOCK_N] (right already transposed)
        b_ptrs = (
            right_ptr
            + batch_idx * stride_right_batch
            + offs_k[:, None] * stride_right_i
            + offs_j[None, :] * stride_right_k
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)  # sum over K

    # store result in FP16
    c_ptrs = (
        out_ptr
        + batch_idx * stride_out_batch
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] (float32)
            - mask:          [B, N, N] (bool or float)
            - weights: dict of model weight tensors
            - config:  dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm on the input ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Mask ----------
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(x.dtype)   # broadcast over hidden
        left = left * mask_f
        right = right * mask_f

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to FP16 for the heavy N³ kernel ----------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Rearrange to [B, hidden, N, N] and make right transposed on the last two axes
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()        # [B, H, N, N]
    right_trans = right_perm.transpose(2, 3).contiguous()          # [B, H, N, N] where dim2 is K, dim3 is i

    # Flatten the batch & hidden dimensions → (B*H, N, N)
    batch_h = B * hidden_dim
    left_flat = left_perm.view(batch_h, N, N)
    right_flat = right_trans.view(batch_h, N, N)

    # Allocate output buffer (FP16)
    out_fp16 = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # Strides for the flattened tensors
    stride_left_batch, stride_left_i, stride_left_k = left_flat.stride()
    stride_right_batch, stride_right_i, stride_right_k = right_flat.stride()
    stride_out_batch, stride_out_i, stride_out_j = out_fp16.stride()

    # ----- Triton launch -----
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        batch_h,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_fp16,
        N,
        stride_left_batch,
        stride_left_i,
        stride_left_k,
        stride_right_batch,
        stride_right_i,
        stride_right_k,
        stride_out_batch,
        stride_out_i,
        stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original shape: [B, N, N, hidden]
    out = (
        out_fp16.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )

    # ---------- Output LayerNorm ----------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ---------- Apply output gate ----------
    out = out_norm * out_gate

    # ---------- Final linear projection back to dim ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
