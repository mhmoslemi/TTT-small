"""TriMul outgoing kernel implementation.

Compute the TriMul operation using a hybrid approach:
- LayerNorm and linear projections are performed with PyTorch.
- An element‑wise Triton kernel fuses the gating (and optional mask) into the
  projected tensors.
- The heavy N³ reduction is performed with a batched matrix multiplication
  (`torch.bmm`), which is highly optimized in cuBLAS.
The final output undergoes a second LayerNorm, gating and linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def gated_mul_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    total_elements,
    BLOCK: tl.constexpr
):
    """
    In‑place elementwise multiplication:
        left = left * left_gate
        right = right * right_gate
    Arguments:
        left_ptr, right_ptr, left_gate_ptr, right_gate_ptr: contiguous float32 tensors
            of shape (total_elements,).
        total_elements: number of elements in each tensor.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load values (use 0.0 for out‑of‑bounds to keep mask well‑defined)
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Fuse the multiplications
    l = l * lg
    r = r * rg

    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data: Tuple (input, mask, weights, config)
            - input: Tensor [B, N, N, dim] (float32)
            - mask:  Tensor [B, N, N]   (float32 or bool)
            - weights: dict of tensors
            - config: dict with keys 'dim' and 'hidden_dim'
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ----- LayerNorm on the input (first norm) ---------------------------------
    norm_weight = weights["norm.weight"].to(dtype=device)
    norm_bias = weights["norm.bias"].to(dtype=device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ----- Linear projections ----------------------------------------------------
    left_proj_weight = weights["left_proj.weight"].to(dtype=device)
    right_proj_weight = weights["right_proj.weight"].to(dtype=device)

    # shape: [B,N,N,hidden_dim]
    left = F.linear(x, left_proj_weight)          # bias is None
    right = F.linear(x, right_proj_weight)

    # ----- Gates -----------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"].to(dtype=device)
    right_gate_weight = weights["right_gate.weight"].to(dtype=device)
    out_gate_weight = weights["out_gate.weight"].to(dtype=device)

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ----- Optional mask ---------------------------------------------------------
    if mask_tensor is not None:
        # broadcast mask to the hidden dimension
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ----- Fuse gating with a Triton kernel ------------------------------------
    # Ensure tensors are contiguous for 1‑D view
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    total_elements = left.numel()
    BLOCK = 1024  # 32‑warps (multiple of 32) works well on B200
    grid = (triton.cdiv(total_elements, BLOCK),)

    gated_mul_kernel[grid](
        left, right,
        left_gate, right_gate,
        total_elements,
        BLOCK=BLOCK,
    )
    # After the kernel `left` and `right` contain the gated values.

    # ----- N³ reduction via batched matrix multiplication ------------------------
    # Reshape tensors to (B*hidden_dim, N, N) and compute (L * Rᵀ) per channel.
    B, N, _, _ = left.shape
    # left: [B,N,N,hidden] -> [B, hidden, N, N]
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Transpose the second matrix on the last two dimensions (k ↔ j)
    right_mat_T = right_mat.transpose(1, 2)   # (B*H, N, N)

    # Batched gemm: (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat_T)   # shape (B*H, N, N)

    # Restore the original layout [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ----- Second LayerNorm ------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype=device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(dtype=device)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # ----- Apply output gate -----------------------------------------------------
    out = out * out_gate

    # ----- Final projection ------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(dtype=device)
    out = F.linear(out, to_out_weight)   # shape [B, N, N, dim]

    # Return float32 tensor (ensure dtype)
    return out.to(torch.float32)
```

"""TriMul outgoing kernel (AlphaFold3).

The implementation follows the reference PyTorch code but accelerates
the element‑wise gating with a Triton kernel and performs the
N³‑reduction via a batched matrix multiplication (torch.bmm),
which is highly optimized in cuBLAS.  The pipeline is:

1. Layer‑norm on the input.
2. Linear projections (left/right) and gating logits.
3. Optional mask multiplication.
4. Triton kernel fuses the gating:   left = left * left_gate,
                                      right = right * right_gate.
5. Cast to bfloat16 and compute out = left @ rightᵀ for each hidden
   channel using torch.bmm.
6. Second Layer‑norm, output‑gate multiplication and final linear
   projection back to the original dimension.
All tensors stay on the GPU; the kernel is launched from the host
function `custom_kernel`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: in‑place elementwise multiplication with gates
# -------------------------------------------------------------
@triton.jit
def gated_mul_kernel(
    left_ptr, right_ptr,               # fp32 tensors [total_elements]
    left_gate_ptr, right_gate_ptr,     # fp32 tensors [total_elements]
    total_elements,                    # int scalar
    BLOCK: tl.constexpr                # block size (multiple of 32)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    l  = tl.load(left_ptr + offs,        mask=mask, other=0.0)
    r  = tl.load(right_ptr + offs,       mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs,   mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs,  mask=mask, other=0.0)

    l = l * lg
    r = r * rg

    tl.store(left_ptr + offs,  l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim]
            - mask  : torch.Tensor [B, N, N] (may be all‑ones)
            - weights: dict of tensors (see code)
            - config : dict with keys 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) First LayerNorm on the input
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # -----------------------------------------------------------------
    # 2) Linear projections
    # -----------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"].to(device, dtype)
    right_proj_weight = weights["right_proj.weight"].to(device, dtype)

    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # 3) Gates (logits -> sigmoid)
    # -----------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"].to(device, dtype)
    right_gate_weight = weights["right_gate.weight"].to(device, dtype)
    out_gate_weight   = weights["out_gate.weight"].to(device, dtype)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # -----------------------------------------------------------------
    # 4) Optional mask (broadcast to hidden_dim)
    # -----------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left  = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 5) Fuse gating with Triton kernel (in‑place)
    # -----------------------------------------------------------------
    left  = left.contiguous()
    right = right.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    total_el = left.numel()
    BLOCK = 1024                       # good default for B200 (multiple of 32)
    grid = (triton.cdiv(total_el, BLOCK),)

    gated_mul_kernel[grid](
        left, right,
        left_gate, right_gate,
        total_el,
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 6) N³ reduction via batched GEMM (left @ rightᵀ) in bfloat16
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape

    # Cast to bf16 as in the reference implementation
    left_bf  = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    # reshape to (B*hidden, N, N)
    left_mat  = left_bf.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right_bf.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # rightᵀ on the last two dimensions (k ↔ j)
    right_mat_T = right_mat.transpose(1, 2)

    # Batched matrix multiply: (B*H, N, N) @ (B*H, N, N)ᵀ -> (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat_T)   # bf16 result

    # Restore original layout and convert back to fp32
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # -----------------------------------------------------------------
    # 7) Second LayerNorm, output gating and final projection
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype)

    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate

    to_out_weight = weights["to_out.weight"].to(device, dtype)   # (dim, hidden_dim)
    out = F.linear(out, to_out_weight)                         # [B, N, N, dim]

    return out.to(torch.float32)
