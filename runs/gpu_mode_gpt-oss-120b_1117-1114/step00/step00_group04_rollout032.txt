    B, N, D,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_d,

@triton.jit
def trimul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, HD,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_d,
    stride_b_batch, stride_b_k, stride_b_i, stride_b_d,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_d,
    # BLOCK_M, BLOCK_N, BLOCK_K compile constants
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    pid = tl.program_id(0)
    batch = pid // HD
    d = pid % HD

    a_ptr = a_ptr + batch * stride_a_batch + d * stride_a_d

    pid_n = tl.program_id(2)

"""
TriMul forward kernel (outgoing version).

The heavy part of the TriMul operation is
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batch of matrix multiplications:
    out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].transpose(-2, -1)

This implementation:
  * Performs LayerNorm, linear projections and gating in PyTorch (fast enough).
  * Casts the gated tensors to float16 and computes the above batched matmul
    with a custom Triton kernel.  The kernel tiles the (i,j) matrix with
    BLOCK_M × BLOCK_N tiles and reduces over k in blocks of size BLOCK_K.
    It treats the batch and hidden dimensions as a single combined axis,
    launching one program instance per (batch, hidden) pair.
  * Finally applies the output LayerNorm, output gate and linear projection
    in PyTorch.  The returned tensor is float32 with shape
    [bs, seq_len, seq_len, dim].
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: batched matmul with right-hand side transposed.
# ----------------------------------------------------------------------------- #
@triton.jit
def trimul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    B, N, HD,                           # runtime scalars: batch size, seq len, hidden dim
    stride_a_batch, stride_a_i, stride_a_k, stride_a_d,
    stride_b_batch, stride_b_k, stride_b_i, stride_b_d,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out = left @ right.T for each (batch, hidden) slice.
    left  : shape (B, N, N, HD)   (row = i, col = k)
    right : shape (B, N, N, HD)   (row = j, col = k) – we read it transposed.
    out   : shape (B, N, N, HD)   (row = i, col = j)
    """
    # --------------------------------------------------------------------- #
    # Identify which (batch, hidden) this program instance works on.
    # --------------------------------------------------------------------- #
    pid = tl.program_id(0)                # combined batch * hidden index
    batch = pid // HD                     # batch index
    d = pid % HD                          # hidden (channel) index

    # Base pointers for this (batch, d) slice
    a_base = a_ptr + batch * stride_a_batch + d * stride_a_d
    b_base = b_ptr + batch * stride_b_batch + d * stride_b_d
    c_base = c_ptr + batch * stride_c_batch + d * stride_c_d

    # --------------------------------------------------------------------- #
    # Program IDs for the 2‑D tiling of the output matrix (i, j).
    # --------------------------------------------------------------------- #
    pid_m = tl.program_id(1)              # tile row index
    pid_n = tl.program_id(2)              # tile col index

    # Offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------------- #
    # Loop over k dimension in tiles of size BLOCK_K.
    # --------------------------------------------------------------------- #
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a tile of left: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)

        # Load a tile of right transposed: (BLOCK_K, BLOCK_N)
        # right is stored as (batch, j, k, d); we access it as (k, j)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0, dtype=tl.float16)

        # Accumulate dot product (result in float32)
        acc += tl.dot(a, b)

    # Write the result tile to output (float32)
    c_ptrs = c_base + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------------- #
# Entry point used by the evaluator.
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : [B, N, N, dim] (float32, on CUDA)
            mask_tensor  : [B, N, N] (float32 or bool, on CUDA)
            weights_dict : mapping of weight names to tensors
            config_dict  : {"dim": int, "hidden_dim": int,
                            "nomask": bool (optional)}

    Returns:
        output tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    import torch
    import triton
    # unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm on input ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]

    # ---------- Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ---------- Optional mask ----------
    if not nomask:
        # mask has shape [B, N, N]; broadcast over hidden dimension
        mask = mask_tensor.unsqueeze(-1).to(x.dtype)
        left = left * mask
        right = right * mask

    # ---------- Gating ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to half for the heavy kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_fp16.shape  # batch, seq_len

    # Allocate output (float32) – will be filled by Triton kernel
    out = torch.empty_like(left_fp16, dtype=torch.float32)

    # ------- Strides (int64) -------
    stride_a_batch = left_fp16.stride(0)
    stride_a_i = left_fp16.stride(1)
    stride_a_k = left_fp16.stride(2)
    stride_a_d = left_fp16.stride(3)

    stride_b_batch = right_fp16.stride(0)
    stride_b_k = right_fp16.stride(2)      # row stride after transposition
    stride_b_i = right_fp16.stride(1)      # col stride after transposition
    stride_b_d = right_fp16.stride(3)

    stride_c_batch = out.stride(0)
    stride_c_i = out.stride(1)
    stride_c_j = out.stride(2)
    stride_c_d = out.stride(3)

    # ------- Tiling parameters -------
    # Small N can use smaller tiles; otherwise stick to 64.
    if N <= 256:
        BLOCK_M = 32
        BLOCK_N = 32
        BLOCK_K = 32
    else:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64

    # ------- Grid launch -------
    grid = (
        B * hidden_dim,                 # one program per (batch, hidden) pair
        triton.cdiv(N, BLOCK_M),       # tiles along i
        triton.cdiv(N, BLOCK_N),       # tiles along j
    )

    # Launch kernel
    trimul_matmul_kernel[grid](
        left_fp16.data_ptr(),
        right_fp16.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_a_batch, stride_a_i, stride_a_k, stride_a_d,
        stride_b_batch, stride_b_k, stride_b_i, stride_b_d,
        stride_c_batch, stride_c_i, stride_c_j, stride_c_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ---------- Output LayerNorm ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    # Apply output gate (same shape as out)
    out = out * out_gate

    # ---------- Final linear projection ----------
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)  # [B, N, N, dim]

    # Ensure dtype is float32
    return out.to(torch.float32)
