@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Abatch, stride_Am, stride_Ak,
    stride_Bbatch, stride_Bk, stride_Bn,
    stride_Cbatch, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # base pointers for this batch
    a_ptrs = A_ptr + pid_batch * stride_Abatch + offs_m[:, None] * stride_Am + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    b_ptrs = B_ptr + pid_batch * stride_Bbatch + tl.arange(0, BLOCK_K)[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load A block
        a = tl.load(a_ptrs, mask= (offs_m < M)[:, None] & (tl.arange(0, BLOCK_K) < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask= (tl.arange(0, BLOCK_K) < K)[:, None] & (offs_n < N)[None, :], other=0.0)
        # dot product
        acc += tl.dot(a, b)
        # advance pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk
    # write back
    c_ptrs = C_ptr + pid_batch * stride_Cbatch + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    mask_c = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc, mask=mask_c)
```

def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Abatch, stride_Am, stride_Ak,
    stride_Bbatch, stride_Bk, stride_Bn,
    stride_Cbatch, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)  # batch_head index
    pid_m = tl.program_id(1)  # tile row
    pid_n = tl.program_id(2)  # tile col

    # compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # create mask to guard out of bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers base for batch
    a_batch_ptr = A_ptr + pid_batch * stride_Abatch
    b_batch_ptr = B_ptr + pid_batch * stride_Bbatch
    c_batch_ptr = C_ptr + pid_batch * stride_Cbatch

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute offset for k block
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # For load, need to guard out-of-range k indices as well

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Ensure float32
    input_tensor = input_tensor.to(torch.float32)
    mask = mask.to(torch.float32)

    # LayerNorm: normalize over last dim (dim)
    eps = 1e-5
    # Compute mean & var across dim axis (-1)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_weight + norm_bias

    # Linear projections
    # F.linear expects weight shape (out_features, in_features)
    left = F.linear(x, left_proj_weight)  # shape [B,N,N,H]
    right = F.linear(x, right_proj_weight)

    # Apply mask (if mask not None)
    # If mask shape is (B,N,N) we broadcast
    mask_unsq = mask.unsqueeze(-1)  # shape [B,N,N,1]
    left = left * mask_unsq
    right = right * mask_unsq

    # Compute gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    B, N, _, H = left.shape
    # Permute for GEMM
    left_perm = left.permute(0,3,1,2).contiguous()   # [B,H,N,N]
    right_perm = right.permute(0,3,1,2).contiguous() # [B,H,N,N]
    # Transpose right for GEMM (swap last two dims)
    right_T = right_perm.transpose(-2, -1).contiguous() # [B,H,N,N]

    # Flatten batch*head
    BH = B * H
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_T.view(BH, N, N)
    out_flat = torch.empty_like(left_flat)  # shape [BH,N,N]

    # Get strides
    stride_Abatch = left_flat.stride(0)
    stride_Am = left_flat.stride(1)
    stride_Ak = left_flat.stride(2)

    stride_Bbatch = right_flat.stride(0)
    stride_Bk = right_flat.stride(1)
    stride_Bn = right_flat.stride(2)

    stride_Cbatch = out_flat.stride(0)
    stride_Cm = out_flat.stride(1)
    stride_Cn = out_flat.stride(2)

    # Block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (BH, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_Abatch, stride_Am, stride_Ak,
        stride_Bbatch, stride_Bk, stride_Bn,
        stride_Cbatch, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape out_flat back
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()  # [B,N,N,H]

    # Apply to_out_norm (layer norm over hidden dim)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Multiply with out_gate
    out = out * out_gate

    # Final linear projection to dim
    out = F.linear(out, to_out_weight)  # [B,N,N,dim]

    return out

"""TriMul (outgoing) forward pass.

The implementation follows the reference PyTorch TriMul module but
offloads the O(N³) pairwise multiplication to a custom Triton kernel.
Steps:
1. Layer‑norm the input tensor.
2. Linear projections (left/right) and gating (left/right/out) using the
   provided weights.
3. Apply the optional mask to the projected tensors.
4. Compute the core update
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched GEMM with the right operand transposed.
   This is done by the Triton kernel `batched_gemm_kernel`.
5. Layer‑norm the result, multiply by the out‑gate, and apply the final
   linear projection back to the original dimensionality.

All tensors stay in float32; the kernel works in float32 for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Abatch, stride_Am, stride_Ak,
    stride_Bbatch, stride_Bk, stride_Bn,
    stride_Cbatch, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM C = A @ B  (A: [batch, M, K], B: [batch, K, N])"""
    pid_batch = tl.program_id(0)   # batch * hidden_dim index
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile col

    # Offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current batch
    a_batch_ptr = A_ptr + pid_batch * stride_Abatch
    b_batch_ptr = B_ptr + pid_batch * stride_Bbatch
    c_batch_ptr = C_ptr + pid_batch * stride_Cbatch

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k_a = cur_k < K
        mask_k_b = cur_k < K

        # Pointers for the current K‑block
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_Am) + (cur_k[None, :] * stride_Ak)
        b_ptrs = b_batch_ptr + (cur_k[:, None] * stride_Bk) + (offs_n[None, :] * stride_Bn)

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k_a[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k_b[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_Cm) + (offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update.
    
    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (binary, can be all‑ones)
            - weights: dict with the module parameters
            - config: dict with "dim" and "hidden_dim"
    
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ----------------------------------------------------------------------
    # 1) LayerNorm on the input (same eps as nn.LayerNorm)
    # ----------------------------------------------------------------------
    eps = 1e-5
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = input_tensor.to(torch.float32)
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x = (x - mean) / torch.sqrt(var + eps)
    x = x * norm_weight + norm_bias

    # ----------------------------------------------------------------------
    # 2) Linear projections
    # ----------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # ----------------------------------------------------------------------
    # 3) Mask (broadcast over hidden dimension)
    # ----------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(torch.float32).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ----------------------------------------------------------------------
    # 4) Gating
    # ----------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 5) Core O(N³) multiplication via Triton
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ----------------------------------------------------------------------
    B, N, _, H = left.shape          # B, seq_len, seq_len, hidden_dim
    # reshape to [B, H, N, N] and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()     # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]

    # Transpose right so that we can do a normal GEMM (K,N) = (K,N) where K = seq_len
    right_T = right_perm.transpose(-2, -1).contiguous()   # [B, H, N, N]

    # Collapse batch * hidden_dim into a single batch dimension for the kernel
    BH = B * H
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_T.view(BH, N, N)
    out_flat = torch.empty_like(left_flat)                # [BH, N, N]

    # Strides for the kernel (all tensors are contiguous)
    stride_Abatch = left_flat.stride(0)
    stride_Am = left_flat.stride(1)
    stride_Ak = left_flat.stride(2)

    stride_Bbatch = right_flat.stride(0)
    stride_Bk = right_flat.stride(1)
    stride_Bn = right_flat.stride(2)

    stride_Cbatch = out_flat.stride(0)
    stride_Cm = out_flat.stride(1)
    stride_Cn = out_flat.stride(2)

    # Blocking sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        BH,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_Abatch, stride_Am, stride_Ak,
        stride_Bbatch, stride_Bk, stride_Bn,
        stride_Cbatch, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------------
    # 6) Post‑processing: LayerNorm, out‑gate, final linear
    # ----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
