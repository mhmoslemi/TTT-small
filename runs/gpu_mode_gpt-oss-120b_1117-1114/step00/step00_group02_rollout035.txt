```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)  # block row (i)
    pid_j = tl.program_id(1)  # block col (j)
    pid_bh = tl.program_id(2)  # batch * hidden

    # decode batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # compute block offsets
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    # base pointers for this batch and hidden channel
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K

        # indices for left: i (rows) and k (cols)
        i_idx = i_offset + tl.arange(0, BLOCK_M)
        k_idx = k_offset + tl.arange(0, BLOCK_K)

        # load left block (M x K)
        left_ptrs = left_base + i_idx[:, None] * stride_left_i + k_idx[None, :] * stride_left_k
        mask_i = i_idx[:, None] < N
        mask_k = k_idx[None, :] < N
        left = tl.load(left_ptrs, mask=mask_i & mask_k, other=0.0)

        # indices for right: j (rows) and k (cols)
        j_idx = j_offset + tl.arange(0, BLOCK_N)

        # load right block (N x K)
        right_ptrs = right_base + j_idx[:, None] * stride_right_i + k_idx[None, :] * stride_right_k
        mask_j = j_idx[:, None] < N
        # reuse mask_k defined above
        right = tl.load(right_ptrs, mask=mask_j & mask_k, other=0.0)

        # compute dot: left (M,K) * right.T (K,N) -> (M,N)
        acc += tl.dot(left, tl.trans(right, (1, 0)))

    # write back
    i_idx = i_offset + tl.arange(0, BLOCK_M)
    j_idx = j_offset + tl.arange(0, BLOCK_N)
    out_ptrs = out_base + i_idx[:, None] * stride_out_i + j_idx[None, :] * stride_out_j
    mask_i = i_idx[:, None] < N
    mask_j = j_idx[None, :] < N
    tl.store(out_ptrs, acc, mask=mask_i & mask_j)


def tri_mul_triton(left: torch.Tensor, right: torch.Tensor):
    """
    Compute the triangular multiplication using a custom Triton kernel.
    left, right: tensors of shape [B, N, N, H] (float32)
    Returns:
        out: tensor of shape [B, N, N, H] (float32)
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape

    # allocate output
    out = torch.empty_like(left)

    # strides
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    # Choose block sizes (tuned for typical GPU)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_left[0],
        s_left[1],
        s_left[2],
        s_left[3],
        s_right[0],
        s_right[1],
        s_right[2],
        s_right[3],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Triton-accelerated forward pass of the Triangle Multiplicative Update (TriMul) operator.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor of shape [B, N, N, dim]
            mask: torch.Tensor of shape [B, N, N] (float or bool)
            weights: dict of named weight tensors
            config: dict containing "dim" and "hidden_dim"
    Returns:
        output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # LayerNorm over last dim
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias = weights["norm.bias"].to(device, dtype)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"].to(device, dtype)
    right_proj_weight = weights["right_proj.weight"].to(device, dtype)

    left = F.linear(x, left_proj_weight)   # shape [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Apply mask if present
    if mask is not None:
        # ensure mask is float32 and broadcastable
        mask_f = mask.to(dtype).unsqueeze(-1)  # shape [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # Gating linear layers
    left_gate_weight = weights["left_gate.weight"].to(device, dtype)
    right_gate_weight = weights["right_gate.weight"].to(device, dtype)
    out_gate_weight = weights["out_gate.weight"].to(device, dtype)

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Triangular multiplication via Triton
    out = tri_mul_triton(left, right)   # shape [B, N, N, hidden_dim]

    # Output LayerNorm and gating
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device, dtype)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate

    # Final linear projection back to original dim
    to_out_weight = weights["to_out.weight"].to(device, dtype)
    out = F.linear(out, to_out_weight)

    # Ensure output is float32
    out = out.to(torch.float32)

    return out
```

"""
Triton‑accelerated forward pass of the Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input over the last dimension.
2. Linear projections (left/right) and gating linear layers (left_gate,
   right_gate, out_gate) without bias.
3. Apply mask (if provided) and element‑wise gates to the projected tensors.
4. Core O(N³) operation:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication left @ rightᵀ for each hidden
   channel *h*.  Implemented with a custom Triton kernel that tiles the
   (i, j) dimensions and reduces over k.
5. Apply a second LayerNorm, multiply by out_gate and project back to the
   original channel dimension.

Only the heavy reduction in step 4 is written in Triton; the surrounding
operations stay in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid_i = tl.program_id(0)          # block row   (i)
    pid_j = tl.program_id(1)          # block col   (j)
    pid_bh = tl.program_id(2)         # batch * hidden index

    # decode batch and hidden channel
    b = pid_bh // H
    h = pid_bh % H

    # block offsets
    i_off = pid_i * BLOCK_M
    j_off = pid_j * BLOCK_N

    # base pointers for this (b, h)
    left_base  = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K

        # indices
        i_idx = i_off + tl.arange(0, BLOCK_M)
        j_idx = j_off + tl.arange(0, BLOCK_N)
        k_idx = k_off + tl.arange(0, BLOCK_K)

        # left block: shape (M, K)
        left_ptrs = left_base + i_idx[:, None] * stride_left_i + k_idx[None, :] * stride_left_k
        mask_l = (i_idx[:, None] < N) & (k_idx[None, :] < N)
        left = tl.load(left_ptrs, mask=mask_l, other=0.0)

        # right block: we load it as (K, N) to match tl.dot requirements
        right_ptrs = right_base + k_idx[:, None] * stride_right_k + j_idx[None, :] * stride_right_i
        mask_r = (k_idx[:, None] < N) & (j_idx[None, :] < N)
        right = tl.load(right_ptrs, mask=mask_r, other=0.0)

        # accumulate product
        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # write results
    i_idx = i_off + tl.arange(0, BLOCK_M)
    j_idx = j_off + tl.arange(0, BLOCK_N)
    out_ptrs = out_base + i_idx[:, None] * stride_out_i + j_idx[None, :] * stride_out_j
    mask_o = (i_idx[:, None] < N) & (j_idx[None, :] < N)
    tl.store(out_ptrs, acc, mask=mask_o)


def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the triangular multiplication using a Triton kernel.
    left, right: tensors of shape [B, N, N, H] (float32)
    Returns: tensor of shape [B, N, N, H] (float32)
    """
    assert left.shape == right.shape, "left/right must have identical shapes"
    B, N, _, H = left.shape

    # allocate output
    out = torch.empty_like(left)

    # strides for pointer arithmetic
    s_l = left.stride()
    s_r = right.stride()
    s_o = out.stride()

    # tile sizes (chosen for typical A100/B200 like hardware)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_o[0], s_o[1], s_o[2], s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of TriMul.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (bool or float)
            - weights: dict of model weights (already on the correct device)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32  # final output required in float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, dtype)
    norm_b = weights["norm.bias"].to(device, dtype)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device, dtype)
    right_proj_w = weights["right_proj.weight"].to(device, dtype)

    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)

    # -----------------------------------------------------------------
    # 3. Optional mask
    # -----------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)      # broadcast over hidden dim
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gates
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device, dtype)
    right_gate_w = weights["right_gate.weight"].to(device, dtype)
    out_gate_w = weights["out_gate.weight"].to(device, dtype)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core TriMul (batched GEMM over the sequence dimension)
    # -----------------------------------------------------------------
    out = tri_mul_triton(left, right)   # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out_gate, final projection
    # -----------------------------------------------------------------
    out_norm_w = weights["to_out_norm.weight"].to(device, dtype)
    out_norm_b = weights["to_out_norm.bias"].to(device, dtype)

    out = F.layer_norm(out, (hidden_dim,), weight=out_norm_w, bias=out_norm_b)
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device, dtype)
    out = F.linear(out, to_out_w)   # back to [B, N, N, dim]

    # ensure float32 output
    out = out.to(torch.float32)
    return out
