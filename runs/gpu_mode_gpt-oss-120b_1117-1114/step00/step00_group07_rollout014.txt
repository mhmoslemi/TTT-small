    right_proj_weight = right_proj_weight.to(torch.float32, device=device)

    right_gate_weight = right_gate_weight.to(torch.float32, device=device)

    to_out_weight = to_out_weight.to(torch.float32, device=device)

    right = right * right_gate

"""
TriMul (outgoing) forward implementation with Triton acceleration.

The heavy O(N³) reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is transformed into a batched matrix multiplication per hidden dimension:
    left[b,:, :, d] @ right[b,:, :, d].T .
A custom Triton kernel computes this GEMM for all (batch, head) pairs.
The remaining linear / layer‑norm / gate operations are performed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,      # pointers
    B, H, N,                           # dimensions
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_i, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out = left @ right.T for each (batch, head) pair.

    left  : [B, H, N, K] (K == N)
    right : [B, H, N, K]
    out   : [B, H, N, N]
    """
    pid_i = tl.program_id(0)      # tile row
    pid_j = tl.program_id(1)      # tile col
    pid_head = tl.program_id(2)   # batch*head index

    # decode batch and head
    b = pid_head // H
    h = pid_head % H

    # compute offsets for the output tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in float32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K (= N)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------- load left tile: shape (BLOCK_M, BLOCK_K) ----------
        a_ptrs = (left_ptr
                  + b * stride_left_b
                  + h * stride_left_h
                  + offs_i[:, None] * stride_left_i
                  + offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # -------- load right tile transposed: shape (BLOCK_K, BLOCK_N) ----------
        # we need right[b,h,j,k] -> shape (k, j)
        b_ptrs = (right_ptr
                  + b * stride_right_b
                  + h * stride_right_h
                  + offs_k[:, None] * stride_right_k
                  + offs_j[None, :] * stride_right_i)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # -------- multiply‑accumulate ----------
        acc += tl.dot(a, b)   # (M,K) @ (K,N) -> (M,N)

    # -------- store the result ----------
    out_ptrs = (out_ptr
                + b * stride_out_b
                + h * stride_out_h
                + offs_i[:, None] * stride_out_i
                + offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask        : torch.Tensor [B, N, N] (optional, broadcastable)
            weights     : dict of model parameters (float32 tensors)
            config      : dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load model weights (ensure float32 dtype and correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(dtype=torch.float32, device=device)
    norm_bias = weights["norm.bias"].to(dtype=torch.float32, device=device)

    left_proj_weight = weights["left_proj.weight"].to(dtype=torch.float32, device=device)
    right_proj_weight = weights["right_proj.weight"].to(dtype=torch.float32, device=device)

    left_gate_weight = weights["left_gate.weight"].to(dtype=torch.float32, device=device)
    right_gate_weight = weights["right_gate.weight"].to(dtype=torch.float32, device=device)
    out_gate_weight = weights["out_gate.weight"].to(dtype=torch.float32, device=device)

    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype=torch.float32, device=device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(dtype=torch.float32, device=device)

    to_out_weight = weights["to_out.weight"].to(dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)    # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # ensure mask is float for multiplication
        mask = mask.to(dtype=dtype, device=device)
        mask = mask.unsqueeze(-1)                 # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core O(N³) reduction via Triton GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    # Cast to fp16 for faster compute
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape  # left shape [B, N, N, H]

    # rearrange to [B, H, N, N] – easier for batched GEMM
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()

    # output buffer for the fused multiplication
    out_perm = torch.empty((B, hidden_dim, N, N), dtype=torch.float32, device=device)

    # strides (in number of elements, not bytes)
    stride_l_b, stride_l_h, stride_l_i, stride_l_k = left_perm.stride()
    stride_r_b, stride_r_h, stride_r_i, stride_r_k = right_perm.stride()
    stride_o_b, stride_o_h, stride_o_i, stride_o_j = out_perm.stride()

    # block sizes – tuned for A100 / B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        B * hidden_dim                    # batch * head
    )

    tri_mul_gemm_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B,
        hidden_dim,
        N,
        stride_l_b, stride_l_h, stride_l_i, stride_l_k,
        stride_r_b, stride_r_h, stride_r_i, stride_r_k,
        stride_o_b, stride_o_h, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4
    )

    # Rearrange back to original layout [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    out = out * out_gate
    out = F.linear(out, to_out_weight)

    return out
