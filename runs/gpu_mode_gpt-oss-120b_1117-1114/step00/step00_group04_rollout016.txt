@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    pid_n = tl.program_id(1)
    pid_z = tl.program_id(2)

    # Compute batch and head indices
    b = pid_z // H
    h = pid_z % H

    # Compute start indices for i and j
    i_start = pid * BLOCK_M
    j_start = pid_n * BLOCK_N

    # Offsets for i and j
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # Mask for valid i and j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute strides for each tile pointer using broadcast
    # Pre-calc base offset: batch and head
    # pointer offsets are in units of elements

    # compute base offset for left and right:
    left_base = b * stride_left_b + h * stride_left_d # h times stride_left_d (should be 1)
    right_base = b * stride_right_b + h * stride_right_d
    out_base = b * stride_out_b + h * stride_out_d

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (i,k)
        # left_ptr has shape [B,N,N,H] but we flatten.
        # Compute offsets: (offs_i[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k) + left_base
        left_ptrs = left_ptr + (offs_i[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k) + left_base
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile (k,j) i.e., right[b, j, k, h]
        # Offsets: (offs_k[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_i) + right_base
        right_ptrs = right_ptr + (offs_k[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_i) + right_base
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Compute dot product using tl.dot
        # accumulates along K dimension (BLOCK_K)
        acc += tl.dot(left, right)

    # Store results
    out_ptrs = out_ptr + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j) + out_base
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward implementation with a Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor on the channel dimension (`dim`).
2. Linear projections to a hidden dimension `hidden_dim`:
   * left = x @ left_proj_weight
   * right = x @ right_proj_weight
3. Apply the optional mask (broadcast over the hidden dimension).
4. Compute per‑hidden‑dimensional gated tensors:
   * left *= sigmoid(x @ left_gate_weight)
   * right *= sigmoid(x @ right_gate_weight)
5. Perform the core O(N³) reduction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication left @ rightᵀ for each
   (batch, hidden) pair.  It is implemented in Triton with tiling over
   the i, j and k dimensions.
6. Layer‑norm the intermediate result on `hidden_dim`, multiply by the
   output gate (sigmoid(x @ out_gate_weight)), and finally project back
   to `dim` with a linear layer.
All operations are carried out in float32; the heavy reduction step is
fully fused in the Triton kernel.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    with left, right, out laid out as [B, N, N, H].
    """
    pid_i = tl.program_id(0)          # block row (i)
    pid_j = tl.program_id(1)          # block col (j)
    pid_z = tl.program_id(2)          # combined batch*head index

    # decode batch and hidden index
    b = pid_z // H
    h = pid_z % H

    # block start offsets
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointer offset for this (b, h)
    left_base = b * stride_left_b + h * stride_left_d
    right_base = b * stride_right_b + h * stride_right_d
    out_base = b * stride_out_b + h * stride_out_d

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension in tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(0, num_k):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + (offs_i[:, None] * stride_left_i) \
                               + (offs_k[None, :] * stride_left_k) \
                               + left_base
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # Load right tile (transposed access): shape (BLOCK_K, BLOCK_N)
        # right[b, j, k, h] -> use j as the first index, k as second
        right_ptrs = right_ptr + (offs_k[:, None] * stride_right_k) \
                                 + (offs_j[None, :] * stride_right_i) \
                                 + right_base
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # Accumulate partial dot product over the K tile
        acc += tl.dot(left, right)

    # Store the result
    out_ptrs = out_ptr + (offs_i[:, None] * stride_out_i) \
                       + (offs_j[None, :] * stride_out_j) \
                       + out_base
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[:, None])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (bool or float)
        - weights      : dict of torch.Tensors containing model weights
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. LayerNorm on input
    # -----------------------------------------------------------------
    x = input_tensor.to(torch.float32)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]    # (hidden_dim, dim)

    left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating
    # -----------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core TriMul reduction via Triton
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # store result of einsum

    # Strides (elements, not bytes)
    stride_l_b = left.stride(0)
    stride_l_i = left.stride(1)
    stride_l_k = left.stride(2)
    stride_l_d = left.stride(3)

    stride_r_b = right.stride(0)
    stride_r_i = right.stride(1)
    stride_r_k = right.stride(2)
    stride_r_d = right.stride(3)

    stride_o_b = out.stride(0)
    stride_o_i = out.stride(1)
    stride_o_j = out.stride(2)
    stride_o_d = out.stride(3)

    # Tile sizes – modest values that fit in registers on B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_d,
        stride_r_b, stride_r_i, stride_r_k, stride_r_d,
        stride_o_b, stride_o_i, stride_o_j, stride_o_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Post‑processing: layer‑norm, output gate, final projection
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight,
        bias=to_out_norm_bias, eps=eps
    )

    out = out * out_gate  # element‑wise gating

    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)

    return out.to(torch.float32)
