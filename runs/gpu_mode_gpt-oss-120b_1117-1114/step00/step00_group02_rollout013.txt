def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, n_elements, BLOCK: tl.constexpr):

    """Element‑wise multiplication using a tiny Triton kernel."""
    assert a.shape == b.shape

    b_cont = b.contiguous()

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Tiny Triton kernel: element‑wise multiplication
# -------------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, n_elements, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


def elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Element‑wise multiplication using a simple Triton kernel."""
    assert a.shape == b.shape, "Shapes must match for element‑wise multiplication"
    a_cont = a.contiguous()
    b_cont = b.contiguous()
    out = torch.empty_like(a_cont)
    n = a_cont.numel()
    BLOCK = 65536  # large block works for a pure multiply
    grid = (triton.cdiv(n, BLOCK),)
    elementwise_mul_kernel[grid](a_cont, b_cont, out, n, BLOCK=BLOCK)
    return out

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of AlphaFold3's TriangleMultiplicativeUpdate
    (the “outgoing” version). The heavy triangular reduction is expressed as a
    batched matrix multiplication (cuBLAS), while the final element‑wise scaling
    by the out‑gate uses a custom Triton kernel.
    """
    input_tensor, mask_tensor, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # Prepare tensors
    # -----------------------------------------------------------------
    x = input_tensor.to(torch.float32)               # [B, N, N, dim]
    mask = mask_tensor.to(torch.float32)             # [B, N, N]

    # -----------------------------------------------------------------
    # Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # -----------------------------------------------------------------
    # Linear projections
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # Apply mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    mask_exp = mask.unsqueeze(-1)                      # [B, N, N, 1]
    left = left * mask_exp
    right = right * mask_exp

    # -----------------------------------------------------------------
    # Gates (sigmoid of linear transforms)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Core triangular multiplicative update:
    # out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # Implemented as left @ rightᵀ for each hidden channel.
    # -----------------------------------------------------------------
    # reshape to (B, H, N, N) and use torch.matmul (cuBLAS)
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # B, H, N, N
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # B, H, N, N (k, j order)
    out_perm = torch.matmul(left_perm, right_perm)            # B, H, N, N
    out = out_perm.permute(0, 2, 3, 1).contiguous()           # B, N, N, H

    # -----------------------------------------------------------------
    # Output LayerNorm
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # -----------------------------------------------------------------
    # Scale by out_gate using Triton
    # -----------------------------------------------------------------
    out = elementwise_mul(out, out_gate)

    # -----------------------------------------------------------------
    # Final linear projection back to original dimension
    # -----------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
