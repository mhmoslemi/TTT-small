@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_am, stride_ak, stride_ai,
    stride_bk, stride_bj, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    batch_idx = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_rem = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_i = pid_rem // tl.cdiv(N, BLOCK_N)
    block_j = pid_rem % tl.cdiv(N, BLOCK_N)

    # compute offsets
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # create masks for boundaries
    mask_i = offs_i < M
    mask_j = offs_j < N

    # pointers
    a_ptrs = a_ptr + batch_idx * stride_cb + offs_i[:,None] * stride_ai + offs_k[None,:] * stride_ak
    b_ptrs = b_ptr + batch_idx * stride_cb + offs_k[:,None] * stride_bk + offs_j[None,:] * stride_bj

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_i[:,None] & (offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:,None] < K) & mask_j[None,:], other=0.0)
        acc += tl.dot(a, b)
        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result
    c_ptrs = c_ptr + batch_idx * stride_cb + offs_i[:,None] * stride_cm + offs_j[None,:] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16) if tl.dtype(acc) == tl.float32 else acc, mask=mask_i[:,None] & mask_j[None,:])
```

@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # compute total number of blocks in i and j dimensions
    num_i = tl.cdiv(M, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    # compute indices
    batch_idx = pid // (num_i * num_j)
    rem = pid % (num_i * num_j)
    pid_i = rem // num_j
    pid_j = rem % num_j

    # offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # base pointers for current batch
    a_base = a_ptr + batch_idx * stride_a_batch
    b_base = b_ptr + batch_idx * stride_b_batch
    c_base = c_ptr + batch_idx * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = a_base + offs_i[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile (K x N)
        b_ptrs = b_base + offs_k[:, None] * stride_b_row + offs_j[None, :] * stride_b_col
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_base + offs_i[:, None] * stride_c_row + offs_j[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

"""
tri_mul_outgoing (custom_kernel)

Implements the outgoing Triangle Multiplicative Update (TriMul) used in AlphaFold3.
The heavy O(N³·H) contraction is off‑loaded to a Triton kernel that performs a
batched matrix multiplication:
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
where `left` and `right` are the gated linear projections of the input tensor.
All other operations (LayerNorm, linear projections, sigmoids, final
LayerNorm/Linear) are carried out with PyTorch for simplicity.

The kernel works on a flattened batch dimension (batch × hidden_dim) and
computes `C = A @ Bᵀ` for each batch entry using tiled loads and `tl.dot`.
Input tensors are cast to float16 before the kernel to reduce memory traffic,
while accumulation is performed in float32 to retain numerical stability.
The final result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    total_batch, M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM kernel for the TriMul contraction.
    A: (total_batch, M, K)   – left tensor (i, k)
    B: (total_batch, K, N)   – right tensor transposed (k, j)
    C: (total_batch, M, N)   – output
    """
    pid = tl.program_id(0)

    # number of blocks in each spatial dimension
    num_i = tl.cdiv(M, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    # decode batch, i‑block and j‑block from linear pid
    batch_idx = pid // (num_i * num_j)
    rem = pid % (num_i * num_j)
    pid_i = rem // num_j
    pid_j = rem % num_j

    # offsets for this block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # base pointers for the current batch entry
    a_base = a_ptr + batch_idx * stride_a_batch
    b_base = b_ptr + batch_idx * stride_b_batch
    c_base = c_ptr + batch_idx * stride_c_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = kk * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_i[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + offs_k[:, None] * stride_b_row + offs_j[None, :] * stride_b_col
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # Store the result tile
    c_ptrs = c_base + offs_i[:, None] * stride_c_row + offs_j[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32 or bool)
            - weights: dict of parameter tensors
            - config: dict containing "dim", "hidden_dim", and optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- 1) LayerNorm on input ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, normalized_shape=(C,),
                     weight=norm_weight, bias=norm_bias)

    # ---------- 2) Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- 3) Optional masking ----------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ---------- 4) Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- 5) Cast to half for the kernel ----------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ---------- 6) Prepare tensors for batched GEMM ----------
    # left: [B, N, N, H]   -> permute to (B, H, N, N) then view as (B*H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 2, 1).contiguous()   # transpose the last two dims (j,k) -> (k,j)

    left_flat = left.view(B * hidden_dim, N, N)
    right_flat = right.view(B * hidden_dim, N, N)

    # Output buffer
    out_flat = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # ---------- 7) Launch Triton kernel ----------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_batch = B * hidden_dim
    grid = (total_batch * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N),)

    tri_mul_gemm_kernel[grid](
        a_ptr=left_flat.data_ptr(),
        b_ptr=right_flat.data_ptr(),
        c_ptr=out_flat.data_ptr(),
        total_batch=total_batch,
        M=N,
        N=N,
        K=N,
        stride_a_batch=left_flat.stride(0),
        stride_a_row=left_flat.stride(1),
        stride_a_col=left_flat.stride(2),
        stride_b_batch=right_flat.stride(0),
        stride_b_row=right_flat.stride(1),
        stride_b_col=right_flat.stride(2),
        stride_c_batch=out_flat.stride(0),
        stride_c_row=out_flat.stride(1),
        stride_c_col=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- 8) Reshape back to [B, N, N, H] ----------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp32

    # ---------- 9) Output LayerNorm ----------
    out = F.layer_norm(out, normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])

    # ---------- 10) Apply output gate ----------
    out = out * out_gate

    # ---------- 11) Final linear projection ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
