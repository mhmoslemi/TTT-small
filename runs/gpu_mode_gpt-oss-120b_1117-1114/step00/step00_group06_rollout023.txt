import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(lhs_ptr, rhs_ptr, out_ptr,
                   N, H, B,
                   lhs_stride_b, lhs_stride_i, lhs_stride_k, lhs_stride_h,
                   rhs_stride_b, rhs_stride_j, rhs_stride_k, rhs_stride_h,
                   out_stride_b, out_stride_i, out_stride_j, out_stride_h,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    # Number of tiles in each dimension
    grid_m = tl.cdiv(N, BLOCK_M)
    grid_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = grid_m * grid_n
    batch_head = pid // tiles_per_batch       # 0 <= batch_head < B*H
    tile_id = pid % tiles_per_batch
    tile_m = tile_id // grid_n
    tile_n = tile_id % grid_n

    # recover batch idx (b) and hidden idx (h)
    b = batch_head // H
    h = batch_head % H

    # base offsets for this (b, h)
    lhs_base = b * lhs_stride_b + h * lhs_stride_h
    rhs_base = b * rhs_stride_b + h * rhs_stride_h
    out_base = b * out_stride_b + h * out_stride_h

    # row and column offsets for the output tile
    i_off = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = i_off < N
    mask_j = j_off < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # pointers for A (left)
        a_ptrs = lhs_ptr + lhs_base + i_off[:, None] * lhs_stride_i + k_off[None, :] * lhs_stride_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # pointers for B (right transposed)
        b_ptrs = rhs_ptr + rhs_base + k_off[:, None] * rhs_stride_k + j_off[None, :] * rhs_stride_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate
        acc += tl.dot(a, b)   # a: (M, K), b: (K, N); acc: float32

    # write out
    c_ptrs = out_ptr + out_base + i_off[:, None] * out_stride_i + j_off[None, :] * out_stride_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[:, None])

def custom_kernel(data):
    """
    Triton-accelerated implementation of the outgoing Triangle Multiplicative Update (TriMul).
    This kernel computes the core einsum sum_k left[i,k,d] * right[j,k,d] using a blocked
    matrix multiplication in Triton. All other linear layers, layer‑norms and gating operations
    are performed with PyTorch for simplicity.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # -----------------------------------------------------------------
    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply mask (if present) and gates
    mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
    left = left * mask_f * left_gate
    right = right * mask_f * right_gate

    # Cast to half for the heavy O(N^3) part
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # -----------------------------------------------------------------
    # Triton kernel for out = sum_k left[i,k,d] * right[j,k,d]
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    out = torch.empty(bs, N, N, hidden_dim, dtype=torch.float32, device=device)

    # strides
    lhs_stride_b = left_h.stride(0)
    lhs_stride_i = left_h.stride(1)
    lhs_stride_k = left_h.stride(2)
    lhs_stride_h = left_h.stride(3)

    rhs_stride_b = right_h.stride(0)
    rhs_stride_j = right_h.stride(1)   # column dimension after transposition
    rhs_stride_k = right_h.stride(2)
    rhs_stride_h = right_h.stride(3)

    out_stride_b = out.stride(0)
    out_stride_i = out.stride(1)
    out_stride_j = out.stride(2)
    out_stride_h = out.stride(3)

    total_batches = bs * hidden_dim
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_programs = total_batches * grid_m * grid_n
    grid = (total_programs,)

    tri_mul_kernel[grid](
        left_h, right_h, out,
        N, hidden_dim, bs,
        lhs_stride_b, lhs_stride_i, lhs_stride_k, lhs_stride_h,
        rhs_stride_b, rhs_stride_j, rhs_stride_k, rhs_stride_h,
        out_stride_b, out_stride_i, out_stride_j, out_stride_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # -----------------------------------------------------------------
    # Post‑processing: layer norm, out‑gate, final linear projection
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=1e-5)
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # shape [B, N, N, dim]

    return out

"""Triton‐accelerated outgoing Triangle Multiplicative Update (TriMul).

The heavy O(N³) part of the module is the einsum

    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication
for each (b,d):  left_{i,k} @ right_{k,j}.  This kernel computes all
batch‑hidden slices in a single launch using a 3‑dimensional tiling
strategy (BLOCK_M×BLOCK_N tiles, BLOCK_K inner reduction).  All
other linear layers, LayerNorms and gating are performed with PyTorch
for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    lhs_ptr, rhs_ptr, out_ptr,
    N, H, B,
    lhs_stride_b, lhs_stride_i, lhs_stride_k, lhs_stride_h,
    rhs_stride_b, rhs_stride_j, rhs_stride_k, rhs_stride_h,
    out_stride_b, out_stride_i, out_stride_j, out_stride_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --- grid layout -------------------------------------------------
    grid_m = tl.cdiv(N, BLOCK_M)          # tiles along M (i dimension)
    grid_n = tl.cdiv(N, BLOCK_N)          # tiles along N (j dimension)
    tiles_per_batch = grid_m * grid_n
    batch_head = pid // tiles_per_batch   # 0 <= batch_head < B*H
    tile_id = pid % tiles_per_batch
    tile_m = tile_id // grid_n
    tile_n = tile_id % grid_n

    # recover (batch, hidden) indices
    b = batch_head // H
    h = batch_head % H

    # base offsets for this (b,h) slice
    lhs_base = b * lhs_stride_b + h * lhs_stride_h
    rhs_base = b * rhs_stride_b + h * rhs_stride_h
    out_base = b * out_stride_b + h * out_stride_h

    # offsets of the output tile
    i_off = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = i_off < N
    mask_j = j_off < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --- reduction over k --------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # left block:   [i, k]
        a_ptrs = lhs_ptr + lhs_base \
                 + i_off[:, None] * lhs_stride_i \
                 + k_off[None, :] * lhs_stride_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right block (transposed): [k, j]
        b_ptrs = rhs_ptr + rhs_base \
                 + k_off[:, None] * rhs_stride_k \
                 + j_off[None, :] * rhs_stride_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # accumulate dot product; a,b are FP16, acc is FP32
        acc += tl.dot(a, b)

    # --- store --------------------------------------------------------
    c_ptrs = out_ptr + out_base \
             + i_off[:, None] * out_stride_i \
             + j_off[None, :] * out_stride_j
    tl.store(c_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑backed forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] torch.Tensor on CUDA
            - mask: [B, N, N] torch.Tensor (bool or 0/1 float)
            - weights: dict of model parameters (LayerNorm, Linear weights, etc.)
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)          # [B,N,N,H]
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # 3) Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # -----------------------------------------------------------------
    # 4) Apply mask and gates
    mask_f = mask.to(x.dtype).unsqueeze(-1)       # [B,N,N,1]
    left = left * mask_f * left_gate
    right = right * mask_f * right_gate

    # -----------------------------------------------------------------
    # 5) Cast to FP16 for the cubic part
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # -----------------------------------------------------------------
    # 6) Triton kernel: batched matmul over the third dimension
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    out = torch.empty(bs, N, N, hidden_dim,
                      dtype=torch.float32, device=device)

    # strides for the tensors (in element units)
    lhs_stride_b = left_h.stride(0)
    lhs_stride_i = left_h.stride(1)
    lhs_stride_k = left_h.stride(2)
    lhs_stride_h = left_h.stride(3)

    rhs_stride_b = right_h.stride(0)
    rhs_stride_j = right_h.stride(1)   # column after transposition
    rhs_stride_k = right_h.stride(2)
    rhs_stride_h = right_h.stride(3)

    out_stride_b = out.stride(0)
    out_stride_i = out.stride(1)
    out_stride_j = out.stride(2)
    out_stride_h = out.stride(3)

    total_batches = bs * hidden_dim
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_programs = total_batches * grid_m * grid_n
    grid = (total_programs,)

    tri_mul_kernel[grid](
        left_h, right_h, out,
        N, hidden_dim, bs,
        lhs_stride_b, lhs_stride_i, lhs_stride_k, lhs_stride_h,
        rhs_stride_b, rhs_stride_j, rhs_stride_k, rhs_stride_h,
        out_stride_b, out_stride_i, out_stride_j, out_stride_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 7) Post‑processing: LayerNorm over hidden dim, out‑gate, final linear
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # shape [B, N, N, dim]

    return out
