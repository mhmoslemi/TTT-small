    device = input_tensor.device
    dtype = input_tensor.dtype

    # get dim and hidden_dim

    hidden_dim = config["hidden_dim"]

    # layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)

    # gates projections (before sigmoid)
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    # mask shape
    mask_unsq = mask.unsqueeze(-1).to(x_norm.dtype)

    # fused multiplication kernel for left and right
    B, N, _, _ = input_tensor.shape  # N repeated
    # allocate left_fused and right_fused
    left_fused = torch.empty_like(left_proj)
    right_fused = torch.empty_like(right_proj)

    # compute strides
    def get_strides(t):

        # t.ndim = 4

"""
TriMul forward implementation (outgoing version).

Algorithm
---------
1. Layer‑norm the input tensor (shape [B, N, N, C]).
2. Apply linear projections to obtain left/right tensors of shape [B, N, N, H].
3. Compute three gating tensors (left_gate, right_gate, out_gate) via a linear
   projection followed by a sigmoid.
4. Fuse element‑wise multiplication of projection, mask and gate using a small
   Triton kernel (the only Triton part required by the task).
5. Perform the triangular multiplicative update:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using an einsum (efficient batched matrix‑multiply under the hood).
6. Apply an output LayerNorm, multiply by the out_gate and a final linear
   projection back to the original channel dimension.

The Triton kernel `fused_mul_kernel` computes
    out = proj * mask * gate
in a single pass, broadcasting the mask over the hidden dimension.
All other computations are done with PyTorch (torch.float32 / torch.bfloat16)
to keep the implementation simple yet correct.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fused_mul_kernel(
    proj_ptr, mask_ptr, gate_ptr, out_ptr,
    B, N, H,
    stride_proj0, stride_proj1, stride_proj2, stride_proj3,
    stride_mask0, stride_mask1, stride_mask2, stride_mask3,
    stride_gate0, stride_gate1, stride_gate2, stride_gate3,
    stride_out0, stride_out1, stride_out2, stride_out3,
    BLOCK: tl.constexpr
):
    """
    out = proj * mask * gate
    proj, gate : [B, N, N, H]
    mask       : [B, N, N, 1] (broadcasted over H)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    m = offs < total

    # Decompose linear offset into (b, i, k, h)
    seq = N * N * H
    i_seq = N * H

    b_idx = offs // seq
    rem = offs - b_idx * seq
    i_idx = rem // i_seq
    rem2 = rem - i_idx * i_seq
    k_idx = rem2 // H
    h_idx = rem2 - k_idx * H

    # Offsets for each tensor
    proj_off = (
        b_idx * stride_proj0 +
        i_idx * stride_proj1 +
        k_idx * stride_proj2 +
        h_idx * stride_proj3
    )
    mask_off = (
        b_idx * stride_mask0 +
        i_idx * stride_mask1 +
        k_idx * stride_mask2
        # h dimension is size 1, we deliberately ignore it to broadcast
    )
    gate_off = (
        b_idx * stride_gate0 +
        i_idx * stride_gate1 +
        k_idx * stride_gate2 +
        h_idx * stride_gate3
    )
    out_off = (
        b_idx * stride_out0 +
        i_idx * stride_out1 +
        k_idx * stride_out2 +
        h_idx * stride_out3
    )

    proj_val = tl.load(proj_ptr + proj_off, mask=m)
    mask_val = tl.load(mask_ptr + mask_off, mask=m)
    gate_val = tl.load(gate_ptr + gate_off, mask=m)

    out_val = proj_val * mask_val * gate_val
    tl.store(out_ptr + out_off, out_val, mask=m)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (bool or float)
            - weights_dict: mapping from weight name to torch.Tensor
            - config_dict: contains at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])            # [B,N,N,H]
    right_proj = F.linear(x, weights["right_proj.weight"])          # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3. Gates (pre‑sigmoid + sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()   # [B,N,N,H]
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid() # [B,N,N,H]
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()    # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4. Fuse mask & gate using Triton
    # ------------------------------------------------------------------
    mask = mask_tensor.unsqueeze(-1).to(dtype)   # [B,N,N,1]

    B, N, _, _ = input_tensor.shape

    # Allocate fused tensors
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    total_elements = B * N * N * hidden_dim
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)

    # Strides for left side
    s_proj = left_proj.stride()
    s_mask = mask.stride()
    s_gate = left_gate.stride()
    s_out = left.stride()
    fused_mul_kernel[grid](
        left_proj, mask, left_gate, left,
        B, N, hidden_dim,
        s_proj[0], s_proj[1], s_proj[2], s_proj[3],
        s_mask[0], s_mask[1], s_mask[2], s_mask[3],
        s_gate[0], s_gate[1], s_gate[2], s_gate[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK=BLOCK,
    )

    # Strides for right side
    s_proj_r = right_proj.stride()
    s_gate_r = right_gate.stride()
    s_out_r = right.stride()
    fused_mul_kernel[grid](
        right_proj, mask, right_gate, right,
        B, N, hidden_dim,
        s_proj_r[0], s_proj_r[1], s_proj_r[2], s_proj_r[3],
        s_mask[0], s_mask[1], s_mask[2], s_mask[3],
        s_gate_r[0], s_gate_r[1], s_gate_r[2], s_gate_r[3],
        s_out_r[0], s_out_r[1], s_out_r[2], s_out_r[3],
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 5. Triangular multiplicative update (einsum)
    # ------------------------------------------------------------------
    # Use bfloat16 for the heavy reduction to improve bandwidth
    left_bf16 = left.to(torch.bfloat16)
    right_bf16 = right.to(torch.bfloat16)

    # out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    out = torch.einsum('b i k d, b j k d -> b i j d', left_bf16, right_bf16).to(dtype)

    # ------------------------------------------------------------------
    # 6. Output normalisation, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
