"""
TriMul (outgoing) forward pass with a fused Triton element‑wise kernel.

The implementation follows the reference PyTorch TriMul module but performs
the heavy element‑wise gating multiplication (`out = out_norm * out_gate`) with
a custom Triton kernel (BF16).  All linear projections, LayerNorms and the
core N×N matrix multiplication are executed with PyTorch (leveraging cuBLAS)
while intermediate tensors are kept in bf16 to reduce memory usage.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: elementwise bf16 multiplication (used for out_norm * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def bf16_mul_kernel(
    a_ptr,                # *pointer* to first input (bf16)
    b_ptr,                # *pointer* to second input (bf16)
    c_ptr,                # *pointer* to output (bf16)
    N,                    # total number of elements
    BLOCK: tl.constexpr,  # block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N

    a = tl.load(a_ptr + offs, mask=mask, other=0.0, dtype=tl.bfloat16)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0, dtype=tl.bfloat16)
    c = a * b
    tl.store(c_ptr + offs, c, mask=mask)


def bf16_elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Element‑wise multiplication of two bf16 tensors using Triton."""
    assert a.shape == b.shape, "Shape mismatch for elementwise mul"
    out = torch.empty_like(a)
    numel = a.numel()
    BLOCK = 1024
    grid = (triton.cdiv(numel, BLOCK),)
    # flatten to 1‑D pointers
    bf16_mul_kernel[grid](
        a.reshape(-1).contiguous(),
        b.reshape(-1).contiguous(),
        out.reshape(-1).contiguous(),
        numel,
        BLOCK=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Execute the outgoing TriMul operation.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (CUDA, float32)
            - mask_tensor: torch.Tensor [B, N, N] (CUDA, bool/int/float)
            - weights: dict with keys:
                'norm.weight', 'norm.bias',
                'left_proj.weight', 'right_proj.weight',
                'left_gate.weight', 'right_gate.weight', 'out_gate.weight',
                'to_out_norm.weight', 'to_out_norm.bias',
                'to_out.weight'
            - config: dict containing at least 'dim' and 'hidden_dim'
    Returns:
        torch.Tensor [B, N, N, dim] (float32, CUDA)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # Cast to bf16 for the remaining heavy work
    x_bf16 = x.to(torch.bfloat16)

    # ------------------------------------------------------------------
    # 2. Linear projections (bf16)
    # ------------------------------------------------------------------
    left = F.linear(x_bf16, weights["left_proj.weight"].to(torch.bfloat16))      # [B, N, N, hidden_dim]
    right = F.linear(x_bf16, weights["right_proj.weight"].to(torch.bfloat16))    # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Apply mask (if provided)
    # ------------------------------------------------------------------
    mask_bf16 = mask.unsqueeze(-1).to(torch.bfloat16)  # [B, N, N, 1]
    left = left * mask_bf16
    right = right * mask_bf16

    # ------------------------------------------------------------------
    # 4. Gating (bf16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x_bf16, weights["left_gate.weight"].to(torch.bfloat16))
    )
    right_gate = torch.sigmoid(
        F.linear(x_bf16, weights["right_gate.weight"].to(torch.bfloat16))
    )
    out_gate = torch.sigmoid(
        F.linear(x_bf16, weights["out_gate.weight"].to(torch.bfloat16))
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core update: out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    #    This can be expressed as batched matrix multiplication:
    #    out = left @ right.transpose(-1, -2)   (per hidden dimension)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left, right: [B, N, N, hidden_dim]
    # reshape to (B*hidden_dim, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)      # (B*H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)    # (B*H, N, N)

    # right.T on the last two dimensions
    right_T = right_flat.transpose(1, 2)  # (B*H, N, N)

    out_flat = torch.bmm(left_flat, right_T)  # (B*H, N, N)
    # reshape back to [B, N, N, hidden_dim]
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (bf16)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.bfloat16),
        bias=weights["to_out_norm.bias"].to(torch.bfloat16),
    )  # [B, N, N, hidden_dim] bf16

    # ------------------------------------------------------------------
    # 7. Fuse with out_gate using Triton (bf16)
    # ------------------------------------------------------------------
    out_fused = bf16_elementwise_mul(out_norm, out_gate)  # [B, N, N, hidden_dim] bf16

    # ------------------------------------------------------------------
    # 8. Final linear projection back to dim (bf16 -> float32)
    # ------------------------------------------------------------------
    out_proj = F.linear(
        out_fused, weights["to_out.weight"].to(torch.bfloat16)
    )  # [B, N, N, dim] bf16

    # Convert to float32 as required by the interface
    return out_proj.to(torch.float32)
