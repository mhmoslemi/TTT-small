def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=[dim],
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    # Projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask (if not nominal)
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1).to(x_norm.dtype)
        left = left * mask_unsq
        right = right * mask_unsq

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute contraction using Triton
    out = torch.empty_like(left)  # shape [B,N,N,hidden_dim]

    # Prepare strides
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # Use triton kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32  # may tune

    total_batch = B * hidden_dim
    nblocks_m = triton.cdiv(N, BLOCK_M)
    nblocks_n = triton.cdiv(N, BLOCK_N)
    grid = (total_batch, nblocks_m, nblocks_n)

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # LayerNorm on out (hidden_dim channel)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=[hidden_dim],
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # Apply out_gate (broadcast)
    out_gate = out_gate  # shape [B,N,N,hidden_dim], broadcast already
    out = out * out_gate

    # Final linear
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)  # batch * hidden
    pid1 = tl.program_id(1)  # block_i
    pid2 = tl.program_id(2)  # block_j

    b = pid0 // H
    h = pid0 % H

    # Offsets base pointers
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Compute ranges
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for i,j out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile [M,K]
        # left[i,k]
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile transposed: we need shape [K,N] where first dim K corresponds to k.
        # Use (k,i) where i is j index.
        right_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Compute dot product
        acc += tl.dot(left_tile, right_tile)

    # Store results
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul implementation with a custom Triton kernel.

The heavy computation is the contraction:
    out[b, i, j, d] = sum_k left[b, i, k, d] * right[b, j, k, d]

For each batch `b` and hidden channel `d` this is a matrix multiplication
L (N×N) @ Rᵀ (N×N).  The kernel treats each (b, d) pair as an independent
batch and tiles the multiplication with block sizes BLOCK_M × BLOCK_N
and a reduction tile BLOCK_K.  All other operations (LayerNorms,
linear projections, sigmoids, gating) are performed with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,                     # pointers
    B, N, H,                                          # tensor dims
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,  # left strides
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,  # right strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,  # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out = left @ rightᵀ per (batch, hidden) slice.
    left:  [B, N, N, H]
    right: [B, N, N, H]
    out:   [B, N, N, H]
    """
    pid_batch = tl.program_id(0)   # B * H
    pid_m = tl.program_id(1)       # block row (i)
    pid_n = tl.program_id(2)       # block col (j)

    # split batch/hidden
    b = pid_batch // H
    h = pid_batch % H

    # base pointers for this (b, h) slice
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    # coordinate vectors for the two output dimensions
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (M, K)   left[b, i, k, h]
        left_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        # right tile transposed: shape (K, N)   right[b, j, k, h] -> accessed as (k, j)
        right_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        # matrix multiply‑accumulate
        acc += tl.dot(left_tile, right_tile)

    # write the result
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul module using Triton for the core contraction.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim] torch.Tensor (cuda)
            - mask_tensor:  [B, N, N] torch.Tensor (cuda) (may be all‑ones)
            - weights_dict: mapping of parameter names to torch.Tensors
            - config_dict: contains "dim", "hidden_dim" (and possibly other entries)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    # ensure float32 computation
    x = input_tensor.to(torch.float32)

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]

    # ----- LayerNorm on the input -----
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ----- Linear projections (no bias) -----
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ----- Apply mask (if any) -----
    # mask: [B, N, N] -> [B, N, N, 1]
    mask_exp = mask.unsqueeze(-1).to(x.dtype)
    left = left * mask_exp
    right = right * mask_exp

    # ----- Gating layers -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----- Core contraction using Triton -----
    out = torch.empty_like(left)  # shape [B, N, N, hidden_dim]

    # strides (in elements)
    l_str = left.stride()
    r_str = right.stride()
    o_str = out.stride()

    # block sizes – chosen to fit shared memory comfortably
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    total_batch = B * hidden_dim
    nblocks_m = triton.cdiv(N, BLOCK_M)
    nblocks_n = triton.cdiv(N, BLOCK_N)
    grid = (total_batch, nblocks_m, nblocks_n)

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        l_str[0],
        l_str[1],
        l_str[2],
        l_str[3],
        r_str[0],
        r_str[1],
        r_str[2],
        r_str[3],
        o_str[0],
        o_str[1],
        o_str[2],
        o_str[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- LayerNorm on the hidden dimension -----
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ----- Apply output gate -----
    out = out * out_gate

    # ----- Final linear projection back to dim -----
    output = F.linear(out, weights["to_out.weight"])

    return output
