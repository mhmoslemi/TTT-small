@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    strideL_b, strideL_i, strideL_k, strideL_h,
    strideR_b, strideR_i, strideR_k, strideR_h,
    strideO_b, strideO_i, strideO_j, strideO_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_z = tl.program_id(2)  # batch*head index
    pid_m = tl.program_id(0)  # row block i
    pid_n = tl.program_id(1)  # col block j

    # compute batch and hidden index
    b = pid_z // H
    h = pid_z % H

    # compute offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute base pointers
    base_l = left_ptr + b * strideL_b + h * strideL_h
    base_r = right_ptr + b * strideR_b + h * strideR_h
    base_o = out_ptr + b * strideO_b + h * strideO_h

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load block of A (left)
        a_ptrs = base_l + (offs_i[:, None] * strideL_i) + (offs_k[None, :] * strideL_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load block of B^T (right transposed)
        # B block shape (BLOCK_K, BLOCK_N) using k as rows and j as cols
        b_ptrs = base_r + (offs_k[:, None] * strideR_k) + (offs_j[None, :] * strideR_i)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate
        acc += tl.dot(a, b)

    # store
    c_ptrs = base_o + (offs_i[:, None] * strideO_i) + (offs_j[None, :] * strideO_j)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right block (k,j) after transpose
        b_ptrs = base_r + (offs_k[:, None] * strideR_k) + (offs_j[None, :] * strideR_i)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

    c_ptrs = base_o + (offs_i[:, None] * strideO_i) + (offs_j[None, :] * strideO_j)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton implementation of the outgoing TriMul operator.

    The kernel computes the contracting sum over the middle sequence dimension
    after applying linear projections, mask and element‑wise gating.
    The remaining LayerNorms and linear layers are kept in PyTorch.
    """
    import torch
    import torch.nn.functional as F
    import triton

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # LayerNorm on input
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # Linear projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Convert mask to float and broadcast
    if mask is None:
        mask_f = torch.ones_like(left[..., :1])
    else:
        mask_f = mask.unsqueeze(-1).to(left.dtype)

    left = left * mask_f
    right = right * mask_f

    # Gating
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to half‑precision for the Triton kernel
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    B, N, _, H = left.shape

    # Output buffer
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # Strides
    strideL_b, strideL_i, strideL_k, strideL_h = left.stride()
    strideR_b, strideR_i, strideR_k, strideR_h = right.stride()
    strideO_b, strideO_i, strideO_j, strideO_h = out.stride()

    # Block sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total = B * H
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), total)

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        strideL_b,
        strideL_i,
        strideL_k,
        strideL_h,
        strideR_b,
        strideR_i,
        strideR_k,
        strideR_h,
        strideO_b,
        strideO_i,
        strideO_j,
        strideO_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Apply the output LayerNorm and gating
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # Final linear projection back to original dim
    out = F.linear(out, weights["to_out.weight"])

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: computes out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
# (right is accessed transposed on the last two axes).
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    strideL_b, strideL_i, strideL_k, strideL_h,
    strideR_b, strideR_i, strideR_k, strideR_h,
    strideO_b, strideO_i, strideO_j, strideO_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_z = tl.program_id(2)            # combined batch * head index
    pid_m = tl.program_id(0)            # block row   (i)
    pid_n = tl.program_id(1)            # block col   (j)

    # Decode batch and head from the combined index
    b = pid_z // H
    h = pid_z % H

    # Offsets for the current tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (b,h) slice
    base_l = left_ptr  + b * strideL_b + h * strideL_h
    base_r = right_ptr + b * strideR_b + h * strideR_h
    base_o = out_ptr    + b * strideO_b + h * strideO_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block: left[b,i,k,h]   (shape BLOCK_M x BLOCK_K)
        a_ptrs = base_l + (offs_i[:, None] * strideL_i) + (offs_k[None, :] * strideL_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B block transposed: right[b,j,k,h] -> we need right_T[k,j,h]
        # B block shape: BLOCK_K x BLOCK_N
        b_ptrs = base_r + (offs_k[:, None] * strideR_k) + (offs_j[None, :] * strideR_i)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Accumulate the dot product over the K dimension
        acc += tl.dot(a, b)

    # Store the result tile
    c_ptrs = base_o + (offs_i[:, None] * strideO_i) + (offs_j[None, :] * strideO_j)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    The heavy contraction Σ_k left[...,i,k,d] * right[...,j,k,d] is performed by a
    custom Triton kernel. All remaining linear layers, LayerNorms and sigmoids are
    executed with PyTorch for simplicity.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if supplied) – broadcast over the hidden dimension
    # ------------------------------------------------------------------
    if mask is None:
        mask_f = torch.ones_like(left[..., :1])
    else:
        mask_f = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
    left = left * mask_f
    right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to half‑precision and launch the Triton kernel
    # ------------------------------------------------------------------
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    B, N, _, H = left.shape   # H == hidden_dim
    assert H == hidden_dim, "hidden dimension mismatch"

    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # strides for the three tensors
    strideL_b, strideL_i, strideL_k, strideL_h = left.stride()
    strideR_b, strideR_i, strideR_k, strideR_h = right.stride()
    strideO_b, strideO_i, strideO_j, strideO_h = out.stride()

    # Tunable block sizes (work well for N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total = B * H                     # combined batch‑head dimension
    grid = (triton.cdiv(N, BLOCK_M),  # i‑tiles
            triton.cdiv(N, BLOCK_N),  # j‑tiles
            total)                    # (b,h) tiles

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        strideL_b,
        strideL_i,
        strideL_k,
        strideL_h,
        strideR_b,
        strideR_i,
        strideR_k,
        strideR_h,
        strideO_b,
        strideO_i,
        strideO_j,
        strideO_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
