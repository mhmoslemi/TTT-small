def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    # ensure float32.
    x = input_tensor.to(torch.float32)  # for LayerNorm.

    # Config reading
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # get eps maybe from weight? Not given; typical LayerNorm epsilon 1e-5.
    eps = 1e-5

    # LayerNorm: using weight and bias from weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # shape (dim)
    # Use torch.nn.functional.layer_norm
    # normalized_shape = (dim,) and weight,bias
    x_norm = torch.nn.functional.layer_norm(x, (dim,), norm_weight, norm_bias, eps)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]  # shape (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask (if mask is None? mask may be passed, but config may have nomask True)
    # In input, mask is given always? Might be all ones? We'll trust it's provided. If config.get('nomask', True) means no mask needed? In reference code they always apply mask. If nomask True means no mask (i.e., mask of all ones). We'll handle:

    nomask = config.get("nomask", True)
    if not nomask:
        # mask shape [B,N,N]
        mask_unsq = mask.unsqueeze(-1).to(x_norm.dtype)  # broadcast
        left = left * mask_unsq
        right = right * mask_unsq
    else:
        # skip mask
        pass

    # Gating
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape
    # Permute and reshape for kernel
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()  # [B, hidden_dim, N, N]
    right_perm_T = right_fp16.permute(0, 3, 2, 1).contiguous()  # [B, hidden_dim, N, N] with (k,j)
    # flatten batch
    batch_total = B * hidden_dim
    left_flat = left_perm.view(batch_total, N, N)
    right_flat = right_perm_T.view(batch_total, N, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)  # [batch_total, N, N]

    stride_batch = left_flat.stride(0)  # offset to next batch
    stride_am = left_flat.stride(1)  # row stride
    stride_ak = left_flat.stride(2)  # col stride
    # right_flat strides:
    stride_bk = right_flat.stride(1)  # row (k) stride
    stride_bn = right_flat.stride(2)  # col (j) stride
    # output strides:
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # Kernel block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid dimensions
    grid = (batch_total,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    # Launch kernel
    batched_matmul_kernel[grid](
        left_flat,  # pointers as tensors (triton will get .data_ptr)
        right_flat,
        out_flat,
        stride_batch,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape output
    out_contracted = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)  # shape [B,N,N,hidden_dim]

    out = out_contracted * out_gate

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), to_out_norm_weight, to_out_norm_bias, eps)

    # Final linear projection to original dim
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)  # shape [B,N,N,dim]

    return out

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_batch,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # get program ids
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask within bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to start of this batch
    a_ptrs = A_ptr + batch_id * stride_batch
    b_ptrs = B_ptr + batch_id * stride_batch
    c_ptrs = C_ptr + batch_id * stride_batch

    # offsets for loading A and B
    # We'll compute ptrs for block of A: (m,k)
    # Offsets for k computed inside loop

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute offsets for k tile
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Compute A load pointers
        a_tile_ptrs = a_ptrs + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        # Load A
        a = tl.load(a_tile_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

import torch
import triton
import triton.language as tl

# Triton kernel for batched matrix multiplication (A @ B)
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_batch,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ B for many independent batch matrices.
    A: (batch, M, K)
    B: (batch, K, N)
    C: (batch, M, N)
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = A_ptr + pid_batch * stride_batch
    b_base = B_ptr + pid_batch * stride_batch
    c_base = C_ptr + pid_batch * stride_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K_tile)
        a_ptrs = a_base + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile (K_tile x N)
        b_ptrs = b_base + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Store tile to C
    c_ptrs = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton-accelerated forward pass of the outgoing TriMul operator.

    Steps:
    1. Layer‑norm on the input.
    2. Linear projections + gating (all in PyTorch, float32).
    3. Cast projections to float16 and rearrange so that each (batch, hidden_dim)
       pair forms an independent matrix of shape [seq_len, seq_len].
    4. Batched GEMM kernel computes out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
    5. Apply the output gate, a second Layer‑Norm and the final linear projection
       back to the original channel dimension.
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim = config["dim"])
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = input_tensor.to(torch.float32)  # work in fp32 for stability
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections + gates (still in fp32)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # mask handling
    nomask = config.get("nomask", True)
    if not nomask:
        mask = mask.to(x.dtype)
        mask = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Prepare tensors for the batched GEMM (float16, contiguous)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape  # B, seq_len, seq_len, hidden_dim

    # Rearrange to [B, hidden_dim, N, N] where rows=i, cols=k
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()          # (B, hidden_dim, N, N)
    # Right needs to be transposed on the last two axes: (k, j)
    right_perm_T = right_fp16.permute(0, 3, 2, 1).contiguous()      # (B, hidden_dim, N, N)

    # Flatten the first two dimensions into one batch dimension
    batch_total = B * hidden_dim
    left_flat = left_perm.view(batch_total, N, N)   # (batch_total, N, N)
    right_flat = right_perm_T.view(batch_total, N, N)

    # Allocate output buffer
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 4. Launch Triton batched GEMM kernel
    # ------------------------------------------------------------------
    # Strides (all in element units)
    stride_batch = left_flat.stride(0)
    stride_am = left_flat.stride(1)   # row stride (M)
    stride_ak = left_flat.stride(2)   # col stride (K)
    stride_bk = right_flat.stride(1)  # row stride (K)
    stride_bn = right_flat.stride(2)  # col stride (N)
    stride_cm = out_flat.stride(1)    # row stride (M)
    stride_cn = out_flat.stride(2)    # col stride (N)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_total,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        stride_batch,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 5. Reshape back and continue with the remaining PyTorch ops
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)  # [B,N,N,hidden_dim]

    # Apply output gate
    out = out * out_gate

    # Second LayerNorm (hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # Final linear projection back to original dim
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)  # [B,N,N,dim]

    return out

"""
TriMul forward (outgoing) with a Triton kernel for the heavy O(N³) contraction.

Algorithm
---------
1. Layer‑norm on the 4‑D input (dim = config["dim"]).
2. Linear projections (left/right) + sigmoid gates (left/right) in FP32.
3. Optionally apply a mask (broadcasted over the hidden dimension).
4. Cast the gated tensors to FP16, permute so that each (batch, hidden_dim)
   pair becomes an independent square matrix [seq_len, seq_len].
5. Batched GEMM kernel (Triton) computes:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a regular matrix multiplication per (b,d) pair:
       out_batch = left_matrix @ right_matrixᵀ
6. Convert the result back to FP32, apply the output gate,
   a second Layer‑norm over the hidden dimension, and the final linear
   projection back to the original channel dimension.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ B) where
#   A : (batch, M, K)   (row‑major)
#   B : (batch, K, N)   (row‑major, i.e. Bᵀ of the original right tensor)
#   C : (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_batch,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)          # batch index (combined batch * hidden_dim)
    pid_m = tl.program_id(1)              # block row index
    pid_n = tl.program_id(2)              # block column index

    # Tile coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = A_ptr + pid_batch * stride_batch
    b_base = B_ptr + pid_batch * stride_batch
    c_base = C_ptr + pid_batch * stride_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K_tile)
        a_ptrs = a_base + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B tile (K_tile x N)
        b_ptrs = b_base + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write back C tile
    c_ptrs = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for the
    O(N³) contraction.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim = config["dim"])
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    x = input_tensor.to(torch.float32)
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections + gates (FP32)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Mask handling (if a mask is present)
    nomask = config.get("nomask", True)
    if not nomask:
        mask = mask.to(x.dtype).unsqueeze(-1)           # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gates (sigmoid)
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Prepare tensors for the batched GEMM (FP16, contiguous)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape

    # Rearrange: (B, hidden_dim, N, N) where rows=i, cols=k
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    # Right must be transposed on the last two axes: (k, j)
    right_perm_T = right_fp16.permute(0, 3, 2, 1).contiguous()

    batch_total = B * hidden_dim
    left_flat = left_perm.view(batch_total, N, N)       # (batch_total, N, N)
    right_flat = right_perm_T.view(batch_total, N, N)   # (batch_total, N, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 4. Launch the Triton batched GEMM kernel
    # ------------------------------------------------------------------
    stride_batch = left_flat.stride(0)   # distance between consecutive batches
    stride_am = left_flat.stride(1)      # row stride (M)
    stride_ak = left_flat.stride(2)      # col stride (K)
    stride_bk = right_flat.stride(1)     # row stride (K)  of Bᵀ
    stride_bn = right_flat.stride(2)     # col stride (N)  of Bᵀ
    stride_cm = out_flat.stride(1)       # row stride (M) of C
    stride_cn = out_flat.stride(2)       # col stride (N) of C

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_total,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        stride_batch,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 5. Remaining PyTorch post‑processing
    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim] and cast to FP32
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # Second LayerNorm (over hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # Apply the output gate (as in the reference implementation)
    out = out * out_gate

    # Final linear projection back to the original dimension
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)  # shape [B, N, N, dim]

    return out
