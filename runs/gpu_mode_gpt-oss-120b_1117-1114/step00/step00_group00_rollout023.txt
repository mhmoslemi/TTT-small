```python
import torch
import triton
import triton.language as tl

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k, stride_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    where left/right have shape (B, N, N, H)
    (i and j are the first two N dimensions, k the middle N dimension,
    h runs over the hidden dimension).
    The kernel tiles the i and j dimensions (BLOCK_M x BLOCK_N) and loops
    over k in tiles of size BLOCK_K.
    """
    pid_m = tl.program_id(0)  # i tile
    pid_n = tl.program_id(1)  # j tile
    pid_hb = tl.program_id(2) # combined batch and hidden index

    # Decode batch and hidden idx
    b = pid_hb // H
    h = pid_hb % H

    # Offsets for i, j, k
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    i_mask = i < N
    j_mask = j < N

    # Base addresses for this (b,h)
    offset_b = b * stride_b
    offset_h = h * stride_h
    left_base = left_ptr + offset_b + offset_h
    right_base = right_ptr + offset_b + offset_h
    out_base = out_ptr + offset_b + offset_h

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k-dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K + k
        k_mask = k_offset < N

        # Load left: shape (BLOCK_M, BLOCK_K) -> left[i, k, h]
        a_ptrs = left_base + (i[:, None] * stride_i) + (k_offset[None, :] * stride_k)
        a = tl.load(a_ptrs, mask = i_mask[:, None] & k_mask[None, :], other=0.0)

        # Load right transposed: shape (BLOCK_K, BLOCK_N) -> right[k, j, h]
        b_ptrs = right_base + (k_offset[:, None] * stride_k) + (j[None, :] * stride_i)
        b = tl.load(b_ptrs, mask = k_mask[:, None] & j_mask[None, :], other=0.0)

        # Matrix multiply-reduce over k
        acc += tl.dot(a, b)

    # Store result: out[i, j, h]
    c_ptrs = out_base + (i[:, None] * stride_i) + (j[None, :] * stride_k)
    tl.store(c_ptrs, acc, mask = i_mask[:, None] & j_mask[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” Triangle
    Multiplicative Update (TriMul) from AlphaFold3.

    Arguments:
        data: tuple (input, mask, weights, config)
            input  – Tensor of shape [B, N, N, dim]
            mask   – Tensor of shape [B, N, N] (float or bool)
            weights – dict of model parameters
            config – dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections and gating
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]  # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    # Linear projections (bias=False)
    left = torch.nn.functional.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask (if any). For “nomask” cases the mask is all ones.
    mask_unsq = mask.unsqueeze(-1).to(left.dtype)
    left = left * mask_unsq
    right = right * mask_unsq

    # Gating values
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) Triton kernel for the tri‑multiplicative update
    # ------------------------------------------------------------------
    # Allocate output tensor for the intermediate result (shape B,N,N,hidden_dim)
    out = torch.empty_like(left)

    # Strides in element units (not bytes)
    stride_b, stride_i, stride_k, stride_h = left.stride()  # (B,N,N,hidden)

    # Choose block sizes. These work well for the provided test configs.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # i tiles
        triton.cdiv(N, BLOCK_N),   # j tiles
        B * hidden_dim,            # batch × hidden dimension
    )

    trimul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_b, stride_i, stride_k, stride_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 4) Post‑processing: Norm on hidden dim, gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)

    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    # Apply output gate
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    B,
    N,
    H,
    stride_b,
    stride_i,
    stride_k,
    stride_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute for every batch b, hidden channel h:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

    left/right/out have shape (B, N, N, H).  The kernel tiles the i‑j
    plane (BLOCK_M × BLOCK_N) and iterates over k in tiles of size
    BLOCK_K.  The hidden dimension is handled by the third launch axis.
    """
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile
    pid_hb = tl.program_id(2)         # combined batch & hidden index

    # decode batch index and hidden channel index
    b = pid_hb // H
    h = pid_hb % H

    # offsets for the tiled dimensions
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    i_mask = i < N
    j_mask = j < N

    # base address for this (b, h)
    offset_b = b * stride_b
    offset_h = h * stride_h
    left_base = left_ptr + offset_b + offset_h
    right_base = right_ptr + offset_b + offset_h
    out_base = out_ptr + offset_b + offset_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K + k
        k_mask = k_offset < N

        # left[b, i, k, h]   -> shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + (i[:, None] * stride_i) + (k_offset[None, :] * stride_k)
        a = tl.load(a_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        # right[b, j, k, h] transposed to (k, j) -> shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + (k_offset[:, None] * stride_k) + (j[None, :] * stride_i)
        b = tl.load(b_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        acc += tl.dot(a, b)

    # write the result: out[b, i, j, h]
    c_ptrs = out_base + (i[:, None] * stride_i) + (j[None, :] * stride_k)
    tl.store(c_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” Triangle
    Multiplicative Update (TriMul) operator.

    Args:
        data: tuple (input, mask, weights, config)
            - input:  torch.Tensor [B, N, N, dim]
            - mask:   torch.Tensor [B, N, N] (float or bool)
            - weights: dict of model parameters
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections + gating
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Mask (if nomask the tensor is all‑ones)
    mask_unsq = mask.unsqueeze(-1).to(left.dtype)
    left = left * mask_unsq
    right = right * mask_unsq

    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) Triton kernel for the O(N³) contraction
    # ------------------------------------------------------------------
    out = torch.empty_like(left)  # shape (B, N, N, hidden_dim)

    # strides in element units (not bytes)
    stride_b, stride_i, stride_k, stride_h = left.stride()

    # block sizes – these work well for the target dimensions
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑tiles
        triton.cdiv(N, BLOCK_N),          # j‑tiles
        B * hidden_dim,                   # batch × hidden
    )

    trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_b,
        stride_i,
        stride_k,
        stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 4) Post‑processing: hidden‑dim LayerNorm, output gate, final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
