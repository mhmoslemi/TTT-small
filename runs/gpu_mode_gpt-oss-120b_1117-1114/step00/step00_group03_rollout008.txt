@triton.jit
def fuse_mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems,
    hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    # load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    if HAS_MASK:

        # but need to compute offset into mask. The mask offset = (offs // hidden_dim) (since mask has no hidden dim)
    else:
        mask_val = 1.0

    right = right * right_gate * mask_val

    # store
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Move weights to correct device & dtype
    device = input_tensor.device
    dtype = input_tensor.dtype

    # layernorm
    norm_weight = weights["norm.weight"].to(device=device, dtype=dtype)
    norm_bias = weights["norm.bias"].to(device=device, dtype=dtype)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    left_proj_weight = weights["left_proj.weight"].to(device=device, dtype=dtype)
    right_proj_weight = weights["right_proj.weight"].to(device=device, dtype=dtype)
    # Linear: output hidden_dim
    left = F.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Gates
    left_gate_weight = weights["left_gate.weight"].to(device=device, dtype=dtype)
    right_gate_weight = weights["right_gate.weight"].to(device=device, dtype=dtype)
    out_gate_weight = weights["out_gate.weight"].to(device=device, dtype=dtype)

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply mask + gates via Triton kernel (in-place)
    B, N, _, _ = input_tensor.shape  # dim final not needed
    # Ensure tensors contiguous
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    if not nomask:
        mask = mask_tensor.to(device=device, dtype=dtype).contiguous()
    else:
        mask = None

    total_elements = left.numel()  # B*N*N*hidden_dim
    BLOCK = 1024  # Tune blocksize
    grid = (triton.cdiv(total_elements, BLOCK),)

    # if mask present flatten it
    mask_flat = mask.view(-1) if mask is not None else torch.empty(0, device=device, dtype=dtype)

    fuse_mask_gate_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_flat,
        total_elements,
        hidden_dim,
        HAS_MASK= (not nomask),
        BLOCK_SIZE=BLOCK,
    )

    # Compute contraction using batched GEMM
    # Permute to [B, hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)  # shape [B, N, N, H]

    to_out_norm_bias = weights["to_out_norm.bias"].to(device=device, dtype=dtype)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply out_gate (multiply)
    out = out * out_gate

    # Final linear
    to_out_weight = weights["to_out.weight"].to(device=device, dtype=dtype)
    out = F.linear(out, to_out_weight)

    return out

"""
TriMul (outgoing) forward pass implemented with Triton.

The operation consists of:
1. Layer‑norm over the last channel of the input tensor.
2. Linear projections to a hidden dimension (left/right) and gating
   (left_gate, right_gate, out_gate).
3. Element‑wise fusion of mask (if present) and gating using a custom
   Triton kernel (in‑place on left/right).
4. Batched matrix multiplication over the sequence dimension:
      out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a PyTorch batched GEMM (torch.bmm) after permuting the
   tensors.
5. Layer‑norm on the hidden dimension, multiplication with out_gate,
   and final linear projection back to the original channel size.

All heavy element‑wise work (mask + gating) is fused in a Triton kernel,
while the O(N³) contraction leverages cuBLAS via torch.bmm.
The result is a float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask and gating (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def fuse_mask_gate_kernel(
    left_ptr,          # *float32
    right_ptr,         # *float32
    left_gate_ptr,     # *float32
    right_gate_ptr,    # *float32
    mask_ptr,          # *float32 (flattened mask, optional)
    total_elements,    # i64
    hidden_dim,        # i64
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Load mask value (broadcasted over hidden_dim)
    if HAS_MASK:
        # mask index = floor(offs / hidden_dim)
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    else:
        m = 1.0

    # Apply gating and mask
    left = left * left_gate * m
    right = right * right_gate * m

    # Store results back (in‑place)
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

# ----------------------------------------------------------------------

        - weights_dict : dict of torch.Tensor with keys:
            'norm.weight', 'norm.bias',
            'left_proj.weight', 'right_proj.weight',
            'left_gate.weight', 'right_gate.weight', 'out_gate.weight',
            'to_out_norm.weight', 'to_out_norm.bias',
            'to_out.weight'
        - config_dict  : dict with at least keys:
            'dim' (int), 'hidden_dim' (int), 'nomask' (bool, optional)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """

    dtype = input_tensor.dtype

    hidden_dim = config["hidden_dim"]

    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=dtype)
    norm_bias = weights["norm.bias"].to(device=device, dtype=dtype)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=dtype)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=dtype)

    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid of linear projects)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=dtype)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=dtype)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 4. Fuse mask and gating with Triton (in‑place on left/right)
    # ------------------------------------------------------------------
    # Ensure contiguity for flat pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # Prepare mask (flattened). If nomask, pass an empty tensor.
    if not nomask:
        mask = mask_tensor.to(device=device, dtype=dtype).contiguous()
        mask_flat = mask.view(-1)
    else:
        mask_flat = torch.empty(0, device=device, dtype=dtype)

    total_elements = left.numel()                     # B * N * N * hidden_dim
    BLOCK = 1024                                      # reasonable block size
    grid = (triton.cdiv(total_elements, BLOCK),)

    fuse_mask_gate_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_flat,
        total_elements,
        hidden_dim,
        HAS_MASK=not nomask,
        BLOCK_SIZE=BLOCK,
    )

    # ------------------------------------------------------------------
    # 5. Batched contraction over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Re‑order to bring hidden dimension to the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)            # [B*H, N, N]
    right_flat = right_perm.view(B * H, N, N)

    # Batched matrix multiply: (N×N) @ (N×N)^T  ->  (N×N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (hidden dimension)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype)
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # ------------------------------------------------------------------
    # 7. Multiply by out_gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype)
    out = F.linear(out, to_out_w)   # [B, N, N, dim]

    return out
```
