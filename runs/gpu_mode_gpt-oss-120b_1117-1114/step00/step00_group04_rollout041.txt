def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # offsets for i and j
    i_idx = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_idx = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_j = j_idx < N

        h_idx = h_start + tl.arange(0, BLOCK_H)
        mask_h = h_idx < H

        # Loop over k dimension
        for k_start in range(0, N, BLOCK_K):

            mask_k = k_idx < N

                pid_b * stride_left_b + \
                i_idx[:, None, None] * stride_left_i + \

                h_idx[None, None, :] * stride_left_h

"""
Optimized Triton implementation of the “outgoing” Triangle Multiplicative Update
(TriMul) used in AlphaFold3. The kernel computes the O(N³) reduction:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
All layer‑norms, linear projections and gating are performed in PyTorch.
Only the expensive inner product is fused into a Triton kernel that tiles the
i‑ and j‑axes and uses a per‑channel matmul (tl.dot) over the k‑axis.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    left_ptr,          # *fp16   left  tensor [B, N, N, H]
    right_ptr,         # *fp16   right tensor [B, N, N, H]
    out_ptr,           # *fp32   output tensor [B, N, N, H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)   # batch dimension
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # i / j indices for this tile
    i_idx = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_idx = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_idx < N
    mask_j = j_idx < N

    # number of tiles over hidden dimension
    num_h_tiles = tl.cdiv(H, BLOCK_H)

    for h_tile in range(0, num_h_tiles):
        h_base = h_tile * BLOCK_H

        # iterate over each channel inside the tile (BLOCK_H is small, e.g. 8)
        for dh in range(0, BLOCK_H):
            h = h_base + dh
            # guard against out‑of‑range channels (should not happen for test values)
            if h >= H:
                continue

            # accumulator for this (i, j, h) tile
            acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

            # loop over k‑tiles
            num_k_tiles = tl.cdiv(N, BLOCK_K)
            for k_tile in range(0, num_k_tiles):
                k_base = k_tile * BLOCK_K
                k_idx = k_base + tl.arange(0, BLOCK_K)
                mask_k = k_idx < N

                # left sub‑matrix [M, K] for channel h
                left_ptrs = left_ptr \
                    + pid_b * stride_left_b \
                    + i_idx[:, None] * stride_left_i \
                    + k_idx[None, :] * stride_left_k \
                    + h * stride_left_h
                left = tl.load(left_ptrs,
                               mask=mask_i[:, None] & mask_k[None, :],
                               other=0.0).to(tl.float32)

                # right sub‑matrix transposed [K, N] for channel h
                # we treat k as the first dimension, j as the second.
                right_ptrs = right_ptr \
                    + pid_b * stride_right_b \
                    + k_idx[:, None] * stride_right_i \
                    + j_idx[None, :] * stride_right_k \
                    + h * stride_right_h
                right_T = tl.load(right_ptrs,
                                  mask=mask_k[:, None] & mask_j[None, :],
                                  other=0.0).to(tl.float32)

                # accumulate matmul result
                acc += tl.dot(left, right_T)

            # write out the result for this channel
            out_ptrs = out_ptr \
                + pid_b * stride_out_b \
                + i_idx[:, None] * stride_out_i \
                + j_idx[None, :] * stride_out_j \
                + h * stride_out_h
            mask_out = mask_i[:, None] & mask_j[None, :]
            tl.store(out_ptrs, acc, mask=mask_out)


def custom_kernel(data):
    """
    Forward pass of TriMul with a Triton kernel for the O(N³) reduction.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch tensor [B, N, N, dim] (float32)
            - mask_tensor : torch tensor [B, N, N] (bool or float)
            - weights_dict : dict of model weights (torch tensors)
            - config_dict  : {"dim": int, "hidden_dim": int}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ----- Load weights ----------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ----- LayerNorm on input ------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ----- Linear projections ------------------------------------------------
    left = F.linear(x, left_proj_weight)      # [B, N, N, H]
    right = F.linear(x, right_proj_weight)    # [B, N, N, H]

    # ----- Apply mask (if provided) ----------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).type_as(x)  # broadcast to [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ----- Gating ------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ----- Cast to fp16 for the kernel (memory bandwidth) -------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ----- Allocate output for the kernel ------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ----- Compute strides (in elements) -------------------------------------
    stride_left = left.stride()   # (B, N, N, H)
    stride_right = right.stride()
    stride_out = out.stride()

    # Launch Triton kernel
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 32
    BLOCK_H = 8   # works because hidden_dim = 128 is divisible by 8

    grid = (B,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))
    trimul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
    )

    # ----- Final LayerNorm, gating and linear projection --------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
