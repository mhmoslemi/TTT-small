                  stride_left_b, stride_left_i, stride_left_k, stride_left_h,
                  stride_right_b, stride_right_j, stride_right_k, stride_right_h,
                  stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                  BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bd = tl.program_id(2)
    b = pid_bd // H
    d = pid_bd % H

    # Compute offsets
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    # Ensure within bounds
    mask_i = i < N
    mask_j = j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over k tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_tiles):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left tile (i,k)
        left_offset = b*stride_left_b + i[:,None]*stride_left_i + k[None,:]*stride_left_k + d*stride_left_h
        a = tl.load(left_ptr + left_offset, mask=mask_i[:,None] & mask_k[None,:], other=0.0).to(tl.float32)
        # right tile (k,j), need shape (BLOCK_K, BLOCK_J)
        right_offset = b*stride_right_b + k[:,None]*stride_right_k + j[None,:]*stride_right_j + d*stride_right_h
        b_tile = tl.load(right_ptr + right_offset, mask=mask_k[:,None] & mask_j[None,:], other=0.0).to(tl.float32)

        # dot product: (BLOCK_I, BLOCK_K) x (BLOCK_K, BLOCK_J) -> (BLOCK_I, BLOCK_J)
        acc += tl.dot(a, b_tile)

    # Store output
    out_offset = b*stride_out_b + i[:,None]*stride_out_i + j[None,:]*stride_out_j + d*stride_out_h
    tl.store(out_ptr + out_offset, acc.to(tl.float32), mask=mask_i[:,None] & mask_j[None,:])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, N, _, D = input_tensor.shape  # D = dim
    hidden_dim = config["hidden_dim"]
    # layernorm: normalized input_x with weight bias
    x = torch.nn.functional.layer_norm(input_tensor, (D,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # mask
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_
    # gating
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    left = left * left_gate
    right = right * right_gate
    # ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    # allocate out hidden
    out_hidden = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=device)
    # Triton kernel for multiplication
    import triton
    import triton.language as tl
    # define kernel above (global)
    # set block sizes
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    grid = (triton.cdiv(N, BLOCK_I), triton.cdiv(N, BLOCK_J), bs * hidden_dim)
    trimul_kernel[grid](
        left, right, out_hidden,
        bs, N, hidden_dim,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        out_hidden.stride(0), out_hidden.stride(1), out_hidden.stride(2), out_hidden.stride(3),
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K,
    )
    # to_out_norm
    out_norm = torch.nn.functional.layer_norm(out_hidden, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out_norm = out_norm * out_gate
    # final output
    out = torch.nn.functional.linear(out_norm, weights["to_out.weight"])
    return out

"""
TriMul (outgoing) forward pass optimized with Triton.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B, N, N, D]).
2. Linear projections → left/right (shape [B, N, N, H]).
3. Apply the binary mask and the per‑position gates.
4. Core O(N³) operation:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a fused batched GEMM in a Triton kernel.
5. Layer‑norm + gate on the hidden result.
6. Final linear projection back to dimension D.

The heavy reduction (step 4) is executed by a Triton kernel that
processes one output tile (BLOCK_I × BLOCK_J) per program instance
and loops over the reduction dimension K in tiles of size BLOCK_K.
The batch index and hidden channel are fused into the third program‑id
dimension (pid_bd), so each kernel launch handles a single (batch,
hidden‑channel) pair.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    left_ptr,           # *fp16   [B, N, N, H]
    right_ptr,          # *fp16   [B, N, N, H]
    out_ptr,            # *fp32   [B, N, N, H]
    B, N, H,            # int32 scalars
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the TriMul operation.

    Each program instance computes a tile (BLOCK_I x BLOCK_J) of the
    output matrix for a specific batch `b` and hidden channel `d`.
    """
    pid_i = tl.program_id(0)                      # tile along i axis
    pid_j = tl.program_id(1)                      # tile along j axis
    pid_bd = tl.program_id(2)                     # combined batch & hidden dim

    # Decode batch and hidden channel from pid_bd
    b = pid_bd // H
    d = pid_bd % H

    # Global coordinates for this tile
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = i < N
    mask_j = j < N

    # Accumulator for the tile (float32)
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over K dimension in tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ----- load left tile : shape (BLOCK_I, BLOCK_K) -----
        left_offset = (
            b * stride_l_b
            + i[:, None] * stride_l_i
            + k[None, :] * stride_l_k
            + d * stride_l_h
        )
        a = tl.load(
            left_ptr + left_offset,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        ).to(tl.float32)

        # ----- load right tile : shape (BLOCK_K, BLOCK_J) -----
        # note the order of indices (k, j) gives the required transpose
        right_offset = (
            b * stride_r_b
            + k[:, None] * stride_r_k
            + j[None, :] * stride_r_j
            + d * stride_r_h
        )
        b_tile = tl.load(
            right_ptr + right_offset,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
            dtype=tl.float16,
        ).to(tl.float32)

        # Accumulate the matrix product for this K‑tile
        acc += tl.dot(a, b_tile)   # (BLOCK_I, BLOCK_J)

    # ----- store the final tile -----
    out_offset = (
        b * stride_o_b
        + i[:, None] * stride_o_i
        + j[None, :] * stride_o_j
        + d * stride_o_h
    )
    tl.store(out_ptr + out_offset, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a fused Triton kernel.
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, D]  (float32)
            - mask: torch.Tensor [B, N, N] (bool / uint8 / float)
            - weights: dict of torch.Tensors (no gradients needed)
            - config: dict with keys "dim" (D) and "hidden_dim" (H)
    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]

    # ---------- 1. LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        (D,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B,N,N,D]

    # ---------- 2. Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])  # [B,N,N,H]

    # ---------- 3. Mask ----------
    if mask is not None:
        m = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * m
        right = right * m

    # ---------- 4. Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for the kernel (better bandwidth)
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # ---------- 5. Core O(N³) via Triton ----------
    out_hidden = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # Block sizes – chosen to fit B200 SM limits and give good occupancy
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
        B * H,                     # combined batch & hidden dimension
    )
    trimul_kernel[grid](
        left,
        right,
        out_hidden,
        B,
        N,
        H,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        out_hidden.stride(0), out_hidden.stride(1), out_hidden.stride(2), out_hidden.stride(3),
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # ---------- 6. Final normalization, gating and projection ----------
    out_norm = F.layer_norm(
        out_hidden,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate  # apply output gate

    output = F.linear(out_norm, weights["to_out.weight"])

    return output
