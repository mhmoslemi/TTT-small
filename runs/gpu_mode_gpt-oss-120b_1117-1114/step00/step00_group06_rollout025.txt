import torch
import triton
import triton.language as tl

@triton.jit
def apply_mask_gate_kernel(x_ptr, mask_ptr, gate_ptr, out_ptr,
                           total_elements, D,
                           BLOCK: tl.constexpr):
    """
    Elementswise kernel: out = x * gate * mask

    x, gate have shape [B, N, N, D] (contiguous, D innermost).
    mask has shape [B, N, N] and is broadcast over the D dimension.
    All tensors are expected in fp16.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # Load x and gate
    x = tl.load(x_ptr + offs, mask=active, other=tl.float16(0.0))
    g = tl.load(gate_ptr + offs, mask=active, other=tl.float16(0.0))

    # broadcast mask: same mask index for each D element
    mask_idx = offs // D
    m = tl.load(mask_ptr + mask_idx, mask=active, other=tl.float16(0.0))

    out = x * g * m
    tl.store(out_ptr + offs, out, mask=active)

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul (outgoing) module.
    Implements elementwise mask+gate fusion in Triton and uses
    batched matrix multiplication (torch.matmul) for the cubic contraction.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # LayerNorm on input (float32)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor,
                                       normalized_shape=(dim,),
                                       weight=norm_weight,
                                       bias=norm_bias)

    # Linear projections (float32 -> then cast to fp16)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Gates (float32 -> fp16)
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Cast to fp16 for the kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)
    out_gate = out_gate.to(torch.float16)
    mask_fp16 = mask.to(torch.float16)

    # Flatten mask for kernel (contiguous)
    mask_flat = mask_fp16.contiguous().view(-1)

    total_elements = bs * N * N * hidden_dim
    BLOCK = 1024

    # Apply mask and per‑position gates to `left`
    left_out = torch.empty_like(left)
    grid = (triton.cdiv(total_elements, BLOCK),)
    apply_mask_gate_kernel[grid](left, mask_flat, left_gate, left_out,
                                 total_elements, hidden_dim,
                                 BLOCK=BLOCK)
    left = left_out

    # Apply mask and per‑position gates to `right`
    right_out = torch.empty_like(right)
    apply_mask_gate_kernel[grid](right, mask_flat, right_gate, right_out,
                                 total_elements, hidden_dim,
                                 BLOCK=BLOCK)
    right = right_out

    # Compute the cubic contraction via batched GEMM:
    #   out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    # This is equivalent to (left_{d}) @ (right_{d})^T for each d.
    # We reshape to (B*hidden_dim, N, N) and use torch.matmul.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, D, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, D, N, N]
    # collapse batch and channel dimensions
    left_mat = left_perm.view(bs * hidden_dim, N, N)
    right_mat = right_perm.view(bs * hidden_dim, N, N)
    out_mat = torch.matmul(left_mat, right_mat)          # [B*D, N, N]
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1) # [B,N,N,D]

    # Post‑norm and output projection (float32)
    out = out.to(torch.float32)

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out,
                                         normalized_shape=(hidden_dim,),
                                         weight=to_out_norm_weight,
                                         bias=to_out_norm_bias)

    # Apply output gate
    out = out * out_gate.to(torch.float32)

    # Final linear projection back to dim
    to_out_weight = weights["to_out.weight"]
    output = torch.nn.functional.linear(out, to_out_weight)

    return output

import torch
import triton
import triton.language as tl


@triton.jit
def apply_mask_gate_kernel(x_ptr, mask_ptr, gate_ptr, out_ptr,
                           total_elements, D,
                           BLOCK: tl.constexpr):
    """
    Element‑wise fusion of mask and gate:
        out = x * gate * mask

    x, gate : [B, N, N, D]  (contiguous, D is the innermost dimension)
    mask    : [B, N, N]      (broadcast over D)
    All tensors are expected in fp16.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)               # linear offsets
    active = offs < total_elements

    # Load the data
    x = tl.load(x_ptr + offs, mask=active, other=tl.float16(0.0))
    g = tl.load(gate_ptr + offs, mask=active, other=tl.float16(0.0))

    # broadcast mask (same mask value for all D elements)
    mask_idx = offs // D
    m = tl.load(mask_ptr + mask_idx, mask=active, other=tl.float16(0.0))

    out = x * g * m
    tl.store(out_ptr + offs, out, mask=active)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Steps:
    1. Layer‑norm the input.
    2. Linear projections (left/right) and gate logits.
    3. Fuse mask & per‑position gates using a Triton kernel.
    4. Perform the cubic contraction via batched GEMM (left @ rightᵀ per channel).
    5. Apply the output LayerNorm, out‑gate and final projection.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -------- layer norm on the input -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # -------- linear projections (float32 -> fp16) ------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # -------- gates (float32 -> fp16) -------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, left_gate_weight)
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, right_gate_weight)
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, out_gate_weight)
    )

    # Cast everything to fp16 for the Triton kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)
    out_gate = out_gate.to(torch.float16)
    mask_fp16 = mask.to(torch.float16)

    # Flatten mask for 1‑D indexing inside the kernel
    mask_flat = mask_fp16.contiguous().view(-1)

    total_elements = bs * N * N * hidden_dim
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)

    # -------- fuse mask & gate for left ------------------------------------------------
    left_fused = torch.empty_like(left)
    apply_mask_gate_kernel[grid](
        left,
        mask_flat,
        left_gate,
        left_fused,
        total_elements,
        hidden_dim,

    # -------- fuse mask & gate for right -----------------------------------------------

        right_fused,
        total_elements,

        BLOCK=BLOCK,

    # This is equivalent to (left_d) @ (right_d)^T for each d.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, D, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, D, N, N]

    left_mat = left_perm.view(bs * hidden_dim, N, N)
    right_mat = right_perm.view(bs * hidden_dim, N, N)

    out_mat = torch.matmul(left_mat, right_mat)          # [B*D, N, N]

    out = (
        out_mat.view(bs, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()

    # -------- post‑norm, out‑gate and final projection ---------------------------------

    to_out_norm_bias = weights["to_out_norm.bias"]

    to_out_weight = weights["to_out.weight"]
    output = torch.nn.functional.linear(out, to_out_weight)  # [B, N, N, dim]
