"""
TriMul (outgoing) forward pass implemented with a fused Triton batched GEMM.
The heavy O(N³·C) reduction  out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
is executed as a batched matrix multiply C = A @ Bᵀ where each channel c is
treated as an independent batch element.  The surrounding LayerNorms,
Linear projections and gating are performed with PyTorch (highly‑optimized
CUDA kernels).  The kernel works for arbitrary batch size (≤2), sequence
length (≤1024) and hidden dimension (≤768).  Masking is optional.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM with B transposed (A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_T_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_az, stride_am, stride_ak,      # batch, M, K strides for A
    stride_bz, stride_bn, stride_bk,      # batch, N, K strides for B (row‑major)
    stride_cz, stride_cm, stride_cn,      # batch, M, N strides for C
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)   # output block row
    pid_n = tl.program_id(1)   # output block col
    pid_z = tl.program_id(2)   # batch * hidden_dim index

    # Offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the current batch slice
    a_batch_ptr = a_ptr + pid_z * stride_az
    b_batch_ptr = b_ptr + pid_z * stride_bz
    c_batch_ptr = c_ptr + pid_z * stride_cz

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile (M x K) – row major
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)

        # Load B tile as column‑major (K x N) so that we compute A @ Bᵀ
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Write out the result
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)

    Arguments:
        data: Tuple (input_tensor, mask, weights, config)
    Returns:
        Tensor with shape [batch, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]      # projection dimension

    # --------------------------------------------------------------
    # 1. LayerNorm on the input (last dimension = dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # --------------------------------------------------------------
    # 2. Linear projections
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (mask shape [B,N,N])
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)                # (B,N,N,1)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core TriMul: batched GEMM (A @ Bᵀ) over the hidden dimension
    # --------------------------------------------------------------
    B, N, _, _ = left.shape                                   # B, seq_len, hidden_dim
    # Cast to fp16 for bandwidth‑limited kernel, accumulate in fp32
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Move hidden dimension into batch axis: (B, hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    # Flatten batch*hidden -> total batches for the kernel
    total_batches = B * hidden_dim
    left = left.view(total_batches, N, N)
    right = right.view(total_batches, N, N)

    # Output buffer (fp16, will be cast back later)
    out = torch.empty_like(left, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        total_batches               # batch*hidden
    )

    _batched_gemm_T_kernel[grid](
        left,
        right,
        out,
        left.stride(0), left.stride(1), left.stride(2),
        right.stride(0), right.stride(1), right.stride(2),
        out.stride(0), out.stride(1), out.stride(2),
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Cast back to float32 and restore original layout
    out = out.to(torch.float32)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
