    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)  # (b * H + h)
    pid1 = tl.program_id(1)  # i tile
    pid2 = tl.program_id(2)  # j tile

    # decode batch and head
    b = pid0 // H
    h = pid0 % H

    # compute start indices
    i_start = pid1 * BLOCK_M
    j_start = pid2 * BLOCK_N

    # Create masks for out of bounds
    i_mask = i_start + tl.arange(0, BLOCK_M) < N
    j_mask = j_start + tl.arange(0, BLOCK_N) < N
    # combine later

    # Initialize accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):

        k_start = k * BLOCK_K
        # compute pointers for left tile (i,k,h)

                      + i_start * stride_left_i
                      + k_start * stride_left_k
                      + h * stride_left_h)

            left_tile_ptr

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid0 = tl.program_id(0)  # (b * H + h)
    pid1 = tl.program_id(1)  # i tile
    pid2 = tl.program_id(2)  # j tile

    b = pid0 // H
    h = pid0 % H

    i_start = pid1 * BLOCK_M
    j_start = pid2 * BLOCK_N

    i_range = tl.arange(0, BLOCK_M)
    j_range = tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    i_mask = i_start + i_range < N
    j_mask = j_start + j_range < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_tiles):
        k_start = kk * BLOCK_K

        # Load left tile (M, K)
        left_offset = (b * stride_left_b
                       + i_start * stride_left_i
                       + k_start * stride_left_k
                       + h * stride_left_h)
        left_ptrs = left_ptr + left_offset
        left_tile = tl.load(
            left_ptrs + (i_range[:, None] * stride_left_i
                         + k_range[None, :] * stride_left_k),
            mask=(i_mask[:, None] & (k_start + k_range[None, :] < N)),
            other=0.0, dtype=tl.float16)
        left_tile = left_tile.to(tl.float32)

        # Load right tile (N, K) (j as rows)
        right_offset = (b * stride_right_b
                        + j_start * stride_right_i
                        + k_start * stride_right_k
                        + h * stride_right_h)
        right_ptrs = right_ptr + right_offset
        right_tile = tl.load(
            right_ptrs + (j_range[:, None] * stride_right_i
                          + k_range[None, :] * stride_right_k),
            mask=(j_mask[:, None] & (k_start + k_range[None, :] < N)),
            other=0.0, dtype=tl.float16)
        right_tile = right_tile.to(tl.float32)

        # Broadcast and accumulate
        # shape (M, N, K)
        prod = left_tile[:, None, :] * right_tile[None, :, :]
        acc += tl.sum(prod, axis=2)

    # Write output
    out_offset = (b * stride_out_b
                  + i_start * stride_out_i
                  + j_start * stride_out_j
                  + h * stride_out_h)
    out_ptrs = out_ptr + out_offset
    out_mask = i_mask[:, None] & j_mask[None, :]
    tl.store(out_ptrs + (i_range[:, None] * stride_out_i
                         + j_range[None, :] * stride_out_j),
             acc, mask=out_mask)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Ensure types
    x = input_tensor.to(torch.float32)
    # LayerNorm
    x = F.layer_norm(x, normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])
    # Linear proj
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])
    # Apply mask
    if mask is None:
        mask_tensor = torch.ones((B, N, N), dtype=x.dtype, device=device)
    else:
        mask_tensor = mask.to(x.dtype)
    mask_tensor = mask_tensor.unsqueeze(-1)  # shape [B,N,N,1]
    left = left * mask_tensor
    right = right * mask_tensor
    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))
    # Apply gates
    left = (left * left_gate).contiguous()
    right = (right * right_gate).contiguous()
    # Cast to half for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    # Allocate output buffer
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)
    # Strides
    stride_left = left_fp16.stride()
    stride_right = right_fp16.stride()
    stride_out = out.stride()
    # Block sizes
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    # Grid dimensions
    num_i_tiles = triton.cdiv(N, BLOCK_M)
    num_j_tiles = triton.cdiv(N, BLOCK_N)
    grid = (B * hidden_dim, num_i_tiles, num_j_tiles)
    # Launch kernel
    tri_mul_kernel[grid](
        left_fp16, right_fp16, out,
        B, N, hidden_dim,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # Post processing
    out = F.layer_norm(out, normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
```

"""
TriMul (outgoing) forward pass accelerated with Triton.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension.
2. Linear projections to obtain `left` and `right` (shape [B,N,N,H]).
3. Apply the optional mask and the three sigmoid gates.
4. Compute the core contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   using a custom Triton kernel that performs a batched GEMM for each
   (batch, head) pair.  The kernel tiles the (i,j) output plane with
   BLOCK_M×BLOCK_N tiles and iterates over k in BLOCK_K steps,
   aggregating the partial results in FP32.
5. Layer‑norm over the hidden dimension, apply `out_gate`,
   and a final linear projection back to `dim`.

The kernel works for any batch size ≤2, sequence length up to 1024,
and hidden dimension 128 (the values used in the test suite).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication for the TriMul contraction.

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    """
    pid0 = tl.program_id(0)  # combined batch * head
    pid1 = tl.program_id(1)  # i‑tile
    pid2 = tl.program_id(2)  # j‑tile

    # decode batch and head
    b = pid0 // H
    h = pid0 % H

    # start indices of the current tile
    i_start = pid1 * BLOCK_M
    j_start = pid2 * BLOCK_N

    i_range = tl.arange(0, BLOCK_M)
    j_range = tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    i_mask = i_start + i_range < N
    j_mask = j_start + j_range < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_tiles):
        k_start = kk * BLOCK_K

        # ----- load left tile (M × K) -----
        left_offset = (
            b * stride_left_b
            + i_start * stride_left_i
            + k_start * stride_left_k
            + h * stride_left_h
        )
        left_ptr_tile = left_ptr + left_offset
        left_tile = tl.load(
            left_ptr_tile
            + (i_range[:, None] * stride_left_i + k_range[None, :] * stride_left_k),
            mask=(i_mask[:, None] & (k_start + k_range[None, :] < N)),
            other=0.0,
            dtype=tl.float32,
        )

        # ----- load right tile (N × K) -----
        right_offset = (
            b * stride_right_b
            + j_start * stride_right_i
            + k_start * stride_right_k
            + h * stride_right_h
        )
        right_ptr_tile = right_ptr + right_offset
        right_tile = tl.load(
            right_ptr_tile
            + (j_range[:, None] * stride_right_i + k_range[None, :] * stride_right_k),
            mask=(j_mask[:, None] & (k_start + k_range[None, :] < N)),
            other=0.0,
            dtype=tl.float32,
        )

        # ----- accumulate partial product -----
        # (M, K) * (N, K) -> (M, N) by broadcasting over K and summing
        prod = left_tile[:, None, :] * right_tile[None, :, :]
        acc += tl.sum(prod, axis=2)

    # ----- write the result -----
    out_offset = (
        b * stride_out_b
        + i_start * stride_out_i
        + j_start * stride_out_j
        + h * stride_out_h
    )
    out_ptr_tile = out_ptr + out_offset
    out_mask = i_mask[:, None] & j_mask[None, :]
    tl.store(
        out_ptr_tile
        + (i_range[:, None] * stride_out_i + j_range[None, :] * stride_out_j),
        acc,
        mask=out_mask,
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module with a Triton kernel for the
    expensive contraction.  The function expects:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension
    x = input_tensor.to(torch.float32)
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Mask (if any) – broadcast to channel dim
    if mask_tensor is None:
        mask = torch.ones((B, N, N), dtype=x.dtype, device=device)
    else:
        mask = mask_tensor.to(x.dtype)
    mask = mask.unsqueeze(-1)          # [B,N,N,1]
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # 4. Compute gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 5. Apply gates
    left = (left * left_gate).contiguous()
    right = (right * right_gate).contiguous()

    # ------------------------------------------------------------------
    # 6. Triton kernel for the O(N³) contraction
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # strides (in elements)
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    # tiling parameters
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    num_i_tiles = triton.cdiv(N, BLOCK_M)
    num_j_tiles = triton.cdiv(N, BLOCK_N)

    grid = (B * H, num_i_tiles, num_j_tiles)
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_left[0],
        s_left[1],
        s_left[2],
        s_left[3],
        s_right[0],
        s_right[1],
        s_right[2],
        s_right[3],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Post‑processing: LayerNorm, out_gate, final projection
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
