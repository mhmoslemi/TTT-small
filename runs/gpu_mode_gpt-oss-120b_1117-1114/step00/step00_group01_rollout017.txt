    B, N, H,

    stride_rb, stride_rj, stride_rk, stride_rh,

"""
TriMul (outgoing) implementation with Triton acceleration.

The heavy O(N³) contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is performed by a custom Triton GEMM kernel that treats each (batch, hidden) pair
as an independent matrix multiplication:
    out[b,i,j,d] = (left[b,:,:,d] @ right[b,:,:,d].T)[i,j]

All other operations (LayerNorm, linear projections, gating) are executed in
PyTorch. The kernel works for any batch size (≤2), sequence length (≤1024) and
hidden dimension (≤384). The final result is returned as a float32 tensor
with shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for each (batch, hidden) slice.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out = left @ rightᵀ for each (batch, hidden) slice.
    left/right shape: [B, N, N, H] (contiguous, last dim = hidden)
    out shape: same as above.
    """
    pid_m = tl.program_id(0)        # i‑tile
    pid_n = tl.program_id(1)        # j‑tile
    pid_z = tl.program_id(2)        # combined batch × hidden index

    # decode batch and hidden index
    batch_id = pid_z // H
    h_id     = pid_z % H

    # starting offsets for the tile
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # index vectors
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (batch, hidden) slice
    a_base = left_ptr  + batch_id * stride_lb + h_id * stride_lh
    b_base = right_ptr + batch_id * stride_rb + h_id * stride_rh
    c_base = out_ptr   + batch_id * stride_ob + h_id * stride_oh

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + (offs_i[:, None] * stride_li) + (offs_k[None, :] * stride_lk)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # B block transposed: we load as (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + (offs_k[:, None] * stride_rk) + (offs_j[None, :] * stride_rj)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # GEMM accumulation
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # store result
    c_ptrs = c_base + (offs_i[:, None] * stride_oi) + (offs_j[None, :] * stride_oj)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Arguments:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones((input_tensor.shape[0],
                           input_tensor.shape[1],
                           input_tensor.shape[2]),
                          device=device, dtype=left_proj.dtype)
    if nomask:
        # Mask might be all‑ones already; still broadcast safely.
        mask_ = mask.unsqueeze(-1)
    else:
        mask_ = mask.unsqueeze(-1)

    left = left_proj * mask_
    right = right_proj * mask_

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel: compute the O(N³) contraction
    # ------------------------------------------------------------------
    # Cast to contiguous tensors (float32) for the kernel
    left = left.contiguous()
    right = right.contiguous()
    B, N, _, _ = left.shape

    # Output placeholder (float32) – same layout as left/right
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    # Strides for the three tensors
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    _trimul_gemm_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Post‑processing: layer‑norm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=1e-5)

    out = out * out_gate
    final_out = F.linear(out, weights["to_out.weight"])   # (dim, hidden_dim)

    return final_out
