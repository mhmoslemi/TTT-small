"""
custom_kernel: Triton-accelerated outgoing triangle multiplicative update (TriMul).

The kernel performs the expensive contraction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where left/right are the projected and gated tensors of shape
[B, N, N, hidden_dim].

Implementation steps:
1. Layer‑norm the input.
2. Linear projections (left/right) + optional mask.
3. Gated element‑wise multiplication.
4. Fuse the O(N³) contraction with a custom Triton batched‑matmul kernel.
5. Apply output LayerNorm, gating and final linear projection back to dim.

All heavy computation (step 4) runs in fp16 with float32 accumulation.
The final result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
# A, B, C : [batch, M, K] , [batch, K, N] , [batch, M, N] (here M=N=K=seq_len)
# -------------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,                     # matrix dimensions
    stride_Ab, stride_Ai, stride_Ak,   # strides for A (batch, i, k)
    stride_Bb, stride_Bk, stride_Bn,   # strides for B (batch, k, n)
    stride_Cb, stride_Ci, stride_Cn,   # strides for C (batch, i, n)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    batch = tl.program_id(2)   # batch index (B * hidden_dim)

    # start offsets of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A[batch, i, k]
        a_ptrs = (
            A_ptr
            + batch * stride_Ab
            + offs_m[:, None] * stride_Ai
            + offs_k[None, :] * stride_Ak
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # B[batch, k, n]  (B is used as‑is, which corresponds to the transposed operand)
        b_ptrs = (
            B_ptr
            + batch * stride_Bb
            + offs_k[:, None] * stride_Bk
            + offs_n[None, :] * stride_Bn
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # accumulate block‑wise product
        acc += tl.dot(a, b)

    # write result back in FP16
    c = acc.to(tl.float16)
    c_ptrs = (
        C_ptr
        + batch * stride_Cb
        + offs_m[:, None] * stride_Ci
        + offs_n[None, :] * stride_Cn
    )
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# -------------------------------------------------------------------------
# Host‑side entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Perform the outgoing TriMul operation using a Triton kernel for the
    high‑cost O(N³) contraction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [batch, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    seq_len = input_tensor.shape[1]  # N
    batch_size = input_tensor.shape[0]

    # -----------------------------------------------------------------
    # 1) LayerNorm on the input (over the last dimension)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias) + optional mask
    # -----------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    if not config.get("nomask", False):
        # mask shape: [B,N,N] -> [B,N,N,1] for broadcast
        mask_f = mask.to(dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 3) Gating
    # -----------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = (left * left_gate).contiguous()
    right = (right * right_gate).contiguous()

    # Cast to fp16 for the heavy contraction
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # -----------------------------------------------------------------
    # 4) Fuse the contraction with Triton
    #    Reshape to (batch * hidden_dim, N, N) to treat each channel as
    #    an independent matrix multiplication.
    # -----------------------------------------------------------------
    # Permute to bring hidden_dim in front of the two N axes
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()

    # Fuse batch and channel dimensions
    total_batches = batch_size * hidden_dim
    left_flat = left_perm.view(total_batches, seq_len, seq_len)
    right_flat = right_perm.view(total_batches, seq_len, seq_len)

    # Allocate output tensor (fp16)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # Strides (in elements)
    stride_Ab = left_flat.stride(0)
    stride_Ai = left_flat.stride(1)
    stride_Ak = left_flat.stride(2)

    stride_Bb = right_flat.stride(0)
    stride_Bk = right_flat.stride(1)   # acts as 'K' stride
    stride_Bn = right_flat.stride(2)

    stride_Cb = out_flat.stride(0)
    stride_Ci = out_flat.stride(1)
    stride_Cn = out_flat.stride(2)

    # Block sizes – tuned for typical B200 dimensions
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(seq_len, BLOCK_M),
        triton.cdiv(seq_len, BLOCK_N),
        total_batches,
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        seq_len,
        seq_len,
        seq_len,
        stride_Ab,
        stride_Ai,
        stride_Ak,
        stride_Bb,
        stride_Bk,
        stride_Bn,
        stride_Cb,
        stride_Ci,
        stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(batch_size, hidden_dim, seq_len, seq_len)
    out = out.permute(0, 2, 3, 1).contiguous().to(torch.float32)   # [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # 5) Output LayerNorm, gating, final linear projection
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate   # broadcast over hidden_dim

    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)   # [B,N,N,dim]

    return out
