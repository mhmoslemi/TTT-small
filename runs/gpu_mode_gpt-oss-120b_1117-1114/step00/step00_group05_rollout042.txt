The heavy pairwise multiplication
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is performed with a batched GEMM via torch.bmm (cuBLAS).  Masking
and gating are fused in a Triton kernel that processes the hidden
dimension in small blocks (default 32).  All other linear layers and
LayerNorms are executed with torch.nn.functional for simplicity.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left/right gates and the optional boolean mask
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    left_proj_ptr, right_proj_ptr,            # [B,N,N,H] raw projections
    left_gate_ptr,  right_gate_ptr,           # [B,N,N,H] gate tensors (sigmoid already)
    mask_ptr,                                 # [B,N,N]    mask (float, 0/1)
    left_out_ptr,  right_out_ptr,             # [B,N,N,H] output tensors
    B, N, H,                                  # dimensions
    stride_b, stride_i, stride_k, stride_h,   # strides for projection tensors
    mask_stride_b, mask_stride_i, mask_stride_k,  # strides for mask
    d_offset: tl.constexpr, BLOCK_D: tl.constexpr
):
    """Mask + gate fusion.
    Each program processes one (b,i,k) element and a BLOCK_D slice of the hidden dimension.
    """
    pid = tl.program_id(0)
    total = B * N * N
    idx = pid
    valid = idx < total

    # ---- decode linear index into (b,i,k) -------------------------
    b = tl.div(idx, N * N)
    rem = idx - b * (N * N)
    i = tl.div(rem, N)
    k = rem - i * N

    # ---- base pointer for (b,i,k) element (ignoring hidden dim) --
    base = b * stride_b + i * stride_i + k * stride_k + d_offset * stride_h

    # ---- hidden‑dim slice -----------------------------------------
    d_range = tl.arange(0, BLOCK_D)
    d_valid = (d_offset + d_range) < H
    mask_vec = valid & d_valid

    # ---- load tensors (masked) ------------------------------------
    left  = tl.load(left_proj_ptr  + base + d_range * stride_h, mask=mask_vec, other=0.0)
    right = tl.load(right_proj_ptr + base + d_range * stride_h, mask=mask_vec, other=0.0)
    lgate = tl.load(left_gate_ptr  + base + d_range * stride_h, mask=mask_vec, other=0.0)
    rgate = tl.load(right_gate_ptr + base + d_range * stride_h, mask=mask_vec, other=0.0)

    # mask scalar (broadcasted later)
    m = tl.load(mask_ptr + b * mask_stride_b + i * mask_stride_i + k * mask_stride_k,
                mask=valid, other=0.0)

    # ---- apply mask and gates --------------------------------------
    left_out  = left  * m * lgate
    right_out = right * m * rgate

    # ---- store results ---------------------------------------------
    tl.store(left_out_ptr  + base + d_range * stride_h, left_out,  mask=mask_vec)
    tl.store(right_out_ptr + base + d_range * stride_h, right_out, mask=mask_vec)

    m_stride_k = mask_f.stride(2)

    # launch kernel in blocks over the hidden dimension
    BLOCK_D = 32  # fits nicely into registers; will loop over hidden dim
    total = B * N * N
    grid = (triton.cdiv(total, 1),)   # one program per (b,i,k) element

    for d_start in range(0, H, BLOCK_D):
        _apply_mask_gate_kernel[grid](
            left_proj, right_proj,
            left_gate, right_gate,
            mask_f,
            left, right,
            B, N, H,
            stride_b, stride_i, stride_k, stride_h,
            m_stride_b, m_stride_i, m_stride_k,
            d_offset=d_start,
            BLOCK_D=BLOCK_D,
        )

    # ------------------------------------------------------------------
    # Core multiplicative update: batched GEMM over the sequence dimension
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) and do left * right^T for each hidden slice
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_t = right.permute(0, 3, 1, 2).contiguous()

    left_flat  = left_t.view(B * H, N, N)            # [BH,N,N]
    right_flat = right_t.view(B * H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # [BH,N,N]

    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # to_out_norm (LayerNorm over hidden dim) and final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate

    # final linear projection back to original channel dim C
    output = F.linear(out_norm, weights["to_out.weight"])

    return output
