def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_j, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Compute pointers
        left_ptrs = left_ptr + batch * stride_left_batch + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        right_ptrs = right_ptr + batch * stride_right_batch + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j

        # Load blocks
        left = tl.load(left_ptrs,
            mask=(offs_i[:, None] < N) & (offs_k[None, :] < K),
            other=0.0)

        right = tl.load(right_ptrs,
            mask=(offs_k[:, None] < K) & (offs_j[None, :] < N),
            other=0.0)

        # Compute dot product across K dimension
        acc += tl.dot(left, right)

        # Advance pointers, but we can just recompute in next iteration

    # Store result
    out_ptrs = out_ptr + batch * stride_out_batch + offs_i[:, None] * stride_out_i + offs_j[None,:] * stride_out_j
    mask = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # Load weights: keep as torch tensors.
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Normalize input: layernorm over last dim
    # shape [B,N,N,dim]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    # Ensure dtype fp32 for linear (but weights are fp32)
    # Compute projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if mask is not None? In given data, mask is always provided. Might be all ones or random.
    # Convert mask to correct dtype.
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1).to(left.dtype)  # broadcast over hidden_dim
        left = left * mask_unsq
        right = right * mask_unsq

@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets within the batch
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for valid indices
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + batch * stride_left_batch + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0)

        # Load right block transposed: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + batch * stride_right_batch + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        right = tl.load(right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0)

        # Accumulate
        acc += tl.dot(left, right)

    # Store result
    out_ptrs = out_ptr + batch * stride_out_batch + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask)

"""
TriMul (outgoing) forward pass implemented with a custom Triton kernel.

Algorithm
---------
1. Layer‑normalize the 4‑D input tensor.
2. Linear projections (left/right) and gating (left/right/out) are performed in
   PyTorch (FP32).
3. After gating, the tensors are cast to FP16 and reshaped so that the
   hidden dimension becomes part of the batch dimension:
       [B, N, N, H] -> [B·H, N, N]
   This turns the core operation into a batched matrix multiplication:
       out[b·h, i, j] = Σ_k left[b·h, i, k] * right[b·h, j, k]
   which is exactly the `einsum('... i k d, ... j k d -> ... i j d')` from the
   reference implementation.
4. The batched matmul is executed by a Triton kernel that tiles the
   (i, j) axes and iterates over K in blocks.  Accumulation is performed in
   FP32 for numerical stability and the result is written back as FP16.
5. The result is reshaped back to [B, N, N, H], converted to FP32, layer‑norm
   over the hidden dimension, gated by the out‑gate, and finally projected
   back to the original dimension.

The kernel is deliberately simple (no shared memory or complex prefetching)
but provides a non‑trivial GPU‑accelerated component required by the task.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    N, K,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication for the TriMul core:
        out = left @ right.T   where left,right ∈ [B·H, N, N]
    The kernel works on FP16 inputs and accumulates in FP32.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets within the current batch
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension (the summation axis)
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a block of left   : shape (BLOCK_M, BLOCK_K)
        left_ptrs = (
            left_ptr
            + batch_id * stride_left_batch
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load a transposed block of right : shape (BLOCK_K, BLOCK_N)
        right_ptrs = (
            right_ptr
            + batch_id * stride_right_batch
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_i
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # Accumulate the dot product over K
        acc += tl.dot(left, right)   # (M×K)·(K×N) → (M×N)

    # Write the result back as FP16
    out_ptrs = (
        out_ptr
        + batch_id * stride_out_batch
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    store_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=store_mask)


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module with a Triton‑accelerated
    O(N³) summation.

    Args:
        data: Tuple (input, mask, weights, config)
            - input:  torch.Tensor of shape [B, N, N, C]  (C == dim)
            - mask:   torch.Tensor of shape [B, N, N] (may be all‑ones)
            - weights: dict containing the model parameters
            - config:  dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (dtype float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # load weights (all are FP32 tensors)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  (FP32)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) – still in FP32
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if present) – broadcast on hidden dimension
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – FP32
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Cast to FP16 before the heavy O(N³) sum
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape

    # ------------------------------------------------------------------
    # 6️⃣ Reshape so that the hidden dimension becomes part of the batch:
    #        [B, N, N, H] -> [B·H, N, N]
    # ------------------------------------------------------------------
    left_h = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_h = right_h.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_bt = left_h.view(B * hidden_dim, N, N)   # [B*H, N, N]
    right_bt = right_h.view(B * hidden_dim, N, N) # [B*H, N, N]

    # Output buffer for the kernel (FP16)
    out_bt = torch.empty_like(left_bt)

    # ------------------------------------------------------------------
    # 7️⃣ Triton kernel launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                     # batch dimension (including hidden)
        triton.cdiv(N, BLOCK_M),           # i‑tiles
        triton.cdiv(N, BLOCK_N),           # j‑tiles
    )

    # Strides (in element counts, not bytes)
    s_left_b, s_left_i, s_left_k = left_bt.stride()
    s_right_b, s_right_i, s_right_k = right_bt.stride()
    s_out_b, s_out_i, s_out_j = out_bt.stride()

    # Launch the kernel
    tri_mul_matmul_kernel[grid](
        left_bt,
        right_bt,
        out_bt,
        s_left_b,
        s_left_i,
        s_left_k,
        s_right_b,
        s_right_i,
        s_right_k,
        s_out_b,
        s_out_i,
        s_out_j,
        N,          # M dimension
        N,          # K dimension (same as N)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Reshape back to [B, N, N, H] and cast to FP32
    # ------------------------------------------------------------------
    out_h = out_bt.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out_h.to(torch.float32)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 9️⃣ Final LayerNorm, gating, and projection back to `dim`
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
