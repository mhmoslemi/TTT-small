def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul (outgoing) operator.
    The heavy O(N^3) reduction is performed by a custom batched GEMM kernel.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ----- 1. LayerNorm over dim -----
    # weight/bias are float32
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ----- 2. Linear projections (no bias) -----
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ----- 3. Mask (if provided) -----
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ----- 4. Gates -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----- 5. Core TriMul reduction via Triton -----
    # Cast to fp16 for the kernel (half precision)
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # Move hidden dimension into the batch axis for a batched matmul
    B, N, _, H = left_f16.shape  # H == hidden_dim
    left_flat = left_f16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right_f16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)  # (B*H, N, N), fp16

    # strides
    stride_am, stride_ai, stride_ak = left_flat.stride()
    stride_bm, stride_bn, stride_bk = right_flat.stride()
    stride_cm, stride_ci, stride_cj = out_flat.stride()

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bn,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ----- 6. Output LayerNorm and gating -----
    out_norm_weight = weights["to_out_norm.weight"]
    out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=out_norm_weight, bias=out_norm_bias)

    out = out * out_gate  # out_gate is float32, broadcast matches

    # ----- 7. Final linear projection -----
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
Triton‑accelerated forward pass of the “outgoing” TriMul operator.

The expensive O(N³) reduction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is performed by a custom batched GEMM kernel.  We first apply the
pre‑projection LayerNorm, linear projections and gating in PyTorch,
reshape the tensors so that the hidden dimension becomes part of the
batch axis and launch a Triton kernel that computes a batched matrix
multiply with the second operand transposed.  The result is reshaped
back, passed through the output LayerNorm, gated and finally linearly
projected to the original feature dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul of A @ Bᵀ  (A,B : [M,N,K] with K=N)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bn, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_b = tl.program_id(2)          # batch‑hidden element (b*hidden + d)

    # Offsets for this tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < K

    # Base pointer for the current batch‑hidden slice
    batch_off_a = pid_b * stride_am
    batch_off_b = pid_b * stride_bm
    batch_off_c = pid_b * stride_cm

    a_ptrs = a_ptr + batch_off_a + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_off_b + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k):
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0, dtype=tl.bfloat16)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0, dtype=tl.bfloat16)
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    c = acc.to(tl.bfloat16)
    c_ptrs = c_ptr + batch_off_c + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[:, None])

def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module with the core
    reduction off‑loaded to Triton.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the feature dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (mask is [B,N,N])
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(x.dtype)
        mask = mask.unsqueeze(-1)                     # -> [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul reduction via Triton
    # ------------------------------------------------------------------
    # Cast to bfloat16 for the kernel (matches the reference implementation)
    left = left.to(torch.bfloat16)
    right = right.to(torch.bfloat16)

    B, N, _, H = left.shape      # H == hidden_dim
    # Move hidden dimension into the batch axis: (B, H, N, N) -> (M, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)          # (M, N, N), bfloat16

    # strides in element units
    stride_am, stride_ai, stride_ak = left_flat.stride()
    stride_bm, stride_bn, stride_bk = right_flat.stride()
    stride_cm, stride_ci, stride_cj = out_flat.stride()

    # kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        B * H,                     # batch‑hidden slices
    )

    _tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bn,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim] and convert to float32
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
