@triton.jit
def layernorm_gate_kernel(
    inp_ptr, gate_ptr, norm_weight_ptr, norm_bias_ptr, out_ptr,
    B, N, H,
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_H: tl.constexpr,
    EPS: tl.constexpr = 1e-5
):
    pid = tl.program_id(0)
    # compute linear index
    total = B * N * N
    if pid >= total: return

    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # Base offsets
    base_in = inp_ptr + b * stride_in_b + i * stride_in_i + j * stride_in_j
    base_gate = gate_ptr + b * stride_gate_b + i * stride_gate_i + j * stride_gate_j
    base_out = out_ptr + b * stride_out_b + i * stride_out_i + j * stride_out_j

    # Compute sum and sum_sq across hidden
    sum_val = tl.zeros((1,), dtype=tl.float32)
    sum_sq = tl.zeros((1,), dtype=tl.float32)
    for offset in range(0, H, BLOCK_H):
        cur = min(BLOCK_H, H - offset)
        # offsets vector
        hid = offset + tl.arange(0, BLOCK_H)
        mask = hid < H

        # load values
        val = tl.load(base_in + hid * stride_in_h, mask=mask, other=0.0)
        sum_val += tl.sum(val, mask=mask)
        sum_sq += tl.sum(val * val, mask=mask)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # Now compute normalized, gated, and write
    for offset in range(0, H, BLOCK_H):
        cur = min(BLOCK_H, H - offset)
        hid = offset + tl.arange(0, BLOCK_H)
        mask = hid < H

        # load
        val = tl.load(base_in + hid * stride_in_h, mask=mask, other=0.0)
        gate = tl.load(base_gate + hid * stride_gate_h, mask=mask, other=0.0)
        # norm weight and bias
        weight = tl.load(norm_weight_ptr + hid, mask=mask, other=0.0)
        bias = tl.load(norm_bias_ptr + hid, mask=mask, other=0.0)

        # normalize
        normed = (val - mean) * inv_std
        # apply weight and bias
        normed = normed * weight + bias
        # apply gate
        out_val = normed * gate

        # store
        tl.store(base_out + hid * stride_out_h, out_val, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Ensure dtype
    dtype = input_tensor.dtype
    if dtype != torch.float32:
        input_tensor = input_tensor.to(torch.float32)

    # LayerNorm on input:
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    eps = 1e-5
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)   # shape (B,N,N,dim)

    # linear projections
    left_proj_weight = weights['left_proj.weight']  # shape (hidden_dim, dim)
    right_proj_weight = weights['right_proj.weight']
    left = F.linear(x, left_proj_weight)  # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)

    # Gate linear weights
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Mask
    if mask is not None:
        # mask shape (B,N,N); cast
        mask_float = mask.to(dtype=x.dtype).unsqueeze(-1)  # shape (B,N,N,1)
        left = left * mask_float
        right = right * mask_float

    # Apply gate
    left = left * left_gate
    right = right * right_gate

    # Compute matmul (left @ right^T). Use torch.bmm
    # reshape
    # left: (B,N,N,hidden) -> (B, hidden, N, N) after permute
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # reshape to (B*hidden, N, N)
    left_flat = left_perm.view(bs * hidden_dim, N, N)
    right_flat = right_perm.view(bs * hidden_dim, N, N)

    out_mat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous() # shape (B,N,N,hidden)

    to_out_norm_bias = weights['to_out_norm.bias']

    Bsize, Nsize = bs, N

    # total elements Bsize*Nsize*Nsize

    # define block size for kernel

    # define strides for out and gate (maybe contiguous memory)
    s_out_b, s_out_i, s_out_j, s_out_h = out.stride()

    s_norm_out_b, s_norm_out_i, s_norm_out_j, s_norm_out_h = out_norm_gated.stride()

@triton.jit
def layernorm_gate_kernel(
    inp_ptr, gate_ptr, norm_weight_ptr, norm_bias_ptr, out_ptr,
    B, N, H,
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_H: tl.constexpr,
    EPS: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return
    # compute indices
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # compute base pointers
    inp_base = inp_ptr + b * stride_in_b + i * stride_in_i + j * stride_in_j
    gate_base = gate_ptr + b * stride_gate_b + i * stride_gate_i + j * stride_gate_j
    out_base = out_ptr + b * stride_out_b + i * stride_out_i + j * stride_out_j

    # compute sum and sum_sq across hidden
    sum_val = tl.zeros((1,), dtype=tl.float32)
    sum_sq = tl.zeros((1,), dtype=tl.float32)

    # loop over hidden dimension
    for start in range(0, H, BLOCK_H):
        cur = tl.minimum(BLOCK_H, H - start)
        # hidden offset vector
        h_offs = start + tl.arange(0, BLOCK_H)
        mask = h_offs < H
        # load values
        val = tl.load(inp_base + h_offs * stride_in_h, mask=mask, other=0.0)
        sum_val += tl.sum(val, mask=mask)
        sum_sq += tl.sum(val * val, mask=mask)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # now compute normalized, bias, gating and store
    for start in range(0, H, BLOCK_H):
        cur = tl.minimum(BLOCK_H, H - start)
        h_offs = start + tl.arange(0, BLOCK_H)
        mask = h_offs < H
        val = tl.load(inp_base + h_offs * stride_in_h, mask=mask, other=0.0)
        gate = tl.load(gate_base + h_offs * stride_gate_h, mask=mask, other=0.0)
        weight = tl.load(norm_weight_ptr + h_offs, mask=mask, other=0.0)
        bias = tl.load(norm_bias_ptr + h_offs, mask=mask, other=0.0)
        normed = (val - mean) * inv_std
        normed = normed * weight + bias
        out_val = normed * gate
        tl.store(out_base + h_offs * stride_out_h, out_val, mask=mask)

"""TriMul forward pass with a fused Triton kernel.

The implementation follows the reference PyTorch TriMul module:
1. Layer‑norm the input (dim‑wise).
2. Linear projections (left/right) and gated projections (left/right/out).
3. Optional mask application.
4. Core multiplicative update: for each hidden channel h,
   out[...,h] = left[...,h] @ right[...,h].T   (batched GEMM via torch.bmm).
5. A Triton kernel fuses the second LayerNorm (over the hidden dimension)
   with the out‑gate multiplication.
6. Final linear projection to the original channel dimension.

Only the last normalization+gate step is implemented in Triton; the heavy
matrix multiplication uses cuBLAS (torch.bmm) for high performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _layernorm_gate_kernel(
    inp_ptr,                # float32*   (B,N,N,H)
    gate_ptr,               # float32*   (B,N,N,H)
    norm_weight_ptr,        # float32*   (H,)
    norm_bias_ptr,          # float32*   (H,)
    out_ptr,                # float32*   (B,N,N,H)  output
    B, N, H,                # dimensions
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,   # strides for inp
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,  # strides for gate
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,      # strides for out
    BLOCK_H: tl.constexpr,  # tile size for hidden dim
    EPS: tl.constexpr = 1e-5,
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode (b,i,j) from linear pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # base pointers for this (b,i,j)
    inp_base = inp_ptr + b * stride_in_b + i * stride_in_i + j * stride_in_j
    gate_base = gate_ptr + b * stride_gate_b + i * stride_gate_i + j * stride_gate_j
    out_base = out_ptr + b * stride_out_b + i * stride_out_i + j * stride_out_j

    # --- first pass: compute mean and variance over hidden dimension ---
    sum_val = tl.zeros((1,), dtype=tl.float32)
    sum_sq = tl.zeros((1,), dtype=tl.float32)

    offset = 0
    while offset < H:
        h_offs = offset + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask = h_offs < H
        # load hidden slice (zero‑pad out‑of‑range entries)
        vals = tl.load(inp_base + h_offs * stride_in_h, mask=mask, other=0.0)
        sum_val += tl.sum(vals)                # scalar
        sum_sq += tl.sum(vals * vals)
        offset += BLOCK_H

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # --- second pass: normalize, apply weight/bias and out_gate ---
    offset = 0
    while offset < H:
        h_offs = offset + tl.arange(0, BLOCK_H)
        mask = h_offs < H

        vals = tl.load(inp_base + h_offs * stride_in_h, mask=mask, other=0.0)
        gate = tl.load(gate_base + h_offs * stride_gate_h, mask=mask, other=0.0)

        w = tl.load(norm_weight_ptr + h_offs, mask=mask, other=0.0)
        b_ = tl.load(norm_bias_ptr + h_offs, mask=mask, other=0.0)

        normed = (vals - mean) * inv_std          # layer‑norm
        normed = normed * w + b_                  # weight & bias
        out_val = normed * gate                    # apply out‑gate

        tl.store(out_base + h_offs * stride_out_h, out_val, mask=mask)
        offset += BLOCK_H


def custom_kernel(data):
    """
    Forward pass of the Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be all‑ones)
            - weights: dict of model parameters (see problem statement)
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ----- LayerNorm over the input channel dimension -----
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # (B,N,N,dim)

    # ----- Linear projections -----
    left = F.linear(x, weights["left_proj.weight"])          # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"])

    # ----- Gates (sigmoid) -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ----- Optional mask -----
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ----- Apply gates to projections -----
    left = left * left_gate
    right = right * right_gate

    # ----- Core multiplicative update (batched GEMM) -----
    # reshape to (B*H, N, N) for torch.bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_perm.view(bs * hidden_dim, N, N)
    right_flat = right_perm.view(bs * hidden_dim, N, N)

    # out_tmp: (B*H, N, N)
    out_tmp = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B,N,N,H)
    out = out_tmp.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----- Fuse to_out_norm + out_gate (Triton) -----
    out_norm_gated = torch.empty_like(out)  # (B,N,N,H)

    # Strides (in elements)
    s_in = out.stride()          # (stride_b, stride_i, stride_j, stride_h)
    s_gate = out_gate.stride()
    s_out = out_norm_gated.stride()

    total_pairs = bs * N * N
    grid = (total_pairs,)

    _layernorm_gate_kernel[grid](
        out,
        out_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_norm_gated,
        bs,
        N,
        hidden_dim,
        s_in[0],
        s_in[1],
        s_in[2],
        s_in[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_H=32,
        EPS=1e-5,
    )

    # ----- Final linear projection to original dim -----
    output = F.linear(out_norm_gated, weights["to_out.weight"])  # (B,N,N,dim)

    return output.to(torch.float32)
