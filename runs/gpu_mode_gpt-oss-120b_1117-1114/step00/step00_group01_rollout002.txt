def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
    M, N, K, BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(axis=0) # batch dimension
    pid_m = tl.program_id(axis=1) # row tile
    pid_n = tl.program_id(axis=2) # col tile

    # compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute mask for out-of-range
    m_mask = offs_m < M
    n_mask = offs_n < N

    # pointers to base for this batch
    a_batch_ptr = a_ptr + pid * batch_stride_a
    b_batch_ptr = b_ptr + pid * batch_stride_b
    c_batch_ptr = c_ptr + pid * batch_stride_c

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):

        offs_k = k_base + tl.arange(0, BLOCK_K)
        k_mask = offs_k < K

        # compute pointers for this tile

@triton.jit
def batched_matmul_kernel(a_ptr, b_ptr, c_ptr,
                          stride_a_batch, stride_a_m, stride_a_k,
                          stride_b_batch, stride_b_k, stride_b_n,
                          stride_c_batch, stride_c_m, stride_c_n,
                          M, N, K,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offset indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask within M, N
    m_mask = offs_m < M
    n_mask = offs_n < N

    # pointer to start of this batch's matrices
    a_ptrs = a_ptr + batch * stride_a_batch
    b_ptrs = b_ptr + batch * stride_b_batch
    c_ptrs = c_ptr + batch * stride_c_batch

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        k_mask = offs_k < K

        # load a tile A[M,K]
        a_tile_ptr = a_ptrs + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k)
        a_tile = tl.load(a_tile_ptr, mask=m_mask[:, None] & k_mask[None, :], other=0.0)

        # load b tile B[K,N]
        b_tile_ptr = b_ptrs + (offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n)
        b_tile = tl.load(b_tile_ptr, mask=k_mask[:, None] & n_mask[None, :], other=0.0)

        acc += tl.dot(a_tile, b_tile)

    # store result
    c_tile_ptr = c_ptrs + (offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n)
    tl.store(c_tile_ptr, acc, mask=m_mask[:, None] & n_mask[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    # Move dims if needed.

    # layernorm
    # Compute x_norm = (x - mean) / sqrt(var+eps) * weight + bias
    eps = 1e-5 (default used by nn.LayerNorm)
    # Use torch.nn.functional.layer_norm
    # Ensure it's applied across last dim C
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # shape [B,N,N, hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if not nomask, but mask is given; config may indicate presence
    # comment: If mask is all ones or no mask, we can skip multiplication.
    # shape mask (B,N,N), unsqueeze -1 => [B,N,N,1], broadcast across hidden_dim
    if mask is not None:
        mask_ = mask.to(dtype=left.dtype).unsqueeze(-1)
        left = left * mask_
        right = right * mask_

    # Gating
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # At this point left/right shape [B,N,N,hidden_dim] fp32

    # Permute to [B, hidden_dim, N, N] for kernel
    left_perm = left.permute(0,3,1,2).contiguous()
    # For right, we need transposed version: (j,k) -> (k,j)
    right_perm = right.permute(0,3,2,1).contiguous()  # shape [B, hidden_dim, N, N] where axes 2=k, 3=j

    # Flatten batch*hidden_dim
    B, H, N, _ = left_perm.shape
    batch_total = B * H
    left_flat = left_perm.view(batch_total, N, N)
    right_flat = right_perm.view(batch_total, N, N)
    # Output allocate
    out_flat = torch.empty_like(left_flat)  # same dtype and device

    # Kernel launch with BLOCK sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (batch_total, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    # Determine strides
    # a_ptr shape (batch_total, M=N, K=N), contiguous memory.
    # Actually left_flat stride: (M*K, K, 1)
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)
    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(1)  # because right_flat shape (batch, K, N)
    stride_b_n = right_flat.stride(2)
    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # Cast to int64? triton expects 64-bit ints.

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape output back to [B,N,N,hidden_dim] but need shape [B,N,N,hidden_dim]? Actually after dot we have output shape [batch_total, N, N] = B*H x N x N. Reshape to [B, H, N, N], then permute back to [B,N,N,H].
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()  # shape [B,N,N,hidden_dim]

    # Apply to_out_norm (layernorm on hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)
    out = out * out_gate   # broadcast out_gate shape [B,N,N,hidden_dim]; out_gate from earlier.

    # final linear projection to original dim
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul (outgoing) forward pass rewritten with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input over the feature dimension.
2. Linear projections (left/right) and gating projections (left_gate,
   right_gate, out_gate) are performed with torch.nn.functional.linear.
   Masking (if present) is applied before gating.
3. The expensive contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix multiplication: for each batch and hidden channel
   d we compute left_d @ right_d^T.  We reshape the tensors to
   (B·hidden, N, N) and launch a custom Triton kernel that
   evaluates many independent GEMMs in parallel.
4. The resulting tensor of shape [B,N,N,hidden] is layer‑normed,
   multiplied by the out‑gate, and finally projected back to the original
   feature dimension.

The Triton kernel is a straightforward block‑wise GEMM with support for
arbitrary batch size, handling out‑of‑bounds tiles via masks.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C = A @ B   (A: [batch, M, K], B: [batch, K, N])"""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the matrices of this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile [M, K]
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile [K, N]
        b_ptrs = b_batch + (offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Execute the outgoing TriMul operation.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask_tensor: torch.Tensor of shape [B, N, N] (may be ignored)
            - weights: dict containing all model weights
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the feature dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (left/right) and gating projections
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_w)          # [B, N, N, hidden]
    right = torch.nn.functional.linear(x, right_proj_w)       # [B, N, N, hidden]

    # Apply mask if required
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # Gating
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) Core contraction via Triton (batched GEMM)
    # ------------------------------------------------------------------
    # Permute to bring hidden dimension to leading batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()        # [B, H, N, N]
    # Right needs a transpose on the last two dimensions (k <-> j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()      # [B, H, N, N] (k, j)

    B, H, N, _ = left_perm.shape
    total_batches = B * H

    # Flatten the leading two dimensions so each (b, h) pair is a separate GEMM
    left_flat = left_perm.view(total_batches, N, N)
    right_flat = right_perm.view(total_batches, N, N)
    out_flat = torch.empty_like(left_flat)

    # Strides for the flattened tensors (they are contiguous in memory)
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)   # M stride (row)
    stride_a_k = left_flat.stride(2)   # K stride (col)

    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(1)  # K stride (row of B)
    stride_b_n = right_flat.stride(2)  # N stride (col of B)

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # Tuning parameters – chosen to fit typical B200 resources
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_gemm_kernel[
        grid
    ](
        left_flat,
        right_flat,
        out_flat,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: [B, N, N, hidden]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4) Output LayerNorm, gate, and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate
    to_out_w = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_w)  # final projection to dim

    return out
