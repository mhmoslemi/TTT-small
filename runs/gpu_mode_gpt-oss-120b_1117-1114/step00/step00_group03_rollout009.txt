def trimul_contraction_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # tile row i
    pid_n = tl.program_id(1)  # tile col j
    pid_h = tl.program_id(2)  # combined batch and hidden index

    b = pid_h // H
    h = pid_h % H

    # compute offsets
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offset = tl.arange(0, BLOCK_K)  # will be used inside loop

    # Mask for out-of-bounds i,j
    i_mask = i_offset < N
    j_mask = j_offset < N

    # Base pointers for left and right
    # left[ b, i, k, h ]
    left_base = left_ptr + b * stride_left_b + h * stride_left_h

    # right[ b, j, k, h ]
    # We'll use transposed indexing via j as row; but we need element at (j,k). For multiplication we need right[b,j,k,h] as right_T[k,j,h] if we want L @ R^T. However we can just load right as shaped (j, k). We'll compute later dot product as left[i,k] * right[j,k].
    # To compute sum_k left[i,k] * right[j,k] we can load right as (j, k) but we need to map to dot (inner dimension K). For dot product we need left tile shape (BLOCK_M, BLOCK_K) and right tile shape (BLOCK_N, BLOCK_K). But Triton dot expects shapes (M, K) and (K, N) for multiplication left @ right. So we could load right tile transposed: column j across k dimension => we need right_T[k, j] shape (BLOCK_K, BLOCK_N). So we can compute pointer offsets such that we treat right as transposed. i.e., right_T_base = right_ptr + b * stride_right_b + h * stride_right_h.

    # right original indexing: (b, j, k, h)
    # to get transposed (k, j) we compute offset = stride_k * k_offset + stride_j * j_offset, with strides swapped: stride_k = stride_right_k, stride_j = stride_right_j.

    # So we can set pointer offset for right_T as:
    # base + k_offset*stride_right_k + j_offset*stride_right_j

    # Use tl.load with pointer offsets.

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):

        cur_k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K) # shape (BLOCK_K)
        k_mask = cur_k_offset < N

        # left offsets: (i_offset, cur_k_offset)
        left_ptrs = left_base + i_offset[:, None] * stride_left_i + cur_k_offset[None, :] * stride_left_k

        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float32)

        # right offsets for transposed: (cur_k_offset, j_offset)
        right_ptrs = right_base + cur_k_offset[:, None] * stride_right_k + j_offset[None, :] * stride_right_j
        # load right_T tile (K,N)
        right = tl.load(right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0).to(tl.float32)

        # accumulate dot product:
        acc += tl.dot(left, right)  # left: (BLOCK_M, BLOCK_K), right: (BLOCK_K, BLOCK_N)
    # end loop

    # store result to out
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + i_offset[:, None] * stride_out_i + j_offset[None, :] * stride_out_j

"""
TriMul outgoing update (AlphaFold3) forward pass with a Triton kernel for the
core N^3 contraction.

Algorithm:
1. LayerNorm over the feature dimension D.
2. Linear projections to hidden dimension H (left/right) and three gating
   linear layers (left_gate, right_gate, out_gate) + sigmoid.
3. Apply (optional) pairwise mask and element‑wise gates.
4. Compute the batch‑wise matrix multiplication
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a Triton kernel that performs a tiled GEMM for each
   (batch, hidden) slice (i.e. out = left @ rightᵀ per hidden channel).
5. LayerNorm over H, multiply by out_gate and a final linear projection back
   to dimension D.

Only the O(N³) contraction is in Triton; the surrounding linear layers,
LayerNorms and sigmoids use PyTorch (highly optimized).
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_contraction_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out = left @ rightᵀ for each (batch, hidden) slice.

    left  : [B, N, N, H]  (i, k)
    right : [B, N, N, H]  (j, k)   → transposed on the fly
    out   : [B, N, N, H]  (i, j)
    """
    pid_m = tl.program_id(0)          # tile row i
    pid_n = tl.program_id(1)          # tile col j
    pid_h = tl.program_id(2)          # combined batch*hidden index

    # decode batch and hidden channel
    b = pid_h // H
    h = pid_h % H

    # tile offsets
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i < N
    j_mask = j < N

    # pointers to the start of the (b, *, *, h) slices
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over reduction dimension k in blocks
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k < N

        # left[i, k, h]  (M x K)
        left_ptrs = left_base + i[:, None] * stride_left_i + k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=i_mask[:, None] & k_mask[None, :],
                       other=0.0).to(tl.float32)

        # right_T[k, j, h] = right[j, k, h]  (K x N)
        right_ptrs = right_base + k[:, None] * stride_right_k + j[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=k_mask[:, None] & j_mask[None, :],
                        other=0.0).to(tl.float32)

        # accumulate matrix product
        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # write out tile
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[:, None])


def custom_kernel(data):
    """
    Forward pass of the TriMul outgoing update.
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, D] (float32/float16)
            - mask: torch.Tensor [B, N, N] (bool or float)
            - weights: dict of parameter tensors
            - config: dict containing "dim", "hidden_dim", "nomask" etc.
    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data

    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the feature dimension D
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(D,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask_ = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left  = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) and gating of left/right
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) N^3 contraction via Triton (left @ rightᵀ per hidden channel)
    # ------------------------------------------------------------------
    # Allocate output tensor
    out = torch.empty_like(left)   # shape [B, N, N, H]

    # Triton launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid_h = B * H                     # one program per (batch, hidden) pair
    grid = (grid_m, grid_n, grid_h)

    # Strides (in elements) for the tensors
    # left/right/out are contiguous with shape (B, N, N, H)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b,   stride_out_i,   stride_out_j,   stride_out_h   = out.stride()

    # Launch kernel
    trimul_contraction_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(H,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    output = torch.nn.functional.linear(out, to_out_weight)   # -> [B, N, N, D]

    # Ensure float32 output
    return output.float()
