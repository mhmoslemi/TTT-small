def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # LayerNorm
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor,
                                       normalized_shape=(dim,),
                                       weight=norm_w,
                                       bias=norm_b,
                                       eps=eps)

    # Linear projections
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_w)   # shape [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # Apply mask
    if mask is not None:
        mask_ = mask.to(dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Reshape for batched matmul
    B, N, _, _ = left_h.shape
    # Permute to [B, hidden_dim, N, N]
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()
    batchdim = B * hidden_dim
    left_reshape = left_perm.view(batchdim, N, N)
    right_reshape = right_perm.view(batchdim, N, N)

    # Launch Triton kernel
    import triton
    import triton.language as tl

    # define kernel earlier in file, not nested.

    # Compute strides
    stride_a_batch, stride_a_m, stride_a_k = left_reshape.stride()
    stride_b_batch, stride_b_m, stride_b_k = right_reshape.stride()
    # Should be (N*N, N, 1)
    # For B_T:
    stride_bk = stride_b_k  # 1
    stride_bn = stride_b_m  # N

    # Output tensor initialised
    out_reshape = torch.empty_like(left_reshape, dtype=torch.float32)

    # Strides for output
    stride_c_batch, stride_c_m, stride_c_n = out_reshape.stride()  # same as left_reshape.

    # Choose block sizes (set constants)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid dimensions
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), batchdim)

    # Launch kernel
    matmul_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        batchdim,
        N,
        N,
        N,
        stride_a_m,
        stride_a_k,
        stride_a_batch,
        stride_bk,
        stride_bn,
        stride_b_batch,
        stride_c_m,
        stride_c_n,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape output back
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # shape [B,N,N,hidden_dim]

    # Apply to_out_norm
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(out,
                                              normalized_shape=(hidden_dim,),
                                              weight=to_out_norm_w,
                                              bias=to_out_norm_b,
                                              eps=eps)

    # Multiply by out_gate
    out_gate = out_gate  # already computed earlier shape [B,N,N,hidden_dim] matches.
    out_norm = out_norm * out_gate

    # Final linear projection
    to_out_w = weights["to_out.weight"]
    output = torch.nn.functional.linear(out_norm, to_out_w)  # shape [B,N,N,dim]

    return output

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for batched matrix multiplication of shape [batch, M, K] @ [batch, K, N]^T -> [batch, M, N]
@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_am, stride_ak, stride_abatch,
    stride_bk, stride_bn, stride_bbatch,
    stride_cm, stride_cn, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matmul where B is accessed in transposed (K,N) layout."""
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block column
    pid_z = tl.program_id(2)  # batch index

    # Offsets within the matrices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + pid_z * stride_abatch + offs_m[:, None] * stride_am
    b_base = b_ptr + pid_z * stride_bbatch + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K
        cur_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M x K)
        a = tl.load(
            a_base + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile transposed (K x N) -- original B is (batch, N, K)
        b = tl.load(
            b_base + cur_k[:, None] * stride_bk,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)

    # Write the result
    c_ptrs = c_ptr + pid_z * stride_cbatch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton-accelerated forward of the TriMul module (outgoing version).
    The heavy O(N³) reduction (left @ rightᵀ per hidden channel) is performed
    by a batched‑matrix‑multiply Triton kernel; all other linear layers,
    LayerNorms and sigmoid gates are kept in PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ----------------------------------------------------------------------
    # LayerNorm on the input tensor
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b, eps=eps)

    # ----------------------------------------------------------------------
    # Linear projections (no bias)
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)

    # ----------------------------------------------------------------------
    # Optional mask (broadcast over hidden dimension)
    if mask is not None:
        mask_b = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_b
        right = right * mask_b

    # ----------------------------------------------------------------------
    # Gating
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # Apply gates
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # ----------------------------------------------------------------------
    # Prepare tensors for the batched matmul kernel
    B, N, _, _ = left.shape  # batch, seq_len, seq_len, hidden_dim
    # Move hidden dimension to the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_dim = B * hidden_dim
    left_reshape = left_perm.view(batch_dim, N, N)   # [B*H, N, N]
    right_reshape = right_perm.view(batch_dim, N, N)

    # Output buffer
    out_reshape = torch.empty(batch_dim, N, N, dtype=torch.float32, device=device)

    # Strides (all in number of elements, not bytes)
    stride_a_batch, stride_a_m, stride_a_k = left_reshape.stride()
    stride_b_batch, stride_b_m, stride_b_k = right_reshape.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_reshape.stride()

    # For the transposed view of B (K x N)
    stride_bk = stride_b_k          # row stride in the transposed view (original column stride)
    stride_bn = stride_b_m          # column stride in the transposed view (original row stride)

    # --------------------------------------------------------------
    # Launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_dim,
    )

    batch_matmul_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        batch_dim,
        N,
        N,
        N,
        stride_a_m,
        stride_a_k,
        stride_a_batch,
        stride_bk,
        stride_bn,
        stride_b_batch,
        stride_c_m,
        stride_c_n,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim]
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------------
    # Output LayerNorm, gating, and final projection
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=eps)

    out = out * out_gate  # out_gate is float32

    to_out_w = weights["to_out.weight"]
    output = F.linear(out, to_out_w)  # shape [B, N, N, dim]

    return output

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_am, stride_ak, stride_abatch,
    stride_bk, stride_bn, stride_bbatch,
    stride_cm, stride_cn, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication:
        C[b, i, j] = sum_k A[b, i, k] * B[b, j, k]   (B is accessed transposed)
    The kernel works on half‑precision/fp32 input tensors and accumulates in fp32.
    """
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_z = tl.program_id(2)   # batch index

    # Offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch (no k offset yet)
    a_base = a_ptr + pid_z * stride_abatch + offs_m[:, None] * stride_am
    b_base = b_ptr + pid_z * stride_bbatch + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K
        cur_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M x K)
        a = tl.load(
            a_base + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile transposed (K x N) – original B is (batch, N, K)
        b = tl.load(
            b_base + cur_k[:, None] * stride_bk,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)

    # Write result
    c_ptrs = c_ptr + pid_z * stride_cbatch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.
    The heavy O(N³) reduction (left @ rightᵀ per hidden channel) is performed
    by a batched‑matrix‑multiply Triton kernel; all other linear layers,
    LayerNorms, and sigmoid gates are executed in PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------ #
    # Input LayerNorm
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b, eps=eps)

    # ------------------------------------------------------------------ #
    # Linear projections (no bias)
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x, left_proj_w)    # [B, N, N, H]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------ #
    # Optional mask (broadcast over hidden dim)
    if mask is not None:
        mask_exp = mask.to(dtype).unsqueeze(-1)        # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------ #
    # Gating
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    left = left * left_gate   # still float32
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # Prepare for batched matmul: move hidden dim into the batch axis
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_dim = B * hidden_dim
    left_reshape = left_perm.view(batch_dim, N, N)      # [B*H, N, N]
    right_reshape = right_perm.view(batch_dim, N, N)

    # Output buffer (float32)
    out_reshape = torch.empty(batch_dim, N, N, dtype=torch.float32, device=device)

    # Strides (in element counts, not bytes)
    stride_a_batch, stride_a_m, stride_a_k = left_reshape.stride()
    stride_b_batch, stride_b_m, stride_b_k = right_reshape.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_reshape.stride()

    # For B transposed: rows correspond to original columns (stride_bk) and
    # columns correspond to original rows (stride_bn)
    stride_bk = stride_b_k   # = 1 for contiguous tensors
    stride_bn = stride_b_m   # = N for contiguous tensors

    # ------------------------------------------------------------------ #
    # Launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_dim,
    )

    batch_matmul_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        batch_dim,
        N,
        N,
        N,
        stride_a_m,
        stride_a_k,
        stride_a_batch,
        stride_bk,
        stride_bn,
        stride_b_batch,
        stride_c_m,
        stride_c_n,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------ #
    # Reshape back to [B, N, N, H]
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm, gating, and final linear projection
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=eps)

    out = out * out_gate

    to_out_w = weights["to_out.weight"]
    output = F.linear(out, to_out_w)   # [B, N, N, dim]

    return output
