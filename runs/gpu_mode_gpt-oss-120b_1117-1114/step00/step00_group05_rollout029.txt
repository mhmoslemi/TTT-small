def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on input (dim)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left_proj_weight = weights["left_proj.weight"]   # shape [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]

    left = F.linear(x, left_proj_weight)   # shape [B,N,N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Mask
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_
    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to float16 for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Permute to (B, hidden_dim, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()
    # Transpose right for matmul (swap last two dims)
    right_h_t = right_h.permute(0, 1, 3, 2).contiguous()   # shape (B, hidden, N, N) where axes (k,j)

    B, hidden, N, _ = left_h.shape
    total_batches = B * hidden

    # Flatten to shape (total_batches, N, N)
    left_flat = left_h.view(total_batches, N, N)
    right_flat = right_h_t.view(total_batches, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)  # same dtype as half

    # Kernel launch parameters
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    # Compute grid dims
    num_i_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_j_blocks = (N + BLOCK_N - 1) // BLOCK_N

    grid = (total_batches * num_i_blocks, num_j_blocks)

    # Determine strides (int64)
    stride_l_batch = left_flat.stride(0)
    stride_l_row = left_flat.stride(1)
    stride_l_col = left_flat.stride(2)

    stride_r_batch = right_flat.stride(0)
    stride_r_row = right_flat.stride(1)
    stride_r_col = right_flat.stride(2)

    stride_o_batch = out_flat.stride(0)
    stride_o_row = out_flat.stride(1)
    stride_o_col = out_flat.stride(2)

    # launch kernel
    tri_mul_kernel[grid](
        left_flat, right_flat, out_flat,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        total_batches=total_batches,
        stride_l_batch=stride_l_batch,
        stride_l_row=stride_l_row,
        stride_l_col=stride_l_col,
        stride_r_batch=stride_r_batch,
        stride_r_row=stride_r_row,
        stride_r_col=stride_r_col,
        stride_o_batch=stride_o_batch,
        stride_o_row=stride_o_row,
        stride_o_col=stride_o_col,
    )

    # out_flat is half; convert to float32
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply out_gate
    out = out * out_gate

    # Final linear projection
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)   # shape [B,N,N,dim]

    return out

    stride_r_batch: tl.constexpr, stride_r_row: tl.constexpr, stride_r_col: tl.constexpr,
    stride_o_batch: tl.constexpr, stride_o_row: tl.constexpr, stride_o_col: tl.constexpr
):

    pid_n = tl.program_id(1)

    num_i_blocks = tl.cdiv(N, BLOCK_M)

    batch_id = pid // num_i_blocks
    i_block = pid % num_i_blocks

    if batch_id >= total_batches:
        return

    i0 = i_block * BLOCK_M
    j0 = pid_n * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile (M x K)
        left_ptrs = left_ptr + batch_id * stride_l_batch + offs_i[:, None] * stride_l_row + offs_k[None, :] * stride_l_col
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile (K x N)
        right_ptrs = right_ptr + batch_id * stride_r_batch + offs_k[:, None] * stride_r_row + offs_j[None, :] * stride_r_col
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left, right)

    out_ptrs = out_ptr + batch_id * stride_o_batch + offs_i[:, None] * stride_o_row + offs_j[None, :] * stride_o_col
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Custom Triton implementation of AlphaFold3's outgoing TriMul operator.
    Performs layer norm, linear projections, gating, a batched N^2 matrix
    multiplication (the O(N^3) core), and final projection.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = 1e-5

    # --------- LayerNorm on input ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # --------- Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # shape [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)

    # --------- Apply mask ----------
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # --------- Gates ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # --------- Core O(N^3) multiplication using Triton ----------
    # Permute to (B, hidden, N, N) where left is (i,k) and right is (k,j)
    left = left.permute(0, 3, 1, 2).contiguous()
    right_trans = right.permute(0, 3, 2, 1).contiguous()   # (B, hidden, k, j)

    B, H, N, _ = left.shape
    total_batches = B * H

    left_flat = left.view(total_batches, N, N)
    right_flat = right_trans.view(total_batches, N, N)

    out_flat = torch.empty_like(left_flat)

    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    num_i_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_j_blocks = (N + BLOCK_N - 1) // BLOCK_N

    grid = (total_batches * num_i_blocks, num_j_blocks)

    stride_l_batch = left_flat.stride(0)
    stride_l_row = left_flat.stride(1)
    stride_l_col = left_flat.stride(2)

    stride_r_batch = right_flat.stride(0)
    stride_r_row = right_flat.stride(1)
    stride_r_col = right_flat.stride(2)

    stride_o_batch = out_flat.stride(0)
    stride_o_row = out_flat.stride(1)
    stride_o_col = out_flat.stride(2)

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        total_batches,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        stride_l_batch=stride_l_batch,
        stride_l_row=stride_l_row,
        stride_l_col=stride_l_col,
        stride_r_batch=stride_r_batch,
        stride_r_row=stride_r_row,
        stride_r_col=stride_r_col,
        stride_o_batch=stride_o_batch,
        stride_o_row=stride_o_row,
        stride_o_col=stride_o_col,
    )

    # Reshape back to original layout [B, N, N, hidden]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------- Output normalization and projection ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    out = out * out_gate

    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)

    return out

"""
Optimized Triton implementation of the outgoing TriMul operator (AlphaFold3).

Algorithm:
1. Layer‑norm the 4‑D input tensor.
2. Linear projections → left/right tensors.
3. Optional mask multiplication.
4. Compute three sigmoid gates and apply them to left/right (and later
   to the output).
5. Core O(N³) operation:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched matrix multiplication using a custom Triton
   kernel that processes (batch*hidden) independent N×N matmuls.
6. Layer‑norm + out‑gate + final linear projection back to the original
   channel dimension.
All heavy computation (step 5) is performed on the GPU with Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,                     # sequence length
    TOTAL_BATCHES: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    stride_l_batch: tl.constexpr, stride_l_row: tl.constexpr, stride_l_col: tl.constexpr,
    stride_r_batch: tl.constexpr, stride_r_row: tl.constexpr, stride_r_col: tl.constexpr,
    stride_o_batch: tl.constexpr, stride_o_row: tl.constexpr, stride_o_col: tl.constexpr,
):
    """
    Batched matrix multiplication:
        out[b,i,j] = Σ_k left[b,i,k] * right[b,k,j]
    `left`  : (B', N, N) where B' = batch * hidden, axes = (i,k)
    `right` : (B', N, N) where axes = (k,j)
    The kernel iterates over K in tiles and accumulates in FP32.
    """
    pid = tl.program_id(0)          # combines batch_id * num_i_blocks + i_block
    pid_n = tl.program_id(1)        # j block index

    # number of i‑blocks per batch dimension
    num_i_blocks = tl.cdiv(N, BLOCK_M)

    batch_id = pid // num_i_blocks
    i_block = pid % num_i_blocks

    # Guard against over‑launch (should not happen with exact grid)
    if batch_id >= TOTAL_BATCHES:
        return

    i0 = i_block * BLOCK_M
    j0 = pid_n * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: (M, K)
        left_ptrs = left_ptr + batch_id * stride_l_batch \
                                 + offs_i[:, None] * stride_l_row \
                                 + offs_k[None, :] * stride_l_col
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right block: (K, N)
        right_ptrs = right_ptr + batch_id * stride_r_batch \
                                   + offs_k[:, None] * stride_r_row \
                                   + offs_j[None, :] * stride_r_col
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        acc += tl.dot(left, right)    # (M,N) result

    # write result
    out_ptrs = out_ptr + batch_id * stride_o_batch \
                       + offs_i[:, None] * stride_o_row \
                       + offs_j[None, :] * stride_o_col
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with a Triton kernel for the
    expensive N³ tensor multiplication.
    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim]
            - mask: torch.Tensor, shape [B, N, N] (0/1 values)
            - weights: dict of torch.Tensor containing model weights
            - config: dict with keys "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ---------- Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]      # [hidden, dim]
    right_proj_weight = weights["right_proj.weight"]    # [hidden, dim]

    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight)

    # ---------- Mask ----------
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ---------- Gates ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ---------- Core O(N³) multiplication (Triton) ----------
    # reshape to (B, hidden, N, N) and transpose right for matmul
    left = left.permute(0, 3, 1, 2).contiguous()          # (B, hidden, i, k)
    right_t = right.permute(0, 3, 2, 1).contiguous()      # (B, hidden, k, j)

    B, H, N, _ = left.shape
    total_batches = B * H

    left_flat = left.view(total_batches, N, N)
    right_flat = right_t.view(total_batches, N, N)

    out_flat = torch.empty_like(left_flat)                # FP32 result

    # Block sizes – tuned for typical N (≤1024) on an NVIDIA B200
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    num_i_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_j_blocks = (N + BLOCK_N - 1) // BLOCK_N

    grid = (total_batches * num_i_blocks, num_j_blocks)

    # strides for the flattened tensors
    stride_l_batch = left_flat.stride(0)
    stride_l_row = left_flat.stride(1)
    stride_l_col = left_flat.stride(2)

    stride_r_batch = right_flat.stride(0)
    stride_r_row = right_flat.stride(1)
    stride_r_col = right_flat.stride(2)

    stride_o_batch = out_flat.stride(0)
    stride_o_row = out_flat.stride(1)
    stride_o_col = out_flat.stride(2)

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        total_batches,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        stride_l_batch=stride_l_batch,
        stride_l_row=stride_l_row,
        stride_l_col=stride_l_col,
        stride_r_batch=stride_r_batch,
        stride_r_row=stride_r_row,
        stride_r_col=stride_r_col,
        stride_o_batch=stride_o_batch,
        stride_o_row=stride_o_row,
        stride_o_col=stride_o_col,
    )

    # reshape back to [B, N, N, hidden]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- Output LayerNorm ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear projection ----------
    to_out_weight = weights["to_out.weight"]          # [dim, hidden]
    out = F.linear(out, to_out_weight)

    return out
