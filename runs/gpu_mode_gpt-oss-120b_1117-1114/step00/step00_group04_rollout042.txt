"""
TriMul (outgoing) forward pass with a Triton-accelerated O(N^3) reduction.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension.
2. Linear projections → left, right tensors (shape [B, N, N, H]).
3. Apply optional pairwise mask and element‑wise sigmoid gates.
4. Fuse the heavy reduction:
      out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   This is a batched matrix multiplication per head:
      out_{b,h} = left_{b,h} @ right_{b,h}.T
   Implemented with a custom Triton kernel that processes all
   (B·H) matrices in a single grid.
5. Layer‑norm over the hidden dimension, multiply by the output gate,
   and a final linear projection back to the original dimension.

The kernel computes a block‑wise GEMM using `tl.dot` and handles
boundary conditions via masking. All other operations stay in
PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                             # matrix sizes (all = seq_len)
    stride_am, stride_ak,                # A strides (row, col)
    stride_bk, stride_bn,                # B strides (row=K, col=N) – transposed view
    stride_cm, stride_cn,                # C strides (row, col)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_z = tl.program_id(2)   # matrix index (batch * hidden)

    # each matrix occupies M*K elements (A) / K*N (B) / M*N (C) = N*N
    offset = pid_z * M * K
    a = a_ptr + offset
    b = b_ptr + offset
    c = c_ptr + offset

    # block coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        a_tile = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b_tile = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a_tile, b_tile)

    c_ptrs = c + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation using Triton.
    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (binary mask; may be all‑ones)
            - weights: dict of model weights (torch.Tensors)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if not nomask:
        mask_unsq = mask.unsqueeze(-1)                # [B,N,N,1]
        left = left * mask_unsq
        right = right * mask_unsq

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare tensors for batched GEMM ----------
    # bring hidden dimension to the 2nd axis and reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)   # (BH, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N) # (BH, N, N)
    out_mat = torch.empty_like(left_mat)

    # strides for the kernel (row‑major layout)
    stride_am = left_mat.stride(1)    # row stride (i) = N
    stride_ak = left_mat.stride(2)    # col stride (k) = 1
    # B is accessed as transposed (K,N) => row stride = 1, col stride = N
    stride_bk = right_mat.stride(2)   # stride over K (k) = 1
    stride_bn = right_mat.stride(1)   # stride over N (j) = N
    stride_cm = out_mat.stride(1)     # row stride = N
    stride_cn = out_mat.stride(2)     # col stride = 1

    BLOCK = 64
    grid = (
        triton.cdiv(N, BLOCK),          # M blocks
        triton.cdiv(N, BLOCK),          # N blocks
        B * hidden_dim,                 # number of matrices
    )

    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,                        # M, N, K
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        BLOCK_M=BLOCK,
        BLOCK_N=BLOCK,
        BLOCK_K=BLOCK,
        num_warps=4,
    )

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- Output LayerNorm, gating, and final projection ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
