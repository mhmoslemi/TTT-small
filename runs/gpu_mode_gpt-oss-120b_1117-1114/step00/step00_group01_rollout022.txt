```
def trimul_gemm_kernel(left_ptr, right_ptr, out_ptr,
                       B, N, H,
                       stride_left_b, stride_left_d, stride_left_i, stride_left_k,
                       stride_right_b, stride_right_d, stride_right_j, stride_right_k,
                       stride_out_b, stride_out_d, stride_out_i, stride_out_j,
                       BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_m = tl.program_id(0)  # i tile
    pid_n = tl.program_id(1)  # j tile
    pid_head = tl.program_id(2) # b*H + d

    # compute b and d
    b = pid_head // H
    d = pid_head % H

    # compute start offsets
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # ranges
    i = i_start + tl.arange(0, BLOCK_M)
    j = j_start + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    mask_i = i < N
    mask_j = j < N
```

```
    B, N, N2, H = left.shape
    assert N == N2
    # Permute to (B, H, N, N)
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    out = torch.empty_like(left_perm)

    # get strides
    stride_left_b, stride_left_d, stride_left_i, stride_left_k = left_perm.stride()
    stride_right_b, stride_right_d, stride_right_j, stride_right_k = right_perm.stride()

    BLOCK_N = 64
    BLOCK_K = 32

    trimul_gemm_kernel[grid](

        stride_out_b, stride_out_d, stride_out_i, stride_out_j,
```

```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # Ensure input dtype is float32
    x = input_tensor
    # LayerNorm
    x = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])
    # Apply mask if not nominal
    if mask is not None and mask.numel() > 0:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Core TriMul via Triton
    out = trimul_gemm(left, right)  # shape [B, N, N, hidden_dim]

    # Output normalization and gate
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out = out * out_gate

    # Final projection
    out = F.linear(out, weights["to_out.weight"])
```

```
def trimul_gemm(left, right):
    # left/right shape (B,N,N,hidden_dim)
    B, N, N2, H = left.shape
    assert N == N2
    # permute to (B, H, N, N) for GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    out = torch.empty_like(left_perm)

    # strides
    stride_left_b, stride_left_h, stride_left_i, stride_left_k = left_perm.stride()
    stride_right_b, stride_right_h, stride_right_j, stride_right_k = right_perm.stride()
    stride_out_b, stride_out_h, stride_out_i, stride_out_j = out.stride()

    # define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)

    trimul_gemm_kernel[grid](
        left_perm, right_perm, out,
        B, N, H,
        stride_left_b, stride_left_h, stride_left_i, stride_left_k,
        stride_right_b, stride_right_h, stride_right_j, stride_right_k,
        stride_out_b, stride_out_h, stride_out_i, stride_out_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # permute back to [B, N, N, H]
    out = out.permute(0, 2, 3, 1).contiguous()
    return out
```

```
@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_j, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # i tile
    pid_n = tl.program_id(1)  # j tile
    pid_head = tl.program_id(2)  # combined batch and head

    # decode batch and head index
    b = pid_head // H
    h = pid_head % H

    # offsets
    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N

    # index ranges
    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)
    # start mask
    mask_i = i < N
    mask_j = j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k):
        k_base = k_tile * BLOCK_K
        k = k_base + tl.arange(0, BLOCK_K)

        # pointer offsets for left (b,h,i,k)
        left_offset = b * stride_left_b + h * stride_left_h + i[:, None] * stride_left_i + k[None, :] * stride_left_k
        left_ptrs = left_ptr + left_offset

        # pointer offsets for right (b,h,k,j) i.e., we need (k,j)
        right_offset = b * stride_right_b + h * stride_right_h + k[:, None] * stride_right_k + j[None, :] * stride_right_j
        right_ptrs = right_ptr + right_offset

        # masks
        mask_ik = (i[:, None] < N) & (k[None, :] < N)
        mask_kj = (k[:, None] < N) & (j[None, :] < N)

        # load
        a = tl.load(left_ptrs, mask=mask_ik, other=0.0)
        b_tile = tl.load(right_ptrs, mask=mask_kj, other=0.0)

        # dot product accumulation
        acc += tl.dot(a, b_tile)
    # store result
    out_offset = b * stride_out_b + h * stride_out_h + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    out_ptrs = out_ptr + out_offset
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

import torch.nn.functional as F
import triton
import triton.language as tl

"""TriMul Triton kernel.

Implements the core “outgoing” triangle multiplicative update:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
for each batch b and hidden channel h.
The kernel performs a batched GEMM over the sequence dimension using
tiling (BLOCK_M, BLOCK_N, BLOCK_K) and accumulates in float32.
All other operations (LayerNorm, linear projections, gating) are done
with PyTorch for simplicity.
"""

@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_j, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr

    mask_i = i < N
    mask_j = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k):

        k = k_base + tl.arange(0, BLOCK_K)

        left_offset = (
            b * stride_left_b +
            h * stride_left_h +
            i[:, None] * stride_left_i +
            k[None, :] * stride_left_k

        j[None, :] * stride_out_j

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

"""TriMul Triton kernel.

Implements the core “outgoing” triangle multiplicative update:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
for each batch b and hidden channel h. The kernel performs a batched GEMM
over the sequence dimension using tiling (BLOCK_M, BLOCK_N, BLOCK_K) and
accumulates in float32. All other operations (LayerNorm, linear projections,
gating) are performed with PyTorch.
"""

@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_j, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # 2‑D tiling over i and j, 1‑D tiling over (batch, head)
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile

    # tile start indices
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # index vectors
    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)

    # bounds masks
    mask_i = i < N
    mask_j = j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k):
