import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel: apply mask and gates to the projected tensors.
@triton.jit
def mask_gate_kernel(
    left_out_ptr, right_out_ptr,
    left_ptr, right_ptr,
    mask_ptr,
    left_gate_ptr, right_gate_ptr,
    # dimensions
    B, N, D,
    # left strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    # right strides
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    # mask strides
    stride_mask_b, stride_mask_i, stride_mask_k,
    # left_gate strides
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_d,
    # right_gate strides
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_d,
    BLOCK: tl.constexpr
):
    """Apply mask and gate to left and right tensors (elementwise).
    The input tensors have shape [B, N, N, D] (left/right/lgate/rgate)
    and mask has shape [B, N, N]. All strides are in elements.
    The kernel writes the gated results to left_out and right_out.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * D
    mask = offs < total

    # linear index -> (b,i,k,d)
    off = offs

    b = off // (N * N * D)
    rem = off - b * (N * N * D)
    i = rem // (N * D)
    rem2 = rem - i * (N * D)
    k = rem2 // D
    d = rem2 - k * D

    # compute flat offsets for each tensor
    left_idx = b * stride_left_b + i * stride_left_i + k * stride_left_k + d * stride_left_d
    right_idx = b * stride_right_b + i * stride_right_i + k * stride_right_k + d * stride_right_d
    mask_idx = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k
    lgate_idx = b * stride_lgate_b + i * stride_lgate_i + k * stride_lgate_k + d * stride_lgate_d
    rgate_idx = b * stride_rgate_b + i * stride_rgate_i + k * stride_rgate_k + d * stride_rgate_d

    # load values
    left_val = tl.load(left_ptr + left_idx, mask=mask, other=0.0)
    right_val = tl.load(right_ptr + right_idx, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # default 1 (no masking)
    lgate_val = tl.load(left_gate_ptr + lgate_idx, mask=mask, other=0.0)
    rgate_val = tl.load(right_gate_ptr + rgate_idx, mask=mask, other=0.0)

    left_out = left_val * mask_val * lgate_val
    right_out = right_val * mask_val * rgate_val

    tl.store(left_out_ptr + left_idx, left_out, mask=mask)
    tl.store(right_out_ptr + right_idx, right_out, mask=mask)

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    A tiny Triton kernel fuses the mask and gate application; the O(N³) contraction
    is performed with a batched matrix multiplication (torch.einsum) for speed.
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    # Extract config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---- LayerNorm on input -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ---- Linear projections --------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]    # (hidden_dim, dim)

    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ---- Gates ---------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # same shape as left
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Cast to half precision for the heavy part
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)
    out_gate = out_gate.to(torch.float16)

    # ---- Prepare mask --------------------------------------------------------
    if nomask:
        mask_f = torch.ones_like(mask, dtype=torch.float16, device=device)
    else:
        mask_f = mask.to(torch.float16)

    # ---- Triton kernel: apply mask and gates ---------------------------------
    # allocate output tensors
    left_masked = torch.empty_like(left)
    right_masked = torch.empty_like(right)

    B, N, _, _ = left.shape  # shape [B,N,N,hidden_dim]

    # strides (in elements) for each tensor
    stride_left = left.stride()          # (B_dim, N_dim, N_dim, D_dim)
    stride_right = right.stride()
    stride_mask = mask_f.stride()
    stride_lgate = left_gate.stride()
    stride_rgate = right_gate.stride()

    BLOCK = 1024  # number of elements processed per program instance
    total_elements = B * N * N * hidden_dim
    grid = (triton.cdiv(total_elements, BLOCK),)

    mask_gate_kernel[grid](
        left_masked, right_masked,
        left, right,
        mask_f,
        left_gate, right_gate,
        B, N, hidden_dim,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_mask[0], stride_mask[1], stride_mask[2],
        stride_lgate[0], stride_lgate[1], stride_lgate[2], stride_lgate[3],
        stride_rgate[0], stride_rgate[1], stride_rgate[2], stride_rgate[3],
        BLOCK=BLOCK
    )

    # ---- Contraction (the O(N³) part) ----------------------------------------
    # Use torch.einsum (batched matmul) on float32 for numeric stability.
    left_fp32 = left_masked.to(torch.float32)
    right_fp32 = right_masked.to(torch.float32)

    # Equivalent to batch matmul per hidden dimension
    # reshape to (B*hidden_dim, N, N) and perform batched GEMM
    B_dim, N_dim, _, H_dim = left_fp32.shape
    left_reshape = left_fp32.permute(0, 3, 1, 2).contiguous().view(B_dim * H_dim, N_dim, N_dim)
    right_reshape = right_fp32.permute(0, 3, 1, 2).contiguous().view(B_dim * H_dim, N_dim, N_dim)

    # out shape (B*H, N, N) after matmul
    out_mat = torch.bmm(left_reshape, right_reshape.transpose(1, 2))
    out = out_mat.view(B_dim, H_dim, N_dim, N_dim).permute(0, 2, 3, 1).contiguous()   # back to [B,N,N,hidden_dim]

    # ---- Output LayerNorm ----------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # ---- Apply output gate ---------------------------------------------------
    out = out * out_gate.to(out.dtype)

    # ---- Final linear projection ---------------------------------------------
    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)
    output = F.linear(out, to_out_weight)     # shape [B,N,N,dim]

    # Ensure final dtype is float32 as required
    return output.to(torch.float32)
```

"""
TriMul (outgoing) forward pass with a tiny Triton kernel.

The heavy computation `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]` is
performed by a batched matrix multiplication (torch.bmm) for speed, while a
custom Triton kernel fuses the element‑wise mask and gating operations.
All other linear layers and LayerNorms are handled by PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------------- #
# Triton kernel: apply mask and the left/right gates element‑wise.
# ----------------------------------------------------------------------------- #
@triton.jit
def mask_gate_kernel(
    left_out_ptr, right_out_ptr,          # output pointers (can be same as inputs)
    left_ptr, right_ptr,                  # input tensors
    mask_ptr,                             # mask tensor (shape [B,N,N])
    left_gate_ptr, right_gate_ptr,        # gate tensors (same layout as left/right)

    # dimensions
    B, N, D,                              # batch, sequence length, hidden_dim

    # strides for left tensor (B,N,N,D)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    # strides for right tensor (same layout)
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    # strides for mask tensor (B,N,N)
    stride_mask_b, stride_mask_i, stride_mask_k,
    # strides for left_gate tensor
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_d,
    # strides for right_gate tensor
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_d,

    BLOCK: tl.constexpr
):
    """
    For each element (b,i,k,d) compute:
        left_out = left * mask * left_gate
        right_out = right * mask * right_gate
    The mask is broadcasted over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * D
    mask = offs < total

    # linear index -> (b,i,k,d)
    off = offs
    b = off // (N * N * D)
    rem = off - b * (N * N * D)
    i = rem // (N * D)
    rem2 = rem - i * (N * D)
    k = rem2 // D
    d = rem2 - k * D

    # flat offsets
    left_idx = b * stride_left_b + i * stride_left_i + k * stride_left_k + d * stride_left_d
    right_idx = b * stride_right_b + i * stride_right_i + k * stride_right_k + d * stride_right_d
    mask_idx = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k
    lgate_idx = b * stride_lgate_b + i * stride_lgate_i + k * stride_lgate_k + d * stride_lgate_d
    rgate_idx = b * stride_rgate_b + i * stride_rgate_i + k * stride_rgate_k + d * stride_rgate_d

    # load values (float16)
    left_val = tl.load(left_ptr + left_idx, mask=mask, other=tl.float16(0.0))
    right_val = tl.load(right_ptr + right_idx, mask=mask, other=tl.float16(0.0))
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))
    lgate_val = tl.load(left_gate_ptr + lgate_idx, mask=mask, other=tl.float16(0.0))
    rgate_val = tl.load(right_gate_ptr + rgate_idx, mask=mask, other=tl.float16(0.0))

    left_out = left_val * mask_val * lgate_val
    right_out = right_val * mask_val * rgate_val

    tl.store(left_out_ptr + left_idx, left_out, mask=mask)
    tl.store(right_out_ptr + right_idx, right_out, mask=mask)


def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.
    A small Triton kernel fuses mask and gating; the cubic reduction is done
    with a batched GEMM (torch.bmm) for speed.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # --------------------------------------------------------------------- #
    # Configuration & weight extraction
    # --------------------------------------------------------------------- #
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)

    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)

    to_out_weight = weights["to_out.weight"].to(device)

    # --------------------------------------------------------------------- #
    # 1. LayerNorm on the input
    # --------------------------------------------------------------------- #
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # --------------------------------------------------------------------- #
    # 2. Projections
    # --------------------------------------------------------------------- #
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # --------------------------------------------------------------------- #
    # 3. Gates
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Cast to half precision for the heavy part (mask & gates & matmul)
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)
    out_gate = out_gate.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 4. Mask (or all‑ones if nomask)
    # --------------------------------------------------------------------- #
    if nomask:
        mask_f = torch.ones_like(mask, dtype=torch.float16, device=device)
    else:
        mask_f = mask.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 5. Triton kernel: apply mask and the per‑position gates in place
    # --------------------------------------------------------------------- #
    B, N, _, _ = left.shape  # left.shape == [B, N, N, hidden_dim]

    # Ensure contiguity (required for stride calculations)
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_f = mask_f.contiguous()

    # Strides (in elements)
    stride_left = left.stride()                # (B*N*N*D, N*D, D, 1)
    stride_right = right.stride()
    stride_mask = mask_f.stride()              # (B*N*N, N, 1)
    stride_lgate = left_gate.stride()
    stride_rgate = right_gate.stride()

    BLOCK = 1024
    total_elements = B * N * N * hidden_dim
    grid = (triton.cdiv(total_elements, BLOCK),)

    # In‑place update: outputs point to the same memory as the inputs
    mask_gate_kernel[grid](
        left,               # left_out_ptr
        right,              # right_out_ptr
        left,               # left_ptr
        right,              # right_ptr
        mask_f,             # mask_ptr
        left_gate,          # left_gate_ptr
        right_gate,         # right_gate_ptr
        B, N, hidden_dim,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_mask[0], stride_mask[1], stride_mask[2],
        stride_lgate[0], stride_lgate[1], stride_lgate[2], stride_lgate[3],
        stride_rgate[0], stride_rgate[1], stride_rgate[2], stride_rgate[3],
        BLOCK=BLOCK
    )

    # --------------------------------------------------------------------- #
    # 6. Cubic contraction (batched GEMM)
    # --------------------------------------------------------------------- #
    # Cast back to float32 for the accumulation
    left_f32 = left.to(torch.float32)
    right_f32 = right.to(torch.float32)

    # Reshape to (B*hidden_dim, N, N) and perform batched matmul
    left_mat = left_f32.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_f32.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]

    # Restore original layout: [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 7. Output LayerNorm, gating, and final projection
    # --------------------------------------------------------------------- #
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate.to(out.dtype)

    output = F.linear(out, to_out_weight)   # back to [B, N, N, dim]

    return output.to(torch.float32)
