"""
TriMul (outgoing) forward implementation with a Triton kernel for the O(N³)
pairwise reduction.

Algorithm
---------
1. Layer‑norm the input tensor x of shape [B, N, N, C].
2. Linear projections -> left, right (shape [B, N, N, H]).
3. Optional mask (broadcasted) applied to left/right.
4. Compute gate sigmoids from x and apply them to left/right.
5. Compute the expensive reduction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using a custom Triton kernel that performs a batched matrix‑multiply of
   left with the transpose of right for each (b, d) slice.
6. Layer‑norm over the hidden dimension, multiply by an output gate, and
   project back to the original dimension C.

Only the reduction step is executed on the GPU via Triton; all other
operations are performed with PyTorch for simplicity and correctness.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def batched_gemm_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N,                                    # M = B*H, N = seq_len
    stride_ah, stride_ai, stride_ak,         # A: (batch, i, k)
    stride_bh, stride_bk, stride_bj,         # B: (batch, k, j)   (B_T)
    stride_ch, stride_ci, stride_cj,         # C: (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication C = A @ Bᵀ
    A, B, C have shape (M, N, N) and are stored in row‑major order.
    The third dimension of the launch grid indexes the combined batch*hidden
    dimension.
    """
    batch_idx = tl.program_id(2)          # 0 <= batch_idx < M
    pid_i = tl.program_id(0)              # i‑block
    pid_j = tl.program_id(1)              # j‑block

    # Offsets for the i‑ and j‑ dimensions within the block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A[i, k]
        a_ptrs = a_ptr + batch_idx * stride_ah + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B_T[k, j] = B[j, k]
        b_ptrs = b_ptr + batch_idx * stride_bh + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Matrix‑multiply the tile
        acc += tl.dot(a, b)

    # Store the result tile
    c_ptrs = c_ptr + batch_idx * stride_ch + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted over the hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gates
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 5. Triton kernel for the O(N³) reduction:
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) and flatten the first two dimensions.
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim                         # total number of (batch, hidden) pairs
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)
    out_flat = torch.empty_like(left_flat)

    # Strides for the flattened tensors (they are contiguous)
    stride_ah = left_flat.stride(0)  # = N * N
    stride_ai = left_flat.stride(1)  # = N
    stride_ak = left_flat.stride(2)  # = 1

    stride_bh = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bj = right_flat.stride(2)

    stride_ch = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # i dimension
        triton.cdiv(N, BLOCK_N),   # j dimension
        BH,                         # batch*hidden
    )

    batched_gemm_transpose_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        M=BH,
        N=N,
        stride_ah=stride_ah,
        stride_ai=stride_ai,
        stride_ak=stride_ak,
        stride_bh=stride_bh,
        stride_bk=stride_bk,
        stride_bj=stride_bj,
        stride_ch=stride_ch,
        stride_ci=stride_ci,
        stride_cj=stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
