    left_ptr, right_ptr, out_ptr,

"""
Implementation of the outgoing Triangle Multiplicative Update (TriMul) operator
using Triton for the core N³ reduction. The surrounding LayerNorms, linear
projections and gating are performed in PyTorch for simplicity. The Triton
kernel computes, for each batch b, pair (i, j) and a block of hidden
dimensions d, the sum over the sequence dimension k:

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

The kernel blocks the i, j and hidden dimensions and iterates over k in
tiles, accumulating in float32 while loading the inputs in float16 for reduced
memory bandwidth.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                   # batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_D: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Triton kernel for the core TriMul reduction."""
    pid_i = tl.program_id(0)   # i‑block
    pid_j = tl.program_id(1)   # j‑block
    pid_z = tl.program_id(2)   # combined batch & channel block

    # ----- compute block start coordinates -----
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # number of channel blocks (computed on host, but needed here)
    num_c_blocks = tl.cdiv(H, BLOCK_D)

    b  = pid_z // num_c_blocks          # batch index
    c_block = pid_z % num_c_blocks       # channel‑block index
    c0 = c_block * BLOCK_D

    # ----- offsets -----
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_c = c0 + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_c = offs_c < H

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ----- loop over the reduction dimension k -----
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, c]
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None, None] * stride_left_i
            + offs_k[None, :, None] * stride_left_k
            + offs_c[None, None, :] * stride_left_h
        )
        # right[b, j, k, c]
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + offs_j[:, None, None] * stride_right_i
            + offs_k[None, :, None] * stride_right_k
            + offs_c[None, None, :] * stride_right_h
        )

        # load tiles as float16 and cast to float32 for accumulation
        left_tile = tl.cast(
            tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_c[None, None, :],
                other=0.0,
                dtype=tl.float16,
            ),
            tl.float32,
        )
        right_tile = tl.cast(
            tl.load(
                right_ptrs,
                mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_c[None, None, :],
                other=0.0,
                dtype=tl.float16,
            ),
            tl.float32,
        )

        # outer‑product accumulation over the k‑tile
        for kk in range(0, BLOCK_K):
            # left_tile[:, kk, :] shape (BLOCK_M, BLOCK_D)
            # right_tile[:, kk, :] shape (BLOCK_N, BLOCK_D)
            l_vec = left_tile[:, kk, :]          # (M, D)
            r_vec = right_tile[:, kk, :]         # (N, D)
            # broadcast to (M, N, D) and accumulate
            acc += l_vec[:, None, :] * r_vec[None, :, :]

    # ----- write the result -----
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + offs_c[None, None, :] * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_c[None, None, :],
    )


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data: (input, mask, weights, config)
            - input: Tensor[bs, N, N, dim]
            - mask:  Tensor[bs, N, N]  (ignored if config['nomask'] is True)
            - weights: dict with keys matching the nn.Linear/LayerNorm parameters
            - config: dict with entries 'dim' and 'hidden_dim'
    Returns:
        Tensor[bs, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to float16 for the kernel (save bandwidth)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    B, N, _, _ = left_fp16.shape
    H = hidden_dim

    # output buffer (float16)
    out_fp16 = torch.empty_like(left_fp16)

    # ------------------------------------------------------------------
    # 6. Launch Triton kernel
    # ------------------------------------------------------------------
    # Block sizes – chosen small enough to fit in registers.
    BLOCK_M = 8
    BLOCK_N = 8
    BLOCK_D = 8
    BLOCK_K = 8

    # Compute grid dimensions
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    c_blocks = triton.cdiv(H, BLOCK_D)
    grid_z = B * c_blocks
    grid = (grid_m, grid_n, grid_z)

    # Strides (in element units)
    s_left = left_fp16.stride()
    s_right = right_fp16.stride()
    s_out = out_fp16.stride()

    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        H,
        s_left[0],
        s_left[1],
        s_left[2],
        s_left[3],
        s_right[0],
        s_right[1],
        s_right[2],
        s_right[3],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_D=BLOCK_D,
        BLOCK_K=BLOCK_K,
    )

    # Cast back to float32 for the remaining ops
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
