def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Extract weights as tensors
    # Possibly weights are stored as torch.Tensor with appropriate dtype (float32)
    # Ensure weight shapes: weight for layernorm (dim,) biases (dim,) etc.

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Linear projections for left and right
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if needed
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)   # shape [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Convert to half precision for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Prepare output tensor
    out = torch.empty_like(left, dtype=torch.float32, device=device)  # shape [B,N,N,hidden_dim]

    # Get strides
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h = right.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h = out.stride()

    # Kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Compute grid dimensions
    # number of blocks in M dimension = ceil(N / BLOCK_M)
    # number of blocks in N dimension = ceil(N / BLOCK_N)
    # batch_hidden = batch_size * hidden_dim
    N = left.shape[1]  # seq_len
    B = left.shape[0]
    batch_hidden = B * hidden_dim

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), batch_hidden)

    tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
        stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
        stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K
    )

    # Now we have out (float32) shape [B,N,N,hidden_dim] but we need to apply final transformations:
    # to_out_norm (LayerNorm) and out_gate gating and final linear projection to dim.

    # out_gate has shape [B,N,N,hidden_dim]
    # Apply to_out_norm: compute layernorm across hidden_dim per each element.

    # Compute to_out_norm: We can use torch.nn.functional.layer_norm, using weight.to_out_norm.weight, bias.

    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # Multiply by out_gate (shape B,N,N,hidden_dim)
    # out_gate shape same as out, from earlier gating
    out_norm = out_norm * out_gate

    # Final linear to map hidden_dim -> dim
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_batch = tl.program_id(2)

    # Compute batch index and hidden index
    batch_idx = pid_batch // H
    hidden_idx = pid_batch % H

    # Compute global offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Compute mask for K
        mask_k = offs_k < N

        # Compute pointers offsets
        a_ptrs = a_ptr \
            + batch_idx * stride_a_batch \
            + hidden_idx * stride_a_h \
            + offs_i[:, None] * stride_a_i \
            + offs_k[None, :] * stride_a_k

        b_ptrs = b_ptr \
            + batch_idx * stride_b_batch \
            + hidden_idx * stride_b_h \
            + offs_k[:, None] * stride_b_k \
            + offs_j[None, :] * stride_b_j

        # Load A and B tiles (float16)
        a = tl.load(a_ptrs, mask=(offs_i[:, None] < N) & mask_k[None, :], other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & (offs_j[None, :] < N), other=0.0, dtype=tl.float16)

        # Compute dot product, result float32
        acc += tl.dot(a, b)

    # Store results
    c_ptrs = c_ptr \
        + batch_idx * stride_c_batch \
        + hidden_idx * stride_c_h \
        + offs_i[:, None] * stride_c_i \
        + offs_j[None, :] * stride_c_j

    mask_store = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_store)

```
import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_batch = tl.program_id(2)

    # Determine the batch and hidden channel indices.
    batch_idx = pid_batch // H
    hidden_idx = pid_batch % H

    # Offsets for the output tile.
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k-tiles.
    num_k = tl.cdiv(N, BLOCK_K)

    for k in range(num_k):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Masks for boundary handling.
        mask_k = offs_k < N
        mask_i = offs_i < N
        mask_j = offs_j < N

        # Pointer offsets.
        a_ptrs = a_ptr + batch_idx * stride_a_batch \
                        + hidden_idx * stride_a_h \
                        + offs_i[:, None] * stride_a_i \
                        + offs_k[None, :] * stride_a_k

        b_ptrs = b_ptr + batch_idx * stride_b_batch \
                        + hidden_idx * stride_b_h \
                        + offs_k[:, None] * stride_b_k \
                        + offs_j[None, :] * stride_b_j

        # Load tiles (float32 for accumulation).
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0, dtype=tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0, dtype=tl.float32)

        # Compute tile product.
        acc += tl.dot(a, b)

    # Store the result.
    c_ptrs = c_ptr + batch_idx * stride_c_batch \
                     + hidden_idx * stride_c_h \
                     + offs_i[:, None] * stride_c_i \
                     + offs_j[None, :] * stride_c_j

    mask_store = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_store)

def custom_kernel(data):
    """
    Triton-accelerated implementation of the TriMul module (outgoing variant).
    The function performs layer‑norm, linear projections, gating, then computes
    the O(N³) pairwise multiply‑sum using a custom Triton kernel, followed by a
    second layer‑norm, gating and linear projection to return the result.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on input
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=eps)

    # Linear projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask if present (mask shape [B,N,N])
    if not config.get("nomask", False):
        # mask may be bool or float; cast to same dtype as x
        mask_f = mask.to(dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to half precision for the kernel (still keep computation in FP32)
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    B, N, _, _ = left.shape

    # Allocate output tensor (float32)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Extract strides (in elements)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h = right.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h = out.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            B * hidden_dim)

    tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
        stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
        stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Second LayerNorm (on hidden dimension)
    out_norm = F.layer_norm(out,
                            (hidden_dim,),
                            weight=weights["to_out_norm.weight"],
                            bias=weights["to_out_norm.bias"],
                            eps=eps)

    # Apply output gate
    out_norm = out_norm * out_gate

    # Final linear projection back to dim
    output = F.linear(out_norm, weights["to_out.weight"])

    return output

"""
Fast Triton implementation of the TriMul (outgoing) operator.

The forward pass consists of:
  1. LayerNorm over the last dimension.
  2. Two linear projections (`left_proj`, `right_proj`).
  3. Optional masking.
  4. Three gating linear layers with sigmoid.
  5. Element‑wise multiplication of projected tensors with their gates.
  6. The heavy O(N³) operation:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
     is realized as a batched matrix‑multiply across the hidden dimension h
     using a custom Triton kernel (`tri_mul_kernel`).
  7. A second LayerNorm (`to_out_norm`), gating (`out_gate`),
     and final linear projection back to the original dimension.
All arithmetic is performed in float32; the Triton kernel loads the
operands as float32 and accumulates in float32 to keep numerical accuracy.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}.

    Each program instance works on a (BLOCK_M × BLOCK_N) tile of the output
    for a specific (batch, hidden) pair.  The K‑dimension is tiled with
    BLOCK_K.  All tensors are stored in row‑major (C‑contiguous) order.
    """
    pid_i = tl.program_id(0)          # tile along i dimension
    pid_j = tl.program_id(1)          # tile along j dimension
    pid_batch = tl.program_id(2)      # combined batch × hidden axis

    batch_idx = pid_batch // H
    hidden_idx = pid_batch % H

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)

    for k in range(num_k):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # A: left  -> shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + batch_idx * stride_a_batch
            + hidden_idx * stride_a_h
            + offs_i[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0, dtype=tl.float32)

        # B: right -> shape (BLOCK_K, BLOCK_N) (note the swapped axes)
        b_ptrs = (
            b_ptr
            + batch_idx * stride_b_batch
            + hidden_idx * stride_b_h
            + offs_k[:, None] * stride_b_k
            + offs_j[None, :] * stride_b_j
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0, dtype=tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # Store the result tile
    c_ptrs = (
        c_ptr
        + batch_idx * stride_c_batch
        + hidden_idx * stride_c_h
        + offs_i[:, None] * stride_c_i
        + offs_j[None, :] * stride_c_j
    )
    mask_store = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_store)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: Tuple of (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask_tensor: torch.Tensor [B, N, N] (may be ignored if `nomask` is True)
            - weights: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with at least "dim" and "hidden_dim" entries
    Returns:
        torch.Tensor: output of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1) LayerNorm over the last dimension
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2) Linear projections
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Optional mask
    # --------------------------------------------------------------
    if not config.get("nomask", False):
        m = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4) Gating
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Core O(N³) multiplication using Triton
    # --------------------------------------------------------------
    B, N, _, _ = left.shape

    # Ensure tensors are contiguous for predictable strides
    left = left.contiguous()
    right = right.contiguous()

    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides in element units
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h = right.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h = out.stride()

    # Tile sizes – chosen to fit well on modern GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        hidden_dim,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_a_h,
        stride_b_batch,
        stride_b_j,
        stride_b_k,
        stride_b_h,
        stride_c_batch,
        stride_c_i,
        stride_c_j,
        stride_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6) Second LayerNorm, gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])
    return output
