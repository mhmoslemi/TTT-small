@triton.jit
def tri_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0) 
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute masks for out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers offsets base for batch
    left_base = left_ptr + pid_b * stride_l_batch
    right_base = right_ptr + pid_b * stride_r_batch
    out_base = out_ptr + pid_b * stride_out_batch

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offsets for this tile
        k_start = k * BLOCK_K
        offs_k_curr = k_start + offs_k
        mask_k = offs_k_curr < N

        # Load tiles
        # left: (M, K)
        a_ptrs = left_base + (offs_m[:, None] * stride_l_m) + (offs_k_curr[None, :] * stride_l_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        # right: (K, N)
        b_ptrs = right_base + (offs_k_curr[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # Store results
    out_ptrs = out_base + (offs_m[:, None] * stride_out_m) + (offs_n[None, :] * stride_out_n)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

@triton.jit
def tri_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)  # batch*channel
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base ptr for this batch
    left_batch_ptr = left_ptr + pid_b * stride_l_batch
    right_batch_ptr = right_ptr + pid_b * stride_r_batch
    out_batch_ptr = out_ptr + pid_b * stride_out_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M x K)
        a_ptrs = left_batch_ptr + (offs_m[:, None] * stride_l_m) + (offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        # load right tile (K x N)
        b_ptrs = right_batch_ptr + (offs_k[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute dot product (M,N)
        acc += tl.dot(a, b)

    # Store result
    out_ptrs = out_batch_ptr + (offs_m[:, None] * stride_out_m) + (offs_n[None, :] * stride_out_n)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

1. Layer‐norm over the last dimension of the input tensor.
2. Linear projections (left, right) and gating projections (left_gate,
   right_gate, out_gate) using the supplied weights.
3. Apply optional mask and element‑wise gating.
4. Compute the core operation:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication per channel:
       left  : (B, D, N, N)  → (B*D, N, N)   (i,k)
       right : (B, D, N, N)  → (B*D, N, N)   (k,j)   (right is transposed)
   The multiplication is performed by a custom Triton kernel.
5. Layer‑norm over the hidden dimension, apply the out‑gate, and a final
   linear projection back to the original dimension.

All operations are performed in float32 (the input is cast if needed) and
the kernel runs fully on the GPU, handling arbitrary batch size, sequence
length and hidden dimension within the specified constraints.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_gemm_kernel(
    left_ptr,          # *float32
    right_ptr,         # *float32
    out_ptr,           # *float32
    N,                 # i32
    stride_l_batch,    # i64
    stride_l_m,        # i64
    stride_l_k,        # i64
    stride_r_batch,    # i64
    stride_r_k,        # i64
    stride_r_n,        # i64
    stride_out_batch,  # i64
    stride_out_m,      # i64
    stride_out_n,      # i64
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)          # batch * hidden_dim
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(2)          # block col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this batch
    left_batch = left_ptr + pid_b * stride_l_batch
    right_batch = right_ptr + pid_b * stride_r_batch
    out_batch = out_ptr + pid_b * stride_out_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (M x K)
        a_ptrs = left_batch + (offs_m[:, None] * stride_l_m) + (offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0, dtype=tl.float32)

        # Load B tile (K x N)
        b_ptrs = right_batch + (offs_k[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0, dtype=tl.float32)

        acc += tl.dot(a, b)

    # Store the result
    out_ptrs = out_batch + (offs_m[:, None] * stride_out_m) + (offs_n[None, :] * stride_out_n)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul "outgoing" operation using a Triton kernel
    for the expensive N‑by‑N matrix multiplication.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be all‑ones)
            - weights: dict of model weights (see code for keys)
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # Ensure float32 computation
    x = input_tensor.to(torch.float32)

    # ----- LayerNorm on input -----
    norm_weight = weights["norm.weight"].to(x.dtype)
    norm_bias = weights["norm.bias"].to(x.dtype)
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # ----- Linear projections -----
    left_proj_w = weights["left_proj.weight"].to(x.dtype)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(x.dtype)  # (hidden_dim, dim)

    left = F.linear(x, left_proj_w)      # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)    # (B,N,N,hidden_dim)

    # ----- Optional mask -----
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)  # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # ----- Gating projections -----
    left_gate_w = weights["left_gate.weight"].to(x.dtype)
    right_gate_w = weights["right_gate.weight"].to(x.dtype)
    out_gate_w = weights["out_gate.weight"].to(x.dtype)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ----- Prepare tensors for the batched GEMM -----
    # Bring hidden dimension to the second axis for easy flattening
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # (B, hidden_dim, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()    # (B, hidden_dim, N, N) where K is first in matmul

    B, H, N, _ = left_perm.shape
    batch_hidden = B * H

    # Collapse the batch*hidden dimension
    left_flat = left_perm.view(batch_hidden, N, N)
    right_flat = right_perm.view(batch_hidden, N, N)

    # Allocate output buffer
    out_flat = torch.empty_like(left_flat)

    # ----- Triton GEMM launch -----
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_hidden,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Strides (convert to int64 for Triton)
    stride_l_batch = left_flat.stride(0)
    stride_l_m = left_flat.stride(1)
    stride_l_k = left_flat.stride(2)

    stride_r_batch = right_flat.stride(0)
    stride_r_k = right_flat.stride(1)   # K dimension
    stride_r_n = right_flat.stride(2)   # N dimension

    stride_out_batch = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    tri_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_k,
        stride_r_n,
        stride_out_batch,
        stride_out_m,
        stride_out_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- Reshape back to (B, N, N, hidden_dim) -----
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)   # (B,N,N,hidden_dim)

    # ----- Output LayerNorm, gating, final linear -----
    to_out_norm_w = weights["to_out_norm.weight"].to(x.dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(x.dtype)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(x.dtype)   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                     # (B,N,N,dim)

    return out.to(torch.float32)
