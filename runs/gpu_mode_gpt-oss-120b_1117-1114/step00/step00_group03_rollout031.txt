    pid_j = tl.program_id(2) # block col

import torch
import triton
import triton.language as tl
import math

@triton.jit
def tri_mul_kernel(A_ptr, B_ptr, C_ptr,
                   N, K,
                   stride_am, stride_ai, stride_ak,
                   stride_bm, stride_bj, stride_bk,
                   stride_cm, stride_ci, stride_cj,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """Compute C = sum_k A[..., i, k] * B[..., j, k] for many batches.

    A shape: [batch, N, K]   (row-major)
    B shape: [batch, N, K]   (row-major)
    Output C shape: [batch, N, N].

    The kernel treats the first dimension as a batch (here batch = B*hidden_dim)
    and performs a regular dense matrix multiplication with the second
    operand transposed on‑the‑fly (B^T).  Accumulation is done in float32.
    """
    batch_id = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # block coordinates
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = A_ptr + batch_id * stride_am + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load B transposed: B[k, j] (original B[j, k])
        b_ptrs = B_ptr + batch_id * stride_bm + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # multiply‑accumulate
        acc += tl.dot(a, b)

    # write out
    c_ptrs = C_ptr + batch_id * stride_cm + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The most expensive primitive is the 4‑D Einstein summation
    out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
which is equivalent to a batched matrix multiplication
    out[b, d] = left[b, d] @ right[b, d].T
for each hidden dimension `d`.  The kernel collapses the batch and hidden
dimension into a single batch axis and computes the GEMM with Triton,
accumulating in float32.  All other operations (LayerNorm, Linear, gating)
are performed with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """C = Σ_k A[i,k] * B[j,k]  (B accessed transposed on‑the‑fly).

    A, B, C are 3‑D tensors with layout [batch, N, K] (row‑major).
    The kernel treats the first dimension as a batch (here batch = B * hidden_dim)
    and performs a dense matrix multiplication with the second operand
    transposed (B⁽ᵀ⁾) while accumulating in float32.
    """
    batch_id = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + batch_id * stride_am \
                 + offs_i[:, None] * stride_ai \
                 + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # B block transposed: we need B[k, j] (original B[j, k])
        b_ptrs = B_ptr + batch_id * stride_bm \
                 + offs_k[:, None] * stride_bk \
                 + offs_j[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    c_ptrs = C_ptr + batch_id * stride_cm \
             + offs_i[:, None] * stride_ci \
             + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[:, None])

    """
    Outgoing TriMul forward pass.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor [B, N, N, dim]
        - mask        : torch.Tensor [B, N, N] (may be all‑ones)
        - weights     : dict of torch.Tensors with the model parameters
        - config      : dict containing at least "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """

    device = input_tensor.device
    eps = config.get("layer_norm_eps", 1e-5)

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # helper to move weights onto the right device / dtype
    # ------------------------------------------------------------------
    def _to_fp32(t):
        return t.to(device=device, dtype=torch.float32)

    norm_weight = _to_fp32(weights["norm.weight"])
    norm_bias   = _to_fp32(weights["norm.bias"])

    left_proj_weight  = _to_fp32(weights["left_proj.weight"])
    right_proj_weight = _to_fp32(weights["right_proj.weight"])

    left_gate_weight  = _to_fp32(weights["left_gate.weight"])
    right_gate_weight = _to_fp32(weights["right_gate.weight"])
    out_gate_weight   = _to_fp32(weights["out_gate.weight"])

    to_out_norm_weight = _to_fp32(weights["to_out_norm.weight"])
    to_out_norm_bias   = _to_fp32(weights["to_out_norm.bias"])

    to_out_weight = _to_fp32(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # LayerNorm on the input pair representation
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    ).to(torch.float32)   # reference casts to float32 after norm

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_weight)   # (B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Apply mask if supplied (mask shape [B, N, N])
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)   # (B, N, N, 1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core multiplicative update using Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape    # B = batch size
    # Move hidden dimension next to batch and make the tensor contiguous
    left_flat  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch * hidden into a single dimension for the kernel
    left_flat  = left_flat.view(-1, N, N)   # (B*hidden, N, N)
    right_flat = right_flat.view(-1, N, N)
    out_flat   = torch.empty_like(left_flat)  # (B*hidden, N, N)

    # Strides (int64)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # original ‘j’ dimension
    stride_bk = right_flat.stride(2)

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # Tile sizes – chosen to fit comfortably in shared memory on a B200 GPU
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        left_flat.shape[0],                # batch * hidden_dim
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Output normalization, gating and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # (B, N, N, dim)

    return out
