def trimul_kernel(L_ptr, R_ptr, O_ptr,
                  B, N, D,
                  stride_LB, stride_LI, stride_LK, stride_LD,
                  stride_RB, stride_RI, stride_RK, stride_RD,
                  stride_OB, stride_OI, stride_OK, stride_OD,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
                  BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr):

    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # offsets

    j_offsets = j_base + tl.arange(0, BLOCK_N)
    d_offsets = tl.arange(0, BLOCK_D)

    # masks for i/j within N
    mask_i = i_offsets < N
    mask_j = j_offsets < N

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_kernel(left_ptr, right_ptr, out_ptr,
                  B, N, D,
                  stride_LB, stride_LI, stride_LK, stride_LD,
                  stride_RB, stride_RI, stride_RK, stride_RD,
                  stride_OB, stride_OI, stride_OK, stride_OD,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
                  BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)  # i block
    pid_j = tl.program_id(1)  # j block

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    i_off = i_base + tl.arange(0, BLOCK_M)
    j_off = j_base + tl.arange(0, BLOCK_N)
    d_off = tl.arange(0, BLOCK_D)

    mask_i = i_off < N
    mask_j = j_off < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    for k in range(0, N, BLOCK_K):
        k_base = k
        k_off = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # left tile (M, K, D)
        ptrs_L = left_ptr + pid_b * stride_LB \
                 + i_off[:, None, None] * stride_LI \
                 + k_off[None, :, None] * stride_LK \
                 + d_off[None, None, :] * stride_LD
        left = tl.load(ptrs_L, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0)

        # right transposed tile (K, N, D) i.e., (K, N, D) where K corresponds to k, N corresponds to j
        ptrs_Rt = right_ptr + pid_b * stride_RB \
                  + j_off[None, :, None] * stride_RI \
                  + k_off[:, None, None] * stride_RK \
                  + d_off[None, None, :] * stride_RD
        right = tl.load(ptrs_Rt, mask=mask_k[:, None, None] & mask_j[None, :, None], other=0.0)

        # compute dot for each d
        for d in range(0, BLOCK_D):
            # (M, K) dot (K, N) -> (M, N)
            acc[:, :, d] += tl.dot(left[:, :, d], right[:, :, d])

    # Store result
    ptrs_O = out_ptr + pid_b * stride_OB \
             + i_off[:, None, None] * stride_OI \
             + j_off[None, :, None] * stride_OK \
             + d_off[None, None, :] * stride_OD
    mask_o = mask_i[:, None, None] & mask_j[None, :, None] & (d_off[None, None, :] < D)
    tl.store(ptrs_O, acc, mask=mask_o)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, _ = input_tensor.shape  # dim
    # LayerNorm on input
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"].to(device), bias=weights["norm.bias"].to(device), eps=1e-5)
    # Projections
    left = F.linear(x, weights["left_proj.weight"].to(device))
    right = F.linear(x, weights["right_proj.weight"].to(device))

    # Apply mask if needed
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(device)  # shape [B,N,N,1]
        left = left * mask_
        right = right * mask_
    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(device)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(device)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(device)))

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()

    # Triangular multiplication via Triton
    out = trmul_triton(left, right)   # shape (B,N,N,hidden_dim)

    # Output layer norm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"].to(device), bias=weights["to_out_norm.bias"].to(device), eps=1e-5)

    # out gate
    out = out * out_gate

    # Final linear projection
    out = F.linear(out, weights["to_out.weight"].to(device))

    return out

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The heavy O(N³) reduction
    out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]
is implemented as a batched matrix multiplication per hidden channel:
    left(b, :, :, d) @ right(b, :, :, d).T
using a custom Triton kernel that tiles the i, j, k and channel dimensions.
All other operations (LayerNorm, Linear projections, gating, final Linear)
are performed with PyTorch for simplicity.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: computes the triangular multiplicative update (outgoing)
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_k, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(2)          # batch axis
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    i_off = i0 + tl.arange(0, BLOCK_M)
    j_off = j0 + tl.arange(0, BLOCK_N)
    d_off = tl.arange(0, BLOCK_D)

    mask_i = i_off < N
    mask_j = j_off < N

    # accumulator for a tile of shape (BLOCK_M, BLOCK_N, BLOCK_D)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # loop over k dimension in blocks
    for k in range(0, N, BLOCK_K):
        k0 = k + tl.arange(0, BLOCK_K)
        mask_k = k0 < N

        # left tile: shape (M, K, D)  -> left[i,k,d]
        ptrs_L = left_ptr + pid_b * stride_l_b \
                 + i_off[:, None, None] * stride_l_i \
                 + k0[None, :, None] * stride_l_k \
                 + d_off[None, None, :] * stride_l_d
        left = tl.load(ptrs_L,
                       mask=mask_i[:, None, None] & mask_k[None, :, None],
                       other=0.0)

        # right transposed tile: shape (K, N, D)  -> right[j,k,d] accessed as (k,j,d)
        ptrs_Rt = right_ptr + pid_b * stride_r_b \
                  + j_off[None, :, None] * stride_r_i \
                  + k0[:, None, None] * stride_r_k \
                  + d_off[None, None, :] * stride_r_d
        right = tl.load(ptrs_Rt,
                        mask=mask_k[:, None, None] & mask_j[None, :, None],
                        other=0.0)

        # dot‑product for each channel slice
        for d in range(0, BLOCK_D):
            # (M,K) @ (K,N) -> (M,N)
            acc[:, :, d] += tl.dot(left[:, :, d], right[:, :, d])

    # write out tile
    ptrs_O = out_ptr + pid_b * stride_o_b \
             + i_off[:, None, None] * stride_o_i \
             + j_off[None, :, None] * stride_o_k \
             + d_off[None, None, :] * stride_o_d
    mask_o = mask_i[:, None, None] & mask_j[None, :, None] & (d_off[None, None, :] < D)
    tl.store(ptrs_O, acc, mask=mask_o)


def _trmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper around `trimul_kernel`. Handles the hidden dimension in
    BLOCK_D‑sized chunks (default 32).
    """
    B, N, _, D = left.shape
    out = torch.empty_like(left)

    # tile sizes – 32 is a safe choice for most GPUs
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    MAX_BLOCK_D = 32

    # strides in element units (not bytes)
    sL = left.stride()
    sR = right.stride()
    sO = out.stride()

    d_start = 0
    while d_start < D:
        cur_D = min(MAX_BLOCK_D, D - d_start)

        # slice the channel dimension
        left_slice = left[..., d_start : d_start + cur_D].contiguous()
        right_slice = right[..., d_start : d_start + cur_D].contiguous()
        out_slice = out[..., d_start : d_start + cur_D]

        grid = (
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            B,
        )
        trimul_kernel[grid](
            left_slice,
            right_slice,
            out_slice,
            B,
            N,
            cur_D,
            sL[0], sL[1], sL[2], sL[3],
            sR[0], sR[1], sR[2], sR[3],
            sO[0], sO[1], sO[2], sO[3],
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=cur_D,
        )
        d_start += cur_D

    return out


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C]
        - mask         : torch.Tensor [B, N, N] (broadcasted over C)
        - weights      : dict of parameter tensors
        - config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(device))
    right = F.linear(x, weights["right_proj.weight"].to(device))

    # ------------------------------------------------------------------
    # 3) Apply optional mask (mask is broadcasted over the channel dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(device).type_as(left)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid on linear transforms of the normalized input)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(device)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(device)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(device)))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triangular multiplicative update (outgoing) via Triton
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = _trmul_triton(left, right)          # shape [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, final gating and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device),
        bias=weights["to_out_norm.bias"].to(device),
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(device))

    return out
