    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    # compute linear indices
    total_elements = B * N * N
    idx = pid
    if idx >= total_elements:
        return
    b = idx // (N * N)
    rem = idx % (N * N)
    i = rem // N
    j = rem % N
    # base offsets
    proj_base = proj_ptr + b * stride_proj_b + i * stride_proj_i + j * stride_proj_j
    gate_base = gate_ptr + b * stride_gate_b + i * stride_gate_i + j * stride_gate_j
    out_base = out_ptr + b * stride_out_b + i * stride_out_i + j * stride_out_j
    mask_val = tl.load(mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j)
    # iterate over hidden dimension

        mask_h = cur_h < H

"""
TriMul forward pass (outgoing version) with a Triton kernel.

Algorithm
---------
1. Layer‑normalize the input tensor over the last dimension.
2. Compute four linear projections of the normalized input:
   * left_proj  : (dim -> hidden_dim)
   * right_proj : (dim -> hidden_dim)
   * left_gate  : (dim -> hidden_dim) + sigmoid
   * right_gate : (dim -> hidden_dim) + sigmoid
3. Fuse mask application, gating and element‑wise multiplication in a
   Triton kernel:
       left  = left_proj  * left_gate  * mask
       right = right_proj * right_gate * mask
   (mask is a scalar 0/1 per pair (b,i,j) and is broadcast over the
    hidden dimension.)
4. Contract over the third axis (k) with a batched einsum:
       out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
5. Compute the output gate from the normalized input (sigmoid(linear)).
6. Apply a layer‑norm on the hidden dimension, multiply by the output
   gate, and finally project back to the original dimension.
7. Return the result in float32.

Only the element‑wise mask/gate fusion (step 3) is implemented in Triton;
all other heavy linear / reduction operations use PyTorch, which already
leverages highly optimized cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def apply_mask_gate_kernel(
    proj_ptr,          # *float16   left/right projection (B,N,N,H)
    gate_ptr,          # *float16   sigmoid gate (B,N,N,H)
    mask_ptr,          # *float16   mask (B,N,N) – 0.0 or 1.0
    out_ptr,           # *float16   destination tensor (B,N,N,H)
    B, N, H,           # runtime scalars
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_H: tl.constexpr
):
    """
    For every (b,i,j) element compute:
        out[b,i,j,:] = proj[b,i,j,:] * gate[b,i,j,:] * mask[b,i,j]
    The hidden dimension H is processed in blocks of size BLOCK_H.
    """
    pid = tl.program_id(0)

    total_pairs = B * N * N
    if pid >= total_pairs:
        return

    # decode (b,i,j) from linear pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # base pointers for this (b,i,j)
    proj_base = proj_ptr + b * stride_proj_b + i * stride_proj_i + j * stride_proj_j
    gate_base = gate_ptr + b * stride_gate_b + i * stride_gate_i + j * stride_gate_j
    out_base  = out_ptr  + b * stride_out_b  + i * stride_out_i  + j * stride_out_j
    mask_val = tl.load(mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j)

    # iterate over the hidden dimension
    for off_h in range(0, H, BLOCK_H):
        h = off_h + tl.arange(0, BLOCK_H)
        mask_h = h < H

        proj = tl.load(proj_base + h * stride_proj_h, mask=mask_h, other=0.0)
        gate = tl.load(gate_base + h * stride_gate_h, mask=mask_h, other=0.0)

        out = proj * gate * mask_val
        tl.store(out_base + h * stride_out_h, out, mask=mask_h)


def custom_kernel(data):
    """
    Triton‑accelerated TriMul forward.

    Args:
        data: Tuple (input_tensor, mask_tensor, weight_dict, config_dict)
            - input_tensor: torch.Tensor [bs, N, N, dim] (float32)
            - mask_tensor : torch.Tensor [bs, N, N] (bool/int/float) or None
            - weight_dict : mapping of weight names to torch.Tensors
            - config_dict : must contain "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [bs, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    bs, N, _, _ = input_tensor.shape

    # ---------- 1. Input LayerNorm ----------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).to(torch.float16)   # fp16 for the rest of the pipeline

    # ---------- 2. Linear projections & gates (fp16) ----------
    # convert all linear weights to fp16 once
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # projections
    left_proj = F.linear(x_norm, left_proj_w)          # (B,N,N,hidden_dim)
    right_proj = F.linear(x_norm, right_proj_w)

    # gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))

    # ---------- 3. Mask handling ----------
    if mask is None:
        mask_fp16 = torch.ones(
            (bs, N, N), dtype=torch.float16, device=device
        )
    else:
        mask_fp16 = mask.to(torch.float16)

    # ---------- 4. Fuse mask + gating (Triton) ----------
    left = torch.empty_like(left_proj)   # (B,N,N,hidden_dim)
    right = torch.empty_like(right_proj)

    # strides (in elements) for the tensors
    proj_strides = left_proj.stride()
    gate_strides = left_gate.stride()
    out_strides = left.stride()
    mask_strides = mask_fp16.stride()

    BLOCK_H = 32                         # compile‑time tile size
    total_pairs = bs * N * N
    grid = (total_pairs,)

    # left side
    apply_mask_gate_kernel[grid](
        proj_ptr=left_proj.data_ptr(),
        gate_ptr=left_gate.data_ptr(),
        mask_ptr=mask_fp16.data_ptr(),
        out_ptr=left.data_ptr(),
        B=bs, N=N, H=hidden_dim,
        stride_proj_b=proj_strides[0],
        stride_proj_i=proj_strides[1],
        stride_proj_j=proj_strides[2],
        stride_proj_h=proj_strides[3],
        stride_gate_b=gate_strides[0],
        stride_gate_i=gate_strides[1],
        stride_gate_j=gate_strides[2],
        stride_gate_h=gate_strides[3],
        stride_mask_b=mask_strides[0],
        stride_mask_i=mask_strides[1],
        stride_mask_j=mask_strides[2],
        stride_out_b=out_strides[0],
        stride_out_i=out_strides[1],
        stride_out_j=out_strides[2],
        stride_out_h=out_strides[3],
        BLOCK_H=BLOCK_H,
    )

    # right side (same mask)
    apply_mask_gate_kernel[grid](
        proj_ptr=right_proj.data_ptr(),
        gate_ptr=right_gate.data_ptr(),
        mask_ptr=mask_fp16.data_ptr(),
        out_ptr=right.data_ptr(),
        B=bs, N=N, H=hidden_dim,
        stride_proj_b=right_proj.stride(0),
        stride_proj_i=right_proj.stride(1),
        stride_proj_j=right_proj.stride(2),
        stride_proj_h=right_proj.stride(3),
        stride_gate_b=right_gate.stride(0),
        stride_gate_i=right_gate.stride(1),
        stride_gate_j=right_gate.stride(2),
        stride_gate_h=right_gate.stride(3),
        stride_mask_b=mask_strides[0],
        stride_mask_i=mask_strides[1],
        stride_mask_j=mask_strides[2],
        stride_out_b=right.stride(0),
        stride_out_i=right.stride(1),
        stride_out_j=right.stride(2),
        stride_out_h=right.stride(3),
        BLOCK_H=BLOCK_H,
    )

    # ---------- 5. O(N³) contraction ----------
    # einsum sums over the third axis (k) while preserving hidden dimension
    out = torch.einsum("...i k d,...j k d->...i j d", left, right)   # (B,N,N,hidden_dim)

    # ---------- 6. Output gate ----------
    out_gate = out_gate.to(torch.float32)   # fp32 for stable multiplication later

    # ---------- 7. Hidden‑dim LayerNorm ----------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float32)
    out = out.to(torch.float32)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # ---------- 8. Apply output gate ----------
    out = out * out_gate

    # ---------- 9. Final linear projection ----------
    to_out_w = weights["to_out.weight"].to(torch.float32)   # (dim, hidden_dim)
    output = F.linear(out, to_out_w)                       # (B,N,N,dim)

    return output
