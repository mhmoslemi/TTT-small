@triton.jit
def tri_mul_out_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)  # i tile
    pid_n = tl.program_id(1)  # j tile
    pid_bh = tl.program_id(2)  # batch * hidden id
    i_start = pid * BLOCK_M
    j_start = pid_n * BLOCK_N
    bh = pid_bh
    b = bh // H
    h = bh % H

    # offset base for each pointer
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Compute output offsets
    offs_m = i_start + tl.arange(0, BLOCK_M)
    offs_n = j_start + tl.arange(0, BLOCK_N)

    # mask for boundaries
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A (left) tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load B (right) tile as transposed: shape (BLOCK_K, BLOCK_N)
        # B's row is k, column is j (j corresponds to j index)
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # accumulate dot product: matrix multiplication of a (M,K) @ b (K,N)
        acc += tl.dot(a, b)

    # store result
    out_ptrs = out_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

Define kernel as:

```
@triton.jit
def tri_mul_out_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    bh = pid_bh
    b = bh // H
    h = bh % H

    # base pointers offset for batch and hidden channel
    left_ptrs = left_ptr + b * stride_left_b + h * stride_left_h
    right_ptrs = right_ptr + b * stride_right_b + h * stride_right_h
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h

    # offsets
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_tiles):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A (left)
        a_ptrs = left_ptrs + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # Convert to float32 if loading as fp16
        a = a.to(tl.float32)

        # Load B (right) transposed (k x j)
        b_ptrs = right_ptrs + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        b = b.to(tl.float32)

        # Accumulate dot
        acc += tl.dot(a, b)

    # Store output
    out_ptrs = out_ptrs + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) implementation with Triton acceleration.

The heavy O(N³) reduction
    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
is rewritten as a batch of matrix multiplications:
    out_h = left_h @ right_hᵀ
for each batch `b` and hidden channel `h`.  The batched matmul is
implemented in a fused Triton kernel.  All other operations (layer‑norm,
linear projections, gating, and final linear) are performed in PyTorch,
using FP16 where possible to reduce memory traffic.  The kernel loads the
inputs as FP16, computes the dot product with a FP32 accumulator, and
writes the result as FP32.  The final output matches the reference
PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_out_kernel(
    left_ptr,          # const half *
    right_ptr,         # const half *
    out_ptr,           #       float *
    B, N, H,           # int64 scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication for the TriMul reduction.

    Computes, for a given (batch, hidden) pair:
        out[i, j] = Σₖ left[i, k] * right[j, k]
    where left and right are viewed as (N, N) matrices.
    """
    pid_i = tl.program_id(0)                    # tile row index
    pid_j = tl.program_id(1)                    # tile col index
    pid_bh = tl.program_id(2)                   # combined batch*hidden id

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Decode batch and hidden index
    b = pid_bh // H
    h = pid_bh - b * H

    # Base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # Offsets within the tile
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile   (M x K)  as half
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :],
                    other=tl.float16(0.0), dtype=tl.float16)

        # Load right tile  (K x N)   (right is transposed on the fly)
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :],
                    other=tl.float16(0.0), dtype=tl.float16)

        # Accumulate with FP32 precision
        acc += tl.dot(a, b)

    # Store the result
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor - torch.Tensor of shape [B, N, N, D] (float32)
            mask         - torch.Tensor of shape [B, N, N] (float32 or bool)
            weights      - dict of model parameters (all float32 tensors)
            config       - dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, D] float32

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for the heavy path
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias) -> [B, N, N, H] FP16
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4. Apply mask if provided (broadcast over the hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_h = mask.unsqueeze(-1).to(x_h.dtype)          # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5. Compute gates (FP16) and apply
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. TriMul reduction via Triton kernel (output in FP32)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left.shape == [B, N, N, H]
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # Tensor strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),        # i‑tiles
        triton.cdiv(N, BLOCK_N),        # j‑tiles
        B * hidden_dim,                 # combined batch*hidden dimension
    )
    tri_mul_out_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Output LayerNorm (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 8. Scale by out_gate (cast gate to float32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)

    # ------------------------------------------------------------------
    # 9. Final linear projection back to original dimension (float32)
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
