import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   N, hidden_dim, batch_size,
                   stride_batch, stride_i, stride_k,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """
    Compute out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    for a batch of matrices, with d flattened into the program id.
    """
    pid = tl.program_id(0)   # flattened batch*hidden
    pid_i = tl.program_id(1) # block row (i)
    pid_j = tl.program_id(2) # block col (j)

    # recover batch index and hidden channel
    b = pid // hidden_dim
    d = pid % hidden_dim

    # base pointer for this (b,d)
    offset_base = b * stride_batch + d

    # offsets for i and j within the tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile A(i,k)
        a_ptrs = left_ptr + offset_base + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile B(k,j)  (note the transpose, we load as (k,j))
        b_ptrs = right_ptr + offset_base + offs_k[:, None] * stride_i + offs_j[None, :] * stride_k
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate
        acc += tl.dot(a, b)

    # store result
    out_ptrs = out_ptr + offset_base + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Forward pass of the TriMul module (AlphaFold3 outgoing version) using Triton
    for the O(N^3 * hidden_dim) multiplication.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # LayerNorm on input
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask if provided (mask may be None or omitted)
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Ensure tensors are contiguous for Triton
    left = left.contiguous()
    right = right.contiguous()

    # Output tensor for the kernel
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # Triton grid and launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    stride_batch = left.stride(0)
    stride_i = left.stride(1)
    stride_k = left.stride(2)

    batchdim = B * hidden_dim
    grid = (batchdim, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    tri_mul_kernel[grid](
        left,
        right,
        out,
        N,
        hidden_dim,
        B,
        stride_batch,
        stride_i,
        stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Apply second LayerNorm (hidden_dim)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Multiply by out gate and final linear projection
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul forward (outgoing) implementation.

The expensive operation is:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We compute it with a custom Triton kernel that performs a batched
matrix multiplication for each hidden channel d (flattened together with the
batch dimension).  All other steps (LayerNorm, linear projections,
gating) are done in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,          # *float32
    right_ptr,         # *float32
    out_ptr,           # *float32
    N,                 # i32  sequence length
    hidden_dim,        # i32
    batch_size,        # i32 (unused, kept for clarity)
    stride_batch,      # i32  stride for batch dimension (B)
    stride_i,          # i32  stride for the first N dimension
    stride_k,          # i32  stride for the second N dimension (k)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    where d is flattened with the batch dimension (program_id 0).
    """
    pid = tl.program_id(0)   # combined batch*hidden index
    pid_i = tl.program_id(1) # block row (i)
    pid_j = tl.program_id(2) # block col (j)

    # recover batch index and hidden channel
    b = pid // hidden_dim
    d = pid % hidden_dim

    # base pointer offset for this (b, d)
    base = b * stride_batch + d

    # offsets for i and j within this tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator for the output tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension in tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile A(i,k)
        a_ptrs = left_ptr + base + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right tile B(k,j) – note the transpose (k rows, j columns)
        b_ptrs = right_ptr + base + offs_k[:, None] * stride_k + offs_j[None, :] * stride_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # accumulate the matrix product of the two tiles
        acc += tl.dot(a, b)

    # write the result tile
    out_ptrs = out_ptr + base + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module using Triton for the
    O(N³·hidden_dim) multiplication.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm on the input ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Apply mask (if present) ----------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous layout for Triton
    left = left.contiguous()
    right = right.contiguous()

    # ---------- Triton kernel for the triple product ----------
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # Tile sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    stride_batch = left.stride(0)   # B dimension stride
    stride_i = left.stride(1)       # first N dimension stride
    stride_k = left.stride(2)       # second N dimension stride (k)

    batch_dim = B * hidden_dim
    grid = (
        batch_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        N,
        hidden_dim,
        B,
        stride_batch,
        stride_i,
        stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Second LayerNorm (over hidden dimension) ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ---------- Apply out gate and final linear ----------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
