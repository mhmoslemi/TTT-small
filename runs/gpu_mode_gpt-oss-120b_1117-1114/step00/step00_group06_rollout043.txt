    stride_bo, stride_io, stride_jo, stride_co,

"""
TriMul (outgoing) kernel optimized with Triton.

The core of TriMul is the cubic‑complexity tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are obtained from a LayerNorm, linear projections and
gating.  The surrounding LayerNorm / Linear ops are cheap, so they are kept in
PyTorch.  The contraction is implemented as a tiled GEMM‑like Triton kernel:
each program processes a (BLOCK_M × BLOCK_N) tile of the output and loops over
the reduction dimension `k` in blocks of `BLOCK_K`.  For every `k` we load the
vectors `left[:,k,:]` and `right[:,k,:]` (size hidden_dim) for the current
tile, compute the outer product across the `i` and `j` dimensions and accumulate
the result.  No synchronization between programs is required; the GPU cache
re‑uses the loaded tiles across different `j` blocks, as in a standard matrix
multiply.

The kernel works for any batch size (≤2), sequence length (≤1024) and hidden
dimension (≤384).  The final output is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    # strides for left  [B, i, k, c]
    stride_bl, stride_il, stride_kl, stride_cl,
    # strides for right [B, j, k, c]
    stride_br, stride_jr, stride_kr, stride_cr,
    # strides for out   [B, i, j, c]
    stride_bo, stride_io, stride_jo, stride_co,
    B, N,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_C: tl.constexpr
):
    pid_b = tl.program_id(2)          # batch
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile

    # ----- offsets for the i and j dimensions -----
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    i_off = i_base + tl.arange(0, BLOCK_M)
    j_off = j_base + tl.arange(0, BLOCK_N)

    i_mask = i_off < N
    j_mask = j_off < N

    # hidden‑dim offsets (compile‑time constant)
    c_off = tl.arange(0, BLOCK_C)

    # accumulator for the (i,j) tile, shape (BLOCK_M, BLOCK_N, BLOCK_C)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_C), dtype=tl.float32)

    # ----- reduction over k -----
    for k_start in range(0, N, BLOCK_K):
        k_off = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # iterate over the inner k‑block explicitly
        for kk in range(0, BLOCK_K):
            cur_k   = k_off[kk]
            cur_k_ok = k_mask[kk]

            # clamp out‑of‑range k to zero (address is still legal, mask will zero it)
            cur_k = tl.where(cur_k_ok, cur_k, 0)

            # ----- load left[b, i, cur_k, c]  (shape BLOCK_M × BLOCK_C) -----
            left_ptrs = left_ptr \
                + pid_b * stride_bl \
                + i_off[:, None] * stride_il \
                + cur_k * stride_kl \
                + c_off[None, :] * stride_cl
            left_mask = i_mask[:, None] & cur_k_ok
            left_vec = tl.load(left_ptrs, mask=left_mask, other=0.0)

            # ----- load right[b, j, cur_k, c] (shape BLOCK_N × BLOCK_C) -----
            right_ptrs = right_ptr \
                + pid_b * stride_br \
                + j_off[:, None] * stride_jr \
                + cur_k * stride_kr \
                + c_off[None, :] * stride_cr
            right_mask = j_mask[:, None] & cur_k_ok
            right_vec = tl.load(right_ptrs, mask=right_mask, other=0.0)

            # outer‑product accumulation: acc[i,j,:] += left[i,:] * right[j,:]
            acc += left_vec[:, None, :] * right_vec[None, :, :]

    # ----- store the result -----
    out_ptrs = out_ptr \
        + pid_b * stride_bo \
        + i_off[:, None, None] * stride_io \
        + j_off[None, :, None] * stride_jo \
        + c_off[None, None, :] * stride_co
    out_mask = i_mask[:, None, None] & j_mask[None, :, None]
    tl.store(out_ptrs, acc, mask=out_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, _ = input_tensor.shape  # input shape [B, N, N, dim]

    # ----- LayerNorm on the input -----
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ----- Linear projections -----
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ----- optional mask (mask may be all‑ones for the “nomask” case) -----
    if mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ----- gating -----
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----- Triton kernel for the cubic contraction -----
    # Ensure tensors are contiguous
    left = left.contiguous()
    right = right.contiguous()
    out_hidden = torch.empty_like(left)  # shape [B,N,N,hidden_dim]

    # strides for each tensor
    slb, sli, slk, slc = left.stride()
    srb, srj, srk, src = right.stride()
    sob, soi, soj, soc = out_hidden.stride()

    # tile sizes (tuned for the target hardware)
    BLOCK_M = 4   # i‑dimension tile
    BLOCK_N = 4   # j‑dimension tile
    BLOCK_K = 32  # reduction tile size

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        bs,
    )

    trimul_kernel[grid](
        left, right, out_hidden,
        slb, sli, slk, slc,
        srb, srj, srk, src,
        sob, soi, soj, soc,
        bs, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_C=hidden_dim,
    )

    # ----- Post‑processing (LayerNorm, gating, final linear) -----
    out_hidden = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out_hidden = out_hidden * out_gate

    output = F.linear(out_hidden, weights["to_out.weight"])
    return output
