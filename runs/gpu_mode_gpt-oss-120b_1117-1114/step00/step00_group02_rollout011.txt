def custom_kernel(data):
    import torch, triton, triton.language as tl
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape  # B,N,N,dim
    hidden_dim = config["hidden_dim"]

    # Load weights:
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Mask handling
    if not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp
    else:
        # optional: no masking; ensure shape matches
        pass

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Prepare for Triton GEMM
    # shape (B, N, N, hidden_dim) -> (B, hidden_dim, N, N)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()
    # reshape to (batch_hidden, N, N)
    batch_hidden = B * hidden_dim
    left_mat = left_perm.view(batch_hidden, N, N)
    right_mat = right_perm.view(batch_hidden, N, N)

    # Allocate output
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)  # shape (batch_hidden,N,N)

    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch_hidden, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    trimul_gemm_kernel[grid](left_mat, right_mat, out_mat,
                             M=N, N=N, K=N,
                             stride_Ab=left_mat.stride(0), stride_Am=left_mat.stride(1), stride_Ak=left_mat.stride(2),
                             stride_Bb=right_mat.stride(0), stride_Bk=right_mat.stride(1), stride_Bn=right_mat.stride(2),
                             stride_Cb=out_mat.stride(0), stride_Cm=out_mat.stride(1), stride_Cn=out_mat.stride(2),
                             BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K)

    # Reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # shape [B,N,N,hidden_dim]
    out = out.to(torch.float32)

    # Output norm and linear projection
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul (outgoing) implementation with a custom Triton kernel for the
pairwise matrix multiplication step.

The heavy O(N³) part
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is rewritten as a batched GEMM:
    out_flat = left_flat @ right_flat.T
where each (b,d) pair is treated as an independent batch. The surrounding
layer‑norms, linear projections and gating are performed with PyTorch.
"""
import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for half‑precision tensors.
# A, B, C have shape (batch, M, K) where M = N, K = N.
# The kernel loops over K in blocks and accumulates in FP32.
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix sizes (M=N, K=N)
    stride_Ab, stride_Am, stride_Ak,   # strides for A (batch, row, col)
    stride_Bb, stride_Bk, stride_Bn,   # strides for B (batch, row=k, col=j) -> transposed load
    stride_Cb, stride_Cm, stride_Cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)                 # batch (B * hidden_dim)
    pid_m = tl.program_id(1)               # block row index
    pid_n = tl.program_id(2)               # block col index

    # Compute absolute offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = A_ptr + pid * stride_Ab
    b_base = B_ptr + pid * stride_Bb
    c_base = C_ptr + pid * stride_Cb

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k‑dimension
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---- Load A block (M×K) ----
        a_ptrs = a_base + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # ---- Load B block transposed (K×N) ----
        # B is stored as (batch, row=j, col=k); we need (k, j) => use stride_Bk for k‑row,
        # stride_Bn for j‑col.
        b_ptrs = b_base + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Cast to FP32 and accumulate
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)                 # (BLOCK_M, BLOCK_K) dot (BLOCK_K, BLOCK_N)

    # ---- Store result (M×N) ----
    c_ptrs = c_base + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask: torch.Tensor [B, N, N] (bool/float) – ignored if config["nomask"]==True
            weights: dict of model weights (torch.Tensor, already on the correct device)
            config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------- LayerNorm ---------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]  FP32

    # --------------------------- Linear projections --------------------
    left = F.linear(x, weights["left_proj.weight"])   # -> [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------- Optional mask ------------------------
    if not nomask:
        # broadcast mask to hidden_dim
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------- Gating ------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------- Cast to half for kernel ------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # ------------------------------------------------------------------
    # Prepare tensors for the batched GEMM: flatten (B, hidden_dim) as batch
    # ------------------------------------------------------------------
    # (B, N, N, hidden_dim) -> (B, hidden_dim, N, N) -> (batch_flat, N, N)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()
    batch_flat = B * hidden_dim
    left_mat = left_perm.view(batch_flat, N, N)
    right_mat = right_perm.view(batch_flat, N, N)

    # Allocate output tensor
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # --------------------------- Triton launch ------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_flat,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_Ab=left_mat.stride(0),
        stride_Am=left_mat.stride(1),
        stride_Ak=left_mat.stride(2),
        stride_Bb=right_mat.stride(0),
        stride_Bk=right_mat.stride(1),   # row stride (original j dimension)
        stride_Bn=right_mat.stride(2),   # col stride (original k dimension)
        stride_Cb=out_mat.stride(0),
        stride_Cm=out_mat.stride(1),
        stride_Cn=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------------------------------------------------------------
    # Reshape back to the original layout [B, N, N, hidden_dim]
    # ---------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------- Output LayerNorm --------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"])

    return out
