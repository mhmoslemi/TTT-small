def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # weights: get .to(device) cast to proper dtype (float32) if needed
    # Maybe the weights are already on device.

    # LayerNorm parameters
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Use torch.nn.functional.layer_norm:
    eps = 1e-5
    x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(input_tensor.var(-1, unbiased=False, keepdim=True) + eps)
    x = x * norm_weight + norm_bias

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Mask
    if mask is not None:
        mask = mask.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Permute for matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] with swapped last dims

    # Kernel matmul
    out_perm = tri_mul_matmul(left_perm, right_perm, bs, N, hidden_dim)

    # Permute back
    out = out_perm.permute(0, 2, 3, 1).contiguous()    # [B, N, N, H]

    # to_out_norm
    out = (out - out.mean(-1, keepdim=True)) / torch.sqrt(out.var(-1, unbiased=False, keepdim=True) + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # Gate
    out = out * out_gate

    # final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

def apply_layer_norm(x, weight, bias, eps=1e-5):
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    # weight and bias should be broadcastable
    return x_norm * weight + bias

"""
Outgoing Triangle Multiplicative Update (TriMul) implementation.
The heavy O(N³) tensor multiplication (sum over k) is performed by a custom
Triton kernel that computes, for each batch element and hidden feature d,

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

All other operations (LayerNorm, linear projections, sigmoids, gating) are
implemented with PyTorch. The final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: core outgoing triangle multiplication
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,            # *float32  [B, N, N, H]   left tensor
    right_ptr,           # *float32  [B, N, N, H]   right tensor
    out_ptr,             # *float32  [B, N, N, H]   output tensor
    N,                   # int       sequence length
    stride_b,            # int       stride for batch dimension
    stride_i,            # int       stride for first N dimension (i or j)
    stride_k,            # int       stride for second N dimension (k)
    out_stride_b,        # int       output stride batch
    out_stride_i,        # int       output stride i
    out_stride_j,        # int       output stride j
    H: tl.constexpr,     # hidden dimension (number of heads/features)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row (i)
    pid_n = tl.program_id(1)   # block col (j)
    pid_z = tl.program_id(2)   # combined batch * hidden index

    # decode batch and hidden index from pid_z
    b = pid_z // H
    h = pid_z % H

    # range of i, j for this block
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointer offset for this (b, h)
    base = b * stride_b + h  # because stride of hidden dim = 1 (contiguous)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K) + k_offset
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + base \
                    + offs_i[:, None] * stride_i \
                    + offs_k[None, :] * stride_k
        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right tile: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + base \
                     + offs_j[:, None] * stride_i \
                     + offs_k[None, :] * stride_k
        b_mat = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0)

        # accumulate dot product
        acc += tl.dot(a, b_mat)

    # write back
    out_ptrs = out_ptr + b * out_stride_b + h \
               + offs_i[:, None] * out_stride_i \
               + offs_j[None, :] * out_stride_j
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper: launch the Triton kernel
# ----------------------------------------------------------------------
def tri_mul_outgoing_tiled(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the outgoing triangle multiplication using Triton.
    left/right: [B, N, N, H] contiguous float32 tensors.
    Returns out of same shape.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # tiling parameters (tuned for typical N up to 1024)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # strides (in elements)
    stride_b = left.stride(0)
    stride_i = left.stride(1)
    stride_k = left.stride(2)

    out_stride_b = out.stride(0)
    out_stride_i = out.stride(1)
    out_stride_j = out.stride(2)

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        N,
        stride_b,
        stride_i,
        stride_k,
        out_stride_b,
        out_stride_i,
        out_stride_j,
        H=H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module with a Triton kernel for the
    O(N³) update.
    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [bs, seq_len, seq_len, dim]
            - mask: torch.Tensor [bs, seq_len, seq_len] (float / bool)
            - weights: dict of torch.Tensors for all linear and norm weights
            - config: dict containing 'dim' and 'hidden_dim'
    Returns:
        torch.Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_unsq
        right = right * mask_unsq

    # ------------------------------------------------------------------
    # Gating values
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguity before launching Triton
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # Core O(N³) multiplication via Triton
    # ------------------------------------------------------------------
    out = tri_mul_outgoing_tiled(left, right)  # shape [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Output LayerNorm (hidden_dim) + gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
