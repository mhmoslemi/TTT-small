@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   B, N, H,
                   stride_l_b, stride_l_i, stride_l_k, stride_l_h,
                   stride_r_b, stride_r_i, stride_r_k, stride_r_h,
                   stride_o_b, stride_o_i, stride_o_j, stride_o_h,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)
    # compute batch and hidden dimensions
    b = pid_z // H
    h = pid_z % H
    # compute range offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (left)
        a_ptrs = left_ptr + b * stride_l_b + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k + h * stride_l_h
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # Cast to float32
        a = a.to(tl.float32)

        # Load B tile (right transposed)
        b_ptrs = right_ptr + b * stride_r_b + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i + h * stride_r_h
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        b = b.to(tl.float32)

        # Compute block multiplication
        acc += tl.dot(a, b)

    # Store result
    out_ptrs = out_ptr + b * stride_o_b + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j + h * stride_o_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) implementation optimized with Triton.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor.
2. Linear projections `left_proj`, `right_proj` (no bias) → shape (B,N,N,H).
3. Compute three gating projections on the normalized input and sigmoid them.
4. Apply optional mask to the projected tensors and then mask‑gates:
   left  = left  * left_gate  * mask
   right = right * right_gate * mask
5. Core O(N³) reduction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply (left @ rightᵀ) for each (b,h) pair.
   Implemented as a Triton kernel that tiles the i/j dimensions and loops
   over k in blocks. Computation is performed in fp32 while data is stored
   in fp16 to reduce bandwidth.
6. Layer‑norm over the hidden dimension, multiply by the out‑gate,
   and a final linear projection back to the original dimension.

The kernel works for any batch size (1‑2), sequence length up to 1024,
and hidden dimensions like 128. All heavy work (the triple‑sum) is done
on the GPU via Triton; the surrounding ops stay in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: computes out = left @ rightᵀ for each batch+head
# --------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    NUM_K: tl.constexpr
):
    pid_i = tl.program_id(0)               # tile over output rows (i)
    pid_j = tl.program_id(1)               # tile over output cols (j)
    pid_z = tl.program_id(2)               # combined batch and hidden index

    # decode batch and hidden indices
    b = pid_z // H
    h = pid_z % H

    # offsets for the current block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k in tiles
    for k_iter in range(NUM_K):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile (i,k) -----
        a_ptrs = (left_ptr + b * stride_l_b
                  + offs_i[:, None] * stride_l_i
                  + offs_k[None, :] * stride_l_k
                  + h * stride_l_h)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = tl.cast(a, tl.float32)

        # ----- load right tile transposed (k,j) -----
        # we read right as (k, j) so that the multiplication yields
        # sum_k left[i,k] * right[j,k]
        b_ptrs = (right_ptr + b * stride_r_b
                  + offs_k[:, None] * stride_r_k
                  + offs_j[None, :] * stride_r_i
                  + h * stride_r_h)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        b = tl.cast(b, tl.float32)

        # matrix‑multiply the two tiles
        acc += tl.dot(a, b)   # (BLOCK_M,BLOCK_K) @ (BLOCK_K,BLOCK_N) → (BLOCK_M,BLOCK_N)

    # ----- write the result -----
    out_ptrs = (out_ptr + b * stride_o_b
                + offs_i[:, None] * stride_o_i
                + offs_j[None, :] * stride_o_j
                + h * stride_o_h)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_forward(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the outgoing TriMul reduction using the Triton kernel.
    Arguments:
        left, right: tensors of shape (B, N, N, H), dtype torch.float16 or float32
    Returns:
        out: tensor of shape (B, N, N, H), dtype torch.float32
    """
    B, N, _, H = left.shape
    assert right.shape == (B, N, N, H)

    # Output buffer (float32 for accumulation stability)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # Ensure contiguous layout (required for correct stride handling)
    left = left.contiguous()
    right = right.contiguous()

    # Strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile sizes – 64 works well for most GPUs, can be tuned.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid configuration
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,                # one program for each (batch, hidden) pair
    )
    num_k_tiles = triton.cdiv(N, BLOCK_K)

    _tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        NUM_K=num_k_tiles,
    )
    return out

    Forward pass of the outgoing TriMul module with Triton acceleration.
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply mask (if present) and gating
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask
        right = right * mask

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy reduction – Triton kernel
    # ------------------------------------------------------------------
    # Reduce memory pressure by casting to half precision before the kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    out = _tri_mul_forward(left, right)          # shape (B,N,N,hidden_dim), fp32

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                       # element‑wise gate

    out = F.linear(out, weights["to_out.weight"])  # back to original dim

    # Ensure final dtype is float32 as required
    out = out.to(torch.float32)
    return out
