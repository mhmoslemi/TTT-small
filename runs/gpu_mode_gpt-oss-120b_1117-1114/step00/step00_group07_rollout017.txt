def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    eps = 1e-5

    # Ensure contiguous
    x = input_tensor
    x = x.contiguous()
    mask = mask.contiguous().float()
    
    # LayerNorm on dim
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # Apply mask and gates via Triton kernel
    B, N, _, _ = left_proj.shape
    # allocate output tensors
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # Launch kernel for left
    total_elements = B * N * N * hidden_dim
    BLOCK = 128
    grid = (triton.cdiv(total_elements, BLOCK),)
    mask_gate_kernel[grid](left_proj.data_ptr(),
                           left_gate.data_ptr(),
                           mask.data_ptr(),
                           left.data_ptr(),
                           B, N, hidden_dim,
                           BLOCK=BLOCK)
    # Right
    mask_gate_kernel[grid](right_proj.data_ptr(),
                           right_gate.data_ptr(),
                           mask.data_ptr(),
                           right.data_ptr(),
                           B, N, hidden_dim,
                           BLOCK=BLOCK)

    # Ensure kernels finished
    torch.cuda.synchronize()

    # Compute out using batched matmul
    # reshape for bmm
    left_reshape = left.permute(0,3,1,2).reshape(B * hidden_dim, N, N)
    right_t = right.permute(0,3,2,1).reshape(B * hidden_dim, N, N)  # transposed over k,j
    out_reshape = torch.bmm(left_reshape, right_t)  # shape (B*H, N, N)
    out = out_reshape.view(B, hidden_dim, N, N).permute(0,2,3,1)  # shape (B,N,N,H)

    # to_out_norm layernorm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Multiply with out_gate
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul forward kernel (outgoing version) with a Triton elementwise
fusion. The heavy O(N³·hidden_dim) reduction is performed with a
batched matrix‑multiply (torch.bmm) after projecting and gating the
input. A small Triton kernel fuses the mask, left/right gates and the
projected tensors, eliminating an extra Python‐level elementwise pass.
The implementation follows the reference PyTorch code but
off‑loads the mask‑gate multiplication to Triton for a demonstrated
use of a custom kernel.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: out = proj * gate * mask   (mask is broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,                # runtime arguments
    BLOCK: tl.constexpr,    # compile‑time tile size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)  # linear offsets for the flattened tensor

    total = B * N * N * H           # total number of elements in proj/gate/out
    mask_total = B * N * N          # total number of elements in mask
    batch_stride = N * N * H        # stride between batches in flattened view

    # Compute (b,i,j,d) indices from the linear offset
    b = offs // batch_stride
    rem = offs % batch_stride
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    # d = rem2 % H   # not needed for mask

    # Index into the mask (broadcasted over d)
    mask_idx = b * (N * N) + i * N + j

    # Load with bounds checking
    mask = tl.load(mask_ptr + mask_idx,
                   mask=mask_idx < mask_total,
                   other=0.0).to(tl.float32)
    proj = tl.load(proj_ptr + offs,
                   mask=offs < total,
                   other=0.0)
    gate = tl.load(gate_ptr + offs,
                   mask=offs < total,
                   other=0.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=offs < total)


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Compute the TriMul forward pass.

    Parameters
    ----------
    data : Tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (float, 0/1)
        - weights      : dict with keys as in the reference module
        - config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack and basic shapes
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Ensure contiguity and correct dtype
    x = input_tensor.contiguous()
    mask = mask.contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # LayerNorm on the input (normalize over last dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(
        x_norm, weights["left_proj.weight"]
    )  # [B, N, N, hidden_dim]
    right_proj = torch.nn.functional.linear(
        x_norm, weights["right_proj.weight"]
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Gates (linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["left_gate.weight"])
    )  # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["right_gate.weight"])
    )  # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["out_gate.weight"])
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Fuse mask, projection and gate using Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left_proj.shape  # batch, seq_len, seq_len, hidden_dim
    total_elements = B * N * N * hidden_dim
    BLOCK = 256  # threads per block (tuned for typical GPUs)
    grid = (triton.cdiv(total_elements, BLOCK),)

    # Allocate buffers for the gated & masked tensors
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # Launch kernel for left side
    _mask_gate_kernel[grid](
        left_proj.data_ptr(),
        left_gate.data_ptr(),
        mask.data_ptr(),
        left.data_ptr(),
        B, N, hidden_dim,
        BLOCK=BLOCK,
    )
    # Launch kernel for right side
    _mask_gate_kernel[grid](
        right_proj.data_ptr(),
        right_gate.data_ptr(),
        mask.data_ptr(),
        right.data_ptr(),
        B, N, hidden_dim,
        BLOCK=BLOCK,
    )
    # Synchronize to make sure the kernels have finished before next ops
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # O(N³) reduction: batch of matrix‑multiplications per hidden dim
    # out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Reshape to (B*H, N, N) where rows correspond to i and columns to k (left)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    # Right needs to be transposed on the (k, j) axes before multiplication
    right_mat_t = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)

    # Batched matrix multiply
    out_mat = torch.bmm(left_mat, right_mat_t)  # (B*H, N, N)

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Final LayerNorm, gating, and output projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, hidden_dim]

    out = out * out_gate  # apply output gate

    # Linear projection back to original dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # [B, N, N, dim]

    return out
