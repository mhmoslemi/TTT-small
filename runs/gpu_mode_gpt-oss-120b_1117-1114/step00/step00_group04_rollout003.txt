def tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    N,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_z = tl.program_id(2)  # batch*hidden_dim index

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out of bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to start
    a_ptrs = A_ptr + pid_z * stride_a_batch + (offs_m[:, None] * stride_a_row) + (offs_k[None, :] * stride_a_col)
    b_ptrs = B_ptr + pid_z * stride_b_batch + (offs_k[:, None] * stride_b_row) + (offs_n[None, :] * stride_b_col)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < N - k*BLOCK_K), other=0.0)  # but using dynamic mask per tile

def custom_kernel(data):

    input_tensor, mask, weights, config = data

    device = input_tensor.device

    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    # Load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Normalize input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # Compute projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if needed
    if not nomask:
        # mask shape [B,N,N]; expand
        mask_exp = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # Compute gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Prepare for kernel: permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose right for matmul
    right_T = right_perm.transpose(-2, -1).contiguous()

    B, H, N, N2 = left_perm.shape
    assert N == N2, "N must be equal"

    # Flatten batch and hidden_dim
    batch_hidden = B * H
    A = left_perm.view(batch_hidden, N, N)
    B_mat = right_T.view(batch_hidden, N, N)
    C = torch.empty_like(A)

    # Compute strides
    stride_a_batch, stride_a_row, stride_a_col = A.stride()
    stride_b_batch, stride_b_row, stride_b_col = B_mat.stride()
    stride_c_batch, stride_c_row, stride_c_col = C.stride()

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # compute grid
    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch_hidden,              # batch*hidden_dim
    )

    # launch
    tri_mul_matmul_kernel[grid](
        A.data_ptr(),
        B_mat.data_ptr(),
        C.data_ptr(),
        N,
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_b_batch,
        stride_b_row,
        stride_b_col,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape output
    out_perm = C.view(B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # shape B,N,N,hidden_dim

    # Final steps
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate  # elementwise gating

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_weight)  # shape B,N,N,dim

    return out

"""
TriMul (outgoing) forward pass with a Triton kernel for the O(N³) reduction.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections to a hidden dimension (left, right) and three gating
   projections (left_gate, right_gate, out_gate).  Apply sigmoid to the gates.
3. Apply the optional mask and multiply the gated projections.
4. Compute the core tensor contraction:
        out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]
   for every batch b and hidden channel d.
   This is a batched matrix‑multiply of shape (B·D, N, N) × (B·D, N, N)ᵀ.
   The reduction is performed by a custom Triton kernel that tiles the i‑j
   plane (BLOCK_M × BLOCK_N) and iterates over k in BLOCK_K tiles.
5. Layer‑norm the result, apply the output gate and a final linear projection
   back to the original dimension.

Only the O(N³) contraction is executed in Triton; all other operations use
standard PyTorch for simplicity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for batched matrix multiplication (A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    A_ptr,                               # *float32
    B_ptr,                               # *float32
    C_ptr,                               # *float32
    N,                                   # sequence length (scalar)
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_z = tl.program_id(2)          # flattened batch*channel index

    # Offsets for the current 2‑D tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension in tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k_tile * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        k_valid = (cur_k + offs_k) < N

        # Pointers to the tiles of A and Bᵀ
        a_ptrs = (A_ptr + pid_z * stride_a_batch
                  + offs_m[:, None] * stride_a_row
                  + (cur_k + offs_k)[None, :] * stride_a_col)

        b_ptrs = (B_ptr + pid_z * stride_b_batch
                  + (cur_k + offs_k)[:, None] * stride_b_row
                  + offs_n[None, :] * stride_b_col)

        # Load with masking; out‑of‑range elements are replaced by 0.0
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_valid[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & mask_n[None, :],
                    other=0.0)

        # Tile multiplication
        acc += tl.dot(a, b)

    # Store the result tile
    c_ptrs = (C_ptr + pid_z * stride_c_batch
              + offs_m[:, None] * stride_c_row
              + offs_n[None, :] * stride_c_col)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (binary float or bool)
            - weights: dict of torch.Tensors containing the model parameters
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same device)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load model parameters (they are already on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)      # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)    # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid on linear projections of the original normalized x)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction using Triton
    #    Compute: out_{i,j,d} = Σₖ left_{i,k,d} * right_{j,k,d}
    # ------------------------------------------------------------------
    # Bring tensors to (B, hidden_dim, N, N) layout and make them contiguous.
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()        # [B, H, N, N]
    # Transpose the last two axes of right for the required A @ Bᵀ product.
    right_T = right_perm.transpose(-2, -1).contiguous()      # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    batch_hidden = B * H

    # Flatten batch and channel into a single dimension for the kernel.
    A = left_perm.view(batch_hidden, N, N)
    B_mat = right_T.view(batch_hidden, N, N)
    C = torch.empty_like(A)

    # Strides for the flattened tensors (they are contiguous).
    stride_a_batch, stride_a_row, stride_a_col = A.stride()
    stride_b_batch, stride_b_row, stride_b_col = B_mat.stride()
    stride_c_batch, stride_c_row, stride_c_col = C.stride()

    # Kernel launch configuration.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        batch_hidden,                     # batch * hidden_dim
    )

    # Launch the Triton kernel.
    tri_mul_matmul_kernel[grid](
        A.data_ptr(),
        B_mat.data_ptr(),
        C.data_ptr(),
        N,
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_b_batch,
        stride_b_row,
        stride_b_col,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, H, N, N) and then to (B, N, N, H)
    out_perm = C.view(B, H, N, N)                 # [B, H, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Final LayerNorm, output gate, and linear projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate                     # element‑wise gating

    # Linear projection to the original dimension (no bias)
    out = F.linear(out, to_out_weight)       # [B, N, N, dim]

    return out
```
