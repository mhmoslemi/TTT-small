"""
TriMul (outgoing) kernel implementation.

The kernel fuses mask application and gating (sigmoid) for the left and right
linear projections.  The heavy O(N³·C_hidden) contraction is performed as a
batched matrix multiplication per hidden channel:
    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
which is equivalent to a GEMM for each hidden dimension.
We use torch.nn.functional for LayerNorm, Linear, and the GEMM, and a small
Triton kernel to apply `mask * gate` without materialising the broadcasted mask.
The final output is LayerNorm‑gated and projected back to the original channel
dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_mask_gate_kernel(
    out_ptr,        # *mut f32   (B,N,N,hidden)
    inp_ptr,        # *mut f32   (B,N,N,hidden) – projected tensor
    gate_ptr,       # *mut f32   (B,N,N,hidden) – sigmoid gate
    mask_ptr,       # *mut f32   (B,N,N)         – flattened mask
    B, N, H,        # runtime sizes
    TOTAL_ELEMS: tl.constexpr,  # total elements in the 4‑D tensors
    BLOCK: tl.constexpr,        # block size (compile‑time)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < TOTAL_ELEMS

    # load projected values and gates
    x = tl.load(inp_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # ----- compute mask index (b,i,j) for the flattened mask -----
    # each element corresponds to (b,i,j,c) in the 4‑D tensor
    # linear index = b*(N*N*H) + i*(N*H) + j*H + c
    # we need mask index = b*(N*N) + i*N + j
    nh = N * N * H
    b = offs // nh
    rem = offs % nh
    i = rem // (N * H)
    rem = rem % (N * H)
    j = rem // H
    mask_idx = b * (N * N) + i * N + j

    m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    # fused masking & gating
    out = x * g * m
    tl.store(out_ptr + offs, out, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, C] (float32/float16/bfloat16)
            - mask_tensor:  torch.Tensor [B, N, N] (float or bool)
            - weights_dict: mapping from string names to weight tensors
            - config_dict: contains at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2. Linear projections + gating pre‑sigmoid
    # -----------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # -----------------------------------------------------------------
    # 3. Apply mask & element‑wise gating via Triton
    # -----------------------------------------------------------------
    # Ensure mask is float32 on the same device
    if mask is None:
        mask = torch.ones((B, N, N), device=device, dtype=torch.float32)
    else:
        mask = mask.to(dtype=torch.float32, device=device)

    mask_flat = mask.contiguous().view(-1)                # (B*N*N,)
    total_elems = left_proj.numel()                       # B*N*N*hidden_dim

    # Destination tensors
    left_gated = torch.empty_like(left_proj)
    right_gated = torch.empty_like(right_proj)

    BLOCK = 1024
    grid = (triton.cdiv(total_elems, BLOCK),)

    # left side
    _apply_mask_gate_kernel[
        grid
    ](
        left_gated,
        left_proj,
        left_gate,
        mask_flat,
        B,
        N,
        hidden_dim,
        total_elems,
        BLOCK=BLOCK,
    )
    # right side
    _apply_mask_gate_kernel[
        grid
    ](
        right_gated,
        right_proj,
        right_gate,
        mask_flat,
        B,
        N,
        hidden_dim,
        total_elems,
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 4. Triangular multiplication (batched GEMM per hidden channel)
    #    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
    # -----------------------------------------------------------------
    # reshape to (B*hidden, N, N) for batched matmul
    left_perm = left_gated.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_perm = right_gated.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_perm, right_perm.transpose(1, 2))            # (B*hidden, N, N)

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)          # (B, N, N, hidden)

    # -----------------------------------------------------------------
    # 5. Output LayerNorm, final gating and projection back to C
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # Ensure final dtype is float32 as required
    out = out.to(dtype=torch.float32)
    return out
