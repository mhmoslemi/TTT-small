```
@triton.jit
def trimul_forward_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_b, stride_i, stride_k, stride_d,
    out_stride_b, out_stride_i, out_stride_j, out_stride_d,
    d_start,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute global indices
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    d = d_start + tl.arange(0, BLOCK_D)

    mask_i = i < N
    mask_j = j < N
    mask_d = d < hidden_dim

    # create broadcast masks for loads
    # These will be used later

    # Set offsets base for batch
    batch_offset = pid_b * stride_b

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # Loop over K dim
    for k_start in range(0, N, BLOCK_K):
        k = k_start + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Load left tile
        left_ptrs = left_ptr + batch_offset \
            + i[:, None, None] * stride_i \
            + k[None, :, None] * stride_k \
            + d[None, None, :] * stride_d
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None],
                       other=0.0,
                       dtype=tl.float16)
        left = left.to(tl.float32)

        # Load right tile
        right_ptrs = right_ptr + batch_offset \
            + j[:, None, None] * stride_i \
            + k[None, :, None] * stride_k \
            + d[None, None, :] * stride_d
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None, None] & mask_k[None, :, None],
                        other=0.0,
                        dtype=tl.float16)

        + i[:, None, None] * out_stride_i \
        + j[None, :, None] * out_stride_j \
        + d[None, None, :] * out_stride_d
    store_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=store_mask)

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The heavy O(N³) component
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is computed on GPU with a custom Triton kernel.  The surrounding
LayerNorms, Linear projections and element‑wise gates are performed in
PyTorch (they are cheap compared to the tensor‑wise multiplication).
The kernel tiles the (i, j) output plane (BLOCK_M × BLOCK_N) and the
hidden dimension (BLOCK_D) while looping over the reduction axis k in
tiles of size BLOCK_K.  Each thread block computes a (BLOCK_M,
BLOCK_N, BLOCK_D) tile in FP32 accumulation and writes FP16 results.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_forward_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_b, stride_i, stride_k, stride_d,
    out_stride_b, out_stride_i, out_stride_j, out_stride_d,
    d_start,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    """Compute out = Σ_k left[...,i,k,d] * right[...,j,k,d].

    Arguments:
        left_ptr, right_ptr, out_ptr: base pointers of the tensors.
        B, N, hidden_dim: tensor dimensions.
        stride_*: strides (in elements) for the input tensors.
        out_stride_*: strides for the output tensor.
        d_start: start index of the hidden‑dim tile handled by this launch.
        BLOCK_M, BLOCK_N: tile sizes for the i and j axes.
        BLOCK_K: tile size for the reduction axis k.
        BLOCK_D: tile size for the hidden dimension.
    """
    pid_b = tl.program_id(0)   # batch axis
    pid_i = tl.program_id(1)   # i‑tile axis
    pid_j = tl.program_id(2)   # j‑tile axis

    # Global coordinates for this tile
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    d = d_start + tl.arange(0, BLOCK_D)

    mask_i = i < N
    mask_j = j < N
    mask_d = d < hidden_dim

    # Base offset for the current batch
    batch_offset = pid_b * stride_b

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # Loop over the k dimension in tiles
    for k_start in range(0, N, BLOCK_K):
        k = k_start + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # -------------------- load left --------------------
        left_off = (
            batch_offset
            + i[:, None, None] * stride_i
            + k[None, :, None] * stride_k
            + d[None, None, :] * stride_d
        )
        left_mask = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
        left = tl.load(left_ptr + left_off,
                       mask=left_mask,
                       other=0.0,
                       dtype=tl.float16)
        left = left.to(tl.float32)   # promote for accumulation

        # -------------------- load right -------------------
        right_off = (
            batch_offset
            + j[:, None, None] * stride_i   # stride for the second dimension is identical
            + k[None, :, None] * stride_k
            + d[None, None, :] * stride_d
        )
        right_mask = mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
        right = tl.load(right_ptr + right_off,
                        mask=right_mask,
                        other=0.0,
                        dtype=tl.float16)
        right = right.to(tl.float32)

        # ------------------ accumulate --------------------
        # Loop over the inner K tile (unrolled, BLOCK_K is constexpr)
        for kk in range(0, BLOCK_K):
            # left[ :, kk, : ] -> (M, D)
            # right[:, kk, :] -> (N, D)
            left_kk = left[:, kk, :]          # (M, D)
            right_kk = right[:, kk, :]        # (N, D)
            # outer product across i and j for this k, broadcast over D
            acc += left_kk[:, None, :] * right_kk[None, :, :]

    # -------------------- store result --------------------
    out_off = (
        batch_offset
        + i[:, None, None] * out_stride_i
        + j[None, :, None] * out_stride_j
        + d[None, None, :] * out_stride_d
    )
    store_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
    tl.store(out_ptr + out_off,
             acc.to(tl.float16),
             mask=store_mask)


def custom_kernel(data):
    """
    Forward pass for the TriMul operator (outgoing version).

    Args:
        data: Tuple (input, mask, weights, config)
            - input: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/int)
            - weights: dict of torch.Tensors containing model parameters
            - config: dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------- LayerNorm on the input -----------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ----------- Linear projections ---------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ----------- Optional masking -----------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).type_as(left)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ----------- Gates ---------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----------- Cast to FP16 for the heavy kernel -----
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape

    # Output buffer (still FP16)
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)

    # Strides (in number of elements)
    stride_b, stride_i, stride_k, stride_d = left_fp16.stride()  # (B, i, k, d)
    out_stride_b, out_stride_i, out_stride_j, out_stride_d = out_fp16.stride()

    # Kernel tile sizes
    BLOCK_M = 8
    BLOCK_N = 8
    BLOCK_K = 16
    BLOCK_D = 32  # hidden‑dim tile; final chunk handled by masking

    grid = (B, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    # Loop over hidden‑dim tiles
    for d_start in range(0, hidden_dim, BLOCK_D):
        trimul_forward_kernel[grid](
            left_fp16,
            right_fp16,
            out_fp16,
            B,
            N,
            hidden_dim,
            stride_b,
            stride_i,
            stride_k,
            stride_d,
            out_stride_b,
            out_stride_i,
            out_stride_j,
            out_stride_d,
            d_start,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )

    # Convert back to FP32 for the remaining layers
    out = out_fp16.to(torch.float32)

    # ----------- Final LayerNorm and projection ---------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
