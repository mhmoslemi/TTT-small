def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bc = tl.program_id(2)

    # compute batch and channel
    b = pid_bc // H
    c = pid_bc % H

"""
Custom Triton kernel for TriMul (outgoing) operation.
...
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """ Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h] """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bc = tl.program_id(2)

    # Decode batch and channel from flattened block id
    b = pid_bc // H
    h = pid_bc % H

    # Index offsets for the tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator for the tile (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_iter * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------- load left tile (M, K) ----------
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k + \
            h * stride_left_h
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0)
        left = left.to(tl.float32)  # promote for accumulation

        # ------- load right tile (N, K) ----------
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            offs_j[:, None] * stride_right_i + \
            offs_k[None, :] * stride_right_k + \
            h * stride_right_h
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0)
        right = right.to(tl.float32)

        # Outer product over i (M) and j (N) for each k, sum across k
        # left: (M, K) -> (M,1,K); right: (N, K) -> (1,N,K)
        # broadcast multiply -> (M,N,K), then sum over K
        prod = left[:, None, :] * right[None, :, :]
        acc += tl.sum(prod, axis=2)

    # Store the result back to global memory
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j + \
        h * stride_out_h
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :]
    )

def custom_kernel(data):
    """
    Compute the outgoing TriMul operation using a fused Triton kernel for the
    O(N³) pairwise multiplication. All other linear / norm / gating operations
    are performed with PyTorch for simplicity.
    Args:
        data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
            (input, mask, weights, config)
    Returns:
        torch.Tensor with shape [bs, seq_len, seq_len, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm over the feature dimension ----------
    # LayerNorm weight and bias
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ---------- Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Cast to half precision for the heavy multiplication
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ---------- Apply mask ----------
    if mask is not None:
        # mask shape [B, N, N]; broadcast to last dim
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_
    # If mask is absent (nomask), treat as all ones (no change)

    # ---------- Gating ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # float32
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates (cast gate to half for multiplication)
    left = (left * left_gate.to(left.dtype)).contiguous()
    right = (right * right_gate.to(right.dtype)).contiguous()

    # ---------- Triton kernel for the O(N^3) multiplication ----------
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # Strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Block sizes – chosen to fit typical GPU registers while covering a range of N.
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    grid_bc = B * hidden_dim   # flattened batch * channel dimension

    grid = (grid_i, grid_j, grid_bc)

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Output transformations ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate   # gating after norm

    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)  # shape [B,N,N,dim]

    return out

"""
Custom Triton kernel for the outgoing TriMul operation.

The core of TriMul is:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

where `left` and `right` have shape [B, N, N, H] (H = hidden_dim).
All other components (LayerNorms, linear projections, gates) are kept in
PyTorch for clarity.  The O(N³) multiplication is executed by a Triton kernel
that processes a tile (BLOCK_M × BLOCK_N) of the output for a single batch
and hidden dimension, looping over the reduction axis `k` in blocks of
size BLOCK_K.  The kernel uses half‑precision loads for `left`/`right`,
accumulates in float32, and writes the result as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]."""
    pid_i = tl.program_id(0)          # tile row
    pid_j = tl.program_id(1)          # tile column
    pid_bc = tl.program_id(2)         # flattened batch*channel index

    # Decode batch index and hidden dimension index
    b = pid_bc // H
    h = pid_bc % H

    # Tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k in blocks of size BLOCK_K
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_iter * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left[i,k] -----
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k + \
            h * stride_left_h
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0)
        left = left.to(tl.float32)

        # ----- load right[j,k] -----
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            offs_j[:, None] * stride_right_i + \
            offs_k[None, :] * stride_right_k + \
            h * stride_right_h
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0)
        right = right.to(tl.float32)

        # Outer‑product over i and j for this k‑block and sum over k
        # left: (M, K) → (M,1,K); right: (N, K) → (1,N,K)
        prod = left[:, None, :] * right[None, :, :]   # (M,N,K)
        acc += tl.sum(prod, axis=2)                 # (M,N)

    # ----- write back -----
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j + \
        h * stride_out_h
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :]
    )

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # -> [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # Use half precision for the heavy O(N³) work
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Optional mask (already broadcasted)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates (cast gate to half for the multiplication)
    left = (left * left_gate.to(left.dtype)).contiguous()
    right = (right * right_gate.to(right.dtype)).contiguous()

    # ------------------------------------------------------------------
    # 5. Triton kernel: Σ_k left[...,k,h] * right[...,k,h]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # Strides in element units (not bytes)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Blocking parameters – work well for N up to 1024 and hidden_dim ≤ 384
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    grid_bc = B * hidden_dim
    grid = (grid_i, grid_j, grid_bc)

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        s_lb,
        s_li,
        s_lk,
        s_lh,
        s_rb,
        s_ri,
        s_rk,
        s_rh,
        s_ob,
        s_oi,
        s_oj,
        s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # -> [B, N, N, dim]

    return out
