                         B, N, H,
                         stride_batch, stride_row, stride_k,
                         BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # derive combined index
    pid_z = tl.program_id(0) # combined batch*hidden
    pid_m = tl.program_id(1) # i tile
    pid_n = tl.program_id(2) # j tile

    # compute batch and hidden index:
    batch_id = pid_z // H
    hidden_id = pid_z % H

    # compute offsets to base
    # base offset for left/right/out includes batch and hidden offset
    batch_offset = batch_id * stride_batch
    hidden_offset = hidden_id

    # compute i,j start
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # compute offsets
    i_offsets = i_start + tl.arange(0, BLOCK_M)
    j_offsets = j_start + tl.arange(0, BLOCK_N)

    # mask for i,j bounds
    mask_i = i_offsets < N
    mask_j = j_offsets < N

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < N

        # compute pointers
        # load left tile
        a_ptrs = left_ptr + batch_offset + hidden_offset + i_offsets[:, None] * stride_row + (k_offsets[None, :] * stride_k)
        # load right tile
        b_ptrs = right_ptr + batch_offset + hidden_offset + (k_offsets[:, None] * stride_k) + j_offsets[None, :] * stride_row

        # load with mask
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

    # mask store

"""
TriMul (outgoing) implementation with a Triton kernel.

The heavy part of the forward pass is the 4‑D reduction

    out[..., i, j, d] = sum_k left[..., i, k, d] * right[..., j, k, d]

which is equivalent to a batch of matrix multiplications per hidden dimension.
The kernel below treats each (batch, hidden) pair as an independent
matrix multiplication (N×N)·(N×N)^T and tiles the i (rows) and j (cols) axes.
All other operations (LayerNorm, Linear projections, gating, final
LayerNorm and output Linear) are performed with PyTorch.
The kernel works in fp16 for the inputs and accumulates in fp32 for
numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: tiled matmul for the TriMul reduction.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # batch, seq_len, hidden_dim
    stride_batch, stride_row, stride_k,         # strides for left/right/out
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # --------------------------------------------------------------
    # Grid layout:
    #   program_id(0) -> combined (batch, hidden) index  b*H + h
    #   program_id(1) -> i‑tile (rows)
    #   program_id(2) -> j‑tile (cols)
    # --------------------------------------------------------------
    pid_z = tl.program_id(0)   # (batch * hidden) dimension
    pid_m = tl.program_id(1)   # i‑tile
    pid_n = tl.program_id(2)   # j‑tile

    # ------------------------------------------------------------------
    # Recover batch and hidden indices from the flattened pid_z
    # ------------------------------------------------------------------
    batch_id  = pid_z // H
    hidden_id = pid_z % H

    # ------------------------------------------------------------------
    # Base address for this (batch, hidden) slice.
    # The hidden dimension is the innermost one, stride = 1.
    # ------------------------------------------------------------------
    batch_offset  = batch_id * stride_batch
    hidden_offset = hidden_id                     # stride_h == 1

    # ------------------------------------------------------------------
    # Tile start indices for the i and j dimensions
    # ------------------------------------------------------------------
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    i_offsets = i_start + tl.arange(0, BLOCK_M)
    j_offsets = j_start + tl.arange(0, BLOCK_N)

    mask_i = i_offsets < N
    mask_j = j_offsets < N

    # ------------------------------------------------------------------
    # Accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k in tiles of size BLOCK_K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < N

        # --------------------------------------------------------------
        # Pointers to the current tile of the left operand (shape MxK)
        # --------------------------------------------------------------
        a_ptrs = left_ptr + batch_offset + hidden_offset \
                 + i_offsets[:, None] * stride_row \
                 + k_offsets[None, :] * stride_k

        # --------------------------------------------------------------
        # Pointers to the current tile of the right operand (shape KxN)
        # --------------------------------------------------------------
        b_ptrs = right_ptr + batch_offset + hidden_offset \
                 + k_offsets[:, None] * stride_k \
                 + j_offsets[None, :] * stride_row

        # --------------------------------------------------------------
        # Load tiles with out‑of‑bounds masking
        # --------------------------------------------------------------
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # --------------------------------------------------------------
        # Matrix‑multiply‑accumulate into fp32 accumulator
        # --------------------------------------------------------------
        acc += tl.dot(a, b)          # a,b are fp16 → acc is fp32

    # ------------------------------------------------------------------
    # Write the result tile to output tensor (convert back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = out_ptr + batch_offset + hidden_offset \
             + i_offsets[:, None] * stride_row \
             + j_offsets[None, :] * stride_row

    mask_store = mask_i[:, None] & mask_j[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_store)


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module (outgoing variant) with a Triton kernel
    for the core bilinear reduction.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
            input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            mask:         torch.Tensor of shape [B, N, N]    (float32 or bool)
            weights_dict: mapping from weight name to torch.Tensor
            config_dict:  contains "dim" and "hidden_dim" (int)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -------------------------------
    # unpack arguments
    # -------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -------------------------------
    # Helper to fetch weight tensors
    # -------------------------------
    def w(name):
        return weights[name].to(device)

    # -------------------------------
    # LayerNorm on the input (dim)
    # -------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
    )  # [B, N, N, dim]  fp32

    # -------------------------------
    # Linear projections (no bias)
    # -------------------------------
    left = F.linear(x, w("left_proj.weight"))   # [B,N,N,hidden]
    right = F.linear(x, w("right_proj.weight"))

    # -------------------------------
    # Mask handling (mask may be all‑ones when nomask=True)
    # -------------------------------
    if mask is None:
        mask_fp = torch.ones_like(left[..., :1])
    else:
        mask_fp = mask.unsqueeze(-1).to(left.dtype)  # [B,N,N,1]

    left = left * mask_fp
    right = right * mask_fp

    # -------------------------------
    # Gating
    # -------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, w("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, w("out_gate.weight")))

    left  = (left * left_gate).to(torch.float16).contiguous()
    right = (right * right_gate).to(torch.float16).contiguous()

    # -------------------------------
    # Triton kernel for the reduction
    # -------------------------------
    B, N, _, _ = left.shape

    # Output buffer (fp16) – same shape as left/right
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Strides of the tensors (all are identical for left/right/out)
    stride_batch = left.stride(0)   # B‑stride = N*N*H
    stride_row   = left.stride(1)   # i‑/j‑stride = N*H
    stride_k     = left.stride(2)   # k‑stride = H

    # Tile sizes (tuned for the B200 – keep product ≤1024)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        B * hidden_dim,               # program_id(0): batch * hidden
        triton.cdiv(N, BLOCK_M),      # program_id(1): i tiles
        triton.cdiv(N, BLOCK_N),      # program_id(2): j tiles
    )

    tri_mul_einsum_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_batch, stride_row, stride_k,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion before moving on (optional, for correctness)
    # torch.cuda.synchronize()

    # -------------------------------
    # Rest of the pipeline (all fp32)
    # -------------------------------
    out_fp32 = out.to(torch.float32)

    out_fp32 = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=w("to_out_norm.weight"),
        bias=w("to_out_norm.bias"),
    )
    out_fp32 = out_fp32 * out_gate  # gating after to_out_norm

    # Final linear projection back to original dimension
    result = F.linear(out_fp32, w("to_out.weight"))  # shape [B,N,N,dim]

    return result
