"""
TriMul custom kernel (outgoing version).

The heavy O(N³·D) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is off‑loaded to a Triton kernel.  The kernel processes one (i, j) pair per
program instance and a tile of the hidden dimension `d`.  It iterates over the
k‑dimension in blocks, multiplies the two (K×D) tiles element‑wise and reduces
over K with `tl.sum`.  All other operations (LayerNorm, linear projections,
gating, final LayerNorm and linear) are performed with PyTorch for simplicity.

The launch grid is (seq_len, seq_len, B·ceil(D/BLOCK_D)).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------- #
# Triton kernel for the core contraction
# --------------------------------------------------------------------- #
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    # --------------------------------------------------------------- #
    # grid mapping
    #   pid_i : output row  i   (0 .. N-1)
    #   pid_j : output col  j   (0 .. N-1)
    #   pid_bd: fused batch index and hidden‑dim tile
    # --------------------------------------------------------------- #
    pid_i = tl.program_id(0)            # i
    pid_j = tl.program_id(1)            # j
    pid_bd = tl.program_id(2)           # batch * d‑tile

    num_d_tiles = tl.cdiv(D, BLOCK_D)
    b = pid_bd // num_d_tiles           # batch index
    d_tile = pid_bd % num_d_tiles        # which hidden‑dim tile

    i = pid_i
    j = pid_j
    d_off = d_tile * BLOCK_D

    # bounds
    mask_i = i < N
    mask_j = j < N
    offs_d = d_off + tl.arange(0, BLOCK_D)
    mask_d = offs_d < D

    # accumulator for the hidden‑dim tile
    acc = tl.zeros([BLOCK_D], dtype=tl.float32)

    # --------------------------------------------------------------- #
    # reduction over k
    # --------------------------------------------------------------- #
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i, k, d]   shape (BLOCK_K, BLOCK_D)
        left_ptrs = left_ptr \
            + b * stride_lb \
            + i * stride_li \
            + offs_k[:, None] * stride_lk \
            + offs_d[None, :] * stride_ld
        left = tl.load(left_ptrs,
                       mask=mask_i & mask_k[:, None] & mask_d[None, :],
                       other=0.0)

        # right[j, k, d]  shape (BLOCK_K, BLOCK_D)
        right_ptrs = right_ptr \
            + b * stride_rb \
            + j * stride_rj \
            + offs_k[:, None] * stride_rk \
            + offs_d[None, :] * stride_rd
        right = tl.load(right_ptrs,
                        mask=mask_j & mask_k[:, None] & mask_d[None, :],
                        other=0.0)

        # element‑wise product + reduction over k
        acc += tl.sum(left * right, axis=0)

    # --------------------------------------------------------------- #
    # store result
    # --------------------------------------------------------------- #
    out_ptrs = out_ptr \
        + b * stride_ob \
        + i * stride_oi \
        + j * stride_oj \
        + offs_d * stride_od
    tl.store(out_ptrs,
             acc,
             mask=mask_i & mask_j & mask_d)

# Host entry point
# --------------------------------------------------------------------- #
def custom_kernel(data):
    """
    TriMul forward pass (outgoing) with a Triton kernel for the heavy contraction.

    Args:
        data (Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]):
            - input_tensor: [B, N, N, dim]
            - mask_tensor : [B, N, N]  (may be all‑ones when no mask)
            - weights: dict with all linear / layer‑norm parameters
            - config: dict with "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------- #
    # unpack weights
    # --------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------- #
    # LayerNorm over the channel dimension (dim)
    # --------------------------------------------------------------- #
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # --------------------------------------------------------------- #
    # Linear projections (no bias)
    # --------------------------------------------------------------- #
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # --------------------------------------------------------------- #
    # Apply mask (if provided)
    # --------------------------------------------------------------- #
    if mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # -> [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------- #
    # Gating (sigmoid of linear)
    # --------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------- #
    # Triton contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------- #
    left = left.contiguous()
    right = right.contiguous()
    B, N, _, _ = left.shape
    out = torch.empty_like(left)

    # block sizes (tunable)
    BLOCK_K = 32
    BLOCK_D = 32

    # stride information (in elements)
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # launch grid
    grid = (
        N,                                           # i dimension
        N,                                           # j dimension
        B * ((hidden_dim + BLOCK_D - 1) // BLOCK_D)  # batch × hidden‑dim tiles
    )

    trimul_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        out_ptr=out.data_ptr(),
        B,
        N,
        hidden_dim,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------- #
    # Post‑processing: LayerNorm, out gating, final linear projection
    # --------------------------------------------------------------- #
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate
    out = F.linear(out, to_out_weight)   # back to [B, N, N, dim]

    return out
```
