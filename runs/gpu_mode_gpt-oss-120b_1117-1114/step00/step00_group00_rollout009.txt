"""
TriMul forward implementation with Triton acceleration.

The core operation is the "outgoing" triangle multiplicative update:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

For each hidden dimension `d` this is a matrix multiplication
    out_d = left_d @ right_d.T
where left_d and right_d have shape (N, N).

We fuse this N×N×N contraction across all hidden dimensions using a
Triton kernel that tiles the i, j and k axes (BLOCK_M, BLOCK_N, BLOCK_K)
and processes a small block of the hidden dimension (BLOCK_D) at a time.
All other operations (LayerNorms, Linear projections and gates) are kept
in PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication over the hidden dimension
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid0 = tl.program_id(0)   # combined batch * d-block
    pid_i = tl.program_id(1)  # i block
    pid_j = tl.program_id(2)  # j block

    # --------------------------------------------------------------
    # Decode batch index and hidden‑dim block
    # --------------------------------------------------------------
    total_d_blocks = (D + BLOCK_D - 1) // BLOCK_D
    b = pid0 // total_d_blocks
    d_block = pid0 % total_d_blocks
    d_start = d_block * BLOCK_D

    # --------------------------------------------------------------
    # Offsets for i and j tiles
    # --------------------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Offsets for hidden dimension block
    # --------------------------------------------------------------
    d_range = d_start + tl.arange(0, BLOCK_D)
    mask_d = d_range < D

    # accumulator for the output tile (M, N, D)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # number of k‑tiles
    num_k_blocks = tl.cdiv(N, BLOCK_K)

    # --------------------------------------------------------------
    # Main k‑loop: accumulate left * right^T
    # --------------------------------------------------------------
    for k_iter in range(0, num_k_blocks):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # --------------------
        # Load left block: shape (M, K, D)
        # --------------------
        left_ptrs = left_ptr \
            + b * stride_left_b \
            + offs_i[:, None, None] * stride_left_i \
            + offs_k[None, :, None] * stride_left_k \
            + d_range[None, None, :] * stride_left_d
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )  # (M, K, D)

        # --------------------
        # Load right block transposed: shape (K, N, D)
        # (right is indexed as (b, j, k, d) in the original tensor;
        #  we load it as (k, j, d) to match tl.dot expectations)
        # --------------------
        right_ptrs = right_ptr \
            + b * stride_right_b \
            + offs_k[:, None, None] * stride_right_k \
            + offs_j[None, :, None] * stride_right_i \
            + d_range[None, None, :] * stride_right_d
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )  # (K, N, D)

        # --------------------
        # Multiply‑accumulate for each hidden dimension slice
        # --------------------
        for d in range(0, BLOCK_D):
            # The mask ensures out‑of‑range d’s stay zero
            acc[:, :, d] += tl.dot(left[:, :, d], right[:, :, d])

    # --------------------------------------------------------------
    # Store the accumulated tile into the output tensor
    # --------------------------------------------------------------
    out_ptrs = out_ptr \
        + b * stride_out_b \
        + offs_i[:, None, None] * stride_out_i \
        + offs_j[None, :, None] * stride_out_j \
        + d_range[None, None, :] * stride_out_d
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )


# ----------------------------------------------------------------------
# Host‑side wrapper
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the TriMul module with Triton‑accelerated
    triangle multiplicative update.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] float32
            - mask:       [B, N, N] (optional, may be all ones)
            - weights:    dict of model parameters
            - config:    dict containing "dim", "hidden_dim", "nomask"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Unpack weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm on the input tensor
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)          # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)        # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Apply mask if required
    # ------------------------------------------------------------------
    if not nomask:
        mask_unsq = mask.unsqueeze(-1).to(dtype)
        left = left * mask_unsq
        right = right * mask_unsq

    # ------------------------------------------------------------------
    # Gates
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triangular multiplicative update via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # Ensure contiguous layout for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # Strides in element units
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # Block sizes (tuned for typical GPUs)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 8   # process 8 hidden channels per program

    num_d_blocks = (hidden_dim + BLOCK_D - 1) // BLOCK_D
    grid = (
        B * num_d_blocks,                     # batch * d‑block dimension
        triton.cdiv(N, BLOCK_M),              # i‑tiles
        triton.cdiv(N, BLOCK_N),              # j‑tiles
    )

    # Launch kernel
    tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_i, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )
    # Ensure kernel completion before next GPU ops
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # Output processing: LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
