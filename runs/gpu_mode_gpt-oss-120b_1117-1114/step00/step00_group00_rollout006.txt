@triton.jit
def tri_mul_kernel(A_ptr, B_ptr, C_ptr,
                   B, N, H,
                   stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
                   stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
                   stride_c_batch, stride_c_i, stride_c_k, stride_c_h,
                   BLOCK_M: tl.constexpr,
                   BLOCK_N: tl.constexpr,
                   BLOCK_K: tl.constexpr):
    pid_x = tl.program_id(0)  # i block
    pid_y = tl.program_id(1)  # j block
    pid_z = tl.program_id(2)  # batch*hidden

    # compute base offsets
    b = pid_z // H  # b in [0,B-1]
    h = pid_z % H   # hidden index

    # compute i and j offsets
    offs_i = pid_x * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_y * BLOCK_N + tl.arange(0, BLOCK_N)

    # check boundaries
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointer for current batch+hidden
    a_ptrs = A_ptr + b * stride_a_batch + h * stride_a_h
    b_ptrs = B_ptr + b * stride_b_batch + h * stride_b_h
    c_ptrs = C_ptr + b * stride_c_batch + h * stride_c_h

    # initialise accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A block shape (BLOCK_M, BLOCK_K) using strides i and k
        a_offs = (offs_i[:, None] * stride_a_i) + (offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs + a_offs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # load B_T block shape (BLOCK_K, BLOCK_N) using strides k (original stride_k) and j (stride_b_i)
        # Since B_T has dimensions (K, N) where K corresponds to original k dimension, and N corresponds to original j dimension
        b_offs = (offs_k[:, None] * stride_b_k) + (offs_j[None, :] * stride_b_i)

        # matmul

    c_offs = (offs_i[:, None] * stride_c_i) + (offs_j[None, :] * stride_c_k) # note: output's second N dimension stride is stride_c_k (same as stride of original k)
    tl.store(c_ptrs + c_offs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # LayerNorm on dim
    eps = 1e-5
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask
    mask_exp = mask.unsqueeze(-1)
    left = left * mask_exp
    right = right * mask_exp

    # Gating
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Output tensor placeholder
    out_h = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Get strides
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left_h.stride()
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h = right_h.stride()
    stride_c_batch, stride_c_i, stride_c_k, stride_c_h = out_h.stride()

    # Define block sizes, maybe adapt to N.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Adjust block sizes if N is smaller than BLOCK size
    BLOCK_M = min(BLOCK_M, N)
    BLOCK_N = min(BLOCK_N, N)

"""
TriMul forward implementation using Triton.

The heavy part of the TriMul operation is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where left and right are obtained from linear projections and
gates on the normalized input. This is a batched matrix multiplication
over the sequence dimension N for each hidden channel d.  The kernel
computes, for every (batch, hidden) pair, a GEMM:
    (N×N) @ (N×N)^T  -> (N×N)
with half‑precision inputs and float32 accumulation.  The rest of the
TriMul pipeline (LayerNorms, linear projections, gating) is performed
in PyTorch.

The kernel tiles the output with BLOCK_M × BLOCK_N tiles and iterates
over the K dimension in BLOCK_K steps.  The batch‑hidden dimension is
handled via the third program ID, mapping pid_z to (batch, hidden).

The function `custom_kernel` follows the required signature and returns
the final tensor of shape [B, N, N, dim] in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to left, right, out
    B, N, H,                           # batch, seq_len, hidden_dim
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_k, stride_c_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out[b, i, j, h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    left (A) layout : (B, N, N, H) -> indices (b,i,k,h)
    right (B) layout: (B, N, N, H) -> indices (b,j,k,h)
    out  (C) layout : (B, N, N, H) -> indices (b,i,j,h)
    """
    pid_i = tl.program_id(0)  # tile over i (row)
    pid_j = tl.program_id(1)  # tile over j (col)
    pid_z = tl.program_id(2)  # batch * hidden

    # Decode batch and hidden indices from pid_z
    b = pid_z // H
    h = pid_z % H

    # Offsets for the i and j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (batch, hidden) slice
    a_base = a_ptr + b * stride_a_batch + h * stride_a_h
    b_base = b_ptr + b * stride_b_batch + h * stride_b_h
    c_base = c_ptr + b * stride_c_batch + h * stride_c_h

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block (M x K)
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # Load B block transposed (K x N) -> we need right[b, j, k, h]
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float16)

        # Matrix multiply-accumulate
        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_k
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul module with a Triton kernel for the
    O(N³) contraction.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]
            - mask         : torch.Tensor [B, N, N] (0/1)
            - weights      : dict of weight tensors
            - config       : dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) LayerNorm on channel dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply mask (if any)
    # ------------------------------------------------------------------
    mask_exp = mask.unsqueeze(-1).to(x.dtype)  # shape [B, N, N, 1]
    left = left * mask_exp
    right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to half precision for the heavy contraction
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # Allocate output in half precision
    out_h = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Strides (int64)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left_h.stride()
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h = right_h.stride()
    stride_c_batch, stride_c_i, stride_c_k, stride_c_h = out_h.stride()

    # ------------------------------------------------------------------
    # 6) Triton kernel launch
    # ------------------------------------------------------------------
    # Choose reasonable block sizes; shrink if N is smaller.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_M = min(BLOCK_M, N)
    BLOCK_N = min(BLOCK_N, N)
    BLOCK_K = min(BLOCK_K, N)

    grid = (
        triton.cdiv(N, BLOCK_M),   # i dimension
        triton.cdiv(N, BLOCK_N),   # j dimension
        B * hidden_dim,            # batch * hidden
    )

    tri_mul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        hidden_dim,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_a_h,
        stride_b_batch,
        stride_b_i,
        stride_b_k,
        stride_b_h,
        stride_c_batch,
        stride_c_i,
        stride_c_k,
        stride_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7) Convert back to float32 and finish with LayerNorm, gate, linear
    # ------------------------------------------------------------------
    out = out_h.to(torch.float32)

    # LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
