        right = right * mask_exp

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()

def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)

"""
TriMul (outgoing) forward pass with a Triton kernel for the core
pairwise multiplication.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ ℝ^{B×N×N×C}.
2. Linear projections → left, right ∈ ℝ^{B×N×N×H} (H = hidden_dim).
3. Optional mask (broadcast over the hidden dimension).
4. Gating:
       left  ← left  * sigmoid(linear(x, left_gate_weight))
       right ← right * sigmoid(linear(x, right_gate_weight))
       out_gate ← sigmoid(linear(x, out_gate_weight))
5. Core update:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication left @ rightᵀ for each hidden
   channel.  We flatten the (batch, hidden) dimensions and compute the
   multiplication with a Triton GEMM‑style kernel.
6. Layer‑norm over the hidden dimension, multiply by out_gate and a final
   linear projection back to C.

The heavy O(N³·H) work is performed inside the Triton kernel; the surrounding
operations use standard PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# -----------------------------------------------------------
# Triton kernel: out = left @ rightᵀ  (batched for each channel)
# -----------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # *float32  [batch_total, M, K]
    right_ptr,         # *float32  [batch_total, N, K] (K is the reduction dimension)
    out_ptr,           # *float32  [batch_total, M, N]
    M, N, K,           # int32     matrix sizes (M = N = seqlen, K = seqlen)
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_n, stride_right_k,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out = left @ rightᵀ for each batch (flattened batch*hidden).
    left : (M, K)
    right: (N, K) – we read it transposed as (K, N) for the GEMM.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load left tile  (M x K)
        left_ptrs = (
            left_ptr
            + batch_id * stride_left_batch
            + offs_m[:, None] * stride_left_m
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load right tile transposed (K x N)
        # right is stored as (N, K); we need (K, N)
        right_ptrs = (
            right_ptr
            + batch_id * stride_right_batch
            + offs_k[:, None] * stride_right_k
            + offs_n[None, :] * stride_right_n
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate GEMM
        acc += tl.dot(left, right, out_dtype=tl.float32)

    # Write back
    out_ptrs = (
        out_ptr
        + batch_id * stride_out_batch
        + offs_m[:, None] * stride_out_m
        + offs_n[None, :] * stride_out_n
    )
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# -----------------------------------------------------------
# Entry point
# -----------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator with a Triton kernel for the
    O(N³·H) pairwise multiplication.
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )  # shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(dtype)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core triangular multiplication via Triton
    #    out = Σ_k left[...,k,:] * right[...,k,:]   (batched GEMM)
    # ------------------------------------------------------------------
    # Rearrange to [B, hidden_dim, N, N] and flatten batch*hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_flat = left_perm.view(-1, N, N)   # [B*H, N, N]  (M = N, K = N)
    right_flat = right_perm.view(-1, N, N)  # same layout

    # Output placeholder
    out_flat = torch.empty_like(left_flat)  # [B*H, N, N]

    # Strides (int64)
    stride_left = left_flat.stride()
    stride_right = right_flat.stride()
    stride_out = out_flat.stride()
    stride_left_batch, stride_left_m, stride_left_k = stride_left
    stride_right_batch, stride_right_n, stride_right_k = stride_right
    stride_out_batch, stride_out_m, stride_out_n = stride_out

    # Block sizes – tuned for NVIDIA H100 (B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_batches = left_flat.shape[0]  # B * hidden_dim

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,               # M
        N,               # N
        N,               # K
        stride_left_batch,
        stride_left_m,
        stride_left_k,
        stride_right_batch,
        stride_right_n,
        stride_right_k,
        stride_out_batch,
        stride_out_m,
        stride_out_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias
    )
    out = out * out_gate  # broadcast over hidden_dim

    to_out_weight = weights["to_out.weight"]   # shape [dim, hidden_dim]
    output = torch.nn.functional.linear(out, to_out_weight)  # [B,N,N,dim]

    return output
