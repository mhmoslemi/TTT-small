"""
TriMul forward (outgoing) implementation with a Triton kernel for the
mask‑gated element‑wise multiplication. All linear projections, layer‑norms,
sigmoids and the final batched matrix multiplication are performed with
PyTorch (cuBLAS) for speed, while the element‑wise gating+masking is fused in
Triton.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: out = left * left_gate * mask ,  right_out = right * right_gate * mask
# ----------------------------------------------------------------------
@triton.jit
def gated_mask_kernel(
    left_ptr,                # const float*  (B,N,N,H)
    left_gate_ptr,           # const float*  (B,N,N,H)
    right_ptr,               # const float*  (B,N,N,H)
    right_gate_ptr,          # const float*  (B,N,N,H)
    mask_ptr,                # const float*  (B,N,N)  (0/1 values)
    out_left_ptr,            #       float*  (B,N,N,H)  result
    out_right_ptr,           #       float*  (B,N,N,H)  result
    B, N, H,                 # int32 scalars
    # strides for left
    stride_left_B, stride_left_I, stride_left_J, stride_left_H,
    # strides for left_gate
    stride_lg_B, stride_lg_I, stride_lg_J, stride_lg_H,
    # strides for right
    stride_right_B, stride_right_I, stride_right_J, stride_right_H,
    # strides for right_gate
    stride_rg_B, stride_rg_I, stride_rg_J, stride_rg_H,
    # strides for mask (no H dim)
    stride_mask_B, stride_mask_I, stride_mask_J,
    # strides for outputs
    stride_outL_B, stride_outL_I, stride_outL_J, stride_outL_H,
    stride_outR_B, stride_outR_I, stride_outR_J, stride_outR_H,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    m = offs < total

    # ---- decode linear index into (b,i,j,h) ----
    # flat = ((b * N + i) * N + j) * H + h
    bn = N * N * H
    b = offs // bn
    tmp = offs % bn
    nij = N * H
    i = tmp // nij
    tmp = tmp % nij
    j = tmp // H
    h = tmp % H

    # ---- compute pointer offsets ----
    left_off = b * stride_left_B + i * stride_left_I + j * stride_left_J + h * stride_left_H
    lg_off   = b * stride_lg_B   + i * stride_lg_I   + j * stride_lg_J   + h * stride_lg_H
    right_off= b * stride_right_B+ i * stride_right_I+ j * stride_right_J+ h * stride_right_H
    rg_off   = b * stride_rg_B   + i * stride_rg_I   + j * stride_rg_J   + h * stride_rg_H
    mask_off = b * stride_mask_B + i * stride_mask_I + j * stride_mask_J

    outL_off = b * stride_outL_B + i * stride_outL_I + j * stride_outL_J + h * stride_outL_H
    outR_off = b * stride_outR_B + i * stride_outR_I + j * stride_outR_J + h * stride_outR_H

    # ---- load ----
    left_val = tl.load(left_ptr + left_off, mask=m, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + lg_off, mask=m, other=0.0)
    right_val = tl.load(right_ptr + right_off, mask=m, other=0.0)
    right_gate_val = tl.load(right_gate_ptr + rg_off, mask=m, other=0.0)
    mask_val = tl.load(mask_ptr + mask_off, mask=m, other=1.0)

    # ---- compute ----
    out_left = left_val * left_gate_val * mask_val
    out_right = right_val * right_gate_val * mask_val

    # ---- store ----
    tl.store(out_left_ptr + outL_off, out_left, mask=m)
    tl.store(out_right_ptr + outR_off, out_right, mask=m)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.
    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # unpack config and weights
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # weights (all expected to be torch tensors on the same device)
    w_norm = weights["norm.weight"]
    b_norm = weights["norm.bias"]
    w_left_proj = weights["left_proj.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]
    w_to_out_norm = weights["to_out_norm.weight"]
    b_to_out_norm = weights["to_out_norm.bias"]
    w_to_out = weights["to_out.weight"]

    bs, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # LayerNorm on the input (over the last dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=w_norm,
        bias=b_norm,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # Linear projections and gates (still in FP32)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, w_left_proj)        # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, w_right_proj)

    left_gate = torch.nn.functional.linear(x, w_left_gate).sigmoid()
    right_gate = torch.nn.functional.linear(x, w_right_gate).sigmoid()
    out_gate = torch.nn.functional.linear(x, w_out_gate).sigmoid()   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # Prepare mask (broadcasted to hidden dim)
    # ------------------------------------------------------------------
    if nomask:
        mask_tensor = torch.ones((bs, N, N), dtype=x.dtype, device=device)
    else:
        mask_tensor = mask.to(dtype=x.dtype, device=device)

    # ------------------------------------------------------------------
    # Fuse mask + gate multiplication with a Triton kernel
    # ------------------------------------------------------------------
    gated_left = torch.empty_like(left)
    gated_right = torch.empty_like(right)

    # compute strides (in element units)
    sL = left.stride()          # (B, I, J, H)
    sLG = left_gate.stride()
    sR = right.stride()
    sRG = right_gate.stride()
    sM = mask_tensor.stride()
    sOutL = gated_left.stride()
    sOutR = gated_right.stride()

    BLOCK = 256
    total_elements = bs * N * N * hidden_dim
    grid = (triton.cdiv(total_elements, BLOCK),)

    gated_mask_kernel[grid](
        left, left_gate, right, right_gate,
        mask_tensor,
        gated_left, gated_right,
        bs, N, hidden_dim,
        # left strides
        sL[0], sL[1], sL[2], sL[3],
        # left_gate strides
        sLG[0], sLG[1], sLG[2], sLG[3],
        # right strides
        sR[0], sR[1], sR[2], sR[3],
        # right_gate strides
        sRG[0], sRG[1], sRG[2], sRG[3],
        # mask strides
        sM[0], sM[1], sM[2],
        # out_left strides
        sOutL[0], sOutL[1], sOutL[2], sOutL[3],
        # out_right strides
        sOutR[0], sOutR[1], sOutR[2], sOutR[3],
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # Core contraction: out[b,i,j,h] = sum_k gated_left[b,i,k,h] * gated_right[b,j,k,h]
    # This is a batched matrix multiplication per hidden channel.
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_mat = gated_left.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)
    right_mat = gated_right.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)

    # batched matmul (left * right^T)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # Output layernorm + output gate + final linear projection
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=1e-5,
    )
    out_gated = out_norm * out_gate
    output = torch.nn.functional.linear(out_gated, w_to_out)  # (B,N,N,dim)

    return output
