def layer_norm(x, weight, bias, eps=1e-5):
    # x: [..., C]
    shape = x.shape
    # flatten leading dims
    x_flat = x.view(-1, x.shape[-1])
    # compute mean and var across last dim
    mean = x_flat.mean(dim=1, keepdim=True)
    var = x_flat.var(dim=1, unbiased=False, keepdim=True)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def apply_mask_kernel(x_ptr, mask_ptr, out_ptr, total_elems, H: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems
    # compute mask index, safe for out-of-range
    mask_idx = offs // H
    mask_idx = tl.where(mask, mask_idx, 0)
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    tl.store(out_ptr + offs, x * m, mask=mask)

def apply_mask(x, mask):
    # x: (B,N,N,H) float32; mask: (B,N,N) float32 (broadcast)
    if mask is None:
        return x
    x = x.contiguous()
    mask = mask.contiguous()
    out = torch.empty_like(x)
    total = x.numel()
    hidden = x.shape[-1]
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    apply_mask_kernel[grid](x, mask, out, total, H=hidden, BLOCK=BLOCK)
    return out

def custom_kernel(data):
    """
    Triton-accelerated implementation of the outgoing Triangle Multiplicative
    Update (TriMul) forward pass. The only part offloaded to Triton is the
    mask‑multiplication of the projected tensors; the heavy O(N³) contraction
    is performed using a batched GEMM (torch.matmul) which already maps to
    cuBLAS. All other linear layers, gates and LayerNorms are computed with
    PyTorch for simplicity.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # LayerNorm on input (dim)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # Linear projections (dim -> hidden_dim)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask (broadcast over hidden dimension) via Triton
    # mask shape: [B, N, N]
    if mask is not None:
        mask_f = mask.to(x.dtype)
        left = apply_mask(left, mask_f)
        right = apply_mask(right, mask_f)
    else:
        # no mask: keep tensors as is
        left = left
        right = right

    # Gating values (dim -> hidden_dim)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Einsum contraction – implemented as batched matrix multiplication:
    # left, right : (B, N, N, hidden_dim)
    # reshape to (B, hidden_dim, N, N)
    left_b = left.permute(0, 3, 1, 2)      # (B, hidden, N, N)
    right_b = right.permute(0, 3, 1, 2)    # (B, hidden, N, N)

    # Compute (L_d @ R_d^T) for each hidden channel via torch.matmul
    out_b = torch.matmul(left_b, right_b.transpose(-1, -2))  # (B, hidden, N, N)

    # Back to (B, N, N, hidden)
    out = out_b.permute(0, 2, 3, 1)

    # Output LayerNorm (hidden_dim)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"])

    return out
```

"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Only the mask‑multiplication of the projected tensors is offloaded to Triton;
the O(N³) contraction is performed with a batched GEMM (torch.matmul) which
already maps to cuBLAS. All linear layers, gates and LayerNorms are computed
with PyTorch for simplicity while preserving exact semantics.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def apply_mask_kernel(x_ptr, mask_ptr, out_ptr, total_elems, H: tl.constexpr, BLOCK: tl.constexpr):
    """Broadcast a [B, N, N] mask over the hidden dimension H.

    Args:
        x_ptr   : pointer to the flattened tensor of shape (B, N, N, H)
        mask_ptr: pointer to the flattened mask of shape (B, N, N)
        out_ptr : pointer to the output tensor (same shape as x)
        total_elems : total number of elements in x (B*N*N*H)
        H       : hidden dimension (compile‑time constant)
        BLOCK   : block size (compile‑time constant)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # safe index into the mask (avoid OOB when mask is false)
    mask_idx = offs // H
    mask_idx = tl.where(mask, mask_idx, 0)

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    tl.store(out_ptr + offs, x * m, mask=mask)


def _apply_mask(x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """Apply a broadcast mask to a 4‑D tensor using Triton."""
    if mask is None:
        return x
    x = x.contiguous()
    mask = mask.contiguous()
    out = torch.empty_like(x)
    total = x.numel()
    hidden = x.shape[-1]

    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    apply_mask_kernel[grid](
        x,
        mask,
        out,
        total,
        H=hidden,
        BLOCK=BLOCK,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask        : torch.Tensor of shape [B, N, N] or None
            - weights     : dict of model weight tensors
            - config      : dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Input layer‑norm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden_dim
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"]) # (B,N,N,hidden)

    # -----------------------------------------------------------------
    # 3. Apply mask (broadcast over hidden) via Triton
    # -----------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(x.dtype)                     # ensure same dtype
        left = _apply_mask(left, mask_f)
        right = _apply_mask(right, mask_f)

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid of linear transforms)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM for each hidden channel.
    # -----------------------------------------------------------------
    # reshape to (B, hidden, N, N) so that hidden becomes part of the batch
    left_b = left.permute(0, 3, 1, 2)   # (B, hidden, N, N)
    right_b = right.permute(0, 3, 1, 2) # (B, hidden, N, N)

    # left @ right^T  -> (B, hidden, N, N)
    out_b = torch.matmul(left_b, right_b.transpose(-1, -2))

    # back to (B, N, N, hidden)
    out = out_b.permute(0, 2, 3, 1)

    # -----------------------------------------------------------------
    # 6. Output layer‑norm on hidden dimension
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 7. Apply output gate
    # -----------------------------------------------------------------
    out = out * out_gate

    # -----------------------------------------------------------------
    # 8. Final linear projection back to dim
    # -----------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
