@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_m = tl.program_id(0)   # i block index
    pid_n = tl.program_id(1)   # j block index
    pid_z = tl.program_id(2)   # determines b and h

    # compute block start indices
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # compute batch and hidden indices
    b = pid_z // H
    h = pid_z % H

    # Compute pointers offsets
    # offset: b*stride_left_b + i*stride_left_i + k*stride_left_k + h*stride_left_h

    # Initiate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K

        # Load A tile (left)
        offs_i = i_start + tl.arange(0, BLOCK_M)
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        # mask for overflow
        mask_i = offs_i < N
        mask_k = offs_k < N

        a_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (right) but we need element (j,k) not (k,j). We need B[j][k] (j as col index)
        offs_j = j_start + tl.arange(0, BLOCK_N)
        # For B we load at indices (b, j, k, h)
        b_ptrs = right_ptr + b*stride_right_b + h*stride_right_h + offs_j[:, None] * stride_right_j + offs_k[None, :] * stride_right_k
        b = tl.load(b_ptrs, mask=(offs_j[:, None] < N) & mask_k[None, :], other=0.0)

        # multiply-accumulate
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)

    # Write back to out
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N
    out_ptrs = out_ptr + b*stride_out_b + h*stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; eps = 1e-5

    # Compute layernorm over dim
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projection
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Mask
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1).type_as(left_proj) # broadcast
        left = left_proj * mask_unsq
        right = right_proj * mask_unsq
    else:
        left = left_proj
        right = right_proj

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # allocate out
    B, N, _, _ = input_tensor.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=left.dtype, device=left.device)

    # kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64  # maybe 32
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * hidden_dim)
    # strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()
    tri_mul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
```

"""
Optimized Triton implementation of the "outgoing" TriMul operator.
The heavy O(N^3) reduction (sum over k) is computed by a fused
batched GEMM kernel where each (batch, hidden) pair performs a
matrix multiplication of shape (N, N) with the k‑dimension summed.
The remaining linear, LayerNorm and gating operations are handled
by PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # i‑block
    pid_n = tl.program_id(1)  # j‑block
    pid_z = tl.program_id(2)  # combined (batch, hidden)

    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # decode batch and hidden index
    mat_id = pid_z
    b = mat_id // H
    h = mat_id % H

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K

        #--- load A (left) tile: shape [M, K] ---
        offs_i = i_start + tl.arange(0, BLOCK_M)
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_i = offs_i < N
        mask_k = offs_k < N
        a_ptrs = left_ptr + \
                 b * stride_left_b + \
                 h * stride_left_h + \
                 offs_i[:, None] * stride_left_i + \
                 offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float32)

        #--- load B (right) tile transposed: shape [K, N] ---
        offs_j = j_start + tl.arange(0, BLOCK_N)
        mask_j = offs_j < N
        b_ptrs = right_ptr + \
                 b * stride_right_b + \
                 h * stride_right_h + \
                 offs_k[:, None] * stride_right_k + \
                 offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0,
                    dtype=tl.float32)

        # accumulate
        acc += tl.dot(a, b)   # (M, K) dot (K, N) -> (M, N)

    # write back
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N
    out_ptrs = out_ptr + \
               b * stride_out_b + \
               h * stride_out_h + \
               offs_i[:, None] * stride_out_i + \
               offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """Forward pass of the outgoing TriMul operator with a fused Triton kernel.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: Tensor[B, N, N, dim]
            - mask: Tensor[B, N, N] (bool / float)
            - weights: dict of model parameters (torch tensors on the same device)
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        Tensor[B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------- LayerNorm over dim -------------------
    # input_tensor shape: [B, N, N, dim]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # Ensure computation in float32 for numerical stability
    x = x.to(torch.float32)

    # ------------------- Linear projections (no bias) -------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ------------------- Apply mask (if any) -------------------
    # mask is assumed broadcastable to [B, N, N, 1]
    mask_ = mask.to(x.dtype).unsqueeze(-1)
    left = left_proj * mask_
    right = right_proj * mask_

    # ------------------- Gating -------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------- Triton kernel for the O(N^3) reduction -------------------
    B, N, _, _ = left.shape
    # allocate output tensor
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # block sizes (tuned for typical B200 GPU)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # strides for each tensor (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------- Output processing -------------------
    # Layer norm over hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
```

"""
Optimized Triton implementation of the outgoing TriMul operator.
The computationally intensive O(N³) reduction (sum over k) is
performed by a fused batched GEMM kernel: for each (batch, hidden)
pair we compute a matrix multiplication of shape (N, N) where the
k‑dimension is summed. All other operations (LayerNorm, linear
projections, gating) are executed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,               # const float*  left  [B, N, N, H]
    right_ptr,              # const float*  right [B, N, N, H]
    out_ptr,                #       float*  out   [B, N, N, H]
    B, N, H,                # dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # i‑block index
    pid_n = tl.program_id(1)          # j‑block index
    pid_z = tl.program_id(2)          # combined (batch, hidden) index

    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # decode batch and hidden indices
    mat_id = pid_z
    b = mat_id // H
    h = mat_id % H

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K

        # ---- load A tile (left) : shape [M, K] ----
        offs_i = i_start + tl.arange(0, BLOCK_M)
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_i = offs_i < N
        mask_k = offs_k < N
        a_ptrs = left_ptr + \
                 b * stride_left_b + \
                 h * stride_left_h + \
                 offs_i[:, None] * stride_left_i + \
                 offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float32)

        # ---- load B tile (right) transposed : shape [K, N] ----
        offs_j = j_start + tl.arange(0, BLOCK_N)
        mask_j = offs_j < N
        b_ptrs = right_ptr + \
                 b * stride_right_b + \
                 h * stride_right_h + \
                 offs_k[:, None] * stride_right_k + \
                 offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0,
                    dtype=tl.float32)

        # accumulate the product
        acc += tl.dot(a, b)   # (M, K) @ (K, N) -> (M, N)

    # ---- write result ----
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N
    out_ptrs = out_ptr + \
               b * stride_out_b + \
               h * stride_out_h + \
               offs_i[:, None] * stride_out_i + \
               offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: Tensor[B, N, N, dim]
            - mask: Tensor[B, N, N] (bool / int / float)
            - weights: dict of torch.Tensors containing all model weights
            - config: dict with keys "dim" and "hidden_dim"
    Returns:
        Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---- LayerNorm over the last dim ----
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x = x.to(torch.float32)

    # ---- Linear projections (no bias) ----
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ---- Apply mask (if any) ----
    mask_ = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
    left = left_proj * mask_
    right = right_proj * mask_

    # ---- Gating ----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---- Triton kernel for the heavy O(N³) sum ----
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # block sizes (tuned for typical NVIDIA B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # strides in elements
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---- Output processing ----
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
