Define Triton kernel for contraction:

```
@triton.jit
def trimul_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    b_h = tl.program_id(2) # combined batch_head index
    batch = b_h // H
    head = b_h % H

    # offsets for this batch and head
    left_batch_head_offset = batch * stride_lb + head * stride_lh
    right_batch_head_offset = batch * stride_rb + head * stride_rh
    out_batch_head_offset = batch * stride_ob + head * stride_oh

    # tile indices
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    # coordinate offsets
    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)

    # mask for i, j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile (i,k)
        left_ptrs = left_ptr + left_batch_head_offset + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right tile (j,k). We need shape (BLOCK_J, BLOCK_K).
        right_ptrs = right_ptr + right_batch_head_offset + offs_j[:, None] * stride_rj + offs_k[None, :] * stride_rk
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # compute dot product over k dimension
        # left shape (BLOCK_I, BLOCK_K), right shape (BLOCK_J, BLOCK_K). We need right transposed.
        acc += tl.dot(left, right, trans_b=True)  # result (BLOCK_I, BLOCK_J)

    # write back
    out_ptrs = out_ptr + out_batch_head_offset + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) forward implementation with a Triton kernel for the
pairwise contraction:

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

The kernel computes a batched matrix multiplication per head using block
tiling (i,j,k). All other operations (LayerNorm, Linear projections,
gates, final LayerNorm and output projection) are performed with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -----------------------------------------------------------
# Triton kernel: batched contraction over the sequence dimension
# -----------------------------------------------------------
@triton.jit
def trimul_contract_kernel(
    left_ptr,               # *float32
    right_ptr,              # *float32
    out_ptr,                # *float32
    B, N, H,                # dimensions
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_i = tl.program_id(0)          # tile over i dimension
    pid_j = tl.program_id(1)          # tile over j dimension
    pid_hb = tl.program_id(2)         # combined batch * head index

    # decode batch and head
    head = pid_hb % H
    batch = pid_hb // H

    # offsets for this tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base offsets that include batch and head
    left_base  = batch * stride_lb + head * stride_lh
    right_base = batch * stride_rb + head * stride_rh
    out_base   = batch * stride_ob + head * stride_oh

    # accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k dimension in blocks
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_I, BLOCK_K)
        left_ptrs = left_ptr + left_base + \
                    offs_i[:, None] * stride_li + \
                    offs_k[None, :] * stride_lk
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile: shape (BLOCK_J, BLOCK_K)
        right_ptrs = right_ptr + right_base + \
                     offs_j[:, None] * stride_rj + \
                     offs_k[None, :] * stride_rk
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0)

        # accumulate dot product over k (right is transposed)
        # left @ right.T -> (BLOCK_I, BLOCK_J)
        acc += tl.dot(left, right, trans_b=True)

    # write the result
    out_ptrs = out_ptr + out_base + \
               offs_i[:, None] * stride_oi + \
               offs_j[None, :] * stride_oj
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# -----------------------------------------------------------
# Entry point
# -----------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        output: Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]  # = H

    # ------------------------------------------------------------------
    # LayerNorm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Optional mask (binary 0/1). If nomask, mask will be all ones.
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(x.shape[:3], dtype=x.dtype, device=device)
    mask = mask.unsqueeze(-1)  # (B,N,N,1) for broadcasting
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # Gates
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton kernel for the contraction over the sequence dimension
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape  # batch, seq_len, seq_len, dim
    # Output tensor (float32)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (in number of elements)
    l_str = left.stride()   # (b, i, k, h)
    r_str = right.stride()
    o_str = out.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
        B * hidden_dim,   # batch * head
    )

    trimul_contract_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        l_str[0], l_str[1], l_str[2], l_str[3],
        r_str[0], r_str[1], r_str[2], r_str[3],
        o_str[0], o_str[1], o_str[2], o_str[3],
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Post‑contraction processing
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # LayerNorm over hidden_dim
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to dim
    out = F.linear(out, to_out_weight)

    return out
