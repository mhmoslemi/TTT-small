@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_batch = tl.program_id(0) # batch index
    pid_m = tl.program_id(1) # tile row
    pid_n = tl.program_id(2) # tile col
    
    # compute block start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask out-of-range
    mask_m = offs_m < M
    mask_n = offs_n < N
    
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # pointer for a and b for this batch
    a_batch_ptr = a_ptr + pid_batch * stride_ab
    b_batch_ptr = b_ptr + pid_batch * stride_bb
    
    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K
        
        # Compute pointers
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        
        # Load
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        
        # dot
        acc += tl.dot(a, b)
        
        # increment pointers? Actually a_ptrs and b_ptrs already incorporate offsets for each iteration.
        # if BLOCK_K loop increments we need to update base pointer? We just compute new pointer each iteration using offs_k.
    # end loop
    
    # write back
    c_ptrs = c_ptr + pid_batch * stride_cb + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    eps = 1e-5  # for layernorm
    # LayerNorm: compute mean, var across dim (last axis)
    # x_norm = (x - mean) / sqrt(var + eps) * weight + bias
    # x shape [B,N,N,dim]
    # compute mean over last dim: x.mean(-1, keepdim=True)
    # var similarly.
    # Use float32 for numeric.
    x = input_tensor
    # ensure float32
    x = x.to(torch.float32)
    # layernorm
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # Projections
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # mask
    if mask is None:
        mask_tensor = torch.ones_like(input_tensor[...,0])  # shape [B,N,N]
    else:
        mask_tensor = mask
    mask_tensor = mask_tensor.unsqueeze(-1)  # shape [B,N,N,1]
    left = left_proj * mask_tensor
    right = right_proj * mask_tensor

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Permute for batched matmul
    # left: [B,N,N,hidden_dim] -> permute to [B, hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # right_T = right_perm.transpose(-2,-1)
    right_T = right_perm.transpose(-2, -1).contiguous()  # shape [B, hidden_dim, N, N]

    # reshape to [batch*hidden_dim, N, N]
    B, H, N_f, N_f2 = left_perm.shape
    assert N_f == N and N_f2 == N
    batch_head = B * H
    left_mat = left_perm.view(batch_head, N, N)
    right_mat = right_T.view(batch_head, N, N)

    # cast to float16
    left_fp16 = left_mat.to(torch.float16)
    right_fp16 = right_mat.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)

    # get strides
    stride_ah, stride_am, stride_ak = left_fp16.stride()  # (batch*H, N, ?). Actually shape [B*H, M, K] => batch stride = M*K; M stride = K; K stride = 1.
    # But calling left_fp16.stride() returns tuple of length 3: (stride0, stride1, stride2)
    # M dimension is dim0? Actually shape is [batch, M, K] with M,N dims. So stride0 = M*K, stride1 = K, stride2 = 1.

    # So assign:
    stride_ab = left_fp16.stride(0)  # batch stride
    stride_am = left_fp16.stride(1)  # row stride (M)
    stride_ak = left_fp16.stride(2)  # col stride (K)

    # For right: shape [batch, K, N]
    # Strides: batch stride = K*N, stride_bk = N, stride_bn = 1
    stride_bb = right_fp16.stride(0)
    stride_bk = right_fp16.stride(1)
    stride_bn = right_fp16.stride(2)

    # For out: shape [batch, M, N]
    stride_cb = out_fp16.stride(0)
    stride_cm = out_fp16.stride(1)
    stride_cn = out_fp16.stride(2)

    # kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (batch_head, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    batched_matmul_kernel[grid](
        left_fp16, right_fp16, out_fp16,
        N, N, N,
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K)

    # reshape output back to [B, hidden_dim, N, N] and permute back to original shape [B,N,N,hidden_dim]
    out_mat = out_fp16.to(torch.float32).view(B, H, N, N)
    out = out_mat.permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim]

    # Apply to_out_norm
    # This is LayerNorm over hidden_dim final dimension
    # Use weights: to_out_norm_weight, bias
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # compute mean/var across last dim hidden_dim
    out_mean = out.mean(-1, keepdim=True)
    out_var = out.var(-1, unbiased=False, keepdim=True)
    eps2 = 1e-5
    out_norm = (out - out_mean) * torch.rsqrt(out_var + eps2)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)  # batch index
    pid_m = tl.program_id(1)  # output row tile index
    pid_n = tl.program_id(2)  # output col tile index

    # Compute offsets for the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for keeping within bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator. Use fp32 accumulation for precision.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Base pointers for this batch
    a_batch_ptr = a_ptr + pid * stride_ab
    b_batch_ptr = b_ptr + pid * stride_bb

    # Loop over K dimension with BLOCK_K tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Check bounds for K
        mask_k = offs_k < K

        # Compute pointers for A and B tiles
        # A: shape [BLOCK_M, BLOCK_K]
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)

    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul forward (outgoing) implementation with a Triton kernel for the
pairwise reduction (einsum). The heavy O(N³) part
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
is performed as a batched matrix multiplication
    out = left @ right.transpose(-2, -1)
using a custom Triton GEMM kernel. All other operations (LayerNorm,
Linear projections, gating) are carried out with PyTorch.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """C[batch, M, N] = A[batch, M, K] @ B[batch, K, N]  (B is already transposed)."""
    pid_batch = tl.program_id(0)   # batch*head
    pid_m = tl.program_id(1)       # row tile
    pid_n = tl.program_id(2)       # col tile

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for this batch element
    a_batch = a_ptr + pid_batch * stride_ab
    b_batch = b_ptr + pid_batch * stride_bb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        # B tile: (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + pid_batch * stride_cb + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    TriMul forward pass (outgoing) – a kernel-accelerated implementation.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, dim]
            mask: torch.Tensor [B, N, N] (may be all‑ones)
            weights: dict of model parameters (see code)
            config: dict with keys "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm on input ----------
    x = input_tensor.to(dtype)
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # ---------- Linear projections ----------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ---------- Mask (optional) ----------
    if config.get("nomask", True):
        mask_tensor = torch.ones(B, N, N, device=device, dtype=dtype)
    else:
        mask_tensor = mask.to(dtype)
    mask_tensor = mask_tensor.unsqueeze(-1)  # [B,N,N,1]

    left = left * mask_tensor
    right = right * mask_tensor

    # ---------- Gates ----------
    left_gate = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare tensors for batched GEMM ----------
    # shape -> [B, hidden, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # right needs to be transposed on the last two dims
    right_t = right_perm.transpose(-2, -1).contiguous()

    # Collapse batch & head dimensions
    batch_head = B * hidden_dim
    left_mat = left_perm.view(batch_head, N, N)
    right_mat = right_t.view(batch_head, N, N)

    # Cast to half for the kernel
    a_fp16 = left_mat.to(torch.float16)
    b_fp16 = right_mat.to(torch.float16)
    c_fp16 = torch.empty_like(a_fp16, dtype=torch.float16)

    # Strides (in elements)
    stride_ab, stride_am, stride_ak = a_fp16.stride()
    stride_bb, stride_bk, stride_bn = b_fp16.stride()
    stride_cb, stride_cm, stride_cn = c_fp16.stride()

    # ---------- Triton GEMM launch ----------
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    grid = (batch_head,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _batched_matmul_kernel[grid](
        a_fp16, b_fp16, c_fp16,
        N, N, N,
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # ---------- Restore shape ----------
    out_mat = c_fp16.to(dtype).view(B, hidden_dim, N, N)
    out = out_mat.permute(0, 2, 3, 1).contiguous()   # [B,N,N,hidden]

    # ---------- Output LayerNorm ----------
    out_mean = out.mean(-1, keepdim=True)
    out_var = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - out_mean) * torch.rsqrt(out_var + eps)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # ---------- Gate and final linear ----------
    out_norm = out_norm * out_gate
    output = F.linear(out_norm, weights["to_out.weight"])   # [B,N,N,dim]

    return output
