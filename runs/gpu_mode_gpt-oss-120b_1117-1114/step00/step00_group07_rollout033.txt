Inside kernel:

```
pid = tl.program_id(0)
num_blocks_m = tl.cdiv(M, BLOCK_M)
num_blocks_n = tl.cdiv(N, BLOCK_N)
blocks_per_batch = num_blocks_m * num_blocks_n
batch_id = pid // blocks_per_batch
block_id = pid % blocks_per_batch
pid_m = block_id // num_blocks_n
pid_n = block_id % num_blocks_n

offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
# ensure within bounds
mask_m = offs_m < M
mask_n = offs_n < N

# init accumulator
acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

# compute pointers for a and b
# Compute offsets for a block: rows = offs_m, cols = kk
for k in range(0, tl.cdiv(K, BLOCK_K)):
    k_base = k * BLOCK_K
    offs_k = k_base + tl.arange(0, BLOCK_K)
    mask_k = offs_k < K

    # load A subblock
    a_ptrs = a_ptr + batch_id * batch_stride_a + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)

    b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

    acc += tl.dot(a, b)

# finalize

tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # Gather weights, ensure dtype
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)
    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    to_out_weight = weights["to_out.weight"].to(device)

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Mask
    if mask is not None:
        mask_ = mask.to(x.dtype).unsqueeze(-1)  # shape (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for matmul
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Permute and reshape for batched matmul
    B, N, _, _ = left_fp16.shape
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()      # (B, hidden_dim, N, N)
    right_perm = right_fp16.permute(0, 3, 2, 1).contiguous()    # (B, hidden_dim, N, N) transposed

    batch_flat = B * hidden_dim
    M = N
    K = N
    N_dim = N

    left_flat = left_perm.view(batch_flat, M, K)
    right_flat = right_perm.view(batch_flat, K, N_dim)

    # Output buffer
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Strides
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)
    batch_stride_a = left_flat.stride(0)

    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)
    batch_stride_b = right_flat.stride(0)

    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)
    batch_stride_c = out_flat.stride(0)

    # Tile sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    blocks_per_batch = grid_m * grid_n
    total_blocks = blocks_per_batch * batch_flat
    grid = (total_blocks,)

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)  # shape (B,N,N,hidden_dim)

"""
TriMul outgoing implementation using Triton.

The forward pass follows the AlphaFold3 reference:
1. LayerNorm over the channel dimension.
2. Two linear projections (left, right) without bias.
3. Optional mask applied to both projections.
4. Three gating linear layers followed by sigmoid.
5. Element‑wise gate application.
6. Core O(N³) reduction:
   out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
   This is realized as a batched matrix multiplication
   (left @ rightᵀ) for each (batch, hidden) pair using a
   custom Triton kernel that operates in bfloat16 for speed
   and accumulates in float32.
7. LayerNorm + output gate.
8. Final linear projection back to the original channel dimension.

All heavy arithmetic (step 6) is executed on the GPU via Triton,
while the remaining lightweight operations are performed with PyTorch.
The final result is a float32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B  (A: [M,K], B: [K,N]) for many batches."""
    pid = tl.program_id(0)

    # Grid decomposition: (batch, block_m, block_n) packed into 1‑D pid
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n
    batch_id = pid // blocks_per_batch
    block_id = pid % blocks_per_batch
    pid_m = block_id // num_blocks_n
    pid_n = block_id % num_blocks_n

    # Offsets for the output block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers including batch offset
        a_ptrs = a_ptr + batch_id * batch_stride_a + \
                 (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        b_ptrs = b_ptr + batch_id * batch_stride_b + \
                 (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

        # Load tiles, convert to float32 for accumulation
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Store the result (cast back to bfloat16)
    c_ptrs = c_ptr + batch_id * batch_stride_c + \
             (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.bfloat16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul "outgoing" update.
    Args:
        data: Tuple (input, mask, weights, config)
            input   – Tensor [B, N, N, dim] (float32, on CUDA)
            mask    – Tensor [B, N, N] (float32 or bool, on CUDA)
            weights – dict of torch.Tensor containing all learned parameters
            config  – dict with keys "dim", "hidden_dim", "nomask", etc.
    Returns:
        Tensor [B, N, N, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------- unpack weights -------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)

    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)

    to_out_weight = weights["to_out.weight"].to(device)

    # ------- layer norm -----------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------- linear projections ---------------------------------------------
    left = F.linear(x, left_proj_weight)      # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------- mask (if present) ---------------------------------------------
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------- gates ----------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()  # used after to_out_norm

    left = left * left_gate
    right = right * right_gate

    # ------- cast to bfloat16 for the heavy reduction -----------------------
    left_bf16 = left.to(torch.bfloat16)
    right_bf16 = right.to(torch.bfloat16)

    B, N, _, _ = left_bf16.shape

    # Permute to (B, hidden, N, N) and transpose right for matmul
    left_perm = left_bf16.permute(0, 3, 1, 2).contiguous()      # (B, hidden, N, N)
    right_perm = right_bf16.permute(0, 3, 2, 1).contiguous()    # (B, hidden, N, N)

    # Flatten batch & hidden into a single batch dimension
    batch_flat = B * hidden_dim
    left_flat = left_perm.view(batch_flat, N, N)    # (B*hidden, M, K)
    right_flat = right_perm.view(batch_flat, N, N)  # (B*hidden, K, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat, dtype=torch.bfloat16)

    # Strides (in elements, not bytes)
    stride_am = left_flat.stride(1)   # stride for M dimension
    stride_ak = left_flat.stride(2)   # stride for K dimension
    batch_stride_a = left_flat.stride(0)

    stride_bk = right_flat.stride(1)  # stride for K dimension
    stride_bn = right_flat.stride(2)  # stride for N dimension
    batch_stride_b = right_flat.stride(0)

    stride_cm = out_flat.stride(1)    # stride for M dimension in output
    stride_cn = out_flat.stride(2)    # stride for N dimension in output
    batch_stride_c = out_flat.stride(0)

    # Tile sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: one program per output block across all batches
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    blocks_per_batch = grid_m * grid_n
    total_blocks = blocks_per_batch * batch_flat
    grid = (total_blocks,)

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim) and convert to float32
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------- output layer norm & output gate --------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate

    # ------- final linear projection to original dim -----------------------
    out = F.linear(out, to_out_weight)   # (B, N, N, dim)

    return out
