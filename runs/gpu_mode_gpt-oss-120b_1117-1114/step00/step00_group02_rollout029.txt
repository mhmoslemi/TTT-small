@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, M, N, K,
    stride_ab: tl.constexpr, stride_ak: tl.constexpr, stride_az: tl.constexpr,
    stride_bk: tl.constexpr, stride_bn: tl.constexpr, stride_bz: tl.constexpr,
    stride_cm: tl.constexpr, stride_cn: tl.constexpr, stride_cz: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for valid indices
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the block for A and B
    a_ptrs = a_ptr + pid_batch * stride_az + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + pid_batch * stride_bz + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load a and b tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K - k*BLOCK_K), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K) < K - k*BLOCK_K)[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store result
    c_ptrs = c_ptr + pid_batch * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k_iter in range(0, num_k_blocks):
        k_offset = k_iter * BLOCK_K
        cur_k = tl.arange(0, BLOCK_K)

        k_idx = k_offset + cur_k
        mask_k = k_idx < K

"""
TriMul forward pass with a custom Triton kernel.

The implementation follows the reference PyTorch TriMul module:
1. LayerNorm on the last channel dimension.
2. Linear projections (left/right) and gating logits (left/right/out).
3. Optional input mask applied before gating.
4. Core triangular multiplicative update:
       out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
   implemented as a batched matrix multiplication using a Triton kernel.
5. LayerNorm on the hidden dimension, out‑gate multiplication, and final linear
   projection back to the original channel dimension.

The Triton kernel computes a batched GEMM for each (batch * hidden_dim) slice.
It loads tiles of the left and transposed‑right matrices in FP16, accumulates in
FP32, and writes the result in FP32.  The rest of the operations remain in
PyTorch for simplicity and correctness.
"""

import torch
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: batched GEMM with A [B, M, K] and B [B, K, N] -> C [B, M, N]
# -------------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)  # batch * hidden_dim dimension
    pid_m = tl.program_id(1)      # output row block
    pid_n = tl.program_id(2)      # output col block

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k_iter in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k_iter * BLOCK_K
        cur_k = tl.arange(0, BLOCK_K)
        k_idx = k_offset + cur_k
        mask_k = k_idx < K

        # Pointers to the current tile of A and B
        a_tile_ptr = a_ptr + pid_batch * stride_a_batch \
                     + offs_m[:, None] * stride_a_m \
                     + k_idx[None, :] * stride_a_k

        b_tile_ptr = b_ptr + pid_batch * stride_b_batch \
                     + k_idx[:, None] * stride_b_k \
                     + offs_n[None, :] * stride_b_n

        # Load FP16 tiles with masking
        a = tl.load(a_tile_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        b = tl.load(b_tile_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Compute dot product, up‑cast to FP32 for accumulation
        a_fp32 = a.to(tl.float32)
        b_fp32 = b.to(tl.float32)
        acc += tl.dot(a_fp32, b_fp32)

    # Write the output tile
    c_tile_ptr = c_ptr + pid_batch * stride_c_batch \
                 + offs_m[:, None] * stride_c_m \
                 + offs_n[None, :] * stride_c_n

    tl.store(c_tile_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# Entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the Triangle Multiplicative Update (outgoing version)
    using a Triton kernel for the core multiplication.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, _ = input_tensor.shape

    # -----------------------------------------------------------------
    # 1. LayerNorm on the input tensor
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (applied before gating)
    # -----------------------------------------------------------------
    if not config.get("nomask", False):
        # mask: [bs, N, N] -> [bs, N, N, 1] and cast to float
        mask_f = mask.to(dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gating logits
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core triangular multiplicative update via Triton
    # -----------------------------------------------------------------
    # Cast to FP16 for the kernel (still accurate enough)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Prepare tensors for batched GEMM:
    #   left:   [bs, N, N, hidden] -> permute -> [bs, hidden, N, N]
    #   right:  [bs, N, N, hidden] -> permute & transpose K/J -> [bs, hidden, N, N]
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()   # (bs, hidden, N, N)
    right_perm = right_fp16.permute(0, 3, 2, 1).contiguous()  # (bs, hidden, N, N)

    batch = bs * hidden_dim  # total number of independent GEMMs

    # Collapse the first two dimensions so each GEMM works on (N, N) matrices
    A = left_perm.reshape(batch, N, N)
    B = right_perm.reshape(batch, N, N)

    # Output buffer (float32)
    C = torch.empty((batch, N, N), dtype=torch.float32, device=device)

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Launch the kernel
    tri_mul_kernel[grid](
        A,
        B,
        C,
        batch,
        N,
        N,
        N,
        A.stride(0), A.stride(1), A.stride(2),   # A strides: batch, M, K
        B.stride(0), B.stride(1), B.stride(2),   # B strides: batch, K, N
        C.stride(0), C.stride(1), C.stride(2),   # C strides: batch, M, N
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to original layout: (bs, N, N, hidden_dim)
    out = C.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, and final projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
