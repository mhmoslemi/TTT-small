@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_am, stride_ak, stride_ac,
    stride_bm, stride_bk, stride_bc,
    stride_cm, stride_cr, stride_cc,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    
    # program ids
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets for M and N
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute K offset for this block
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Load A tile (batch, M, K)
        a_ptrs = a_ptr + pid_batch * stride_am + offs_m[:, None] * stride_ak + k_offset[None, :] * stride_ac
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (batch, K, N)
        b_ptrs = b_ptr + pid_batch * stride_bm + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bc
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate

def batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    # expects left shape (B,N,N,H) and right shape (B,N,N,H)
    # compute left @ right^T per channel using Triton
    device = left.device
    B, N, _, H = left.shape
    # Permute
    left_perm = left.permute(0,3,1,2).contiguous()  # (B,H,N,N)
    right_perm = right.permute(0,3,2,1).contiguous()  # (B,H,N,N) transposed

    right_fp16 = right_flat.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16)  # (M_total,N,N)
    # Strides
    stride_am = left_fp16.stride(0)
    stride_ak = left_fp16.stride(1)
    stride_ac = left_fp16.stride(2)
    stride_bm = right_fp16.stride(0)
    stride_bk = right_fp16.stride(1)
    stride_bc = right_fp16.stride(2)
    stride_cm = out_fp16.stride(0)
    stride_cr = out_fp16.stride(1)
    stride_cc = out_fp16.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

"""
TriMul outgoing implementation (AlphaFold3).

The heavy contraction
    out[i, j, d] = Σ_k left[i, k, d] * right[j, k, d]
is performed with a custom Triton kernel that treats the hidden
dimension as an extra batch dimension and computes many small
matrix multiplications (left @ rightᵀ) in parallel.
All other operations (LayerNorm, linear projections, gating) are
handled with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton batched GEMM kernel (left @ rightᵀ per hidden channel)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                         # pointers
    total_batch, M, N, K,                         # dimensions (M=N=K=seq_len)
    stride_am, stride_ak, stride_ac,              # A strides: batch, row, col
    stride_bm, stride_bk, stride_bc,              # B strides: batch, row(K), col(N)
    stride_cm, stride_cr, stride_cc,              # C strides: batch, row, col
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)      # which (batch * hidden) we are processing
    pid_m = tl.program_id(1)          # tile row index
    pid_n = tl.program_id(2)          # tile col index

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < K

        # A tile: (M x K)
        a_ptrs = a_ptr + pid_batch * stride_am \
                       + offs_m[:, None] * stride_ak \
                       + k_off[None, :] * stride_ac
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B tile: (K x N) – note B is already transposed (k dimension first)
        b_ptrs = b_ptr + pid_batch * stride_bm \
                       + k_off[:, None] * stride_bk \
                       + offs_n[None, :] * stride_bc
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_ptr + pid_batch * stride_cm \
                   + offs_m[:, None] * stride_cr \
                   + offs_n[None, :] * stride_cc
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Computes `out = left @ rightᵀ` for each hidden channel using Triton.
    left, right: (B, N, N, H) float32
    Returns out: (B, N, N, H) float32
    """
    B, N, _, H = left.shape
    device = left.device

    # Bring hidden dimension into the batch dimension and make tensors contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) with swapped axes

    total_batch = B * H
    left_flat = left_perm.view(total_batch, N, N)
    right_flat = right_perm.view(total_batch, N, N)

    # Use FP16 inside the kernel to improve bandwidth
    a_fp16 = left_flat.to(torch.float16)
    b_fp16 = right_flat.to(torch.float16)
    c_fp16 = torch.empty_like(a_fp16)

    # strides (batch, row, col)
    stride_am, stride_ak, stride_ac = a_fp16.stride()
    stride_bm, stride_bk, stride_bc = b_fp16.stride()
    stride_cm, stride_cr, stride_cc = c_fp16.stride()

    # tile sizes – reasonable defaults for B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        total_batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        a_fp16, b_fp16, c_fp16,
        total_batch, N, N, N,
        stride_am, stride_ak, stride_ac,
        stride_bm, stride_bk, stride_bc,
        stride_cm, stride_cr, stride_cc,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # a modest number of warps works well on B200
    )

    # reshape back to (B, N, N, H) and cast to FP32
    out = c_fp16.to(torch.float32).view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ----- LayerNorm on the input -----
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # ----- Linear projections -----
    left = F.linear(x, left_proj_weight)   # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_weight)

    # ----- Mask handling -----
    if nomask:
        mask_f = torch.ones_like(mask, dtype=x.dtype, device=device)
    else:
        mask_f = mask.to(dtype=x.dtype, device=device)
    mask_f = mask_f.unsqueeze(-1)  # (B, N, N, 1)

    left = left * mask_f
    right = right * mask_f

    # ----- Gating -----
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----- Core contraction via Triton -----
    out = _batched_matmul_triton(left, right)               # (B, N, N, hidden_dim)

    # ----- Output LayerNorm, gating and final projection -----
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=1e-5)

    out = out * out_gate
    out = F.linear(out, to_out_weight)                      # (B, N, N, dim)

    return out
```
