def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # extract config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Mask may be none? Actually mask always provided maybe but may be all ones if nomask.
    # Ensure mask shape matches.

    # LayerNorm: from weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # normalizing across last dim (dim)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # compute linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    # bias? All have bias=False except layernorm and to_out_norm's bias, and to_out weight has no bias.

    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)
    # Compute gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # Apply mask if any; mask is bool or float. mask shape [B,N,N]
    if mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # broadcast to hidden
        left = left * mask
        right = right * mask
    else:
        # No mask: just skip
        pass

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Now compute core multiplication via custom kernel
    # Permute to [B, hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, N2 = left_perm.shape
    assert N == N2, "Non-square??" In our case square.

    # flatten batch*hidden for kernel
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Compute multiplication
    out_flat = triton_matmul(left_flat, right_flat)  # implement function wrapping kernel
    # Reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)  # shape [B,N,N,hidden]

    # Apply to_out_norm (LayerNorm over hidden_dim)

    out = out * out_gate

    out = torch.nn.functional.linear(out, to_out_weight)  # shape [B,N,N,dim]

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _triton_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    N,
    stride_ab, stride_ai, stride_ak,
    stride_bb, stride_bi, stride_bk,
    stride_cb, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    # Tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # A: shape (BLOCK_M, BLOCK_K) – A[pid_z, i, k]
        a_ptrs = A_ptr + pid_z * stride_ab + offs_i[:, None] * stride_ai + k_offset[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # B_T: shape (BLOCK_K, BLOCK_N) – B_T[k, j] = B[pid_z, j, k]
        b_ptrs = B_ptr + pid_z * stride_bb + k_offset[:, None] * stride_bk + offs_j[None, :] * stride_bi
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write the result
    c_ptrs = C_ptr + pid_z * stride_cb + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def _triton_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B.T for tensors of shape
    (batch, N, N).  The input tensors must be contiguous and have the same
    size.  Returns C of dtype torch.float32.
    """
    assert A.is_contiguous() and B.is_contiguous()
    assert A.shape == B.shape
    batch, N, _ = A.shape

    # Allocate output in float32
    C = torch.empty((batch, N, N), dtype=torch.float32, device=A.device)

    # Tile sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), batch)

    # Strides (in elements)
    stride_ab = A.stride(0)
    stride_ai = A.stride(1)
    stride_ak = A.stride(2)

    stride_bb = B.stride(0)
    stride_bi = B.stride(1)
    stride_bk = B.stride(2)

    stride_cb = C.stride(0)
    stride_ci = C.stride(1)
    stride_cj = C.stride(2)

    _triton_matmul_kernel[grid](
        A, B, C,
        N,
        stride_ab, stride_ai, stride_ak,
        stride_bb, stride_bi, stride_bk,
        stride_cb, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C

def custom_kernel(data):
    """
    Triton‑accelerated implementation of the TriMul “outgoing” update.
    The heavy N³ contraction is performed with a custom Triton kernel
    that computes C[b,i,j] = Σₖ left[b,i,k]·right[b,j,k] for every hidden
    channel (batch*hidden is treated as the kernel’s third grid dimension).
    All other operations (LayerNorm, linear projections, gating) are kept
    in PyTorch for simplicity.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # ------------------------------------------------------------------
    # 1. Layer‑norm over the feature dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # ------------------------------------------------------------------
    # 2. Linear projections and gating pre‑activations
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 3. Apply mask (if supplied) and gates
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core contraction: out = left @ rightᵀ  (batched GEMM per hidden)
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape  # batch, seq_len, seq_len, dim
    # Rearrange to (B, hidden, N, N) and flatten batch*hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_perm.view(B * hidden_dim, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)

    # Triton GEMM (A @ Bᵀ) -> shape (B*H, N, N)
    out_flat = _triton_matmul(left_flat, right_flat)

    # Reshape back to [B, N, N, hidden]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 5. Output normalisation, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out

        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # B_T[k, j] = B[pid_z, j, k]
        b_ptrs = B_ptr + pid_z * stride_bb \
                        + k_off[:, None] * stride_bk \
                        + offs_j[None, :] * stride_bi
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = C_ptr + pid_z * stride_cb \
                    + offs_i[:, None] * stride_ci \
                    + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def _triton_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM: C = A @ Bᵀ
    A/B shape: (batch, N, N), dtype float32, contiguous.
    Returns C of dtype torch.float32.
    """
    assert A.is_contiguous() and B.is_contiguous()
    assert A.shape == B.shape
    batch, N, _ = A.shape

    C = torch.empty((batch, N, N), dtype=torch.float32, device=A.device)

    # tile sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )

    stride_ab = A.stride(0)
    stride_ai = A.stride(1)
    stride_ak = A.stride(2)

    stride_bb = B.stride(0)
    stride_bi = B.stride(1)
    stride_bk = B.stride(2)

    stride_cb = C.stride(0)
    stride_ci = C.stride(1)
    stride_cj = C.stride(2)

    _triton_matmul_kernel[grid](
        A, B, C,
        N,
        stride_ab, stride_ai, stride_ak,
        stride_bb, stride_bi, stride_bk,
        stride_cb, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Triton‑accelerated TriMul (outgoing) forward pass.

    Steps:
    1. Layer‑norm over the feature dimension.
    2. Linear projections for left/right and their gates.
    3. Apply optional mask and the sigmoid gates.
    4. Core O(N³) contraction via a custom Triton kernel:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d].
    5. Layer‑norm over the hidden dimension, apply the output gate
       and final linear projection back to the original feature size.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch, seq_len, _, _ = input_tensor.shape
    N = seq_len

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Linear projections and gate pre‑activations
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # --------------------------------------------------------------
    # 3. Mask (if present) and element‑wise gating
    # --------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 4. Core contraction using Triton (batched GEMM per hidden channel)
    # --------------------------------------------------------------
    # reshape to (B, hidden, N, N) and flatten batch*hidden for the kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_perm.view(batch * hidden_dim, N, N)
    right_flat = right_perm.view(batch * hidden_dim, N, N)

    # Triton matrix multiplication: C = left_flat @ right_flatᵀ
    out_flat = _triton_matmul(left_flat, right_flat)  # [B*H, N, N] float32

    # reshape back to [B, N, N, H]
    out = out_flat.view(batch, hidden_dim, N, N).permute(0, 2, 3, 1)

    # --------------------------------------------------------------
    # 5. Output LayerNorm, gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
