    left_ptr, right_ptr, out_ptr,
    B, N,

    stride_right_batch, stride_right_j, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,

    # using torch.nn.functional.linear (no bias)
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # Apply mask if present
    if mask is not None:
        mask_ = mask.unsqueeze(-1).type_as(left)  # shape [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # Apply elementwise gates
    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # Allocate output half tensor
    out = torch.empty_like(left, dtype=torch.float16, device=device)

    # Precompute strides
    stride_left_batch = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)

    stride_right_batch = right.stride(0)
    stride_right_i = right.stride(1)
    stride_right_k = right.stride(2)

    stride_out_batch = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)

    # Triton kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    for d in range(hidden_dim):
        left_d = left[..., d].contiguous()
        right_d = right[..., d].contiguous()
        out_d = out[..., d].contiguous()
        # Launch kernel
        tr_mul_one_channel_kernel[grid](
            left_d,
            right_d,
            out_d,
            B,
            N,
            stride_left_batch,
            stride_left_i,
            stride_left_k,
            stride_right_batch,
            stride_right_i,
            stride_right_k,
            stride_out_batch,
            stride_out_i,
            stride_out_j,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )
    # Cast out back to float32
    out = out.to(torch.float32)

    # Apply to_out_norm (LayerNorm over hidden_dim)
    # out: [B,N,N,hidden_dim]
    mean_out = out.mean(-1, keepdim=True)
    var_out = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_out) / torch.sqrt(var_out + eps)
    out = out * to_out_norm_weight + to_out_norm_bias

    # Multiply with out_gate (shape same)
    out = out * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

@triton.jit
def tr_mul_one_channel_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # coordinate ranges
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

"""
TriMul forward kernel (outgoing version) implemented with Triton.

The heavy O(N³) contraction (einsum '... i k d, ... j k d -> ... i j d')
is performed by a custom Triton kernel that computes, for each
batch and for each hidden dimension, the matrix multiplication
   out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d].
All other operations (LayerNorm, linear projections, gates) are
executed in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tr_mul_one_channel_kernel(
    left_ptr,          # *fp16
    right_ptr,         # *fp16
    out_ptr,           # *fp16
    B, N,              # int
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k] for a single channel."""
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)   # block row (i)
    pid_n = tl.program_id(2)   # block col (j)

    # Coordinate ranges for i and j
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left: shape (BLOCK_M, BLOCK_K)
        left_offset = (
            batch * stride_left_batch
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptr + left_offset,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right: shape (BLOCK_K, BLOCK_N) – k as leading dimension
        right_offset = (
            batch * stride_right_batch
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_i
        )
        right = tl.load(
            right_ptr + right_offset,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # Accumulate the dot product over the current k‑block
        acc += tl.dot(left, right)

    # Store the result (cast back to fp16)
    out_offset = (
        batch * stride_out_batch
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(
        out_ptr + out_offset,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the TriMul module using a Triton kernel for the
    O(N³) reduction.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] torch.Tensor (float32)
            - mask:        [B, N, N]      torch.Tensor (float32 or bool)
            - weights:     dict of torch.Tensors containing all model weights
            - config:      dict with keys "dim", "hidden_dim"

    Returns:
        output_tensor: [B, N, N, dim] torch.Tensor (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # Load weights (all are stored as FP32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float32, device=device)
    norm_bias = weights["norm.bias"].to(torch.float32, device=device)

    left_proj_weight = weights["left_proj.weight"].to(torch.float32, device=device)   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"].to(torch.float32, device=device)

    left_gate_weight = weights["left_gate.weight"].to(torch.float32, device=device)
    right_gate_weight = weights["right_gate.weight"].to(torch.float32, device=device)
    out_gate_weight = weights["out_gate.weight"].to(torch.float32, device=device)

    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32, device=device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32, device=device)

    to_out_weight = weights["to_out.weight"].to(torch.float32, device=device)      # (dim, hidden_dim)

    # -----------------------------------------------------------------
    # LayerNorm over the last dimension (C = dim)
    # -----------------------------------------------------------------
    x = input_tensor
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x = (x - mean) / torch.sqrt(var + eps)
    x = x * norm_weight + norm_bias

    # -----------------------------------------------------------------
    # Linear projections (no bias) and gates
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)          # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    # -----------------------------------------------------------------
    # Apply mask if supplied
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)        # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # Apply element‑wise gates
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Cast to FP16 for the heavy reduction
    # -----------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # Output tensor (still FP16 at this stage)
    out = torch.empty_like(left, dtype=torch.float16, device=device)

    # Strides needed by the Triton kernel
    # left/right shape: (B, N, N) after slicing a channel
    stride_left_batch = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)

    stride_right_batch = right.stride(0)
    stride_right_i = right.stride(1)
    stride_right_k = right.stride(2)

    stride_out_batch = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)

    # Triton block sizes – chosen to fit typical B200 resources
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # -----------------------------------------------------------------
    # Run the Triton kernel for each hidden dimension independently
    # -----------------------------------------------------------------
    for d in range(hidden_dim):
        # Views for a single channel (shape [B, N, N])
        left_ch = left[..., d].contiguous()
        right_ch = right[..., d].contiguous()
        out_ch = out[..., d].contiguous()

        tr_mul_one_channel_kernel[grid](
            left_ch,
            right_ch,
            out_ch,
            B,
            N,
            stride_left_batch,
            stride_left_i,
            stride_left_k,
            stride_right_batch,
            stride_right_i,
            stride_right_k,
            stride_out_batch,
            stride_out_i,
            stride_out_j,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )

    # Convert back to FP32 for the remaining operations
    out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # LayerNorm over hidden dimension (to_out_norm)
    # -----------------------------------------------------------------
    mean_out = out.mean(-1, keepdim=True)
    var_out = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean_out) / torch.sqrt(var_out + eps)
    out = out * to_out_norm_weight + to_out_norm_bias

    # Multiply by the output gate (broadcasted)
    out = out * out_gate

    # Final linear projection back to model dimension
    out = F.linear(out, to_out_weight)   # shape [B, N, N, dim]

    return out
