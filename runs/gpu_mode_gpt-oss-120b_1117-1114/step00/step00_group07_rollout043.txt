@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_b_left, stride_i_left, stride_k_left, stride_d_left,
    stride_b_right, stride_i_right, stride_k_right, stride_d_right,
    stride_b_out, stride_i_out, stride_j_out, stride_d_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(2)  # batch
    pid_m = tl.program_id(0)  # block row (i)
    pid_n = tl.program_id(1)  # block col (j)

    # Compute offsets for i and j base indices
    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N

    # Offsets range
    i_offsets = i_base + tl.arange(0, BLOCK_M)
    j_offsets = j_base + tl.arange(0, BLOCK_N)
    k_offsets = tl.arange(0, BLOCK_K)
    d_offsets = tl.arange(0, BLOCK_D)

    # Masks
    mask_i = i_offsets < N
    mask_j = j_offsets < N

    # Compute base pointers for given batch
    left_batch_ptr = left_ptr + pid_b * stride_b_left
    right_batch_ptr = right_ptr + pid_b * stride_b_right
    out_batch_ptr = out_ptr + pid_b * stride_b_out

    # Initialize accumulator tile
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # Loop over k dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        k_valid = k_base + k_offsets < N

        # Load left block: shape [BLOCK_M, BLOCK_K, BLOCK_D]
        # Compute pointer for left: left + i_offsets[:, None, None] * stride_i_left + (k_base + k_offsets)[None, :, None] * stride_k_left + d_offsets[None, None, :] * stride_d_left
        left_ptrs = left_batch_ptr + i_offsets[:, None, None] * stride_i_left + (k_base + k_offsets)[None, :, None] * stride_k_left + d_offsets[None, None, :] * stride_d_left
        left_block = tl.load(left_ptrs, mask=mask_i[:, None, None] & k_valid[None, :, None] & (d_offsets < D), other=0.0)

        # Load right block: shape [BLOCK_N, BLOCK_K, BLOCK_D] with index j,k,d.
        right_ptrs = right_batch_ptr + j_offsets[:, None, None] * stride_i_right + (k_base + k_offsets)[None, :, None] * stride_k_right + d_offsets[None, None, :] * stride_d_right
        right_block = tl.load(right_ptrs, mask=mask_j[:, None, None] & k_valid[None, :, None] & (d_offsets < D), other=0.0)

        # Compute contribution:
        # For each d, we have outer product across i and j: left_block[i,k,d] * right_block[j,k,d], sum over k.
        # We can compute using broadcasting:
        # left_block: (M,K,D)
        # right_block: (N,K,D)
        # We want per (i,j,d): sum_k left[i,k,d] * right[j,k,d]
        # => We can loop over k block: use inner loop over k within tile:
        for kk in range(BLOCK_K):
            # Get vector for left at (i, kk) across D
            left_vec = left_block[:, kk, :]   # shape (M, D)
            # Get vector for right at (j, kk) across D
            right_vec = right_block[:, kk, :] # shape (N, D)

            # Multiply and accumulate:
            # Need to broadcast left_vec (M,D) and right_vec (N,D) to (M,N,D)
            # Use tl.broadcast? Use tl.expand_dims? In Triton we can compute via broadcasting: left_vec[:, None, :] * right_vec[None, :, :] yields shape (M,N,D)

            # Use tl.dot? No, but we can compute elementwise multiplies with broadcasting.
            # But Triton elementwise broadcasting may not be direct; we can use outer addition like:
            # left_vec[:, None, :] * right_vec[None, :, :]

            # However left_vec and right_vec are pointers to registers; we can compute using tl.broadcast in Python style?
            # In Triton, arithmetic supports broadcasting if dimensions are compatible: e.g., left_vec[:, None, :] (shape M,1,D) multiplied by right_vec[None, :, :] (1,N,D) yields shape M,N,D. That may work.

            acc += left_vec[:, None, :] * right_vec[None, :, :]

    # Write output tile
    out_ptrs = out_batch_ptr + i_offsets[:, None, None] * stride_i_out + j_offsets[None, :, None] * stride_j_out + d_offsets[None, None, :] * stride_d_out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fuse_gate_mask_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elems,
    hidden_dim,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # Load mask repeated across hidden_dim: each hidden element uses same mask entry.
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    left_out = lp * mask_val * lg
    tl.store(left_out_ptr + offs, left_out, mask=mask)

    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    right_out = rp * mask_val * rg
    tl.store(right_out_ptr + offs, right_out, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # cast mask to float32
    mask = mask.to(dtype=torch.float32, device=device)

    # extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ensure contiguous
    inp = input_tensor.contiguous()
    mask = mask.contiguous()

    # LayerNorm on input
    x = F.layer_norm(inp, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections (float32)
    left_proj = F.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right_proj = F.linear(x, right_proj_weight)

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Ensure tensors are contiguous for kernel
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    out_gate = out_gate.contiguous()

    # Prepare output tensors for gated+masked left/right
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # Launch fuse kernel
    total = left_proj.numel()
    BLOCK = 1024 * 4  # 4096
    grid = (triton.cdiv(total, BLOCK),)
    fuse_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left,
        right,
        total,
        hidden_dim,
        BLOCK=BLOCK
    )

    # Now perform batched matrix multiplication across hidden dimension
    # left: [B,N,N,D] indices (i,k); right: [B,N,N,D] indices (j,k)
    # Need to transpose right's second and first N dims
    # Reshape for batched gemm
    B_, N_, _, D_ = left.shape
    # left_mat shape (B*D, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * D, N, N)
    right_T = right.permute(0, 2, 1, 3)  # (B, N, N, D) -> (B, k, j, d)
    right_mat = right_T.permute(0, 3, 1, 2).reshape(B * D, N, N)
    # Batch matmul
    out_mat = torch.bmm(left_mat, right_mat)
    # reshape back
    out = out_mat.view(B, D, N, N).permute(0, 2, 3, 1)  # shape [B,N,N,D]

    # to_out_norm (LayerNorm over hidden_dim)
    out_norm = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply out_gate
    out_gate = out_gate  # already computed
    out_gated = out_norm * out_gate

    # Final linear projection to dim
    output = F.linear(out_gated, to_out_weight)

    return output

"""
TriMul forward implementation with a fused Triton kernel.

The algorithm follows the reference PyTorch implementation:

1. Layer‑norm the input tensor.
2. Linear projections (`left_proj`, `right_proj`) and linear gates
   (`left_gate`, `right_gate`, `out_gate`) are computed with torch.nn.functional.
3. A small Triton kernel fuses the mask, the projection and the gate:
      left  = left_proj  * mask * left_gate
      right = right_proj * mask * right_gate
4. The core O(N³) reduction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is performed as a batched matrix multiplication:
      left  → (B·D, N, N)  with axes (i,k)
      right → (B·D, N, N)  with axes (k,j)   (right is transposed on the two
      sequence dimensions before reshaping)
      out   = torch.bmm(left, right)  → (B·D, N, N)
      reshaped back to (B, N, N, D).
5. Apply the output LayerNorm, elementwise multiply by `out_gate`,
   and finally a linear projection back to the original dimension `dim`.

Only the mask‑gate‑fusion is performed in Triton; the remaining
operations use highly‑optimized torch kernels (cuBLAS, cuDNN).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _fuse_gate_mask_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elems,
    hidden_dim,
    BLOCK: tl.constexpr,
):
    """Fuse mask, projection and gate.  Operates on flattened tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    # mask is broadcast on the hidden dimension -> each hidden entry shares the same mask value
    mask_idx = offs // hidden_dim
    m_val = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)

    lp = tl.load(left_proj_ptr + offs, mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    tl.store(left_out_ptr + offs, lp * m_val * lg, mask=valid)

    rp = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    tl.store(right_out_ptr + offs, rp * m_val * rg, mask=valid)


def custom_kernel(data):
    """
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (float or bool)
            - weights: dict of weight tensors
            - config: dict with keys "dim" and "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 0) Prepare tensors / cast types
    # ------------------------------------------------------------------
    inp = input_tensor.contiguous()
    mask = mask.to(dtype=torch.float32, device=device).contiguous()

    # ------------------------------------------------------------------
    # 1) Input layer norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections and gates (float32)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Ensure contiguity for the Triton kernel
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    out_gate = out_gate.contiguous()

    # ------------------------------------------------------------------
    # 3) Fuse mask, projection and gate using Triton
    # ------------------------------------------------------------------
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    total_elements = left_proj.numel()
    BLOCK = 4096  # tunable; fits well in registers
    grid = (triton.cdiv(total_elements, BLOCK),)
    _fuse_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left,
        right,
        total_elements,
        hidden_dim,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 4) Core O(N³) reduction via batched GEMM over the hidden dimension
    # ------------------------------------------------------------------
    # left: (B, N, N, D)   -> (B·D, N, N)   axes (i, k)
    # right: (B, N, N, D)  -> after transposition -> (B·D, N, N) axes (k, j)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_T = right.permute(0, 2, 1, 3)               # (B, k, j, D)
    right_mat = right_T.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat)        # (B·D, N, N)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B, N, N, D)

    # ------------------------------------------------------------------
    # 5) Output LayerNorm, gate and final linear projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_gated = out_norm * out_gate
    output = F.linear(out_gated, weights["to_out.weight"])

    return output
```
