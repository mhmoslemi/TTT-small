def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,

    stride_rb, stride_ri, stride_rj, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)  # block_i
    pid1 = tl.program_id(1)  # block_j
    pid2 = tl.program_id(2)  # batch*H index

    # Determine actual batch and hidden index
    bh = pid2
    batch = bh // H
    hidden = bh % H

"""
TriMul "outgoing" update kernel (AlphaFold3).

The implementation follows the reference PyTorch code but off‑loads the
O(N³) triangular multiplication
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
to a custom Triton kernel.  The rest of the workflow (layer‑norms,
linear projections, gating and final projection) is kept in PyTorch.
The kernel works on a per‑batch‑and‑hidden‑slice basis:
each block computes a tile of the output matrix for a fixed (batch,
hidden) pair using a classic GEMM tiling strategy.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel for the core triangular multiplication
# -------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # data pointers
    B, N, H,                               # ints: batch, seq_len, hidden
    stride_lb, stride_li, stride_lj, stride_lh,   # left strides
    stride_rb, stride_ri, stride_rj, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid_i = tl.program_id(0)          # tile over i (rows)
    pid_j = tl.program_id(1)          # tile over j (cols)
    pid_hb = tl.program_id(2)         # combined batch & hidden index

    # decode batch and hidden slice
    hid = pid_hb % H
    batch = pid_hb // H

    # base offsets for the selected batch & hidden dimension
    left_base  = left_ptr  + batch * stride_lb + hid * stride_lh
    right_base = right_ptr + batch * stride_rb + hid * stride_rh
    out_base   = out_ptr   + batch * stride_ob + hid * stride_oh

    # block start indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # offsets within the block
    i_offsets = i0 + tl.arange(0, BLOCK_M)
    j_offsets = j0 + tl.arange(0, BLOCK_N)

    mask_i = i_offsets < N
    mask_j = j_offsets < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        k_offsets = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < N

        # ----- load left tile (M x K) -----
        left_ptrs = left_base + i_offsets[:, None] * stride_li + k_offsets[None, :] * stride_lj
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)          # shape (M, K)

        # ----- load right tile transposed (K x N) -----
        # we need right[k, j] layout, i.e. (K, N)
        right_ptrs = right_base + k_offsets[:, None] * stride_rj + j_offsets[None, :] * stride_ri
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)         # shape (K, N)

        # compute dot product and accumulate (convert to fp32)
        acc += tl.dot(left, right).to(tl.float32)

    # store the resulting tile
    out_ptrs = out_base + i_offsets[:, None] * stride_oi + j_offsets[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# -------------------------------------------------------------
# Host entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul "outgoing" module with a Triton kernel
    for the heavy O(N³) multiplication.
    Arguments:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    # -----------------------------------------------------------------
    # Extract config and weight tensors
    # -----------------------------------------------------------------
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # LayerNorm on the input
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # -----------------------------------------------------------------
    # Linear projections
    # -----------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]     # (hidden_dim, dim)

    left = F.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_unsq = mask.unsqueeze(-1).to(left.dtype)   # (B,N,N,1)
        left = left * mask_unsq
        right = right * mask_unsq

    # -----------------------------------------------------------------
    # Gating
    # -----------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = F.linear(x, left_gate_weight).sigmoid()   # (B,N,N,hidden_dim)
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate   = F.linear(x, out_gate_weight).sigmoid()    # (B,N,N,hidden_dim)

    left = (left * left_gate).to(torch.float16)    # cast to fp16 for kernel
    right = (right * right_gate).to(torch.float16)

    # -----------------------------------------------------------------
    # Triton kernel: triangular multiplication
    # -----------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=input_tensor.device)

    # strides (in elements, not bytes)
    stride_lb, stride_li, stride_lj, stride_lh = left.stride()
    stride_rb, stride_ri, stride_rj, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # block sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            B * hidden_dim)

    tri_mul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lj, stride_lh,
        stride_rb, stride_ri, stride_rj, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # Post‑processing: LayerNorm, final gate, and projection
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate                               # apply output gate

    to_out_weight = weights["to_out.weight"]           # (dim, hidden_dim)
    out = F.linear(out, to_out_weight)                 # final projection -> (B,N,N,dim)

    return out
