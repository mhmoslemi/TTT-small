        stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_d,
        BLOCK=BLOCK
    )

    # After kernel, masked_left and masked_right hold left*mask*gate etc.

    # Use torch.bmm for batched matrix multiplication per channel
    left_reshape = masked_left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_reshape = masked_right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)  # note transpose

    out_reshape = torch.bmm(left_reshape, right_reshape)  # shape (B*hidden_dim,N,N)

    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,hidden_dim)

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # out gate multiplication
    out = out * out_gate

    # final linear
    output = torch.nn.functional.linear(out, to_out_weight)  # shape (B,N,N,dim)

    return output

"""
TriMul (outgoing) forward pass with a small Triton kernel.

The heavy O(N³·C) reduction is expressed as a batched matrix multiplication
per hidden channel:
    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
which is equivalent to (left_c @ right_cᵀ) for each channel c.
All LayerNorms, linear projections, and sigmoid gates are performed in
PyTorch.  The only Triton‑accelerated part is the mask+gate fusion,
which multiplies the projected tensors by the optional mask and the
corresponding sigmoid gates in a single kernel.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_kernel(
    out_left_ptr, out_right_ptr,          # output buffers
    in_left_ptr, in_right_ptr,            # input projected tensors
    mask_ptr,                             # mask tensor (float)
    left_gate_ptr, right_gate_ptr,        # sigmoid gates
    B: tl.constexpr, N: tl.constexpr, C: tl.constexpr,          # batch, seq, hidden
    stride_in_left_b, stride_in_left_i, stride_in_left_j, stride_in_left_c,
    stride_in_right_b, stride_in_right_i, stride_in_right_j, stride_in_right_c,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_gate_left_b, stride_gate_left_i, stride_gate_left_j, stride_gate_left_c,
    stride_gate_right_b, stride_gate_right_i, stride_gate_right_j, stride_gate_right_c,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                     # one program per (b,i,j)
    total_ij = N * N
    b = pid // total_ij
    ij = pid % total_ij
    i = ij // N
    j = ij % N

    # --------------------------------------------------------------------
    # Load the scalar mask value for this (b,i,j) location
    # --------------------------------------------------------------------
    mask_off = mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    mask_val = tl.load(mask_off, mask=tl.full([1], True, dtype=tl.int1), other=0.0)

    # --------------------------------------------------------------------
    # Base offsets for the 4‑D tensors (b,i,j,channel)
    # --------------------------------------------------------------------
    left_base = in_left_ptr + b * stride_in_left_b + i * stride_in_left_i + j * stride_in_left_j
    right_base = in_right_ptr + b * stride_in_right_b + i * stride_in_right_i + j * stride_in_right_j
    left_gate_base = left_gate_ptr + b * stride_gate_left_b + i * stride_gate_left_i + j * stride_gate_left_j
    right_gate_base = right_gate_ptr + b * stride_gate_right_b + i * stride_gate_right_i + j * stride_gate_right_j

    out_left_base = out_left_ptr + b * stride_in_left_b + i * stride_in_left_i + j * stride_in_left_j
    out_right_base = out_right_ptr + b * stride_in_right_b + i * stride_in_right_i + j * stride_in_right_j

    # --------------------------------------------------------------------
    # Process the hidden dimension in blocks
    # --------------------------------------------------------------------
    for c_off in range(0, C, BLOCK):
        offs = c_off + tl.arange(0, BLOCK)
        mask = offs < C

        left = tl.load(left_base + offs * stride_in_left_c, mask=mask, other=0.0)
        right = tl.load(right_base + offs * stride_in_right_c, mask=mask, other=0.0)

        left_gate = tl.load(left_gate_base + offs * stride_gate_left_c, mask=mask, other=0.0)
        right_gate = tl.load(right_gate_base + offs * stride_gate_right_c, mask=mask, other=0.0)

        # apply sigmoid gates and mask
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val

        tl.store(out_left_base + offs * stride_in_left_c, left, mask=mask)
        tl.store(out_right_base + offs * stride_in_right_c, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] torch.Tensor (float32/float16)
            - mask:        [B, N, N] torch.Tensor (bool or float)
            - weights: dict with all linear and norm parameters (torch.Tensor)
            - config:  dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    torch.cuda.set_device(torch.cuda.current_device())
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------
    # Load model weights (convert to the correct device / dtype)
    # --------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias = weights["norm.bias"].to(device, dtype)

    left_proj_weight = weights["left_proj.weight"].to(device, dtype)
    right_proj_weight = weights["right_proj.weight"].to(device, dtype)

    left_gate_weight = weights["left_gate.weight"].to(device, dtype)
    right_gate_weight = weights["right_gate.weight"].to(device, dtype)
    out_gate_weight = weights["out_gate.weight"].to(device, dtype)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device, dtype)

    to_out_weight = weights["to_out.weight"].to(device, dtype)

    # --------------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension
    # --------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # --------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) -> shape [B,N,N,hidden_dim]
    # --------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)
    right = F.linear(x, right_proj_weight)

    # --------------------------------------------------------------------
    # 3️⃣ Sigmoid gates
    # --------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # --------------------------------------------------------------------
    # 4️⃣ Apply mask AND gates in a Triton kernel (in‑place fused)
    # --------------------------------------------------------------------
    # Ensure mask is float for multiplication
    mask_fp = mask.to(dtype)

    # Allocate output buffers
    masked_left = torch.empty_like(left)
    masked_right = torch.empty_like(right)

    # Strides (in element counts, not bytes)
    s_left = left.stride()
    s_right = right.stride()
    s_mask = mask_fp.stride()
    s_lgate = left_gate.stride()
    s_rgate = right_gate.stride()

    # Kernel launch configuration
    total_ij = N * N
    grid = (B * total_ij,)

    # Choose a block size that divides hidden_dim nicely (32 works for 128, 384)
    BLOCK = 32

    _mask_gate_kernel[grid](
        masked_left, masked_right,
        left, right,
        mask_fp,
        left_gate, right_gate,
        B, N, hidden_dim,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_mask[0], s_mask[1], s_mask[2],
        s_lgate[0], s_lgate[1], s_lgate[2], s_lgate[3],
        s_rgate[0], s_rgate[1], s_rgate[2], s_rgate[3],
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------------
    # 5️⃣ O(N³·C) reduction: per‑channel batched matrix multiplication
    #     out_c = left_c @ right_cᵀ
    # --------------------------------------------------------------------
    # Reshape to (B*hidden_dim, N, N) where each row is a separate channel
    left_mat = masked_left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    # right needs to be transposed on the last two axes (k,j)
    right_mat = masked_right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat)          # (B*hidden_dim, N, N)

    # Restore original layout: [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # --------------------------------------------------------------------
    # 6️⃣ Output LayerNorm, gate, and final linear projection
    # --------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    output = F.linear(out, to_out_weight)

    return output.to(torch.float32)
