"""
TriMul "outgoing" kernel (AlphaFold3) – Triton implementation.

The forward pass consists of:
1. LayerNorm over the channel dimension (dim).
2. Two linear projections (left/right) followed by optional masking.
3. Gated linear projections (left/right gates) and an output gate.
4. Core O(N³) operation:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to a batched matrix multiplication per hidden channel.
   This is implemented in a Triton kernel that tiles the i, j and hidden (d)
   dimensions (M×N×D) while reducing over k.
5. LayerNorm over the hidden dimension, gating, and final linear projection.

The kernel computes a tile of shape (BLOCK_M, BLOCK_N, BLOCK_D) per program
instance, accumulating a vector of length BLOCK_D for each (i,j) pair.
Masking and gating are applied in PyTorch before launching the kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel for the O(N³) reduction
# ----------------------------------------------------------------------------- #
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # batch, seq_len, hidden_dim
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,   # right strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)

    # number of tiles per dimension
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    num_d = tl.cdiv(H, BLOCK_D)
    blocks_per_batch = num_m * num_n * num_d

    # decode batch index
    b_idx = pid // blocks_per_batch
    pid = pid % blocks_per_batch

    # decode i, j, d tile indices
    i_idx = pid // (num_n * num_d)
    pid = pid % (num_n * num_d)
    j_idx = pid // num_d
    d_idx = pid % num_d

    # compute offsets inside the tile
    offs_i = i_idx * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_idx * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_d = d_idx * BLOCK_D + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < H

    # accumulator for the output tile (M, N, D)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # loop over reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile -----
        # shape (M, K, D)
        left_ptrs = (
            left_ptr
            + b_idx * stride_l_b
            + offs_i[:, None, None] * stride_l_i
            + offs_k[None, :, None] * stride_l_k
            + offs_d[None, None, :] * stride_l_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )

        # ----- load right tile -----
        # shape (N, K, D)
        right_ptrs = (
            right_ptr
            + b_idx * stride_r_b
            + offs_j[:, None, None] * stride_r_i
            + offs_k[None, :, None] * stride_r_k
            + offs_d[None, None, :] * stride_r_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )

        # ----- reduction across the K tile -----
        # loop over the inner K dimension (unrolled)
        for kk in range(0, BLOCK_K):
            left_kk = left[:, kk, :]          # (M, D)
            right_kk = right[:, kk, :]        # (N, D)
            # broadcast to (M, N, D) and accumulate
            acc += left_kk[:, None, :] * right_kk[None, :, :]

    # ----- write back the result -----
    out_ptrs = (
        out_ptr
        + b_idx * stride_o_b
        + offs_i[:, None, None] * stride_o_i
        + offs_j[None, :, None] * stride_o_j
        + offs_d[None, None, :] * stride_o_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )

# ----------------------------------------------------------------------------- #
# Host entry point
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul "outgoing" module with a Triton kernel for the
    expensive pairwise reduction. All other operations (LayerNorm, Linear,
    gating) are performed with PyTorch.

    Args:
        data: Tuple of (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask_tensor : torch.Tensor of shape [B, N, N] (may be unused)
            - weights_dict: mapping from weight name to torch.Tensor
            - config_dict : contains "dim", "hidden_dim", "nomask"

    Returns:
        output tensor of shape [B, N, N, dim] (float32)
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------- LayerNorm (input) --------------------------- #
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # --------------------------- Linear projections --------------------------- #
    left_proj_w = weights["left_proj.weight"].to(device)
    right_proj_w = weights["right_proj.weight"].to(device)

    left = F.linear(x, left_proj_w, bias=None)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w, bias=None)

    # --------------------------- Optional mask ------------------------------- #
    if not nomask:
        # mask shape [B, N, N] -> [B, N, N, 1]
        mask_exp = mask.unsqueeze(-1).to(dtype)
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------- Gating -------------------------------------- #
    left_gate_w = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w = weights["out_gate.weight"].to(device)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w, bias=None))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w, bias=None))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w, bias=None))

    left = left * left_gate
    right = right * right_gate

    # --------------------------- Triton reduction ----------------------------- #
    B, N, _, H = left.shape  # H == hidden_dim
    out = torch.empty_like(left)

    # compute strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile sizes – chosen to stay within register limits
    BLOCK_M = 8   # i dimension
    BLOCK_N = 8   # j dimension
    BLOCK_K = 32  # reduction over k
    BLOCK_D = 8   # hidden dimension slice

    # total number of program instances
    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    num_d = (H + BLOCK_D - 1) // BLOCK_D
    total_blocks = B * num_m * num_n * num_d
    grid = (total_blocks,)

    trimul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------- Post‑processing ----------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    to_out_w = weights["to_out.weight"].to(device)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w,
                       bias=to_out_norm_b, eps=1e-5)
    out = out * out_gate
    output = F.linear(out, to_out_w, bias=None)   # final projection to dim

    return output
