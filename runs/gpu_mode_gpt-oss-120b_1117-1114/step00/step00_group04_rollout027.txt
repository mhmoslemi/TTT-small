def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps
    )

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask
    # mask shape: [B, N, N]
    mask_ = mask.unsqueeze(-1)  # shape [B, N, N, 1]
    left = left * mask_
    right = right * mask_

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Compute out via Triton
    out = trimul_triton(left, right)  # defined below

    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps
    )
    # out_gate scaling
    out = out * out_gate

    # final linear projection
    output = torch.nn.functional.linear(out, weights["to_out.weight"])
    return output

import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_bs, stride_left_i, stride_left_k, stride_left_d,
    stride_right_bs, stride_right_i, stride_right_k, stride_right_d,
    stride_out_bs, stride_out_i, stride_out_j, stride_out_d,
    d_offset,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr):
    # Program IDs
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Offsets within block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_d = tl.arange(0, BLOCK_D) + d_offset

    # Masks for bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # Loop over k dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        # Compute k indices
        k_start = kb * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block [M, K, D]
        left_ptrs = left_ptr + \
            pid_b * stride_left_bs + \
            offs_i[:, None, None] * stride_left_i + \
            offs_k[None, :, None] * stride_left_k + \
            offs_d[None, None, :] * stride_left_d
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0)

        # Load right block [K, N, D] (note j dimension)
        right_ptrs = right_ptr + \
            pid_b * stride_right_bs + \
            offs_j[None, :, None] * stride_right_i + \
            offs_k[:, None, None] * stride_right_k + \
            offs_d[None, None, :] * stride_right_d
        right = tl.load(right_ptrs, mask=mask_j[None, :, None] & mask_k[:, None, None] & mask_d[None, None, :], other=0.0)

        # Compute dot for each d
        for d in range(0, BLOCK_D):
            # left_slice: (M, K), right_slice: (K, N)
            left_slice = left[:, :, d]    # shape (M, K)
            right_slice = right[:, :, d]  # shape (K, N)
            # tl.dot returns (M,N)
            acc[:, :, d] += tl.dot(left_slice, right_slice)

    # Store results
    out_ptrs = out_ptr + \
        pid_b * stride_out_bs + \
        offs_i[:, None, None] * stride_out_i + \

"""
TriMul (outgoing) implementation with the cubic contraction
performed in a custom Triton kernel. The heavy `einsum` 
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is off‑loaded to Triton. All other linear layers, LayerNorms and gating
are executed with PyTorch (torch.nn.functional). The kernel tiles the
(i, j) pairwise dimensions (BLOCK_M×BLOCK_N) and the hidden dimension
into small blocks to keep register pressure low while iterating over the
k‑dimension in blocks (BLOCK_K). The kernel works for any batch size,
sequence length and hidden dimension that fit in GPU memory.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the cubic contraction (outgoing update)
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_bs, stride_left_i, stride_left_k, stride_left_d,
    stride_right_bs, stride_right_i, stride_right_k, stride_right_d,
    stride_out_bs, stride_out_i, stride_out_j, stride_out_d,
    d_offset,                           # runtime offset for the hidden dim tile
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] for a
    tile of (i,j) and a slice of d (hidden dimension)."""

    pid_b = tl.program_id(0)               # batch index
    pid_i = tl.program_id(1)               # i‑tile index
    pid_j = tl.program_id(2)               # j‑tile index

    # --------------------------------------------------------------
    # Offsets for this tile
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)          # (M,)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)          # (N,)
    offs_d = tl.arange(0, BLOCK_D) + d_offset                # (D_tile,)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # accumulator: (M, N, D_tile) in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k
    # --------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_start = kb * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)          # (K,)
        mask_k = offs_k < N

        # ------------------ load left[ b, i, k, d ] -----------------
        # shape after broadcasting: (M, K, D_tile)
        left_ptrs = left_ptr + \
            pid_b * stride_left_bs + \
            offs_i[:, None, None] * stride_left_i + \
            offs_k[None, :, None] * stride_left_k + \
            offs_d[None, None, :] * stride_left_d
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )   # (M, K, D_tile)

        # ------------------ load right[ b, j, k, d ] ----------------
        # we need the layout (K, N, D_tile) → k is first axis
        right_ptrs = right_ptr + \
            pid_b * stride_right_bs + \
            offs_j[None, :, None] * stride_right_i + \
            offs_k[:, None, None] * stride_right_k + \
            offs_d[None, None, :] * stride_right_d
        right = tl.load(
            right_ptrs,
            mask=mask_j[None, :, None] & mask_k[:, None, None] & mask_d[None, None, :],
            other=0.0,
        )   # (K, N, D_tile)

        # ------------------ per‑d dot product -----------------------
        # left[:, :, d] : (M, K)   right[:, :, d] : (K, N)
        for d in range(0, BLOCK_D):
            acc[:, :, d] += tl.dot(left[:, :, d], right[:, :, d])

    # --------------------------------------------------------------
    # Write back the result
    # --------------------------------------------------------------
    out_ptrs = out_ptr + \
        pid_b * stride_out_bs + \
        offs_i[:, None, None] * stride_out_i + \
        offs_j[None, :, None] * stride_out_j + \
        offs_d[None, None, :] * stride_out_d

    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )
# ----------------------------------------------------------------------


def _trimul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """Launches `trimul_outgoing_kernel` to compute the cubic contraction.

    Args:
        left:  Tensor of shape [B, N, N, D] (float32, contiguous)
        right: Tensor of shape [B, N, N, D] (float32, contiguous)

    Returns:
        out: Tensor of shape [B, N, N, D] (float32)
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, D = left.shape
    out = torch.empty_like(left)

    # --------------------------------------------------------------
    # Strides (in elements, not bytes)
    # --------------------------------------------------------------
    sL_bs, sL_i, sL_k, sL_d = left.stride()
    sR_bs, sR_i, sR_k, sR_d = right.stride()
    sO_bs, sO_i, sO_j, sO_d = out.stride()

    # ------------------------------------------------------------------
    # Tiling parameters – chosen to keep register usage modest.
    # ------------------------------------------------------------------
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 16
    BLOCK_D = 8   # hidden‑dim tile size

    grid = (B, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    for d_off in range(0, D, BLOCK_D):
        cur_D = min(BLOCK_D, D - d_off)
        trimul_outgoing_kernel[
            grid
        ](
            left,
            right,
            out,
            B,
            N,
            D,
            sL_bs,
            sL_i,
            sL_k,
            sL_d,
            sR_bs,
            sR_i,
            sR_k,
            sR_d,
            sO_bs,
            sO_i,
            sO_j,
            sO_d,
            d_off,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=cur_D,
        )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args:
        data: Tuple[input_tensor, mask, weights, config]
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (float, 0/1)
            - weights: dict with named weight tensors
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        output: torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Mask (if any)
    # ------------------------------------------------------------------
    mask_f = mask.to(x.dtype).unsqueeze(-1)                 # [B,N,N,1]
    left = left * mask_f
    right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cubic contraction via Triton
    # ------------------------------------------------------------------
    # Ensure contiguous tensors for deterministic stride handling
    left = left.contiguous()
    right = right.contiguous()
    out = _trimul_outgoing_triton(left, right)   # shape [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 6) Output LayerNorm + gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7) Final linear projection back to original dim
    # ------------------------------------------------------------------
    output = F.linear(out, weights["to_out.weight"])
    return output
