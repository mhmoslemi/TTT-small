"""
TriMul forward pass (outgoing version) with the O(N³·H) contraction off‑loaded to a
custom Triton kernel.

Steps
-----
1. Layer‑norm over the input channel dimension.
2. Linear projections (left/right) and gating linear layers (left/right/out).
3. Apply mask (if provided) and element‑wise gating.
4. Fuse the core contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   using a Triton kernel.  The batch and hidden dimensions are merged into a
   single “head” dimension, so the kernel performs a batched matrix‑multiply
   left @ rightᵀ for each head.
5. Layer‑norm over the hidden dimension, multiply by the output gate and a final
   linear projection back to the original channel dimension.
All tensors stay on the CUDA device; only the heavy N³ contraction is executed
in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,          # const pointer to left tensor   [heads, N, N] (fp16)
    right_ptr,         # const pointer to right tensor  [heads, N, N] (fp16)
    out_ptr,           # pointer to output tensor       [heads, N, N] (fp32)
    N,                 # sequence length (int)
    heads,             # number of heads = B * hidden_dim (int)
    stride_lh, stride_li, stride_lk,  # left strides
    stride_rh, stride_rj, stride_rk,  # right strides
    stride_oh, stride_oi, stride_oj,  # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication: out[h, i, j] = Σ_k left[h, i, k] * right[h, j, k]
    with left/right stored as (heads, N, N) and right accessed in transposed
    order to avoid an explicit transpose.
    """
    pid_h = tl.program_id(0)      # head index
    pid_m = tl.program_id(1)      # block-row index (i)
    pid_n = tl.program_id(2)      # block-col index (j)

    # base pointers for this head
    left_base  = left_ptr  + pid_h * stride_lh
    right_base = right_ptr + pid_h * stride_rh
    out_base   = out_ptr   + pid_h * stride_oh

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A tile: left[h, i, k] -> shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_m[:, None] * stride_li + offs_k[None, :] * stride_lk
        a = tl.load(a_ptrs, mask=(mask_m[:, None] & mask_k[None, :]), other=0.0)
        a = a.to(tl.float32)

        # load B tile transposed: we need right[h, j, k] as (k, j)
        # original layout is (heads, j, k); we load with k as leading dim
        b_ptrs = right_base + offs_k[:, None] * stride_rk + offs_n[None, :] * stride_rj
        b = tl.load(b_ptrs, mask=(mask_k[:, None] & mask_n[None, :]), other=0.0)
        b = b.to(tl.float32)

        # acc += a @ b   (a: MxK, b: KxN)
        acc += tl.dot(a, b)

    # write back result
    c_ptrs = out_base + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------- #
    #   Load weights (they are already on the correct device)
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # --------------------------------------------------------------------- #
    #   1. Input layer‑norm
    # --------------------------------------------------------------------- #
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # --------------------------------------------------------------------- #
    #   2. Linear projections
    # --------------------------------------------------------------------- #
    left  = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B,N,N,hidden_dim]

    # --------------------------------------------------------------------- #
    #   3. Apply mask (if any)
    # --------------------------------------------------------------------- #
    if mask is None:
        mask = torch.ones_like(input_tensor[..., 0], device=device, dtype=x.dtype)
    mask = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
    left  = left * mask
    right = right * mask

    # --------------------------------------------------------------------- #
    #   4. Gating
    # --------------------------------------------------------------------- #
    left_gate  = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate   = F.linear(x, out_gate_weight).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    #   5. Core contraction via Triton (FP16 inputs, FP32 accumulation)
    # --------------------------------------------------------------------- #
    # Cast to FP16 to reduce memory bandwidth
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape          # H == hidden_dim
    heads = B * H

    # reshape to [heads, N, N] where head = (batch, hidden)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat  = left_perm.view(heads, N, N)
    right_flat = right_perm.view(heads, N, N)

    out_flat = torch.empty((heads, N, N), dtype=torch.float32, device=device)

    # block sizes (tuned for typical A100 / B200 GPUs)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (heads,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        heads,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    #   6. Output layer‑norm, gating and final linear projection
    # --------------------------------------------------------------------- #
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
