def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # retrieve weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    B, N, _, _ = x.shape

    # flatten for linear
    x_flat = x.reshape(-1, dim)  # (B*N*N, dim)

    left_flat = F.linear(x_flat, left_proj_weight)  # (B*N*N, hidden_dim)
    right_flat = F.linear(x_flat, right_proj_weight)
    left_gate_flat = torch.sigmoid(F.linear(x_flat, left_gate_weight)).reshape(B, N, N, hidden_dim)
    right_gate_flat = torch.sigmoid(F.linear(x_flat, right_gate_weight)).reshape(B, N, N, hidden_dim)
    out_gate_flat = torch.sigmoid(F.linear(x_flat, out_gate_weight)).reshape(B, N, N, hidden_dim)

    left = left_flat.view(B, N, N, hidden_dim)
    right = right_flat.view(B, N, N, hidden_dim)

    # Apply mask and gates
    if not nomask:
        mask_exp = mask.unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate_flat
    right = right * right_gate_flat

    # Cast to half for matmul
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to bring hidden dim as batch for matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Transpose right for matmul (swap last two dims)
    right_T = right_perm.permute(0, 1, 3, 2).contiguous()  # (B, H, N, N) with axes (batch, hidden, k, j)

    # flatten batch*hidden
    B_h = B * hidden_dim
    left_mat = left_perm.view(B_h, N, N)
    right_mat = right_T.view(B_h, N, N)

    # Output buffer
    out_mat = torch.empty_like(left_mat)  # fp16

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B_h, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    batched_matmul_kernel[grid](
        left_mat, right_mat, out_mat,
        B_h, N, N, N,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,hidden_dim)

    # To_out_norm (layernorm over hidden_dim)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Multiply by out_gate
    out = out * out_gate_flat

    # Final linear projection to dim
    out_flat = out.reshape(-1, hidden_dim)
    out_proj_flat = F.linear(out_flat, to_out_weight)  # shape (B*N*N, dim)
    out_proj = out_proj_flat.view(B, N, N, dim)

"""
TriMul (outgoing) implementation with a Triton‑accelerated batched matrix multiplication.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B, N, N, D]).
2. Linear projections → left/right (to hidden_dim) and three gate
   projections (left_gate, right_gate, out_gate).  Gates use a sigmoid.
3. Apply optional mask, then element‑wise multiply left/right with their
   respective gates.
4. Cast left/right to float16 and permute so that the hidden dimension becomes
   part of the batch dimension: shape (B*H, N, N).
5. Compute the contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to a batched matrix multiplication
        (B*H, N, N) @ (B*H, N, N)^T .
   This is performed by a custom Triton kernel `batched_matmul_kernel`.
6. Reshape back to (B, N, N, H), apply a second LayerNorm over H,
   multiply by the out‑gate, and finally linearly project back to D.
7. Return the result in float32.

Only the O(N³·H) contraction is done in Triton; all other ops use
high‑performance PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: batched matrix multiplication (C = A @ B)                       #
# A : (batch, M, K)   B : (batch, K, N)   C : (batch, M, N)                     #
# Here M = N = K = sequence length.                                            #
# ----------------------------------------------------------------------------- #
@triton.jit
def batched_matmul_kernel(
    A, B, C,
    batch, M, N, K,
    stride_aba, stride_ai, stride_ak,
    stride_bba, stride_bk, stride_bj,
    stride_cca, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the output block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for A and B
        a_ptr = A + pid_batch * stride_aba + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        b_ptr = B + pid_batch * stride_bba + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj

        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0, dtype=tl.float16)

        # Multiply‑accumulate in FP32
        acc += tl.dot(a, b)

    # Store the result
    c_ptr = C + pid_batch * stride_cca + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptr, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------------- #
# Public entry point                                                             #
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Executes the TriMul forward pass.  Arguments:

    data = (input_tensor, mask, weights, config)

    - input_tensor: [B, N, N, D] torch tensor (cuda)
    - mask        : [B, N, N]   torch tensor (bool/int, cuda)
    - weights     : dict of model parameters (torch tensors)
    - config      : dict containing at least "dim", "hidden_dim", "nomask"
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------------- #
    # 1. Load weights
    # --------------------------------------------------------------------- #
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # --------------------------------------------------------------------- #
    # 2. LayerNorm on the input
    # --------------------------------------------------------------------- #
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    B, N, _, _ = x.shape

    # --------------------------------------------------------------------- #
    # 3. Linear projections and gates (float32)
    # --------------------------------------------------------------------- #
    x_flat = x.view(-1, dim)                                 # (B*N*N, D)

    left_flat = F.linear(x_flat, left_proj_w)                # (B*N*N, H)
    right_flat = F.linear(x_flat, right_proj_w)

    left_gate_flat = torch.sigmoid(F.linear(x_flat, left_gate_w))
    right_gate_flat = torch.sigmoid(F.linear(x_flat, right_gate_w))
    out_gate_flat = torch.sigmoid(F.linear(x_flat, out_gate_w))

    left = left_flat.view(B, N, N, hidden_dim)
    right = right_flat.view(B, N, N, hidden_dim)

    left_gate = left_gate_flat.view(B, N, N, hidden_dim)
    right_gate = right_gate_flat.view(B, N, N, hidden_dim)
    out_gate = out_gate_flat.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------------- #
    # 4. Apply mask and gates (still float32)
    # --------------------------------------------------------------------- #
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(dtype=left.dtype)   # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Cast to float16 for the heavy contraction
    # --------------------------------------------------------------------- #
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to put hidden dimension in front of the two spatial axes
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)  -> (i,k)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)  -> (j,k)

    # Transpose the last two axes of right so that we can do a regular matmul:
    # left: (B, H, i, k) ; right_T: (B, H, k, j)
    right_T = right_perm.permute(0, 1, 3, 2).contiguous()

    # Collapse batch and hidden dim for a single batched matmul
    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)      # (BH, M=N, K=N)
    right_mat = right_T.view(BH, N, N)      # (BH, K=N, N=j)

    # Allocate output tensor (float16)
    out_mat = torch.empty_like(left_mat)

    # --------------------------------------------------------------------- #
    # 6. Triton kernel launch: batched matmul C = A @ B
    # --------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        BH,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        BH,
        N,
        N,
        N,
        left_mat.stride(0),
        left_mat.stride(1),
        left_mat.stride(2),
        right_mat.stride(0),
        right_mat.stride(1),
        right_mat.stride(2),
        out_mat.stride(0),
        out_mat.stride(1),
        out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------------- #
    # 7. Reshape back to (B, N, N, H) and continue in FP32
    # --------------------------------------------------------------------- #
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)
    out = out.to(torch.float32)

    # --------------------------------------------------------------------- #
    # 8. Post‑contraction LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------------- #
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    out_flat = out.view(-1, hidden_dim)
    out_proj_flat = F.linear(out_flat, to_out_w)               # (B*N*N, D)
    out_final = out_proj_flat.view(B, N, N, dim)

    return out_final
