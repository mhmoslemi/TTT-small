import torch
import torch.nn.functional as F
import triton
import triton.language as tl

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    The core O(N^3) multiplication is fused in a Triton kernel that computes
    out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h].
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, D = x.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    # Cast to half for projections
    x_fp16 = x.to(torch.float16)

    # Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    # Use torch.nn.functional.linear: input shape (..., dim), weight shape (out_features, in_features)
    left = F.linear(x_fp16, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = F.linear(x_fp16, right_proj_weight)

    # Apply mask if provided
    # mask shape [B,N,N], dtype bool or float
    if mask is not None:
        mask = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask
    else:
        mask = torch.ones_like(left)  # no mask case (just for gating? Not needed)
    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    # Compute gate values (sigmoid)

        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,

    # Apply out_gate (should be broadcastable)
    out = out * out_gate

    # Final linear projection
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)  # shape [B,N,N,dim]

    return out

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * H
    if pid >= total:
        return
    b = pid // H
    h = pid % H

    # Base pointer offsets for this (b,h)
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block
        # left[i,k] at offsets: i*stride_l_i + k*stride_l_k

        acc += tl.dot(left, right)

    # write out
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)                 # index over (batch, hidden)
    # map linear pid to (b, h)
    b = pid // H
    h = pid % H

    # base pointers for this (b, h) instance
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # 2‑D grid over the i / j output matrix
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction axis k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right block (k, j) required for the matmul: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate the dot product
        acc += tl.dot(left, right)

    # write the result
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------
# Triton‑accelerated TriMul forward.
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [B, N, N, D] torch.float32
            - mask_tensor  : [B, N, N] torch.float32 / torch.bool
            - weights_dict : mapping from weight name to torch.Tensor
            - config_dict  : contains "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # unpack inputs
    x, mask, weights, config = data
    device = x.device

    B, N, _, D = x.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -------------------------------------------------
    # 1. Input LayerNorm
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # -------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]

    left  = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)  # [B, N, N, hidden_dim]

    # -------------------------------------------------
    # 3. Optional mask
    # -------------------------------------------------
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)        # [B, N, N, 1]
        left  = left * mask
        right = right * mask

    # -------------------------------------------------
    # 4. Gating values (sigmoid)
    # -------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5. Core O(N³) multiplication via Triton
    # -------------------------------------------------
    left  = left.contiguous()
    right = right.contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # block sizes (tuned for Ampere B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_instances = B * hidden_dim
    grid = (
        total_instances,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # strides (torch uses row‑major layout)
    s_l_b, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_b, s_r_i, s_r_k, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        s_l_b, s_l_i, s_l_k, s_l_h,
        s_r_b, s_r_i, s_r_k, s_r_h,
        s_o_b, s_o_i, s_o_j, s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------
    # 6. Post‑processing: LayerNorm, gating, final projection
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w,
                       bias=to_out_norm_b, eps=eps)

    out = out * out_gate

    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)      # [B, N, N, dim]

    return out
