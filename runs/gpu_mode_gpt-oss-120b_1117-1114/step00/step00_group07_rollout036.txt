@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak, stride_ad,
    stride_bm, stride_bi, stride_bk, stride_bd,
    stride_cm, stride_ci, stride_cj, stride_cd,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HIDDEN: tl.constexpr
):
    pid_m = tl.program_id(0)  # i tile
    pid_n = tl.program_id(1)  # j tile
    pid_z = tl.program_id(2)  # batch*hidden index
    
    # Compute batch and channel index
    b = pid_z // HIDDEN
    d = pid_z % HIDDEN
    
    # Compute base pointers for this batch-channel
    a_base = a_ptr + b * stride_am + d * stride_ad
    b_base = b_ptr + b * stride_bm + d * stride_bd
    c_base = c_ptr + b * stride_cm + d * stride_cd
    
    # Compute tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    
    # mask for out-of-range
    mask_m = offs_m < M
    mask_n = offs_n < N
    
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # Loop over K dimension
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_blocks):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K
        
        # Load A tile [M, K]
        a_ptrs = a_base + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak, stride_ad,
    stride_bm, stride_bi, stride_bk, stride_bd,
    stride_cm, stride_ci, stride_cj, stride_cd,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HIDDEN: tl.constexpr
):
    """
    Triton kernel for the core O(N^3) reduction of the outgoing TriMul operator.
    Computes:
        C[b,i,j,d] = sum_k A[b,i,k,d] * B[b,j,k,d]
    where A == left, B == right (already gated and masked).
    """
    pid_m = tl.program_id(0)  # i tile
    pid_n = tl.program_id(1)  # j tile
    pid_z = tl.program_id(2)  # combined batch * hidden index

    # decompose pid_z into batch and hidden index
    batch_idx = pid_z // HIDDEN
    hidden_idx = pid_z % HIDDEN

    # base pointers for this (batch, hidden) pair
    a_base = a_ptr + batch_idx * stride_am + hidden_idx * stride_ad
    b_base = b_ptr + batch_idx * stride_bm + hidden_idx * stride_bd
    c_base = c_ptr + batch_idx * stride_cm + hidden_idx * stride_cd

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for k in range(tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # load B tile transposed: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bi
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        acc += tl.dot(a, b)

    # store result tile
    c_ptrs = c_base + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul) operator.
    Heavy O(N^3) reduction (einsum) is performed by a Triton kernel.
    Arguments:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        output_tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -------------------------------------------------
    # LayerNorm on the input (last dimension = dim)
    # -------------------------------------------------
    x = input_tensor
    x = F.layer_norm(x,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=eps)

    # -------------------------------------------------
    # Linear projections (no bias)
    # -------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------
    # Apply mask if present
    # -------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)   # broadcast over hidden
        left = left * mask
        right = right * mask

    # -------------------------------------------------
    # Gating (sigmoid)
    # -------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # Cast gates to same dtype as left/right for elementwise multiplication
    left_gate = left_gate.to(left.dtype)
    right_gate = right_gate.to(right.dtype)

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # Prepare tensors for the Triton reduction
    # -------------------------------------------------
    # Cast to half-precision for the kernel (saves bandwidth)
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    bs, N, _, _ = left_h.shape  # N == seq_len
    # Allocate output accumulator (float32)
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=x.device)

    # Strides (in elements)
    stride_am = left_h.stride(0)
    stride_ai = left_h.stride(1)
    stride_ak = left_h.stride(2)
    stride_ad = left_h.stride(3)

    # Right tensor has identical strides
    stride_bm = stride_am
    stride_bi = stride_ai
    stride_bk = stride_ak
    stride_bd = stride_ad

    stride_cm = out.stride(0)
    stride_ci = out.stride(1)
    stride_cj = out.stride(2)
    stride_cd = out.stride(3)

    # Triton launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        bs * hidden_dim,
    )
    tri_mul_kernel[grid](
        left_h,
        right_h,
        out,
        N, N, N,
        stride_am, stride_ai, stride_ak, stride_ad,
        stride_bm, stride_bi, stride_bk, stride_bd,
        stride_cm, stride_ci, stride_cj, stride_cd,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN=hidden_dim,
    )

    # -------------------------------------------------
    # Output post‑processing (layer‑norm, gating, final linear)
    # -------------------------------------------------
    # Normalize across hidden dimension
    out_norm = F.layer_norm(out,
                            (hidden_dim,),
                            weight=weights["to_out_norm.weight"],
                            bias=weights["to_out_norm.bias"],
                            eps=eps)

    # Apply out gate (float32)
    out_gated = out_norm * out_gate

    # Final projection back to original dimension
    final = F.linear(out_gated, weights["to_out.weight"])

    return final

import torch

import triton.language as tl

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak, stride_ad,
    stride_bm, stride_bi, stride_bk, stride_bd,
    stride_cm, stride_ci, stride_cj, stride_cd,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HIDDEN: tl.constexpr,
):
    """
    Triton kernel implementing the core O(N³) reduction of the outgoing
    Triangle Multiplicative Update (TriMul) operator.

    Computes for each batch `b` and hidden channel `d`:
        C[b,i,j,d] = Σₖ A[b,i,k,d] * B[b,j,k,d]
    where A == left, B == right (already masked and gated).
    """
    pid_m = tl.program_id(0)      # i‑tile
    pid_n = tl.program_id(1)      # j‑tile
    pid_z = tl.program_id(2)      # combined (batch, hidden) index

    # Decompose pid_z into batch and hidden indices
    batch_idx = pid_z // HIDDEN
    hidden_idx = pid_z % HIDDEN

    # Base pointers for the current (batch, hidden) plane
    a_base = a_ptr + batch_idx * stride_am + hidden_idx * stride_ad
    b_base = b_ptr + batch_idx * stride_bm + hidden_idx * stride_bd
    c_base = c_ptr + batch_idx * stride_cm + hidden_idx * stride_cd

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K in BLOCK_K sized chunks
    for k in range(tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (M, K)   -> left[b,i,k,d]
        a_ptrs = a_base + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # B tile transposed: (K, N)   -> right[b,j,k,d] accessed as (k,j)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bi
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        acc += tl.dot(a, b)

    # Store the computed tile into C
    c_ptrs = c_base + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    The expensive einsum is executed by a custom Triton kernel.
    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
          - input_tensor: [B, N, N, dim] (float32)
          - mask_tensor:  [B, N, N] (bool or float)
          - weights_dict: contains all linear / layer‑norm parameters
          - config_dict:  {"dim": int, "hidden_dim": int, ...}
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5  # same default as torch.nn.LayerNorm

    # ---------- LayerNorm on the input ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ---------- Linear projections (no bias) ----------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # broadcast over hidden
        left = left * mask
        right = right * mask

    # ---------- Gates ----------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # Cast gates to match left/right dtype before element‑wise multiply
    left_gate = left_gate.to(left.dtype)
    right_gate = right_gate.to(right.dtype)

    right = right * right_gate

    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    stride_am = left_h.stride(0)

    stride_bm = stride_am
    stride_bi = stride_ai
    stride_bk = stride_ak
    stride_bd = stride_ad

    stride_ci = out.stride(1)
    stride_cj = out.stride(2)
    stride_cd = out.stride(3)

    BLOCK_N = 64
    BLOCK_K = 64

        triton.cdiv(N, BLOCK_M),

        N,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_ad,
        stride_bm,
        stride_bi,
        stride_bk,
        stride_bd,
        stride_cm,
        stride_ci,
        stride_cj,
        stride_cd,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN=hidden_dim,
    )

    # LayerNorm over hidden dimension
    out_norm = F.layer_norm(

        (hidden_dim,),

        bias=weights["to_out_norm.bias"],
