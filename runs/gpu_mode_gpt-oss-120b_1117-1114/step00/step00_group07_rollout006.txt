"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,D]).
2. Linear projections (no bias) → `left` and `right`  (shape [B,N,N,H]).
3. Optional mask multiplication (broadcasted on the hidden dim).
4. Gating: sigmoid(linear(x)) for left, right and output gates.
   Apply left/right gates element‑wise to `left`/`right`.
5. Compute the core O(N³·H) tensor contraction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix‑multiply per hidden channel using a
   Triton kernel that tiles the (i,j) plane and the reduction dimension k.
6. Apply a second Layer‑norm on the hidden dimension, multiply by the
   output gate, and finally a linear projection back to the original
   dimension D.
All operations are performed in float32; the kernel works on a single
batch at a time (B ≤ 2) to keep the launch simple.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_i, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    """Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] for a single batch."""
    pid_i = tl.program_id(0)   # tile row   (i)
    pid_j = tl.program_id(1)   # tile col   (j)
    pid_h = tl.program_id(2)   # tile hidden channel block

    # Offsets for the current tile
    off_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    off_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    off_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_i = off_i < N
    mask_j = off_j < N
    mask_h = off_h < H

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        off_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = off_k < N

        # Masks for loads (broadcasted)
        mask_l = mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]
        mask_r = mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]

        # Pointers for left[i,k,h]
        left_ptrs = (left_ptr
                     + off_i[:, None, None] * stride_left_i
                     + off_k[None, :, None] * stride_left_k
                     + off_h[None, None, :] * stride_left_h)
        # Pointers for right_T[k,j,h]  (right accessed as (k,j,h) to avoid a transpose)
        right_ptrs = (right_ptr
                      + off_k[:, None, None] * stride_right_k
                      + off_j[None, :, None] * stride_right_i
                      + off_h[None, None, :] * stride_right_h)

        # Load tiles
        left_tile = tl.load(left_ptrs, mask=mask_l, other=0.0)   # (M, K, H)
        right_tile = tl.load(right_ptrs, mask=mask_r, other=0.0) # (K, N, H)

        # For each hidden channel in the tile, perform a small matrix‑multiply
        for h_idx in range(0, BLOCK_H):
            # Guard against out‑of‑bounds hidden channels
            if not mask_h[h_idx]:
                continue
            l_mat = left_tile[:, :, h_idx]   # (M, K)
            r_mat = right_tile[:, :, h_idx]   # (K, N)
            acc[:, :, h_idx] += tl.dot(l_mat, r_mat)   # (M, N)

    # Store the result with proper masking
    out_ptrs = (out_ptr
                + off_i[:, None, None] * stride_out_i
                + off_j[None, :, None] * stride_out_j
                + off_h[None, None, :] * stride_out_h)
    mask_o = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
    tl.store(out_ptrs, acc, mask=mask_o)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D]
        - mask         : torch.Tensor [B, N, N] (may be all‑ones)
        - weights      : dict of weight tensors
        - config       : dict containing 'dim', 'hidden_dim', ...

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32, on the same device).
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------
    # 1. Basic shape info
    # ------------------------------------------------------------------
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]          # hidden dimension (c_z in paper)
    device = input_tensor.device
    dtype = torch.float32

    # ------------------------------------------------------------------
    # 2. LayerNorm on the input (dim = D)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(D,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left  = F.linear(x, left_proj_weight)   # [B,N,N,H]
    right = F.linear(x, right_proj_weight)  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(dtype)   # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 5. Gates (sigmoid of linear layers, no bias)
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # [B,N,N,H]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core contraction via Triton kernel (batchwise)
    # ------------------------------------------------------------------
    out = torch.empty_like(left, dtype=dtype, device=device)   # [B,N,N,H]

    # Choose tile sizes (tuned for the A100/B200 architecture)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    # Hidden‑dim tile – small enough to fit in registers
    if H <= 128:
        BLOCK_H = 8
    elif H <= 256:
        BLOCK_H = 8
    else:
        BLOCK_H = 4

    # Strides for left/right/out (in elements, not bytes)
    stride_left_i = left.stride(1)   # over axis i
    stride_left_k = left.stride(2)   # over axis k
    stride_left_h = left.stride(3)   # over hidden

    stride_right_i = right.stride(1) # original i (will become j after transpose)
    stride_right_k = right.stride(2) # original k
    stride_right_h = right.stride(3)

    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    for b in range(B):
        # Slice the batch dimension; Triton kernel works on a single batch
        left_batch  = left[b]
        right_batch = right[b]
        out_batch   = out[b]

        grid = (
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            triton.cdiv(H, BLOCK_H),
        )
        trimul_outgoing_kernel[grid](
            left_batch,
            right_batch,
            out_batch,
            N,
            H,
            stride_left_i,
            stride_left_k,
            stride_left_h,
            stride_right_i,
            stride_right_k,
            stride_right_h,
            stride_out_i,
            stride_out_j,
            stride_out_h,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H,
        )

    # ------------------------------------------------------------------
    # 7. Post‑processing: second LayerNorm, output gate, final linear
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]   # shape [D, H]

    # LayerNorm over hidden dimension H
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                 # element‑wise gating
    output = F.linear(out, to_out_weight)  # [B,N,N,D]

    return output
```
