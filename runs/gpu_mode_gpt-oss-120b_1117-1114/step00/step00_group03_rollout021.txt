def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # row tile id
    pid_n = tl.program_id(1)   # col tile id
    pid_z = tl.program_id(2)   # batch*hidden id
    
    # compute block offsets
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    
    # compute batch and hidden index
    b = pid_z // H
    h = pid_z % H
    
    # compute pointers base
    # compute term for batch offset
    left_batch_offset = b * stride_left_b
    right_batch_offset = b * stride_right_b
    out_batch_offset = b * stride_out_b
    
    # base pointer for this hidden dimension
    left_base = left_ptr + left_batch_offset + h * stride_left_d
    right_base = right_ptr + right_batch_offset + h * stride_right_d
    out_base = out_ptr + out_batch_offset + h * stride_out_d
    
    # Determine bounds for M,N
    N_M = N
    N_N = N
    
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # Loop over K dimension
    for k in range(0, N, BLOCK_K):
        k_offset = k + tl.arange(0, BLOCK_K)
        # mask for k
        k_mask = k_offset < N
        # Load left block
        a_ptrs = left_base + (i_offset[:, None] * stride_left_i) + (k_offset[None, :] * stride_left_k)

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    # Norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    # compute mean and var
    x = input_tensor
    # LayerNorm across last dim
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    inv_std = torch.rsqrt(var + eps)
    x_norm = (x - mean) * inv_std * norm_weight + norm_bias
    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    # using torch.nn.functional.linear
    left = torch.nn.functional.linear(x_norm, left_proj_weight)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)
    # Apply mask
    mask = mask_tensor.unsqueeze(-1).type_as(left)
    left = left * mask
    right = right * mask
    # Gating
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate
    # cast half for kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    # Allocate output float32
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32, device=device)
    # strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left_fp16.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right_fp16.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out_fp32.stride()

        stride_left_d,

        stride_right_k,

        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

"""
TriMul (outgoing) forward implementation with a fused Triton kernel.

The heavy part of TriMul is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched matrix multiplication
    out_{b,d} = left_{b,:, :, d} @ right_{b,:, :, d}^T .
The kernel computes this matmul for every batch and hidden dimension
using tiled loads and fp32 accumulation (inputs are fp16 for bandwidth).
All other operations (LayerNorm, linear projections, gating) are done in
PyTorch.
"""
import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    Grid:
        (grid_m, grid_n, grid_z) where
            grid_m = ceil(N / BLOCK_M)
            grid_n = ceil(N / BLOCK_N)
            grid_z = B * H   (each (b,d) pair)
    """
    pid_m = tl.program_id(0)               # tile row
    pid_n = tl.program_id(1)               # tile col
    pid_z = tl.program_id(2)               # combined batch & hidden index

    # compute offsets within the output matrix tile
    i_off = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # decode batch and hidden indices
    b = pid_z // H
    d = pid_z % H

    # base pointers for this (b,d)
    left_base  = left_ptr  + b * stride_left_b  + d * stride_left_d
    right_base = right_ptr + b * stride_right_b + d * stride_right_d
    out_base   = out_ptr   + b * stride_out_b   + d * stride_out_d

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, N, BLOCK_K):
        k_off = k + tl.arange(0, BLOCK_K)

        # masks for block boundaries
        mask_i = i_off < N
        mask_j = j_off < N
        mask_k = k_off < N

        # left block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + i_off[:, None] * stride_left_i + k_off[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=(mask_i[:, None] & mask_k[None, :]), other=0.0).to(tl.float32)

        # right block (transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + j_off[None, :] * stride_right_j + k_off[:, None] * stride_right_k
        b = tl.load(b_ptrs, mask=(mask_j[None, :] & mask_k[:, None]), other=0.0).to(tl.float32)

        # matrix multiplication for this K‑tile
        acc += tl.dot(a, b)

    # store the result
    c_ptrs = out_base + i_off[:, None] * stride_out_i + j_off[None, :] * stride_out_j
    mask_store = (i_off[:, None] < N) & (j_off[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_store)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on input (dim = config["dim"])
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # compute mean/var over the last dimension
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    inv_std = torch.rsqrt(var + eps)
    x = (input_tensor - mean) * inv_std
    x = x * norm_weight + norm_bias  # shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)      # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)    # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Apply mask (if any)
    # ------------------------------------------------------------------
    mask = mask_tensor.unsqueeze(-1).type_as(left)  # broadcast to hidden_dim
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()   # will be used after matmul

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to fp16 and launch Triton kernel for the cubic contraction
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    # allocate output in fp32 (to avoid extra cast before LayerNorm)
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32, device=device)

    # strides (in elements) for each tensor
    slb, sli, slk, sld = left_fp16.stride()
    srb, srj, srk, srd = right_fp16.stride()
    sob, soi, soj, sod = out_fp32.stride()

    # block size configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # launch grid
    grid = (
        triton.cdiv(input_tensor.shape[1], BLOCK_M),   # N // BLOCK_M
        triton.cdiv(input_tensor.shape[1], BLOCK_N),   # N // BLOCK_N
        input_tensor.shape[0] * hidden_dim            # B * hidden_dim
    )

    trimul_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        input_tensor.shape[0],          # B
        input_tensor.shape[1],          # N
        hidden_dim,                     # H
        slb, sli, slk, sld,
        srb, srj, srk, srd,
        sob, soi, soj, sod,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    out = out_fp32  # shape [B,N,N,hidden_dim] in fp32

    # ------------------------------------------------------------------
    # 6. LayerNorm over hidden_dim (to_out_norm)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]

    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) * torch.rsqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # ------------------------------------------------------------------
    # 7. Apply output gate
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original dim
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]
    final = F.linear(out_norm, to_out_weight)  # bias is None

    return final
