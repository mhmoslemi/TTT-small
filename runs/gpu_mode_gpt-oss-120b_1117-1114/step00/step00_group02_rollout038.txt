@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_jr, stride_kr, stride_hr,
    stride_ob, stride_io, stride_jo, stride_ho,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # offsets
    offs_h = tl.arange(0, H)

    # masks for handling boundary? i, j always within N because pid covers only valid, but safe includes mask.

    # initialize accumulator
    acc = tl.zeros((H,), dtype=tl.float32)

    # loop over k dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        cur_k = tl.minimum(N - k_start, BLOCK_K)
        # k offsets with mask
        k_offs = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_offs < N

        # load left: shape (BLOCK_K, H)
        left_ptrs = left_ptr \
                    + b * stride_bl \
                    + i * stride_il \
                    + k_offs[:, None] * stride_kl \
                    + offs_h[None, :] * stride_hl
        left = tl.load(left_ptrs, mask=k_mask[:, None] & (offs_h[None, :] < H), other=0.0)
        # shape: (BLOCK_K, H)

        # load right: shape (BLOCK_K, H)
        right_ptrs = right_ptr \
                    + b * stride_br \
                    + j * stride_jr \
                    + k_offs[:, None] * stride_kr \
                    + offs_h[None, :] * stride_hr
        right = tl.load(right_ptrs, mask=k_mask[:, None] & (offs_h[None, :] < H), other=0.0)

        # elementwise multiply and accumulate sum over k
        # sum over K dimension: tl.sum
        prod = left * right  # (BLOCK_K, H)
        # sum across K dimension
        acc += tl.sum(prod, axis=0)  # (H,)
    # end loop

    # store result: shape (H,)
    out_ptrs = out_ptr \
              + b * stride_ob \
              + i * stride_io \
              + j * stride_jo \
              + offs_h * stride_ho
    tl.store(out_ptrs, acc, mask=(offs_h < H))

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_jr, stride_kr, stride_hr,
    stride_ob, stride_io, stride_jo, stride_ho,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    b = pid // (N * N)
    res = pid % (N * N)
    i = res // N
    j = res % N

    offs_h = tl.arange(0, H)
    # accumulator
    acc = tl.zeros((H,), dtype=tl.float32)

    for start_k in range(0, N, BLOCK_K):
        # compute actual block size
        cur_K = tl.minimum(N - start_k, BLOCK_K)
        k_offsets = start_k + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < N

        # left
        left_ptrs = left_ptr \
          + b * stride_bl \
          + i * stride_il \
          + k_offsets[:, None] * stride_kl \
          + offs_h[None, :] * stride_hl
        left = tl.load(left_ptrs, mask=k_mask[:, None], other=0.0)

        # right
        right_ptrs = right_ptr \
          + b * stride_br \
          + j * stride_jr \
          + k_offsets[:, None] * stride_kr \
          + offs_h[None, :] * stride_hr
        right = tl.load(right_ptrs, mask=k_mask[:, None], other=0.0)
        # elementwise multiply (BLOCK_K, H) and sum across K
        prod = left * right
        # sum across K dimension (axis 0)
        acc += tl.sum(prod, axis=0)

    # write out
    out_ptrs = out_ptr \
        + b * stride_ob \
        + i * stride_io \
        + j * stride_jo \
        + offs_h * stride_ho
    tl.store(out_ptrs, acc, mask=(offs_h < H))

def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, n_elements, BLOCK: tl.constexpr):

    start = pid * BLOCK

    b = tl.load(b_ptr + offs, mask=mask, other=0.0)

    # Gate weights
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    # Output Norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # Final projection weight
    to_out_weight = weights["to_out.weight"]

    # Apply layernorm on input
    x_norm = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )

    # Linear projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if present
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x_norm.dtype)
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to float16 for cheap einsum
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Triangular multiplicative update (full sum across k)
    out = torch.einsum('... i k d, ... j k d -> ... i j d', left, right)

    # Convert back to float32
    out = out.to(torch.float32)

    # Output layernorm
    out = torch.nn.functional.layer_norm(

import torch
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, n_elements, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < n_elements
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.
    Computes the Triangle Multiplicative Update based on the reference
    PyTorch implementation, using a small Triton kernel for the final
    element‑wise gating. All other operations (LayerNorms, linear
    projections, Einstein summation) are performed with PyTorch.
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # Extract weights (already on the correct device)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1) Pre‑normalisation
    # -----------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )

    # -----------------------------------------------------------------
    # 2) Linear projections
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, left_proj_weight)    # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcast over the hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x_norm.dtype)   # shape [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4) Gating
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to half‑precision before the cubic‑cost einsum
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # -----------------------------------------------------------------
    # 5) Triangle multiplicative update (full sum over k)
    # -----------------------------------------------------------------
    # out shape: [B, N, N, hidden_dim]
    out = torch.einsum('... i k d, ... j k d -> ... i j d', left, right)
    out = out.to(torch.float32)       # back to fp32 for the remaining ops

    # -----------------------------------------------------------------
    # 6) Output layer‑norm
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias
    )

    # -----------------------------------------------------------------
    # 7) Element‑wise gating (performed with a tiny Triton kernel)
    # -----------------------------------------------------------------
    # Flatten tensors so the kernel works on a 1‑D view.
    out_flat = out.contiguous().view(-1)
    gate_flat = out_gate.contiguous().view(-1)

    result_flat = torch.empty_like(out_flat)

    BLOCK = 1024
    grid = (triton.cdiv(out_flat.numel(), BLOCK),)
    elementwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        result_flat,
        out_flat.numel(),
        BLOCK=BLOCK,
    )
    out = result_flat.view_as(out)

    # -----------------------------------------------------------------
    # 8) Final linear projection back to the original dimensionality
    # -----------------------------------------------------------------
    output = torch.nn.functional.linear(out, to_out_weight)

    return output

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Tiny Triton kernel: element‑wise multiplication of two flat tensors.
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, n_elements, BLOCK: tl.constexpr):
    """
    out[i] = a[i] * b[i]   for i in [0, n_elements)
    """
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < n_elements

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)

    Forward pass of the “outgoing” Triangle Multiplicative Update (TriMul)
    as described in AlphaFold‑3. Most of the work is carried out with
    PyTorch; a small Triton kernel is used for the final element‑wise
    gating (out = out * out_gate).
    """
    # Unpack the input tuple
    input_tensor, mask, weights, config = data

    # Configuration parameters
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # Load model weights (they already reside on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2) Linear projections
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # --------------------------------------------------------------
    # 3) Apply optional mask (broadcast over the hidden dimension)
    # --------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x_norm.dtype)  # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4) Gating
    # --------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Triangle multiplicative update (full sum over k)
    #    Use half‑precision for the cubic‑cost einsum.
    # --------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    out = torch.einsum('... i k d, ... j k d -> ... i j d', left, right)  # [B, N, N, hidden_dim]
    out = out.to(torch.float32)  # back to fp32 for the following ops

    # --------------------------------------------------------------
    # 6) Output LayerNorm
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias
    )  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 7) Element‑wise gating (out = out * out_gate) via Triton
    # --------------------------------------------------------------
    # Flatten tensors to 1‑D views; they are already contiguous after the
    # LayerNorm call, but `contiguous()` makes this explicit.
    out_flat = out.contiguous().view(-1)
    gate_flat = out_gate.contiguous().view(-1)

    out_gated_flat = torch.empty_like(out_flat)

    BLOCK = 1024
    grid = (triton.cdiv(out_flat.numel(), BLOCK),)
    elementwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        out_gated_flat,
        out_flat.numel(),
        BLOCK=BLOCK,
    )
    out = out_gated_flat.view_as(out)  # shape [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 8) Final linear projection back to the original dimensionality
    # --------------------------------------------------------------
    output = torch.nn.functional.linear(out, to_out_weight)  # [B, N, N, dim]

    return output
