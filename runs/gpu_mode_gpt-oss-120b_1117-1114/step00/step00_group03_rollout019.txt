    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    left = F.linear(x, left_proj_weight)  # shape [..., hidden_dim]

"""
TriMul outgoing kernel implementation using Triton.

The heavy O(N^3) contraction out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
is computed as a batched matrix multiplication L @ R^T for each (b,d) pair,
leveraging a custom Triton kernel with block tiling.
All other operations (LayerNorm, Linear projections, gates) are performed in PyTorch.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_br, stride_bc,
    stride_cm, stride_cn,
    batch_stride,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ B^T for many independent (M,N,K) problems.

    A : [batch, M, K]   (row‑major)
    B : [batch, N, K]   (row‑major, we treat the K dimension as the leading one)
    C : [batch, M, N]
    The kernel is launched with a 3‑D grid (batch, M tiles, N tiles).
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the start of the tile for this (batch, m) row and (batch, n) column
    a_base = a_ptr + pid_batch * batch_stride + offs_m[:, None] * stride_am
    b_base = b_ptr + pid_batch * batch_stride + offs_n[None, :] * stride_bc

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for ki in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = ki * BLOCK_K
        a = tl.load(
            a_base + (cur_k + offs_k) * stride_ak,
            mask=(offs_m[:, None] < M) & ((cur_k + offs_k)[None, :] < K),
            other=0.0,
            dtype=tl.float16
        )
        b = tl.load(
            b_base + (cur_k + offs_k)[:, None] * stride_br,
            mask=((cur_k + offs_k)[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
            dtype=tl.float16
        )
        acc += tl.dot(a, b)

    # Write back result (store as fp16, later cast to fp32 in Python)
    c = c_ptr + pid_batch * batch_stride + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c, acc.to(tl.float16), mask=mask_c)

    """

        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, S, S, C]
            - mask: torch.Tensor [B, S, S] (may be all‑ones)
            - weights: dict of torch.Tensors containing all model parameters
            - config: dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor of shape [B, S, S, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5  # LayerNorm epsilon (same as torch default)

    # -------------------------------------------------
    #  Load weight tensors onto the same device and dtype
    # -------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_bias = weights["norm.bias"].to(device, dtype=torch.float32)

    left_proj_weight = weights["left_proj.weight"].to(device, dtype=torch.float32)
    right_proj_weight = weights["right_proj.weight"].to(device, dtype=torch.float32)

    left_gate_weight = weights["left_gate.weight"].to(device, dtype=torch.float32)
    right_gate_weight = weights["right_gate.weight"].to(device, dtype=torch.float32)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device, dtype=torch.float32)
    to_out_weight = weights["to_out.weight"].to(device, dtype=torch.float32)

    # -------------------------------------------------
    #  LayerNorm on the input (dim last)
    # -------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # -------------------------------------------------
    #  Linear projections (no bias)
    # -------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, S, S, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, S, S, hidden_dim]

    # -------------------------------------------------
    #  Apply optional mask (mask shape [B, S, S])
    # -------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).type_as(x)  # [B, S, S, 1]
        left = left * mask_exp
        right = right * mask_exp

    # -------------------------------------------------
    #  Gating values (sigmoid)
    # -------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    #  Heavy O(N^3) contraction with Triton
    # -------------------------------------------------
    # Cast to half for kernel (smaller bandwidth)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Rearrange to [B, hidden_dim, S, S] and flatten batch dimensions
    B, S, _, _ = left_h.shape
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, S, S]
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(-1, S, S)    # [B*H, S, S]
    right_flat = right_perm.view(-1, S, S)

    batch = left_flat.shape[0]      # B * hidden_dim
    M = N = K = S

    # Output tensor (half)
    out_flat = torch.empty_like(left_flat)

    # Strides – all tensors are contiguous, so strides are simple integers
    stride_am = left_flat.stride(1)   # stride for M dimension (row)
    stride_ak = left_flat.stride(2)   # stride for K dimension (col)
    stride_br = right_flat.stride(2)  # after transpose: row‑stride = original col‑stride
    stride_bc = right_flat.stride(1)  # col‑stride = original row‑stride
    stride_cm = left_flat.stride(1)   # output uses same layout as left
    stride_cn = left_flat.stride(2)
    batch_stride = left_flat.stride(0)

    # Tile sizes – chosen to fit typical B200 resources
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch,
        triton.cdiv(S, BLOCK_M),
        triton.cdiv(S, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        M, N, K,
        stride_am, stride_ak,
        stride_br, stride_bc,
        stride_cm, stride_cn,
        batch_stride,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape back to [B, S, S, hidden_dim] and cast to float32 for the remaining ops
    out = out_flat.view(B, hidden_dim, S, S).permute(0, 2, 3, 1).to(torch.float32)

    # -------------------------------------------------
    #  Post‑processing (LayerNorm, out‑gate, final linear)
    # -------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)
    out = out * out_gate
    output = F.linear(out, to_out_weight)

    return output

"""
TriMul outgoing kernel – Triton acceleration.

The dominant O(N³) operation
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix multiplication L @ Rᵀ for every (batch, d) pair.
We implement this contraction with a custom Triton kernel that tiles the
M‑, N‑ and K‑dimensions (M = N = seq_len) and accumulates in FP32.
All surrounding LayerNorms, Linear projections and gating are performed with
PyTorch for simplicity.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_br, stride_bc,
    stride_cm, stride_cn,
    batch_stride,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ Bᵀ for many independent problems.

    A : [batch, M, K] (row‑major)
    B : [batch, N, K] (row‑major, we treat the K dimension as the leading one)
    C : [batch, M, N]

    Launched with a 3‑D grid (batch, tile_M, tile_N).
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Base pointers for this tile
    a_base = a_ptr + pid_batch * batch_stride + offs_m[:, None] * stride_am
    b_base = b_ptr + pid_batch * batch_stride + offs_n[None, :] * stride_bc

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K steps
    for ki in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = ki * BLOCK_K
        # Load A tile (M × K)
        a = tl.load(
            a_base + (cur_k + offs_k) * stride_ak,
            mask=(offs_m[:, None] < M) & ((cur_k + offs_k)[None, :] < K),
            other=0.0,
            dtype=tl.float16,
        )
        # Load B tile (N × K) – we want the K‑dimension first, so we load with
        # stride_br (original column stride) as the row stride.
        b = tl.load(
            b_base + (cur_k + offs_k)[:, None] * stride_br,
            mask=((cur_k + offs_k)[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
            dtype=tl.float16,
        )
        acc += tl.dot(a, b)

    # Write C tile (M × N) as FP16; later we cast to FP32 in Python.
    c = c_ptr + pid_batch * batch_stride + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing’’ TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, S, S, C] (float32)
            - mask: torch.Tensor of shape [B, S, S] (bool or float)
            - weights: dict of torch.Tensors containing all model parameters
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, S, S, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -------------------- Load weights --------------------
    norm_weight = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_bias = weights["norm.bias"].to(device, dtype=torch.float32)

    right_proj_weight = weights["right_proj.weight"].to(device, dtype=torch.float32)

    right_gate_weight = weights["right_gate.weight"].to(device, dtype=torch.float32)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device, dtype=torch.float32)
    to_out_weight = weights["to_out.weight"].to(device, dtype=torch.float32)

    # -------------------- LayerNorm on input --------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # -------------------- Linear projections (no bias) --------------------
    left = F.linear(x, left_proj_weight)     # [B, S, S, hidden_dim]
    right = F.linear(x, right_proj_weight)   # [B, S, S, hidden_dim]

    # -------------------- Apply mask if provided --------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).type_as(x)  # broadcast to [B, S, S, 1]
        left = left * mask_
        right = right * mask_

    # -------------------- Gating (sigmoid) --------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -------------------- Heavy contraction with Triton --------------------
    # Cast to FP16 for bandwidth‑limited kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, S, _, _ = left_h.shape
    # Rearrange to [B, hidden_dim, S, S] and flatten the first two dims
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(-1, S, S)   # [B*hidden_dim, S, S]
    right_flat = right_perm.view(-1, S, S)

    batch = left_flat.shape[0]   # B * hidden_dim
    M = N = K = S

    # Output buffer (FP16)
    out_flat = torch.empty_like(left_flat)

    # Strides (contiguous tensors)
    stride_am = left_flat.stride(1)   # row stride for A (M)
    stride_ak = left_flat.stride(2)   # col stride for A (K)
    stride_br = right_flat.stride(2)  # row stride for Bᵀ (K)
    stride_bc = right_flat.stride(1)  # col stride for Bᵀ (N)
    stride_cm = left_flat.stride(1)   # row stride for C (M)
    stride_cn = left_flat.stride(2)   # col stride for C (N)
    batch_stride = left_flat.stride(0)

    # Tile sizes – chosen to balance shared‑memory usage and occupancy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch,
        triton.cdiv(S, BLOCK_M),
        triton.cdiv(S, BLOCK_N),
    )

    batched_matmul_kernel[grid](

    )

    out = out_flat.view(B, hidden_dim, S, S).permute(0, 2, 3, 1).to(torch.float32)

    # -------------------- Post‑processing --------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)
    out = out * out_gate
    output = F.linear(out, to_out_weight)   # final projection to original dim

    return output
