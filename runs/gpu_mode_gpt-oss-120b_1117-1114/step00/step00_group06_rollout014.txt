@triton.jit
def trimul_matmul_kernel(left_ptr, right_ptr, out_ptr,
                         B, N, H,
                         stride_left_b, stride_left_i, stride_left_k, stride_left_h,
                         stride_right_b, stride_right_k, stride_right_j, stride_right_h,
                         stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                         BLOCK_M: tl.constexpr,
                         BLOCK_N: tl.constexpr,
                         BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)  # row block
    pid_j = tl.program_id(1)  # col block
    pid_bh = tl.program_id(2) # batch*hidden

    b = pid_bh // H
    h = pid_bh % H

    # offset of row and col
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for valid rows / cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute left pointer offsets
        # left_ptr + b*stride_left_b + offs_i[:,None]*stride_left_i + offs_k[None,:]*stride_left_k + h*stride_left_h
        left_ptrs = left_ptr + b * stride_left_b + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k + h * stride_left_h

        # compute right pointer offsets
        # right_ptr + b*stride_right_b + offs_k[:,None]*stride_right_k + offs_j[None,:]*stride_right_j + h*stride_right_h
        right_ptrs = right_ptr + b * stride_right_b + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j + h * stride_right_h

        # load with mask
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0) # shape (BLOCK_M, BLOCK_K)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0) # shape (BLOCK_K, BLOCK_N)

        # cast to float32 for accumulation
        left = left.to(tl.float32)
        right = right.to(tl.float32)

        acc += tl.dot(left, right)

    # compute final pointers for output
    out_ptrs = out_ptr + b * stride_out_b + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j + h * stride_out_h

    # store result
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

```python
import torch
import triton
import triton.language as tl

# Triton kernel for batched hidden-dim matrix multiplication
@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_k, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ B^T per batch b and hidden h.
    A = left[b, i, k, h]; B = right[b, k, j, h]; C[b,i,j,h] = sum_k A * B.
    """
    pid_i = tl.program_id(0)   # row tile index
    pid_j = tl.program_id(1)   # col tile index
    pid_bh = tl.program_id(2)  # combined batch * hidden index

    # decode batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # Compute offsets for rows (i) and columns (j)
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers for A (left)
        a_ptrs = left_ptr + \
                 b * stride_left_b + \
                 offs_i[:, None] * stride_left_i + \
                 offs_k[None, :] * stride_left_k + \
                 h * stride_left_h

        # Pointers for B (right) – note that k corresponds to original i dimension
        b_ptrs = right_ptr + \
                 b * stride_right_b + \
                 offs_k[:, None] * stride_right_k + \
                 offs_j[None, :] * stride_right_j + \
                 h * stride_right_h

        # Load tiles with boundary masking
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Convert to float32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        # Accumulate matrix product for this K tile
        acc += tl.dot(a, b)

    # Store result
    out_ptrs = out_ptr + \
               b * stride_out_b + \
               offs_i[:, None] * stride_out_i + \
               offs_j[None, :] * stride_out_j + \
               h * stride_out_h

    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul (outgoing) operator.
    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (float or bool)
            - weights: dict of parameter tensors
            - config: dict with keys "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -------------------------------------------------
    # LayerNorm on the input (dim)
    # -------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -------------------------------------------------
    # Linear projections
    # -------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------
    # Apply mask (if any)
    # -------------------------------------------------
    # mask is assumed to be float or bool; expand to last dim
    if mask is None:
        mask_exp = torch.ones_like(left[..., :1])
    else:
        mask_exp = mask.unsqueeze(-1).type_as(left)
    left = left * mask_exp
    right = right * mask_exp

    # -------------------------------------------------
    # Gating
    # -------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = (left.to(torch.float16)) * left_gate
    right = (right.to(torch.float16)) * right_gate

    # -------------------------------------------------
    # TriMul core: sum_k left[...,i,k,h] * right[...,k,j,h]
    # Implemented as a batched GEMM per hidden dim using Triton
    # -------------------------------------------------
    B, N, _, _ = left.shape  # left shape is [B, N, N, hidden_dim]

    # Allocate output tensor (float16 for kernel)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Strides in elements (not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_j, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Triton block sizes – tuned for typical GPU
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    trimul_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,   # k uses original i stride
        stride_right_j,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------
    # Remaining post‑processing in PyTorch (float32)
    # -------------------------------------------------
    out = out.to(torch.float32)

    # LayerNorm over hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Multiply by output gate
    out = out * out_gate

    # Final linear projection back to `dim`
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass with a Triton‑accelerated O(N³) core.

Algorithm
---------
1. Layer‑norm the input tensor (last dimension = dim).
2. Project to a hidden dimension `H` using two weight‑only linear layers:
       left  = x @ W_left   (shape [B,N,N,H])
       right = x @ W_right
3. Apply the optional mask (broadcast on the hidden axis).
4. Compute sigmoid gates for left, right and for the final output.
   Multiply left/right with their respective gates.
5. Core operation:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,k,j,h]
   This is a batched matrix multiply for each `(b,h)` pair.
   Implemented as a tiled GEMM in Triton (`trimul_matmul_kernel`).
6. Convert the result back to FP32, apply a hidden‑dim LayerNorm,
   multiply by the output gate, and project back to `dim`.

The kernel only handles step 5; all other operations stay in PyTorch.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_matmul_kernel(
    left_ptr,          # *fp16
    right_ptr,         # *fp16
    out_ptr,           # *fp16
    B, N, H,           # int32
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b,    stride_right_k, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out = left @ right^T per batch b and hidden h.
    left  : [B, N, N, H]  (i,k)
    right : [B, N, N, H]  (k,j)
    out   : [B, N, N, H]  (i,j)
    """
    pid_i = tl.program_id(0)          # tile row index (i)
    pid_j = tl.program_id(1)          # tile col index (j)
    pid_bh = tl.program_id(2)         # combined batch * hidden index

    # decode batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # row / column offsets for this tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulate in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile (i,k) -----
        a_ptrs = left_ptr + \
                 b * stride_left_b + \
                 offs_i[:, None] * stride_left_i + \
                 offs_k[None, :] * stride_left_k + \
                 h * stride_left_h
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)
        a = a.to(tl.float32)

        # ----- load right tile (k,j) -----
        b_ptrs = right_ptr + \
                 b * stride_right_b + \
                 offs_k[:, None] * stride_right_k + \
                 offs_j[None, :] * stride_right_j + \
                 h * stride_right_h
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)
        b = b.to(tl.float32)

        # ----- accumulate -----
        acc += tl.dot(a, b)

    # ----- store result (i,j) -----
    out_ptrs = out_ptr + \
               b * stride_out_b + \
               offs_i[:, None] * stride_out_i + \
               offs_j[None, :] * stride_out_j + \
               h * stride_out_h
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul (outgoing) operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float)
            - weights: dict with the required parameter tensors
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # -------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------
    # 3. Mask (if any) – broadcast on the hidden axis
    # -------------------------------------------------
    mask_exp = mask.unsqueeze(-1).type_as(left)   # shape [B,N,N,1]
    left = left * mask_exp
    right = right * mask_exp

    # -------------------------------------------------
    # 4. Gating (sigmoid of linear projections of the normalized input)
    # -------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))   # keep FP32 for later

    left = left.to(torch.float16) * left_gate
    right = right.to(torch.float16) * right_gate

    # -------------------------------------------------
    # 5. Core TriMul via Triton (batched GEMM per hidden dim)
    # -------------------------------------------------
    B, N, _, _ = left.shape

    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Strides in element units (not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_j, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Block sizes – chosen to fit typical SMEM limits & enable WMMA
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * hidden_dim,            # batch * hidden
    )

    trimul_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,   # as stride over the k dimension (original i)
        stride_right_j,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------
    # 6. Post‑processing in PyTorch (float32)
    # -------------------------------------------------
    out = out.to(torch.float32)

    # Hidden‑dim LayerNorm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply output gate
    out = out * out_gate

    # Final projection back to `dim`
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
