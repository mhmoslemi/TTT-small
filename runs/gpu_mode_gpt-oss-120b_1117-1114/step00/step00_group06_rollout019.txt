import triton.language as tl

@triton.jit
def apply_mask_gate_kernel(src_ptr, mask_ptr, gate_ptr, out_ptr, 
                           B, N, D,
                           stride_src_b, stride_src_i, stride_src_j, stride_src_c,
                           stride_mask_b, stride_mask_i, stride_mask_j,
                           stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_c,
                           BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    pid_d = tl.program_id(1)

    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    total = B * N * N
    mask_m = offs_m < total
    mask_d = offs_d < D
    mask = mask_m[:, None] & mask_d[None, :]

    # compute (b,i,j) indices
    b = offs_m // (N * N)
    rem = offs_m % (N * N)
    i = rem // N
    j = rem % N

    # compute base offsets
    src_offset = b * stride_src_b + i * stride_src_i + j * stride_src_j
    gate_offset = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j
    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # load mask scalar

"""
TriMul forward (outgoing) implemented with a fused Triton kernel for the
mask‑gate multiplication.  The heavy O(N³) part is carried out by a
batched half‑precision matrix multiplication (torch.bmm) which utilizes
cuBLAS/Tensor Cores.  All layer‑norms and linear layers follow the
reference implementation.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: out = src * mask * gate   (src and gate are [B,N,N,D])
# mask is [B,N,N] and is broadcast over D.
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_gate_kernel(
    src_ptr, mask_ptr, gate_ptr, out_ptr,
    B, N, D,
    stride_src_b, stride_src_i, stride_src_j, stride_src_c,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_c,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)          # blocks over (b,i,j)
    pid_d = tl.program_id(1)        # blocks over hidden dimension D

    # ------------------------------------------------------------------
    # Compute linear offsets for the (b,i,j) block
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)               # (BLOCK_M,)
    total = B * N * N
    mask_m = offs_m < total                                      # valid (b,i,j)

    b = offs_m // (N * N)
    rem = offs_m % (N * N)
    i = rem // N
    j = rem % N

    # ------------------------------------------------------------------
    # Offsets for the hidden dimension block
    # ------------------------------------------------------------------
    offs_d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)              # (BLOCK_D,)
    mask_d = offs_d < D
    mask = mask_m[:, None] & mask_d[None, :]                     # (BLOCK_M, BLOCK_D)

    # ------------------------------------------------------------------
    # Base pointers for src, gate and mask
    # ------------------------------------------------------------------
    src_off = b * stride_src_b + i * stride_src_i + j * stride_src_j
    gate_off = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # load scalar mask (broadcast later)
    mask_val = tl.load(mask_ptr + mask_off, mask=mask_m, other=0.0)
    mask_val = mask_val[:, None]            # (BLOCK_M, 1)

    # load src and gate vectors
    src = tl.load(src_ptr + src_off[:, None] + offs_d[None, :] * stride_src_c,
                  mask=mask, other=0.0)
    gate = tl.load(gate_ptr + gate_off[:, None] + offs_d[None, :] * stride_gate_c,
                   mask=mask, other=0.0)

    # fused multiply
    out = src * mask_val * gate

    # store result back (in‑place)
    tl.store(out_ptr + src_off[:, None] + offs_d[None, :] * stride_src_c,
             out, mask=mask)

# Entrypoint expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward pass (no gradients).  The heavy mask‑gate operation is
    executed in a custom Triton kernel; the O(N³) product is performed with a
    batched half‑precision matrix multiplication.
    """
    # unpack inputs -------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = input_tensor
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # ------------------------------------------------------------------
    # Linear projections and gates (float32)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # Cast to fp16 for the Triton kernel (saves bandwidth)
    # ------------------------------------------------------------------
    left_proj_h = left_proj.to(torch.float16)
    right_proj_h = right_proj.to(torch.float16)
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)
    mask_h = mask.to(torch.float16)

    B, N, _, D = left_proj_h.shape  # D == hidden_dim

    # strides for the tensors (contiguous layout)
    s_src_b, s_src_i, s_src_j, s_src_c = left_proj_h.stride()
    s_gate_b, s_gate_i, s_gate_j, s_gate_c = left_gate_h.stride()
    s_mask_b, s_mask_i, s_mask_j = mask_h.stride()

    # -----------------------------------------------------------------------------
    # Apply mask and gate in a fused Triton kernel (in‑place)
    # -----------------------------------------------------------------------------
    BLOCK_M = 256            # number of (b,i,j) elements per program
    BLOCK_D = 32             # hidden‑dim tile
    total = B * N * N
    grid = (triton.cdiv(total, BLOCK_M), triton.cdiv(D, BLOCK_D))

    # left side
    apply_mask_gate_kernel[
        grid
    ](
        left_proj_h, mask_h, left_gate_h, left_proj_h,
        B, N, D,
        s_src_b, s_src_i, s_src_j, s_src_c,
        s_mask_b, s_mask_i, s_mask_j,
        s_gate_b, s_gate_i, s_gate_j, s_gate_c,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # right side
    apply_mask_gate_kernel[
        grid
    ](
        right_proj_h, mask_h, right_gate_h, right_proj_h,
        B, N, D,
        s_src_b, s_src_i, s_src_j, s_src_c,
        s_mask_b, s_mask_i, s_mask_j,
        s_gate_b, s_gate_i, s_gate_j, s_gate_c,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # Core O(N³) update: batched half‑precision matrix multiplication
    #   out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    # Equivalent to (B*D) independent matmuls of (N×N) @ (N×N)^T
    # ------------------------------------------------------------------
    # reshape to (B*D, N, N)
    left_mat = left_proj_h.permute(0, 3, 1, 2).reshape(B * D, N, N)
    right_mat = right_proj_h.permute(0, 3, 2, 1).reshape(B * D, N, N)

    # batched GEMM in fp16 (Tensor Cores)
    out_mat = torch.bmm(left_mat, right_mat)          # (B*D, N, N)

    # restore original layout: (B,N,N,D)
    out = out_mat.view(B, D, N, N).permute(0, 2, 3, 1)   # [B, N, N, D]

    # ------------------------------------------------------------------
    # to_out_norm (LayerNorm over hidden dimension D)
    # ------------------------------------------------------------------
    eps_out = 1e-5
    out_norm = (out - out.mean(-1, keepdim=True)) / torch.sqrt(out.var(-1, unbiased=False, keepdim=True) + eps_out)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # ------------------------------------------------------------------
    # Apply output gate (broadcast over D)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # Final linear projection back to dim (float32 output)
    # ------------------------------------------------------------------
    out_fp32 = torch.nn.functional.linear(out_norm.to(torch.float32), weights["to_out.weight"])

    return out_fp32
