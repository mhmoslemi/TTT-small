    stride_br, stride_jr, stride_kr, stride_dr,

    BLOCK_N: tl.constexpr,

    BLOCK_D: tl.constexpr,

    pid_j = tl.program_id(2)

@triton.jit
def tri_mul_kernel(...):
    # Program IDs
    batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    pid_d = tl.program_id(3)

    # Offsets
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)   # Typically BLOCK_D=1 -> array length 1

    # Masks
    mask_i = i < N
    mask_j = j < N
    mask_d = d < H

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, N, BLOCK_K):
        k_range = k + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # Load left tile (M,K)
        left_ptrs = left_ptr + batch * stride_bl \
                + i[:, None] * stride_il \
                + k_range[None, :] * stride_kl \
                + d[0] * stride_dl   # Since BLOCK_D=1 we can take d[0]
        # If BLOCK_D > 1 we need broadcasting across d dimension inside accumulator. We'll ignore.
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile (K,N) => transposed
        right_ptrs = right_ptr + batch * stride_br \
                + j[None, :] * stride_jr \
                + k_range[:, None] * stride_kr \
                + d[0] * stride_dr
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Dot accumulate
        acc += tl.dot(left, right)

    # Store output
    out_ptrs = out_ptr + batch * stride_bo \
                + i[:, None] * stride_io \
                + j[None, :] * stride_jo \
                + d[0] * stride_do
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :] & mask_d[None, None])

    input_tensor, mask, weights, config = data

    # Transfer to correct dtype
    x = input_tensor.to(torch.float32)  # ensure FP32 maybe required for layernorm etc.

    hidden_dim = config['hidden_dim']

    # get weights
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    left_proj_weight = weights['left_proj.weight']
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    to_out_weight = weights['to_out.weight']

    # Normalize
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Ensure mask dtype
    mask = mask.to(x.dtype)  # convert
    mask = mask.unsqueeze(-1)  # shape B,N,N,1

    # Apply mask
    left = left * mask
    right = right * mask

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute out via Triton
    # allocate output
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=left.dtype, device=left.device)

    # define kernel
    # (kernel defined at top-level)

    # compute strides
    stride_bl = left.stride(0)
    stride_il = left.stride(1)
    stride_kl = left.stride(2)
    stride_dl = left.stride(3)

    stride_br = right.stride(0)
    stride_jr = right.stride(1)  # (j dimension) but second axis is also N.
    stride_kr = right.stride(2)
    stride_dr = right.stride(3)

    stride_bo = out.stride(0)
    stride_io = out.stride(1)
    stride_jo = out.stride(2)
    stride_do = out.stride(3)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_D = 1

    out = out * out_gate

    out = torch.nn.functional.linear(out, to_out_weight)  # shape (B,N,N,dim)

    out = out.to(torch.float32)

    return out

TriMul “outgoing” implementation with Triton acceleration.

The heavy O(N³) part `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`
is executed in a custom Triton kernel that performs a batched
matrix‑multiply for each hidden dimension slice. The rest of the
TriMul pipeline (LayerNorm, linear projections, gating and final
projection) remains in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -----------------------------------------------------------------------------
# Triton kernel: compute one hidden‑dim slice of the cubic sum.
# Each program (block) handles a tile (BLOCK_M x BLOCK_N) of the
# output matrix for a single hidden‑dimension index.
# -----------------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_dl,
    stride_br, stride_jr, stride_kr, stride_dr,
    stride_bo, stride_io, stride_jo, stride_do,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,   # kept for generality, we launch with BLOCK_D = 1
):
    # -------------------------------------------------------------------------
    # Program IDs (grid dimensions): (batch, i_tile, j_tile, hidden_tile)
    # -------------------------------------------------------------------------
    batch_id = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    pid_d = tl.program_id(3)               # hidden‑dim tile index

    # -------------------------------------------------------------------------
    # Offsets for the i, j and d dimensions
    # -------------------------------------------------------------------------
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    d_offset = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    mask_i = i_offset < N
    mask_j = j_offset < N
    mask_d = d_offset < H

    # -------------------------------------------------------------------------
    # Accumulator for the (i,j) tile – one accumulator per hidden slice.
    # With BLOCK_D == 1 the accumulator is a scalar per (i,j).
    # -------------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------------------
    # Loop over the reduction dimension k in tiles of size BLOCK_K
    # -------------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # ---------------------------------------------------------------------
        # Load left tile: shape (BLOCK_M, BLOCK_K)
        # left[b, i, k, d]   -> stride dimensions (batch, i, k, d)
        # ---------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + batch_id * stride_bl
            + i_offset[:, None] * stride_il
            + k_offset[None, :] * stride_kl
            + d_offset[0] * stride_dl   # BLOCK_D == 1, use the first element
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )

        # ---------------------------------------------------------------------
        # Load right tile transposed: shape (BLOCK_K, BLOCK_N)
        # right[b, j, k, d] -> we need rows = k, cols = j
        # ---------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + batch_id * stride_br
            + k_offset[:, None] * stride_kr
            + j_offset[None, :] * stride_jr
            + d_offset[0] * stride_dr
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )

        # ---------------------------------------------------------------------
        # Accumulate dot product for this k‑tile
        # ---------------------------------------------------------------------
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N)

    # -------------------------------------------------------------------------
    # Write the result back.  The mask makes sure we do not write out‑of‑bounds.
    # -------------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch_id * stride_bo
        + i_offset[:, None] * stride_io
        + j_offset[None, :] * stride_jo
        + d_offset[0] * stride_do
    )
    store_mask = mask_i[:, None] & mask_j[None, :] & mask_d
    tl.store(out_ptrs, acc, mask=store_mask)


# -----------------------------------------------------------------------------
# Public entry point.
# -----------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator with the cubic
    reduction accelerated by Triton. The function expects:

        data = (input_tensor, mask, weights, config)

    where `weights` contains all LayerNorm and Linear parameters.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    # -----------------------------------------------------------------
    # Unpack configuration
    # -----------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # Load model parameters (all are assumed to be on the same device)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1) LayerNorm over the last dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    ).to(dtype)

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)    # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # 3) Apply mask (if any).  The mask comes as [B,N,N] – expand to hidden_dim.
    # -----------------------------------------------------------------
    mask = mask.to(dtype).unsqueeze(-1)    # [B,N,N,1]
    left = left * mask
    right = right * mask

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid activations)
    # -----------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Triton kernel: cubic reduction
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # Strides (in element counts, not bytes)
    stride_bl, stride_il, stride_kl, stride_dl = left.stride()
    stride_br, stride_jr, stride_kr, stride_dr = right.stride()
    stride_bo, stride_io, stride_jo, stride_do = out.stride()

    # Tile sizes – chosen for good occupancy on NVIDIA B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_D = 1  # one hidden dimension per program

    grid = (
        B,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        hidden_dim // BLOCK_D,
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_bl,
        stride_il,
        stride_kl,
        stride_dl,
        stride_br,
        stride_jr,
        stride_kr,
        stride_dr,
        stride_bo,
        stride_io,
        stride_jo,
        stride_do,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=4,  # a modest number of warps works well
    )

    # -----------------------------------------------------------------
    # 6) Output LayerNorm + gating + final Linear
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # maps hidden_dim -> dim

    return out
