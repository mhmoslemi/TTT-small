import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, D,
    stride_li, stride_lk, stride_ld,
    stride_rj, stride_rk, stride_rd,
    stride_oi, stride_oj, stride_od,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_d = tl.program_id(2)

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    d_start = pid_d * BLOCK_D

    # indexes
    i_offset = i_start + tl.arange(0, BLOCK_I)
    j_offset = j_start + tl.arange(0, BLOCK_J)
    d_offset = d_start + tl.arange(0, BLOCK_D)

    mask_i = i_offset < N
    mask_j = j_offset < N
    mask_d = d_offset < D

    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, N, BLOCK_K):
        k_offset = k + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # masks for loads
        mask_left = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
        mask_right = mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]

        # pointer arithmetic
        left_ptrs = left_ptr + i_offset[:, None, None] * stride_li + k_offset[None, :, None] * stride_lk + d_offset[None, None, :] * stride_ld
        right_ptrs = right_ptr + k_offset[:, None, None] * stride_rk + j_offset[None, :, None] * stride_rj + d_offset[None, None, :] * stride_rd

        # loads
        left_block = tl.load(left_ptrs, mask=mask_left, other=0.0)   # (BLOCK_I, BLOCK_K, BLOCK_D)
        right_block = tl.load(right_ptrs, mask=mask_right, other=0.0)  # (BLOCK_K, BLOCK_J, BLOCK_D)

        # inner reduction over K for each d slice
        # unroll d loop
        for d in range(0, BLOCK_D):
            a = left_block[:, :, d]    # (BLOCK_I, BLOCK_K)
            b = right_block[:, :, d]   # (BLOCK_K, BLOCK_J)
            acc[:, :, d] += tl.dot(a, b)

    # store output
    out_ptrs = out_ptr + i_offset[:, None, None] * stride_oi + j_offset[None, :, None] * stride_oj + d_offset[None, None, :] * stride_od
    mask_out = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
    tl.store(out_ptrs, acc, mask=mask_out)

def custom_kernel(data):
    """
    TriMul "outgoing" implementation that offloads the O(N^3) reduction to a Triton kernel.
    The surrounding LayerNorms, linear projections, and gating are performed in PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -------------------------------------------------------------------------
    # 1. LayerNorm on the input (over the channel dim `dim`)
    # -------------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -------------------------------------------------------------------------
    # 2. Linear projections (no bias) to hidden dimension
    # -------------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])    # [B,N,N,H]

    # -------------------------------------------------------------------------
    # 3. Optional mask (broadcasted on hidden dimension)
    # -------------------------------------------------------------------------
    if mask is not None:
        mask_uns = mask.unsqueeze(-1).type_as(left)   # shape [B,N,N,1]
        left = left * mask_uns
        right = right * mask_uns

    # -------------------------------------------------------------------------
    # 4. Gating (sigmoid of a linear layer on the normalized input)
    # -------------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous memory for the kernel
    left = left.contiguous()
    right = right.contiguous()

    # -------------------------------------------------------------------------
    # 5. Triton kernel: out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    # -------------------------------------------------------------------------
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Block sizes (tuned for typical N and hidden_dim)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_D = 8   # hidden_dim will be tiled by 8; adjust if hidden_dim < 8

    grid = (
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
        triton.cdiv(hidden_dim, BLOCK_D),
    )

    # Strides (in elements) for left/right/out tensors
    # left/right have shape [B, N, N, H]
    stride_li = left.stride(1)   # stride over i dimension
    stride_lk = left.stride(2)   # stride over k dimension
    stride_ld = left.stride(3)   # stride over hidden dimension
    stride_rj = right.stride(1)  # stride over j dimension (same as i stride)
    stride_rk = right.stride(2)  # stride over k dimension
    stride_rd = right.stride(3)  # stride over hidden dim
    stride_oi = out.stride(1)
    stride_oj = out.stride(2)
    stride_od = out.stride(3)

    for b in range(bs):
        # Compute base pointers for this batch element
        left_base = left.data_ptr() + b * left.stride(0)
        right_base = right.data_ptr() + b * right.stride(0)
        out_base = out.data_ptr() + b * out.stride(0)

        trimul_outgoing_kernel[grid](
            left_base,
            right_base,
            out_base,
            N,
            hidden_dim,
            stride_li,
            stride_lk,
            stride_ld,
            stride_rj,
            stride_rk,
            stride_rd,
            stride_oi,
            stride_oj,
            stride_od,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )

    # -------------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # -------------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],

"""
TriMul (outgoing) forward pass with the O(N³) reduction offloaded to a Triton kernel.
The surrounding LayerNorms, linear projections and gating are performed in PyTorch,
while the core einsum `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]` is computed
by a custom Triton kernel that tiles the i, j and hidden (d) dimensions and reduces
over k in blocks. This achieves better memory locality and uses TL's fast dot
instruction for each hidden slice.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, D,
    stride_li, stride_lk, stride_ld,
    stride_rj, stride_rk, stride_rd,
    stride_oi, stride_oj, stride_od,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].

    Arguments
    ----------
    left_ptr, right_ptr, out_ptr : pointers to the start of a single batch element
    N, D                         : sequence length and hidden dimension
    stride_*                     : strides (in elements) for the tensors
    BLOCK_*                      : compile‑time tile sizes
    """
    pid_i = tl.program_id(0)   # block on output i dimension
    pid_j = tl.program_id(1)   # block on output j dimension
    pid_d = tl.program_id(2)   # block on hidden dimension

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    d_start = pid_d * BLOCK_D

    i_offset = i_start + tl.arange(0, BLOCK_I)
    j_offset = j_start + tl.arange(0, BLOCK_J)
    d_offset = d_start + tl.arange(0, BLOCK_D)

    mask_i = i_offset < N
    mask_j = j_offset < N
    mask_d = d_offset < D

    # accumulator for the (i,j,d) tile
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # reduction over k
    for k in range(0, N, BLOCK_K):
        k_offset = k + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # masks for left (i,k,d) and right (k,j,d)
        mask_left  = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
        mask_right = mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]

        # pointer arithmetic
        left_ptrs = (left_ptr
                     + i_offset[:, None, None] * stride_li
                     + k_offset[None, :, None] * stride_lk
                     + d_offset[None, None, :] * stride_ld)

        right_ptrs = (right_ptr
                      + k_offset[:, None, None] * stride_rk
                      + j_offset[None, :, None] * stride_rj
                      + d_offset[None, None, :] * stride_rd)

        # load tiles
        left_tile  = tl.load(left_ptrs,  mask=mask_left,  other=0.0)   # (BLOCK_I, BLOCK_K, BLOCK_D)
        right_tile = tl.load(right_ptrs, mask=mask_right, other=0.0)   # (BLOCK_K, BLOCK_J, BLOCK_D)

        # inner dot for each hidden slice in the tile
        for d in range(0, BLOCK_D):
            a = left_tile[:, :, d]    # (BLOCK_I, BLOCK_K)
            b = right_tile[:, :, d]   # (BLOCK_K, BLOCK_J)
            acc[:, :, d] += tl.dot(a, b)

    # write the result
    out_ptrs = (out_ptr
                + i_offset[:, None, None] * stride_oi
                + j_offset[None, :, None] * stride_oj
                + d_offset[None, None, :] * stride_od)

    mask_out = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
    tl.store(out_ptrs, acc, mask=mask_out)


def custom_kernel(data):
    """
    Forward pass of the TriMul module (outgoing variant).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (may be all‑ones)
        - weights      : dict of model parameters
        - config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension `dim`
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 2) Linear projections to hidden dimension (no bias)
    # -----------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"]) # [B,N,N,H]

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_uns = mask.unsqueeze(-1).type_as(left)   # [B,N,N,1]
        left  = left * mask_uns
        right = right * mask_uns

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid of a linear layer on the normalized input)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Triton kernel for the heavy O(N³) reduction
    # -----------------------------------------------------------------
    left  = left.contiguous()
    right = right.contiguous()
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Tile sizes (tuned for the given workloads)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_D = 8   # hidden dimension is tiled by 8; mask handles leftovers

    grid = (
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
        triton.cdiv(hidden_dim, BLOCK_D),
    )

    # Strides in *elements* (not bytes)
    stride_li = left.stride(1)   # over i
    stride_lk = left.stride(2)   # over k
    stride_ld = left.stride(3)   # over hidden
    stride_rj = right.stride(1)  # over j
    stride_rk = right.stride(2)  # over k
    stride_rd = right.stride(3)  # over hidden
    stride_oi = out.stride(1)    # over i
    stride_oj = out.stride(2)    # over j
    stride_od = out.stride(3)    # over hidden

    for b in range(bs):
        # Base pointers for batch `b`
        left_base  = left.data_ptr()  + b * left.stride(0)
        right_base = right.data_ptr() + b * right.stride(0)
        out_base   = out.data_ptr()   + b * out.stride(0)

        trimul_outgoing_kernel[grid](
            left_base,
            right_base,
            out_base,
            N,
            hidden_dim,
            stride_li,
            stride_lk,
            stride_ld,
            stride_rj,
            stride_rk,
            stride_rd,
            stride_oi,
            stride_oj,
            stride_od,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )

    # -----------------------------------------------------------------
    # 6) Output norm, output gate and final projection back to `dim`
    # -----------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate
    output = F.linear(out_norm, weights["to_out.weight"])

    return output
