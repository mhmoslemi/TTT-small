@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, TASKS,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_task = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    # Bounds
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    # base pointer offsets
    # each element dtype size given by tl.float32
    # offset for task in flattened memory: task_offset = pid_task * N * N
    task_offset = pid_task * N * N
    # pointers
    left_base = left_ptr + task_offset
    right_base = right_ptr + task_offset
    out_base = out_ptr + task_offset

    # compute offsets
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate k blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left_tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + (offs_i[:, None] * N + offs_k[None, :])
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right_tile transposed: (BLOCK_K, BLOCK_N)
        # right[task, k, j]
        right_ptrs = right_base + (offs_k[:, None] * N + offs_j[None, :])
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate dot product
        acc += tl.dot(left, right)

    # store output
    out_ptrs = out_base + (offs_i[:, None] * N + offs_j[None, :])
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; batch_size, seq_len, _, _ = input_tensor.shape

    # LayerNorm
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    eps = 1e-5
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask if present (if config.get("nomask", False) maybe)
    if not config.get("nomask", False):
        mask_f = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_f
        right = right * mask_f
    else:
        # no mask: maybe do nothing
        pass

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Prepare for kernel
    # Shape: [B, N, N, hidden_dim]
    # Permute to [B, hidden_dim, N, N]
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()

    # Flatten tasks dimension
    B, H, N, _ = left_perm.shape

import torch
import triton
import triton.language as tl

@triton.jit
def trimul_kernel(left_ptr, right_ptr, out_ptr,
                  N,
                  BLOCK_M: tl.constexpr,
                  BLOCK_N: tl.constexpr,
                  BLOCK_K: tl.constexpr):
    pid_task = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Compute offsets
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # Masks for i and j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointer for this task (flat matrix)
    task_offset = pid_task * N * N
    left_base = left_ptr + task_offset
    right_base = right_ptr + task_offset
    out_base = out_ptr + task_offset

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + (offs_i[:, None] * N + offs_k[None, :])
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile transposed: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + (offs_k[:, None] * N + offs_j[None, :])
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left, right)

    # Store result
    out_ptrs = out_base + (offs_i[:, None] * N + offs_j[None, :])
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

def custom_kernel(data):
    """
    Triton implementation of the outgoing Triangle Multiplicative Update (TriMul) operator.
    Performs LayerNorm, linear projections, gating, masked multiplication, and
    a batched matrix multiplication with a custom Triton kernel.
    Returns a tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    import torch.nn.functional as F
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # LayerNorm on the last dimension
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask if provided (mask shape: [B,N,N])
    mask_f = mask.unsqueeze(-1).to(x.dtype)  # shape [B,N,N,1]
    left = left * mask_f
    right = right * mask_f

    # Gates
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Rearrange to [B, hidden_dim, N, N] and make contiguous for Triton kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape  # H == hidden_dim, N == seq_len
    total_tasks = B * H

    left_flat = left_perm.view(total_tasks, N, N)
    right_flat = right_perm.view(total_tasks, N, N)

    out_flat = torch.empty_like(left_flat)

    # Triton launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (total_tasks,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    trimul_kernel[grid](
        left_flat.data_ptr(),
        right_flat.data_ptr(),
        out_flat.data_ptr(),
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Final LayerNorm over hidden dimension
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to original dimension
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) Triton implementation.

The forward pass consists of:
1. LayerNorm over the feature dimension.
2. Linear projections (left/right) without bias.
3. Optional mask application.
4. Sigmoid gating (left, right, out).
5. Batched matrix multiplication over the sequence dimension:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This O(N³) step is performed by a custom Triton kernel.
6. LayerNorm over the hidden dimension, out‑gate multiplication,
   and final linear projection back to the original feature dimension.

All tensors are kept in float32 for numerical stability; the kernel
operates directly on float32 data. The kernel processes each (batch,
hidden) slice as an independent matrix multiplication, tiled over
M/N/K dimensions.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_kernel(left_ptr, right_ptr, out_ptr,
                   N,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """Batched matrix multiplication for TriMul:
    out[t,i,j] = Σ_k left[t,i,k] * right[t,k,j]
    where t = batch * hidden_dim.
    """
    pid_task = tl.program_id(0)   # task = batch * hidden_dim
    pid_i = tl.program_id(1)      # tile over rows (i)
    pid_j = tl.program_id(2)      # tile over cols (j)

    # Starting indices of the current tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Offsets within the tile
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointer for this task (flattened matrix of size N×N)
    task_offset = pid_task * N * N
    left_base = left_ptr + task_offset
    right_base = right_ptr + task_offset
    out_base = out_ptr + task_offset

    # Accumulator for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[t, i, k]  -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + (offs_i[:, None] * N + offs_k[None, :])
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right[t, k, j] (transposed view of right[t, j, k]) -> shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + (offs_k[:, None] * N + offs_j[None, :])
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # Accumulate dot product over k block
        acc += tl.dot(left, right)

    # Write out the computed tile
    out_ptrs = out_base + (offs_i[:, None] * N + offs_j[None, :])
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton implementation of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple of (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (binary mask, may be all ones)
            - weights: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with at least "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    # Ensure all weight tensors are on the correct device/dtype
    weights = {k: v.to(device, dtype=dtype) for k, v in weights.items()}

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # -------------------------------------------------------------
    # 1. LayerNorm over the feature dimension
    # -------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # -------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------------------
    # 3. Mask (if any)
    # -------------------------------------------------------------
    mask_f = mask.unsqueeze(-1).to(dtype)                # [B, N, N, 1]
    left = left * mask_f
    right = right * mask_f

    # -------------------------------------------------------------
    # 4. Gating (sigmoid)
    # -------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------
    # 5. Batched matrix multiplication via Triton kernel
    # -------------------------------------------------------------
    # Rearrange to [B, hidden_dim, N, N] for contiguous layout
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Flatten the first two dimensions: total_tasks = B * hidden_dim
    total_tasks = B * hidden_dim
    left_flat = left_perm.view(total_tasks, N, N)
    right_flat = right_perm.view(total_tasks, N, N)

    out_flat = torch.empty_like(left_flat)  # allocate output

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (total_tasks,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _trimul_kernel[grid](
        left_flat.data_ptr(),
        right_flat.data_ptr(),
        out_flat.data_ptr(),
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to original layout [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------------------
    # 6. Final LayerNorm, out‑gate, and projection back to dim
    # -------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
