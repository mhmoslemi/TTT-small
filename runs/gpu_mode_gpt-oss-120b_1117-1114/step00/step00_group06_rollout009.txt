def tri_mul_core(left, right, mask, out_gate, config):
    B, N, N, H = left.shape

    # left and right have dims [B,N,N,H] after gating and mask.
    # reshape for batched GEMM
    left_mat = left.permute(0,3,1,2).reshape(B*H, N, N)  # shape (B*H,N,N)
    right_mat = right.permute(0,3,2,1).reshape(B*H, N, N)  # swap N dims and reshape

    # Cast to fp16 for speed
    left_mat = left_mat.to(torch.float16)
    right_mat = right_mat.to(torch.float16)

    out_mat = torch.bmm(left_mat, right_mat) # shape (B*H,N,N)

    # Reshape back to [B,N,N,H] 
    out = out_mat.view(B, H, N, N).permute(0,2,3,1) # shape (B,N,N,H)

    # Convert back to float32 for subsequent layers
    out = out.to(torch.float32)

    # layernorm + out_gate + final linear
    out = to_out_norm(out)
    out = out * out_gate
    out = to_out(out)
    return out

def fuse_gate_mask_kernel(left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
                         out_left_ptr, out_right_ptr,
                         B, N, H,
                         stride_left_b, stride_left_n1, stride_left_n2, stride_left_h,
                         stride_right_b, stride_right_n1, stride_right_n2, stride_right_h,
                         stride_gate_b, stride_gate_n1, stride_gate_n2, stride_gate_h,
                         stride_mask_b, stride_mask_n1, stride_mask_n2,
                         BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    total = B * N * N
    # Compute linear index for each "element" (positions of i,j)
    # We'll compute per (b, i, j) block and iterate over hidden dim inside.
    # For simplicity consider each thread processes one hidden dimension element.
    # Or each block processes a chunk of hidden dimension.
    # We'll index linear offset = pid * BLOCK + offset
    # offset < total*H
    off = pid * BLOCK + tl.arange(0, BLOCK)
    # Convert to global indices:
    total_hw = total * H
    mask = off < total_hw
    # Compute b,i,j,h from flat index: 
    # pos = off[mask] // H gives index into (b,i,j)
    # h = off[mask] % H
    pos = off // H
    h = off % H

    # decode pos into b,i,j using integer division
    b = pos // (N * N)
    rem = pos % (N * N)
    i = rem // N
    j = rem % N

    # compute pointer offsets
    left_offset = b * stride_left_b + i * stride_left_n1 + j * stride_left_n2 + h * stride_left_h
    right_offset = b * stride_right_b + i * stride_right_n1 + j * stride_right_n2 + h * stride_right_h
    left_gate_offset = b * stride_gate_b + i * stride_gate_n1 + j * stride_gate_n2 + h * stride_gate_h
    right_gate_offset = b * stride_gate_b + i * stride_gate_n1 + j * stride_gate_n2 + h * stride_gate_h
    mask_offset = b * stride_mask_b + i * stride_mask_n1 + j * stride_mask_n2  # mask scalar

    # load values
    left_val = tl.load(left_ptr + left_offset, mask=mask, other=0.0)
    right_val = tl.load(right_ptr + right_offset, mask=mask, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + left_gate_offset, mask=mask, other=0.0)

    tl.store(out_left_ptr + left_offset, out_left_val, mask=mask)

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel to broadcast mask over hidden dimension and multiply left/right.
@triton.jit
def mask_apply_kernel(left_ptr, right_ptr, mask_ptr,
                      total_elements, H: tl.constexpr, BLOCK: tl.constexpr):
    """
    In-place fuse mask (shape [B, N, N]) into left and right tensors
    (shape [B, N, N, H]). Each thread handles one element of hidden dimension.
    """
    pid = tl.program_id(0)
    # linear offset for this block
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    # guard
    mask = offs < total_elements

    # Load left and right values
    left_val = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right_val = tl.load(right_ptr + offs, mask=mask, other=0.0)

    # Compute index into mask tensor (mask has size total_elements // H)
    mask_idx = offs // H
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    left_val = left_val * mask_val
    right_val = right_val * mask_val

    tl.store(left_ptr + offs, left_val, mask=mask)
    tl.store(right_ptr + offs, right_val, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) operation.
    Steps:
      1. Layer‑norm the input.
      2. Linear projections for left/right and their gates.
      3. Fuse mask (if present) into the gated tensors using a Triton kernel.
      4. Batched matrix multiplication per hidden dimension:
         out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
      5. Layer‑norm, out‑gate and final linear projection.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Load model weights (they are already torch tensors on the correct device)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1. LayerNorm over dim
    x = input_tensor.to(torch.float32)
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # 2. Projections and gates (all in float32)
    proj_left = F.linear(x, left_proj_weight)          # [B,N,N,hidden_dim]
    proj_right = F.linear(x, right_proj_weight)

    gate_left = torch.sigmoid(F.linear(x, left_gate_weight))
    gate_right = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = proj_left * gate_left    # shape [B,N,N,H]
    right = proj_right * gate_right

    # 3. Mask fusion via Triton (if mask is present)
    if not nomask:
        # Ensure tensors are contiguous and on GPU
        left = left.contiguous()
        right = right.contiguous()
        mask_f = mask.to(torch.float32).contiguous()
        total_elements = left.numel()            # B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)
        mask_apply_kernel[grid](left, right, mask_f,
                                total_elements, hidden_dim,
                                BLOCK=BLOCK)

    # 4. Batched GEMM per hidden dimension (L @ R^T)
    B, N, _, H = left.shape
    # reshape for batched matmul
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * H, N, N)  # transpose the two N axes
    # compute in half precision for speed
    left_mat = left_mat.to(torch.float16)
    right_mat = right_mat.to(torch.float16)
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N)
    # restore shape [B,N,N,H]
    out = out_mat.to(torch.float32).view(B, H, N, N).permute(0, 2, 3, 1)

    # 5. Output LayerNorm over hidden_dim, gating and final projection
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # final shape [B,N,N,dim]

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: in‑place mask application.
# Operates on flattened tensors with shape [B, N, N, H] (contiguous)
# and a mask of shape [B, N, N].  Each thread processes one element of the
# hidden dimension, broadcasting the scalar mask value.
# -------------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    left_ptr,          # float32*  left tensor (B,N,N,H) – in‑place updated
    right_ptr,         # float32*  right tensor (B,N,N,H) – in‑place updated
    mask_ptr,          # float32*  mask tensor (B,N,N)
    total_elements,   # i64      total number of elements in left/right
    H: tl.constexpr,   # hidden dimension (compile‑time constant)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)               # linear offsets
    mask_off = offs < total_elements                        # bounds check

    # Load values
    left_val  = tl.load(left_ptr + offs,  mask=mask_off, other=0.0)
    right_val = tl.load(right_ptr + offs, mask=mask_off, other=0.0)

    # Compute corresponding mask index (mask is H‑times smaller)
    mask_idx = offs // H
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask_off, other=0.0)

    # Apply mask
    left_val  = left_val  * mask_val
    right_val = right_val * mask_val

    # Store results back in place
    tl.store(left_ptr + offs,  left_val,  mask=mask_off)
    tl.store(right_ptr + offs, right_val, mask=mask_off)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) operation.

    Steps:
      1. Layer‑norm the input tensor.
      2. Linear projections for left/right and their sigmoid gates.
      3. Fuse the optional pair‑wise mask into the gated tensors via a Triton kernel.
      4. Perform a batched GEMM per hidden dimension:
         out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]   (i.e. L @ Rᵀ).
      5. Apply a second LayerNorm, the output gate, and the final linear projection.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # Load model weights (all tensors are already on the correct device)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # -----------------------------------------------------------------
    x = input_tensor.to(torch.float32)
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # -----------------------------------------------------------------
    # 2. Linear projections + sigmoid gates (still in FP32)
    # -----------------------------------------------------------------
    proj_left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    proj_right = F.linear(x, right_proj_weight)

    gate_left  = torch.sigmoid(F.linear(x, left_gate_weight))
    gate_right = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left  = proj_left  * gate_left   # shape [B, N, N, hidden_dim]
    right = proj_right * gate_right

    # -----------------------------------------------------------------
    # 3. Optional mask fusion via Triton (in‑place)
    # -----------------------------------------------------------------
    if not nomask:
        left  = left.contiguous()
        right = right.contiguous()
        mask_f = mask.to(torch.float32).contiguous()

        total_elements = left.numel()                     # B * N * N * hidden_dim
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        mask_apply_kernel[grid](
            left,
            right,
            mask_f,
            total_elements,
            H=hidden_dim,
            BLOCK=BLOCK,
        )

    # -----------------------------------------------------------------
    # 4. Batched GEMM: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a series of independent matrix multiplications
    #    L @ Rᵀ for each hidden channel.
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) matrices
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * H, N, N)   # transpose the two N axes

    # compute in half‑precision for speed
    left_mat = left_mat.to(torch.float16)
    right_mat = right_mat.to(torch.float16)

    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N)

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.to(torch.float32).view(B, H, N, N).permute(0, 2, 3, 1)

    # -----------------------------------------------------------------
    # 5. Output normalization, gating and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # final shape [B, N, N, dim]

    return out
