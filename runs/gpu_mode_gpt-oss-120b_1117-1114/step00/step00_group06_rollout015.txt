def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k,
    stride_right_b, stride_right_i, stride_right_k,
    stride_out_b, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # i tile
    pid_n = tl.program_id(1)  # j tile
    pid_bh = tl.program_id(2)  # batch*hidden index

    # Compute which batch and hidden dim
    bh = pid_bh
    b = bh // H
    h = bh % H

    # Offsets for i and j start
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Mask for valid k
        k_valid = offs_k < N

        # Pointers for this tile
        left_offset = (b * stride_left_b) + (h) + (offs_m[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        # left shape (BLOCK_M, BLOCK_K)
        left = tl.load(left_ptr + left_offset, mask=offs_m[:, None] < N and k_valid[None, :], other=0.0)

        # Right is accessed as (k, j)
        right_offset = (b * stride_right_b) + (h) + (offs_k[:, None] * stride_right_k) + (offs_n[None, :] * stride_right_i)

        acc += tl.dot(left, right)  # shapes (M,K) dot (K,N) -> (M,N)

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out of bounds

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile shape (M,K)
        left_offset = left_base + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_offset, mask=mask_m[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)

        # load right tile shape (K,N)

    out_offset = out_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_offset, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) implementation with a fused Triton kernel.

The heavy part of the TriMul module is the einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We rewrite this as a batched matrix multiplication
    out[b,d] = left[b,d] @ right[b,d].T
where left/right are reshaped to [B, D, N, N] (D = hidden_dim).
The Triton kernel computes the multiplication tile‑wise (M×N blocks)
with an inner reduction over K, handling an optional mask and using
float16 for the bulk of the work while accumulating in float32.
All other operations (LayerNorm, Linear projections, gates) are done
with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid_m = tl.program_id(0)          # i tile
    pid_n = tl.program_id(1)          # j tile
    pid_bh = tl.program_id(2)         # combined batch- hidden index

    # Decode batch and hidden dimension
    bh = pid_bh
    b = bh // H
    h = bh % H

    # Offsets for the i and j dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this (b, h)
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile [M, K] (float16)
        left_offset = left_base + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_offset,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.float16)

        # Load right tile [K, N] (float16). This is the transposed view.
        right_offset = right_base + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_i
        right = tl.load(right_offset,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0,
                        dtype=tl.float16)

        # Accumulate the matrix product; result automatically promoted to FP32
        acc += tl.dot(left, right)

    # Store the result
    out_offset = out_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_offset, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, dim]
            mask:          torch.Tensor [B, N, N] (bool or numeric)
            weights:       dict of torch.Tensors containing the model parameters
            config:        dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(device))
    right = F.linear(x, weights["right_proj.weight"].to(device))

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid) and optional mask
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(device)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(device)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(device)).sigmoid()

    # apply mask if present (mask is already on device)
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(x.dtype)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Cast to FP16 and launch the Triton kernel
    # ------------------------------------------------------------------
    left = left.contiguous().to(torch.float16)
    right = right.contiguous().to(torch.float16)

    B, N, _, H = left.shape  # H == hidden_dim
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # Block sizes – chosen to give good occupancy on the B200 GPU
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (i-tiles, j-tiles, batch*hidden)
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    # Strides for the tensors
    l_stride_b, l_stride_i, l_stride_k, l_stride_h = left.stride()
    r_stride_b, r_stride_i, r_stride_k, r_stride_h = right.stride()
    o_stride_b, o_stride_i, o_stride_j, o_stride_h = out.stride()

    # Launch kernel
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        l_stride_b,
        l_stride_i,
        l_stride_k,
        l_stride_h,
        r_stride_b,
        r_stride_i,
        r_stride_k,
        r_stride_h,
        o_stride_b,
        o_stride_i,
        o_stride_j,
        o_stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 5) Post‑processing: LayerNorm (hidden), gating, final Linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device),
        bias=weights["to_out_norm.bias"].to(device),
        eps=eps,
    )

    out = out * out_gate  # broadcast over hidden dim

    out = F.linear(out, weights["to_out.weight"].to(device))  # -> [B,N,N,dim]

    return out
