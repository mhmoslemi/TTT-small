import torch.nn.functional as F

import triton.language as tl

@triton.jit
def fuse_mask_gate_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    mask_ptr, left_out_ptr, right_out_ptr,
    B, N, H,
    total_elems,
    BLOCK: tl.constexpr
):

def custom_kernel(data):
    """
    Triton-accelerated TriMul (outgoing) forward pass.
    Performs LayerNorm, linear projections, applies mask & sigmoid gates via Triton,
    batched matrix multiplication, output LayerNorm, gating and final linear projection.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # LayerNorm over dim
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)
    # Linear projections
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])
    left_gate = F.linear(x, weights["left_gate.weight"])
    right_gate = F.linear(x, weights["right_gate.weight"])
    out_gate = F.linear(x, weights["out_gate.weight"])
    # Prepare mask
    mask = mask_tensor.to(torch.float32).contiguous()
    # Allocate fused tensors
    left_fused = torch.empty_like(left_proj)
    right_fused = torch.empty_like(right_proj)
    total_elements = left_proj.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)
    # Launch Triton kernel
    fuse_mask_gate_kernel[grid](
        left_proj.data_ptr(),
        left_gate.data_ptr(),
        right_proj.data_ptr(),
        right_gate.data_ptr(),
        mask.view(-1).data_ptr(),
        left_fused.data_ptr(),
        right_fused.data_ptr(),
        bs,
        seq_len,
        hidden_dim,
        total_elements,
        BLOCK=BLOCK,
    )
    # Batched GEMM (left_fused @ right_fused^T) across hidden_dim
    # reshape to (B*H, N, N)
    left_mat = left_fused.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, seq_len, seq_len)
    right_mat = right_fused.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, seq_len, seq_len)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # reshape back to (B, N, N, H)
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()
    # Output LayerNorm
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # Apply out gate
    out_gate_sig = torch.sigmoid(out_gate)
    out_gated = out_norm * out_gate_sig
    # Final linear projection back to dim
    output = F.linear(out_gated, weights["to_out.weight"])
    return output

"""
Triton‑accelerated implementation of the outgoing TriMul operator.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension.
2. Linear projections to a hidden dimension (left/right) and linear
   gates (left/right/out) without bias.
3. Fuse mask multiplication and sigmoid‑gate application with a tiny
   Triton kernel (avoids an extra Python pass).
4. Perform the O(N³) reduction as a batched matrix multiplication:
   for each hidden channel d we compute  out[b,i,j,d] =
   Σ_k left[b,i,k,d] * right[b,j,k,d]  =  left @ rightᵀ .
5. Apply a second LayerNorm on the hidden dimension, multiply by the
   output gate, and project back to the original channel dimension.

Only the element‑wise mask‑gate fusion is written in Triton; the heavy
batched GEMM uses PyTorch's highly‑optimized `torch.bmm`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask and sigmoid gates for left/right tensors
# ----------------------------------------------------------------------
@triton.jit
def fuse_mask_gate_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    mask_ptr, left_out_ptr, right_out_ptr,
    B, N, H,
    total_elems,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total_elems

    # Each H consecutive elements share the same mask entry.
    mask_val = tl.load(mask_ptr + offs // H, mask=valid)

    # ----- left branch -------------------------------------------------
    left_val = tl.load(left_ptr + offs, mask=valid, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate_val))
    left_out = left_val * mask_val * left_gate_sig
    tl.store(left_out_ptr + offs, left_out, mask=valid)

    # ----- right branch ------------------------------------------------
    right_val = tl.load(right_ptr + offs, mask=valid, other=0.0)
    right_gate_val = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate_val))
    right_out = right_val * mask_val * right_gate_sig
    tl.store(right_out_ptr + offs, right_out, mask=valid)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑enabled forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [B, N, N, dim] torch.Tensor (float32, CUDA)
            - mask_tensor  : [B, N, N]        torch.Tensor (bool/float, CUDA)
            - weights_dict : mapping from weight name to torch.Tensor
            - config_dict  : {"dim": int, "hidden_dim": int}

    Returns:
        output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]          # hidden dimension

    # --------------------------------------------------------------
    # 1. Input LayerNorm (over the last dimension)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])          # [B,N,N,H]
    right_proj = F.linear(x, weights["right_proj.weight"])
    left_gate_pre = F.linear(x, weights["left_gate.weight"])
    right_gate_pre = F.linear(x, weights["right_gate.weight"])
    out_gate_pre = F.linear(x, weights["out_gate.weight"])

    # --------------------------------------------------------------
    # 3. Fuse mask and sigmoid gates via Triton
    # --------------------------------------------------------------
    mask = mask_tensor.to(torch.float32).contiguous()            # [B,N,N]
    left_fused = torch.empty_like(left_proj)
    right_fused = torch.empty_like(right_proj)

    total_elements = left_proj.numel()      # B * N * N * H
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)

    fuse_mask_gate_kernel[grid](
        left_proj.data_ptr(),
        left_gate_pre.data_ptr(),
        right_proj.data_ptr(),
        right_gate_pre.data_ptr(),
        mask.view(-1).data_ptr(),
        left_fused.data_ptr(),
        right_fused.data_ptr(),
        B,
        N,
        H,
        total_elements,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 4. Batched GEMM: for each hidden channel compute left @ rightᵀ
    # --------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_mat = left_fused.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_fused.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # out_mat = left_mat @ right_matᵀ   -> shape (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate, and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    out_gate = torch.sigmoid(out_gate_pre)          # [B,N,N,H]
    out = out * out_gate

    output = F.linear(out, weights["to_out.weight"])   # back to [B,N,N,dim]
    return output
