def mask_gate_kernel(
    inp_ptr, gate_ptr, mask_ptr, out_ptr,
    N, H,
    stride_in_i, stride_in_j, stride_in_h,
    stride_gate_i, stride_gate_j, stride_gate_h,

    BLOCK_H: tl.constexpr

    j = tl.program_id(1)
    # compute base offsets
    in_base = inp_ptr + i * stride_in_i + j * stride_in_j
    gate_base = gate_ptr + i * stride_gate_i + j * stride_gate_j

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Ensure dtype
    input_tensor = input_tensor.to(torch.float32)
    mask = mask.to(torch.float32)  # ensures broadcast
    # LayerNorm
    eps = 1e-5
    x_norm = torch.nn.functional.layer_norm(input_tensor,
                                            normalized_shape=(dim,),
                                            weight=weights["norm.weight"],
                                            bias=weights["norm.bias"],
                                            eps=eps)
    # Projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # Allocate output tensors for masked gated left/right
    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    # Determine block size for hidden dimension processing
    BLOCK_H = 64  # chosen based on hidden_dim <= 384

    # Launch kernel for each batch
    for b in range(bs):
        # get per-batch slices
        left_b = left[b]
        left_gate_b = left_gate[b]
        right_b = right[b]
        right_gate_b = right_gate[b]
        mask_b = mask[b]
        # output slices
        left_out_b = left_out[b]
        right_out_b = right_out[b]
        # strides for left
        stride_in_i = left_b.stride(0)  # N*H
        stride_in_j = left_b.stride(1)  # H
        stride_in_h = left_b.stride(2)  # 1
        stride_gate_i = left_gate_b.stride(0)
        stride_gate_j = left_gate_b.stride(1)
        stride_gate_h = left_gate_b.stride(2)
        stride_mask_i = mask_b.stride(0)
        stride_mask_j = mask_b.stride(1)
        # grid
        grid = (seq_len, seq_len)
        mask_gate_kernel[grid](
            left_b, left_gate_b, mask_b, left_out_b,
            N=seq_len, H=hidden_dim,
            stride_in_i=stride_in_i,
            stride_in_j=stride_in_j,
            stride_in_h=stride_in_h,
            stride_gate_i=stride_gate_i,
            stride_gate_j=stride_gate_j,
            stride_gate_h=stride_gate_h,
            stride_mask_i=stride_mask_i,
            stride_mask_j=stride_mask_j,
            BLOCK_H=BLOCK_H
        )
        mask_gate_kernel[grid](
            right_b, right_gate_b, mask_b, right_out_b,
            N=seq_len, H=hidden_dim,
            stride_in_i=stride_in_i,
            stride_in_j=stride_in_j,
            stride_in_h=stride_in_h,
            stride_gate_i=stride_gate_i,
            stride_gate_j=stride_gate_j,
            stride_gate_h=stride_gate_h,
            stride_mask_i=stride_mask_i,
            stride_mask_j=stride_mask_j,
            BLOCK_H=BLOCK_H
        )
    # After kernel, left_out and right_out contain left*mask*gate and similarly for right

    # Contraction using batched matrix multiplication
    # reshape to (bs*hidden_dim, seq_len, seq_len)
    left_perm = left_out.permute(0, 3, 1, 2).contiguous()
    right_perm = right_out.permute(0, 3, 1, 2).contiguous()
    B_H = bs * hidden_dim
    left_mat = left_perm.view(B_H, seq_len, seq_len)
    right_mat = right_perm.view(B_H, seq_len, seq_len)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # reshape back to (bs, seq_len, seq_len, hidden_dim)
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1)

    # to_out_norm
    out = torch.nn.functional.layer_norm(out,
                                         normalized_shape=(hidden_dim,),
                                         weight=weights["to_out_norm.weight"],
                                         bias=weights["to_out_norm.bias"],
                                         eps=eps)

    # apply out_gate
    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
Optimized TriMul forward pass.

The heavy O(N³·H) contraction is performed with a fused
mask‑gate kernel written in Triton (applies mask and per‑channel
gate in‑place) and the remaining matrix multiplications are done
with batched GEMM (torch.bmm). All other linear / layer‑norm
operations stay in PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: apply mask and gate (elementwise) to a [N, N, H] tensor.
# One thread processes one (i, j) element and loops over the hidden
# dimension H in blocks.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    inp_ptr,               # float32*   input tensor (N,N,H)
    gate_ptr,              # float32*   gate tensor (N,N,H)
    mask_ptr,              # float32*   mask tensor (N,N)
    out_ptr,               # float32*   output (can be same as inp_ptr)
    N,                     # tl.constexpr   sequence length
    H,                     # tl.constexpr   hidden dimension
    stride_in_i,           # tl.constexpr   stride of i dimension in inp
    stride_in_j,           # tl.constexpr   stride of j dimension in inp
    stride_in_h,           # tl.constexpr   stride of h dimension in inp
    stride_gate_i,         # tl.constexpr   stride of i dim in gate
    stride_gate_j,         # tl.constexpr   stride of j dim in gate
    stride_gate_h,         # tl.constexpr   stride of h dim in gate
    stride_mask_i,         # tl.constexpr   stride of i dim in mask
    stride_mask_j,         # tl.constexpr   stride of j dim in mask
    BLOCK_H: tl.constexpr  # block size along hidden dimension
):
    i = tl.program_id(0)
    j = tl.program_id(1)

    # base pointers for this (i, j) element
    inp_base   = inp_ptr   + i * stride_in_i   + j * stride_in_j
    gate_base  = gate_ptr  + i * stride_gate_i  + j * stride_gate_j
    out_base   = out_ptr   + i * stride_in_i   + j * stride_in_j
    mask_off   = i * stride_mask_i + j * stride_mask_j
    mask_val   = tl.load(mask_ptr + mask_off, mask=True)

    # iterate over hidden dimension
    for start in range(0, H, BLOCK_H):
        offs_h = tl.arange(0, BLOCK_H) + start
        active = offs_h < H

        # load inputs
        val   = tl.load(inp_base  + offs_h * stride_in_h,   mask=active, other=0.0)
        gate  = tl.load(gate_base + offs_h * stride_gate_h, mask=active, other=0.0)

        # compute masked gated value
        out = val * gate * mask_val

        # store result (in‑place possible)
        tl.store(out_base + offs_h * stride_in_h, out, mask=active)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module with a Triton kernel for mask‑gate fusion.

    Args:
        data: Tuple of (input, mask, weights, config)
            - input: torch.Tensor [B, N, N, D] (float32, on CUDA)
            - mask:  torch.Tensor [B, N, N] (float32 or bool, on CUDA)
            - weights: dict of torch.Tensors for all linear / layer‑norm params
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim = D)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(D,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid) based on the same normalized input
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Apply mask and gates using Triton (in‑place on left/right)
    # ------------------------------------------------------------------
    # Ensure mask is float32 for the kernel
    mask_f = mask.to(torch.float32)

    BLOCK_H = 64  # chosen to cover hidden dimensions up to 384 efficiently
    grid = (N, N)  # one program for each (i, j) pair

    for b in range(B):
        # slice view for the current batch
        left_b   = left[b]
        right_b  = right[b]
        left_g_b = left_gate[b]
        right_g_b = right_gate[b]
        mask_b   = mask_f[b]

        # strides for left/right tensors (shape [N, N, H])
        s_in_i = left_b.stride(0)   # N*H
        s_in_j = left_b.stride(1)   # H
        s_in_h = left_b.stride(2)   # 1

        s_gate_i = left_g_b.stride(0)
        s_gate_j = left_g_b.stride(1)
        s_gate_h = left_g_b.stride(2)

        s_mask_i = mask_b.stride(0)   # N
        s_mask_j = mask_b.stride(1)   # 1

        # left = left * mask * left_gate
        mask_gate_kernel[grid](
            left_b, left_g_b, mask_b, left_b,
            N, H,
            s_in_i, s_in_j, s_in_h,
            s_gate_i, s_gate_j, s_gate_h,
            s_mask_i, s_mask_j,
            BLOCK_H=BLOCK_H,
        )
        # right = right * mask * right_gate
        mask_gate_kernel[grid](
            right_b, right_g_b, mask_b, right_b,
            N, H,
            s_in_i, s_in_j, s_in_h,
            s_gate_i, s_gate_j, s_gate_h,
            s_mask_i, s_mask_j,
            BLOCK_H=BLOCK_H,
        )

    # ------------------------------------------------------------------
    # 5. Contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM where each (batch,channel) pair
    #    is a separate matrix multiplication.
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for left and right
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # B, H, N, N
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # B, H, N, N

    BH = B * H
    left_mat  = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    # batched matrix multiplication (left @ right^T)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output layernorm over hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7. Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original dimension D
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # ensure float32 output
    return out.to(torch.float32)
