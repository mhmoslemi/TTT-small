"""
TriMul forward pass with Triton‑accelerated contraction.

Algorithm
---------
1. Layer‑norm the input tensor using provided weights.
2. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) are computed with torch.nn.functional.linear.
   Mask (if present) and gates are applied in PyTorch.
3. The core O(N³·D) contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is performed by a custom Triton kernel.  For each batch‑channel pair
   (b,d) the kernel executes a blocked matrix multiplication
   `left[b,:,:,d] @ right[b,:,:,d].T`.
4. Apply the output gate, a second LayerNorm, and a final linear
   projection to obtain the result tensor of shape [B, N, N, dim].

Only the heavy contraction (step 3) runs on Triton; the surrounding
operations stay in PyTorch for simplicity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,          # const float*  (B,N,N,D)
    right_ptr,         # const float*  (B,N,N,D)
    out_ptr,           #       float*  (B,N,N,D)
    B, N, D,
    stride_left_bc, stride_left_i, stride_left_k, stride_left_d,
    stride_right_bc, stride_right_j, stride_right_k, stride_right_d,
    stride_out_bc, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # block index over rows (i)
    pid_n = tl.program_id(1)          # block index over cols (j)
    pid_z = tl.program_id(2)          # combined batch and channel index

    # Decode batch and hidden dimension from pid_z
    batch = pid_z // D
    d = pid_z % D

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i coordinates
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j coordinates
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator for the tile (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k in blocks
    num_blocks_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_blocks_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)   # k coordinates
        mask_k = offs_k < N

        # ---- Load left tile: shape (BLOCK_M, BLOCK_K) ----
        # left[b, i, k, d]
        left_ptrs = (
            left_ptr
            + batch * stride_left_bc
            + d * stride_left_d
            + offs_m[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # float32

        # ---- Load right tile (transposed): shape (BLOCK_K, BLOCK_N) ----
        # We need right_T[k, j] = right[b, j, k, d]
        right_ptrs = (
            right_ptr
            + batch * stride_right_bc
            + d * stride_right_d
            + offs_k[:, None] * stride_right_k
            + offs_n[None, :] * stride_right_j
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # float32

        # Dot product over k dimension
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N)

    # Store the result tile
    out_ptrs = (
        out_ptr
        + batch * stride_out_bc
        + d * stride_out_d
        + offs_m[:, None] * stride_out_i
        + offs_n[None, :] * stride_out_j
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the Triangle Multiplicative Update (TriMul) module.
    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask:        torch.Tensor [B, N, N] (ignored if config['nomask'] is True)
            - weights:     dict of weight tensors
            - config:      dict with keys 'dim', 'hidden_dim', optionally 'nomask'
    Returns:
        output: torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------- LayerNorm on input ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous tensors for Triton
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # Strides (in element counts, not bytes)
    stride_left_bc = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_d = left.stride(3)

    stride_right_bc = right.stride(0)
    stride_right_j = right.stride(1)      # same as left_i
    stride_right_k = right.stride(2)
    stride_right_d = right.stride(3)

    stride_out_bc = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_d = out.stride(3)

    # ---------- Triton kernel launch ----------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left_bc,
        stride_left_i,
        stride_left_k,
        stride_left_d,
        stride_right_bc,
        stride_right_j,
        stride_right_k,
        stride_right_d,
        stride_out_bc,
        stride_out_i,
        stride_out_j,
        stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Post‑processing ----------
    # Apply output gate after the second LayerNorm (as in reference)
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out_norm * out_gate
    output = F.linear(out, weights["to_out.weight"])   # final projection to [dim]

    return output
