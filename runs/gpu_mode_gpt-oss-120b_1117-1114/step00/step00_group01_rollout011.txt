def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    # Access weights
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]; right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]; right_gate_weight = weights["right_gate.weight"]; out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]; to_out_norm_bias = weights["to_out_norm.bias"]; to_out_weight = weights["to_out.weight"]
    # Compute normalized x
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)
    # Compute left and right projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)
    # Apply mask if not nomask; mask shape [B,N,N], unsqueeze last dim
    mask_unsqueeze = mask.unsqueeze(-1) # shape [B,N,N,1]
    left = left * mask_unsqueeze
    right = right * mask_unsqueeze
    # Compute gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate
    # Cast to half for kernel
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)
    # Output allocate
    out_f16 = torch.empty_like(left_f16) # shape B,N,N,H
    # Launch kernel
    B, N, _, H = left_f16.shape
    # strides:
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_f16.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_f16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_f16.stride()
    # Choose block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64  # might adjust
    # grid: (ceil(N/BLOCK_M), ceil(N/BLOCK_N), B*H)
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B*H)
    tri_mul_kernel[grid](
        left_f16, right_f16, out_f16,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # convert out to float32
    out = out_f16.to(torch.float32)
    # Apply to_out_norm: layer norm across hidden_dim
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    # Multiply by out_gate (already shape [B,N,N,H])
    out = out * out_gate
    # Final linear projection to dim: shape [B,N,N,C] where C=dim
    out = torch.nn.functional.linear(out, to_out_weight) # weight shape [dim, hidden_dim]
    return out

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(2)
    b_idx = pid_bh // H
    h_idx = pid_bh % H

    pid_m = tl.program_id(0) # block row i
    pid_n = tl.program_id(1) # block col j

    # start indices
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # Offsets for i and j
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # Mask for boundary
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Compute pointers base for this batch and hidden
    left_base = left_ptr + b_idx*stride_l_b + h_idx*stride_l_h
    right_base = right_ptr + b_idx*stride_r_b + h_idx*stride_r_h
    out_base = out_ptr + b_idx*stride_o_b + h_idx*stride_o_h

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, N, BLOCK_K):
        # Compute k offsets
        k_start = k
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile A: shape (BLOCK_M, BLOCK_K)
        # Offsets: row i, column k
        a_ptrs = left_base + (offs_i[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile B_T: shape (BLOCK_K, BLOCK_N) using transposed indices
        # Right index original: [b, k, j, h] (i swapped with k)
        b_ptrs = right_base + (offs_k[:, None] * stride_r_i) + (offs_j[None, :] * stride_r_j)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Compute dot product over k dimension
        acc += tl.dot(a, b)   # a: (M,K), b: (K,N) => (M,N)

    # Write output
    out_ptrs = out_base + (offs_i[:, None] * stride_o_i) + (offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑normalize the input tensor `x` (shape [B, N, N, C]).
2. Linear projections:
   - left  = Linear(x, left_proj_weight)   # → [B,N,N,H]
   - right = Linear(x, right_proj_weight) # → [B,N,N,H]
3. Optional mask multiplication (broadcasted over the hidden dimension).
4. Gating:
   - left_gate  = sigmoid(Linear(x, left_gate_weight))
   - right_gate = sigmoid(Linear(x, right_gate_weight))
   - out_gate   = sigmoid(Linear(x, out_gate_weight))
   Apply gates: left  *= left_gate ; right *= right_gate
5. Core N³ operation:
   out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,k,j,h}
   Implemented as a batched matrix‑multiply (A @ Bᵀ) using a Triton kernel.
   The kernel loops over the shared dimension `k` in tiles (BLOCK_K) and
   accumulates in float32 while loading from float16.
6. Convert the kernel output back to float32.
7. Layer‑normalize the result (hidden dimension H) and apply `out_gate`.
8. Final linear projection back to `C` dimensions.

Only the heavy N³ multiplication is performed on the GPU with Triton;
all other operations use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for left   (B,N,i,k,H)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # strides for right  (B,N,i,j,H) – we will access as (k,j)
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    # strides for out    (B,N,i,j,H)
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(2)                      # combined batch & hidden index
    b_idx = pid_bh // H
    h_idx = pid_bh % H

    pid_m = tl.program_id(0)                       # tile row (i)
    pid_n = tl.program_id(1)                       # tile col (j)

    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (batch, hidden) pair
    left_base  = left_ptr  + b_idx * stride_l_b + h_idx * stride_l_h
    right_base = right_ptr + b_idx * stride_r_b + h_idx * stride_r_h
    out_base   = out_ptr   + b_idx * stride_o_b + h_idx * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the shared dimension k
    for k in range(0, N, BLOCK_K):
        k_start = k
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + (offs_i[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right tile Bᵀ: shape (BLOCK_K, BLOCK_N)
        # we need right[b, k, j, h] → use stride_r_i for k and stride_r_j for j
        b_ptrs = right_base + (offs_k[:, None] * stride_r_i) + (offs_j[None, :] * stride_r_j)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # Accumulate dot product over k
        acc += tl.dot(a, b)   # a:(M,K), b:(K,N) → (M,N)
    # Store the result
    out_ptrs = out_base + (offs_i[:, None] * stride_o_i) + (offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        output tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # ------------------------------------------------------------------
    # Load weights onto the correct device and dtype
    # ------------------------------------------------------------------
    norm_weight      = weights["norm.weight"].to(device, dtype)
    norm_bias        = weights["norm.bias"].to(device, dtype)
    left_proj_weight = weights["left_proj.weight"].to(device, dtype)
    right_proj_weight= weights["right_proj.weight"].to(device, dtype)
    left_gate_weight = weights["left_gate.weight"].to(device, dtype)
    right_gate_weight= weights["right_gate.weight"].to(device, dtype)
    out_gate_weight  = weights["out_gate.weight"].to(device, dtype)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype)
    to_out_weight    = weights["to_out.weight"].to(device, dtype)

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2) Projections
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Mask (if provided)
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton kernel for the N³ multiply‑accumulate
    # ------------------------------------------------------------------
    # Cast to half for the kernel (more bandwidth efficient)
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    B, N, _, H = left_f16.shape
    out_f16 = torch.empty_like(left_f16)   # same shape as left/right

    # strides (int64)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_f16.stride()
    stride_r_b, stride_r_i, stride_r_j, stride_r_h = right_f16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_f16.stride()

    # Tile sizes – chosen to give good occupancy on B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows i
        triton.cdiv(N, BLOCK_N),   # cols j
        B * H                      # batch * hidden dimension
    )

    tri_mul_kernel[grid](
        left_f16, right_f16, out_f16,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_j, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    out = out_f16.to(dtype)           # back to float32

    # ------------------------------------------------------------------
    # 6) Post‑processing: layer‑norm, output gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # → [B,N,N,dim]

    return out
