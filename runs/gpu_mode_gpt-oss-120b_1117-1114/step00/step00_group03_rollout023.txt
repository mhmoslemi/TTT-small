def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # retrieve weight tensors and biases

    # Apply LayerNorm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Apply linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if needed
    if not config.get("nomask", False) and mask is not None:
        mask_ = mask.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_
        right = right * mask_
    else:
        mask_ = None

    # Compute gating
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast left/right to fp16 for kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Prepare output buffer
    out_fp16 = torch.empty((bs, N, N, hidden_dim), dtype=torch.float16, device=left_fp16.device)

    # Compute contraction using Triton
    _contract_triton(left_fp16, right_fp16, out_fp16, BLOCK_M=64, BLOCK_N=64, BLOCK_K=32)

    # Cast to float32 for remaining ops
    out = out_fp16.to(torch.float32)

    # Apply to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # Apply out_gate
    out = out * out_gate

    # Apply final linear to dim
    final_out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return final_out

@triton.jit
def kernel_contract(
    left_ptr, right_ptr, out_ptr,
    bs, N, H,
    stride_b_l, stride_i_l, stride_k_l, stride_h_l,
    stride_b_r, stride_i_r, stride_k_r, stride_h_r,
    stride_b_o, stride_i_o, stride_j_o, stride_h_o,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Get program ids
    bh = tl.program_id(0)    # combined batch*H
    i_block = tl.program_id(1)
    j_block = tl.program_id(2)

    # Decompose batch and hidden index
    batch_idx = bh // H
    hidden_idx = bh % H

    # Compute linear offsets for batch and hidden base
    # Base pointers for this batch and hidden
    left_base = left_ptr + batch_idx * stride_b_l + hidden_idx * stride_h_l
    right_base = right_ptr + batch_idx * stride_b_r + hidden_idx * stride_h_r
    out_base = out_ptr + batch_idx * stride_b_o + hidden_idx * stride_h_o

    # Offsets for i and j
    i_range = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j_range = j_block * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for boundary conditions
    mask_i = i_range < N
    mask_j = j_range < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, N, BLOCK_K):
        k_range = k + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # Load left block: shape [BLOCK_M, BLOCK_K]
        # left index: [i, k] with hidden fixed
        # compute pointer offsets
        # left_ptrs = left_base + i_range[:, None] * stride_i_l + k_range[None, :] * stride_k_l
        left_ptrs = left_base + i_range[:, None] * stride_i_l + k_range[None, :] * stride_k_l
        a = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right block (transposed) shape [BLOCK_K, BLOCK_N]
        # right index: [k, j]
        right_ptrs = right_base + k_range[:, None] * stride_i_r + j_range[None, :] * stride_k_r
        b = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Compute dot product (matrix multiplication)
        acc += tl.dot(a, b)

    # Store result to out: shape [BLOCK_M, BLOCK_N]
    out_ptrs = out_base + i_range[:, None] * stride_i_o + j_range[None, :] * stride_j_o
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul forward (outgoing) implementation with a Triton kernel for the
core N‑by‑N contraction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

The kernel also fuses the element‑wise mask and gate multiplications,
so that only a single GPU kernel handles the most expensive O(N³) work.
All other operations (LayerNorms, Linear projections, final LayerNorm
and output projection) are performed with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched matmul with fused mask and gate
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    # tensors
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr, out_ptr,
    # dimensions
    B, N, H,
    # strides for left
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # strides for left_gate
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    # strides for right
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    # strides for right_gate
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    # strides for mask (shape [B,N,N,1])
    stride_m_b, stride_m_i, stride_m_j, stride_m_h,
    # strides for out
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out = (left*left_gate*mask) @ (right*right_gate*mask)^T"""
    pid = tl.program_id(0)       # combined batch * hidden dimension
    pid_i = tl.program_id(1)     # block index for i (rows)
    pid_j = tl.program_id(2)     # block index for j (cols)

    # decompose combined batch*hidden id
    batch_idx = pid // H
    hid_idx   = pid % H

    # base pointers for this batch and hidden channel
    left_base      = left_ptr      + batch_idx * stride_l_b   + hid_idx * stride_l_h
    left_gate_base = left_gate_ptr + batch_idx * stride_lg_b  + hid_idx * stride_lg_h
    right_base     = right_ptr     + batch_idx * stride_r_b   + hid_idx * stride_r_h
    right_gate_base= right_gate_ptr+ batch_idx * stride_rg_b  + hid_idx * stride_rg_h
    out_base       = out_ptr       + batch_idx * stride_o_b   + hid_idx * stride_o_h
    mask_base      = mask_ptr      + batch_idx * stride_m_b

    # tile offsets
    i_offsets = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offsets = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # boundary masks
    mask_i = i_offsets < N
    mask_j = j_offsets < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, N, BLOCK_K):
        k_offsets = k + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < N

        # ---------- left side (i,k) ----------
        left_ptrs = left_base + i_offsets[:, None] * stride_l_i + k_offsets[None, :] * stride_l_k
        left_gate_ptrs = left_gate_base + i_offsets[:, None] * stride_lg_i + k_offsets[None, :] * stride_lg_k
        mask_l_ptrs = mask_base + i_offsets[:, None] * stride_m_i + k_offsets[None, :] * stride_m_j

        a = tl.load(left_ptrs,      mask=mask_i[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)
        ag = tl.load(left_gate_ptrs,mask=mask_i[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)
        am = tl.load(mask_l_ptrs,   mask=mask_i[:, None] & mask_k[None, :], other=1.0, dtype=tl.float16)

        a = a * ag * am                          # (BLOCK_M, BLOCK_K)

        # ---------- right side (j,k) but transposed ----------
        # we need right[j,k] → we load as (k, j) to get (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + k_offsets[:, None] * stride_r_i + j_offsets[None, :] * stride_r_k
        right_gate_ptrs = right_gate_base + k_offsets[:, None] * stride_rg_i + j_offsets[None, :] * stride_rg_k
        mask_r_ptrs = mask_base + j_offsets[:, None] * stride_m_i + k_offsets[None, :] * stride_m_j

        b = tl.load(right_ptrs,      mask=mask_k[:, None] & mask_j[None, :], other=0.0, dtype=tl.float16)
        bg = tl.load(right_gate_ptrs,mask=mask_k[:, None] & mask_j[None, :], other=0.0, dtype=tl.float16)
        bm = tl.load(mask_r_ptrs,   mask=mask_k[:, None] & mask_j[None, :], other=1.0, dtype=tl.float16)

        b = b * bg * bm                          # (BLOCK_K, BLOCK_N)

        # ---- accumulate ----
        acc += tl.dot(a, b)                       # (BLOCK_M, BLOCK_N) in fp32

    # write out
    out_ptrs = out_base + i_offsets[:, None] * stride_o_i + j_offsets[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper to launch the above kernel
# ----------------------------------------------------------------------
def _tri_mul_contract(left, left_gate, right, right_gate, mask):
    """
    left, right, left_gate, right_gate: torch.float16 tensors of shape
        [B, N, N, H]
    mask: torch.float16 tensor of shape [B, N, N, 1] (all‑ones when no mask)
    Returns:
        out: torch.float16 tensor of shape [B, N, N, H]
    """
    B, N, _, H = left.shape
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32

    # strides
    stride_l = left.stride()
    stride_lg = left_gate.stride()
    stride_r = right.stride()
    stride_rg = right_gate.stride()
    stride_m = mask.stride()
    stride_o = left.new_empty(0).stride()   # placeholder; we'll allocate out separately

    # allocate output
    out = torch.empty_like(left)

    # recompute out strides after allocation
    stride_o = out.stride()

    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    grid = (B * H, grid_i, grid_j)

    _tri_mul_kernel[grid](
        left,
        left_gate,
        right,
        right_gate,
        mask,
        out,
        B, N, H,
        # left strides
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        # left_gate strides
        stride_lg[0], stride_lg[1], stride_lg[2], stride_lg[3],
        # right strides
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        # right_gate strides
        stride_rg[0], stride_rg[1], stride_rg[2], stride_rg[3],
        # mask strides (shape [B,N,N,1])
        stride_m[0], stride_m[1], stride_m[2], stride_m[3],
        # out strides
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------- LayerNorm on input -----------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ----------------- Linear projections -----------------
    left = F.linear(x, weights["left_proj.weight"])            # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ----------------- Gates (sigmoid) -----------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ----------------- Prepare mask -----------------
    if nomask or mask is None:
        # all‑ones mask
        mask_fp16 = torch.ones(
            (bs, seq_len, seq_len, 1), dtype=torch.float16, device=device
        )
    else:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)

    # Cast to fp16 for the heavy kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    left_gate_fp16 = left_gate.to(torch.float16).contiguous()
    right_gate_fp16 = right_gate.to(torch.float16).contiguous()

    # ----------------- Contraction (core O(N³) work) -----------------
    contracted_fp16 = _tri_mul_contract(
        left_fp16, left_gate_fp16,
        right_fp16, right_gate_fp16,
        mask_fp16,
    )  # shape [B, N, N, hidden_dim] in fp16

    # Convert back to fp32 for the remaining layers
    out = contracted_fp16.to(torch.float32)

    # ----------------- Output LayerNorm + gating -----------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate  # element‑wise

    # ----------------- Final linear projection -----------------
    output = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return output
