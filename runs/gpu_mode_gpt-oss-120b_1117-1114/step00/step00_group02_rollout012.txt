    pid_z = tl.program_id(0)   # batch_h index (0..batch_h-1)
    pid_m = tl.program_id(1)   # tile row
    pid_n = tl.program_id(2)   # tile col
    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers for A tile
    a_ptrs = left_ptr + pid_z * stride_lbh + offs_m[:, None] * stride_li + tl.arange(0, BLOCK_K)[None, :] * stride_lk
    # pointers for B_T tile (transposed)
    b_ptrs = right_ptr + pid_z * stride_rbh + tl.arange(0, BLOCK_K)[:, None] * stride_rk + offs_n[None, :] * stride_rj
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k*BLOCK_K+tl.arange(0,BLOCK_K) < N), other=0.0)

        acc += tl.dot(a, b)
        # advance pointers
        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk

    out_ptrs = out_ptr + pid_z * stride_obh + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    B,N,N2,dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    assert N==N2

    # Extract weight tensors
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)
    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    to_out_weight = weights["to_out.weight"].to(device)

    eps = 1e-5

    # LayerNorm input
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Optional mask
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)  # broadcast
        left = left * mask
        right = right * mask
    # else no mask (do nothing)

    # Compute gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    right = right * right_gate

    # Triangular multiplication via Triton
    # Permute to (B, hidden_dim, N, N) and flatten batch-h
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    B_, H_, _, _ = left_perm.shape
    # flatten batch and hidden_dim
    left_flat = left_perm.view(-1, N, N)
    right_flat = right_perm.view(-1, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (left_flat.shape[0], triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape back to (B, H, N, N)
    out = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()  # shape B,N,N,H

    # Apply to_out_norm and gating
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)

    return out
```

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

The heavy O(N²·H) pairwise multiplication
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is performed as a batched GEMM (left @ rightᵀ) for each hidden channel h.
All other operations (LayerNorm, linear projections, gating) are done in PyTorch.
The kernel works on a flattened batch‑hidden dimension (B·H) and
uses tiling (BLOCK_M×BLOCK_N) with a dynamic K‑loop.
Result shape: [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_gemm_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    N,                                        # sequence length (scalar)
    stride_A_batch, stride_A_i, stride_A_k,   # A strides
    stride_B_batch, stride_B_j, stride_B_k,   # B strides (we access Bᵀ)
    stride_C_batch, stride_C_i, stride_C_j,   # C strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_z = tl.program_id(0)   # batch*hidden index
    pid_m = tl.program_id(1)   # tile row
    pid_n = tl.program_id(2)   # tile col

    # ------------------------------
    # offsets for the current tile
    # ------------------------------
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # rows   (i)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # cols   (j)

    mask_i = i < N
    mask_j = j < N

    # base pointers for this (batch*hidden) slice
    a_base = A_ptr + pid_z * stride_A_batch
    b_base = B_ptr + pid_z * stride_B_batch
    c_base = C_ptr + pid_z * stride_C_batch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)   # absolute k indices
        mask_k = k < N

        # A tile: rows i, cols k
        a_ptr = a_base + i[:, None] * stride_A_i + k[None, :] * stride_A_k
        a = tl.load(
            a_ptr,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Bᵀ tile: rows k, cols j   (original B is [j, k])
        b_ptr = b_base + k[:, None] * stride_B_k + j[None, :] * stride_B_j
        b = tl.load(
            b_ptr,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        acc += tl.dot(a, b)

    # write the result
    c_ptr = c_base + i[:, None] * stride_C_i + j[None, :] * stride_C_j
    tl.store(c_ptr, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim]
            - mask: torch.Tensor or None, shape [B, N, N]
            - weights: dict of torch.Tensor containing all model weights
            - config: dict with keys "dim", "hidden_dim" (and optional metadata)

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load weights (ensure they live on the same device & dtype)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias = weights["norm.bias"].to(device, dtype)

    left_proj_weight = weights["left_proj.weight"].to(device, dtype)
    right_proj_weight = weights["right_proj.weight"].to(device, dtype)

    left_gate_weight = weights["left_gate.weight"].to(device, dtype)
    right_gate_weight = weights["right_gate.weight"].to(device, dtype)
    out_gate_weight = weights["out_gate.weight"].to(device, dtype)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device, dtype)

    to_out_weight = weights["to_out.weight"].to(device, dtype)

    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # same shape

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core triangular multiplication using Triton
    # ------------------------------------------------------------------
    # transpose to [B, hidden, N, N] and flatten batch*hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(-1, N, N)   # [B*H, N, N]
    right_flat = right_perm.view(-1, N, N)

    out_flat = torch.empty_like(left_flat)  # will hold [B*H, N, N]

    # Triton launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        left_flat.shape[0],               # batch*hidden
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
    )

    _trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output normalisation, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # back to original dim

    return out
