import torch
import triton
import triton.language as tl
from torch.nn import functional as F

@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k,
    stride_right_b, stride_right_i, stride_right_k,
    stride_out_b, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # tile for i dimension
    pid_n = tl.program_id(1)  # tile for j dimension
    pid_hb = tl.program_id(2) # combined batch*hidden

    # start offsets
    i_offset = pid_m * BLOCK_M
    j_offset = pid_n * BLOCK_N

    # compute batch and hidden channel from pid_hb
    b = pid_hb // H
    h = pid_hb % H

    # compute base addresses for this batch+hidden
    left_base = left_ptr + b * stride_left_b + h * 1   # stride for hidden dimension is 1
    right_base = right_ptr + b * stride_right_b + h * 1
    out_base = out_ptr + b * stride_out_b + h * 1

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # i indices
        i_idx = i_offset + tl.arange(0, BLOCK_M)
        mask_i = i_idx < N

        # j indices
        j_idx = j_offset + tl.arange(0, BLOCK_N)
        mask_j = j_idx < N

        # load left tile (i,k)
        left_ptrs = left_base + i_idx[:, None] * stride_left_i + cur_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right tile (k,j) i.e. transposed
        right_ptrs = right_base + cur_k[:, None] * stride_right_k + j_idx[None, :] * stride_right_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(left, right)

    # store result
    i_idx = i_offset + tl.arange(0, BLOCK_M)
    j_idx = j_offset + tl.arange(0, BLOCK_N)
    mask_i = i_idx < N
    mask_j = j_idx < N

    out_ptrs = out_base + i_idx[:, None] * stride_out_i + j_idx[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # layernorm x
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # compute mean and var over last dim
    x = input_tensor
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_w + norm_b

    # Projections
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # out_gate
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # Apply mask if needed
    if mask_tensor is not None and not config.get("nomask", False):

        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Now compute triangular multiplicative update using Triton kernel

    # allocate output
    out = torch.empty_like(left)  # same shape, dtype float32

    # Get strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    B, N, _, H = left.shape  # shape [B,N,N,H] after projection

    # define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32  # chosen

    # compute grid

        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H
    )

        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,

        H=H,

        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,

    to_out_norm_w = weights["to_out_norm.weight"]

"""
TriMul (outgoing) forward implementation with a Triton kernel.

The kernel computes the core O(N³·H) contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
for every batch `b` and hidden channel `h`.  This is a batched matrix
multiplication of `left` with the transpose of `right`.  All other
operations (LayerNorms, linear projections, gates, masking) are performed
with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k,
    stride_right_b, stride_right_i, stride_right_k,
    stride_out_b, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out = left @ rightᵀ for every (batch, hidden) pair."""
    pid_m = tl.program_id(0)          # tile row (i) index
    pid_n = tl.program_id(1)          # tile col (j) index
    pid_hb = tl.program_id(2)         # combined batch*hidden index

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # Decode batch and hidden channel
    b = pid_hb // H
    h = pid_hb % H

    # Base pointers for this (b, h)
    left_base = left_ptr + b * stride_left_b + h
    right_base = right_ptr + b * stride_right_b + h
    out_base = out_ptr + b * stride_out_b + h

    # Accumulator for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension (the shared sequence axis)
    for k_start in range(0, N, BLOCK_K):
        k = k_start + tl.arange(0, BLOCK_K)
        mask_k = k < N

        i = i0 + tl.arange(0, BLOCK_M)
        j = j0 + tl.arange(0, BLOCK_N)
        mask_i = i < N
        mask_j = j < N

        # left[i, k]  (shape M x K)
        left_ptrs = left_base + i[:, None] * stride_left_i + k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # rightᵀ[k, j]  (shape K x N) – we load the transposed view
        right_ptrs = right_base + k[:, None] * stride_right_k + j[None, :] * stride_right_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # Accumulate the dot product over K
        acc += tl.dot(left, right)

    # Store the result tile
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)
    mask_i = i < N
    mask_j = j < N
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Args:
        data: Tuple (input, mask, weights, config)
            input  – torch.Tensor [B, N, N, dim]
            mask   – torch.Tensor [B, N, N] (may be ignored if config["nomask"]==True)
            weights – dict of torch.Tensors with model parameters
            config – dict with keys "dim", "hidden_dim", and optional "nomask"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ---------- LayerNorm on input ----------
    x = input_tensor
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # ---------- Linear projections ----------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # ---------- Optional masking ----------
    if mask is not None and not config.get("nomask", False):
        mask = mask.to(dtype).unsqueeze(-1)            # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ---------- Apply gating ----------
    left = left * left_gate
    right = right * right_gate

    # ---------- Triton GEMM for the triangle multiplicative update ----------
    # Ensure contiguous layout
    left = left.contiguous()
    right = right.contiguous()
    B, N, _, H = left.shape  # H == hidden_dim

    out = torch.empty_like(left)  # shape [B, N, N, H]

    # Strides (in elements) for the tensors
    s_left_b, s_left_i, s_left_k, s_left_h = left.stride()
    s_right_b, s_right_i, s_right_k, s_right_h = right.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = out.stride()

    # Tile sizes – chosen for good occupancy on Hopper (B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    trimul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=s_left_b,
        stride_left_i=s_left_i,
        stride_left_k=s_left_k,
        stride_right_b=s_right_b,
        stride_right_i=s_right_i,
        stride_right_k=s_right_k,
        stride_out_b=s_out_b,
        stride_out_i=s_out_i,
        stride_out_j=s_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- LayerNorm on the result ----------
    mean_out = out.mean(dim=-1, keepdim=True)
    var_out = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_out) / torch.sqrt(var_out + eps)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # ---------- Apply output gate ----------
    out_norm = out_norm * out_gate

    # ---------- Final linear projection ----------
    output = F.linear(out_norm, weights["to_out.weight"])

    return output
