    left_ptr, right_ptr, out_ptr,
    batch_stride, i_stride, k_stride, d_stride,
    batch_stride_r, j_stride_r, k_stride_r, d_stride_r,
    batch_stride_out, i_stride_out, j_stride_out, d_stride_out,
    N, hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    pid_n = tl.program_id(1) # j tile
    pid_z = tl.program_id(2) # batch*hidden_dim channel

    d = pid_z % hidden_dim

    j_offset = pid_n * BLOCK_N

    offs_j = j_offset + tl.arange(0, BLOCK_N)

    left = left * left_gate
    right = right * right_gate

    right_fp16 = right.to(torch.float16)

    stride_b_r = stride_b_l

    stride_k_r = stride_k_l
    stride_d_r = stride_d

    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

        stride_b_r, stride_j_r, stride_k_r, stride_d_r,
        stride_b_o, stride_i_o, stride_k_o, stride_d_o,
        N, hidden_dim,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Convert out back to float32
    out = out_fp16.to(torch.float32)

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]

    out = out * out_gate

    # final linear
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul (outgoing) forward implementation with a custom Triton kernel.

The heavy part of TriMul is the O(N³) reduction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We compute the reduction with a Triton kernel that performs a batched
matrix multiplication for each (batch, hidden) pair:
    left[b,:,:,d]  (N×N)  @  right[b,:,:,d].T  (N×N) → out[b,:,:,d] (N×N)

All other operations (LayerNorm, linear projections, gates) are carried out
with PyTorch for simplicity and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # left strides: (batch, i, k, d)
    stride_b_l, stride_i_l, stride_k_l, stride_d_l,
    # right strides: (batch, j, k, d)
    stride_b_r, stride_j_r, stride_k_r, stride_d_r,
    # out strides: (batch, i, j, d)
    stride_b_o, stride_i_o, stride_j_o, stride_d_o,
    # problem size
    N, hidden_dim,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)          # tile index over i
    pid_j = tl.program_id(1)          # tile index over j
    pid_z = tl.program_id(2)          # encodes batch and hidden channel

    batch = pid_z // hidden_dim
    d = pid_z % hidden_dim

    # tile start coordinates
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # offsets inside the tile
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k_start in range(0, N, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile:  (i, k)   shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr \
            + batch * stride_b_l \
            + offs_i[:, None] * stride_i_l \
            + offs_k[None, :] * stride_k_l \
            + d * stride_d_l
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile: we need (k, j) → transpose of the stored (j, k)
        right_ptrs = right_ptr \
            + batch * stride_b_r \
            + offs_k[:, None] * stride_k_r \
            + offs_j[None, :] * stride_j_r \
            + d * stride_d_r
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # dot‑product over k
        acc += tl.dot(left, right)

    # write result back (store as FP16 to save bandwidth)
    out_ptrs = out_ptr \
        + batch * stride_b_o \
        + offs_i[:, None] * stride_i_o \
        + offs_j[None, :] * stride_j_o \
        + d * stride_d_o
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (binary, may be all‑ones)
            - weights: dict of torch.Tensors containing all model parameters
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # -----------------------------------------------------------------
    # 1) LayerNorm over the feature dimension `dim`
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])    # [B, N, N, hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Apply optional mask (broadcast on hidden_dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4) Gated activations
    # -----------------------------------------------------------------
    left_gate = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Triton reduction kernel (matrix multiplication per channel)
    # -----------------------------------------------------------------
    # Cast to FP16 for the kernel (accumulation is FP32 inside)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16)  # [B, N, N, hidden_dim]

    # contiguous tensors → easy stride extraction
    stride_b, stride_i, stride_j, stride_d = left_fp16.stride()  # (B,N,N,hidden)

    # tile sizes – chosen to fit registers on the B200
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # i dimension
        triton.cdiv(N, BLOCK_N),          # j dimension
        B * hidden_dim,                   # batch * hidden channel
    )

    _trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        # left strides
        stride_b,
        stride_i,
        stride_j,
        stride_d,
        # right strides (batch, j, k, d)
        stride_b,
        stride_i,   # j uses the same stride as i in the contiguous layout
        stride_j,   # k stride
        stride_d,
        # output strides (batch, i, j, d)
        stride_b,
        stride_i,
        stride_j,
        stride_d,
        N,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to FP32 for the remaining operations
    out = out_fp16.to(torch.float32)

    # -----------------------------------------------------------------
    # 6) Output layer‑norm, final gating and projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
