@triton.jit
def batched_gemm_kernel(A_ptr, B_ptr, C_ptr,
                        M, N, K,
                        stride_am, stride_ak, stride_ab,
                        stride_bk, stride_bn, stride_bb,
                        stride_cm, stride_cn, stride_cb,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets for rows and cols
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for valid rows and cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute pointer base for this batch
    a_ptrs = A_ptr + batch_id * stride_ab
    b_ptrs = B_ptr + batch_id * stride_bb
    c_ptrs = C_ptr + batch_id * stride_cb

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A and B tiles
        a = tl.load(a_ptrs + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak), mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn), mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # matmul accumulate
        acc += tl.dot(a, b)

    # Write back result
    tl.store(c_ptrs + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn), acc, mask=mask_m[:, None] & mask_n[None, :])

def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    a_ptrs = A_ptr + batch * stride_ab
    b_ptrs = B_ptr + batch * stride_bb
    c_ptrs = C_ptr + batch * stride_cb

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul (outgoing) operator.

    The kernel fuses the cubic N^3 reduction:
        out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    where left/right are linear projections of the input, masked and gated.
    All other operations (layer‑norms, gating, final projection) are performed
    in PyTorch. The reduction is executed as a batched matrix multiplication
    using a Triton kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----- LayerNorm on the input -----
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ----- Linear projections -----
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ----- Optional mask (broadcast over hidden) -----
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(dtype)  # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ----- Gates (sigmoid) -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # keep FP32 for later

    # ----- Apply gates -----
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # ----- Prepare tensors for batched GEMM -----
    # left: [B,N,N,hidden] -> [B,hidden,N,N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right transposed on the two N axes: right_T[b, hidden, k, j]
    right_T = right.permute(0, 3, 2, 1).contiguous()

"""TriMul (outgoing) forward pass with Triton‑accelerated cubic reduction.

The heavy N³ reduction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is rewritten as a batched matrix multiplication
    out_d = left_d @ right_dᵀ   (for each hidden channel d).
A Triton kernel computes all batch × hidden products in FP16 (inputs) → FP32
(accumulator) while the surrounding LayerNorms, gates and final projection stay
in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)          # which (B * hidden) matrix
    pid_m = tl.program_id(1)             # tile row
    pid_n = tl.program_id(2)             # tile col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_ptrs = A_ptr + batch_id * stride_ab
    b_ptrs = B_ptr + batch_id * stride_bb
    c_ptrs = C_ptr + batch_id * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a = tl.load(
            a_ptrs + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        b = tl.load(
            b_ptrs + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        acc += tl.dot(a, b)

    tl.store(
        c_ptrs + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        acc,
        mask=mask_m[:, None] & mask_n[None, :]
    )


def custom_kernel(data):
    """Forward pass of the TriMul “outgoing” module using Triton."""
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])    # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(dtype)       # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # stays FP32

    # ------------------------------------------------------------------
    # 5) Apply gates (cast to FP16 for the cubic reduction)
    # ------------------------------------------------------------------
    left = (left * left_gate).to(torch.float16)   # [B,N,N,hidden] FP16
    right = (right * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # 6) Rearrange tensors to (B*hidden, N, N) for batched GEMM
    # ------------------------------------------------------------------
    # left:  [B,N,N,hidden] -> [B,hidden,N,N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right_T = rightᵀ on the two N axes: [B,hidden,N,N]
    right_T = right.permute(0, 3, 2, 1).contiguous()

    B, N, _, H = left.shape
    left_mat = left_perm.view(-1, N, N)          # (B*H, N, N) FP16
    right_mat = right_T.view(-1, N, N)           # (B*H, N, N) FP16

    # Output accumulator in FP32
    out_mat = torch.empty_like(left_mat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 7) Strides (in elements) needed by the kernel
    # ------------------------------------------------------------------
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)
    stride_ab = left_mat.stride(0)

    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)
    stride_bb = right_mat.stride(0)

    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)
    stride_cb = out_mat.stride(0)

    # ------------------------------------------------------------------
    # 8) Triton launch
    # ------------------------------------------------------------------
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    batch = left_mat.shape[0]                     # B * hidden_dim
    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 9) Restore original layout: [B, N, N, hidden]
    # ------------------------------------------------------------------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)   # FP32

    # ------------------------------------------------------------------
    # 10) Output gating, hidden LayerNorm, final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate                                 # gate (FP32)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    out = F.linear(out, weights["to_out.weight"])        # → [B, N, N, dim]

    return out
