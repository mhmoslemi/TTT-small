def trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,

    stride_bm, stride_bk, stride_bn,
    stride_cm, stride_cn, stride_co,  # but C shape same as A, strides: cm row, cn col

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    # (maybe confirm dtype)
    # layernorm on input
    eps = 1e-5
    # Use stored weight and bias from weights dict
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Normalize
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    # if not, do manual

    # linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # if mask present and not nomask
    if not config.get("nomask", False):
        # mask shape [B,N,N]; unsqueeze last dim
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # apply gates (cast to fp16)
    left = left * left_gate
    right = right * right_gate

    # Cast left, right to float16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Permute and reshape for kernel
    B, N, _, H = left_fp16.shape
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]
    right_perm = right_fp16.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    M = B * H
    left_mat = left_perm.view(M, N, N)
    right_mat = right_perm.view(M, N, N)

    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        M,
    )
    trimul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M,
        N,
        N,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Convert out_mat to float32
    out_fp32 = out_mat.to(torch.float32)

    # Reshape back
    out_perm = out_fp32.view(B, H, N, N)  # shape [B, H, N, N]
    out = out_perm.permute(0, 2, 3, 1)  # [B, N, N, H]

    # Apply to_out_norm
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # Multiply with out_gate (out_gate is float32)
    out_norm = out_norm * out_gate

    # Final linear projection
    final_output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])
    return final_output

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_batch_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)  # batch index (M)
    pid_m = tl.program_id(0)      # block row
    pid_n = tl.program_id(1)      # block column

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for kb in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A block [BLOCK_M, BLOCK_K]
        a_ptrs = A_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load B block [BLOCK_K, BLOCK_N]
        b_ptrs = B_ptr + pid_batch * stride_b_batch + offs_k[:, None] * stride_b_row + offs_n[None, :] * stride_b_col
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = C_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    It performs layer‑norm, linear projections, optional masking, gating, a batched
    matrix multiplication that implements the N³ einsum, a second layer‑norm and a final
    linear projection. The expensive inner product is executed with a custom Triton kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----- LayerNorm on input -----
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ----- Linear projections -----
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ----- Optional mask -----
    if not config.get("nomask", False):
        # mask shape: [B,N,N] -> broadcast to hidden_dim
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ----- Gates (sigmoid) -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # [B,N,N,hidden_dim]

    # ----- Apply gates -----
    left = left * left_gate
    right = right * right_gate

    # ----- Cast to fp16 for the heavy N³ product -----
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Free up FP32 intermediates to keep memory footprint low
    del left, right, left_gate, right_gate, x
    torch.cuda.empty_cache()

    # ----- Rearrange tensors for batched matmul (i,k) @ (k,j) -----
    # left: (B,N,N,hidden) -> (B,hidden,N,N) with rows=i, cols=k
    # right: (B,N,N,hidden) -> (B,hidden,N,N) with rows=k, cols=j
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_fp16.permute(0, 3, 2, 1).contiguous() # [B, H, N, N]

    del left_fp16, right_fp16
    torch.cuda.empty_cache()

    B, H, N, _ = left_perm.shape
    batch_dim = B * H

    # flatten batch and hidden dims
    A = left_perm.view(batch_dim, N, N)
    B_mat = right_perm.view(batch_dim, N, N)

    # output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # ----- Triton kernel launch -----
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_dim,
    )

    trimul_batch_matmul_kernel[grid](
        A,
        B_mat,
        C,
        batch_dim,
        N,
        N,
        A.stride(0), A.stride(1), A.stride(2),
        B_mat.stride(0), B_mat.stride(1), B_mat.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- Restore original tensor layout -----
    # C shape [batch_dim, N, N] (fp16) -> fp32
    out_fp16 = C.view(B, H, N, N)            # [B, H, N, N]
    out = out_fp16.permute(0, 2, 3, 1).to(torch.float32)  # [B, N, N, H]

    # ----- Second LayerNorm over hidden dimension -----
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ----- Output gating and final projection -----
    out_norm = out_norm * out_gate
    final_output = F.linear(out_norm, weights["to_out.weight"])

    return final_output

# Triton kernel: batched MatMul for the N³ einsum in TriMul
# Computes C = A @ B where A, B, C have shape (M, N, N)
# A : (i, k)   B : (k, j)   -> C : (i, j)  for every batch m
# ----------------------------------------------------------------------
@triton.jit
def trimul_batch_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_m = tl.program_id(0)              # tile row
    pid_n = tl.program_id(1)              # tile col

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

                 + offs_m[:, None] * stride_a_row \
                 + offs_k[None, :] * stride_a_col
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B block [BLOCK_K, BLOCK_N]
        b_ptrs = B_ptr + pid_batch * stride_b_batch \
                 + offs_k[:, None] * stride_b_row \
                 + offs_n[None, :] * stride_b_col
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Store result as float16
    c_ptrs = C_ptr + pid_batch * stride_c_batch \
             + offs_m[:, None] * stride_c_row \
             + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    Steps:
      1. Layer‑norm the input.
      2. Linear projections (left/right) and optional mask.
      3. Compute three sigmoid gates.
      4. Apply gates to the projected tensors.
      5. Cast to float16 and rearrange for a batched mat‑mul that implements the
         einsum '... i k d, ... j k d -> ... i j d'.
      6. Run the custom Triton kernel for the heavy N³ product.
      7. Restore the original (B, N, N, hidden) layout.
      8. Second layer‑norm over the hidden dimension, apply the output gate,
         and a final linear projection back to the original feature dimension.
    Returns a float32 tensor of shape [bs, seq_len, seq_len, dim].
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask
    # --------------------------------------------------------------
    if not config.get("nomask", False):
        mask_f = mask.to(x.dtype).unsqueeze(-1)       # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4. Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 5. Apply gates and cast to float16
    # --------------------------------------------------------------
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # Release unused tensors early
    del left_gate, right_gate, x
    torch.cuda.empty_cache()

    # --------------------------------------------------------------
    # 6. Rearrange for batched matmul: (i,k) @ (k,j)
    # --------------------------------------------------------------
    # left:  (B,N,N,H) -> (B,H,N,N) rows=i, cols=k
    # right: (B,N,N,H) -> (B,H,N,N) rows=k, cols=j (transpose k<->j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N]

    del left, right
    torch.cuda.empty_cache()

    B, H, N, _ = left_perm.shape
    batch_dim = B * H

    A = left_perm.view(batch_dim, N, N)   # [M, N, N]
    B_mat = right_perm.view(batch_dim, N, N)
    C = torch.empty_like(A, dtype=torch.float16)   # output buffer

    # --------------------------------------------------------------
    # 7. Triton kernel launch
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch_dim,                 # batch (M)
    )

    trimul_batch_matmul_kernel[grid](
        A,
        B_mat,
        C,
        batch_dim,
        N,
        N,
        A.stride(0), A.stride(1), A.stride(2),
        B_mat.stride(0), B_mat.stride(1), B_mat.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 8. Restore original layout (B, N, N, hidden_dim)
    # --------------------------------------------------------------
    out_fp16 = C.view(B, H, N, N)                     # [B, H, N, N]
    out = out_fp16.permute(0, 2, 3, 1).to(torch.float32)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 9. Second LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------
    # 10. Output gate and final linear projection back to dim
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate
    final_output = F.linear(out_norm, weights["to_out.weight"])

    return final_output
