def tri_mul_kernel(lhs_ptr, rhs_ptr, out_ptr,
                   batch_size, N,
                   stride_lhs_batch, stride_lhs_m, stride_lhs_k,
                   stride_rhs_batch, stride_rhs_m, stride_rhs_k,
                   stride_out_batch, stride_out_i, stride_out_j,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_b = tl.program_id(2)

    # Compute block offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile [M,K]
        lhs_ptrs = lhs_ptr + pid_b * stride_lhs_batch + offs_i[:, None] * stride_lhs_m + offs_k[None, :] * stride_lhs_k
        lhs = tl.load(lhs_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile [N,K]
        rhs_ptrs = rhs_ptr + pid_b * stride_rhs_batch + offs_j[:, None] * stride_rhs_m + offs_k[None, :] * stride_rhs_k
        rhs = tl.load(rhs_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # Compute dot product and accumulate
        acc += tl.dot(lhs, rhs, trans_b=True)

    # Write result
    out_ptrs = out_ptr + pid_b * stride_out_batch + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward implementation with a Triton kernel for the
core N³ multiplication.

Algorithm:
1. LayerNorm over the last dimension of the input.
2. Linear projections (left/right) and gating (left_gate/right_gate/out_gate).
3. Apply mask and gating, cast projected tensors to half‑precision.
4. Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] using a
   Triton GEMM kernel (batched matmul of left (M×K) with rightᵀ (K×N)).
5. LayerNorm on the hidden dimension, gate with out_gate and final linear
   projection back to the original dimension.

Only the heavy O(N³) contraction is performed in Triton; all other
operations use PyTorch (which is already highly optimized).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(lhs_ptr, rhs_ptr, out_ptr,
                   batch, N,
                   stride_lhs_batch, stride_lhs_m, stride_lhs_k,
                   stride_rhs_batch, stride_rhs_m, stride_rhs_k,
                   stride_out_batch, stride_out_i, stride_out_j,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """
    Batched matmul kernel computing:
        out[b, i, j] = Σ_k lhs[b, i, k] * rhs[b, j, k]
    where lhs and rhs are of shape (batch, N, N) for each hidden channel.
    The computation uses a dot product with trans_b=True to treat rhs as
    transposed on the fly.
    """
    pid_i = tl.program_id(0)   # tile index over rows (i)
    pid_j = tl.program_id(1)   # tile index over cols (j)
    pid_b = tl.program_id(2)   # batch*hidden index

    # Offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a tile of lhs: shape (BLOCK_M, BLOCK_K)
        lhs_ptrs = (lhs_ptr
                    + pid_b * stride_lhs_batch
                    + offs_i[:, None] * stride_lhs_m
                    + offs_k[None, :] * stride_lhs_k)
        lhs = tl.load(lhs_ptrs,
                      mask=mask_i[:, None] & mask_k[None, :],
                      other=0.0)

        # Load a tile of rhs: shape (BLOCK_N, BLOCK_K)
        rhs_ptrs = (rhs_ptr
                    + pid_b * stride_rhs_batch
                    + offs_j[:, None] * stride_rhs_m
                    + offs_k[None, :] * stride_rhs_k)
        rhs = tl.load(rhs_ptrs,
                      mask=mask_j[:, None] & mask_k[None, :],
                      other=0.0)

        # Accumulate dot product. rhs is logically transposed (K,N) via trans_b=True.
        acc += tl.dot(lhs, rhs, trans_b=True)

    # Write the result tile back as float16
    out_ptrs = (out_ptr
                + pid_b * stride_out_batch
                + offs_i[:, None] * stride_out_i
                + offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask:          torch.Tensor [B, N, N] (may be all‑ones)
            - weights:       dict of weight tensors
            - config:        dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    # ------------------------------------------------------------------
    # 1. Extract configuration and weights
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5  # default LayerNorm epsilon

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 2. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # 3. Linear projections (float32) and gating
    # ------------------------------------------------------------------
    # Projections
    left = F.linear(x, left_proj_weight)          # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)       # [B,N,N,hidden]

    # Apply mask if provided (mask is broadcastable)
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(dtype)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # Gates (float32)
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Cast to half‑precision and prepare for Triton kernel
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to [B, hidden, N, N] and flatten the batch*hidden dimension
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, hidden, N, N]

    lhs = left.view(B * hidden_dim, N, N)
    rhs = right.view(B * hidden_dim, N, N)

    # Output buffer
    out = torch.empty_like(lhs, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 5. Triton kernel launch (batched GEMM for the reduction over k)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    batch_hidden = B * hidden_dim
    grid = (triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            batch_hidden)

    # Strides in element units
    stride_lhs_batch, stride_lhs_m, stride_lhs_k = lhs.stride()
    stride_rhs_batch, stride_rhs_m, stride_rhs_k = rhs.stride()
    stride_out_batch, stride_out_i, stride_out_j = out.stride()

    tri_mul_kernel[grid](
        lhs, rhs, out,
        batch_hidden, N,
        stride_lhs_batch, stride_lhs_m, stride_lhs_k,
        stride_rhs_batch, stride_rhs_m, stride_rhs_k,
        stride_out_batch, stride_out_i, stride_out_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # ------------------------------------------------------------------
    # 6. Restore original layout ([B,N,N,hidden]) and continue in FP32
    # ------------------------------------------------------------------
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]
    out = out.to(dtype)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, gating, and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    out = out * out_gate  # element‑wise gating (float32)

    # Final linear projection back to dim
    out = F.linear(out, to_out_weight)  # [B,N,N,dim]

    return out
