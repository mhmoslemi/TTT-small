"""
TriMul (outgoing) implementation with a Triton kernel.

The kernel fuses the mask multiplication and the two per‑element gates
(left and right) into a single pass:
    left   = (x_norm @ W_left)   * mask * sigmoid(x_norm @ W_left_gate)
    right  = (x_norm @ W_right)  * mask * sigmoid(x_norm @ W_right_gate)

All heavy matrix–matrix products are performed with torch.bmm on half‑precision
tensors, taking advantage of Tensor‑cores.  The final LayerNorms and the
output Linear are done in FP32 to match the reference implementation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mask_gate_fused_kernel(
    # input pointers
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    # output pointers
    out_left_ptr, out_right_ptr,
    # sizes
    B, N, D,
    # strides for left projection
    stride_lproj_b, stride_lproj_i, stride_lproj_j, stride_lproj_d,
    # strides for left gate
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_d,
    # strides for right projection
    stride_rproj_b, stride_rproj_i, stride_rproj_j, stride_rproj_d,
    # strides for right gate
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_d,
    # strides for mask (no d dimension)
    stride_mask_b, stride_mask_i, stride_mask_j,
    # strides for output tensors (same as left/right layouts)
    stride_out_left_b, stride_out_left_i, stride_out_left_j, stride_out_left_d,
    stride_out_right_b, stride_out_right_i, stride_out_right_j, stride_out_right_d,
    # compile‑time constant
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)            # global element IDs
    total = B * N * N * D
    mask = offs < total

    # ---- decode linear index into (b,i,j,d) ----
    d = offs % D
    tmp = offs // D
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ---- compute pointer offsets ----
    # left projection
    off_lproj = (
        b * stride_lproj_b
        + i * stride_lproj_i
        + j * stride_lproj_j
        + d * stride_lproj_d
    )
    # left gate
    off_lgate = (
        b * stride_lgate_b
        + i * stride_lgate_i
        + j * stride_lgate_j
        + d * stride_lgate_d
    )
    # right projection
    off_rproj = (
        b * stride_rproj_b
        + i * stride_rproj_i
        + j * stride_rproj_j
        + d * stride_rproj_d
    )
    # right gate
    off_rgate = (
        b * stride_rgate_b
        + i * stride_rgate_i
        + j * stride_rgate_j
        + d * stride_rgate_d
    )
    # mask (no d dimension)
    off_mask = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # ---- loads ----
    left_val = tl.load(left_proj_ptr + off_lproj, mask=mask, other=tl.float16(0.0))
    left_gate_val = tl.load(left_gate_ptr + off_lgate, mask=mask, other=tl.float16(0.0))

    right_val = tl.load(right_proj_ptr + off_rproj, mask=mask, other=tl.float16(0.0))
    right_gate_val = tl.load(right_gate_ptr + off_rgate, mask=mask, other=tl.float16(0.0))

    mask_val = tl.load(mask_ptr + off_mask, mask=mask, other=tl.float16(0.0))

    # ---- fused computation ----
    out_left = left_val * mask_val * left_gate_val
    out_right = right_val * mask_val * right_gate_val

    # ---- stores ----
    tl.store(out_left_ptr + off_lproj, out_left, mask=mask)   # same layout as left_proj
    tl.store(out_right_ptr + off_rproj, out_right, mask=mask)   # same layout as right_proj


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for the
    mask & gate fusion and torch.bmm for the heavy N³ contraction.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (float32, 0/1)
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor: output of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) LayerNorm on the input (float32)
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # Cast to half for the compute‑intensive part
    x_norm_h = x_norm.half()

    # -----------------------------------------------------------------
    # 2) Linear projections & gates (half precision)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_proj = F.linear(x_norm_h, left_proj_w)          # [B, N, N, H]
    right_proj = F.linear(x_norm_h, right_proj_w)       # [B, N, N, H]

    left_gate = torch.sigmoid(F.linear(x_norm_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm_h, out_gate_w))

    # -----------------------------------------------------------------
    # 3) Fuse mask and gates via a Triton kernel
    # -----------------------------------------------------------------
    # Convert mask to half and keep its original shape
    mask_h = mask.to(torch.float16)

    left_fused = torch.empty_like(left_proj)
    right_fused = torch.empty_like(right_proj)

    # Strides (in elements) for all tensors
    stride_lproj_b, stride_lproj_i, stride_lproj_j, stride_lproj_d = left_proj.stride()
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_d = left_gate.stride()
    stride_rproj_b, stride_rproj_i, stride_rproj_j, stride_rproj_d = right_proj.stride()
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_d = right_gate.stride()
    stride_mask_b, stride_mask_i, stride_mask_j = mask_h.stride()

    # Output strides (identical to input layouts)
    stride_out_left_b, stride_out_left_i, stride_out_left_j, stride_out_left_d = left_fused.stride()
    stride_out_right_b, stride_out_right_i, stride_out_right_j, stride_out_right_d = right_fused.stride()

    total_elements = B * N * N * hidden_dim
    BLOCK = 256
    grid = (triton.cdiv(total_elements, BLOCK),)

    mask_gate_fused_kernel[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask_h,
        left_fused,
        right_fused,
        B,
        N,
        hidden_dim,
        stride_lproj_b,
        stride_lproj_i,
        stride_lproj_j,
        stride_lproj_d,
        stride_lgate_b,
        stride_lgate_i,
        stride_lgate_j,
        stride_lgate_d,
        stride_rproj_b,
        stride_rproj_i,
        stride_rproj_j,
        stride_rproj_d,
        stride_rgate_b,
        stride_rgate_i,
        stride_rgate_j,
        stride_rgate_d,
        stride_mask_b,
        stride_mask_i,
        stride_mask_j,
        stride_out_left_b,
        stride_out_left_i,
        stride_out_left_j,
        stride_out_left_d,
        stride_out_right_b,
        stride_out_right_i,
        stride_out_right_j,
        stride_out_right_d,
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 4) Contract over the sequence dimension (batched GEMM in half)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # -----------------------------------------------------------------
    # Bring hidden dimension to the batch axis for torch.bmm
    left_perm = left_fused.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_fused.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    # Batched matrix multiply: (i,k) @ (j,k)^T -> (i,j)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N), half

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # -----------------------------------------------------------------
    # 5) Output normalization, gating and final linear projection
    # -----------------------------------------------------------------
    # to_out_norm (float32)
    out_fp32 = out.to(torch.float32)

    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # apply out gate (convert to float32 first)
    out_norm = out_norm * out_gate.to(torch.float32)

    # final linear back to `dim`
    output = F.linear(out_norm, weights["to_out.weight"])  # bias is None

    return output
