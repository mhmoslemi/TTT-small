@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H, B,
    stride_b_l, stride_i_l, stride_k_l,
    stride_b_r, stride_i_r, stride_k_r,
    stride_b_o, stride_i_o, stride_j_o,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_h = tl.program_id(2)   # head index (combined batch and hidden)
    
    # compute batch and hidden indices
    b = pid_h // H
    h = pid_h % H
    
    # base offsets for left/right/out for this head
    # Add head offset (h) to base pointer
    left_base = left_ptr + b * stride_b_l + h * stride_h_l   # stride_h_l = 1
    right_base = right_ptr + b * stride_b_r + h * stride_h_r
    out_base = out_ptr + b * stride_b_o + h * stride_h_o
    
    # compute offsets within tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    
    mask_i = offs_i < N
    mask_j = offs_j < N
    
    # create offsets for K dimension - will be used in loop
    offs_k = tl.arange(0, BLOCK_K)
    
    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute actual number of k elements
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < N
        
        # load left tile (M,K)

TriMul (outgoing) kernel optimized with Triton.

The heavy part of the TriMul forward is:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

We treat each (batch, hidden) pair as an independent matrix multiplication

BLOCK_M×BLOCK_N tiles and iterates over K in BLOCK_K steps, loading left
tiles (M×K) and transposed right tiles (K×N) directly from global memory.
All arithmetic is accumulated in FP32 while the inputs are stored in FP16
to reduce memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched matrix multiply for the outgoing TriMul update.
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, hidden_dim, batch_size,
    stride_b_l, stride_i_l, stride_k_l,
    stride_b_r, stride_i_r, stride_k_r,
    stride_b_o, stride_i_o, stride_j_o,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Computes, for a given head (batch * hidden) index:
        out[i, j] = Σ_k left[i, k] * right[j, k]

    The tensors have shape:
        left  : [B, N, N, H]   (stored as B-i-k-h)
        right : [B, N, N, H]   (stored as B-j-k-h)
        out   : [B, N, N, H]   (stored as B-i-j-h)
    """
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_h = tl.program_id(2)          # combined batch*head index

    # --------------------------------------------------------------- #
    # De‑compose pid_h into batch and hidden indices.
    # --------------------------------------------------------------- #
    b = pid_h // hidden_dim
    h = pid_h % hidden_dim

    # Base pointer for this (b, h) slice.
    left_base  = left_ptr  + b * stride_b_l + h              # stride_h == 1
    right_base = right_ptr + b * stride_b_r + h
    out_base   = out_ptr   + b * stride_b_o + h

    # --------------------------------------------------------------- #
    # Offsets inside the tile.
    # --------------------------------------------------------------- #
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------- #
    # Loop over K dimension.
    # --------------------------------------------------------------- #
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)  = left[i, k]
        left_ptrs = left_base + offs_i[:, None] * stride_i_l + cur_k[None, :] * stride_k_l
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.float16).to(tl.float32)

        # right tile (transposed view):
        # we need right_T[k, j] == right[j, k]
        right_ptrs = right_base + cur_k[:, None] * stride_k_r + offs_j[None, :] * stride_i_r
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0,
                        dtype=tl.float16).to(tl.float32)

        # Accumulate (M×K)·(K×N) -> (M×N)
        acc += tl.dot(left, right)

    # --------------------------------------------------------------- #
    # Write back the result.
    # --------------------------------------------------------------- #
    out_ptrs = out_base + offs_i[:, None] * stride_i_o + offs_j[None, :] * stride_j_o
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# --------------------------------------------------------------------------- #
# Entry point called by the evaluation harness.
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Implements the forward pass of the outgoing TriMul operation.

    Args:
        data : tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask         : torch.Tensor [B, N, N]   (float32 or bool)
            - weights      : dict of torch.Tensor parameters
            - config       : dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    torch.set_grad_enabled(False)

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------- #
    # LayerNorm on the input.
    # --------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim], float32

    # --------------------------------------------------------------- #
    # Linear projections.
    # --------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])    # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask if supplied (mask may be all‑ones when nomask=True)
    if mask is not None:
        mask_f = mask.to(torch.float32).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------- #
    # Gating signals.
    # --------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------- #
    # Cast projections to fp16 for the GEMM kernel.
    # --------------------------------------------------------------- #
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # Output buffer (float32, contiguous)
    out = torch.empty_like(left_h, dtype=torch.float32, device=device)

    # --------------------------------------------------------------- #
    # Strides for the kernel (contiguous layout => stride_h == 1).
    # --------------------------------------------------------------- #
    stride_b_l, stride_i_l, stride_k_l, stride_h_l = left_h.stride()
    stride_b_r, stride_i_r, stride_k_r, stride_h_r = right_h.stride()
    stride_b_o, stride_i_o, stride_j_o, stride_h_o = out.stride()

    # ------------------------------------------------------------------- #
    # Kernel launch configuration.
    # ------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑tiles
        triton.cdiv(N, BLOCK_N),          # j‑tiles
        B * hidden_dim                    # (batch, hidden) heads
    )

    tri_mul_outgoing_kernel[grid](
        left_h, right_h, out,
        N, hidden_dim, B,
        stride_b_l, stride_i_l, stride_k_l,
        stride_b_r, stride_i_r, stride_k_r,
        stride_b_o, stride_i_o, stride_j_o,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # --------------------------------------------------------------- #
    # Post‑processing: layer‑norm, gate, final linear projection.
    # --------------------------------------------------------------- #
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )  # [B, N, N, hidden_dim], float32

    out_gated = out_norm * out_gate  # broadcast over hidden dim

    # Final projection back to the original channel dimension.
    output = F.linear(out_gated, weights["to_out.weight"])  # [B, N, N, dim]

    return output
