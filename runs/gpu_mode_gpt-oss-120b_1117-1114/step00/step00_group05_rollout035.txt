@triton.jit
def trimul_gemm_kernel(L_ptr, R_ptr, O_ptr,
                       B, N, # maybe combined batch dimension?
                       stride_L_batch, stride_L_m, stride_L_k,
                       stride_R_batch, stride_R_n, stride_R_k,
                       stride_O_batch, stride_O_m, stride_O_n,
                       BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
                       BLOCK_K: tl.constexpr, num_batch: tl.constexpr):
    batch_id = tl.program_id(0) # total number of (B*H) batches
    # compute offset base for L,R,O
    # compute tile indices for m,n:
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # compute coordinate offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        L_ptrs = L_ptr + batch_id * stride_L_batch + offs_m[:, None] * stride_L_m + offs_k[None, :] * stride_L_k
        R_ptrs = R_ptr + batch_id * stride_R_batch + offs_n[:, None] * stride_R_n + offs_k[None, :] * stride_R_k
        # Load with mask if out-of-bounds
        L = tl.load(L_ptrs, mask=offs_m < N and offs_k < N, other=0.0)
        R = tl.load(R_ptrs, mask=offs_n < N and offs_k < N, other=0.0)

def trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,

"""
TriMul forward pass using Triton for the heavy N³ matrix multiplication.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension.
2. Project the normalized tensor to a hidden dimension with three linear layers
   (left, right) and compute three sigmoid gates.
3. Apply mask (if any) and element‑wise gating.
4. Compute the core contraction
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is equivalent to a batched matrix multiplication
   `out = left @ rightᵀ` for each hidden channel `d`.  We fuse many
   of these small GEMMs using a custom Triton kernel that loops over the
   `k` dimension in blocks, accumulates in FP32 and writes the result in
   FP16.
5. Layer‑norm the result over the hidden dimension, apply the output gate,
   and linearly project back to the original dimension.

The kernel processes a flattened batch dimension `B * hidden_dim`,
so each Triton program instance computes one `(i, j)` tile of a single
matrix product `L @ Rᵀ`.  Block sizes are chosen to fit well on an NVIDIA
B200 GPU (BLOCK_M = BLOCK_N = 64, BLOCK_K = 32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the N³ contraction (batched matmul L @ Rᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,                     # left  matrix, shape (M, K)
    B_ptr,                     # right matrix transposed, shape (K, N)
    C_ptr,                     # output matrix, shape (M, N)
    M, N, K,                   # dimensions (all int)
    stride_Ab, stride_Am, stride_Ak,   # strides for A: batch, M, K
    stride_Bb, stride_Bk, stride_Bn,   # strides for B: batch, K, N (transposed view)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C: batch, M, N
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)      # which (batch * hidden) we are processing
    pid_m   = tl.program_id(1)      # tile row index
    pid_n   = tl.program_id(2)      # tile column index

    # ------------------------------- offsets -------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = A_ptr + batch_id * stride_Ab
    b_batch = B_ptr + batch_id * stride_Bb
    c_batch = C_ptr + batch_id * stride_Cb

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------- loop over K ---------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A : (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B : (BLOCK_K, BLOCK_N)  (we treat B as transposed, i.e. K × N)
        b_ptrs = b_batch + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot product accumulates into FP32
        acc += tl.dot(a, b)   # a:(M,K) , b:(K,N) -> (M,N)

    # ------------------------------- store result -------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module with Triton‑accelerated core contraction.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights: dict of model parameters (torch.Tensor)
            - config: dict containing 'dim' (C) and 'hidden_dim' (H)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # Extract configuration
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Load weights – keep linear weights in FP16 for speed, norms in FP32
    # ------------------------------------------------------------------
    # LayerNorm over `dim`
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float32)
    norm_bias   = weights["norm.bias"].to(device=device, dtype=torch.float32)

    # Linear projection weights (no bias)
    left_proj_weight  = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    # Gate weights
    left_gate_weight  = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_weight   = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    # Output normalization and projection
    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=torch.float32)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device=device, dtype=torch.float32)
    to_out_weight      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (C)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # Cast to FP16 for the heavy linear work
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections
    # ------------------------------------------------------------------
    left  = F.linear(x_fp16, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x_fp16, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3) Apply mask (if present) – mask is broadcast on the channel axis
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(device=device, dtype=torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Compute gates and apply them
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_weight))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))  # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_weight))    # [B,N,N,H]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core contraction using Triton: out = left @ rightᵀ  (per hidden channel)
    # ------------------------------------------------------------------
    # Rearrange to [B, H, N, N] and flatten batch*H dimension
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    batch_hidden = B * H

    left_flat  = left_perm.view(batch_hidden, N, N)
    right_flat = right_perm.view(batch_hidden, N, N)

    out_flat = torch.empty_like(left_flat)   # FP16 output

    # Strides (in elements) for the flattened tensors
    stride_A_batch = left_flat.stride(0)   # = N*N
    stride_Am       = left_flat.stride(1)   # = N
    stride_Ak       = left_flat.stride(2)   # = 1

    stride_B_batch = right_flat.stride(0)   # = N*N
    # For B we need transposed view: (K,N)
    stride_Bk = right_flat.stride(2)   # original column stride = 1  -> stride over K
    stride_Bn = right_flat.stride(1)   # original row stride    = N -> stride over N

    stride_C_batch = out_flat.stride(0)
    stride_Cm       = out_flat.stride(1)
    stride_Cn       = out_flat.stride(2)

    # Block sizes – chosen for the target GPU
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_hidden,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _trimul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                     # M, N, K (all equal to seq_len)
        stride_A_batch, stride_Am, stride_Ak,
        stride_B_batch, stride_Bk, stride_Bn,
        stride_C_batch, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out_perm = out_flat.view(B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_fp32 = out.to(torch.float32)                  # cast back to FP32 for the following layers

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(out_fp32,
                            normalized_shape=(hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias,
                            eps=eps)

    out_norm = out_norm * out_gate.to(torch.float32)

    # Final linear projection back to dim (use FP16 for speed)
    out_norm_fp16 = out_norm.to(torch.float16)
    out_final_fp16 = F.linear(out_norm_fp16, to_out_weight)   # [B, N, N, C]
    out_final = out_final_fp16.to(torch.float32)

    return out_final
