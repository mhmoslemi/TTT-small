def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # maybe also "seqlen", "bs"? Not needed.
    eps = 1e-5
    
    device = input_tensor.device
    
    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    
    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    
    # Apply mask if provided (assuming mask shape [B,N,N])
    if mask is not None:
        mask_f = mask.to(input_tensor.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f
    
    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()
    
    # Apply gates
    left = left * left_gate
    right = right * right_gate
    
    # Triton contraction: out = left @ right^T per hidden dim
    B, N, _, _ = left.shape
    
    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # shape [B,N,N,hidden_dim]
    
    # Compute strides
    stride_l_batch = left.stride(0)
    stride_l_i = left.stride(1)
    stride_l_k = left.stride(2)
    # For right same strides, using same variable names
    stride_r_batch = right.stride(0)
    stride_r_i = right.stride(1)  # i dimension (row) corresponds to j in contraction
    stride_r_k = right.stride(2)
    stride_o_batch = out.stride(0)
    stride_o_i = out.stride(1)
    stride_o_j = out.stride(2)
    
    # Block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * hidden_dim)
    
    tri_mul_gemm_kernel[grid](
        left, right, out,
        N,
        hidden_dim,
        stride_l_batch, stride_l_i, stride_l_k,
        stride_r_batch, stride_r_i, stride_r_k,
        stride_o_batch, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    
    # Now out is shape [B,N,N,hidden_dim]
    # Apply to_out_norm
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    
    # Apply out gate
    out_norm = out_norm * out_gate
    
    # Final Linear to dim
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])
    
    return output

"""
TriMul (outgoing) implementation with Triton.

Algorithm overview
------------------
1. Layer‑normalize the input tensor over the feature dimension (`dim`).
2. Project the normalized tensor to a hidden dimension using two linear layers
   (`left_proj`, `right_proj`).  Apply the optional mask.
3. Compute three sigmoid gates from the normalized tensor and apply them to
   the projected tensors.
4. Core O(N³) contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication of shape (N×N) @ (N×N)ᵀ for each
   hidden head.  The computation is done in a Triton kernel that tiles the
   M, N and K dimensions (block sizes 64×64×32) and processes each
   (batch, hidden) pair independently.
5. Apply a second LayerNorm on the hidden dimension, multiply with the output
   gate, and finally linearly project back to the original `dim`.

All operations except the contraction are performed with PyTorch; the
contraction is the performance‑critical part and is fully fused in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(lhs_ptr, rhs_ptr, out_ptr,
                        N, hidden_dim,
                        stride_l_batch, stride_l_i, stride_l_k,
                        stride_r_batch, stride_r_i, stride_r_k,
                        stride_o_batch, stride_o_i, stride_o_j,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """Batched GEMM for the TriMul contraction.

    Computes for each (batch, head):
        out[i, j] = Σₖ lhs[i, k] * rhs[j, k]   (rhs is accessed transposed)
    """
    pid_m = tl.program_id(0)                     # block row index
    pid_n = tl.program_id(1)                     # block column index
    batch_head = tl.program_id(2)                # combined batch * hidden_dim index

    # split combined index into batch and hidden head
    b_idx = batch_head // hidden_dim
    h_idx = batch_head % hidden_dim

    # base pointers for this batch & hidden head
    lhs = lhs_ptr + b_idx * stride_l_batch + h_idx
    rhs = rhs_ptr + b_idx * stride_r_batch + h_idx
    out = out_ptr + b_idx * stride_o_batch + h_idx

    # coordinate ranges for this tile
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_base = k * BLOCK_K
        k_range = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # Load left block: shape (BLOCK_M, BLOCK_K)
        lhs_ptrs = lhs + i[:, None] * stride_l_i + k_range[None, :] * stride_l_k
        lhs_tile = tl.load(lhs_ptrs,
                           mask=mask_i[:, None] & mask_k[None, :],
                           other=0.0)

        # Load right block transposed: shape (BLOCK_K, BLOCK_N)
        # Access right[j, k] -> we treat k as row, j as column
        rhs_ptrs = rhs + k_range[:, None] * stride_r_k + j[None, :] * stride_r_i
        rhs_tile = tl.load(rhs_ptrs,
                           mask=mask_k[:, None] & mask_j[None, :],
                           other=0.0)

        # Accumulate dot product
        acc += tl.dot(lhs_tile, rhs_tile)

    # Store the result
    out_ptrs = out + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Execute the TriMul forward pass.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (float/bool) or None
            - weights: dict with the model's parameters
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input layer norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Projections and mask
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    if mask is not None:
        m = mask.to(x.dtype).unsqueeze(-1)              # [B,N,N,1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 3. Gates
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core contraction via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # shape [B,N,N,hidden_dim]

    # Strides for the contiguous layout [B,N,N,hidden_dim]
    stride_l_batch = left.stride(0)   # N*N*hidden_dim
    stride_l_i = left.stride(1)       # N*hidden_dim
    stride_l_k = left.stride(2)       # hidden_dim

    stride_r_batch = right.stride(0)
    stride_r_i = right.stride(1)      # N*hidden_dim (row dimension = j)
    stride_r_k = right.stride(2)      # hidden_dim

    stride_o_batch = out.stride(0)
    stride_o_i = out.stride(1)        # N*hidden_dim
    stride_o_j = out.stride(2)        # hidden_dim

    # Tile sizes – tuned for the B200 memory hierarchy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        N,
        hidden_dim,
        stride_l_batch,
        stride_l_i,
        stride_l_k,
        stride_r_batch,
        stride_r_i,
        stride_r_k,
        stride_o_batch,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 5. Output layer norm, output gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return output
