import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def triangular_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k, stride_l,
    stride_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    # program ids
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    b = pid_b

    # tile offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_offset < N
    mask_j = j_offset < N

    d_start = 0
    while d_start < H:
        d_offset = d_start + tl.arange(0, BLOCK_D)
        mask_d = d_offset < H

        # accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        num_k_tiles = tl.cdiv(N, BLOCK_K)
        for k_tile in range(num_k_tiles):
            k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = k_offset < N

            # load left: shape (BLOCK_M, BLOCK_K, BLOCK_D)
            left_ptrs = left_ptr + b * stride_b \
                        + i_offset[:, None, None] * stride_i \
                        + k_offset[None, :, None] * stride_k \
                        + d_offset[None, None, :] * stride_l
            mask_left = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
            left_tile = tl.load(left_ptrs, mask=mask_left, other=0.0)

            # load right: shape (BLOCK_N, BLOCK_K, BLOCK_D)
            right_ptrs = right_ptr + b * stride_b \
                         + j_offset[:, None, None] * stride_i \
                         + k_offset[None, :, None] * stride_k \
                         + d_offset[None, None, :] * stride_l
            mask_right = mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
            right_tile = tl.load(right_ptrs, mask=mask_right, other=0.0)

            # accumulate over k dimension
            for kk in range(0, BLOCK_K):
                left_vec = left_tile[:, kk, :]          # (BLOCK_M, BLOCK_D)
                right_vec = right_tile[:, kk, :]        # (BLOCK_N, BLOCK_D)
                # broadcast multiply and sum
                acc += left_vec[:, None, :] * right_vec[None, :, :]

        # store result
        out_ptrs = out_ptr + b * stride_b \
                    + i_offset[:, None, None] * stride_i \
                    + j_offset[None, :, None] * stride_j \
                    + d_offset[None, None, :] * stride_l
        mask_out = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
        tl.store(out_ptrs, acc, mask=mask_out)

        d_start += BLOCK_D

def custom_kernel(data):
    """
    Implements the outgoing Triangle Multiplicative Update (TriMul) module.
    Core N³ reduction `out[i,j,:] = Σ_k left[i,k,:] * right[j,k,:]` is
    executed in a Triton kernel that tiles the i, j and hidden dimensions.
    All other linear / LayerNorm / gating operations are performed in PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ----- LayerNorm on input -----
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ----- Linear projections -----
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ----- Apply optional mask -----
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(x.dtype)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ----- Gating -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----- Core Tensor Contraction in Triton -----
    out = torch.empty_like(left)  # shape [B,N,N,hidden_dim]

    # strides (in elements)
    stride_b, stride_i, stride_k, stride_l = left.stride()
    stride_j = out.stride(2)  # stride for the second N dimension in the output

    # tiling parameters
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 16
    BLOCK_D = 16  # must be <= hidden_dim; will be masked for the tail

    # grid dimensions: (batch, i_tiles, j_tiles)
    grid = (
        bs,
        triton.cdiv(seq_len, BLOCK_M),
        triton.cdiv(seq_len, BLOCK_N),
    )

    triangular_mul_kernel[
        grid
    ](
        left,
        right,
        out,
        bs,
        seq_len,
        hidden_dim,
        stride_b,
        stride_i,
        stride_k,
        stride_l,
        stride_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ----- Output LayerNorm and linear -----
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output

    stride_j,

):

                + i_base[:, None, None] * stride_i
                + k_base[None, :, None] * stride_k
                + d_base[None, None, :] * stride_l
            )
            mask_left = (
                mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
            )
            left_tile = tl.load(left_ptrs, mask=mask_left, other=0.0)  # (BM,BK,BD)

            # ---------- load right ----------
            right_ptrs = (
                right_ptr
                + b * stride_b
                + j_base[:, None, None] * stride_i  # right’s first N uses stride_i
                + k_base[None, :, None] * stride_k
                + d_base[None, None, :] * stride_l
            )
            mask_right = (
                mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
            )
            right_tile = tl.load(right_ptrs, mask=mask_right, other=0.0)  # (BN,BK,BD)

            # ---------- accumulate ----------
            for kk in range(0, BLOCK_K):
                # (BM, BD) and (BN, BD)
                lv = left_tile[:, kk, :]
                rv = right_tile[:, kk, :]
                # broadcast multiply -> (BM, BN, BD)
                acc += lv[:, None, :] * rv[None, :, :]

        # ---------- store ----------
        out_ptrs = (
            out_ptr
            + b * stride_b
            + i_base[:, None, None] * stride_i
            + j_base[None, :, None] * stride_j
            + d_base[None, None, :] * stride_l
        )
        mask_out = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
        tl.store(out_ptrs, acc, mask=mask_out)

        d_start += BLOCK_D


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments
    ---------
    data: tuple (input_tensor, mask, weights, config)
        - input_tensor: [B, N, N, dim] torch.Tensor
        - mask:        [B, N, N] torch.Tensor (0/1)
        - weights:     dict with all linear / layer‑norm parameters
        - config:     dict containing "dim" and "hidden_dim"

    Returns
    -------
    output: [B, N, N, dim] torch.Tensor
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -------------------------------------------------
    # 1. Input LayerNorm
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # -------------------------------------------------
    # 2. Linear projections to hidden dimension
    # -------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------
    # 3. Optional mask (mask is always provided)
    # -------------------------------------------------
    m = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
    left = left * m
    right = right * m

    # -------------------------------------------------
    # 4. Gating
    # -------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5. Core contraction in Triton
    # -------------------------------------------------
    out = torch.empty_like(left)  # [B, N, N, hidden_dim]

    # strides (in elements) for the left/right tensors
    stride_b, stride_i, stride_k, stride_l = left.stride()
    # output stride for the second N dimension (j)
    stride_j = out.stride(2)

    # tiling parameters (tuned for modest register pressure)
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 16
    BLOCK_D = 16  # hidden_dim ≤ 128 ⇒ 16 works; tail handled by masking

    grid = (
        bs,
        triton.cdiv(seq_len, BLOCK_M),
        triton.cdiv(seq_len, BLOCK_N),
    )

    triangular_mul_kernel[
        grid
    ](
        left,
        right,
        out,
        bs,
        seq_len,
        hidden_dim,
        stride_b,
        stride_i,
        stride_k,
        stride_l,
        stride_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # -------------------------------------------------
    # 6. Output LayerNorm, gating, final linear
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
