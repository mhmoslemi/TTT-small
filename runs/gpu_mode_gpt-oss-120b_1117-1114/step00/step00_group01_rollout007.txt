def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          M, N, K,
                          stride_am, stride_ak, stride_ak2,
                          stride_bm, stride_bk, stride_bn,
                          stride_cm, stride_cn, stride_ck,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_b = tl.program_id(0)  # batch index
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(2)  # block col

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    # N, N dims are unchanged; torch.nn.functional.linear expects (..., in_features) -> (..., out_features)
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # mask handling
    if mask is not None:
        # mask shape [B,N,N]
        # Ensure mask dtype matches (e.g., float32)
        mask_f = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_f
        right = right * mask_f

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Prepare for batched GEMM
    # Permute to batch hidden dimension
    left_perm = left.permute(0,3,1,2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0,3,1,2).contiguous() # [B, hidden, N, N]
    # Transpose right for GEMM
    right_perm_T = right_perm.permute(0,1,3,2).contiguous() # [B, hidden, N, N] with last dims swapped

    B, hidden, N, _ = left_perm.shape
    batch_hidden = B * hidden

    left_mat = left_perm.view(batch_hidden, N, N)
    right_mat_T = right_perm_T.view(batch_hidden, N, N)

    # allocate output
    out_mat = torch.empty_like(left_mat)   # shape [batch_hidden, N, N]

    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch_hidden, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    # compute strides
    # Strides for left_mat:
    stride_am = left_mat.stride(0)
    stride_ak = left_mat.stride(1)
    stride_ak2 = left_mat.stride(2)
    # Strides for right_mat_T:
    stride_bm = right_mat_T.stride(0)
    stride_bk = right_mat_T.stride(1)
    stride_bn = right_mat_T.stride(2)
    # Strides for out_mat:
    stride_cm = out_mat.stride(0)
    stride_cn = out_mat.stride(1)
    stride_ck = out_mat.stride(2)

    batched_matmul_kernel[grid](
        left_mat, right_mat_T, out_mat,
        N, N, N,
        stride_am, stride_ak, stride_ak2,
        stride_bm, stride_bk, stride_bn,
        stride_cm, stride_cn, stride_ck,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape out_mat back
    out_perm = out_mat.view(B, hidden, N, N)
    # permute back to [B,N,N,hidden]
    out = out_perm.permute(0,2,3,1)

    # To_Out Norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # out gate multiplication
    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, to_out_weight)  # shape [B,N,N,dim]

    return out

"""
Outgoing TriMul implementation.

The module performs:
1. LayerNorm on the input tensor.
2. Two linear projections (left/right) and three linear gates.
3. Element‑wise gating and optional masking.
4. Core N³ contraction: for each hidden dimension d,
   compute C₍b,d₎ = left₍b,d₎ @ right₍b,d₎ᵀ.
   This is realized as a batched GEMM using a custom Triton kernel.
5. LayerNorm + output gating + final linear projection back to `dim`.

Only the contraction (step 4) is executed on Triton; all other
operations use native PyTorch (which is already optimized on GPU).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ak2,
    stride_bm, stride_bk, stride_bn,
    stride_cm, stride_cn, stride_ck,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C[batch, m, n] = A[batch, m, k] @ B[batch, k, n]  (float32)"""
    pid_b = tl.program_id(0)   # batch * hidden index
    pid_m = tl.program_id(1)   # row block
    pid_n = tl.program_id(2)   # col block

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = tl.arange(0, BLOCK_K)

        # A block
        a_ptr = (
            A_ptr
            + pid_b * stride_am
            + offs_m[:, None] * stride_ak
            + (k_offset + k_range)[None, :] * stride_ak2
        )
        a_mask = (offs_m < M)[:, None] & (k_range < K - k_offset)[None, :]
        a = tl.load(a_ptr, mask=a_mask, other=0.0)

        # B block (B is already transposed, shape [batch, k, n])
        b_ptr = (
            B_ptr
            + pid_b * stride_bm
            + (k_offset + k_range)[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b_mask = (k_range < K - k_offset)[:, None] & (offs_n < N)[None, :]
        b = tl.load(b_ptr, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)

    # write back
    c_ptr = (
        C_ptr
        + pid_b * stride_cm
        + offs_m[:, None] * stride_cn
        + offs_n[None, :] * stride_ck
    )
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptr, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float)
            - weights: dict of all learnable parameters
            - config: dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---- unpack weights -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ---- LayerNorm on the input -----------------------------------------
    x = F.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )  # [B, N, N, dim]

    # ---- Linear projections (no bias) ------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # ---- Mask (if present) -----------------------------------------------
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ---- Gating -----------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ---- Prepare tensors for batched GEMM --------------------------------
    # bring hidden dimension to the second axis and make data contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, hidden, N, N]

    # transpose the last two dimensions of right for A @ B^T
    right_perm_T = right_perm.permute(0, 1, 3, 2).contiguous()  # [B, hidden, N, N]

    # merge batch and hidden dims -> treat as independent GEMM batches
    batch_hidden = B * hidden_dim
    left_mat = left_perm.view(batch_hidden, N, N)      # [BH, N, N]
    right_mat_T = right_perm_T.view(batch_hidden, N, N)  # [BH, N, N]

    out_mat = torch.empty_like(left_mat)  # [BH, N, N]

    # ---- Triton launch ----------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_hidden,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # strides (int)
    stride_am, stride_ak, stride_ak2 = left_mat.stride()
    stride_bm, stride_bk, stride_bn = right_mat_T.stride()
    stride_cm, stride_cn, stride_ck = out_mat.stride()

    _batched_matmul_kernel[grid](
        left_mat,
        right_mat_T,
        out_mat,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_ak2,
        stride_bm,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---- Reshape back to original layout ---------------------------------
    out_perm = out_mat.view(B, hidden_dim, N, N)            # [B, hidden, N, N]
    out = out_perm.permute(0, 2, 3, 1)                     # [B, N, N, hidden]

    # ---- final LayerNorm, output gating and projection --------------------
    out = F.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
```
