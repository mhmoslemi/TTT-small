def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    # 1. LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor,
                                      normalized_shape=(dim,),
                                      weight=weights["norm.weight"],
                                      bias=weights["norm.bias"])
    # 2. Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # 3. Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # 4. Apply mask and gate via Triton kernel
    has_mask = (mask_tensor is not None) and (not config.get("nomask", False))
    if has_mask:
        mask = mask_tensor.to(torch.float32)
    else:
        # dummy mask of ones (not used)
        mask = torch.empty(0, dtype=torch.float32, device=device)

    # Allocate output gated tensors
    left_gated = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    # mask contiguous
    if has_mask:
        mask = mask.contiguous()
    else:
        # still need a tensor placeholder of appropriate shape for pointer; let's use left as dummy.
        mask = left  # not used

    total = left.numel()  # B * N * N * hidden_dim

    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    mask_gate_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask,
        left_gated, right_gated,
        total,
        hidden_dim=hidden_dim,
        has_mask=int(has_mask),
        BLOCK=BLOCK,
    )
    # free unneeded tensors?
    # 5. Reduce over k dimension using batched matmul
    B, N, _, H = left_gated.shape

    # reshape for batch matmul
    left_mat = left_gated.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right_gated.permute(0, 3, 2, 1).reshape(B * H, N, N)

def mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,

    total_elements,

def custom_kernel(data):
    """
    Triton-accelerated TriMul (outgoing) forward implementation.
    
    Args:
        data: Tuple (input, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = torch.float32

    # 1. Input LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # 2. Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # 3. Gate linear layers + sigmoid
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Ensure tensors are contiguous for Triton
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # 4. Apply mask and gates in a Triton kernel
    has_mask = mask_tensor is not None and not config.get("nomask", False)
    if has_mask:
        # mask shape [B, N, N], values 0/1 (or any float)
        mask = mask_tensor.to(dtype).contiguous()
    else:
        # dummy placeholder (won't be accessed)
        mask = left  # any tensor with same dtype/device

    left_gated = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    total_elements = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)

    mask_gate_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask,
        left_gated, right_gated,
        total_elements,
        hidden_dim=hidden_dim,
        has_mask=int(has_mask),
        BLOCK=BLOCK,
    )

    # 5. Reduction over the third dimension via batched GEMM
    B, N, _, H = left_gated.shape  # H == hidden_dim
    # reshape for batched matmul: (B*H, N, N) @ (B*H, N, N)^T
    left_mat = left_gated.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right_gated.permute(0, 3, 2, 1).reshape(B * H, N, N)

    out = torch.bmm(left_mat, right_mat)  # (B*H, N, N)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1)  # (B, N, N, H)

    # 6. Output LayerNorm over hidden_dim
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # 7. Apply output gate and final projection
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out.to(dtype)

"""
TriMul (outgoing) forward pass with Triton acceleration.

The heavy elementwise part (masking + gate application) is fused into a
Triton kernel. The cubic reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed via a batched GEMM (torch.bmm) after the gating step.
All other linear layers and LayerNorms are executed with PyTorch.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    total_elements,
    hidden_dim: tl.constexpr,
    has_mask: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    left/right        : [B, N, N, H] (contiguous)
    left_gate/right_gate: same shape
    mask               : [B, N, N] (contiguous) – ignored if has_mask==0
    Produces out_left/out_right = left*left_gate*mask, right*right_gate*mask
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < total_elements

    # load raw values
    left_val = tl.load(left_ptr + offs, mask=m, other=0.0)
    right_val = tl.load(right_ptr + offs, mask=m, other=0.0)
    lgate_val = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    rgate_val = tl.load(right_gate_ptr + offs, mask=m, other=0.0)

    if has_mask:
        mask_idx = offs // hidden_dim                # map (b,i,k,h) -> (b,i,k)
        mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=0.0)
    else:
        mask_val = 1.0

    out_left = left_val * lgate_val * mask_val
    out_right = right_val * rgate_val * mask_val

    tl.store(out_left_ptr + offs, out_left, mask=m)
    tl.store(out_right_ptr + offs, out_right, mask=m)

    Triton‑accelerated TriMul forward operator.

            - weights_dict: mapping of weight names to tensors
            - config_dict: must contain "dim", "hidden_dim", and optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = torch.float32

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(

        normalized_shape=(dim,),

    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Gates + sigmoid
    # --------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # --------------------------------------------------------------
    # 4. Apply mask and gates in a Triton kernel
    # --------------------------------------------------------------
    has_mask = mask_tensor is not None and not config.get("nomask", False)

    if has_mask:

    else:

        mask = left

    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    left_gated = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    total_elems = left.numel()          # B * N * N * hidden_dim
    BLOCK = 1024
    grid = (triton.cdiv(total_elems, BLOCK),)

    mask_gate_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask,
        left_gated,
        right_gated,
        total_elems,
        hidden_dim=hidden_dim,
        has_mask=int(has_mask),
        BLOCK=BLOCK,
    )
