def custom_kernel(data):
   input_tensor, mask_tensor, weights, config = data
   bs = input_tensor.shape[0]; seq_len = input_tensor.shape[1]; dim = config["dim"]; hidden_dim = config["hidden_dim"]
   # Ensure mask shape matches or no mask
   if mask_tensor is None: # maybe mask missing, but defined? The spec says mask always provided. Might have random mask but still shape.

   # Get weights
   norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
   left_proj_weight = weights["left_proj.weight"]; right_proj_weight = weights["right_proj.weight"]
   left_gate_weight = weights["left_gate.weight"]; right_gate_weight = weights["right_gate.weight"]; out_gate_weight = weights["out_gate.weight"]
   to_out_norm_weight = weights["to_out_norm.weight"]; to_out_norm_bias = weights["to_out_norm.bias"]
   to_out_weight = weights["to_out.weight"]

   # Compute normalized x
   # Use torch.nn.functional.layer_norm: shape normalized_shape = (dim,)
   x_norm = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)

   # Compute projections
   left = torch.nn.functional.linear(x_norm, left_proj_weight) # out shape [B,N,N,hidden_dim]
   right = torch.nn.functional.linear(x_norm, right_proj_weight)

   # Cast mask
   if mask_tensor is None:
        # mask all ones
        mask = torch.ones((bs, seq_len, seq_len), dtype=left.dtype, device=left.device)
   else:
        mask = mask_tensor
   mask = mask.unsqueeze(-1) # shape [B,N,N,1]

   # Apply mask (elementwise) using a Triton kernel (to satisfy requirement)
   # We'll implement kernel for elementwise multiplication left,right with mask.
   # But we can simply multiply in pyTorch; to satisfy we need a kernel, we could have a small kernel for mask.

   # But we can postpone; for simplicity to have at least one kernel, we will use the tri_mul_kernel which is a kernel.

   # Cast left,right to fp16
   left = left.to(torch.float16)
   right = right.to(torch.float16)

   mask_fp = mask.to(torch.float16)

   left = left * mask_fp
   right = right * mask_fp

   # Compute gates
   left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight)).to(torch.float16)
   right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight)).to(torch.float16)
   out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

   left = left * left_gate
   right = right * right_gate

   # Compute tri-mul using kernel
   out = torch.empty((bs, seq_len, seq_len, hidden_dim), dtype=torch.float32, device=left.device)

   # Get strides
   stride_left = left.stride()
   stride_right = right.stride()
   stride_out = out.stride()
   # stride tuple order: (B_dim, N_dim_i, N_dim_k, H_dim). Since shape [B,N,N,H].
   stride_left_b, stride_left_i, stride_left_k, stride_left_h = stride_left
   stride_right_b, stride_right_i, stride_right_k, stride_right_h = stride_right
   stride_out_b, stride_out_i, stride_out_j, stride_out_h = stride_out

   # Block sizes
   BLOCK_M = 64
   BLOCK_N = 64
   BLOCK_K = 32

   total_hb = bs * hidden_dim
   grid = (total_hb, triton.cdiv(seq_len, BLOCK_M), triton.cdiv(seq_len, BLOCK_N))
   # Launch kernel
   tri_mul_kernel[grid](
       left.data_ptr(),
       right.data_ptr(),
       out.data_ptr(),
       bs,
       seq_len,
       hidden_dim,
       stride_left_b, stride_left_i, stride_left_k, stride_left_h,
       stride_right_b, stride_right_i, stride_right_k, stride_right_h,
       stride_out_b, stride_out_i, stride_out_j, stride_out_h,
       BLOCK_M=BLOCK_M,
       BLOCK_N=BLOCK_N,
       BLOCK_K=BLOCK_K,
   )

   # After multiplication, out is in float32.

   # Layernorm to hidden dim
   out_norm = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

   # Scale by out_gate
   out_scaled = out_norm * out_gate

   # Final linear projection
   out_final = torch.nn.functional.linear(out_scaled, to_out_weight)

   return out_final

"""
TriMul (outgoing) forward pass with a Triton kernel for the core
N³‑hidden_dim reduction.  The implementation follows the reference
PyTorch code but off‑loads the sum‑over‑k operation
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
to a custom Triton kernel that does a batched matrix multiply over the
(hidden_dim, batch) dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out = left @ right^T for each (batch, hidden) slice.
    left/right are half‑precision, out is single‑precision.
    Grid:
        program_id(0) -> combined batch * hidden index
        program_id(1) -> i‑tile
        program_id(2) -> j‑tile
    """
    pid_hb = tl.program_id(0)          # combined (batch, hidden)
    pid_i = tl.program_id(1)           # tile over rows
    pid_j = tl.program_id(2)           # tile over cols

    # decode batch index and hidden channel
    b = pid_hb // H
    h = pid_hb % H

    # start offsets for this tile
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of tiles along the reduction dimension
    num_k = tl.cdiv(N, BLOCK_K)

    for k_tile in range(0, num_k):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (M, K)
        a_ptrs = left_ptr + \
                 b * stride_left_b + \
                 h * stride_left_h + \
                 offs_i[:, None] * stride_left_i + \
                 offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right tile (transposed): shape (K, N)
        b_ptrs = right_ptr + \
                 b * stride_right_b + \
                 h * stride_right_h + \
                 offs_k[:, None] * stride_right_k + \
                 offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # dot: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(a, b)

    # write the result
    out_ptrs = out_ptr + \
               b * stride_out_b + \
               h * stride_out_h + \
               offs_i[:, None] * stride_out_i + \
               offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update.
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (float32) → cast to fp16 for the heavy part
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Apply mask (if any)
    # ------------------------------------------------------------------
    if mask_tensor is None:
        mask = torch.ones((bs, seq_len, seq_len), dtype=torch.float16, device=device)
    else:
        mask = mask_tensor.to(torch.float16)
    mask = mask.unsqueeze(-1)          # [B,N,N,1]
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # keep FP32 for later scaling

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul reduction via Triton
    # ------------------------------------------------------------------
    out = torch.empty((bs, seq_len, seq_len, hidden_dim), dtype=torch.float32, device=device)

    # stride information (B, N_i, N_k, H)
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    stride_left_b, stride_left_i, stride_left_k, stride_left_h = s_left
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = s_right
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = s_out

    # Block sizes – chosen to stay within shared memory limits
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_hb = bs * hidden_dim
    grid = (
        total_hb,
        triton.cdiv(seq_len, BLOCK_M),
        triton.cdiv(seq_len, BLOCK_N),
    )
    _tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        bs,
        seq_len,
        hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, out‑gate scaling, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate               # broadcast over hidden_dim

    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
