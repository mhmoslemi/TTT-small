def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # Load weights
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"] # shape (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm input (dim)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    # Projections (no bias)
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask
    mask_exp = mask.unsqueeze(-1)  # shape B,N,N,1
    left = left * mask_exp
    right = right * mask_exp

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Allocate output
    out = torch.empty_like(left_fp16, dtype=torch.float32).contiguous()  # shape B,N,N,hidden_dim

    # Prepare strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_fp16.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_fp16.stride()
    stride_out_b, stride_out_i, stride_out_k, stride_out_h = out.stride()  # using same naming conv

    # For right transposition we need stride_right_j = stride_right_i
    stride_right_j = stride_right_i

    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), (input_tensor.shape[0] * hidden_dim))
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_k, stride_out_h,
        N,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Now out shape [B,N,N,hidden_dim] float32
    # Apply final layernorm and gating
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out_norm = out_norm * out_gate

    # Final linear
    output = torch.nn.functional.linear(out_norm, to_out_weight)
    return output
```

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    N, H,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(2)  # combined batch and hidden dim
    pid_m = tl.program_id(0)   # i block
    pid_n = tl.program_id(1)   # j block

    # batch and hidden index
    b = pid_bh // H
    d = pid_bh % H

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this batch and hidden channel
    left_base = left_ptr + b * stride_left_b + d * stride_left_h
    right_base = right_ptr + b * stride_right_b + d * stride_right_h
    out_base = out_ptr + b * stride_out_b + d * stride_out_h

    # K offset base range
    offs_k = tl.arange(0, BLOCK_K)

    # Compute pointer offsets
    left_ptrs = left_base + (offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k)
    right_ptrs = right_base + (offs_k[:, None] * stride_right_j + offs_n[None, :] * stride_right_k)

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for _ in range(0, tl.cdiv(N, BLOCK_K)):
        # Load tiles (fp16)
        left_tile = tl.load(left_ptrs, mask=mask_m[:, None] & mask_n[None, :], other=0.0)  # mask only for i/j? Actually we need mask for i and k but we ignore k mask as N divisible by BLOCK_K.

        # Compute dot product
        acc += tl.dot(left_tile, right_tile)

        # Increment pointers
        left_ptrs += BLOCK_K * stride_left_k
        right_ptrs += BLOCK_K * stride_right_j

"""
TriMul forward kernel (outgoing update) implemented with Triton.

Algorithm
---------
1. Layer‐normalize the input tensor.
2. Apply linear projections (left/right) and gating linear layers.
3. Mask the projected tensors and multiply by the corresponding gates.
4. Compute the triangular multiplicative update:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication where each hidden dimension `d`
   is treated as an independent batch.  The core contraction is performed by
   a Triton kernel that computes C = A @ Bᵀ for each (batch, hidden) pair.
5. Layer‑normalize the result, apply the output gate and a final linear
   projection back to the original hidden dimension.

The Triton kernel is a standard blocked GEMM with:
    BLOCK_M = rows per tile (i dimension)
    BLOCK_N = columns per tile (j dimension)
    BLOCK_K = reduction block size (k dimension)
Hidden dimension is folded into the kernel's third grid axis, so each
kernel instance works on one (batch, hidden) slice.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,               # fp16   [B, N, N, H]
    right_ptr,              # fp16   [B, N, N, H]
    out_ptr,                # fp32   [B, N, N, H]
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    N, H,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Program IDs
    pid_bh = tl.program_id(2)   # combined batch * hidden dim
    pid_m  = tl.program_id(0)   # i‑tile
    pid_n  = tl.program_id(1)   # j‑tile

    # De‑compose batch and hidden indices
    b = pid_bh // H
    d = pid_bh %  H

    # Offsets within the i and j dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this (batch, hidden) slice
    left_base  = left_ptr  + b * stride_left_b  + d * stride_left_h
    right_base = right_ptr + b * stride_right_b + d * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + d * stride_out_h

    # Offsets for the reduction dimension k
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the current tile of A (left) and Bᵀ (right)
    # left: rows = i, columns = k
    left_ptrs = left_base + (offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k)
    # rightᵀ: rows = k, columns = j   (original right[j,k,d])
    right_ptrs = right_base + (offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j)

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the k dimension (N is always a multiple of BLOCK_K in the test set)
    for _ in range(0, tl.cdiv(N, BLOCK_K)):
        left_tile  = tl.load(left_ptrs,  mask=mask_m[:, None], other=0.0)   # fp16
        right_tile = tl.load(right_ptrs, mask=mask_n[None, :], other=0.0)   # fp16
        acc += tl.dot(left_tile, right_tile)   # (BLOCK_M, BLOCK_N)  fp32
        # Advance to the next k‑block
        left_ptrs  += BLOCK_K * stride_left_k
        right_ptrs += BLOCK_K * stride_right_k

    # Store the computed tile
    out_ptrs = out_base + (offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask:        torch.Tensor of shape [B, N, N]
            - weights:     dict of parameter tensors
            - config:      dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ----- Load model parameters ------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]    # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]   # (hidden_dim, dim)

    left_gate_weight  = weights["left_gate.weight"]    # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]   # (hidden_dim, dim)
    out_gate_weight   = weights["out_gate.weight"]     # (hidden_dim, dim)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (hidden_dim,)
    to_out_norm_bias   = weights["to_out_norm.bias"]    # (hidden_dim,)

    to_out_weight = weights["to_out.weight"]           # (dim, hidden_dim)

    # ----- LayerNorm -----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ----- Linear projections (no bias) ----------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ----- Apply mask -----------------------------------------------------------
    mask_exp = mask.unsqueeze(-1)            # [B, N, N, 1]
    left  = left * mask_exp
    right = right * mask_exp

    # ----- Gating --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left  * left_gate
    right = right * right_gate

    # ----- Cast to half‑precision for the kernel -------------------------------
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp32   = torch.empty_like(left_fp16, dtype=torch.float32).contiguous()  # [B, N, N, hidden_dim]

    # ----- Strides (in elements) ------------------------------------------------
    slb, sli, slk, slh = left_fp16.stride()
    srb, srj, srk, srh = right_fp16.stride()
    sob, soi, soj, soh = out_fp32.stride()

    # ----- Triton launch parameters -------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑tiles
        triton.cdiv(N, BLOCK_N),          # j‑tiles
        B * hidden_dim,                   # batch * hidden dimension
    )

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        slb, sli, slk, slh,
        srb, srj, srk, srh,
        sob, soi, soj, soh,
        N,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- Post‑processing ------------------------------------------------------
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out_norm = out_norm * out_gate

    # Final linear projection back to `dim`
    output = F.linear(out_norm, to_out_weight)   # [B, N, N, dim]

    return output
