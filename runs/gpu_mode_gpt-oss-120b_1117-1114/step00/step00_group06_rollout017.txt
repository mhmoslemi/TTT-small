import triton
import triton.language as tl

@triton.jit
def batched_matmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # Compute pointer offsets
        a_ptrs = a_ptr + pid_batch * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_batch * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        # Load with mask
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)
        # Compute dot product
        acc += tl.dot(a, b)
    # Write back
    c_ptrs = c_ptr + pid_batch * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    # LayerNorm for input
    # For normalization along dim (last axis)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # apply mask if present and not nomask
    if not nomask:
        # ensure mask dtype matches
        mask_ = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_
        right = right * mask_
    else:
        # mask ignored; or apply ones for consistency.
        pass

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute out = sum_k left[i,k,d] * right[j,k,d] efficiently via Triton.
    bs, N, _, _ = left.shape   # left shape [B,N,N, hidden_dim]
    # reshape to (batch*hidden_dim, N, N)
    left_reshape = left.permute(0, 3, 1, 2).contiguous()
    right_reshape = right.permute(0, 3, 1, 2).contiguous()
    batch_hidden = bs * hidden_dim
    left_reshape = left_reshape.view(batch_hidden, N, N)
    right_reshape = right_reshape.view(batch_hidden, N, N)
    # Prepare output
    out_reshape = torch.empty_like(left_reshape)

    # Strides
    a_batch_stride = left_reshape.stride(0)
    b_batch_stride = right_reshape.stride(0)
    c_batch_stride = out_reshape.stride(0)

    stride_am = left_reshape.stride(1)  # row stride (i)
    stride_ak = left_reshape.stride(2)  # col stride (k)

    # For B_T: row stride is original k stride = right_reshape.stride(2) (should be 1)
    stride_bk = right_reshape.stride(2)  # k stride
    stride_bn = right_reshape.stride(1)  # j stride

    stride_cm = out_reshape.stride(1)
    stride_cn = out_reshape.stride(2)

    # Block sizes: choose powers of 2, tune based on N.
    # We'll set BLOCK_M = BLOCK_N = 64 (or 128) if N is larger.
    # Choose BLOCK_K = 32 or 64.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch_hidden, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    batched_matmul_outgoing_kernel[grid](
        left_reshape, right_reshape, out_reshape,
        a_batch_stride, b_batch_stride, c_batch_stride,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape back to [B,N,N,hidden_dim]
    out = out_reshape.view(bs, hidden_dim, N, N).permute(0,2,3,1).contiguous()

    # Apply to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

import torch
import triton
import triton.language as tl

def _layer_norm(x, weight, bias):
    return torch.nn.functional.layer_norm(x, (x.shape[-1],), weight=weight, bias=bias)

@triton.jit
def batched_matmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        a_ptrs = a_ptr + pid_batch * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_batch * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_ptr + pid_batch * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)

def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.
    The core pairwise multiplication over the sequence dimension is performed
    with a tiled Triton kernel, while all LayerNorms, linear projections and
    gating operations are executed in PyTorch.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --- LayerNorm on input (over channel dim) ---
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # --- Linear projections (no bias) ---
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --- Apply mask if present ---
    if not nomask:
        # Cast mask to same dtype as activations
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # --- Gates ---
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --- Triton kernel for the O(N^3) update ---
    B, N, _, _ = left.shape  # N is seq_len
    # reshape to (B*hidden_dim, N, N) where hidden dim is the last dim
    left_ = left.permute(0, 3, 1, 2).contiguous()
    right_ = right.permute(0, 3, 1, 2).contiguous()
    batch_hidden = B * hidden_dim
    left_ = left_.view(batch_hidden, N, N)
    right_ = right_.view(batch_hidden, N, N)

    # output buffer
    out_ = torch.empty_like(left_)

    # strides (in elements, not bytes)
    a_batch_stride = left_.stride(0)
    b_batch_stride = right_.stride(0)
    c_batch_stride = out_.stride(0)

    stride_am = left_.stride(1)   # row stride (i)
    stride_ak = left_.stride(2)   # column stride (k)

    # For right we want to read it as transposed (k,j)
    stride_bk = right_.stride(2)  # original k stride, becomes row stride after transpose
    stride_bn = right_.stride(1)  # original j stride, becomes column stride

    stride_cm = out_.stride(1)
    stride_cn = out_.stride(2)

    # Tile sizes – choose a reasonable default; they fit most test sizes.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_hidden,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_outgoing_kernel[grid](
        left_, right_, out_,
        a_batch_stride, b_batch_stride, c_batch_stride,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --- Output LayerNorm over hidden dimension ---
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to original dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

import torch
import triton
import triton.language as tl

# ------------------------------------------------------------
# Triton kernel: batched matmul for the outgoing TriMul update
# Computes, for each batch*hidden channel:
#   C[i, j] = sum_k A[i, k] * B_T[k, j]
# where B_T is the transposed view of the original right tensor.
# ------------------------------------------------------------
@triton.jit
def batched_matmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)

        # Pointers with proper strides
        a_ptrs = a_ptr + pid_batch * batch_stride_a \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_batch * batch_stride_b \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)

    c_ptrs = c_ptr + pid_batch * batch_stride_c \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.
    All LayerNorms, linear projections and gating are performed in PyTorch.
    The expensive O(N³) pairwise multiplication is carried out by a
    tiled Triton kernel that performs a batched matrix multiplication
    between the left tensor and the transpose of the right tensor.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the feature dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask application (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) based on the normalized input
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel: compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    This is a batched matmul of shape (B*D, N, N) × (B*D, N, N)^T.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                     # left shape: [B, N, N, hidden_dim]

    # reshape to (batch*hidden, N, N) where the hidden dim is the leading channel
    left_ = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_ = right.permute(0, 3, 1, 2).contiguous()
    batch_hidden = B * hidden_dim
    left_ = left_.view(batch_hidden, N, N)
    right_ = right_.view(batch_hidden, N, N)

    out_ = torch.empty_like(left_)                 # placeholder for the result

    # strides in element units (not bytes)
    a_batch_stride = left_.stride(0)
    b_batch_stride = right_.stride(0)
    c_batch_stride = out_.stride(0)

    stride_am = left_.stride(1)    # row stride (i)
    stride_ak = left_.stride(2)    # column stride (k)

    # right tensor is accessed as transposed (k, j)
    stride_bk = right_.stride(2)   # k stride becomes row stride after transpose
    stride_bn = right_.stride(1)   # j stride becomes column stride

    stride_cm = out_.stride(1)
    stride_cn = out_.stride(2)

    # Tile sizes – good defaults for the tested sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_hidden,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_outgoing_kernel[grid](
        left_, right_, out_,
        a_batch_stride, b_batch_stride, c_batch_stride,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape back to original layout [B, N, N, hidden_dim]
    out = out_.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # ------------------------------------------------------------------
    # 7. Apply the output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
