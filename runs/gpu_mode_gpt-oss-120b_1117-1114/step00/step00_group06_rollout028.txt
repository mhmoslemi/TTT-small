@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, 
    FP16: tl.constexpr
):
    pid = tl.program_id(0) # batch index (combined)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block start
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute pointers for A and B for this batch entry
    a_batch_ptr = a_ptr + pid * stride_ah
    b_batch_ptr = b_ptr + pid * stride_bh
    c_batch_ptr = c_ptr + pid * stride_ch

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        cur_k = min(BLOCK_K, K - k)
        # offsets for k
        offs_k = k + tl.arange(0, cur_k)
        # load A sub-block: shape (BLOCK_M, cur_k)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        # load B sub-block: shape (cur_k, BLOCK_N) (since B_T: rows=K, cols=N)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        # compute dot product: update acc
        acc += tl.dot(a, b)
    # write result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16) if FP16 else acc, mask=mask_m[:, None] & mask_n[None, :])

import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    FP16: tl.constexpr
):
    pid = tl.program_id(0) # batch*hidden index
    pid_m = tl.program_id(1) # i tile
    pid_n = tl.program_id(2) # j tile

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    a_batch = a_ptr + pid * stride_ah
    b_batch = b_ptr + pid * stride_bh
    c_batch = c_ptr + pid * stride_ch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        cur_k = tl.minimum(BLOCK_K, K - k) # not used if we mask
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        if FP16:

        # Load B_T block

        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, _ = input_tensor.shape

    # layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)          # (bs,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)

    # mask
    if mask is not None and mask.numel()>0:
        mask_exp = mask.unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Convert to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to (bs, hidden_dim, N, N) and reshape to (bs*hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_h = bs * hidden_dim
    A = left_perm.view(batch_h, N, N)
    B = right_perm.view(batch_h, N, N)
    C = torch.empty_like(A)  # float16

    # strides
    stride_ah = A.stride(0)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)
    stride_bh = B.stride(0)
    stride_bk = B.stride(2)   # original col stride
    stride_bn = B.stride(1)   # original row stride
    stride_ch = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    BLOCK_M = min(BLOCK_M, N)
    BLOCK_N = min(BLOCK_N, N)
    BLOCK_K = min(BLOCK_K, N)

    grid = (batch_h, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    batched_matmul_kernel[grid](
        A, B, C,
        N, N, N,
        stride_am, stride_ak, stride_ah,
        stride_bk, stride_bn, stride_bh,
        stride_cm, stride_cn, stride_ch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        FP16=True
    )
    # Reshape back
    out = C.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # final layernorm and linear
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)
    out = out * out_gate
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)  # shape (bs,N,N,dim)
    return out

"""
TriMul forward implementation with Triton acceleration.

Algorithm
---------

1. Layer‑normalize the input tensor (shape [B, N, N, C]).
2. Linear projections (`left_proj`, `right_proj`) and gated linear layers are
   computed in PyTorch (the cost is negligible compared to the cubic term).
3. The expensive contraction

       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

   is performed as a batched matrix multiplication:
   for each (b, h) we compute  A = left[b, :, :, h]   (N×N)
                             B = right[b, :, :, h]  (N×N)
   and obtain   C = A @ Bᵀ .
   The batch dimension is flattened (B·H) and a custom Triton kernel
   executes many independent GEMMs in parallel (one per (b, h) pair).

4. The result (still in FP16) is reshaped back to [B, N, N, H],
   cast to FP32, passed through a second LayerNorm, gated,
   and finally projected back to the original channel dimension `C`.

The kernel uses FP16 arithmetic with a FP32 accumulator (Tensor‑core friendly)
and works for any supported sequence length and hidden dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix sizes (all = seq_len)
    stride_am, stride_ak, stride_ah,    # A strides: (row, col, batch*hid)
    stride_bk, stride_bn, stride_bh,    # Bᵀ strides (row = K, col = N, batch*hid)
    stride_cm, stride_cn, stride_ch,    # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    FP16: tl.constexpr,
):
    # program IDs
    pid = tl.program_id(0)          # flattened (batch * hidden)
    pid_m = tl.program_id(1)        # i‑tile
    pid_n = tl.program_id(2)        # j‑tile

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this (batch, hidden) instance
    a_batch = a_ptr + pid * stride_ah
    b_batch = b_ptr + pid * stride_bh
    c_batch = c_ptr + pid * stride_ch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the K dimension in tiles
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A block: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        if FP16:
            a = a.to(tl.float16)

        # Bᵀ block: (BLOCK_K, BLOCK_N)   (original B is transposed on‑the‑fly)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        if FP16:
            b = b.to(tl.float16)

        # matrix multiplication for the tile
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    if FP16:
        tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
    else:
        tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of TriMul with a Triton‑accelerated cubic contraction.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, C]   (float32, on CUDA)
            - mask_tensor : [B, N, N]    (float32 or bool)
            - weights_dict: model parameters (see code)
            - config_dict : {"dim": C, "hidden_dim": H}

    Returns:
        Tensor of shape [B, N, N, C] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ----- 1. LayerNorm on the input -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ----- 2. Linear projections ----------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)          # [B, N, N, H]
    right = F.linear(x, right_proj_weight)

    # ----- 3. Apply mask (if any) ----------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ----- 4. Gating ----------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ----- 5. Cast to FP16 for the heavy cubic part -------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # move hidden dimension to second axis and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch and hidden dimension -> (B*H, N, N)
    batch_h = bs * hidden_dim
    A = left_perm.view(batch_h, N, N)
    B = right_perm.view(batch_h, N, N)

    # allocate output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # ----- 6. Strides for the Triton kernel ---------------------------------------
    stride_ah = A.stride(0)
    stride_am = A.stride(1)   # row stride (M)
    stride_ak = A.stride(2)   # col stride (K)

    stride_bh = B.stride(0)
    # For Bᵀ we swap the last two strides
    stride_bk = B.stride(2)   # original col stride -> stride of K dimension
    stride_bn = B.stride(1)   # original row stride -> stride of N dimension

    stride_ch = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    # ----- 7. Launch Triton kernel --------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_M = min(BLOCK_M, N)
    BLOCK_N = min(BLOCK_N, N)
    BLOCK_K = min(BLOCK_K, N)

    grid = (
        batch_h,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A,
        B,
        C,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_ah,
        stride_bk,
        stride_bn,
        stride_bh,
        stride_cm,
        stride_cn,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        FP16=True,
    )

    # ----- 8. Reshape back to original layout ---------------------------------------
    out = C.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ----- 9. Final LayerNorm, gating and projection --------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)   # -> [B, N, N, C]

    return out
