"""
TriMul (outgoing) forward kernel.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ ℝ^{B×N×N×D}.
2. Linear projections (no bias):
   * left  = x @ W_left   ∈ ℝ^{B×N×N×H}
   * right = x @ W_right  ∈ ℝ^{B×N×N×H}
3. Gating vectors (sigmoid):
   * left_gate  = sigmoid(x @ W_left_gate)   ∈ ℝ^{B×N×N×H}
   * right_gate = sigmoid(x @ W_right_gate)  ∈ ℝ^{B×N×N×H}
   * out_gate   = sigmoid(x @ W_out_gate)    ∈ ℝ^{B×N×N×H}
4. Fuse mask, left‑gate and right‑gate with a Triton kernel:
   left  ← left  * left_gate  * mask
   right ← right * right_gate * mask
   (mask is broadcast along the hidden dimension.)
5. Contract over the sequence dimension k:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiply: for each (b,h) we compute
   out_{b,h} = left_{b,h} @ right_{b,h}ᵀ.
   Implemented efficiently with torch.bmm on a reshaped view.
6. Layer‑norm the contraction result, apply out_gate and a final linear
   projection to the original dimension D.

The heavy element‑wise masking/gating step is performed in Triton,
while the rest of the work (layer‑norms, linear layers, and the
contraction) uses highly‑optimized PyTorch/CUDA kernels.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    mask_ptr,
    # strides for left/right tensors (shape B,N,N,H)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_g_b, stride_g_i, stride_g_k, stride_g_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    # strides for mask tensor (shape B,N,N)
    stride_m_b, stride_m_i, stride_m_k,
    B, N, H,                # constexprs
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # Decompose linear offset into (b,i,k,h)
    # offset = ((b*N + i)*N + k)*H + h
    h = offs % H
    idx = offs // H                      # idx = ((b*N + i)*N + k)
    k = idx % N
    idx_i = idx // N                     # idx_i = b*N + i
    i = idx_i % N
    b = idx_i // N

    # Compute pointers
    left_off   = b * stride_l_b + i * stride_l_i + k * stride_l_k + h * stride_l_h
    left_gate_off = b * stride_g_b + i * stride_g_i + k * stride_g_k + h * stride_g_h
    right_off  = b * stride_r_b + i * stride_r_i + k * stride_r_k + h * stride_r_h
    right_gate_off = b * stride_g_b + i * stride_g_i + k * stride_g_k + h * stride_g_h

    mask_off = b * stride_m_b + i * stride_m_i + k * stride_m_k

    # Load values (masked loads return 0.0 for out‑of‑bounds)
    left_val       = tl.load(left_ptr   + left_off,   mask=mask, other=0.0)
    left_gate_val  = tl.load(left_gate_ptr + left_gate_off, mask=mask, other=0.0)
    right_val      = tl.load(right_ptr  + right_off,  mask=mask, other=0.0)
    right_gate_val = tl.load(right_gate_ptr + right_gate_off, mask=mask, other=0.0)
    mask_val       = tl.load(mask_ptr + mask_off, mask=mask, other=0.0)

    # Apply gating and mask
    left_out  = left_val * left_gate_val * mask_val
    right_out = right_val * right_gate_val * mask_val

    # Store results back (in‑place)
    tl.store(left_ptr   + left_off,   left_out,   mask=mask)
    tl.store(right_ptr  + right_off,  right_out,  mask=mask)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul module using a Triton kernel
    for mask‑gate fusion.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, D] (float32)
            - mask_tensor:  torch.Tensor of shape [B, N, N] (bool/float)
            - weights: dictionary with the following keys (all tensors are float32):
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict containing at least "dim", "hidden_dim".
    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, D = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. LayerNorm on the input
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias   = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (D,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device)   # (H, D)
    right_proj_w = weights["right_proj.weight"].to(device) # (H, D)

    left  = F.linear(x, left_proj_w)   # [B, N, N, H]
    right = F.linear(x, right_proj_w)  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 3. Gating vectors (sigmoid)
    # -----------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device)
    right_gate_w = weights["right_gate.weight"].to(device)
    out_gate_w   = weights["out_gate.weight"].to(device)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # [B, N, N, H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # [B, N, N, H]

    # -----------------------------------------------------------------
    # 4. Apply mask and gates with a Triton kernel
    # -----------------------------------------------------------------
    # Ensure mask is float and broadcastable
    mask_f = mask.to(dtype).contiguous()
    # Flatten tensors are already contiguous after linear
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    total_elements = B * N * N * hidden_dim
    BLOCK = 1024  # threads per program; fits within hardware limits
    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_kernel[grid](
        left_ptr=left.data_ptr(),
        left_gate_ptr=left_gate.data_ptr(),
        right_ptr=right.data_ptr(),
        right_gate_ptr=right_gate.data_ptr(),
        mask_ptr=mask_f.data_ptr(),
        stride_l_b=left.stride(0), stride_l_i=left.stride(1),
        stride_l_k=left.stride(2), stride_l_h=left.stride(3),
        stride_g_b=left_gate.stride(0), stride_g_i=left_gate.stride(1),
        stride_g_k=left_gate.stride(2), stride_g_h=left_gate.stride(3),
        stride_r_b=right.stride(0), stride_r_i=right.stride(1),
        stride_r_k=right.stride(2), stride_r_h=right.stride(3),
        stride_m_b=mask_f.stride(0), stride_m_i=mask_f.stride(1), stride_m_k=mask_f.stride(2),
        B=B, N=N, H=hidden_dim,
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 5. Contraction over k: out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    #    Implemented as a batched GEMM (left @ rightᵀ) with batch size B*H.
    # -----------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    # batched matmul: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device)
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate  # broadcasted element‑wise

    to_out_w = weights["to_out.weight"].to(device)  # (D, H)
    out = F.linear(out, to_out_w)  # [B, N, N, D]

    return out
