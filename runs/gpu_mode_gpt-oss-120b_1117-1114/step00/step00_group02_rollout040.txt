"""
TriMul (outgoing) forward pass with a fused Triton kernel for the heavy
pairwise contraction:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

The kernel computes this for a block of (i,j) and a block of the hidden
dimension d using a tiled GEMM style algorithm (loop over k in BLOCK_K
chunks).  The rest of the module (LayerNorm, Linear projections, gating,
final Linear) is performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------------- #
# Triton kernel: block (i,j) × hidden slice (d) multiplication
# ----------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel(
    left_ptr,          # *mut* left tensor   [B,N,N,H]
    right_ptr,         # *mut* right tensor  [B,N,N,H]
    out_ptr,           # *mut* output tensor [B,N,N,H]
    B,                 # batch size (runtime)
    N,                 # sequence length (compile‑time)
    H,                 # hidden dimension (runtime)
    d_offset,          # start offset of hidden block (runtime)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """
    Compute a tile of the TriMul contraction.
    Each program instance handles one (batch) × block_i × block_j tile.
    """
    pid_b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)          # block over i (row)
    pid_j = tl.program_id(2)          # block over j (col)

    # --------------------------------------------------------------- #
    # Offsets for the i and j dimensions of this block
    # --------------------------------------------------------------- #
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    i = i0 + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    j = j0 + tl.arange(0, BLOCK_N)          # [BLOCK_N]
    d = d_offset + tl.arange(0, BLOCK_D)    # [BLOCK_D]

    i_mask = i < N
    j_mask = j < N
    d_mask = d < H

    # --------------------------------------------------------------- #
    # Base pointers for this batch (add batch stride)
    # --------------------------------------------------------------- #
    left_base  = left_ptr  + pid_b * stride_left_b
    right_base = right_ptr + pid_b * stride_right_b
    out_base   = out_ptr   + pid_b * stride_out_b

    # --------------------------------------------------------------- #
    # Accumulator for a (BLOCK_M, BLOCK_N, BLOCK_D) tile
    # --------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # --------------------------------------------------------------- #
    # Loop over the k dimension in BLOCK_K chunks
    # --------------------------------------------------------------- #
    for k_start in range(0, N, BLOCK_K):
        k = k_start + tl.arange(0, BLOCK_K)          # [BLOCK_K]
        k_mask = k < N

        # ---- left tile: shape (BLOCK_M, BLOCK_K, BLOCK_D) -----------------
        left_ptrs = (left_base
                     + i[:, None, None] * stride_left_i
                     + k[None, :, None] * stride_left_k
                     + d[None, None, :] * stride_left_h)
        mask_left = (i_mask[:, None, None] &
                     k_mask[None, :, None] &
                     d_mask[None, None, :])
        left_tile = tl.load(left_ptrs, mask=mask_left, other=0.0)

        # ---- right tile: shape (BLOCK_K, BLOCK_N, BLOCK_D) ----------------
        right_ptrs = (right_base
                      + j[None, :, None] * stride_right_i
                      + k[:, None, None] * stride_right_k
                      + d[None, None, :] * stride_right_h)
        mask_right = (j_mask[None, :, None] &
                      k_mask[:, None, None] &
                      d_mask[None, None, :])
        right_tile = tl.load(right_ptrs, mask=mask_right, other=0.0)

        # ---- Accumulate using a dot for each hidden slice ---------------
        for dh in range(BLOCK_D):
            # left_tile[:, :, dh] -> (M, K)
            # right_tile[:, :, dh] -> (K, N)
            acc[:, :, dh] += tl.dot(left_tile[:, :, dh], right_tile[:, :, dh])

    # --------------------------------------------------------------- #
    # Write out the accumulated tile
    # --------------------------------------------------------------- #
    out_ptrs = (out_base
                + i[:, None, None] * stride_out_i
                + j[None, :, None] * stride_out_j
                + d[None, None, :] * stride_out_h)
    mask_out = (i_mask[:, None, None] &
                j_mask[None, :, None] &
                d_mask[None, None, :])
    tl.store(out_ptrs, acc, mask=mask_out)


# ----------------------------------------------------------------------------- #
# Python entry point
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul module using the Triton kernel for the
    O(N³·H) contraction.  All other linear / LayerNorm / gating steps are
    performed with PyTorch.
    """
    input_tensor, mask, weights, config = data

    # --------------------------------------------------------------------- #
    # Unpack configuration and weights
    # --------------------------------------------------------------------- #
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = torch.float32

    # LayerNorm for the input
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask if present (mask shape [B,N,N])
    if mask is not None:
        mask_expanded = mask.unsqueeze(-1).to(dtype)          # [B,N,N,1]
        left = left * mask_expanded
        right = right * mask_expanded

    # Gating
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape

    # --------------------------------------------------------------------- #
    # Triton kernel launch for the O(N³·H) contraction
    # --------------------------------------------------------------------- #
    # Ensure contiguity (required for correct stride values)
    left = left.contiguous()
    right = right.contiguous()

    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # Block sizes – chosen to keep register usage reasonable
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 32
    BLOCK_D = 32

    # Compile‑time constants for N (required for the loop inside the kernel)
    N_const = int(N)   # Triton expects a python int for constexpr arguments

    grid = (B,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    # Strides (in element units, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Loop over the hidden dimension in blocks of BLOCK_D
    for d_start in range(0, hidden_dim, BLOCK_D):
        # Pad the last block with a mask inside the kernel
        tri_mul_kernel[grid](
            left, right, out,
            B, N_const, hidden_dim, d_start,
            stride_left_b, stride_left_i, stride_left_k, stride_left_h,
            stride_right_b, stride_right_i, stride_right_k, stride_right_h,
            stride_out_b, stride_out_i, stride_out_j, stride_out_h,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )

    # --------------------------------------------------------------------- #
    # Final LayerNorm, gating and projection back to dim
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
