@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_k, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
    # program id for batch*hidden
    pid = tl.program_id(0)  # flatten over B*H
    # compute block row i and col j ids from program_id(1) and (2)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute base indices
    b_h = pid  # index in flatten
    b = b_h // H
    h = b_h % H

    # offsets for rows and cols
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask within N
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the beginning of the matrices for this batch and hidden dim
    left_ptrs = left_ptr + b * stride_left_b + h * stride_left_h
    right_ptrs = right_ptr + b * stride_right_b + h * stride_right_h
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # k offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptrs + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0) # convert to float32

        # load right tile: shape (BLOCK_K, BLOCK_N)
        # right stored as permuted: shape (k,j) where k is row, j is column
        # pointer: stride_right_k for row dimension, stride_right_j for col dimension
        b_ptrs = right_ptrs + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # store results
    c_ptrs = out_ptrs + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    stride_right_k = stride_right_j
    stride_right_n = stride_right_i

    stride_out_b = out_fp16.stride(0)

    stride_out_j = out_fp16.stride(2)
    stride_out_h = out_fp16.stride(3)

    BLOCK_N = 64
    BLOCK_K = 32

        stride_right_h,

        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    out = out_fp16.to(torch.float32)

        bias=to_out_norm_bias,
    )

    out = torch.nn.functional.linear(out, to_out_weight)

"""TriMul custom kernel.

Fast implementation of the "outgoing" triangle multiplicative update (TriMul)
used in AlphaFold3.  The expensive O(N³) reduction
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
is performed by a fused Triton kernel that treats the operation as a batch of
matrix multiplications per hidden channel.  All other linear, gating and
layer‑norm operations are handled by PyTorch.

The kernel works for batch sizes 1‑2, sequence lengths up to 1024 and hidden
dimension 128.  Input tensors are expected on CUDA; the output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,   # *fp16
    right_ptr,  # *fp16 (original right tensor, accessed as transposed)
    out_ptr,    # *fp16 output
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_k, stride_right_n, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul kernel for TriMul.
    For each (batch, hidden) pair we compute:
        out[i, j] = sum_k left[i, k] * right_T[k, j]
    where right_T is right transposed on the two inner dimensions.
    """
    pid = tl.program_id(0)   # batch * hidden index
    pid_m = tl.program_id(1)  # row block
    pid_n = tl.program_id(2)  # col block

    # derive batch and hidden indices
    b = pid // H
    h = pid % H

    # offsets within the N x N matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this (b, h)
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # right tile (already treated as transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # write result back as fp16
    c = acc.to(tl.float16)
    c_ptrs = out_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Fast forward pass of the TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, dim]
            mask: torch.Tensor [B, N, N] (bool/int/float)
            weights: dict of pre‑loaded tensors
            config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ----- Load weights ----------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ----- LayerNorm on input ---------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ----- Projections ------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ----- Mask (if provided) -----------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ----- Gating -----------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----- Convert to fp16 for the heavy O(N³) kernel -----------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, H = left_fp16.shape

    # Output buffer (fp16)
    out_fp16 = torch.empty_like(left_fp16)

    # ----- Strides ---------------------------------------------------------
    stride_left_b = left_fp16.stride(0)
    stride_left_i = left_fp16.stride(1)
    stride_left_k = left_fp16.stride(2)
    stride_left_h = left_fp16.stride(3)

    stride_right_b = right_fp16.stride(0)
    # original strides for right
    stride_right_i = right_fp16.stride(1)  # original i
    stride_right_j = right_fp16.stride(2)  # original j
    stride_right_h = right_fp16.stride(3)

    # In the kernel we treat right as transposed (k as rows, j as cols)
    stride_right_k = stride_right_j
    stride_right_n = stride_right_i

    stride_out_b = out_fp16.stride(0)
    stride_out_i = out_fp16.stride(1)
    stride_out_j = out_fp16.stride(2)
    stride_out_h = out_fp16.stride(3)

    # ----- Triton kernel launch --------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_k,
        stride_right_n,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Cast back to fp32 for the remaining layers
    out = out_fp16.to(torch.float32)

    # ----- Output LayerNorm and gating ---------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate

    # ----- Final projection -------------------------------------------------
    out = F.linear(out, to_out_weight)

    return out

"""TriMul custom kernel.

Fast implementation of the “outgoing” triangle multiplicative update (TriMul)
used in AlphaFold3 and related models.  The costly O(N³) reduction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is performed by a fused Triton kernel that treats each (batch, hidden)
pair as an independent matrix multiplication.  All other linear, gating and
layer‑norm operations are done with PyTorch.  The kernel works for batch
sizes 1‑2, sequence lengths up to 1024 and hidden dimension 128.  Input and
mask tensors are expected on CUDA; the final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,   # *fp16
    right_ptr,  # *fp16 (original right tensor, accessed as transposed)
    out_ptr,    # *fp16 output
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_k, stride_right_n, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul kernel for TriMul.

    For each (batch, hidden) pair we compute:
        out[i, j] = Σ_k left[i, k] * right_T[k, j]
    where right_T is the view of `right` transposed on its inner two axes.
    """
    pid = tl.program_id(0)   # flattened (batch * hidden) index
    pid_m = tl.program_id(1)  # row‑block index
    pid_n = tl.program_id(2)  # col‑block index

    # Recover batch and hidden‑channel indices
    b = pid // H
    h = pid % H

    # Offsets inside the N×N matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this (b, h) slice
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a tile of `left` (M×K)
        a_ptrs = left_base + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load a tile of `right` treated as transposed (K×N)
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Store the result back as fp16
    c = acc.to(tl.float16)
    c_ptrs = out_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul module using a Triton kernel for the O(N³) core.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be all‑ones when no mask)
            - weights: dict of torch.Tensors containing model parameters
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ----------- Load weights ------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ----------- Input layer‑norm -------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ----------- Linear projections -----------------------------------------
    left = F.linear(x, left_proj_weight)    # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ----------- Optional mask ----------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ----------- Gating ------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----------- Cast to fp16 for the heavy O(N³) kernel --------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, H = left_fp16.shape
    out_fp16 = torch.empty_like(left_fp16)

    # ----------- Strides -----------------------------------------------------
    stride_left_b = left_fp16.stride(0)
    stride_left_i = left_fp16.stride(1)
    stride_left_k = left_fp16.stride(2)
    stride_left_h = left_fp16.stride(3)

    stride_right_b = right_fp16.stride(0)
    # original strides of `right`
    stride_right_i = right_fp16.stride(1)  # i dimension
    stride_right_j = right_fp16.stride(2)  # j dimension
    stride_right_h = right_fp16.stride(3)

    # In the kernel we view `right` as transposed: rows = k (original j), cols = j (original i)
    stride_right_k = stride_right_j
    stride_right_n = stride_right_i

    stride_out_b = out_fp16.stride(0)
    stride_out_i = out_fp16.stride(1)
    stride_out_j = out_fp16.stride(2)
    stride_out_h = out_fp16.stride(3)

    # ----------- Launch Triton kernel ---------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (B * H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_k,
        stride_right_n,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------- Convert back to fp32 ---------------------------------------
    out = out_fp16.to(torch.float32)

    # ----------- Output layer‑norm, gating and final linear -----------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)

    return out
