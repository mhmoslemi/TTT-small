import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def apply_mul_mask_gate_kernel(proj_ptr, mask_ptr, gate_ptr, out_ptr,
                               total_elements, hidden_dim,
                               BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements
    # Compute index in mask (broadcast over hidden_dim)
    idx_mask = offs // hidden_dim
    # Load values (use 0.0 for out-of-range)
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + idx_mask, mask=mask, other=0.0)
    out = proj * m * gate
    tl.store(out_ptr + offs, out, mask=mask)

def custom_kernel(data):
    """
    Triton-accelerated implementation of the outgoing Triangle Multiplicative
    Update (TriMul) used in AlphaFold3. The heavy tensor contraction is performed
    with a batched GEMM (torch.bmm) while the element‑wise mask and gating
    multiplication is fused into a small Triton kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----- LayerNorm on input -----
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ----- Linear projections -----
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ----- Gates -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ----- Prepare mask -----
    # mask shape: [B, N, N], broadcast over hidden_dim
    mask_f = mask.to(dtype=dtype)

    # ----- Fuse mask and gate using Triton -----
    # Flatten tensors to 1‑D for the kernel
    B, N, _, _ = input_tensor.shape
    total = B * N * N * hidden_dim

    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    # flatten
    left_proj_flat = left_proj.view(-1)
    right_proj_flat = right_proj.view(-1)
    left_gate_flat = left_gate.view(-1)
    right_gate_flat = right_gate.view(-1)
    mask_flat = mask_f.view(-1)          # shape B*N*N
    left_out_flat = left_out.view(-1)
    right_out_flat = right_out.view(-1)

    # Triton launch configuration
    BLOCK = 8192  # number of elements processed per program
    grid = (triton.cdiv(total, BLOCK),)

    # left side
    apply_mul_mask_gate_kernel[grid](
        left_proj_flat,
        mask_flat,
        left_gate_flat,
        left_out_flat,
        total,
        hidden_dim,
        BLOCK=BLOCK,
    )
    # right side
    apply_mul_mask_gate_kernel[grid](
        right_proj_flat,
        mask_flat,
        right_gate_flat,
        right_out_flat,
        total,
        hidden_dim,
        BLOCK=BLOCK,
    )

    # Reshape back to [B, N, N, H]
    left = left_out
    right = right_out

    # ----- Batched matrix multiplication: out = left @ right^T -----
    # Bring hidden dimension to batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    # Compute batch GEMM (right transposed)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore [B, N, N, H]

    # ----- Output LayerNorm -----
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ----- Output gate -----
    out = out * out_gate

    # ----- Final linear projection back to dim -----
    out = F.linear(out, weights["to_out.weight"])

    # Cast to float32 as required
    return out.to(torch.float32)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: fuse mask broadcasting, element‑wise gating and multiplication
# -------------------------------------------------------------------------
@triton.jit
def apply_mul_mask_gate_kernel(
    proj_ptr,            # pointer to projected tensor  (B,N,N,H) flattened
    mask_ptr,            # pointer to mask tensor        (B,N,N)   flattened
    gate_ptr,            # pointer to gate tensor        (B,N,N,H) flattened
    out_ptr,             # pointer to output tensor      (B,N,N,H) flattened
    total_elements,      # int: B*N*N*H
    hidden_dim,          # int: H
    BLOCK: tl.constexpr  # number of elements processed per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # index of the corresponding mask element (broadcast over hidden dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    out = proj * m * gate
    tl.store(out_ptr + offs, out, mask=mask)

# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul.
    The heavy contraction (left @ rightᵀ) is performed with a batched GEMM
    (torch.bmm).  The element‑wise mask & gate multiplication is fused
    into a tiny Triton kernel to satisfy the “use Triton” requirement.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps        = 1e-5

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2) Linear projections
    # -----------------------------------------------------------------
    left_proj  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right_proj = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Gating (sigmoid of linear)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 4) Fuse mask & gates using Triton
    # -----------------------------------------------------------------
    # ensure mask is float for multiplication
    mask_f = mask.to(dtype=dtype)

    B, N, _, _ = input_tensor.shape
    total = B * N * N * hidden_dim                # total elements in (B,N,N,H)

    left  = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    left_proj_flat  = left_proj.view(-1)
