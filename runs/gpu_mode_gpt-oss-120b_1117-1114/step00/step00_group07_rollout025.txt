    B, N, _, H = left.shape
    assert right.shape == left.shape
    # left,right are half
    out = torch.empty_like(left)  # same dtype and shape
    stride_b = left.stride(0)
    stride_i = left.stride(1)

    right = right.contiguous()

        pid_j = tl.program_id(1)

        off_hidden = hidden_idx * stride_left_h

            acc += tl.dot(a, b)

    out_fp16 = torch.empty_like(left_fp16)  # same shape

    stride_li = left_fp16.stride(1)
    stride_lk = left_fp16.stride(2)
    stride_lh = left_fp16.stride(3)

    stride_rb = right_fp16.stride(0)
    stride_ri = right_fp16.stride(1)
    stride_rk = right_fp16.stride(2)
    stride_rh = right_fp16.stride(3)

    stride_ob = out_fp16.stride(0)
    stride_oi = out_fp16.stride(1)
    stride_oj = out_fp16.stride(2)
    stride_oh = out_fp16.stride(3)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H

    trimul_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_ri,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # convert out to float32
    out = out_fp16.to(torch.float32)

    # Apply to_out_norm and gating
    # out_gate shape [B,N,N,H] float32
    # torch.nn.functional.layer_norm expects normalized shape (hidden_dim)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, to_out_w)

    return out

"""
TriMul forward kernel (outgoing version) using Triton.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B,N,N,C]).
2. Linear projections to hidden dimension H (no bias):
   left = x @ W_left_proj , right = x @ W_right_proj .
3. Optional masking (mask shape [B,N,N] -> broadcast over H).
4. Gating: left_gate = sigmoid(x @ W_left_gate) , right_gate = sigmoid(x @ W_right_gate) .
   Apply gates: left *= left_gate , right *= right_gate .
5. Compute the core O(N³) reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   This is a batched matrix‑multiply with the second operand transposed.
   The heavy computation is performed by a custom Triton kernel that:
   * processes each (batch, hidden) pair as an independent matrix,
   * tiles the i‑ and j‑dimensions (64×64) and reduces over k in 32‑element chunks,
   * loads FP16 data, accumulates in FP32, and stores FP32 results.
6. Apply a second LayerNorm over the hidden dimension, multiply by the output gate
   (sigmoid(x @ W_out_gate)), and project back to the original channel dimension C
   with a final linear layer.
All operations after the kernel are kept in FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # runtime sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)          # tile row index
    pid_j = tl.program_id(1)          # tile col index
    pid_z = tl.program_id(2)          # (batch, hidden) index

    hidden = H
    batch_idx    = pid_z // hidden
    hidden_idx   = pid_z %  hidden

    # Offsets for the selected batch and hidden channel
    off_batch_left   = batch_idx * stride_left_b
    off_batch_right  = batch_idx * stride_right_b
    off_batch_out    = batch_idx * stride_out_b
    off_hidden_left  = hidden_idx * stride_left_h
    off_hidden_right = hidden_idx * stride_right_h
    off_hidden_out   = hidden_idx * stride_out_h

    # Tile coordinates (may exceed N, masked later)
    row = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    col = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_row = row < N
    mask_col = col < N
    mask_out = mask_row[:, None] & mask_col[None, :]

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k_start in range(0, N, BLOCK_K):
        k = k_start + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ----- Load left tile: shape (BLOCK_M, BLOCK_K) -----
        a_ptrs = (left_ptr + off_batch_left + off_hidden_left
                  + row[:, None] * stride_left_i
                  + k[None, :] * stride_left_k)
        a = tl.load(a_ptrs,
                    mask=mask_row[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ----- Load right tile (transposed): shape (BLOCK_K, BLOCK_N) -----
        # right is indexed as (batch, i=j, k, hidden)
        b_ptrs = (right_ptr + off_batch_right + off_hidden_right
                  + col[None, :] * stride_right_i
                  + k[:, None] * stride_right_k)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_col[None, :],
                    other=0.0).to(tl.float32)

        # ----- Multiply‑accumulate -----
        acc += tl.dot(a, b)

    # Store the result (FP32)
    out_ptrs = (out_ptr + off_batch_out + off_hidden_out
                + row[:, None] * stride_out_i
                + col[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_out)


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Compute the outgoing TriMul operation.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor : torch.Tensor of shape [B, N, N] (bool/float)
            - weights_dict : mapping str -> torch.Tensor containing all model weights
            - config_dict  : contains "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------- LayerNorm -------------------
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------- Projections -------------------
    left_proj_w  = weights["left_proj.weight"].to(device, dtype)
    right_proj_w = weights["right_proj.weight"].to(device, dtype)
    left  = F.linear(x, left_proj_w)      # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)

    # ------------------- Masking (if any) -------------------
    # maskTensor is always provided; when nomask=True it consists of ones.
    mask = mask_tensor.to(dtype).unsqueeze(-1)          # [B,N,N,1]
    left  = left * mask
    right = right * mask

    # ------------------- Gating -------------------
    left_gate_w  = weights["left_gate.weight"].to(device, dtype)
    right_gate_w = weights["right_gate.weight"].to(device, dtype)
    out_gate_w   = weights["out_gate.weight"].to(device, dtype)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------- Triton matmul reduction -------------------
    # Cast to FP16 for the kernel (memory‑bandwidth friendly)
    left_fp16  = left.contiguous().to(torch.float16)
    right_fp16 = right.contiguous().to(torch.float16)

    B, N, _, H = left_fp16.shape

    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32, device=device)

    # Strides (int values)
    s_lb = left_fp16.stride(0)
    s_li = left_fp16.stride(1)
    s_lk = left_fp16.stride(2)
    s_lh = left_fp16.stride(3)

    s_rb = right_fp16.stride(0)
    s_ri = right_fp16.stride(1)
    s_rk = right_fp16.stride(2)
    s_rh = right_fp16.stride(3)

    s_ob = out_fp32.stride(0)
    s_oi = out_fp32.stride(1)
    s_oj = out_fp32.stride(2)
    s_oh = out_fp32.stride(3)

    # Tile sizes – chosen to fit typical NVIDIA tensor cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H
    )

    _trimul_matmul_kernel[grid](
        left_fp16, right_fp16, out_fp32,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------- Second LayerNorm + output gate -------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype)

    out = F.layer_norm(out_fp32, (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    out = out * out_gate

    # ------------------- Final projection back to C -------------------
    to_out_w = weights["to_out.weight"].to(device, dtype)
    out = F.linear(out, to_out_w)      # shape [B,N,N,dim]

    return out
