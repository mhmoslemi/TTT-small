def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config['hidden_dim']
    # Convert to fp32 for norm and linear
    # but may use fp16 for speed. Let's use fp32 for linear weights.
    # compute x_norm
    x_norm = torch.nn.functional.layer_norm(input_tensor,
                                            normalized_shape=(dim,),
                                            weight=weights['norm.weight'],
                                            bias=weights['norm.bias'],
                                            eps=1e-5)
    # Projections + gates
    # linear: weight shape [hidden_dim, dim]
    left = torch.nn.functional.linear(x_norm, weights['left_proj.weight'])
    right = torch.nn.functional.linear(x_norm, weights['right_proj.weight'])
    # mask
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_
    # gate
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights['left_gate.weight']))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights['right_gate.weight']))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights['out_gate.weight']))
    left = left * left_gate
    right = right * right_gate
    # Cast to float16 for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    # Prepare output tensor
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=input_tensor.device)
    # Compute with triton kernel
    # Compute strides
    # left strides (in elements):
    stride_left_b = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_d = left.stride(3)
    # right strides
    stride_right_b = right.stride(0)
    stride_right_j = right.stride(1)
    stride_right_k = right.stride(2)
    stride_right_d = right.stride(3)
    # out strides
    stride_out_b = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_d = out.stride(3)
    # tile sizes
    BLOCK_M = 64 # or 32
    BLOCK_N = 64
    BLOCK_K = 32
    # grid
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), bs * hidden_dim)
    trimul_kernel[grid](left, right, out,
                        N,
                        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
                        stride_right_b, stride_right_j, stride_right_k, stride_right_d,
                        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
                        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K)
    # Apply To_out_norm, out_gate, to_out projection
    # Compute to_out_norm: layer_norm over hidden_dim
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,),
                                         weight=weights['to_out_norm.weight'],
                                         bias=weights['to_out_norm.bias'],
                                         eps=1e-5)
    # Multiply with out_gate (same shape)
    # out_gate is shape [bs,N,N,hidden_dim], we computed earlier as float32
    out = out * out_gate
    # Final linear to dim
    output = torch.nn.functional.linear(out, weights['to_out.weight'])
    return output

"""
TriMul (outgoing) implementation with a Triton kernel for the heavy
i‑k‑j reduction. The kernel computes, for each batch b and hidden channel d,

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where left/right already contain the projection, mask and gating.
The rest of the module (LayerNorms, Linear projections, gates) is kept
in PyTorch for simplicity. The kernel works on float16 tensors and
accumulates in float32, producing a float32 output ready for the
subsequent LayerNorm and final projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, hidden_dim,
    # strides for left tensor
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    # strides for right tensor
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    # strides for out tensor
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out = Σ_k left[...,i,k,d] * right[...,j,k,d] for a single
    (batch, hidden) pair.
    """
    pid_m = tl.program_id(0)   # block row (i)
    pid_n = tl.program_id(1)   # block col (j)
    pid_z = tl.program_id(2)   # combined batch * hidden index

    # ----- decode batch and hidden index -----
    b = pid_z // hidden_dim
    d = pid_z % hidden_dim

    # ----- base pointers for this (b,d) -----
    left_base  = left_ptr  + b * stride_left_b  + d * stride_left_d
    right_base = right_ptr + b * stride_right_b + d * stride_right_d
    out_base   = out_ptr   + b * stride_out_b   + d * stride_out_d

    # ----- coordinate ranges -----
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k):
        k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_mask = mask_i[:, None] & mask_k[None, :]
        left_tile = tl.load(
            left_base + i[:, None] * stride_left_i + k[None, :] * stride_left_k,
            mask=left_mask, other=0.0
        )

        # right tile transposed: we need (k, j) layout → shape (BLOCK_K, BLOCK_N)
        right_mask = mask_k[:, None] & mask_j[None, :]
        right_tile = tl.load(
            right_base + k[:, None] * stride_right_k + j[None, :] * stride_right_j,
            mask=right_mask, other=0.0
        )

        # multiply‑accumulate (cast to float32 after dot)
        acc += tl.dot(left_tile, right_tile).to(tl.float32)

    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(
        out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j,
        acc, mask=out_mask
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask_tensor : torch.Tensor [B, N, N] (bool/float)
            - weights_dict: dict with all linear / layer‑norm parameters
            - config_dict : dict containing "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -------------------- unpack --------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------- LayerNorm on dim --------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -------------------- Linear projections --------------------
    # x shape: [B,N,N,dim] → keep last dim as feature
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])        # [B,N,N,hidden]

    # -------------------- Optional masking --------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(x.dtype)        # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -------------------- Gates (sigmoid) --------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate   # still float32
    right = right * right_gate

    # Cast to half for the heavy reduction
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # -------------------- Allocate output for reduction --------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # -------------------- Compute strides (in elements) --------------------
    # left/right are contiguous because they were created by linear → they are
    # standard .contiguous() tensors.
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # -------------------- Triton launch parameters --------------------
    BLOCK_M = 64   # rows of the output tile
    BLOCK_N = 64   # cols of the output tile
    BLOCK_K = 32   # reduction tile size

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    trimul_kernel[grid](
        left,
        right,
        out,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_d,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_d,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # -------------------- Post‑processing --------------------
    # Normalize over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to original dim
    output = F.linear(out, weights["to_out.weight"])

    return output
