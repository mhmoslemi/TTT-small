Now implement kernel signature:

@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)  # combined batch*hid

    # Compute batch and hidden channel
    hidden_blocks = H
    batch_idx = pid_hb // H
    hid = pid_hb % H  # hidden channel index

    # compute i offset range
    offs_i = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds for i, j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute base pointers for left: left[batch_idx, i, k, hid]
    # left_ptr: pointer to start of left tensor
    # offset calculation:
    # ptr = left_ptr + batch_idx*stride_left_b + offs_i*stride_left_i + k*stride_left_k + hid*stride_left_h

    # Similarly for right: right[batch_idx, j, k, hid] (note j dimension first)
    # right_ptr + batch_idx*stride_right_b + offs_j*stride_right_j + k*stride_right_k + hid*stride_right_h

    # out_ptr: out[batch_idx, i, j, hid]
    # out_ptr + batch_idx*stride_out_b + offs_i*stride_out_i + offs_j*stride_out_j + hid*stride_out_h

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offset
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load left tile
        # left_ptrs shape: (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + batch_idx * stride_left_b + \
                    offs_i[:, None] * stride_left_i + \
                    k_offset[None, :] * stride_left_k + \
                    hid * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right tile
        right_ptrs = right_ptr + batch_idx * stride_right_b + \
                     offs_j[:, None] * stride_right_j + \
                     k_offset[None, :] * stride_right_k + \
                     hid * stride_right_h
        # shape (BLOCK_N, BLOCK_K)
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
        # transpose right to (BLOCK_K, BLOCK_N)
        right_T = tl.trans(right)

        # dot product
        acc += tl.dot(left, right_T)  # (BLOCK_M, BLOCK_N)

    # apply mask for store
    out_ptrs = out_ptr + batch_idx * stride_out_b + \
               offs_i[:, None] * stride_out_i + \
               offs_j[None, :] * stride_out_j + \
               hid * stride_out_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton-accelerated forward pass of TriMul (outgoing version).
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # linear projections
    left = F.linear(x, weights["left_proj.weight"])

"""
TriMul (outgoing) forward pass with Triton acceleration.

The core expensive operation is the 4‑D einsum:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

We compute this reduction using a custom Triton kernel that tiles the
(i, j) output plane and iterates over the k‑dimension in blocks.
Each kernel instance processes a single batch element and a single hidden
channel (h), so the accumulator size is modest (BLOCK_M×BLOCK_N) and fits
in registers. All other operations (LayerNorm, Linear projections,
gating, and the final output projection) are performed with PyTorch
functional APIs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out = Σ_k left[..., i, k, h] * right[..., j, k, h].

    Each program processes a tile (BLOCK_M, BLOCK_N) of the (i, j) plane
    for a specific batch element b and hidden channel h.
    """
    pid_i = tl.program_id(0)          # tile over i
    pid_j = tl.program_id(1)          # tile over j
    pid_hb = tl.program_id(2)         # combined batch * hidden index

    # --------------------------------------------------------------
    # decode batch index and hidden channel index
    # --------------------------------------------------------------
    b = pid_hb // H
    h = pid_hb % H

    # --------------------------------------------------------------
    # compute i/j offsets for the current tile
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # reduction over k
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + b * stride_l_b \
                    + offs_i[:, None] * stride_l_i \
                    + offs_k[None, :] * stride_l_k \
                    + h * stride_l_h
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile: shape (BLOCK_N, BLOCK_K)
        right_ptrs = right_ptr + b * stride_r_b \
                     + offs_j[:, None] * stride_r_j \
                     + offs_k[None, :] * stride_r_k \
                     + h * stride_r_h
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0)

        # transpose right to (BLOCK_K, BLOCK_N) for dot product
        right_T = tl.trans(right)

        # accumulate dot product over k
        acc += tl.dot(left, right_T)   # (BLOCK_M, BLOCK_N)

    # --------------------------------------------------------------
    # write result
    # --------------------------------------------------------------
    out_ptrs = out_ptr + b * stride_o_b \
               + offs_i[:, None] * stride_o_i \
               + offs_j[None, :] * stride_o_j \
               + h * stride_o_h
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.Tensor of shape [B, N, N] (may be all‑ones)
        - weights      : dict of model weights (torch.Tensors)
        - config       : dict with keys:
            - "dim"         : input channel dimension C
            - "hidden_dim"  : hidden dimension H
            - "nomask"      : bool, True => ignore mask

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    # ------------------------------------------------------------------
    # configuration
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor.to(dtype),
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask dtype must match left/right
        m = mask.to(left.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy reduction kernel
    # ------------------------------------------------------------------
    left_f16 = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_f16.shape

    # output tensor for the einsum result (float32)
    out = torch.empty((B, N, N, hidden_dim), device=device, dtype=dtype)

    # ------------------------------------------------------------------
    # Strides (in elements, not bytes)
    # ------------------------------------------------------------------
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_f16.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right_f16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # ------------------------------------------------------------------
    # Triton launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # i dimension
        triton.cdiv(N, BLOCK_N),   # j dimension
        B * hidden_dim,            # batch * hidden
    )

    _trimul_einsum_kernel[grid](
        left_f16,
        right_f16,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_j,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
