@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_i, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # Compute num blocks
    num_i_blocks = tl.cdiv(N, BLOCK_M)
    num_j_blocks = tl.cdiv(N, BLOCK_N)
    # compute indices
    total_i_j = num_i_blocks * num_j_blocks
    hd_idx = pid // total_i_j
    ij_idx = pid % total_i_j
    i_block = ij_idx // num_j_blocks
    j_block = ij_idx % num_j_blocks

    b = hd_idx // H
    d = hd_idx % H

    # compute start offset
    i_off = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    k_off = tl.arange(0, BLOCK_K)

    # mask within bounds
    mask_i = i_off < N
    mask_j = j_off < N

    # pointers for A and B
    # A: left[b, d, i, k] 
    a_ptrs = left_ptr + b * stride_left_b + d * stride_left_h + i_off[:, None] * stride_left_i + k_off[None, :] * stride_left_k
    # B: right[b, d, j, k]
    b_ptrs = right_ptr + b * stride_right_b + d * stride_right_h + j_off[:, None] * stride_right_i + k_off[None, :] * stride_right_k

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for start_k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute offset for k
        cur_k = start_k * BLOCK_K
        k_range = cur_k + k_off
        # mask for k out of bounds
        mask_k = k_range < N

        # load A and B blocks with mask
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b_mat = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
        # transpose b_mat to shape (BLOCK_K, BLOCK_N)
        b_T = tl.trans(b_mat)  # shape (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b_T)

        # increment pointers for next k block
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

"""
TriMul "outgoing" implementation with a Triton kernel for the core
pairwise multiplication over the sequence dimension.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension (dim).
2. Linear projections (left/right) and gating (left/right/out) are applied
   using the provided weights.  If a mask is present (and nomask=False) it
   is broadcasted and multiplied into the projected tensors.
3. The heavy N³ update
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is performed as a batched matrix multiplication:
      left_d  = left[..., d]  (shape B×N×N)
      right_d = right[..., d] (shape B×N×N)
      out_d   = left_d @ right_dᵀ
   The kernel processes each (batch, hidden) pair as an independent
   matrix‑multiply using block‑wise tiling (BLOCK_M×BLOCK_N) and loops
   over the K dimension in chunks (BLOCK_K).  Accumulators are float32,
   while the inputs are stored in float16 for bandwidth efficiency.
4. The resulting tensor (still in float16) is permuted to shape
   [B, N, N, hidden_dim] and cast to float32.
5. Layer‑norm (to_out_norm), element‑wise gating (out_gate) and the final
   linear projection (to_out) produce the output [B, N, N, dim].

Only the N³ multiplication (the true bottleneck) is offloaded to Triton;
all other linear / norm / sigmoid operations stay in PyTorch.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched matmul for each (batch, hidden) pair.
# out[b, d, i, j] = Σ_k left[b, d, i, k] * right[b, d, j, k]
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, H, N,                               # dimensions
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_i, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # number of tiles per dimension
    num_i_tiles = tl.cdiv(N, BLOCK_M)
    num_j_tiles = tl.cdiv(N, BLOCK_N)
    tiles_per_hd = num_i_tiles * num_j_tiles

    # decode program id -> (batch, hidden, i_tile, j_tile)
    hd_idx = pid // tiles_per_hd
    tile_idx = pid % tiles_per_hd
    i_tile = tile_idx // num_j_tiles
    j_tile = tile_idx % num_j_tiles

    b = hd_idx // H
    d = hd_idx % H

    # offsets within the tile
    i_off = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    k_off = tl.arange(0, BLOCK_K)

    mask_i = i_off < N
    mask_j = j_off < N

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_iter in range(0, num_k_tiles):
        cur_k = k_iter * BLOCK_K
        k_idx = cur_k + k_off
        mask_k = k_idx < N

        # pointers for the current K block
        a_ptrs = (left_ptr
                  + b * stride_left_b
                  + d * stride_left_h
                  + i_off[:, None] * stride_left_i
                  + k_idx[None, :] * stride_left_k)
        b_ptrs = (right_ptr
                  + b * stride_right_b
                  + d * stride_right_h
                  + j_off[:, None] * stride_right_i
                  + k_idx[None, :] * stride_right_k)

        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0, dtype=tl.float16)
        b_mat = tl.load(b_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0, dtype=tl.float16)

        # transpose B so its inner dimension aligns with A
        b_T = tl.trans(b_mat)               # (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b_T)               # (BLOCK_M, BLOCK_N) in float32

    # store result (back to float16)
    out_ptrs = (out_ptr
                + b * stride_out_b
                + d * stride_out_h
                + i_off[:, None] * stride_out_i
                + j_off[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple containing
            - input: torch.Tensor of shape [B, N, N, dim]
            - mask:  torch.Tensor of shape [B, N, N] (may be all ones)
            - weights: dict with the model weights (see problem statement)
            - config: dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # 2. Projections
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] -> [B,N,N,1] for broadcasting
        mask_exp = mask.to(dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare tensors for Triton kernel
    # ------------------------------------------------------------------
    # Use float16 for the heavy multiplication
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to [B, hidden_dim, N, N] (contiguous)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()

    B, H, N, N2 = left_perm.shape
    assert N == N2, "Non‑square pairwise tensor"

    # output buffer in the same layout
    out_perm = torch.empty_like(left_perm, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 6. Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_i_tiles = (N + BLOCK_M - 1) // BLOCK_M
    num_j_tiles = (N + BLOCK_N - 1) // BLOCK_N
    total_tiles = B * H * num_i_tiles * num_j_tiles

    grid = (total_tiles,)

    left_ptr = left_perm
    right_ptr = right_perm
    out_ptr = out_perm

    # strides
    stride_left_b, stride_left_h, stride_left_i, stride_left_k = left_perm.stride()
    stride_right_b, stride_right_h, stride_right_i, stride_right_k = right_perm.stride()
    stride_out_b, stride_out_h, stride_out_i, stride_out_j = out_perm.stride()

    trimul_outgoing_kernel[grid](
        left_ptr, right_ptr, out_ptr,
        B, H, N,
        stride_left_b, stride_left_h, stride_left_i, stride_left_k,
        stride_right_b, stride_right_h, stride_right_i, stride_right_k,
        stride_out_b, stride_out_h, stride_out_i, stride_out_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Post‑kernel processing (PyTorch)
    # ------------------------------------------------------------------
    # back to shape [B, N, N, hidden_dim] and float32
    out = out_perm.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # 7a. Output LayerNorm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    # 7b. Output gate
    out = out * out_gate

    # 7c. Final linear projection to dim
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)   # shape [B,N,N,dim]

    return out
