def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs = input_tensor.shape[0]
    seqlen = input_tensor.shape[1]  # N
    # (expected shape: [bs, N, N, dim])
    # Get weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    # Compute normalized input
    # Use torch.nn.functional.layer_norm
    # We assume layernorm along last dim (dim)
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)
    # Linear projections:
    # left = x @ left_proj_weight.t()
    # right similar
    left = torch.nn.functional.linear(x, left_proj_weight)  # weight shape (hidden_dim, dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # mask handling
    if mask is not None:
        # mask shape [bs, N, N], unsqueeze to (...,1) to broadcast
        left = left * mask.unsqueeze(-1)
        right = right * mask.unsqueeze(-1)
    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()
    # Apply gating to left/right
    left = left * left_gate
    right = right * right_gate

    # Compute triangular multiplication via Triton kernel
    # left/right are [B,N,N,H], dtype float32
    # output shape same
    B = bs
    N = seqlen
    H = hidden_dim
    out = torch.empty((B, N, N, H), dtype=left.dtype, device=left.device)

    # Strides
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Determine block sizes:
    BLOCK_M = 64  # can tune
    BLOCK_N = 64
    BLOCK_K = 32

    # grid
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)
    # launch kernel
    trimul_matmul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_ri, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Now out is shape [B,N,N,H] result of left@right^T per hidden dim.

    # Apply to_out_norm (layernorm over hidden dim)
    out = torch.nn.functional.layer_norm(out, normalized_shape=(H,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr

    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

The heavy O(N³) part
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is expressed as a batch of independent matrix multiplications
    out_d = left_d @ right_dᵀ
for each hidden channel d (and each batch).  The Triton kernel computes
these multiplications for all (batch, hidden) pairs in one launch, using
standard 2‑D tiling (BLOCK_M × BLOCK_N) and a K‑axis tile (BLOCK_K).

All remaining operations (LayerNorm, linear projections, gating, masking)
are performed with PyTorch for simplicity and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_matmul_kernel(
    left_ptr,          # *float32
    right_ptr,         # *float32
    out_ptr,           # *float32
    B, N, H,           # scalar ints
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]"""

    pid_i = tl.program_id(0)          # block row   (i)
    pid_j = tl.program_id(1)          # block col   (j)
    pid_hb = tl.program_id(2)         # packed (batch, hidden)

    # unpack batch and hidden index
    b = pid_hb // H
    d = pid_hb % H

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K
        k = k_base + offs_k
        mask_k = k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + offs_i[:, None] * stride_li
            + k[None, :] * stride_lk
            + d * stride_lh
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right block transposed to shape (BLOCK_K, BLOCK_N)
        # original layout right[b, j, k, d]; we load as (k, j)
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + k[:, None] * stride_rk
            + offs_j[None, :] * stride_ri
            + d * stride_rh
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # accumulate dot product over K
        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # write results
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
        + d * stride_oh
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim]
            mask         : torch.Tensor [B, N, N] (binary 0/1)
            weights      : dict of model parameters
            config       : dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # Load weights (move to correct device & dtype)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)

    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)

    to_out_weight = weights["to_out.weight"].to(device)

    # -----------------------------------------------------------------
    # 1) LayerNorm on input
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 2) Linear projections to hidden dimension
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)      # [B, N, N, H]
    right = F.linear(x, right_proj_weight)    # [B, N, N, H]

    # -----------------------------------------------------------------
    # 3) Optional mask (mask may be all‑ones when nomask=True)
    # -----------------------------------------------------------------
    if mask is not None:
        # mask: [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # -----------------------------------------------------------------
    # 4) Gating
    # -----------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Triangular multiplication via Triton (batched GEMM per hidden channel)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    out = torch.empty((B, N, N, H), dtype=left.dtype, device=device)

    # strides for the 4‑D tensors (contiguous layout)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Block sizes (tuned for typical B200 capabilities)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,                       # one program for each (batch, hidden) pair
    )

    _trimul_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_ri,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6) Post‑processing: LayerNorm over hidden dim, out‑gate, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # back to [B, N, N, dim]

    return out
```
