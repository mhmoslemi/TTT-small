4. Compute left/right gates (sigmoid of linear projections) and apply them.
5. Perform the triangular multiplication:
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   This is implemented as a batched GEMM between `left` and the
   transpose of `right` on the (i, j, k) axes.  A Triton kernel tiles
   the i/j dimensions (BLOCK_M × BLOCK_N) and performs a reduction over
   k in BLOCK_K chunks.  Each kernel instance processes one hidden‑channel
   slice for one batch, so the launch grid is (N/BlockM, N/BlockN, B*H).
6. Apply a second LayerNorm, an output gate, and a final linear projection
   back to the original dimension.

All tensors are on CUDA; the heavy O(N³·H) multiplication is executed
entirely in Triton.  The remaining operations stay in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # tensor pointers
    B, N, H,                               # sizes
    stride_batch_left, stride_i_left, stride_k_left,
    stride_batch_right, stride_k_right, stride_i_right,
    stride_batch_out, stride_i_out, stride_j_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # ------------------------------------------------------------------
    # program IDs
    pid_m = tl.program_id(0)   # block row   (i)
    pid_n = tl.program_id(1)   # block col   (j)
    pid_c = tl.program_id(2)   # batch * hidden channel index

    # map pid_c → batch and hidden channel
    batch = pid_c // H
    h = pid_c % H

    # base pointers for this batch & hidden channel (stride_h == 1)
    left_base  = left_ptr  + batch * stride_batch_left  + h
    right_base = right_ptr + batch * stride_batch_right + h
    out_base   = out_ptr   + batch * stride_batch_out   + h

    # start offsets for this tile
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # range vectors
    i_idx = i0 + tl.arange(0, BLOCK_M)
    j_idx = j0 + tl.arange(0, BLOCK_N)

    mask_i = i_idx < N
    mask_j = j_idx < N

    # accumulator (FP32 for accuracy)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # reduction over k
    for kp in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = kp * BLOCK_K
        k_idx = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # left block: (i, k)
        left_ptrs = left_base + i_idx[:, None] * stride_i_left + k_idx[None, :] * stride_k_left
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right block (transposed): we need right[k, j]
        # row stride = stride_k_right (original k axis)
        # col stride = stride_i_right (original i axis, which becomes column after transpose)
        right_ptrs = right_base + k_idx[:, None] * stride_k_right + j_idx[None, :] * stride_i_right
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # accumulate matrix product for this k‑tile
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_K) × (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # write back
    out_ptrs = out_base + i_idx[:, None] * stride_i_out + j_idx[None, :] * stride_j_out
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Execute the outgoing Triangle Multiplicative Update (TriMul) using a
    Triton kernel for the cubic multiplication.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be None)
            - weights: dict of torch.Tensors containing all model parameters
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # LayerNorm on the input channel dimension
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # Linear projections (float32)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (broadcasted on the hidden dim)
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)   # shape (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gates (sigmoid of linear projections)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to half for the heavy kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # allocate output buffer in half precision
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Strides (int values) for the kernel
    # left/right/out are contiguous (B,N,N,H) tensors
    stride_batch_l = left_h.stride(0)
    stride_i_l = left_h.stride(1)   # step in the i (row) axis
    stride_k_l = left_h.stride(2)   # step in the k (col) axis

    stride_batch_r = right_h.stride(0)
    stride_i_r = right_h.stride(1)  # original i axis (will be used as column after transpose)
    stride_k_r = right_h.stride(2)  # original k axis (row after transpose)

    stride_batch_o = out_h.stride(0)
    stride_i_o = out_h.stride(1)
    stride_j_o = out_h.stride(2)    # j axis (same as k stride before transpose)

    # ------------------------------------------------------------------
    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    _tri_mul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        hidden_dim,
        stride_batch_l,
        stride_i_l,
        stride_k_l,
        stride_batch_r,
        stride_k_r,
        stride_i_r,
        stride_batch_o,
        stride_i_o,
        stride_j_o,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Convert back to float32 for the remaining operations
    out = out_h.to(torch.float32)

    # to_out_norm (LayerNorm over hidden dimension)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # output gate
    out = out * out_gate

    # final linear projection back to original dim
    out = F.linear(out, weights["to_out.weight"])

    return out
