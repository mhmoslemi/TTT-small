@triton.jit
def tri_mul_kernel(
    left_ptr,  # * fp16
    right_ptr,
    out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    
    # compute block start indices
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute batch and hidden idx
    bh = pid_bh
    b = bh // H
    h = bh % H

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32) # better accumulate in fp32

    # pointers base
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offset range
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # mask if out of range
        a_i = i
        a_k = k_offset
        b_j = j
        b_k = k_offset

        # masks for bounds
        a_mask = (a_i < N) & (a_k < N)
        b_mask = (b_j < N) & (b_k < N)

        # compute offsets for A: left[b, i, k, h] -> i*stride_i + k*stride_k
        a_ptrs = left_base + a_i[:, None] * stride_left_i + a_k[None, :] * stride_left_k
        # B: right[b, j, k, h] with transposed order as B[k,j] -> k*stride_k + j*stride_j
        b_ptrs = right_base + b_k[:, None] * stride_right_k + b_j[None, :] * stride_right_j

        # load tiles
        a = tl.load(a_ptrs, mask=a_mask[:, None] & a_mask[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=b_mask[:, None] & b_mask[None, :], other=0.0)

        # dot product accumulate
        acc += tl.dot(a, b)

        # increment? Not needed since using offset loops.

    # store result
    # compute out ptr offsets: out[b,i,j,h] -> i*stride_i + j*stride_j
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    # mask store
    out_mask = (i < N)[:, None] & (j < N)[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=out_mask)

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)   # block row
    pid_j = tl.program_id(1)   # block col
    pid_bh = tl.program_id(2)  # batch*hidden

    # compute start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # compute i,j ranges
    i = i_start + tl.arange(0, BLOCK_M)
    j = j_start + tl.arange(0, BLOCK_N)

    # bounds for i,j
    i_valid = i < N
    j_valid = j < N

    # decode B and H from pid_bh
    b = pid_bh // H
    h = pid_bh % H

    # base pointers for this (b,h)
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k):
        k_start = kk * BLOCK_K
        k = k_start + tl.arange(0, BLOCK_K)
        k_valid = k < N

        # A tile: left[b,i,k,h] shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + i[:, None] * stride_left_i + k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=i_valid[:, None] & k_valid[None, :], other=0.0)

        # B tile: right[b,j,k,h] but we need transposed version B[k,j]
        b_ptrs = right_base + k[:, None] * stride_right_k + j[None, :] * stride_right_j
        b = tl.load(b_ptrs, mask=k_valid[:, None] & j_valid[None, :], other=0.0)

        # multiply-accumulate
        acc += tl.dot(a, b)

    # store result
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_valid[:, None] & j_valid[None, :])
```

Note: We used stride_right_j for col stride (j). For our pointer B we used k as first dimension and j as second. That matches required B[k, j] shape.

```python
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config['dim']
    hidden_dim = config['hidden_dim']

    # LayerNorm
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    # cast to fp16 for projections/gates
    x_fp16 = x.to(torch.float16)

    # Linear projections
    left_proj_weight = weights['left_proj.weight'].to(torch.float16)
    right_proj_weight = weights['right_proj.weight'].to(torch.float16)
    left = torch.nn.functional.linear(x_fp16, left_proj_weight)  # shape [B,N,N,H]
    right = torch.nn.functional.linear(x_fp16, right_proj_weight)

    # Gates
    left_gate_weight = weights['left_gate.weight'].to(torch.float16)
    right_gate_weight = weights['right_gate.weight'].to(torch.float16)
    out_gate_weight = weights['out_gate.weight'].to(torch.float16)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_weight))

    # Mask
    if mask is not None and mask.numel() > 0:
        mask_ = mask.unsqueeze(-1).to(torch.float16)
        left = left * left_gate * mask_
        right = right * right_gate * mask_
    else:
        left = left * left_gate
        right = right * right_gate

    # Prepare output tensor
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Get strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_k, stride_out_h = out.stride()

    # Strides for right need to map correctly: we treat j dimension as stride_right_i (original i) and k as stride_right_k.
    # In kernel we use stride_right_j for column stride (i.e., original i stride) and stride_right_k for row stride.
    # So we define:
    stride_right_j = stride_right_i  # original i stride = N*H
    # stride_right_k remains stride_right_k = H

    # Define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_k,  # NOTE: stride_out_k corresponds to original k dimension; but out shape same as left/right so we use same mapping
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Convert output to float32 for remaining ops
    out_fp32 = out.to(torch.float32)

    # to_out_norm
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    # Apply out_gate (convert to float32)
    out_gate_fp32 = out_gate.to(torch.float32)
    out_norm = out_norm * out_gate_fp32

    # Final linear
    to_out_weight = weights['to_out.weight'].to(torch.float32)
    output = torch.nn.functional.linear(out_norm, to_out_weight)

    return output

"""TriMul “outgoing” update (AlphaFold3) implemented with a fused Triton GEMM.

The routine follows the reference PyTorch implementation:
1. Layer‑norm over the last dimension.
2. Linear projections (dim → hidden) for left/right and three gates.
3. Apply mask and sigmoid gates.
4. Core O(N³) operation: for each batch and hidden channel compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiply L @ Rᵀ (L,R are N×N).
   This is performed by a custom Triton kernel that processes (batch,hidden)
   as the third grid dimension.
5. Layer‑norm over the hidden dimension, gated by out_gate, and final linear
   back to dim.

All heavy arithmetic (step 4) runs in FP16 with a FP32 accumulator,
while the surrounding layers use FP32 for numerical stability.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: out = left @ rightᵀ for each (batch, hidden) pair."""
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile
    pid_bh = tl.program_id(2)         # batch*hidden

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    i = i_start + tl.arange(0, BLOCK_M)
    j = j_start + tl.arange(0, BLOCK_N)

    i_mask = i < N
    j_mask = j < N

    # decode batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # base pointers for this (b,h)
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulator in FP32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k):
        k_start = kk * BLOCK_K
        k = k_start + tl.arange(0, BLOCK_K)
        k_mask = k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + i[:, None] * stride_left_i + k[None, :] * stride_left_k
        a = tl.load(
            a_ptrs,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0
        )

        # right tile transposed: we need B[k,j] = right[b,j,k,h]
        b_ptrs = right_base + k[:, None] * stride_right_k + j[None, :] * stride_right_j
        b = tl.load(
            b_ptrs,
            mask=k_mask[:, None] & j_mask[None, :],
            other=0.0
        )

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N) accumulation

    # write result (convert back to FP16)
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=i_mask[:, None] & j_mask[None, :]
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (CUDA)
            - mask: torch.Tensor [B, N, N] (CUDA) – may be all‑ones
            - weights: dict of model parameters (torch.Tensors on CUDA)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Layer‑norm over the feature dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Linear projections (dim → hidden) and gates
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)          # [B,N,N,H]
    right = F.linear(x_fp16, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    # -----------------------------------------------------------------
    # 3. Apply mask and elementwise gates
    # -----------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)   # [B,N,N,1]
        left = left * left_gate * mask_fp16
        right = right * right_gate * mask_fp16
    else:
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 4. Core multiplicative update via Triton GEMM
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    out_half = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Strides (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_k, stride_out_h = out_half.stride()

    # right uses swapped axes for the multiplication: we pass the stride of
    # the original 'i' dimension as stride_right_j (columns after transpose)
    stride_right_j = stride_right_i          # original i‑stride = N*H
    # stride_right_k stays the same (original k‑stride = H)

    # output uses the same layout as left/right
    stride_out_j = stride_out_k              # original k‑stride = H

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out_half,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 5. Post‑processing (LayerNorm, gating, final linear)
    # -----------------------------------------------------------------
    out_fp32 = out_half.to(torch.float32)

    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate.to(torch.float32)

    final = F.linear(out_norm, weights["to_out.weight"])
    return final
