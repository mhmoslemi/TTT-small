"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The heavy O(N³·C) contraction is performed by:
  1. Linear projections and sigmoid gates (PyTorch).
  2. A Triton kernel that fuses projection, gate and optional mask:
       out = proj * gate * mask   (mask broadcast over the hidden dim)
  3. Batched matrix multiplication (torch.bmm) over the sequence dimension
     for each hidden channel (implemented as a reshape+transpose trick).
  4. Output LayerNorm, output gate, and final linear projection (PyTorch).

All tensors stay on the GPU, the final output is float32.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: fuse a projection, its sigmoid gate and the optional mask.
# ----------------------------------------------------------------------
@triton.jit
def _fuse_proj_gate_mask_kernel(
    proj_ptr,          # float32*   projected tensor (B,N,N,D) flattened
    gate_ptr,          # float32*   corresponding gate tensor (B,N,N,D) flattened
    mask_ptr,          # float32*   mask tensor (B,N,N) flattened
    out_ptr,           # float32*   output tensor (B,N,N,D) flattened
    total_elements,    # i64        total number of elements in proj/gate (B*N*N*D)
    hidden_dim,        # i64        hidden dimension (D)
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask_offs = offs < total_elements

    # load projection and gate
    proj = tl.load(proj_ptr + offs, mask=mask_offs, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask_offs, other=0.0)

    # broadcast mask over hidden dim
    mask_idx = offs // hidden_dim          # index in mask (B,N,N)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask_offs, other=1.0)

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out, mask=mask_offs)


# ----------------------------------------------------------------------
# Entry point that will be called by the harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (binary 0/1)
        - weights      : dict of parameter tensors
        - config       : dict with keys "dim", "hidden_dim", "nomask" ...

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # unpack weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = input_tensor.contiguous()
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and gate computations (sigmoid)
    # ------------------------------------------------------------------
    # shape: [B, N, N, hidden_dim]
    left_proj  = F.linear(x_norm, left_proj_weight)   # (dim->hidden_dim)
    right_proj = F.linear(x_norm, right_proj_weight)

    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    # ------------------------------------------------------------------
    # 3. Apply mask (or all‑ones) and fuse with gate using Triton
    # ------------------------------------------------------------------
    # Ensure mask is float32 and contiguous
    if nomask:
        mask_tensor = torch.ones_like(mask, dtype=dtype, device=device)
    else:
        mask_tensor = mask.to(dtype).contiguous()

    mask_flat = mask_tensor.view(-1).contiguous()       # [B*N*N]

    total = left_proj.numel()                           # B*N*N*hidden_dim
    BLOCK = 1024                                         # good warp multiple
    grid = (triton.cdiv(total, BLOCK),)

    # Allocate fused tensors
    left_fused  = torch.empty_like(left_proj)
    right_fused = torch.empty_like(right_proj)

    # left
    _fuse_proj_gate_mask_kernel[grid](
        left_proj,
        left_gate,
        mask_flat,
        left_fused,
        total,
        hidden_dim,
        BLOCK=BLOCK,
    )
    # right
    _fuse_proj_gate_mask_kernel[grid](
        right_proj,
        right_gate,
        mask_flat,
        right_fused,
        total,
        hidden_dim,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 4. Batched matrix multiplication over the sequence dimension
    #    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, D = left_fused.shape   # D == hidden_dim

    # Convert to BF16 for higher throughput (B200 supports BF16 well)
    left_b = left_fused.to(torch.bfloat16)
    right_b = right_fused.to(torch.bfloat16)

    # reshape to (B*D, N, N)
    left_mat = left_b.permute(0, 3, 1, 2).contiguous().view(B * D, N, N)
    right_mat = right_b.permute(0, 3, 1, 2).contiguous().view(B * D, N, N)

    # batched gemm: (B*D, N, N) @ (B*D, N, N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, D)
    out = out_mat.view(B, D, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, output gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate  # apply output gate

    # final projection back to dim
    output = F.linear(out, to_out_weight)   # (hidden_dim -> dim)

    return output
