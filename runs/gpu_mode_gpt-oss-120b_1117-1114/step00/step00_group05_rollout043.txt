    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_bi, stride_bk, stride_bh,
    stride_br, stride_bj, stride_bk2, stride_bh2,
    stride_ol, stride_oi, stride_oj, stride_oh,

    i_tile = tl.program_id(1)
    j_tile = tl.program_id(2)

    i_start = i_tile * BLOCK_I
    j_start = j_tile * BLOCK_J

    # Offsets
    i_offsets = i_start + tl.arange(0, BLOCK_I)

def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_bl, stride_bi, stride_bk, stride_bh,
    stride_br, stride_bj, stride_bk_r, stride_bh_r,
    stride_ol, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    H: tl.constexpr,  # hidden dimension compile time
    ):
    b = tl.program_id(0)
    i_tile = tl.program_id(1)
    j_tile = tl.program_id(2)

    i_start = i_tile * BLOCK_I
    j_start = j_tile * BLOCK_J

    # Offsets
    i_off = i_start + tl.arange(0, BLOCK_I)
    j_off = j_start + tl.arange(0, BLOCK_J)
    h_off = tl.arange(0, H)

    i_mask = i_off < N
    j_mask = j_off < N

    i_mask = i_mask[:, None, None]  # (I,1,1)
    j_mask = j_mask[None, :, None]  # (1,J,1)

```python
import torch
import triton
import triton.language as tl

@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_bl, stride_bi, stride_bk, stride_bh,
    stride_br, stride_bj, stride_bk2, stride_bh2,
    stride_ol, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    H: tl.constexpr
):
    # program ids
    b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    i_off = i_start + tl.arange(0, BLOCK_I)
    j_off = j_start + tl.arange(0, BLOCK_J)
    h_off = tl.arange(0, H)

    # range checks
    i_valid = i_off < N
    j_valid = j_off < N

    # shape masks for loads (broadcast H dimension)
    mask_i = i_valid[:, None]   # (I,1) broadcast to (I,H)
    mask_j = j_valid[None, :]   # (1,J) broadcast to (J,H)

    # accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, H), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, N):
        # load left slice: [I, H]
        left_ptrs = left_ptr + b * stride_bl + i_off[:, None] * stride_bi + k * stride_bk + h_off[None, :] * stride_bh
        left_tile = tl.load(left_ptrs, mask=mask_i, other=0.0).to(tl.float32)  # (I, H)

        # load right slice: [J, H]
        right_ptrs = right_ptr + b * stride_br + j_off[:, None] * stride_bj + k * stride_bk2 + h_off[None, :] * stride_bh2
        right_tile = tl.load(right_ptrs, mask=mask_j, other=0.0).to(tl.float32)  # (J, H)

        # outer product and accumulate
        # left_tile: (I, H) -> (I, 1, H)
        # right_tile: (J, H) -> (1, J, H)
        acc += left_tile[:, None, :] * right_tile[None, :, :]
    # combine i and j validity for store mask (broadcast over H)
    store_mask = (i_valid[:, None, None] & j_valid[None, :, None])
    out_ptrs = out_ptr + b * stride_ol + i_off[:, None, None] * stride_oi + j_off[None, :, None] * stride_oj + h_off[None, None, :] * stride_oh
    tl.store(out_ptrs, acc.to(tl.float16), mask=store_mask)

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The heavy O(N³·H) operation
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is implemented in Triton.  The kernel processes a tile of
the (i, j) plane (BLOCK_I × BLOCK_J) and computes the full
hidden dimension H (fixed to 128 in all test configs) for each
(i, j) pair.  The input tensors are first layer‑normalized,
projected and gated using regular PyTorch operators;
the projected tensors are cast to float16 and passed to the
kernel.  The kernel accumulates in float32 for numeric stability
and writes the result back as float16.  The remaining
normalization, output gating and final linear projection are
performed again in PyTorch.

The kernel launches a 3‑D grid:
    batch × ceil(N / BLOCK_I) × ceil(N / BLOCK_J)
and handles boundary conditions with masks.
"""
import torch
import triton
import triton.language as tl


@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_bl, stride_bi, stride_bk, stride_bh,
    stride_br, stride_bj, stride_bk2, stride_bh2,
    stride_ol, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    H: tl.constexpr,               # hidden dimension (compile‑time constant, 128)
):
    """Compute out = Σ_k left[...,k,:] * right[...,k,:]."""
    b = tl.program_id(0)           # batch index
    pid_i = tl.program_id(1)       # i‑tile
    pid_j = tl.program_id(2)       # j‑tile

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    i_off = i_start + tl.arange(0, BLOCK_I)
    j_off = j_start + tl.arange(0, BLOCK_J)
    h_off = tl.arange(0, H)

    i_valid = i_off < N
    j_valid = j_off < N

    # masks for loads (broadcast over H)
    mask_i = i_valid[:, None]      # (BLOCK_I, 1) → broadcast to (BLOCK_I, H)
    mask_j = j_valid[None, :]      # (1, BLOCK_J) → broadcast to (BLOCK_J, H)

    # accumulator in float32
    acc = tl.zeros((BLOCK_I, BLOCK_J, H), dtype=tl.float32)

    # reduction over the sequence dimension k
    for k in range(0, N):
        # left[b, i, k, h]   shape (BLOCK_I, H)
        left_ptrs = (
            left_ptr
            + b * stride_bl
            + i_off[:, None] * stride_bi
            + k * stride_bk
            + h_off[None, :] * stride_bh
        )
        left_tile = tl.load(left_ptrs, mask=mask_i, other=0.0).to(tl.float32)

        # right[b, j, k, h]  shape (BLOCK_J, H)
        right_ptrs = (
            right_ptr
            + b * stride_br
            + j_off[:, None] * stride_bj
            + k * stride_bk2
            + h_off[None, :] * stride_bh2
        )
        right_tile = tl.load(right_ptrs, mask=mask_j, other=0.0).to(tl.float32)

        # outer product over (i, j) for each hidden channel
        # left_tile[:, None, :] → (BLOCK_I, 1, H)
        # right_tile[None, :, :] → (1, BLOCK_J, H)
        acc += left_tile[:, None, :] * right_tile[None, :, :]

    # store only for valid i/j positions (broadcast over H)
    store_mask = i_valid[:, None, None] & j_valid[None, :, None]
    out_ptrs = (
        out_ptr
        + b * stride_ol
        + i_off[:, None, None] * stride_oi
        + j_off[None, :, None] * stride_oj
        + h_off[None, None, :] * stride_oh
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=store_mask)

    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] float32
            - mask:          [B, N, N]        (bool or float)
            - weights: dict of torch tensors containing all model parameters
            - config:  dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        out: [B, N, N, dim] float32
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]            # always 128 in the test suite
    nomask = config.get("nomask", True)

    # ---------- LayerNorm on input ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ---------- Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ---------- Optional mask ----------
    if not nomask:
        mask_uns = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask_uns
        right = right * mask_uns

    # ---------- Gating ----------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Triton kernel for the O(N³·H) einsum ----------
    # Cast to float16 to reduce memory traffic inside the kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    B, N, _, _ = left_fp16.shape
    out_fp16 = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Tile sizes (tuned for register pressure)
    BLOCK_I = 4
    BLOCK_J = 4
    # hidden dimension is a compile‑time constant for the kernel
    H = hidden_dim

    # Strides (in elements, not bytes)
    l_stride = left_fp16.stride()
    r_stride = right_fp16.stride()
    o_stride = out_fp16.stride()

    grid = (
        B,
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
    )

    trimul_einsum_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        l_stride[0],
        l_stride[1],
        l_stride[2],
        l_stride[3],
        r_stride[0],
        r_stride[1],
        r_stride[2],
        r_stride[3],
        o_stride[0],
        o_stride[1],
        o_stride[2],
        o_stride[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        H=H,
        num_warps=4,
    )

    # ---------- Continue in PyTorch (float32) ----------
    out = out_fp16.to(torch.float32)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # output gating
    out = out * out_gate

    # final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
