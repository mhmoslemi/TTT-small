@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    # compute batch and hidden block indices
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)
    pid2 = tl.program_id(2)

    # compute batch and h block mapping
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    b = pid0 // num_h_blocks
    h_block = pid0 % num_h_blocks

    # current block start indices for i and j
    i_start = pid1 * BLOCK_I
    j_start = pid2 * BLOCK_J
    h_start = h_block * BLOCK_H

    # compute i, j, h coordinates arrays
    i_range = i_start + tl.arange(0, BLOCK_I)
    j_range = j_start + tl.arange(0, BLOCK_J)
    h_range = h_start + tl.arange(0, BLOCK_H)

    # predicates for bounds
    mask_i = i_range < N
    mask_j = j_range < N
    mask_h = h_range < H

    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # Loop over k dimension in blocks
    for k_block in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_block * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # Load left tile: shape (BLOCK_I, BLOCK_K, BLOCK_H)
        # left_ptr + b*stride_lb + i_range[:, None, None]*stride_li + k_range[None, :, None]*stride_lk + h_range[None, None, :]*stride_lh
        left_ptrs = left_ptr + b * stride_lb + i_range[:, None, None] * stride_li + k_range[None, :, None] * stride_lk + h_range[None, None, :] * stride_lh
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0)

        # Load right tile transposed: shape (BLOCK_K, BLOCK_J, BLOCK_H)
        # pointer uses k_range as first dim, j_range as second dim
        right_ptrs = right_ptr + b * stride_rb + k_range[:, None, None] * stride_rk + j_range[None, :, None] * stride_rj + h_range[None, None, :] * stride_rh
        right = tl.load(right_ptrs, mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :], other=0.0)

        # For each h in block, compute dot product
        # We'll loop over h_offset
        for h_offset in range(0, BLOCK_H):
            # check if this h is within bounds
            valid_h = mask_h[h_offset]
            if valid_h:
                left_h = left[:, :, h_offset]   # (BLOCK_I, BLOCK_K)
                right_h = right[:, :, h_offset] # (BLOCK_K, BLOCK_J)
                # compute dot
                dot = tl.dot(left_h, right_h)   # (BLOCK_I, BLOCK_J)
                acc[:, :, h_offset] += dot

    # Store result
    out_ptrs = out_ptr + b * stride_ob + i_range[:, None, None] * stride_oi + j_range[None, :, None] * stride_oj + h_range[None, None, :] * stride_oh
    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

"""
Triangular Multiplicative Update (TriMul) – outgoing version.

The forward pass consists of:
1) Layer‑norm on the input.
2) Two linear projections (left, right) and three gating linear layers.
3) Mask application and gating.
4) Core update: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   Implemented as a batched GEMM over the “k” dimension using a Triton kernel.
5) Layer‑norm on the hidden result, out‑gate multiplication,
   and final linear projection back to the original dim.

Only the O(N³·H) multiplication is performed on Triton; the surrounding
operations use PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h].

    Grid layout (program_id):
        pid0 = batch * num_h_blocks + h_block
        pid1 = i block
        pid2 = j block
    """
    pid0 = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Number of hidden‑dim blocks
    num_h_blocks = tl.cdiv(H, BLOCK_H)

    b = pid0 // num_h_blocks
    h_block = pid0 % num_h_blocks

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    h_start = h_block * BLOCK_H

    # Ranges for each dimension
    i_range = i_start + tl.arange(0, BLOCK_I)
    j_range = j_start + tl.arange(0, BLOCK_J)
    h_range = h_start + tl.arange(0, BLOCK_H)

    mask_i = i_range < N
    mask_j = j_range < N
    mask_h = h_range < H

    # Accumulator for a tile (BLOCK_I, BLOCK_J, BLOCK_H)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # Loop over k in blocks
    for k_block in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_block * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # ----- load left[b, i, k, h] -----
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + i_range[:, None, None] * stride_li
            + k_range[None, :, None] * stride_lk
            + h_range[None, None, :] * stride_lh
        )
        left_mask = mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]
        left = tl.load(left_ptrs, mask=left_mask, other=0.0)  # (I, K, H)

        # ----- load right[b, j, k, h] transposed to (K, J, H) -----
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + k_range[:, None, None] * stride_rk
            + j_range[None, :, None] * stride_rj
            + h_range[None, None, :] * stride_rh
        )
        right_mask = mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
        right = tl.load(right_ptrs, mask=right_mask, other=0.0)  # (K, J, H)

        # ----- compute dot products for each hidden slice -----
        for h_off in range(0, BLOCK_H):
            # left slice (I, K), right slice (K, J)
            left_h = left[:, :, h_off]   # (I, K)
            right_h = right[:, :, h_off] # (K, J)
            # dot -> (I, J)
            dot = tl.dot(left_h, right_h)
            acc[:, :, h_off] += dot

    # ----- write back -----
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + i_range[:, None, None] * stride_oi
        + j_range[None, :, None] * stride_oj
        + h_range[None, None, :] * stride_oh
    )
    out_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for the
    O(N³·H) multiplication.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ----- LayerNorm on input -----
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ----- Linear projections -----
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ----- Mask (broadcast to hidden) -----
    mask_exp = mask.unsqueeze(-1).to(dtype)  # [B, N, N, 1]
    left = left * mask_exp
    right = right * mask_exp

    # ----- Gates -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape

    # ----- Triton kernel for the triangular multiplication -----
    out = torch.empty_like(left)  # shape [B, N, N, hidden_dim]

    # strides (in elements)
    l_strides = left.stride()
    r_strides = right.stride()
    o_strides = out.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64
    BLOCK_H = 32  # hidden dimension tiling

    grid = (
        B * triton.cdiv(hidden_dim, BLOCK_H),   # pid0 : batch * hidden‑block
        triton.cdiv(N, BLOCK_I),                # pid1 : i‑blocks
        triton.cdiv(N, BLOCK_J),                # pid2 : j‑blocks
    )

    trimul_outgoing_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        l_strides[0], l_strides[1], l_strides[2], l_strides[3],
        r_strides[0], r_strides[1], r_strides[2], r_strides[3],
        o_strides[0], o_strides[1], o_strides[2], o_strides[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ----- Post‑processing -----
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
