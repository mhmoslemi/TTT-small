@triton.jit
def tri_mul_matmul_kernel(
    L_ptr, R_ptr, O_ptr,
    M, N, K,
    stride_L_g, stride_L_i, stride_L_k,
    stride_R_g, stride_R_k, stride_R_n,
    stride_O_g, stride_O_i, stride_O_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    gid = tl.program_id(0)   # group
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute masks for bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Pointers for L and R and O
    L_ptrs = L_ptr + gid * stride_L_g + offs_m[:, None] * stride_L_i + offs_k[None, :] * stride_L_k
    R_ptrs = R_ptr + gid * stride_R_g + offs_k[:, None] * stride_R_k + offs_n[None, :] * stride_R_n
    O_ptrs = O_ptr + gid * stride_O_g + offs_m[:, None] * stride_O_i + offs_n[None, :] * stride_O_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load tiles
        L_tile = tl.load(L_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        R_tile = tl.load(R_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product accumulate
        acc += tl.dot(L_tile, R_tile)

        # move pointers to next k block
        L_ptrs += BLOCK_K * stride_L_k
        R_ptrs += BLOCK_K * stride_R_k

    # Store result
    store_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(O_ptrs, acc, mask=store_mask)

    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # Retrieve weights as tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ensure all weights are on same device dtype
    # maybe they are already, but to be safe:
    norm_weight = norm_weight.to(device=device, dtype=dtype)
    norm_bias = norm_bias.to(device=device, dtype=dtype)
    left_proj_weight = left_proj_weight.to(device=device, dtype=dtype)
    right_proj_weight = right_proj_weight.to(device=device, dtype=dtype)
    left_gate_weight = left_gate_weight.to(device=device, dtype=dtype)
    right_gate_weight = right_gate_weight.to(device=device, dtype=dtype)
    out_gate_weight = out_gate_weight.to(device=device, dtype=dtype)
    to_out_norm_weight = to_out_norm_weight.to(device=device, dtype=dtype)
    to_out_norm_bias = to_out_norm_bias.to(device=device, dtype=dtype)
    to_out_weight = to_out_weight.to(device=device, dtype=dtype)

    # LayerNorm on last dimension
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Mask
    if mask is None or config.get("nomask", True):
        # mask is all ones: no effect
        pass
    else:
        mask_ = mask.unsqueeze(-1).to(dtype)
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Rearrange for kernel
    # Permute to (B, hidden_dim, N, N)

    stride_L_i = L.stride(1)
    stride_L_k = L.stride(2)

    stride_O_n = O.stride(2)

    # Kernel config
    BLOCK_M = 128

    BLOCK_K = 32

    grid = (G, triton.cdiv(seq_len, BLOCK_M), triton.cdiv(seq_len, BLOCK_N))

"""
TriMul (outgoing) forward pass implemented with Triton.

The heavy O(N³) part is the einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched matrix multiplication
    out[g] = L[g] @ R[g].T   with   g = b * hidden_dim + d.

We fuse this multiplication into a custom Triton kernel that
processes each (batch, hidden) group independently and uses
blocked matmul (BLOCK_M×BLOCK_N tiles, BLOCK_K inner tiles). The rest
(layer‑norms, linear projections, gating) is performed with PyTorch
operations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    L_ptr, R_ptr, O_ptr,
    M, N, K,                         # all equal to seq_len
    stride_L_g, stride_L_i, stride_L_k,
    stride_R_g, stride_R_k, stride_R_n,
    stride_O_g, stride_O_i, stride_O_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matmul for the TriMul update:
    O[g,i,j] = Σ_k L[g,i,k] * R[g,j,k]   (R is read transposed).
    """
    gid = tl.program_id(0)            # group = batch * hidden_dim
    pid_m = tl.program_id(1)          # tile over rows i
    pid_n = tl.program_id(2)          # tile over cols j

    # tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # bounds
    mask_i = offs_i < M
    mask_j = offs_j < N
    mask_k = offs_k < K

    # pointers for the current group
    L_base = L_ptr + gid * stride_L_g
    R_base = R_ptr + gid * stride_R_g
    O_base = O_ptr + gid * stride_O_g

    L_tile_ptr = L_base + offs_i[:, None] * stride_L_i + offs_k[None, :] * stride_L_k
    R_tile_ptr = R_base + offs_k[:, None] * stride_R_k + offs_j[None, :] * stride_R_n
    O_tile_ptr = O_base + offs_i[:, None] * stride_O_i + offs_j[None, :] * stride_O_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        L_tile = tl.load(L_tile_ptr,
                         mask=mask_i[:, None] & mask_k[None, :],
                         other=0.0)
        R_tile = tl.load(R_tile_ptr,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0)

        # Accumulate fp32 dot product
        acc += tl.dot(L_tile, R_tile)

        # advance k pointers
        L_tile_ptr += BLOCK_K * stride_L_k
        R_tile_ptr += BLOCK_K * stride_R_k

    # store the result
    store_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(O_tile_ptr, acc, mask=store_mask)


def custom_kernel(data):
    """
    Forward of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be all‑ones)
            - weights: dict of torch.Tensors with model parameters
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same device)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Load weights (all are bias‑free Linear layers except LayerNorms)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=dtype)
    norm_bias   = weights["norm.bias"].to(device=device, dtype=dtype)

    left_proj_weight  = weights["left_proj.weight"].to(device=device, dtype=dtype)
    right_proj_weight = weights["right_proj.weight"].to(device=device, dtype=dtype)

    left_gate_weight  = weights["left_gate.weight"].to(device=device, dtype=dtype)
    right_gate_weight = weights["right_gate.weight"].to(device=device, dtype=dtype)
    out_gate_weight   = weights["out_gate.weight"].to(device=device, dtype=dtype)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device=device, dtype=dtype)
    to_out_weight      = weights["to_out.weight"].to(device=device, dtype=dtype)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight,
                     bias=norm_bias, eps=1e-5)

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Mask (if provided)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", True):
        mask_exp = mask.unsqueeze(-1).to(dtype=dtype)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) multiplication via Triton
    # ------------------------------------------------------------------
    # reshape to (B, hidden_dim, N, N) → (G, N, N) where G = B*hidden_dim
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    G = B * hidden_dim

    L = left_perm.view(G, N, N)
    R = right_perm.view(G, N, N)
    O = torch.empty_like(L)  # will hold the matmul result

    # strides for Triton (int64)
    stride_L_g = L.stride(0)
    stride_L_i = L.stride(1)
    stride_L_k = L.stride(2)

    stride_R_g = R.stride(0)
    stride_R_k = R.stride(2)          # column stride (original k axis) = 1
    stride_R_n = R.stride(1)          # row stride (original j axis) = N

    stride_O_g = O.stride(0)
    stride_O_i = O.stride(1)
    stride_O_n = O.stride(2)

    # Triton launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        G,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _tri_mul_matmul_kernel[grid](
        L.data_ptr(),
        R.data_ptr(),
        O.data_ptr(),
        N, N, N,                       # M, N, K
        stride_L_g, stride_L_i, stride_L_k,
        stride_R_g, stride_R_k, stride_R_n,
        stride_O_g, stride_O_i, stride_O_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # reshape back to [B, N, N, hidden_dim]
    out = O.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=1e-5)
    out = out * out_gate
    output = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return output
