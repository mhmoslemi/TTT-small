"""
TriMul (outgoing) forward pass with a fused Triton reduction.
The heavy O(N³·C) “einsum” is replaced by a batched GEMM kernel that
computes, for every channel c,   out[..., i, j, c] = Σₖ left[..., i, k, c] *
right[..., j, k, c].
All other operations (LayerNorm, Linear projections, gates, final LayerNorm
and projection) are performed with PyTorch for simplicity and numerical
stability.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C = A @ B   (A: [B, M, K], B: [B, K, N])"""
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile column
    pid_b = tl.program_id(2)          # batch * channel index

    # tile coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointers to the beginning of the current block (k = 0)
    a_ptrs = A + pid_b * stride_Ab + offs_m[:, None] * stride_Am + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    b_ptrs = B + pid_b * stride_Bb + tl.arange(0, BLOCK_K)[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        mask_a = (offs_m[:, None] < M) & (cur_k[None, :] < K)
        mask_b = (cur_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)          # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)          # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)                                 # (BLOCK_M, BLOCK_N)

        # advance k‑pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # write result back (store as half‑precision)
    c_ptrs = C + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operation.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: Tensor of shape [B, N, N, dim]
            - mask: Tensor of shape [B, N, N] (may be all‑ones when nomask=True)
            - weights: dict of model weights (see problem statement)
            - config: dict containing "dim", "hidden_dim", and "nomask"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------------------------------------------------------------------
    # LayerNorm on the input (dim)
    # ---------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # mean/var over the last dimension
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias   # broadcast over B,N,N

    # ---------------------------------------------------------------------
    # Linear projections (no bias)
    # ---------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x_norm, left_proj_weight)          # [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_weight)

    # ---------------------------------------------------------------------
    # Optional mask (applied after projection, before gating)
    # ---------------------------------------------------------------------
    if not nomask:
        m = mask.unsqueeze(-1).to(left.dtype)         # [B,N,N,1]
        left = left * m
        right = right * m

    # ---------------------------------------------------------------------
    # Gating (sigmoid)
    # ---------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------------------
    # Triton kernel for the reduction:  out = left @ rightᵀ per channel
    # ---------------------------------------------------------------------
    # convert to half precision for the kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape

    # reshape to [B*hidden_dim, N, N] and prepare rightᵀ
    left_flat = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_t = right_flat.transpose(1, 2).contiguous()

    out_flat = torch.empty_like(left_flat)  # half‑precision buffer

    # block sizes (tuned for the B200 – keep them modest)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    tri_mul_matmul_kernel[grid](
        left_flat,
        right_t,
        out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_t.stride(0),   right_t.stride(1),   right_t.stride(2),
        out_flat.stride(0),  out_flat.stride(1),  out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim] and cast to fp32 for the rest
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ---------------------------------------------------------------------
    # LayerNorm on the hidden dimension
    # ---------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)

    # ---------------------------------------------------------------------
    # Apply output gate and final linear projection back to dim
    # ---------------------------------------------------------------------
    out = out * out_gate
    to_out_weight = weights["to_out.weight"]
    output = F.linear(out, to_out_weight)   # shape [B, N, N, dim]

    return output
