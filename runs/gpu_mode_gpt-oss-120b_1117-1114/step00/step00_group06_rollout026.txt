"""
TriMul forward (outgoing version) accelerated with Triton.

Algorithm
---------
1. Layer‑normalize the input tensor.
2. Apply two linear projections (left/right) and three gating linear layers.
3. Fuse mask (if provided) and per‑position gates via a tiny Triton kernel.
   The kernel multiplies each element of a tensor of shape
   [B, N, N, H] by a scalar mask (broadcasted over the hidden dimension)
   and by a gate of the same shape.
4. Compute the core triangular multiplicative update:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   This is equivalent to a batched matrix multiplication per hidden channel,
   implemented efficiently with torch.bmm after reshaping.
5. Layer‑normalize the result, apply the output gate, and a final linear
   projection back to the original feature dimension.
All tensors remain on the GPU, and the heavy element‑wise masking/gating
is off‑loaded to a custom Triton kernel.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: in‑place mask * gate multiplication
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    x_ptr,            # pointer to tensor [B,N,N,H] flattened, float32
    mask_ptr,         # pointer to mask [B,N,N] flattened, float32
    gate_ptr,         # pointer to gate [B,N,N,H] flattened, float32
    total_elements,   # total number of elements in x (B*N*N*H)
    hidden_dim: tl.constexpr,   # H (known at launch)
    BLOCK: tl.constexpr,        # block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    active = offs < total_elements

    # Load values
    x = tl.load(x_ptr + offs, mask=active, other=0.0)
    g = tl.load(gate_ptr + offs, mask=active, other=0.0)

    # Mask index = element index // hidden_dim (broadcast over H)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)

    # Apply mask and gate in‑place
    out = x * m * g
    tl.store(x_ptr + offs, out, mask=active)


# ----------------------------------------------------------------------
# Helper to launch the mask‐gate kernel
# ----------------------------------------------------------------------
def apply_mask_and_gate(tensor: torch.Tensor, mask: torch.Tensor, gate: torch.Tensor, hidden_dim: int):
    """
    In‑place multiplication: tensor = tensor * mask * gate,
    where mask is broadcasted over the hidden dimension.
    """
    assert tensor.is_contiguous()
    assert mask.is_contiguous()
    assert gate.is_contiguous()
    total = tensor.numel()
    BLOCK = 65536  # reasonable block size for 1‑D launch
    grid = (triton.cdiv(total, BLOCK),)
    mask_gate_kernel[grid](
        tensor,
        mask,
        gate,
        total,
        hidden_dim=hidden_dim,
        BLOCK=BLOCK,
    )
    return tensor


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.
    Arguments:
        data: Tuple (input_tensor, mask, weights, config)
    Returns:
        output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -------------------------
    # 1. LayerNorm on input
    # -------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -------------------------
    # 2. Linear projections
    # -------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------
    # 3. Gating values
    # -------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # Ensure mask is float32 for multiplication
    mask_f = mask.to(dtype=torch.float32)

    # -------------------------
    # 4. Fuse mask + gates via Triton (in‑place)
    # -------------------------
    # Make tensors contiguous for safe pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_f = mask_f.contiguous()

    apply_mask_and_gate(left, mask_f, left_gate, hidden_dim)
    apply_mask_and_gate(right, mask_f, right_gate, hidden_dim)

    # -------------------------
    # 5. Core TriMul: batched matrix multiplication per hidden channel
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # -------------------------
    B, N, _, _ = left.shape  # left shape: (B, N, N, H)

    # Reshape to (B*H, N, N) for left and right^T
    left_reshape = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (i, k)
    right_reshape = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)  # (j, k) -> (k, j) after permute

    # Batched GEMM: (B*H, N, N) @ (B*H, N, N) -> (B*H, N, N)
    out_reshape = torch.bmm(left_reshape, right_reshape)

    # Restore original layout: (B, N, N, H)
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # -------------------------
    # 6. Output LayerNorm, gating and final projection
    # -------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate  # element‑wise gate

    # Final linear: hidden_dim -> dim
    out = F.linear(out, weights["to_out.weight"])

    return out.to(dtype=torch.float32)
