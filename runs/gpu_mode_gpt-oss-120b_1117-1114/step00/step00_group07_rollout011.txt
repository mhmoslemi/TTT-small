    a_ptr, b_ptr, c_ptr,
    batch, N, hidden_dim,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_c,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_c,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bc = tl.program_id(2)

    # Decode batch and channel
    batch_idx = pid_bc // hidden_dim
    chan_idx = pid_bc % hidden_dim

    # compute offsets
    offs_m = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers offset by batch and channel
    a_base = a_ptr + batch_idx * stride_a_batch + chan_idx * stride_a_c
    b_base = b_ptr + batch_idx * stride_b_batch + chan_idx * stride_b_c
    c_base = c_ptr + batch_idx * stride_c_batch + chan_idx * stride_c_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):

        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        a = tl.load(a_base + offs_m[:, None] * stride_a_i + offs_k[None, :] * stride_a_k,

"""
TriMul (outgoing) forward pass implemented with Triton.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Two linear projections (left/right) and three sigmoids (gates).
3. Apply mask (if present) and multiply by gates.
4. Compute the expensive contraction
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   as a batched half‑precision GEMM using a custom Triton kernel.
5. Layer‑norm the result, multiply by the output gate,
   and apply a final linear projection back to the input dimension.

The heavy O(N³·C) reduction is offloaded to the Triton kernel,
while the remaining lightweight operations are performed with PyTorch.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _triton_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, N, hidden_dim,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_c,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_c,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched half‑precision GEMM: C = A @ Bᵀ  (A,B : [B,N,N,H])"""
    pid_i = tl.program_id(0)          # block row (i)
    pid_j = tl.program_id(1)          # block col (j)
    pid_bc = tl.program_id(2)         # combined batch & hidden channel

    # decode batch index and hidden channel index
    batch_idx = pid_bc // hidden_dim
    chan_idx = pid_bc % hidden_dim

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this batch & channel
    a_base = a_ptr + batch_idx * stride_a_batch + chan_idx * stride_a_c
    b_base = b_ptr + batch_idx * stride_b_batch + chan_idx * stride_b_c
    c_base = c_ptr + batch_idx * stride_c_batch + chan_idx * stride_c_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile (M x K) – left[b,i,k,chan]
        a = tl.load(
            a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # B tile transposed (K x N) – right[b,j,k,chan]ᵀ
        b = tl.load(
            b_base + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_j,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # accumulate partial product
        acc += tl.dot(a, b)   # (M,K) @ (K,N) -> (M,N)

    # write the block of C
    tl.store(
        c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: (input, mask, weights, config)
            - input: Tensor[B, N, N, dim] (float32, CUDA)
            - mask:  Tensor[B, N, N] (any dtype, CUDA)
            - weights: dict of model weights
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        Tensor[B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # move/convert all weight tensors to the correct device & dtype
    # -----------------------------------------------------------------
    def _to(w):
        return w.to(device=device, dtype=torch.float32)

    norm_weight = _to(weights["norm.weight"])
    norm_bias = _to(weights["norm.bias"])
    left_proj_weight = _to(weights["left_proj.weight"])
    right_proj_weight = _to(weights["right_proj.weight"])
    left_gate_weight = _to(weights["left_gate.weight"])
    right_gate_weight = _to(weights["right_gate.weight"])
    out_gate_weight = _to(weights["out_gate.weight"])
    to_out_norm_weight = _to(weights["to_out_norm.weight"])
    to_out_norm_bias = _to(weights["to_out_norm.bias"])
    to_out_weight = _to(weights["to_out.weight"])

    # -----------------------------------------------------------------
    # 1. Input layer‑norm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 2. Projections
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # 3. Mask (if any) – broadcast on the hidden dimension
    # -----------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=torch.float32).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gating
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Half‑precision GEMM using Triton (the O(N³·H) part)
    # -----------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    out = torch.empty(B, N, N, hidden_dim, dtype=torch.float32, device=device)

    # strides (in elements) for the tensors
    stride_a = left_h.stride()   # (B, i, k, c)
    stride_b = right_h.stride()
    stride_c = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    _triton_matmul_kernel[grid](
        left_h,
        right_h,
        out,
        B,
        N,
        hidden_dim,
        stride_a[0],
        stride_a[1],
        stride_a[2],
        stride_a[3],
        stride_b[0],
        stride_b[1],
        stride_b[2],
        stride_b[3],
        stride_c[0],
        stride_c[1],
        stride_c[2],
        stride_c[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Output layer‑norm + output gate
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate

    # -----------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    # -----------------------------------------------------------------
    output = torch.nn.functional.linear(out, to_out_weight)   # [B,N,N,dim]
    return output
