@triton.jit
def fuse_kernel(a_ptr, b_ptr, c_ptr, out_ptr,
                B, N, H,
                stride_a_b, stride_a_i, stride_a_j, stride_a_h,
                stride_b_b, stride_b_i, stride_b_j, stride_b_h,
                stride_c_b, stride_c_i, stride_c_j,
                stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                BLOCK_B: tl.constexpr, BLOCK_I: tl.constexpr,
                BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    # compute offsets for each dimension
    offs_b = pid_b * BLOCK_B + tl.arange(0, BLOCK_B)
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    # loop over h
    for off_h in range(0, H, BLOCK_H):
        offs_h = off_h + tl.arange(0, BLOCK_H)
        # compute masks for out-of-bounds
        mask_b = offs_b < B
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_h = offs_h < H
        mask = mask_b[:, None, None, None] & mask_i[None, :, None, None] & mask_j[None, None, :, None] & mask_h[None, None, None, :]
        # compute pointers
        a = tl.load(a_ptr + offs_b[:, None, None, None] * stride_a_b
                    + offs_i[None, :, None, None] * stride_a_i
                    + offs_j[None, None, :, None] * stride_a_j
                    + offs_h[None, None, None, :] * stride_a_h,
                    mask=mask, other=0.0)
        b = tl.load(b_ptr + offs_b[:, None, None, None] * stride_b_b

                       stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

"""
TriMul (outgoing) forward implementation.

The algorithm follows the reference PyTorch module:
1. Layer‑norm the input.
2. Linear projections: left, right.
3. Linear gates (sigmoid) for left, right and the output.
4. Fuse mask and gates with a custom Triton kernel:
   left_fused = left_proj * left_gate * mask
   right_fused = right_proj * right_gate * mask
5. Compute the multiplicative update:
   out[b,i,j,d] = sum_k left_fused[b,i,k,d] * right_fused[b,j,k,d]
   which is a batched GEMM left @ right^T for every hidden dimension.
6. Layer‑norm the result, apply the output gate and a final linear.
All operations are performed in float32; the heavy element‑wise
fusing (mask + gates) is accelerated with Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _fuse_triple_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_B: tl.constexpr, BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr
):
    """Fuse projection, gate and mask:
       out = proj * gate * mask  (mask broadcast over H).
    """
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_b = pid_b * BLOCK_B + tl.arange(0, BLOCK_B)
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # iterate over the hidden dimension
    for off_h in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = off_h * BLOCK_H + tl.arange(0, BLOCK_H)

        # 4‑D mask for loads/stores
        mask_b = offs_b < B
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_h = offs_h < H
        mask4d = (mask_b[:, None, None, None] &
                  mask_i[None, :, None, None] &
                  mask_j[None, None, :, None] &
                  mask_h[None, None, None, :])

        # load projection and gate
        proj = tl.load(
            proj_ptr + offs_b[:, None, None, None] * stride_proj_b
                    + offs_i[None, :, None, None] * stride_proj_i
                    + offs_j[None, None, :, None] * stride_proj_j
                    + offs_h[None, None, None, :] * stride_proj_h,
            mask=mask4d, other=0.0)

        gate = tl.load(
            gate_ptr + offs_b[:, None, None, None] * stride_gate_b
                    + offs_i[None, :, None, None] * stride_gate_i
                    + offs_j[None, None, :, None] * stride_gate_j
                    + offs_h[None, None, None, :] * stride_gate_h,
            mask=mask4d, other=0.0)

        # load 3‑D mask and broadcast to H
        mask3d = tl.load(
            mask_ptr + offs_b[:, None, None] * stride_mask_b
                    + offs_i[None, :, None] * stride_mask_i
                    + offs_j[None, None, :] * stride_mask_j,
            mask=mask_b[:, None, None] & mask_i[None, :, None] & mask_j[None, None, :],
            other=0.0)

        mask4d_val = mask3d[:, :, :, None]  # (B, I, J, 1)

        # element‑wise fusion
        out = proj * gate * mask4d_val

        tl.store(
            out_ptr + offs_b[:, None, None, None] * stride_out_b
                    + offs_i[None, :, None, None] * stride_out_i
                    + offs_j[None, None, :, None] * stride_out_j
                    + offs_h[None, None, None, :] * stride_out_h,
            out, mask=mask4d)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with a Triton‑accelerated
    mask‑gate fusion.
    Args:
        data: tuple (input, mask, weights, config)
            - input: torch.Tensor [B, N, N, C]
            - mask: torch.Tensor [B, N, N] (float or bool)
            - weights: dict of weight tensors
            - config: dict containing "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor of shape [B, N, N, C]
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim = C)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias = weights["norm.bias"].to(dtype)

    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_weight + norm_bias   # broadcast over B,N,N

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"].to(dtype))
    right_proj = F.linear(x, weights["right_proj.weight"].to(dtype))

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))

    # ------------------------------------------------------------------
    # 4. Mask handling
    # ------------------------------------------------------------------
    if config.get("nomask", False):
        mask_f = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask_f = mask.to(dtype)

    # ------------------------------------------------------------------
    # 5. Fuse mask and gates with Triton
    # ------------------------------------------------------------------
    left_fused = torch.empty_like(left_proj)
    right_fused = torch.empty_like(right_proj)

    # strides for the tensors
    lp_s = left_proj.stride()    # (B, N, N, H)
    lg_s = left_gate.stride()
    rp_s = right_proj.stride()
    rg_s = right_gate.stride()
    m_s = mask_f.stride()
    lo_s = left_fused.stride()
    ro_s = right_fused.stride()

    # tiling parameters (tuned for typical sizes)
    BLOCK_B = 1
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_H = 32

    grid = (
        triton.cdiv(B, BLOCK_B),
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
    )

    # left side
    _fuse_triple_kernel[grid](
        left_proj, left_gate, mask_f, left_fused,
        B, N, hidden_dim,
        lp_s[0], lp_s[1], lp_s[2], lp_s[3],
        lg_s[0], lg_s[1], lg_s[2], lg_s[3],
        m_s[0], m_s[1], m_s[2],
        lo_s[0], lo_s[1], lo_s[2], lo_s[3],
        BLOCK_B=BLOCK_B, BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J, BLOCK_H=BLOCK_H,
    )

    # right side
    _fuse_triple_kernel[grid](
        right_proj, right_gate, mask_f, right_fused,
        B, N, hidden_dim,
        rp_s[0], rp_s[1], rp_s[2], rp_s[3],
        rg_s[0], rg_s[1], rg_s[2], rg_s[3],
        m_s[0], m_s[1], m_s[2],
        ro_s[0], ro_s[1], ro_s[2], ro_s[3],
        BLOCK_B=BLOCK_B, BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J, BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 6. Batched matrix multiplication: out[b,i,j,d] = sum_k left[b,i,k,d]*right[b,j,k,d]
    #    Equivalent to (B, H, N, N) @ (B, H, N, N)^T
    # ------------------------------------------------------------------
    # bring hidden dimension to the second axis for torch.bmm
    left_perm = left_fused.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_fused.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # batched GEMM (B*H, N, N) @ (B*H, N, N)^T
    out_perm = torch.matmul(left_perm, right_perm.transpose(-1, -2))  # (B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(dtype)

    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean_o) * torch.rsqrt(var_o + eps)
    out = out * to_out_norm_weight + to_out_norm_bias

    # ------------------------------------------------------------------
    # 8. Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"].to(dtype))

    return output
```
