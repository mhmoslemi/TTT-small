@triton.jit
def batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_k, stride_right_j,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)  # tile row index (i)
    pid_j = tl.program_id(1)  # tile col index (j)
    pid_bd = tl.program_id(2) # batch+head
    # Compute batch and hidden (d) from pid_bd
    b = pid_bd // H
    d = pid_bd % H
    # Compute offsets
    # Offsets for left: (b,d,i_start,k)
    # Offsets for right: (b,d,k,j_start)
    # Offsets for out: (b,d,i_start,j_start)

    # Compute i,j ranges
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask to avoid out-of-bounds
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load left tile
        left_ptrs = left_ptr + b*stride_left_b + d*stride_left_h + i_offset[:, None]*stride_left_i + k_offset[None, :]*stride_left_k
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right tile
        right_ptrs = right_ptr + b*stride_right_b + d*stride_right_h + k_offset[:, None]*stride_right_k + j_offset[None, :]*stride_right_j
        # shape (BLOCK_K, BLOCK_N)
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(left_tile, right_tile)

    # Store result
    out_ptrs = out_ptr + b*stride_out_b + d*stride_out_h + i_offset[:, None]*stride_out_i + j_offset[None, :]*stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) kernel.

The heavy O(N³) part of the TriMul forward pass is the Einstein summation:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where left/right are the projected and gated pairwise tensors.
We fuse this computation with a custom Triton kernel that performs a batched
matrix multiplication over the batch and hidden dimensions:
    (B, H, N, N) @ (B, H, N, N) → (B, H, N, N)
with the second operand transposed on the last two axes.
All other operations (layer‑norms, linear projections, gating, masking) are
executed in PyTorch. The kernel works in FP16 for the multiply‑accumulate,
accumulates in FP32 and writes the result back in FP16, after which the
tensor is cast to FP32 for the remaining post‑processing.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for batched GEMM: (B,H,N,N) @ (B,H,N,N) → (B,H,N,N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_k, stride_right_j,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_i = tl.program_id(0)         # tile row
    pid_j = tl.program_id(1)         # tile col
    pid_bd = tl.program_id(2)        # combined batch * head id

    # decode batch and head from pid_bd
    b = pid_bd // H
    d = pid_bd % H

    # offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for left: shape (BLOCK_M, BLOCK_K)
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + d * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # pointers for right: shape (BLOCK_K, BLOCK_N)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + d * stride_right_h
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j
        )
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(left, right)

    # store result (convert to FP16)
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + d * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operation.
    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, C] torch.float32
            - mask_tensor : [B, N, N]  (bool / float, broadcastable)
            - weights_dict: mapping of weight names to tensors (already on device)
            - config_dict : contains "dim", "hidden_dim", "nomask" (bool) etc.
    Returns:
        output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # unpack weights (they are already on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input layer‑norm (over dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)    # B,N,N,hidden
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Gating values (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 4. Mask (if present)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool; convert to same dtype as x
        mask_f = mask.to(dtype).unsqueeze(-1)  # B,N,N,1
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Cast to FP16 before the heavy O(N³) compute
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Permute to (B, H, N, N) and make contiguous for the kernel
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N)
    # Right needs its second N dimension (k) first, then j
    right_perm = right_h.permute(0, 3, 2, 1).contiguous()    # (B, H, N, N) (k, j)

    B, H, N, _ = left_perm.shape

    # ------------------------------------------------------------------
    # 7. Allocate output buffer (FP16)
    # ------------------------------------------------------------------
    out_perm = torch.empty_like(left_perm, dtype=torch.float16, device=device)

    # Strides (in elements)
    stride_l_b, stride_l_h, stride_l_i, stride_l_k = left_perm.stride()
    stride_r_b, stride_r_h, stride_r_k, stride_r_j = right_perm.stride()
    stride_o_b, stride_o_h, stride_o_i, stride_o_j = out_perm.stride()

    # ------------------------------------------------------------------
    # 8. Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    grid_bd = B * H

    _batched_matmul_kernel[(grid_i, grid_j, grid_bd)](
        left_perm,
        right_perm,
        out_perm,
        B,
        N,
        H,
        stride_l_b,
        stride_l_h,
        stride_l_i,
        stride_l_k,
        stride_r_b,
        stride_r_h,
        stride_r_k,
        stride_r_j,
        stride_o_b,
        stride_o_h,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 9. Restore original ordering: (B, N, N, H) and cast back to FP32
    # ------------------------------------------------------------------
    out = out_perm.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 10. Output layer‑norm, gating and final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)  # (B,N,N,dim)

    return out
