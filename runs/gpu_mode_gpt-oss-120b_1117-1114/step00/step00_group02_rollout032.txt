@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_i = tl.program_id(0)   # i tile
    pid_j = tl.program_id(1)   # j tile
    pid_z = tl.program_id(2)   # combined batch and hidden block

    # compute batch and hidden block offsets
    hidden_blocks = tl.cdiv(H, BLOCK_D)
    b = pid_z // hidden_blocks
    hb = pid_z % hidden_blocks   # hidden block index

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    d0 = hb * BLOCK_D

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # calculate offsets for i and j ranges
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    offs_d = d0 + tl.arange(0, BLOCK_D)

    # mask for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N
    mask_d = offs_d < H

    # iterate over k dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_iter * BLOCK_K
        # update offsets for current block
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left block: shape (M, K, D)
        # left_ptr + b*stride_left_batch + offs_i[:,None,None]*stride_left_i + offs_k[None,:,None]*stride_left_k + offs_d[None,None,:]*stride_left_d

                               + offs_i[:, None, None] * stride_left_i \
                               + offs_k[None, :, None] * stride_left_k \
                               + offs_d[None, None, :] * stride_left_d

                                + offs_k[:, None, None] * stride_right_k \
                                + offs_j[None, :, None] * stride_right_i \
                                + offs_d[None, None, :] * stride_right_d
        right = tl.load(right_ptrs, mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :], other=0.0).to(tl.float32)

        # Compute multiplication over k dimension using batch dot
        # left: (M, K, D), right: (K, N, D)
        # For each d, compute tl.dot of left[:,:,d] and right[:,:,d]
        for d in range(0, BLOCK_D):

            left_slice = left[:, :, d]   # (M, K)
            right_slice = right[:, :, d] # (K, N)
            # dot
            acc[:, :, d] += tl.dot(left_slice, right_slice)
    # store acc to out
    # out pointer: shape (M,N,D)
    out_ptrs = out_ptr + b * stride_out_batch \
                        + offs_i[:, None, None] * stride_out_i \
                        + offs_j[None, :, None] * stride_out_j \
                        + offs_d[None, None, :] * stride_out_d
    # mask for storing

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # Load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm on input
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [bs,N,N,hd]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # mask
    mask_exp = mask.unsqueeze(-1)   # shape [bs,N,N,1] broadcast
    left = left * mask_exp
    right = right * mask_exp

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to float16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Allocate output for kernel
    out_fp16 = torch.empty_like(left_fp16)  # shape [bs,N,N,hd]

    # Setup kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_D = 32

    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    hidden_blocks = triton.cdiv(hidden_dim, BLOCK_D)
    grid_z = bs * hidden_blocks
    grid = (grid_i, grid_j, grid_z)

    # Compute strides
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d = left_fp16.stride()
    stride_right_batch, stride_right_i, stride_right_k, stride_right_d = right_fp16.stride()
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d = out_fp16.stride()

    # Launch kernel
    tri_mul_kernel[grid](
        left_fp16, right_fp16, out_fp16,
        bs, N, hidden_dim,
        stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
        stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
        stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D
    )

    # Convert output to float32
    out = out_fp16.to(torch.float32)

    # Apply to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply out_gate and final linear
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)

    return out
```

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Compute out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    using blocked matrix multiplication per hidden channel block.
    """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    # split pid_z into batch index and hidden-block index
    hidden_blocks = tl.cdiv(H, BLOCK_D)
    b = pid_z // hidden_blocks
    hb = pid_z % hidden_blocks

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    d0 = hb * BLOCK_D

    # Offsets for each dimension
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_d = d0 + tl.arange(0, BLOCK_D)

    # Masks for boundary conditions
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < H

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # Loop over k dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k0 = kb * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left: shape (M, K, D)
        left_ptrs = left_ptr \
            + b * stride_left_batch \
            + offs_i[:, None, None] * stride_left_i \
            + offs_k[None, :, None] * stride_left_k \
            + offs_d[None, None, :] * stride_left_d
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                       other=0.0).to(tl.float32)

        # Load right transposed: shape (K, N, D) where K dimension first
        right_ptrs = right_ptr \
            + b * stride_right_batch \
            + offs_k[:, None, None] * stride_right_k \
            + offs_j[None, :, None] * stride_right_i \
            + offs_d[None, None, :] * stride_right_d
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
                        other=0.0).to(tl.float32)

        # For every hidden sub‑channel, perform a small matmul
        for d in range(0, BLOCK_D):
            # left slice (M, K)
            a = left[:, :, d]
            # right slice (K, N)
            bmat = right[:, :, d]
            acc[:, :, d] += tl.dot(a, bmat)

    # Store the result
    out_ptrs = out_ptr \
        + b * stride_out_batch \
        + offs_i[:, None, None] * stride_out_i \
        + offs_j[None, :, None] * stride_out_j \
        + offs_d[None, None, :] * stride_out_d
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])

def custom_kernel(data):
    """
    Triton‑accelerated implementation of the “outgoing” Triangle Multiplicative
    Update (TriMul) from AlphaFold3.  All linear projections, layer‑norms and
    gating are performed in PyTorch; the expensive contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d]·right[b,j,k,d] is evaluated in a fused
    Triton kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # fetch weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ---------- Phase 1: pre‑processing in PyTorch ----------
    # LayerNorm over last dimension
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hd]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if present
    mask_exp = mask.unsqueeze(-1)  # [B,N,N,1]
    left = left * mask_exp
    right = right * mask_exp

    # Gating
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to half‑precision for the kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Allocate output buffer
    out_fp16 = torch.empty_like(left_fp16)   # shape [B,N,N,hd]

    # ---------- Phase 2: Triton kernel ----------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_D = 32

    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    hidden_blocks = triton.cdiv(hidden_dim, BLOCK_D)
    grid_z = bs * hidden_blocks

    # stride arguments (in elements, not bytes)
    s_lb, s_li, s_lk, s_ld = left_fp16.stride()
    s_rb, s_ri, s_rk, s_rd = right_fp16.stride()
    s_ob, s_oi, s_oj, s_od = out_fp16.stride()

    tri_mul_kernel[ (grid_i, grid_j, grid_z) ](
        left_fp16,
        right_fp16,
        out_fp16,
        bs,
        N,
        hidden_dim,
        s_lb,
        s_li,
        s_lk,
        s_ld,
        s_rb,
        s_ri,
        s_rk,
        s_rd,
        s_ob,
        s_oi,
        s_oj,
        s_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ---------- Phase 3: post‑processing ----------
    out = out_fp16.to(torch.float32)

    # LayerNorm on hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # Apply output gate and final linear projection
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

):
    """Compute out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]."""
    pid_i = tl.program_id(0)          # tile index over i
    pid_j = tl.program_id(1)          # tile index over j
    pid_z = tl.program_id(2)          # combined batch & hidden‑block index

    # Split pid_z into batch index and hidden‑block index
    hidden_blocks = tl.cdiv(H, BLOCK_D)
    b = pid_z // hidden_blocks
    hb = pid_z % hidden_blocks

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    d0 = hb * BLOCK_D

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_d = d0 + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < H

    # accumulator (float32) for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k0 = kb * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- left slice : (M, K, D) -------------------------------------------------
        left_ptrs = left_ptr + b * stride_left_batch \
                    + offs_i[:, None, None] * stride_left_i \
                    + offs_k[None, :, None] * stride_left_k \
                    + offs_d[None, None, :] * stride_left_d
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- right slice (transposed) : (K, N, D) ----------------------------------
        right_ptrs = right_ptr + b * stride_right_batch \
                     + offs_k[:, None, None] * stride_right_k \
                     + offs_j[None, :, None] * stride_right_i \
                     + offs_d[None, None, :] * stride_right_d
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- per‑channel matmul ----------------------------------------------------
        for d in range(0, BLOCK_D):
            a = left[:, :, d]      # (M, K)
            bmat = right[:, :, d]  # (K, N)
            acc[:, :, d] += tl.dot(a, bmat)

    # ----- write back ---------------------------------------------------------------
    out_ptrs = out_ptr + b * stride_out_batch \
               + offs_i[:, None, None] * stride_out_i \
               + offs_j[None, :, None] * stride_out_j \
               + offs_d[None, None, :] * stride_out_d
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )


def custom_kernel(data):
    """
    Triton‑based forward pass for the outgoing TriMul operation.

    Args:
        data: Tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim]
            - mask  : torch.Tensor [B, N, N] (binary, broadcastable)
            - weights: dict mapping weight names to tensors
            - config : dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ----- unpack weights ---------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ----- Phase 1: PyTorch preprocessing ---------------------------------------
    # LayerNorm over the feature dimension
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask (mask may be all‑ones)
    mask_exp = mask.unsqueeze(-1).to(left.dtype)
    left = left * mask_exp
    right = right * mask_exp

    # Gating
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to half‑precision for the Triton kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Output buffer for the kernel
    out_fp16 = torch.empty_like(left_fp16)   # [B, N, N, H]

    # ----- Phase 2: Triton kernel launch ---------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_D = 32

    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    hidden_blocks = triton.cdiv(hidden_dim, BLOCK_D)
    grid_z = bs * hidden_blocks
    grid = (grid_i, grid_j, grid_z)

    # Strides in element units (not bytes)
    s_lb, s_li, s_lk, s_ld = left_fp16.stride()
    s_rb, s_ri, s_rk, s_rd = right_fp16.stride()
    s_ob, s_oi, s_oj, s_od = out_fp16.stride()

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        bs,
        N,
        hidden_dim,
        s_lb,
        s_li,
        s_lk,
        s_ld,
        s_rb,
        s_ri,
        s_rk,
        s_rd,
        s_ob,
        s_oi,
        s_oj,
        s_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ----- Phase 3: PyTorch postprocessing ---------------------------------------
    out = out_fp16.to(torch.float32)

    # Hidden‑dim layer‑norm
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias
    )

    # Apply output gate and final linear projection
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)

    return out
