def mask_gate_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                     total_elements, H, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK
    offs = block_start + tl.arange(0, BLOCK)
    mask_offs = offs // H
    mask_condition = offs < total_elements
    proj = tl.load(proj_ptr + offs, mask=mask_condition, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask_condition, other=0.0)
    mask = tl.load(mask_ptr + mask_offs, mask=mask_condition, other=0.0)
    out = proj * mask * gate
    tl.store(out_ptr + offs, out, mask=mask_condition)
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mask_gate_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                     total, H,
                     BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK
    offs = block_start + tl.arange(0, BLOCK)
    mask_valid = offs < total
    proj = tl.load(proj_ptr + offs, mask=mask_valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask_valid, other=0.0)
    mask_idx = offs // H
    mask = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=0.0)
    out = proj * mask * gate
    tl.store(out_ptr + offs, out, mask=mask_valid)

def apply_mask_and_gate(proj, gate, mask):
    # proj, gate: [B,N,N,H]; mask: [B,N,N,1] broadcastable
    B, N, N2, H = proj.shape
    assert N == N2
    total = proj.numel()
    proj_flat = proj.view(-1)
    gate_flat = gate.view(-1)
    mask_flat = mask.squeeze(-1).contiguous().view(-1)  # shape B*N*N
    out = torch.empty_like(proj)
    out_flat = out.view(-1)
    BLOCK = 1024  # threads per block (must be multiple of warp size 32)
    grid = (triton.cdiv(total, BLOCK),)
    mask_gate_kernel[grid](proj_flat, gate_flat, mask_flat, out_flat,
                           total, H, BLOCK=BLOCK, num_warps=4)
    return out

def custom_kernel(data):
    """
    Triton-accelerated implementation of the outgoing TriMul operator.
    Performs LayerNorm, linear projections, gating, mask application (via a Triton kernel),
    and the core N^3 multiplicative update using a batched matrix multiplication.
    Returns a tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # Load weights
    norm_weight = weights["norm.weight"].to(device, dtype=torch.float32)
    norm_bias = weights["norm.bias"].to(device, dtype=torch.float32)
    left_proj_weight = weights["left_proj.weight"].to(device, dtype=torch.float32)
    right_proj_weight = weights["right_proj.weight"].to(device, dtype=torch.float32)
    left_gate_weight = weights["left_gate.weight"].to(device, dtype=torch.float32)
    right_gate_weight = weights["right_gate.weight"].to(device, dtype=torch.float32)
    out_gate_weight = weights["out_gate.weight"].to(device, dtype=torch.float32)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device, dtype=torch.float32)
    to_out_weight = weights["to_out.weight"].to(device, dtype=torch.float32)

    # 1. LayerNorm over dim
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Ensure contiguous for linear ops
    x = x.contiguous()

    # 2. Linear projections (no bias)
    left_proj = F.linear(x, left_proj_weight)       # [B,N,N,hidden_dim]
    right_proj = F.linear(x, right_proj_weight)     # [B,N,N,hidden_dim]

    # 3. Linear gates + sigmoid
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # 4. Apply mask and gates using Triton kernel
    # mask shape: [B,N,N]; broadcast to [B,N,N,1]
    mask_f = mask.to(dtype=torch.float32).unsqueeze(-1)  # [B,N,N,1]
    left = apply_mask_and_gate(left_proj, left_gate, mask_f)
    right = apply_mask_and_gate(right_proj, right_gate, mask_f)

    # 5. Core multiplicative update: out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    # Use batched matrix multiplication via torch.matmul
    # reshape to (B, hidden_dim, N, N) with appropriate transposes
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) where last two dims are (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) where last two dims are (k,j)
    out_perm = torch.matmul(left_perm, right_perm)      # (B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()    # (B, N, N, H)

    # 6. Output layer norm over hidden_dim
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # 7. Multiply by out_gate
    out = out * out_gate

    # 8. Final linear projection back to dim
    out = F.linear(out, to_out_weight)  # (B, N, N, dim)

    # Ensure output dtype is float32 as required
    out = out.to(torch.float32)

    return out
```

"""
TriMul (outgoing) implementation with a Triton‐accelerated mask+gate fusion.
The heavy N^3 reduction is performed via a batched matrix multiplication
(using PyTorch’s highly‑optimized GEMM).  The mask application and gating are
fused in a custom Triton kernel to satisfy the “part of the operations in a
kernel” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mask_gate_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                     total, hidden_dim,
                     BLOCK: tl.constexpr):
    """
    out = proj * mask * gate   (mask broadcasted over the hidden dimension)
    proj, gate : [total] float32 (flattened [B, N, N, H])
    mask       : [total // H] float32 (flattened [B, N, N])
    """
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask_ok = offs < total

    # Load projected tensor and gate
    proj = tl.load(proj_ptr + offs, mask=mask_ok, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask_ok, other=0.0)

    # Compute mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim
    # Guard against out‑of‑bounds mask accesses for padded threads
    mask_idx = tl.where(mask_ok, mask_idx, 0)
    mask = tl.load(mask_ptr + mask_idx, mask=mask_ok, other=0.0)

    out = proj * mask * gate
    tl.store(out_ptr + offs, out, mask=mask_ok)


def apply_mask_and_gate(proj: torch.Tensor,
                        gate: torch.Tensor,
                        mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse ``proj * mask * gate`` with a Triton kernel.
    All tensors are expected on the same CUDA device.
    proj, gate : [B, N, N, H]   (float32)
    mask       : [B, N, N, 1]   (float32, broadcastable)
    Returns a tensor of shape [B, N, N, H].
    """
    B, N, N2, H = proj.shape
    assert N == N2, "Tensor must be square in the two middle dimensions"

    total = proj.numel()                     # B * N * N * H
    proj_f = proj.contiguous().view(-1)
    gate_f = gate.contiguous().view(-1)
    mask_f = mask.squeeze(-1).contiguous().view(-1)  # [B*N*N]

    out = torch.empty_like(proj)
    out_f = out.view(-1)

    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    mask_gate_kernel[grid](
        proj_f,
        gate_f,
        mask_f,
        out_f,
        total,
        H,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul.
    Args:
        data = (input_tensor, mask, weights, config)
            input_tensor : [B, N, N, dim]   (float32 CUDA tensor)
            mask         : [B, N, N]        (bool/float tensor, broadcastable)
            weights      : dict of torch tensors (model parameters)
            config       : dict with keys "dim" and "hidden_dim"
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ---- Load weights onto the correct device & dtype ---------------------------------
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias = weights["norm.bias"].to(device, dtype)

    left_proj_weight = weights["left_proj.weight"].to(device, dtype)
    right_proj_weight = weights["right_proj.weight"].to(device, dtype)

    left_gate_weight = weights["left_gate.weight"].to(device, dtype)
    right_gate_weight = weights["right_gate.weight"].to(device, dtype)
    out_gate_weight = weights["out_gate.weight"].to(device, dtype)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device, dtype)

    to_out_weight = weights["to_out.weight"].to(device, dtype)

    # ---- 1. Input LayerNorm ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    x = x.contiguous()

    # ---- 2. Linear projections (no bias) -------------------------------------------
    left_proj = F.linear(x, left_proj_weight)      # [B, N, N, hidden_dim]
    right_proj = F.linear(x, right_proj_weight)    # [B, N, N, hidden_dim]

    # ---- 3. Gate projections + sigmoid ---------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ---- 4. Apply mask and gate via Triton -----------------------------------------
    mask_f = mask.to(dtype).unsqueeze(-1)                # [B, N, N, 1]
    left = apply_mask_and_gate(left_proj, left_gate, mask_f)
    right = apply_mask_and_gate(right_proj, right_gate, mask_f)

    # ---- 5. Core N³ multiplicative update (batched GEMM) -------------------------
    # left : [B, N, N, H]  -> (B, H, N, N) where last two dims are (i, k)
    # right: [B, N, N, H]  -> (B, H, N, N) where last two dims are (k, j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)
    out_perm = torch.matmul(left_perm, right_perm)      # (B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()    # (B, N, N, H)

    # ---- 6. Output LayerNorm over hidden dimension ---------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # ---- 7. Apply output gate -------------------------------------------------------
    out = out * out_gate

    # ---- 8. Final linear back to original dimension ---------------------------------
    out = F.linear(out, to_out_weight)   # (B, N, N, dim)

    return out.to(torch.float32)
