"""
TriMul forward pass using Triton for the cubic O(N³) reduction.
The heavy part (einsum over the sequence dimension) is implemented as a
batched matrix multiplication kernel:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where left/right are obtained from linear projections, gating and masking.
All other operations (LayerNorm, Linear, sigmoid) are performed with
torch.nn.functional for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_az, stride_am, stride_ak,   # A: (batch, M, K)  -> (z, i, k)
    stride_bz, stride_bk, stride_bn,   # B: (batch, K, N)  -> (z, k, n) (transposed view)
    stride_cz, stride_cm, stride_cn,   # C: (batch, M, N)
    M, N, K,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)          # batch*hidden index
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block column

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the current batch slice
    a_batch_ptr = A_ptr + pid_z * stride_az
    b_batch_ptr = B_ptr + pid_z * stride_bz

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a = tl.load(
            a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (K x N) – note B is stored transposed, so stride_bk corresponds to k-step
        b = tl.load(
            b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # Write back result
    c_ptrs = C_ptr + pid_z * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [bs, seq_len, seq_len, dim]
            - mask: torch.Tensor [bs, seq_len, seq_len] (float or bool)
            - weights: dict of torch.Tensors containing all model parameters
            - config: dict with keys "dim", "hidden_dim", "nomask", ...
    Returns:
        torch.Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -------------------------------------------------------------------------
    # 1. LayerNorm on the input
    # -------------------------------------------------------------------------
    eps = 1e-5
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(dtype)

    # -------------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # -------------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # -------------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # 5. Cubic reduction via Triton batched GEMM
    #    left/right : [B, N, N, hidden_dim]
    #    We treat each (batch, hidden) slice as a separate matrix.
    # -------------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Rearrange to (B, hidden_dim, N, N) and make contiguous for easy view
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    batch_hidden = B * hidden_dim
    # Flatten to (batch_hidden, N, N)
    A = left_perm.view(batch_hidden, N, N)
    Bmat = right_perm.view(batch_hidden, N, N)

    # Prepare output container
    C = torch.empty_like(A)

    # Block sizes – tuned for typical B200 hardware
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch_hidden,              # batch*hidden
    )

    _tri_matmul_kernel[grid](
        A,
        Bmat,
        C,
        A.stride(0), A.stride(1), A.stride(2),
        Bmat.stride(0), Bmat.stride(2), Bmat.stride(1),  # transposed view: (k stride, n stride)
        C.stride(0), C.stride(1), C.stride(2),
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection back to dim
    # -------------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
