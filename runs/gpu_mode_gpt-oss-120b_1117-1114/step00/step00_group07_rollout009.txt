@triton.jit
def triton_tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                         B, N, hidden_dim,
                         stride_left_batch, stride_left_i, stride_left_k,
                         stride_right_batch, stride_right_i, stride_right_k,
                         stride_out_batch, stride_out_i, stride_out_j,
                         BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bd = tl.program_id(2)

    # compute batch and channel index
    batch = pid_bd // hidden_dim
    d = pid_bd % hidden_dim

    # compute start offsets
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for i,j
    mask_i = i < N
    mask_j = j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute offset to left base for batch and d
    left_base = left_ptr + batch * stride_left_batch + d * tl.constexpr(1)  # d offset is just d
    right_base = right_ptr + batch * stride_right_batch + d * tl.constexpr(1)
    out_base = out_ptr + batch * stride_out_batch + d * tl.constexpr(1)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k range
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + (i[:, None] * stride_left_i) + (k_offset[None, :] * stride_left_k)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right tile transposed: shape (BLOCK_K, BLOCK_N)
        # we need offs_k as rows, offs_j as columns after transpose
        right_ptrs = right_base + (k_offset[:, None] * stride_right_k) + (j[None, :] * stride_right_i)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product accumulate
        acc += tl.dot(left, right)  # left (M,K) dot right (K,N) -> (M,N)

    # store output
    out_ptrs = out_base + (i[:, None] * stride_out_i) + (j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   B, N, hidden_dim,
                   stride_left_batch, stride_left_i, stride_left_k,
                   stride_right_batch, stride_right_i, stride_right_k,
                   stride_out_batch, stride_out_i, stride_out_j,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bd = tl.program_id(2)

    # decode batch and channel
    batch = pid_bd // hidden_dim
    d = pid_bd % hidden_dim

    # compute global i,j
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # base pointers for this (b,d) pair
    left_base = left_ptr + batch * stride_left_batch + d
    right_base = right_ptr + batch * stride_right_batch + d
    out_base = out_ptr + batch * stride_out_batch + d

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load left tile (i,k)
        left_ptrs = left_base + i[:, None] * stride_left_i + k_offset[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right tile transposed (k,j)
        right_ptrs = right_base + k_offset[:, None] * stride_right_k + j[None, :] * stride_right_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N)

    # store result
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # config values
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B = input_tensor.shape[0]
    N = input_tensor.shape[1]
    # Ensure contiguous
    input_tensor = input_tensor.contiguous()
    mask = mask.contiguous()

    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -- LayerNorm on last dim
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # -- Linear projections (no bias)
    left = F.linear(x, left_proj_weight)   # shape (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)

    # -- Apply mask (if provided)
    mask_unsq = mask.unsqueeze(-1)   # shape (B,N,N,1)
    left = left * mask_unsq
    right = right * mask_unsq

    # -- Gates
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -- Reduction via Triton kernel
    out = torch.empty_like(left)   # same shape

    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # get strides (in elements)
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_batch, stride_right_i, stride_right_k, stride_right_d = right.stride()
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # Block sizes (tuned for typical N)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * hidden_dim)

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        stride_left_batch, stride_left_i, stride_left_k,
        stride_right_batch, stride_right_i, stride_right_k,
        stride_out_batch, stride_out_i, stride_out_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # -- Output layernorm and gating
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate

    # -- Final linear projection back to dim
    output = F.linear(out, to_out_weight)
    return output

"""
TriMul (outgoing) forward pass with a fused Triton kernel for the
O(N³) reduction.

The implementation follows the reference PyTorch module:
1. Layer‑norm the input.
2. Linear projections (left/right) and gating linear layers.
3. Apply mask and element‑wise gates.
4. Compute  out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   using a custom Triton kernel that performs a tiled matrix
   multiplication for each (batch, hidden) pair.
5. Layer‑norm, output gate and final linear projection back to the
   original feature dimension.

Only the reduction (step 4) is performed in Triton; the remaining
operations use regular PyTorch kernels.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d].

    Each program works on a tile (BLOCK_M×BLOCK_N) of the output for a
    specific (batch, hidden) pair.
    """
    pid_i = tl.program_id(0)          # tile row   (i)
    pid_j = tl.program_id(1)          # tile col   (j)
    pid_bd = tl.program_id(2)         # combined batch & hidden index

    # decode batch and hidden channel from pid_bd
    batch = pid_bd // hidden_dim
    d = pid_bd % hidden_dim

    # global i / j indices for this tile
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # base pointers that already include batch and hidden offsets
    left_base = left_ptr + batch * stride_left_batch + d
    right_base = right_ptr + batch * stride_right_batch + d
    out_base = out_ptr + batch * stride_out_batch + d

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction dimension k in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # left[i, k, d]   shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + i[:, None] * stride_left_i + k_off[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right[j, k, d] transposed to (k, j) -> shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + k_off[:, None] * stride_right_k + j[None, :] * stride_right_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N)

    # write the tile back
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[:, None])


def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask        : torch.Tensor [B, N, N] (may be all‑ones)
            - weights     : dict of model parameters
            - config      : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, N2, _ = input_tensor.shape
    assert N == N2, "input must be square in the two middle dimensions"

    # -----------------------------------------------------------------
    # unpack weights (all tensors are already on the correct device)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1. Layer‑norm over the feature dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)     # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # 3. Apply mask (if any)
    # -----------------------------------------------------------------
    mask_unsq = mask.unsqueeze(-1)            # [B,N,N,1]
    left = left * mask_unsq
    right = right * mask_unsq

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid on linear transforms of the same x)
    # -----------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Reduction via Triton (out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d])
    # -----------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float32)

    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # strides in elements (C‑contiguous layout)
    s_lb, s_li, s_lk, s_ld = left.stride()   # (batch, i, k, d)
    s_rb, s_ri, s_rk, s_rd = right.stride()
    s_ob, s_oi, s_oj, s_od = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),            # i‑tiles
        triton.cdiv(N, BLOCK_N),            # j‑tiles
        B * hidden_dim,                     # (batch, channel) tiles
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        s_lb, s_li, s_lk,
        s_rb, s_ri, s_rk,
        s_ob, s_oi, s_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Output layer‑norm, gating and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias)
    out = out * out_gate
    output = F.linear(out, to_out_weight)    # back to [B,N,N,dim]

    return output
