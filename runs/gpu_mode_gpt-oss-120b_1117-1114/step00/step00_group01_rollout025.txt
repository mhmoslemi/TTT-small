"""
TriMul forward (outgoing) implementation with Triton acceleration.

The heavy N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is rewriten as a batch of matrix multiplications:
    out_slice = left_slice @ right_slice.T
for each (batch, hidden) pair.  A custom Triton kernel computes these
matrix products in fp16 (inputs) with fp32 accumulation, while the rest
(LayerNorms, linear projections, gating) are performed with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ  where A,B ∈ (M, N, K)  (M = B*hidden_dim)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_bm, stride_bn, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # index over (batch * hidden)
    pid_i = tl.program_id(1)          # block row of output
    pid_j = tl.program_id(2)          # block column of output

    # Offsets within the N dimension
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_valid = i < N
    j_valid = j < N

    # Base pointers for the current (m) slice
    a_base = a_ptr + pid_m * stride_am + i[:, None] * stride_an          # (BLOCK_M, 1)
    b_base = b_ptr + pid_m * stride_bm + j[None, :] * stride_bn          # (1, BLOCK_N)

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in tiles
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)            # absolute k indices
        k_valid = k < K

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_base + k[None, :] * stride_ak,
            mask=i_valid[:, None] & k_valid[None, :],
            other=0.0,
        )
        # Load Bᵀ block: shape (BLOCK_K, BLOCK_N)   (note the transpose)
        b = tl.load(
            b_base + k[:, None] * stride_bk,
            mask=k_valid[:, None] & j_valid[None, :],
            other=0.0,
        )
        # Dot product over K dimension
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Store the computed block
    c = c_ptr + pid_m * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    tl.store(c, acc, mask=i_valid[:, None] & j_valid[None, :])


# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple(input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] (bool / int / float)
            - weights: dict containing all model parameters
            - config: dict with "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1. LayerNorm over the channel dimension
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B, N, N, hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Apply mask (if any)
    # --------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)   # (B, N, N, 1)
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gating values (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Rearrange for batch of matmuls:  (B, hidden, N, N) → (M, N, N)
    #    where M = B * hidden_dim
    # --------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    M = B * hidden
    left_flat = left.reshape(M, N, N)
    right_flat = right.reshape(M, N, N)

    # Cast to fp16 for the kernel (still keep fp32 copy for safety)
    left_fp16 = left_flat.to(torch.float16).contiguous()
    right_fp16 = right_flat.to(torch.float16).contiguous()

    # Output buffer (fp32 accumulation)
    out_flat = torch.empty((M, N, N), dtype=torch.float32, device=device)

    # --------------------------------------------------------------
    # 6. Launch Triton kernel
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        M,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_flat,
        M,
        N,
        N,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 7. Reshape back to (B, N, N, hidden)
    # --------------------------------------------------------------
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Final LayerNorm + output gating
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # broadcast over hidden dimension

    # --------------------------------------------------------------
    # 9. Linear projection back to original dimension
    # --------------------------------------------------------------
    output = F.linear(out, weights["to_out.weight"])

    return output
