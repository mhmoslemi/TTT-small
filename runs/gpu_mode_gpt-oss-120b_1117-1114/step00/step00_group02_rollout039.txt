@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, D,
    stride_left_i, stride_left_k, stride_left_d,
    stride_right_j, stride_right_k, stride_right_d,
    stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    num_blocks_i = tl.cdiv(N, BLOCK_M)
    num_blocks_j = tl.cdiv(N, BLOCK_N)
    pid_i = pid // num_blocks_j
    pid_j = pid % num_blocks_j

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    offs_d = tl.arange(0, BLOCK_D)
    mask_d = offs_d < D

    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    k = tl.zeros([], tl.int64)  # scalar
    while k < num_k:
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M, K, D)
        left_ptrs = left_ptr + (offs_i[:, None, None] * stride_left_i
                                + offs_k[None, :, None] * stride_left_k
                                + offs_d[None, None, :] * stride_left_d)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                       other=0.0)

        # Load right tile (K, N, D) = right[:, j, k] transposed
        right_ptrs = right_ptr + (offs_k[:, None, None] * stride_right_k
                                  + offs_j[None, :, None] * stride_right_j
                                  + offs_d[None, None, :] * stride_right_d)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
                        other=0.0)

        # Compute dot for each d
        for d in range(BLOCK_D):
            acc[:, :, d] += tl.dot(left[:, :, d], right[:, :, d])

        k += 1

    # Store out
    out_ptrs = out_ptr + (offs_i[:, None, None] * stride_out_i
                          + offs_j[None, :, None] * stride_out_j
                          + offs_d[None, None, :] * stride_out_d)
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])

"""TriMul kernel (outgoing) implementation.

This file defines a Triton kernel that computes the core O(N³) update:
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]

The surrounding operations (LayerNorm, linear projections, gating,
output LayerNorm and linear) are performed with PyTorch for simplicity.
The kernel processes a block of the (i,j) plane together with a block
of the hidden dimension d, accumulating in float32.  The kernel is
launched per‑batch and per‑hidden‑dimension block.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,          # Pointer to left tensor [N, N, D] (float32)
    right_ptr,         # Pointer to right tensor [N, N, D] (float32)
    out_ptr,           # Pointer to output tensor [N, N, D] (float32)
    N,                 # Sequence length (int64)
    D,                 # Hidden dimension of this block (int64)
    stride_left_i, stride_left_k, stride_left_d,   # strides of left
    stride_right_j, stride_right_k, stride_right_d, # strides of right (j is first N dim)
    stride_out_i, stride_out_j, stride_out_d,     # strides of out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Blocked GEMM where the inner dimension k is summed and d is kept.

    For each block (i0:i0+M, j0:j0+N, d0:d0+D) we compute:
        out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    """
    pid = tl.program_id(0)
    num_blocks_i = tl.cdiv(N, BLOCK_M)
    num_blocks_j = tl.cdiv(N, BLOCK_N)

    pid_i = pid // num_blocks_j
    pid_j = pid % num_blocks_j

    # Coordinates of the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    offs_d = tl.arange(0, BLOCK_D)
    mask_d = offs_d < D

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # Loop over the reduction axis k
    num_k = tl.cdiv(N, BLOCK_K)
    k = tl.zeros([], tl.int64)
    while k < num_k:
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile [M, K, D]
        left_ptrs = left_ptr + (
            offs_i[:, None, None] * stride_left_i
            + offs_k[None, :, None] * stride_left_k
            + offs_d[None, None, :] * stride_left_d
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )

        # Load right tile transposed to [K, N, D]
        right_ptrs = right_ptr + (
            offs_k[:, None, None] * stride_right_k
            + offs_j[None, :, None] * stride_right_j
            + offs_d[None, None, :] * stride_right_d
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )

        # Compute dot-products for each hidden channel in the block
        for d in range(BLOCK_D):
            # left[:, :, d] -> (M, K), right[:, :, d] -> (K, N)
            acc[:, :, d] += tl.dot(left[:, :, d], right[:, :, d])

        k += 1

    # Write the result
    out_ptrs = out_ptr + (
        offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + offs_d[None, None, :] * stride_out_d
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, dim]
            mask: torch.Tensor [B, N, N] (binary 0/1)
            weights: dict of torch.Tensor containing layer‑norm, linear and gating weights
            config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---- LayerNorm on the input ------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ---- Linear projections ----------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---- Apply mask ------------------------------------------------------------
    if mask is not None:
        # mask shape [B, N, N] -> broadcast to [B, N, N, 1]
        mask_ = mask.unsqueeze(-1)
        left = left * mask_
        right = right * mask_

    # ---- Gating ----------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---- Core O(N³) update (Triton) -------------------------------------------
    B, N, _, _ = left.shape  # left shape [B, N, N, hidden_dim]
    # Output tensor from the triangular multiplication, float32
    out_intermediate = torch.empty_like(left, dtype=torch.float32, device=device)

    # Triton block sizes (tuned for typical N up to 1024, hidden_dim <= 384)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_D = 16  # process 16 hidden channels per kernel launch

    # Strides for a tensor of shape [N, N, D] (after removing batch dim)
    stride_i = left.stride(1)          # N * D
    stride_k = left.stride(2)          # D
    stride_d = left.stride(3)          # 1

    # Right‑tensor strides (j corresponds to the first N dimension)
    stride_rj = right.stride(1)        # N * D
    stride_rk = right.stride(2)        # D
    stride_rd = right.stride(3)        # 1

    # Output strides
    stride_oi = out_intermediate.stride(1)
    stride_oj = out_intermediate.stride(2)
    stride_od = out_intermediate.stride(3)

    # Grid definition (covers the i‑j plane)
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid = (grid_m * grid_n,)

    # Loop over batches and hidden‑channel blocks
    for b in range(B):
        # Slice the batch dimension once (pointer arithmetic handled by Triton)
        left_batch = left[b]
        right_batch = right[b]
        out_batch = out_intermediate[b]

        # Process hidden channels in blocks of BLOCK_D
        for d_start in range(0, hidden_dim, BLOCK_D):
            d_end = min(d_start + BLOCK_D, hidden_dim)
            # Python slicing creates a view with the same stride (stride_d == 1)
            left_slice = left_batch[:, :, d_start:d_end]
            right_slice = right_batch[:, :, d_start:d_end]
            out_slice = out_batch[:, :, d_start:d_end]

            # Launch the kernel
            tri_mul_kernel[grid](
                left_slice,
                right_slice,
                out_slice,
                N,
                d_end - d_start,               # actual D for this launch
                stride_i,
                stride_k,
                stride_d,
                stride_rj,
                stride_rk,
                stride_rd,
                stride_oi,
                stride_oj,
                stride_od,
                BLOCK_M=BLOCK_M,
                BLOCK_N=BLOCK_N,
                BLOCK_K=BLOCK_K,
                BLOCK_D=BLOCK_D,
            )

    # ---- Output LayerNorm -------------------------------------------------------
    out_norm = F.layer_norm(
        out_intermediate,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply the output gate
    out_norm = out_norm * out_gate

    # Final linear projection back to dim
    output = F.linear(out_norm, weights["to_out.weight"])

    # Ensure final dtype is float32
    return output.to(torch.float32)
