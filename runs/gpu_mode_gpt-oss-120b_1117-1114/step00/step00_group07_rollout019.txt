def triton_tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr

    pid_m = tl.program_id(0)  # i block
    pid_n = tl.program_id(1)  # j block
    pid_z = tl.program_id(2)  # batch*channel
    
    # compute base offset for this slice
    a_base = a_ptr + pid_z * stride_a_batch
    b_base = b_ptr + pid_z * stride_b_batch
    c_base = c_ptr + pid_z * stride_c_batch
    
    # offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config['hidden_dim']
    eps = 1e-5
    
    # LayerNorm on dim
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    
    # Linear projections
    left = torch.nn.functional.linear(x, weights['left_proj.weight'])   # shape [B,N,N,hidden]
    right = torch.nn.functional.linear(x, weights['right_proj.weight'])
    
    # Apply mask if provided
    if mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask
    else:
        mask = None
    
    # Gating
    left_gate = torch.nn.functional.linear(x, weights['left_gate.weight']).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights['right_gate.weight']).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights['out_gate.weight']).sigmoid()
    
    left = left * left_gate
    right = right * right_gate
    
    # matrix multiplication per channel using Triton
    # Reshape to (B*hidden, N, N)
    left_t = left.permute(0,3,1,2).contiguous()
    right_t = right.permute(0,3,1,2).contiguous()
    B, hidden, N, _ = left_t.shape
    P = B * hidden
    a = left_t.view(P, N, N)
    b = right_t.view(P, N, N)
    c = torch.empty_like(a)  # shape [P,N,N], dtype same as a (float32)
    
    # Get strides
    stride_a_batch = a.stride(0)
    stride_a_row = a.stride(1)
    stride_a_col = a.stride(2)
    stride_b_batch = b.stride(0)
    stride_b_row = b.stride(1)
    stride_b_col = b.stride(2)
    stride_c_batch = c.stride(0)
    stride_c_row = c.stride(1)
    stride_c_col = c.stride(2)
    
    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), P)
    triton_tri_mul_kernel[grid](
        a_ptr=a.data_ptr(),
        b_ptr=b.data_ptr(),
        c_ptr=c.data_ptr(),
        N=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    
    # Reshape back to original shape
    out = c.view(B, hidden, N, N).permute(0,2,3,1)  # [B,N,N,hidden]
    
    # to_out_norm
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    out = torch.nn.functional.layer_norm(out, (hidden,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)
    
    # apply out_gate
    out = out * out_gate
    
    # final linear to dim
    out = torch.nn.functional.linear(out, weights['to_out.weight'])
    
    return out

"""
custom_kernel for the TriMul "outgoing" operation.

Algorithm
---------
1. Layer‑norm the input tensor (dim‑last) using provided weights.
2. Apply linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) with sigmoid activation.
3. Mask the projected tensors (if a mask is supplied) and multiply by the
   gate values.
4. Compute the core reduction:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   This is a batch of matrix multiplications left @ rightᵀ per hidden
   channel.  The reduction is performed by a Triton kernel that tiles the
   i‑, j‑, and k‑dimensions (BLOCK_M × BLOCK_N × BLOCK_K).  The kernel works
   on a flattened view of shape (B·H, N, N) to keep the implementation simple.
5. Apply a second layer‑norm (to_out_norm) on the hidden dimension,
   multiply by the out‑gate, and finally linearly project back to the original
   dimension.
All operations are carried out in float32; the Triton kernel accumulates in
float32 as well.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ where A and B have shape (P, N, N).
    The kernel works on a single slice with index `pid_z` = batch * hidden.
    """
    pid_m = tl.program_id(0)      # block row index (i)
    pid_n = tl.program_id(1)      # block column index (j)
    pid_z = tl.program_id(2)      # flattened batch*hidden index

    # Base pointers for this slice
    a_base = a_ptr + pid_z * stride_a_batch
    b_base = b_ptr + pid_z * stride_b_batch
    c_base = c_ptr + pid_z * stride_c_batch

    # Offsets within the block
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k):
        k_offset = kk * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Masks for bounds
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_i[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load Bᵀ block: shape (BLOCK_K, BLOCK_N)
        # Bᵀ[k, j] = B[j, k] in original layout
        b_ptrs = b_base + offs_k[:, None] * stride_b_col + offs_j[None, :] * stride_b_row
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Accumulate dot product
        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # Write the result
    c_ptrs = c_base + offs_i[:, None] * stride_c_row + offs_j[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul module.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask_tensor: torch.Tensor of shape [B, N, N] (may be all‑ones)
            - weights_dict: dict with all required weight tensors
            - config_dict: contains "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])    # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden)
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core reduction (batch of matrix multiplications) via Triton
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) and flatten the first two dims.
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()
    P = B * hidden  # total number of independent matrix multiplies

    a = left_t.view(P, N, N)
    b = right_t.view(P, N, N)
    c = torch.empty_like(a)

    # Strides in element units (contiguous layout => (N*N, N, 1))
    stride_a_batch = a.stride(0)
    stride_a_row = a.stride(1)
    stride_a_col = a.stride(2)
    stride_b_batch = b.stride(0)
    stride_b_row = b.stride(1)
    stride_b_col = b.stride(2)
    stride_c_batch = c.stride(0)
    stride_c_row = c.stride(1)
    stride_c_col = c.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        P,
    )
    _tri_mul_kernel[grid](
        a_ptr=a.data_ptr(),
        b_ptr=b.data_ptr(),
        c_ptr=c.data_ptr(),
        N=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Reshape result back to [B, N, N, H]
    out = c.view(B, hidden, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6) Output layer‑norm, out‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
