def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    
    # layernorm x
    # using provided weight and bias
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # x_norm = (x - mean) / sqrt(var+eps) * weight + bias
    # compute mean and variance over last dim dim.
    x = input_tensor
    # x is float32 presumably.
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias
    # Ensure dtype float32
    x_norm = x_norm.to(torch.float32)

    # linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # mask
    # shape [B,N,N] -> unsqueeze
    mask_ = mask.unsqueeze(-1)
    left = left * mask_
    right = right * mask_

    # gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # cast to fp16 for kernel (optionally)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output tensor
    out = torch.empty_like(left_fp16, dtype=torch.float32, device=device)  # shape B,N,N,H

    # strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right_fp16.stride()  # same dims but j/k reversed? we will treat same ordering.
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    B, N, _, H = left_fp16.shape   # left shape [B,N,N,hidden_dim] shape matches.

    # define block sizes
    BLOCK_M = 64  # i dimension
    BLOCK_N = 64  # j dimension
    BLOCK_K = 64  # k reduction
    # maybe adjust for small N to avoid too large blocks; but mask will handle.
    
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Now out has result shape [B,N,N,H], dtype float32.

    # Now apply to_out_norm (layernorm over hidden dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # compute mean, var along last dim (H)
    mean2 = out.mean(dim=-1, keepdim=True)
    var2 = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean2) * torch.rsqrt(var2 + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # gating
    out_norm = out_norm * out_gate  # out_gate shape matches.

    # final linear to dim
    to_out_weight = weights["to_out.weight"]
    # shape: (dim, hidden_dim)
    final_out = torch.nn.functional.linear(out_norm, to_out_weight)  # B,N,N,dim
    # final_out is float32

    return final_out
```

Implementation details for kernel:

```
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_x = tl.program_id(0)  # i block
    pid_y = tl.program_id(1)  # j block
    pid_z = tl.program_id(2)  # combined batch*head

    # decode batch and hidden index
    b = pid_z // H
    h = pid_z % H

    # compute offsets
    i_start = pid_x * BLOCK_M
    j_start = pid_y * BLOCK_N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K

        # compute mask for valid loads
        i_bound = i_start + tl.arange(0, BLOCK_M) < N
        j_bound = j_start + tl.arange(0, BLOCK_N) < N
        k_bound = k_start + tl.arange(0, BLOCK_K) < N

        # offsets for left: (i, k)
        left_ptrs = left_ptr + \
            b * stride_l_b + \
            (i_start + tl.arange(0, BLOCK_M))[:, None] * stride_l_i + \
            (k_start + tl.arange(0, BLOCK_K))[None, :] * stride_l_k + \
            h * stride_l_h

        left = tl.load(left_ptrs, mask=i_bound[:, None] & k_bound[None, :], other=0.0)

        # offsets for right transposed: (k, j)
        right_ptrs = right_ptr + \
            b * stride_r_b + \
            (k_start + tl.arange(0, BLOCK_K))[:, None] * stride_r_k + \
            (j_start + tl.arange(0, BLOCK_N))[None, :] * stride_r_j + \
            h * stride_r_h

        right = tl.load(right_ptrs, mask=k_bound[:, None] & j_bound[None, :], other=0.0)

        # accumulate
        acc += tl.dot(left, right)

    # write back
    out_ptrs = out_ptr + \
        b * stride_o_b + \
        (i_start + tl.arange(0, BLOCK_M))[:, None] * stride_o_i + \
        (j_start + tl.arange(0, BLOCK_N))[None, :] * stride_o_j + \
        h * stride_o_h

    tl.store(out_ptrs, acc, mask=i_bound[:, None] & j_bound[None, :])
```

import triton
import triton.language as tl

epsilon = 1e-5

@triton.jit
def tri_mul_kernel(...):
   ...

def custom_kernel(data):
    """
    Triton-accelerated implementation of the 'outgoing' Triangle Multiplicative Update.
    ...
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # optional config for nomask but we treat mask always.

    # LayerNorm over dim using provided parameters
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=epsilon)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if provided
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gating
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for Triton matmul
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, H = left_fp16.shape  # H == hidden_dim

    # Allocate output tensor (float32)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # Strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Block sizes (tuned for GPU)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Second LayerNorm over hidden_dim
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_ln = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=epsilon)

    # Apply output gate
    out_ln = out_ln * out_gate

    # Final linear projection back to dim
    to_out_weight = weights["to_out.weight"]
    final = torch.nn.functional.linear(out_ln, to_out_weight)

    return final

"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ [B,N,N,D] using the supplied
   `norm.weight` and `norm.bias`.
2. Linear projections (no bias) produce left/right tensors of shape
   [B,N,N,H] where H = hidden_dim.
3. Optional mask (broadcasted on the feature dimension) is applied.
4. Gating vectors are computed by a sigmoid‑activated linear projection on
   the normalized input and multiplied element‑wise with left/right.
5. The core operation:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is a batched matrix multiplication L @ Rᵀ for each hidden channel.
   This is performed by a custom Triton kernel that tiles the (i,j)
   dimensions (BLOCK_M×BLOCK_N) and loops over k in blocks (BLOCK_K).
6. The result is layer‑normed across the hidden dimension, multiplied by
   the output gate, and finally projected back to the original dimension D.
All heavy computation (step 5) runs on the GPU via Triton; the remaining
operations use PyTorch (float32) for simplicity and numerical stability.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# -------------------------------------------------------------
# Triton kernel for the O(N³) reduction:
#   out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
# -------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_x = tl.program_id(0)          # block row (i)
    pid_y = tl.program_id(1)          # block col (j)
    pid_z = tl.program_id(2)          # combined batch * hidden channel

    # decode batch index and hidden channel
    b = pid_z // H
    h = pid_z % H

    i0 = pid_x * BLOCK_M
    j0 = pid_y * BLOCK_N

    i_range = tl.arange(0, BLOCK_M)
    j_range = tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    mask_i = i0 + i_range < N
    mask_j = j0 + j_range < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = kk * BLOCK_K
        mask_k = k0 + k_range < N

        # ----- load left tile (i, k) -----
        left_ptrs = left_ptr + \
            b * stride_l_b + \
            (i0 + i_range)[:, None] * stride_l_i + \
            (k0 + k_range)[None, :] * stride_l_k + \
            h * stride_l_h
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # ----- load right tile transposed (k, j) -----
        # original layout is (b, j, k, h); we access it as (k, j)
        right_ptrs = right_ptr + \
            b * stride_r_b + \
            (k0 + k_range)[:, None] * stride_r_k + \
            (j0 + j_range)[None, :] * stride_r_j + \
            h * stride_r_h
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # ----- accumulate multiplication -----
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ----- write back the result -----
    out_ptrs = out_ptr + \
        b * stride_o_b + \
        (i0 + i_range)[:, None] * stride_o_i + \
        (j0 + j_range)[None, :] * stride_o_j + \
        h * stride_o_h
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------
# Public entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Args:
        data: Tuple containing
            - input_tensor : torch.Tensor of shape [B, N, N, D]
            - mask         : torch.Tensor of shape [B, N, N] (may be None or all‑ones)
            - weights      : dict of weight tensors (see implementation)
            - config       : dict with keys "dim" (D) and "hidden_dim" (H)

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    eps = 1e-5

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    H = hidden_dim

    # ------------------- LayerNorm on input -------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,D]

    # ------------------- Linear projections --------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------- Apply mask (if any) ------------------
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)       # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------- Gating -------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------- Cast to fp16 for the kernel ----------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ------------------- Allocate output -----------------------
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # ------------------- Strides --------------------------------
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # ------------------- Triton launch -------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            B * H)  # each (batch, hidden) pair gets its own matmul

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------- Second LayerNorm (hidden dim) ---------
    out_ln = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------- Output gate ---------------------------
    out_ln = out_ln * out_gate

    # ------------------- Final linear back to dim -------------
    final = F.linear(out_ln, weights["to_out.weight"])

    return final
