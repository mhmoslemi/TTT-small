    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_bzl, stride_bzl_i, stride_bzl_j, stride_bzl_d,
    stride_bzr, stride_bzr_i, stride_bzr_j, stride_bzr_d,
    stride_bzo, stride_bzo_i, stride_bzo_j, stride_bzo_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch = tl.program_id(0)  # range B*D
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # compute b, d from batch
    b = batch // D
    d = batch % D
    # Base pointers for left slice
    left_base = left_ptr + b * stride_bzl + d * stride_bzl_d
    right_base = right_ptr + b * stride_bzr + d * stride_bzr_d
    out_base = out_ptr + b * stride_bzo + d * stride_bzo_d

    mask_m = offs_m < N
    mask_n = offs_n < N

        right_vals = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left_vals, right_vals)

def custom_kernel(data):
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    eps = 1e-5

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Use torch.nn.functional.layer_norm
    # shape normalized over last dim
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left_proj_w = weights["left_proj.weight"]  # shape (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # Compute projections
    left = F.linear(x, left_proj_w)  # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)

    # Apply mask if needed
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)  # shape (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # Compute gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # convert to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Prepare right transposed for multiplication
    right_T = right.permute(0,3,2,1)  # (B, hidden, N, N)

    # Reshape left, right_T to (B*hidden, N, N)
    left_mat = left.permute(0,3,1,2).contiguous().view(bs * hidden_dim, N, N)
    right_mat = right_T.contiguous().view(bs * hidden_dim, N, N)

    out_mat = torch.empty_like(left_mat)  # fp16

    # kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (bs * hidden_dim, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        bs * hidden_dim,
        N,
        left_mat.stride(0),
        left_mat.stride(1),
        left_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # out_mat shape (B*hidden, N, N) fp16
    # Reshape back to (B,N,N,hidden)
    out = out_mat.view(bs, hidden_dim, N, N).permute(0,2,3,1).to(torch.float32)

    # to_out_norm and gating
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=eps)

    out = out * out_gate
    # final linear
    out = F.linear(out, to_out_w)

    return out

                          batch, N,

                          stride_r_batch, stride_r_row, stride_r_col,
                          stride_o_batch, stride_o_row, stride_o_col,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: (i,k)

    # store output

"""
TriMul (outgoing) implementation with a custom Triton kernel.

The heavy O(N³·C) reduction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is performed as a batched matrix multiplication using a Triton
kernel (one batch = one pair (b, d)).  The surrounding LayerNorms,
linear projections and gating are kept in PyTorch for simplicity.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the core TriMul reduction.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    batch, N,                                    # sizes
    stride_l_batch, stride_l_row, stride_l_col, # left strides
    stride_r_batch, stride_r_row, stride_r_col, # right strides
    stride_o_batch, stride_o_row, stride_o_col, # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes, for each ``batch`` (i.e. a specific (b, d) pair),
    out = left @ right_T  where both left and right_T are N×N matrices.
    """
    batch_idx = tl.program_id(0)                      # 0 ≤ batch_idx < batch
    pid_m = tl.program_id(1)                          # block row index
    pid_n = tl.program_id(2)                          # block column index

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block  (i, k)
        left_ptrs = (left_ptr
                     + batch_idx * stride_l_batch
                     + offs_m[:, None] * stride_l_row
                     + offs_k[None, :] * stride_l_col)
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # Load right_T block  (k, j)
        right_ptrs = (right_ptr
                      + batch_idx * stride_r_batch
                      + offs_k[:, None] * stride_r_row
                      + offs_n[None, :] * stride_r_col)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # Multiply‑accumulate
        acc += tl.dot(left, right)

    # Store the result tile
    out_ptrs = (out_ptr
                + batch_idx * stride_o_batch
                + offs_m[:, None] * stride_o_row
                + offs_n[None, :] * stride_o_col)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entrypoint used by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim]  (float32)
            - mask:         [B, N, N]       (float32 or bool)
            - weights: dict of model parameters
            - config:  dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # Unpack configuration
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # LayerNorm on the input (over the last dimension)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])    # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply mask if present (mask is broadcast over the hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)        # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to half for the heavy reduction
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # Prepare right transposed (j ↔ k) and flatten batch×hidden for the kernel
    # ------------------------------------------------------------------
    # left: (B,N,N,hidden) -> (B, hidden, N, N) -> (B*hidden, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # right_T: (B,N,N,hidden) -> permute to (B, hidden, N, N) where axes are (k,j)
    right_T = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Output buffer for the batched matmul (still half)
    out_mat = torch.empty_like(left_mat)

    # ------------------------------------------------------------------
    # Triton kernel launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                # batch dimension (b * d)
        triton.cdiv(N, BLOCK_M),      # block rows
        triton.cdiv(N, BLOCK_N),      # block columns
    )

    # All three tensors have the same contiguous layout: (batch, row, col)
    stride_l_batch, stride_l_row, stride_l_col = left_mat.stride()
    stride_r_batch, stride_r_row, stride_r_col = right_T.stride()
    stride_o_batch, stride_o_row, stride_o_col = out_mat.stride()

    tri_mul_matmul_kernel[grid](
        left_mat,
        right_T,
        out_mat,
        B * hidden_dim,
        N,
        stride_l_batch,
        stride_l_row,
        stride_l_col,
        stride_r_batch,
        stride_r_row,
        stride_r_col,
        stride_o_batch,
        stride_o_row,
        stride_o_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Restore the original 4‑D layout and convert back to FP32
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
