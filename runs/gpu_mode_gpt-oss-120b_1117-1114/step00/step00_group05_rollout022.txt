@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr, 
                   B, N, H,
                   stride_left_b, stride_left_i, stride_left_k, stride_left_h,
                   stride_right_b, stride_right_j, stride_right_k, stride_right_h,
                   stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                   BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)  # combined batch and hidden dim

    # Compute batch index and hidden dimension index
    batch = pid_z // H
    hd = pid_z % H

    # Compute starting indices
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    # Offsets for i and j
    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)

    # Mask for i and j (boundary)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over k dimension
    for k_start in range(0, N, BLOCK_K):
        # k offsets
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Compute pointers for left block: shape (BLOCK_I, BLOCK_K)
        # left index: (batch, i, k, hd)
        left_ptrs = left_ptr + \
            batch * stride_left_b + \
            hd * stride_left_h + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k

        # Load left block with mask

        # Compute pointers for right block transposed: shape (BLOCK_K, BLOCK_J)
        # right index: (batch, j, k, hd). We want (k,j) ordering.

            batch * stride_right_b + \
            hd * stride_right_h + \
            offs_k[:, None] * stride_right_k + \
            offs_j[None, :] * stride_right_j

        # Load right block
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Multiply-accumulate

    # Write back output
    out_ptrs = out_ptr + \
        batch * stride_out_b + \
        hd * stride_out_h + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j

    # Convert acc to fp16 for storage
    out = acc.to(tl.float16)

    tl.store(out_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])

    input_tensor, mask, weights, config = data
    # device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim: int = config["hidden_dim"]
    eps = 1e-5

    # Retrieve weight tensors from dict
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Ensure all weights are float32
    norm_weight = norm_weight.to(torch.float32)
    norm_bias = norm_bias.to(torch.float32)
    left_proj_weight = left_proj_weight.to(torch.float32)
    right_proj_weight = right_proj_weight.to(torch.float32)
    left_gate_weight = left_gate_weight.to(torch.float32)
    right_gate_weight = right_gate_weight.to(torch.float32)
    out_gate_weight = out_gate_weight.to(torch.float32)
    to_out_norm_weight = to_out_norm_weight.to(torch.float32)
    to_out_norm_bias = to_out_norm_bias.to(torch.float32)
    to_out_weight = to_out_weight.to(torch.float32)

    # 1) LayerNorm on input x
    x_norm = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # 2) Linear projections + gating
    left = F.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_weight)

    # Apply mask if needed: mask shape [B,N,N]; unsqueeze
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x_norm.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for kernel compute
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Allocate output buffer
    out_fp16 = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=input_tensor.device)

    # Strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_fp16.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right_fp16.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_fp16.stride()

    # Define block sizes
    BLOCK_I = 128  # may tune
    BLOCK_J = 128
    BLOCK_K = 32  # smaller for caching

    # Grid dimensions
    grid_i = triton.cdiv(N, BLOCK_I)
    grid_j = triton.cdiv(N, BLOCK_J)
    grid_z = B * hidden_dim

    # Launch kernel
    tri_mul_kernel[ (grid_i, grid_j, grid_z) ](
        left_fp16, right_fp16, out_fp16,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K
    )

    # Convert back to float32 for the rest
    out = out_fp16.to(torch.float32)

    # apply to_out_norm (LayerNorm over hidden_dim)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # multiply by out_gate (already float32)
    out = out * out_gate

    # final linear projection to dim
    output = F.linear(out, to_out_weight)  # shape [B,N,N,dim]

    return output

"""
TriMul (outgoing) forward pass using Triton.

Algorithm
---------
1. Layer‑norm the input tensor `x` with learned weights.
2. Linear projections (left/right) to the hidden dimension.
3. Apply the binary mask (if present) to the projected tensors.
4. Compute three sigmoid gates from the normalized input.
5. Gated left/right tensors: left *= left_gate, right *= right_gate.
6. Core O(N³·H) update:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a fused Triton kernel that performs a blocked
   matrix multiplication for each (batch, hidden) pair.
7. Layer‑norm over the hidden dimension, multiply by the final gate,
   and linearly project back to the original feature dimension.

All operations are performed on the GPU, the heavy reduction (step 6) is
fully fused in Triton, while the surrounding linear/normalisation steps
use PyTorch for simplicity.  The final output is a float‑32 tensor of
shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,          # *float32
    right_ptr,         # *float32
    out_ptr,           # *float32
    B, N, H,           # scalar sizes
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid_i = tl.program_id(0)          # block row (i)
    pid_j = tl.program_id(1)          # block col (j)
    pid_z = tl.program_id(2)          # combined batch & hidden index

    # ----- decode batch and hidden index -----
    batch = pid_z // H
    hid   = pid_z % H

    # ----- compute start indices for this block -----
    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J

    # ----- range vectors -----
    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ----- base pointers for this (batch, hidden) pair -----
    left_base  = left_ptr  + batch * stride_lb + hid * stride_lh
    right_base = right_ptr + batch * stride_rb + hid * stride_rh
    out_base   = out_ptr   + batch * stride_ob + hid * stride_oh

    # ----- accumulator (float32) -----
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # ----- loop over k dimension in blocks -----
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_I, BLOCK_K)
        left_ptrs = left_base + \
            offs_i[:, None] * stride_li + \
            offs_k[None, :] * stride_lk
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right block (transposed): shape (BLOCK_K, BLOCK_J)
        right_ptrs = right_base + \
            offs_k[:, None] * stride_rk + \
            offs_j[None, :] * stride_rj
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # accumulate dot product over the K‑block
        acc += tl.dot(left, right)   # (BLOCK_I,BLOCK_K)@(BLOCK_K,BLOCK_J)

    # ----- write back the result -----
    out_ptrs = out_base + \
        offs_i[:, None] * stride_oi + \
        offs_j[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float32),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module with a Triton kernel
    for the heavy multiplicative update.

    Args:
        data: Tuple (input, mask, weights, config)
            - input: torch.Tensor of shape [B, N, N, dim]
            - mask:  torch.Tensor of shape [B, N, N] (binary)
            - weights: dict with all Linear/LayerNorm parameters
            - config: dict with keys 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. LayerNorm on the input
    norm_w = weights["norm.weight"].to(torch.float32)
    norm_b = weights["norm.bias"].to(torch.float32)
    x_norm = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b, eps=eps)

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    left_proj_w  = weights["left_proj.weight"].to(torch.float32)
    right_proj_w = weights["right_proj.weight"].to(torch.float32)

    left  = F.linear(x_norm, left_proj_w)   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_w)

    # -----------------------------------------------------------------
    # 3. Apply mask (if any)
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(x_norm.dtype)   # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Compute gates (sigmoid of a linear projection)
    left_gate_w  = weights["left_gate.weight"].to(torch.float32)
    right_gate_w = weights["right_gate.weight"].to(torch.float32)
    out_gate_w   = weights["out_gate.weight"].to(torch.float32)

    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_w))

    # -----------------------------------------------------------------
    # 5. Apply gates to projected tensors
    left  = left  * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Core O(N³·H) multiplication via Triton
    # Ensure tensors are contiguous for correct strides
    left  = left.contiguous()
    right = right.contiguous()

    # Allocate output buffer
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (must match left/right/out layout)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Block sizes – can be tuned; chosen to fit typical B200 resources
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32

    grid_i = triton.cdiv(N, BLOCK_I)
    grid_j = triton.cdiv(N, BLOCK_J)
    grid_z = B * hidden_dim                      # (batch * hidden)

    tri_mul_kernel[(grid_i, grid_j, grid_z)](
        left,
        right,
        out,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )
    # -----------------------------------------------------------------
    # 7. Post‑processing: LayerNorm over hidden dim, gating, final linear
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float32)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w,
                       bias=to_out_norm_b, eps=eps)

    out = out * out_gate      # broadcast over hidden dimension

    to_out_w = weights["to_out.weight"].to(torch.float32)   # [dim, hidden_dim]
    output = F.linear(out, to_out_w)   # final shape [B,N,N,dim]

    return output
