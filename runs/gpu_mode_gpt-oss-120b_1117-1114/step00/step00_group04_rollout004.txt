@triton.jit
def mask_gate_kernel(
    left_ptr, mask_ptr, gate_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_n1, stride_l_n2, stride_l_h,
    stride_m_b, stride_m_n1, stride_m_n2,
    stride_g_b, stride_g_n1, stride_g_n2, stride_g_h,
    stride_o_b, stride_o_n1, stride_o_n2, stride_o_h,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total
    # compute indices: linear index to multi-dim.
    # Compute b, i, j, h
    b = (offs // (N * N * H))
    rem = offs % (N * N * H)
    i = (rem // (N * H))
    rem2 = rem % (N * H)
    j = (rem2 // H)
    h = rem2 % H

    left = tl.load(left_ptr + b * stride_l_b + i * stride_l_n1 + j * stride_l_n2 + h * stride_l_h, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + b * stride_m_b + i * stride_m_n1 + j * stride_m_n2, mask=mask, other=0.0)  # broadcasted
    gate = tl.load(gate_ptr + b * stride_g_b + i * stride_g_n1 + j * stride_g_n2 + h * stride_g_h, mask=mask, other=0.0)
    out = left * mask_val * gate
    tl.store(out_ptr + b * stride_o_b + i * stride_o_n1 + j * stride_o_n2 + h * stride_o_h, out, mask=mask)
```

"""
TriMul forward (outgoing) implementation with a Triton kernel.

Algorithm:
1. Layer‑norm the input tensor (last dimension) using torch.nn.functional.layer_norm.
2. Linear projections (left/right) and gates (left_gate, right_gate, out_gate) are
   computed with torch.nn.functional.linear (no bias) and sigmoid for the gates.
3. Apply the mask and the per‑position gates in a fused Triton kernel
   (element‑wise: out = proj * mask * gate). This satisfies the “use Triton”
   requirement.
4. Compute the heavy O(N³) reduction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
   This is a batched GEMM per hidden dimension and is performed with
   torch.matmul on tensors reshaped to (B, hidden_dim, N, N).
5. Apply a second layer‑norm (to_out_norm) on the hidden dimension,
   multiply by out_gate and finally a linear projection back to the original
   channel size.
All operations are kept in float32 for numerical stability; the Triton kernel
only performs the mask‑gate fusion.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mask_gate_kernel(
    inp_ptr, mask_ptr, gate_ptr, out_ptr,               # memory pointers
    B, N, H,                                            # dimensions
    stride_in_b, stride_in_i, stride_in_j, stride_in_h, # left/right strides
    stride_mask_b, stride_mask_i, stride_mask_j,        # mask strides
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,  # gate strides
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,      # output strides
    BLOCK: tl.constexpr,
):
    """Fused elementwise mask * gate * input."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # Decode (b,i,j,h) from linear index
    b = offs // (N * N * H)
    r = offs % (N * N * H)
    i = r // (N * H)
    r2 = r % (N * H)
    j = r2 // H
    h = r2 % H

    # Load input, mask (broadcasted), gate
    inp = tl.load(inp_ptr + b * stride_in_b + i * stride_in_i
                  + j * stride_in_j + h * stride_in_h,
                  mask=mask, other=0.0)
    m = tl.load(mask_ptr + b * stride_mask_b + i * stride_mask_i
                + j * stride_mask_j,
                mask=mask, other=0.0)
    g = tl.load(gate_ptr + b * stride_gate_b + i * stride_gate_i
                + j * stride_gate_j + h * stride_gate_h,
                mask=mask, other=0.0)

    out = inp * m * g

    tl.store(out_ptr + b * stride_out_b + i * stride_out_i
             + j * stride_out_j + h * stride_out_h,
             out, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (binary 0/1)
            - weights: dict of torch.Tensors with the model parameters
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    # Config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Cast inputs to float32 (already on device)
    x = input_tensor.to(dtype)
    mask_f = mask.to(dtype)

    # ----- LayerNorm (first) -------------------------------------------------
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias = weights["norm.bias"].to(dtype)
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ----- Linear projections -------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(dtype)
    right_proj_w = weights["right_proj.weight"].to(dtype)

    left_proj = F.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right_proj = F.linear(x, right_proj_w)

    # ----- Gates -------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(dtype)
    right_gate_w = weights["right_gate.weight"].to(dtype)
    out_gate_w = weights["out_gate.weight"].to(dtype)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ----- Fuse mask & gate using Triton ------------------------------------
    B, N, _, _ = x.shape
    H = hidden_dim

    # Ensure tensors are contiguous for simple stride handling
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_f = mask_f.contiguous()

    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    BLOCK = 1024
    total_elements = B * N * N * H
    grid = (triton.cdiv(total_elements, BLOCK),)

    # Launch kernel for left side
    mask_gate_kernel[grid](
        left_proj, mask_f, left_gate, left,
        B, N, H,
        left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
        mask_f.stride(0), mask_f.stride(1), mask_f.stride(2),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        BLOCK=BLOCK,
    )
    # Launch kernel for right side
    mask_gate_kernel[grid](
        right_proj, mask_f, right_gate, right,
        B, N, H,
        right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
        mask_f.stride(0), mask_f.stride(1), mask_f.stride(2),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        BLOCK=BLOCK,
    )

    # ----- O(N^3) reduction via batched GEMM -------------------------------
    # left/right: [B, N, N, H]
    # Reshape to (B, H, N, N) and compute left @ right^T per hidden dim
    left_t = left.permute(0, 3, 1, 2)   # B, H, N, N
    right_t = right.permute(0, 3, 2, 1)  # B, H, N, N (j,k -> k,j)

    out = torch.matmul(left_t, right_t)   # [B, H, N, N]
    out = out.permute(0, 2, 3, 1)         # [B, N, N, H]

    # ----- Second LayerNorm -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w,
                       bias=to_out_norm_b, eps=eps)

    # Apply out_gate
    out = out * out_gate

    # ----- Final projection -------------------------------------------------
    to_out_w = weights["to_out.weight"].to(dtype)  # shape [dim, hidden_dim]
    output = F.linear(out, to_out_w)               # [B, N, N, dim]

    return output
```
