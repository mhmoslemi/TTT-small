    stride_br, stride_ir, stride_kr, stride_hr,

"""
TriMul (outgoing) forward pass implemented with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ [B,N,N,D].
2. Linear projections to hidden dimension H:
   left  = x @ W_left   (shape [B,N,N,H])
   right = x @ W_right
3. (Optional) apply a binary mask of shape [B,N,N].
4. Compute per‑position gates (sigmoids) from x and multiply:
   left  = left  * sigmoid(x @ W_left_gate)
   right = right * sigmoid(x @ W_right_gate)
5. Core O(N³) operation:
   out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]
   This is implemented by a Triton kernel where each program (thread)
   processes a single (b,i,j) triplet and loops over k, accumulating
   a vector of length H in FP32.
6. Layer‑norm the hidden result, apply the output gate
   (sigmoid(x @ W_out_gate)) and a final linear projection back to D.
7. Return the final tensor of shape [B,N,N,D] (FP32).

Only the heavy O(N³) reduction is executed in Triton; the surrounding
linear, norm, and gate operations use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # *fp16  [B,N,N,H]
    right_ptr,         # *fp16  [B,N,N,H]
    out_ptr,           # *fp32  [B,N,N,H]
    B: tl.constexpr,   # batch size (constexpr)
    N: tl.constexpr,   # sequence length (constexpr)
    stride_bl, stride_il, stride_kl, stride_hl,   # left strides
    stride_br, stride_ir, stride_kr, stride_hr,   # right strides
    stride_bo, stride_io, stride_jo, stride_ho,   # out strides
    BLOCK_H: tl.constexpr                       # hidden dimension (constexpr)
):
    pid = tl.program_id(0)

    # map 1‑D program ID to (b,i,j)
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base offsets for this program
    b_off_l = b * stride_bl
    b_off_r = b * stride_br
    b_off_o = b * stride_bo

    i_off_l = i * stride_il
    i_off_o = i * stride_io

    j_off_r = j * stride_ir
    j_off_o = j * stride_jo

    # vector of h‑indices and corresponding byte offsets
    h_range = tl.arange(0, BLOCK_H)
    h_off = h_range * stride_hl  # stride_hl == stride_hr == stride_ho (contiguous)

    # accumulator in FP32
    acc = tl.zeros([BLOCK_H], dtype=tl.float32)

    # reduction over k (static loop, unrolled at compile time)
    for k in range(N):
        # load left[b,i,k,:]   (fp16)
        left_ptr_k = left_ptr + b_off_l + i_off_l + k * stride_kl
        left_vec = tl.load(left_ptr_k + h_off, dtype=tl.float16)

        # load right[b,j,k,:]  (fp16)
        right_ptr_k = right_ptr + b_off_r + j_off_r + k * stride_kr
        right_vec = tl.load(right_ptr_k + h_off, dtype=tl.float16)

        # promote to fp32 and accumulate
        acc += left_vec.to(tl.float32) * right_vec.to(tl.float32)

    # write result out[b,i,j,:]  (fp32)
    out_base = out_ptr + b_off_o + i_off_o + j_off_o
    tl.store(out_base + h_off, acc)


def custom_kernel(data):
    """
    Executes the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input, mask, weights, config)
            input   – Tensor [bs, seq_len, seq_len, dim] (float32)
            mask    – Tensor [bs, seq_len, seq_len] (optional)
            weights – Dict of model parameters (float32)
            config  – Dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm on the input ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if mask is not None and mask.numel() > 0 and not config.get("nomask", False):
        mask_unsq = mask.unsqueeze(-1).to(x.dtype)  # [B,N,N,1]
        left = left * mask_unsq
        right = right * mask_unsq

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast projections to half for the Triton kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ---------- Triton kernel for the O(N³) reduction ----------
    out_hidden = torch.empty(
        (bs, N, N, hidden_dim), dtype=torch.float32, device=device
    )

    # Strides (in elements)
    stride_bl, stride_il, stride_kl, stride_hl = left_fp16.stride()
    stride_br, stride_ir, stride_kr, stride_hr = right_fp16.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out_hidden.stride()

    total_progs = bs * N * N
    grid = (total_progs,)

    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_hidden,
        bs,
        N,
        stride_bl,
        stride_il,
        stride_kl,
        stride_hl,
        stride_br,
        stride_ir,
        stride_kr,
        stride_hr,
        stride_bo,
        stride_io,
        stride_jo,
        stride_ho,
        BLOCK_H=hidden_dim,  # compile‑time constant
    )

    # ---------- Output LayerNorm ----------
    out_norm = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply the output gate (already FP32)
    out_norm = out_norm * out_gate

    # ---------- Final linear projection ----------
    output = F.linear(out_norm, weights["to_out.weight"])

    return output
