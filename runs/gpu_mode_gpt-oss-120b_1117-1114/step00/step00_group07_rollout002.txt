def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # layernorm
    eps = 1e-5
    # compute mean, var over last dim
    x = input_tensor
    # x_norm = (x - mean) / sqrt(var+eps) * weight + bias
    # compute per-element mean/var across dim (last)
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = x_norm * norm_weight + norm_bias   # broadcasting

    # linear projections from dim->hidden_dim

    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # Use F.linear
    left = torch.nn.functional.linear(x_norm, left_proj_w) # shape [B,N,N,hidden]
    right = torch.nn.functional.linear(x_norm, right_proj_w)

    # mask unsqueeze
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # reshape for batch multiplication
    # Permute: (B,N,N,H) -> (B,H,N,N)
    left_perm = left_fp16.permute(0,3,1,2).contiguous()
    right_perm = right_fp16.permute(0,3,1,2).contiguous()
    # Reshape to (B*H, N, N)
    batch_h = bs * hidden_dim
    left_mat = left_perm.view(batch_h, N, N)
    right_mat = right_perm.view(batch_h, N, N)

    # allocate output matrix
    out_mat_fp16 = torch.empty_like(left_mat, dtype=torch.float16)

    # compute strides
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_m, stride_b_n = right_mat.stride()  # but we will treat b as transposed
    # for transposed, we need stride_b_k = stride_b_m, stride_b_n = stride_b_n (original k dimension)
    stride_b_k = stride_b_m
    stride_b_n = stride_b_n

    stride_c_batch, stride_c_m, stride_c_n = out_mat_fp16.stride()

    # set block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (batch_h, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    batched_matmul_kernel[grid](
        left_mat, right_mat, out_mat_fp16,
        batch_h, N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape back to (B,N,N,H)
    out_mat = out_mat_fp16.view(bs, hidden_dim, N, N).permute(0,2,3,1).to(torch.float32)

    # to_out_norm
    out_norm = torch.nn.functional.layer_norm(out_mat, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=1e-5)

    # gating
    out_norm = out_norm * out_gate

    # final linear
    out = torch.nn.functional.linear(out_norm, to_out_w)  # shape [B,N,N,dim]
    return out

"""
TriMul forward (outgoing) implementation with a Triton kernel for the
core N^3 reduction.

Algorithm
---------
1. Layer‑norm over the last dimension of the input tensor.
2. Linear projections to hidden_dim (left/right) and to gates (left_gate,
   right_gate, out_gate).  Apply the optional mask and the sigmoid‑gates.
3. The reduction
     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is realised as a batched matrix multiplication:
     out_h[b] = left_h[b] @ right_h[b].T
   for each hidden dimension h.  A custom Triton kernel performs the
   batched matmul (accumulating in FP32) on FP16 data for speed.
4. Layer‑norm over the hidden dimension, apply the out‑gate, and a final
   linear projection back to the original dimension.

All operations run on GPU; the heavy N‑cube reduction is handled by the
Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from triton import cdiv

# ----------------------------------------------------------------------
# Triton kernel: batched matmul (A @ B^T) where A,B have shape (BATCH, N, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    BATCH, M, N, K,               # dimensions (BATCH = total_batches)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)          # which batch*head
    pid_m = tl.program_id(1)             # block row
    pid_n = tl.program_id(2)             # block column

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile [M, K]
        a_ptrs = a_ptr + batch_id * stride_a_batch \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load B tile transposed: we need B^T -> shape [K, N]
        b_ptrs = b_ptr + batch_id * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # Store the result
    c_ptrs = c_ptr + batch_id * stride_c_batch \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward pass (outgoing) with a Triton kernel for the N^3 reduction.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool/float)
            - weights_dict: contains all required weight/bias tensors
            - config_dict: has keys "dim" (C) and "hidden_dim" (H)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input layer norm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (C,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # ------------------------------------------------------------------
    # 2. Linear projections and gates
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left = F.linear(x, left_proj_w)          # [B, N, N, H]
    right = F.linear(x, right_proj_w)        # [B, N, N, H]

    # Apply mask if provided (mask may be all‑ones when nomask=True)
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Core O(N^3) reduction via Triton batched matmul
    # ------------------------------------------------------------------
    # Convert to half for bandwidth‑limited kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Rearrange to (batch, head, N, N) -> (batch*head, N, N)
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()
    total_batches = B * H
    left_mat = left_perm.view(total_batches, N, N)
    right_mat = right_perm.view(total_batches, N, N)

    # Output container (FP16, will be cast back later)
    out_mat_fp16 = torch.empty_like(left_mat, dtype=torch.float16)

    # Strides
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_m, stride_b_n = right_mat.stride()
    # For B^T we swap the first two spatial strides
    stride_b_k = stride_b_m   # step for the reduction dimension (k)
    stride_b_n = stride_b_n   # step for the j dimension

    stride_c_batch, stride_c_m, stride_c_n = out_mat_fp16.stride()

    # Tile sizes – reasonable defaults for the NVIDIA B200 (Ampere)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_batches,
        cdiv(N, BLOCK_M),
        cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat_fp16,
        total_batches,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H] and promote to float32
    out = out_mat_fp16.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 4. Output layer‑norm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b, eps=1e-5)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # shape [B, N, N, C]

    return out
