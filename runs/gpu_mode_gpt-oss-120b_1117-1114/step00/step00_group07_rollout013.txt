import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0) # batch_hidden
    pid1 = tl.program_id(1) # i block
    pid2 = tl.program_id(2) # j block

    # compute batch index b and hidden dim h
    b = pid0 // H
    h = pid0 % H

    # Offsets
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for i and j within N
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointers
        left_ptrs = left_ptr + b*stride_lb + offs_i[:,None]*stride_li + offs_k[None,:]*stride_lk + h*stride_lh
        right_ptrs = right_ptr + b*stride_rb + offs_j[:,None]*stride_ri + offs_k[None,:]*stride_rk + h*stride_rh

        left = tl.load(left_ptrs, mask=mask_i[:,None] & mask_k[None,:], other=0.0)
        right = tl.load(right_ptrs, mask=mask_j[:,None] & mask_k[None,:], other=0.0)

        # dot product: left (M,K) and right (N,K)
        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_ptr + b*stride_ob + offs_i[:,None]*stride_oi + offs_j[None,:]*stride_oj + h*stride_oh
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:,None] & mask_j[None,:])
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)
    pid2 = tl.program_id(2)

    # Compute batch index and hidden dimension index
    b = pid0 // H
    h = pid0 % H

    # Offsets for i and j
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (M, K)
        left_ptrs = left_ptr + b * stride_lb + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk + h * stride_lh
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile: shape (N, K)
        right_ptrs = right_ptr + b * stride_rb + offs_j[:, None] * stride_ri + offs_k[None, :] * stride_rk + h * stride_rh
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # Accumulate dot product over k
        acc += tl.dot(left, right)

    # Store the result
    out_ptrs = out_ptr + b * stride_ob + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj + h * stride_oh
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul (outgoing) operator.
    Performs layer normalization, projections, gating, the O(N^3) pairwise
    multiplication via a tiled Triton GEMM kernel, followed by a second
    LayerNorm, gating and final linear projection.
    """
    import torch
    import torch.nn.functional as F

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Step 1: LayerNorm on input
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Step 2: Linear projections
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Step 3: Optional mask
    if not config.get("nomask", True):
        # mask shape: [B, N, N] -> broadcast to hidden_dim
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_

    # Step 4: Gating
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Step 5: Triton kernel for the O(N^3) reduction
    out = tri_mul_kernel_launcher(left, right)  # shape [B, N, N, hidden_dim]

    # Step 6: Second LayerNorm, gating and final projection
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)
    return out

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
is implemented as a tiled batched GEMM using a Triton kernel.
All other operations (LayerNorms, linear projections, gating) are performed
with regular PyTorch for simplicity.

import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    Computes, for a fixed batch b and hidden index h:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    using a block‑wise dot product.

    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (M, K)
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + offs_i[:, None] * stride_li
            + offs_k[None, :] * stride_lk
            + h * stride_lh
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right tile: shape (N, K)
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + offs_j[:, None] * stride_ri
            + offs_k[None, :] * stride_rk
            + h * stride_rh
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )

        # accumulate dot product over the k dimension
        acc += tl.dot(left, right)

    # write result
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
        + h * stride_oh
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper for the Triton kernel.  Both tensors must have shape
    (B, N, N, H) and be contiguous in memory.
    """
    B, N, _, H = left.shape
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # strides (in elements) for each dimension
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # block sizes – 64 works well for the tested sequence lengths
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                     # batch * hidden
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
    )
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_ri,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for the
    cubic reduction.  The function receives a tuple
        (input_tensor, mask, weights, config)
    and returns the processed tensor of shape [bs, seq_len, seq_len, dim].

    Arguments:
        input_tensor : torch.Tensor [B, N, N, dim]
        mask         : torch.Tensor [B, N, N]  (ignored when config["nomask"] is True)
        weights      : dict of model parameters
        config       : dict with keys "dim", "hidden_dim", "nomask", etc.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ---- Load weights ----------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ---- Step 1: Input LayerNorm -----------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ---- Step 2: Linear projections ---------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ---- Step 3: Optional mask --------------------------------------------
    if not config.get("nomask", True):
        # mask shape: [B, N, N, 1] -> broadcast over hidden_dim
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_

    # ---- Step 4: Gating ---------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ---- Step 5: Triton kernel for the O(N³) reduction --------------------
    out = _tri_mul_triton(left, right)   # shape [B, N, N, hidden_dim]

    # ---- Step 6: Output LayerNorm, gating, final linear -------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # final projection back to dim

    return out
