@triton.jit
def trimul_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0) # i block
    pid_j = tl.program_id(1) # j block

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Offsets for i, j
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    # mask check: only compute if < N
    mask_i = offs_i < N
    mask_j = offs_j < N

    # use grid's block shape for iteration

    # Initialize accumulator for each hidden dim
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32) 

    # but we need hidden dimension; we could accumulate for each hidden dim within a loop. Let's loop over h.

    # Loop over hidden dimensions
    for h in range(0, H):
        # Create accumulator per hidden dimension
        acc_h = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        # iterate over k dimension in tiles
        for k in range(0, tl.cdiv(N, BLOCK_K)):
            # compute K tile start
            k_start = k * BLOCK_K
            offs_k = k_start + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # load left tile: shape (BLOCK_M, BLOCK_K)
            left_ptrs = left_ptr + pid_b*stride_left_b \
                + offs_i[:,None] * stride_left_i \
                + offs_k[None,:] * stride_left_k \
                + h * stride_left_h
            left = tl.load(left_ptrs, mask=mask_i[:,None] & mask_k[None,:], other=0.0)

            # load right tile: shape (BLOCK_N, BLOCK_K) as we need right[j,k]
            # right should be transposed: we need right[j,k] but we want view (N, K) with j along rows, k along columns; So we load shape (BLOCK_N, BLOCK_K)
            right_ptrs = right_ptr + pid_b*stride_right_b \
                + offs_j[:,None] * stride_right_j \
                + offs_k[None,:] * stride_right_k \
                + h * stride_right_h
            right = tl.load(right_ptrs, mask=mask_j[:,None] & mask_k[None,:], other=0.0)

            # compute dot: left (M,K) dot right (N,K) -> (M,N): need right transposed to (K,N) for dot; But tl.dot expects (M,K) @ (K,N). So we need to load right_T: we can store right as (K,N) by swapping dims.
            # Option: load right_T: shape (BLOCK_K, BLOCK_N).
            # So load right tile with indexes swapped: use offsets (k,j). Actually right[j,k] we need right_T[k,j] = right[j,k].
            # So we compute right_T_ptrs using offs_k (rows) and offs_j (columns):
            right_T_ptrs = right_ptr + pid_b*stride_right_b \
                + offs_k[:,None] * stride_right_k \
                + offs_j[None,:] * stride_right_j \
                + h * stride_right_h
            right_T = tl.load(right_T_ptrs, mask=mask_k[:,None] & mask_j[None,:], other=0.0)

            # now right_T shape (BLOCK_K, BLOCK_N)
            # dot:
            acc_h += tl.dot(left, right_T)  # (M,N)
        # store to out for this hidden dimension
        out_ptrs = out_ptr + pid_b*stride_out_b \
            + offs_i[:,None] * stride_out_i \
            + offs_j[None,:] * stride_out_j \
            + h * stride_out_h
        tl.store(out_ptrs, acc_h, mask=mask_i[:,None] & mask_j[None,:])
    # end for h

    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # decode batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # compute i and j ranges
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute base pointers for this batch and hidden
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)

        offs_k = k_start + tl.arange(0, BLOCK_K)

import torch
import triton
import triton.language as tl

@triton.jit
def trimul_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_i = tl.program_id(0)   # block index for output row (i)
    pid_j = tl.program_id(1)   # block index for output col (j)
    pid_bh = tl.program_id(2)  # batch * hidden index

    b = pid_bh // H
    h = pid_bh % H

    # offsets for i and j within this block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this batch and hidden slice
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_tiles):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (M, K)
        left_ptrs = left_base \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0, dtype=tl.float16).to(tl.float32)

        # right tile transposed: shape (K, N)
        right_ptrs = right_base \
            + offs_k[:, None] * stride_right_k \
            + offs_j[None, :] * stride_right_j
        right_T = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :],
                          other=0.0, dtype=tl.float16).to(tl.float32)

        # dot product
        acc += tl.dot(left, right_T)

    # store result
    out_ptrs = out_base \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    """
    Triton-accelerated forward pass of the TriMul (outgoing) update.
    The heavy N^3 contraction is performed on GPU via a custom kernel.
    """

    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # Retrieve weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm on input (last dimension)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # Linear projections
    left_proj = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right_proj = torch.nn.functional.linear(x, right_proj_weight)

    # Gating scalars
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # Apply mask if provided (mask shape [B,N,N])
    # mask may contain 0/1 values; ensure dtype matches x
    if mask is None:
        mask_unsq = torch.ones_like(x[..., :1])
    else:
        mask_unsq = mask.unsqueeze(-1).type_as(x)

    # Apply mask and gates
    left = left_proj * mask_unsq * left_gate
    right = right_proj * mask_unsq * right_gate

    # Cast to half precision for the kernel to reduce memory bandwidth
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    B, N, _, _ = input_tensor.shape

    # Output tensor for the contraction (float32)
    out = torch.empty((B, N, N, hidden_dim), device=device, dtype=torch.float32)

    # Strides (convert to Python ints)
    stride_left_b = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

    stride_right_b = right.stride(0)
    stride_right_j = right.stride(1)
    stride_right_k = right.stride(2)
    stride_right_h = right.stride(3)

    stride_out_b = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    # Triton block sizes (tuned for typical N up to 1024)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),

        left,
        right,
        out,
        B,
        N,
        hidden_dim,

        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Post‑processing: layer‑norm, gating and final linear projection
    out = torch.nn.functional.layer_norm(

        bias=to_out_norm_bias,

    )
    out = out * out_gate
    output = torch.nn.functional.linear(out, to_out_weight)

    return output

import torch

@triton.jit
def trimul_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute the outgoing triangle‑multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    The kernel is launched on a 3‑D grid:
        dim0 → i‑tiles, dim1 → j‑tiles, dim2 → batch*hidden.
    """
    pid_i = tl.program_id(0)          # i‑tile index
    pid_j = tl.program_id(1)          # j‑tile index
    pid_bh = tl.program_id(2)         # combined batch & hidden index

    # Decode batch and hidden dimension from pid_bh
    b = pid_bh // H
    h = pid_bh % H

    # Tile offsets for the i and j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this batch & hidden slice
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_tiles):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i,k] tile  (M × K)
        left_ptrs = left_base \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        ).to(tl.float32)

        # right[j,k] transposed to (k,j) for the dot product (K × N)
