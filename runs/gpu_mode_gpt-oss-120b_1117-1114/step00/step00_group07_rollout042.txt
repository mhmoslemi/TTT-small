def elementwise_mul_kernel(x_ptr, mask_ptr, gate_ptr, out_ptr, N_ELEMS, BLOCK: tl.constexpr):
   pid = tl.program_id(0)
   offs = pid * BLOCK + tl.arange(0, BLOCK)
   mask = offs < N_ELEMS
   a = tl.load(x_ptr + offs, mask=mask, other=0.0)
   m = tl.load(mask_ptr + offs, mask=mask, other=0.0)
   g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
   result = a * m * g
   tl.store(out_ptr + offs, result, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Cast to half for compute
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)
    out_gate = out_gate.to(torch.float16)
    mask = mask.to(torch.float16)

    # Flatten
    left_flat = left.reshape(-1).contiguous()
    right_flat = right.reshape(-1).contiguous()
    left_gate_flat = left_gate.reshape(-1).contiguous()
    right_gate_flat = right_gate.reshape(-1).contiguous()
    mask_flat = mask.reshape(-1).contiguous()

    total = left_flat.shape[0] # B*N*N*hidden_dim
    BLOCK = 1024

    grid = (triton.cdiv(total, BLOCK),)

    # Multiply left
    mul_three_kernel[grid](left_flat.data_ptr(), left_gate_flat.data_ptr(), mask_flat.data_ptr(),
                           left_flat.data_ptr(),
                           total, hidden_dim, BLOCK=BLOCK)
    # Multiply right
    mul_three_kernel[grid](right_flat.data_ptr(), right_gate_flat.data_ptr(), mask_flat.data_ptr(),
                           right_flat.data_ptr(),
                           total, hidden_dim, BLOCK=BLOCK)
    # reshape back
    left = left_flat.view(input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2], hidden_dim)
    right = right_flat.view(input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2], hidden_dim)

    # TriMul via batched matmul (half)
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,2,1).contiguous()
    B = left.shape[0]
    N = left.shape[1]
    left_mat = left_perm.reshape(B*hidden_dim, N, N)
    right_mat = right_perm.reshape(B*hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat)  # half precision
    out = out_mat.reshape(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()  # shape B,N,N,hidden_dim

    # Convert to float32 for norm and final projection
    out_f32 = out.to(torch.float32)

    # to_out_norm
    out_norm = torch.nn.functional.layer_norm(out_f32, (hidden_dim,),
                                              weight=weights["to_out_norm.weight"],
                                              bias=weights["to_out_norm.bias"])

    # multiply by out_gate (convert to float32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # final linear
    out_final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out_final

import triton

@triton.jit
def mul_three_kernel(a_ptr, b_ptr, mask_ptr, out_ptr,
                     N_ELEMS: tl.int64,
                     D: tl.int64,
                     BLOCK: tl.constexpr):
    """
    Elementwise multiply three tensors:
    out = a * b * mask
    mask is broadcasted across the last dimension of size D.
    All tensors are 1-D flattened views (contiguous), dtype can be half.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK, dtype=tl.int64)
    mask = offs < N_ELEMS
    a = tl.load(a_ptr + offs, mask=mask, other=tl.float16(0.0))
    b = tl.load(b_ptr + offs, mask=mask, other=tl.float16(0.0))
    # broadcast mask: each D consecutive elements share the same mask entry
    mask_idx = offs // D
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))
    tl.store(out_ptr + offs, a * b * m, mask=mask)

def custom_kernel(data):
    """
    Triton-accelerated outgoing Triangular Multiplicative Update (TriMul).
    The heavy pairwise reduction is performed via a batched half-precision
    matrix multiply (torch.bmm) while the elementwise mask & gate fusion
    is executed in a Triton kernel.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (float32 -> half)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # Gates (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"])).to(torch.float16)

    # ------------------------------------------------------------------
    # Fuse mask and gates with a Triton kernel.
    #   left  = left  * left_gate  * mask
    #   right = right * right_gate * mask
    # mask is broadcasted on the last dimension (hidden_dim).
    # ------------------------------------------------------------------
    mask = mask.to(torch.float16)

    # flatten tensors to 1-D contiguous for the kernel
    B, N, _, _ = input_tensor.shape
    left_flat = left.reshape(-1).contiguous()
    right_flat = right.reshape(-1).contiguous()
    left_gate_flat = left_gate.reshape(-1).contiguous()
    right_gate_flat = right_gate.reshape(-1).contiguous()
    mask_flat = mask.reshape(-1).contiguous()   # shape B*N*N

    total = left_flat.numel()                   # B * N * N * hidden_dim
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    # left = left * left_gate * mask
    mul_three_kernel[grid](
        left_flat.data_ptr(),
        left_gate_flat.data_ptr(),
        mask_flat.data_ptr(),
        left_flat.data_ptr(),
        total,
        hidden_dim,
        BLOCK=BLOCK,
    )
    # right = right * right_gate * mask
    mul_three_kernel[grid](
        right_flat.data_ptr(),
        right_gate_flat.data_ptr(),
        mask_flat.data_ptr(),
        right_flat.data_ptr(),
        total,
        hidden_dim,
        BLOCK=BLOCK,
    )

    # reshape back to 4-D tensors
    left = left_flat.view(B, N, N, hidden_dim)
    right = right_flat.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # Triangular multiplicative update via batched half‑precision matmul.
    #   out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    # This is equivalent to a batch of (hidden_dim) matrix multiplies.
    # ------------------------------------------------------------------
    # left: (B, N, N, H) -> (B, H, N, N)   (i,k)
    # right: (B, N, N, H) -> (B, H, N, N) with (k,j) order
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()    # (B, H, N, N)

    BH = B * hidden_dim
    left_mat = left_perm.reshape(BH, N, N)
    right_mat = right_perm.reshape(BH, N, N)

    # torch.bmm supports half precision and uses tensor cores.
    out_mat = torch.bmm(left_mat, right_mat)               # (BH, N, N)

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # out shape: (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # Output block: layer norm, gating, final linear projection.
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)   # to_out_norm operates in fp32

    out_norm = F.layer_norm(
        out_f32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # Apply out_gate (already sigmoid‑ed)
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final projection back to original dim
    output = F.linear(out_norm, weights["to_out.weight"])

    return output

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask multiplication and gating (three‑way product)
# ----------------------------------------------------------------------
@triton.jit
def mul_three_kernel(a_ptr, b_ptr, mask_ptr, out_ptr,
                     N_ELEMS, D,
                     BLOCK: tl.constexpr):
    """
    out = a * b * mask
    `mask` is 1‑D (B·N·N) and is broadcast over the last dimension of size D.
    All tensors are flattened contiguous 1‑D views, stored in fp16.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK, dtype=tl.int64)
    mask = offs < N_ELEMS

    a = tl.load(a_ptr + offs, mask=mask, other=tl.float16(0.0))
    b = tl.load(b_ptr + offs, mask=mask, other=tl.float16(0.0))

    # broadcast the mask: each group of D consecutive entries share the same mask value
    mask_idx = offs // D
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))

    tl.store(out_ptr + offs, a * b * m, mask=mask)


def custom_kernel(data):
    """
    Outgoing Triangular Multiplicative Update (TriMul) forward pass.

    The main O(N³·C) reduction is performed as a batched half‑precision
    matrix multiplication (torch.bmm).  The element‑wise mask and gate
    fusion is accelerated with a small Triton kernel, satisfying the
    requirement to use a custom kernel.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # --------------------------------------------------------------
    # 1. Input layer‑norm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Linear projections (float32 → fp16)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # --------------------------------------------------------------
    # 3. Gating (sigmoid of linear, fp16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"])).to(torch.float16)

    # --------------------------------------------------------------
    # 4. Fuse mask and gates with Triton (in‑place)
    # --------------------------------------------------------------
    mask = mask.to(torch.float16)

    B, N, _, _ = input_tensor.shape

    # flatten for the kernel (contiguous views)
    left_flat = left.reshape(-1).contiguous()
    right_flat = right.reshape(-1).contiguous()
    left_gate_flat = left_gate.reshape(-1).contiguous()
    right_gate_flat = right_gate.reshape(-1).contiguous()
    mask_flat = mask.reshape(-1).contiguous()   # shape B·N·N

    total = left_flat.numel()                    # B·N·N·hidden_dim
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    # left = left * left_gate * mask
    mul_three_kernel[grid](
        left_flat.data_ptr(),
        left_gate_flat.data_ptr(),
        mask_flat.data_ptr(),
        left_flat.data_ptr(),
        total,
        hidden_dim,
        BLOCK=BLOCK,
    )
    # right = right * right_gate * mask
    mul_three_kernel[grid](
        right_flat.data_ptr(),
        right_gate_flat.data_ptr(),
        mask_flat.data_ptr(),
        right_flat.data_ptr(),
        total,
        hidden_dim,
        BLOCK=BLOCK,
    )

    # reshape back to 4‑D tensors
    left = left_flat.view(B, N, N, hidden_dim)
    right = right_flat.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5. Core TriMul reduction via batched half‑precision GEMM
    #    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    implemented as a batch of (hidden_dim) matmuls.
    # --------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N) for both operands
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)  i‑k
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)  k‑j

    BH = B * hidden_dim
    left_mat = left_perm.reshape(BH, N, N)   # (B·H, N, N)
    right_mat = right_perm.reshape(BH, N, N) # (B·H, N, N)

    # half‑precision batched matmul (tensor‑core accelerated)
    out_mat = torch.bmm(left_mat, right_mat)  # (B·H, N, N)

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 6. Output block: layer‑norm → out‑gate → final linear (float32)
    # --------------------------------------------------------------
    out_f32 = out.to(torch.float32)

    out_norm = F.layer_norm(
        out_f32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    out_norm = out_norm * out_gate.to(torch.float32)

    # final projection back to original dimension
    output = F.linear(out_norm, weights["to_out.weight"])
