def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_h = tl.program_id(0)   # combined batch*hidden index
    pid_m = tl.program_id(1)  # tile over M (i)
    pid_n = tl.program_id(2)  # tile over N (j)

    b = batch_h // H
    h = batch_h % H

    # Compute offsets:
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Mask for within bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Compute pointers for output tile

"""
TriMul (outgoing) forward pass with a Triton‑accelerated batched GEMM.

Algorithm
---------
1. Layer‑normalize the input tensor x ∈ ℝ^{B×N×N×D}.
2. Project x to a hidden dimension H (left_proj, right_proj) without bias.
3. If a mask is provided, multiply the projected tensors by mask[...,None].
4. Compute three sigmoid gates (left_gate, right_gate, out_gate) via linear layers.
5. Apply left/right gates element‑wise: left←left⋅left_gate, right←right⋅right_gate.
6. Compute the O(N³) reduction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a batch of H independent matrix multiplications per batch using a
   Triton kernel (batched GEMM on the (i,k)×(j,k) axes).
7. Layer‑normalize the result over the hidden dimension H.
8. Multiply by the out_gate.
9. Project back to the original dimension D with a final linear layer.

Only the heavy O(N³) part (step 6) runs on Triton; all other
operations are performed with PyTorch.  The final output is float‑32.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_gemm_kernel(
    left_ptr,                    # *fp16: [B, N, N, H]
    right_ptr,                   # *fp16: [B, N, N, H]
    out_ptr,                     # *fp32: [B, N, N, H]
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for TriMul.
    Computes, for each batch b and hidden index h:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    """
    batch_h = tl.program_id(0)      # combined batch * hidden index
    pid_m = tl.program_id(1)        # tile over i (M dimension)
    pid_n = tl.program_id(2)        # tile over j (N dimension)

    # Recover batch and hidden indices
    b = batch_h // H
    h = batch_h % H

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointer to the output tile
    out_tile_ptr = (
        out_ptr
        + b * stride_ob
        + h * stride_oh
        + offs_m[:, None] * stride_oi
        + offs_n[None, :] * stride_oj
    )
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        left_tile_ptr = (
            left_ptr
            + b * stride_lb
            + h * stride_lh
            + offs_m[:, None] * stride_li
            + offs_k[None, :] * stride_lk
        )
        left = tl.load(
            left_tile_ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # Load right tile (transposed for GEMM): shape (BLOCK_K, BLOCK_N)
        right_tile_ptr = (
            right_ptr
            + b * stride_rb
            + h * stride_rh
            + offs_k[:, None] * stride_rk
            + offs_n[None, :] * stride_rj
        )
        right = tl.load(
            right_tile_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # Accumulate product
        acc += tl.dot(left, right)

    # Write the computed tile
    tl.store(out_tile_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, D]
            - mask: torch.Tensor, shape [B, N, N] (binary 0/1)
            - weights: dict of weight tensors
            - config: dict with keys "dim", "hidden_dim", "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]  # hidden dimension (e.g., 128)

    # -----------------------------------------------------------------
    # Unpack weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(D,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B,N,N,D]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # 3. Optional mask
    # -----------------------------------------------------------------
    if not config.get("nomask", False):
        mask_ = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4. Gates
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Triton batched GEMM over the hidden dimension
    # -----------------------------------------------------------------
    # Cast to half‑precision for the kernel (accumulates in FP32)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    out_fp32 = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # Strides (make them Python ints)
    stride_lb, stride_li, stride_lk, stride_lh = left_fp16.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right_fp16.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out_fp32.stride()

    # Tile sizes – chosen for good occupancy on B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                     # batch * hidden
        triton.cdiv(N, BLOCK_M),  # i‑tiles
        triton.cdiv(N, BLOCK_N),  # j‑tiles
    )
    trimul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Output LayerNorm over hidden dimension
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out_fp32,
        normalized_shape=(H,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # [B,N,N,H]

    # -----------------------------------------------------------------
    # 7. Apply out_gate
    # -----------------------------------------------------------------
    out = out * out_gate

    # -----------------------------------------------------------------
    # 8. Final linear projection back to D
    # -----------------------------------------------------------------
    output = torch.nn.functional.linear(out, to_out_weight)  # [B,N,N,D]

    return output
