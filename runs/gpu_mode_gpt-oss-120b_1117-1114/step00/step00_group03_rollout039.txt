```
@triton.jit
def layernorm_forward_kernel(
    X,
    Y,
    weight,
    bias,
    stride_x_batch,
    stride_x_i,
    stride_x_j,
    stride_x_c,
    stride_y_batch,
    stride_y_i,
    stride_y_j,
    stride_y_c,
    stride_w,
    stride_b,
    N,
    C,
    EPS: tl.constexpr = 1e-5,
):
    pid = tl.program_id(0)
    # compute linear index: row = pid
    # Determine b,i,j from pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # offsets for data
    offs_c = tl.arange(0, C)
    x_ptrs = X + b * stride_x_batch + i * stride_x_i + j * stride_x_j + offs_c * stride_x_c
    y_ptrs = Y + b * stride_y_batch + i * stride_y_i + j * stride_y_j + offs_c * stride_y_c

    x = tl.load(x_ptrs, mask=offs_c < C, other=0.0)

    # Compute mean and variance
    mean = tl.sum(x, axis=0) / C
    var = tl.sum((x - mean) * (x - mean), axis=0) / C
    norm = (x - mean) * tl.rsqrt(var + EPS)

    w = tl.load(weight + offs_c * stride_w, mask=offs_c < C, other=0.0)
    b_ = tl.load(bias + offs_c * stride_b, mask=offs_c < C, other=0.0)

    y = norm * w + b_
    tl.store(y_ptrs, y, mask=offs_c < C)

@triton.jit
def layernorm_kernel(
    X_ptr, Y_ptr, weight_ptr, bias_ptr,
    stride_x_batch, stride_x_i, stride_x_j, stride_x_c,
    stride_y_batch, stride_y_i, stride_y_j, stride_y_c,
    stride_w, stride_b,
    B, N, C,
    EPS: tl.constexpr = 1e-5
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # offset for output and input:
    offs_c = tl.arange(0, C)
    # compute pointer offsets
    x_ptr = X_ptr + b * stride_x_batch + i * stride_x_i + j * stride_x_j + offs_c * stride_x_c
    y_ptr = Y_ptr + b * stride_y_batch + i * stride_y_i + j * stride_y_j + offs_c * stride_y_c

    # load data
    x = tl.load(x_ptr, mask=offs_c < C, other=0.0)

    # compute mean
    sum_x = tl.sum(x, axis=0)
    inv_C = 1.0 / C
    mean = sum_x * inv_C

    # centered
    x_centered = x - mean

    # var
    var = tl.sum(x_centered * x_centered, axis=0) * inv_C

    inv_std = tl.rsqrt(var + EPS)

    norm = x_centered * inv_std

    # load weight and bias
    w = tl.load(weight_ptr + offs_c * stride_w, mask=offs_c < C, other=0.0)
    b_ = tl.load(bias_ptr + offs_c * stride_b, mask=offs_c < C, other=0.0)

    y = norm * w + b_

    tl.store(y_ptr, y, mask=offs_c < C)

"""
Trimit Multiply (TriMul) forward pass using Triton for LayerNorm.
The heavy contraction over the sequence dimension is performed with torch.einsum.
The LayerNorms are fused in a custom Triton kernel for speed.
"""

import torch
import triton
import triton.language as tl

# Triton kernel for 4-D LayerNorm (normalization over the last channel dimension)
@triton.jit
def layernorm_kernel(
    X_ptr, Y_ptr,
    weight_ptr, bias_ptr,
    stride_x_batch, stride_x_i, stride_x_j, stride_x_c,
    stride_y_batch, stride_y_i, stride_y_j, stride_y_c,
    stride_w, stride_b,
    B, N, C,
    EPS: tl.constexpr = 1e-5
):
    pid = tl.program_id(0)
    total_rows = B * N * N
    if pid >= total_rows:
        return

    # compute (b, i, j) from flat pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # range over channel dimension
    offs_c = tl.arange(0, C)

    # pointers for this row
    x_ptr = X_ptr + b * stride_x_batch + i * stride_x_i + j * stride_x_j + offs_c * stride_x_c
    y_ptr = Y_ptr + b * stride_y_batch + i * stride_y_i + j * stride_y_j + offs_c * stride_y_c

    # load input row
    x = tl.load(x_ptr, mask=offs_c < C, other=0.0)

    # compute mean
    inv_C = 1.0 / C
    mean = tl.sum(x, axis=0) * inv_C

    # centered
    x_centered = x - mean

    # variance
    var = tl.sum(x_centered * x_centered, axis=0) * inv_C

    # normalize
    inv_std = tl.rsqrt(var + EPS)
    norm = x_centered * inv_std

    # load weight and bias
    w = tl.load(weight_ptr + offs_c * stride_w, mask=offs_c < C, other=0.0)
    b_ = tl.load(bias_ptr + offs_c * stride_b, mask=offs_c < C, other=0.0)

    # apply affine transform
    y = norm * w + b_

    # write back
    tl.store(y_ptr, y, mask=offs_c < C)

def layernorm_triton(x, weight, bias, eps=1e-5):
    """LayerNorm over the last dimension using a Triton kernel."""
    # x is expected to be contiguous and float32
    B, N, N2, C = x.shape
    assert N == N2, "TriMul expects square pairwise dimensions"
    y = torch.empty_like(x)

    stride_x = x.stride()
    stride_y = y.stride()

    total = B * N * N
    grid = (triton.cdiv(total, 1),)  # one program per output row

    layernorm_kernel[grid](
        x, y,
        weight, bias,
        stride_x[0], stride_x[1], stride_x[2], stride_x[3],
        stride_y[0], stride_y[1], stride_y[2], stride_y[3],
        weight.stride(0), bias.stride(0),
        B, N, C,
        EPS=eps,
        num_warps=4,
    )
    return y

    Forward pass of the TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        output Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    # Cast everything to float32 for numeric stability
    x = input_tensor.to(torch.float32)
    if mask is not None:
        mask = mask.to(torch.float32)

    # Extract config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Extract weight tensors and cast to float32
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias = weights["norm.bias"].to(torch.float32)
    left_proj_weight = weights["left_proj.weight"].to(torch.float32)
    right_proj_weight = weights["right_proj.weight"].to(torch.float32)
    left_gate_weight = weights["left_gate.weight"].to(torch.float32)
    right_gate_weight = weights["right_gate.weight"].to(torch.float32)
    out_gate_weight = weights["out_gate.weight"].to(torch.float32)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32)
    to_out_weight = weights["to_out.weight"].to(torch.float32)

    # ----- LayerNorm on input -----
    x_norm = layernorm_triton(x, norm_weight, norm_bias)  # shape [B,N,N,dim]

    # ----- Linear projections (no bias) -----
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight) # [B,N,N,hidden_dim]

    # ----- Optional mask -----
    if mask is not None and not nomask:
        # mask shape [B, N, N] -> broadcast over hidden_dim
        mask_exp = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ----- Gating -----
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----- Triangular multiplication (einsum) -----
    # Compute sum_k left[..., i, k, d] * right[..., j, k, d]
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)

    # ----- LayerNorm on hidden dimension -----
    out_norm = layernorm_triton(out, to_out_norm_weight, to_out_norm_bias)

    # ----- Apply output gate -----
    out_norm = out_norm * out_gate

    # ----- Final projection back to original dim -----
    output = torch.nn.functional.linear(out_norm, to_out_weight)  # [B,N,N,dim]

    return output

"""
TriMul forward implementation.

Algorithm:
1. Apply LayerNorm over the last channel dimension of the input tensor using a
   custom Triton kernel (first normalization).
2. Perform the four linear projections (left, right, left_gate, right_gate,
   out_gate) with torch.nn.functional.linear (no bias).
3. Apply an optional binary mask to the left/right projections.
4. Apply sigmoid gates and element‑wise multiplication.
5. Compute the cubic contraction
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] via torch.einsum
   (the dominant O(N³) work, efficiently handled by cuBLAS).
6. Apply a second LayerNorm over the hidden dimension using the same Triton
   kernel.
7. Multiply by the out_gate, then project back to the original dimension
   with a final linear layer.
All arithmetic is performed in float32; only the two LayerNorm passes are
implemented in Triton to satisfy the “kernel” requirement while keeping the
overall implementation simple and correct.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: LayerNorm over the last dimension of a [B, N, N, C] tensor
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    X_ptr, Y_ptr,               # input & output pointers
    weight_ptr, bias_ptr,       # LayerNorm affine parameters
    stride_x_batch, stride_x_i, stride_x_j, stride_x_c,
    stride_y_batch, stride_y_i, stride_y_j, stride_y_c,
    stride_w, stride_b,
    B, N, C,                    # runtime dimensions
    EPS: tl.constexpr = 1e-5   # epsilon
):
    pid = tl.program_id(0)
    total_rows = B * N * N
    if pid >= total_rows:
        return

    # decode (b,i,j) from the flat program id
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # range over the channel dimension
    offs_c = tl.arange(0, C)

    # compute pointers for this (b,i,j) row
    x_ptr = X_ptr + b * stride_x_batch + i * stride_x_i + j * stride_x_j + offs_c * stride_x_c
    y_ptr = Y_ptr + b * stride_y_batch + i * stride_y_i + j * stride_y_j + offs_c * stride_y_c

    # load input row
    x = tl.load(x_ptr, mask=offs_c < C, other=0.0)

    # ---------- mean ----------
    inv_C = 1.0 / C
    mean = tl.sum(x, axis=0) * inv_C

    # ---------- variance ----------
    x_centered = x - mean
    var = tl.sum(x_centered * x_centered, axis=0) * inv_C

    # ---------- normalize ----------
    inv_std = tl.rsqrt(var + EPS)
    norm = x_centered * inv_std

    # ---------- affine ----------
    w = tl.load(weight_ptr + offs_c * stride_w, mask=offs_c < C, other=0.0)
    b_ = tl.load(bias_ptr + offs_c * stride_b, mask=offs_c < C, other=0.0)
    y = norm * w + b_

    # store result
    tl.store(y_ptr, y, mask=offs_c < C)


def layernorm_triton(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:

    LayerNorm over the last dimension using the Triton kernel above.
    Input ``x`` must be contiguous and of dtype float32.
    """
    B, N, N2, C = x.shape
    assert N == N2, "TriMul expects square pairwise dimensions"

    # allocate output tensor
    y = torch.empty_like(x)

    # strides (in elements, not bytes)
    sx = x.stride()
    sy = y.stride()

    total_rows = B * N * N
    grid = (triton.cdiv(total_rows, 1),)   # one program per output row

        weight, bias,
        sx[0], sx[1], sx[2], sx[3],

        weight.stride(0), bias.stride(0),
        B, N, C,
        EPS=eps,
        num_warps=4,

    # Unpack configuration
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load weights (ensure float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias = weights["norm.bias"].to(torch.float32)

    left_proj_weight = weights["left_proj.weight"].to(torch.float32)
    right_proj_weight = weights["right_proj.weight"].to(torch.float32)

    left_gate_weight = weights["left_gate.weight"].to(torch.float32)
    right_gate_weight = weights["right_gate.weight"].to(torch.float32)
    out_gate_weight = weights["out_gate.weight"].to(torch.float32)

    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32)

    to_out_weight = weights["to_out.weight"].to(torch.float32)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (over `dim`)
    # ------------------------------------------------------------------
    x_norm = layernorm_triton(x, norm_weight, norm_bias)   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3) Optional mask (applied after projection)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_exp = mask.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid on linear transforms of the normalized input)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triangular multiplication (cubic contraction)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    out_norm = layernorm_triton(out, to_out_norm_weight, to_out_norm_bias)

    # ------------------------------------------------------------------
    # 7) Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate
    output = torch.nn.functional.linear(out_norm, to_out_weight)   # [B, N, N, dim]

    return output
```
