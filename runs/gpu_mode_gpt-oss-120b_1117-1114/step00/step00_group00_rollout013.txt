@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                           M, N, K,
                           stride_am, stride_ak, stride_ab,  # one stride for batch offset? Actually we can treat stride_ab as stride between matrices (batch)
                           stride_bk, stride_bn, stride_bb,
                           stride_cm, stride_cn, stride_cb,
                           BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_m = tl.program_id(0) # row block
    pid_n = tl.program_id(1) # col block
    pid_b = tl.program_id(2) # batch index

    # Compute global offsets for the start of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute pointers for A and B
    a_ptrs = A_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + pid_b * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load A and B tiles
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)

        # Cast to appropriate compute type if needed
        a = a.to(tl.float16) if a.dtype != tl.float16 else a
        b = b.to(tl.float16) if b.dtype != tl.float16 else b
        # Multiply-accumulate
        acc += tl.dot(a, b, out_dtype=tl.float32)

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back to C
    c_ptrs = C_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C = A @ B  where A,B are fp16 and accumulation is fp32."""
    pid_b = tl.program_id(2)   # batch index (B * hidden_dim)
    pid_m = tl.program_id(0)   # row tile
    pid_n = tl.program_id(1)   # column tile

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Pointers for current tile
        a_ptrs = (A_ptr + pid_b * stride_ab
                  + offs_m[:, None] * stride_am
                  + offs_k[None, :] * stride_ak)
        b_ptrs = (B_ptr + pid_b * stride_bb
                  + offs_k[:, None] * stride_bk
                  + offs_n[None, :] * stride_bn)

        # Load with bounds checking
        a = tl.load(a_ptrs, mask=(mask_m[:, None] & (offs_k[None, :] < K)), other=0.0)
        b = tl.load(b_ptrs, mask=((offs_k[:, None] < K) & mask_n[None, :]), other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Write back
    c_ptrs = (C_ptr + pid_b * stride_cb
              + offs_m[:, None] * stride_cm
              + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be ignored if config["nomask"] is True)
            - weights: dict of parameter tensors
            - config: dict containing "dim", "hidden_dim", "nomask" etc.

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ---------------------- Hyper‑parameters ----------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------------------- LayerNorm on input --------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ---------------------- Linear projections --------------------
    left_proj_w = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]                # (dim, hidden_dim)

    left = F.linear(x, left_proj_w)        # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)

    if not nomask:
        # mask broadcasted on the channel dimension
        mask_f = mask.to(dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ---------------------- Cast to fp16 for GEMM -----------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # transpose right for the "outgoing" contraction (B @ B^T)
    right_t_fp16 = right_fp16.transpose(-2, -1)

    B, N, _, _ = left_fp16.shape  # B=batch_size, N=seq_len

    # Reshape to (B*hidden, N, N) for batched GEMM
    left_reshape = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_reshape = right_t_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Allocate output buffer in fp16
    out_reshape = torch.empty_like(left_reshape)

    # ---------------------- Triton GEMM launch -------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Strides for left (A)
    stride_am = left_reshape.stride(1)  # row stride (N)
    stride_ak = left_reshape.stride(2)  # col stride (1)
    stride_ab = left_reshape.stride(0)  # batch stride (N * N)

    # Strides for right (B) – note B is already transposed
    stride_bk = right_reshape.stride(1)  # now row stride = 1 (because of transpose)
    stride_bn = right_reshape.stride(2)  # column stride = N
    stride_bb = right_reshape.stride(0)  # batch stride = N * N

    # Strides for output (C)
    stride_cm = out_reshape.stride(1)    # row stride = N
    stride_cn = out_reshape.stride(2)    # col stride = 1
    stride_cb = out_reshape.stride(0)    # batch stride = N * N

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * hidden_dim,            # batch*hidden
    )
    batched_gemm_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        N, N, N,
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ---------------------- Reshape back to original layout -----
    out_fp16 = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out_fp16.to(torch.float32)

    # ---------------------- Output LayerNorm and gating ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate

    # Final linear projection back to dim
    output = F.linear(out, to_out_w)   # shape [B, N, N, dim]

    return output
