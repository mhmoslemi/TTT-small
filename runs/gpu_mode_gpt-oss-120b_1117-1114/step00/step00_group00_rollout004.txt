import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(A, B, C,
                        M, N, K,
                        stride_Ab, stride_Am, stride_Ak,
                        stride_Bb, stride_Bk, stride_Bn,
                        stride_Cb, stride_Cm, stride_Cn,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
   pid = tl.program_id(0) # block row
   pid_n = tl.program_id(1) # block col
   pid_b = tl.program_id(2) # batch id
   # offsets
   offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
   offs_k = tl.arange(0, BLOCK_K)
   # compute pointer offsets with batching
   a_ptrs = A + pid_b * stride_Ab + (offs_m[:, None] * stride_Am) + (offs_k[None, :] * stride_Ak)
   b_ptrs = B + pid_b * stride_Bb + (offs_k[:, None] * stride_Bk) + (offs_n[None, :] * stride_Bn)
   # accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
   # iterate over K dimension
   for k in range(0, tl.cdiv(K, BLOCK_K)):
       a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)
       b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)
       acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) dot (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

"""
TriMul (outgoing) forward pass with Triton-accelerated contraction.

The module performs:
  1. LayerNorm on the input tensor.
  2. Linear projections + gating (left/right) and optional masking.
  3. Core N‑dimensional contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
     implemented as a batched GEMM using a custom Triton kernel.
  4. LayerNorm on the contracted tensor, elementwise out‑gate scaling,
     and a final linear projection back to the original dimension.

All heavy tensor operations stay in PyTorch except the O(N³·D) contraction,
which is the performance bottleneck and is executed on the GPU via Triton.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: Batched GEMM (C = A @ B) where A, B, C have a common batch dim.
# Shapes: A [B, M, K], B [B, K, N], C [B, M, N]
# ----------------------------------------------------------------------------- #
@triton.jit
def batched_gemm_kernel(
    A, B, C,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)        # block row
    pid_n = tl.program_id(1)        # block col
    pid_b = tl.program_id(2)        # batch id

    # offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # pointers for A and B with batch offset
    a_ptrs = A + pid_b * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B + pid_b * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # load tiles, masking out-of-bounds elements
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # write back C
    c_ptrs = C + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


# ----------------------------------------------------------------------------- #
# Host‑side entry point
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Compute the outgoing TriMul forward pass.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be all‑ones when `nomask` is True)
            - weights: dict of torch.Tensor containing all linear/ln weights & biases
            - config: dict with keys:
                * "dim" (int) – input channel dimension
                * "hidden_dim" (int) – projection dimension
                * "nomask" (bool) – if True mask can be ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------------- #
    # unpack
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ensure contiguous and float32
    x = input_tensor.contiguous().float()
    B, N, _, _ = x.shape

    # --------------------------------------------------------------------- #
    # 1) LayerNorm on the input
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------------- #
    # 2) Linear projections + gates
    # --------------------------------------------------------------------- #
    left_proj = F.linear(x, weights["left_proj.weight"])          # [B, N, N, hidden_dim]
    right_proj = F.linear(x, weights["right_proj.weight"])

    # optional mask (broadcast on hidden_dim)
    if not nomask:
        mask_unsq = mask.to(x.dtype).unsqueeze(-1)               # [B, N, N, 1]
        left_proj = left_proj * mask_unsq
        right_proj = right_proj * mask_unsq

    # gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # --------------------------------------------------------------------- #
    # 3) Core contraction using Triton (batched GEMM)
    # --------------------------------------------------------------------- #
    # permute so hidden_dim becomes a batch dimension
    # left:   [B, N, N, hidden_dim] -> [B, hidden_dim, N, N] (i,k)
    # right:  [B, N, N, hidden_dim] -> [B, hidden_dim, N, N] (k,j) after transpose
    left_perm = left.permute(0, 3, 1, 2).contiguous()            # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()          # (B, H, N, N) with k,j ordering

    # fuse batch and hidden_dim into a single batch for the kernel
    batch_dim = B * hidden_dim
    left_mat = left_perm.view(batch_dim, N, N)                   # (B*H, M=N, K=N)
    right_mat = right_perm.view(batch_dim, N, N)                 # (B*H, K=N, N)

    # output placeholder
    out_mat = torch.empty_like(left_mat)

    # strides (batch, row, col)
    stride_Ab, stride_Am, stride_Ak = left_mat.stride()
    stride_Bb, stride_Bk, stride_Bn = right_mat.stride()
    stride_Cb, stride_Cm, stride_Cn = out_mat.stride()

    # block sizes – tuned for typical N (e.g., 256‑1024) and hidden_dim=128
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch_dim,                 # batch
    )
    batched_gemm_kernel[grid](
        left_mat, right_mat, out_mat,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 4) Post‑contraction LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
