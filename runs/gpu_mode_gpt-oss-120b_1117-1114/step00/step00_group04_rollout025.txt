@triton.jit
def tri_mul_kernel(a_ptr, b_ptr, c_ptr,
                   N,
                   stride_am, stride_ai, stride_ak,
                   stride_bm, stride_bi, stride_bk,
                   stride_cm, stride_ci, stride_cj,
                   BLOCK_M: tl.constexpr,
                   BLOCK_N: tl.constexpr,
                   BLOCK_K: tl.constexpr):
    # program ids
    pid_m = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute offsets for i and j
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for OOB
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Compute base pointer offsets for each block
    a_ptrs = a_ptr + pid_m * stride_am + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_m * stride_bm + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bi
    # Explanation: b_ptr loads B as (k,j) shape.

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension
    for k in range(0, N, BLOCK_K):
        # Compute current k offsets
        cur_k = k + offs_k
        # Mask for K dimension
        mask_k = cur_k < N

        # Load tiles
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # get config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # extract weights and cast
    norm_weight = weights["norm.weight"].to(device=device, dtype=dtype)
    norm_bias = weights["norm.bias"].to(device=device, dtype=dtype)
    left_proj_weight = weights["left_proj.weight"].to(device=device, dtype=dtype)
    right_proj_weight = weights["right_proj.weight"].to(device=device, dtype=dtype)
    left_gate_weight = weights["left_gate.weight"].to(device=device, dtype=dtype)
    right_gate_weight = weights["right_gate.weight"].to(device=device, dtype=dtype)
    out_gate_weight = weights["out_gate.weight"].to(device=device, dtype=dtype)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device=device, dtype=dtype)
    to_out_weight = weights["to_out.weight"].to(device=device, dtype=dtype)

    # LayerNorm on input (dim)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    # Projections
    # flatten for linear
    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)  # (B*N*N, dim)

    # Left proj
    left = torch.nn.functional.linear(x_flat, left_proj_weight)  # (B*N*N, hidden)
    left = left.view(B, N, N, hidden_dim)

    # Right proj
    right = torch.nn.functional.linear(x_flat, right_proj_weight)
    right = right.view(B, N, N, hidden_dim)

    # Gates (pre-sigmoid)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, left_gate_weight)).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, right_gate_weight)).view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, out_gate_weight)).view(B, N, N, hidden_dim)

    # mask handling
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)  # shape (B,N,N,1)
        left = left * mask
        right = right * mask

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # heavy TriMul = einsum over k using Triton kernel
    # reshape tensors for kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    M = B * hidden_dim
    left_flat = left_perm.view(M, N, N)
    right_flat = right_perm.view(M, N, N)

    # output placeholder
    out_flat = torch.empty_like(left_flat)

    # kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (M,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,hidden)

    # to_out_norm (LayerNorm over hidden dim)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # multiply with out_gate
    out = out * out_gate

    # final linear to dim
    out_flat2 = out.reshape(-1, hidden_dim)
    out = torch.nn.functional.linear(out_flat2, to_out_weight)  # (B*N*N, dim)
    out = out.view(B, N, N, dim)

    return out

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(a_ptr, b_ptr, c_ptr,
                   N,
                   stride_am, stride_ai, stride_ak,
                   stride_bm, stride_bi, stride_bk,
                   stride_cm, stride_ci, stride_cj,
                   BLOCK_M: tl.constexpr,
                   BLOCK_N: tl.constexpr,
                   BLOCK_K: tl.constexpr):
    """Compute out[m,i,j] = sum_k a[m,i,k] * b[m,j,k].

    a and b have shape (M, N, N). The reduction is performed over the last
    dimension k. The kernel tiles the i and j axes and iterates over k in
    blocks of size BLOCK_K.
    """
    pid_m = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # offsets within the tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for start_k in range(0, N, BLOCK_K):
        cur_k = start_k + offs_k
        mask_k = cur_k < N

        # load a[m,i,k]
        a = tl.load(
            a_ptr
            + pid_m * stride_am
            + offs_i[:, None] * stride_ai
            + cur_k[None, :] * stride_ak,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load b[m,j,k] but interpreted as (k,j) for dot product
        b = tl.load(
            b_ptr
            + pid_m * stride_bm
            + cur_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bi,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # write result
    mask_ij = mask_i[:, None] & mask_j[None, :]
    c = c_ptr + pid_m * stride_cm + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c, acc, mask=mask_ij)

    module (the “outgoing” version from AlphaFold3). The heavy pairwise reduction

    everything else (layer_norm, linear projections, gating, final projection)
    is handled with PyTorch primitives.
    """

    right_proj_weight = weights["right_proj.weight"].to(device=device, dtype=dtype)

    left_gate_weight = weights["left_gate.weight"].to(device=device, dtype=dtype)

    out_gate_weight = weights["out_gate.weight"].to(device=device, dtype=dtype)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device=device, dtype=dtype)
    to_out_weight = weights["to_out.weight"].to(device=device, dtype=dtype)

    # ----- LayerNorm on input -----------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    B, N, _, _ = x.shape

    # ----- Linear projections (no bias) --------------------------------
    x_flat = x.reshape(-1, dim)  # (B*N*N, dim)

    left = torch.nn.functional.linear(x_flat, left_proj_weight).view(B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, right_proj_weight).view(B, N, N, hidden_dim)

    # ----- Gating -------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, left_gate_weight)
    ).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, right_gate_weight)
    ).view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, out_gate_weight)
    ).view(B, N, N, hidden_dim)

    # ----- Apply optional mask -----------------------------------------
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ----- Apply gates --------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ----- Core O(N^3) reduction via Triton kernel ----------------------
    # Rearrange for kernel: (B, hidden, N, N) -> (M, N, N) where M = B*hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    M = B * hidden_dim
    left_flat = left_perm.view(M, N, N)
    right_flat = right_perm.view(M, N, N)

    out_flat = torch.empty_like(left_flat)

    # Launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        M,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,hidden)

    # ----- Output layernorm ------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # ----- Apply output gate ----------------------------------------------
    out = out * out_gate

    # ----- Final linear projection -----------------------------------------
    out = torch.nn.functional.linear(out.reshape(-1, hidden_dim), to_out_weight).view(B, N, N, dim)

    return out

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr,
    b_ptr,
    c_ptr,
    N,
    stride_am,
    stride_ai,
    stride_ak,
    stride_bm,
    stride_bi,
    stride_bk,
    stride_cm,
    stride_ci,
    stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute   out[m, i, j] = Σ_k a[m, i, k] * b[m, j, k]

    a, b, c have shape (M, N, N) where M = batch * hidden_dim.
    The reduction over k is tiled with size BLOCK_K, while i and j are tiled
    with BLOCK_M / BLOCK_N respectively.
    """
    pid_m = tl.program_id(0)        # M dimension (batch * hidden)
    pid_i = tl.program_id(1)        # tile along i
    pid_j = tl.program_id(2)        # tile along j

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for start_k in range(0, N, BLOCK_K):
        cur_k = start_k + offs_k
        mask_k = cur_k < N

        # a[m, i, k]  -> shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_m * stride_am
            + offs_i[:, None] * stride_ai
            + cur_k[None, :] * stride_ak,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # b[m, j, k] interpreted as (k, j) for dot product -> shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + pid_m * stride_bm
            + cur_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bi,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # store result
    mask_ij = mask_i[:, None] & mask_j[None, :]
    c = (
        c_ptr
        + pid_m * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(c, acc, mask=mask_ij)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” Triangle Multiplicative Update
    (TriMul) module from AlphaFold3. The heavy O(N³) reduction over the sequence
    dimension k is performed by a custom Triton kernel; the remaining
    normalisations, projections, gating and final linear projection are done
    with PyTorch primitives.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=dtype)
    norm_bias = weights["norm.bias"].to(device=device, dtype=dtype)

    left_proj_weight = weights["left_proj.weight"].to(device=device, dtype=dtype)
    right_proj_weight = weights["right_proj.weight"].to(device=device, dtype=dtype)

    left_gate_weight = weights["left_gate.weight"].to(device=device, dtype=dtype)
    right_gate_weight = weights["right_gate.weight"].to(device=device, dtype=dtype)
    out_gate_weight = weights["out_gate.weight"].to(device=device, dtype=dtype)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device=device, dtype=dtype)
    to_out_weight = weights["to_out.weight"].to(device=device, dtype=dtype)

    # ------------------------------------------------------------------
    # Input layer‑norm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    x_flat = x.reshape(-1, dim)  # (B*N*N, dim)

    left = torch.nn.functional.linear(x_flat, left_proj_weight).view(
        B, N, N, hidden_dim
    )
    right = torch.nn.functional.linear(x_flat, right_proj_weight).view(
        B, N, N, hidden_dim
    )

    # ------------------------------------------------------------------
    # Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, left_gate_weight)
    ).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, right_gate_weight)
    ).view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, out_gate_weight)
    ).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # Optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core heavy O(N³) part via Triton
    # ------------------------------------------------------------------
    # reshape to (M, N, N) where M = B * hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    M = B * hidden_dim
    left_flat = left_perm.view(M, N, N)
    right_flat = right_perm.view(M, N, N)

    out_flat = torch.empty_like(left_flat)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        M,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # Output layer‑norm, gate and final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate

    out = torch.nn.functional.linear(
        out.reshape(-1, hidden_dim), to_out_weight
    ).view(B, N, N, dim)

    return out
