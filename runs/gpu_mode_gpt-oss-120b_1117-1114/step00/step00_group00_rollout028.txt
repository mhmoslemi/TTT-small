@triton.jit
def batched_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)  # batch idx
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(2)  # block col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    a_ptrs = A + pid_b * stride_ab + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    # We'll load K tiles in loop; need to compute offset pointer gradually.
    # Actually compute base pointer for current k tile later.
    b_ptrs = B + pid_b * stride_bb + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

"""
Custom Triton implementation of the "outgoing" Triangle Multiplicative Update (TriMul)
operator.

Algorithm
---------
1. Layer‑norm the input tensor `x` (last dimension = dim).
2. Linear projections `left` and `right` (dim → hidden_dim) without bias.
3. Apply the pairwise mask and gating (sigmoid) per position.
4. Compute the heavy O(N³) contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using a fused batched matrix‑multiply kernel written in Triton.
   The contraction is a batched GEMM: for each (batch,head) we multiply
   `L` (N×N) with `Rᵀ` (N×N).  The kernel tiles the M,N,K dimensions
   (BLOCK_M, BLOCK_N, BLOCK_K) and accumulates in FP32 while loading FP16.
5. Apply a second layer‑norm on the result, multiply by `out_gate`,
   and project back to the original dimension with a final linear layer.

The Triton kernel only covers step 4; all other operations are performed
with native PyTorch to keep the implementation simple and correct.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ B   (A: M×K, B: K×N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_am, stride_ak, stride_ab,      # A: (batch, M, K)
    stride_bk, stride_bn, stride_bb,      # B: (batch, K, N)
    stride_cm, stride_cn, stride_cb,      # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)               # batch * head index
    pid_m = tl.program_id(1)               # tile row
    pid_n = tl.program_id(2)               # tile column

    # Offsets inside the M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch/head
    a_ptr = A + pid_b * stride_ab
    b_ptr = B + pid_b * stride_bb
    c_ptr = C + pid_b * stride_cb

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile of A (M×K) – dtype FP16
        a = tl.load(
            a_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load a tile of B (K×N) – dtype FP16
        b = tl.load(
            b_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Dot product (FP16 × FP16 → FP32 accumulation)
        acc += tl.dot(a, b)

    # Write back the result tile
    tl.store(
        c_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )

# ----------------------------------------------------------------------
# Helper to launch the batched GEMM kernel
# ----------------------------------------------------------------------
def batched_matmul(A: torch.Tensor, B: torch.Tensor, C: torch.Tensor,
                   BLOCK_M: int = 64, BLOCK_N: int = 64, BLOCK_K: int = 64) -> None:
    """
    A: (B, M, K)   FP16
    B: (B, K, N)   FP16 (already transposed as needed)
    C: (B, M, N)   FP16 – output buffer
    Performs C = A @ B.
    """
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "Incompatible inner dimensions for matrix multiplication"

    # Strides (in elements)
    stride_ab, stride_am, stride_ak = A.stride()
    stride_bb, stride_bk, stride_bn = B.stride()
    stride_cb, stride_cm, stride_cn = C.stride()

    grid = (batch, triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N))
    batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_ab,
        stride_bk,
        stride_bn,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator with the core O(N³) contraction
    off‑loaded to a custom Triton kernel.

    Args:
        data: (input, mask, weights, config)
            - input:  Tensor of shape [bs, seq_len, seq_len, dim]
            - mask:   Tensor of shape [bs, seq_len, seq_len] (float or bool)
            - weights: dict containing all linear / layer‑norm parameters
            - config:  dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype torch.float32)
    """
    # --------------------------- unpack ---------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------ LayerNorm --------------------------------
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ----------------------- Linear projections -------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])      # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------- Mask -----------------------------------
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------- Gates ----------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------- Cast to FP16 -------------------------
    # FP16 reduces memory pressure for the O(N³) kernel.
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Rearrange to (batch, head, N, N) and flatten batch*head.
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()          # [B, hidden_dim, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()        # [B, hidden_dim, N, N]

    left_flat = left.view(-1, N, N)                       # [B*hidden_dim, N, N]
    right_flat = right.view(-1, N, N)                     # [B*hidden_dim, N, N]
    right_flat_T = right_flat.transpose(1, 2).contiguous()  # transpose K dimension

    # Output buffer for the contraction (still FP16)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # -------------------- Batched matmul via Triton --------------------
    # C = left_flat @ right_flat_T   (equivalent to the required einsum)
    batched_matmul(left_flat, right_flat_T, out_flat,
                   BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)

    # Restore original layout and convert back to FP32 for the remaining ops
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------ Output LayerNorm ------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # -------------------------- Final Linear -------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
