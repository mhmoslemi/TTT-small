@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          M, N, K,
                          stride_am, stride_ak,
                          stride_bk, stride_bn,
                          stride_cm, stride_cn,
                          stride_batch_a, stride_batch_b, stride_batch_c,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute row and column offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute base pointers for this batch
    a_base = A_ptr + batch * stride_batch_a
    b_base = B_ptr + batch * stride_batch_b
    c_base = C_ptr + batch * stride_batch_c

    # offsets for A and B
    a_ptrs = a_base + (offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak)  # shape (BLOCK_M, BLOCK_K)
    b_ptrs = b_base + (tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn)  # shape (BLOCK_K, BLOCK_N)

    # Initialize accum
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load tiles
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (tl.arange(0, BLOCK_K) < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N), other=0.0)

        # dot
        acc += tl.dot(a, b)

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store
    c_ptrs = c_base + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    mask = (offs_m < M)[:, None] & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16) if ??? else acc, mask=mask)

def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          M, N, K,
                          stride_am, stride_ak,
                          stride_bk, stride_bn,
                          stride_cm, stride_cn,
                          stride_batch_a, stride_batch_b, stride_batch_c,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # ptr base for this batch
    a_ptr = A_ptr + pid_batch * stride_batch_a
    b_ptr = B_ptr + pid_batch * stride_batch_b
    c_ptr = C_ptr + pid_batch * stride_batch_c

    # compute pointers for the tiles
    a_ptrs = a_ptr + (offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_ptr + (tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load A and B tiles
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (tl.arange(0, BLOCK_K) < K)[None, :], other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N), other=0.0).to(tl.float16)
        # dot product
        acc += tl.dot(a, b)
        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result
    c_ptrs = c_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    mask = (offs_m < M)[:, None] & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask)

    device = input_tensor.device
    dim = config["dim"]

    # projection

"""
TriMul outgoing kernel (AlphaFold3 style) using Triton for the core
pairwise matrix multiplication over the sequence dimension.

Algorithm:
1. Layer‑norm the input tensor over its last dimension (dim).
2. Project the normalized tensor to a hidden dimension with two linear layers
   (left_proj, right_proj).  Cast to float16 for speed.
3. Apply optional mask and multiplicative gates (left_gate, right_gate) that
   are computed from the normalized tensor (float32 -> sigmoid -> cast).
4. The main update is:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which for each hidden feature d is a matrix multiplication:
      out_d = left_d @ right_dᵀ
   We flatten the batch and hidden dimensions and launch a batched GEMM
   kernel written in Triton.  The kernel computes C = A @ B where
   A has shape (batch*hidden, N, N) and B is the transposed right tensor.
5. Reshape the result back to [B, N, N, hidden_dim] (float32).
6. Apply a final LayerNorm (to_out_norm), multiply by the output gate
   (out_gate) and a final linear projection back to the original dimension.

All heavy arithmetic (the Σ_k product) runs on the GPU via Triton,
while LayerNorms and linear transformations use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ B) where
#   A : (batch, M, K)   (row‑major)
#   B : (batch, K, N)   (row‑major)
#   C : (batch, M, N)
# M, N, K are the sequence length (seq_len).  Batch dimension = B * hidden_dim.
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)      # batch * hidden
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(2)          # block col

    # Compute the absolute row/col offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for the current batch
    a_base = A_ptr + pid_batch * stride_batch_a
    b_base = B_ptr + pid_batch * stride_batch_b
    c_base = C_ptr + pid_batch * stride_batch_c

    # Pointers for the current tile (k dimension will be stepped inside the loop)
    a_ptrs = a_base + (offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_base + (tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # Accumulator in float32 (higher precision than the fp16 inputs)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K sized chunks
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k_iter in range(0, num_k_blocks):
        k_offset = k_iter * BLOCK_K

        # Masks to avoid out‑of‑bounds loads
        a_mask = (offs_m < M)[:, None] & ((tl.arange(0, BLOCK_K) + k_offset) < K)[None, :]
        b_mask = ((tl.arange(0, BLOCK_K) + k_offset) < K)[:, None] & (offs_n < N)[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # fp16

        # Dot product accumulates into fp32
        acc += tl.dot(a, b)

        # Move pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write the result back to C
    c_ptrs = c_base + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc, mask=c_mask)


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data (Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]):
            - input_tensor: shape [B, N, N, dim]
            - mask: shape [B, N, N] (may be None or all‑ones)
            - weights: dict with model parameters
            - config: dict containing 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor: shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)                               # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2) Linear projections (float16 for speed)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight).to(torch.float16)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight).to(torch.float16) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(left.dtype)           # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating signals (computed in float32, then cast)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid().to(left.dtype)   # [B, N, N, hidden_dim]
    right_gate = F.linear(x, right_gate_weight).sigmoid().to(left.dtype) # [B, N, N, hidden_dim]
    out_gate = F.linear(x, out_gate_weight).sigmoid()                    # [B, N, N, hidden_dim] float32

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul via batched matrix multiplication
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape  # batch size, seq_len

    # Bring hidden dimension to axis 1 and make tensors contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Transpose the last two dims of right so that we multiply left @ right.T
    right_T = right_perm.transpose(2, 3).contiguous()   # [B, H, N, N] where dim0=N (k), dim1=N (j)

    # Collapse batch & hidden dimensions for batched GEMM
    batch_hidden = B * hidden_dim
    A = left_perm.view(batch_hidden, N, N)   # fp16
    B_t = right_T.view(batch_hidden, N, N)   # fp16

    # Allocate output buffer (float32)
    C = torch.empty((batch_hidden, N, N), dtype=torch.float32, device=device)

    # Block sizes – chosen to fit typical B200 SM resources
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_hidden,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        A_ptr=A,
        B_ptr=B_t,
        C_ptr=C,
        M=N,
        N=N,
        K=N,
        stride_am=A.stride(1),      # = N
        stride_ak=A.stride(2),      # = 1
        stride_bk=B_t.stride(1),    # = N
        stride_bn=B_t.stride(2),    # = 1
        stride_cm=C.stride(1),      # = N
        stride_cn=C.stride(2),      # = 1
        stride_batch_a=A.stride(0), # = N*N
        stride_batch_b=B_t.stride(0),
        stride_batch_c=C.stride(0),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # float32

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating, and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)                     # float32

    out = out * out_gate                               # gating (float32)
    out = F.linear(out, to_out_weight)                 # final projection to dim

    return out
