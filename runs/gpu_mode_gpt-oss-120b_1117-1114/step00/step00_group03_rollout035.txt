```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Get all weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    # Normalize input
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # Compute projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)
    # Compute gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()
    # Apply mask if needed
    if mask is not None:
        mask_uns = mask.unsqueeze(-1).to(x_norm.dtype)
        left = left * mask_uns
        right = right * mask_uns
    # Apply gates
    left = left * left_gate
    right = right * right_gate
    # Allocate output tensor for contraction
    out = torch.empty((bs, seq_len, seq_len, hidden_dim), dtype=left.dtype, device=left.device)
    # Perform contraction using Triton
    _tri_mul_contraction(left, right, out, BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)
    # Apply output layernorm and out_gate
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    # Final linear
    output = torch.nn.functional.linear(out, to_out_weight)
    return output
```

```
def _tri_mul_contraction(left, right, out, BLOCK_M=64, BLOCK_N=64, BLOCK_K=64):
    import triton
    import triton.language as tl

    @triton.jit
    def kernel(l_ptr, r_ptr, o_ptr,
               B, N, H,
               stride_l_b, stride_l_i, stride_l_k, stride_l_h,
               stride_r_b, stride_r_i, stride_r_k, stride_r_h,
               stride_o_b, stride_o_i, stride_o_k, stride_o_h,
               BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
        pid_bh = tl.program_id(2)
        b = pid_bh // H
        h = pid_bh % H
        pid_i = tl.program_id(0)
        pid_j = tl.program_id(1)
        i_base = pid_i * BLOCK_M
        j_base = pid_j * BLOCK_N
        # loadi pointers base offsets
        offset_batch = b * stride_l_b
        offset_h = h * stride_l_h  # stride_h is 1
        # pointers for left and right.
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        # compute K blocks loop
        num_k = tl.cdiv(N, BLOCK_K)
        for k_iter in range(num_k):
            k_base = k_iter * BLOCK_K
            # compute offsets
            offs_i = i_base + tl.arange(0, BLOCK_M)
            offs_j = j_base + tl.arange(0, BLOCK_N)
            offs_k = k_base + tl.arange(0, BLOCK_K)
            # mask for valid indices
            mask_i = offs_i < N
            mask_j = offs_j < N
            mask_k = offs_k < N
            # load L tile shape (BLOCK_M, BLOCK_K)
            l_ptrs = l_ptr + offset_batch + offset_h + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
            L = tl.load(l_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
            # load R tile shape (BLOCK_N, BLOCK_K)
            r_ptrs = r_ptr + b * stride_r_b + h * stride_r_h + offs_j[:, None] * stride_r_i + offs_k[None, :] * stride_r_k
            R = tl.load(r_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
            # accumulate
            acc += tl.dot(L, R, trans_b=True)
        # store
        o_ptrs = o_ptr + b * stride_o_b + h * stride_o_h + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_k
        tl.store(o_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
    # compute strides
    B = left.shape[0]
    N = left.shape[1]  # seq_len
    H = left.shape[3]  # hidden_dim
    # compute strides in elements
    stride_l_b = left.stride(0)  # B stride
    stride_l_i = left.stride(1)  # N stride for i dimension (should be N*H)
    stride_l_k = left.stride(2)  # stride for j dimension (should be H)
    stride_l_h = left.stride(3)  # 1
    stride_r_b = right.stride(0)
    stride_r_i = right.stride(1)
    stride_r_k = right.stride(2)
    stride_r_h = right.stride(3)
    stride_o_b = out.stride(0)
    stride_o_i = out.stride(1)
    stride_o_k = out.stride(2)
    stride_o_h = out.stride(3)
    # launch grid
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)
    kernel[grid](left, right, out,
                 B, N, H,
                 stride_l_b, stride_l_i, stride_l_k, stride_l_h,
                 stride_r_b, stride_r_i, stride_r_k, stride_r_h,
                 stride_o_b, stride_o_i, stride_o_k, stride_o_h,
                 BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K)
```

"""
TriMul forward pass with a Triton kernel for the O(N³·H) contraction.

Algorithm
1. Layer‑norm the input tensor.
2. Linear projections (left/right) and three gating linear layers.
3. Apply optional mask and element‑wise gates.
4. Contract over the sequence dimension k:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (left @ rightᵀ) for each hidden channel h.
   Implemented as a tiled Triton kernel.
5. Layer‑norm the contracted result, apply the output gate and final linear
   projection back to the original feature dimension.

All heavy arithmetic (step 4) runs on the GPU via Triton; the remaining
operations use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_contraction_kernel(
    l_ptr, r_ptr, o_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_k, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C = A @ Bᵀ  where A,B ∈ [B·H, N, N]"""
    pid_bh = tl.program_id(2)               # batch*hidden index
    b = pid_bh // H                         # batch id
    h = pid_bh % H                          # hidden channel id

    pid_i = tl.program_id(0)                # tile row index
    pid_j = tl.program_id(1)                # tile col index
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # base offsets for the selected batch & channel
    offset_b = b * stride_l_b
    offset_h = h * stride_l_h                # stride_l_h == stride_r_h == stride_o_h == 1

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_base = k_iter * BLOCK_K

        offs_i = i_base + tl.arange(0, BLOCK_M)
        offs_j = j_base + tl.arange(0, BLOCK_N)
        offs_k = k_base + tl.arange(0, BLOCK_K)

        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # ---- left tile (i,k) ------------------------------------------------
        l_ptrs = l_ptr + offset_b + offset_h \
                 + offs_i[:, None] * stride_l_i \
                 + offs_k[None, :] * stride_l_k
        L = tl.load(l_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # ---- right tile transposed (k,j) ------------------------------------
        # we load it directly as (k, j) so that we can use tl.dot without extra transpose
        r_ptrs = r_ptr + offset_b + offset_h \
                 + offs_k[:, None] * stride_r_k \
                 + offs_j[None, :] * stride_r_i
        R_T = tl.load(r_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # ---- accumulate -----------------------------------------------------
        acc += tl.dot(L, R_T)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # store the result block
    o_ptrs = o_ptr + offset_b + offset_h \
             + offs_i[:, None] * stride_o_i \
             + offs_j[None, :] * stride_o_k
    tl.store(o_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_contraction(left: torch.Tensor, right: torch.Tensor, out: torch.Tensor,
                         BLOCK_M: int = 32, BLOCK_N: int = 32, BLOCK_K: int = 32) -> None:
    """Launch the Triton kernel for the O(N³·H) contraction."""
    B, N, _, H = left.shape
    # ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # strides in elements (not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_k, stride_o_h = out.stride()

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        B * H                      # batch * hidden
    )
    tri_mul_contraction_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_k, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )


def custom_kernel(data):
    """
    Forward pass of the TriMul module using Triton for the contraction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim]
            - mask: torch.Tensor, shape [B, N, N] (may be all‑ones)
            - weights: dict of parameter tensors
            - config: dict with keys "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # -----------------------------------------------------------------
    # Load weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1. Layer‑norm on the input
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)          # [B,N,N,H]
    right = F.linear(x, right_proj_weight)        # [B,N,N,H]

    # -----------------------------------------------------------------
    # 3. Gates (sigmoid) and optional mask
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # apply mask if present (mask is float or bool)
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4. Contraction over the sequence dimension (k) via Triton
    # -----------------------------------------------------------------
    out = torch.empty_like(left)   # shape [B,N,N,H]
    _tri_mul_contraction(left, right, out, BLOCK_M=32, BLOCK_N=32, BLOCK_K=32)

    # -----------------------------------------------------------------
    # 5. Output layer‑norm, gate, final linear projection back to dim
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    output = F.linear(out, to_out_weight)   # shape [B,N,N,dim]

    return output
