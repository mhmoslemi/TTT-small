def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, dim_check = input_tensor.shape
    assert dim == dim_check

    # layer norm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask
    mask = mask.unsqueeze(-1).type_as(left)  # broadcast
    left = left * mask
    right = right * mask

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # cast to half for matmul
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # reshape for batched matmul
    # bring hidden dim to batch dimension
    left_mat = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N).transpose(1, 2).contiguous()

    # output buffer
    out_mat = torch.empty_like(left_mat)

    # block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # compute strides
    stride_a_batch = left_mat.stride(0)
    stride_a_m = left_mat.stride(1)
    stride_a_k = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_m = right_mat.stride(1)  # corresponds to K
    stride_b_n = right_mat.stride(2)  # corresponds to N

    stride_c_batch = out_mat.stride(0)
    stride_c_m = out_mat.stride(1)
    stride_c_n = out_mat.stride(2)

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * hidden_dim)
    matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_m, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # shape [B,N,N,hidden_dim]
    out = out.to(torch.float32)

    # final layernorm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass with a Triton‐accelerated batched GEMM.

The module follows the reference PyTorch implementation:
1. Layer‑norm the input tensor.
2. Linear projections to a hidden dimension (no bias).
3. Optional masking and sigmoid gating.
4. Core O(N³) contraction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is implemented as a batched matrix multiplication
   (left @ rightᵀ) using a custom Triton kernel.
5. Layer‑norm, gating and final linear projection back to `dim`.

All heavy computation (step 4) is performed in the Triton kernel,
while the remaining element‑wise work stays in PyTorch.
The kernel works on half‑precision tensors (fp16) and accumulates in fp32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B) for square matrices.
# A : (batch, M, K)   – fp16
# B : (batch, K, N)   – fp16 (right matrix already transposed)
# C : (batch, M, N)   – fp16
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_m, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_b = tl.program_id(2)   # batch id

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base address for the current batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        # k offsets for this tile
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for the K dimension
        mask_k = offs_k < K

        # ------------------ Load A tile (M x K) ----------------------
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m
                            + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # ------------------ Load B tile (K x N) ----------------------
        b_ptrs = b_batch + (offs_k[:, None] * stride_b_m
                            + offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # ------------------ Matrix multiply & accumulate -------------
        # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)          # fp16·fp16 → fp32 accumulator

    # ------------------------------------------------------------------
    # Write back the result tile
    # ------------------------------------------------------------------
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m
                        + offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Input:
        data = (input_tensor, mask, weights, config)
    Output:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, dim_check = input_tensor.shape
    assert dim == dim_check, "Config dim does not match input."

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the last dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Mask (if provided) – mask shape [B,N,N] -> broadcast to hidden dim
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).type_as(left)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to fp16 and reshape for batched GEMM
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # bring hidden dimension into batch dimension: [B, H, N, N]
    left_mat = left_fp16.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    # right^T : swap its last two axes before flattening
    right_mat = right_fp16.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_mat = right_mat.permute(0, 2, 1).contiguous()   # (batch, K, N)

    # output buffer
    out_mat = torch.empty_like(left_mat)

    # ------------------------------------------------------------------
    # 6. Triton batched GEMM
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # strides (in elements)
    stride_a_batch = left_mat.stride(0)
    stride_a_m = left_mat.stride(1)   # row stride (M → K)
    stride_a_k = left_mat.stride(2)   # col stride

    stride_b_batch = right_mat.stride(0)
    stride_b_m = right_mat.stride(1)  # row stride (K)
    stride_b_n = right_mat.stride(2)  # col stride (N)

    stride_c_batch = out_mat.stride(0)
    stride_c_m = out_mat.stride(1)
    stride_c_n = out_mat.stride(2)

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        bs * hidden_dim,
    )
    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_m, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to [B,N,N,hidden_dim] and convert to fp32
    # ------------------------------------------------------------------
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).to(dtype)

    # ------------------------------------------------------------------
    # 8. Final layer‑norm, gating and output projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
