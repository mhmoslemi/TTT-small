@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(axis=0)   # block row (i)
    pid_n = tl.program_id(axis=1) # block column (j)
    pid_b = tl.program_id(axis=2) # batch

    # Offsets for M, N axes
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointers
    a_ptrs = a_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + pid_b * stride_bb + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)   # accumulate in fp32

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load a block
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (tl.arange(0, BLOCK_K)[None, :] + k*BLOCK_K < K), other=0.0)
        # load b block (NOTE: b should be transposed)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K)[:, None] + k*BLOCK_K < K) & (offs_n[None, :] < N), other=0.0)
        # dot product
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result
    c_ptrs = c_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0) # block row
    pid_n = tl.program_id(1) # block column
    pid_b = tl.program_id(2) # batch

    # Compute offsets
    # offm: (BLOCK_M,) each thread's row index
    offm = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offn = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # k offset within block
        offk = k * BLOCK_K + tl.arange(0, BLOCK_K)

        a = tl.load(a_ptr + pid_b * stride_ab + offm[:, None] * stride_am + offk[None, :] * stride_ak,

def custom_kernel(data):
    """
    Triton-accelerated TriMul forward pass (outgoing version).
    ...

    Returns:
        Tensor of shape (bs, seq_len, seq_len, dim) float32
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # LayerNorm (norm)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Normalize over last dim: dim
    x_norm = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear proj
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # P cast to half for speed
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Apply mask if provided (mask shape [bs,N,N]; unsqueeze to broadcast)
    if mask is not None:
        mask = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # Gates (float32 optional)
    # Use float32 for gate computation for better stability
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"])).to(left.dtype)
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"])).to(right.dtype)
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"])).to(torch.float16) # we will convert later

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Compute out using Triton kernel
    # reshape for batched matmul
    left_ = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, N_check = left_.shape
    assert N == N_check

    # Collapse batch and heads
    left_mat = left_.view(-1, N, N)   # (B*H, N, N)
    right_mat = right_.view(-1, N, N) # (B*H, N, N)
    # Transpose right_mat for matmul
    right_mat_T = right_mat.transpose(1, 2)

    # Output tensor
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # Get strides
    stride_ab = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_bb = right_mat_T.stride(0)
    stride_bk = right_mat_T.stride(1)
    stride_bn = right_mat_T.stride(2)

    stride_cb = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    # Choose block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        left_mat.shape[0],  # batch dimension
    )
    tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat_T,
        out_mat,
        N, N, N,
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)
    # Convert to float32
    out = out.to(torch.float32)

    # LayerNorm on hidden (to_out_norm)
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # Apply out_gate (convert to float32)
    out_gate = out_gate.to(torch.float32)  # shape (B,N,N,hidden_dim)
    out = out * out_gate

    # Final linear projection
    out = F.linear(out, weights["to_out.weight"])

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for batched matrix multiplication
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B where:
        A : (batch, M, K)
        B : (batch, K, N)
        C : (batch, M, N)
    Batch dimension is encoded in program id axis 2.
    Result is accumulated in FP32 and stored as FP16.
    """
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block column
    pid_b = tl.program_id(2)  # batch

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulators
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a = tl.load(
            a_ptr
            + pid_b * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        )
        # Load B tile (K x N)
        b = tl.load(
            b_ptr
            + pid_b * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )
        acc += tl.dot(a, b)

        # Advance pointers (optional as we recompute offsets each iter)

    # Write back C tile
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_b * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    Arguments:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, D] (float32)
            - mask  : torch.Tensor [B, N, N] (float32 or bool)
            - weights: dict of weight tensors
            - config : dict containing "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ----- LayerNorm on input -----
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B,N,N,dim), float32

    # ----- Linear projections -----
    left = F.linear(x_norm, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x_norm, weights["right_proj.weight"])

    # Cast to half for the heavy matmul
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ----- Optional masking -----
    if mask is not None:
        mask_fp = mask.to(left.dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_fp
        right = right * mask_fp

    # ----- Gating tensors (computed in FP32 for stability) -----
    left_gate_fp = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate_fp = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate_fp = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))  # (B,N,N,hidden_dim)

    # Cast gates to half to match left/right dtype
    left_gate = left_gate_fp.to(left.dtype)
    right_gate = right_gate_fp.to(right.dtype)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ----- Triton batched matrix multiplication: out = left @ right^T -----
    # Rearrange to (batch, hidden_dim, N, N) and collapse batch*hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Collapse leading dimensions
    batch_dim = left_perm.shape[0] * left_perm.shape[1]   # B * H
    left_mat = left_perm.view(batch_dim, N, N)            # (B*H, N, N)
    right_mat = right_perm.view(batch_dim, N, N)          # (B*H, N, N)

    # Transpose the second matrix so we compute A @ B^T
    right_mat_T = right_mat.transpose(1, 2)               # (B*H, N, N), non‑contiguous

    # Allocate output
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # Strides for Triton (int64)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)
    stride_ab = left_mat.stride(0)

    stride_bk = right_mat_T.stride(1)
    stride_bn = right_mat_T.stride(2)
    stride_bb = right_mat_T.stride(0)

    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)
    stride_cb = out_mat.stride(0)

    # Block sizes – tuned for the B200 GPU
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_dim,
    )

    batched_matmul_kernel[grid](
        left_mat,
        right_mat_T,
        out_mat,
        N,          # M
        N,          # N
        N,          # K
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # now FP32 for subsequent LN and final linear

    # ----- Output layer norm -----
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ----- Apply output gate -----
    out = out * out_gate_fp

    # ----- Final linear projection back to dim -----
    out = F.linear(out, weights["to_out.weight"])

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ B   (A: (B,M,K), B: (B,K,N))
# In TriMul we need C[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d],
# which is exactly a batched matmul of left with right^T.
# -------------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matmul: C = A @ B  (A: batch×M×K, B: batch×K×N)."""
    pid_m = tl.program_id(0)  # row block
    pid_n = tl.program_id(1)  # column block
    pid_b = tl.program_id(2)  # batch

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile (M×K)
        a = tl.load(
            a_ptr
            + pid_b * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        )
        # Load B tile (K×N)
        b = tl.load(
            b_ptr
            + pid_b * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )

    Forward pass of the outgoing Triangle Multiplicative Update (TriMul) using
    Triton for the O(N³) core matmul.

    Args:
        data: (input, mask, weights, config)
            - input  : Tensor [B, N, N, D] (float32)
            - mask   : Tensor [B, N, N] (float32 or bool)
            - weights: dict of model parameters
            - config : dict containing 'dim' and 'hidden_dim'

    Returns:
        Tensor [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim), float32

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x_norm, weights["right_proj.weight"])

    # Cast to half for the heavy matmul
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # -----------------------------------------------------------------
    # 3) Masking (if a mask is supplied)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_fp = mask.to(left.dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_fp
        right = right * mask_fp

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    left_gate_fp = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate_fp = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate_fp = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))  # (B,N,N,hidden_dim)

    # Cast gates to half to match left/right dtype before element‑wise multiplication
    left_gate = left_gate_fp.to(left.dtype)
    right_gate = right_gate_fp.to(right.dtype)

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Core O(N³) multiplication via Triton
    #    Compute out = left @ rightᵀ  for each hidden channel.
    # -----------------------------------------------------------------
    # Rearrange tensors to (B, H, N, N) then collapse B*H → batch dimension.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    batch_dim = left_perm.shape[0] * left_perm.shape[1]      # B * H

    right_mat_T = right_mat.transpose(1, 2)  # (B*H, N, N), non‑contiguous

    stride_ab = left_mat.stride(0)   # stride for batch

    stride_bk = right_mat_T.stride(1)  # K stride
    stride_bn = right_mat_T.stride(2)  # N stride
    stride_bb = right_mat_T.stride(0)  # batch stride

    stride_cm = out_mat.stride(1)   # M stride in output
    stride_cn = out_mat.stride(2)   # N stride in output
    stride_cb = out_mat.stride(0)   # batch stride in output

    # Tuned block sizes for the B200 GPU
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_dim,
    )

    batched_matmul_kernel[grid](
        left_mat,
        right_mat_T,
        out_mat,
        N,                # M
        N,                # N
        N,                # K
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # FP32 for the remaining ops

    # -----------------------------------------------------------------
    # 6) Output LayerNorm over hidden dim
    # -----------------------------------------------------------------
    out = F.layer_norm(

        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],

    # 7) Apply output gate and final linear projection
    # -----------------------------------------------------------------
    out = out * out_gate_fp          # broadcasted over hidden dim
    out = F.linear(out, weights["to_out.weight"])

    return out
