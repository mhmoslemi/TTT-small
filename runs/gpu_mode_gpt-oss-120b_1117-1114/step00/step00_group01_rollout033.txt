@triton.jit
def tr_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, M,  # N seq length, M typed as B*H
    stride_left_m, stride_left_i, stride_left_k,
    stride_right_m, stride_right_i, stride_right_k,
    stride_out_m, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
   # program_id(0) -> i block, program_id(1) -> j block, program_id(2) -> m block (i.e. (b,h) index)
   pid_i = tl.program_id(0)
   pid_j = tl.program_id(1)
   pid_m = tl.program_id(2)

   # compute offsets for i , j
   offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

   mask_i = offs_i < N
   mask_j = offs_j < N

   # compute offsets for m dimension = flatten batch+hid
   # each program handles a single (m) index (since program_id(2) is m)
   # else we can treat m as block dimension when M > block size; but we keep one per kernel instance.
   # Works because M up to 256 (B=2,H=128) which is okay. 
   # Prepare pointer offsets base:
   left_base = left_ptr + pid_m * stride_left_m
   right_base = right_ptr + pid_m * stride_right_m
   out_base = out_ptr + pid_m * stride_out_m

   # accumulate
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # loop over K dimension tiles
   for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N
        # load A: left[i,k]
        a_ptrs = left_base + (offs_i[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # load B: right[k,j] (we need transpose for matmul)
        b_ptrs = right_base + (offs_k[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_i)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        # dot
        acc += tl.dot(a, b)

   # store output
   out_ptrs = out_base + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
   tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

   # Done

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # fetch weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Ensure weights are on correct device and dtype
    # Already presumably on correct device (same as input) but convert dtype if needed.
    # Use dtype torch.float32 for all but we will cast left/right later.

    # LayerNorm on input
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    left_proj = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x, right_proj_weight)

    # Gating
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # Apply mask if provided (mask shape [B,N,N]).
    # Ensure mask dtype matches left_proj dtype
    if mask is not None:
        mask = mask.to(input_tensor.dtype).unsqueeze(-1)
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Convert to half for kernel
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    B, N, _, _ = input_tensor.shape
    # Permute to shape [B, hidden_dim, N, N]
    left_perm = left_f16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_f16.permute(0, 3, 1, 2).contiguous()
    M = B * hidden_dim

    left_flat = left_perm.view(M, N, N)
    right_flat = right_perm.view(M, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Strides
    stride_left_m, stride_left_i, stride_left_k = left_flat.stride()
    stride_right_m, stride_right_i, stride_right_k = right_flat.stride()
    stride_out_m, stride_out_i, stride_out_j = out_flat.stride()  # same as left

    # Triton launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), M)

    tr_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        M,
        stride_left_m, stride_left_i, stride_left_k,
        stride_right_m, stride_right_i, stride_right_k,
        stride_out_m, stride_out_i, stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # shape B,N,N,hidden_dim
    # Convert to float32 for subsequent ops
    out = out.to(torch.float32)

    # Apply final layer norm on hidden dimension
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

@triton.jit
def tr_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, M,
    stride_left_m, stride_left_i, stride_left_k,
    stride_right_m, stride_right_i, stride_right_k,
    stride_out_m, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_m = tl.program_id(2)

    # offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointer for current m
    left_base = left_ptr + pid_m * stride_left_m
    right_base = right_ptr + pid_m * stride_right_m
    out_base = out_ptr + pid_m * stride_out_m

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile A (i,k)
        a_ptr = left_base + (offs_i[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right tile B (k,j) i.e., transpose of right[i,k]? Actually we need B shape (k,j)
        b_ptr = right_base + (offs_k[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_i)

"""
TriMul (outgoing) – fast forward pass with a Triton kernel.

The heavy N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed as a batched matrix multiplication on the GPU.
We reshape the batch and hidden dimension into one “M” dimension
(M = batch * hidden_dim) and compute, for each M,
    out[M,i,j] = left[M,i,:] @ right[M,j,:]ᵀ
using a tiled GEMM kernel written in Triton. All other
operations (LayerNorm, linear projections, gating) are kept in
PyTorch for simplicity. The kernel works on bfloat16 tensors,
matching the reference implementation, and accumulates in float32
to preserve numerical quality.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def tr_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, M,
    stride_left_m, stride_left_i, stride_left_k,
    stride_right_m, stride_right_i, stride_right_k,
    stride_out_m, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul kernel:
    for each m in [0, M):
        out[m, i, j] = Σ_k left[m, i, k] * right[m, j, k]
    """
    pid_i = tl.program_id(0)   # block index for output row   (i)
    pid_j = tl.program_id(1)   # block index for output col   (j)
    pid_m = tl.program_id(2)   # which (batch, hidden) slice

    # offsets within the output matrix
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this m‑slice
    left_base = left_ptr + pid_m * stride_left_m
    right_base = right_ptr + pid_m * stride_right_m
    out_base = out_ptr + pid_m * stride_out_m

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the k dimension in tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_tiles):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile:   shape (BLOCK_M, BLOCK_K)   = left[i, k]
        a_ptrs = left_base + (offs_i[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=tl.bfloat16(0.0))

        # right tile (transposed): shape (BLOCK_K, BLOCK_N) = right[k, j]
        b_ptrs = right_base + (offs_k[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_i)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=tl.bfloat16(0.0))

        # accumulate
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # write result back in bfloat16
    out_ptrs = out_base + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.bfloat16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data: Tuple(input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # unpack weights (all are float32 tensors)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections (float32)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_weight)          # [B, N, N, hidden_dim]
    right_proj = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Gating (float32)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply mask (if any) and gating
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(input_tensor.shape[:3], device=device, dtype=dtype)
    mask = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]

    left = left_proj * left_gate * mask
    right = right_proj * right_gate * mask

    # ------------------------------------------------------------------
    # 5. Cast to bfloat16 and launch Triton kernel for the N³ reduction
    # ------------------------------------------------------------------
    left_bf = left.to(torch.bfloat16)
    right_bf = right.to(torch.bfloat16)

    B, N, _, _ = input_tensor.shape

    # reshape to (M, N, N) where M = B * hidden_dim
    left_perm = left_bf.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right_bf.permute(0, 3, 1, 2).contiguous()

    M = B * hidden_dim
    left_flat = left_perm.view(M, N, N)
    right_flat = right_perm.view(M, N, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.bfloat16)

    # strides in element units
    s_lm, s_li, s_lk = left_flat.stride()
    s_rm, s_ri, s_rk = right_flat.stride()
    s_om, s_oi, s_oj = out_flat.stride()

    # tile sizes (tuned for typical B200 GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑blocks
        triton.cdiv(N, BLOCK_N),   # j‑blocks
        M,                         # one program per (batch, hidden) slice
    )

    tr_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        M,
        s_lm, s_li, s_lk,
        s_rm, s_ri, s_rk,
        s_om, s_oi, s_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim] (float32)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # maps hidden_dim -> dim

    return out
