@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)  # batch index
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(2)  # block col

    # offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointers
        left_offset = pid_b * stride_left_batch + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        right_offset = pid_b * stride_right_batch + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i

        # load blocks
        left = tl.load(left_ptr + left_offset, mask=mask_i[:, None] & mask_k[None, :], other=tl.float16(0.0))
        right = tl.load(right_ptr + right_offset, mask=mask_k[:, None] & mask_j[None, :], other=tl.float16(0.0))

        # cast to float32 for accumulation
        left = left.to(tl.float32)
        right = right.to(tl.float32)

        # accumulate
        acc += tl.dot(left, right)

    # store result
    out_offset = pid_b * stride_out_batch + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptr + out_offset, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    # Ensure weights are correct dtype (float32)
    # Possibly need to move to device.
    norm_weight = norm_weight.to(device)
    norm_bias = norm_bias.to(device)
    left_proj_weight = left_proj_weight.to(device)
    right_proj_weight = right_proj_weight.to(device)
    left_gate_weight = left_gate_weight.to(device)
    right_gate_weight = right_gate_weight.to(device)
    out_gate_weight = out_gate_weight.to(device)
    to_out_norm_weight = to_out_norm_weight.to(device)
    to_out_norm_bias = to_out_norm_bias.to(device)
    to_out_weight = to_out_weight.to(device)

    # LayerNorm
    x_norm = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left = F.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_weight)

    # Apply mask if provided (mask may be None)
    if mask is not None:
        mask_ = mask.unsqueeze(-1).type_as(x_norm)
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

"""
TriMul (outgoing) kernel implementation.

The 4‑D input tensor x ∈ ℝ^{B×N×N×C} is normalized, linearly projected to a
hidden dimension H, masked and gated.  The heavy part
    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
is carried out by a custom Triton kernel that performs a batched matrix
multiplication for each hidden channel h.  The kernel accumulates in FP32
while loading the projections in FP16 to reduce bandwidth.  After the
contraction a final LayerNorm, gating, and a linear projection back to C
produce the output of shape (B, N, N, C).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel for a single hidden channel: matmul L @ Rᵀ
# -------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N,                                       # dimensions
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)          # batch index
    pid_m = tl.program_id(1)          # block row (i)
    pid_n = tl.program_id(2)          # block col (j)

    # ----------- offsets within the i/j dimensions ----------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------- loop over the reduction dimension k --------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block:   shape (M, K)  -> indices (i, k)
        left_off = (
            pid_b * stride_left_batch +
            offs_i[:, None] * stride_left_i +
            offs_k[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptr + left_off,
            mask=mask_i[:, None] & mask_k[None, :],
            other=tl.float16(0.0)
        )
        left = left.to(tl.float32)

        # right block transposed: shape (K, N) -> indices (k, j)
        # we load right as (k, j) so that a dot with left yields (i, j)
        right_off = (
            pid_b * stride_right_batch +
            offs_k[:, None] * stride_right_k +
            offs_j[None, :] * stride_right_i
        )
        right = tl.load(
            right_ptr + right_off,
            mask=mask_k[:, None] & mask_j[None, :],
            other=tl.float16(0.0)
        )
        right = right.to(tl.float32)

        # accumulate
        acc += tl.dot(left, right)

    # store the result (M, N) tile
    out_off = (
        pid_b * stride_out_batch +
        offs_i[:, None] * stride_out_i +
        offs_j[None, :] * stride_out_j
    )
    tl.store(
        out_ptr + out_off,
        acc,
        mask=mask_i[:, None] & mask_j[None, :]
    )

# -------------------------------------------------------------
# Public entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Compute the outgoing TriMul operation using a custom Triton kernel.

    Args:
        data: Tuple of (input, mask, weights, config)
            - input: torch.Tensor of shape [B, N, N, C] (float32)
            - mask:  torch.Tensor of shape [B, N, N] (optional, broadcastable)
            - weights: dict of model weight tensors
            - config: dict with keys "dim" (C) and "hidden_dim" (H)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # Load model weights (assume they are already on the correct device)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # LayerNorm on the input
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias
    )  # shape [B, N, N, C], dtype float32

    # -----------------------------------------------------------------
    # Linear projections (float32)
    # -----------------------------------------------------------------
    left  = F.linear(x_norm, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x_norm, right_proj_weight)  # [B, N, N, H]

    # -----------------------------------------------------------------
    # Optional mask (broadcast over the hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_f = mask.unsqueeze(-1).type_as(x_norm)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # Gating (sigmoid after a linear)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_weight))   # [B, N, N, H]

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Cast projections to FP16 for the kernel
    # -----------------------------------------------------------------
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape

    # -----------------------------------------------------------------
    # Allocate output tensor (float32)
    # -----------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # -----------------------------------------------------------------
    # Triton kernel launch
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (B, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    # Launch one kernel per hidden channel (hidden_dim ≤ 384, loop overhead negligible)
    for h in range(hidden_dim):
        # Views for a single channel
        left_h_chan  = left_h[..., h]   # [B, N, N]
        right_h_chan = right_h[..., h]  # [B, N, N]
        out_chan     = out[..., h]      # [B, N, N]

        tri_mul_kernel[grid](
            left_h_chan,
            right_h_chan,
            out_chan,
            B,
            N,
            left_h_chan.stride(0), left_h_chan.stride(1), left_h_chan.stride(2),
            right_h_chan.stride(0), right_h_chan.stride(1), right_h_chan.stride(2),
            out_chan.stride(0), out_chan.stride(1), out_chan.stride(2),
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )

    # -----------------------------------------------------------------
    # Final LayerNorm, gating and projection back to original dim
    # -----------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias
    )  # [B, N, N, H], float32

    out_norm = out_norm * out_gate                # apply output gate
    output = F.linear(out_norm, to_out_weight)    # [B, N, N, C]

    return output
