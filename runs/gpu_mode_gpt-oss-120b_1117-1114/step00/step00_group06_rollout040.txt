import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_kernel(left_ptr, right_ptr, out_ptr,
                  N, H,
                  stride_bl, stride_il, stride_kl, stride_hl,
                  stride_br, stride_jr, stride_kr, stride_hr,
                  stride_bo, stride_io, stride_jo, stride_ho,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # decode batch and hidden dimension
    b = pid_bh // H
    d = pid_bh % H

    # row and column offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile (M x K)
        left_ptrs = left_ptr + b * stride_bl + offs_i[:, None] * stride_il + offs_k[None, :] * stride_kl + d * stride_hl
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right tile transposed (K x N) – note k is rows, j is columns
        right_ptrs = right_ptr + b * stride_br + offs_k[:, None] * stride_kr + offs_j[None, :] * stride_jr + d * stride_hr
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # matrix multiplication (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(left, right)

    # write result
    out_ptrs = out_ptr + b * stride_bo + offs_i[:, None] * stride_io + offs_j[None, :] * stride_jo + d * stride_ho
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, N, _, _ = input_tensor.shape

    # fetch weights
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)
    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    to_out_weight = weights["to_out.weight"].to(device)

    eps = 1e-5

    # LayerNorm on input (dim)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections (no bias)
    left = F.linear(x, left_proj_weight)    # shape [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Apply mask (mask is [B,N,N] possibly bool or float)
    mask_f = mask.to(x.dtype).unsqueeze(-1)   # shape [B,N,N,1]
    left = left * mask_f
    right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel (saving memory bandwidth)
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)

    # allocate output tensor
    out = torch.empty((batch_size, N, N, hidden_dim), dtype=torch.float32, device=device)

    # get strides after making tensors contiguous (just in case)
    left_half = left_half.contiguous()
    right_half = right_half.contiguous()
    out = out.contiguous()

    # strides as Python ints
    l_str = left_half.stride()
    r_str = right_half.stride()
    o_str = out.stride()

    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), batch_size * hidden_dim)

"""
TriMul (outgoing) implementation with Triton.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Project the normalized tensor with three linear layers (left_proj, right_proj, gates).
3. Apply the pairwise mask and gating.
4. Compute the heavy O(N³·H) contraction:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   via a tiled matrix‑multiply kernel written in Triton.  The kernel processes a
   tile (BLOCK_M × BLOCK_N) for a single batch element *and* a single hidden
   dimension `d`.  For each `k` block it loads a (M×K) tile from `left` and a
   transposed (K×N) tile from `right` and accumulates with `tl.dot`.
5. Apply a second LayerNorm, multiply by the output‑gate and a final linear
   projection to restore the original feature dimension.

The Triton kernel only performs the O(N³·H) core; everything else stays in
PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,               # [B, N, N, H]  (float16)
    right_ptr,              # [B, N, N, H]  (float16)
    out_ptr,                # [B, N, N, H]  (float32)
    N, H,                   # sequence length, hidden dimension
    stride_bl, stride_il, stride_kl, stride_hl,   # left strides
    stride_br, stride_jr, stride_kr, stride_hr,   # right strides
    stride_bo, stride_io, stride_jo, stride_ho,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]."""

    pid_i = tl.program_id(0)          # tile row i
    pid_j = tl.program_id(1)          # tile col j
    pid_bh = tl.program_id(2)         # combined batch * hidden index

    b = pid_bh // H                   # batch index
    d = pid_bh % H                    # hidden dimension index

    # Offsets of the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator for the (M x N) tile, stored in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (M, K)
        left_ptrs = (
            left_ptr
            + b * stride_bl
            + offs_i[:, None] * stride_il
            + offs_k[None, :] * stride_kl
            + d * stride_hl
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # right tile transposed: shape (K, N)
        # we load right as (k, j) so that it is already transposed
        right_ptrs = (
            right_ptr
            + b * stride_br
            + offs_k[:, None] * stride_kr
            + offs_j[None, :] * stride_jr
            + d * stride_hr
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # Matrix multiplication for this K‑block
        acc += tl.dot(left, right)   # (M, K) @ (K, N) -> (M, N)

    # Store the result tile
    out_ptrs = (
        out_ptr
        + b * stride_bo
        + offs_i[:, None] * stride_io
        + offs_j[None, :] * stride_jo
        + d * stride_ho
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Executes the outgoing TriMul operation.
    Args:
        data: Tuple[input, mask, weights, config]
            input  - torch.Tensor [B, N, N, C] (float32)
            mask   - torch.Tensor [B, N, N] (bool or float)
            weights- Dict[str, torch.Tensor] containing all model parameters
            config - Dict with keys "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, N, _, _ = input_tensor.shape
    eps = 1e-5

    # ----- LayerNorm on input -----
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
        eps=eps,
    )

    # ----- Linear projections (no bias) -----
    left = F.linear(x, weights["left_proj.weight"].to(device))
    right = F.linear(x, weights["right_proj.weight"].to(device))

    # ----- Apply mask -----
    mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
    left = left * mask_f
    right = right * mask_f

    # ----- Gates -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(device)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(device)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(device)))

    left = left * left_gate
    right = right * right_gate

    # ----- Triton multiplication kernel -----
    # Cast to half precision for bandwidth‑limited load
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    out = torch.empty(
        (batch_size, N, N, hidden_dim), dtype=torch.float32, device=device
    ).contiguous()

    # Strides (int) after contiguity
    l_str = left_h.stride()   # (B, i, k, d)
    r_str = right_h.stride()
    o_str = out.stride()

    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_size * hidden_dim,
    )

    trimul_outgoing_kernel[grid](
        left_h,
        right_h,
        out,
        N,
        hidden_dim,
        l_str[0],
        l_str[1],
        l_str[2],
        l_str[3],
        r_str[0],
        r_str[1],
        r_str[2],
        r_str[3],
        o_str[0],
        o_str[1],
        o_str[2],
        o_str[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- Post‑processing -----
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device),
        bias=weights["to_out_norm.bias"].to(device),
        eps=eps,
    )
    out = out_norm * out_gate
    out = F.linear(out, weights["to_out.weight"].to(device))  # -> [B, N, N, dim]

    return out
