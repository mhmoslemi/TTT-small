@triton.jit
def triangular_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH, M, N, K,
    stride_left_b, stride_left_i, stride_left_k,
    stride_right_b, stride_right_i, stride_right_k,
    stride_out_b, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
   pid_batch = tl.program_id(0)
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)

   # Offsets
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
   # Initialize accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   for k in range(0, tl.cdiv(K, BLOCK_K)):
       offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
       # masks
       mask_m = offs_m < M
       mask_n = offs_n < N
       mask_k = offs_k < K
       # Load left tile: shape (BLOCK_M, BLOCK_K)
       left_ptrs = left_ptr + pid_batch * stride_left_b + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
       left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
       # Load right tile transposed: shape (BLOCK_K, BLOCK_N)
       # Right memory layout: (batch, j, k), we need to load as (k, j)
       right_ptrs = right_ptr + pid_batch * stride_right_b + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_i
       right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
       # Compute dot
       acc += tl.dot(left, right)  # left (M,K) dot right (K,N) => (M,N)
   # Write out
   out_ptrs = out_ptr + pid_batch * stride_out_b + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
   mask_o = (offs_m[:, None] < M) & (offs_n[None, :] < N)
   tl.store(out_ptrs, acc.to(tl.float16), mask=mask_o)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Use float32 dtype throughout
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if needed
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1).type_as(x_norm) # shape [B,N,N,1]
        left = left * mask_unsq
        right = right * mask_unsq

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Compute outer product via Triton
    # Convert to half for kernel
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)

    # Permute and reshape to (B*hidden_dim, N, N)
    B = left.shape[0]
    N = left.shape[1]
    # Permute to bring hidden_dim to batch dimension
    left_perm = left_half.permute(0, 3, 1, 2).contiguous()
    right_perm = right_half.permute(0, 3, 1, 2).contiguous()
    batch_dim = B * hidden_dim
    left_mat = left_perm.view(batch_dim, N, N)
    right_mat = right_perm.view(batch_dim, N, N)

    out = out_mat.to(torch.float32).view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # final linear
    to_out_weight = weights["to_out.weight"]

@triton.jit
def triangular_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_left_b, stride_left_i, stride_left_k,
    stride_right_b, stride_right_i, stride_right_k,
    stride_out_b, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
   batch = tl.program_id(0)
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)

   # Offsets
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
   # Initialize acc
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # Loop over K
   for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask for dimension bounds
        mask_m = offs_m < M
        mask_n = offs_n < N
        mask_k = offs_k < K
        # Load left tile (M,K)
        left_ptrs = left_ptr + batch * stride_left_b + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load right tile transposed (K,N) -> we want shape (K,N)
        right_ptrs = right_ptr + batch * stride_right_b + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(left, right)
   # Store result
   out_ptrs = out_ptr + batch * stride_out_b + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
   mask_o = (offs_m[:, None] < M) & (offs_n[None, :] < N)
   tl.store(out_ptrs, acc.to(tl.float16), mask=mask_o)

"""
TriMul (outgoing) forward implementation with a custom Triton kernel.

The heavy computation is the O(N³·C) reduction:
    out[b, i, j, c] = Σ_k left[b, i, k, c] * right[b, j, k, c]
which is a batched matrix multiplication of the form
    out_batch = left_batch @ right_batchᵀ
for each (batch, channel) pair.
The kernel implements this batched GEMM in pure Triton while the
remaining linear + layer‑norm + gating steps are performed with
torch.nn.functional for simplicity.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
# A : (B, M, K)   (here M = K = N)
# B : (B, N, K)   (the second operand before transpose)
# C : (B, M, N)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the output tile (m,n)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑range accesses
        mask_m = offs_m < M
        mask_n = offs_n < N
        mask_k = offs_k < K

        # -------- Load A (shape M×K) --------
        a_ptrs = (
            a_ptr
            + batch_id * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # -------- Load Bᵀ (shape K×N) --------
        # B is stored as (batch, N, K); we want (K, N) for the dot.
        b_ptrs = (
            b_ptr
            + batch_id * stride_b_batch
            + offs_k[:, None] * stride_b_k   # K‑dimension first
            + offs_n[None, :] * stride_b_n   # N‑dimension second
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # -------- Multiply‑accumulate --------
        acc += tl.dot(a, b)   # a : (M,K), b : (K,N) → (M,N)

    # -------- Store the result --------
    c_ptrs = (
        c_ptr
        + batch_id * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    mask_o = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_o)


def _batched_matmul_triton(a: torch.Tensor, b: torch.Tensor,
                           BLOCK_M=128, BLOCK_N=128, BLOCK_K=32) -> torch.Tensor:
    """
    Batched GEMM C = A @ Bᵀ using the Triton kernel above.
    A, B: (batch, N, N) float32 tensors (contiguous).
    Returns C with the same dtype/shape.
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, M, K = a.shape
    _, N, _ = b.shape
    assert M == K and N == K, "TriMul expects square inner dimensions"

    out = torch.empty((B, M, N), dtype=a.dtype, device=a.device)

    # Strides (in number of elements)
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_n, stride_b_k = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = out.stride()

    grid = (
        B,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _tri_mul_kernel[grid](
        a, b, out,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_n, stride_b_k,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor:   [B, N, N, dim]   torch.float32
            - mask:           [B, N, N]        torch.float32 (0/1) or bool
            - weights: dict of all linear / layernorm weights
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------- LayerNorm on the input (over last dim) -------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------- Linear projections (no bias) -------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------- Optional mask (mask is already on device) -------
    # The reference code multiplies mask before gating.
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # shape [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------- Gates -------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    #   Heavy O(N³·C) reduction with a Triton batched GEMM
    #   left/right : [B, N, N, hidden_dim]
    #   we bring hidden_dim into the batch dimension:
    #   shape -> (B * hidden_dim, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Permute to (B, hidden_dim, N, N) and flatten batch*hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    # Triton GEMM
    out_mat = _batched_matmul_triton(left_mat, right_mat)

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------- Output layer‑norm -------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------- Apply out‑gate and final linear -------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
