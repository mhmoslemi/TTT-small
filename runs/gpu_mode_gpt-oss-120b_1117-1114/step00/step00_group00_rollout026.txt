    right_proj_weight = weights["right_proj.weight"]

    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    right = right * right_gate

    # Permute and reshape
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transposed, [B, hidden_dim, N, N]

    # reshape
    left_mat = left_perm.view(batch, seq_len, seq_len)
    right_mat = right_perm.view(batch, seq_len, seq_len)

    out_mat = torch.empty_like(left_mat)

    import triton.language as tl

    BLOCK_N = 64
    BLOCK_K = 32

    b_ptr = right_mat
    c_ptr = out_mat

    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_mat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    batched_matmul_kernel[grid](
        a_ptr, b_ptr, c_ptr,
        seq_len, seq_len, seq_len,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # Reshape back
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1)
    # Now out has shape [bs, seq_len, seq_len, hidden_dim]

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # LayerNorm on hidden_dim
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    out = out * out_gate

    # final linear to original dim
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)
    # Ensure dtype float32
    out = out.to(torch.float32)

    return out

import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_n = tl.program_id(1)
    pid_b = tl.program_id(2)

    # Compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # over M dimension
    pid_n = tl.program_id(1)  # over N dimension
    pid_b = tl.program_id(2)  # batch * head dimension

    # Compute offsets within the matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current tile for A and B (k=0)
    a_tile_ptr = a_ptr + pid_b * stride_a_batch + offs_m[:, None] * stride_a_m + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    b_tile_ptr = b_ptr + pid_b * stride_b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # Load tiles with bounds checking
        a = tl.load(a_tile_ptr + k_offset * stride_a_k,
                    mask=mask_m[:, None] & (k_offset + tl.arange(0, BLOCK_K) < K),
                    other=0.0)
        b = tl.load(b_tile_ptr + k_offset * stride_b_k,
                    mask=(k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
                    other=0.0)
        # Accumulate product
        acc += tl.dot(a, b)

    # Write back
    c_tile_ptr = c_ptr + pid_b * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_tile_ptr, acc, mask=mask)


def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul (outgoing) operator.
    Performs LayerNorm, linear projections, gating, a batched matrix multiplication
    via a custom Triton kernel (the equivalent of the einsum in the reference),
    followed by a second LayerNorm, gating and final linear projection.
    """
    input_tensor, mask, weights, config = data
    # Resolve configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----------------------------------------------------------------------
    # 1. LayerNorm on the input tensor
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, normalized_shape=(dim,),
                     weight=norm_weight, bias=norm_bias, eps=eps)

    # ----------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ----------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # ----------------------------------------------------------------------
    # 3. Apply mask if present (mask shape [B,N,N])
    # ----------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ----------------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise modulation
    # ----------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 5. Batched matrix multiplication using Triton
    #    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    # ----------------------------------------------------------------------
    # Permute to bring the hidden dimension in front and make the
    # multiplication resemble a standard GEMM.
    # left:  [B, N, N, H] -> [B, H, N, N]
    # right: [B, N, N, H] -> [B, H, N, N] and transpose the last two axes.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (transposed)

    B, H, N, _ = left_perm.shape
    batch = B * H

    # Collapse batch and head dimensions for the kernel
    left_mat = left_perm.view(batch, N, N)
    right_mat = right_perm.view(batch, N, N)

    # Output buffer
    out_mat = torch.empty_like(left_mat)

    # Strides for Triton (batch, M, K) and (batch, K, N)
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_mat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), batch)

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,      # M
        N,      # N
        N,      # K
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to original 4‑D layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)   # [B, N, N, H]

    # ----------------------------------------------------------------------
    # 6. Second LayerNorm, output gate and final projection
    # ----------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    out = out * out_gate

    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    # Ensure fp32 output
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass with Triton acceleration.

The heavy part of the module is the einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We rewrite this as a batched matrix multiplication:
    out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d]ᵀ

For every batch and hidden channel we launch a Triton kernel that
computes a GEMM with tiling (BLOCK_M × BLOCK_N × BLOCK_K).  All
other operations (LayerNorm, Linear projections, gating) are kept in
PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched mat‑mul C = A @ B  (A: (batch,M,K), B: (batch,K,N))."""
    pid_m = tl.program_id(0)               # output row block
    pid_n = tl.program_id(1)               # output col block
    pid_b = tl.program_id(2)               # batch * head index

    # Offsets inside the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current (m,n) tile at k = 0
    a_tile_ptr = a_ptr + pid_b * stride_a_batch + offs_m[:, None] * stride_a_m + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    b_tile_ptr = b_ptr + pid_b * stride_b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        # Load tiles with masking for the tail blocks
        a = tl.load(
            a_tile_ptr + k_off * stride_a_k,
            mask=mask_m[:, None] & (k_off + tl.arange(0, BLOCK_K) < K),
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr + k_off * stride_b_k,
            mask=(k_off + tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # Write the result
    c_tile_ptr = c_ptr + pid_b * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_tile_ptr, acc, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.
    Arguments:
        data = (input_tensor, mask, weights, config)
        - input_tensor: [B, N, N, dim]
        - mask        : [B, N, N] (float or bool)
        - weights     : dict of torch.Tensors matching the module's parameters
        - config      : {"dim": int, "hidden_dim": int}
    Returns:
        output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (mask is broadcast on the hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise modulation
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Batched GEMM via Triton (the core einsum)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    #   a) bring hidden dimension to the leading axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transposed, [B, H, N, N]

    B, H, N, _ = left_perm.shape
    batch = B * H

    #   b) collapse batch and head dimensions for the kernel
    a_mat = left_perm.view(batch, N, N)   # (batch, M=N, K=N)
    b_mat = right_perm.view(batch, N, N)  # (batch, K=N, N)
    c_mat = torch.empty_like(a_mat)

    #   c) strides for the kernel
    stride_a_batch, stride_a_m, stride_a_k = a_mat.stride()
    stride_b_batch, stride_b_k, stride_b_n = b_mat.stride()
    stride_c_batch, stride_c_m, stride_c_n = c_mat.stride()

    #   d) launch Triton kernel
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), batch)
    batched_matmul_kernel[grid](
        a_mat,
        b_mat,
        c_mat,
        N,      # M
        N,      # N
        N,      # K
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    #   e) restore original 4‑D layout
    out = c_mat.view(B, H, N, N).permute(0, 2, 3, 1)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Second LayerNorm, output gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out.to(torch.float32)
