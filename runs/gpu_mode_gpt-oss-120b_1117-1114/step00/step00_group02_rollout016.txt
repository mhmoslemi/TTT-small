    left_ptr, right_ptr, out_ptr,
    batch_h, N,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_i, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    pid_n = tl.program_id(1)
    pid_b = tl.program_id(2)

    # compute offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    k_tiles = tl.cdiv(N, BLOCK_K)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_tile in range(k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # mask for bounds
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # load right tile (B) as transposed: shape (BLOCK_K, BLOCK_N)
        # right index: right[batch, j, k] -> strides: batch*stride_r_batch + j*stride_r_i + k*stride_r_k

    mask = (offs_i[:, None] < N) & (offs_j[None, :] < N)

"""
TriMul forward pass with the core triangular multiplication performed in Triton.

Algorithm
---------
1. Layer‑norm the input tensor over the last channel dimension.
2. Linear projections (left/right) and gating linear layers are applied in
   PyTorch (FP32).  The mask (if provided) is broadcasted and multiplied
   before gating.
3. The main operation  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batch of matrix multiplications L @ Rᵀ for each hidden channel d.
   We reshape the tensors to (B·H, N, N) and launch a Triton kernel that
   computes the batched GEMM with block‑wise accumulation (BLOCK_M/N/K = 64).
4. The result is reshaped back to (B,N,N,H) and cast to FP32.
5. Layer‑norm over the hidden dimension, multiplication by the output gate,
   and a final linear projection back to the original dimension are done
   in PyTorch.

Only the O(N³) triangular multiplication is off‑loaded to Triton; the rest
remains in PyTorch for simplicity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,        # [B*H, N, N]   (float16)
    right_ptr,       # [B*H, N, N]   (float16)
    out_ptr,         # [B*H, N, N]   (float16)
    batch_h, N,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_i, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)  # block row
    pid_j = tl.program_id(1)  # block col
    pid_b = tl.program_id(2)  # batch*hidden

    # Offsets for the M and N dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of tiles along K
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for k_tile in range(num_k_tiles):
        k_off = k_tile * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)

        # Load a tile of left: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + pid_b * stride_l_batch \
                 + offs_i[:, None] * stride_l_i \
                 + offs_k[None, :] * stride_l_k
        a_mask = (offs_i[:, None] < N) & (offs_k[None, :] < N)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # Load a tile of right transposed: shape (BLOCK_K, BLOCK_N)
        # right[b, j, k] -> we need (k, j)
        b_ptrs = right_ptr + pid_b * stride_r_batch \
                 + offs_k[:, None] * stride_r_k \
                 + offs_j[None, :] * stride_r_i
        b_mask = (offs_k[:, None] < N) & (offs_j[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write the result tile
    c_ptrs = out_ptr + pid_b * stride_o_batch \
             + offs_i[:, None] * stride_o_i \
             + offs_j[None, :] * stride_o_j
    c_mask = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the TriMul module with Triton‑accelerated multiplication.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (float or bool)
            - weights: dict containing:
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Config parameters
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. LayerNorm over the channel dimension
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias) and gates (with sigmoid)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Optional mask (broadcasted over the hidden dimension)
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(dtype)
        left = left * mask_exp
        right = right * mask_exp

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # used later after norm

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 3. Triton kernel for the batched multiplication over K
    #    (left @ right^T for each hidden channel)
    # Cast to half for the kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Put hidden dimension as the second axis for easy flattening
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    batch_h = B * H

    # Merge batch and hidden dimensions
    left_flat = left_perm.view(batch_h, N, N)
    right_flat = right_perm.view(batch_h, N, N)

    # Output buffer
    out_flat = torch.empty((batch_h, N, N), dtype=torch.float16, device=device)

    # Strides (in elements)
    stride_l_batch = left_flat.stride(0)
    stride_l_i = left_flat.stride(1)
    stride_l_k = left_flat.stride(2)

    stride_r_batch = right_flat.stride(0)
    stride_r_i = right_flat.stride(1)
    stride_r_k = right_flat.stride(2)

    stride_o_batch = out_flat.stride(0)
    stride_o_i = out_flat.stride(1)
    stride_o_j = out_flat.stride(2)

    # Kernel launch parameters
    BLOCK = 64
    grid = (
        triton.cdiv(N, BLOCK),   # i dimension
        triton.cdiv(N, BLOCK),   # j dimension
        batch_h,                 # batch*hidden
    )
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_h,
        N,
        stride_l_batch, stride_l_i, stride_l_k,
        stride_r_batch, stride_r_i, stride_r_k,
        stride_o_batch, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK,
        BLOCK_N=BLOCK,
        BLOCK_K=BLOCK,
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 4. Output LayerNorm, gating and final linear projection
    out = out.to(torch.float32)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # shape [B, N, N, dim]

    return out
