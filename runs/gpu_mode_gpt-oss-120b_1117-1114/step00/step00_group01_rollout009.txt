"""
TriMul (outgoing) kernel implementation.

The operator works on an input tensor `x` of shape [B, N, N, C] and
computes a triangle multiplicative update:

    left  = Linear_left (LN(x)) * mask
    right = Linear_right(LN(x)) * mask
    left  = left  * sigmoid(Linear_left_gate (x))
    right = right * sigmoid(Linear_right_gate(x))

    out   = Σ_k left[..., i, k, d] * right[..., j, k, d]   (d is hidden_dim)
            = (left @ rightᵀ)  for each hidden dimension d

    out   = LayerNorm(out) * sigmoid(Linear_out_gate(x))
    out   = Linear_out(out)                # projects hidden_dim → C

The heavy O(N³) contraction `out = left @ rightᵀ` is performed by a
custom Triton kernel that treats each (batch, hidden_dim) pair as an
independent matrix multiplication.  All other operations are carried
out with PyTorch (float32) for simplicity and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton GEMM kernel for one (batch, hidden_dim) slice
# --------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,          # *fp16  [B,N,N,H]
    right_ptr,         # *fp16  [B,N,N,H]
    out_ptr,           # *fp16  [B,N,N,H]
    B, N, H,           # int32 dimensions
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""

    # --------------------------------------------------------------
    # Identify the (batch, hidden) slice this program instance works on
    # --------------------------------------------------------------
    batch_h = tl.program_id(2)                # 0 .. B*H-1
    b = batch_h // H                           # batch index
    h = batch_h % H                            # hidden‑dim index

    pid_m = tl.program_id(0)                   # row block index (i)
    pid_n = tl.program_id(1)                   # col block index (j)

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # accumulator in fp32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block : left[b, i, k, h]
        a_ptrs = left_base + (offs_m[:, None] * stride_li) + (offs_k[None, :] * stride_lk)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B_T block : rightᵀ[k, j, h] == right[b, j, k, h]
        b_ptrs = right_base + (offs_k[:, None] * stride_rk) + (offs_n[None, :] * stride_rj)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Accumulate the dot product
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Store the result (cast back to fp16)
    # --------------------------------------------------------------
    c_ptrs = out_base + (offs_m[:, None] * stride_oi) + (offs_n[None, :] * stride_oj)
    mask_c = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


# --------------------------------------------------------------
# Entry point used by the evaluator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul layer with a fused Triton kernel
    for the O(N³) contraction.

    Args:
        data (Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]):
            - input_tensor: [B, N, N, C]   (float32 or float16)
            - mask:         [B, N, N]      (float32, 0/1) or None
            - weights: dictionary with all linear / layer‑norm parameters
            - config:  dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Extract configuration
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load weights (all are stored as float32 on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]

    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # Layer‑norm over the feature dimension
    # ------------------------------------------------------------------
    x = input_tensor.to(torch.float32)
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # ------------------------------------------------------------------
    # Projections (float32) and gating
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_weight)    # [B,N,N,hidden_dim]
    right_proj = F.linear(x, right_proj_weight)

    # Mask (if required)
    if (mask is not None) and (not nomask):
        mask_ = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask_
        right_proj = right_proj * mask_

    # Gates (float32)
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # Fuse projection, mask and gate; cast to fp16 for the kernel
    # ------------------------------------------------------------------
    left  = (left_proj * left_gate).to(torch.float16)
    right = (right_proj * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # Allocate output buffer for the GEMM (fp16)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    out_fp16 = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # Prepare strides for the Triton kernel
    # ------------------------------------------------------------------
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out_fp16.stride()

    # Kernel block sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid definition: (row blocks, col blocks, batch*hidden)
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    # Launch the Triton kernel
    tri_mul_gemm_kernel[grid](
        left, right, out_fp16,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Post‑processing in float32
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)                     # [B,N,N,hidden_dim]

    # Layer‑norm over the hidden dimension
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=1e-5)

    # Apply the output gate (out_gate is float32)
    out = out * out_gate

    # Final linear projection back to original dimension C (=dim)
    out = F.linear(out, to_out_weight)                 # [B,N,N,dim]

    return out
