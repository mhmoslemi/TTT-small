    WRP_ptr,  # right_proj weight T
    WRG_ptr,  # right_gate weight T

    stride_wglk, stride_wglh,
    stride_wprk, stride_wprh,
    stride_wgrk, stride_wgrh,
    stride_maskm,
    stride_outlm, stride_outlh,
    stride_outrm, stride_outrh,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)
    # Compute block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    # Mask for valid M and H
    mask_m = offs_m < M
    mask_h = offs_h < H
    mask_mh = mask_m[:, None] & mask_h[None, :]

    # Initialize accumulators
    acc_LP = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_LG = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_RP = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_RG = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    a_ptrs = X_ptr + (offs_m[:, None] * stride_xm + tl.arange(0, BLOCK_K)[None, :] * stride_xk)

    bRP_ptrs = WRP_ptr + (tl.arange(0, BLOCK_K)[:, None] * stride_wprk + offs_h[None, :] * stride_wprh)
    bRG_ptrs = WRG_ptr + (tl.arange(0, BLOCK_K)[:, None] * stride_wgrk + offs_h[None, :] * stride_wgrh)

        bLP = tl.load(bLP_ptrs + k_offs[:, None] * stride_wplk, mask=k_valid[:, None] & mask_h[None, :], other=0.0)
        bLG = tl.load(bLG_ptrs + k_offs[:, None] * stride_wglk, mask=k_valid[:, None] & mask_h[None, :], other=0.0)
        bRP = tl.load(bRP_ptrs + k_offs[:, None] * stride_wprk, mask=k_valid[:, None] & mask_h[None, :], other=0.0)
        bRG = tl.load(bRG_ptrs + k_offs[:, None] * stride_wgrk, mask=k_valid[:, None] & mask_h[None, :], other=0.0)

        # Dot products
        acc_LP += tl.dot(a, bLP)
        acc_LG += tl.dot(a, bLG)
        acc_RP += tl.dot(a, bRP)
        acc_RG += tl.dot(a, bRG)

        # advance pointers for next block? Actually we recompute a_ptrs with offsets each iteration. Using a_ptrs + k_offs * stride_xk works.

"""
TriMul custom kernel (outgoing version).

Algorithm:
1. Layer‑Norm the 4‑D input tensor.
2. Fuse the four linear transformations (left_proj, left_gate, right_proj, right_gate)
   and the optional mask into a single Triton GEMM kernel.
   For each element (b,i,j) we compute:
       left = (x·W_left) * mask * sigmoid(x·W_left_gate)
       right = (x·W_right) * mask * sigmoid(x·W_right_gate)
3. Compute the triangular multiplicative update:
       out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   This is a batched matrix multiplication for each hidden dimension d.
4. Apply a second LayerNorm, multiply with the out‑gate (sigmoid(x·W_out_gate)),
   and project back to the original channel dimension.

Only the projection/gating step is implemented in Triton; the rest uses PyTorch
(which already provides highly‑optimized kernels for the heavy batched GEMM and
layer‑norms).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def proj_gate_kernel(
    X_ptr,                     # float* [M, K]
    WLP_ptr, WLG_ptr,         # float* [K, H] left proj / gate weights (transposed)
    WRP_ptr, WRG_ptr,         # float* [K, H] right proj / gate weights (transposed)
    mask_ptr,                 # float* [M]
    outL_ptr, outR_ptr,       # float* [M, H] gated left / right outputs
    M, K, H,                  # tensor dimensions
    stride_xm, stride_xk,     # strides for X
    stride_wk, stride_wh,     # strides for all weight matrices (identical shape)
    stride_maskm,              # stride for mask
    stride_outlm, stride_outlh,  # strides for outL
    stride_outrm, stride_outrh,  # strides for outR
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    # Offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_m = offs_m < M
    mask_h = offs_h < H
    mask_mh = mask_m[:, None] & mask_h[None, :]

    # Accumulators for the four dot‑products
    acc_LP = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_LG = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_RP = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_RG = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    # Constant ranges within a block
    offs_k = tl.arange(0, BLOCK_K)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        k_valid = cur_k < K

        # Load a tile of X
        a = tl.load(
            X_ptr + (offs_m[:, None] * stride_xm + cur_k[None, :] * stride_xk),
            mask=mask_m[:, None] & k_valid[None, :],
            other=0.0,
        )

        # Load tiles of the four weight matrices
        bLP = tl.load(
            WLP_ptr + (cur_k[:, None] * stride_wk + offs_h[None, :] * stride_wh),
            mask=k_valid[:, None] & mask_h[None, :],
            other=0.0,
        )
        bLG = tl.load(
            WLG_ptr + (cur_k[:, None] * stride_wk + offs_h[None, :] * stride_wh),
            mask=k_valid[:, None] & mask_h[None, :],
            other=0.0,
        )
        bRP = tl.load(
            WRP_ptr + (cur_k[:, None] * stride_wk + offs_h[None, :] * stride_wh),
            mask=k_valid[:, None] & mask_h[None, :],
            other=0.0,
        )
        bRG = tl.load(
            WRG_ptr + (cur_k[:, None] * stride_wk + offs_h[None, :] * stride_wh),
            mask=k_valid[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Accumulate dot products
        acc_LP += tl.dot(a, bLP)
        acc_LG += tl.dot(a, bLG)
        acc_RP += tl.dot(a, bRP)
        acc_RG += tl.dot(a, bRG)

    # Load mask values and broadcast
    mask_vals = tl.load(mask_ptr + offs_m, mask=mask_m, other=0.0)
    mask_vals = mask_vals[:, None]  # shape (BLOCK_M, 1)

    # Sigmoid for the gates
    gate_sig_L = 1.0 / (1.0 + tl.exp(-acc_LG))
    gate_sig_R = 1.0 / (1.0 + tl.exp(-acc_RG))

    # Apply mask and gates
    left_out = acc_LP * mask_vals * gate_sig_L
    right_out = acc_RP * mask_vals * gate_sig_R

    # Store results
    tl.store(
        outL_ptr + (offs_m[:, None] * stride_outlm + offs_h[None, :] * stride_outlh),
        left_out,
        mask=mask_mh,
    )
    tl.store(
        outR_ptr + (offs_m[:, None] * stride_outrm + offs_h[None, :] * stride_outrh),
        right_out,
        mask=mask_mh,
    )

def custom_kernel(data):
    """
    TriMul forward pass with a fused Triton kernel for projection & gating.
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        output: torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    # Config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) Layer‑Norm on the input
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    B, N, _, _ = x_norm.shape
    M = B * N * N  # total number of (b,i,j) positions

    # ------------------------------------------------------------------
    # 2) Fuse linear projections & gating with optional mask (Triton)
    # ------------------------------------------------------------------
    # Flatten to [M, dim]
    x_flat = x_norm.reshape(M, dim).contiguous()
    mask_flat = mask.reshape(M).contiguous().to(dtype)

    # Transpose weight matrices to [dim, hidden_dim] -> [K, H]
    WLP_T = weights["left_proj.weight"].t().contiguous()
    WRP_T = weights["right_proj.weight"].t().contiguous()
    WLG_T = weights["left_gate.weight"].t().contiguous()
    WRG_T = weights["right_gate.weight"].t().contiguous()

    left_out = torch.empty((M, hidden_dim), dtype=dtype, device=device)
    right_out = torch.empty((M, hidden_dim), dtype=dtype, device=device)

    # Strides
    stride_xm, stride_xk = x_flat.stride()
    stride_wk, stride_wh = WLP_T.stride()   # same for all weight tensors
    stride_maskm = mask_flat.stride(0)
    stride_outlm, stride_outlh = left_out.stride()
    stride_outrm, stride_outrh = right_out.stride()

    # Block sizes (tuned for typical hidden_dim <= 384)
    BLOCK_M = 128
    BLOCK_H = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(hidden_dim, BLOCK_H),
    )

    proj_gate_kernel[grid](
        x_flat,
        WLP_T,
        WLG_T,
        WRP_T,
        WRG_T,
        mask_flat,
        left_out,
        right_out,
        M,
        dim,
        hidden_dim,
        stride_xm,
        stride_xk,
        stride_wk,
        stride_wh,
        stride_maskm,
        stride_outlm,
        stride_outlh,
        stride_outrm,
        stride_outrh,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    left = left_out.view(B, N, N, hidden_dim)
    right = right_out.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3) Compute out‑gate (sigmoid of a linear projection of x_norm)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4) Triangular multiplicative update (batched matrix multiplication)
    # ------------------------------------------------------------------
    # Transform to shape [B, H, N, N] for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Collapse batch & hidden dimensions → [B*H, N, N]
    left_flat = left_perm.view(B * hidden_dim, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)

    # out_flat[b*h, i, j] = Σ_k left_flat[b*h, i, k] * right_flat[b*h, j, k]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5) Final LayerNorm, gate, and projection back to dim
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_gated = out_norm * out_gate
    output = F.linear(out_gated, weights["to_out.weight"])

    return output
