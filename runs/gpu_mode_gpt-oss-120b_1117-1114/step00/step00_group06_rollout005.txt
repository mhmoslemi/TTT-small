    left_ptr, right_ptr, out_ptr,
    B_D, N,

    stride_right_batch, stride_right_j, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,

    # compute number of tiles per output per batch
    tiles_m = tl.cdiv(N, BLOCK_M)
    tiles_n = tl.cdiv(N, BLOCK_N)
    total_tiles_per_batch = tiles_m * tiles_n

    batch_idx = pid // total_tiles_per_batch
    tile_idx = pid % total_tiles_per_batch

    tile_i = tile_idx // tiles_n
    tile_j = tile_idx % tiles_n

    # row and col offsets
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)

"""
TriMul forward (outgoing) implementation with Triton acceleration.

The heavy O(N³) part:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

is performed as a batched matrix multiplication
    out_{b,d} = L_{b,d} @ R_{b,d}.T
where L_{b,d}, R_{b,d} ∈ ℝ^{N×N} are the left/right tensors for each
(batch, hidden_dim) pair.  The multiplication is tiled using a Triton
kernel to keep all three tensors (L, Rᵀ, out) in registers while streaming
over the k‑dimension.  All other operations (LayerNorm, Linear projections,
gating) are done in PyTorch.  Computations are carried out in bfloat16
inside the kernel and accumulated in float32 to match the reference
implementation (which uses bfloat16 for the einsum).
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

@triton.jit
def batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B_D, N,                                     # int scalars
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_j, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out = left @ right.T for each (batch, hidden_dim) pair.
    left, right, out have shape (B_D, N, N) and are stored in
    bfloat16.  Accumulation uses float32.
    """
    pid = tl.program_id(0)

    # tiling on the output matrix (i, j) for each batch entry
    tiles_m = tl.cdiv(N, BLOCK_M)
    tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = tiles_m * tiles_n

    batch_idx = pid // tiles_per_batch
    tile_idx  = pid % tiles_per_batch

    tile_i = tile_idx // tiles_n
    tile_j = tile_idx % tiles_n

    # coordinate offsets
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile (M x K) -----
        left_ptrs = left_ptr + \
            batch_idx * stride_left_batch + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ----- load right tile transposed (K x N) -----
        # We load Rᵀ so that its shape matches (K, N) for tl.dot
        right_ptrs = right_ptr + \
            batch_idx * stride_right_batch + \
            offs_k[:, None] * stride_right_k + \
            offs_j[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # ----- accumulate -----
        acc += tl.dot(left, right)   # (M,K) x (K,N) -> (M,N)

    # ----- store result (as bfloat16) -----
    out_ptrs = out_ptr + \
        batch_idx * stride_out_batch + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.bfloat16),
             mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # ---------- LayerNorm on input ----------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ---------- Linear projections ----------
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left  = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B,N,N,hidden_dim]

    # ---------- Apply mask ----------
    if mask_tensor is None:
        mask = torch.ones((batch_size, seq_len, seq_len), dtype=x.dtype, device=device)
    else:
        mask = mask_tensor
    mask_exp = mask.unsqueeze(-1)            # [B,N,N,1]
    left  = left  * mask_exp
    right = right * mask_exp

    # ---------- Gating ----------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left  * left_gate
    right = right * right_gate

    # ---------- Prepare for Triton kernel ----------
    # Convert to bfloat16 for the heavy O(N³) part
    left = left.to(torch.bfloat16)
    right = right.to(torch.bfloat16)

    # reshape to (B*hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hdim, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(batch_size * hidden_dim, seq_len, seq_len)
    right_mat = right_perm.view(batch_size * hidden_dim, seq_len, seq_len)

    out_mat = torch.empty_like(left_mat, dtype=torch.bfloat16)

    # strides (in elements)
    stride_left_batch, stride_left_i, stride_left_k = left_mat.stride()
    stride_right_batch, stride_right_j, stride_right_k = right_mat.stride()
    stride_out_batch, stride_out_i, stride_out_j = out_mat.stride()

    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    tiles_m = (seq_len + BLOCK_M - 1) // BLOCK_M
    tiles_n = (seq_len + BLOCK_N - 1) // BLOCK_N
    total_blocks = (batch_size * hidden_dim) * tiles_m * tiles_n
    grid = (total_blocks,)

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch_size * hidden_dim,
        seq_len,
        stride_left_batch,
        stride_left_i,
        stride_left_k,
        stride_right_batch,
        stride_right_j,
        stride_right_k,
        stride_out_batch,
        stride_out_i,
        stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Reshape back ----------
    out = out_mat.view(batch_size, hidden_dim, seq_len, seq_len) \
                 .permute(0, 2, 3, 1).contiguous()            # [B,N,N,hidden_dim]
    out = out.to(torch.float32)

    # ---------- Output LayerNorm ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear ----------
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)   # -> [B,N,N,dim]

    return out
```
