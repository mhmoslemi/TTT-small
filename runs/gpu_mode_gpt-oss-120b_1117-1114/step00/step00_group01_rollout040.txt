def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # Get config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # Ensure mask is correct
    if mask is None:
        mask = torch.ones((batch_size, seq_len, seq_len), device=input_tensor.device, dtype=input_tensor.dtype)
    # Expand mask to ... x 1 for broadcast
    mask_exp = mask.unsqueeze(-1).to(input_tensor.dtype)  # same dtype as input (float32)

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections (without bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask
    left = left * mask_exp
    right = right * mask_exp

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bd = tl.program_id(2)

    # decode batch and hidden index
    b = pid_bd // H
    d = pid_bd % H

    # base offsets for this batch and hidden channel
    b_offset = b * stride_left_b
    d_offset = d * stride_left_h   # usually 1

    # tile origin
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # pre-compute i and j indices and masks
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # clamp indices to zero when out of range (address safe)
    i_safe = tl.where(mask_i, i, 0)
    j_safe = tl.where(mask_j, j, 0)

    i_offset = i_safe * stride_left_i   # shape (BLOCK_M)
    j_offset = j_safe * stride_right_i  # shape (BLOCK_N)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_iter * BLOCK_K
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N
        k_safe = tl.where(mask_k, k, 0)
        k_offset = k_safe * stride_left_k   # shape (BLOCK_K)

        # left block (M x K)
        left_ptrs = left_ptr + b_offset + d_offset + i_offset[:, None] + k_offset[None, :]
        mask_left = mask_i[:, None] & mask_k[None, :]
        left = tl.load(left_ptrs, mask=mask_left, other=0.0).to(tl.float32)

        # right block (K x N)
        right_ptrs = right_ptr + b_offset + d_offset + k_offset[:, None] + j_offset[None, :]
        mask_right = mask_k[:, None] & mask_j[None, :]
        right = tl.load(right_ptrs, mask=mask_right, other=0.0).to(tl.float32)

        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N)

    # write out
    out_ptrs = out_ptr + b_offset + d_offset + i_offset[:, None] + j_offset[None, :]
    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)

def custom_kernel(data):
    """
    Outgoing triangle multiplicative update (TriMul) forward pass.
    Implements the heavy O(N^3) contraction with a fused Triton kernel.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # Ensure mask tensor exists and broadcastable
    if mask is None:
        mask = torch.ones((batch_size, seq_len, seq_len), device=input_tensor.device, dtype=input_tensor.dtype)
    mask = mask.to(input_tensor.dtype).unsqueeze(-1)   # [B,N,N,1]

    # LayerNorm on the last channel dimension
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Apply mask to projections
    left = left * mask
    right = right * mask

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply elementwise gates
    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for the kernel, ensure contiguity
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # Allocate output buffer for the contraction
    out = torch.empty((batch_size, seq_len, seq_len, hidden_dim), dtype=torch.float16, device=input_tensor.device)

    # Strides (in elements)
    sL_b, sL_i, sL_k, sL_h = left.stride()
    sR_b, sR_i, sR_k, sR_h = right.stride()
    sO_b, sO_i, sO_j, sO_h = out.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(seq_len, BLOCK_M),
        triton.cdiv(seq_len, BLOCK_N),
        batch_size * hidden_dim,
    )

    trimul_outgoing_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        out_ptr=out.data_ptr(),
        B=batch_size,
        N=seq_len,
        H=hidden_dim,
        stride_left_b=sL_b,
        stride_left_i=sL_i,
        stride_left_k=sL_k,
        stride_left_h=sL_h,
        stride_right_b=sR_b,
        stride_right_i=sR_i,
        stride_right_k=sR_k,
        stride_right_h=sR_h,
        stride_out_b=sO_b,
        stride_out_i=sO_i,
        stride_out_j=sO_j,
        stride_out_h=sO_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert contraction result back to FP32
    out_fp32 = out.to(torch.float32)

    # Output LayerNorm (hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_weight,
                            bias=to_out_norm_bias, eps=1e-5)

    # Apply output gate
    out_gated = out_norm * out_gate

    # Final projection back to original channel dimension
    to_out_weight = weights["to_out.weight"]
    output = F.linear(out_gated, to_out_weight)

    return output

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Fused triangle‐multiplicative update (outgoing) kernel.

    Computes, for each batch b and hidden channel d:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    The implementation is a block‑wise GEMM over the k‑dimension.
    """
    pid_i = tl.program_id(0)          # tile along the i axis
    pid_j = tl.program_id(1)          # tile along the j axis
    pid_bd = tl.program_id(2)         # combined batch‐hidden index

    # Decode batch and hidden channel
    b = pid_bd // H
    d = pid_bd % H

    # Base offsets for this (b,d) slice
    b_off = b * stride_left_b
    d_off = d * stride_left_h          # usually 1 (contiguous in last dim)

    # Origin of the current tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # ----------------------------------------------------------------------
    # Pre‑compute i and j indices, masks and safe offsets (clamped to 0)
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    i_safe = tl.where(mask_i, i, 0)
    j_safe = tl.where(mask_j, j, 0)

    i_off = i_safe * stride_left_i      # (BLOCK_M)
    j_off = j_safe * stride_right_i     # (BLOCK_N)

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------------------------------------------
    # Main loop over the reduction dimension k
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_iter * BLOCK_K
        k = k0 + tl.arange(0, BLOCK_K)

        mask_k = k < N
        k_safe = tl.where(mask_k, k, 0)
        k_off = k_safe * stride_left_k   # (BLOCK_K)

        # -------------------- load left block (i,k)
        left_ptrs = left_ptr + b_off + d_off + i_off[:, None] + k_off[None, :]
        mask_left = mask_i[:, None] & mask_k[None, :]
        left = tl.load(left_ptrs, mask=mask_left, other=0.0).to(tl.float32)

        # -------------------- load right block (k,j)
        right_ptrs = right_ptr + b_off + d_off + k_off[:, None] + j_off[None, :]
        mask_right = mask_k[:, None] & mask_j[None, :]
        right = tl.load(right_ptrs, mask=mask_right, other=0.0).to(tl.float32)

        # -------------------- block‑wise dot product
        acc += tl.dot(left, right)      # (BLOCK_M, BLOCK_N)

    # ----------------------------------------------------------------------
    # Write the result back (cast to FP16)
    out_ptrs = out_ptr + b_off + d_off + i_off[:, None] + j_off[None, :]
    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)


def custom_kernel(data):
    """
    Outgoing TriMul forward pass.
    The heavy O(N³) contraction is off‑loaded to a Triton kernel,
    while LayerNorms, linear projections and gates are handled in PyTorch.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # Prepare mask (if missing treat as all‑ones)
    if mask is None:
        mask = torch.ones((batch_size, seq_len, seq_len), device=input_tensor.device,
                         dtype=input_tensor.dtype)
    mask = mask.to(input_tensor.dtype).unsqueeze(-1)  # [B,N,N,1]

    # ------------------------------------------------------------------
    # LayerNorm on the input tensor
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Apply the pairwise mask
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # Gating pathways
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to FP16 and ensure contiguity for the Triton kernel
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # Allocate output buffer for the contraction
    out = torch.empty(
        (batch_size, seq_len, seq_len, hidden_dim),
        dtype=torch.float16,
        device=input_tensor.device,
    )

    # Strides (in elements, not bytes)
    sL_b, sL_i, sL_k, sL_h = left.stride()
    sR_b, sR_i, sR_k, sR_h = right.stride()
    sO_b, sO_i, sO_j, sO_h = out.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(seq_len, BLOCK_M),          # i‑tiles
        triton.cdiv(seq_len, BLOCK_N),          # j‑tiles
        batch_size * hidden_dim,                # (b,d) combos
    )

    # Launch the Triton kernel
    trimul_outgoing_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        out_ptr=out.data_ptr(),
        B=batch_size,
        N=seq_len,
        H=hidden_dim,
        stride_left_b=sL_b,
        stride_left_i=sL_i,
        stride_left_k=sL_k,
        stride_left_h=sL_h,
        stride_right_b=sR_b,
        stride_right_i=sR_i,
        stride_right_k=sR_k,
        stride_right_h=sR_h,
        stride_out_b=sO_b,
        stride_out_i=sO_i,
        stride_out_j=sO_j,
        stride_out_h=sO_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Post‑processing in FP32
    out_fp32 = out.to(torch.float32)

    # LayerNorm on the hidden dimension
    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply the output gate
    out_gated = out_norm * out_gate

    # Final linear projection back to the original channel dimension
    output = F.linear(out_gated, weights["to_out.weight"])

    return output
