    right_gate_weight = weights["right_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # x Norm
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left = F.linear(x, left_proj_weight)  # shape [bs,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Mask handling
    if mask is None:
        mask_tensor = torch.ones_like(input_tensor[...,0:1])  # shape [bs,N,N,1]
    else:
        mask_tensor = mask.unsqueeze(-1).to(x.dtype)  # broadcast

    left = left * mask_tensor
    right = right * mask_tensor

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Prepare for matmul
    bs, N, _, _ = left.shape

    # Permute to bring hidden dim to batch dimension
    left_perm = left.permute(0,3,1,2).contiguous()      # (bs, hidden_dim, N, N)
    right_perm = right.permute(0,3,2,1).contiguous()    # (bs, hidden_dim, N, N) where second N is k dimension, third N is j dimension

    M = bs * hidden_dim
    left_flat = left_perm.view(M, N, N)   # (M,N,N)
    right_flat = right_perm.view(M, N, N) # (M,N,N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)  # dtype float32

    # Strides
    stride_am = left_flat.stride(0)
    stride_an = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # Choose block sizes based on N
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), M)

        left_flat,
        right_flat,
        out_flat,
        M, N, N,
        stride_am, stride_an, stride_ak,
        stride_bm, stride_bk, stride_bn,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    out = out_flat.view(bs, hidden_dim, N, N).permute(0,2,3,1).contiguous()

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension.
2. Apply left/right linear projections, mask, and element‑wise sigmoid gates.
3. Compute the batch of matrix multiplications
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using a custom Triton kernel that treats each (b,d) pair as an
   independent matrix and tiles the computation along the i/j axes.
   The kernel performs a blocked GEMM in FP32.
4. Reshape the result back to [B, N, N, H] (H = hidden_dim).
5. Apply a second LayerNorm, multiply by the output gate and project back
   to the original channel dimension.

The kernel handles the full “outgoing” triangular update and delivers the
final tensor in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,                       # M = batch*hidden, N = seq_len, K = seq_len
    stride_am, stride_an, stride_ak,
    stride_bm, stride_bk, stride_bn,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication C = A @ B   (A: MxNxK, B: MxKxN)"""
    pid_batch = tl.program_id(2)   # corresponds to (batch * hidden_dim) index
    pid_i = tl.program_id(0)       # tile row over N
    pid_j = tl.program_id(1)       # tile col over N

    # Offsets for the i and j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for cur_block_k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = cur_block_k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load a tile of A (shape BLOCK_M x BLOCK_K)
        a_ptrs = (A_ptr
                  + pid_batch * stride_am
                  + offs_i[:, None] * stride_an
                  + cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load a tile of B (shape BLOCK_K x BLOCK_N)
        b_ptrs = (B_ptr
                  + pid_batch * stride_bm
                  + cur_k[:, None] * stride_bk
                  + offs_j[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Accumulate dot product
        acc += tl.dot(a, b)

    # Store the result tile
    c_ptrs = (C_ptr
              + pid_batch * stride_cm
              + offs_i[:, None] * stride_ci
              + offs_j[None, :] * stride_cj)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a fused Triton kernel
    for the O(N³) tensor contraction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (float or bool)
            - weights: dict of model weight tensors
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----- unpack weights ----------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ----- LayerNorm over the last dimension ---------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ----- Linear projections -------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ----- Apply mask --------------------------------------------------------
    if mask is None:
        mask_tensor = torch.ones_like(x[..., :1])
    else:
        mask_tensor = mask.unsqueeze(-1).to(dtype)   # [B, N, N, 1]

    left = left * mask_tensor
    right = right * mask_tensor

    # ----- Gates --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ----- Prepare tensors for the Triton GEMM -------------------------------
    B, N, _, _ = left.shape   # B = batch size
    # Bring the hidden dimension into the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()     # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # (B, H, N, N) where dim2 is the reduction axis

    M = B * hidden_dim                                 # total independent matrices
    left_flat = left_perm.view(M, N, N)                # (M, N, N)
    right_flat = right_perm.view(M, N, N)              # (M, N, N)  (rows=K, cols=N)

    # Allocate output tensor
    out_flat = torch.empty_like(left_flat)             # (M, N, N)

    # ----- Strides (in element units) ----------------------------------------
    stride_am = left_flat.stride(0)
    stride_an = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # ----- Triton launch ------------------------------------------------------
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # tiles along i
        triton.cdiv(N, BLOCK_N),   # tiles along j
        M                           # one launch per (batch * hidden_dim)
    )

    _tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M, N, N,
        stride_am, stride_an, stride_ak,
        stride_bm, stride_bk, stride_bn,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- Reshape back to [B, N, N, hidden_dim] ----------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----- Second LayerNorm ---------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ----- Apply output gate and final linear projection ----------------------
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # shape [B, N, N, dim]

    return out
