@triton.jit
def trimul_batched_matmul_kernel(a_ptr, b_ptr, c_ptr,
                                 M, N, K,
                                 stride_am, stride_ak, stride_ab,
                                 stride_bm, stride_bn, stride_bk,
                                 stride_cm, stride_cn, stride_cb,
                                 BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets for M, N
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # (tile) predicates
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator to 0
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # compute pointers
        a_ptrs = a_ptr + batch_id * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch_id * stride_cb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # load
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(a, b)

    # store results
    c_ptrs = c_ptr + batch_id * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    dim = config["dim"]  # presumably equals C

    # (1) LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    x = torch.nn.functional.layer_norm(input_tensor, (C,), weight=norm_weight, bias=norm_bias, eps=eps)

    # (2) Projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # (3) Mask (if nomask maybe not needed, but we multiply anyway)
    # mask shape [B,N,N]
    mask_exp = mask.unsqueeze(-1).to(x.dtype)
    left = left * mask_exp
    right = right * mask_exp

    # (4) Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # (5) Cast to fp16 for matmul
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # (6) Prepare for Triton kernel
    # reshape to [B*hidden_dim, N, N]
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()
    batch_dim = B * hidden_dim
    left_flat = left_perm.view(batch_dim, N, N)
    right_flat = right_perm.view(batch_dim, N, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float32)  # result shape [batch_dim, N, N]

    # (7) Launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch_dim, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    trimul_batched_matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1), # note swapped usage
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # (8) Reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim]

    # (9) to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # multiply by out_gate
    out = out * out_gate

    # to_out linear
    to_out_weight = weights["to_out.weight"]  # shape [dim, hidden_dim]
    output = torch.nn.functional.linear(out, to_out_weight)  # shape [B,N,N,dim]

    return output
```

"""
TriMul outgoing implementation with Triton acceleration.

The forward pass consists of:
1. Layer‑norm on the input tensor.
2. Linear projections and sigmoid gates (implemented in PyTorch).
3. A mask applied to the projected tensors.
4. The expensive O(N³) contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication for each (b,d) pair.
   We fuse it into a Triton kernel that computes many (i,j) tiles while
   reducing over k.  The kernel processes the batch dimension (B×hidden_dim)
   as independent matrices, using float16 inputs and float32 accumulation.
5. Layer‑norm on the intermediate result, gating, and final linear projection.
All tensor dimensions are handled generically; the kernel works for any
batch size ≤2, sequence length up to 1024 and hidden dimension up to 384.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication for the TriMul einsum.

    Computes, for each batch index `b` (here batch = B * hidden_dim):
        C[b] = A[b] @ B[b].T
    where
        A : (M, K)   ← left tensor (i,k)
        B : (N, K)   ← right tensor (j,k)  (transposed on‑the‑fly)
        C : (M, N)   ← output (i,j)

    Accumulates in float32 even when inputs are float16.
    """
    batch_id = tl.program_id(0)          # combined batch * hidden_dim
    pid_m = tl.program_id(1)             # tile row
    pid_n = tl.program_id(2)             # tile column

    # Offsets for the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to A[b, m, k] and B[b, k, n] (note B is accessed transposed)
        a_ptrs = a_ptr + batch_id * stride_a_batch \
                       + offs_m[:, None] * stride_a_m \
                       + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + batch_id * stride_b_batch \
                       + offs_k[:, None] * stride_b_k \
                       + offs_n[None, :] * stride_b_n

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Block‑wise dot product
        acc += tl.dot(a, b)

    # Store the result tile
    c_ptrs = c_ptr + batch_id * stride_c_batch \
                    + offs_m[:, None] * stride_c_m \
                    + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or uint8/float)
            - weights: dict with all linear / layer‑norm parameters
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # Unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    dim = config["dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. Input layer‑norm
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (C,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Apply mask (mask may be all‑ones when `nomask` is True)
    # -----------------------------------------------------------------
    mask_exp = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
    left = left * mask_exp
    right = right * mask_exp

    # -----------------------------------------------------------------
    # 4. Compute sigmoid gates
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates to the projected tensors
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Triton‑accelerated contraction (einsum)
    # -----------------------------------------------------------------
    # Cast to float16 for bandwidth‑limited kernels
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Rearrange to [B, hidden_dim, N, N] and flatten batch*hidden_dim
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()
    batch_dim = B * hidden_dim

    left_flat = left_perm.view(batch_dim, N, N)   # (B*H, M, K)
    right_flat = right_perm.view(batch_dim, N, N) # (B*H, N, K) – will be read transposed

    # Output buffer (float32 accumulation)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32, device=device)

    # Choose tile sizes (tuned for typical N ≤ 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _trimul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,               # M
        N,               # N
        N,               # K
        left_flat.stride(0),   # stride_a_batch
        left_flat.stride(1),   # stride_a_m
        left_flat.stride(2),   # stride_a_k
        right_flat.stride(0),  # stride_b_batch
        right_flat.stride(2),  # stride_b_k (col → reduction dim)
        right_flat.stride(1),  # stride_b_n (row after transposition)
        out_flat.stride(0),    # stride_c_batch
        out_flat.stride(1),    # stride_c_m
        out_flat.stride(2),    # stride_c_n
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output layer‑norm, gate and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
