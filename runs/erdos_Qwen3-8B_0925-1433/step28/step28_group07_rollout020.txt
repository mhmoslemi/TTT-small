import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    learning_rate = 0.1
    learning_rate_decay = 0.999
    max_iter = 1000
    perturbation_prob = 0.1
    perturbation_scale = 0.05
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        lag = max_corr_idx - (n_points - 1)
        gradient = np.zeros(n_points)
        for j in range(n_points):
            j_plus_l = j + lag
            term1 = 1 - h[j_plus_l] if j_plus_l < n_points else 0
            j_minus_l = j - lag
            term2 = h[j_minus_l] if j_minus_l >= 0 else 0
            gradient[j] = dx * (term1 - term2)
        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
