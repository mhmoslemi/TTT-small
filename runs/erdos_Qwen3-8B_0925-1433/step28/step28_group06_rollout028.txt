import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()

    def compute_overlap(h_array):
        n = len(h_array)
        h_fft = np.fft.fft(h_array, 2*n)
        one_minus_h = 1 - h_array
        one_minus_h_fft = np.fft.fft(one_minus_h, 2*n)
        corr_fft = h_fft * np.conj(one_minus_h_fft)
        corr = np.fft.ifft(corr_fft).real
        full_corr = corr[n-1:2*n-1]
        return np.max(full_corr) * dx

    def rescale_to_sum(x):
        s = np.sum(x)
        if s == 0:
            return np.zeros_like(x)
        return x * target_sum / s

    population = []
    population.append(h)
    h_mirrored = np.concatenate([h[::-1], h[1:]])
    population.append(h_mirrored[:n_points])
    x = np.linspace(0, 1, n_points)
    population.append(h * (1 + 0.15 * np.sin(2 * np.pi * x)))
    population.append(np.random.permutation(h))

    for _ in range(200):
        fitness = np.array([compute_overlap(ind) for ind in population])
        parents = [population[i] for i in np.argsort(fitness)[:10]]
        new_pop = []
        for _ in range(len(population)):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = (p1 + p2) / 2
            if np.random.rand() < 0.2:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = rescale_to_sum(child)
            new_pop.append(child)
        population = new_pop
        if time.time() - start_time > budget_s:
            break
        current_best = population[np.argmin(fitness)]
        if compute_overlap(current_best) < best_c5:
            best_h = current_best
            best_c5 = compute_overlap(current_best)

    top_candidates = [rescale_to_sum(np.clip(p, 0, 1)) for p in population[:5]]
    for cand in top_candidates:
        result = minimize(lambda x: compute_overlap(x), cand, method='SLSQP',
                          bounds=[(0, 1)]*n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          options={'maxiter': 200, 'ftol': 1e-6})

            best_h = result.x
            best_c5 = result.fun

    best_h = rescale_to_sum(np.clip(best_h, 0, 1))
    return (best_h, best_c5, n_points)
