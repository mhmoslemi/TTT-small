    best_c5 = min(best_c5, 0.381)
    start_time = time.time()
    
    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(h)
        j = np.arange(n_points) + k
        valid = (j >= 0) & (j < n_points)
        grad[valid] = 1.0 - h[j[valid]]
        return grad
    
    class PSOParticle:
        def __init__(self, pos, vel):
            self.pos = pos
            self.vel = vel
            self.pbest = pos.copy()
            self.pbest_val = objective(pos)
    
    pop_size = 20
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    particles = [PSOParticle(
        np.clip(h + 0.1 * np.random.randn(n_points), 0, 1) * target_sum / np.sum(np.clip(h + 0.1 * np.random.randn(n_points), 0, 1)), 
        np.zeros(n_points)
    ) for _ in range(pop_size)]
    global_best = min(particles, key=lambda p: p.pbest_val)
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        for p in particles:
            r1, r2 = np.random.rand(), np.random.rand()
            p.vel = inertia * p.vel + cognitive * r1 * (p.pbest - p.pos) + social * r2 * (global_best.pbest - p.pos)
            p.pos += p.vel
            p.pos = np.clip(p.pos, 0, 1)
            p.pos = p.pos * target_sum / np.sum(p.pos)
            curr_val = objective(p.pos)
            if curr_val < p.pbest_val:
                p.pbest = p.pos.copy()
                p.pbest_val = curr_val
        if global_best.pbest_val < best_c5:
            best_c5 = global_best.pbest_val
            best_h = global_best.pbest.copy()
        if np.random.rand() < 0.05:
            for _ in range(5):
                idx = np.random.randint(0, pop_size)
                particles[idx].pos = np.clip(
                    particles[idx].pos + 0.1 * np.random.randn(n_points), 0, 1
                ) * target_sum / np.sum(np.clip(particles[idx].pos + 0.1 * np.random.randn(n_points), 0, 1))
        global_best = min(particles, key=lambda p: p.pbest_val)
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        top_idx = np.argsort([p.pbest_val for p in particles])[:5]
        for idx in top_idx:
            p = particles[idx]
            res = minimize(objective, p.pbest, method='SLSQP', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], jac=compute_gradient, tol=1e-6, options={'maxiter': 100})
            if res.success and res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x.copy()
    
    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
