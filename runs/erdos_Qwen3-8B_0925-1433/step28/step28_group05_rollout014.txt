        alpha = learning_rate
        h_new = h - alpha * gradient

        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)

        # Perturb with some probability
        if np.random.rand() < perturb_prob:
            # Random permutation
            h_new = h_new[np.random.permutation(n_points)]
            h_new = h_new * target_sum / np.sum(h_new)
        elif np.random.rand() < perturb_prob * 0.5:
            # Random scale
            scale = np.random.uniform(0.8, 1.2)
            h_new = h_new * scale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)

        h = h_new

    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
