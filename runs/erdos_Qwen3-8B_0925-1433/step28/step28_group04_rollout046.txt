        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):

            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    def gen_initial_guesses(base_h):
