import numpy as np
from scipy.optimize import differential_evolution, minimize
import time
from scipy.stats.qmc import Halton

def run(seed=42, budget_s=600):

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(60)
        h[:30] = 1.0
        h = h * 30.0 / np.sum(h)
    n_points = len(h)

    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    def objective_de(x):
        h_clipped = np.clip(x, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    def objective(x):
        h_clipped = np.clip(x, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    def generate_evenly_spaced(n, phase=0):
        spacing = n // 2
        h = np.zeros(n)
        for i in range(n):
            if (i - phase) % spacing == 0:
                h[i] = 1.0
        return h
    
    def generate_sine(n):
        h = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n) / (n - 1)))
        h = np.clip(h, 0, 1)
        return h
    
    def generate_block(n, size=5):
        h = np.zeros(n)
        for i in range(0, n, 2 * size):
            h[i:i + size] = 1.0
        return h
    
    def generate_halton(n):
        m = n // 2
        if m == 0:
            return np.zeros(n)
        sampler = Halton(d=1, scramble=False)
        points = sampler.random(m)
        points *= 2.0
        h = np.zeros(n)
        for i, x in enumerate(points):
            idx = np.argmin(np.abs(x - np.linspace(0, 2, n)))
            h[idx] = 1.0
        h = h * target_sum / np.sum(h)
        return h
    
    initial_guesses = []
    for phase in [0, 1, 2]:
        guess = generate_evenly_spaced(n_points, phase)
        guess = guess * target_sum / np.sum(guess)

    initial_guesses.append(generate_sine(n_points))
    initial_guesses.append(generate_block(n_points))
    if not 'initial_h_values' in globals():
