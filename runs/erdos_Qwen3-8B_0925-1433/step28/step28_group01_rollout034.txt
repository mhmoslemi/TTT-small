from scipy.optimize import minimize
from scipy.stats.qmc import Halton

def run(seed=42, budget_s=600):

        for x in points:
            idx = np.argmin(np.abs(x - np.linspace(0, 2, n_points)))
            h[idx] = 1.0
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = h * target_sum / np.sum(h)
    
    def objective(x):
        h_clipped = np.clip(x, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    initial_guesses = [h + np.random.normal(0, 0.05, n_points) for _ in range(3)]
    for guess in initial_guesses:
        res = minimize(objective, guess, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 100})
        if res.success:

            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5
    
    res = minimize(objective, best_h, method='L-BFGS-B',
                   bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 100})
