from scipy.optimize import differential_evolution, minimize
from scipy.stats.qmc import Halton
import time

def run(seed=42, budget_s=600):

    else:

        points = halton.random(n=n_points // 2) * 2.0
        h = np.zeros(n_points)
        dx = 2.0 / n_points
        for x in points:
            idx = np.argmin(np.abs(x - np.linspace(0, 2, n_points)))
            h[idx] = 1.0
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    def objective(x):
        h_clipped = np.clip(x, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    initial_guesses = [np.random.uniform(0, 1, n_points) for _ in range(3)]
    times = 0
    
    for guess in initial_guesses:
        if times > budget_s:
            break
        start = time.time()
        res_de = differential_evolution(objective, [(0, 1)] * n_points, popsize=20, maxiter=100, tol=1e-5, mutation=(0.5, 1), recombination=0.7)
        if time.time() - start < budget_s:
            h_candidate = res_de.x.copy()
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5
        times += time.time() - start
    
    for _ in range(3):
        if times > budget_s:
            break
        start = time.time()
        res_slsqp = minimize(objective, best_h, method='SLSQP',
                            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                            bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 100})
        if time.time() - start < budget_s:
            h_candidate = res_slsqp.x.copy()
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5
        times += time.time() - start
    
    for _ in range(2):

        start = time.time()
        res_l_bfgs = minimize(objective, best_h, method='L-BFGS-B',
