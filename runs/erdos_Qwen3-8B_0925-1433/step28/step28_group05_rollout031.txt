import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_c5 = min(best_c5, 0.381)

    def obj(h_flat):
        h = h_flat.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(h_flat):
        h = h_flat.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_idx = np.argmax(corr)
        k = k_max_idx - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        result = minimize(
            obj, h, method='L-BFGS-B',
            bounds=[(0, 1)]*n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            jac=grad, tol=1e-6, options={'maxiter': 80}
        )
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = h * target_sum / np.sum(h)
