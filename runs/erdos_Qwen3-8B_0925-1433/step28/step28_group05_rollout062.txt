import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def objective_grad(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr_index = np.argmax(corr)
        k = max_corr_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = -(1 - h_vec[j])
        return grad

    def constraint(h_vec):
        return np.sum(h_vec) - target_sum

    def generate_patterns():
        sine = 0.5 + 0.5 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))
        sine = sine * target_sum / np.sum(sine)
        regular = np.zeros(n_points)
        for i in range(n_points // 2):
            regular[i * 2] = 1
        regular = regular * target_sum / np.sum(regular)
        return [h, sine, regular]

    initial_guesses = generate_patterns()

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        h_guess = guess.copy()
        result = minimize(
            objective, h_guess, method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': constraint}],
            jac=objective_grad, tol=1e-5, options={'maxiter': 300}
        )
        if result.success and result.fun < best_c5:
            best_h, best_c5 = result.x.copy(), result.fun
        if np.random.rand() < 0.3:
            h_guess = np.clip(h_guess + 0.15 * np.random.randn(n_points), 0, 1)
            h_guess = h_guess * target_sum / np.sum(h_guess)
            result = minimize(
                objective, h_guess, method='L-BFGS-B',
                bounds=[(0, 1)] * n_points,
                constraints=[{'type': 'eq', 'fun': constraint}],
                jac=objective_grad, tol=1e-5, options={'maxiter': 150}
            )
            if result.success and result.fun < best_c5:
                best_h, best_c5 = result.x.copy(), result.fun

    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
