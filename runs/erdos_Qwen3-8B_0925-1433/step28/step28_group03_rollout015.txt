import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial = np.clip(initial_h_values, 0, 1)
    initial = initial * target_sum / np.sum(initial)
    best_h = np.copy(initial)
    best_c5 = np.max(np.correlate(initial, 1 - initial, mode='full')) * dx

    def objective(x):
        return np.max(np.correlate(x, 1 - x, mode='full')) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    def generate_variants(base):
        yield base
        perm = np.random.permutation(n_points)
        yield np.clip(initial_h_values[perm], 0, 1) * target_sum / np.sum(np.clip(initial_h_values[perm], 0, 1))
        sine_vals = 0.5 + 0.2 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))
        yield sine_vals * target_sum / np.sum(sine_vals)
        rand_vals = np.random.rand(n_points)
        yield rand_vals * target_sum / np.sum(rand_vals)

    start_time = time.time()
    population = list(generate_variants(initial))
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        evals = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            evals.append((c5, h))
        evals.sort()
        top = evals[:int(len(evals)*0.2)]
        new_pop = []
        for i in range(len(top)):
            for j in range(i+1, len(top)):
                h1, h2 = top[i][1], top[j][1]
                h_child = 0.5 * h1 + 0.5 * h2 + 0.05 * np.random.randn(n_points)
                h_child = np.clip(h_child, 0, 1) * target_sum / np.sum(h_child)
                new_pop.append(h_child)
        for _ in range(len(top)):
            h_rand = np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))
            new_pop.append(h_rand)
        population = new_pop
        for h in population:
            res = minimize(objective, h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=[{'type':'eq','fun':constraint}], tol=1e-6, options={'maxiter':50})
            if res.success and res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x.copy()

    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
