from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_rescale(arr):
        arr = np.clip(arr, 0, 1)
        return arr * target_sum / np.sum(arr)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        return evaluate(x)

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')

        k = k_max_idx - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    def gen_initial_guesses(base):
        yield base.copy()
        yield clip_rescale(base * np.random.rand(n_points))
