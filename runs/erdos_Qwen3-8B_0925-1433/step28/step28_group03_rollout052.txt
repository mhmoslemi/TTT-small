import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        corr = np.correlate(x, 1 - x, mode='full')
        k_max_idx = np.argmax(corr)
        k = k_max_idx - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):

            if 0 <= j < n_points:
                grad[i] = 1.0 - x[j]
        return grad

        else:
            base = np.random.rand(n_points)
            base = base * target_sum / np.sum(base)
