import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - x[j]
        return grad

    def generate_guesses(base):
        yield rescale(base * (1 + 0.2 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))))
        yield rescale(base[np.random.permutation(n_points)])
        yield rescale(np.clip(base + np.random.normal(0, 0.1, n_points), 0, 1))
        yield rescale(base * (1 + 0.15 * np.cos(2 * np.pi * np.linspace(0, 1, n_points))))
        yield rescale(np.clip(base * (1 + 0.1 * np.random.randn(n_points)), 0, 1))
        yield rescale(base * (1 + 0.2 * np.random.randn(n_points)))
        yield rescale(base * (1 - 0.1 * np.random.randn(n_points)))

    base = np.array(initial_h_values, dtype=float)
    base = rescale(base)
    best_h = base.copy()
    best_c5 = objective(best_h)

    start_time = time.time()
    initial_guesses = list(generate_guesses(base))
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result_de = differential_evolution(objective, bounds=[(0, 1)]*n_points, strategy='best1bin', maxiter=100, popsize=30, tol=1e-6, mutation=(0.5, 1.0), recombination=0.8, disp=False)
        h_de = result_de.x.copy()
        h_de = rescale(h_de)
        res_slsqp = minimize(objective, h_de, method='SLSQP', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], jac=compute_gradient, tol=1e-6, options={'maxiter': 200})
        if res_slsqp.success and res_slsqp.fun < best_c5:

            best_c5 = res_slsqp.fun
        if time.time() - start_time > budget_s:
            break

    best_h = rescale(best_h)
    return (best_h, best_c5, n_points)
