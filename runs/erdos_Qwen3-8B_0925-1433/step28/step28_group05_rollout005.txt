import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = np.min([best_c5, 0.381])  # Conservative fallback

    # Generate diverse initial guesses
    initial_guesses = []
    t = np.linspace(0, 1, n_points)
    guess1 = 0.5 + 0.5 * np.sin(2 * np.pi * t)
    guess1 = guess1 * target_sum / np.sum(guess1)
    initial_guesses.append(guess1)
    spacing = n_points // 3
    guess2 = np.zeros(n_points)
    for i in range(n_points):
        if i % spacing == 0:
            guess2[i] = 1.0
    guess2 = guess2 * target_sum / np.sum(guess2)
    initial_guesses.append(guess2)

    start_time = time.time()
    for _ in range(400):  # 400 iterations with time budget
        if time.time() - start_time > budget_s:
            break
        current_guess = initial_guesses[np.random.choice(len(initial_guesses))]
        corr = np.correlate(current_guess, 1 - current_guess, mode='full')
        k_opt = np.argmax(corr)
        lag = k_opt - (n_points - 1)
        grad = np.zeros_like(current_guess)
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                grad[i] = -(1 - current_guess[j])
        # Gradient descent step with adaptive learning rate
        step_size = 0.1 / (1 + np.linalg.norm(grad) / 100)
        new_h = current_guess + step_size * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        new_corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(new_corr) * dx
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
        # Adaptive perturbations to escape local optima
        if np.random.rand() < 0.05:
            best_h = np.random.permutation(best_h)
            best_h = best_h * target_sum / np.sum(best_h)
        if np.random.rand() < 0.05:
            scale = np.random.uniform(0.8, 1.2)
            best_h = best_h * scale
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * target_sum / np.sum(best_h)

    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
