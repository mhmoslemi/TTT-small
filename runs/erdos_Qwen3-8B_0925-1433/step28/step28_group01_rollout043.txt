import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 60
        h = np.zeros(n_points)
        idxs = np.random.choice(n_points, size=n_points//2, replace=False)
        h[idxs] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)

    def compute_gradient(h, lag):
        grad = np.zeros_like(h)
        for j in range(n_points):
            if j + lag < n_points:
                grad[j] += 1 - h[j + lag]
            if j - lag >= 0:
                grad[j] -= h[j - lag]
        return grad

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    # Gradient descent steps
    step_size = 0.001
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        i_max = np.argmax(corr)
        lag = i_max - (n_points - 1)
        grad = compute_gradient(best_h, lag)
        best_h -= step_size * grad
        best_h = np.clip(best_h, 0, 1)

    # L-BFGS refinement
    res_l_bfgs = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-6, options={'maxiter': 100})
