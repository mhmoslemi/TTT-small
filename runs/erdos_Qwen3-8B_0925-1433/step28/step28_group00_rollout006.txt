from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    target_sum = n_points / 2.0

        return np.correlate(x, 1 - x, mode='full')[k] * dx

        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] = 1.0 - x[j]
        return grad
