import numpy as np
from scipy.optimize import differential_evolution, minimize
import time
from scipy.stats.qmc import Halton

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(60)
        for i in range(0, 60, 6):
            h[i:i+5] = 1.0
        h = h * 30.0 / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    def objective(x):
        h_clipped = np.clip(x, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    def generate_evenly_spaced(n, phase=0):
        spacing = n // 4
        h = np.zeros(n)
        for i in range(n):
            if (i - phase) % spacing == 0:
                h[i] = 1.0
        return h
    
    def generate_sine(n):
        h = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n) / (n - 1)))
        h = np.clip(h, 0, 1)
        return h
    
    def generate_block(n, size=5):
        h = np.zeros(n)
        for i in range(0, n, 2 * size):
            h[i:i + size] = 1.0
        return h
    
    def generate_halton(n):
        m = n // 2
        if m == 0:
            return np.zeros(n)
        sampler = Halton(d=1, scramble=False)
        points = sampler.random(m)
        points *= 2.0
        h = np.zeros(n)
        for i, x in enumerate(points):
            idx = np.argmin(np.abs(x - np.linspace(0, 2, n)))
            h[idx] = 1.0
        h = h * target_sum / np.sum(h)
        return h
    
    initial_guesses = [generate_evenly_spaced(n_points, phase) for phase in range(4)]
    initial_guesses.append(generate_sine(n_points))
    initial_guesses.append(generate_block(n_points))
    initial_guesses.append(generate_halton(n_points))
    
    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = differential_evolution(objective, [(0, 1)] * n_points, popsize=50, maxiter=100, tol=1e-5, mutation=(0.8, 1.0), recombination=0.9)
        h_candidate = res.x.copy()
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        current_c5 = objective(h_candidate)
        if current_c5 < best_c5:
            best_h = h_candidate.copy()
            best_c5 = current_c5
    
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='SLSQP', bounds=[(0, 1)] * n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-6, options={'maxiter': 100})
        if res.success:
            h_candidate = res.x.copy()
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5
    
    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 100})
