    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()

    learning_rate = 0.1
    max_iter = 500
    perturb_prob = 0.1

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
            j = i - lag
            if 0 <= j < n_points:
                grad[i] -= h[j] * dx
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        if np.random.rand() < perturb_prob:
            h += np.random.uniform(-0.05, 0.05, size=n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        current_c5 = max_corr * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        learning_rate *= 0.9995

    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    return (best_h, best_c5, n_points)
