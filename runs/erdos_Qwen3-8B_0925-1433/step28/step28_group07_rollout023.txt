import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    learning_rate = 0.1
    max_iter = 500
    perturb_prob = 0.1
    early_stop_iter = 50
    iter_count = 0

    start_time = time.time()

    while time.time() - start_time < budget_s:
        iter_count += 1
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        lag_index = np.argmax(corr)
        lag = lag_index - (n_points - 1)

        gradient = np.zeros(n_points)
        for j in range(n_points):
            j_plus_k = j + lag
            j_minus_k = j - lag
            term1 = 0.0
            if j_plus_k < n_points:
                term1 = 1.0 - h[j_plus_k]
            term2 = 0.0
            if j_minus_k >= 0:
                term2 = h[j_minus_k]
            gradient[j] = (term1 - term2) * dx

        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()

        if np.random.rand() < perturb_prob:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)

        if iter_count % 100 == 0:
            learning_rate = max(learning_rate - 0.001, 0.0001)

        if iter_count > early_stop_iter and np.isnan(best_c5):
            break

    return (best_h, best_c5, n_points)
