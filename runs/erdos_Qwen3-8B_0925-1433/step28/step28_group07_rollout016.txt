import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    # PSO Parameters
    population_size = 100
    inertia_weight = 0.8
    cognitive = 2.0
    social = 2.0
    max_iter = 300
    local_refine_pop = 20

    # Initialize swarm
    swarm = np.random.rand(population_size, n_points) * 2 - 1
    swarm = np.clip(swarm, 0, 1)
    swarm = np.array([s * target_sum / np.sum(s) for s in swarm], dtype=float)
    velocities = np.random.rand(population_size, n_points) * 0.1 - 0.05
    personal_best = swarm.copy()
    personal_best_values = np.array([objective(s) for s in swarm])
    global_best = swarm[np.argmin(personal_best_values)]
    global_best_value = np.min(personal_best_values)

    start_time = time.time()
    no_improvement_count = 0
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(population_size):
            # Update velocity with dynamic inertia
            w = inertia_weight * (1 - (_ / max_iter))
            velocities[i] = (w * velocities[i] +
                             cognitive * np.random.rand() * (personal_best[i] - swarm[i]) +
                             social * np.random.rand() * (global_best - swarm[i]))
            # Update position
            swarm[i] += velocities[i]
            swarm[i] = clip_normalize(swarm[i])
            # Evaluate
            curr_val = objective(swarm[i])
            if curr_val < personal_best_values[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_values[i] = curr_val
        # Update global best
        current_global = np.min(personal_best_values)
        if current_global < global_best_value:
            global_best_value = current_global
            global_best = personal_best[np.argmin(personal_best_values)]
        # Local refinement
        top_indices = np.argsort(personal_best_values)[:local_refine_pop]
        for idx in top_indices:
            h_refine = personal_best[idx].copy()
            # SLSQP optimization with tighter tolerance
