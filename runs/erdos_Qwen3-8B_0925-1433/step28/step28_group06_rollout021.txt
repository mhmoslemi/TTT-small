import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = _compute_overlap_fft(h)

    def _compute_overlap_fft(h):
        n = len(h)
        h_fft = np.fft.fft(h, 2 * n)
        oh = 1 - h
        oh_fft = np.fft.fft(oh, 2 * n)
        cross_corr = np.fft.ifft(h_fft * np.conj(oh_fft)).real
        return np.max(cross_corr) * dx

    def _objective(x):
        return _compute_overlap_fft(x)

    def _gradient(x):
        h = x
        corr = _compute_overlap_fft(h)
        grad = np.zeros_like(h)
        eps = 1e-6
        for i in range(len(h)):
            h_eps = h.copy()
            h_eps[i] += eps
            grad[i] = (_compute_overlap_fft(h_eps) - corr) / eps
        return grad

    start_time = time.time()
    perturbations = []

    # Generate perturbations
    x = np.linspace(0, 1, n_points)
    sin_perturb = h * (1 + 0.2 * np.sin(2 * np.pi * x))
    sin_perturb = np.clip(sin_perturb, 0, 1) * target_sum / np.sum(sin_perturb)
    perturbations.append(sin_perturb)

    perm_perturb = np.random.permutation(h)
    perm_perturb = np.clip(perm_perturb, 0, 1) * target_sum / np.sum(perm_perturb)
    perturbations.append(perm_perturb)

    noise_perturb = h + np.random.normal(0, 0.1, n_points)
    noise_perturb = np.clip(noise_perturb, 0, 1) * target_sum / np.sum(noise_perturb)
    perturbations.append(noise_perturb)

    for p in perturbations:
        if time.time() - start_time > budget_s:
            break
        # L-BFGS-B optimization
        res = minimize(_objective, p, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       jac=_gradient, tol=1e-6, options={'maxiter': 100})
        if res.success and res.fun < best_c5:

            best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
