        from scipy.optimize import minimize
        def objective(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            return compute_overlap(h)
        res = minimize(objective, best_h, method='SLSQP', 
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], 
                       bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 100})
        if res.success:
            h_candidate = res.x.copy()
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            c5_candidate = compute_overlap(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
    
    return (best_h, best_c5, n_points)
