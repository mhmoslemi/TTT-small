from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    start_time = time.time()
    perturbations = []

    # Generate four diverse perturbations
    x = np.linspace(0, 1, n_points)
    sin_perturb = h * (1 + 0.3 * np.sin(2 * np.pi * x))

    mirror_perturb = np.concatenate((h[:n_points//2], h[:n_points//2][::-1]))
    mirror_perturb = np.clip(mirror_perturb, 0, 1) * target_sum / np.sum(mirror_perturb)

    for p in perturbations:
        if time.time() - start_time > budget_s:

        result = minimize(objective, p, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
