import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    best_c5 = min(best_c5, 0.381)
    
    def compute_c5(h, dx):
        n = len(h)
        fft_h = np.fft.fft(h, 2 * n)
        fft_1h = np.fft.fft(1 - h, 2 * n)
        corr = np.fft.ifft(fft_h * np.conj(fft_1h)).real
        corr_values = corr[n-1:2*n-1]
        max_corr = np.max(corr_values)
        return max_corr * dx
    
    def compute_gradient(x):
        h = x
        n = len(h)
        fft_h = np.fft.fft(h, 2 * n)
        fft_1h = np.fft.fft(1 - h, 2 * n)
        corr = np.fft.ifft(fft_h * np.conj(fft_1h)).real
        corr_values = corr[n-1:2*n-1]
        k_max_index = np.argmax(corr_values)
        k = k_max_index - (n - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad
    
    start_time = time.time()
    perturbations = []
    
    # Symmetric mirroring
    h_mirrored = np.concatenate([h[::-1], h[1:-1]])
    h_mirrored = h_mirrored[:n_points]
    h_mirrored = h_mirrored * target_sum / np.sum(h_mirrored)
    perturbations.append(h_mirrored)
    
    # Sinusoidal modulation
    x = np.linspace(0, 1, n_points)
    sin_perturb = h * (1 + 0.2 * np.sin(2 * np.pi * x))
    sin_perturb = np.clip(sin_perturb, 0, 1) * target_sum / np.sum(sin_perturb)
    perturbations.append(sin_perturb)
    
    # Balanced segments
    seg_perturb = np.zeros(n_points)
    for i in range(n_points):
        seg_perturb[i] = 0.5 if (i % 5 == 0) else 0.0
    seg_perturb = seg_perturb * target_sum / np.sum(seg_perturb)
    perturbations.append(seg_perturb)
    
    # Random permutation
    perm_perturb = np.random.permutation(h)
    perm_perturb = np.clip(perm_perturb, 0, 1) * target_sum / np.sum(perm_perturb)
    perturbations.append(perm_perturb)
    
    for p in perturbations:
        if time.time() - start_time > budget_s:
            break
        result = minimize(compute_c5, p, method='SLSQP', bounds=[(0, 1)]*n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          jac=compute_gradient, tol=1e-6, options={'maxiter': 200})
        if result.success and result.fun < best_c5:
            best_h = result.x.copy()
            best_c5 = result.fun
        result_l_bfgs = minimize(compute_c5, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points,

                                 jac=compute_gradient, tol=1e-6, options={'maxiter': 100})
        if result_l_bfgs.success and result_l_bfgs.fun < best_c5:
