        grad = np.zeros(n_points)

        sum_new = np.sum(h_new)
        if sum_new != 0:
            h = h_new * target_sum / sum_new
        else:
            h = h_new
        # Check if this h is better
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
