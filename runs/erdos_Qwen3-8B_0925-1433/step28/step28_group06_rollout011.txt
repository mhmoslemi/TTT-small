import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad
    
    start_time = time.time()
    perturbations = []
    
    # Symmetric perturbation
    symmetric = np.array(initial_h_values[::-1])
    symmetric = np.clip(symmetric, 0, 1) * target_sum / np.sum(symmetric)
    perturbations.append(symmetric)
    
    # Sinusoidal perturbation
    x = np.linspace(0, 2, n_points)
    sin_perturb = initial_h_values * (1 + 0.2 * np.sin(2 * np.pi * x))
    sin_perturb = np.clip(sin_perturb, 0, 1) * target_sum / np.sum(sin_perturb)
    perturbations.append(sin_perturb)
    
    # Comb pattern
    comb = np.zeros(n_points)
    comb[::3] = 1.0
    comb = np.clip(comb, 0, 1) * target_sum / np.sum(comb)
    perturbations.append(comb)
    
    # Permutation
    perm_perturb = np.random.permutation(initial_h_values).copy()
    perm_perturb = np.clip(perm_perturb, 0, 1) * target_sum / np.sum(perm_perturb)
    perturbations.append(perm_perturb)
    
    # Noise perturbation
    noise_perturb = initial_h_values + np.random.normal(0, 0.1, n_points)
    noise_perturb = np.clip(noise_perturb, 0, 1) * target_sum / np.sum(noise_perturb)
    perturbations.append(noise_perturb)
    
    for p in perturbations:
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, p, method='SLSQP', bounds=[(0, 1)]*n_points, 
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], 
                          jac=compute_gradient, tol=1e-6, options={'maxiter': 200})
        if result.success and result.fun < best_c5:
            best_h = result.x.copy()
            best_c5 = result.fun
        result_l_bfgs = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                                 constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], 
                                 jac=compute_gradient, tol=1e-6, options={'maxiter': 100})
        if result_l_bfgs.success and result_l_bfgs.fun < best_c5:
            best_h = result_l_bfgs.x.copy()
            best_c5 = result_l_bfgs.fun
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
