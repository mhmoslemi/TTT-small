from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()
    
    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad
    
    class PSOParticle:
        def __init__(self, pos, vel):
            self.pos = pos
            self.vel = vel
            self.pbest = pos.copy()
            self.pbest_val = objective(pos)
    
    pop_size = 20
    inertia = 0.7
    cognitive = 2.0
    social = 1.5
    particles = [PSOParticle(
        np.clip(h + 0.2 * np.random.randn(n_points), 0, 1) * target_sum / np.sum(np.clip(h + 0.2 * np.random.randn(n_points), 0, 1)), 
        np.zeros(n_points)
    ) for _ in range(pop_size)]
    global_best = min(particles, key=lambda p: p.pbest_val)
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        for p in particles:
            r1, r2 = np.random.rand(), np.random.rand()
            p.vel = inertia * p.vel + cognitive * r1 * (p.pbest - p.pos) + social * r2 * (global_best.pbest - p.pos)
            p.pos += p.vel
            p.pos = np.clip(p.pos, 0, 1)
            p.pos = p.pos * target_sum / np.sum(p.pos)
            curr_val = objective(p.pos)
            if curr_val < p.pbest_val:
                p.pbest = p.pos.copy()
                p.pbest_val = curr_val
        if global_best.pbest_val < best_c5:

    for _ in range(150):
