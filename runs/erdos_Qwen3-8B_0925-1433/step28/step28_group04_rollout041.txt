        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

        yield np.clip(base_h * (1 + 0.5 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))), 0, 1) * (target_sum / np.sum(np.clip(base_h * (1 + 0.5 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))), 0, 1)))
