        current_c5 = objective(h.copy())
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Find worst lag and compute its gradient
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = compute_gradient(h)
        
        # Gradient descent step
        h -= learning_rate * grad
        # Clip values to [0, 1]
        h = np.clip(h, 0, 1)
        # Normalize to target_sum
        h = h * target_sum / np.sum(h)
        
        # Occasionally apply random perturbation
        if np.random.rand() < 0.05:
            h += np.random.randn(n_points) * perturbation_factor
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
    
    # After iterations, return the best solution
    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
