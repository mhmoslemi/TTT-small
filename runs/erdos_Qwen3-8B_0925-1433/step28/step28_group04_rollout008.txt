        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)

            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

            base = base * target_sum / np.sum(base)

            yield base + np.random.normal(0, 0.15, n_points)

            yield base * (1 + 0.25 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)))

    initial_guesses = []

    start_time = time.time()

        result_de = differential_evolution(objective, bounds=[(0, 1)]*n_points, strategy='best1bin', maxiter=100, popsize=25, tol=1e-6, mutation=(0.5, 1.0), recombination=0.8, disp=False)

        res_slsqp = minimize(objective, h_de, method='SLSQP', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], jac=compute_gradient, tol=1e-6, options={'maxiter': 200})
