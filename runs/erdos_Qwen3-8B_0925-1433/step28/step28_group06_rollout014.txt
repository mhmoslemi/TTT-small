from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad
    
    start_time = time.time()
    perturbations = []
    
    # Sinusoidal perturbation
    x = np.linspace(0, 1, n_points)

    perturbations.append(sin_perturb)
    
    # Permutation perturbation
    perm_perturb = np.random.permutation(h).copy()
    perm_perturb = np.clip(perm_perturb, 0, 1) * target_sum / np.sum(perm_perturb)
    perturbations.append(perm_perturb)

    def swap_perturb(h):

        h_new = np.clip(h_new, 0, 1) * target_sum / np.sum(h_new)
        return h_new
    perturbations.append(swap_perturb(h))
    
    # Block-based perturbation
    def block_perturb(h):
