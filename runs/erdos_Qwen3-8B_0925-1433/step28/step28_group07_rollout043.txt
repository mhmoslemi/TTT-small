    # Initialize population with initial and random samples
    pop = [np.array(initial_h_values, dtype=float)]
    for _ in range(pop_size - 1):
        rand = np.random.rand(n_points)
        rand.clip(0, 1, out=rand)
        rand *= target_sum / np.sum(rand)
        pop.append(rand)

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = obj(best_h)

    start_time = time.time()
    max_iter = 300
    F, CR = 0.8, 0.9

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(pop_size):
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            trial = pop[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial.clip(0, 1, out=trial)
            trial *= target_sum / np.sum(trial)
            if obj(trial) < obj(pop[i]):
                pop[i] = trial.copy()

        # Track global best
        for h in pop:
            if obj(h) < best_c5:
                best_c5 = obj(h)
                best_h = h.copy()

    # Local refinement
    top_idx = np.argsort([obj(h) for h in pop])[:10]
    for i in top_idx:
        h_refine = pop[i].copy()
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(obj, h_refine, method='SLSQP', bounds=[(0, 1)] * n_points, constraints=cons, tol=1e-6, options={'maxiter': 50})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    best_h.clip(0, 1, out=best_h)
    best_h *= target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
