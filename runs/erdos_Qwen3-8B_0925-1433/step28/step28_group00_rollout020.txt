        grad = np.zeros_like(x)
        for idx in top_indices:
            lag = idx - (n_points - 1)
            for i in range(n_points):
                j = i + lag
                if 0 <= j < n_points:
                    grad[i] += (1.0 - x[j])
        return grad
    
    def constraint(x):
        return np.sum(x) - target_sum
    
    while time.time() - start_time < budget_s:
        current_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(current_corr) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if np.random.rand() < 0.1:
            perturbation = np.random.normal(0, 0.05, n_points)
            h += perturbation
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        else:
            grad = compute_gradient(h)
            res = minimize(
                objective,
                h,
                method='L-BFGS-B',
                bounds=[(0, 1)] * n_points,
                constraints=[{'type': 'eq', 'fun': constraint}],
                jac=grad,
                options={'maxiter': 10, 'ftol': 1e-6}
            )
            if res.success:
                h = res.x
                h = h * target_sum / np.sum(h)
    
    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
