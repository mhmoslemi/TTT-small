import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def compute_gradient(h, top_lags, corr_vals):
        grad = np.zeros_like(h)
        for k, corr_val in zip(top_lags, corr_vals):
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    grad[i] += (1 - h[j]) * corr_val
        return grad

    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        top_indices = np.argsort(corr)[-5:]
        top_lags = [i - (n_points - 1) for i in top_indices]
        corr_vals = corr[top_indices]
        grad = compute_gradient(h, top_lags, corr_vals)
        res = minimize(
            objective,
            h,
            method='L-BFGS-B',
            jac=grad,
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            tol=1e-6,
            options={'maxiter': 100}
        )
        if res.success:
            new_h = res.x
            new_h = np.clip(new_h, 0, 1)
            s = np.sum(new_h)
            if s > 1e-12:
                new_h = new_h * target_sum / s
            else:
                new_h = np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)
            current_c5 = objective(new_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_h.copy()
            h = new_h
        if np.random.rand() < 0.1:
            noise = 0.1 * np.random.randn(n_points)
            h = np.clip(h + noise, 0, 1)
            s = np.sum(h)
            h = h * target_sum / s if s > 1e-12 else np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)
    return (best_h, best_c5, n_points)
