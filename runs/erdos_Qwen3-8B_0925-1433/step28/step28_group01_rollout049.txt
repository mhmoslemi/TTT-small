            idx = np.argmin(np.abs(x - np.linspace(0, 2, n)))
            h[idx] = 1.0
        return h
    
    initial_guesses = []
    if 'initial_h_values' in globals():
        initial_guesses.append(h)
    else:
        initial_guesses.append(generate_halton(n_points))
        for _ in range(2):
            initial_guesses.append(np.random.rand(n_points) > 0.5)
        for _ in range(2):
            h = np.zeros(n_points)
            for i in range(n_points):
                if i % 3 == 0:

            h_candidate = res_de.x.copy()

            h_candidate = h_candidate * target_sum / np.sum(h_candidate)

            if res_slsqp.success:
