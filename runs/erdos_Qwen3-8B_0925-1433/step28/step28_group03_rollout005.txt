import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx

    initial_guesses = []
    initial_guesses.append(initial_h.copy())
    initial_guesses.append(np.clip(initial_h + 0.2 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)), 0, 1) * (target_sum / np.sum(np.clip(initial_h + 0.2 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)), 0, 1))))
    initial_guesses.append(np.random.permutation(initial_h) * (target_sum / np.sum(np.random.permutation(initial_h))))

    λ = 10.0
    for guess in initial_guesses:
        h = guess.copy()
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        def objective(x):
            conv = np.correlate(x, 1 - x, mode='full')
            integrals = conv * dx
            return np.log(np.mean(np.exp(-λ * integrals)))
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if result.success:
            c5 = np.max(np.correlate(result.x, 1 - result.x, mode='full')) * dx
            if c5 < best_c5:
                best_h = result.x.copy()
                best_c5 = c5

    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
