    best_c5 = min(best_c5, 0.381)  # Conservative fallback

    # Define objective and gradient
    def objective(x):

        return np.max(corr) * dx

        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:

            return new_h, new_c5

import time

    dx = 2.0 / n_points
