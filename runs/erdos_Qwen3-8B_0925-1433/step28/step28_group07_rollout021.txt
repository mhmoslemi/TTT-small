        if np.sum(h) == 0:
            h = np.full(n_points, target_sum / n_points)
        else:
            h = h * target_sum / np.sum(h)
        if iterations % 100 == 0 and np.random.rand() < 0.1:
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            if np.sum(h) == 0:
                h = np.full(n_points, target_sum / n_points)
            else:
                h = h * target_sum / np.sum(h)
        new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        learning_rate = max(0.0001, learning_rate - 0.00001)
        iterations += 1
    
    return (best_h, best_c5, n_points)
