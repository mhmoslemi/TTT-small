        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    for h_candidate in candidates:
        current_c5 = compute_c5(h_candidate)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_candidate.copy()

    result = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, 
                      constraints=constraints, jac=compute_gradient, tol=1e-6, options={'maxiter': 200})
    if result.success and result.fun < best_c5:
        best_h = result.x
        best_c5 = result.fun

    pop_size = 50
    generations = 20
    population = [h.copy() for h in candidates]
    for gen in range(generations):
        fitness = [compute_c5(h) for h in population]
        top_indices = np.argsort(fitness)[:int(pop_size * 0.2)]
        top_candidates = [population[i] for i in top_indices]
        new_population = []
        for _ in range(pop_size - len(top_candidates)):
            parent1 = np.random.choice(top_candidates)
            parent2 = np.random.choice(top_candidates)
            child = (parent1 + parent2) / 2
            mutation = np.random.normal(0, 0.05, n_points)
            child += mutation
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = top_candidates + new_population
        for h_candidate in population:
            current_c5 = compute_c5(h_candidate)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_candidate.copy()

    result = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, 
                      constraints=constraints, jac=compute_gradient, tol=1e-6, options={'maxiter': 200})
    if result.success and result.fun < best_c5:
        best_h = result.x
        best_c5 = result.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)

    return (best_h, best_c5, n_points)
