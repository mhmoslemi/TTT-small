import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * (2.0 / n_points)
    best_c5 = min(best_c5, 0.381)

    def generate_guesses(base):
        yield base.copy()[np.random.permutation(n_points)]
        t = np.linspace(0, 1, n_points)
        sin_val = 0.5 + 0.2 * np.sin(2 * np.pi * t)
        yield np.clip(sin_val, 0, 1) * (target_sum / np.sum(np.clip(sin_val, 0, 1)))
        kernel = np.exp(-0.1 * np.arange(n_points)**2)
        yield np.clip(np.convolve(base, kernel, mode='same'), 0, 1) * (target_sum / np.sum(np.clip(np.convolve(base, kernel, mode='same'), 0, 1)))
        greedy = np.zeros(n_points)
        greedy[::2] = 1
        greedy = greedy * target_sum / np.sum(greedy)
        yield greedy

    guesses = []
    for _ in range(2):
        if np.random.rand() < 0.5:
            base = initial_h.copy()
            guesses.extend(generate_guesses(base))
        else:
            base = np.random.rand(n_points)
            base = base * target_sum / np.sum(base)
            guesses.extend(generate_guesses(base))

    start = time.time()
    for guess in guesses:
        if time.time() - start > budget_s:
            break
        def obj(x):
            h = x.copy()
            corr = np.correlate(h, 1 - h, mode='full')
            lags = corr[n_points - 1:]
            λ = 10
            return -np.log(np.mean(np.exp(-λ * lags)))

        res = minimize(obj, guess, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if res.success:

            c5 = np.max(np.correlate(h_opt, 1 - h_opt, mode='full')) * (2.0 / n_points)

    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
