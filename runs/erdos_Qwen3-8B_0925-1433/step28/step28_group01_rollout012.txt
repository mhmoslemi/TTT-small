import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    else:
        n_points = 45
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                h[i] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    def objective(x):
        h_clipped = np.clip(x, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    res_de = differential_evolution(objective, [(0, 1)] * n_points, popsize=50, maxiter=150, tol=1e-5, mutation=(0.5, 1), recombination=0.7)
    h_candidate = res_de.x.copy()
    h_candidate = np.clip(h_candidate, 0, 1)
    h_candidate = h_candidate * target_sum / np.sum(h_candidate)
    current_c5 = objective(h_candidate)
    if current_c5 < best_c5:
        best_h = h_candidate.copy()
        best_c5 = current_c5
    
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res_slsqp = minimize(objective, best_h, method='SLSQP',
                            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                            bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 150})
        if res_slsqp.success:
            h_candidate = res_slsqp.x.copy()
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5
    
    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        res_l_bfgs = minimize(objective, best_h, method='L-BFGS-B',
                             bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 150})
        if res_l_bfgs.success:
            h_candidate = res_l_bfgs.x.copy()
            h_candidate = np.clip(h_candidate, 0, 1)
