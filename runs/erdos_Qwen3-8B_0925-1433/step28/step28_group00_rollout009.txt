        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad
    
    class PSOParticle:
        def __init__(self, pos, vel):
            self.pos = pos.copy()
            self.vel = vel.copy()
            self.pbest = pos.copy()
            self.pbest_val = objective(pos)
    
    pop_size = 50
    inertia = 0.8
    cognitive = 1.5
    social = 1.5
    particles = [PSOParticle(

    ) for _ in range(pop_size)]
    global_best = min(particles, key=lambda p: p.pbest_val)
