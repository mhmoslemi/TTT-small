            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            grad = compute_gradient(h)
            res = minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, jac=grad, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-5, options={'maxiter': 20})
            if res.success:
                h = res.x
            h = np.clip(h, 0, 1) * target_sum / np.sum(h)
            if _ % 10 == 0:
                h = perturb(h)

    h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    final_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    return (h, final_c5, n_points)
