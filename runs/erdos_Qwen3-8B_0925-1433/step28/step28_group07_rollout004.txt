    # Initialize h
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)  # initial check against current record

    # Parameters
    learning_rate = 0.1
    learning_decay = 0.001
    max_iter = 500
    perturb_prob = 0.1
    perturb_amp = 0.05
    early_stop = 50

    # Track best
    best_c5 = best_c5
    best_h = best_h.copy()

    start_time = time.time()

    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        max_idx = np.argmax(corr)
        lag = (max_idx - (n_points - 1))  # compute actual lag
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + lag < n_points:
                gradient[i] = (1 - h[i + lag]) * dx
            else:
                gradient[i] = 0.0
        # Update h
        h = h - learning_rate * gradient
        # Clip and normalize
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        # Add perturbation
        if np.random.rand() < perturb_prob:
            noise = np.random.uniform(-perturb_amp, perturb_amp, size=n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        # Evaluate new c5
        new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        # Decay learning rate
        learning_rate = max(learning_rate - learning_decay, 0.0001)
        # Early stopping
        if iter > 0 and (new_c5 == best_c5 or (abs(new_c5 - best_c5) < 1e-8)):

                break
    # Final check and return
    best_h = np.clip(best_h, 0, 1)

import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    learning_rate = 0.1
    learning_decay = 0.001
    max_iter = 500
    perturb_prob = 0.1
    perturb_amp = 0.05
    early_stop = 50
    start_time = time.time()

    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + lag < n_points:
                grad[i] = (1 - h[i + lag]) * dx
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
