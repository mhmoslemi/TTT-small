    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x, top_lags, n_points):
        grad = np.zeros_like(x)
        for k in top_lags:
            s = k - (n_points - 1)
            for i in range(n_points):
                j = i + s
                if 0 <= j < n_points:
                    grad[i] += 1.0 - x[j]
        return grad

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        top_lags = np.argsort(corr)[-5:]
        grad = compute_gradient(h, top_lags, n_points)
        res = minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       jac=grad, tol=1e-6, options={'maxiter': 100})
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        else:
            h = np.clip(h + 0.1 * np.random.randn(n_points), 0, 1)
            h = h * target_sum / np.sum(h)
        if np.random.rand() < 0.05:
            h = np.clip(h + 0.1 * np.random.randn(n_points), 0, 1)
            h = h * target_sum / np.sum(h)
    return (best_h, best_c5, n_points)
