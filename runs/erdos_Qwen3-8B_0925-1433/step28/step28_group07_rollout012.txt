    dx = 2.0 / n_points

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    learning_rate = 0.1
    perturbation_prob = 0.1
    perturbation_amp = 0.05
    max_iter = 500
    early_stop_iter = 50
    update_freq = 100
    start_time = time.time()

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(cross_corr)
        m = np.argmax(cross_corr)
        k = m - (n_points - 1)
        grad = np.zeros(n_points)
        for j in range(n_points):
            j_plus = j + k
            if j_plus < n_points:
                grad[j] += 1 - h[j_plus]
            j_minus = j - k
            if j_minus >= 0:
                grad[j] -= h[j_minus]
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        if np.random.rand() < perturbation_prob:
            noise = np.random.uniform(-perturbation_amp, perturbation_amp, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        if iteration % update_freq == 0:
            learning_rate = max(learning_rate - 0.001, 0.0001)
        if iteration > 0 and abs(new_c5 - best_c5) < 1e-6:
            break
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    return (h, best_c5, n_points)
