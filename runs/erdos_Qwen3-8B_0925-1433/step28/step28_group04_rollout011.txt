from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    def gen_initial_guesses(base_h):
        base = base_h.copy() * target_sum / np.sum(base_h)

    initial_guesses = []

        result_de = differential_evolution(objective, bounds=[(0, 1)]*n_points, strategy='best1bin', maxiter=100, popsize=20, tol=1e-6, mutation=(0.5, 1.0), recombination=0.8, disp=False)
