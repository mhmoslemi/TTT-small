from scipy.optimize import minimize
import time

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k_candidates = np.where(corr == corr[k_max_index])[0]
        grads = np.zeros_like(h)
        for k_idx in k_candidates:
            k = k_idx - (n_points - 1)
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    grads[i] += 1.0 - h[j]
        return grads
    
    start_time = time.time()
    perturbations = []
    x = np.linspace(0, 1, n_points)
    sin_perturb = h * (1 + 0.25 * np.sin(2 * np.pi * x))

    perturbations.append(sin_perturb)
    perm_perturb = np.random.permutation(h).copy()

    perturbations.append(noise_perturb)
    band_perturb = h.copy()
    band_perturb += np.random.uniform(-0.2, 0.2, n_points)
    band_perturb = np.clip(band_perturb, 0, 1) * target_sum / np.sum(band_perturb)

        result = minimize(objective, p, method='SLSQP', bounds=[(0, 1)]*n_points,
