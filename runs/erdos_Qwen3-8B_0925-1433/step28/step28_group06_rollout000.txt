from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    initial_guesses = []
    mirrored_h = np.array(initial_h_values[::-1], dtype=float)
    mirrored_h = mirrored_h * target_sum / np.sum(mirrored_h)
    initial_guesses.append(mirrored_h)

        swapped_h[i] = np.random.rand()
    swapped_h = swapped_h * target_sum / np.sum(swapped_h)
    initial_guesses.append(swapped_h)
    
    population = [np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points)) for _ in range(20)]
    population.extend(initial_guesses)
    population = [np.clip(p, 0, 1) for p in population]

    start_time = time.time()
