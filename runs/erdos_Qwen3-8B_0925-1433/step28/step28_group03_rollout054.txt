import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2 / n_points
    target_sum = n_points / 2

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_idx = np.argmax(corr)
        k = k_max_idx - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    start_time = time.time()

    def generate_guesses(base):
        yield base * target_sum / np.sum(base)
        yield np.clip(base[np.random.permutation(n_points)], 0, 1) * target_sum / np.sum(np.clip(base[np.random.permutation(n_points)], 0, 1))

    initial_guesses = list(generate_guesses(h))
    for guess in initial_guesses:
