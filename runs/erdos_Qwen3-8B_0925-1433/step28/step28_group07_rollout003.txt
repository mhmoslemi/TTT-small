import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def clip_normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    # Initialize population
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = clip_normalize(h_initial)
    population = [h_initial]
    perturbed = h_initial + np.random.normal(0, 0.1, n_points)
    population.extend([clip_normalize(p) for p in perturbed[:20]])
    for _ in range(20):
        rand = np.random.rand(n_points)
        population.append(clip_normalize(rand))
    population = np.array(population)
    if len(population) < 50:
        population = np.vstack([population, np.random.rand(50 - len(population), n_points)])
        population = np.array([clip_normalize(p) for p in population])

    F = 0.8
    CR = 0.9
    max_iter = 300
    best_h = population[0]
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(len(population)):
            a, b, c = population[np.random.choice(len(population), 3, replace=False)]
            mutant = a + F * (b - c)
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = clip_normalize(trial)
            if evaluate(trial) < evaluate(population[i]):
                population[i] = trial
        objectives = np.array([evaluate(h) for h in population])
        best_idx = np.argmin(objectives)
        if objectives[best_idx] < best_c5:
            best_c5 = objectives[best_idx]
            best_h = population[best_idx]

        top_indices = np.argsort(objectives)[:10]
        for idx in top_indices:
            h_refine = population[idx].copy()
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            res = minimize(evaluate, h_refine, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-6, options={'maxiter':50})
            if res.success and evaluate(res.x) < best_c5:
                best_c5 = evaluate(res.x)
                best_h = res.x

    best_h = clip_normalize(best_h)
    return (best_h, best_c5, n_points)
