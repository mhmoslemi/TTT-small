import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] = 1.0 - h[j]
        return grad

    def generate_diverse_guesses(base):
        yield base.copy()
        yield np.clip(base + 0.15 * np.random.randn(n_points), 0, 1) * (target_sum / np.sum(np.clip(base + 0.15 * np.random.randn(n_points), 0, 1)))
        yield np.clip(base + 0.1 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)), 0, 1) * (target_sum / np.sum(np.clip(base + 0.1 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)), 0, 1)))
        yield np.clip(np.random.permutation(base), 0, 1) * (target_sum / np.sum(np.clip(np.random.permutation(base), 0, 1)))

    start_time = time.time()
    initial_guesses = list(generate_diverse_guesses(h))
    initial_guesses += [np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points)) for _ in range(2)]

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1) for _ in range(n_points)]
        result = differential_evolution(objective, bounds, strategy='best1bin', maxiter=50, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.7, disp=False)
        h_de = result.x.copy()
        h_de = np.clip(h_de, 0, 1) * target_sum / np.sum(h_de)
        res_slsqp = minimize(objective, h_de, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], jac=compute_gradient, tol=1e-6, options={'maxiter': 150})
        if res_slsqp.success and res_slsqp.fun < best_c5:
            best_h = res_slsqp.x.copy()
            best_c5 = res_slsqp.fun
        res_l_bfgs = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], jac=compute_gradient, tol=1e-6, options={'maxiter': 80})
        if res_l_bfgs.success and res_l_bfgs.fun < best_c5:
            best_h = res_l_bfgs.x.copy()
            best_c5 = res_l_bfgs.fun

    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
