from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600):

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    def gen_initial_guesses(base_h):
        yield base_h.copy()
        yield base_h.copy() * (target_sum / np.sum(base_h))
        yield np.clip(base_h + 0.1 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)), 0, 1) * (target_sum / np.sum(np.clip(base_h + 0.1 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)), 0, 1)))
        yield np.clip(base_h[np.random.permutation(n_points)], 0, 1) * (target_sum / np.sum(np.clip(base_h[np.random.permutation(n_points)], 0, 1)))

    initial_guesses = []
    for _ in range(2):
        base = np.array(initial_h_values, dtype=float)
        base = base * target_sum / np.sum(base)
        initial_guesses.extend(gen_initial_guesses(base))

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result_de = differential_evolution(
