import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale(h):
        h = np.clip(h, 0, 1)
        return h * target_sum / np.sum(h)

    def objective(x):
        return np.max(np.correlate(x, 1 - x, mode='full')) * dx

    def compute_gradient(x):
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - x[j]
        return grad

    h = np.array(initial_h_values, dtype=float)
    h = rescale(h)
    best_h = h.copy()
    best_c5 = objective(h)
    best_c5 = min(best_c5, 0.381)

    def gen_initial_guesses(base):
        yield rescale(base * (1 + 0.2 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))))
        yield rescale(base * np.random.rand(n_points))
        yield rescale(np.clip(base + np.random.normal(0, 0.1, n_points), 0, 1))
        yield rescale(base[np.random.permutation(n_points)])
        yield rescale(base * (1 + 0.1 * np.cos(2 * np.pi * np.linspace(0, 1, n_points))))
        yield rescale(np.clip(base * (1 - 0.1 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))), 0, 1))
        yield rescale(base * (1 + 0.15 * np.random.rand(n_points)))
        yield rescale(base * (1 + 0.05 * np.random.rand(n_points)))

    initial_guesses = []
    for _ in range(5):
        base = np.array(initial_h_values, dtype=float)
        base = rescale(base)
        initial_guesses.extend(gen_initial_guesses(base))

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = differential_evolution(objective, bounds=[(0, 1)]*n_points, strategy='best1bin', maxiter=300, popsize=20, tol=1e-5, mutation=(0.5, 0.8), recombination=0.9, disp=False)
        h_de = rescale(result.x)
        res_slsqp = minimize(objective, h_de, method='SLSQP', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], jac=compute_gradient, tol=1e-5, options={'maxiter': 150})
        if res_slsqp.success and res_slsqp.fun < best_c5:
            best_h = res_slsqp.x
            best_c5 = res_slsqp.fun
        res_l_bfgs = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], jac=compute_gradient, tol=1e-5, options={'maxiter': 100})
        if res_l_bfgs.success and res_l_bfgs.fun < best_c5:
            best_h = res_l_bfgs.x
            best_c5 = res_l_bfgs.fun

    best_h = rescale(best_h)
    return (best_h, best_c5, n_points)
