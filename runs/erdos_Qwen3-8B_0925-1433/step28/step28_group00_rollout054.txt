import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()
    
    def objective(x):
        return np.max(np.correlate(x, 1 - x, mode='full')) * dx
    
    def compute_gradient(x):
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] = 1.0 - x[j]
        return grad
    
    def rescale(x):
        x = np.clip(x, 0, 1)
        return x * target_sum / np.sum(x)
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, h, method='L-BFGS-B', jac=compute_gradient, bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-7, options={'maxiter': 100})
        if res.success:
            h = res.x
            h = rescale(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        if np.random.rand() < 0.01:
            h = rescale(np.clip(h + 0.15 * np.random.randn(n_points), 0, 1))
    
    h = rescale(h)
    return (h, best_c5, n_points)
