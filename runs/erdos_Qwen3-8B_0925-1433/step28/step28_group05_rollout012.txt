            j_neg = (i - k_opt) % n_points
            grad[i] = (1 - h[j]) - h[j_neg]
        return grad

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_opt = np.argmax(corr) - (n_points - 1)
        grad = compute_gradient(best_h, k_opt)
        learning_rate = 0.1
        new_h = best_h - learning_rate * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        if np.random.rand() < 0.1:
            new_h = np.random.permutation(new_h)
            new_h = new_h * target_sum / np.sum(new_h)
        elif np.random.rand() < 0.1:
            scale = np.random.uniform(0.8, 1.2)
            new_h = np.clip(new_h * scale, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h)
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
