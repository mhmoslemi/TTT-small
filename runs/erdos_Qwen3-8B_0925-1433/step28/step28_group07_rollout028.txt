import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

        x = x * target_sum / np.sum(x)
        return x

        corr = np.correlate(h_vals, 1 - h_vals, mode='full')

    eps = 1e-7
    start_time = time.time()
