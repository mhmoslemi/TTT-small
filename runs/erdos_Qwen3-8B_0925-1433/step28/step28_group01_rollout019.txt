from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

            x = np.random.uniform(0, 2)

            h[idx] = 1.0
        h = h * target_sum / np.sum(h)

    def objective(x):
        h_clipped = np.clip(x, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = objective(best_h)
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)] * n_points, tol=1e-5, options={'maxiter': 150})
        if res.success:
            h_candidate = res.x.copy()
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:

    for _ in range(3):

        res = minimize(objective, best_h, method='SLSQP', bounds=[(0, 1)] * n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-5, options={'maxiter': 150})
        if res.success:
