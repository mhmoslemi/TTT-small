from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)  # Conservative fallback

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    def gen_initial_guesses(base_h):
        yield base_h.copy() * (1 + 0.3 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)))
        yield base_h.copy() * (1 + 0.3 * np.sin(4 * np.pi * np.linspace(0, 1, n_points)))
        yield base_h.copy() * (1 + 0.3 * np.sin(6 * np.pi * np.linspace(0, 1, n_points)))
        yield base_h[np.random.permutation(n_points)] * target_sum / np.sum(base_h[np.random.permutation(n_points)])
        yield np.clip(base_h + np.random.normal(0, 0.15, n_points), 0, 1) * target_sum / np.sum(np.clip(base_h + np.random.normal(0, 0.15, n_points), 0, 1))

    initial_guesses = []
    for _ in range(3):
        if np.random.rand() < 0.5:
            base = np.array(initial_h_values, dtype=float)
            base = base * target_sum / np.sum(base)

    start_time = time.time()

        result_de = differential_evolution(

        h_de = result_de.x.copy()
        h_de = np.clip(h_de, 0, 1) * target_sum / np.sum(h_de)
        res_slsqp = minimize(
