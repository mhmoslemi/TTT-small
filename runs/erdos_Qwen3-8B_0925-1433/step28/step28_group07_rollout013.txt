import time

def run(seed=42, budget_s=600):

    decay = 0.001
    max_iter = 400

    perturb_scale = 0.05
    time_start = time.time()

    for iter in range(max_iter):
        if time.time() - time_start > budget_s:

        corr = np.correlate(h, 1 - h, mode='full')
