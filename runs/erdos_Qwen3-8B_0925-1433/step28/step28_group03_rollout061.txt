            h_fallback = h_fallback * target_sum / np.sum(h_fallback)
            lags_fallback = np.correlate(h_fallback, 1 - h_fallback, mode='full')
            c5_fallback = np.max(lags_fallback) * dx
            best_c5 = c5_fallback
            best_h = h_fallback
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
