        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        grad = compute_gradient(h)
        h -= 0.1 * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        if np.random.rand() < 0.1:
            h = h[np.random.permutation(n_points)]
            h = h * target_sum / np.sum(h)
        elif np.random.rand() < 0.1:
            scale = 1.0 + 0.2 * (np.random.rand() - 0.5)
            h = np.clip(h * scale, 0, 1)
            h = h * target_sum / np.sum(h)

    h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (h, best_c5, n_points)
