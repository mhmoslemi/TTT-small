import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) - h[i]
        return grad

    def gen_initial_guesses(base_h):
        yield base_h.copy()
        yield base_h.copy() * (target_sum / np.sum(base_h))
        yield np.clip(base_h + 0.2 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)), 0, 1) * (target_sum / np.sum(np.clip(base_h + 0.2 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)), 0, 1)))
        yield np.clip(base_h[np.random.permutation(n_points)], 0, 1) * (target_sum / np.sum(np.clip(base_h[np.random.permutation(n_points)], 0, 1)))
        yield np.clip(0.5 + 0.2 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)), 0, 1) * (target_sum / np.sum(np.clip(0.5 + 0.2 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)), 0, 1)))

    initial_guesses = []
    for _ in range(3):
        base = np.array(initial_h_values, dtype=float)
        base = base * target_sum / np.sum(base)
        initial_guesses.extend(gen_initial_guesses(base))
        base = np.random.rand(n_points)
        base = base * target_sum / np.sum(base)
        initial_guesses.extend(gen_initial_guesses(base))

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res_de = differential_evolution(objective, bounds=[(0, 1)]*n_points, strategy='best1bin', maxiter=50, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.7, disp=False)
        h_de = res_de.x.copy()
        h_de = np.clip(h_de, 0, 1) * target_sum / np.sum(h_de)

        if res_slsqp.success and res_slsqp.fun < best_c5:

            best_c5 = res_slsqp.fun
