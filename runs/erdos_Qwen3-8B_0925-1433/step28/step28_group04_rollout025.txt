from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600):

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_idx = np.argmax(corr)
        k = k_max_idx - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    initial_guesses = []

    # Generate initial guesses
    base = np.array(initial_h_values, dtype=float)
    base = base * target_sum / np.sum(base)

    initial_guesses.append(np.clip(base + np.random.normal(0, 0.15, n_points), 0, 1) * (target_sum / np.sum(np.clip(base + np.random.normal(0, 0.15, n_points), 0, 1))))

    initial_guesses.append(base[np.random.permutation(n_points)] * (target_sum / np.sum(base[np.random.permutation(n_points)])))

    start_time = time.time()

        # Differential Evolution
        bounds = [(0, 1)] * n_points

        h_de = result_de.x.copy()
        h_de = np.clip(h_de, 0, 1) * target_sum / np.sum(h_de)
        # SLSQP
        res_slsqp = minimize(objective, h_de, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], jac=compute_gradient, tol=1e-6, options={'maxiter': 200})
        if res_slsqp.success and res_slsqp.fun < best_c5:
