import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    time_start = time.time()
    best_c5 = min(best_c5, 0.381)

    for _ in range(500):
        if time.time() - time_start > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        step_size = 0.1 / (1 + _ // 10)
        h -= step_size * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        if np.random.rand() < 0.1:
            h = h[np.random.permutation(n_points)]
            h = h * target_sum / np.sum(h)
            h = np.clip(h, 0, 1)
    return (best_h, best_c5, n_points)
