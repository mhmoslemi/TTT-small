    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1) * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    def generate_population(base):
        def mirror(x):
            return np.concatenate([x[::-1], x])[:n_points]
        def sinusoidal(x):
            x = x * (1 + 0.15 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)))
            return np.clip(x, 0, 1) * target_sum / np.sum(x)
        def segmented(x):
            x = np.zeros(n_points)
            x[::n_points//4] = 0.5
            return np.clip(x, 0, 1) * target_sum / np.sum(x)
        def permuted(x):
            return np.random.permutation(x) * target_sum / np.sum(x)
        return [mirror(base), sinusoidal(base), segmented(base), permuted(base)]
    
    population = generate_population(h)
    while len(population) < 50:
        noise = np.random.normal(0, 0.1, n_points)
        temp = h + noise
        temp = np.clip(temp, 0, 1) * target_sum / np.sum(temp)
        population.append(temp)
    
    def compute_c5(h_vec):
        n = len(h_vec)
        h_fft = np.fft.fft(h_vec, 2*n)
        one_minus_h_fft = np.fft.fft(1 - h_vec, 2*n)
        cross_corr = np.fft.ifft(h_fft * np.conj(one_minus_h_fft)).real
        max_val = np.max(cross_corr)
        return max_val * dx
    
    start_time = time.time()
    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        fitness = [compute_c5(p) for p in population]
        parents = []
        for _ in range(50):
            idxs = np.random.choice(len(population), 5, replace=False)
            fitness_vals = [fitness[i] for i in idxs]
            winner_idx = idxs[np.argmin(fitness_vals)]
            parents.append(population[winner_idx])
        new_pop = []
        for i in range(0, 50, 2):
            p1, p2 = parents[i], parents[i+1]
            child = (p1 + p2) / 2 + np.random.normal(0, 0.05, n_points)
            child = np.clip(child, 0, 1) * target_sum / np.sum(child)
            new_pop.append(child)
            if i+1 < 50:
                new_pop.append((p1 + p2) / 2 + np.random.normal(0, 0.05, n_points))
        population = new_pop
    
    top = sorted(population, key=lambda x: compute_c5(x))[:5]
    for h_candidate in top:
        h_candidate = np.clip(h_candidate, 0, 1) * target_sum / np.sum(h_candidate)
        res = minimize(lambda x: compute_c5(x), h_candidate,
                       method='SLSQP',
                       bounds=[(0, 1)]*n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       tol=1e-6,
                       options={'maxiter': 200})
        if res.success and res.fun < best_c5:
            best_h = res.x.copy()
            best_c5 = res.fun
    
    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
