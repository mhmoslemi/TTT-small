import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    learning_rate = 0.1
    max_iterations = 500
    perturb_freq = 100
    early_stop = 50
    
    start_time = time.time()
    for iter in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_max_val = np.max(corr)
        max_index = np.argmax(corr)
        shift_k = max_index - (n_points - 1)
        
        gradient = np.zeros_like(h)
        for j in range(n_points):
            j_plus_k = (j + shift_k) % n_points
            j_minus_k = (j - shift_k) % n_points
            gradient[j] = (1 - h[j_plus_k]) - h[j_minus_k]
        
        h = h - learning_rate * gradient
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        if iter % perturb_freq == 0:
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        
        new_c5 = current_max_val * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        if iter > 0 and new_c5 >= best_c5 and iter >= early_stop:
            break
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
