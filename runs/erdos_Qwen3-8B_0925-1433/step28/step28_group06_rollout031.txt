from scipy.optimize import minimize, Bounds
import time
import cma

def run(seed=42, budget_s=600):

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def evaluate(x):

        np.clip(h, 0, 1, out=h)
        h = h * target_sum / np.sum(h)
        return objective(h)
    
    start_time = time.time()
    
    # CMA-ES global search
    es = cma.CMAEvolutionStrategy(h, 0.3 * (1.0 - 0.0), {'popsize': 40, 'bounds': (0.0, 1.0)})
    while time.time() - start_time < budget_s * 0.6:
        solutions = es.ask()
        obj_vals = np.array([evaluate(sol) for sol in solutions])
        es.tell(solutions, obj_vals)
        es.update()
    
    # Local refinement
    candidates = [es.result.x] + [es.ask()[0] for _ in range(4)]
    for candidate in candidates:
