import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    start_time = time.time()
    
    def compute_c5(h_val):
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def objective(x):
        return compute_c5(x)
    
    def gradient(x):
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - x[j]
        return grad
    
    class PSOParticle:
        def __init__(self, pos, vel):
            self.pos = pos.copy()
            self.vel = vel.copy()
            self.pbest = pos.copy()
            self.pbest_val = objective(pos)
    
    pop_size = 30
    inertia = 0.8
    cognitive = 1.5
    social = 1.5
    particles = []
    for _ in range(pop_size):
        pos = h + 0.15 * np.random.randn(n_points)
        pos = np.clip(pos, 0, 1)
        pos = pos * target_sum / np.sum(pos)
        vel = 0.01 * np.random.randn(n_points)
        particles.append(PSOParticle(pos, vel))
    global_best = min(particles, key=lambda p: p.pbest_val)
    
    for _ in range(600):
        if time.time() - start_time > budget_s:
            break
        for p in particles:
            r1, r2 = np.random.rand(), np.random.rand()
            p.vel = inertia * p.vel + cognitive * r1 * (p.pbest - p.pos) + social * r2 * (global_best.pbest - p.pos)
            p.pos += p.vel
            p.pos = np.clip(p.pos, 0, 1)
            p.pos = p.pos * target_sum / np.sum(p.pos)
            curr_val = objective(p.pos)
            if curr_val < p.pbest_val:
                p.pbest = p.pos.copy()
                p.pbest_val = curr_val
        global_best = min(particles, key=lambda p: p.pbest_val)
        if global_best.pbest_val < best_c5:
            best_c5 = global_best.pbest_val
            best_h = global_best.pbest.copy()
        if np.random.rand() < 0.01:
            for _ in range(5):
                idx = np.random.randint(0, pop_size)
                particles[idx].pos = np.clip(
                    particles[idx].pos + 0.15 * np.random.randn(n_points), 0, 1
                ) * target_sum / np.sum(np.clip(particles[idx].pos + 0.15 * np.random.randn(n_points), 0, 1))
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        top_idx = np.argsort([p.pbest_val for p in particles])[:5]
        for idx in top_idx:
            p = particles[idx]
            res = minimize(objective, p.pbest, method='L-BFGS-B', jac=gradient, bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-6, options={'maxiter': 50})
            if res.success and res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x.copy()
    
    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
