import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] = 1.0 - h[j]
        return grad

    initial_guesses = []
    base = h.copy()
    initial_guesses.append(base)
    for _ in range(3):
        permuted = base[np.random.permutation(n_points)]
        initial_guesses.append(permuted * target_sum / np.sum(permuted))
        sin_pattern = 0.5 + 0.3 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))
        initial_guesses.append(sin_pattern * target_sum / np.sum(sin_pattern))
        noise = np.random.normal(0, 0.05, n_points)
        initial_guesses.append((base + noise) * target_sum / np.sum(base + noise))

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = differential_evolution(objective, bounds=[(0, 1)]*n_points, strategy='best1bin', maxiter=50, popsize=20, tol=1e-6, mutation=(0.5, 1), recombination=0.7, disp=False)
        h_de = result.x.copy()
        h_de = np.clip(h_de, 0, 1) * target_sum / np.sum(h_de)
        res = minimize(objective, h_de, method='SLSQP', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], jac=compute_gradient, tol=1e-6, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_h = res.x.copy()
            best_c5 = res.fun
        res_l_bfgs = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], jac=compute_gradient, tol=1e-6, options={'maxiter': 50})
        if res_l_bfgs.success and res_l_bfgs.fun < best_c5:

            best_c5 = res_l_bfgs.fun

    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
