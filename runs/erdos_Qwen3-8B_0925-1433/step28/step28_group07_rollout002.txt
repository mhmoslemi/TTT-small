import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def clip_normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    population_size = 50
    F = 0.8
    CR = 0.9
    max_iter = 300
    local_refine_pop = 10
    local_refine_iter = 50

    swarm = np.random.rand(population_size, n_points) * 2 - 1
    swarm = np.clip(swarm, 0, 1)
    swarm = np.array([s * target_sum / np.sum(s) for s in swarm], dtype=float)
    best_swarm = swarm.copy()
    best_swarm_val = np.array([objective(s) for s in swarm])

    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = swarm[a] + F * (swarm[b] - swarm[c])
            trial = np.clip(swarm[i] + np.random.rand(n_points) * (mutant - swarm[i]), 0, 1)
            trial = trial * target_sum / np.sum(trial)
            curr_val = objective(trial)
            if curr_val < best_swarm_val[i]:
                swarm[i] = trial
                best_swarm_val[i] = curr_val
        best_swarm = swarm[np.argmin(best_swarm_val)]
        best_swarm_val = best_swarm_val[np.argmin(best_swarm_val)]
        if best_swarm_val < best_c5:
            best_c5 = best_swarm_val
            best_h = best_swarm.copy()

    top_indices = np.argsort(best_swarm_val)[:local_refine_pop]
    for idx in top_indices:
        h_refine = best_swarm[idx].copy()
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(objective, h_refine, method='SLSQP', bounds=[(0, 1)]*n_points,
                       constraints=cons, tol=1e-6, options={'maxiter': local_refine_iter})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x.copy()

    best_h = clip_normalize(best_h)
    return (best_h, best_c5, n_points)
