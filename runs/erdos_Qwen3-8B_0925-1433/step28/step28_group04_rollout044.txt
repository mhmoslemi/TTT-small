import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    def generate_initial_guesses(base):
        guesses = []
        guesses.append(base.copy())
        guesses.append(base[np.random.permutation(n_points)].copy())
        guesses.append(np.clip(base + np.random.normal(0, 0.1, n_points), 0, 1))
        guesses.append(base * (1 + 0.2 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))))
        guesses.append(base * (1 + 0.15 * np.cos(2 * np.pi * np.linspace(0, 1, n_points))))
        guesses.append((1 - base) * (np.random.rand(n_points) < 0.5))
        guesses.append(np.clip(base * (1 + 0.1 * np.random.rand(n_points)), 0, 1))
        return guesses

    base = np.array(initial_h_values, dtype=float)
    base = base * target_sum / np.sum(base)
    best_h = base.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    start_time = time.time()
    initial_guesses = generate_initial_guesses(base)
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        guess = np.clip(guess, 0, 1)
        guess = guess * (target_sum / np.sum(guess))
        res = minimize(objective, guess, method='SLSQP', bounds=[(0, 1)]*n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       jac=compute_gradient, options={'maxiter': 200})
        if res.success:
            current_c5 = evaluate_erdos_solution(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x.copy()
    best_h = np.clip(best_h, 0, 1) * (target_sum / np.sum(best_h))
    return (best_h, best_c5, n_points)
