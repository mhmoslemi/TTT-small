import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in locals() and len(initial_h_values) > 0:
        h_initial = np.array(initial_h_values, dtype=float)
    else:
        h_initial = np.random.rand(45)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * 45 / 2 / np.sum(h_initial)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def clip_norm(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x
    
    population_size = 50
    F = 0.8
    CR = 0.9
    max_iter = 300
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    population = [h_initial.copy() for _ in range(population_size)]
    
    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = clip_norm(trial)
            if objective(trial) < objective(population[i]):
                population[i] = trial.copy()
        current = [objective(x) for x in population]
        idx = np.argmin(current)
        if current[idx] < best_c5:
            best_c5 = current[idx]
            best_h = population[idx].copy()
    
    top = np.argsort([objective(x) for x in population])[:10]
    for i in top:
        h_refine = population[i].copy()
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(objective, h_refine, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-6, options={'maxiter':50})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x.copy()
    
    best_h = clip_norm(best_h)
    return (best_h, best_c5, n_points)
