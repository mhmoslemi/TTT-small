import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()
    
    def compute_gradient(k, h):
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] = 1.0 - h[j]
        return grad
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr) - (n_points - 1)
        k = np.clip(k, 0, n_points - 1)
        grad = compute_gradient(k, h)
        result = minimize(lambda x: np.correlate(x, 1 - x, mode='full')[k] * dx,

                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          jac=grad,
                          tol=1e-6)
        if result.success:
            h = result.x
            current_c5 = np.correlate(h, 1 - h, mode='full')[k] * dx

            h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    return (best_h, best_c5, n_points)
