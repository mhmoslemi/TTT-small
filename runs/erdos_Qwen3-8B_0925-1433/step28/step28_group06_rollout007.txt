import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = np.clip(h_initial, 0.0, 1.0)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx

    def objective_de(x):
        sum_h = np.sum(x)
        if sum_h == 0:
            return float('inf')
        h_scaled = x * target_sum / sum_h
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    # Run differential evolution for global search
    result_de = differential_evolution(
        objective_de,
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        strategy='best1bin',
        popsize=20,
        maxiter=50,
        tol=1e-6,
        polish=True,
        disp=False
    )
    if time.time() - start_time < budget_s:
        scaled_x_de = result_de.x * target_sum / np.sum(result_de.x)
        best_h = scaled_x_de.copy()
        best_c5 = result_de.fun

    # Run local refinement with L-BFGS-B
    def objective_l_bfgs(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds_l_bfgs = [(0.0, 1.0) for _ in range(n_points)]
    result_l_bfgs = minimize(
        objective_l_bfgs,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds_l_bfgs,
        constraints=constraints,
        tol=1e-6,
        options={'maxiter': 100}
    )
    if result_l_bfgs.success:
        best_h = result_l_bfgs.x.copy()
        best_c5 = result_l_bfgs.fun

    # Final constraint enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
