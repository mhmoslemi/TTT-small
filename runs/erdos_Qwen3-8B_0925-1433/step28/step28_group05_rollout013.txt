import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        dx_val = 2.0 / n_points
        for j in range(n_points):
            term1 = 0.0
            if j + k < n_points:
                term1 = (1.0 - h[j + k]) * dx_val
            term2 = 0.0
            if j - k >= 0:
                term2 = -h[j - k] * dx_val
            grad[j] = term1 + term2
        return grad

    def gen_initial_guesses(base_h):
        yield base_h.copy() * (1 + 0.3 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)))
        yield base_h.copy() * (1 + 0.3 * np.sin(4 * np.pi * np.linspace(0, 1, n_points)))
        yield base_h.copy() * (1 + 0.3 * np.sin(6 * np.pi * np.linspace(0, 1, n_points)))
        yield np.clip(base_h + np.random.normal(0, 0.15, n_points), 0, 1) * target_sum / np.sum(np.clip(base_h + np.random.normal(0, 0.15, n_points), 0, 1))
        yield base_h[np.random.permutation(n_points)] * target_sum / np.sum(base_h[np.random.permutation(n_points)])
        yield np.clip(base_h * (1 + np.random.normal(0, 0.1, n_points)), 0, 1) * target_sum / np.sum(np.clip(base_h * (1 + np.random.normal(0, 0.1, n_points)), 0, 1))

    initial_guesses = []
    for _ in range(4):
        if np.random.rand() < 0.5:
            base = np.array(initial_h_values, dtype=float)
            base = base * target_sum / np.sum(base)
            initial_guesses.extend(gen_initial_guesses(base))
        else:
            initial_guesses.append(h.copy())

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            jac=compute_gradient,
            tol=1e-6,
            options={'maxiter': 100}

        if result.success:
            h_candidate = np.clip(result.x, 0, 1) * target_sum / np.sum(np.clip(result.x, 0, 1))
            current_c5 = np.max(np.correlate(h_candidate, 1 - h_candidate, mode='full')) * dx
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
