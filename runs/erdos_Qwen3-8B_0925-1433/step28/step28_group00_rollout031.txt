def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    h = initial_h * target_sum / np.sum(initial_h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()
    for _ in range(1000):  # Maximum iterations
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k_index = np.argmax(corr)
        k = k_index - (n_points - 1)

        # Compute gradient for this k
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] = 1.0 - h[j]

        # Define objective function
        def objective(x):
            shifted = np.roll(x, -k)
            return np.dot(x, 1 - shifted) * dx

        # Define gradient function
        def gradient(x):
            grad = np.zeros_like(x)
            for i in range(n_points):
                j = (i + k) % n_points
                grad[i] = 1.0 - x[j]
            return grad

        # Optimization
        try:
            res = minimize(
                objective,
                h,
                method='L-BFGS-B',
                bounds=[(0, 1)] * n_points,
                jac=gradient,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                tol=1e-6
            )
            if res.success:
                h = res.x
                # Clip and rescale to ensure bounds and sum
                h = np.clip(h, 0, 1)
                h = h * target_sum / np.sum(h)
                # Compute new C5
                new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
        except:
            pass

        # Random mutation with 5% chance
        if np.random.rand() < 0.05:
            perturb = 0.15 * np.random.randn(n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)

    return (best_h, best_c5, n_points)
