def run(seed=42, budget_s=600):

    h = np.array(initial_h_values, dtype=float)

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()
    
    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def compute_gradient(h, corr_values, top_lags):
        grad = np.zeros_like(h)
        for lag in top_lags:
            corr_val = corr_values[lag]
            k = lag
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    grad[i] += corr_val * (1 - h[j])
        return grad
