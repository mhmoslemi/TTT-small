    best_c5 = min(best_c5, 0.381)

    def objective(x):

    def compute_gradient(x):

        k_opt = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):

            if 0 <= j < n_points:
