import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad
    
    start_time = time.time()
    
    def generate_transformed_h(h_base):
        h_sym = h_base[::-1].copy()
        h_sym = np.concatenate([h_sym[:n_points//2], h_sym[n_points//2:]]) if n_points % 2 == 0 else h_sym
        h_periodic = np.zeros(n_points)
        for i in range(n_points):
            h_periodic[i] = h_base[i % (n_points//2)]
        h_periodic = h_periodic * target_sum / np.sum(h_periodic)
        h_swap = h_base.copy()
        indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
        h_swap[indices] = h_swap[np.random.permutation(indices)]
        h_swap = h_swap * target_sum / np.sum(h_swap)
        h_sine = h_base * (1 + 0.25 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)))
        h_sine = np.clip(h_sine, 0, 1) * target_sum / np.sum(h_sine)
        return [h_sym, h_periodic, h_swap, h_sine]
    
    transformed_h = generate_transformed_h(h)
    
    for t_h in transformed_h:
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, t_h, method='SLSQP', bounds=[(0, 1)]*n_points, 
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], 
                          jac=compute_gradient, tol=1e-6, options={'maxiter': 200})
        if result.success and result.fun < best_c5:
            best_h = result.x.copy()
            best_c5 = result.fun
        result_l_bfgs = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                                 constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], 
                                 jac=compute_gradient, tol=1e-6, options={'maxiter': 100})
        if result_l_bfgs.success and result_l_bfgs.fun < best_c5:
            best_h = result_l_bfgs.x.copy()
            best_c5 = result_l_bfgs.fun
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
