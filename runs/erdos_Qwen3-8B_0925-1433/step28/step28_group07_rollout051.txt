import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h, best_c5 = h.copy(), np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def clip_norm(x):
        return np.clip(x, 0, 1) * target_sum / np.sum(np.clip(x, 0, 1))

    def objective(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    pop = [clip_norm(initial_h_values + 0.1 * np.random.randn(n_points)) for _ in range(50)]
    pop = [clip_norm(p) for p in pop]
    pop.append(clip_norm(initial_h_values))
    F, CR = 0.8, 0.9
    max_iter = 300
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break

            a, b, c = np.random.choice(pop, 3, replace=False)
            mutant = a + F * (b - c)
            trial = np.where(np.random.rand(n_points) < CR, mutant, pop[i])
            trial = clip_norm(trial)
            if objective(trial) < objective(pop[i]):
                pop[i] = trial
        curr_min = min(pop, key=objective)
        curr_val = objective(curr_min)
        if curr_val < best_c5:
