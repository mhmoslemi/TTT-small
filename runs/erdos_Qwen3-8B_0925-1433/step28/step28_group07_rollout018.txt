import time

        return np.max(np.correlate(x, 1 - x, mode='full')) * dx

    population_size = 50

    CR = 0.9
    max_iter = 300
    local_refine_pop = 10
    local_iter = 50

    global_best_val = np.min(personal_best_vals)

            mutant = swarm[a] + F * (swarm[b] - swarm[c])
