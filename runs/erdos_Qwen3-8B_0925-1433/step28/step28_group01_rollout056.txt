import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.stats.qmc import Halton
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 60
        h = np.zeros(n_points)
        dx = 2.0 / n_points
        halton = Halton(d=1, scramble=False)
        points = halton.random(n=30) * 2
        for x in points:
            idx = np.argmin(np.abs(x - np.linspace(0, 2, n_points)))
            h[idx] = 1.0
    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    def objective(x):
        h_clip = np.clip(x, 0, 1)
        h_scaled = h_clip * target_sum / np.sum(h_clip)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    best_c5 = min(best_c5, 0.381)
    
    def generate_guesses():
        guesses = []
        # Halton
        h_halton = np.zeros(n_points)
        points = Halton(d=1, scramble=False).random(n=30) * 2
        for x in points:
            idx = np.argmin(np.abs(x - np.linspace(0, 2, n_points)))
            h_halton[idx] = 1.0
        guesses.append(h_halton)
        # Random
        h_rand = np.zeros(n_points)
        idxs = np.random.choice(n_points, 30, replace=False)
        h_rand[idxs] = 1.0
        guesses.append(h_rand)
        # Periodic
        h_per = np.zeros(n_points)
        h_per[::2] = 1.0
        guesses.append(h_per)
        return [np.clip(g * target_sum / np.sum(g), 0, 1) for g in guesses]
    
    initial_guesses = generate_guesses()
    
    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = differential_evolution(
            objective, [(0, 1)] * n_points, popsize=20, maxiter=50,
            tol=1e-8, mutation=(0.5, 1), recombination=0.7
        )
        h_candidate = res.x.copy()
        h_candidate = np.clip(h_candidate * target_sum / np.sum(h_candidate), 0, 1)
        current_c5 = objective(h_candidate)
        if current_c5 < best_c5:
            best_h = h_candidate.copy()
            best_c5 = current_c5
    
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(
            objective, best_h, method='SLSQP',
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 100}
        )
        if res.success:
            h_candidate = res.x.copy()
            h_candidate = np.clip(h_candidate * target_sum / np.sum(h_candidate), 0, 1)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5
    
    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        res = minimize(
            objective, best_h, method='L-BFGS-B',
            bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 100}

        if res.success:
            h_candidate = res.x.copy()

    return (best_h, best_c5, n_points)
