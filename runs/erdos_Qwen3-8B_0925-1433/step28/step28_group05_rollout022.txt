    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

    best_h = h.copy()

        return np.max(corr) * dx

        return grad

        h_periodic = 0.8 + 0.2 * np.sin(2 * np.pi * np.random.uniform(1, 4) * x)
        h_periodic = np.clip(h_periodic, 0, 1)
        h_periodic = h_periodic * target_sum / np.sum(h_periodic)
