import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        return np.max(np.correlate(x, 1 - x, mode='full')) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    def compute_gradient(x):
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - x[j]
        return grad

    initial_guesses = []
    base = np.array(initial_h_values, dtype=float)
    base = base * target_sum / np.sum(base)
    initial_guesses.append(base)
    initial_guesses.append(np.roll(base, int(n_points // 3)))
    initial_guesses.append(base * (1 + 0.1 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))))
    initial_guesses.append(base[np.random.permutation(n_points)])
    initial_guesses.append(base + 0.1 * np.random.rand(n_points))

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1)] * n_points
        result_de = differential_evolution(lambda x: objective(x), bounds, strategy='best1bin', maxiter=50, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.7, disp=False)
        h_de = result_de.x
        h_de = np.clip(h_de, 0, 1) * target_sum / np.sum(h_de)
        res_sqp = minimize(objective, h_de, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}], jac=compute_gradient, tol=1e-6, options={'maxiter': 50})
        if res_sqp.success and res_sqp.fun < best_c5:
            best_c5 = res_sqp.fun
            best_h = res_sqp.x.copy()

    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
