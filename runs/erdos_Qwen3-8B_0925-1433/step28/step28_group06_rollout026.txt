    h_sin = h_initial * (1 + 0.15 * np.sin(2 * np.pi * x / 2))
    h_sin = np.clip(h_sin, 0, 1)
    h_sin = h_sin * target_sum / np.sum(h_sin)
    variations.append(h_sin)
    h_perm = np.random.permutation(h_initial).copy()
    h_perm = h_perm * target_sum / np.sum(h_perm)
    variations.append(h_perm)
    k = 3
    h_periodic = np.zeros(n_points)
    for i in range(n_points):
        if i % k == 0:
            h_periodic[i] = 1.0
    h_periodic = h_periodic * target_sum / np.sum(h_periodic)
    variations.append(h_periodic)

    for h in variations:
        res = minimize(objective, h, method='SLSQP', bounds=[(0, 1)]*n_points,
                       constraints=[{'type': 'eq', 'fun': constraint}],
                       jac=compute_gradient, tol=1e-6, options={'maxiter': 200})
        if res.success and res.fun < best_c5:
            best_h = res.x.copy()
            best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
