import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values, dtype=float)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h.copy()
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.max(np.fft.ifft(np.fft.fft(h, 2*n_points) * np.conj(np.fft.fft(1-h, 2*n_points)))[-n_points:].real) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        h = x
        fft_len = 2 * n_points
        h_fft = np.fft.fft(h, fft_len)
        g_fft = np.fft.fft(1 - h, fft_len)
        corr = np.fft.ifft(h_fft * np.conj(g_fft)).real
        return np.max(corr[-n_points:].real) * dx

    def compute_gradient(x):
        h = x
        fft_len = 2 * n_points
        h_fft = np.fft.fft(h, fft_len)
        g_fft = np.fft.fft(1 - h, fft_len)
        corr = np.fft.ifft(h_fft * np.conj(g_fft)).real
        k_max = np.argmax(corr[-n_points:].real)
        k = k_max - n_points + 1
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    start_time = time.time()
    # Generate structured initial guesses
    guesses = []
    # Symmetric mirroring
    h_mirrored = np.concatenate([h[::-1], h[1:]])
    h_mirrored = h_mirrored * target_sum / np.sum(h_mirrored)
    guesses.append(h_mirrored)
    # Sinusoidal modulation
    sin_perturb = h * (1 + 0.25 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)))
    sin_perturb = np.clip(sin_perturb, 0, 1) * target_sum / np.sum(sin_perturb)
    guesses.append(sin_perturb)
    # Random permutation
    perm_perturb = np.random.permutation(h)
    perm_perturb = np.clip(perm_perturb, 0, 1) * target_sum / np.sum(perm_perturb)
