import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    base_h = np.array(initial_h_values, dtype=float)
    base_h = base_h * target_sum / np.sum(base_h)
    best_h = base_h.copy()
    best_c5 = np.max(np.correlate(base_h, 1 - base_h, mode='full')) * dx

    initial_guesses = []
    for _ in range(5):
        perturb = np.random.normal(0, 0.15, n_points)
        guess = base_h + perturb
        guess = np.clip(guess, 0, 1) * target_sum / np.sum(np.clip(guess, 0, 1))
        initial_guesses.append(guess)

        perm = np.random.permutation(base_h)
        perm = np.clip(perm, 0, 1) * target_sum / np.sum(np.clip(perm, 0, 1))
        initial_guesses.append(perm)

    start_time = time.time()

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break

        bounds = [(0, 1)] * n_points
        result = differential_evolution(lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx, bounds, strategy='best1bin', maxiter=100, popsize=15, tol=1e-6, mutation=(0.6, 1.0), recombination=0.8, disp=False)
        h_de = result.x.copy()
        h_de = np.clip(h_de, 0, 1) * target_sum / np.sum(h_de)

        def objective(x):
            return np.max(np.correlate(x, 1 - x, mode='full')) * dx

        def constraint(x):
            return np.sum(x) - target_sum

        cons = [{'type': 'eq', 'fun': constraint}]
        result_slqp = minimize(objective, h_de, method='SLSQP', bounds=[(0, 1)] * n_points, constraints=cons, tol=1e-6, options={'maxiter': 100})
        h = result_slqp.x.copy()
        h = np.clip(h, 0, 1) * target_sum / np.sum(h)

        c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
