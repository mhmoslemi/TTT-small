from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    
    # Generate initial population
    pop_size = 50
    pop = []
    for _ in range(pop_size):
        if np.random.rand() < 0.5:
            p = np.roll(h * (1 + 0.2 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))), np.random.randint(n_points))
        elif np.random.rand() < 0.3:
            p = np.zeros(n_points)
            p[np.random.choice(n_points, size=n_points // 3, replace=False)] = 1
        else:
            p = np.random.permutation(h)
        p = np.clip(p, 0, 1) * target_sum / np.sum(p)
        pop.append(p)
    
    # Genetic algorithm
    for gen in range(150):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = [objective(p) for p in pop]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = pop[best_idx].copy()
        
        # Selection and crossover
        indices = np.argsort(fitness)
        selected = [pop[i] for i in indices[:20]]
        new_pop = []
        for _ in range(pop_size):
            p1, p2 = np.random.choice(selected, 2, replace=False)
            child = (p1 + np.random.normal(0, 0.1, n_points)) * 0.5 + (p2 * 0.5)
            child = np.clip(child, 0, 1) * target_sum / np.sum(child)
            new_pop.append(child)
        pop = new_pop
    
    # Local refinement
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)] * n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
