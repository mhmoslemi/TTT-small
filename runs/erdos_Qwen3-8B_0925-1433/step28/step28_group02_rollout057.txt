import numpy as np
from scipy.optimize import differential_evolution, minimize
import time
from scipy.stats.qmc import Halton

def run(seed=42, budget_s=600):

    else:
        n_points = 70
        h = np.random.rand(n_points) * 2
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    def objective(x):
        h_clipped = np.clip(x, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    best_c5 = min(best_c5, 0.381)
    
    def generate_evenly_spaced(n, phase=0):
        spacing = n // 2
        h = np.zeros(n)
        for i in range(n):
            if (i - phase) % spacing == 0:
                h[i] = 1.0
        return h
    
    def generate_sine(n):
        h = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n) / (n - 1)))
        h = np.clip(h, 0, 1)
        return h
    
    def generate_block(n, size=5):
        h = np.zeros(n)
        for i in range(0, n, 2 * size):
            h[i:i+size] = 1.0
        return h
    
    def generate_halton(n):
        m = n // 2
        if m == 0:
            return np.zeros(n)
        sampler = Halton(d=1, scramble=False)
        points = sampler.random(m)
        points *= 2.0
        h = np.zeros(n)
        for i, x in enumerate(points):
            idx = np.argmin(np.abs(x - np.linspace(0, 2, n)))
            h[idx] = 1.0
        h = h * target_sum / np.sum(h)
        return h
    
    initial_guesses = []
    for phase in [0, 1, 2]:
        guess = generate_evenly_spaced(n_points, phase)
        guess = guess * target_sum / np.sum(guess)
        initial_guesses.append(guess)
    initial_guesses.append(generate_sine(n_points))
    initial_guesses.append(generate_block(n_points))
    initial_guesses.append(generate_halton(n_points))
    
    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = differential_evolution(objective, [(0, 1)]*n_points, popsize=40, maxiter=100, tol=1e-5)
        h_candidate = res.x.copy()
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        current_c5 = objective(h_candidate)
        if current_c5 < best_c5:
            best_h = h_candidate.copy()
            best_c5 = current_c5
    
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       bounds=[(0, 1)]*n_points, tol=1e-6, options={'maxiter': 100})
        if res.success:
            h_candidate = res.x.copy()
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:

                best_c5 = current_c5
    
    for _ in range(2):
        if time.time() - start_time > budget_s:
