import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    best_h = np.array(initial_h_values, dtype=float)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad
    
    start_time = time.time()
    perturbations = []
    # Symmetric mirroring
    mirrored = np.array(initial_h_values)
    mirrored = mirrored[::-1]
    mirrored = mirrored * target_sum / np.sum(mirrored)
    perturbations.append(mirrored)
    # Periodic extension
    periodic = np.array(initial_h_values)
    periodic = np.repeat(periodic[:n_points//2], 2)
    periodic = periodic[:n_points]
    periodic = periodic * target_sum / np.sum(periodic)
    perturbations.append(periodic)
    # Random swap
    swapped = np.array(initial_h_values)
    indices = np.random.choice(n_points, size=int(0.2*n_points), replace=False)
    np.random.shuffle(indices)
    for i in indices:
        swapped[i] = np.random.rand()
    swapped = swapped * target_sum / np.sum(swapped)
    perturbations.append(swapped)
    
    for p in perturbations:
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, p, method='SLSQP', bounds=[(0, 1)]*n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          jac=compute_gradient, tol=1e-6, options={'maxiter': 200})
        if result.success and result.fun < best_c5:
            best_h = result.x.copy()
            best_c5 = result.fun
        result_l_bfgs = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                                 constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                                 jac=compute_gradient, tol=1e-6, options={'maxiter': 100})
        if result_l_bfgs.success and result_l_bfgs.fun < best_c5:
            best_h = result_l_bfgs.x.copy()
            best_c5 = result_l_bfgs.fun
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
