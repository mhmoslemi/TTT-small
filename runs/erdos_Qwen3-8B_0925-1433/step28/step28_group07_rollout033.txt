import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    population_size = 100
    inertia_weight = 0.8
    cognitive = 2.0
    social = 2.0
    max_iter = 300
    local_refine_pop = 20
    local_refine_iter = 100

    # Initialize swarm
    swarm = np.random.rand(population_size, n_points) * 2 - 1
    swarm = np.clip(swarm, 0, 1)
    swarm = np.array([s * target_sum / np.sum(s) for s in swarm], dtype=float)
    velocities = np.random.rand(population_size, n_points) * 0.1 - 0.05
    personal_best = swarm.copy()
    personal_best_values = np.array([objective(s) for s in swarm])
    global_best = swarm[np.argmin(personal_best_values)]
    global_best_value = np.min(personal_best_values)

    start_time = time.time()
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Update velocities and positions
        r1, r2 = np.random.rand(2, population_size, n_points)
        velocities = (inertia_weight * velocities +
                      cognitive * r1 * (personal_best - swarm) +
                      social * r2 * (global_best[None, :] - swarm))
        swarm += velocities
        swarm = np.array([clip_normalize(s) for s in swarm])
        # Evaluate and update personal best
        for i in range(population_size):
            curr_val = objective(swarm[i])
            if curr_val < personal_best_values[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_values[i] = curr_val
        # Update global best
        current_global = np.min(personal_best_values)
        if current_global < global_best_value:
            global_best_value = current_global
            global_best = personal_best[np.argmin(personal_best_values)]
        # Local refinement on top particles
        top_indices = np.argsort(personal_best_values)[:local_refine_pop]
        for idx in top_indices:
            h_refine = personal_best[idx].copy()
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            res = minimize(objective, h_refine, method='SLSQP', bounds=[(0, 1)]*n_points,
                           constraints=cons, tol=1e-8, options={'maxiter': local_refine_iter})
            if res.success and res.fun < global_best_value:
                global_best_value = res.fun
                global_best = res.x.copy()
        # Track best
        if global_best_value < best_c5:
            best_c5 = global_best_value
            best_h = global_best.copy()

    best_h = clip_normalize(best_h)
    return (best_h, best_c5, n_points)
