import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()
    
    def objective(x):
        return np.max(np.correlate(x, 1 - x, mode='full')) * dx
    
    # Generate diverse initial candidates
    variations = []
    # Symmetric pattern
    symmetric = np.concatenate([h[::-1], h[1:]])
    variations.append(symmetric * target_sum / np.sum(symmetric))
    # Sinusoidal modulation
    x = np.linspace(0, 1, n_points)
    sin_perturb = h * (1 + 0.15 * np.sin(2 * np.pi * x))
    sin_perturb = np.clip(sin_perturb, 0, 1) * target_sum / np.sum(sin_perturb)
    variations.append(sin_perturb)
    # Evenly spaced 1s
    spaced = np.zeros(n_points)
    spaced[::n_points//3] = 1
    spaced = spaced * target_sum / np.sum(spaced)
    variations.append(spaced)
    # Permuted version
    perm_perturb = np.random.permutation(h).copy()
    perm_perturb = np.clip(perm_perturb, 0, 1) * target_sum / np.sum(perm_perturb)
    variations.append(perm_perturb)
    
    for var in variations:
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, var, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          tol=1e-6, options={'maxiter': 100})
        if result.success and result.fun < best_c5:
            best_h = result.x.copy()
            best_c5 = result.fun
    
    # Random perturbations
    for _ in range(3):

            break
        noise = np.random.normal(0, 0.1, n_points)
        perturb = h + noise
        perturb = np.clip(perturb, 0, 1) * target_sum / np.sum(perturb)
        result = minimize(objective, perturb, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
