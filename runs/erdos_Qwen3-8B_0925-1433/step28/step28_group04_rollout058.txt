from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        h = x.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    def generate_initial_guesses(base):

        yield base[np.random.permutation(n_points)] * (target_sum / np.sum(base[np.random.permutation(n_points)]))
        yield base.copy() * (target_sum / np.sum(base)) * np.random.rand(n_points)

    initial_guesses = []

    start_time = time.time()

        result_de = differential_evolution(objective, bounds=[(0, 1)]*n_points, strategy='best1bin', maxiter=150, popsize=30, tol=1e-6, mutation=(0.4, 1.1), recombination=0.8, disp=False)
        h_de = result_de.x.copy()

        if res_slsqp.success and res_slsqp.fun < best_c5:
