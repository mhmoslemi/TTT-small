            if m + k < n_points:
                term2 = h[m + k]
            else:
                term2 = 0.0
            grad[m] = term1 - term2

        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        if np.random.rand() < perturb_prob:
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)

        learning_rate *= decay_rate

    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    final_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    return (h, final_c5, n_points)
