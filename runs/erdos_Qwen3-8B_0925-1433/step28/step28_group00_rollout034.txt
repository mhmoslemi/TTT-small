import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()

    def compute_gradient(x, k, n_points):
        grad = np.zeros_like(x)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] = 1.0 - x[j]
        return grad

    max_iterations = 200
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k_index = np.argmax(corr)
        k = k_index - (n_points - 1)
        current_c5 = corr[k_index] * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        def objective(x):
            corr_opt = np.correlate(x, 1 - x, mode='full')
            return corr_opt[k_index] * dx
        def grad_func(x):
            return compute_gradient(x, k, n_points)
        result = minimize(
            fun=objective,
            x0=h,
            method='L-BFGS-B',
            jac=grad_func,
            bounds=[(0.0, 1.0)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            tol=1e-6,
            options={'maxiter': 100}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()
        h = np.clip(result.x, 0, 1)
        h = h * target_sum / np.sum(h)
    return (np.clip(best_h, 0, 1) * target_sum / np.sum(best_h), best_c5, n_points)
