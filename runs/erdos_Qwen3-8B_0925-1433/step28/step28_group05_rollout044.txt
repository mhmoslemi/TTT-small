        new_h = np.clip(new_h, 0, 1)
        new_h_sum = np.sum(new_h)
        if new_h_sum > 0:
            new_h = new_h * target_sum / new_h_sum
        else:
            new_h = np.zeros_like(new_h) + target_sum / n_points
        new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
        if np.random.rand() < 0.1:
            scale = 1.0 + 0.2 * np.random.normal(0, 0.5)
            perturbed_h = best_h * scale
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h_sum = np.sum(perturbed_h)
            if perturbed_h_sum > 0:
                perturbed_h = perturbed_h * target_sum / perturbed_h_sum
            else:
                perturbed_h = np.zeros_like(perturbed_h) + target_sum / n_points
            new_c5_perturbed = np.max(np.correlate(perturbed_h, 1 - perturbed_h, mode='full')) * dx
            if new_c5_perturbed < best_c5:
                best_h = perturbed_h.copy()
                best_c5 = new_c5_perturbed
    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
