from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)  # Conservative fallback

    def compute_c5(h_array):
        corr = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    def objective(x):
        return compute_c5(x)

    def constraint(x):
        return np.sum(x) - target_sum

    start_time = time.time()
    iterations = 0
    while time.time() - start_time < budget_s:
        iterations += 1
        if iterations > 1000:
            break

            h,

            constraints=[{'type': 'eq', 'fun': constraint}],

        if not result.success:
