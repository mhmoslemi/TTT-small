from scipy.optimize import minimize

    def compute_c5(h, n_points):
        corr = np.fft.ifft(np.fft.fft(h, 2*n_points) * np.conj(np.fft.fft(1-h, 2*n_points)))[n_points-1:2*n_points-1].real
        return np.max(corr) * (2.0 / n_points)
    
    def objective(x):
        return compute_c5(x, n_points)
    
    def compute_gradient(x):
        h = x

        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = k_max - (n_points - 1) + i
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

        pops = []
