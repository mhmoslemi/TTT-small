import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    start_time = time.time()
    
    start_points = [h.copy()]
    # Mirrored version
    mirrored = h[::-1].copy()
    start_points.append(mirrored)
    # Sinusoidal modulated
    x = np.linspace(0, 2, n_points)
    sin_perturb = h * (1 + 0.2 * np.sin(2 * np.pi * x / n_points))
    sin_perturb = np.clip(sin_perturb, 0, 1) * target_sum / np.sum(sin_perturb)
    start_points.append(sin_perturb)
    # Permutation
    perm_perturb = np.random.permutation(h).copy()
    perm_perturb = np.clip(perm_perturb, 0, 1) * target_sum / np.sum(perm_perturb)
    start_points.append(perm_perturb)
    # Evenly spaced 1s and 0s
    even = np.zeros(n_points)
    step = n_points // 3
    for i in range(n_points):
        if i % step == 0:
            even[i] = 1.0
    even = np.clip(even, 0, 1) * target_sum / np.sum(even)
    start_points.append(even)
    
    for sp in start_points:
        if time.time() - start_time > budget_s:
            break
        sp = sp.copy()
        sp = sp * target_sum / np.sum(sp)
        result = minimize(objective, sp, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          options={'maxiter': 200, 'ftol': 1e-8})
        if result.success and result.fun < best_c5:
            best_h = result.x.copy()
            best_c5 = result.fun
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
