from scipy.optimize import minimize
import time

        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    population_size = 100
    inertia_start = 0.8
    inertia_end = 0.4
    cognitive = 2.0
    social = 2.0
    max_iter = 300
    local_refine_pop = 10

    # Initialize swarm
    swarm = np.random.rand(population_size, n_points) * 2 - 1
    swarm = np.clip(swarm, 0, 1)

    personal_best = swarm.copy()

    global_best_value = np.min(personal_best_values)

    start_time = time.time()
