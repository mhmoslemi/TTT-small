def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = np.min([best_c5, 0.381])

    def objective(x):
        h = x.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        lag = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        if lag >= 0:
            grad[:n_points - lag] = 1 - h[lag:]
        else:
            grad[-lag:] = 1 - h[:n_points + lag]
        return grad

    def generate_initial_guesses(base):

        yield np.random.permutation(base)

        yield base[np.random.permutation(n_points)] * target_sum / np.sum(base[np.random.permutation(n_points)])

    initial_guesses = []

        initial_guesses.extend(generate_initial_guesses(base))

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(

            guess,
