from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = np.array(initial_h_values, dtype=float)

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        # Symmetric version
        sym_h = np.concatenate([initial_h_values[::-1], initial_h_values[1:-1]])
        sym_h = np.clip(sym_h, 0, 1)
        sym_h = sym_h * target_sum / np.sum(sym_h)

        # Sinusoidal
        x = np.linspace(0, 1, n_points)

        sin_h = np.clip(sin_h, 0, 1)
        sin_h = sin_h * target_sum / np.sum(sin_h)
        candidates.append(sin_h)
        # Evenly spaced

        spacing = n_points // 3

        perm_h = np.random.permutation(initial_h_values)
        perm_h = np.clip(perm_h, 0, 1)
        perm_h = perm_h * target_sum / np.sum(perm_h)
        candidates.append(perm_h)
        return candidates
    
    population = generate_candidates()
    for _ in range(150):
        if time.time() - start_time > budget_s:

        population = [np.clip(h, 0, 1) * target_sum / np.sum(np.clip(h, 0, 1)) for h in population]
        objectives = [evaluate(h) for h in population]
        best_idx = np.argmin(objectives)
        if objectives[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = objectives[best_idx]
        new_population = []
