import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)  # Initial upper bound

    learning_rate = 0.1
    max_iterations = 500
    start_time = time.time()
    perturb_count = 0

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Compute cross-correlation to find max lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)

        # Compute gradient for the maximum lag
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                gradient[i] += (1 - h[j]) * dx
                gradient[j] -= h[i] * dx

        # Update h
        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        # Evaluate new C5
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Decay learning rate
        learning_rate *= 0.9999

        # Add perturbation every 100 iterations
        if iteration % 100 == 0:
            perturb_count += 1
            if perturb_count % 2 == 0:
                noise = np.random.normal(0, 0.05, size=n_points)
                h += noise
                h = np.clip(h, 0, 1)
                h = h * target_sum / np.sum(h)

    return (best_h, best_c5, n_points)
