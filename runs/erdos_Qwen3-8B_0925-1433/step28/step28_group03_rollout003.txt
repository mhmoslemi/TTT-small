from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    h = h * target_sum / np.sum(h)

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    def generate_initial_guesses(base):
        yield base.copy()
        yield base.copy() * (target_sum / np.sum(base))

        yield np.clip(base[np.random.permutation(n_points)], 0, 1) * (target_sum / np.sum(np.clip(base[np.random.permutation(n_points)], 0, 1)))
        yield np.clip(base * 0.8 + 0.1 * np.random.rand(n_points), 0, 1) * (target_sum / np.sum(np.clip(base * 0.8 + 0.1 * np.random.rand(n_points), 0, 1)))

    initial_guesses = []

            base = base * target_sum / np.sum(base)

    start_time = time.time()

        result_de = differential_evolution(objective, bounds=[(0, 1)]*n_points, strategy='best1bin', maxiter=100, popsize=20, tol=1e-6, mutation=(0.5, 1), recombination=0.7, disp=False, polish=True, polish_kwargs={'method': 'L-BFGS-B', 'bounds': [(0, 1)]*n_points})
        h_de = result_de.x.copy()

        if res_slsqp.success and res_slsqp.fun < best_c5:
