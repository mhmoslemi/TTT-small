        return np.sum(x) - target_sum

    start_time = time.time()
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        guess = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
        guess = guess * target_sum / np.sum(guess)
        result = minimize(objective, guess, method='L-BFGS-B', jac=compute_gradient,
                          bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': constraint}],
                          options={'maxiter': 100, 'disp': False})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()
        h = best_h

    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
