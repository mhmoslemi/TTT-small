import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad
    
    start_time = time.time()
    pop_size = 50
    generations = 150
    mutation_std = 0.1
    elite_ratio = 0.2
    
    def generate_initial_population():
        pop = [h.copy()]
        x = np.linspace(0, 1, n_points)
        pop.append(h * (1 + 0.15 * np.sin(2 * np.pi * x)))
        pop.append(np.tile([1.0, 0.0, 0.0], n_points // 3))
        pop.append(np.random.permutation(h))
        while len(pop) < pop_size:
            perm = np.random.permutation(h)
            pop.append(perm)
        for i in range(len(pop)):
            pop[i] = np.clip(pop[i], 0, 1) * target_sum / np.sum(pop[i])
        return pop
    
    pop = generate_initial_population()
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitness = [objective(ind) for ind in pop]
        elite_idx = np.argsort(fitness)[:int(elite_ratio * pop_size)]
        elites = [pop[i] for i in elite_idx]
        new_pop = elites[:]
        while len(new_pop) < pop_size:
            parent1, parent2 = np.random.choice(elite_idx, 2, replace=False)
            child = (pop[parent1] + pop[parent2]) / 2.0

                child += mutation_std * np.random.randn(n_points)
            child = np.clip(child, 0, 1) * target_sum / np.sum(child)
            new_pop.append(child)
        pop = new_pop
    
    candidates = [np.clip(ind, 0, 1) * target_sum / np.sum(ind) for ind in pop]
    for ind in candidates:
        if time.time() - start_time > budget_s:

        result = minimize(objective, ind, method='L-BFGS-B', bounds=[(0, 1)]*n_points,

    return (best_h, best_c5, n_points)
