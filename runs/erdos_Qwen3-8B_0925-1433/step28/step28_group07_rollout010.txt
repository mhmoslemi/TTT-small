import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    population_size = 50
    F = 0.8
    CR = 0.9
    max_iter = 300
    local_refine_pop = 10

    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        perturb = np.random.normal(0, 0.1, n_points)
        tmp = initial_h_values * (1 + perturb)
        tmp = np.clip(tmp, 0, 1)
        tmp = tmp * target_sum / np.sum(tmp)
        population.append(tmp)
    population = np.array(population)

    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        mutant = np.zeros_like(population)
        for i in range(population_size):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant[i] = population[a] + F * (population[b] - population[c])
        trial = np.zeros_like(population)
        for i in range(population_size):
            for j in range(n_points):
                if np.random.rand() < CR or j == np.random.randint(n_points):
                    trial[i][j] = mutant[i][j]
                else:
                    trial[i][j] = population[i][j]
        trial = np.clip(trial, 0, 1)
        trial = trial * target_sum / np.sum(trial, axis=1, keepdims=True)
        obj_trial = np.array([objective(t) for t in trial])
        obj_pop = np.array([objective(p) for p in population])
        mask = obj_trial < obj_pop
        population[mask] = trial[mask]

    top_indices = np.argsort([objective(p) for p in population])[:local_refine_pop]
    for idx in top_indices:
        h_refine = population[idx].copy()
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(objective, h_refine, method='SLSQP', bounds=[(0, 1)]*n_points,
                       constraints=cons, tol=1e-6, options={'maxiter': 50})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x.copy()

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
