import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize with parent
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    def generate_candidates(base):
        yield base.copy()
        yield base.copy() * (target_sum / np.sum(base))
        yield np.clip(base + 0.1 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)), 0, 1) * (target_sum / np.sum(np.clip(base + 0.1 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)), 0, 1)))
        yield np.clip(base[np.random.permutation(n_points)], 0, 1) * (target_sum / np.sum(np.clip(base[np.random.permutation(n_points)], 0, 1)))

    start_time = time.time()
    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        base = np.array(initial_h_values, dtype=float)
        base = base * target_sum / np.sum(base)
        for candidate in generate_candidates(base):
            if time.time() - start_time > budget_s:
                break
            result = minimize(objective, candidate, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], jac=compute_gradient, tol=1e-6, options={'maxiter': 100})
            if result.success and result.fun < best_c5:
                best_h = result.x.copy()
                best_c5 = result.fun

    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
