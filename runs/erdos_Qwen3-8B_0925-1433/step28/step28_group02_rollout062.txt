import numpy as np
from scipy.optimize import minimize
import time
from scipy.stats.qmc import Halton

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 60
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        target_sum = n_points / 2.0
        h = h * target_sum / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        h_clipped = np.clip(x, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    def generate_evenly_spaced(n, phase=0):
        spacing = n // 2
        h = np.zeros(n)
        for i in range(n):
            if (i - phase) % spacing == 0:
                h[i] = 1.0
        return h * target_sum / np.sum(h)

    def generate_sine(n):
        h = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n) / (n - 1)))
        return np.clip(h, 0, 1) * target_sum / np.sum(h)

    def generate_block(n, size=5):
        h = np.zeros(n)
        for i in range(0, n, 2 * size):
            h[i:i+size] = 1.0
        return h * target_sum / np.sum(h)

    def generate_halton(n):
        m = n // 2
        if m == 0:
            return np.zeros(n)
        sampler = Halton(d=1, scramble=False)
        points = sampler.random(m)
        points *= 2.0
        h = np.zeros(n)
        for i, x in enumerate(points):
            idx = np.argmin(np.abs(x - np.linspace(0, 2, n)))
            h[idx] = 1.0
        return h * target_sum / np.sum(h)

    initial_guesses = []
    for phase in [0, 1, 2]:
        initial_guesses.append(generate_evenly_spaced(n_points, phase))
    initial_guesses.append(generate_sine(n_points))
    initial_guesses.append(generate_block(n_points))
    if not 'initial_h_values' in globals():
        initial_guesses.append(generate_halton(n_points))

    population = initial_guesses
    best_h = population[0]
    best_c5 = objective(best_h)
    start_time = time.time()

    for _ in range(30):
        if time.time() - start_time > budget_s:
            break
        fitness = [objective(ind) for ind in population]
        ranked = sorted(zip(fitness, population), key=lambda x: x[0])

            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        population = new_pop

        current_c5 = objective(current_best)
