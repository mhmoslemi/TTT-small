import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.stats.qmc import Halton
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 60
        m = n_points // 2
        halton = Halton(d=1, scramble=False)
        points = halton.random(n=m) * 2.0
        h = np.zeros(n_points)
        dx = 2.0 / n_points
        for x in points:
            idx = np.argmin(np.abs(x - np.linspace(0, 2, n_points)))
            h[idx] = 1.0
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    def objective(x):

        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()
    
    # Differential Evolution
    res_de = differential_evolution(objective, [(0, 1)] * n_points, popsize=50, maxiter=100, tol=1e-5, mutation=(0.5, 1), recombination=0.7)
    h_candidate = res_de.x.copy()
    h_candidate = np.clip(h_candidate, 0, 1)
    h_candidate = h_candidate * target_sum / np.sum(h_candidate)
    current_c5 = objective(h_candidate)
    if current_c5 < best_c5:
        best_h = h_candidate.copy()
        best_c5 = current_c5
    
    # SLSQP Refinement
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res_slsqp = minimize(objective, best_h, method='SLSQP',
                            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                            bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 100})
        if res_slsqp.success:
            h_candidate = res_slsqp.x.copy()
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5
    
    # Critical Lag Adjustment
    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
        k_worst = np.argmax(corr)
        if k_worst < n_points:
            h = best_h.copy()
            h[k_worst] = np.clip(h[k_worst] + 0.01, 0, 1)
            h = h * target_sum / np.sum(h)
            best_h = h
        if k_worst >= n_points:
            h = best_h.copy()
            h[k_worst - n_points] = np.clip(h[k_worst - n_points] + 0.01, 0, 1)
            h = h * target_sum / np.sum(h)
            best_h = h
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
