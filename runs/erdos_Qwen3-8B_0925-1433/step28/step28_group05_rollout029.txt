    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_c5 = min(best_c5, 0.381)

    initial_guesses = []
    initial_guesses.append(h.copy())
    reg_spacing = np.zeros(n_points)
    reg_spacing[::2] = 1.0
    reg_spacing = reg_spacing * target_sum / np.sum(reg_spacing)
    initial_guesses.append(reg_spacing)
    sine_pattern = 0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 1, n_points))
    sine_pattern = sine_pattern * target_sum / np.sum(sine_pattern)
    initial_guesses.append(sine_pattern)

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        current_h = np.clip(guess, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        res = minimize(
            objective,
            current_h,
            method='L-BFGS-B',
            bounds=[(0, 1)]*n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            jac=compute_gradient,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if res.success:
            candidate_h = res.x
            candidate_c5 = compute_c5(candidate_h)
            if candidate_c5 < best_c5:
                best_h = candidate_h.copy()
                best_c5 = candidate_c5

    perturbed = np.random.permutation(best_h)
    perturbed = perturbed * target_sum / np.sum(perturbed)
    res = minimize(
        objective,
        perturbed,
        method='L-BFGS-B',
        bounds=[(0, 1)]*n_points,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
        jac=compute_gradient,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    if res.success:
        candidate_h = res.x
        candidate_c5 = compute_c5(candidate_h)
        if candidate_c5 < best_c5:
            best_h = candidate_h.copy()
            best_c5 = candidate_c5

    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
