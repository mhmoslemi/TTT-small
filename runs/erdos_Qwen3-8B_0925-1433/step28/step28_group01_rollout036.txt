from scipy.optimize import minimize, Bounds
from scipy.stats.qmc import Halton

        return np.max(corr) * dx
    
    best_h = h.copy()

                self.max_iter = max_iter
                self.w = w
                self.c1 = c1
                self.c2 = c2
                self.best_pos = None
                self.best_cost = float('inf')
                self.history = []
            
            def optimize(self, bounds, objective):
                particles = np.random.rand(self.n_particles, len(bounds)) * (bounds[:,1]-bounds[:,0]) + bounds[:,0]
