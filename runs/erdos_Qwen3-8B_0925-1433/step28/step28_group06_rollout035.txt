import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    best_h = h_initial
    best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx

    def rescale_h(h):
        s = np.sum(h)
        if s == 0: return np.zeros_like(h) + 1
        scale = target_sum / s
        return np.clip(h * scale, 0, 1)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def eval_h(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    # Generate initial variations
    candidates = [h_initial]
    x = np.linspace(0, 1, n_points)
    candidates.append(rescale_h(h_initial * (1 + 0.15 * np.sin(2 * np.pi * x))))
    candidates.append(rescale_h(np.random.permutation(h_initial)))
    for i in range(5):
        h = np.zeros(n_points)
        h[::n_points//3] = 1
        candidates.append(rescale_h(h))

    start = time.time()
    while time.time() - start < budget_s:
        best = min(candidates, key=eval_h)
        best_c5 = eval_h(best)
        if best_c5 < best_c5: best_h = best; best_c5 = best_c5

        # Local refinement
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(objective, best, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                       constraints=cons, tol=1e-6, options={'maxiter': 100})
        if res.success:
            best_h = res.x
            best_c5 = res.fun

        # Generate new candidates
        new_cands = []
        for h in candidates:
            noise = np.random.normal(0, 0.1, n_points)
            mut = rescale_h(np.clip(h + noise, 0, 1))
            new_cands.append(mut)
            new_cands.append(rescale_h(np.roll(h, np.random.randint(n_points))))
        candidates += new_cands
        if len(candidates) > 50:
            candidates = candidates[:50]

    best_h = rescale_h(np.clip(best_h, 0, 1))
    return (best_h, best_c5, n_points)
