from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def objective(x):
        h = x.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(x):
        h = x.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad
    
    start_time = time.time()
    perturbations = []
    
    # Symmetric mirroring
    mirrored = np.concatenate([h[:n_points//2], h[::-1][:n_points//2]])
    mirrored = mirrored * target_sum / np.sum(mirrored)
    perturbations.append(mirrored)
    
    # Periodic extension
    periodic = np.tile(h[:n_points//2], 2)
    periodic = periodic[:n_points]
    periodic = periodic * target_sum / np.sum(periodic)
    perturbations.append(periodic)
    
    # Random swap
    swapped = h.copy()
    idx = np.random.choice(n_points, size=n_points//5, replace=False)

    perturbations.append(swapped)
    
    # Sinusoidal modulation
    for freq in [1, 2, 3, 4]:
