            grad[i] = (value_plus - value_minus) / (2 * epsilon)
        
        h_candidate = h - learning_rate * grad
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        
        corr_candidate = compute_corr(h_candidate)
        current_c5_candidate = np.max(corr_candidate)
        
        if current_c5_candidate < current_c5:
            h = h_candidate
            best_c5 = current_c5_candidate
        else:
            learning_rate *= 0.9
    
    best_h = np.clip(h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
