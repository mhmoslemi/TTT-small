import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        corr = np.correlate(x, 1 - x, mode='full')
        k = np.argmax(corr) - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - x[j]
        return grad

    initial_guesses = []
    for _ in range(2):
        perm = np.random.permutation(n_points)
        guess = initial_h_values[perm] * target_sum / np.sum(initial_h_values[perm])
        initial_guesses.append(guess)
        shift = np.roll(initial_h_values, 1)
        guess = shift * target_sum / np.sum(shift)
        initial_guesses.append(guess)
        mod = initial_h_values * (1 + 0.1 * np.sin(2 * np.pi * np.arange(n_points) / n_points))
        mod = np.clip(mod, 0, 1) * target_sum / np.sum(mod)
        initial_guesses.append(mod)

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = differential_evolution(objective, bounds=[(0, 1)]*n_points, strategy='best1bin', maxiter=50, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.7, disp=False)
        h_de = np.clip(result.x, 0, 1) * target_sum / np.sum(np.clip(result.x, 0, 1))
        res = minimize(objective, h_de, method='SLSQP', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], jac=compute_gradient, tol=1e-6, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x.copy(), res.fun

    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(np.clip(best_h, 0, 1))
    return (best_h, best_c5, n_points)
