        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_c5 = np.max(corr)
        k = np.argmax(corr)
        l_correct = k - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_correct) % n_points
            grad[i] = 1.0 - h[j]
