import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    start_time = time.time()
    perturbations = []
    x = np.linspace(0, 1, n_points)
    perturbations.append(h * (1 + 0.3 * np.sin(2 * np.pi * x)))
    perturbations.append(np.random.permutation(h))
    perturbations.append(h + np.random.normal(0, 0.2, n_points))
    perturbations.append(h * (1 + 0.2 * np.sin(4 * np.pi * x)))
    perturbations.append(h * (1 + 0.25 * np.sin(6 * np.pi * x)))

    for p in perturbations:
        if time.time() - start_time > budget_s:
            break
        p = np.clip(p, 0, 1) * target_sum / np.sum(p)
        result = minimize(objective, p, method='L-BFGS-B', bounds=[(0, 1)]*n_points, 
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], 
                          jac=compute_gradient, tol=1e-6, options={'maxiter': 100})
        if result.success and result.fun < best_c5:
            best_h = result.x.copy()
            best_c5 = result.fun

            pert = np.random.normal(0, 0.1, n_points)
