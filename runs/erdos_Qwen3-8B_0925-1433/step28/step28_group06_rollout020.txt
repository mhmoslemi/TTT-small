def run(seed=42, budget_s=600):
    np.random.seed(seed)

    target_sum = n_points / 2.0
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)

    def compute_corr_fft(x):

        fft_1mh = np.fft.fft(1 - x, fft_size)
        corr = np.fft.ifft(fft_h * np.conj(fft_1mh)).real
        return corr[n-1:2*n-1]
    
    def normalize_h(x, target):
        s = np.sum(x)
        if s == 0:
            return np.zeros_like(x) + target / len(x)
        scaling = target / s
        x = x * scaling
        return np.clip(x, 0, 1)
    
    def fitness_func(x):

        return grad
    
    def generate_initial_structures():
        structures = []

        structures.append(initial_h_values * (1 + 0.2 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))))

            rand = np.random.rand(n_points)
            rand = normalize_h(rand, target_sum)
            structures.append(rand)
        return structures
    
    initial_population = generate_initial_structures()
