    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    learning_rate = 0.1
    learning_rate_decay = 0.001
    max_iter = 500
    perturb_prob = 0.1
    perturb_scale = 0.05
    early_stop = 50
    start_time = time.time()
    improvement_iter = 0

    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            improvement_iter = 0
        else:
            improvement_iter += 1
            if improvement_iter > early_stop:
                break

        lag_idx = np.argmax(corr)
        L = lag_idx - (n_points - 1)
        gradient = np.zeros(n_points)
        for j in range(n_points):
            if 0 <= j + L < n_points:
                gradient[j] += (1 - h[j + L])
            if 0 <= j - L < n_points:
                gradient[j] -= h[j - L]
        h = h - learning_rate * gradient
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        if np.random.rand() < perturb_prob:
            noise = np.random.normal(0, perturb_scale, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)

        learning_rate = max(learning_rate - learning_rate_decay, 0.0001)

    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    final_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    return (best_h, best_c5, n_points)
