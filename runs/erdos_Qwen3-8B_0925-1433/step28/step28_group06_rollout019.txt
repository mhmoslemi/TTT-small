import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def compute_c5(x):
        return np.max(np.correlate(x, 1 - x, mode='full')) * dx

    def clip_scale(arr):
        arr = np.clip(arr, 0, 1)
        return arr * target_sum / np.sum(arr)

    pop_size = 50
    generations = 150
    mutation_rate = 0.1
    population = [h_initial]
    x = np.linspace(0, 1, n_points)
    population.append(h_initial * (1 + 0.2 * np.sin(2 * np.pi * x)))
    population.append(np.zeros(n_points))
    population.append(np.random.permutation(h_initial))
    for i in range(pop_size - 3):
        population.append(np.random.rand(n_points))

    start_time = time.time()
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
        pop_fitness = [compute_c5(p) for p in population]
        parents = [p for _, p in sorted(zip(pop_fitness, population))]
        new_pop = []
        for _ in range(pop_size):
            parent1, parent2 = np.random.choice(parents, 2, replace=False)
            child = (parent1 + parent2) / 2
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child += noise
            new_pop.append(clip_scale(child))
        population = new_pop
        for p in population:
            if compute_c5(p) < best_c5:
                best_h = p.copy()
                best_c5 = compute_c5(p)

    top_candidates = [p for p in population if compute_c5(p) < best_c5]
    top_candidates.sort(key=compute_c5)
    top_candidates = top_candidates[:5]

    for p in top_candidates:
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        result = minimize(compute_c5, p, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success and result.fun < best_c5:
            best_h = result.x.copy()
            best_c5 = result.fun

    best_h = clip_scale(best_h)
    return (best_h, best_c5, n_points)
