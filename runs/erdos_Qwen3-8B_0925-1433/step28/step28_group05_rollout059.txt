    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    initial_guesses = []
    initial_guesses.append(h[np.random.permutation(n_points)])
    for f in [2, 3, 4, 5]:
        t = np.linspace(0, 1, n_points)
        mod = 1 + 0.2 * np.sin(2 * np.pi * f * t)
        initial_guesses.append(h * mod * target_sum / np.sum(h * mod))
    initial_guesses.append(np.clip(h + np.random.normal(0, 0.15, n_points), 0, 1) * target_sum / np.sum(np.clip(h + np.random.normal(0, 0.15, n_points), 0, 1)))

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        def objective(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx

        def grad(x):
            corr = np.correlate(x, 1 - x, mode='full')
            k_max_index = np.argmax(corr)
            k = k_max_index - (n_points - 1)
            j = np.arange(n_points) + k
            valid = (j >= 0) & (j < n_points)
            grad = np.zeros_like(x)
            grad[valid] = 1 - x[j[valid]]
            return grad

        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        result = minimize(objective, guess, method='SLSQP', jac=grad, bounds=bounds, constraints=constraints, tol=1e-5, options={'maxiter': 200})
        if result.success and result.fun < best_c5:
            best_h = result.x.copy()
            best_c5 = result.fun

    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
