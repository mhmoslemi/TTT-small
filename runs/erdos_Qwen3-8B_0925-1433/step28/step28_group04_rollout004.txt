import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale(h):
        return np.clip(h, 0, 1) * target_sum / np.sum(np.clip(h, 0, 1))

    h = np.array(initial_h_values, dtype=float)
    h = rescale(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    initial_guesses = []
    for _ in range(15):
        h_pert = h.copy()
        if np.random.rand() < 0.5:
            h_pert += 0.2 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))
            h_pert += 0.15 * np.cos(2 * np.pi * np.linspace(0, 1, n_points))
        elif np.random.rand() < 0.3:
            h_pert = np.random.permutation(h_pert)
        else:
            h_pert += 0.15 * np.random.randn(n_points)
        initial_guesses.append(rescale(h_pert))

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s - 60:
            break
        result = differential_evolution(
            objective, bounds=[(0, 1)]*n_points,
            strategy='best1bin', maxiter=150, popsize=30,
            tol=1e-6, mutation=(0.3, 0.9), recombination=0.8, disp=False
        )
        h_de = rescale(result.x)
        result_slsqp = minimize(
            objective, h_de, method='SLSQP',
            bounds=[(0, 1)]*n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            jac=compute_gradient, tol=1e-6, options={'maxiter': 250}
        )
        if result_slsqp.success and result_slsqp.fun < best_c5:
            best_h = result_slsqp.x.copy()
            best_c5 = result_slsqp.fun

    best_h = rescale(best_h)
    return (best_h, best_c5, n_points)
