import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale(h):
        h = np.clip(h, 0, 1)
        return h * target_sum / np.sum(h)

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] = 1.0 - h[j]
        return grad

    initial_guesses = []
    base = np.array(initial_h_values, dtype=float)
    base = rescale(base)
    initial_guesses.append(base)
    initial_guesses.append(base * (1 + 0.1 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))))
    initial_guesses.append(base[np.random.permutation(n_points)])
    initial_guesses.append(0.5 + 0.2 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)))

    best_h = base.copy()
    best_c5 = objective(base)
    start_time = time.time()

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result_de = differential_evolution(
            objective, bounds=[(0, 1)]*n_points, strategy='best1bin',
            maxiter=50, popsize=20, tol=1e-6, mutation=(0.5, 1),
            recombination=0.7, disp=False, seed=seed
        )
        h_de = rescale(result_de.x)
        res_slsqp = minimize(
            objective, h_de, method='SLSQP', bounds=[(0, 1)]*n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            jac=compute_gradient, tol=1e-6, options={'maxiter': 100}
        )
        if res_slsqp.success and res_slsqp.fun < best_c5:
            best_h = res_slsqp.x.copy()
            best_c5 = res_slsqp.fun
        res_l_bfgs = minimize(
            objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            jac=compute_gradient, tol=1e-6, options={'maxiter': 50}
        )
        if res_l_bfgs.success and res_l_bfgs.fun < best_c5:
            best_h = res_l_bfgs.x.copy()
            best_c5 = res_l_bfgs.fun

    best_h = rescale(best_h)
    return (best_h, best_c5, n_points)
