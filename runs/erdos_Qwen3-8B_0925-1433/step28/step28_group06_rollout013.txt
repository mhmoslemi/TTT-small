import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def evaluate_candidate(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return np.inf

        corr = np.fft.ifft(h_fft * np.conj(one_minus_h_fft)).real
        max_corr = np.max(corr[n_points-1:2*n_points-1])
        return max_corr * dx
    
    bounds = [(0, 1)] * n_points
    
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = evaluate_candidate(best_h)

    candidates = [np.array(initial_h_values, dtype=float)]

    candidates.append(h_sin)

    res = differential_evolution(
