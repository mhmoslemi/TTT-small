import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    initial_guesses = []
    base = np.array(initial_h_values, dtype=float)
    base = base * target_sum / np.sum(base)
    for _ in range(3):

        initial_guesses.append(np.clip(base + np.random.normal(0, 0.1, n_points), 0, 1) * target_sum / np.sum(np.clip(base + np.random.normal(0, 0.1, n_points), 0, 1)))
        initial_guesses.append(base.copy() * (target_sum / np.sum(base)))

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result_de = differential_evolution(
            objective,

            maxiter=100,
            popsize=20,
            tol=1e-6,
            mutation=(0.5, 0.9),
            recombination=0.8,
            disp=False
        )
        h_de = result_de.x.copy()
        h_de = np.clip(h_de, 0, 1) * target_sum / np.sum(h_de)
        res_slsqp = minimize(
            objective,
            h_de,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            jac=compute_gradient,

        if res_slsqp.success and res_slsqp.fun < best_c5:
