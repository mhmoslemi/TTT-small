    half = n_points // 2
    symmetric_h[:half] = h_initial[:half]
    symmetric_h[half:] = h_initial[:half][::-1]
    symmetric_h = symmetric_h * (target_sum / np.sum(symmetric_h))
    perturbations.append(symmetric_h)
    
    # Periodic extension
    periodic_h = np.tile(h_initial[:n_points//2], 2)
    periodic_h = periodic_h[:n_points]
    periodic_h = periodic_h * (target_sum / np.sum(periodic_h))
    perturbations.append(periodic_h)
    
    # Random swap
    random_swap = np.random.permutation(h_initial)
    random_swap = random_swap * (target_sum / np.sum(random_swap))
    perturbations.append(random_swap)
    
    # Original initial guess
    perturbations.append(h_initial)
    
    for p in perturbations:
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, p, method='SLSQP', 
                          bounds=[(0, 1)] * n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          jac=compute_gradient, tol=1e-6, options={'maxiter': 200})
        if result.success and result.fun < best_c5:
            best_h = result.x.copy()
            best_c5 = result.fun
        result_l_bfgs = minimize(objective, best_h, method='L-BFGS-B',
                                 bounds=[(0, 1)] * n_points,
                                 constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], 
                                 jac=compute_gradient, tol=1e-6, options={'maxiter': 100})
        if result_l_bfgs.success and result_l_bfgs.fun < best_c5:
            best_h = result_l_bfgs.x.copy()
            best_c5 = result_l_bfgs.fun
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
