import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    pop_size = 50
    F = 0.8
    CR = 0.9
    max_iter = 300
    local_pop = 10
    local_iter = 50
    t0 = time.time()

    def perturb(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.random.rand(n_points) * target_sum / n_points
        return h * target_sum / s

    pop = [perturb(h_initial + 0.1 * np.random.randn(n_points)) for _ in range(pop_size)]
    best = h_initial.copy()
    best_obj = obj(best)

    for _ in range(max_iter):
        if time.time() - t0 > budget_s - 30:
            break
        a, b, c = np.random.choice(pop_size, 3, replace=False)
        mutant = pop[a] + F * (pop[b] - pop[c])
        mutant = perturb(mutant)
        trial = np.copy(pop[np.random.randint(pop_size)])
        for i in range(n_points):
            if np.random.rand() < CR or trial[i] < 1e-6:
                trial[i] = mutant[i]
        trial = perturb(trial)
        if obj(trial) < obj(pop[np.random.randint(pop_size)]):
            pop[np.random.randint(pop_size)] = trial
        curr_best = min(pop, key=obj)
        if obj(curr_best) < best_obj:
            best, best_obj = curr_best, obj(curr_best)

    top = sorted(pop, key=obj)[:local_pop]
    for h in top:
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-6, options={'maxiter': local_iter})

            best, best_obj = res.x, res.fun
