import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        corr = np.correlate(x, 1 - x, mode='full')
        k_max_idx = np.argmax(corr)
        k = k_max_idx - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - x[j]
        return grad

    start_time = time.time()
    # Generate transformed initial guesses
    guesses = []
    # 1. Symmetric mirroring
    h_mirrored = np.concatenate([h[::-1], h[1:]])
    h_mirrored = np.clip(h_mirrored, 0, 1) * target_sum / np.sum(h_mirrored)
    guesses.append(h_mirrored)
    # 2. Periodic extension
    period = n_points // 2
    h_periodic = np.tile(h[:period], 2)
    h_periodic = np.clip(h_periodic, 0, 1) * target_sum / np.sum(h_periodic)
    guesses.append(h_periodic)
    # 3. Random swaps
    h_swapped = h.copy()
    idxs = np.random.choice(n_points, size=n_points//5, replace=False)
    for i in idxs:
        h_swapped[i] = h_swapped[np.random.choice(n_points, size=1, replace=False)]
    h_swapped = np.clip(h_swapped, 0, 1) * target_sum / np.sum(h_swapped)
    guesses.append(h_swapped)

    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, guess, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          jac=compute_gradient, tol=1e-6, options={'maxiter': 100})
        if result.success and result.fun < best_c5:
            best_h = result.x.copy()
            best_c5 = result.fun
        # Explore perturbations
        perturb = np.random.normal(0, 0.1, n_points)
        perturb = np.clip(perturb, -0.2, 0.2)
        new_guess = best_h + perturb
        new_guess = np.clip(new_guess, 0, 1) * target_sum / np.sum(new_guess)
        result_perturb = minimize(objective, new_guess, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                                 constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                                 jac=compute_gradient, tol=1e-6, options={'maxiter': 100})
        if result_perturb.success and result_perturb.fun < best_c5:
            best_h = result_perturb.x.copy()
            best_c5 = result_perturb.fun

    # Final refinement
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
