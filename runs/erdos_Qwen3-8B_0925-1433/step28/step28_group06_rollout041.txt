import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate initial population with transformations
    def generate_initial_pop():
        h = np.array(initial_h_values, dtype=float) * (target_sum / np.sum(initial_h_values))
        mirrored = np.array(initial_h_values[::-1], dtype=float) * (target_sum / np.sum(initial_h_values[::-1]))
        permuted = np.random.permutation(h) * (target_sum / np.sum(np.random.permutation(h)))
        return [h, mirrored, permuted]

    population = generate_initial_pop()
    best_h = population[0]
    best_c5 = compute_c5(best_h)

    # Genetic algorithm
    start_time = time.time()
    pop_size = 50
    generations = 300
    for gen in range(generations):
        if time.time() - start_time > budget_s - 150:
            break
        # Evaluate fitness
        fitness = [compute_c5(h) for h in population]
        # Select top 50
        top_idx = np.argsort(fitness)[:pop_size]
        top_pop = [population[i] for i in top_idx]
        # Generate new population
        new_pop = []
        for _ in range(pop_size):
            if np.random.rand() < 0.8:
                # Crossover
                p1, p2 = np.random.choice(top_pop, size=2, replace=False)
                child = np.zeros_like(p1)
                for i in range(n_points):
                    if np.random.rand() < 0.5:
                        child[i] = p1[i]
                    else:
                        child[i] = p2[i]
            else:
                # Mutation
                child = np.random.rand(n_points)
                child = np.clip(child, 0, 1) * (target_sum / np.sum(child))
            # Clip and normalize
            child = np.clip(child, 0, 1)
            child = child * (target_sum / np.sum(child))
            new_pop.append(child)
        population = new_pop
        # Update best
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Local optimization
    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_idx = np.argmax(corr)
        k = k_max_idx - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    # Final refine
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h))
    result = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                      jac=gradient, tol=1e-6, options={'maxiter': 100})
    if result.success:
        best_h = result.x
        best_c5 = result.fun

    return (best_h, best_c5, n_points)
