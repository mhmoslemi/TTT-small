        grad = np.zeros_like(h)
        for idx in top_lags:
            k = idx - (n_points - 1)
            corr_val = corr[idx]
            for i in range(n_points):
                j = (i + k) % n_points
                grad[i] += corr_val * (1 - h[j])
        return grad
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        top_lags = np.argsort(corr)[-5:]
        grad = compute_gradient(h, top_lags, corr, n_points)
        res = minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1)] * n_points, 
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], 
                       jac=grad, tol=1e-6)
        if res.success:
            h = res.x
            new_c5 = res.fun
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        if np.random.rand() < 0.1:
            perturb = 0.1 * np.random.randn(n_points)
            h = np.clip(h + perturb, 0, 1)
            h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    
    return (best_h, best_c5, n_points)
