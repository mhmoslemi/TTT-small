import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()
    
    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(h, top_lags, n_points):
        grad = np.zeros_like(h)
        for k in top_lags:
            for i in range(n_points):
                j_plus = (i + k) % n_points
                j_minus = (i - k) % n_points
                grad[i] += (1 - h[j_plus]) - h[j_minus]
        return grad
    
    def constraint(x):
        return np.sum(x) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    iteration = 0
    while time.time() - start_time < budget_s:

        top_lags_indices = np.argpartition(corr, -5)[-5:]
        top_lags = []
        for idx in top_lags_indices:
            k = idx - (n_points - 1)
            top_lags.append(k)
        
        grad = compute_gradient(h, top_lags, n_points)
        grad /= len(top_lags)
        
        cons = {'type': 'eq', 'fun': constraint}
        res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, jac=grad, tol=1e-6, options={'maxiter': 100})
