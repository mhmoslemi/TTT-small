import numpy as np
from scipy.optimize import minimize
import time
import matplotlib.pyplot as plt

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def objective(x):
        h = x
        n_fft = 2 * n_points
        ifft_result = np.fft.ifft(np.fft.fft(h, n_fft) * np.conj(np.fft.fft(1 - h, n_fft)))
        corr = ifft_result.real[n_points - 1:2*n_points - 1]
        return np.max(corr) * dx
    
    def compute_gradient(x):
        h = x
        n_fft = 2 * n_points
        ifft_result = np.fft.ifft(np.fft.fft(h, n_fft) * np.conj(np.fft.fft(1 - h, n_fft)))
        corr = ifft_result.real[n_points - 1:2*n_points - 1]
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad
    
    start_time = time.time()
    perturbations = []
    
    # Symmetric mirroring
    h_mirrored = np.concatenate([h[:n_points//2], h[n_points//2:][::-1]])
    h_mirrored = np.clip(h_mirrored, 0, 1) * target_sum / np.sum(h_mirrored)
    perturbations.append(h_mirrored)
    
    # Sinusoidal modulation
    x = np.linspace(0, 1, n_points)
    sin_perturb = h * (1 + 0.15 * np.sin(2 * np.pi * x))
    sin_perturb = np.clip(sin_perturb, 0, 1) * target_sum / np.sum(sin_perturb)
    perturbations.append(sin_perturb)
    
    # Balanced segmentation
    seg_perturb = np.zeros(n_points)
    n_segment = n_points // 5
    for i in range(n_points):
        if i % n_segment == 0:
            seg_perturb[i] = 0.5
    seg_perturb = np.clip(seg_perturb, 0, 1) * target_sum / np.sum(seg_perturb)
    perturbations.append(seg_perturb)
    
    # Random permutation
    perm_perturb = np.random.permutation(h).copy()
    perm_perturb = np.clip(perm_perturb, 0, 1) * target_sum / np.sum(perm_perturb)
    perturbations.append(perm_perturb)
    
    for p in perturbations:
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, p, method='SLSQP', bounds=[(0, 1)]*n_points, 
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], 
                          jac=compute_gradient, tol=1e-6, options={'maxiter': 200})
        if result.success and result.fun < best_c5:
            best_h = result.x.copy()
            best_c5 = result.fun
        result_l_bfgs = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
