    result = differential_evolution(objective, bounds, strategy='best1bin', maxiter=100, popsize=30, tol=1e-6, mutation=(0.5, 1.0), recombination=0.8, disp=False)
    best_coeffs = result.x
    a_opt = best_coeffs[::2]
    b_opt = best_coeffs[1::2]
    best_h = 0.5 * np.ones(n_points)
    for n in range(N):
        cos_terms = np.cos((n + 1) * np.pi * x_i)
        sin_terms = np.sin((n + 1) * np.pi * x_i)
        best_h += a_opt[n] * cos_terms + b_opt[n] * sin_terms
    best_h = np.clip(best_h, 0, 1)
    best_h *= (target_sum / np.sum(best_h))
    best_c5 = objective(best_coeffs)

    h_initial_scaled = h_initial * (target_sum / np.sum(h_initial))
    initial_c5 = np.max(np.correlate(h_initial_scaled, 1 - h_initial_scaled, mode='full')) * dx
    if best_c5 >= initial_c5:
        best_h = h_initial_scaled
        best_c5 = initial_c5
    return (best_h, best_c5, n_points)
