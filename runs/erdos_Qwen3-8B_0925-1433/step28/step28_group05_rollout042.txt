import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        h_current = best_h.copy()
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        result = minimize(
            objective,
            h_current,
            method='L-BFGS-B',
            jac=compute_gradient,
            bounds=[(0, 1)] * n_points,
            options={'maxiter': 100, 'disp': False}
        )
        if not result.success:
            continue
        h_new = np.clip(result.x, 0, 1)
        sum_new = np.sum(h_new)
        if sum_new == 0:
            continue
        h_new = h_new * target_sum / sum_new
        c5_new = objective(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        if np.random.rand() < 0.05:
            perturb = np.random.normal(0, 0.1, n_points)
            h_perturb = h_new + perturb
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * target_sum / np.sum(h_perturb)
            c5_perturb = objective(h_perturb)
            if c5_perturb < best_c5:
                best_h = h_perturb
                best_c5 = c5_perturb

    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
