    if elapsed < budget_s:
        best_h = result_de.x.copy()
        best_c5 = result_de.fun

        def compute_gradient(x):
            h = x
            corr = np.correlate(h, 1 - h, mode='full')
            k_max_index = np.argmax(corr)
            k = k_max_index - (n_points - 1)
            grad = np.zeros_like(h)
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    grad[i] = 1.0 - h[j]
            return grad

        constraints = [{
            'type': 'eq',
            'fun': lambda x: np.sum(x) - target_sum
        }]
        bounds_l_bfgs = [(0, 1) for _ in range(n_points)]

        result_l_bfgs = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds_l_bfgs,
            constraints=constraints,
            jac=compute_gradient,
            tol=1e-6,
            options={'maxiter': 100}
        )

        if result_l_bfgs.success and result_l_bfgs.fun < best_c5:
            best_h = result_l_bfgs.x
            best_c5 = result_l_bfgs.fun

        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
    else:
        best_h = h_initial
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    return (best_h, best_c5, n_points)
