from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

            yield base * (1 + 0.5 * (np.random.rand(n_points) < 0.5).astype(float))

        h_de = np.clip(result_de.x, 0, 1) * target_sum / np.sum(np.clip(result_de.x, 0, 1))
