import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    n_iter = 500
    start_time = time.time()

    for _ in range(n_iter):
        if time.time() - start_time > budget_s:
            break
        current_corr = np.correlate(h, 1 - h, mode='full') * dx
        current_max = current_corr.max()
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        
        k_max = np.argmax(current_corr)
        h_fft = np.fft.fft(h)
        h_fft[k_max] *= 0.9
        h = np.fft.ifft(h_fft).real
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

    final_corr = np.correlate(h, 1 - h, mode='full') * dx
    final_max = final_corr.max()
    best_c5 = min(best_c5, final_max)
    return (h, best_c5, n_points)
