import numpy as np
import time

def run(seed=42, budget_s=600):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    learning_rate = 0.01
    decay_rate = 0.99
    start_time = time.time()

    for _ in range(1000):
        corr = np.correlate(h, 1 - h, mode='full')
        k_index = np.argmax(corr)
        lag = k_index - (n_points - 1)

        i_plus = np.arange(n_points) + lag
        i_plus = np.clip(i_plus, 0, n_points-1)
        term1 = (1.0 - h[i_plus]) * dx

        i_minus = np.arange(n_points) - lag
        i_minus = np.clip(i_minus, 0, n_points-1)
        term2 = -h[i_minus] * dx

        grad = term1 + term2

        h = h - learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
        if current_sum != 0:
            h = h * target_sum / current_sum
        else:
            h = np.zeros_like(h)

        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        learning_rate *= decay_rate
        if time.time() - start_time > budget_s:
            break

    current_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    if current_c5 < best_c5:
        best_c5 = current_c5

    return (best_h, best_c5, n_points)
