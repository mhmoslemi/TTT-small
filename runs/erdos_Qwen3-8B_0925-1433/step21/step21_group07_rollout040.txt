import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        shift = k_max_index - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            if 0 <= i_plus < n_points:
                term1 = 1 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - shift
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    def obj(x):
        return compute_c5(x)

    # Generate initial guesses
    initial_guesses = []
    base = initial_h_values.copy()
    for _ in range(5):
        noise = np.random.normal(0, 0.15, n_points)
        guess = base + noise
        guess = np.clip(guess, 0, 1)
        guess = guess * target_sum / np.sum(guess)
        initial_guesses.append(guess)

    best_c5 = float('inf')
    best_h = initial_h_values.copy()

    start_time = time.time()

    for guess in initial_guesses:
        h = guess.copy()
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        for _ in range(50):
            if time.time() - start_time > budget_s:
                break
            grad = compute_gradient(h)
            h = h - 0.3 * grad
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6, jac=compute_gradient)
    if res.success and res.fun < best_c5:
        best_h = res.x
        best_c5 = res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
