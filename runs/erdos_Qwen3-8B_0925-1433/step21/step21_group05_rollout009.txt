import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(h_vec)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h_vec[j]
        return grad

    def generate_initial_guesses(h_initial, n_points, target_sum):
        guesses = []
        guesses.append(h_initial.copy())
        for _ in range(4):
            if np.random.rand() < 0.5:
                idx = np.random.choice(n_points, size=int(0.3 * n_points), replace=False)
                h_perturbed = h_initial.copy()
                h_perturbed[idx] = 1.0 - h_perturbed[idx]
                h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
                guesses.append(h_perturbed)
            else:
                freq = 2 * np.pi * np.random.rand()
                phase = np.random.rand() * 2 * np.pi
                h_perturbed = h_initial.copy()
                h_perturbed += 0.5 * np.sin(freq * np.arange(n_points) + phase)
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
                guesses.append(h_perturbed)
        return guesses

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()

    guesses = generate_initial_guesses(h, n_points, target_sum)
    for guess in guesses[:5]:
        if time.time() - start_time > budget_s:
            break
        result = differential_evolution(
            compute_c5,
            bounds,
            popsize=20,
            maxiter=100,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            polish=True,
            polish_bounds=bounds
        )
        h_candidate = np.clip(result.x, 0, 1)
        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(
            compute_c5,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            jac=compute_gradient,
            tol=1e-6,
            options={'maxiter': 100}
        )
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(
            compute_c5,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            jac=compute_gradient,
            tol=1e-6,
            options={'maxiter': 100}
        )
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
