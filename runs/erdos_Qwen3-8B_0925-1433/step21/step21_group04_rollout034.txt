def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

    dx = 2.0 / n_points

    # Initialize h
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)  # Scale to target_sum

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()

    iterations = 0
    while time.time() - start_time < budget_s:
        iterations += 1
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        j = np.argmax(corr)
        k = j - (n_points - 1)
        # Find valid i's for the lag k
        valid_indices = []
        for i in range(n_points):
            if 0 <= i + k < n_points:
                valid_indices.append(i)
        if not valid_indices:
            break  # no valid indices, skip
        # Find the i with max contribution
        contrib = [h[i] * (1 - h[i + k]) for i in valid_indices]
        max_contrib_idx = valid_indices[np.argmax(contrib)]
        i = max_contrib_idx
        # Adjust h[i] and h[i + k]
        delta = 0.01
        h[i] -= delta
        h[i + k] += delta
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        # Rescale to target_sum
        sum_h = np.sum(h)
        scale = target_sum / sum_h
        h = h * scale
        # Compute current c5 and check improvement
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

        # Early termination if improvement is negligible
        if iterations > 1000 or abs(current_c5 - best_c5) < 1e-6:

    return (best_h, best_c5, n_points)

def run(seed=42, budget_s=600):

    target_sum = n_points / 2.0

    h = h * target_sum / np.sum(h)

    start_time = time.time()
    max_iterations = 1000

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        j = np.argmax(corr)
        k = j - (n_points - 1)
        valid_indices = [i for i in range(n_points) if 0 <= i + k < n_points]
        if not valid_indices:
            break
        contrib = [h[i] * (1 - h[i + k]) for i in valid_indices]
        i = valid_indices[np.argmax(contrib)]
        # Adjust h[i] and h[i + k]

        h = np.clip(h, 0, 1)
