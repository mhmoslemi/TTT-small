    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    H = np.fft.fft(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    iterations = 0
    max_iterations = 500
    while time.time() - start_time < budget_s:
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        current_max = np.max(overlaps)
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()

        k_max = np.argmax(overlaps)
        delta = 0.006
        for m in range(max(0, k_max - 2), min(n_points, k_max + 3)):
            sym_m = n_points - m - 1
            H[m] -= delta
            H[sym_m] -= delta
            H[m] = np.clip(H[m], -1e6, 1e6)
            H[sym_m] = np.clip(H[sym_m], -1e6, 1e6)

        h_new = np.fft.ifft(H)
        h_new = np.real(h_new)
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        scale = target_sum / current_sum if current_sum != 0 else 1.0
        h_new *= scale
        h = h_new.copy()
        iterations += 1
        if iterations >= max_iterations:
            break

    final_overlaps = np.correlate(h, 1 - h, mode='full') * dx
    best_c5 = np.max(final_overlaps)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    scale = target_sum / current_sum
    h *= scale
    return (h, best_c5, n_points)
