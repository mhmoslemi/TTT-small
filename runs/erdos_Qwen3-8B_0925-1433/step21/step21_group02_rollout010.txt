import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if abs(sum_h - target_sum) > 1e-6:
        h *= target_sum / sum_h
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    eta = 0.02
    iteration = 0
    prev_best = best_c5
    start_time = time.time()
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        m = np.argmax(corr)
        k = m - n_points + 1
        gradient = np.zeros_like(h)
        for i in range(n_points):
            if i + k < n_points:
                gradient[i] += 1 - h[i + k]
            if i - k >= 0:
                gradient[i] -= h[i - k]
        h -= eta * gradient
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h *= target_sum / sum_h
        current_c5 = max_corr * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if abs(best_c5 - prev_best) < 1e-6:
            break
        prev_best = best_c5
        iteration += 1
    return (best_h, best_c5, n_points)
