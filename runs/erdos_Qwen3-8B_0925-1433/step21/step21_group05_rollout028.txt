import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
    best_c5 = min(best_c5, 0.381)

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * (2.0 / n_points)

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if total == 0:
            return np.zeros_like(x)
        return x * target_sum / total

    bounds = [(0, 1)] * n_points

    def de_objective(x):
        x = clip_rescale(x)
        return objective(x)

    def slsqp_objective(x):
        x = clip_rescale(x)
        return objective(x)

    def lbfgs_objective(x):
        x = clip_rescale(x)
        return objective(x)

    def de_bounds_transform(x):
        return np.clip(x, 0, 1)

    def de_fitness(x):
        return de_objective(x)

    def slsqp_fitness(x):
        return slsqp_objective(x)

    def lbfgs_fitness(x):
        return lbfgs_objective(x)

    def de_generator(pop_size=25, iters=100):
        result = differential_evolution(
            de_fitness,
            bounds,
            strategy='best1bin',
            popsize=pop_size,
            mutation=(0.8, 1.0),
            recombination=0.9,
            tol=1e-6,
            maxiter=iters,
            disp=False,
            polish=False,
            vectorized=True,
            mutation_func=lambda x, bounds, *args: x + np.random.uniform(-1, 1, x.shape) * 0.5,
            mutation_probs=None,
            crossover_func=None,
            crossover_prob=None,
            seed=seed,
            workers=-1
        )
        return result.x

    def slsqp_refine(x0):
        result = minimize(
            slsqp_fitness,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            tol=1e-6,
            options={'maxiter': 200}
        )
        return result.x

    def lbfgs_refine(x0):
        result = minimize(
            lbfgs_fitness,
            x0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            tol=1e-6,
            options={'maxiter': 100}
        )
        return result.x

    start_time = time.time()
    candidates = []

    # Generate diverse initial candidates
    for _ in range(3):
        if np.random.rand() < 0.3:
            idx = np.random.choice(n_points, size=int(0.3 * n_points), replace=False)
            h_perturbed = h.copy()
            h_perturbed[idx] = 1 - h_perturbed[idx]
            h_perturbed = clip_rescale(h_perturbed)
            candidates.append(h_perturbed)
        else:
            h_perturbed = np.random.rand(n_points)
            h_perturbed = clip_rescale(h_perturbed)
            candidates.append(h_perturbed)

    # DE global search
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        try:
            de_result = de_generator(pop_size=25, iters=100)
            candidates.append(de_result)
        except:
            pass

    # Local refinement
    for candidate in candidates[:5]:
        if time.time() - start_time > budget_s:
            break
        refined = slsqp_refine(candidate)
        candidates.append(refined)

    # Final optimization
    for candidate in candidates[:3]:

        refined = lbfgs_refine(candidate)
        candidates.append(refined)

    # Evaluate all candidates
    for candidate in candidates:
        if time.time() - start_time > budget_s:
            break
        c5 = objective(candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate

    best_h = clip_rescale(best_h)
    return (best_h, best_c5, n_points)
