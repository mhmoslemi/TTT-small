import numpy as np
from scipy.optimize import minimize, SLSQP
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        h_opt = x * target_sum / np.sum(x)
        return evaluate(h_opt)

    def local_optimize(h):
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
        return res.x if res.success else h

    def perturb(h):
        noise = np.random.normal(0, 0.1, n_points)
        h_perturb = h + noise
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        return h_perturb

    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    initial_h = np.clip(initial_h, 0, 1)

    population = [initial_h]
    for _ in range(4):
        population.append(perturb(initial_h))

    best_h = initial_h
    best_c5 = evaluate(best_h)

    start_time = time.time()
    pop_size = 15
    elite_count = 5
    generations = 200
    mutation_std = 0.06
    crossover_prob = 0.92

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        evals = np.array([evaluate(h) for h in population])
        elite_indices = np.argsort(evals)[:elite_count]
        elites = population[elite_indices]

        new_population = []
        for _ in range(pop_size):
            if np.random.rand() < crossover_prob:
                p1, p2 = elites[np.random.choice(elite_count, 2, replace=False)]
                idx = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:idx], p2[idx:]])
            else:
                child = elites[np.random.choice(elite_count)].copy()
            noise = np.random.normal(0, mutation_std, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = new_population

        for h in population:
            current_c5 = evaluate(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

    best_h = local_optimize(best_h)
    return (best_h, evaluate(best_h), n_points)
