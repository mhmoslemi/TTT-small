from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)

        initial_guesses.append(perturbed)
    initial_guesses = np.array(initial_guesses)
    
    best_h = h_initial.copy()

    # Initial optimization with COBYLA
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
