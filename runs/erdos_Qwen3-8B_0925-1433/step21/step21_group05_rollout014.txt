import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    max_lag = 2 * n_points - 2

    def validate_h(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def objective(h_vec):
        h = validate_h(h_vec)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def structured_initial_guesses():
        h1 = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                h1[i] = 1
        h1 = validate_h(h1)
        yield h1

        h2 = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                h2[i] = 1
        h2 = validate_h(h2)
        yield h2

        h3 = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / (n_points - 1)))
        h3 = np.clip(h3, 0, 1)
        h3 = h3 * target_sum / np.sum(h3)
        yield h3

    initial_guesses = list(structured_initial_guesses())
    initial_guesses.append(np.random.rand(n_points))
    initial_guesses.append(validate_h(initial_h_values))

    best_h = initial_h_values * target_sum / np.sum(initial_h_values)
    best_c5 = objective(best_h)

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1)] * n_points
        result = differential_evolution(
            lambda x: objective(x),
            bounds,
            strategy='best1bin',
            popsize=25,
            maxiter=80,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.9,
            seed=seed
        )
        if result.success:
            candidate = result.x
            candidate = validate_h(candidate)
            current_c5 = objective(candidate)
            if current_c5 < best_c5:
                best_h = candidate
                best_c5 = current_c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            tol=1e-6,
            options={'maxiter': 200}
        )
        if result.success:
            best_h = result.x
            best_h = validate_h(best_h)
            best_c5 = objective(best_h)

    for _ in range(2):
        if time.time() - start_time > budget_s:

        result = minimize(

        )
        if result.success:
            best_h = result.x

    return (best_h, best_c5, n_points)
