    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()

    # Run from initial_h_values
    res = minimize(evaluate, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 500})
    if res.success:
        current_h = res.x
    else:
        current_h = h
    current_c5 = evaluate(current_h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()

    # Final refinement
    def obj(x):
        h = x * target_sum / np.sum(x)
        return evaluate(h)
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 200})
    final_h = res.x if res.success else best_h
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * target_sum / np.sum(final_h)
    return (final_h, best_c5, n_points)
