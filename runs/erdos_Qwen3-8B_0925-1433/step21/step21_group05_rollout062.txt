from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def clip_rescale(v):
        v = np.clip(v, 0, 1)
        v = v * target_sum / np.sum(v)
        return v
    
    def structured_initial():
        base = np.zeros(n_points)
        n_blocks = int(n_points / 4)
        base[::n_blocks] = 1
        shifts = [0, int(n_points / 4), int(n_points / 2)]
        for shift in shifts:
            h_shift = np.roll(base, shift)
            yield h_shift.copy()
        sin_base = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / (n_points - 1)))
        yield np.clip(sin_base, 0, 1) * target_sum / np.sum(np.clip(sin_base, 0, 1))
    
    start_time = time.time()
    best_candidates = [h.copy()]
    
    # Differential Evolution
    bounds = [(0, 1) for _ in range(n_points)]
    for init in structured_initial():
        if time.time() - start_time > budget_s:
            break
        result = differential_evolution(objective, bounds, strategy='best1bin', popsize=25, tol=1e-6, mutation=(0.8, 1), recombination=0.9, maxiter=80, seed=seed)
        if result.success:
            best_candidates.append(clip_rescale(result.x))
    
    # SLSQP Refinement
    for candidate in best_candidates[:5]:
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, candidate, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       bounds=bounds, tol=1e-6, options={'maxiter': 200})
        if res.success:
            best_candidates.append(clip_rescale(res.x))
    
    # L-BFGS-B Final Optimization
    for candidate in best_candidates[:3]:
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, candidate, method='L-BFGS-B',
                       bounds=bounds, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       tol=1e-6, options={'maxiter': 100})
        if res.success:
            best_h = clip_rescale(res.x)
            best_c5 = res.fun
    
    best_h = clip_rescale(best_h)
    return (best_h, best_c5, n_points)
