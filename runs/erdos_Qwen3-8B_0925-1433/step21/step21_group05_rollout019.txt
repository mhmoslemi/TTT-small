import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vec):
        h = np.array(h_vec)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def rescale(x):
        x = np.clip(x, 0, 1)
        return x * target_sum / np.sum(x)

    def generate_sine(n):
        x = np.linspace(0, 2, n)
        return 0.5 * (1 + np.sin(np.pi * x / 2)) + 1e-6

    def generate_block(n):
        return np.tile([1, 0, 1, 0], n // 4)[:n]

    def generate_lattice(n, phase=0.25):
        pattern = np.zeros(n)
        for i in range(n):
            pattern[i] = 1 if (i * 2 % n) < phase * n else 0
        return pattern

    def generate_greedy(n):
        h = np.zeros(n)
        for i in range(n):
            if np.sum(h) >= target_sum:
                break
            h[i] = 1
        for i in reversed(range(n)):
            if np.sum(h) >= target_sum:
                break
            h[i] = 1
        return h

    def generate_random(n):
        return np.random.rand(n)

    initial_guesses = []
    for _ in range(3):
        guess = rescale(generate_sine(n_points))
        initial_guesses.append(guess)
    for _ in range(3):
        guess = rescale(generate_block(n_points) + np.random.rand(n_points) * 0.1)
        initial_guesses.append(guess)
    for _ in range(3):
        guess = rescale(generate_lattice(n_points, np.random.rand()))
        initial_guesses.append(guess)
    for _ in range(3):
        guess = rescale(generate_greedy(n_points) + np.random.rand(n_points) * 0.1)
        initial_guesses.append(guess)
    initial_guesses.append(rescale(np.random.rand(n_points)))

    best_h = rescale(initial_h_values)
    best_c5 = objective(best_h)

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1)] * n_points
        result = differential_evolution(
            objective, bounds, strategy='best1bin', popsize=15, maxiter=50,
            tol=1e-6, mutation=(0.5, 1.0), recombination=0.9, seed=seed)
        if result.success:
            h_candidate = rescale(result.x)
            c5 = objective(h_candidate)
            if c5 < best_c5:
                best_h, best_c5 = h_candidate, c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, best_h, method='L-BFGS-B',
                          bounds=[(0, 1)] * n_points,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                          tol=1e-6, options={'maxiter': 100})
        if result.success and result.fun < best_c5:
            best_h = rescale(result.x)
            best_c5 = result.fun

    best_h = rescale(best_h)
    return (best_h, best_c5, n_points)
