import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    population_size = 50
    population = [np.array(initial_h_values, dtype=float)]

        h = h * target_sum / np.sum(h)
        population.append(h)

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, 0.381)

    tournament_size = 5
    crossover_prob = 0.8
    mutation_rate = 0.15
    generations = 200
    start_time = time.time()

            break
        fitness = [compute_c5(h) for h in population]
        new_population = []
        for _ in range(population_size):

            idxs2 = np.random.choice(len(population), size=tournament_size, replace=False)

            parent2 = population[best_idx2]
