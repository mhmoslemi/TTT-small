import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    eta = 0.05
    iteration = 0
    max_iterations = 1000  # Tuned for balance between speed and convergence
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    while iteration < max_iterations and time.time() - start_time < budget_s:
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h[:]
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + shift < n_points:
                grad[i] = (1 - h[i + shift]) * dx
        h = h - eta * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        eta *= 0.9995
        iteration += 1
    
    h = np.clip(best_h, 0, 1)
    h = h * target_sum / np.sum(h)
    return (h, best_c5, n_points)
