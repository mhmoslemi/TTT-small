import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    # Generate initial guesses with perturbations
    guesses = [initial_h.copy()]
    for _ in range(3):
        perturb = np.random.normal(0, 0.08, n_points)
        guess = initial_h.copy() + perturb
        guess = np.clip(guess, 0, 1)
        guess = guess * target_sum / np.sum(guess)
        guesses.append(guess)
    
    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            current_c5 = objective(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
