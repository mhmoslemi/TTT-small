                best_h_full = np.zeros(n_points)
                best_h_full[:half_size] = current_h_half
                best_h_full[half_size:] = 1 - current_h_half[::-1]

    def gradient_descent(h_half, steps=100, learning_rate=0.1):
        for _ in range(steps):
            h_clipped = np.clip(h_half, 0, 1)
            sum_clipped = np.sum(h_clipped)
            if sum_clipped == 0:
                continue
            scaling = target_sum / sum_clipped
            h_scaled = h_clipped * scaling
            h_full = np.zeros(n_points)
            h_full[:half_size] = h_scaled
            h_full[half_size:] = 1 - h_scaled[::-1]
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            max_corr_index = np.argmax(corr)
            grad = np.zeros_like(h_scaled)
            for i in range(half_size):
                h_plus = h_scaled.copy()
                h_plus[i] += 1e-5
                h_plus_clipped = np.clip(h_plus, 0, 1)
                sum_plus = np.sum(h_plus_clipped)
                if sum_plus == 0:
                    continue
                scaling_plus = target_sum / sum_plus
                h_plus_scaled = h_plus_clipped * scaling_plus
                h_full_plus = np.zeros(n_points)
                h_full_plus[:half_size] = h_plus_scaled
                h_full_plus[half_size:] = 1 - h_plus_scaled[::-1]
                corr_plus = np.correlate(h_full_plus, 1 - h_full_plus, mode='full')
                grad[i] = (np.max(corr_plus) - np.max(corr)) / 1e-5
            h_scaled -= learning_rate * grad
            h_clipped = np.clip(h_scaled, 0, 1)
            sum_clipped = np.sum(h_clipped)
            if sum_clipped == 0:
                continue
            scaling = target_sum / sum_clipped
            h_scaled = h_clipped * scaling
            h_half = h_scaled[:half_size]
        return h_half

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        refined_half = gradient_descent(best_h_half.copy())
        refined_c5 = objective(refined_half)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h_half = refined_half
            best_h_full = np.zeros(n_points)
            best_h_full[:half_size] = best_h_half
            best_h_full[half_size:] = 1 - best_h_half[::-1]

    return (best_h_full, best_c5, n_points)
