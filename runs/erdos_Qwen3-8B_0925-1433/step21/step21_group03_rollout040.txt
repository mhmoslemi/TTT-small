    start_time = time.time()
    for variant in variants:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1)] * n_points
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=25,
            mutation=(0.5, 0.8),
            recombination=0.9,
            tol=1e-6,
            maxiter=300,
            seed=seed
        )
        if result.success:
            current_h = clip_and_rescale(result.x)
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

    # Local refinement with gradient descent
    def local_objective(h):
        h_clip = np.clip(h, 0, 1)
        h_rescale = h_clip * target_sum / np.sum(h_clip)
        corr = np.correlate(h_rescale, 1 - h_rescale, mode='full')
        return np.max(corr) * dx

    h_refined = best_h.copy()
    for _ in range(100):
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_perturb = h_refined.copy()
            h_perturb[i] += 1e-5
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * target_sum / np.sum(h_perturb)
            c_perturb = local_objective(h_perturb)
            c_original = local_objective(h_refined)
            grad[i] = (c_perturb - c_original) / 1e-5
        step = 0.1
        h_refined -= step * grad
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * target_sum / np.sum(h_refined)
        if np.max(np.abs(grad)) < 1e-3:
            break
    final_c5 = local_objective(h_refined)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h_refined

    return (best_h, best_c5, n_points)
