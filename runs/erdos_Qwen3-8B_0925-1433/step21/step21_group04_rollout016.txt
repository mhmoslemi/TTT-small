import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    population_size = 50
    population = [np.array(initial_h_values, dtype=float)]

    for _ in range(population_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = target_sum / sum_h
        h *= scale
        population.append(h)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    c5_values = [compute_c5(h) for h in population]
    best_index = np.argmin(c5_values)
    best_h = population[best_index]
    best_c5 = c5_values[best_index]

    def tournament_selection(pop, c5, tournament_size=5):
        indices = np.random.choice(len(pop), tournament_size, replace=False)
        best_idx = indices[np.argmin(c5[indices])]
        return pop[best_idx]

    start_time = time.time()
    max_generations = 200
    tournament_size = 5
    crossover_prob = 0.8
    mutation_prob = 0.15
    mutation_strength = 0.05

    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break

        new_population = []

        for _ in range(population_size):
            parent1 = tournament_selection(population, c5_values, tournament_size)
            parent2 = tournament_selection(population, c5_values, tournament_size)

            if np.random.rand() < crossover_prob:
                child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            else:
                child = parent1.copy()

            for i in range(n_points):
                if np.random.rand() < mutation_prob:
                    child[i] += np.random.normal(0, mutation_strength)
                    child[i] = np.clip(child[i], 0, 1)

            sum_child = np.sum(child)
            scale = target_sum / sum_child
            child *= scale

            new_population.append(child)

        combined = population + new_population
        combined_c5 = c5_values + [compute_c5(h) for h in new_population]
        combined_sorted = sorted(zip(combined, combined_c5), key=lambda x: x[1])
        selected = combined_sorted[:int(0.75 * len(combined_sorted))]
        population = [h for h, _ in selected[:population_size]]
        c5_values = [c for _, c in selected[:population_size]]

        current_best_index = np.argmin(c5_values)
        if c5_values[current_best_index] < best_c5:
            best_c5 = c5_values[current_best_index]
            best_h = population[current_best_index]

    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
