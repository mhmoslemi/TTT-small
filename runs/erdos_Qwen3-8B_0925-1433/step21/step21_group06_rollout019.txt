from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_guess = np.array(initial_h_values, dtype=float)
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraints(h):
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraints}

    def generate_perturbations(h):
        perturbed = h.copy()
        noise = np.random.normal(0, 0.1, n_points)
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        return perturbed

    def refine(h):
        res = minimize(obj, h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-8, options={'max iterations': 100})
        if res.success:
            h = np.clip(res.x, 0, 1)
            h = h * target_sum / np.sum(h)
            return h
        return h

    start_time = time.time()
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        h = generate_perturbations(best_h)
        h = refine(h)
        c5 = obj(h)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
