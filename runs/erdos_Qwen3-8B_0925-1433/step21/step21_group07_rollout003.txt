import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = evaluate(best_h)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    def obj(x):
        return evaluate(x)

    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            if 0 <= i_plus < n_points:
                term1 = 1 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - shift
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    start_time = time.time()
    h = h * target_sum / np.sum(h)
    learning_rate = 0.15
    momentum = 0.8
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        grad_val = grad(h)
        h -= learning_rate * grad_val
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = evaluate(h)
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, bounds=bounds, constraints=constraints, method='L-BFGS-B', jac=grad, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
