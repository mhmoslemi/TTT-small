    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    global_best = best_h
    global_best_c5 = best_c5

    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for _ in range(len(population) // 2):
            p1 = np.random.choice(population, 5, replace=False)
            p1 = p1[np.argmin([np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in p1])]
            p2 = np.random.choice(population, 5, replace=False)
            p2 = p2[np.argmin([np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in p2])]
            child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
            if np.random.rand() < 0.15:
                noise = 0.1 * np.random.randn(n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child *= target_sum / np.sum(child)
                child = np.clip(child, 0, 1)
            new_population.append(child)
        
        combined = population + new_population
        evaluated = [(h, np.max(np.correlate(h, 1 - h, mode='full')) * dx) for h in combined]
        evaluated.sort(key=lambda x: x[1])
        population = [h for h, _ in evaluated[:len(population)]]
        
        for h in population:
            c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if c5 < global_best_c5:
                global_best = h.copy()
                global_best_c5 = c5

    global_best = np.clip(global_best, 0, 1)
    global_best *= target_sum / np.sum(global_best)
    global_best = np.clip(global_best, 0, 1)
    return (global_best, global_best_c5, n_points)
