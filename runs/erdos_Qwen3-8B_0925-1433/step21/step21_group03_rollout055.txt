import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_rescale(x):
        x_clipped = np.clip(x, 0, 1)
        sum_clipped = np.sum(x_clipped)
        if sum_clipped == 0:
            return x_clipped
        scaling = target_sum / sum_clipped
        x_scaled = x_clipped * scaling
        # Enforce symmetry
        n = len(x_scaled)
        for i in range(n // 2):
            x_scaled[2*n - i - 2] = 1 - x_scaled[i]
        # Clip again to ensure [0,1]
        x_scaled = np.clip(x_scaled, 0, 1)
        sum_scaled = np.sum(x_scaled)
        if sum_scaled != 0:
            scaling = target_sum / sum_scaled
            x_scaled = x_scaled * scaling
        return x_scaled

    def objective(x):
        h = clip_and_rescale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate initial variants
    variants = []
    h_initial = np.array(initial_h_values, dtype=float)

    variants.append(h_initial.copy())

    # Add some perturbations
    for _ in range(4):
        h = h_initial.copy()

        h[:n_points//2] += noise
        h = clip_and_rescale(h)
        variants.append(h)

    best_c5 = min(best_c5, 0.381)
