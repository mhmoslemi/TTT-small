    max_iterations = int(budget_s / 10)
    while time.time() - start_time < budget_s:
        iteration += 1
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        lag = max_index - (n_points - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + lag < n_points:
                grad[i] += (1 - h[i + lag]) * dx
            if 0 <= i - lag < n_points:
                grad[i] -= h[i - lag] * dx
        
        eta = 0.5 * (0.99) ** iteration
        h -= eta * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        else:
            h = np.zeros(n_points)
        
        current_c5 = corr[max_index] * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        if iteration > 200 and abs(current_c5 - best_c5) < 1e-8:
            break
    
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(final_corr) * dx
    return (best_h, best_c5, n_points)
