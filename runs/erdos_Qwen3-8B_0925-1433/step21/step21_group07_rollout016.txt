import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            if 0 <= i_plus < n_points:
                term1 = 1.0 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - shift
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    def optimize_guess(guess):
        h = guess.copy()
        h = h * target_sum / np.sum(h)
        start_time = time.time()
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            grad = compute_gradient(h)
            h += 0.3 * grad
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6, jac=compute_gradient, options={'maxiter': 100})
        if res.success:
            h = res.x
        return h

    initial_guesses = []
    for _ in range(5):
        h = initial_h_values.copy()
        noise = np.random.uniform(-0.15, 0.15, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        initial_guesses.append(h)

    best_h = initial_guesses[0]
    best_c5 = compute_c5(best_h)

    for guess in initial_guesses:
        optimized = optimize_guess(guess)
        current_c5 = compute_c5(optimized)
        if current_c5 < best_c5:
            best_h = optimized
            best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
