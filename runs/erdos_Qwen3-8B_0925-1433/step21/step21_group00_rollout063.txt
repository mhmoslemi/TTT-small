import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    learning_rate = 0.15
    momentum = 0.95
    decay = 0.995
    max_iterations = 250
    start_time = time.time()
    m = np.zeros_like(h)

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        i_plus = (np.arange(n_points) + shift) % n_points
        i_minus = (np.arange(n_points) - shift) % n_points
        term1 = 1 - h[i_plus]
        term2 = h[i_minus]
        grad = term1 - term2
        m = momentum * m + (1 - momentum) * grad
        h = h - learning_rate * m
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=Bounds(0, 1), constraints=cons, tol=1e-6)
    best_h, best_c5 = res.x, res.fun

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        noise = np.random.normal(0, 0.015, size=n_points)
        h_perturb = np.clip(best_h + noise, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        res = minimize(obj, h_perturb, method='L-BFGS-B', bounds=Bounds(0, 1), constraints=cons, tol=1e-6)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
