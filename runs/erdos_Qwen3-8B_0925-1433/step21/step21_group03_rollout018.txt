from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0

    def clip_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return h_clipped
        scaling = target_sum / sum_clipped
        return h_clipped * scaling

    def objective(x):
        h = clip_rescale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * 2.0 / n_points

    variants = [initial_h_values.copy()]
    transforms = [
        lambda h: h[::-1],  # Mirror
        lambda h: 1 - h,    # Invert
        lambda h: np.roll(h, n_points // 2),  # Shift
        lambda h: (h - 0.5) * 2 + 0.5,  # Normalize
    ]
    for _ in range(3):
        h = np.array(initial_h_values, dtype=float)
        h += np.random.normal(0, 0.2, n_points)
        h = clip_rescale(h)
        variants.append(h)
        for t in transforms:
            h_t = t(h)
            h_t = clip_rescale(h_t)
            variants.append(h_t)

    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)

    start_time = time.time()
    for i, variant in enumerate(variants):
        if time.time() - start_time > budget_s:
            break
        h_init = np.array(variant, dtype=float)
        bounds = [(0, 1)] * n_points
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=20,
            mutation=(0.6, 0.9),
            recombination=0.7,
            tol=1e-6,
            maxiter=100,
            seed=seed + i
        )
        if result.success:
            current_h = clip_rescale(result.x)
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

    # Local refinement
    for _ in range(5):
        if time.time() - start_time > budget_s:

        h = best_h.copy()
