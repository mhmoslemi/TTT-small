        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)

    max_iterations = 1000
    for _ in range(max_iterations):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        if lag >= 0:
            indices = np.arange(0, n_points - lag)
        else:
            m = -lag
            indices = np.arange(m, n_points)
        if len(indices) == 0:
            continue
        contrib = h[indices] * (1 - h[indices + lag])
        i = indices[np.argmax(contrib)]
        delta = 0.05
        h[i] = max(0, h[i] - delta)
        current_sum = np.sum(h)
        scale = target_sum / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(corr) * dx
    best_c5 = min(best_c5, final_c5)
    return (best_h, best_c5, n_points)
