import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    h = h_initial * target_sum / np.sum(h_initial)
    
    def obj(x):
        h_scaled = x * target_sum / np.sum(x)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    def evaluate(h):
        h_scaled = h * target_sum / np.sum(h)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    # Generate initial variations
    def generate_variants(h):
        variants = []
        variants.append(h.copy())
        variants.append(1 - h.copy())

        # Sparsify
        mask = np.random.choice([True, False], size=len(h), p=[0.5, 0.5])
        sparse = h.copy()
        sparse[~mask] = 0
        scale = target_sum / (np.sum(sparse[mask]) if np.sum(sparse[mask]) != 0 else 1)
        sparse[mask] *= scale
        variants.append(sparse)
        return variants
    
    initial_pop = generate_variants(h_initial)
    pop = [np.array(v, dtype=float) for v in initial_pop]
    
    start_time = time.time()
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        fitness = [(-evaluate(p), p) for p in pop]
        fitness.sort()
        top5 = [p for _, p in fitness[:5]]
