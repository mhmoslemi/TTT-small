import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_sine():
        h = 0.5 * (1 + np.sin(np.pi * np.arange(n_points) / n_points))
        return h * target_sum / np.sum(h)

    def generate_lattice():
        h = np.zeros(n_points)
        h[::2] = 1.0
        return h * target_sum / np.sum(h)

    def generate_block():
        block_size = n_points // 4
        h = np.zeros(n_points)
        for i in range(0, n_points, 2 * block_size):
            h[i:i+block_size] = 1.0
        return h * target_sum / np.sum(h)

    def generate_greedy():
        h = np.zeros(n_points)
        used = np.zeros(n_points, dtype=bool)
        for i in range(n_points // 2):
            idx = np.random.choice(np.where(~used)[0])
            h[idx] = 1.0
            used[idx] = True
        return h * target_sum / np.sum(h)

    initial_guesses = [np.array(initial_h_values, dtype=float), generate_sine(), generate_lattice(), generate_block(), generate_greedy()]

    best_h = initial_guesses[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1)] * n_points
        def obj(x):
            return np.max(np.correlate(x, 1 - x, mode='full')) * dx
        res = differential_evolution(obj, bounds, strategy='best1bin', popsize=15, maxiter=50, tol=1e-4, mutation=(0.5, 1), recombination=0.8, polish=True)
        if res.success:
            x = res.x
            def obj_slqp(x):
                return np.max(np.correlate(x, 1 - x, mode='full')) * dx
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            res_slqp = minimize(obj_slqp, x, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
            if res_slqp.success and res_slqp.fun < best_c5:
                best_c5 = res_slqp.fun
                best_h = res_slqp.x.copy()

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
