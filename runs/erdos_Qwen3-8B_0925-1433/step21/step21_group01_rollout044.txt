from scipy.fft import fft, ifft

import time

def run(seed=42, budget_s=600):

        initial_h_values = np.random.rand(45) * 2 - 1
        initial_h_values = np.clip(initial_h_values, 0, 1)

        noise = np.random.normal(0, 0.1, n_points)
        h += noise

        h = h * target_sum / np.sum(h)
        population.append(h)
    population = np.array(population)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_smooth.copy()
    best_c5 = evaluate(best_h)
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()
    pop_size = 20
    elite_count = 5
    generations = 200
    mutation_std = 0.05
    crossover_prob = 0.8
    no_improvement = 0

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        evals = np.array([evaluate(h) for h in population])

        elites = population[sorted_indices[:elite_count]]
        
        new_population = []

            if np.random.rand() < crossover_prob:
                p1, p2 = elites[np.random.choice(elite_count, 2, replace=False)]
