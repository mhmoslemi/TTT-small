    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx
    def objective(h_vec):
        h_clamped = np.clip(h_vec, 0, 1)
        h_scaled = h_clamped * target_sum / np.sum(h_clamped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    def compute_gradient(h_vec):
        h_clamped = np.clip(h_vec, 0, 1)
        h_scaled = h_clamped * target_sum / np.sum(h_clamped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h_vec)
        for i in range(n_points):
            j_add = (i + k) % n_points
            j_sub = (i - k) % n_points
            grad[i] = (1 - h_scaled[j_add]) - h_scaled[j_sub]
        return grad
    structured_guesses = [h_initial]
    n = n_points
    # Greedy sparse
    greedy = np.zeros(n)
    for i in range(n):
        idx = int(i * 1.5)
        if idx < n:
            greedy[idx] = 1
    greedy = greedy * target_sum / np.sum(greedy)
    structured_guesses.append(greedy)
    # Sinusoidal
    sine = 0.5 * (1 + np.sin(np.pi * np.arange(n) / n))
    sine = np.clip(sine, 0, 1)
    sine = sine * target_sum / np.sum(sine)
    structured_guesses.append(sine)
    # Lattice
    phase = 0.3
    lattice = np.zeros(n)
    for i in range(n):
        if (i * 2 + phase) % 2 < 1:
            lattice[i] = 1
    lattice = lattice * target_sum / np.sum(lattice)
    structured_guesses.append(lattice)
    # Block
    block_size = n // 4
    block = np.tile([1]*block_size + [0]*block_size, n // (2*block_size) + 1)[:n]
    block = block * target_sum / np.sum(block)
    structured_guesses.append(block)
    # Random
    random_guess = np.random.rand(n) * target_sum / np.sum(np.random.rand(n))
    structured_guesses.append(random_guess)
    initial_guesses = structured_guesses
    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = differential_evolution(objective, [(0, 1)] * n, popsize=15, tol=1e-6, maxiter=100)
        if res.success:
            current_c5 = objective(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       bounds=[(0, 1)] * n, jac=compute_gradient, tol=1e-6, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B',
                       bounds=[(0, 1)] * n, jac=compute_gradient, tol=1e-6, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
