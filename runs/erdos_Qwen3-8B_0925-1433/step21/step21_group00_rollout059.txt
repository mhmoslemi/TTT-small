from scipy.optimize import minimize
import time

        corr = np.correlate(h, 1 - h, mode='full')

            shift = k - (n_points - 1)

                term1 = 1 - h[i_plus]
                i_minus = (i - shift) % n_points
                term2 = h[i_minus]
                grad[i] += term1 - term2
        return grad / len(k_indices)
