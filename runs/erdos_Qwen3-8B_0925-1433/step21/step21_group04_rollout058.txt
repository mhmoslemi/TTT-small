    h_sum = np.sum(h)

    start_time = time.time()
    iterations = 0
    no_improvement = 0
    while time.time() - start_time < budget_s and iterations < 10000:
        iterations += 1
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        start_i = max(0, 0 - k)
        end_i = min(n_points - 1, n_points - 1 - k)
        if start_i > end_i:
            no_improvement += 1
            continue
        contributions = []
        for i in range(start_i, end_i + 1):
            j = i + k
            contributions.append(h[i] * (1 - h[j]))
        if not contributions:
            no_improvement += 1
            continue
        i_to_adjust = np.argmax(contributions)
        i = start_i + i_to_adjust
        j = i + k
        tau = 0.01
        h[i] -= tau
