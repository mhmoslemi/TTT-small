import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_initial_guesses():
        guesses = [np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)]
        binary = np.random.rand(n_points) > 0.3
        guesses.append(binary * target_sum / np.sum(binary))
        cosine = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n_points))
        guesses.append(cosine * target_sum / np.sum(cosine))
        shuffled = np.random.permutation(initial_h_values)
        guesses.append(shuffled * target_sum / np.sum(shuffled))
        return guesses

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = float('inf')

    for initial in generate_initial_guesses():
        h = initial.copy()
        h = h * target_sum / np.sum(h)
        best_h_local = h.copy()
        best_c5_local = np.max(np.correlate(h, 1 - h, mode='full')) * dx

        def obj(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        def compute_gradient(x):
            h = x * target_sum / np.sum(x)
            corr = np.correlate(h, 1 - h, mode='full')
            k_max_index = np.argmax(corr)
            shift = k_max_index - (n_points - 1)
            grad = np.zeros_like(x)
            for i in range(n_points):
                i_plus = i + shift
                term1 = 1.0 if 0 <= i_plus < n_points else 0.0
                i_minus = i - shift
                term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
                grad[i] = term1 - term2
            return grad

        max_iterations = 200
        start_time = time.time()
        learning_rate = 0.3
        momentum = 0.8
        decay_rate = 0.99
        for _ in range(max_iterations):
            if time.time() - start_time > budget_s:
                break
            grad = compute_gradient(h)
            m = np.zeros_like(grad)
            v = np.zeros_like(grad)
            for i in range(n_points):
                m[i] = momentum * m[i] + (1 - momentum) * grad[i]
                v[i] = decay_rate * v[i] + (1 - decay_rate) * grad[i]**2
                biased_m = m[i] / (1 - momentum**(_ + 1))
                biased_v = v[i] / (1 - decay_rate**(_ + 1))
                h[i] = h[i] + learning_rate * biased_m / (np.sqrt(biased_v) + 1e-8)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5_local:
                best_h_local, best_c5_local = h.copy(), current_c5

        for _ in range(10):
            if time.time() - start_time > budget_s:
                break
            res = minimize(obj, best_h_local, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6, jac=compute_gradient)
            if res.success and res.fun < best_c5_local:
                best_h_local, best_c5_local = res.x, res.fun

        if best_c5_local < best_c5:
            best_h, best_c5 = best_h_local.copy(), best_c5_local

    def apply_fourier_regularization(h):
        h_fft = np.fft.fft(h)
        h_fft[1:-1] = 0
        h_reg = np.fft.ifft(h_fft).real
        h_reg = np.clip(h_reg, 0, 1)
        h_reg = h_reg * target_sum / np.sum(h_reg)
        return h_reg

    best_h = apply_fourier_regularization(best_h)
    return (np.clip(best_h, 0, 1), best_c5, n_points)
