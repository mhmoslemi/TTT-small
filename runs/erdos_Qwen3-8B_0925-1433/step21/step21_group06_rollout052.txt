import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (len(initial_h)/2) / np.sum(initial_h)
    n_points = len(initial_h)
    target_sum = n_points / 2.0
    
    def clip_normalize(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    def update_velocities(particles, velocities, pbest, gbest):
        r1, r2 = np.random.rand(2)
        velocities = 0.7 * velocities + 1.5 * r1 * (pbest - particles) + 1.5 * r2 * (gbest - particles)
        return velocities
    
    particles = np.random.rand(50, n_points)
    particles = clip_normalize(particles)
    velocities = np.random.uniform(-0.1, 0.1, size=particles.shape)
    pbest = particles.copy()
    pbest_fitness = np.full(50, np.inf)
    gbest = particles[0].copy()
    gbest_fitness = np.inf
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        for i in range(50):
            h = particles[i]
            fit = evaluate_erdos_solution(h)
            if fit < pbest_fitness[i]:
                pbest_fitness[i] = fit
                pbest[i] = h.copy()
            if fit < gbest_fitness:
                gbest_fitness = fit
                gbest = h.copy()
        velocities = update_velocities(particles, velocities, pbest, gbest)
        particles += velocities
        particles = np.array([clip_normalize(p) for p in particles])
        if gbest_fitness < best_c5:
            best_h, best_c5 = gbest.copy(), gbest_fitness
    
    def obj(x):
        return evaluate_erdos_solution(x)
    
    def constraints(x):
        return np.sum(x) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraints}
    
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = clip_normalize(res.x)
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
