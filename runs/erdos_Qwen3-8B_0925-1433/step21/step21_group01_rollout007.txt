import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    start_time = time.time()
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        h_perturb = initial_h.copy()
        noise = np.random.normal(0, 0.05, n_points)
        h_perturb += noise
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        result = minimize(objective, h_perturb, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8)
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-8)
    if result.success:
        best_h = result.x
        best_c5 = result.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
