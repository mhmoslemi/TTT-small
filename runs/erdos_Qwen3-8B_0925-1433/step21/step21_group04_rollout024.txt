import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    population = [initial_h]
    for _ in range(49):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, 'full')
        return np.max(corr) * dx

    best_h = population[0].copy()
    best_c5 = evaluate(best_h)

    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for _ in range(50):
            t1 = np.random.choice(population, 5)
            p1 = min(t1, key=evaluate)
            t2 = np.random.choice(population, 5)
            p2 = min(t2, key=evaluate)
            child = np.random.rand(n_points)
            for i in range(n_points):
                child[i] = p1[i] if np.random.rand() < 0.5 else p2[i]
            for i in range(n_points):
                child[i] += 0.1 * np.random.randn()
                child[i] = np.clip(child[i], 0, 1)
            child_sum = np.sum(child)
            scale = target_sum / child_sum
            child *= scale
            new_pop.append(child)
        for h in new_pop:
            c5 = evaluate(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        population = new_pop[:]

    best_h = np.clip(best_h, 0, 1)
    current_sum = np.sum(best_h)
    scale = target_sum / current_sum
    best_h *= scale
    return (best_h, best_c5, n_points)
