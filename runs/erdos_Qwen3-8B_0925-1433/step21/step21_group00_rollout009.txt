from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    best_h = h_init.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            if 0 <= i + k < n_points:
                grad[i] += 1 - h[i + k]
            if 0 <= i - k < n_points:
                grad[i] -= h[i - k]
        return grad

        h_init,
        1 - h_init,
        np.roll(h_init, 1),

    ]
    population = variations
