import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return h_clipped
        scaling = target_sum / sum_clipped
        return h_clipped * scaling

    def objective(x):
        h = clip_rescale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    variants = [initial_h_values.copy()]
    for _ in range(4):
        h = np.array(initial_h_values, dtype=float)
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        variants.append(h)
    for _ in range(2):
        h = np.array(initial_h_values, dtype=float)
        idxs = np.random.choice(n_points, 10, replace=False)
        h[idxs] += np.random.uniform(-0.2, 0.2, 10)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        variants.append(h)

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = objective(best_h)
    start_time = time.time()

    def local_refine(h, max_iter=200, lr=0.05):
        for _ in range(max_iter):
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                h_plus = clip_rescale(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c_plus = np.max(corr_plus) * dx
                h_minus = h.copy()
                h_minus[i] -= 1e-5
                h_minus = clip_rescale(h_minus)
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
                c_minus = np.max(corr_minus) * dx
                grad[i] = (c_plus - c_minus) / (2e-5)
            h -= lr * grad
            h = clip_rescale(h)
            new_c5 = objective(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h
            if time.time() - start_time > budget_s:
                return h
        return h

    for variant in variants:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1)] * n_points
        result = differential_evolution(

            if new_c5 < best_c5:

                best_h = h_candidate
            h_refined = local_refine(h_candidate)

            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_refined

    best_h = clip_rescale(best_h)
    return (best_h, best_c5, n_points)
