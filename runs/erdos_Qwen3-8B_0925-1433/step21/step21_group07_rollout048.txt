import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            term1 = 1 - h[i_plus] if 0 <= i_plus < n_points else 0.0
            i_minus = i - shift
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        return grad

    # Generate multiple initial guesses
    num_initial = 5
    initial_guesses = []
    for _ in range(num_initial):
        h_perturb = initial_h_values + np.random.normal(0, 0.15, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)

        initial_guesses.append(h_perturb)

    best_h = initial_h_values.copy()
    best_c5 = float('inf')

    # Global search
    for guess in initial_guesses:
        h = guess.copy()
        max_iter = 200
        lr = 0.3
        momentum = 0.8
        decay = 0.99
        start_time = time.time()
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            grad = compute_gradient(h)
            m = np.zeros_like(grad)
            v = np.zeros_like(grad)
            for i in range(n_points):
