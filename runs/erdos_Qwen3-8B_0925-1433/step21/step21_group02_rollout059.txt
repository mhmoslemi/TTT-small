import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h from initial_h_values, clip and rescale
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        h *= target_sum / sum_h
    best_h = h.copy()
    best_c5 = 0.0

    # Function to compute cross-correlation and find the worst lag
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_c5 = np.max(corr) * dx
        return max_c5, corr

    # Initial evaluation
    current_c5, corr = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()

    # Gradient descent parameters
    learning_rate = 0.05
    decay_rate = 0.995  # decay learning rate each iteration
    max_iterations = 1000
    iteration = 0
    start_time = time.time()

    while iteration < max_iterations and time.time() - start_time < budget_s:
        # Compute cross-correlation and find worst lag
        current_c5, corr = compute_c5(h)
        if current_c5 >= best_c5:
            # No improvement, maybe break early
            break
        # Find the index of the maximum cross-correlation
        k_index = np.argmax(corr)
        # Compute actual lag
        lag = k_index - (n_points - 1)
        # Compute gradient based on lag
        gradient = np.zeros_like(h)
        if lag >= 0:
            # Positive lag: shift left by lag steps
            for i in range(n_points - lag):
                j = i + lag
                gradient[i] = 1.0 - h[j]
        else:
            # Negative lag: shift right by -lag steps
            for i in range(-lag, n_points):
                j = i + lag
                gradient[i] = 1.0 - h[j]
        # Update h
        h -= learning_rate * gradient
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        # Rescale to meet the sum constraint
        sum_h = np.sum(h)
        if sum_h != 0:
            h *= target_sum / sum_h
        else:
            h = np.zeros(n_points)
        # Update best solution
        current_c5, _ = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Decay learning rate
        learning_rate *= decay_rate
        iteration += 1

    # Final evaluation
    current_c5, _ = compute_c5(best_h)
    best_c5 = min(best_c5, current_c5)

    return (best_h, best_c5, n_points)

import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        h *= target_sum / sum_h
    best_h = h.copy()
    best_c5 = 0.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_c5 = np.max(corr) * dx
        return max_c5, corr

    current_c5, corr = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()

    learning_rate = 0.05
    decay_rate = 0.995
    max_iterations = 1000
    iteration = 0
    start_time = time.time()

    while iteration < max_iterations and time.time() - start_time < budget_s:
        current_c5, corr = compute_c5(h)
        if current_c5 >= best_c5:
            break
        k_index = np.argmax(corr)
        lag = k_index - (n_points - 1)
        gradient = np.zeros_like(h)
        if lag >= 0:
            for i in range(n_points - lag):
                gradient[i] = 1.0 - h[i + lag]
        else:
            for i in range(-lag, n_points):
                gradient[i] = 1.0 - h[i + lag]
        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            h *= target_sum / sum_h
        else:
            h = np.zeros(n_points)
        current_c5, _ = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        learning_rate *= decay_rate
        iteration += 1

    current_c5, _ = compute_c5(best_h)
    best_c5 = min(best_c5, current_c5)
    return (best_h, best_c5, n_points)
