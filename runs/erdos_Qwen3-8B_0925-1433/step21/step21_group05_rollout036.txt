import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h0 = np.array(initial_h_values, dtype=float)

    h0 = h0 * target_sum / np.sum(h0)
    best_h, best_c5 = h0.copy(), np.max(np.correlate(h0, 1 - h0, mode='full')) * dx

    def obj(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    guesses = [
        0.5 * (1 + np.sin(np.linspace(0, np.pi, n_points))),
        0.5 * (1 + np.sin(2 * np.linspace(0, np.pi, n_points))),
        np.zeros(n_points), np.ones(n_points), np.random.rand(n_points)
    ]

    start_time = time.time()
    for g in guesses:
        g = np.clip(g, 0, 1) * target_sum / np.sum(np.clip(g, 0, 1))
        res = differential_evolution(
            obj, bounds=[(0,1)]*n_points, strategy='best1bin', mutation=(0.5,1),
            recombination=0.9, popsize=20, maxiter=20, tol=1e-6, seed=seed, disp=False
        )
        if res.success and res.fun < best_c5:
            best_c5, best_h = res.fun, res.x.copy()

    def slsqp(x): return obj(x)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res1 = minimize(slsqp, best_h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-6, options={'maxiter':200})
    if res1.success and res1.fun < best_c5:
        best_c5, best_h = res1.fun, res1.x
    res2 = minimize(slsqp, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-6, options={'maxiter':100})
    if res2.success and res2.fun < best_c5:
        best_c5, best_h = res2.fun, res2.x
    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(np.clip(best_h, 0, 1))
