import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def constraint(h_vec):
        return np.sum(h_vec) - target_sum

    def clip_rescale(x):
        return x.clip(0, 1) * target_sum / np.sum(x.clip(0, 1))

    def generate_initial_guesses():
        guesses = []
        for _ in range(5):
            if np.random.rand() < 0.3:
                idx = np.random.choice(n_points, size=int(0.3 * n_points), replace=False)
                h_perturbed = h.copy()
                h_perturbed[idx] = 1.0 - h_perturbed[idx]
                h_perturbed = clip_rescale(h_perturbed)
            else:
                h_perturbed = h.copy() + np.random.normal(0, 0.1, size=n_points)
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed = clip_rescale(h_perturbed)
            guesses.append(h_perturbed)
        guesses.append(np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points)))
        return guesses

    initial_guesses = generate_initial_guesses()

    start_time = time.time()
    best_de = None
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1)] * n_points
        result = differential_evolution(
            lambda x: objective(x),
            bounds,
            strategy='best1bin',
            popsize=25,
            maxiter=100,
            tol=1e-6,
            mutation=(0.8, 1.0),
            recombination=0.9,
            disp=False,
            polish=True,
            init=guess
        )
        if result.success:
            if result.fun < best_c5:
                best_c5 = result.fun
                best_de = result.x.copy()

    if best_de is not None:
        best_h = best_de
        best_h = clip_rescale(best_h)

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': constraint}],
            tol=1e-6,
            options={'maxiter': 200}
        )
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x.copy()

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': constraint}],
            tol=1e-6,
            options={'maxiter': 100}
        )
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x.copy()

    best_h = clip_rescale(best_h)
    return (best_h, best_c5, n_points)
