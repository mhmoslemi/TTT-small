import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    # Initialize population with initial_h_values and 49 random candidates
    population = [np.array(initial_h_values, dtype=float)]
    for _ in range(49):
        h = np.random.uniform(0, 1, n_points)
        h = h * target_sum / np.sum(h)
        population.append(h.copy())
    
    # Helper to compute c5
    def compute_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    # Track best solution
    best_h = population[0]
    best_c5 = compute_c5(best_h)
    
    tournament_size = 5
    mutation_rate = 0.1
    mutation_scale = 0.15
    max_generations = 200
    start_time = time.time()
    
    for _ in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        # Generate new candidates
        for _ in range(50):
            # Tournament selection for two parents
            parents = []
            for _ in range(2):
                tournament = np.random.choice(population, tournament_size, replace=False)
                scores = [compute_c5(h) for h in tournament]
                best_idx = np.argmin(scores)
                parents.append(tournament[best_idx])
            # Crossover
            child = np.where(np.random.rand(n_points) < 0.5, parents[0], parents[1])
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_scale, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                current_sum = np.sum(child)
                child = child * target_sum / current_sum
            # Clip and scale to ensure constraints
            child = np.clip(child, 0, 1)
            child_sum = np.sum(child)
            child = child * target_sum / child_sum
            new_population.append(child)
        # Select top 50 from new_population based on c5
        new_population.sort(key=compute_c5)
        population = new_population[:50]
        # Update best solution
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    # Final check and scaling
    best_h = np.clip(best_h, 0, 1)
    best_sum = np.sum(best_h)
    best_h = best_h * target_sum / best_sum
    return (best_h, best_c5, n_points)
