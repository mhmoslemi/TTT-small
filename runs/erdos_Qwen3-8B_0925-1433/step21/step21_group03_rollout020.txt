import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_rescale(x):
        x_clipped = np.clip(x, 0, 1)
        sum_clipped = np.sum(x_clipped)
        if sum_clipped == 0:
            return x_clipped
        scaling = target_sum / sum_clipped
        return x_clipped * scaling

    def objective(x):
        h = clip_and_rescale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    variants = []
    h = initial_h_values.copy()
    variants.append(h)
    h_sym = np.concatenate([h[::-1], h[1:-1]])  # Symmetric variant
    variants.append(clip_and_rescale(h_sym))
    h_spaced = np.zeros(n_points)
    for i in range(n_points):
        h_spaced[i] = 1 if (i % 3 == 0) else 0
    variants.append(clip_and_rescale(h_spaced))
    h_noise = initial_h_values.copy() + np.random.normal(0, 0.1, n_points)
    variants.append(clip_and_rescale(h_noise))
    h_sym_noise = np.concatenate([h[::-1], h[1:-1]]) + np.random.normal(0, 0.1, n_points)
    variants.append(clip_and_rescale(h_sym_noise))

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    for variant in variants:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1)] * n_points
        result = differential_evolution(
            objective, bounds, strategy='best1bin', popsize=40, mutation=(0.8, 1.2),
            recombination=0.9, tol=1e-6, maxiter=100, seed=seed, polish=False
        )
        if result.success:
            current_h = clip_and_rescale(result.x)
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

    best_h = clip_and_rescale(best_h)
    return (best_h, best_c5, n_points)
