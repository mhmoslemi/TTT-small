from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    best_h = initial_guess.copy()

    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraints(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraints}

    def generate_initial_guesses(num, base):

            noise = np.random.normal(0, 0.05, n_points)

            flipped = normalize(flipped)
            guesses.append(flipped)
        for _ in range(num // 2):
            uniform = np.random.rand(n_points)
            uniform = normalize(uniform)
            guesses.append(uniform)
        return guesses

        if res.success:
