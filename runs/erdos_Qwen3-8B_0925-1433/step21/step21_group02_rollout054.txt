import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    start_time = time.time()
    eta = 0.01
    decay_rate = 0.995
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        k_index = np.argmax(corr)
        m = k_index - (n_points - 1)
        
        grad = np.zeros(n_points)
        for j in range(n_points):
            j_plus_m = j + m
            j_minus_m = j - m
            term1 = 0.0
            if 0 <= j_plus_m < n_points:
                term1 = 1.0 - h[j_plus_m]
            term2 = 0.0
            if 0 <= j_minus_m < n_points:
                term2 = h[j_minus_m]
            grad[j] = (term1 - term2) * dx
        
        h -= eta * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.zeros(n_points)
        else:
            h = h * target_sum / h_sum
        
        new_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(new_corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        eta *= decay_rate
    
    return (best_h, best_c5, n_points)
