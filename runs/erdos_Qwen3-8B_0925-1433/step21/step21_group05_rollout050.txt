import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    def bounds_func(_):
        return [(0, 1)] * n_points

    def generate_sine(n):
        h = 0.5 + 0.5 * np.sin(np.pi * np.linspace(0, n-1, n) / n)
        h = np.clip(h, 0, 1)
        return h * target_sum / np.sum(h)

    def generate_block(n):
        w = n // 4
        h = np.zeros(n)
        for i in range(0, n, w*2):
            h[i:i+w] = 1.0
        h = h * target_sum / np.sum(h)
        return h

    def generate_lattice(n):
        h = np.zeros(n)
        for i in range(n):
            if (i % 2 == 0) or (i % 3 == 0):
                h[i] = 1.0
        h = h * target_sum / np.sum(h)
        return h

    initial_guesses = [initial_h.copy() * target_sum / np.sum(initial_h)]
    initial_guesses.append(generate_sine(n_points))
    initial_guesses.append(generate_block(n_points))
    initial_guesses.append(generate_lattice(n_points))
    initial_guesses.append(np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points)))

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = differential_evolution(
            objective,
            bounds_func,
            args=(),
            strategy='best1bin',
            popsize=20,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.9,
            maxiter=100,
            constraints=[{'type': 'eq', 'fun': constraint_sum}]
        )
        if res.success:
            current_h = res.x.copy()
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds_func,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            tol=1e-6,
            options={'maxiter': 100}
        )
        if res.success:
            current_h = res.x.copy()
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
