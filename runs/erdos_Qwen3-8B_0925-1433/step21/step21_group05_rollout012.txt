import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def create_sine_guess(n):
        i = np.arange(n)
        return np.clip(0.5 * (1 + np.sin(np.pi * i / n)), 0, 1) * (target_sum / np.sum(0.5 * (1 + np.sin(np.pi * i / n))))

    def create_lattice_guess(n):
        i = np.arange(n)
        shift = np.random.rand()
        return np.clip(0.5 * (1 + np.cos(2 * np.pi * (i + shift) / n)), 0, 1) * (target_sum / np.sum(0.5 * (1 + np.cos(2 * np.pi * (i + shift) / n))))

    def create_block_guess(n):
        block_size = n // 4
        h = np.zeros(n)
        for i in range(0, n, block_size * 2):
            h[i:i+block_size] = 1.0
        return h * (target_sum / np.sum(h))

    def create_random_guess(n):
        h = np.random.rand(n)
        return h * (target_sum / np.sum(h))

    def objective(h_vec):
        h_clipped = np.clip(h_vec, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.inf
        h_rescaled = h_clipped * (target_sum / sum_clipped)
        corr = np.correlate(h_rescaled, 1 - h_rescaled, mode='full')
        return np.max(corr) * dx

    initial_guesses = [
        np.array(initial_h_values, dtype=float),
        create_sine_guess(n_points),
        create_lattice_guess(n_points),
        create_block_guess(n_points),
        create_random_guess(n_points)
    ]

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.inf
    start_time = time.time()

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = differential_evolution(
            objective,
            bounds=[(0, 1)] * n_points,
            strategy='best1bin',
            popsize=25,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.9,
            maxiter=100,
            polish=True,
            disp=False

            h_candidate = result.x
            h_clipped = np.clip(h_candidate, 0, 1)
            sum_clipped = np.sum(h_clipped)
            if sum_clipped > 0:
                h_candidate = h_clipped * (target_sum / sum_clipped)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_candidate.copy()

    if best_h is not None:
        res_slsqp = minimize(objective, best_h, method='SLSQP',

        if res_slsqp.success:
            best_h = res_slsqp.x
            best_c5 = res_slsqp.fun

        res_lbfgs = minimize(objective, best_h, method='L-BFGS-B',
