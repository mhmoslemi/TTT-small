from scipy.optimize import minimize
import time

            noise = np.random.normal(0, 0.1, n_points)

            h = scale_h(h)
            candidates.append(h)
        return candidates

    initial_candidates = generate_initial_candidates(initial_h_values)
    best_h = np.copy(initial_h_values)

    def compute_c5(h):
