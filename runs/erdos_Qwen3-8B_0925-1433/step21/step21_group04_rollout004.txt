def run(seed=42, budget_s=600):

        h = np.clip(h, 0, 1)

        h *= scale

        new_c5 = new_cor.max() * dx
        if new_c5 < best_c5 - improvement_threshold:

                scale = target_sum / sum_h
                h *= scale
                no_improvement_steps = 0

    # Final check
    final_cor = np.correlate(h, 1 - h, mode='full')
    final_c5 = final_cor.max() * dx
    if final_c5 < best_c5:
        best_c5 = final_c5
