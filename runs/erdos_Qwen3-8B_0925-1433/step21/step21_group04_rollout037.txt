            best_c5 = new_c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1

        if stagnation_count >= max_stagnation:
            perturb = np.random.normal(0, 0.05, size=n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            scale_factor = target_sum / current_sum
            h *= scale_factor
            stagnation_count = 0

    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    scale_factor = target_sum / current_sum
    h *= scale_factor
    return (h, best_c5, n_points)
