import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Validate and normalize initial_h_values

    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    best_h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    # PSO parameters
    n_particles = 50
    omega = 0.7
    c1, c2 = 1.5, 1.5
    max_iter = int(budget_s * 0.66)
    local_refine_iters = 5

    # Initialize particles
    particles = np.random.uniform(0, 1, (n_particles, n_points))
    particles = np.array([p * target_sum / np.sum(p) for p in particles])
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    pbest = particles.copy()
    pbest_fitness = np.full(n_particles, np.inf)
    gbest = particles[0].copy()
    gbest_fitness = np.inf

    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = np.array([evaluate_erdos_solution(p) for p in particles])
        # Update personal best
        for i in range(n_particles):
            if fitness[i] < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fitness[i]
        # Update global best
        if np.min(fitness) < gbest_fitness:
            gbest = particles[np.argmin(fitness)].copy()
            gbest_fitness = np.min(fitness)
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2, n_points)
            velocities[i] = omega * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
        # Early check for best
        if gbest_fitness < best_c5:
            best_h, best_c5 = gbest.copy(), gbest_fitness

    # Local refinement
    for _ in range(local_refine_iters):
        if time.time() - start_time > budget_s:
            break
        res = minimize(lambda x: evaluate_erdos_solution(x), best_h, method='SLSQP',
                       bounds=[(0, 1) for _ in range(n_points)],
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5
