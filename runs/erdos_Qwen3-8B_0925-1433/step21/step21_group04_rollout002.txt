    dx = 2.0 / n_points

    sum_h = np.sum(h)
    scale = target_sum / sum_h
    h *= scale
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    # Iterative refinement
    start_time = time.time()
    for _ in range(1000):  # Arbitrary number of iterations
        if time.time() - start_time > budget_s:
            break
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        index_max = np.argmax(corr)
        lag = index_max - (n_points - 1)
        # Find the best i for this lag
        max_contribution = -1
        best_i = -1
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                contribution = h[i] * (1 - h[j])
                if contribution > max_contribution:
                    max_contribution = contribution
                    best_i = i

        j = best_i + lag
        a = h[best_i]
        b = h[j]
        delta = min(0.1 * a, 0.1 * (1 - b))
        h[best_i] -= delta
        h[j] += delta
        # Clip to [0, 1] (though delta ensures this)
        h = np.clip(h, 0, 1)
        # Recompute C5

    h = np.clip(best_h, 0, 1)
    sum_h = np.sum(h)

    return (h, best_c5, n_points)

import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    h = np.array(initial_h_values, dtype=float)
    sum_h = np.sum(h)
    scale = target_sum / sum_h
    h *= scale
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        index_max = np.argmax(corr)
        lag = index_max - (n_points - 1)
        max_contribution = -1
        best_i = -1
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                contribution = h[i] * (1 - h[j])
                if contribution > max_contribution:
                    max_contribution = contribution
                    best_i = i
        if best_i == -1:
            continue
        j = best_i + lag
        a, b = h[best_i], h[j]
        delta = min(0.1 * a, 0.1 * (1 - b))
        h[best_i] -= delta
