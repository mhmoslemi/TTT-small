import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    h0 = initial_h_values
    n_point = len(h0)
    dx = 2.0 / n_point
    target_sum = n_point / 2.0

    def clip_and_rescale(x):
        x_c = np.clip(x, 0, 1)
        s = np.sum(x_c)
        if s == 0:
            return x_c
        return x_c * target_sum / s

    def obj(x):
        h = clip_and_rescale(x)
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    var = [h0.copy()]
    for _ in range(4):
        h = h0.copy()
        h += np.random.normal(0, 0.1, n_point)
        var.append(clip_and_rescale(h))
        h = h0.copy()
        high = np.where(h > 0.5)[0]
        if high.size:
            r = np.random.uniform(0.1, 0.2, len(high))
            for i, idx in enumerate(high):
                h[idx] *= (1 - r[i])
                if idx > 0:
                    h[idx - 1] *= (1 + r[i] * 0.5)
                if idx < n_point - 1:
                    h[idx + 1] *= (1 + r[i] * 0.5)
        var.append(clip_and_rescale(h))
        h = h0.copy()
        idx = np.random.choice(n_point, int(n_point / 2), replace=False)
        h[idx] = np.random.permutation(h[idx])
        var.append(clip_and_rescale(h))
        h = h0.copy()
        h[:n_point // 2] = h[n_point // 2:][::-1]
        var.append(clip_and_rescale(h))

    best_h = h0.copy()
    best_val = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    t0 = time.time()
    for x in var:
        if time.time() - t0 > budget_s:
            break
        res = differential_evolution(obj, [(0, 1)] * n_point, strategy='best1bin', popsize=25,
                                    mutation=(0.5, 0.8), recombination=0.9, tol=1e-6,
                                    maxiter=150, seed=seed)
        if res.success:
            h = clip_and_rescale(res.x)
            val = res.fun
            if val < best_val:
                best_val = val
                best_h = h

    def refine(h):
        for _ in range(100):
            grad = np.zeros(n_point)
            for i in range(n_point):
                h_dup = h.copy()
                h_dup[i] += 1e-4
                h_dup = clip_and_rescale(h_dup)
                v1 = obj(h_dup)
                h_dup = h.copy()
                h_dup[i] -= 1e-4
                h_dup = clip_and_rescale(h_dup)
                v2 = obj(h_dup)
                grad[i] = (v1 - v2) / 2e-4
            h += -0.1 * grad
            h = clip_and_rescale(h)
        return h

    final_h = refine(best_h)
    return (final_h, best_val, n_point)
