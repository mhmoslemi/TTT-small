            h = h * target_sum / total
        return h

    h = np.array(initial_h_values, dtype=float)
    h = rescale_h(h)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    learning_rate = 0.01
    decay_rate = 0.98
    max_iterations = 1000
    iterations = 0
    start_time = time.time()

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)

        grad = np.zeros(n_points)
        for j in range(n_points):
            if 0 <= j + k < n_points:
                grad[j] += 1.0 - h[j + k]
            if 0 <= j - k < n_points:
                grad[j] -= h[j - k]

        h = h - learning_rate * grad
        h = rescale_h(h)

        current_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(current_corr) * dx

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        learning_rate *= decay_rate
        iterations += 1
        if iterations > max_iterations:
            break

    return (best_h, best_c5, n_points)
