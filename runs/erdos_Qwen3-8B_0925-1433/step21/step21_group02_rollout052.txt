import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_rescale(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return h * target_sum / s

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_h = np.array(initial_h_values, dtype=float)
    population = [initial_h.copy()]
    for _ in range(20):
        h = np.random.rand(n_points)
        population.append(clip_rescale(h))

    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()

    while time.time() - start_time < budget_s:
        ranked = []
        for h in population:
            c5 = compute_c5(h)
            ranked.append((c5, h))
        ranked.sort()
        best_c5, best_h = ranked[0]

        top_indices = [i for i, (c, _) in enumerate(ranked) if i < len(ranked)//10]
        top_candidates = [ranked[i][1] for i in top_indices]
        for h in top_candidates:
            h = h.copy()
            eps = 1e-4
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = clip_rescale(h_plus)
                c5_plus = compute_c5(h_plus)
                h_minus = h.copy()
                h_minus[i] -= eps

                c5_minus = compute_c5(h_minus)
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            eta = 0.1 * (1 - (best_c5 - ranked[0][0]) / best_c5)
