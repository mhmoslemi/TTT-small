import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_rescale(x):
        x_clipped = np.clip(x, 0, 1)
        sum_clipped = np.sum(x_clipped)
        if sum_clipped == 0:
            return x_clipped
        scaling = target_sum / sum_clipped
        return x_clipped * scaling

    def objective(x):
        h = clip_and_rescale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate initial variants
    variants = [initial_h_values.copy()]
    h_sym = np.array(initial_h_values, dtype=float)
    for i in range(n_points):
        h_sym[i] = 1 - h_sym[n_points - 1 - i]
    variants.append(h_sym)
    h_spaced = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            h_spaced[i] = 1
    variants.append(clip_and_rescale(h_spaced))
    h_rand = np.zeros(n_points)
    num_ones = int(target_sum)
    indices = np.random.choice(n_points, num_ones, replace=False)
    h_rand[indices] = 1
    variants.append(clip_and_rescale(h_rand))
    h_noise = np.array(initial_h_values, dtype=float)
    noise = np.random.normal(0, 0.1, n_points)
    h_noise += noise
    variants.append(clip_and_rescale(h_noise))
    h_sym_noise = h_sym.copy()
    noise = np.random.normal(0, 0.1, n_points)
    h_sym_noise += noise
    variants.append(clip_and_rescale(h_sym_noise))

    best_h = initial_h_values.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    for variant in variants:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1)] * n_points
        result = differential_evolution(objective, bounds, strategy='best1bin', popsize=30, mutation=(0.8, 1.2), recombination=0.9, tol=1e-6, maxiter=100, seed=seed)
        if result.success:
            current_h = clip_and_rescale(result.x)
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

    # Local refinement
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        h = best_h.copy()
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = clip_and_rescale(h_plus)
            c_plus = objective(h_plus)
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = clip_and_rescale(h_minus)
            c_minus = objective(h_minus)
            grad[i] = (c_plus - c_minus) / (2 * eps)
        h_update = np.clip(h - 0.1 * grad, 0, 1)
        h_update = clip_and_rescale(h_update)
        c_update = objective(h_update)
        if c_update < best_c5:
            best_h = h_update
            best_c5 = c_update
        else:
            break

    return (best_h, best_c5, n_points)
