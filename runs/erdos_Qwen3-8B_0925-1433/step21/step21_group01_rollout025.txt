                no_improvement = 0
            else:
                no_improvement += 1
            if no_improvement >= 5:
                break

    # Final refinement with SLSQP
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1)] * n_points
    def obj(x):
        h = x * target_sum / np.sum(x) if np.sum(x) != 0 else x
        return evaluate(h)
    
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    final_h = res.x if res.success else best_h
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * target_sum / np.sum(final_h)
    return (final_h, best_c5, n_points)
