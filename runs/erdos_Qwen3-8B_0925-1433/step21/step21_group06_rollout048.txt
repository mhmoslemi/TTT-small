import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) != 0 else x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    start_time = time.time()

    # DE for global search
    result = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=30, maxiter=100,
        tol=1e-6, polish=True, x0=np.array(initial_h_values),
        mutation=(0.5, 1.0), recombination=0.9, disp=False
    )
    best_h = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time

    # SLSQP refinement
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds_slsqp = [(0, 1) for _ in range(n_points)]

    def obj_slsqp(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) != 0 else x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj_slsqp, best_h, method='SLSQP',
                       bounds=bounds_slsqp, constraints=constraints, tol=1e-6)
        if res.success:
            best_h = res.x
            best_c5 = res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
