import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = [(0.0, 1.0)] * n_points
    constraints = [{'type': 'eq', 'fun': constraint_sum}]

    # Generate initial candidates
    initial_candidates = [h_initial]

        perturbed = h_initial + noise

        perturbed = perturbed * target_sum / np.sum(perturbed)
        initial_candidates.append(perturbed)

    best_c5 = float('inf')
    best_h = h_initial

            fun=objective,
            x0=h_pert,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 150, 'ftol': 1e-8, 'eps': 1e-8}
        )
