import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_rescale(x):
        x_clipped = np.clip(x, 0, 1)
        sum_clipped = np.sum(x_clipped)
        scaling = target_sum / sum_clipped if sum_clipped > 0 else 1.0
        return x_clipped * scaling

    def objective(x):
        h = clip_and_rescale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    variants = [initial_h_values.copy()]
    for _ in range(4):
        h = initial_h_values.copy()
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = clip_and_rescale(h)
        variants.append(h)
        h = h[::-1]
        h = clip_and_rescale(h)
        variants.append(h)

    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)

    start_time = time.time()
    for variant in variants:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1)] * n_points
        result = differential_evolution(objective, bounds, strategy='best1bin', popsize=30, mutation=(0.8, 1.2), recombination=0.9, tol=1e-6, maxiter=50, seed=seed)
        if result.success:
            current_h = clip_and_rescale(result.x)
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        worst_lag = np.argmax(corr)
        perturb = np.zeros(n_points)
        perturb[np.random.randint(n_points)] = 0.1
        h_perturbed = clip_and_rescale(best_h + perturb)
        new_c5 = objective(h_perturbed)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_perturbed
        else:
            break

    return (best_h, best_c5, n_points)
