import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(h_vec):
        h_clipped = np.clip(h_vec, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h_vec):
        h_clipped = np.clip(h_vec, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        k_max = np.argmax(corr) - (n_points - 1)
        grad = np.zeros_like(h_vec)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                grad[i] = 1.0 - h_scaled[j]
        return grad

    initial_guesses = []
    for _ in range(5):
        if np.random.rand() < 0.3:
            idx = np.random.choice(n_points, size=int(0.3 * n_points), replace=False)
            h_perturbed = h.copy()
            h_perturbed[idx] = 1.0 - h_perturbed[idx]
            h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
        else:
            h_perturbed = h.copy() + np.random.normal(0, 0.1, size=n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
        initial_guesses.append(h_perturbed)

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1) for _ in range(n_points)]
        res_de = differential_evolution(
            objective, bounds, strategy='best1bin', popsize=20, mutation=(0.5, 1), 
            recombination=0.9, tol=1e-6, maxiter=100, seed=seed
        )
        if res_de.success:
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            res_slsqp = minimize(
                objective, res_de.x, method='SLSQP', bounds=bounds, constraints=cons,
                jac=compute_gradient, tol=1e-6, options={'maxiter': 200}
            )
            if res_slsqp.success:
                res_lbfgs = minimize(
                    objective, res_slsqp.x, method='L-BFGS-B', bounds=bounds,
                    jac=compute_gradient, tol=1e-6, options={'maxiter': 100}
                )
                if res_lbfgs.success and res_lbfgs.fun < best_c5:
                    best_h = res_lbfgs.x.copy()
                    best_c5 = res_lbfgs.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
