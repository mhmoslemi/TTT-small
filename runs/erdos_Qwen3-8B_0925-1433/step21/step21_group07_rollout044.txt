import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def create_variants(base):
        base = np.array(base, dtype=float)
        base = base * target_sum / np.sum(base)
        variants = [base]
        np.random.seed(seed + 1)
        for _ in range(3):
            noise = np.random.uniform(-0.25, 0.25, n_points)
            variants.append(base + noise)
            variants.append(base[::-1] + noise)
            variants.append(base + 0.05 * np.sin(2 * np.pi * np.arange(n_points) / n_points))
        return variants

    variants = create_variants(initial_h_values)
    best_h, best_c5 = variants[0].copy(), np.max(np.correlate(variants[0], 1 - variants[0], mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + k
            i_plus %= n_points
            term1 = 1.0 - h[i_plus]
            i_minus = i - k
            i_minus %= n_points
            term2 = h[i_minus]
            grad[i] = term1 - term2
        return grad

    for h in variants:
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        for _ in range(200):
            if time.time() - time.time() > budget_s:
                break
            grad_val = grad(h)
            h += 0.2 * grad_val / (np.sqrt(np.sum(grad_val**2)) + 1e-8)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            curr_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if curr_c5 < best_c5:
                best_h, best_c5 = h.copy(), curr_c5

    for _ in range(5):
        if time.time() - time.time() > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints={'type':'eq','fun':lambda x: np.sum(x)-target_sum}, jac=grad, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
