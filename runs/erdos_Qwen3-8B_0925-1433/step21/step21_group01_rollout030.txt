from scipy.optimize import minimize
import time
from scipy.ndimage import gaussian_filter1d

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        variants.append(h_noise)

    for _ in range(2):
        smoothed = gaussian_filter1d(initial_h, sigma=1.0)
        smoothed = np.clip(smoothed, 0, 1)
        smoothed = smoothed * target_sum / np.sum(smoothed)

        variants.append(h_freq)

    variants = np.array(variants)
