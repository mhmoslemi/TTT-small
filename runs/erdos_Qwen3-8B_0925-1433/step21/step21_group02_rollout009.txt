from scipy.optimize import minimize
import time

        s = np.sum(h)
        if s == 0:
            return np.inf
        h_scaled = h * target_sum / s
        corr = np.correlate(h_scaled, 1.0 - h_scaled, mode='full')
        return np.max(corr) * dx

    def repair(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return h * target_sum / s

    pop_size = 50

        if i < 20:
            noise = np.random.normal(0, 0.1, n_points)
            pop[i] += noise
        else:
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
        pop[i] = repair(pop[i])

    def tournament_select(pop, k=5):
        idxs = np.random.choice(len(pop), k, replace=False)
        return pop[np.argmin([eval_candidate(p) for p in pop[idxs]])]

    def crossover(parent1, parent2):
        if np.random.rand() < 0.8:
            pt = np.random.randint(1, n_points)
            child = np.concatenate([parent1[:pt], parent2[pt:]])

            child = np.random.rand(n_points)
            child = child * target_sum / np.sum(child)
        return repair(child)

    def mutate(h):
