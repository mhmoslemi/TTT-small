    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)

        for i in range(n_points):
            pos = (i + k) % n_points
            neg = (i - k) % n_points
            grad[i] = (1 - h[pos]) - h[neg]
        return grad

    max_iterations = 200
    start_time = time.time()
