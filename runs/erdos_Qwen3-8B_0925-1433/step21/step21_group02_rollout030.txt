    # Initialize h
    h = np.array(initial_h_values, dtype=float)

        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode='full')
        k_index = np.argmax(corr)
        lag_shift = k_index - (n_points - 1)
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + lag_shift
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) - h[i]
            else:
                grad[i] = 0
        # Update h
        h += -eta * grad
        # Clip and rescale
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        # Compute current_c5

        # Update best
        if current_c5 < best_c5:

            best_h = np.copy(h)
        # Decay eta

        iteration += 1
        # Early stopping

            break

    # Final check
    h = np.clip(best_h, 0, 1)
    h = h * target_sum / np.sum(h)
    return (h, best_c5, n_points)
