import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_rescale(x):
        x_clipped = np.clip(x, 0, 1)
        if np.sum(x_clipped) == 0:
            return x_clipped
        scaling = target_sum / np.sum(x_clipped)
        return x_clipped * scaling

    def objective(x):
        h = clip_rescale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    variants = [np.array(initial_h_values, dtype=float)]
    # Symmetric variant
    variants.append(initial_h_values[::-1])
    # Block pattern with spacing
    block_width = 0.3
    spacing = 0.6
    block = np.zeros(n_points)
    for i in range(n_points):
        pos = i * dx
        if (pos >= block_width and pos < block_width + spacing):
            block[i] = 1
        elif (pos >= 2 - block_width - spacing and pos < 2 - block_width):
            block[i] = 1
    variants.append(block)
    # Random noise variant
    noise = np.random.normal(0, 0.1, n_points)
    variants.append(clip_rescale(initial_h_values + noise))
    # Symmetric + noise
    variants.append(clip_rescale(initial_h_values[::-1] + np.random.normal(0, 0.1, n_points)))

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    for variant in variants:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1)] * n_points
        result = differential_evolution(
            objective, bounds, strategy='best1bin',
            popsize=40, mutation=(0.8, 1.2), recombination=0.9,
            tol=1e-6, maxiter=200, seed=seed)
        if result.success:
            current_h = clip_rescale(result.x)
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

    best_h = clip_rescale(best_h)
    return (best_h, best_c5, n_points)
