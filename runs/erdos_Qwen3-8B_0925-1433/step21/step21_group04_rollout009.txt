            p2 = population[p2_idx].copy()
            if np.random.rand() < crossover_rate:
                offspring = np.random.choice([p1, p2], size=n_points)
            else:
                offspring = p1.copy()
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                offspring += noise
                offspring = np.clip(offspring, 0, 1)
            current_sum = np.sum(offspring)
            scale = target_sum / current_sum
            offspring *= scale
            new_population.append(offspring)
        population = new_population

    current_sum = np.sum(best_h)
    scale = target_sum / current_sum
    best_h *= scale
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h, best_c5, n_points)
