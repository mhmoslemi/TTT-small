def run(seed=42, budget_s=600):

        variants.append(h)

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = objective(best_h)
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    for variant in variants:
        if time.time() - start_time > budget_s:

        result = differential_evolution(
