    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    initial_guesses.append(h_initial)

    h_sine = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
    h_sine = np.clip(h_sine, 0, 1)
    h_sine = h_sine * target_sum / np.sum(h_sine)
    initial_guesses.append(h_sine)

    h_alt = np.zeros(n_points)
    h_alt[::2] = 1.0
    h_alt = h_alt * target_sum / np.sum(h_alt)
    initial_guesses.append(h_alt)

    h_block = np.zeros(n_points)
    h_block[:n_points//2] = 1.0
    h_block = h_block * target_sum / np.sum(h_block)
    initial_guesses.append(h_block)

    h_rand = np.random.rand(n_points)
    h_rand = h_rand * target_sum / np.sum(h_rand)
    initial_guesses.append(h_rand)

    best_h, best_c5 = process_guess(h_initial)
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = differential_evolution(objective, [(0, 1)]*n_points, 
                                       strategy='best1bin', maxiter=100,
                                       popsize=20, tol=1e-6, mutation=(0.5, 1),
                                       recombination=0.9, seed=seed)
        h_de, c5 = process_guess(result.x)
        if c5 < best_c5:
            best_h, best_c5 = h_de, c5

    def refine(x):
        res = minimize(objective, x, method='SLSQP', bounds=[(0,1)]*n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       jac=compute_gradient, tol=1e-6, options={'maxiter': 100})
        return res.x if res.success else x

    best_h = refine(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
