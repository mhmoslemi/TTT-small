import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)

    def obj(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def gen_sine():
        return 0.5 * (1 + np.sin(np.pi * np.arange(n_points) / n_points))

    def gen_lattice():
        return np.array([(i % 2) * 0.5 for i in range(n_points)])

    def gen_block():
        return np.array([(i % 10 < 5) for i in range(n_points)]).astype(float)

    guesses = [h, gen_sine(), gen_lattice(), gen_block(), np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))]

    start = time.time()
    best_h = h
    best_c5 = obj(h)
    for g in guesses:
        if time.time() - start > budget_s:
            break
        res = differential_evolution(obj, bounds=[(0,1)]*n_points, popsize=15, strategy='best1bin', maxiter=50, tol=1e-6)
        if res.success:
            h_cand = res.x
            c5_cand = obj(h_cand)
            if c5_cand < best_c5:
                best_h = h_cand
                best_c5 = c5_cand

    def obj_slsqp(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res_slsqp = minimize(obj_slsqp, best_h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, options={'maxiter': 200})
    if res_slsqp.success:
        best_h = res_slsqp.x
        best_c5 = obj_slsqp(best_h)

    res_lbgfs = minimize(obj_slsqp, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=cons, options={'maxiter': 100})
    if res_lbgfs.success:
        best_h = res_lbgfs.x
        best_c5 = obj_slsqp(best_h)

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
