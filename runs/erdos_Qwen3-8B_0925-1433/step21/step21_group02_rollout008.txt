import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    population = []

        h += noise

        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.full(n_points, 0.5)
        population.append(h)
    for _ in range(30):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.full(n_points, 0.5)
        population.append(h)
    population = np.array(population, dtype=float)

    def eval_candidate(h):
        h_clipped = np.clip(h, 0, 1)
        S = np.sum(h_clipped)
        if S == 0:
            return np.inf
        h_scaled = h_clipped * target_sum / S
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    best_h = initial_h_values.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()
    for gen in range(150):
        if time.time() - start_time > budget_s:
            break
        # Tournament selection
        parents = []
        for _ in range(20):

            scores = [eval_candidate(c) for c in candidates]

            parents.append(candidates[best_idx])
        # Crossover
        offspring = []

            child1 = np.concatenate((p1[:split], p2[split:]))

            offspring.append(child1)

        # Mutation
        for child in offspring:
            for j in range(n_points):
