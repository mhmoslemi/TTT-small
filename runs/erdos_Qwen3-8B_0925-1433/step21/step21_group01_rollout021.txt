import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate initial population with Fourier-smoothed variation
    h_initial = np.array(initial_h_values, dtype=float)
    fft_h = fft(h_initial)
    cutoff = int(0.2 * n_points)
    fft_h[cutoff:] = 0
    fft_h[-cutoff:] = 0
    h_smoothed = np.fft.ifft(fft_h).real
    h_smoothed = np.clip(h_smoothed, 0, 1)
    h_smoothed = h_smoothed * target_sum / np.sum(h_smoothed)
    population = [h_initial, h_smoothed]
    
    for _ in range(4):
        h = h_smoothed.copy()
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    population = np.array(population)

    best_h = h_initial.copy()
    best_c5 = evaluate(best_h)
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()
    pop_size = 20
    elite_count = 5
    generations = 200
    mutation_std = 0.05
    crossover_prob = 0.8
    no_improvement = 0

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        evals = np.array([evaluate(h) for h in population])
        sorted_indices = np.argsort(evals)
        elites = population[sorted_indices[:elite_count]]
        
        new_population = []
        for _ in range(pop_size):
            if np.random.rand() < crossover_prob:
                p1, p2 = elites[np.random.choice(elite_count, 2, replace=False)]
                idx = np.random.randint(1, n_points-1)
                child = np.concatenate([p1[:idx], p2[idx:]])
            else:
                child = elites[np.random.choice(elite_count)].copy()
            noise = np.random.normal(0, mutation_std, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = np.array(new_population)

        for h in population:
            current_c5 = evaluate(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                no_improvement = 0
            else:
                no_improvement += 1
            if no_improvement >= 5:
                break

    # Final refinement with L-BFGS-B
    def obj(x):
        h = x * target_sum / np.sum(x)
        return evaluate(h)
    
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                   constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
    final_h = res.x if res.success else best_h
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * target_sum / np.sum(final_h)
