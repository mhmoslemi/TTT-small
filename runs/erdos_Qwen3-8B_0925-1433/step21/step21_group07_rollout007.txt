import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in locals():
        n_points = len(initial_h_values)
        h_initial = np.array(initial_h_values)
    else:
        n_points = 45
        h_initial = np.random.rand(n_points)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = h_initial * target_sum / np.sum(h_initial)

    def compute_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_gradient(h, k_max):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_idx = np.argmax(corr)
        shift = k_max_idx - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            i_plus = i + shift
            if 0 <= i_plus < n_points:
                term1 = 1 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - shift
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    def fourier_regularization(h):
        fft_h = np.fft.fft(h)
        cutoff = int(n_points / 4)
        fft_h[cutoff:] = 0
        h_filtered = np.fft.ifft(fft_h).real
        h_filtered = np.clip(h_filtered, 0, 1)
        h_filtered = h_filtered * target_sum / np.sum(h_filtered)
        return h_filtered

    def objective(x):
        h = x * target_sum / np.sum(x)
        return compute_c5(h)

    def grad_objective(x):
        h = x * target_sum / np.sum(x)
        grad = compute_gradient(h, np.argmax(np.correlate(h, 1 - h, mode='full')))
        return grad

    def optimize_guess(h_guess):
        h = h_guess.copy()
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            grad = compute_gradient(h, np.argmax(np.correlate(h, 1 - h, mode='full')))
            h -= 0.3 * grad
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
        h = fourier_regularization(h)
        return h, compute_c5(h)

    start_time = time.time()
    initial_guesses = [
        h_initial,
        np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points)),
        0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n_points) / n_points)
    ]
    initial_guesses = [g * target_sum / np.sum(g) for g in initial_guesses]

    best_h = initial_guesses[0]
    best_c5 = compute_c5(best_h)

    for guess in initial_guesses[1:]:
        h, c5 = optimize_guess(guess)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    h = best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, h, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                       jac=grad_objective, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun
        h = best_h

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
