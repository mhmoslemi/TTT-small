import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_rescale(x):
        x_clipped = np.clip(x, 0, 1)
        sum_clipped = np.sum(x_clipped)
        if sum_clipped == 0:
            return x_clipped
        scaling = target_sum / sum_clipped
        scaled = x_clipped * scaling
        return np.clip(scaled, 0, 1)

    def objective(x):
        h = clip_and_rescale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    variants = [initial_h_values.copy(), initial_h_values[::-1].copy()]
    size = int(n_points / 2) + 1
    spaced = np.zeros(n_points)
    idx = np.random.choice(n_points, size=size, replace=False)
    spaced[idx] = 1.0
    variants.append(spaced)
    rand = np.random.rand(n_points)
    variants.append(rand)

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = objective(best_h)

    start_time = time.time()
    for variant in variants:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1)] * n_points
        res = differential_evolution(objective, bounds, strategy='best1bin', popsize=30, mutation=(0.8, 1.2), recombination=0.9, tol=1e-6, maxiter=150, seed=seed)
        if res.success:
            current_h = clip_and_rescale(res.x)
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

    def refine(h, steps=50, lr=0.05):
        h = h.copy()
        for _ in range(steps):
            perturb = np.random.normal(0, 0.01, h.shape)
            h_perturb = clip_and_rescale(h + perturb)
            corr_perturb = np.max(np.correlate(h_perturb, 1 - h_perturb, mode='full')) * dx
            corr = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            grad = (corr_perturb - corr) / 0.01
            if grad == 0:
                continue
            h += lr * grad
            h = clip_and_rescale(h)
            new_corr = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if abs(new_corr - corr) < 1e-6:
                break
        return h

    best_h = refine(best_h, steps=100, lr=0.05)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h, best_c5, n_points)
