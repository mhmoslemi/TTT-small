import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Process initial_h_values
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = np.clip(h_initial, 0, 1)
    sum_h = np.sum(h_initial)
    scale = target_sum / sum_h
    h_initial = h_initial * scale
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h, dx)
    best_c5 = min(best_c5, 0.381)

    # Generate initial guesses
    initial_guesses = []
    # Original initial guess
    initial_guesses.append(h_initial.copy())
    
    # Sine-like pattern
    h_sine = 0.5 * (1 + np.sin(np.pi * np.arange(n_points) / (n_points)))
    h_sine = np.clip(h_sine, 0, 1)
    sum_h = np.sum(h_sine)
    scale = target_sum / sum_h
    h_sine = h_sine * scale
    initial_guesses.append(h_sine.copy())
    
    # Block pattern: 1s and 0s with some jitter
    h_block = np.zeros(n_points)
    for i in range(n_points):
        if (i % 4 == 0 or i % 4 == 1):
            h_block[i] = 1.0
        else:
            h_block[i] = 0.0
    sum_h = np.sum(h_block)
    scale = target_sum / sum_h
    h_block = h_block * scale
    h_block = np.clip(h_block, 0, 1)
    initial_guesses.append(h_block.copy())
    
    # Random guess
    h_random = np.random.rand(n_points)
    sum_h = np.sum(h_random)
    scale = target_sum / sum_h
    h_random = h_random * scale
    h_random = np.clip(h_random, 0, 1)
    initial_guesses.append(h_random.copy())
    
    # Alternating 1 and 0 with jitter
    h_alt = np.zeros(n_points)
    for i in range(n_points):
        h_alt[i] = 1.0 if (i % 2 == 0) else 0.0
    sum_h = np.sum(h_alt)
    scale = target_sum / sum_h
    h_alt = h_alt * scale
    h_alt = np.clip(h_alt, 0, 1)
    initial_guesses.append(h_alt.copy())
    
    def objective_for_de(h_candidate):
        h = np.clip(h_candidate, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.inf
        scale = target_sum / sum_h
        h_scaled = h * scale
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = differential_evolution(objective_for_de, bounds, 
                                       popsize=20, strategy='best1bin', 
                                       maxiter=100, tol=1e-6, 
                                       mutation=(0.5, 1), 
                                       recombination=0.9, 
                                       seed=seed,
                                       args=())
        if result.success:
            current_h = np.clip(result.x, 0, 1)
            sum_current = np.sum(current_h)
            scale_current = target_sum / sum_current
            current_h_scaled = current_h * scale_current
            current_c5 = np.max(np.correlate(current_h_scaled, 1 - current_h_scaled, mode='full')) * dx
            if current_c5 < best_c5:
                best_h = current_h_scaled.copy()
                best_c5 = current_c5
    
    # SLSQP refinement
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h_vec)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h_vec[j]
        return grad
    
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       bounds=[(0, 1)] * n_points,
                       jac=compute_gradient, tol=1e-6, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_h = res.x.copy()
            best_c5 = res.fun
    
    # Final L-BFGS-B optimization
    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B',

        if res.success and res.fun < best_c5:
            best_h = res.x.copy()
            best_c5 = res.fun
    
    # Ensure the best_h is feasible
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    scale_best = target_sum / sum_best
    best_h = best_h * scale_best
    return (best_h, best_c5, n_points)
