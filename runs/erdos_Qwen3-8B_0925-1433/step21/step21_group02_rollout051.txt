        lag = max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if 0 <= i + lag < n_points:
                grad[i] += 1.0 - h[i + lag]
            if 0 <= i - lag < n_points:
                grad[i] -= h[i - lag]
        h -= eta * grad
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * target_sum / h_sum
        else:
            h = np.zeros(n_points)
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        eta *= 0.99

    return (best_h, best_c5, n_points)
