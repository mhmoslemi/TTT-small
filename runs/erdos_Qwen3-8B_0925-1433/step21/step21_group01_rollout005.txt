import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_h = np.array(initial_h_values, dtype=float)
    fft_h = np.fft.fft(initial_h)
    cut_off = int(0.3 * n_points)
    filtered_fft = np.zeros_like(fft_h)
    filtered_fft[:cut_off] = fft_h[:cut_off]
    filtered_h = np.fft.ifft(filtered_fft).real
    filtered_h = np.clip(filtered_h, 0, 1)
    filtered_h = filtered_h * target_sum / np.sum(filtered_h)
    population = [filtered_h]
    for _ in range(4):
        h = filtered_h.copy()
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    population = np.array(population)

    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        evals = np.array([evaluate(h) for h in population])
        elite_idx = np.argsort(evals)[:5]
        elites = population[elite_idx]
        new_pop = []
        for _ in range(20):
            if np.random.rand() < 0.8:
                p1, p2 = np.random.choice(elites, 2, replace=False)
                idx = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:idx], p2[idx:]])
            else:
                child = elites[np.random.choice(5)].copy()
            mutation = np.random.normal(0, 0.05 * (1 - gen / 200), n_points)
            child += mutation
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        population = np.array(new_pop)
        for h in population:
            current_c5 = evaluate(h)
