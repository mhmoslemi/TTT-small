import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_rescale(x):
        x_clipped = np.clip(x, 0, 1)
        if np.sum(x_clipped) == 0:
            return x_clipped
        return x_clipped * target_sum / np.sum(x_clipped)

    def objective(x):
        h = clip_rescale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def local_refine(h):
        def local_obj(x):
            h_clipped = clip_rescale(x)
            corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
            return np.max(corr) * dx
        result = minimize(local_obj, h, method='L-BFGS-B', jac=None, options={'maxiter': 100, 'ftol': 1e-7})
        return clip_rescale(result.x)

    # Generate diverse initial variants
    variants = [initial_h_values]
    for _ in range(5):

        noise = np.random.normal(0, 0.1, n_points)

        h = h * target_sum / np.sum(h)
        variants.append(h)
        # Symmetry perturbation
        h_sym = np.concatenate([h[n_points//2:], h[:n_points//2]])
        h_sym = np.clip(h_sym, 0, 1)
        h_sym = h_sym * target_sum / np.sum(h_sym)
        variants.append(h_sym)

            h[idx] *= 0.8
            h[np.where(h < 0.3)[0]] *= 1.2
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            variants.append(h)

    best_h = np.array(initial_h_values)
    best_c5 = objective(best_h)

    start_time = time.time()
    for variant in variants:

        bounds = [(0, 1)] * n_points
