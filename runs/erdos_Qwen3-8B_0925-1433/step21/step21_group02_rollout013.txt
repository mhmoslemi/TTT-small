        initial_h = initial_h * target_sum / np.sum(initial_h)
    else:
        initial_h = np.clip(initial_h, 0, 1)

    population = [initial_h]
    for _ in range(49):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return corr.max()

    def compute_gradient(h):
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * target_sum / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            max_corr_plus = corr_plus.max()
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * target_sum / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full') * dx
            max_corr_minus = corr_minus.max()
            
            grad[i] = (max_corr_plus - max_corr_minus) / (2 * eps)
        return grad

    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        evaluations = []
        for h in population:
            if np.sum(h) == 0:
                c5 = np.inf
            else:
                c5 = compute_c5(h)
            evaluations.append((c5, h))
        evaluations.sort()
        
        top_candidates = evaluations[:int(len(evaluations)*0.1)]
        for c5, h in top_candidates:
            grad = compute_gradient(h)
            step_size = 0.1 * (1 - (c5 - compute_c5(best_h)) / compute_c5(best_h))
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            new_c5 = compute_c5(h_new)
            for i, (curr_c5, curr_h) in enumerate(evaluations):
                if np.array_equal(curr_h, h):
                    evaluations[i] = (new_c5, h_new)
                    break
        
        for i in range(len(evaluations)):
            c5, h = evaluations[i]
            if np.random.rand() < 0.2:
                corr = np.correlate(h, 1 - h, mode='full') * dx
                worst_k = np.argmax(corr)
                h_perturb = h.copy()
                noise = np.random.normal(0, 0.05, n_points)
                h_perturb += noise
                h_perturb = np.clip(h_perturb, 0, 1)
                h_perturb = h_perturb * target_sum / np.sum(h_perturb)
                evaluations[i] = (compute_c5(h_perturb), h_perturb)
        
        for c5, h in evaluations:
            if c5 < best_c5:
                best_c5 = c5
                best_h = h

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
