import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    learning_rate = 0.1
    momentum = 0.9
    decay_rate = 0.995
    max_iterations = 200
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        k = max_corr_idx - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            pos = (i + k) % n_points
            term1 = 1.0 - h[pos]
            neg = (i - k) % n_points
            term2 = h[neg]
            grad[i] = term1 - term2
        m = momentum * m + (1 - momentum) * grad
        h = h - learning_rate * m
        learning_rate *= decay_rate
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        if np.random.rand() < 0.1:
            noise = np.random.normal(0, 0.05, size=n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * target_sum / np.sum(h)

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=Bounds(0, 1), constraints=cons, tol=1e-6)
    if res.success and res.fun < best_c5:
        best_h, best_c5 = res.x, res.fun

    for _ in range(5):
        if time.time() - start_time > budget_s:

        h_perturb = np.clip(best_h + noise, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        res = minimize(obj, h_perturb, method='L-BFGS-B', bounds=Bounds(0, 1), constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
