import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj_de(x):
        x = np.clip(x, 0, 1)
        x_norm = x * target_sum / np.sum(x)
        corr = np.correlate(x_norm, 1 - x_norm, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]

    def obj_slsqp(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    cons_slsqp = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds_slsqp = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    # Initial DE run
    if time.time() - start_time < budget_s * 0.6:
        de_result = differential_evolution(obj_de, bounds, popsize=50, strategy='best1bin', tol=1e-6, maxiter=150)
        best_h = de_result.x
        best_c5 = obj_de(best_h)

    # SLSQP refinement
    if time.time() - start_time < budget_s:
        res = minimize(obj_slsqp, best_h, method='SLSQP', bounds=bounds_slsqp, constraints=cons_slsqp, tol=1e-6)
        if res.success:
            best_h = res.x
            best_c5 = res.fun

    # Local perturbations
    for _ in range(3):
        if time.time() - start_time >= budget_s:
            break
        perturbation = np.random.normal(0, 0.05, n_points)
        trial = np.clip(best_h + perturbation, 0, 1)
        trial = trial * target_sum / np.sum(trial)
        res = minimize(obj_slsqp, trial, method='SLSQP', bounds=bounds_slsqp, constraints=cons_slsqp, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
