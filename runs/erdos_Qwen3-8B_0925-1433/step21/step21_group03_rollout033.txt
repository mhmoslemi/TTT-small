from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def clip_and_rescale(x):
        x_clipped = np.clip(x, 0, 1)
        sum_clipped = np.sum(x_clipped)
        if sum_clipped == 0:
            return x_clipped
        scaling = target_sum / sum_clipped
        return x_clipped * scaling

    def objective(x):
        h = clip_and_rescale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate diverse initial variants
    variants = [initial_h_values.copy()]
    for _ in range(7):
        h = np.array(initial_h_values, dtype=float)
        # Add noise
        noise = np.random.normal(0, 0.15, n_points)
        h += noise
        h = clip_and_rescale(h)
        variants.append(h)
        # Region adjustment
        pos = np.where(h > 0.6)
        if len(pos[0]) > 0:
            h[pos] *= 0.8
            adj = np.zeros(n_points)
            adj[pos] = 1.0
            h += adj * (target_sum - np.sum(h)) / n_points
            h = clip_and_rescale(h)
        variants.append(h)
        # Symmetry perturbation
        mid = n_points // 2
        h[:mid], h[mid:] = h[mid:].copy(), h[:mid].copy()
        variants.append(h)

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    for variant in variants:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1)] * n_points
        result = differential_evolution(
            objective, bounds, strategy='best1bin', popsize=25,
            mutation=(0.5, 0.8), recombination=0.9, tol=1e-6,
            maxiter=150, seed=seed + variants.index(variant)
        )
        if result.success:
            current_h = clip_and_rescale(result.x)
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

    # Local refinement with gradient descent
    for _ in range(100):
