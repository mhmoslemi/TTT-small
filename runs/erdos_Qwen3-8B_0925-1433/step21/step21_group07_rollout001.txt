import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        shift = k_max_index - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            if 0 <= i_plus < n_points:
                term1 = 1 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - shift
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    # Generate initial guesses with perturbations
    n_initial = 5
    initial_guesses = []
    base = np.array(initial_h_values, dtype=float)
    base = base * target_sum / np.sum(base)
    for _ in range(n_initial):
        perturb = np.random.uniform(-0.15, 0.15, n_points)
        guess = base + perturb
        guess = np.clip(guess, 0, 1)
        guess = guess * target_sum / np.sum(guess)
        initial_guesses.append(guess)

    best_c5 = float('inf')
    best_h = initial_guesses[0].copy()

    start_time = time.time()
    for h_start in initial_guesses:
        h = h_start.copy()
        # Global search phase
        for iter in range(200):
            if time.time() - start_time > budget_s:
                break
            grad = compute_gradient(h)
            lr = 0.3 * (0.995 ** iter)
            m = np.zeros_like(grad)
            v = np.zeros_like(grad)
            for i in range(n_points):
                m[i] = 0.9 * m[i] + 0.1 * grad[i]
                v[i] = 0.999 * v[i] + 0.001 * grad[i]**2
                h[i] = h[i] + lr * m[i] / (np.sqrt(v[i]) + 1e-8)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

        # Local refinement
        res = minimize(obj, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6, jac=compute_gradient)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x.copy()

    return (np.clip(best_h, 0, 1), best_c5, n_points)
