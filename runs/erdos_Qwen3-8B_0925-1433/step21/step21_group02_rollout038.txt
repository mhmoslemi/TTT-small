import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    eta = 0.01
    beta = 0.99
    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k_idx = np.argmax(corr)
        k = k_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) - h[i]
            else:
                grad[i] = 0.0
        h = h - eta * grad
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            h = np.zeros(n_points)
        else:
            h = h * target_sum / np.sum(h)
        new_corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(new_corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        eta *= beta

    h = np.clip(best_h, 0, 1)
    if np.sum(h) == 0:
        h = np.zeros(n_points)
    else:
        h = h * target_sum / np.sum(h)
    return (h, best_c5, n_points)
