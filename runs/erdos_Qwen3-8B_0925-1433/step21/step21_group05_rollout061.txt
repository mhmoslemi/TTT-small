    best_c5 = res_slsqp.fun

    res_l_bfgs = minimize(
        objective_slsqp,
        best_h,
        method='L-BFGS-B',
        bounds=[(0, 1)] * n_points,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        tol=1e-6,
        options={'maxiter': 100}
    )

    best_h = res_l_bfgs.x
    best_c5 = res_l_bfgs.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
