import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            term1 = 1 - x[i_plus] if 0 <= i_plus < n_points else 0
            i_minus = i - shift
            term2 = x[i_minus] if 0 <= i_minus < n_points else 0
            grad[i] = term1 - term2
        return grad

    start_time = time.time()
    for _ in range(5):  # 5 initial perturbations
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.15, n_points)
        h_pert = h + perturb
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)
        best_h_pert = h_pert.copy()
        best_c5_pert = best_c5

        # Gradient descent phase
        h_opt = h_pert.copy()
        learning_rate = 0.5
        momentum = 0.9
        decay = 0.999
        m = np.zeros(n_points)
        v = np.zeros(n_points)
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            grad = compute_gradient(h_opt)
            m = momentum * m + (1 - momentum) * grad
            v = decay * v + (1 - decay) * grad ** 2
            h_opt = h_opt + learning_rate * m / np.sqrt(v + 1e-8)
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * target_sum / np.sum(h_opt)
            curr_c5 = np.max(np.correlate(h_opt, 1 - h_opt, mode='full')) * dx
            if curr_c5 < best_c5_pert:
                best_h_pert, best_c5_pert = h_opt.copy(), curr_c5

        # L-BFGS refinement
        res = minimize(obj, best_h_pert, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6,
                       jac=compute_gradient, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
