import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_h = h.copy()

    n_iter = 0
    max_no_improve = 0
    eta = 0.05

    start_time = time.time()

    while time.time() - start_time < budget_s:
        n_iter += 1
        # Compute current C5
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        # Find the lag with maximum overlap
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        # Compute gradient for lag k
        g = np.zeros(n_points)
        for j in range(n_points):
            i_plus = j + k
            i_minus = j - k
            if 0 <= i_plus < n_points and 0 <= i_minus < n_points:
                g[j] = (1 - h[i_plus]) - h[i_minus]
        # Update h
        h += eta * g
        # Clip and scale
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        # Check if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            max_no_improve = 0
        else:
            max_no_improve += 1
        if max_no_improve > 20:
            # Apply random perturbation
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            max_no_improve = 0
        # If no improvement for 100 iterations, break
        if n_iter > 1000:
            break

    # Final clipping and scaling
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    return (h, best_c5, n_points)
