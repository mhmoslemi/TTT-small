import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def eval_candidate(h):
        h = np.clip(h, 0, 1)
        S = np.sum(h)
        if S == 0:
            return np.inf
        h_scaled = h * target_sum / S
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    pop_size = 50
    population = []
    for _ in range(pop_size):
        if _ < 20:
            h = np.array(initial_h_values, dtype=float)
            noise = np.random.normal(0, 0.1, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            population.append(h)
        else:
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
            population.append(h)
    population = np.array(population)
    
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.inf
    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        parents = []
        for _ in range(pop_size // 2):
            tournament = np.random.choice(population, size=5, replace=False)
            scores = np.array([eval_candidate(t) for t in tournament])
            best_tournament = tournament[np.argmin(scores)]
            parents.append(best_tournament)
        parents = np.array(parents)
        
        offspring = []
        for i in range(len(parents) // 2):
            p1, p2 = parents[2*i], parents[2*i + 1]
            if np.random.rand() < 0.8:
                idx = np.random.randint(1, n_points - 1)
                c1 = np.concatenate((p1[:idx], p2[idx:]))
                c2 = np.concatenate((p2[:idx], p1[idx:]))
            else:
                c1, c2 = p1.copy(), p2.copy()
            for j in range(n_points):
                if np.random.rand() < 0.2:
                    c1[j] += np.random.normal(0, 0.1)
                    c2[j] += np.random.normal(0, 0.1)
            c1 = np.clip(c1, 0, 1)
            c1 = c1 * target_sum / np.sum(c1)
            c2 = np.clip(c2, 0, 1)
            c2 = c2 * target_sum / np.sum(c2)
            offspring.append(c1)
            offspring.append(c2)
        offspring = np.array(offspring)
        
        combined = np.vstack((population, offspring))
        combined_values = np.array([eval_candidate(h) for h in combined])
        sorted_indices = np.argsort(combined_values)
        population = combined[sorted_indices][:pop_size]
        
        top_candidates = population[:5]

            res = minimize(eval_candidate, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, options={'maxiter': 50})

                    best_h = res.x

    S = np.sum(best_h)
    if S != 0:
        best_h = best_h * target_sum / S
    else:
        best_h = np.zeros(n_points)
    return (best_h, best_c5, n_points)
