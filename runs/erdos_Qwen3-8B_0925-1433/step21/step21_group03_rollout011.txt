import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return h_clipped
        scaling = target_sum / sum_clipped
        return h_clipped * scaling

    def objective(x):
        h = clip_rescale(np.array(x, dtype=float))
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h, lag, epsilon=1e-6):
        grad = np.zeros_like(h)
        h = h.copy()
        for i in range(len(h)):
            h_plus = clip_rescale(h.copy())
            h_plus[i] += epsilon
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[lag]
            h_minus = clip_rescale(h.copy())
            h_minus[i] -= epsilon
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[lag]
            grad[i] = (corr_plus - corr_minus) / (2 * epsilon)
        return grad

    def refine_solution(h, max_iter=200, lr=0.05, tol=1e-6):
        h = h.copy()
        for _ in range(max_iter):
            corr = np.correlate(h, 1 - h, mode='full')
            best_lag = np.argmax(corr)
            grad = compute_gradient(h, best_lag)
            h -= lr * grad
            h = clip_rescale(h)
            if np.max(np.abs(grad)) < tol:
                break
        return h

    variants = [
        initial_h_values.copy(),
        1 - initial_h_values[::-1].copy(),
        np.random.normal(0, 0.1, n_points) + initial_h_values.copy(),
        np.random.choice([0, 1], size=n_points, p=[0.1, 0.9]) * initial_h_values.copy(),
    ]
    for v in variants:
        v = clip_rescale(np.array(v, dtype=float))
        variants.append(v)

    best_h = initial_h_values.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    for variant in variants:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1)] * n_points
        result = differential_evolution(
            objective,
            bounds,
            strategy='rand1bin',
            popsize=30,
            mutation=(0.5, 1.0),
            recombination=0.9,
            tol=1e-6,
            maxiter=100,
            seed=seed
        )
        if result.success:
            refined_h = refine_solution(result.x)
            current_c5 = objective(refined_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = refined_h

    best_h = clip_rescale(best_h)
    return (best_h, best_c5, n_points)
