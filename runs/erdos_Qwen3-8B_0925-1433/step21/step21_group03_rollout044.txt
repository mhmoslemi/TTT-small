from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):

    target_sum = n_points / 2.0

    def clip_rescale(x):
        x_clipped = np.clip(x, 0, 1)
        sum_clipped = np.sum(x_clipped)
        if sum_clipped == 0:
            return x_clipped
        scaling = target_sum / sum_clipped
        return x_clipped * scaling

    def objective(x):
        h = clip_rescale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    variants = [h.copy()]
    for _ in range(4):
        h_sym = np.concatenate((h[::-1], h[1:-1])) if n_points > 1 else h
        variants.append(h_sym)
        h_noise = h + np.random.normal(0, 0.1, n_points)
        h_noise = clip_rescale(h_noise)
        variants.append(h_noise)
        h_block = np.zeros(n_points)
        for i in range(n_points // 5):
            h_block[i * 5] = 1.0
        h_block = clip_rescale(h_block)
        variants.append(h_block)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()

    for variant in variants:
        if time.time() - start_time > budget_s:
            break

        result = differential_evolution(objective, bounds, strategy='best1bin',
