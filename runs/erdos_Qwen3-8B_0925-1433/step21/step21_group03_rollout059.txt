from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

    def clip_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return h_clipped
        scaling = target_sum / sum_clipped
        return h_clipped * scaling

    def objective(x):
        h = clip_rescale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def local_refine(h):
        h = h.copy()
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            max_corr = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if max_corr < 1e-10:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr)
            dt = 0.01
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += dt
                h_plus = clip_rescale(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                grad[i] = (np.max(corr_plus) * dx - max_corr) / dt
            h -= 0.05 * grad
            h = clip_rescale(h)
        return h

    variants = [initial_h_values.copy()]
    for _ in range(4):
        h = np.array(initial_h_values, dtype=float)
        h = np.roll(h, np.random.randint(n_points))

        h = 1 - h
        variants.append(h)

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    for variant in variants:
