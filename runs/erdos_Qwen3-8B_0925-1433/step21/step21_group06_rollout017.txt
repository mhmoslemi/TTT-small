import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

        perturbed = perturbed * target_sum / np.sum(perturbed)
        initial_guesses.append(perturbed)
    for _ in range(15):
        uniform = np.random.rand(n_points)
        uniform = uniform * target_sum / np.sum(uniform)
        initial_guesses.append(uniform)
    for _ in range(15):
        flipped = np.array(initial_h_values, dtype=float)
        flip_mask = np.random.rand(n_points) < 0.3
        flipped[flip_mask] = 1 - flipped[flip_mask]
        flipped = np.clip(flipped, 0, 1)
        flipped = flipped * target_sum / np.sum(flipped)
        initial_guesses.append(flipped)

    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = objective(best_h)

    start_time = time.time()

    # Differential evolution
    result = differential_evolution(objective, bounds, strategy='best1bin', popsize=20, tol=1e-6, mutation=0.5, recombination=0.9, maxiter=100, disp=False)
    best_h = result.x
    best_c5 = objective(best_h)

    # Local refinement
    def obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')

    bounds_slsqp = [(0, 1) for _ in range(n_points)]

    # Refine initial guesses
    for guess in initial_guesses:
