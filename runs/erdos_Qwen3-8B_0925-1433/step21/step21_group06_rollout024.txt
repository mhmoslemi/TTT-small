import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        return x * target_sum / np.sum(x)

    def objective(x):
        h = normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    remaining_time = budget_s - (time.time() - start_time)
    
    # Phase 1: Differential Evolution
    if remaining_time > 0:
        result_de = differential_evolution(
            objective, bounds, strategy='best1bin', 
            popsize=25, maxiter=50, tol=1e-6,
            mutation=(0.5, 0.8), recombination=0.9,
            options={'mutation_scale': 0.5, 'atol': 1e-4}
        )
        best_h = normalize(result_de.x)
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
        elapsed = time.time() - start_time
        remaining_time = budget_s - elapsed

    # Phase 2: Local refinement
    if remaining_time > 0:
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(
            objective, best_h, method='SLSQP',
            bounds=bounds, constraints=constraints,
            tol=1e-6, options={'max_iter': 100}
        )
        if res.success:
            best_h = normalize(res.x)
            best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
        elapsed = time.time() - start_time
        remaining_time = budget_s - elapsed

    # Phase 3: Final refinement
    if remaining_time > 0:
        for _ in range(3):
            if time.time() - start_time >= budget_s:
                break
            res = minimize(
                objective, best_h, method='SLSQP',
                bounds=bounds, constraints=constraints,
                tol=1e-6, options={'max_iter': 20}
            )
            if res.success:
                best_h = normalize(res.x)
                best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    return (np.clip(best_h, 0, 1), best_c5, n_points)
