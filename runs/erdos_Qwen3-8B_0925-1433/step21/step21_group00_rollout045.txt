import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    learning_rate = 0.1
    decay = 0.99

    start_time = time.time()
    m1 = np.zeros(n_points)
    m2 = np.zeros(n_points)

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        current_c5 = max_corr * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
        k = np.argmax(corr) - (n_points - 1)
        k = np.clip(k, -n_points + 1, n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            pos = (i + k) % n_points
            neg = (i - k) % n_points
            grad[i] = (1 - h[pos]) - h[neg]
        m1 = beta1 * m1 + (1 - beta1) * grad
        m2 = beta2 * m2 + (1 - beta2) * (grad ** 2)
        m1_hat = m1 / (1 - beta1 ** (_ + 1))
        m2_hat = m2 / (1 - beta2 ** (_ + 1))
        h = h - learning_rate * m1_hat / (np.sqrt(m2_hat) + eps)
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)

    # Refinement with L-BFGS-B
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = Bounds(0, 1)
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    best_h = res.x
    current_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    if current_c5 < best_c5:
