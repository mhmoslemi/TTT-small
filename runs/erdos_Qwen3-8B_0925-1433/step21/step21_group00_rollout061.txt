        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            pos = (i + k) % n_points
            neg = (i - k) % n_points
            grad[i] = (1 - h[pos]) - h[neg]
        return grad
