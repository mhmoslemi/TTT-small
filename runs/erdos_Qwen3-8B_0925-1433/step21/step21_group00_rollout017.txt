    h_noise = h_noise * target_sum / np.sum(h_noise)
    variations.append(h_noise)
    
    # Invert
    h_invert = 1 - initial_h
    h_invert = h_invert * target_sum / np.sum(h_invert)
    variations.append(h_invert)
    
    # Shift
    shift = n_points // 4
    h_shift = np.roll(initial_h, shift)
    h_shift = h_shift * target_sum / np.sum(h_shift)
    variations.append(h_shift)
    
    # Sparsify
    sparsify_mask = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
    h_sparse = initial_h * sparsify_mask
    sum_sparse = np.sum(h_sparse)
    if sum_sparse == 0:
        sum_sparse = 1e-6
    h_sparse = h_sparse * target_sum / sum_sparse
    variations.append(h_sparse)
    
    start_time = time.time()
    
    for h in variations:
        bounds = Bounds(0, 1)
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_c5 = compute_c5(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x.copy()
    
    # Additional refinement
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, size=n_points)
        h_perturb = np.clip(best_h + perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        res = minimize(obj, h_perturb, method='L-BFGS-B', bounds=Bounds(0, 1), constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success:
            current_c5 = compute_c5(res.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x.copy()
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
