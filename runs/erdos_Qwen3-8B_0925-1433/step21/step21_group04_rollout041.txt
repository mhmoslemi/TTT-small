            return np.zeros_like(v)
        scale = target_sum / s
        v_scaled = v * scale
        v_clipped = np.clip(v_scaled, 0, 1)
        return v_clipped

    # Initialize population
    population = []
    initial_h = np.array(initial_h_values, dtype=float)
    population.append(initial_h)
    for _ in range(49):
        h = np.random.uniform(0, 1, n_points)
        h = scale_and_clip(h, target_sum)
        population.append(h)

    best_h = initial_h.copy()
    best_c5 = evaluate(initial_h)

    tournament_size = 5
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 200
    max_time = budget_s
    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > max_time:
            break

        # Evaluate fitness
        fitness = [evaluate(ind) for ind in population]
        if np.min(fitness) < best_c5:
            best_c5 = np.min(fitness)
            best_h = population[np.argmin(fitness)]

        new_population = []
        for _ in range(len(population) // 2):
            # Tournament selection
            tournament = np.random.choice(len(population), tournament_size)
            parent1 = population[np.argmin([fitness[i] for i in tournament])]
            tournament = np.random.choice(len(population), tournament_size)
            parent2 = population[np.argmin([fitness[i] for i in tournament])]

            # Crossover
            if np.random.rand() < crossover_rate:
                child = np.random.choice([parent1, parent2], size=n_points)
                child = scale_and_clip(child, target_sum)
            else:
                child = parent1.copy()

            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child += noise
                child = scale_and_clip(child, target_sum)

            new_population.append(child)

        # Replace worst half
        fitness_new = [evaluate(ind) for ind in new_population]
        combined = list(zip(population, fitness)) + list(zip(new_population, fitness_new))
        combined.sort(key=lambda x: x[1])
        population = [x[0] for x in combined[:len(population)]]

    return (best_h, best_c5, n_points)
