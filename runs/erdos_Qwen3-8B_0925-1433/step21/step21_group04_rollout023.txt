        contrib = [h[x] * (1 - h[(x + k) % n_points]) for x in range(n_points)]
        max_idx = np.argmax(contrib)
        x = max_idx
        x_next = (x + k) % n_points

        alpha = 0.05
        h[x] = np.clip(h[x] - alpha, 0, 1)
        h[x_next] = np.clip(h[x_next] + alpha, 0, 1)
        scale = target_sum / np.sum(h)
        h *= scale
        h = np.clip(h, 0, 1)

    corr_final = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = corr_final.max()
    return (best_h, final_c5, n_points)
