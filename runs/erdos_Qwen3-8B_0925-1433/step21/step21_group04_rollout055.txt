import numpy as np
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()

    n_particles = 20
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []

    for _ in range(n_particles):
        if _ == 0:
            h = np.array(initial_h_values, dtype=float)
            noise = np.zeros(n_points)
        else:
            h = np.array(initial_h_values, dtype=float)
            noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, size=n_points))
        c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        pbest.append(h)
        pbest_c5.append(c5)

    global_best = best_h.copy()
    global_best_c5 = best_c5

    start_time = time.time()

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] * 0.7 + 1.49 * r1 * (pbest[i] - particles[i]) + 1.49 * r2 * (global_best - particles[i])
            velocities[i] = vel
            new_pos = particles[i] + vel
            new_pos = np.clip(new_pos, 0, 1)
            current_sum = np.sum(new_pos)
            scale = target_sum / current_sum
            new_pos *= scale
            particles[i] = new_pos
            c5 = np.max(np.correlate(particles[i], 1 - particles[i], mode='full')) * dx
            if c5 < pbest_c5[i]:

    global_best = np.clip(global_best, 0, 1)
    current_sum = np.sum(global_best)
    scale = target_sum / current_sum
    global_best *= scale
    return (global_best, global_best_c5, n_points)
