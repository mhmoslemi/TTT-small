import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vec):
        h = np.clip(h_vec, 0, 1)
        sum_h = np.sum(h)
        if sum_h < 1e-10:
            return float('inf')
        h = h * (target_sum / sum_h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    initial_guess = np.array(initial_h_values, dtype=float)
    initial_guess = initial_guess * target_sum / np.sum(initial_guess)

    start_time = time.time()

    # Differential Evolution
    res_de = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=200,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        disp=False
    )

    best_h = res_de.x
    best_c5 = objective(best_h)

    # Local refinement with L-BFGS-B
    def obj(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h < 1e-10:
            return float('inf')
        h = h * (target_sum / sum_h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h < 1e-10:
            return np.zeros_like(x)
        h = h * (target_sum / sum_h)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        shift = k_max_index - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            if 0 <= i_plus < n_points:
                term1 = 1 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - shift
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    res_local = minimize(
        obj,
        best_h,
        method='L-BFGS-B',
        bounds=[(0, 1)] * n_points,
        jac=grad,
        constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
        tol=1e-6
    )

    best_h = res_local.x
    best_c5 = obj(best_h)
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h != 0:

    else:
        best_h = np.zeros_like(best_h)
        best_h = best_h * (target_sum / np.sum(best_h))

    return (best_h, best_c5, n_points)
