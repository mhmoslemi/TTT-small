import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    population = []

    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitness = [np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in population]
        best_index = np.argmin(fitness)
        if fitness[best_index] < best_c5:
            best_h = population[best_index].copy()

        new_population = []

            t1 = np.random.choice(50, 5, replace=False)
            c1 = fitness[t1]
            w1 = population[t1[np.argmin(c1)]]
            t2 = np.random.choice(50, 5, replace=False)

            child = np.where(np.random.rand(n_points) < 0.5, w1, w2)

                noise = np.random.normal(0, 0.1, n_points)
                child += noise
                child = np.clip(child, 0, 1)

        population = np.array(new_population)
