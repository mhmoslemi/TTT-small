import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):

    def clip_and_rescale(x):

        sum_clipped = np.sum(x_clipped)
        if sum_clipped == 0:
            return x_clipped
        scaling = target_sum / sum_clipped
        return x_clipped * scaling

    def objective(x):
        h = clip_and_rescale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    variants = [initial_h_values.copy()]
    for _ in range(4):
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        variants.append(h)
        h = np.array(initial_h_values, dtype=float)
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        variants.append(clip_and_rescale(h))
        h = np.array(initial_h_values, dtype=float)
        h = h[::-1]  # symmetric
        variants.append(clip_and_rescale(h))

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()
    for variant in variants:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1)] * n_points
        result = differential_evolution(
