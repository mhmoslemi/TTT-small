import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        x = x * target_sum / np.sum(x)
        return x

        return np.max(corr) * dx

    n_particles = 50
    particles = []
    initial_guess = np.array(initial_h_values, dtype=float)
    initial_guess = normalize(initial_guess)
    particles.append(initial_guess.copy())
    for _ in range(n_particles - 1):
        if np.random.rand() < 0.5:
            perturbed = initial_guess.copy()
            noise = np.random.normal(0, 0.1, n_points)
            perturbed += noise
        else:
            perturbed = np.random.rand(n_points)
        perturbed = normalize(perturbed)
        particles.append(perturbed.copy())

    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(n_particles)]
    best_h = initial_guess.copy()
    best_c5 = compute_c5(best_h, dx)

    omega, c1, c2 = 0.7, 1.5, 1.5
    pbest = [part.copy() for part in particles]

    start_time = time.time()
