import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def compute_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        shift = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            i_plus = i + shift
            term1 = 1.0 if 0 <= i_plus < n_points else 0.0
            i_minus = i - shift
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        return grad

    def optimize_candidate(h):
        h = h.copy()
        for _ in range(100):
            grad = compute_gradient(h)
            h += 0.3 * grad
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                nonlocal best_c5, best_h
                best_c5, best_h = current_c5, h.copy()
        res = minimize(lambda x: compute_c5(x), h, method='L-BFGS-B',
                       bounds=[(0, 1)] * n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                       jac=compute_gradient, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_c5, best_h = res.fun, res.x.copy()
        return best_c5, best_h

    candidates = [h.copy()]
    h_rand = np.random.rand(n_points) > 0.3
    h_rand = h_rand * target_sum / np.sum(h_rand)
    h_rand = np.clip(h_rand, 0, 1)
    candidates.append(h_rand)
    h_periodic = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n_points) / n_points))

    for cand in candidates:
