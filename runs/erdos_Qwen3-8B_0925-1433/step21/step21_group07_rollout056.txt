import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            term1 = 1.0 if 0 <= i_plus < n_points else 0.0
            i_minus = i - shift
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        return grad

    initial_guess = np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)
    num_initial = 5
    initial_guesses = [initial_guess + np.random.normal(0, 0.15, n_points) for _ in range(num_initial)]

    best_h = initial_guess.copy()
    best_c5 = obj(best_h)

    def update_h(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    start_time = time.time()
    for guess in initial_guesses:
        candidates = [guess]
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            h = update_h(candidates[-1])
            grad = compute_gradient(h)
            h += 0.1 * grad
            candidates.append(h)
        best_candidate = min(candidates, key=obj)
        result = minimize(obj, best_candidate, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                          constraints={'type': 'eq', 'fun': lambda x: np.sum(x)-target_sum},
                          jac=compute_gradient, tol=1e-6)
        if result.success and result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun

    best_h = update_h(best_h)
    return (best_h, best_c5, n_points)
