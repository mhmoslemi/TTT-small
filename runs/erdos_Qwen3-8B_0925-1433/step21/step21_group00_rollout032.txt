import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values, dtype=float)
    else:
        h_initial = np.ones(45) * 0.5
        h_initial = h_initial * 45 / 2.0 / np.sum(h_initial)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(h):
        return compute_c5(h)

    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    bounds = Bounds(0, 1)

    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    def generate_variants(h):
        yield h.copy()
        noisy = h + 0.05 * np.random.randn(len(h))
        yield np.clip(noisy, 0, 1) * target_sum / np.sum(np.clip(noisy, 0, 1))
        yield 1 - h
        yield np.roll(h, 1)
        sparsified = h.copy()
        sparsified[np.random.rand(len(h)) < 0.5] = 0
        yield sparsified * target_sum / np.sum(sparsified)

    for variation in generate_variants(h_initial):
        variation = variation.copy()
        for _ in range(2):
            if time.time() - start_time > budget_s:
                break
            res = minimize(objective, variation, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8)
            if res.success:
                current_h = res.x
                current_c5 = compute_c5(current_h)
                if current_c5 < best_c5:
                    best_h, best_c5 = current_h, current_c5
                variation = current_h

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        perturb = 0.05 * np.random.randn(n_points)
        perturbed = best_h + perturb
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        res = minimize(objective, perturbed, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-8)
        if res.success:
            current_h = res.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_h, best_c5 = current_h, current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
