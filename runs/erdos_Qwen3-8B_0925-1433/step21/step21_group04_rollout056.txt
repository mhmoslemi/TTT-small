            mut = np.random.normal(0, 0.05, n_points)
            child += mut
            child = np.clip(child, 0, 1)
            sum_current = np.sum(child)
            if sum_current > 0:
                child *= target_sum / sum_current
            new_population.append(child)

        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        population[:replace_count] = new_population

    best_h = np.clip(best_h, 0, 1)
    sum_current = np.sum(best_h)
    if sum_current > 0:
        best_h *= target_sum / sum_current
    return (best_h, best_c5, n_points)
