    best_h = best_h.copy()

    start_time = time.time()

    for _ in range(150):  # Max iterations to avoid infinite loop
        if time.time() - start_time > budget_s:
            break
        # Evaluate all candidates
        current_candidates = []
        for h in population:
            c5_h = compute_c5(h)
            current_candidates.append((c5_h, h))
        current_candidates.sort()
        top_candidates = current_candidates[:int(len(current_candidates)*0.1)]
        
        # Refine top candidates
        for c5_h, h in top_candidates:
            grad = compute_gradient(h)
            h_new = h - 0.1 * grad
            h_new = np.clip(h_new, 0, 1)
            S = np.sum(h_new)
            if S == 0:
                h_scaled = np.zeros(n_points)
            else:
                h_scaled = h_new * target_sum / S
            population.append(h_scaled)
        
        # Generate new candidates via crossover
        new_candidates = []
        for _ in range(10):
            h1 = np.random.choice(population)
            h2 = np.random.choice(population)
            h_cross = (h1 + h2) / 2
            h_cross = np.clip(h_cross, 0, 1)
            S = np.sum(h_cross)
            if S == 0:
                h_scaled = np.zeros(n_points)
            else:
                h_scaled = h_cross * target_sum / S
            new_candidates.append(h_scaled)
        population.extend(new_candidates)
        
        # Keep only top 50% of population
        population.sort(key=lambda x: compute_c5(x))
        population = population[:int(len(population)*0.5)]
        
        # Update best solution
        for c5, h in current_candidates:
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Ensure best_h meets constraints
    best_h = np.clip(best_h, 0, 1)
    S = np.sum(best_h)
    if S != 0:
        best_h = best_h * target_sum / S
    else:
        best_h = np.zeros(n_points)
    return (best_h, best_c5, n_points)
