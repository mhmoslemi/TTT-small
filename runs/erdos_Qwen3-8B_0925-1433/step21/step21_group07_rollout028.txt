from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

        variations = []
        for i in range(5):
            perturbed = base + 0.1 * np.random.randn(n_points)
            shifted = np.roll(perturbed, i)
            sin_perturbed = perturbed + 0.05 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
            variations.append(np.clip(perturbed, 0, 1))
            variations.append(np.clip(shifted, 0, 1))
            variations.append(np.clip(sin_perturbed, 0, 1))
        return variations

    initial_guesses = generate_initial_guesses()
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):

        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            i_plus = i + shift
            if 0 <= i_plus < n_points:
                term1 = 1 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - shift
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    start_time = time.time()
    for guess in initial_guesses:
