def run(seed=42, budget_s=600):

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h_vec)
        for i in range(n_points):
            j = i - k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h_vec[j]
        return grad

    def generate_sine_guess(n):

        h = np.clip(y, 0, 1)
        total = np.sum(h)
        h = h * target_sum / total
        return h

    def generate_block_guess(n):

        total = np.sum(h)
        h = h * target_sum / total
        return h

    def generate_greedy_guess(n):
        h = np.zeros(n)

            if cnt < n // 2:
                h[i] = 1.0
                cnt += 1
        total = np.sum(h)
        h = h * target_sum / total
        return h

    initial_guesses = [generate_sine_guess(n_points), generate_block_guess(n_points), generate_greedy_guess(n_points)]
