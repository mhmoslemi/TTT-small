import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    half = n_points // 2
    dx = 2.0 / n_points

    def mirror_array(h):
        n = len(h)
        mirrored = np.zeros(n)
        mirrored[:n//2] = h[:n//2]
        mirrored[n//2 + 1:] = 1 - h[:n//2][::-1]
        if n % 2 == 1:
            mirrored[n//2] = 0.5
        return mirrored

    # Mirror the initial_h_values if needed
    h_initial = initial_h_values.copy()
    h_initial = mirror_array(h_initial)
    left_initial = h_initial[:half]

    def objective(left_half):
        h = np.concatenate([left_half, 1 - left_half[::-1]])
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return float('inf')
        scaling = (n_points / 2.0) / sum_clipped
        h_scaled = h_clipped * scaling
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    # Generate initial variants
    variants = [left_initial.copy()]
    for _ in range(4):
        h = left_initial.copy()
        noise = np.random.normal(0, 0.1, half)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * (n_points/4.0) / np.sum(h)
        variants.append(h)

    best_left = left_initial.copy()
    best_c5 = objective(best_left)

    start_time = time.time()
    for variant in variants:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0.0, 1.0)] * half
        result = differential_evolution(objective, bounds, strategy='best1bin',
                                       popsize=25, mutation=(0.6, 1.0),
                                       recombination=0.9, tol=1e-5, maxiter=150,
                                       seed=seed)
        if result.success:
            current_left = result.x
            current_c5 = objective(current_left)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_left = current_left

    # Local refinement
    for _ in range(100):
        left = best_left.copy()
        perturb = np.random.uniform(-0.01, 0.01, half)
        left_perturb = np.clip(left + perturb, 0, 1)
        left_perturb = left_perturb * (n_points/4.0) / np.sum(left_perturb)
        current_c5 = objective(left_perturb)
        if current_c5 < best_c5:
            best_left = left_perturb
            best_c5 = current_c5
        else:
            break

    # Convert best_left to full array
    h_best = np.concatenate([best_left, 1 - best_left[::-1]])
    h_best_clipped = np.clip(h_best, 0, 1)
    sum_clipped = np.sum(h_best_clipped)
    scaling = (n_points / 2.0) / sum_clipped
    h_best_scaled = h_best_clipped * scaling
    # Re-evaluate C5 to ensure it's correct
    corr = np.correlate(h_best_scaled, 1 - h_best_scaled, mode='full')
    final_c5 = np.max(corr) * dx

    return (h_best_scaled, final_c5, n_points)
