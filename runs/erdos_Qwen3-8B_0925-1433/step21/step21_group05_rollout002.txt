    def generate_lattice_guess(n):
        pattern = np.array([1, 0] * (n // 2 + 1))
        pattern = pattern[:n]
        pattern = pattern * target_sum / np.sum(pattern)
        return pattern

    # Create initial guesses
    initial_guesses = [h]
    initial_guesses.append(generate_sine_guess(n_points))
    initial_guesses.append(generate_block_guess(n_points))
    initial_guesses.append(generate_lattice_guess(n_points))
    initial_guesses.append(np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points)))

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()

    def objective_de(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = differential_evolution(objective_de, bounds, strategy='best1bin', maxiter=100, popsize=15, mutation=0.5, recombination=0.9, tol=1e-6)
        h_de = np.clip(res.x, 0, 1)
        h_de = h_de * target_sum / np.sum(h_de)
        current_c5 = np.max(np.correlate(h_de, 1 - h_de, mode='full')) * dx
        if current_c5 < best_c5:
            best_h = h_de.copy()
            best_c5 = current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res_slsqp = minimize(objective, best_h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], jac=compute_gradient, tol=1e-6, options={'maxiter': 100})
        if res_slsqp.success and res_slsqp.fun < best_c5:
            best_h = res_slsqp.x.copy()
            best_c5 = res_slsqp.fun

        res_lbfgs = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], jac=compute_gradient, tol=1e-6, options={'maxiter': 100})
        if res_lbfgs.success and res_lbfgs.fun < best_c5:
            best_h = res_lbfgs.x.copy()
            best_c5 = res_lbfgs.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
