            if i + k < n_points:
                grad[i] += 1 - h[i + k]
            if i - k >= 0:
                grad[i] -= h[i - k]
        
        h += step_size * (-grad)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        scale = target_sum / current_sum
        h *= scale
        
        if iteration % perturb_interval == 0:
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            scale = target_sum / current_sum
            h *= scale
        
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        iteration += 1
    
    h_final = np.clip(best_h, 0, 1)
    current_sum = np.sum(h_final)
    scale = target_sum / current_sum
    h_final *= scale
    current_c5_final = np.max(np.correlate(h_final, 1 - h_final, mode='full')) * dx
    return (h_final, current_c5_final, n_points)
