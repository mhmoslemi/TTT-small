import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_variants(h_base):
        variants = [h_base.copy()]
        noise = 0.15 * np.random.normal(size=n_points)
        variants.append(np.clip(h_base + noise, 0, 1))
        variants.append(np.roll(h_base, 1))
        variants.append(h_base + 0.05 * np.sin(2 * np.pi * np.arange(n_points) / n_points))
        return variants

    h_list = generate_variants(np.array(initial_h_values, dtype=float))
    h_list = [h * target_sum / np.sum(h) for h in h_list]
    best_h = h_list[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        shift = k_max_index - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            if 0 <= i_plus < n_points:
                term1 = 1 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - shift
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    start_time = time.time()
    for h in h_list:
        h = h * target_sum / np.sum(h)
        h_current = h.copy()
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            grad = compute_gradient(h_current)
            h_current = h_current + 0.3 * grad
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * target_sum / np.sum(h_current)
            current_c5 = np.max(np.correlate(h_current, 1 - h_current, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h_current.copy(), current_c5

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6, jac=compute_gradient)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
