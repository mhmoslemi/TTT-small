import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h_clipped = np.clip(h, 0, 1)
        S = np.sum(h_clipped)
        if S == 0:
            return np.inf
        h_scaled = h_clipped * target_sum / S
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h):
        eps = 1e-5
        n = len(h)
        grad = np.zeros(n)
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * target_sum / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * target_sum / np.sum(h_minus)
            c5_minus = compute_c5(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad

    best_h = np.array(initial_h_values, dtype=float)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = compute_c5(best_h)

    population = [best_h]
    for _ in range(20):
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)

    start_time = time.time()

            candidates.append((c5, h))
