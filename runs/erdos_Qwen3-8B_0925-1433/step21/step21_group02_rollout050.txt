        i_plus_lag_valid = i_plus_lag[valid_plus]
        grad = np.zeros(n_points)
        grad[valid_plus] += 1 - h[i_plus_lag_valid]

        i_minus_lag = np.arange(n_points) - lag
        valid_minus = (i_minus_lag >= 0) & (i_minus_lag < n_points)
        i_minus_lag_valid = i_minus_lag[valid_minus]
        grad[valid_minus] -= h[i_minus_lag_valid]

        # Update h
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)

        if current_sum != 0:
            h = h * target_sum / current_sum
        else:
            h = np.zeros(n_points)

        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        # Update best_c5 and best_h
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        # Decay learning rate
        learning_rate *= decay_rate
