        k = np.argmax(current_corr)
        target_freq = k / n_points
        freq_indices = np.arange(n_points) / n_points
        idx = np.argmin(np.abs(freq_indices - target_freq))
        delta = 0.05 * np.exp(1j * np.random.rand() * 2 * np.pi)
        H[idx] += delta
        h_new = np.fft.ifft(H).real
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        h = h_new.copy()
        if it > 100 and current_max < best_c5 + 1e-4:
            break

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h, final_c5, n_points)
