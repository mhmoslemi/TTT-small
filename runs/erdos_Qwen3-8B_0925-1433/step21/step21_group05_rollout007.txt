from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def constraint(h_vec):
        return np.sum(h_vec) - target_sum

    # Generate structured guesses
    def gen_even_spaced(phase=0):
        return np.zeros(n_points)
    def gen_sine():
        return 0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / (n_points - 1)))
    def gen_block_pattern():
        return np.zeros(n_points)
    initial_guesses = [
        gen_even_spaced(0), gen_even_spaced(np.pi/4), gen_even_spaced(np.pi/2),
        gen_sine(), gen_block_pattern()
    ]

    start_time = time.time()
    candidates = []

    # Global search with DE
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1)] * n_points
        result = differential_evolution(
            objective, bounds, strategy='best1bin', popsize=25, tol=1e-8,
            maxiter=80, mutation=(0.5, 1), recombination=0.9, seed=seed, disp=False
        )
        if result.success:
            candidates.append((result.x, result.fun))

    # Local refinement with SLSQP
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        if not candidates:
            break
        guess, _ = min(candidates, key=lambda x: x[1])
        res = minimize(
            objective, guess, method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': constraint}],
            tol=1e-6, options={'maxiter': 200}
        )
        if res.success:
            candidates.append((res.x, res.fun))

    # Final optimization with L-BFGS-B
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        if not candidates:
            break
        guess, _ = min(candidates, key=lambda x: x[1])
        res = minimize(
            objective, guess, method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': constraint}],
            tol=1e-6, options={'maxiter': 100}
        )
        if res.success:
            candidates.append((res.x, res.fun))

    # Find best candidate
    best_candidate = min(candidates, key=lambda x: x[1])
    best_h, best_c5 = best_candidate
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
