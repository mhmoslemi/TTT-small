from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def eval_candidate(h):
        h = np.clip(h, 0, 1)
        S = np.sum(h)
        if S == 0:
            return np.inf
        h_scaled = h * (target_sum / S)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    # Initialize population
    def init_pop():
        pop = []
        # Perturb initial_h_values
        base = np.array(initial_h_values, dtype=float)

        base += noise
        base = np.clip(base, 0, 1)
        base = base * target_sum / np.sum(base)
        pop.append(base)
        # Add 4 more perturbations
        for _ in range(3):
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
            pop.append(h)
        # Generate 30 random valid vectors
        for _ in range(30):
            h = np.random.rand(n_points)
            h = h * target_sum / np.sum(h)
            pop.append(h)
        return pop

    pop = init_pop()
    best_h = np.array(initial_h_values, dtype=float)

    start_time = time.time()

    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        # Tournament selection
        parents = []
        for _ in range(20):
            candidates = np.random.choice(pop, size=5)
            best = min(candidates, key=eval_candidate)
            parents.append(best)
        # Crossover and mutation
        new_pop = []
        for i in range(0, len(parents), 2):
            h1 = parents[i]
            h2 = parents[i+1] if i+1 < len(parents) else parents[i]
            if np.random.rand() < 0.8:
                # Single-point crossover
                idx = np.random.randint(1, n_points - 1)
                child = np.concatenate([h1[:idx], h2[idx:]])
            else:
                # Uniform crossover
                child = np.random.rand(n_points)
                child = child * target_sum / np.sum(child)
            # Mutate
            noise = np.random.normal(0, 0.1, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)

        # Replace 25 worst
        new_pop += pop[:25]
        # Evaluate and sort
        scores = [eval_candidate(h) for h in new_pop]

        combined.sort(key=lambda x: x[1])
        pop = [h for h, _ in combined[:50]]

        top = sorted(pop, key=eval_candidate)[:5]
