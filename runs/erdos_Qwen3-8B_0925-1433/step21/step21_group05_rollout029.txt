                                             maxiter=100, seed=seed)
            if res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x.copy()
        except:
            pass

    # Local refinement with SLSQP
    if best_c5 == float('inf'):
        best_h = initial_guesses[0].copy()

    def objective_local(h_vec):
        h = np.clip(h_vec, 0, 1)
        if np.sum(h) == 0:
            h = np.full_like(h, 0.5)
        else:
            h = h * (target_sum / np.sum(h))
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res_local = opt.minimize(objective_local, best_h, method='SLSQP',
                             bounds=bounds, constraints=cons, tol=1e-6,
                             options={'maxiter': 200})

    best_h = res_local.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)

    return (best_h, res_local.fun, n_points)
