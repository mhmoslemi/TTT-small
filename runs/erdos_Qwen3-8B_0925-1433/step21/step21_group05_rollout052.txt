    for i in range(n_points):
        if i % 5 == 0:
            greedy_guess[i] = 1.0
    greedy_guess = greedy_guess * target_sum / np.sum(greedy_guess)
    structured_guesses.append(greedy_guess)

    start_time = time.time()
    for guess in structured_guesses:
        guess = np.clip(guess, 0, 1)
        guess = guess * target_sum / np.sum(guess)
        bounds = [(0, 1)] * n_points
        result = differential_evolution(
            compute_c5, bounds, strategy='best1bin', popsize=20, maxiter=50,
            mutation=(0.8, 1), recombination=0.9, tol=1e-6, polish=True, disp=False
        )
        if result.success and result.fun < best_c5:
            best_h = result.x.copy()
            best_c5 = result.fun
        if time.time() - start_time > budget_s:
            break

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    res_slsqp = minimize(compute_c5, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 150})
    if res_slsqp.success and res_slsqp.fun < best_c5:
        best_h = res_slsqp.x.copy()
        best_c5 = res_slsqp.fun

    res_lbfgs = minimize(compute_c5, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 80})
    if res_lbfgs.success and res_lbfgs.fun < best_c5:
        best_h = res_lbfgs.x.copy()
        best_c5 = res_lbfgs.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
