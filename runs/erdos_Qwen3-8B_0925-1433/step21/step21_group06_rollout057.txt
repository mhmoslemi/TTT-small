import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        scale = target_sum / np.sum(x) if np.sum(x) > 0 else 1.0
        return x * scale

    def objective(x):
        h = normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def generate_initial_guesses(num):
        guesses = []
        for _ in range(num // 2):
            perturbed = initial_h_values.copy()
            noise = np.random.normal(0, 0.15, n_points)
            perturbed += noise
            guesses.append(normalize(perturbed))
        for _ in range(num // 2):
            uniform = np.random.rand(n_points)
            guesses.append(normalize(uniform))
        return guesses

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()

    # Global exploration with DE
    initial_guesses = generate_initial_guesses(30)
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = differential_evolution(
            objective, bounds, popsize=15, strategy='best1bin', tol=1e-5,
            mutation=0.5, recombination=0.9, seed=seed, maxiter=100)
        if res.success:
            current_h = normalize(res.x)
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

    # Local refinement with SLSQP
    def local_refine(h):
        def obj(x):
            return objective(x)
        def constraints(x):
            return np.sum(x) - target_sum
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraints}, tol=1e-6)
        if res.success:
            return res.x
        return h

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        best_h = local_refine(best_h)
        current_c5 = objective(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
