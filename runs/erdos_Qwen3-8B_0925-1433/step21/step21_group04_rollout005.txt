    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_h = h.copy()
    
    eta = 0.05
    perturb_freq = 20
    perturb_amp = 0.05
    max_iter = 1000
    iter_count = 0
    no_imp_iter = 0
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        corr_old = np.correlate(h, 1 - h, mode='full')
        max_pos = np.argmax(corr_old)
        lag_k = max_pos - (n_points - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            term1 = 1.0 - h[i + lag_k] if 0 <= i + lag_k < n_points else 0.0
            term2 = h[i - lag_k] if 0 <= i - lag_k < n_points else 0.0
            grad[i] = term1 - term2
        
        h -= eta * grad
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = target_sum / sum_h
        h *= scale
        
        corr_new = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr_new) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_imp_iter = 0
        else:
            no_imp_iter += 1
        
        if no_imp_iter >= perturb_freq:
            perturb_mask = np.random.rand(n_points) < 0.1
            h[perturb_mask] += np.random.uniform(-perturb_amp, perturb_amp, size=sum(perturb_mask))
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            scale = target_sum / sum_h
            h *= scale
            no_imp_iter = 0
    
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    scale = target_sum / sum_h
    h *= scale
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    return (h, best_c5, n_points)
