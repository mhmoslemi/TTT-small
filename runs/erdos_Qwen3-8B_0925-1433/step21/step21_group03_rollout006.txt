import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_and_rescale(x):
        x_clipped = np.clip(x, 0, 1)
        sum_clipped = np.sum(x_clipped)
        if sum_clipped == 0:
            return x_clipped
        scaling = target_sum / sum_clipped
        x_scaled = x_clipped * scaling
        for i in range(n_points):
            x_scaled[2 * n_points - i - 1] = 1 - x_scaled[i]
        return x_scaled

    def objective(x):
        h = clip_and_rescale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h_initial = np.array(initial_h_values, dtype=float)
    for i in range(n_points):
        h_initial[2 * n_points - i - 1] = 1 - h_initial[i]
    h_initial = clip_and_rescale(h_initial)

    variants = [h_initial.copy()]
    for _ in range(4):
        noise = np.random.normal(0, 0.1, n_points)
        h = h_initial + noise
        h = clip_and_rescale(h)
        variants.append(h)

    best_h = h_initial
    best_c5 = objective(best_h)

    start_time = time.time()
    bounds = [(0, 1)] * n_points
    result = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=25, mutation=(0.6, 1.0),
        recombination=0.9, tol=1e-6, maxiter=150, seed=seed, x0=variants[0]
    )
    if result.success:
        candidate_h = clip_and_rescale(result.x)
        candidate_c5 = objective(result.x)
        if candidate_c5 < best_c5:
            best_h = candidate_h
            best_c5 = candidate_c5

    top_candidates = [best_h]
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        h = top_candidates[0].copy()
        for _ in range(100):
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_perturbed = h.copy()
                h_perturbed[i] += 1e-6
                grad[i] = (objective(h_perturbed) - objective(h)) / 1e-6
            h += 0.1 * grad
            h = clip_and_rescale(h)
            current_c5 = objective(h)
            if current_c5 >= best_c5:
                break
        if objective(h) < best_c5:
            best_h = h
            best_c5 = objective(h)

    return (best_h, best_c5, n_points)
