from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return h_clipped
        scaling = target_sum / sum_clipped
        return h_clipped * scaling

    def obj(x):
        h = clip_rescale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    variants = [initial_h_values.copy()]
    for _ in range(10):
        h = np.array(initial_h_values, dtype=float)
        # Mirror
        if _ % 2 == 0:
            h = h[::-1]
        # Invert
        if _ % 3 == 0:
            h = 1.0 - h
        # Shift
        if _ % 5 == 0:
            h = np.roll(h, np.random.randint(0, n_points))
        variants.append(h)

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    for variant in variants:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1)] * n_points
        result = differential_evolution(
            obj, bounds, strategy='best1bin', popsize=30, mutation=(0.6, 1.2), recombination=0.9,
            tol=1e-6, maxiter=150, seed=seed + variants.index(variant)
        )
        if result.success:
            curr_h = clip_rescale(result.x)
            curr_c5 = obj(result.x)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = curr_h

    def grad_descent(h, steps=200, lr=0.05):
        for _ in range(steps):
            h = clip_rescale(h)
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr)
