import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h_scaled = h * (target_sum / np.sum(h))
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    def objective_slsqp(x):
        return compute_c5(x)

    def objective_de(x):
        h = np.clip(x, 0, 1)
        h_scaled = h * (target_sum / np.sum(h))
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    initial_guesses = [h_initial]

    def generate_sine_pattern(n):
        x = np.linspace(0, 2, n)
        h = 0.5 * (1 + np.sin(np.pi * x))
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        return h

    initial_guesses.append(generate_sine_pattern(n_points))

    def generate_periodic_pattern(n):
        h = np.zeros(n)
        for i in range(n):
            h[i] = 0.5 + 0.5 * np.sin(np.pi * (i - 0.5) / n)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        return h

    initial_guesses.append(generate_periodic_pattern(n_points))

    def generate_random_pattern(n):
        h = np.random.rand(n)
        h = h * (target_sum / np.sum(h))
        return h

    initial_guesses.append(generate_random_pattern(n_points))

    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:

        bounds = [(0, 1)] * n_points

            current_c5 = compute_c5(current_h)
