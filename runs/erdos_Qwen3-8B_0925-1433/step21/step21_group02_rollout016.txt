import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        h = np.clip(h, 0, 1)
        S = np.sum(h)

        h += np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    for _ in range(30):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = eval_candidate(best_h)

    start_time = time.time()
    max_generations = 100

        if time.time() - start_time > budget_s:
            break
        parents = []
        for _ in range(20):
            candidates = np.random.choice(population, 5)
            best = min(candidates, key=eval_candidate)
            parents.append(best)
        offspring = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i + 1] if i + 1 < len(parents) else parents[i]
            if np.random.rand() < 0.8:

            else:
                c1, c2 = p1.copy(), p2.copy()
