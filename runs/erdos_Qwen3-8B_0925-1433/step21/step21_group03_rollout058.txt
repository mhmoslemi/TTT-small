import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_rescale(h):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return h_clipped
        scaling = target_sum / sum_clipped
        return h_clipped * scaling

    def objective(x):
        h = clip_rescale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate diverse initial variants
    variants = [initial_h_values.copy()]
    for _ in range(3):
        h = np.array(initial_h_values, dtype=float)
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = clip_rescale(h)
        variants.append(h)
    h_mirror = np.concatenate([initial_h_values[::-1], initial_h_values])
    variants.append(h_mirror)

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = objective(best_h)
    start_time = time.time()

    # Global optimization with multiple strategies
    params = {
        'popsize': 30,
        'mutation': [(0.6, 1.0), (0.8, 1.2), (0.4, 0.8)],
        'recombination': [0.7, 0.5, 0.9],
        'strategy': ['best1bin', 'rand1bin', 'best2bin'],
        'maxiter': 100
    }

    for i, (mut, rec, stra) in enumerate(zip(params['mutation'], params['recombination'], params['strategy'])):
        if time.time() - start_time > budget_s:
            break

        result = differential_evolution(
            objective, bounds, strategy=stra, popsize=params['popsize'],

            seed=seed + i

        if result.success:

            current_c5 = result.fun

    temp_h = best_h.copy()
