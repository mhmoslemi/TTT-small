import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return np.inf
        scale = target_sum / sum_h
        h_scaled = h_clipped * scale
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    # Initialize population
    population = []
    for _ in range(10):
        h = np.array(initial_h_values, dtype=float)
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.zeros(n_points)
        else:
            h *= target_sum / sum_h
        population.append(h)
    for _ in range(15):
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.zeros(n_points)
        else:
            h *= target_sum / sum_h
        population.append(h)
    
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Tournament selection
        parents = []
        for _ in range(20):
            idxs = np.random.choice(len(population), 5, replace=False)
            idx = min(idxs, key=lambda x: evaluate(population[x]))
            parents.append(population[idx])
        # Crossover
        offspring = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1] if i+1 < len(parents) else parents[i]
            if np.random.rand() < 0.8:
                pt = np.random.randint(1, n_points)
                child = np.concatenate([p1[:pt], p2[pt:]])
            else:
                child = p1.copy()

            for j in range(n_points):
