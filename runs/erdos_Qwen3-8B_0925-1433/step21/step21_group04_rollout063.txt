import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

        current_max = np.max(corr)
        k_max = np.argmax(corr)
        H = np.fft.fft(h)

        H[freq] *= 0.9995
        H[-freq-1] *= 0.9995
        h_new = np.fft.ifft(H).real

        h_new = h_new * target_sum / np.sum(h_new)
        new_c5 = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
        if new_c5 < current_max:
            h = h_new.copy()
            best_c5 = new_c5
    
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
