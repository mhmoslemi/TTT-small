import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_variations(h):
        h = h.copy()
        variations = [h]
        noise = np.random.normal(0, 0.1, n_points)
        variations.append(h + noise)
        variations.append(h - noise)
        variations.append(np.roll(h, 1))
        variations.append(np.random.permutation(h))
        variations.append(h + 0.05 * np.sin(2 * np.pi * np.arange(n_points) / n_points))
        return variations

    initial_guesses = generate_variations(np.array(initial_h_values, dtype=float))
    initial_guesses = [g * target_sum / np.sum(g) for g in initial_guesses]

    best_h = initial_guesses[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        shift = k_max_index - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            if 0 <= i_plus < n_points:
                term1 = 1 - h[i_plus]
            else:
                term1 = 0.0
            i_minus = i - shift
            if 0 <= i_minus < n_points:
                term2 = h[i_minus]
            else:
                term2 = 0.0
            grad[i] = term1 - term2
        return grad

    start_time = time.time()
    for guess in initial_guesses:
        h = guess.copy()
        learning_rate = 0.5
        momentum = 0.9
        decay_rate = 0.99
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            grad = compute_gradient(h)
            m = np.zeros_like(grad)
            v = np.zeros_like(grad)
            for i in range(n_points):
                m[i] = momentum * m[i] + (1 - momentum) * grad[i]
                v[i] = decay_rate * v[i] + (1 - decay_rate) * grad[i] ** 2
                biased_m = m[i] / (1 - momentum ** (_ + 1))
