from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def optimize_start(h_start):
        start_time = time.time()
        res = differential_evolution(objective, bounds=[(0, 1)]*n_points, strategy='best1bin', popsize=20, mutation=(0.8, 1.0), recombination=0.9, tol=1e-6, maxiter=100, disp=False)
        for _ in range(5):
            if time.time() - start_time > budget_s:
                break
            res = minimize(objective, res.x, method='SLSQP', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-6, options={'maxiter': 100})
            if res.success and res.fun < best_c5:
                best_h = res.x.copy()
                best_c5 = res.fun
        return best_h, best_c5

    initial_guesses = [h]
    n = n_points

    initial_guesses.append(0.5 * (1 + np.sin(np.pi * np.arange(n) / n)))
