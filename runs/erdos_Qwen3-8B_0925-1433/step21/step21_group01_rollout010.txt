from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population with initial_h_values and Fourier-based pattern
    population = []

    base = 0.5 + 0.5 * np.sin(np.pi * x / 2)
    smoothed = np.convolve(base, np.ones(5), mode='same') / 5
    smoothed = np.clip(smoothed, 0, 1)
    smoothed = smoothed * target_sum / np.sum(smoothed)
    population.append(smoothed)
    # Add noise perturbations
    for _ in range(2):
        h = initial_h.copy()
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    population = np.array(population)

    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()
    pop_size = 15
    elite_count = 3
    generations = 150
    mutation_std = 0.05
    crossover_prob = 0.8
    no_improvement = 0

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        evals = np.array([evaluate(h) for h in population])
        sorted_indices = np.argsort(evals)
        elites = population[sorted_indices[:elite_count]]
        
        new_population = []
        for _ in range(pop_size):
            if np.random.rand() < crossover_prob:
                p1, p2 = elites[np.random.choice(elite_count, 2, replace=False)]
                idx = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:idx], p2[idx:]])
            else:
                child = elites[np.random.choice(elite_count)].copy()
            noise = np.random.normal(0, mutation_std, n_points)

            else:
