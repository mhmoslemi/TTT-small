import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    result_de = differential_evolution(obj, bounds, strategy='best1bin',
                                      popsize=50, tol=1e-6, mutation=(0.5, 1),
                                      recombination=0.9, maxiter=200, polish=False,
                                      disp=False)
    de_time = time.time() - start_time

    best_x = result_de.x
    best_c5 = result_de.fun

    def obj_slsqp(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds_slsqp = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    result_slsqp = minimize(obj_slsqp, best_x, method='SLSQP',
                            bounds=bounds_slsqp, constraints=cons,
                            tol=1e-6, options={'maxiter': 100})
    slsqp_time = time.time() - start_time

    if result_slsqp.success:
        best_x = result_slsqp.x
        best_c5 = result_slsqp.fun

    for _ in range(5):
        if time.time() - start_time > budget_s - de_time - slsqp_time:

        res = minimize(obj_slsqp, best_x, method='SLSQP',
                       bounds=bounds_slsqp, constraints=cons,
                       tol=1e-6, options={'maxiter': 10})
        if res.success:
            best_x = res.x
            best_c5 = res.fun
