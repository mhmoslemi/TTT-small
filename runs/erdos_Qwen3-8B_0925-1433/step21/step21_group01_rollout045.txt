import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population with initial_h_values and variants
    population = np.array([np.array(initial_h_values, dtype=float)])
    # Add Fourier-smoothed variants
    for _ in range(3):
        fft_h = np.fft.fft(population[0])
        freqs = np.fft.fftfreq(n_points)
        window = np.exp(-((freqs * n_points / 2) ** 2) / (2 * 50.**2))
        fft_h_filtered = fft_h * window
        h_filtered = np.fft.ifft(fft_h_filtered).real
        h_filtered = np.clip(h_filtered, 0, 1)
        h_filtered = h_filtered * target_sum / np.sum(h_filtered)
        population = np.vstack((population, h_filtered))

    best_h = population[0].copy()
    best_c5 = evaluate(best_h)
    best_c5 = min(best_c5, 0.381)
    start_time = time.time()
    pop_size = 20
    elite_count = 5
    generations = 200
    mutation_lr = 0.1
    crossover_prob = 0.6
    no_improvement = 0

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        evals = np.array([evaluate(h) for h in population])
        sorted_idx = np.argsort(evals)
        elites = population[sorted_idx[:elite_count]]
        new_pop = []

        for _ in range(pop_size):
            if np.random.rand() < crossover_prob:
                p1, p2 = elites[np.random.choice(elite_count, 2, replace=False)]
                idx = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:idx], p2[idx:]])
            else:
                child = elites[np.random.choice(elite_count)].copy()

            # Gradient-based mutation
            corr = np.correlate(child, 1 - child, mode='full')
            k_worst = np.argmax(corr)
            grad = np.zeros_like(child)
            grad[:n_points - k_worst] = (1 - child[k_worst:]) * dx
            grad[k_worst:] = -child[:n_points - k_worst] * dx
            child -= mutation_lr * grad
            # Clip and rescale
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        population = np.array(new_pop)

        for h in population:
            current_c5 = evaluate(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                no_improvement = 0
            else:
                no_improvement += 1
            if no_improvement >= 5:
                break

    # Final refinement with L-BFGS-B
    def obj(x):
        h = x * target_sum / np.sum(x)
        return evaluate(h)

                   constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)

    return (final_h, best_c5, n_points)
