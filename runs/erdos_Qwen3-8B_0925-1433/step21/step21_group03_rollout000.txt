def run(seed=42, budget_s=600):

        sum_clipped = np.sum(x_clipped)
        if sum_clipped == 0:
            return x_clipped
        scaling = target_sum / sum_clipped
        return x_clipped * scaling

    # Objective function
    def objective(x):
        h = clip_and_rescale(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate initial variants
    variants = [initial_h_values.copy()]
    for _ in range(3):
        # Noise variant
        h = np.array(initial_h_values, dtype=float)
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = clip_and_rescale(h)
        variants.append(h)
        # Region adjustment variant
        h = np.array(initial_h_values, dtype=float)
        over = h > 0.5
        if np.any(over):
            h[over] -= 0.1 * h[over]
