import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Clip and scale initial_h_values
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def gradient_step(h):
        eps = 1e-5
        grad = np.zeros(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            c5_plus = compute_c5(h_plus)
            c5_minus = compute_c5(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    # Generate initial population
    population = [h.copy()]
    for _ in range(20):
        rand_h = np.random.rand(n_points)
        rand_h = np.clip(rand_h, 0, 1)
        rand_h = rand_h * target_sum / np.sum(rand_h)
        population.append(rand_h)

    while time.time() - start_time < budget_s:
        # Evaluate all candidates
        candidates = []
        for h_candidate in population:
            candidates.append((compute_c5(h_candidate), h_candidate))
        # Sort candidates by C5
        candidates.sort()
        # Select top 10%
        top_candidates = candidates[:int(len(candidates)*0.1)]
        # Local refinement on top candidates
        for c5_val, h_candidate in top_candidates:
            grad = gradient_step(h_candidate)
            h_new = h_candidate - 0.1 * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            if compute_c5(h_new) < best_c5:
                best_c5 = compute_c5(h_new)
                best_h = h_new.copy()
            candidates.append((compute_c5(h_new), h_new))
        # Mutate remaining candidates
        for c5_val, h_candidate in candidates[int(len(candidates)*0.1):]:
            # Find max lag
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
