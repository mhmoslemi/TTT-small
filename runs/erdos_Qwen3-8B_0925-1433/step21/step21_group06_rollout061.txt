import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def normalize(h, target):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        scaling = target / total
        return h * scaling

    # Generate initial particles
    n_particles = 50
    particles = []
    velocities = []
    pbest = []
    
    # Perturb initial_h_values
    for _ in range(10):
        h = initial_h_values.copy()
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = normalize(h, target_sum)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(h.copy())
    
    # Random samples
    for _ in range(40):
        h = np.random.rand(n_points)
        h = normalize(h, target_sum)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(h.copy())

    # Initial global best
    gbest = particles[0].copy()
    best_c5 = evaluate_erdos_solution(gbest)
    
    start_time = time.time()
    for _ in range(300):  # Max iterations
        if time.time() - start_time > budget_s:
            break
        # Evaluate all particles
        current_c5s = [evaluate_erdos_solution(p) for p in particles]
        # Update pbest
        for i in range(n_particles):
            if current_c5s[i] < evaluate_erdos_solution(pbest[i]):
                pbest[i] = particles[i].copy()
        # Update gbest
        for p in pbest:
            p_c5 = evaluate_erdos_solution(p)
            if p_c5 < best_c5:
                best_c5 = p_c5
                gbest = p.copy()
        # Update velocities and positions
        for i in range(n_particles):
            vel = velocities[i]
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            vel += 0.7 * (pbest[i] - particles[i]) * r1
            vel += 1.5 * (gbest - particles[i]) * r2
            velocities[i] = vel
            # Update position
            particles[i] += vel
            # Clip and normalize
            particles[i] = normalize(particles[i], target_sum)

    # Local refinement
    best_h = gbest.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(lambda x: evaluate_erdos_solution(x), best_h, 
                       method='SLSQP', 
                       bounds=[(0,1) for _ in range(n_points)], 
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success:
            best_h = res.x
            best_c5 = evaluate_erdos_solution(best_h)
        else:
            break
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
