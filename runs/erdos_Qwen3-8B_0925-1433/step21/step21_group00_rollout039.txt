from scipy.optimize import minimize, Bounds

    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_idx = np.argmax(corr)
        shift = k_max_idx - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = (i + shift) % n_points
            term1 = 1 - h[i_plus]
            i_minus = (i - shift) % n_points
            term2 = h[i_minus]
            grad[i] = term1 - term2
        return grad

    # Adam optimizer parameters
    learning_rate = 0.05
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    m = np.zeros(n_points)

            break
