                best_c5 = compute_c5(p)
        for i in range(len(pop)):
            if i not in top:
                p = pop[i].copy()
                noise = np.random.normal(0, 0.05, n)
                p += noise
                p = np.clip(p, 0, 1)
                p *= target_sum / np.sum(p)
                pop[i] = p

    return (best_h, best_c5, n)
