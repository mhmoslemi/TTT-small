        np.roll(h0, 2),
        np.random.permutation(h0),
        h0 + 0.05 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
    ]

    for guess in variants:
        h = guess.copy()
        learning_rate = 0.3
        momentum = 0.9
        decay_rate = 0.99
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            grad = compute_gradient(h)
            m, v = np.zeros_like(grad), np.zeros_like(grad)
            for i in range(n_points):
                m[i] = momentum * m[i] + (1 - momentum) * grad[i]
                v[i] = decay_rate * v[i] + (1 - decay_rate) * grad[i] ** 2
                h[i] = h[i] + learning_rate * m[i] / (np.sqrt(v[i]) + 1e-8)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    start_local = time.time()
    for _ in range(10):
        if time.time() - start_local > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)] * n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6, jac=compute_gradient)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    return (np.clip(best_h, 0, 1), best_c5, n_points)
