import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    population = []

    # Track best solution
    best_h = population[0].copy()

    for h in population:
        current_c5 = evaluate(h)
        if current_c5 < best_c5:

            tournament = np.random.choice(population, size=tournament_size)
            parent1 = tournament[np.argmin([evaluate(h) for h in tournament])]

            tournament = np.random.choice(population, size=tournament_size)
            parent2 = tournament[np.argmin([evaluate(h) for h in tournament])]

            child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
