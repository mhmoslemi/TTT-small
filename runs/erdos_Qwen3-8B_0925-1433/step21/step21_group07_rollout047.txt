import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        shift = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus = i + shift
            term1 = 1.0 if 0 <= i_plus < n_points else 0.0
            i_minus = i - shift
            term2 = h[i_minus] if 0 <= i_minus < n_points else 0.0
            grad[i] = term1 - term2
        return grad
    
    max_iterations = 500
    start_time = time.time()
    h = h * target_sum / np.sum(h)
    learning_rate = 0.3
    momentum = 0.8
    decay_rate = 0.99
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        grad = compute_gradient(h)
        m = np.zeros_like(grad)
        v = np.zeros_like(grad)
        for i in range(n_points):
            m[i] = momentum * m[i] + (1 - momentum) * grad[i]
            v[i] = decay_rate * v[i] + (1 - decay_rate) * grad[i]**2
            biased_m = m[i] / (1 - momentum**( _ + 1))
            biased_v = v[i] / (1 - decay_rate**( _ + 1))
            h[i] = h[i] + learning_rate * biased_m / (np.sqrt(biased_v) + 1e-8)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h, best_c5 = h.copy(), current_c5
    
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},

        if res.success and res.fun < best_c5:
