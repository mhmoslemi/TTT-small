import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)

    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_h = h.copy()

    start_time = time.time()
    iteration = 0
    learning_rate = 0.03
    decay_rate = 0.999
    max_iterations = int(budget_s / 10)

    while time.time() - start_time < budget_s and iteration < max_iterations:
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        lag_shift = max_index - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + lag_shift < n_points:
                grad[i] += (1 - h[i + lag_shift])
            if 0 <= i - lag_shift < n_points:
                grad[i] -= h[i - lag_shift]
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        learning_rate *= decay_rate
        iteration += 1

    return (best_h, best_c5, n_points)
