from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def rescale(h):
        return np.clip(h * target_sum / np.sum(h), 0, 1)
    
    def generate_structured():
        structured = []
        for phase in [0, np.pi/4, np.pi/2]:
            positions = np.round(np.linspace(0, n_points-1, n_points//2 + 1) + phase).astype(int)
            h = np.zeros(n_points)
            for p in np.unique(positions % n_points):
                h[p] = 1.0
            structured.append(rescale(h))
        return structured
    
    def objective(h):
        return evaluate(h)
    
    def bounds():
        return [(0, 1) for _ in range(n_points)]
    
    def constraint(h):
        return np.sum(h) - target_sum
    
    best_h = np.array(initial_h_values, dtype=float)
    best_h = rescale(best_h)
    best_c5 = evaluate(best_h)
    
    start_time = time.time()

        result = differential_evolution(objective, bounds(), strategy='best1bin',

        if result.success:

                best_h, best_c5 = candidate, c5
    
    # Local refinement
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, best_h, method='SLSQP',
