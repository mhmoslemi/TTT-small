from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def fourier_smooth(h, cutoff=0.3):
        n = len(h)
        fft_h = np.fft.fft(h)
        freqs = np.fft.fftfreq(n)
        mask = np.abs(freqs) <= cutoff
        fft_h = fft_h * mask
        smoothed = np.fft.ifft(fft_h).real
        smoothed = np.clip(smoothed, 0, 1)
        return smoothed * target_sum / np.sum(smoothed)

    initial_h = np.array(initial_h_values, dtype=float)
    smoothed = fourier_smooth(initial_h)
    population = [smoothed]
    for _ in range(3):
        h = smoothed + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
    population = np.array(population)

    best_h = smoothed.copy()
    best_c5 = evaluate(best_h)

    start_time = time.time()
    pop_size = 20
    elite_count = 5
    generations = 200
    mutation_std = 0.05
    crossover_prob = 0.8
    no_improvement = 0

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        evals = np.array([evaluate(h) for h in population])
        sorted_indices = np.argsort(evals)
        elites = population[sorted_indices[:elite_count]]
        
        new_population = []
        for _ in range(pop_size):
            if np.random.rand() < crossover_prob:
                p1, p2 = elites[np.random.choice(elite_count, 2, replace=False)]
                idx = np.random.randint(1, n_points - 1)
                child = np.concatenate([p1[:idx], p2[idx:]])
            else:
                child = elites[np.random.choice(elite_count)].copy()
            noise = np.random.normal(0, mutation_std, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = np.array(new_population)

        for h in population:
            current_c5 = evaluate(h)
            if current_c5 < best_c5:

                best_h = h.copy()

    def obj(x):
        h = x * target_sum / np.sum(x)
