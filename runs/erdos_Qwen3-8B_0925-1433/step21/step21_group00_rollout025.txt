import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values, dtype=float)
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def obj(x):
        h = x * target_sum / np.sum(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    variations = [initial_h]
    for _ in range(3):
        variations.append(initial_h + np.random.normal(0, 0.1, n_points))
        variations.append(1.0 - initial_h)
        variations.append(np.roll(initial_h, 1))
        sparsified = initial_h.copy()
        sparsified[np.random.rand(n_points) < 0.5] = 0
        sparsified = sparsified * target_sum / np.sum(sparsified)
        variations.append(sparsified)
    population = [clip_normalize(v) for v in variations[:8]]
    
    start_time = time.time()
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        scores = []
        for p in population:
            scores.append(obj(p))
        idxs = np.argsort(scores)[:4]
        new_pop = []
        for i in range(4):
            p1, p2 = population[idxs[i]], population[idxs[(i+1)%4]]
            child = (p1 + p2) / 2 + np.random.normal(0, 0.1, n_points)
            new_pop.append(clip_normalize(child))
        population = new_pop

    best_h = min(population, key=obj)
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    best_h = res.x
    best_h = clip_normalize(best_h)
    c5_bound = obj(best_h)
    return (np.clip(best_h, 0, 1), c5_bound, n_points)
