import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_guesses = []

    # Parent guess
    parent = initial_h_values.copy()
    sum_parent = np.sum(parent)
    scale_parent = (n_points / 2) / sum_parent
    initial_guesses.append(parent * scale_parent)

    # Alternating pattern
    alt = np.zeros(n_points)
    for i in range(n_points):
        alt[i] = 1 if i % 2 == 0 else 0
    sum_alt = np.sum(alt)
    scale_alt = (n_points / 2) / sum_alt
    initial_guesses.append(alt * scale_alt)

    # Random pattern
    num_ones = int(np.floor(n_points / 2))
    rand = np.zeros(n_points)
    rand[:num_ones] = 1
    np.random.shuffle(rand)
    sum_rand = np.sum(rand)
    scale_rand = (n_points / 2) / sum_rand
    initial_guesses.append(rand * scale_rand)

    best_h = None
    best_c5 = float('inf')

    for guess in initial_guesses:
        guess_clipped = np.clip(guess, 0.0, 1.0)
        result = minimize(
            fun=lambda h: np.max(np.correlate(h, 1 - h, mode='full')) * dx,
            x0=guess_clipped,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}],
            options={'maxiter': 200, 'ftol': 1e-10, 'eps': 1e-8}
        )
