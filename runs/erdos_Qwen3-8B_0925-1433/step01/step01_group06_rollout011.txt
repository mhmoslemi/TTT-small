import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Predefined initial guess if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 160  # More detailed grid for better optimization
        h = np.zeros(n_points)
        # Create a structured initial guess with alternating blocks
        block_size = 4
        for i in range(n_points):
            if (i // block_size) % 2 == 0:
                h[i] = 1.0
            else:
                h[i] = 0.0
        # Normalize to meet integral constraint
        sum_h = np.sum(h)
        scale = (n_points / 2.0) / sum_h if sum_h != 0 else 1.0
        h *= scale
    n_points = len(h)
    dx = 2.0 / n_points

    def objective(h):
        cc = correlate(h, 1 - h, mode='full')
        max_cc = np.max(cc) * dx
        return max_cc

    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    x0 = h.copy()

    # Optimize with tighter tolerances and extended iterations
    options = {
        'maxiter': 300,
        'ftol': 1e-8,
        'eps': 1e-8,
        'disp': False,
        'gtol': 1e-9
    }
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, options=options)

    best_h = result.x
    cc = correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(cc) * dx

    # Final clipping, scaling, and constraint enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    scale = (n_points / 2.0) / np.sum(best_h) if np.sum(best_h) != 0 else 1.0
    best_h *= scale

    # Final check for constraint accuracy
    final_sum = np.sum(best_h) * dx
    if abs(final_sum - 1.0) > 1e-6:
        best_h = (best_h * (1.0 / final_sum) * 1.0) * (n_points / 2.0)

    return (best_h, c5_bound, n_points)
