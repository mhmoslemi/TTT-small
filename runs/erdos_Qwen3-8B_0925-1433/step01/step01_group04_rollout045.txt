import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 480
        h = np.random.rand(n_points)
        required_sum = n_points / 2
        h *= required_sum / np.sum(h)
    dx = 2.0 / n_points
    required_sum = n_points / 2
    current_h = h.copy()
    current_energy = np.max(correlate(current_h, 1 - current_h, mode='full') * dx)
    best_h = current_h.copy()
    best_energy = current_energy
    T = 1.0
    cooling_rate = 0.995
    max_iterations = 1000
    iteration = 0
    start_time = time.time()

    while iteration < max_iterations and time.time() - start_time < budget_s - 10:
        delta = np.random.normal(0, 0.05, n_points)
        h_candidate = current_h + delta
        h_candidate = np.clip(h_candidate, 0, 1)
        current_sum = np.sum(h_candidate)
        if current_sum == 0:
            h_candidate = np.zeros_like(h_candidate)
        else:
            scale = required_sum / current_sum
            h_candidate *= scale
        cc_candidate = correlate(h_candidate, 1 - h_candidate, mode='full')
        c_candidate = cc_candidate * dx
        candidate_energy = np.max(c_candidate)
        if candidate_energy < current_energy:
            current_h = h_candidate.copy()
            current_energy = candidate_energy
        else:
            prob = np.exp(-(candidate_energy - current_energy) / T)
            if np.random.rand() < prob:
                current_h = h_candidate.copy()
                current_energy = candidate_energy
        if candidate_energy < best_energy:
            best_h = h_candidate.copy()
            best_energy = candidate_energy
        T *= cooling_rate
        iteration += 1

    best_h = np.clip(best_h, 0, 1)
    current_sum = np.sum(best_h)
    if current_sum > 0:
        scale = required_sum / current_sum
        best_h *= scale
    cc_final = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(cc_final * dx)
    return (best_h, final_c5, n_points)
