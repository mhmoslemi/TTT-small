import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)

        n_points = 320
        h = np.random.rand(n_points)
        h *= (n_points / 2) / np.sum(h)
    dx = 2.0 / n_points
    required_sum = n_points / 2.0

    def compute_c5(h_flat):
        h = h_flat.reshape(n_points)
        cc = correlate(h, 1 - h, mode='full')
        return np.max(cc * dx)

    pop_size = 20
    h_pop = [h.copy() for _ in range(pop_size)]
    vel = np.random.uniform(-0.1, 0.1, (pop_size, n_points))
    personal_best = np.array([compute_c5(h) for h in h_pop])
    global_best = np.min(personal_best)
    best_idx = np.argmin(personal_best)

    def clip_and_scale(h):
        h = np.clip(h, 0, 1)
        scale = required_sum / np.sum(h)
        return h * scale

    start_time = time.time()
    for _ in range(5000):
        if time.time() - start_time > budget_s - 10:
            break
        new_vel = 0.8 * vel + 1.5 * np.random.rand(pop_size, n_points) * (np.array(h_pop) - h_pop) + 1.5 * np.random.rand(pop_size, n_points) * (np.array([global_best]*n_points) - h_pop)
        new_pos = h_pop + new_vel
        new_pos = np.array([clip_and_scale(p) for p in new_pos])
        new_c5 = np.array([compute_c5(p) for p in new_pos])
        for i in range(pop_size):
            if new_c5[i] < personal_best[i]:
                personal_best[i] = new_c5[i]
                h_pop[i] = new_pos[i]
        global_best = np.min(personal_best)
        best_idx = np.argmin(personal_best)
        h_pop, vel = new_pos, new_vel
    best_h = h_pop[best_idx]
    cc = correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(cc * dx)
    return (best_h, c5_bound, n_points)
