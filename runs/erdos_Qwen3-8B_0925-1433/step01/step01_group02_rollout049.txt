    h_random = np.zeros(n_points, dtype=np.float64)
    rng = np.random.default_rng(seed)
    indices = rng.choice(n_points, size=n_points // 2, replace=False)
    h_random[indices] = 1.0
    initial_guesses.append(h_random)

    scaled_guesses = []
    for guess in initial_guesses:
        sum_g = np.sum(guess)
        if sum_g != 0:
            scaled_guess = guess * (n_points / 2) / sum_g
        else:
            scaled_guess = guess.copy()
        scaled_guesses.append(scaled_guess)

    best_h = None
    best_c5 = float('inf')
    for guess in scaled_guesses:
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        def constraint_sum(h_vals):
            return np.sum(h_vals) - n_points / 2
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        result = minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-10}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    if best_h is not None:
        best_h = np.clip(best_h, 0.0, 1.0)
        sum_best = np.sum(best_h)
        if abs(sum_best - n_points / 2) > 1e-6:
            best_h = best_h * (n_points / 2) / sum_best

    return (best_h if best_h is not None else initial_h_values), best_c5, n_points
