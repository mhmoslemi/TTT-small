import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for efficiency
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        dx = 2.0 / n_points
        sum_h = np.sum(h) * dx
        scale = (n_points / 2.0) / sum_h if sum_h != 0 else 1.0
        h *= scale
    dx = 2.0 / n_points

    def objective(h):
        cc = correlate(h, 1 - h, mode='full')
        return np.max(cc) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Global optimization with DE
    result_de = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=20, tol=1e-6,
        mutation=(0.5, 1), recombination=0.8, maxiter=100, seed=seed
    )
    best_de = result_de.x

    # Clip & rescale to enforce constraint
    best_de = np.clip(best_de, 0.0, 1.0)
    sum_h = np.sum(best_de) * dx
    scale = (n_points / 2.0) / sum_h if sum_h != 0 else 1.0
    best_de *= scale

    # Local refinement with SLSQP
    x0 = best_de.copy()
    result_slsqp = minimize(
        objective, x0, method='SLSQP', bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
    )

    best_h = result_slsqp.x
    cc = correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(cc) * dx

    # Final clip & rescale
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h) * dx
    scale = (n_points / 2.0) / sum_h if sum_h != 0 else 1.0
    best_h *= scale

    return (best_h, c5_bound, n_points)
