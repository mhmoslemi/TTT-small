    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def grad_objective(h):

        max_idx = np.argmax(corr)
        grad = np.zeros_like(h)
        if max_idx < n_points:
            lag = max_idx - (n_points - 1)
            for i in range(n_points):
                j = i + lag
                if 0 <= j < n_points:
                    grad[i] += (1 - h[j]) * dx
                    grad[j] -= h[i] * dx
        else:
            lag = max_idx - (n_points - 1)
            for i in range(n_points):
                j = i - lag
                if 0 <= j < n_points:
                    grad[i] += (1 - h[j]) * dx
                    grad[j] -= h[i] * dx
        return grad

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
