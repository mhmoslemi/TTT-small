import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600, **kwargs):
    n_points = 50
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = initial_h_values if 'initial_h_values' in globals() else None

    if initial_h is not None:
        initial_h = np.array(initial_h, dtype=float)
    else:
        initial_h = np.random.rand(n_points).astype(float)

    if initial_h is not None:
        initial_sum = np.sum(initial_h)
        scaling = target_sum / initial_sum
        initial_h = np.clip(initial_h * scaling, 0.0, 1.0)
    else:
        initial_h = np.random.rand(n_points).astype(float)
        initial_sum = np.sum(initial_h)
        scaling = target_sum / initial_sum
        initial_h = np.clip(initial_h * scaling, 0.0, 1.0)

    def objective(h_vector):
        h = np.array(h_vector, dtype=np.float64)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.inf
        scaling = target_sum / current_sum
        h_scaled = h * scaling
        h_scaled = np.clip(h_scaled, 0.0, 1.0)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        c5 = np.max(corr) * dx
        return c5

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=25,
        maxiter=1000,
        mutation=(0.5, 1.0),
        recombination=0.9,
        tol=1e-6,
        seed=seed,
        polish=True
    )
    best_h = result.x
    h_clipped = np.clip(best_h, 0.0, 1.0)
    current_sum_clipped = np.sum(h_clipped)
    scaling_factor = target_sum / current_sum_clipped
    best_h_scaled = h_clipped * scaling_factor
    final_c5 = objective(best_h_scaled)
