import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 160
        h = np.random.uniform(0, 1, n_points)
        dx = 2.0 / n_points
        sum_h = np.sum(h) * dx
        if sum_h == 0:
            h = np.full(n_points, 0.5)
        else:
            scale = (1.0 / sum_h) * (n_points / 2.0)
            h *= scale
    n_points = len(h)
    dx = 2.0 / n_points

    def objective(h):
        cc = correlate(h, 1 - h, mode='full')
        max_cc = np.max(cc) * dx
        return max_cc

    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    x0 = h.copy()

    options = {'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-8, 'disp': False}
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, options=options)

    best_h = result.x
    cc = correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(cc) * dx

    best_h = np.clip(best_h, 0.0, 1.0)
    scale = (n_points / 2.0) / np.sum(best_h)
    best_h *= scale

    final_sum = np.sum(best_h) * dx
    if abs(final_sum - 1.0) > 1e-6:
        best_h = (best_h * (1.0 / final_sum) * 1.0) * (n_points / 2.0)

    return (best_h, c5_bound, n_points)
