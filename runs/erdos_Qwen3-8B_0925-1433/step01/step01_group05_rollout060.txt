    T = 1.0
    cooling_rate = 0.995
    iterations = 5000
    for _ in range(iterations):
        if time.time() - start_time > budget_s:
            break
        # Generate a new candidate
        perturb = np.random.normal(0, 0.1, size=n_points)

import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h *= scale
    dx = 2.0 / n_points

    def objective(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}

    def sa_step(h_curr):
        nonlocal best_c5, best_h
        T = 1.0
        cooling = 0.999
        for _ in range(1000):
            if time.time() - start_time > budget_s:
                return
            perturb = np.random.normal(0, 0.1, size=n_points)
            new_h = h_curr + perturb
            new_h = np.clip(new_h, 0.0, 1.0)
            sum_new = np.sum(new_h)
            if sum_new > 0:
                new_h *= (n_points / 2) / sum_new
            new_c5 = objective(new_h)
            if new_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - new_c5) / T):
                h_curr = new_h.copy()
                best_c5 = new_c5
            T *= cooling

    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    sa_step(best_h)

    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x

    return (best_h, best_c5, n_points)
