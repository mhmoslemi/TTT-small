import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Ensure initial_h_values meets the integral constraint
    init_sum = np.sum(initial_h_values)
    scale = (n_points / 2) / init_sum if init_sum != 0 else 1.0
    best_h = initial_h_values * scale
    best_c5 = _compute_c5(best_h, dx)
    
    start_time = time.time()
    max_amplitude = 0.2
    iteration = 0
    total_iter = 20
    
    while time.time() - start_time < budget_s and iteration < total_iter:
        iteration += 1
        # Generate sinusoidal perturbation
        perturb = max_amplitude * np.sin(2 * np.pi * np.arange(n_points) / n_points)
        guess = best_h + perturb
        guess = np.clip(guess, 0.0, 1.0)
        guess_sum = np.sum(guess)
        if guess_sum != 0:
            guess *= (n_points / 2) / guess_sum
        
        # Objective function
        def obj(x):
            h = x.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        # Constraints
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Perform optimization
        result = minimize(obj, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        # Reduce perturbation amplitude
        max_amplitude *= 0.95
    
    return (best_h, best_c5, n_points)

def _compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
