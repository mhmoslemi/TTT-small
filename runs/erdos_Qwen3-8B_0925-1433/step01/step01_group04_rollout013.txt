import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 480
        h = np.random.rand(n_points)
        required_sum = n_points / 2
        h *= required_sum / np.sum(h)
    required_sum = n_points / 2
    dx = 2.0 / n_points
    current_h = h.copy()
    best_c5 = np.max(correlate(current_h, 1 - current_h, mode='full') * dx)
    best_h = current_h.copy()
    temp = 1.0
    cooling_rate = 0.995
    max_iterations = 1000
    start_time = time.time()
    for _ in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break
        delta = np.random.normal(0, 0.05, n_points)
        delta -= np.mean(delta)
        h_candidate = current_h + delta
        h_candidate = np.clip(h_candidate, 0, 1)
        sum_candidate = np.sum(h_candidate)
        scale = required_sum / sum_candidate
        h_candidate *= scale
        cc = correlate(h_candidate, 1 - h_candidate, mode='full')
        new_c5 = np.max(cc * dx)
        if new_c5 < best_c5:
            current_h = h_candidate.copy()
            best_c5 = new_c5
        else:
            prob = np.exp((best_c5 - new_c5) / temp)
            if np.random.rand() < prob:
                current_h = h_candidate.copy()
                best_c5 = new_c5
        temp *= cooling_rate
    best_h = np.clip(current_h, 0, 1)
    sum_best = np.sum(best_h)
    best_h *= required_sum / sum_best
    return (best_h, best_c5, n_points)
