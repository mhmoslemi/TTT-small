import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    pop_size = 32
    max_gen = 200
    elite_ratio = 0.2
    mutation_rate = 0.2

    def normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full(n_points, 1./n_points)
        return h * (n_points / 2) / total

    # Initialize population
    population = [initial_h_values.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = normalize(h)
        population.append(h)

    best_h = initial_h_values.copy()
    best_c5 = float('inf')

    for gen in range(max_gen):
        # Evaluate fitness
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        
        # Update best
        min_idx = np.argmin(fitness)
        current_c5 = fitness[min_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[min_idx].copy()

        # Select elites
        elite_idx = np.argsort(fitness)[:int(pop_size * elite_ratio)]
        elites = [population[i] for i in elite_idx]

        # Generate new population
        new_pop = elites[:]
        while len(new_pop) < pop_size:
            # Crossover
            p1, p2 = np.random.choice(elites, 2, replace=False)
            child = (p1 + p2) / 2
            # Mutate
            if np.random.rand() < mutation_rate:
                child += np.random.normal(0, 0.05, n_points)
            new_pop.append(normalize(child))
        
        population = new_pop

        # Early stopping
        if gen % 10 == 0:
            if abs(best_c5 - current_c5) < 1e-6:
                break
    
    return (best_h, best_c5, n_points)
