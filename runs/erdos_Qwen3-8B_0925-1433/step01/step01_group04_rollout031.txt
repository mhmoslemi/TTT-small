from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    else:
        n_points = 320
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
    
    dx = 2.0 / n_points
    required_sum = n_points / 2
    
    # PSO parameters
    num_particles = 20
    inertia = 0.8
    c1, c2 = 1.5, 1.5
    vel_clip = 0.1
    
    # Initialize population
    particles = np.clip(h[np.newaxis, :], 0, 1).repeat(num_particles, axis=0)
    velocities = np.random.uniform(-vel_clip, vel_clip, (num_particles, n_points))
    
    # Bests
    personal_best = particles.copy()
    personal_scores = np.full(num_particles, np.inf)
    global_best = particles[0].copy()
    global_score = np.inf
    
    def compute_c5(h_flat):
        cc = correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(cc * dx)
    
    start_time = time.time()
    for _ in range(5000):
        if time.time() - start_time > budget_s - 10:
            break
        
        # Evaluate fitness
        scores = np.array([compute_c5(p) for p in particles])
        for i in range(num_particles):
            if scores[i] < personal_scores[i]:
                personal_scores[i] = scores[i]
                personal_best[i] = particles[i].copy()
        
        # Update global best
        if np.min(scores) < global_score:
            global_score = np.min(scores)
            global_best = particles[np.argmin(scores)].copy()
        
        # Update velocities and positions
        r1, r2 = np.random.rand(num_particles), np.random.rand(num_particles)
        velocities = inertia * velocities + c1 * r1 * (personal_best - particles) + c2 * r2 * (global_best[np.newaxis, :] - particles)
        velocities = np.clip(velocities, -vel_clip, vel_clip)
        particles += velocities
        
        # Clip and normalize
        particles = np.clip(particles, 0, 1)
        for i in range(num_particles):
            particles[i] = particles[i] * required_sum / np.sum(particles[i])
    
    # Final evaluation
    best_h = global_best
    cc = correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(cc * dx)
    return (best_h, c5_bound, n_points)
