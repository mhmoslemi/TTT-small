        max_corr = np.max(corr)
        c5 = max_corr * dx
        sum_h = np.sum(h)
        penalty = 1e3 * (sum_h - n_points / 2)**2
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result_de = differential_evolution(objective_de, bounds, strategy='best1bin', popsize=20, tol=1e-6, mutation=(0.5, 1), recombination=0.9, maxiter=100)
    best_h_de = result_de.x

    def objective_slsqp(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    cons_slsqp = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})
