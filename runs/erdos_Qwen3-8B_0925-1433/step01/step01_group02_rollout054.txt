        best_h[0] = 1.0
    else:
        best_h = best_h * (n_points / 2) / sum_h

    c5_bound = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * (2.0 / n_points)
    return best_h, c5_bound, n_points
