import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}

    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x

    elapsed = time.time() - start_time
    remaining = budget_s - elapsed

    for _ in range(8):
        if remaining < 10:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        grad = np.zeros_like(best_h, dtype=np.float64)
        for i in range(n_points):
            if 0 <= i + max_k < n_points:
                grad[i] += (1 - best_h[i + max_k])

        result = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)

            best_h = result.x
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
