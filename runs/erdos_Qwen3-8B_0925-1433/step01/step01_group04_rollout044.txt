import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)

        required_sum = n_points / 2
        h *= required_sum / np.sum(h)
    dx = 2.0 / n_points
    required_sum = n_points / 2

    def compute_c5(h_flat):
        cc = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(cc * dx)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    T = 1.0
    cooling_rate = 0.995
    max_iter = 1000
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
        delta = np.random.normal(0, 0.05, n_points)
        h_candidate = h + delta
        h_candidate = np.clip(h_candidate, 0, 1)
        current_sum = np.sum(h_candidate)
        scale = required_sum / current_sum
        h_candidate *= scale
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            h = h_candidate.copy()
            best_c5 = c5_candidate
        else:
            prob = np.exp(-(c5_candidate - best_c5) / T)
            if np.random.rand() < prob:
                h = h_candidate.copy()
                best_c5 = c5_candidate
        T *= cooling_rate

    final_sum = np.sum(final_h)
    final_h *= required_sum / final_sum
    final_c5 = compute_c5(final_h)
    return (final_h, final_c5, n_points)
