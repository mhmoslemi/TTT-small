        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        # Scale to meet integral constraint
        perturbed_sum = np.sum(perturbed_h)
        scale_factor = (n_points / 2) / perturbed_sum
        perturbed_h = perturbed_h * scale_factor
        
        # Optimization
        result = minimize(objective, perturbed_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8)
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                prev_best_c5 = best_c5
        
        # Check for convergence
        if abs(best_c5 - prev_best_c5) < 1e-6:
            break
        
        perturbation_amplitude *= 0.99  # Adaptive amplitude reduction
    
    return (best_h, best_c5, n_points)
