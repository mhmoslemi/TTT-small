        m = np.argmax(corr)
        lag = m - (n_points - 1)
        
        # Compute gradient
        g = np.zeros(n_points)
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                g[i] = 1.0 - h[j]
        
        # Gradient descent
        h_new = h - learning_rate * g
        h_new = np.clip(h_new, 0, 1)
        sum_new = np.sum(h_new)
        if sum_new == 0:
            h_new = np.ones(n_points) * 0.5
        else:
            scale = (n_points / 2) / sum_new
            h_new = np.clip(h_new * scale, 0, 1)
        
        # Check improvement
        c5_new = compute_c5(h_new, dx, n_points)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        else:
            learning_rate *= 0.99
        
        # Check time
        if time.time() - start_time > budget_s:
            break
        h = h_new
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
