import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    import numpy as np
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    # Generate initial guesses
    initial_guesses = [initial_h_values]
    # Binary alternating pattern
    binary_pattern = np.zeros(n_points, dtype=np.float64)
    for i in range(n_points):
        if i % 2 == 0:
            binary_pattern[i] = 1.0
        else:
            binary_pattern[i] = 0.0
    initial_guesses.append(binary_pattern)
    # Scale each guess
    scaled_guesses = []
    for guess in initial_guesses:
        sum_guess = np.sum(guess)
        if sum_guess == 0:
            scale_factor = 1.0
        else:
            scale_factor = (n_points / 2.0) / sum_guess
        scaled_guess = guess * scale_factor
        scaled_guess = np.clip(scaled_guess, 0.0, 1.0)
        scaled_guesses.append(scaled_guess)
    # Prepare for optimization
    def objective(h_values):
        c = np.correlate(h_values, 1.0 - h_values, mode='full')
        return np.max(c) * dx
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2.0
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_result = None
    best_c5 = float('inf')
    for guess in scaled_guesses:
        # Run optimization
        result = minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-10}
        )
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_result = result
    if best_result is None:
        # Fallback to initial_guesses[0] if no success
        best_result = minimize(
            fun=objective,
            x0=initial_guesses[0],
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-10}
        )
    # Final check to ensure the best is valid
    final_h = best_result.x
    # Ensure h is within bounds and sum is correct
    final_h = np.clip(final_h, 0.0, 1.0)
    if np.abs(np.sum(final_h) - n_points / 2.0) > 1e-6:
        raise ValueError("Final sum does not match constraint")
    return final_h, best_c5, n_points
