        return np.sum(x[m:]) - 1.0

    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    for j in range(m-1):
        def constraint_overlap(x, j=j):
            a_prev = x[:m]
            l_prev = x[m:]
            return a_prev[j+1] - (a_prev[j] + l_prev[j])
        constraints.append({'type': 'ineq', 'fun': lambda x, j=j: constraint_overlap(x, j)})
    for j in range(m):
        def constraint_end(x, j=j):
            a_prev = x[:m]
            l_prev = x[m:]
            return 2.0 - (a_prev[j] + l_prev[j])
        constraints.append({'type': 'ineq', 'fun': lambda x, j=j: constraint_end(x, j)})

    bounds = [(0.0, 2.0) for _ in range(2*m)]
    x0 = np.concatenate([a, l])
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 200, 'ftol': 1e-8})
    best_x = result.x
    a_opt = best_x[:m]
    l_opt = best_x[m:]
    intervals_opt = list(zip(a_opt, a_opt + l_opt))
    intervals_opt.sort()
    h_values = np.zeros(n_points)
    for (a_i, b_i) in intervals_opt:
        start = int(np.floor(a_i / dx))
        end = int(np.ceil(b_i / dx))
        if start < n_points and end > 0:
            h_values[start:end] = 1.0
    cc = correlate(h_values, 1 - h_values, mode='full')
    c5_bound = np.max(cc) * dx
    h_values = np.clip(h_values, 0.0, 1.0)
    sum_h = np.sum(h_values) * dx
    if abs(sum_h - 1.0) > 1e-6:
        scale = (1.0 / sum_h) * (n_points / 2.0)
        h_values *= scale
    return (h_values, c5_bound, n_points)
