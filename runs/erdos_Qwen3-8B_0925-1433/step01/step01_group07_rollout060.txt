import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_sum = np.sum(initial_h_values)
    scale_factor = (n_points / 2.0) / initial_sum
    h = initial_h_values * scale_factor
    best_h = h.copy()
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    
    start_time = time.time()
    temp = 1.0
    while time.time() - start_time < budget_s:
        noise = np.random.normal(0, 0.1 * temp, size=n_points)
        new_h = best_h + noise
        new_h = np.clip(new_h, 0.0, 1.0)
        sum_new = np.sum(new_h)
        if sum_new == 0:
            continue
        scale = (n_points / 2.0) / sum_new
        new_h_scaled = new_h * scale
        if np.max(new_h_scaled) > 1.0:
            continue
        new_h = new_h_scaled
        corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(corr) * dx
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
        else:
            delta = new_c5 - best_c5
            prob = np.exp(-delta / temp)
            if np.random.rand() < prob:
                best_h = new_h.copy()
                best_c5 = new_c5
        temp *= 0.99
        if temp < 1e-5:
            break
