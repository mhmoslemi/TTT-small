import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraints = ({'type': 'eq', 'fun': lambda h_vec: np.sum(h_vec) - n_points / 2})
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    for _ in range(10):
        # Generate structured sinusoidal perturbation
        freq = np.random.randint(1, 4)
        amplitude = np.random.uniform(0.01, 0.1)
        perturb = amplitude * np.sin(2 * np.pi * np.arange(n_points) / n_points * freq)
        initial_guess = best_h + perturb
        initial_guess = np.clip(initial_guess, 0.0, 1.0)
        
        # Rescale to meet constraint
        sum_guess = np.sum(initial_guess)
        if np.abs(sum_guess - n_points / 2) > 1e-6:
            scale = (n_points / 2) / sum_guess
            initial_guess *= scale
        
        # Optimization with L-BFGS-B
        result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-6)
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return (best_h, best_c5, n_points)
