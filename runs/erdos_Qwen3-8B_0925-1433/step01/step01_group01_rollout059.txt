import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600, **kwargs):
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points).astype(float)
    if initial_h is not None:
        initial_h = np.clip(initial_h, 0.0, 1.0)
        initial_sum = np.sum(initial_h)
        scaling = target_sum / initial_sum
        initial_h *= scaling
    else:
        initial_h = np.random.rand(n_points).astype(float)
        initial_h = np.clip(initial_h, 0.0, 1.0)
        initial_sum = np.sum(initial_h)
        scaling = target_sum / initial_sum
        initial_h *= scaling

    def objective_de(h_vector):
        h = np.array(h_vector, dtype=np.float64)
        current_sum = np.sum(h)
        scaling = target_sum / current_sum if current_sum != 0 else 1.0
        h_scaled = h * scaling
        h_scaled = np.clip(h_scaled, 0.0, 1.0)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_de = differential_evolution(
        objective_de, bounds, strategy='best1bin', popsize=50, maxiter=300,
        mutation=(0.6, 0.9), recombination=0.8, tol=1e-6, seed=seed, polish=True
    )
    best_h_de = result_de.x
    best_h_de_clipped = np.clip(best_h_de, 0.0, 1.0)
    sum_clipped = np.sum(best_h_de_clipped)
    scaling_final = target_sum / sum_clipped
    best_h_de_scaled = best_h_de_clipped * scaling_final

    def objective_local(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h_clipped)
        penalty = 1e5 * np.abs(sum_h - target_sum)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        alpha = 100
        soft_max = np.log(np.sum(np.exp(alpha * corr))) / alpha
        return soft_max + penalty

    result_local = minimize(objective_local, best_h_de_scaled, method='L-BFGS-B', bounds=[(0.0, 1.0)]*n_points,
                           options={'maxiter': 200, 'ftol': 1e-8})
    best_h_local = result_local.x
    best_h_local_clipped = np.clip(best_h_local, 0.0, 1.0)
    sum_local = np.sum(best_h_local_clipped)
    scaling_final_local = target_sum / sum_local
    best_h_local_scaled = best_h_local_clipped * scaling_final_local

    final_corr = np.correlate(best_h_local_scaled, 1 - best_h_local_scaled, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (best_h_local_scaled, final_c5, n_points)
