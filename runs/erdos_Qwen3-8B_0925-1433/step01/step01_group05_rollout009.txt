import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective_de(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    # DE phase
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    de_result = differential_evolution(
        objective_de, bounds, strategy='best1bin', popsize=20, maxiter=150,
        tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed, disp=False
    )
    best_h_de = de_result.x
    sum_h = np.sum(best_h_de)
    scale_factor = (n_points / 2) / np.clip(sum_h, 1e-12, None)
    best_h_de = np.clip(best_h_de * scale_factor, 0, 1)
    current_c5 = np.max(np.correlate(best_h_de, 1 - best_h_de, mode='full')) * dx
    if current_c5 < best_c5:
        best_h = best_h_de
        best_c5 = current_c5

    # SLSQP refinement
    def objective_slsqp(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    result_slsqp = minimize(
        objective_slsqp, best_h, method='SLSQP', bounds=bounds_slsqp,
        constraints=cons, tol=1e-8, options={'maxiter': 500}
    )
    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun

    elapsed = time.time() - start_time
    if elapsed > budget_s - 100:
        return (best_h, best_c5, n_points)

    # Local search with perturbations
    for _ in range(3):
        if elapsed > budget_s - 50:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        top_lags = np.argsort(corr)[-5:]
        perturb = np.zeros_like(best_h)
        for lag in top_lags:
            if lag > n_points - 1:
                continue
            for i in range(n_points):
                x = i * dx
                x_k = x + (lag - n_points / 2) * dx
                if 0 <= x_k <= 2:
                    perturb[i] += np.exp(-((x - 0.5) ** 2 + (x_k - 0.5) ** 2))
        perturb *= 0.05
        perturbed = best_h + perturb
        perturbed = np.clip(perturbed, 0, 1)
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed > 0:
            perturbed *= (n_points / 2) / sum_perturbed
        result = minimize(objective_slsqp, perturbed, method='SLSQP',
                          bounds=bounds_slsqp, constraints=cons, tol=1e-6)
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        elapsed = time.time() - start_time

    return (best_h, best_c5, n_points)
