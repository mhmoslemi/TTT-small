    dx = 2.0 / n_points
    required_sum = n_points / 2

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        cc = correlate(h, 1 - h, mode='full')
        c_values = cc * dx
        return np.max(c_values)

    def constraint_eq(h_flat):
        return np.sum(h_flat.reshape(n_points)) - required_sum

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    result = minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_eq}],
        options={'maxiter': 500, 'ftol': 1e-8, 'disp': False}
    )
