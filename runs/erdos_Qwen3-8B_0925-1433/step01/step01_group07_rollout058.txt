import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_sum = np.sum(initial_h_values)
    if abs(initial_sum - n_points / 2) > 1e-6:
        scale = (n_points / 2.0) / initial_sum
        initial_h_values = initial_h_values * scale

    n_particles = 50
    particles = [initial_h_values.copy() + np.random.normal(0, 0.05, n_points) for _ in range(n_particles)]
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(n_particles)]

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    personal_best = [p.copy() for p in particles]
    personal_c5 = [evaluate_c5(p) for p in particles]
    best_h = initial_h_values.copy()
    best_c5 = evaluate_c5(best_h)

    w = 0.8
    c1 = c2 = 2.0
    max_iter = 500
    iter_count = 0
    start_time = time.time()

    while iter_count < max_iter and time.time() - start_time < budget_s:
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i] * w + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (best_h - particles[i])
            velocities[i] = vel
            new_pos = particles[i] + vel
            new_pos = np.clip(new_pos, 0.0, 1.0)
            new_sum = np.sum(new_pos)
            if new_sum != 0:
                new_pos = new_pos * (n_points / 2.0) / new_sum
            particles[i] = new_pos
            current_c5 = evaluate_c5(particles[i])
            if current_c5 < personal_c5[i]:
                personal_c5[i] = current_c5
                personal_best[i] = particles[i].copy()
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = particles[i].copy()
        w = max(0.4, w - 0.001)
        iter_count += 1

    return (best_h, best_c5, n_points)
