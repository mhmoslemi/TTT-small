from scipy.optimize import minimize
import time

    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}

    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x

    elapsed = time.time() - start_time
    remaining = budget_s - elapsed

    for _ in range(5):
        if remaining < 10:
            break
