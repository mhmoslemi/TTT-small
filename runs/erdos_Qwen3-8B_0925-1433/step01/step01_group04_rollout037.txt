    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 480
        h = np.random.rand(n_points)
        required_sum = n_points / 2
        h *= required_sum / np.sum(h)
    dx = 2.0 / n_points
    required_sum = n_points / 2

    def compute_c5(h_flat):
        h = h_flat.reshape(n_points)
        cc = correlate(h, 1 - h, mode='full')
        return np.max(cc * dx)

    # Initialize best_h
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Parameters for simulated annealing
    T = 1.0  # Starting temperature
    cooling_rate = 0.995
    max_iterations = 1000
    start_time = time.time()

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break

        # Generate perturbation
        i = np.random.randint(n_points)
        j = np.random.randint(n_points)
        while j == i:
            j = np.random.randint(n_points)
        delta = np.random.uniform(-0.1, 0.1) * T
        h_candidate = h.copy()
        h_candidate[i] += delta
        h_candidate[j] -= delta

        # Clip to [0, 1]
        h_candidate = np.clip(h_candidate, 0, 1)

        # Compute sum of clipped h_candidate
        sum_clipped = np.sum(h_candidate)
        if sum_clipped == 0:
            continue  # avoid division by zero, though unlikely

        # Scale to required_sum
        scaling_factor = required_sum / sum_clipped
        h_candidate *= scaling_factor
        h_candidate = np.clip(h_candidate, 0, 1)

        # Compute new_c5
        new_c5 = compute_c5(h_candidate)

        # Acceptance probability
        if new_c5 < best_c5:
            h = h_candidate
            best_c5 = new_c5
        else:
            # Probability to accept worse solution
            prob = np.exp( (best_c5 - new_c5) / T )
            if np.random.rand() < prob:

        h *= scaling_factor

    best_c5 = compute_c5(h)

    return (h, best_c5, n_points)
