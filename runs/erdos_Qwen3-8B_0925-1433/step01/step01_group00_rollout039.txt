import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    h0 = initial_h_values.copy()
    dx = 2.0 / n_points
    required_ones = n_points / 2

    def generate_initial_guesses():
        guesses = [h0]
        # Periodic pattern with period 3
        h_per = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                h_per[i] = 1.0
        current_ones = np.sum(h_per)
        diff = required_ones - current_ones
        if diff > 0:
            for i in range(n_points):
                if h_per[i] == 0:
                    h_per[i] = 1.0
                    diff -= 1
                    if diff == 0:
                        break
        elif diff < 0:
            for i in range(n_points):
                if h_per[i] == 1:
                    h_per[i] = 0.0
                    diff += 1
                    if diff == 0:
                        break
        guesses.append(h_per)
        # Random guess
        indices = np.random.choice(n_points, size=int(required_ones), replace=False)
        h_rand = np.zeros(n_points)
        h_rand[indices] = 1.0
        if required_ones > int(required_ones):
            idx = np.random.choice(n_points, size=1, replace=False)
            h_rand[idx] = 0.5
        h_rand = h_rand * (required_ones / np.sum(h_rand))
        guesses.append(h_rand)
        # Jittered guess
        h_jitter = h0 + 0.05 * np.random.randn(n_points)
        h_jitter = np.clip(h_jitter, 0, 1)
        h_jitter = h_jitter * (required_ones / np.sum(h_jitter))
        guesses.append(h_jitter)
        # Additional guess with spacing
        h_space = np.zeros(n_points)
        for i in range(n_points):
            if i % 4 == 0 or i % 5 == 0:
                h_space[i] = 1.0
        current_ones = np.sum(h_space)
        diff = required_ones - current_ones
        if diff > 0:
            for i in range(n_points):
                if h_space[i] == 0:
                    h_space[i] = 1.0
                    diff -= 1
                    if diff == 0:
                        break
        elif diff < 0:
            for i in range(n_points):
                if h_space[i] == 1:
                    h_space[i] = 0.0
                    diff += 1
                    if diff == 0:
                        break
        guesses.append(h_space)
        return guesses[:5]

    initial_guesses = generate_initial_guesses()

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - required_ones}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h0.copy()
    best_c5 = obj(h0)

    for initial_h in initial_guesses:
        result = minimize(obj, initial_h, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'ftol': 1e-8, 'maxiter': 300, 'disp': False})
        if result.success:
            current_h = result.x
            current_c5 = obj(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        else:
            current_h = initial_h
            current_c5 = obj(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5

    if np.any(best_h < 0) or np.any(best_h > 1):
        best_h = h0
        best_c5 = obj(h0)

    return (best_h, best_c5, n_points)
