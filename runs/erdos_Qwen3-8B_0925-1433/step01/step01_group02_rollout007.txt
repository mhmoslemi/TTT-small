    for _ in range(max_iter):
        np.random.shuffle(population)
        curr_best = population[0]
        curr_corr = np.correlate(curr_best, 1 - curr_best, mode='full')
        curr_obj = np.max(curr_corr) * dx
        if curr_obj < best_c5:
            best_h = curr_best.copy()
            best_c5 = curr_obj
        for i in range(pop_size):
            a, b, c = population[np.random.choice(pop_size, 3, replace=False)]
            mutant = a + F * (b - c)
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            trial = np.clip(trial, 0, 1)
            trial_sum = np.sum(trial)
            if trial_sum == 0:
                trial = np.zeros_like(trial) + 0.5
            else:
                trial = trial * (n_points / 2) / trial_sum
            corr = np.correlate(trial, 1 - trial, mode='full')
            obj = np.max(corr) * dx
            if obj < np.max(np.correlate(population[i], 1 - population[i], mode='full')) * dx:
                population[i] = trial
            if obj < best_c5:
                best_h = trial.copy()
                best_c5 = obj
    return best_h, best_c5, n_points
