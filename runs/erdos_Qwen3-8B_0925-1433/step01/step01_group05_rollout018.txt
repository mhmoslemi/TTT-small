import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def rescale_h(h):
        total = np.sum(h)
        if total == 0:
            return h
        scale = (n_points / 2) / total
        return h * scale

    def perturb_h(h, std):
        perturb = np.random.normal(0, std, n_points)
        perturbed = h + perturb
        perturbed = np.clip(perturbed, 0, 1)
        return rescale_h(perturbed)

    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]
    for i in range(population_size):
        population[i] = perturb_h(population[i], 0.1)

    c5_values = np.array([compute_c5(h) for h in population])
    best_index = np.argmin(c5_values)
    best_h = population[best_index].copy()
    best_c5 = c5_values[best_index]

    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        tournament = np.random.choice(range(population_size), size=5, replace=False)
        tournament_c5 = c5_values[tournament]
        parent_idx = tournament[np.argmin(tournament_c5)]
        parent = population[parent_idx]

        if np.random.rand() < 0.2:
            mutated = perturb_h(parent, 0.15)
        else:
            mutated = parent.copy()

        worst_idx = np.argmax(c5_values)
        population[worst_idx] = mutated
        c5_values[worst_idx] = compute_c5(mutated)

        if c5_values[worst_idx] < best_c5:
