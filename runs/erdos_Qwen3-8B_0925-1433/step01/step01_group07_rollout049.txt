import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Scale initial h_values to meet the sum constraint
    initial_sum = np.sum(initial_h_values)
    scale_factor = (n_points / 2.0) / initial_sum
    h_initial = initial_h_values * scale_factor
    h_initial = np.clip(h_initial, 0.0, 1.0)
    new_sum = np.sum(h_initial)
    scale_factor_2 = (n_points / 2.0) / new_sum
    h_initial = h_initial * scale_factor_2
    best_h = h_initial.copy()
    best_c5 = objective(best_h, n_points, dx)
    
    start_time = time.time()
    max_amp = 0.15
    amp_decay = 0.995
    max_iterations = 100
    for iteration in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        i = np.arange(n_points)
        # Generate perturbation
        perturb = max_amp * np.sin(2 * np.pi * i / n_points)
        perturbed_h = best_h + perturb
        # Clip and rescale
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_sum = np.sum(perturbed_h)
        scale = (n_points / 2.0) / perturbed_sum
        perturbed_h = perturbed_h * scale
        # Optimization
        result = minimize(objective, perturbed_h, args=(n_points, dx), method='L-BFGS-B',
                          bounds=[(0.0, 1.0) for _ in range(n_points)],
                          constraints=[{'type': 'eq', 'fun': lambda h_vec: np.sum(h_vec) - n_points / 2}],
                          tol=1e-6)
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        max_amp *= amp_decay
    
    return (best_h, best_c5, n_points)

def objective(h_vec, n_points, dx):
    h = h_vec.reshape(n_points)
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
