    bounds_SLSQP = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result_slsqp = minimize(obj_SLSQP, best_h_de, method='SLSQP', bounds=bounds_SLSQP, constraints=cons, tol=1e-8, options={'maxiter': 200})
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed

    best_h_slsqp = result_slsqp.x
    best_c5_slsqp = obj_SLSQP(best_h_slsqp)

    best_h = best_h_de if best_c5_de < best_c5_slsqp else best_h_slsqp
    best_c5 = best_c5_de if best_c5_de < best_c5_slsqp else best_c5_slsqp

    best_c5_initial = obj_DE(initial_h)
    if best_c5 < best_c5_initial:
        return (best_h, best_c5, n_points)
    else:
        return (initial_h, best_c5_initial, n_points)
