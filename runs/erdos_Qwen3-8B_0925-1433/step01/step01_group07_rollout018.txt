import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points
    bounds = [(0.0, 1.0) for _ in range(n_points)]

        corr = np.correlate(h, 1 - h, mode='full')

    population_size = 50
    population = [initial_h_values.copy() for _ in range(population_size)]
    for i in range(population_size):
        perturb = np.random.normal(0, 0.05, n_points)
        cand = initial_h_values + perturb
        cand = np.clip(cand, 0, 1)
        total = np.sum(cand)
        cand *= (n_points / 2) / total
        population[i] = cand

    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    start_time = time.time()

    for gen in range(50):
        scores = [(obj(cand), cand) for cand in population]
        scores.sort()
        best_c5 = min(best_c5, scores[0][0])
        best_h = scores[0][1].copy()

        # Select parents

        new_pop = []

                alpha = np.random.rand()

                new_pop.append(child)
