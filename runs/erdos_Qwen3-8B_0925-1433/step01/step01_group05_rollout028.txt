import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Rescale initial_h
    sum_initial = np.sum(initial_h)
    if np.abs(sum_initial - n_points / 2) > 1e-8:
        initial_h = initial_h * (n_points / 2) / sum_initial

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    # Simulated Annealing
    T = 1.0
    cooling_rate = 0.995
    n_sa_steps = 500
    for step in range(n_sa_steps):
        perturb = np.random.normal(0, 0.1, n_points)
        new_h = np.clip(best_h + perturb, 0.0, 1.0)
        new_sum = np.sum(new_h)
        new_h = new_h * (n_points / 2) / new_sum
        new_c5 = objective(new_h)
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h = new_h
            best_c5 = new_c5
        T *= cooling_rate
        elapsed = time.time() - start_time
        if elapsed > budget_s - 100:
            break

    # SLSQP refinement
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x

    return (best_h, best_c5, n_points)
