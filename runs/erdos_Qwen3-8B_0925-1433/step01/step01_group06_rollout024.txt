    def constraint_sum(h_vals):
        return np.sum(h_vals) - required_sum
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_slsqp = minimize(objective, h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8, 'disp': False})
    best_h_slsqp = result_slsqp.x
    def objective_de(candidate):
        candidate = np.clip(candidate, 0.0, 1.0)
        sum_candidate = np.sum(candidate)
        if np.isclose(sum_candidate, 0):
            scaled = np.zeros_like(candidate)
        else:
            scale = required_sum / sum_candidate
            scaled = candidate * scale
        cc = np.correlate(scaled, 1 - scaled, mode='full')
        return np.max(cc) * dx
    result_de = differential_evolution(objective_de, bounds, strategy='best1bin', popsize=20, maxiter=100, tol=1e-5, seed=seed)
    best_h_de = result_de.x
    best_h_de = np.clip(best_h_de, 0.0, 1.0)
    sum_h_de = np.sum(best_h_de)
    if np.isclose(sum_h_de, 0):
        best_h_final = np.full(n_points, 0.5)
    else:
        scale_final = required_sum / sum_h_de
        best_h_final = best_h_de * scale_final
    final_c5 = compute_c5(best_h_final)
    return (best_h_final, final_c5, n_points)
