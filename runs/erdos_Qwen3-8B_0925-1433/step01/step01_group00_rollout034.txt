import numpy as np
from scipy.optimize import minimize
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h0 = initial_h_values.copy()
    else:
        n_points = 200
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1
        h0 = np.roll(h0, n_points//4)
    n_points = len(h0)
    dx = 2.0 / n_points
    sum_h = n_points / 2

    def obj(h):
        corr = signal.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_and_rescale(h):
        h = np.clip(h, 0, 1)
        h = h * sum_h / np.sum(h)
        return h

    def eval_obj(h):
        return obj(clip_and_rescale(h))

    def generate_patterns():
        patterns = []
        if 'initial_h_values' in globals():
            patterns.append(initial_h_values.copy())
        patterns.append(np.random.uniform(0.8, 1.2, n_points))
        patterns.append(np.tile([1, 0, 0], n_points//3))
        patterns.append(0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n_points) / n_points))
        patterns.append(np.random.choice([0, 1], size=n_points, p=[0.4, 0.6]))
        return [clip_and_rescale(p) for p in patterns]

    candidates = generate_patterns()
    best_h = candidates[0]
    best_c5 = obj(best_h)

    for init in candidates[1:]:
        h = init.copy()
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - sum_h}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = minimize(eval_obj, h, method='COBYLA', bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'disp': False})
        h_res = result.x
        if result.success:
            c5 = obj(h_res)
            if c5 < best_c5:
                best_h = h_res
                best_c5 = c5

    best_h = clip_and_rescale(best_h)
    final_c5 = obj(best_h)
    return (best_h, final_c5, n_points)
