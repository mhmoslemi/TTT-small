import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    pop_size = 50
    elites = int(0.2 * pop_size)
    mut_rates = [0.5, 0.3, 0.2]
    max_gen = 1000
    early_stop = 5
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    start_time = time.time()
    
    def eval_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def norm(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / s
    
    # Initialize population
    pop = np.vstack([initial_h_values.copy() + 0.1 * np.random.randn(n_points) for _ in range(pop_size)])
    pop = np.vstack([norm(h) for h in pop])
    
    for gen in range(max_gen):
        c5s = np.array([eval_c5(h) for h in pop])
        elite_idx = np.argsort(c5s)[:elites]
        elites = pop[elite_idx]
        
        # Generate offspring
        offspring = []
        for _ in range(pop_size - elites.shape[0]):
            p1, p2 = np.random.choice(elites, 2, replace=False)
            alpha = np.maximum(0.5 * np.random.rand(n_points), 0.01)
            beta = p1 * (1 - alpha) + p2 * alpha
            beta = norm(beta)
            offspring.append(beta)
        
        # Apply mutations
        for h in offspring:
            if np.random.rand() < 0.5:
                h += np.random.normal(0, 0.05, n_points)
            elif np.random.rand() < 0.3:
                h = np.where(np.random.rand(n_points) < 0.1, 1 - h, h)
            elif np.random.rand() < 0.2:
                h += 0.1 * np.sin(2 * np.pi * np.random.rand(n_points))
            h = norm(h)
        
        pop = np.vstack([elites, offspring])
        c5s = np.array([eval_c5(h) for h in pop])
        best_idx = np.argmin(c5s)
        best_h = pop[best_idx]
        best_c5 = c5s[best_idx]
        
        if time.time() - start_time > budget_s:
            break
        
        # Local refinement
        if gen % 10 == 0:
            h = best_h.copy()
            def obj(x):
                return eval_c5(x)
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, options={'ftol': 1e-9, 'maxiter': 100})
            if res.success:
                best_h = res.x
                best_c5 = eval_c5(best_h)
    
    return (best_h, best_c5, n_points)
