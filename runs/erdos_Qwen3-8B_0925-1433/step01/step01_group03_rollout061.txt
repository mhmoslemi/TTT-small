            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.ones(n_points) * 0.5
            else:
                scale = (n_points / 2) / sum_h
                h = np.clip(h * scale, 0, 1)
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
