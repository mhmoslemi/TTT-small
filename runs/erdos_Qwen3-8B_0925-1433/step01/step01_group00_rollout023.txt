import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from numpy.random import normal, uniform, choice

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = globals().get('initial_h_values', np.random.rand(95))
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def clip_and_scale(h):
        h = np.clip(h, 0, 1)
        scale = target_sum / np.sum(h)
        return h * scale
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def mutate(h, mu=0.1, sigma=0.05, prob=0.5):
        h = h.copy()
        if np.random.rand() < prob:
            h += sigma * normal(size=n_points)
        else:
            idx = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
            h[idx] = 1 - h[idx]
        return clip_and_scale(h)
    
    def crossover(parent1, parent2, eta=10.0):
        alpha = np.random.rand(n_points)
        beta = np.random.rand(n_points)
        child = (alpha * parent1 + beta * parent2) / (alpha + beta)
        return clip_and_scale(child)
    
    def generate_population(size):
        if initial_h_values.size == n_points:
            pop = [initial_h_values.copy()]
        else:
            pop = [np.random.rand(n_points) for _ in range(size)]
        for i in range(len(pop)):
            pop[i] = clip_and_scale(pop[i])
        return pop
    
    population = generate_population(50)
    best_h = population[0]
    best_c5 = evaluate(best_h)
    elapsed = 0
    
    while elapsed < budget_s:
        fitness = [evaluate(ind) for ind in population]
        elites = [np.array(population[i]) for i in np.argsort(fitness)[:int(len(population)*0.2)]]
        
        new_population = elites[:]
        for _ in range(len(population) - len(elites)):
            p1, p2 = np.random.choice(elites, size=2, replace=False)
            child = crossover(p1, p2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
        current_best = min(zip(population, fitness))
        if current_best[1] < best_c5:
            best_h, best_c5 = current_best
        elapsed += 10
