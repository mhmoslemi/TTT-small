import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}

    start_time = time.time()
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x

    elapsed = time.time() - start_time
    remaining = budget_s - elapsed

    for _ in range(5):
        if remaining < 10:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        perturb = np.zeros_like(best_h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                perturb[i] -= 0.05 * best_h[i]
                perturb[j] += 0.05 * best_h[i]
        perturbed = best_h + perturb
        perturbed = (perturbed * (n_points / 2) / np.sum(perturbed))
        result = minimize(objective, perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed

    return (best_h, best_c5, n_points)
