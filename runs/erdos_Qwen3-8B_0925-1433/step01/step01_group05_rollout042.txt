from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}

    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    # Initial population
    pop_size = 50
    population = [np.clip(initial_h + np.random.normal(0, 0.1, n_points), 0, 1) * (n_points / 2) / np.sum(np.clip(initial_h + np.random.normal(0, 0.1, n_points), 0, 1)) for _ in range(pop_size)]

    # Evaluate and sort
    fitness = [objective(p) for p in population]
    sorted_idx = np.argsort(fitness)
    population = [population[i] for i in sorted_idx]
    fitness = [fitness[i] for i in sorted_idx]

    # GA phase
    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s * 0.7:
            break
        # Tournament selection
        tour = np.random.choice(len(population), 5)
        best_idx = tour[np.argmin(fitness[tour])]
        best_p = population[best_idx]

        # Find high-overlap lags
        corr = np.correlate(best_p, 1 - best_p, mode='full')
        high_lags = np.argsort(corr)[::-1][:5]

        # Mutation: perturb high-overlap regions
        noise = np.random.normal(0, 0.1, n_points)
        mutated = best_p + noise
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points / 2) / np.sum(mutated)

        # Replace worst individual
        worst_idx = np.argmax(fitness)
        population[worst_idx] = mutated
        fitness[worst_idx] = objective(mutated)
        if fitness[worst_idx] < best_c5:
            best_c5 = fitness[worst_idx]
            best_h = mutated.copy()

    # SQP refinement
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        idx = np.argmin(fitness)
        res = minimize(objective, population[idx], method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    return (best_h, best_c5, n_points)
