import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_pop = [initial_h_values.copy() + np.random.normal(0, 0.1, n_points) for _ in range(50)]
    for i in range(50):
        initial_pop[i] = np.clip(initial_pop[i], 0, 1)
        initial_pop[i] *= (n_points / 2) / np.sum(initial_pop[i])

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h):
        noise = np.random.normal(0, 0.05, n_points)
        child = h + noise
        child = np.clip(child, 0, 1)
        child *= (n_points / 2) / np.sum(child)
        return child

    best_h = initial_h_values.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()

    population = initial_pop
    for gen in range(100):
        fitness = np.array([evaluate(h) for h in population])
        elites = population[np.argsort(fitness)[:5]]
        new_pop = elites
        for _ in range(len(population) - 5):
            idx1, idx2 = np.random.choice(len(population), 2, replace=False)
            parent1, parent2 = population[idx1], population[idx2]
            child = (parent1 + parent2) / 2
            if np.random.rand() < 0.2:
                child = mutate(child)
            new_pop.append(child)
        population = new_pop
        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break

    for _ in range(3):
        if time.time() - start_time > budget_s - 10:
            break
        fitness = np.array([evaluate(h) for h in population])
        h_candidate = population[np.argmin(fitness)]
        cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        result = minimize(evaluate, h_candidate, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-8)
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    best_h = np.clip(best_h, 0, 1)
    best_h *= (n_points / 2) / np.sum(best_h)
    return (best_h, best_c5, n_points)
