import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def rescale(h):
        s = np.sum(h)
        return h * (n_points / 2) / s if s > 0 else h

    # Generate initial population
    pop_size = 50
    pop = [initial_h.copy()]
    for _ in range(pop_size - 1):
        perturb = np.random.normal(0, 0.1, n_points)
        h = initial_h + perturb
        h = np.clip(h, 0, 1)
        h = rescale(h)
        pop.append(h)

    fitness = np.array([evaluate(h) for h in pop])
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx]
    best_c5 = fitness[best_idx]

    start_time = time.time()
    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        # Tournament selection
        selected = []
        for _ in range(pop_size):
            candidates = np.random.choice(pop_size, 5, replace=False)
            best = candidates[np.argmin(fitness[candidates])]
            selected.append(pop[best])
        selected = np.array(selected)

        # Crossover
        offspring = []
        for i in range(0, pop_size, 2):
            if i + 1 < pop_size:
                p1, p2 = selected[i], selected[i+1]
                alpha = np.random.rand()
                child = alpha * p1 + (1 - alpha) * p2
                child = np.clip(child, 0, 1)
                child = rescale(child)
                offspring.append(child)
            else:
                offspring.append(selected[i])
        offspring = np.array(offspring)

        # Mutation
        for h in offspring:
            corr = np.correlate(h, 1 - h, mode='full')
            top_k = np.argsort(corr)[-5:]
            noise = np.random.normal(0, 0.05, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = rescale(h)
        offspring = np.array([rescale(h) for h in offspring])

        # Evaluate
        fit = np.array([evaluate(h) for h in offspring])
        pop = offspring
        fitness = fit
        current_best = np.argmin(fitness)
        if fitness[current_best] < best_c5:
            best_c5 = fitness[current_best]
            best_h = pop[current_best]
