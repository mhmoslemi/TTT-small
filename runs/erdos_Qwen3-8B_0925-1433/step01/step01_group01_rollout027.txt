        polish=True
    )
    best_h = result.x
    h_clipped = np.clip(best_h, 0.0, 1.0)
    current_sum_clipped = np.sum(h_clipped)
    scaling_factor = target_sum / current_sum_clipped
    best_h_scaled = h_clipped * scaling_factor

    def local_objective(h_vector):
        h = np.clip(np.array(h_vector), 0.0, 1.0)
        current_sum = np.sum(h)
        scaling = target_sum / current_sum
        h_scaled = h * scaling
        corr = correlate(h_scaled, 1 - h_scaled, mode='full')
        alpha = 100
        soft_max = np.log(np.sum(np.exp(alpha * corr))) / alpha
        return soft_max

    result_local = minimize(
        local_objective,
        best_h_scaled,
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        method='L-BFGS-B',
        tol=1e-8,
        options={'maxiter': 100}
    )
    best_h_local = result_local.x
    h_clipped_local = np.clip(best_h_local, 0.0, 1.0)
    current_sum_clipped_local = np.sum(h_clipped_local)
    scaling_factor_local = target_sum / current_sum_clipped_local
    best_h_scaled_local = h_clipped_local * scaling_factor_local

    corr_final = correlate(best_h_scaled_local, 1 - best_h_scaled_local, mode='full')
    final_c5 = np.max(corr_final) * dx
    return (best_h_scaled_local, final_c5, n_points)
