import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def rescale(h):
        return h * (n_points / 2) / np.sum(h)

    # Initialize population
    pop_size = 50
    population = [initial_h.copy() for _ in range(pop_size)]
    for i in range(pop_size):
        perturb = np.random.normal(0, 0.1, n_points)
        population[i] += perturb
        population[i] = np.clip(population[i], 0, 1)
        population[i] = rescale(population[i])

    def tournament_select():
        a, b = np.random.choice(len(population), 2, replace=False)
        return population[a] if fitness(population[a]) < fitness(population[b]) else population[b]

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for _ in range(pop_size // 2):
            p1 = tournament_select()
            p2 = tournament_select()
            child = (p1 + p2) / 2
            perturb = np.random.normal(0, 0.1, n_points)
            child += perturb
            child = np.clip(child, 0, 1)
            child = rescale(child)
            new_pop.append(child)
        population = new_pop + [best_h] * (pop_size - len(new_pop))
        fitnesses = [fitness(h) for h in population]
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_c5 = fitnesses[best_idx]
            best_h = population[best_idx].copy()
    
    result = minimize(fitness, best_h, method='SLSQP', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - n_points/2})
    if result.success:
        best_h = result.x
        best_c5 = fitness(best_h)
    return (best_h, best_c5, n_points)
