from scipy.signal import correlate
import time

        n_points = len(h_initial)

    else:
        n_points = 320
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    dx = 2.0 / n_points

    population_size = 20
    particles = []
    velocities = []
    particles.append(h_initial.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, size=n_points))
    for _ in range(population_size - 1):
        new_particle = np.random.rand(n_points)
        new_particle = new_particle * (n_points / 2) / np.sum(new_particle)
        particles.append(new_particle)
        velocities.append(np.random.uniform(-0.1, 0.1, size=n_points))

    def evaluate(p):
        c = correlate(p, 1 - p, mode='full')
        return np.max(c * dx)

    personal_best = [p.copy() for p in particles]
    personal_best_values = [evaluate(p) for p in particles]
    global_best = np.array(personal_best[np.argmin(personal_best_values)])
    global_best_value = np.min(personal_best_values)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        current_values = [evaluate(p) for p in particles]
        for i in range(population_size):
            if current_values[i] < personal_best_values[i]:
                personal_best[i] = particles[i].copy()
                personal_best_values[i] = current_values[i]
        best_index = np.argmin(personal_best_values)
        if personal_best_values[best_index] < global_best_value:
            global_best_value = personal_best_values[best_index]
            global_best = personal_best[best_index].copy()
        for i in range(population_size):
            inertia, cognitive, social = 0.8, 1.5, 1.5
            r1, r2 = np.random.rand(), np.random.rand()
            vel = velocities[i].copy()
            vel = inertia * vel + cognitive * r1 * (personal_best[i] - particles[i]) + social * r2 * (global_best - particles[i])
            vel = np.clip(vel, -0.1, 0.1)
            velocities[i] = vel
            new_pos = particles[i] + vel
            new_pos = np.clip(new_pos, 0, 1)
            sum_new = np.sum(new_pos)
            if sum_new > 1e-10:
                new_pos = new_pos * (n_points / 2) / sum_new
            particles[i] = new_pos
    return (global_best, global_best_value, n_points)
