import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    pop_size = 20

    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        pop.append(h)
    
    def calc_c5(h):
        c = np.correlate(h, 1 - h, mode='full')
        return np.max(c) * dx
    
    best_c5 = float('inf')
    best_h = np.copy(pop[0])
    for gen in range(100):
        fitness = [(calc_c5(h), h) for h in pop]
        fitness.sort()
        parents = [h for _, h in fitness[:int(len(pop)*0.3)]]
        offspring = []
        for _ in range(len(parents)//2):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = (p1 + p2) / 2
            mask = np.random.rand(n_points) < 0.1
            child[mask] += np.random.normal(0, 0.05, np.sum(mask))

    def objective(x):

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    result = minimize(objective, best_h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, options={'maxiter': 200})
