from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Simulated Annealing global search
    def sa_optimize():
        current_h = initial_h.copy()
        best_h = current_h.copy()
        best_c5 = objective(current_h)
        temp = 100.0
        cooling_rate = 0.995
        iterations = 1000
        for _ in range(iterations):
            perturbation = np.random.normal(0, 0.1, size=n_points)
            perturbed = current_h + perturbation
            perturbed = np.clip(perturbed, 0.0, 1.0)
            perturbed_sum = np.sum(perturbed)

                    best_c5 = new_c5
                    best_h = current_h.copy()
            temp *= cooling_rate
        return best_h, best_c5

    best_h, best_c5 = sa_optimize()

    # Local refinement with SLSQP
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
