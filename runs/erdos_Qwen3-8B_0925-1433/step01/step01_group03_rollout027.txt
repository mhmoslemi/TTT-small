    while time.time() - start_time < budget_s and iter_count < max_iterations:
        iter_count += 1
        
        # Compute worst shift and gradient
        shifted = 1.0 - h
        corr = np.correlate(h, shifted, mode='full')
        max_corr_pos = np.argmax(corr)
        k_shift = max_corr_pos - (n_points - 1)
        
        # Compute gradient at the worst shift
        g = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_shift) % n_points
            g[i] = 1.0 - h[j]
        
        # Gradient descent step
        alpha = 0.05  # learning rate
        h_new = h - alpha * g
        h_new = np.clip(h_new, 0, 1)
        sum_new = np.sum(h_new)
        if sum_new != 0:
            scale_new = (n_points / 2) / sum_new
            h_new = h_new * scale_new
        else:
            h_new = np.ones(n_points) * 0.5
        
        # Evaluate and update best
        current_c5 = compute_c5(h_new, dx, n_points)
        if current_c5 < best_c5_val:
            best_c5_val = current_c5
            best_h = h_new.copy()
        
        # Update h for next iteration
        h = h_new.copy()
    
    return (best_h, best_c5_val, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1.0 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
