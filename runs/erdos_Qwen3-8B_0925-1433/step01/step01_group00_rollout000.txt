    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    initial_guesses = [initial_h_values.copy()]

    # Random pattern
    h_rand = np.random.rand(n_points)
    h_rand = (h_rand < 0.5).astype(float)
    s = np.sum(h_rand)
    diff = target_sum - s
    if diff > 1e-6:
        idxs = np.argsort(h_rand)
        h_rand[idxs[:int(diff)]] = 1.0
    elif diff < -1e-6:
        idxs = np.argsort(h_rand)[::-1]
        h_rand[idxs[:int(-diff)]] = 0.0
    initial_guesses.append(h_rand)

    # Alternating pattern
    h_alt = np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n_points)])
    s = np.sum(h_alt)
    if s > target_sum:
        h_alt[np.argmax(h_alt)] = 0.5
    initial_guesses.append(h_alt)

    best_h = initial_guesses[0].copy()
    best_c5 = obj(best_h)
    total_time = 0.0

    for guess in initial_guesses:
        if total_time >= budget_s:
            break
        start_time = time.time()
        result = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'ftol': 1e-8, 'maxiter': 500, 'disp': False})
        elapsed_time = time.time() - start_time
        total_time += elapsed_time

        if result.success:
            h_candidate = result.x
            c5_candidate = obj(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
        else:
            h_candidate = guess
            c5_candidate = obj(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate

    best_h = np.clip(best_h, 0.0, 1.0)
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-5:
        best_h[np.argmax(best_h)] += diff

    return (best_h, best_c5, n_points)
