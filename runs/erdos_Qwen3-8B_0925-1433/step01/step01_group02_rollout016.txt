import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values, dtype=np.float64) if 'initial_h_values' in locals() else None
    
    def generate_initial_guesses(n_points):
        if initial_h is not None:
            yield initial_h.copy()
        # Alternating pattern
        alt = np.zeros(n_points)
        alt[::2] = 1.0
        alt = alt * (n_points / 2) / np.sum(alt)
        yield alt
        # Random binary
        rand = np.random.rand(n_points)
        rand = np.sort(rand)
        rand = (rand <= 0.5).astype(float)
        rand = rand * (n_points / 2) / np.sum(rand)
        yield rand
    
    n_points = 70  # Use parent's n_points if available, else default
    if initial_h is not None:
        n_points = len(initial_h)
    
    best_h = None
    best_c5 = float('inf')
    
    for guess in generate_initial_guesses(n_points):
        try:
            h = guess.copy()
            dx = 2.0 / n_points
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            
            def objective(h_values):
                c = np.correlate(h_values, 1 - h_values, mode='full')
                return np.max(c) * dx
            
            def constraint_sum(h_values):
                return np.sum(h_values) - n_points / 2
            
            cons = [{'type': 'eq', 'fun': constraint_sum}]
            result = minimize(
                fun=objective,
                x0=h,
                method='SLSQP',
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-10}
            )
            
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
        except:
            pass
    
    if best_h is None:
        best_h = np.zeros(n_points)
        best_c5 = 1.0
    
    return best_h, best_c5, n_points
