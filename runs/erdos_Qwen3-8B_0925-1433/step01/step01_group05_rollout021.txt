import numpy as np
from scipy.optimize import minimize, COBYLA
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}

    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = objective(best_h)

    start_time = time.time()
    elapsed = 0

    # Initial optimization with COBYLA
    result = minimize(objective, best_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-5)
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x

    # Perturbation strategies
    for _ in range(4):
        if elapsed > budget_s - 10:
            break
        perturb = np.random.normal(0, 0.15, size=n_points)
        perturbed = best_h + perturb
        perturbed = np.clip(perturbed, 0.0, 1.0)
        sum_perturbed = np.sum(perturbed)
        if sum_perturbed > 0:
            perturbed *= (n_points / 2) / sum_perturbed
        result = minimize(objective, perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        elapsed = time.time() - start_time

    # Final refinement with increased precision
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if result.success and result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x

    return (best_h, best_c5, n_points)
