import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        scale = (n_points / 2.0) / h_sum if h_sum != 0 else 1.0
        h *= scale
    else:
        h = np.random.uniform(0, 1, n_points)
        h_sum = np.sum(h)
        scale = (n_points / 2.0) / h_sum if h_sum != 0 else 1.0
        h *= scale
    dx = 2.0 / n_points
    
    def obj_de(x):
        h = np.clip(x, 0.0, 1.0)
        h_sum = np.sum(h)
        scale = (n_points / 2.0) / h_sum
        h_scaled = h * scale
        cc = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(cc) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_de = differential_evolution(obj_de, bounds, strategy='best1bin', maxiter=100, popsize=20, tol=1e-6, mutation=(0.5, 1), recombination=0.7, disp=False)
    best_h_de = result_de.x
    
    def obj_slsqp(h):
        cc = np.correlate(h, 1 - h, mode='full')
        return np.max(cc) * dx
    
    def cons_sum(h):
        return np.sum(h) - n_points / 2.0
    
    result_slsqp = minimize(obj_slsqp, best_h_de, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': cons_sum}, tol=1e-8, options={'maxiter': 200})
    best_h = result_slsqp.x
    
    best_h = np.clip(best_h, 0.0, 1.0)
    h_sum = np.sum(best_h)
    scale = (n_points / 2.0) / h_sum
    best_h *= scale
    
    cc = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(cc) * dx
    
    return (best_h, c5_bound, n_points)
