import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 160  # Increased resolution for better approximation
    dx = 2.0 / n_points
    
    # Initialize h using initial_h_values if available, else use a structured grid
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Use evenly spaced intervals for structured initial guess
        m = 5  # Number of intervals
        intervals = np.linspace(0, 2, m + 1)
        values = np.zeros(n_points)
        for i in range(m):
            start, end = intervals[i], intervals[i+1]
            idx = np.logical_and(h >= start, h <= end)
            values[idx] = 0.5
        h = values
    
    # Ensure integral constraint is respected
    sum_h = np.sum(h) * dx
    scale = (n_points / 2.0) / sum_h
    h *= scale
    
    def objective(h_values):
        cc = correlate(h_values, 1 - h_values, mode='full')
        return np.max(cc) * dx
    
    def constraint_sum(h_values):

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    x0 = h.copy()
    
    # Optimization settings with tighter tolerances
    options = {'maxiter': 400, 'ftol': 1e-9, 'eps': 1e-9, 'disp': False}
