def run(seed=42, budget_s=600, **kwargs):

    for gen in range(max_gen):
        fit = np.array([evaluate_erdos_solution(h) for h in population])
        best_idx = np.argmin(fit)

            child = (p1 + p2) / 2 + np.random.rand(n_points) * 0.05
