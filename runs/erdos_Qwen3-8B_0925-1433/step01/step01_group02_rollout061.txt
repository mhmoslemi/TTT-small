    best_c5 = objective(best_h)
    
    for _ in range(500):
        pop = de_step(pop)
        current = [objective(h) for h in pop]
        idx = np.argmin(current)
        if current[idx] < best_c5:
            best_h = np.copy(pop[idx])
            best_c5 = current[idx]
        if _ > 10 and abs(current[idx] - best_c5) < 1e-6:
            break
    
    return best_h, best_c5, n_points
