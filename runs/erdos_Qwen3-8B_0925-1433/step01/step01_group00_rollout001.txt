        return np.array([1 if i % 3 == 0 else 0 for i in range(n)])
    guesses.append(gen_p3(n_points))

    def gen_rand(n):
        h = np.random.choice([0, 1], size=n)
        s = np.sum(h)
        t = n / 2
        if s < t:
            hc = np.where(h == 0)[0]
            if len(hc) > 0:
                h[hc[:t - s]] = 1
        elif s > t:
            hc = np.where(h == 1)[0]
            if len(hc) > 0:
                h[hc[:s - t]] = 0
        return h
    guesses.append(gen_rand(n_points))

    def gen_rnd(n):
        return np.random.rand(n).astype(float)
    guesses.append(gen_rnd(n_points))

    best_c5 = float('inf')
    best_h = h0.copy()
    start_time = time.time()

    for guess in guesses:
        guess = np.array(guess)
        if np.any(guess < 0) or np.any(guess > 1):
            guess = np.clip(guess, 0, 1)
        result = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'ftol': 1e-8, 'maxiter': 500, 'disp': False})
        if result.success:
            ch = result.x
            c5 = obj(ch)
            if c5 < best_c5:
                best_c5 = c5
                best_h = ch
        else:
            c5 = obj(guess)
            if c5 < best_c5:
                best_c5 = c5
                best_h = guess
        if time.time() - start_time > budget_s:
            break

    if np.any(best_h < 0) or np.any(best_h > 1):
        best_h = h0.copy()
        best_c5 = obj(best_h)

    return (best_h, best_c5, n_points)
