import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx
    pop_size = 50
    population = [initial_h.copy() for _ in range(pop_size)]
    start = time.time()
    gen = 0
    while time.time() - start < budget_s:
        gen += 1
        fitness = [np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in population]
        best_idx = np.argmin(fitness)
        candidates = []
        for _ in range(pop_size // 10):
            a, b = np.random.choice(len(population), 2, replace=False)
            p1, p2 = population[a], population[b]
            child = (p1 + p2) / 2
            mut = np.random.normal(0, 0.05, size=n_points)
            child += mut
            child = np.clip(child, 0, 1)
            child *= (n_points / 2) / np.sum(child)
            candidates.append(child)
        new_pop = [population[best_idx]] + candidates
        population = new_pop
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx]
    return (best_h, best_c5, n_points)
