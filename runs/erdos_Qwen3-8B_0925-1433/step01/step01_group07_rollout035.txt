import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    initial_h = initial_h_values.copy()
    initial_h_sum = np.sum(initial_h)
    scale = target_sum / initial_h_sum
    initial_h = np.clip(initial_h * scale, 0.0, 1.0)
    
    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    pop_size = 50
    population = [initial_h.copy() for _ in range(pop_size)]
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for gen in range(150):  # 150 generations
        pop_c5 = np.array([compute_c5(h) for h in population])
        best_idx = np.argmin(pop_c5)
        worst_idx = np.argmax(pop_c5)
        
        parents = [population[best_idx]]
        for _ in range(pop_size - 1):
            idx1, idx2 = np.random.choice(pop_size, 2, replace=False)
            parent1, parent2 = population[idx1], population[idx2]
            child = (parent1 + parent2) / 2
            if np.random.rand() < 0.2:
                mutate_mask = np.random.rand(n_points) < 0.1
                child[mutate_mask] = np.random.uniform(0.0, 1.0, np.sum(mutate_mask))
            child_sum = np.sum(child)
            scale_factor = target_sum / child_sum
            child = np.clip(child * scale_factor, 0.0, 1.0)
            parents.append(child)
        
        population = parents
        current_best_c5 = np.min(pop_c5)
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[np.argmin(pop_c5)]
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    constraints = ({'type': 'eq', 'fun': lambda h_vec: np.sum(h_vec) - target_sum})
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
    
    if result.success:
        final_h = result.x
        final_c5 = result.fun
    else:
        final_h = best_h
        final_c5 = best_c5
    
    return (final_h, final_c5, n_points)
