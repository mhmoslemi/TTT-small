import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    import numpy as np
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    target_sum = n_points / 2
    h_sum = np.sum(h)
    h = h * target_sum / h_sum

    def evaluate_fitness(h_values):
        dx = 2.0 / n_points
        c = np.correlate(h_values, 1.0 - h_values, mode='full')
        return np.max(c) * dx

    def repair_h(h_values):
        h_clip = np.clip(h_values, 0.0, 1.0)
        h_sum = np.sum(h_clip)
        if abs(h_sum - target_sum) > 1e-8:
            scale = target_sum / h_sum
            h_clip *= scale
        return h_clip

    population = [h.copy()]
    for _ in range(5):
        h_rand = np.random.rand(n_points)
        h_rand = repair_h(h_rand)
        population.append(h_rand)

    pop_size = 20
    gen = 200
    best_h = h.copy()
    best_c5 = evaluate_fitness(h)

    for _ in range(gen):
        fitness = [(evaluate_fitness(ind), ind) for ind in population]
        fitness.sort()
        selected = [f[1] for f in fitness[:int(0.4 * len(population))]]
        
        offspring = []
        for _ in range(pop_size - len(selected)):
            p1, p2 = np.random.choice(selected, 2, replace=False)
            child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
            if np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
            child = repair_h(child)
            offspring.append(child)
        
        population = selected + offspring
        current_best = min(population, key=lambda x: evaluate_fitness(x))
        if evaluate_fitness(current_best) < best_c5:
            best_h, best_c5 = current_best, evaluate_fitness(current_best)
    
    final_h = repair_h(best_h)
    final_c5 = evaluate_fitness(final_h)
    return final_h, final_c5, n_points
