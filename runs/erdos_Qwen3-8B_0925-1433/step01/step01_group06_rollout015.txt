        a_initial = [0.2 * j for j in range(m)]
        l_initial = [0.4 for _ in range(m)]

    # Flatten a_j and l_j for optimization
    variables = a_initial + l_initial
    num_vars = len(variables)

    # Add bounds for each variable
    bounds = []
    for i in range(m):
        a_bounds = (0.0, 2.0 - l_initial[i])
        bounds.append(a_bounds)
        l_bounds = (1e-6, 2.0)
        bounds.append(l_bounds)

    # Objective: maximize minimum pairwise min(l_i, l_j)
    def objective(x):
        a = x[:m]
        l = x[m:]
        # Ensure a is sorted and non-overlapping
        sorted_indices = np.argsort(a)
        a_sorted = a[sorted_indices]
        l_sorted = l[sorted_indices]
        max_overlap = 0.0
        for i in range(m):
            for j in range(i + 1, m):
                max_overlap = max(max_overlap, min(l_sorted[i], l_sorted[j]))
        return max_overlap

    # Constraints: sum(l) = 1, a_{j+1} >= a_j + l_j
    cons = [
        {'type': 'eq', 'fun': lambda x: np.sum(x[m:]) - 1.0},
    ]
    for i in range(m - 1):
        cons.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[i + m] + x[i] - x[i + m + 1]
        })

    # Optimize
    res = minimize(
        objective,
        variables,
        method='SLSQP',
        bounds=Bounds(bounds),
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )

    # Extract optimized intervals
    a_opt = res.x[:m]
    l_opt = res.x[m:]
    # Sort by positions
    sorted_indices = np.argsort(a_opt)
    a_sorted = a_opt[sorted_indices]
    l_sorted = l_opt[sorted_indices]

    # Generate h_values
    h_values = np.zeros(n_points)
    for i in range(m):
        a = a_sorted[i]
        l = l_sorted[i]
        start = int(a / dx)
        end = int((a + l) / dx)
        h_values[start:end] = 1.0

    # Ensure sum(h) * dx ≈ 1
    sum_h = np.sum(h_values) * dx
    scale = (n_points / 2) / sum_h
    h_values *= scale

    # Final sanity check
    h_values = np.clip(h_values, 0.0, 1.0)
    sum_h = np.sum(h_values) * dx
    if abs(sum_h - 1.0) > 1e-6:
        h_values = (h_values * (1.0 / sum_h) * (n_points / 2.0))

    # Compute C5 using exact correlation
    cc = correlate(h_values, 1 - h_values, mode='full')
    c5_bound = np.max(cc) * dx
    return (h_values, c5_bound, n_points)
