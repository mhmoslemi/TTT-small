    target_sum = n_points / 2.0

        initial_h = np.random.rand(n_points).astype(float)

        scaling = target_sum / current_sum
        h_scaled = h * scaling

        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = differential_evolution(
        de_objective,
        bounds,
        strategy='best1bin',
        popsize=100,
        maxiter=200,
        mutation=(0.5, 1.0),
        recombination=0.9,
        tol=1e-6,
        seed=seed,
        polish=True
    )
    best_h = result.x
    best_h = np.clip(best_h, 0.0, 1.0)
