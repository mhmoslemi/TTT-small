            h = np.clip(h * scale, 0, 1)
        else:
            h = np.ones(n_points) * 0.5
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
