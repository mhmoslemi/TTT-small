import numpy as np
from scipy.optimize import minimize
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def compute_c5(h):
        c = np.correlate(h, 1 - h, mode='full')
        return np.max(c) * dx

    def normalize(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return h * target_sum / s

    # Initialize population with initial guess, random samples, and alternating pattern
    population = [normalize(initial_h_values)]
    for _ in range(5):
        rand_h = np.random.rand(n_points)
        rand_h = normalize(rand_h)
        population.append(rand_h)
    alt_pattern = np.zeros(n_points)
    for i in range(n_points):
        alt_pattern[i] = 1 if i % 2 == 0 else 0
    population.append(normalize(alt_pattern))

    best_h = population[0]
    best_c5 = compute_c5(best_h)

    max_generations = 50
    pop_size = len(population)
    for gen in range(max_generations):
        candidates = [(h, compute_c5(h)) for h in population]
        candidates.sort(key=lambda x: x[1])
        top_half = candidates[:int(0.5 * pop_size)]

        # Generate offspring via crossover and mutation
        offspring = []
        for p1, p2 in zip(candidates[:len(candidates)//2], candidates[1:]):
            child = (p1[0] + np.random.rand(n_points) * (p2[0] - p1[0])) / 2
            child = normalize(child)
            offspring.append(child)

        # Add random perturbations to exploration
        for h in population[:len(population)//2]:
            perturb = np.random.normal(0, 0.1, n_points)
            h = normalize(np.clip(h + perturb, 0, 1))
            offspring.append(h)

        # Replace worst individuals
        population = [c[0] for c in top_half] + offspring
        population = population[:pop_size]

        if compute_c5(population[0]) < best_c5:
            best_h = population[0]
            best_c5 = compute_c5(best_h)

    # Final refinement with local optimization
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(lambda h: compute_c5(h), best_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 200})
    return result.x, result.fun, n_points
