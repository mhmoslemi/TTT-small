import numpy as np
def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def eval(h):
        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr) * dx

    def rescale(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        return h * (n_points/2) / s if s else h

    pop = [initial_h.copy()]
    for _ in range(49):
        h = initial_h + np.random.normal(0, 0.1, n_points)
        pop.append(rescale(h))

    best = initial_h.copy()
    best_val = eval(best)

    for _ in range(100):
        fit = [eval(h) for h in pop]
        fit_idx = np.argsort(fit)
        pop = [pop[i] for i in fit_idx]
        best = pop[0] if fit[0] < best_val else best
        best_val = min(best_val, fit[0])

        new_pop = []
        for _ in range(len(pop)//2):
            a, b = pop[np.random.choice(len(pop), 2, replace=False)]
            c = (a + b) / 2
            if np.random.rand() < 0.2:
                idxs = np.random.choice(n_points, 5, replace=False)
                for i in idxs:
                    c[i] += np.random.normal(0, 0.05)
                    c[i] = np.clip(c[i], 0, 1)
                c = rescale(c)
            new_pop.append(c)
        for _ in range(len(pop)//2):
            new_pop.append(pop[np.random.choice(len(pop))])
        pop = new_pop + pop[:len(pop)//2]

    return (best, best_val, n_points)
