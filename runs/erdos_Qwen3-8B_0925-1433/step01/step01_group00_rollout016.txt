def run(seed=42, budget_s=600, **kwargs):
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        n_points = 95
        initial_h = np.zeros(n_points)
        initial_h[::2] = 1

    dx = 2.0 / n_points
    
    def obj(h):

        return np.max(corr) * dx
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    initial_guesses = [initial_h.copy()]
    
    def generate_periodic(period):
        pattern = np.zeros(n_points)
        for i in range(n_points):
            if i % period == 0:
                pattern[i] = 1
        sum_p = np.sum(pattern)
        pattern = pattern * (n_points / 2) / sum_p
        return pattern
    
    initial_guesses.append(generate_periodic(3))
    initial_guesses.append(generate_periodic(2))
    
    np.random.seed(42)
    jitter = initial_h.copy() + np.random.normal(0, 0.1, n_points)
    jitter = np.clip(jitter, 0, 1)

    best_h = initial_h.copy()

        best_c5 = obj(best_h)
    
    return (best_h, best_c5, n_points)
