import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initialize h based on available initial_h_values
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        n_points = 60
        h = np.random.choice([0, 1], size=n_points, p=[0.75, 0.25])
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    
    dx = 2.0 / len(h)
    n_points = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        
        # Optimize current h
        def obj(x):
            h_vec = x.copy()
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            return np.max(corr) * dx
        
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=constraints)
        if res.success:
            h = res.x.copy()
            current_c5 = compute_c5(h, dx, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Perturb solution every 10 steps
        if _ % 10 == 0:
            perturb = np.random.uniform(-0.1, 0.1, size=n_points)
            h = np.clip(h + perturb, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                scale = (n_points / 2) / sum_h
                h = np.clip(h * scale, 0, 1)
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
