import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    population = [initial_h_values.copy() for _ in range(50)]
    for h in population:
        h += np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)

    start_time = time.time()
    for gen in range(150):
        fitness = [objective(h) for h in population]
        if np.min(fitness) < best_c5:
            best_c5 = np.min(fitness)
            best_h = population[np.argmin(fitness)]
        if time.time() - start_time > budget_s - 30:
            break
        new_pop = []
        for _ in range(len(population) // 2):
            idx1, idx2 = np.random.choice(len(population), 2, replace=False)
            child = (population[idx1] + population[idx2]) / 2
            mutate = np.random.choice(n_points, 5, replace=False)
            for i in mutate:
                child[i] += np.random.normal(0, 0.05)
                child[i] = np.clip(child[i], 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        population = new_pop + [population[np.argmin(fitness)] * 5]  # elitism

    # SQP refinement
    candidates = [population[i] for i in np.argsort([objective(h) for h in population])[:5]]
    for h in candidates:
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        result = minimize(objective, h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-8)
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        if time.time() - start_time > budget_s - 10:
            break

    return (best_h, best_c5, n_points)
