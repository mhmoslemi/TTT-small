import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 480
        h = np.random.rand(n_points)
        required_sum = n_points / 2
        h *= required_sum / np.sum(h)
    dx = 2.0 / n_points
    required_sum = n_points / 2

    def compute_c5(h_val):
        cc = correlate(h_val, 1 - h_val, mode='full')
        return np.max(cc * dx)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    T = 1.0
    cooling_rate = 0.995
    max_iterations = 1000
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
        sigma = 0.05
        delta = np.random.normal(0, sigma, n_points)
        delta -= np.mean(delta)
        h_candidate = h + delta
        h_candidate = np.clip(h_candidate, 0, 1)
        sum_candidate = np.sum(h_candidate)
        if sum_candidate == 0:
            continue
        h_candidate = h_candidate * required_sum / sum_candidate
        c_candidate = compute_c5(h_candidate)
        if c_candidate < best_c5:
            best_c5 = c_candidate
            best_h = h_candidate.copy()
        else:
            prob = np.exp((best_c5 - c_candidate) / T)
            if np.random.rand() < prob:
                best_c5 = c_candidate
                best_h = h_candidate.copy()
        T *= cooling_rate
        h = best_h.copy()

    best_h = np.clip(best_h, 0, 1)
    sum_check = np.sum(best_h)
    if sum_check != required_sum:
        best_h = best_h * required_sum / sum_check
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
