def run(seed=42, budget_s=600, **kwargs):

        initial_h = initial_h_values
    else:
        n_points = 50

        return np.max(corr) * dx

    def rescale(h):
        total = np.sum(h)
        scale = (n_points / 2) / total if total != 0 else 0
        return h * scale

    pop_size = 50
    population = [initial_h.copy() for _ in range(pop_size)]
    for i in range(pop_size):

        h = initial_h + perturb

        population[i] = rescale(h)

    fitness = np.array([compute_fitness(h) for h in population])
    best_index = np.argmin(fitness)
    best_h = population[best_index].copy()

            selected.append(population[np.argmin(fitness[idxs])])
