import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    if abs(sum_h - n_points / 2) > 1e-5:
        initial_h *= (n_points / 2) / sum_h
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()
    for _ in range(1000):
        perturb = np.random.normal(0, 0.1, n_points)
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h *= (n_points / 2) / sum_new
        current_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
        if current_c5 < best_c5:
            best_h = new_h
            best_c5 = current_c5
        if time.time() - start_time > budget_s - 10:
            break

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    result = minimize(lambda h: np.max(np.correlate(h, 1 - h, mode='full')) * dx, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if result.success and result.fun < best_c5:
        best_h = result.x
        best_c5 = result.fun

    return (best_h, best_c5, n_points)
