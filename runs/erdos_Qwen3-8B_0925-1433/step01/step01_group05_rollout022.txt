import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    initial_sum = np.sum(initial_h)
    if abs(initial_sum - n_points/2) > 1e-6:
        initial_h *= (n_points/2) / initial_sum
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    elapsed = 0
    remaining = budget_s

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.min(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}

    for _ in range(10):
        if remaining < 10:
            break
        pop = [initial_h.copy()]
        for _ in range(49):
            perturb = np.random.normal(0, 0.05, n_points)
            p = initial_h + perturb
            p = np.clip(p, 0, 1)
            p_sum = np.sum(p)
            if p_sum > 0:
                p *= (n_points / 2) / p_sum
            pop.append(p)
        pop_fitness = [fitness(ind) for ind in pop]
        pop.sort(key=lambda x: pop_fitness.pop(pop.index(x)))
        best_pop = pop[:5]
        for ind in best_pop:
            res = minimize(objective, ind, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x
        elapsed = time.time() - time.time()
        remaining = budget_s - elapsed
