    best_c5 = obj(best_h)
    start_time = time.time()

    for guess in initial_guesses:
        result = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'ftol': 1e-8, 'maxiter': 500, 'disp': False})
        if result.success:
            current_h = result.x
            current_c5 = obj(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        else:
            current_c5 = obj(guess)
            if current_c5 < best_c5:
                best_h = guess.copy()
                best_c5 = current_c5
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
