import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    import numpy as np
    from scipy.optimize import minimize
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    def objective(h_values):
        c = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(c) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    guesses = [initial_h_values.copy()]
    h_alt = np.zeros(n_points)
    h_alt[::2] = 1
    guesses.append(h_alt)
    
    np.random.seed(seed)
    indices = np.random.choice(n_points, size=n_points // 2, replace=False)
    h_rand = np.zeros(n_points)
    h_rand[indices] = 1.0
    guesses.append(h_rand)
    
    for guess in guesses:
        s = np.sum(guess)
        if s == 0: scaled = np.full_like(guess, 0.5)
        else: scaled = guess * (n_points / 2) / s
        
        res = minimize(objective, scaled, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 500, 'ftol': 1e-10})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
    
    return best_h, best_c5, n_points
