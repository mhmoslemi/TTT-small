import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_sum = np.sum(initial_h_values)
    scale_factor = (n_points / 2) / initial_sum
    h = initial_h_values * scale_factor
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    start_time = time.time()
    temp = 1.0
    perturb_std = 0.05
    iteration = 0
    max_iterations = int(budget_s * 10)
    
    while iteration < max_iterations:
        if time.time() - start_time >= budget_s:
            break
        perturb = np.random.normal(0, perturb_std, n_points)
        new_h = h + perturb
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        scale_factor = (n_points / 2) / sum_new
        new_h = new_h * scale_factor
        current_c5 = compute_c5(h)
        new_c5 = compute_c5(new_h)
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h = new_h.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        temp *= 0.995
        perturb_std *= 0.99
        iteration += 1
    
    return (best_h, best_c5, n_points)
