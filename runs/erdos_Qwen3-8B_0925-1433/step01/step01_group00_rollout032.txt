    required = n_points / 2
    delta = current_sum - required
    if delta > 0:
        h_rand[np.argsort(h_rand)[::-1][:int(delta)]] = 0
    elif delta < 0:
        h_rand[np.argsort(h_rand)[:int(-delta)]] = 1
    initial_guesses.append(h_rand)

    best_h = initial_h_values.copy()
    best_c5 = obj(best_h)
    best_time = 0

    for guess in initial_guesses:
        start_time = time.time()
        guess = np.clip(guess, 0.0, 1.0)
        current_sum = np.sum(guess)
        required = n_points / 2
        delta = current_sum - required
        if delta != 0:
            if delta > 0:
                guess[np.argsort(guess)[::-1][:int(delta)]] = 0
            else:
                guess[np.argsort(guess)[:int(-delta)]] = 1
        result = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'ftol': 1e-9, 'maxiter': 500, 'disp': False})
        elapsed = time.time() - start_time
        best_time += elapsed
        if best_time >= budget_s:
            break
        if result.success:
            h_opt = result.x
            if np.any(h_opt < 0) or np.any(h_opt > 1):
                h_opt = guess
            c5 = obj(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
        else:
            c5 = obj(guess)
            if c5 < best_c5:
                best_c5 = c5
                best_h = guess

    best_h = np.clip(best_h, 0.0, 1.0)
    current_sum = np.sum(best_h)
    required = n_points / 2
    delta = current_sum - required
    if delta != 0:
        if delta > 0:
            best_h[np.argsort(best_h)[::-1][:int(delta)]] = 0
        else:
            best_h[np.argsort(best_h)[:int(-delta)]] = 1

    return (best_h, best_c5, n_points)
