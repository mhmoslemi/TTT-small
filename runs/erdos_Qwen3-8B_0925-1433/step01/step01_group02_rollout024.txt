import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    import numpy as np
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    np.random.shuffle(rand)
    guesses.append(rand.astype(np.float64))
    best_c5 = float('inf')
    best_h = np.zeros(n_points)

        h = h.astype(np.float64)

            h = h * (n_points / 2) / h_sum
        h = np.clip(h, 0, 1)

            c = np.correlate(x, 1 - x, mode='full')
            return np.max(c) * dx

        res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=[{'type':'eq','fun':cons}], options={'maxiter':500, 'ftol':1e-10})
        if res.success and res.fun < best_c5:
