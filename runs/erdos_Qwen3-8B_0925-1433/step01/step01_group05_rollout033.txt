import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    if not np.isclose(sum_h, n_points / 2):
        h *= (n_points / 2) / sum_h
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}

    def evaluate(h):
        return objective(h)

    best_h = h
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    remaining = budget_s - (time.time() - start_time)
    
    # First phase: DE for global exploration
    if remaining > 50:
        result = differential_evolution(evaluate, bounds, strategy='best1bin', 
                                       popsize=20, tol=1e-5, maxiter=50, 
                                       polish=True, mutation=(0.8, 1.0), 
                                       recombination=0.9, seed=seed)
        if result.success:
            best_h = result.x
            best_c5 = result.fun
    
    # Second phase: SLSQP refinement
    if remaining > 50:
        result = minimize(evaluate, best_h, method='SLSQP', bounds=bounds, 
                          constraints=cons, tol=1e-8, options={'maxiter': 200})
        if result.success and result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
    
    # Third phase: targeted perturbations
    for _ in range(3):
        if remaining < 10:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr_idx = np.argmax(corr)
        lag = (max_corr_idx - (n_points - 1)) / dx
        shift = int(round(lag))
        overlap_vals = best_h * (1 - best_h[shift:])
        overlap_vals = np.pad(overlap_vals, (0, n_points - len(overlap_vals)), 'constant')
        h_idx = np.argsort(overlap_vals)[::-1][:10]
        
        perturb = np.zeros(n_points)
        for i in h_idx:
            perturb[i] += np.random.normal(0, 0.1)
        perturb = np.clip(perturb, -0.2, 0.2)
        perturbed = best_h + perturb

        sum_perturbed = np.sum(perturbed)

        result = minimize(evaluate, perturbed, method='SLSQP', bounds=bounds,
