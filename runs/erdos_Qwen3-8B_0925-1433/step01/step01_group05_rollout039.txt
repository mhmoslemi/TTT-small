import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    pop_size = 50
    pop = [initial_h.copy() for _ in range(pop_size)]
    for i in range(pop_size):
        perturb = np.random.normal(0, 0.1, n_points)
        perturbed = initial_h + perturb
        perturbed = np.clip(perturbed, 0, 1)
        if np.sum(perturbed):
            perturbed *= (n_points / 2) / np.sum(perturbed)
        pop[i] = perturbed

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        fits = [evaluate(h) for h in pop]
        best_idx = np.argmin(fits)
        if fits[best_idx] < best_c5:
            best_c5 = fits[best_idx]
            best_h = pop[best_idx].copy()
        new_pop = [pop[best_idx]]
        for _ in range(pop_size - 1):
            t = random.sample(range(pop_size), 5)
            tidx = min(t, key=lambda x: fits[x])
            p1 = pop[tidx]
            p2 = random.choice(pop)
            child = np.random.rand() * p1 + (1 - np.random.rand()) * p2
            child = np.clip(child, 0, 1)
            if np.sum(child):
                child *= (n_points / 2) / np.sum(child)
            new_pop.append(child)
        pop = new_pop

    return (best_h, best_c5, n_points)
