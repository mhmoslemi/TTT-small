def run(seed=42, budget_s=600, **kwargs):
    n_points = 50
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    alpha = 100

    initial_h = initial_h_values if 'initial_h_values' in globals() else None
    if initial_h is not None:
        initial_h = np.array(initial_h, dtype=float)
        initial_sum = np.sum(initial_h)
        scaling = target_sum / initial_sum
        initial_h *= scaling
        initial_h = np.clip(initial_h, 0.0, 1.0)
    else:
        initial_h = np.random.rand(n_points).astype(float)
        initial_h = np.clip(initial_h, 0.0, 1.0)
        initial_sum = np.sum(initial_h)
        scaling = target_sum / initial_sum
        initial_h *= scaling

    def objective(h_vector):
        h = np.clip(h_vector, 0.0, 1.0)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        log_sum = logsumexp(alpha * (corr - max_corr))
        return dx * (max_corr + log_sum / alpha)

    def constraint_sum(h):

    cons = {'type': 'eq', 'fun': constraint_sum}

    result = minimize(
        fun=objective,
        x0=initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 1000, 'eps': 1e-8}
    )

    best_h = result.x

    c5_bound = np.max(corr) * dx

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (target_sum / np.sum(best_h))
    best_h = np.clip(best_h, 0.0, 1.0)

    return (best_h, c5_bound, n_points)
