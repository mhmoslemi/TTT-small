import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_initial = np.sum(initial_h_values)
    if np.abs(sum_initial - n_points / 2) > 1e-6:
        initial_h_values = initial_h_values * (n_points / 2) / sum_initial

    n_particles = 50
    max_iter = 500
    w = 0.8
    c1 = c2 = 2.0

    particles = np.random.rand(n_particles, n_points)
    particles = initial_h_values + particles * 0.1
    particles = np.clip(particles, 0, 1)
    for i in range(n_particles):
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))

    personal_best = particles.copy()
    personal_fitness = np.zeros(n_particles)
    for i in range(n_particles):
        h = particles[i]
        corr = np.correlate(h, 1 - h, mode='full')
        personal_fitness[i] = np.max(corr) * dx
    global_best = particles[np.argmin(personal_fitness)].copy()
    best_fitness = np.min(personal_fitness)

    start_time = time.time()

    for _ in range(max_iter):
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = w * velocities[i] + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            sum_p = np.sum(particles[i])
            particles[i] = particles[i] * (n_points / 2) / sum_p

        current_fitness = np.zeros(n_particles)
        for i in range(n_particles):
            h = particles[i]
            corr = np.correlate(h, 1 - h, mode='full')
            current_fitness[i] = np.max(corr) * dx

        for i in range(n_particles):
            if current_fitness[i] < personal_fitness[i]:

                if personal_fitness[i] < best_fitness:

        elapsed = time.time() - start_time
