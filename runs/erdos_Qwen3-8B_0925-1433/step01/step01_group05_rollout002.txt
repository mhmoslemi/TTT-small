from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    n_points = len(h_initial)

    total = np.sum(h_initial)
    if total > 0:
        h_initial *= (n_points / 2) / total

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def ga_phase(h_pop, pop_size=50, gen=100, mut_strength=0.2):
        for _ in range(gen):
            fitness = np.array([objective(ind) for ind in h_pop])
            elite = h_pop[np.argsort(fitness)[:5]]
            new_pop = []
            for _ in range(pop_size - 5):
                pa, pb = np.random.choice(pop_size, 2, replace=False)
                parent_a = h_pop[pa]
                parent_b = h_pop[pb]
                child = (parent_a + parent_b) / 2
                idxs = np.argsort(fitness)[::-1][:5]
                for idx in idxs:
                    if np.random.rand() < mut_strength:
                        child[idx] += np.random.normal(0, 0.1)
                child = np.clip(child, 0, 1)
                total = np.sum(child)
                if total > 0:
                    child *= (n_points / 2) / total
                new_pop.append(child)
            h_pop = np.vstack((elite, new_pop))
            h_pop = h_pop[np.argsort(fitness)[:pop_size]]
            if np.max(fitness) < 1e-6:
                break
        return h_pop

    def sqp_refine(h_init, max_iter=200, tol=1e-6):
        result = minimize(objective, h_init, method='SLSQP', bounds=bounds, constraints=cons, tol=tol, options={'max_iter': max_iter})
        if result.success:
            return result.x
        return h_init

    start_time = time.time()
    best_h = h_initial.copy()

    h_pop = [np.clip(h, 0, 1) for h in h_pop]
    for i in range(len(h_pop)):

        if total > 0:
            h_pop[i] *= (n_points / 2) / total
    h_pop = np.array(h_pop)

    elapsed = time.time() - start_time
