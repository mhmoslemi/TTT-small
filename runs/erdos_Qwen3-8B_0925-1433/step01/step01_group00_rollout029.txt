import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Assume this is provided by the parent
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_sum = np.sum(initial_h_values)
    target_sum = n_points / 2

    # Initialize population
    pop_size = 50
    particles = np.array([initial_h_values + 0.1 * np.random.randn(n_points) for _ in range(pop_size)])
    velocities = 0.1 * np.random.randn(pop_size, n_points)
    personal_best = particles.copy()
    personal_best_vals = np.full(pop_size, np.inf)
    global_best = particles[0].copy()
    global_best_val = np.inf

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    for _ in range(1000):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break

        # Evaluate all particles
        current_vals = np.array([evaluate(p) for p in particles])
        for i in range(pop_size):
            if current_vals[i] < personal_best_vals[i]:
                personal_best_vals[i] = current_vals[i]
                personal_best[i] = particles[i].copy()
        # Update global best
        best_idx = np.argmin(personal_best_vals)
        if personal_best_vals[best_idx] < global_best_val:
            global_best_val = personal_best_vals[best_idx]
            global_best = personal_best[best_idx].copy()

        # Update velocities and positions
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = 0.7 * velocities[i] + 1.5 * r1 * (personal_best[i] - particles[i]) + 1.5 * r2 * (global_best - particles[i])
            particles[i] += velocities[i]
            # Clip and normalize
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] *= (target_sum / np.sum(particles[i]))

        # Early termination if no improvement
        if np.abs(global_best_val - np.min(personal_best_vals)) < 1e-6:
            break

    # Final evaluation
    final_val = evaluate(global_best)
    return (global_best, final_val, n_points)
