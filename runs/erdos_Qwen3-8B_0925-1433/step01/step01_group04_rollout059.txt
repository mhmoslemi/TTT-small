import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_points = len(h_initial)
    else:
        n_points = 320
        x = np.linspace(0, 2, n_points)
        h_initial = 0.5 * (1 + np.cos(np.pi * x))
        required_sum = n_points / 2
        h_initial *= required_sum / np.sum(h_initial)
    
    def evaluate(h_flat):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        c_values = corr * (2.0 / n_points)
        return np.max(c_values)
    
    def constraint(h_flat):
        return np.sum(h_flat.reshape(n_points)) - n_points / 2
    
    bounds = [(0, 1)] * n_points
    initial_guess = h_initial.flatten()
    
    start_time = time.time()
    
    result = minimize(
        fun=evaluate,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        best_h = initial_guess.reshape(n_points)
        best_c5 = evaluate(best_h)
        return (best_h, best_c5, n_points)
    
    best_h = result.x.reshape(n_points)
    best_c5 = evaluate(best_h)
    
    for _ in range(3):
        noise = np.random.normal(0, 0.05, n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        required_sum = n_points / 2
        perturbed *= required_sum / np.sum(perturbed)
        result_perturbed = minimize(
            fun=evaluate,
            x0=perturbed.flatten(),
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
        )
        if result_perturbed.fun < best_c5:
            best_h = result_perturbed.x.reshape(n_points)
            best_c5 = result_perturbed.fun
    
    best_h = np.clip(best_h, 0, 1)
    required_sum = n_points / 2
    best_h *= required_sum / np.sum(best_h)
    
    return (best_h, best_c5, n_points)
