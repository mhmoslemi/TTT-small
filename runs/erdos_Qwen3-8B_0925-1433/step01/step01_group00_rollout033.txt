    initial_guesses.append(half_pattern)

    # Random guess
    rng = np.random.RandomState(seed)
    random_guess = rng.rand(n_points)
    random_sum = np.sum(random_guess)
    random_guess = np.clip(random_guess * (n_points / 2) / random_sum, 0.0, 1.0)
    initial_guesses.append(random_guess)

    best_h = h0.copy()
    best_c5 = obj(h0)

    for guess in initial_guesses:
        try:
            result = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons,
                              options={'ftol': 1e-9, 'maxiter': 500, 'disp': False})
            h_candidate = result.x
            c5_candidate = obj(h_candidate)
            if np.any(h_candidate < 0) or np.any(h_candidate > 1):
                h_candidate = np.clip(h_candidate, 0.0, 1.0)
                sum_candidate = np.sum(h_candidate)
                scale = (n_points / 2) / sum_candidate
                h_candidate = h_candidate * scale
            else:
                sum_candidate = np.sum(h_candidate)
                if abs(sum_candidate - n_points / 2) > 1e-6:
                    scale = (n_points / 2) / sum_candidate
                    h_candidate = h_candidate * scale
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
        except:
            pass

    return (best_h, best_c5, n_points)
