from scipy.optimize import minimize
import time

    dx = 2.0 / n_points

    n_particles = 50
    particles = np.array([h_initial + np.random.normal(0, 0.05, n_points) for _ in range(n_particles)])
    particles = np.clip(particles, 0, 1)
    particles = particles * (n_points / 2) / np.sum(particles, axis=1, keepdims=True)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    inertia = 0.8
    inertia_decay = 0.001
    c1, c2 = 2.0, 2.0
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    personal_best = particles.copy()
    personal_best_fitness = np.array([objective(p) for p in particles])
    global_best = particles[np.argmin(personal_best_fitness)]
    global_best_fitness = np.min(personal_best_fitness)
    
    for _ in range(500):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        r1, r2 = np.random.rand(n_particles), np.random.rand(n_particles)

        particles = particles * (n_points / 2) / np.sum(particles, axis=1, keepdims=True)
        
        current_fitness = np.array([objective(p) for p in particles])
