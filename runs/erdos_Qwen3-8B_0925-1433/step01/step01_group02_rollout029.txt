from scipy.optimize import minimize
from copy import deepcopy

        h = np.random.rand(n_points)
        h = (h / np.sum(h) * n_points / 2).clip(0, 1)
        return h
    
    # Fill population with random patterns
    for _ in range(4):
        h_pop.append(generate_rand_pattern())
    
    # Genetic algorithm parameters
    max_gen = 150
    pop_size = 15
    elite_frac = 0.2
    mutation_rate = 0.15
    
    best_h = deepcopy(h_pop[0])
    best_c5 = np.correlate(best_h, 1 - best_h, mode='full').max() * dx
    
    for gen in range(max_gen):
        # Evaluate fitness
        fitness = []
        for h in h_pop:
            c = np.correlate(h, 1 - h, mode='full')
            fit = c.max() * dx
            fitness.append((fit, h))
        fitness.sort()
        
        # Select parents
        selected = [f[1] for f in fitness[:int(pop_size * elite_frac)]]
        new_pop = selected[:]
        
        # Crossover and mutation
        while len(new_pop) < pop_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child = np.zeros_like(parent1)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = parent1[i]

                    child[i] = parent2[i]
            # Mutation
            for i in range(n_points):

                    child[i] += np.random.normal(0, 0.05)
                    child[i] = np.clip(child[i], 0, 1)
            # Normalize sum
