def run(seed=42, budget_s=600, **kwargs):

        sm = np.sum(particles[i])

    velocities = np.random.uniform(-0.1, 0.1, (pop_size, n_points))
    pbest = particles.copy()
    pbest_c5 = np.full(pop_size, np.inf)

    gbest_c5 = np.inf
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
