import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    # Genetic Algorithm setup
    pop_size = 50
    gen_limit = 100
    mut_rate = 0.2
    pop = np.random.normal(0, 0.1, (pop_size, n_points))
    pop = np.clip(pop, 0, 1)
    pop = (pop * (n_points / 2) / np.sum(pop, axis=1, keepdims=True))
    pop_fitness = np.array([objective(ind) for ind in pop])

    for _ in range(gen_limit):
        if time.time() - start_time > budget_s - 10:
            break
        # Tournament selection
        parents = np.empty((pop_size, n_points))
        for i in range(pop_size):
            candidates = np.random.choice(range(pop_size), 5, replace=False)
            idx = candidates[np.argmin(pop_fitness[candidates])]
            parents[i] = pop[idx]
        # Mutation
        for i in range(pop_size):
            if np.random.rand() < mut_rate:
                noise = np.random.normal(0, 0.1, n_points)
                parents[i] += noise
                parents[i] = np.clip(parents[i], 0, 1)
                parents[i] = (parents[i] * (n_points / 2)) / np.sum(parents[i])
        # Crossover
        new_pop = np.empty((pop_size, n_points))
        for i in range(pop_size):
            p1, p2 = np.random.choice(range(pop_size), 2, replace=False)
            new_pop[i] = (parents[p1] + parents[p2]) / 2
        new_pop = np.clip(new_pop, 0, 1)
        new_pop = (new_pop * (n_points / 2)) / np.sum(new_pop, axis=1, keepdims=True)
        # Evaluate
        new_fitness = np.array([objective(ind) for ind in new_pop])
        # Selection
        combined = np.vstack((pop, new_pop))
        fitness = np.hstack((pop_fitness, new_fitness))
        indices = np.argsort(fitness)
