import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scale = (n_points / 2) / total if total > 0 else 1.0
        return h * scale

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h, mutation_type):
        h = h.copy()
        if mutation_type == 'additive':
            noise = np.random.normal(0, 0.05, len(h))
            h += noise
        elif mutation_type == 'binary':
            mask = np.random.rand(len(h)) < 0.1
            h[mask] = np.random.rand(np.sum(mask))
        elif mutation_type == 'periodic':
            h = h * np.sin(0.5 * np.pi * np.random.rand(len(h)))
        return clip_and_normalize(h)

    pop_size = 50
    population = [initial_h_values.copy() + np.random.normal(0, 0.05, n_points)]
    for _ in range(pop_size - 1):
        h_rand = np.random.rand(n_points)
        population.append(clip_and_normalize(h_rand))

    best_h = population[0]
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()

    for gen in range(1000):
        c5s = [evaluate_c5(h) for h in population]
        best_idx = np.argmin(c5s)
        if c5s[best_idx] < best_c5:
            best_c5 = c5s[best_idx]
            best_h = population[best_idx]

        elite_size = int(pop_size * 0.2)
        elites = np.argsort(c5s)[:elite_size]
        new_pop = [population[i] for i in elites]
