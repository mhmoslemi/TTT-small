import numpy as np
from scipy.optimize import minimize, differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    h = np.array([])  # Placeholder
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 42
        h = np.random.uniform(0, 1, n_points)
        dx = 2.0 / n_points
        sum_h = np.sum(h) * dx
        if np.isclose(sum_h, 0):
            h = np.full(n_points, 0.5)
        else:
            scale = (1.0 / sum_h) * (n_points / 2.0)
            h *= scale
    n_points = len(h)
    dx = 2.0 / n_points

    def obj(h_vals):
        cc = correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(cc) * dx

    def constraint(h_vals):
        return np.sum(h_vals) - n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Local optimization
    result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8})
    h_slsqp = result.x
    h_slsqp = np.clip(h_slsqp, 0.0, 1.0)
    sum_clipped = np.sum(h_slsqp)
    scale = (n_points / 2.0) / sum_clipped if sum_clipped != 0 else 1.0
    h_slsqp *= scale

    # Global optimization
    result_de = differential_evolution(obj, bounds, strategy='best1bin', maxiter=100, popsize=20, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
    h_de = result_de.x
    h_de = np.clip(h_de, 0.0, 1.0)
    sum_clipped = np.sum(h_de)
    scale = (n_points / 2.0) / sum_clipped if sum_clipped != 0 else 1.0
    h_de *= scale

    # Select better solution
    h_vals = [h_slsqp, h_de]
    best = h_slsqp
    best_c5 = obj(h_slsqp)
    for h_cand in h_vals:
        c5 = obj(h_cand)
        if c5 < best_c5:
            best = h_cand
            best_c5 = c5

    # Final clipping and scaling
    best = np.clip(best, 0.0, 1.0)
    sum_clipped = np.sum(best)
    scale = (n_points / 2.0) / sum_clipped if sum_clipped != 0 else 1.0
    best *= scale

    return (best, best_c5, n_points)
