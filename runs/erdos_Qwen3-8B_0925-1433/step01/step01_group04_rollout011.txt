        options={'maxiter': 200, 'ftol': 1e-9, 'disp': False}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h *= required_sum / np.sum(best_h)

    cc = correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(cc * dx)
    return (best_h, c5_bound, n_points)
