from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

        return np.max(corr) * dx
    
    def normalize(h):
        s = np.sum(h)
        if abs(s) < 1e-10:
            return np.zeros_like(h) + 0.5
        return h * (n_points / 2) / s
    
    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    for _ in range(5):

        h = best_h + perturb
        h = np.clip(h, 0.0, 1.0)
        h = normalize(h)
        
        res = minimize(objective, h, method='SLSQP', bounds=[(0,1)]*n_points, 
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2})
        
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
