        indices = np.argsort(combined_fitness)
        population = [combined[i] for i in indices[:len(population)]]
        
        # Local refinement
        candidates = [population[i] for i in np.argsort(combined_fitness)[:10]]
        for h_candidate in candidates:
            # Find worst k
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            max_idx = np.argmax(corr)
            k = (max_idx - (n_points - 1)) * dx
            # Optimize for this k
            def obj(x):
                h = x.copy()
                sum_h = np.sum(h)
                if sum_h == 0:
                    return 1e6
                scale = (n_points / 2) / sum_h
                h = np.clip(h * scale, 0, 1)
                # Compute overlap for specific k
                if np.abs(k) > 2:
                    return 1e6
                shifted = h_candidate.copy()
                # Handle shift
                if k < 0:
                    shifted = np.roll(shifted, int(-k / dx))
                elif k > 0:
                    shifted = np.roll(shifted, int(k / dx))
                # Truncate to [0, 2]
                long = np.zeros(n_points)
                long[:(n_points - int(k / dx))] = shifted[int(k / dx):]
                overlap = np.sum(h * long)
                return overlap
            
            result = minimize(obj, h_candidate, method='SLSQP', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}])
            if result.success:
                h_opt = result.x
                c5_opt = compute_c5(h_opt, dx, n_points)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
