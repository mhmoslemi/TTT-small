from scipy.optimize import differential_evolution, minimize

        initial_h = np.clip(initial_h, 0.0, 1.0)
        initial_sum = np.sum(initial_h)
        scaling = target_sum / initial_sum
        initial_h *= scaling
    else:
        initial_h = np.random.rand(n_points).astype(float)
        initial_h = np.clip(initial_h, 0.0, 1.0)
        initial_sum = np.sum(initial_h)
        scaling = target_sum / initial_sum
        initial_h *= scaling

    def objective(h_vector):
        h = np.array(h_vector, dtype=np.float64)
        current_sum = np.sum(h)
        scaling = target_sum / current_sum
        h_scaled = h * scaling
        h_scaled = np.clip(h_scaled, 0.0, 1.0)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        c5 = np.max(corr) * dx
        return c5

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=50,
        maxiter=300,
        mutation=(0.5, 1.0),
        recombination=0.9,
        tol=1e-6,
        seed=seed,
        polish=True
    )
    best_h = result.x
    h_clipped = np.clip(best_h, 0.0, 1.0)
    current_sum_clipped = np.sum(h_clipped)
    scaling_factor = target_sum / current_sum_clipped
    best_h_scaled = h_clipped * scaling_factor

    # Local refinement with L-BFGS-B
    def objective_refine(x):
        h = x.copy()
        h = np.clip(h, 0.0, 1.0)
        scaling = target_sum / np.sum(h)
        h_scaled = h * scaling
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds_refine = [(0.0, 1.0) for _ in range(n_points)]
    result_refine = minimize(
        objective_refine,
        best_h_scaled,
        method='L-BFGS-B',
        bounds=bounds_refine,
        tol=1e-8,
        options={'maxiter': 100}
    )
    refined_h = result_refine.x
    refined_h = np.clip(refined_h, 0.0, 1.0)
    refined_sum = np.sum(refined_h)
    refined_scaling = target_sum / refined_sum
    final_h = refined_h * refined_scaling

    final_c5 = objective(final_h)
    return (final_h, final_c5, n_points)
