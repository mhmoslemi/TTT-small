from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    import numpy as np
    np.random.seed(seed)
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 70
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        h = np.clip(h, 0, 1)

    population_size = 20
    population = [h]
    for _ in range(population_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand_sum = np.sum(h_rand)
        h_rand = h_rand * (n_points / 2) / h_rand_sum
        h_rand = np.clip(h_rand, 0, 1)
        population.append(h_rand)
    
    max_generations = 100
    mutation_rate = 0.1
    generations = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    def evaluate(h_vals):

    while generations < max_generations:

        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:

        top_indices = np.argsort(fitness)[:int(len(population)*0.5)]
        parents = [population[i] for i in top_indices]

        for _ in range(len(population)):

            child_sum = np.sum(child)
