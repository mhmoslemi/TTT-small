import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

        h = h_vec.reshape(n_points)

        perturb = amplitude * np.sin(2 * np.pi * t)

                best_h = result.x
