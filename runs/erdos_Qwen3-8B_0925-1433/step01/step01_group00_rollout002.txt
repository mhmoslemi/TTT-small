from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):

    np.random.seed(seed)
    guesses = [h0.copy()]
    guesses.append(h0 + np.random.uniform(-0.1, 0.1, size=n_points))
    period = 3
    guesses.append(np.tile([1.0, 0.0, 0.5], n_points // period + 1)[:n_points])

    guesses.append(np.random.choice([0, 1], size=n_points, p=[0.5, 0.5]))

    scaled_guesses = []
    for g in guesses:
        g_clipped = np.clip(g, 0.0, 1.0)
        total = np.sum(g_clipped)
        scale = (n_points / 2) / total if total else 1.0
        scaled_guesses.append(g_clipped * scale)

    best_h = None
    best_c5 = float('inf')
    for guess in scaled_guesses:
