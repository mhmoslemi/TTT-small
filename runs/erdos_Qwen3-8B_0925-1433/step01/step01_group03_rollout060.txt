import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:

        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        lag_index = np.argmax(corr)
        k = lag_index - n_points + 1
        def objective(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            if sum_x == 0:

            scale = (n_points / 2) / sum_x
            x = np.clip(x * scale, 0, 1)
            shifted_idx = (np.arange(n_points) + k) % n_points

        if res.success:
            h = res.x.copy()
            current_c5 = compute_c5(h, dx, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            h += np.random.uniform(-0.1, 0.1, n_points)

            sum_h = np.sum(h)
            if sum_h != 0:
                scale = (n_points / 2) / sum_h
                h = np.clip(h * scale, 0, 1)
            current_c5 = compute_c5(h, dx, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    return (best_h, best_c5, n_points)
