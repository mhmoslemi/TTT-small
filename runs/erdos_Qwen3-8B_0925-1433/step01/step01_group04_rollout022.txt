            if current_value < personal_best_values[i]:
                personal_best[i] = population[i].copy()
                personal_best_values[i] = current_value
            if current_value < global_best_value:
                global_best = population[i].copy()
                global_best_value = current_value

        for i in range(population_size):
            vel = velocities[i]
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            vel = inertia * vel + cognitive * r1 * (personal_best[i] - population[i]) + social * r2 * (global_best - population[i])
            vel = np.clip(vel, -velocity_limit, velocity_limit)
            velocities[i] = vel
            population[i] = population[i] + vel
            population[i] = np.clip(population[i], 0, 1)
            sum_pop = np.sum(population[i])
            population[i] = population[i] * required_sum / sum_pop

    best_c5 = evaluate_C5(global_best)
    return (global_best, best_c5, n_points)
