import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def scale_vec(v):
        v = np.clip(v, 0, 1)
        s = np.sum(v)
        return v * (n_points / 2) / s if s != 0 else v
    
    # Generate initial guesses
    h_guesses = [initial_h_values.copy()]
    
    # Random jitter
    h_guesses.append(scale_vec(initial_h_values + np.random.uniform(-0.1, 0.1, n_points)))
    
    # Periodic pattern
    h_guesses.append(scale_vec(np.array([1,1,0]*n_points[:n_points//3] + [1,1,0][:n_points%3])))
    
    # Cosine pattern
    theta = 2 * np.pi * np.arange(n_points) / n_points
    h_guesses.append(scale_vec(0.5 + 0.5 * np.cos(theta)))
    
    # Random binary
    h_bin = np.zeros(n_points)
    h_bin[:n_points//2] = 1.0
    np.random.shuffle(h_bin)
    h_guesses.append(scale_vec(h_bin))
    
    best_h = h_guesses[0]
    best_c5 = obj(best_h)
    
    for guess in h_guesses:
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points/2}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        try:
            res = minimize(obj, guess, method='COBYLA', bounds=bounds, constraints=cons,
                          options={'ftol': 1e-8, 'maxiter': 200})
            if res.success:
                h = res.x
            else:
                h = scale_vec(guess)
        except:
            h = scale_vec(guess)
            
        current_c5 = obj(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
    
    best_h = scale_vec(best_h)
    return (best_h, best_c5, n_points)
