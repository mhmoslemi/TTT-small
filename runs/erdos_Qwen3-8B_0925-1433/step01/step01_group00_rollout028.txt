        for i in range(N):
            h = population[i]
            v = velocities[i]
            r1, r2 = np.random.rand(2)
            v_new = omega * v + c1 * r1 * (pbest[i] - h) + c2 * r2 * (best_global - h)
            velocities[i] = v_new
            h_new = h + v_new
            h_clipped = np.clip(h_new, 0, 1)
            sum_h_clipped = np.sum(h_clipped)
            scaling = (n_points / 2) / sum_h_clipped
            h_normalized = h_clipped * scaling
            population[i] = h_normalized

        for i in range(N):
            h = population[i]
            c5 = compute_c5(h)
            if c5 < pbest_c5[i]:
                pbest[i] = h.copy()
                pbest_c5[i] = c5
                if c5 < best_global_c5:
                    best_global_c5 = c5
                    best_global = h.copy()

        if best_global_c5 < best_c5:
            best_c5 = best_global_c5
            best_solution = best_global.copy()

    sum_h = np.sum(best_solution)
    if abs(sum_h - n_points / 2) > 1e-6:
        best_solution = h_initial.copy()
        best_c5 = compute_c5(best_solution)

    return (best_solution, best_c5, n_points)
