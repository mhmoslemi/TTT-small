import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        sum_h = np.sum(h)
        penalty = 1e4 * abs(sum_h - n_points / 2)
        return c5 + penalty
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Run differential evolution for global search
    result_de = differential_evolution(objective, bounds, strategy='best1bin', 
                                      popsize=20, tol=1e-6, maxiter=100, mutation=(0.5, 1), 
                                      recombination=0.9, seed=seed)
    best_h_de = result_de.x
    best_c5_de = objective(best_h_de)
    
    # Local refinement with SLSQP
    def objective_slsqp(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    cons_slsqp = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    
    result_slsqp = minimize(objective_slsqp, best_h_de, method='SLSQP', 
                            bounds=bounds_slsqp, constraints=cons_slsqp, 
                            tol=1e-8, options={'maxiter': 500})
    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun
    
    if best_c5 < best_c5_de:
        best_c5 = best_c5
        best_h = best_h
    
    # Additional refinement with targeted perturbations
    # Analyze top correlations and adjust
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    top_lag_idx = np.argsort(corr)[::-1][:5]
    for i in top_lag_idx:
        shift = i - len(best_h) + 1
        if shift > 0:
            best_h[shift:] += 0.05
            best_h[shift:] = np.clip(best_h[shift:], 0, 1)
        else:
            best_h[:shift] += 0.05
            best_h[:shift] = np.clip(best_h[:shift], 0, 1)
        # Re-normalize
        norm = np.sum(best_h) / (n_points / 2)
        best_h /= norm
    
    # Final optimization with perturbed candidate
    result_final = minimize(objective_slsqp, best_h, method='SLSQP', 
                            bounds=bounds_slsqp, constraints=cons_slsqp, 
                            tol=1e-8, options={'maxiter': 500})
    best_h = result_final.x
    best_c5 = result_final.fun
    
    return (best_h, best_c5, n_points)
