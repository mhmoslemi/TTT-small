import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h, dx)

    start_time = time.time()
    temp = 1.0
    cooling_rate = 0.995
    iteration = 0
    max_iter = int(budget_s / 0.1)  # ~10,000 iterations

    while iteration < max_iter and time.time() - start_time < budget_s:
        perturbation = np.random.normal(0, 0.05, n_points)
        new_h = best_h + perturbation
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        if np.abs(new_sum) < 1e-6:
            new_h = best_h.copy()
        else:
            scale = target_sum / new_sum
            new_h = new_h * scale
        new_c5 = compute_c5(new_h, dx)
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            best_h = new_h.copy()
            best_c5 = new_c5
        temp *= cooling_rate
        iteration += 1

    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
