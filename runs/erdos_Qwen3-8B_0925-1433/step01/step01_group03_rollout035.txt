    start_time = time.time()
    iterations = 0
    while time.time() - start_time < budget_s and iterations < 1000:
        iterations += 1
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        displacement = idx_max - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + displacement
            if 0 <= j < n_points:
                grad[i] = 1 - h[j]
        h_new = h - 0.05 * grad
        h_new = np.clip(h_new, 0, 1)
        sum_h_new = np.sum(h_new)
        if sum_h_new != 0:
            scale = (n_points / 2) / sum_h_new
            h_new = np.clip(h_new * scale, 0, 1)
        else:
            h_new = np.ones(n_points) * 0.5
        c5 = compute_c5(h_new, dx, n_points)
        if c5 < best_c5:
            h = h_new
            best_c5 = c5
        best_h = h.copy()
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
