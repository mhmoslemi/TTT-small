from skopt import BayesianOptimization
from skopt.utils import use_named_args
import time

def run(seed=42, budget_s=600, **kwargs):

    n_points = 50
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = initial_h_values if 'initial_h_values' in globals() else None
    if initial_h is not None:
        initial_h = np.array(initial_h, dtype=float)
    else:

    initial_sum = np.sum(initial_h)
    scaling = target_sum / initial_sum
    initial_h *= scaling

    scaling = target_sum / initial_sum
    initial_h *= scaling

    def objective(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        h_scaled = h_clipped * (target_sum / np.sum(h_clipped))
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        c5 = np.max(corr) * dx
        penalty = 1e4 * np.abs(np.sum(h_scaled) - target_sum)
        return c5 + penalty

    bo = BayesianOptimization(func_eval, bounds, n_jobs=1)
    best_solution = initial_h.copy()

    for i in range(300):
        if time.time() - start_time > budget_s - 10:

    final_h = np.clip(best_solution, 0.0, 1.0)
