import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 50
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        initial_h = np.random.rand(n_points).astype(float)
    
    initial_sum = np.sum(initial_h)
    scaling = target_sum / initial_sum
    initial_h = np.clip(initial_h * scaling, 0.0, 1.0)
    
    alpha = 100.0
    
    def objective(h_vector):
        h = np.clip(h_vector, 0.0, 1.0)
        corr = np.correlate(h, 1 - h, mode='full')
        integrals = corr * dx
        max_int = np.max(integrals)
        return (1 / alpha) * np.log(np.sum(np.exp(alpha * (integrals - max_int))))
    
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
    
    result = minimize(
        objective,
        initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-8,
        options={'maxiter': 1000, 'disp': False}
    )
    
    best_h = np.clip(result.x, 0.0, 1.0)
    current_sum = np.sum(best_h)
    scaling_factor = target_sum / current_sum
    best_h_scaled = best_h * scaling_factor
    
    corr = np.correlate(best_h_scaled, 1 - best_h_scaled, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (best_h_scaled, c5_bound, n_points)
