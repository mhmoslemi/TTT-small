        c = np.correlate(h_values, 1.0 - h_values, mode='full')
        return np.max(c) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2

    initial_guesses = []

    # First guess: scaled initial guess
    h_sum = np.sum(initial_h_values)
    h_scaled = initial_h_values * (n_points / 2) / h_sum
    initial_guesses.append(h_scaled)

    # Second guess: binary alternating pattern
    binary_pattern = np.zeros(n_points)
    for i in range(n_points):
        binary_pattern[i] = 1 if i % 2 == 0 else 0
    sum_binary = np.sum(binary_pattern)
    if sum_binary != 0:
        binary_scaled = binary_pattern * (n_points / 2) / sum_binary
    else:
        binary_scaled = np.zeros(n_points)
    initial_guesses.append(binary_scaled)

    # Third guess: random binary pattern
    while True:
        random_pattern = np.random.randint(0, 2, size=n_points)
        sum_random = np.sum(random_pattern)
        if sum_random != 0:
            random_scaled = random_pattern * (n_points / 2) / sum_random
            initial_guesses.append(random_scaled)
            break

    best_c5 = float('inf')
    best_h = None
    for guess in initial_guesses:
        result = minimize(
            fun=objective,
            x0=guess,
            method='SLSQP',
            bounds=[(0., 1.) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 500, 'ftol': 1e-10}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            current_c5 = objective(guess)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = guess

    return best_h, best_c5, n_points
