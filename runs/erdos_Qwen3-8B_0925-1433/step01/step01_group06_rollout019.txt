import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 160
    dx = 2.0 / n_points
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_initial = np.array(initial_h_values)
        m = 5
        x = np.linspace(0, 2, n_points)
        positions = np.linspace(0.2, 1.8, m)
        widths = np.full(m, 0.4)
        weights = np.zeros(m)
        for i in range(m):
            start = positions[i] - widths[i] / 2.0
            end = positions[i] + widths[i] / 2.0
            idx = np.where((x >= start) & (x <= end))[0]
            weights[i] = np.mean(h_initial[idx])
        sum_int = np.sum(weights * widths)
        weights *= 1.0 / sum_int
    else:
        m = 5
        positions = np.linspace(0.2, 1.8, m)
        widths = np.full(m, 0.4)
        weights = np.full(m, 0.2)
        sum_int = np.sum(weights * widths)
        weights *= 1.0 / sum_int

    def generate_h(weights):
        h = np.zeros(n_points)
        for i in range(m):
            start = positions[i] - widths[i] / 2.0
            end = positions[i] + widths[i] / 2.0
            idx = np.where((x >= start) & (x <= end))[0]
            h[idx] = weights[i]
        return h

    def objective(weights):
        h = generate_h(weights)
        cc = np.correlate(h, 1 - h, mode='full')
        return np.max(cc) * dx

    def constraint(weights):
        return np.sum(weights * widths) - 1.0

    result = minimize(objective, weights, method='SLSQP', bounds=[(0, 1)]*m, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 100, 'ftol': 1e-8})
    best_weights = result.x
    h_best = generate_h(best_weights)
    cc = np.correlate(h_best, 1 - h_best, mode='full')
    c5_bound = np.max(cc) * dx
    h_best = np.clip(h_best, 0, 1)
    scale = (n_points / 2.0) / np.sum(h_best)
    h_best *= scale
    return (h_best, c5_bound, n_points)
