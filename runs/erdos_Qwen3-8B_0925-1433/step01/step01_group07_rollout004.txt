        p = particles[i]
        p = np.clip(p, 0, 1)
        sum_p = np.sum(p)
        if sum_p == 0:
            particles[i] = np.zeros_like(p)
        else:
            particles[i] = p * (n_points / 2) / sum_p

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    personal_bests = [p.copy() for p in particles]
    global_best = particles[0]
    best_c5 = compute_c5(global_best)

    start_time = time.time()
    no_improvement = 0

    while time.time() - start_time < budget_s:
        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            vel = velocities[i] * w + c1 * r1 * (personal_bests[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            velocities[i] = vel
            particles[i] += vel
            p = particles[i]
            p = np.clip(p, 0, 1)
            sum_p = np.sum(p)
            if sum_p == 0:
                particles[i] = np.zeros_like(p)
            else:
                particles[i] = p * (n_points / 2) / sum_p

        for i in range(n_particles):
            current_c5 = compute_c5(particles[i])
            if current_c5 < compute_c5(personal_bests[i]):
                personal_bests[i] = particles[i].copy()
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    global_best = particles[i].copy()

        if best_c5 == compute_c5(global_best):
            no_improvement += 1
        else:
            no_improvement = 0

        if no_improvement > 50:
            break

    return (global_best, best_c5, n_points)
