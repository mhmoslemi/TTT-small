        result = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'ftol': 1e-8, 'maxiter': 500, 'disp': False})
        h_best = result.x
        h_best = np.clip(h_best, 0, 1)
        sum_h = np.sum(h_best)
        if abs(sum_h - sum_required) > 1e-6:
            h_best = h_best * (sum_required / sum_h)
            h_best = np.clip(h_best, 0, 1)
        c5 = obj(h_best)
        if c5 < best_c5:
            best_h = h_best
            best_c5 = c5

    sum_h = np.sum(best_h)
    if abs(sum_h - sum_required) > 1e-6:
        best_h = initial_h_values.copy()
        best_c5 = obj(best_h)
    return (best_h, best_c5, n_points)
