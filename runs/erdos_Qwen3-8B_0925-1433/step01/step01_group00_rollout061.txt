            scaling = (n_points / 2) / sum_h_refined
            h_refined *= scaling
            
            population[idx] = h_refined
            c5_values[idx] = obj_refine(h_refined)
            if c5_values[idx] < best_c5:
                best_c5 = c5_values[idx]
                best_h = population[idx]
        
        elapsed = time.time() - start_time
        time_remaining = budget_s - elapsed
        if time_remaining < 0:
            break
    
    return (best_h, best_c5, n_points)
