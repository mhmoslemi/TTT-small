import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600, **kwargs):
    n_points = 80
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        initial_h = np.random.rand(n_points).astype(float)

    initial_sum = np.sum(initial_h)
    scaling = target_sum / initial_sum
    initial_h = np.clip(initial_h, 0.0, 1.0) * scaling
    initial_h = np.clip(initial_h, 0.0, 1.0)
    initial_sum_clipped = np.sum(initial_h)
    scaling = target_sum / initial_sum_clipped
    initial_h = initial_h * scaling
    initial_h = np.clip(initial_h, 0.0, 1.0)

    def de_objective(h_vector):
        h = np.array(h_vector, dtype=np.float64)
        h = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h)
        scaling = target_sum / h_sum
        h_scaled = h * scaling
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        c5 = np.max(corr) * dx
        return c5

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = differential_evolution(
        de_objective,
        bounds,
        strategy='best1bin',
        popsize=50,
        maxiter=200,
        mutation=(0.5, 1.0),
        recombination=0.8,
        tol=1e-7,
        seed=seed,
        polish=True
    )

    best_h_de = result.x
    best_h_de = np.clip(best_h_de, 0.0, 1.0)
    best_sum_de = np.sum(best_h_de)
    scaling_de = target_sum / best_sum_de
    best_h_de_scaled = best_h_de * scaling_de
    best_h_de_scaled = np.clip(best_h_de_scaled, 0.0, 1.0)

    alpha = 100.0

    def local_objective(h_vector):
        h = np.clip(h_vector, 0.0, 1.0)
        h_sum = np.sum(h)
        if h_sum == 0:
            return np.inf
        h_scaled = h * (target_sum / h_sum)

        return soft_max_val * dx

    result_local = minimize(
        local_objective,
        best_h_de_scaled,
        method='L-BFGS-B',
        bounds=bounds,
        tol=1e-8,
        options={'maxiter': 200}
    )

    best_h_local = result_local.x
