import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = evaluate_erdos_solution()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_required = n_points / 2.0

    initial_sum = np.sum(initial_h_values)
    scale_factor = sum_required / initial_sum
    initial_h_values = initial_h_values * scale_factor

    n_particles = 50
    particles = np.random.randn(n_particles, n_points) * 0.05 + initial_h_values
    particles = np.clip(particles, 0, 1)
    particles = particles * (sum_required / np.sum(particles, axis=1, keepdims=True))

    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    pbest = particles.copy()
    gbest = particles[0].copy()
    best_c5 = np.max(np.correlate(gbest, 1 - gbest, mode='full')) * dx

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        w = 0.8 - 0.001 * _
        c1, c2 = 2.0, 2.0
        r1, r2 = np.random.rand(n_particles), np.random.rand(n_particles)
        velocities = w * velocities + c1 * r1 * (pbest - particles) + c2 * r2 * (gbest - particles)
        particles = particles + velocities
        particles = np.clip(particles, 0, 1)
        for i in range(n_particles):
            particles[i] = particles[i] * (sum_required / np.sum(particles[i]))
            curr_c5 = compute_c5(particles[i])
            if curr_c5 < compute_c5(pbest[i]):
                pbest[i] = particles[i].copy()
                if curr_c5 < best_c5:
                    best_c5 = curr_c5
                    gbest = pbest[i].copy()
    return (gbest, best_c5, n_points)
