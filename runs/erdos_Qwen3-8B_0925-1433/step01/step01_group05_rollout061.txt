    initial_h = initial_h_values

    sum_h = np.sum(initial_h)

    initial_h = initial_h * scale
    initial_h = np.clip(initial_h, 0.0, 1.0)

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # SA phase
    current_h = initial_h.copy()

    best_h = current_h.copy()
    best_c5 = current_c5

    # SA parameters
    n_iterations = 1000
    initial_temperature = 1.0
    cooling_rate = 0.99

    start_time = time.time()

    for iteration in range(n_iterations):
        # Generate a new candidate
        perturb = np.random.normal(0, 0.05, size=n_points)
        new_h = current_h + perturb
        new_h = np.clip(new_h, 0.0, 1.0)
        sum_new = np.sum(new_h)
        if sum_new > 0:
            new_h *= (n_points / 2) / sum_new
        new_c5 = objective(new_h)

        # Acceptance probability
        delta = new_c5 - best_c5
        if delta < 0:
            current_h = new_h.copy()
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            temp = initial_temperature * (cooling_rate ** iteration)
            if np.random.rand() < np.exp(-delta / temp):
                current_h = new_h.copy()
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
        # Check time

    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
