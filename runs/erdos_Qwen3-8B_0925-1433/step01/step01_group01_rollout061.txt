import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    n_points = 50
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = initial_h_values if 'initial_h_values' in globals() else None
    if initial_h is not None:
        initial_h = np.array(initial_h, dtype=float)
        initial_h = np.clip(initial_h, 0.0, 1.0)
        initial_sum = np.sum(initial_h)
        scaling = target_sum / initial_sum
        initial_h = initial_h * scaling
    else:
        initial_h = np.random.rand(n_points).astype(float)
        initial_h = np.clip(initial_h, 0.0, 1.0)
        initial_sum = np.sum(initial_h)
        scaling = target_sum / initial_sum
        initial_h = initial_h * scaling

    def objective(h_vector):
        h = np.array(h_vector)
        corr = np.correlate(h, 1 - h, mode='full')
        corr_vals = corr * dx
        max_corr = np.max(corr_vals)
        corr_norm = corr_vals - max_corr
        alpha = 100.0
        soft_max_val = (1.0 / alpha) * np.log(np.sum(np.exp(alpha * corr_norm)))
        return soft_max_val

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def equality_constraint(h_vector):
        return np.sum(h_vector) - target_sum

    cons = [{'type': 'eq', 'fun': equality_constraint}]
    result = minimize(
        objective,
        initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h_sum = np.sum(best_h)
    scaling = target_sum / best_h_sum
    best_h_scaled = best_h * scaling
    corr = np.correlate(best_h_scaled, 1 - best_h_scaled, mode='full')
    final_c5 = np.max(corr) * dx
    return (best_h_scaled, final_c5, n_points)
