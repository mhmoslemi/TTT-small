import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    
    for i in range(n_particles):
        pos = initial_h_values.copy()
        noise = np.random.normal(0, 0.05, n_points)
        pos += noise
        pos = np.clip(pos, 0, 1)
        pos *= (n_points / 2) / np.sum(pos)
        particles[i] = pos
        velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
    
    personal_best = particles.copy()
    global_best = np.min(particles, axis=0)
    best_c5 = np.inf
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    w = 0.8
    c1 = c2 = 2.0
    max_iter = 500
    start_time = time.time()
    
    for iter in range(max_iter):
        # Evaluate objectives
        c5_values = np.array([compute_c5(p) for p in particles])
        for i in range(n_particles):
            if c5_values[i] < compute_c5(personal_best[i]):
                personal_best[i] = particles[i].copy()
                if c5_values[i] < best_c5:
                    best_c5 = c5_values[i]
                    global_best = particles[i].copy()
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = w * velocities[i] + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] *= (n_points / 2) / np.sum(particles[i])
        
        # Adjust inertia weight
        if iter % 100 == 0 and w > 0.4:
            w -= 0.01
        
        # Check time budget
        if time.time() - start_time >= budget_s:
            break
    
    best_h = global_best
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
