    initial_h = initial_h_values * scale
    initial_h = np.clip(initial_h, 0.0, 1.0)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx
    start_time = time.time()
    temp = 1.0
    cooling_rate = 0.99
    
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        noise = np.random.normal(0, temp * 0.1, n_points)
        new_h = initial_h + noise
        new_h = np.clip(new_h, 0.0, 1.0)
        new_sum = np.sum(new_h)
        if new_sum != 0:
            scale_factor = (n_points / 2.0) / new_sum
            new_h *= scale_factor
        current_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx
        new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            initial_h = new_h.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
        temp *= cooling_rate
    
    return (best_h, best_c5, n_points)
