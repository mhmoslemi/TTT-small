        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        s = max_corr_idx - (n_points - 1)

        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + s
            if 0 <= j < n_points:
                grad[i] = 1 - h[j]
            else:
                grad[i] = 0

        h_new = h - 0.05 * grad
        h_new = np.clip(h_new, 0, 1)
        sum_new = np.sum(h_new)
        if sum_new == 0:
            h_new = np.ones(n_points) * 0.5
        else:
            scale = (n_points / 2) / sum_new
            h_new = np.clip(h_new * scale, 0, 1)

        current_c5 = compute_c5(h_new, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
        h = h_new.copy()

        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
