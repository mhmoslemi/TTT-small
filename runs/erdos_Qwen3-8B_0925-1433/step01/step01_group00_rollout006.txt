import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(95)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def normalize(h):
        h = np.clip(h, 0, 1)
        scale = target_sum / np.sum(h)
        return h * scale
    
    def mutate(h, method):
        if method == 'add':
            return h + 0.1 * (np.random.rand(n_points) * 2 - 1)
        elif method == 'bin':
            return np.where(np.random.rand(n_points) < 0.1, 1 - h, h)
        elif method == 'per':
            return h * np.sin(0.1 * np.random.rand(n_points) + 1.0)
        return h
    
    def sbx(parent1, parent2, eta):
        alpha = np.random.rand(n_points) * 2 - 1
        beta = (1 + 2 * np.abs(alpha)) ** (-eta)
        child = (parent1 * alpha + parent2 * (1 - alpha)) * beta
        return np.clip(child, 0, 1)
    
    def local_refine(h):
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        res = minimize(evaluate, h, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-9, 'maxiter': 100})
        return normalize(res.x)
    
    pop_size = 50
    population = [normalize(initial_h + 0.1 * np.random.rand(n_points))] + [normalize((np.random.rand(n_points) > 0.5).astype(float)) for _ in range(pop_size - 1)]
    best_h = population[0]
    best_c5 = evaluate(best_h)
    no_improvement = 0
    
    for gen in range(2000):
        if gen % 10 == 0:
            elapsed = gen * 0.1
            if elapsed > budget_s:
                break
        
        fit = [evaluate(h) for h in population]
        sorted_idx = np.argsort(fit)
        elites = [population[i] for i in sorted_idx[:int(pop_size * 0.2)]]
        
        new_pop = elites[:]
        for _ in range(pop_size - len(elites)):
            p1, p2 = np.random.choice(elites, 2, replace=False)
            child = sbx(p1, p2, 10)
            new_pop.append(child)
        
        for i in range(len(new_pop)):
            method = np.random.choice(['add', 'bin', 'per'])
            new_pop[i] = mutate(new_pop[i], method)
            new_pop[i] = normalize(new_pop[i])
        
        population = new_pop
        current_best = min(fit)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[np.argmin(fit)]
            no_improvement = 0
        else:
            no_improvement += 1
        
        if no_improvement >= 5:
            break
    
    refined = local_refine(best_h)
    final_c5 = evaluate(refined)
    return (refined, final_c5, n_points)
