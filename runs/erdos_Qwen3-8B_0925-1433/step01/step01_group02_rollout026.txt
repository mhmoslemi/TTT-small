from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):

    if 'initial_h_values' in globals():

        return np.sum(h_vals) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

        res = minimize(obj, scaled, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 200, 'ftol': 1e-10})
