    # Generate binary alternating pattern
    binary_pattern = np.zeros(n_points)
    binary_pattern[::2] = 1.0
    sum_binary = np.sum(binary_pattern)
    scaling = (n_points / 2.0) / sum_binary
    binary_pattern = binary_pattern * scaling
    initial_guesses.append(binary_pattern)

    best_c5 = float('inf')
    best_h = None

    for guess in initial_guesses:
        sum_guess = np.sum(guess)
        if sum_guess == 0:
            guess[0] = 1.0
            sum_guess = 1.0
        scaling = (n_points / 2.0) / sum_guess
        scaled_guess = guess * scaling

        def objective(h_values):
            corr = np.correlate(h_values, 1 - h_values, mode='full')
            return np.max(corr) * dx

        def constraint_sum(h_values):
            return np.sum(h_values) - n_points / 2.0

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraint_sum}]

        result = minimize(
            fun=objective,
            x0=scaled_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-10}
        )

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
