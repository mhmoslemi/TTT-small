import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    h0 = initial_h_values.copy()
    dx = 2.0 / n_points

    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h0
    best_c5 = obj(h0)
    initial_guesses = [h0]

    # Generate structured patterns
    for p in [3, 4, 5, 6]:
        h_p = np.zeros(n_points)
        for i in range(n_points):
            if (i % p) < (p // 2):
                h_p[i] = 1
        curr_sum = np.sum(h_p)
        target = n_points / 2
        h_p[:int(target - curr_sum)] = 1
        initial_guesses.append(h_p)

    np.random.seed(seed)
    h_rand = np.zeros(n_points)
    h_rand[np.random.permutation(n_points)[:int(n_points // 2)]] = 1
    initial_guesses.append(h_rand)

    start_time = time.time()
    for guess in initial_guesses:
        if np.any(guess < 0) or np.any(guess > 1):
            guess = np.clip(guess, 0, 1)
            curr_sum = np.sum(guess)
            target = n_points / 2
            h_p = np.copy(guess)
            h_p[:int(target - curr_sum)] = 1
            guess = h_p
        result = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'ftol': 1e-8, 'maxiter': 500, 'disp': False})
        if result.success:

            curr_c5 = obj(curr_h)
            if curr_c5 < best_c5:
                best_h = curr_h
                best_c5 = curr_c5
        else:
            curr_c5 = obj(guess)
            if curr_c5 < best_c5:
                best_h = guess
                best_c5 = curr_c5
        elapsed = time.time() - start_time
