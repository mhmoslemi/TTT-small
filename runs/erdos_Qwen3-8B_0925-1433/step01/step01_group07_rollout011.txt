import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_sum = np.sum(initial_h_values)
    if np.abs(initial_sum - n_points / 2) > 1e-6:
        initial_h_values *= (n_points / 2) / initial_sum
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h, dx)

    def normalize(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h) + 0.5
        scaled = h * (n_points / 2) / s
        return np.clip(scaled, 0, 1)

    def compute_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    pop_size = 50
    generations = 50
    mutation_rate = 0.2
    population = [normalize(initial_h_values + np.random.normal(0, 0.05, n_points)) for _ in range(pop_size)]
    start_time = time.time()

    for gen in range(generations):
        fitness = np.array([compute_c5(ind, dx) for ind in population])
        top_indices = np.argsort(fitness)[:int(pop_size * 0.5)]
        parents = [population[i] for i in top_indices]
        new_pop = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                p1 = parents[i]
                p2 = parents[i + 1]
                alpha = np.random.rand()
                child = alpha * p1 + (1 - alpha) * p2
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[j] += np.random.normal(0, 0.1)
                        child[j] = np.clip(child[j], 0, 1)
                child = normalize(child)
                new_pop.append(child)
            else:
                new_pop.append(parents[i])
        population = new_pop
        if np.any(np.abs(fitness - fitness[0]) < 1e-6):
            break
        current_best = np.argmin(fitness)
        if fitness[current_best] < best_c5:
            best_c5 = fitness[current_best]
            best_h = population[current_best].copy()
        if time.time() - start_time > budget_s:
            break

    constraints = ({'type': 'eq', 'fun': lambda h_vec: np.sum(h_vec) - n_points / 2})
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
    if result.success:
        best_c5 = result.fun
        best_h = result.x
    return (best_h, best_c5, n_points)
