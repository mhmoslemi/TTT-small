import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_guesses = []
    np.random.seed(seed)
    # Original initial guess
    initial_guesses.append(initial_h_values.copy())
    # Random jitter
    jitter = np.random.uniform(-0.1, 0.1, n_points)
    guess = initial_h_values + jitter
    guess = np.clip(guess, 0, 1)
    guess = guess * (n_points / 2) / np.sum(guess)
    initial_guesses.append(guess)
    # Periodic pattern (period 3)
    guess = np.array([1 if i % 3 == 0 else 0 for i in range(n_points)])
    guess = np.clip(guess, 0, 1)
    guess = guess * (n_points / 2) / np.sum(guess)
    initial_guesses.append(guess)
    # Cosine-based pattern
    h = 0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n_points) / n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    initial_guesses.append(h)
    # Random binary
    guess = np.random.binomial(1, 0.5, n_points)
    guess = np.clip(guess, 0, 1)
    guess = guess * (n_points / 2) / np.sum(guess)
    initial_guesses.append(guess)
    # Optimization setup
    bounds = [(0.0, 1.0) * np.ones(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    best_h = initial_h_values.copy()
    best_c5 = float('inf')
    for guess in initial_guesses:
        h_clipped = np.clip(guess, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            continue
        h_scaled = h_clipped * (n_points / 2) / sum_clipped
        def obj(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        result = minimize(obj, h_scaled, method='COBYLA', bounds=bounds, constraints=cons,
                          options={'ftol': 1e-8, 'maxiter': 200})
        if result.success:
            current_h = result.x
            current_c5 = result.fun
        else:
            current_h = h_scaled
            current_c5 = obj(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    # Final feasibility check
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if abs(sum_best - n_points / 2) > 1e-6:
        scale_factor = (n_points / 2) / sum_best
        best_h = best_h * scale_factor
    return (best_h, best_c5, n_points)
