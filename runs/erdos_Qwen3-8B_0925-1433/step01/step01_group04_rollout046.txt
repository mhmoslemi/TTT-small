from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
        if np.sum(h) != n_points / 2:
            h *= (n_points / 2) / np.sum(h)
    else:
        n_points = 320
        h = np.random.rand(n_points)
        h *= (n_points / 2) / np.sum(h)
    dx = 2.0 / n_points
    required_sum = n_points / 2

    pop_size = 20
    inertia = 0.8
    cogn_coeff = 1.5
    soc_coeff = 1.5
    vel_limit = 0.1

    particles = h.copy().reshape(1, -1)
    velocities = np.random.uniform(-vel_limit, vel_limit, (1, n_points))
    pbest = particles.copy()
    pbest_values = np.full(pop_size, np.inf)
    gbest = particles.copy()
    gbest_value = np.inf

    def evaluate(h_flat):
        cc = correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(cc * dx)

    start_time = time.time()
    for _ in range(10000):
        if time.time() - start_time > budget_s - 10:
            break
        for i in range(pop_size):
            h_flat = particles[i]
            curr_val = evaluate(h_flat)
            if curr_val < pbest_values[i]:
                pbest[i] = h_flat.copy()
                pbest_values[i] = curr_val
        gbest_val = np.min(pbest_values)
        if gbest_val < gbest_value:
            gbest_value = gbest_val
            gbest = pbest[pbest_values == gbest_val][0]
        for i in range(pop_size):
            r1, r2 = np.random.rand(2)
            vel = velocities[i] * inertia + cogn_coeff * r1 * (pbest[i] - particles[i]) + soc_coeff * r2 * (gbest - particles[i])
            velocities[i] = np.clip(vel, -vel_limit, vel_limit)
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] *= required_sum / np.sum(particles[i])
    best_h = gbest
    cc = correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(cc * dx)
    return (best_h, c5_bound, n_points)
