    dx = 2.0 / n_points
    for i in range(n_points):
        x = i * dx
        for a, l in intervals:
            if a <= x < a + l:
                h_values[i] = 1
                break
    
    sum_h = np.sum(h_values) * dx
    if abs(sum_h - 1.0) > 1e-6:
        scale = (1.0 / sum_h) * (n_points / 2.0)
        h_values *= scale
    
    cc = correlate(h_values, 1 - h_values, mode='full')
    c5_bound = np.max(cc) * dx
    return (h_values, c5_bound, n_points)
