import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    def rescale(h):
        return h * (n_points / 2) / np.sum(h) if np.sum(h) > 0 else h

    def feasible(h):
        return np.clip(h, 0, 1)

    population_size = 50
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = initial_h + np.random.normal(0, 0.1, n_points)
        h = rescale(feasible(h))
        population.append(h)

    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        fitness = [evaluate(ind) for ind in population]
        sorted_indices = np.argsort(fitness)
        sorted_pop = np.array(population)[sorted_indices]
        elite_size = max(1, population_size // 10)
        elites = sorted_pop[:elite_size]
        new_pop = elites.copy()
        while len(new_pop) < population_size:
            parent1, parent2 = np.random.choice(sorted_pop, 2, replace=False)
            child = (parent1 + parent2) / 2
            if np.random.rand() < 0.3:
                noise = np.random.normal(0, 0.1, n_points)
                child += noise
            child = rescale(feasible(child))
            new_pop.append(child)
        population = new_pop.tolist()
        current_best = sorted_pop[0]
        if evaluate(current_best) < best_c5:
            best_c5 = evaluate(current_best)
            best_h = current_best.copy()

    top = np.sort(population)[:5]
    for h in top:
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        result = minimize(evaluate, h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-6)
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    best_h = rescale(feasible(best_h))
