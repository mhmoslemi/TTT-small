import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 160
        h = np.zeros(n_points)
        h[:8] = 1
        h[16:24] = 1
        h[32:40] = 1
        h[48:56] = 1
        h[64:72] = 1
        h = h / np.sum(h) * (n_points / 2)
    n_points = len(h)
    dx = 2.0 / n_points
    h = np.clip(h, 0, 1)
    
    def interval_overlap(a, l, shift):
        overlaps = []
        for i in range(len(a)):
            for j in range(len(a)):
                if i == j:
                    continue
                x_start = a[i]
                x_end = x_start + l[i]
                y_start = a[j] + shift
                y_end = y_start + l[j]
                overlap_start = max(x_start, y_start)
                overlap_end = min(x_end, y_end)
                if overlap_start < overlap_end:
                    overlaps.append(overlap_end - overlap_start)
        return max(overlaps) if overlaps else 0
    
    def objective(x):
        a = x[:len(x)//2]
        l = x[len(x)//2:]
        max_overlap = 0
        for shift in range(1, int(2 / dx) + 1):
            curr = interval_overlap(a, l, shift * dx)
            if curr > max_overlap:
                max_overlap = curr
        return max_overlap
    
    def constraints(x):
        a = x[:len(x)//2]
        l = x[len(x)//2:]
        total_length = np.sum(l)
        sorted_idx = np.argsort(a)
        a_sorted = a[sorted_idx]
        l_sorted = l[sorted_idx]
        non_overlap = []
        for i in range(len(a_sorted) - 1):
            if a_sorted[i+1] < a_sorted[i] + l_sorted[i]:
                non_overlap.append(1)
        return [total_length - 1.0, np.sum(non_overlap)]
    
    n_intervals = 5
    x0 = np.zeros(2 * n_intervals)
    x0[:n_intervals] = np.linspace(0.2, 1.8, n_intervals)
    x0[n_intervals:] = np.full(n_intervals, 0.2)
    bounds = [(0, 2) for _ in range(n_intervals)] + [(0, 2) for _ in range(n_intervals)]
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraints}, options={'maxiter': 200, 'ftol': 1e-8})
    best_a = res.x[:n_intervals]
    best_l = res.x[n_intervals:]
    
    h_values = np.zeros(n_points)
    for i in range(n_points):

        for j in range(n_intervals):
            if best_a[j] <= x < best_a[j] + best_l[j]:
                h_values[i] = 1
                break
    h_values = np.clip(h_values, 0, 1)
    scale = (n_points / 2) / np.sum(h_values)
    h_values *= scale
    h_values = np.clip(h_values, 0, 1)
    
    cc = correlate(h_values, 1 - h_values, mode='full')
    c5_bound = np.max(cc) * dx
