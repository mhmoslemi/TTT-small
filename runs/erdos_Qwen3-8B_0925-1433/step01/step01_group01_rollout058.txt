import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_h = initial_h_values if 'initial_h_values' in globals() else None
    
    if initial_h is None:
        h = np.random.rand(n_points).astype(float)
    else:
        h = np.array(initial_h, dtype=float)
    
    # Scale initial guess
    h = np.clip(h, 0.0, 1.0)
    h_sum = np.sum(h)
    scaling = target_sum / h_sum
    h *= scaling

    def de_objective(h_vec):
        h = np.array(h_vec, dtype=np.float64)
        h_sum = np.sum(h)
        scaling = target_sum / h_sum
        h_scaled = h * scaling
        h_scaled = np.clip(h_scaled, 0.0, 1.0)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_de = differential_evolution(
        de_objective, bounds, strategy='best1bin',
        popsize=50, maxiter=100, mutation=(0.7, 1.0),
        recombination=0.9, tol=1e-6, seed=seed, polish=True,
        disp=False
    )
    best_h = result_de.x

    def local_objective(h_vec):
        h = np.clip(h_vec, 0.0, 1.0)
        h_sum = np.sum(h)
        scaling = target_sum / h_sum
        h_scaled = h * scaling
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        alpha = 100
        soft_max = np.log(np.sum(np.exp(alpha * corr))) / alpha
        penalty = 1e6 * (h_sum - target_sum)**2
        return soft_max * dx + penalty

    result_local = minimize(
        local_objective, best_h, method='L-BFGS-B',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        tol=1e-6, options={'maxiter': 100}
    )
    best_h = result_local.x

    h_clipped = np.clip(best_h, 0.0, 1.0)
    h_sum = np.sum(h_clipped)
    scaling = target_sum / h_sum
    final_h = h_clipped * scaling

    corr = np.correlate(final_h, 1 - final_h, mode='full')
    c5 = np.max(corr) * dx
