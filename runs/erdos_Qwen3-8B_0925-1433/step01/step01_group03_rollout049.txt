So, the new_c5 is computed by this function, which would find the maximum over all k. Hence, the current best solution is the one with the lowest new_c5.

import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Process initial_h_values to ensure valid
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    
    start_time = time.time()
    max_iterations = 200  # Sufficient for convergence within 600s
    
    for _ in range(max_iterations):
        # Compute cross-correlation to find the current worst k
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        k_index = np.argmax(corr)
        
        # Define the objective function
        def obj_func(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return corr[k_index] * dx
        
        # Set up constraints
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        bounds = [(0, 1)] * n_points
        
        # Perform optimization
        res = minimize(obj_func, h, method='SLSQP', bounds=bounds, constraints=constraints)
        if res.success:
            h = res.x
            # Compute new C5
            new_c5 = compute_c5(h, dx, n_points)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        # Check time
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
