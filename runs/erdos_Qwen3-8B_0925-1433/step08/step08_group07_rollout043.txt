            if i > 0:
                grad[i] -= np.sign(h_vec[i] - h_vec[i-1])
            if i < n - 1:
                grad[i] += np.sign(h_vec[i+1] - h_vec[i])
        return grad

    def clip_rescale(h_vec):
        h_clipped = np.clip(h_vec, 0.0, 1.0)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.zeros_like(h_vec)
        return h_clipped * (target_sum / sum_clipped)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    lambda_reg = 0.1
    step_size = 0.5
    max_iter = 300

    start_time = time.time()
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        grad_c5 = compute_gradient_c5(best_h)
        grad_tv = compute_gradient_tv(best_h)
        total_grad = grad_c5 + lambda_reg * grad_tv

        h_new = best_h - step_size * total_grad
        h_new = clip_rescale(h_new)

        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
        else:
            step_size *= 0.95

        if iter % 50 == 0 and iter != 0:
            if new_c5 < best_c5:
                lambda_reg *= 1.1
            else:
                lambda_reg *= 0.9

    return (best_h, best_c5, n_points)
