import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial guess
    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)

    H = np.fft.fft(h)
    f_c = int(0.1 * n_points)
    H_filtered = H.copy()
    H_filtered[f_c+1:] = 0  # zero out higher frequencies
    h_filtered = np.fft.ifft(H_filtered).real
    h_filtered = np.clip(h_filtered, 0, 1)
    h_filtered = h_filtered * target_sum / np.sum(h_filtered) if np.sum(h_filtered) > 0 else h_filtered

    # Define objective function
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    # Constraints
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1)] * n_points

    # Perform SLSQP optimization
    res = minimize(obj, h_filtered, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    best_h = res.x
    best_c5 = res.fun

    # Additional refinement using perturbations
    # Compute worst lag
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    max_k = np.argmax(corr)
    if max_k < n_points - 1:
        # Apply sinusoidal perturbation
        freq = 2 * np.pi * max_k / n_points
        perturb = 0.01 * np.sin(freq * np.arange(n_points))

        if res_perturb.success and res_perturb.fun < best_c5:

            best_c5 = res_perturb.fun

import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)

    H = np.fft.fft(h)
    f_c = int(0.15 * n_points)
    H_filtered = H.copy()
    H_filtered[f_c + 1:] = 0
    h_filtered = np.fft.ifft(H_filtered).real
    h_filtered = np.clip(h_filtered, 0, 1)
    h_filtered = h_filtered * target_sum / np.sum(h_filtered) if np.sum(h_filtered) > 0 else h_filtered

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    start_time = time.time()
    best_h = h_filtered
    best_c5 = obj(best_h)

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if not res.success:
            break
        best_h = res.x
        best_c5 = res.fun

        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        if max_k < n_points - 1:
            freq = 2 * np.pi * max_k / n_points
            perturb = 0.005 * np.sin(freq * np.arange(n_points))
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            res_perturb = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
            if res_perturb.success and res_perturb.fun < best_c5:
                best_h = res_perturb.x
                best_c5 = res_perturb.fun

    return (best_h, best_c5, n_points)
