import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if n_points == 0:
        h = np.array([])
    else:
        if initial_h_values is not None:
            h = initial_h_values.copy()

            h = np.zeros(n_points)

            h[:n_half] = 1.0
            if n_points % 2 == 1:
                h[-1] += 0.5
            h = np.clip(h, 0, 1)
            if np.sum(h) > 0:
                h = h * target_sum / np.sum(h)
            else:
                h = np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    best_h, best_c5 = result.x, result.fun

    for _ in range(15):

        perturb = alpha * np.sin(2 * np.pi * max_lag * np.arange(n_points) / n_points)
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
