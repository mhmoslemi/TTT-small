        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return 1e10
        h_norm = h_clipped * target_sum / sum_h
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    
    start_time = time.time()
    best_h = h_initial
    best_c5 = objective(best_h)
    
    # Differential Evolution phase
    result_de = differential_evolution(
        objective, bounds, popsize=20, strategy='best1bin', tol=1e-6, maxiter=100,
        mutation=(0.5, 1.0), recombination=0.9, disp=False
    )
    best_h = result_de.x
    best_c5 = result_de.fun
    
    # Local refinement with L-BFGS-B
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        result_lbgf = minimize(
            objective, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-8
        )
        if result_lbgf.fun < best_c5:
            best_h, best_c5 = result_lbgf.x, result_lbgf.fun
    
    # Final refinement
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        result_lbgf = minimize(
            objective, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-9
        )
        if result_lbgf.fun < best_c5:
            best_h, best_c5 = result_lbgf.x, result_lbgf.fun
    
    # Final check
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * target_sum / final_sum
    final_c5 = objective(final_h)
    
    return (final_h, final_c5, n_points)
