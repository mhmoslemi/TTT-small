from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600):

    np.random.seed(seed)
    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    # Differential Evolution for global search
    result_de = differential_evolution(obj, bounds, strategy='best1bin',

    # Local refinement with L-BFGS-B
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result_lbgf = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
