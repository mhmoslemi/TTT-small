            grad[i] = (1 - h[j]) * dx

        # Use L-BFGS-B with constraints
        bounds = [(0, 1)] * n_points
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx), h, method='L-BFGS-B',
                       bounds=bounds, constraints=cons, jac=lambda x: grad, tol=1e-8)
        h = res.x
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

        if _ % 50 == 0:
            perturb = np.sin(np.linspace(0, 2 * np.pi, n_points) * 2) * 0.05 * (1 + np.abs(np.roll(h, lag) - h))
            h = np.clip(h + perturb, 0, 1)
            h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h

    return (best_h, best_c5, n_points)
