import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Generate diverse initializations
    h0 = initial_h_values.copy()
    init_list = [h0, 1 - h0]
    x = np.linspace(0, 2, n_points)
    phase = np.random.uniform(0, 2 * np.pi)
    mod = h0 * (1 + 0.5 * np.sin(2 * np.pi * x / 2 + phase))
    mod = np.clip(mod, 0, 1)
    mod = mod * target_sum / np.sum(mod) if np.sum(mod) > 0 else np.zeros(n_points)
    init_list.append(mod)
    np.random.shuffle(h0)
    permut = h0 * target_sum / np.sum(h0) if np.sum(h0) > 0 else np.zeros(n_points)
    init_list.append(permut)
    rand = np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points))
    init_list.append(rand)

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for h in init_list:
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    # Adaptive refinement
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        pert = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + pert, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        result = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    return (best_h, best_c5, n_points)
