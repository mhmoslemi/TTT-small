    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    best_h = h.copy()
    best_c5 = np.inf

    def compute_autocorr(h):
        autocorr = np.correlate(h, h, mode='full')
        return autocorr[n:]

    def compute_gradient(h, k):
        grad = np.zeros_like(h)
        for m in range(len(h)):
            if m + k < len(h):
                grad[m] += h[m + k]
            if m - k >= 0:
                grad[m] += h[m - k]
        return grad

    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        autocorr = compute_autocorr(h)
        if np.allclose(autocorr, autocorr[0]):
            break
        min_idx = np.argmin(autocorr)
        k = min_idx + 1  # Lag 1 to n-1
        grad = compute_gradient(h, k)
        h += 0.05 * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

        curr_corr = compute_autocorr(h)
        curr_c5 = np.max(curr_corr) * dx
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()

    def objective(x):
        corr = np.correlate(x, x, mode='full')
        return np.max(corr[n:]) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    final_h = result.x
    final_c5 = result.fun

    return (final_h, final_c5, n_points)
