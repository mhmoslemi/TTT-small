    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        h = h * scale_factor

    h_fft = np.fft.fft(h)

        h_filtered = h_filtered * scale_factor
    h = h_filtered.copy()
