        m = np.argmax(corr)
        shift_k = m - (n_points - 1)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + shift_k
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * dx
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, jac=grad, tol=1e-8)
        h = res.x
        if res.fun < best_c5:
            best_c5 = res.fun
            best_h = h.copy()
        if _ % 50 == 0:
            perturb = np.sin(np.linspace(0, 2 * np.pi, n_points) * 2) * 0.05 * (1 + np.abs(np.roll(h, m) - h))
            h = np.clip(h + perturb, 0, 1)
            h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h

    return (best_h, best_c5, n_points)
