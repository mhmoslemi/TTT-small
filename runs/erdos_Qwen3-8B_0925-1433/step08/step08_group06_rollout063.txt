import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Rescale initial guess
    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def jac_obj(x, k):
        # Finite difference Jacobian
        eps = 1e-6
        grad = np.zeros_like(x)
        corr = np.correlate(x, 1 - x, mode='full')
        max_k_val = np.max(corr)
        for i in range(n_points):
            x_plus = x.copy()
            x_plus[i] += eps
            corr_plus = np.correlate(x_plus, 1 - x_plus, mode='full')
            grad[i] = (np.max(corr_plus) - max_k_val) / eps
        return grad

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    best_h = h
    best_c5 = obj(h)
    start_time = time.time()

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        # Use L-BFGS-B for faster convergence
        res = minimize(obj, best_h, method='L-BFGS-B',
                      bounds=bounds, constraints=cons,
