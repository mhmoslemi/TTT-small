import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def optimize_pattern(p):
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result = minimize(compute_c5, p, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5, options={'maxiter': 100})
        return result.x

    patterns = [
        np.zeros(n_points),  # baseline
        np.clip(initial_h_values, 0, 1),
        np.zeros(n_points),
        np.zeros(n_points),
        np.zeros(n_points),
    ]

    def even_spaced():
        m = int(n_points / 2)
        points = np.linspace(0, 2, m + 1)[:-1]
        idx = np.searchsorted(np.linspace(0, 2, n_points), points)
        res = np.zeros(n_points)
        res[idx] = 1.0
        res = res * target_sum / np.sum(res)
        return res

    def alternating():
        res = np.zeros(n_points)
        res[::2] = 1.0
        res = res * target_sum / np.sum(res)
        return res

    def gaussian():
        x = np.linspace(0, 2, n_points)
        gauss = np.exp(-((x - 1)**2) / (2 * 0.5**2))
        norm = np.sum(gauss)
        res = gauss * target_sum / norm
        return res

    def random_place():
        idx = np.random.choice(n_points, size=int(n_points / 2), replace=False)
        res = np.zeros(n_points)
        res[idx] = 1.0
        res = res * target_sum / np.sum(res)
        return res

    patterns[0] = even_spaced()
    patterns[1] = alternating()
    patterns[2] = gaussian()
    patterns[3] = random_place()
    patterns[4] = (patterns[0] + patterns[1] + patterns[2] + patterns[3]) / 4

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    for p in patterns:
        optimized = optimize_pattern(p.copy())
        c5 = compute_c5(optimized)
        if c5 < best_c5:
            best_h = optimized.copy()
            best_c5 = c5

    start_time = time.time()
    for _ in range(30):

            break
        pert = np.random.normal(0, 0.05, n_points)

        c5 = compute_c5(new_h)
        if c5 < best_c5:

    return (best_h, best_c5, n_points)
