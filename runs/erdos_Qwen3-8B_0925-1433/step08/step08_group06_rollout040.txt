import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def grad_obj(x, k):
        eps = 1e-6
        grad = np.zeros_like(x)
        for i in range(n_points):
            x_p = x.copy()
            x_p[i] += eps
            corr_p = np.correlate(x_p, 1 - x_p, mode='full')
            grad[i] = (np.max(corr_p) - obj(x)) / eps
        return grad

    def perturb(h, k, amp):
        freq = 2 * np.pi * k / n_points
        pert = amp * np.sin(freq * np.arange(n_points))
        new_h = np.clip(h + pert, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        return new_h

    best_h = h
    best_c5 = obj(best_h)
    start = time.time()

    for _ in range(1200):
        if time.time() - start > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k = np.argmax(corr)
        grad = grad_obj(best_h, k)
        res = minimize(obj, best_h, method='SLSQP',
                       jac=grad, bounds=[(0,1)]*n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
        if res.success:
            best_h = res.x
            best_c5 = res.fun
        else:
            break
        if k < n_points - 1:
            best_h = perturb(best_h, k, 0.02)
            if np.sum(best_h) > 0:
                best_h = best_h * target_sum / np.sum(best_h)
            best_h = np.clip(best_h, 0, 1)
            res_perturb = minimize(obj, best_h, method='SLSQP',
                                   jac=grad_obj(best_h, k), bounds=[(0,1)]*n_points,
                                   constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
            if res_perturb.success and res_perturb.fun < best_c5:
                best_h = res_perturb.x
                best_c5 = res_perturb.fun

    return (best_h, best_c5, n_points)
