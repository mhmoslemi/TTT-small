import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h_initial = np.clip(initial_h_values, 0, 1)
    h_initial = h_initial * target_sum / np.sum(h_initial) if np.sum(h_initial) > 0 else np.zeros(n_points)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx
    
    def objective(h_candidate):
        h = np.clip(h_candidate, 0, 1)
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h > 0 else 1.0
        h = h * scale
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    
    candidates = [h_initial]
    m = int(n_points / 2)
    spacing = 2 / (m + 1)
    even_h = np.zeros(n_points)
    for i in range(m):
        pos = i * spacing
        idx = int(np.round(pos * n_points / 2))
        even_h[idx] = 1
    even_h = even_h * target_sum / np.sum(even_h)
    candidates.append(even_h)
    
    x = np.linspace(0, 2, n_points)
    sine_h = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_h = np.clip(sine_h, 0, 1)
    sine_h = sine_h * target_sum / np.sum(sine_h)
    candidates.append(sine_h)
    
    random_h = np.random.rand(n_points)
    random_h = np.clip(random_h, 0, 1)
    random_h = random_h * target_sum / np.sum(random_h)
    candidates.append(random_h)
    
    start_time = time.time()
    for i, candidate in enumerate(candidates):
        if time.time() - start_time > budget_s * 0.8:
            break
        res = differential_evolution(objective, bounds, strategy='best1bin', 
                                    popsize=15, mutation=(0.5, 1), recombination=0.8,
                                    tol=1e-6, maxiter=50, seed=seed + i)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
