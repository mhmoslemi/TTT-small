import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        H = np.fft.fft(best_h)
        cutoff = int((0.1 + _ * 0.05) * n_points)
        filter_mask = np.zeros_like(H)
        filter_mask[:cutoff] = 1.0
        H_filtered = H * filter_mask
        h_filtered = np.fft.ifft(H_filtered).real
        h_filtered = np.clip(h_filtered, 0, 1)
        h_filtered = h_filtered * target_sum / np.sum(h_filtered) if np.sum(h_filtered) > 0 else h_filtered

        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx

        def cons(x):
            return np.sum(x) - target_sum

        bounds = [(0.0, 1.0)] * n_points
        res = minimize(obj, h_filtered, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': cons}, tol=1e-8)
        if res.success:
            best_h = res.x
            best_c5 = res.fun
        else:
            best_h = h_filtered
            best_c5 = np.max(np.correlate(h_filtered, 1 - h_filtered, mode='full')) * dx

        best_h = np.clip(best_h, 0, 1)
        if np.sum(best_h) < target_sum - 1e-6:
            best_h = best_h * target_sum / np.sum(best_h)
        best_h = np.clip(best_h, 0, 1)

    return (best_h, best_c5, n_points)
