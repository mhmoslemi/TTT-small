def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Rescale initial guess
    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def jac_obj(x, k):
        # Approximate Jacobian using finite differences
        eps = 1e-6
        grad = np.zeros_like(x)
        for i in range(n_points):
            x_plus = np.copy(x)
            x_plus[i] += eps
            corr_plus = np.correlate(x_plus, 1 - x_plus, mode='full')
            grad[i] = (np.max(corr_plus) - obj(x)) / eps
        return grad

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    best_h = h
    best_c5 = obj(h)
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Trust-constr optimization with targeted Jacobian
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        res = minimize(obj, best_h, method='trust-constr',
                      bounds=bounds, constraints=cons,
                      jac=lambda x: jac_obj(x, max_k),
                      tol=1e-6)
        if res.success:
            best_h = res.x
            best_c5 = res.fun
        else:
            break

        # Apply structured perturbations
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        if max_k < n_points - 1:
            # Sinusoidal modulation
            freq = 2 * np.pi * max_k / n_points
            perturb = 0.01 * np.sin(freq * np.arange(n_points))
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            res_perturb = minimize(obj, new_h, method='trust-constr',
                                bounds=bounds, constraints=cons,
                                jac=lambda x: jac_obj(x, max_k),
                                tol=1e-6)
            if res_perturb.success and res_perturb.fun < best_c5:
                best_h = res_perturb.x
                best_c5 = res_perturb.fun

    return (best_h, best_c5, n_points)
