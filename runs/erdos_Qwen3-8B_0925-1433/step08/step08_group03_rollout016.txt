import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_patterns():
        patterns = [initial_h_values.copy()]
        x = np.linspace(0, 2, n_points)
        sin_h = 0.5 + 0.5 * np.sin(3 * np.pi * x)
        sin_h = np.clip(sin_h, 0, 1)
        sin_h = sin_h * target_sum / np.sum(sin_h)
        patterns.append(sin_h)
        num_ones = n_points // 2
        rand_indices = np.random.choice(n_points, size=num_ones, replace=False)
        rand_h = np.zeros(n_points)
        rand_h[rand_indices] = 1
        rand_h = rand_h * target_sum / np.sum(rand_h)
        patterns.append(rand_h)
        return patterns

    patterns = generate_patterns()
    best_h = patterns[0].copy()
    best_c5 = np.inf
    start_time = time.time()

    for h in patterns:
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.03, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
