                best_h, best_c5 = res.x, res.fun
        else:
            non_overlap_start = end_idx + 1
            non_overlap_end = n_points - 1
            if non_overlap_start <= non_overlap_end:
                non_swap_start = np.random.randint(non_overlap_start, non_overlap_end - swap_len + 1)
                non_swap_end = non_swap_start + swap_len
                new_h = best_h.copy()
                new_h[seg_start:seg_end], new_h[non_swap_start:non_swap_end] = new_h[non_swap_start:non_swap_end], new_h[seg_start:seg_end]
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
                res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
                if res.fun < best_c5:
                    best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
