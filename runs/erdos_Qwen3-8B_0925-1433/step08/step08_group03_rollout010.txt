import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population with variations
    population_size = 30
    population = []
    h_base = initial_h_values.copy()
    h_base = h_base * target_sum / np.sum(h_base)
    population.append(h_base)
    for _ in range(population_size - 1):
        h = initial_h_values.copy()
        h += np.random.normal(0, 0.1, h.shape)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)

    best_h = h_base
    best_c5 = compute_C5(best_h)

    max_generations = 200
    start_time = time.time()

    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = []
        for h in population:
            fitness.append(compute_C5(h))
        # Select elite
        elite_idx = np.argsort(fitness)[:int(0.2 * len(population))]
        elite = [population[i] for i in elite_idx]
        # Crossover and mutate
        new_population = []
        for i in range(0, len(elite), 2):
            if i + 1 < len(elite):
                parent1 = elite[i]
                parent2 = elite[i + 1]
                child = 0.5 * (parent1 + parent2)
            else:
                child = elite[i]
            mutation = np.random.normal(0, 0.05, child.shape)
            child += mutation
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_population.append(child)
        population = new_population
        # Track best
        for h in population:
            current_c5 = compute_C5(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    # Local refinement
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if result.success:
        best_h = result.x
        best_c5 = result.fun

    return (best_h, best_c5, n_points)
