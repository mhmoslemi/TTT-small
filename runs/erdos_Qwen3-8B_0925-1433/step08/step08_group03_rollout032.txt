import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_init():
        h = np.random.rand(n_points)
        h = np.sort(h)
        h[1:-1] = 0
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    initial_guesses = [initial_h_values, generate_init()]

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = initial_h_values.copy()
    best_c5 = float('inf')

    start_time = time.time()
    for init in initial_guesses:
        h = np.clip(init, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        if time.time() - start_time > budget_s:
            break

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.03, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    return (best_h, best_c5, n_points)
