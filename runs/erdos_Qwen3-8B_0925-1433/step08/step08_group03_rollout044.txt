    result = minimize(obj, x0, method='SLSQP', bounds=bounds, tol=1e-8)
    best_x = result.x
    best_c5 = result.fun
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Add structured perturbations
        perturb = np.random.normal(0, 0.03, n_vars)
        new_x = np.clip(best_x + perturb, 0.0, 1.0)
        res = minimize(obj, new_x, method='SLSQP', bounds=bounds, tol=1e-8)
        if res.fun < best_c5:
            best_x = res.x
            best_c5 = res.fun
    
    h = np.zeros(n_points)
    for i in range(n_vars):
        if i < n_vars - 1:
            h[i] = best_x[i]
            h[n_points - 1 - i] = 1 - best_x[i]
        elif n_points % 2 == 1 and i == n_vars - 1:
            h[i] = 0.5
    return (h, best_c5, n_points)
