import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def prepare_guess(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        else:
            h = np.zeros_like(h)
        return h

    def optimize_guess(guess):
        h = prepare_guess(guess)
        def obj(x):
            x = prepare_guess(x)
            return compute_c5(x)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        result = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        return result.x, result.fun

    def perturb(h, scale=0.05):
        h = np.clip(h, 0, 1)
        h += np.random.normal(0, scale, n_points)
        h = prepare_guess(h)
        return h

    initial_guesses = [initial_h_values]
    for _ in range(2):
        x = np.linspace(0, 2, n_points)
        freq = 1.2 * (n_points / 2)
        sin_guess = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x)
        sin_guess = prepare_guess(sin_guess)
        initial_guesses.append(sin_guess)
        sin_guess = np.roll(sin_guess, int(n_points * 0.25))
        initial_guesses.append(sin_guess)

    best_h, best_c5 = initial_h_values, compute_c5(initial_h_values)
    start = time.time()

    for guess in initial_guesses:
        if time.time() - start > budget_s:
            break
        h_opt, c5_opt = optimize_guess(guess)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    for _ in range(50):
        if time.time() - start > budget_s:
            break
        best_h = perturb(best_h)
        h_opt, c5_opt = optimize_guess(best_h)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    return (best_h, best_c5, n_points)
