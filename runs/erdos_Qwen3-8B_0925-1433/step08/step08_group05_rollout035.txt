from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    best_h, best_c5 = result.x, result.fun

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Generate structured patterns with spaced 1s
        spaced = np.zeros(n_points)
        spaced[::3] = 1
        spaced = spaced * target_sum / np.sum(spaced) if np.sum(spaced) > 0 else np.zeros(n_points)
        res = minimize(objective, spaced, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(80):
