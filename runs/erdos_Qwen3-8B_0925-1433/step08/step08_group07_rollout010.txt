import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    import numpy as np
    from scipy.optimize import minimize
    import time

    global initial_h_values
    h = np.array(initial_h_values)

    best_c5 = obj(best_h)
    lambda_reg = 0.1

        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-9)
        if res.success:
            curr_c5 = obj(res.x)
            if curr_c5 < best_c5:
                best_h, best_c5 = res.x, curr_c5
                lambda_reg *= 1.05
            else:
                lambda_reg *= 0.95

            perturb = np.random.normal(0, 0.01, n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
