import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective_de(h_candidate):
        h = np.clip(h_candidate, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            return np.inf
        h_normalized = h * target_sum / h_sum
        corr = np.correlate(h_normalized, 1 - h_normalized, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result_de = differential_evolution(
        objective_de, bounds, strategy='best1bin',
        popsize=20, tol=1e-6, mutation=(0.4, 0.8), recombination=0.9,
        maxiter=200, callback=None
    )
    elapsed = time.time() - start_time
    best_h = result_de.x

    if elapsed < budget_s:
        def objective_lbfgs(h_candidate):
            h = np.clip(h_candidate, 0, 1)
            h_sum = np.sum(h)
            if h_sum == 0:
                return np.inf
            h_normalized = h * target_sum / h_sum
            corr = np.correlate(h_normalized, 1 - h_normalized, mode='full')
            return np.max(corr) * dx

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds_lbfgs = [(0.0, 1.0) for _ in range(n_points)]
        result_lbfgs = minimize(
            objective_lbfgs, best_h, method='L-BFGS-B',
            bounds=bounds_lbfgs, constraints=cons, tol=1e-8
        )
        best_h = result_lbfgs.x

    h_final = np.clip(best_h, 0, 1)
    h_sum = np.sum(h_final)
    c5_bound = np.inf if h_sum == 0 else np.max(np.correlate(
        h_final * target_sum / h_sum, 1 - h_final * target_sum / h_sum, mode='full'
    )) * dx

    return (best_h, c5_bound, n_points)
