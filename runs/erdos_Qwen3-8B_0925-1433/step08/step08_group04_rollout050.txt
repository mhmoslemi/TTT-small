import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if n_points == 0:
        h = np.array([])
    else:
        if initial_h_values is not None:
            h = initial_h_values.copy()
        else:
            x = np.linspace(0, 2, n_points)
            h = np.zeros(n_points)
            h[:n_points//2] = 1
            h = h * target_sum / np.sum(h)
            h = np.clip(h, 0, 1)
    
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h
    best_c5 = obj(h)
    iterations = 0

    while time.time() - start_time < budget_s:
        iterations += 1
        if iterations > 50:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag = np.argmax(corr)
        lag = max_lag
        perturbation = 0.05 * np.sin(2 * np.pi * lag * np.linspace(0, 2, n_points) / 2)
        new_h = best_h + perturbation
        new_h = np.clip(new_h, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        else:
            new_h = best_h

        result = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h, best_c5 = result.x, current_c5
        else:
            result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
            if result.success and result.fun < best_c5:
                best_h, best_c5 = result.x, result.fun

    return (best_h, best_c5, n_points)
