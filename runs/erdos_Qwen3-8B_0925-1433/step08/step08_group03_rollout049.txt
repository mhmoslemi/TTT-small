import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    variants = [h.copy()]
    for _ in range(4):
        noise = np.random.normal(0, 0.1, n_points)
        variant = h + noise
        variant = np.clip(variant, 0, 1)
        variant = variant * target_sum / np.sum(variant) if np.sum(variant) > 0 else h
        variants.append(variant)

    best_h = h
    best_c5 = obj(h)

    start_time = time.time()

    for variant in variants:
        try:
            res = minimize(obj, variant, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
            if res.success:
                current_h = res.x
                current_c5 = res.fun
                if current_c5 < best_c5:
                    best_h, best_c5 = current_h, current_c5
        except:
            pass

            pass
