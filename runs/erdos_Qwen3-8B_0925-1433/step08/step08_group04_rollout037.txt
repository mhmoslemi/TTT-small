import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values) if initial_h_values is not None else 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[::2] = 1
        h[0] = 0.5
        h = h * target_sum / np.sum(h)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h
    best_c5 = obj(h)

    np.random.seed(seed)
    variations = []
    for _ in range(3):
        noise = np.random.normal(0, 0.05, n_points)
        temp = h + noise
        temp = np.clip(temp, 0, 1)
        temp = temp * target_sum / np.sum(temp)
        variations.append(temp)

    for f in [0.3, 0.4, 0.5]:
        x = np.linspace(0, 2, n_points)
        sin_h = 0.5 + 0.5 * np.sin(2 * np.pi * f * x)
        sin_h = np.clip(sin_h, 0, 1)
        sin_h = sin_h * target_sum / np.sum(sin_h)
        variations.append(sin_h)

    for guess in variations:
        result = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    for _ in range(5):

        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        result = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success and result.fun < best_c5:
