import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    # Differential Evolution Phase
    pop_size = 20
    iterations = 200
    start_time = time.time()
    result_de = differential_evolution(
        obj, bounds, strategy='best1bin', popsize=pop_size, tol=1e-8,
        mutation=(0.8, 1), recombination=0.9, maxiter=iterations,
        disp=False, seed=seed
    )
    best_h = result_de.x
    best_c5 = result_de.fun
    elapsed = time.time() - start_time

    # Local Refinement with L-BFGS-B
    if elapsed < budget_s:
        result_lbgf = minimize(
            obj, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-8,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        if result_lbgf.fun < best_c5:
            best_h, best_c5 = result_lbgf.x, result_lbgf.fun

    # Final Refinement
    if elapsed < budget_s:
        result_final = minimize(

    final_c5 = obj(h_final)
