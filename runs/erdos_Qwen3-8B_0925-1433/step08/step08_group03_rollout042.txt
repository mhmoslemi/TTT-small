import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    pop_size = 30
    max_gen = 200
    local_refine = 10

    base = initial_h_values.copy()
    pop = []
    for _ in range(pop_size):
        h = np.random.permutation(base)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        pop.append(h)
    for _ in range(pop_size // 2):
        h = np.random.choice([0,1], size=n_points, p=[0.5,0.5])
        h = h * target_sum / np.sum(h)
        pop.append(h)
    pop = np.array(pop)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    best_h = pop.mean(axis=0)
    best_c5 = compute_c5(best_h)

    for _ in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([compute_c5(h) for h in pop])
        idx = np.argsort(fitness)
        top = pop[idx[:pop_size//2]]
        new_pop = []
        for i in range(0, len(top), 2):
            if i+1 < len(top):
                h1, h2 = top[i], top[i+1]
                new_pop.append(0.5 * h1 + 0.5 * h2)
        while len(new_pop) < pop_size:
            h = new_pop[-1] + np.random.normal(0, 0.1, n_points)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            new_pop.append(h)
        pop = np.array(new_pop)
        curr_best = pop[np.argmin(fitness)]
        if compute_c5(curr_best) < best_c5:
            best_c5 = compute_c5(curr_best)
            best_h = curr_best

    bounds = [(0,1)]*n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result = minimize(lambda x: compute_c5(x), best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    best_h = result.x
    best_c5 = result.fun

    return (best_h, best_c5, n_points)
