import numpy as np
import time

def run(seed=42, budget_s=600):
    import numpy as np
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_rescale(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return np.zeros_like(h)
        scale = target_sum / np.sum(h)
        return h * scale

    h_init = clip_rescale(initial_h_values)
    swarm = [np.copy(h_init) + np.random.normal(0, 0.1, n_points) for _ in range(20)]
    swarm = [clip_rescale(s) for s in swarm]

    best_h = np.copy(h_init)
    best_c5 = np.max(np.correlate(h_init, 1 - h_init, mode='full')) * dx

    pos = [np.copy(s) for s in swarm]
    vel = [np.zeros(n_points) for _ in range(20)]
    pbest = [np.copy(s) for s in swarm]
    pbest_c5 = [np.max(np.correlate(s, 1 - s, mode='full')) * dx for s in swarm]
    gbest = np.copy(pbest[np.argmin(pbest_c5)])
    gbest_c5 = np.min(pbest_c5)

    w = 0.8
    c1 = 1.5
    c2 = 1.5
    max_vel = 0.2 * dx
    max_iter = 500
    early_stop = 50
    iter_count = 0
    no_improve = 0
    start_time = time.time()

    while iter_count < max_iter and time.time() - start_time < budget_s:
        iter_count += 1
        for i in range(20):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            vel[i] = w * vel[i] + c1 * r1 * (pbest[i] - pos[i]) + c2 * r2 * (gbest - pos[i])
            vel[i] = np.clip(vel[i], -max_vel, max_vel)
            pos[i] = pos[i] + vel[i]
            pos[i] = clip_rescale(pos[i])
            curr_c5 = np.max(np.correlate(pos[i], 1 - pos[i], mode='full')) * dx
            if curr_c5 < pbest_c5[i]:
                pbest[i] = np.copy(pos[i])
                pbest_c5[i] = curr_c5
        curr_gbest_c5 = np.min(pbest_c5)
        if curr_gbest_c5 < gbest_c5:
            gbest = np.copy(pbest[np.argmin(pbest_c5)])
            gbest_c5 = curr_gbest_c5
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= early_stop:
            break
        if gbest_c5 < best_c5:
            best_h = np.copy(gbest)
            best_c5 = gbest_c5

    return (best_h, best_c5, n_points)
