import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def cons(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0)] * n_points
    cons_dict = {'type': 'eq', 'fun': cons}

    h_filtered = np.fft.ifft(np.fft.fft(h) * np.heaviside(np.fft.fftfreq(n_points, d=dx) * n_points, 0.5)).real
    h_filtered = np.clip(h_filtered, 0, 1)
    h_filtered = h_filtered * target_sum / np.sum(h_filtered) if np.sum(h_filtered) > 0 else h_filtered

    best_h = h_filtered
    best_c5 = obj(best_h)

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons_dict, tol=1e-8)
        if result.success:
            best_h = result.x
            best_c5 = result.fun
        else:
            break

        # Perturb with sinusoidal modulation
        freq = 2 * np.pi * np.argmax(np.correlate(best_h, 1 - best_h, mode='full')) / n_points

        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        result_perturb = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons_dict, tol=1e-8)
        if result_perturb.success and result_perturb.fun < best_c5:
            best_h = result_perturb.x
