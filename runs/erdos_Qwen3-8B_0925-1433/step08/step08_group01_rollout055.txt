        res = minimize(lambda x: 0, current_h, method='SLSQP',
                       bounds=[(0, 1)] * n_points,
                       constraints=constraints)
        return res.success, res.x if res.success else None

    low, high = 0.3, 0.4
    best_h = h.copy()
    best_c5 = high
    start_time = time.time()

    while high - low > 1e-7 and time.time() - start_time < budget_s:
        mid = (low + high) / 2
        success, new_h = feasibility_check(mid, h)
        if success:
            h = new_h
            high = mid
            best_h = h.copy()
            best_c5 = mid
        else:
            low = mid

    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_actual = np.max(corr) * dx
    return (best_h, c5_actual, n_points)
