            feasible = False
        
        if feasible:
            high = mid
        else:
            low = mid
    
    # Final evaluation
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    c5_final = np.max(corr_final) * dx
    return (best_h, c5_final, n_points)
