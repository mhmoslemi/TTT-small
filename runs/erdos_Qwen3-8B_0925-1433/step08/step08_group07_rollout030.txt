def run(seed=42, budget_s=600):
    import numpy as np
    from scipy.optimize import differential_evolution
    import time

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        objective,
        bounds,

        maxiter=100,
