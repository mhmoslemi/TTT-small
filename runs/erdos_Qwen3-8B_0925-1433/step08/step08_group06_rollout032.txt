import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def jac(x, k):
        eps = 1e-5
        grad = np.zeros_like(x)
        for i in range(n_points):
            x_plus = np.copy(x)
            x_plus[i] += eps
            corr_plus = np.correlate(x_plus, 1 - x_plus, mode='full')
            x_minus = np.copy(x)
            x_minus[i] -= eps
            corr_minus = np.correlate(x_minus, 1 - x_minus, mode='full')
            grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        return grad

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    best_h = h
    best_c5 = obj(h)
    start_time = time.time()

    for _ in range(1200):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=lambda x: jac(x, max_k), tol=1e-8)
        if res.success:
            best_h = res.x
            best_c5 = res.fun
        else:
            break

        perturb = np.random.normal(0, 0.008, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res_perturb = minimize(obj, new_h, method='SLSQP', bounds=bounds,
                               constraints=cons, jac=lambda x: jac(x, np.argmax(np.correlate(new_h, 1 - new_h, mode='full'))), tol=1e-8)
        if res_perturb.success and res_perturb.fun < best_c5:
            best_h = res_perturb.x
            best_c5 = res_perturb.fun

    return (best_h, best_c5, n_points)
