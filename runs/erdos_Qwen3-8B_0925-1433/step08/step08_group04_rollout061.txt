import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values) if initial_h_values is not None else 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        x = np.linspace(0, 2, n_points)
        f = 0.25
        h = 0.5 + 0.5 * np.sin(2 * np.pi * f * x)
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        else:
            h = np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    initial_guesses = [h]
    np.random.seed(seed)
    for _ in range(4):
        if initial_h_values is not None:
            perm = np.random.permutation(h)
        else:
            perm = np.random.rand(n_points)
        perm = np.clip(perm, 0, 1)
        if np.sum(perm) > 0:
            perm = perm * target_sum / np.sum(perm)
        initial_guesses.append(perm)

    best_h = h
    best_c5 = float('inf')
    start_time = time.time()

    for guess in initial_guesses:
        result = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h, best_c5 = result.x, current_c5

    for _ in range(4):

        max_lag = np.argmax(corr)
        perturbation = 0.03 * np.sin(2 * np.pi * max_lag * np.linspace(0, 2, n_points) / 2)
        new_h = best_h + perturbation

        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)

        if result.success:
