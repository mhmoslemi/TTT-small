    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h with the initial values
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def jac(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr_idx = np.argmax(corr)
        shift = max_corr_idx - (n_points - 1)
        i_plus = (np.arange(n_points) + shift) % n_points
        i_minus = (np.arange(n_points) - shift) % n_points
        grad = (1 - h_vec[i_plus]) - h_vec[i_minus]
        return grad * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Initial optimization with SLSQP
    result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, jac=jac, tol=1e-6)
    best_h, best_c5 = result.x, result.fun

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Add structured perturbations
        if n_points < 10:
            continue
        seg_len = int(n_points * 0.3)
        seg1 = np.random.choice(n_points - seg_len, 1)[0]
        seg2 = np.random.choice(n_points - seg_len, 1)[0]
        new_h = best_h.copy()
        new_h[seg1:seg1+seg_len], new_h[seg2:seg2+seg_len] = new_h[seg2:seg2+seg_len], new_h[seg1:seg1+seg_len]
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, jac=jac, tol=1e-6)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
