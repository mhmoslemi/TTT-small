import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    if 'initial_h_values' in locals():
        h = initial_h_values
    else:
        n_points = 200
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = h * (n_points/2) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def optimize(start):
        res = minimize(obj, start, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and np.isfinite(res.fun):
            return res.x, res.fun
        return start, np.inf

    def perturb(h, alpha=0.05):
        shift = np.random.uniform(0, 1, n_points)
        return np.clip(h * (1 + alpha * np.sin(2 * np.pi * shift)), 0, 1)

    h_candidates = [
        h,
        np.concatenate([h, h])[::2],
        np.concatenate([h, 1 - h])[::2],
        np.roll(h, n_points//4),
        np.roll(h, n_points//2)
    ]
    h_candidates = [np.clip(c, 0, 1) * target_sum / np.sum(np.clip(c, 0, 1)) for c in h_candidates]

    best_h, best_c5 = h_candidates[0], np.inf
    for h0 in h_candidates:
        if np.isfinite(np.sum(h0)) and np.isfinite(obj(h0)):
            h0 = np.clip(h0, 0, 1)
            h0 = h0 * target_sum / np.sum(h0)
            curr_h, curr_c5 = optimize(h0)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = curr_h

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        h = perturb(best_h)
        h = h * target_sum / np.sum(h)
        curr_h, curr_c5 = optimize(h)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = curr_h

    return (best_h, best_c5, n_points)
