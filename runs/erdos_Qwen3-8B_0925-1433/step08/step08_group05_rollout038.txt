import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    # Compute autocorrelations for all lags
    def compute_autocorrs(h):
        n = len(h)
        autocorrs = np.zeros(n)
        for k in range(n):
            ac = 0.0
            for i in range(n - k):
                ac += h[i] * h[i + k]
            autocorrs[k] = ac * dx
        return autocorrs

    # Compute gradient for a specific lag
    def compute_gradient(h, k):
        n = len(h)
        grad = np.zeros(n)
        for m in range(n):
            if m + k < n:
                grad[m] += h[m + k] * dx
            if m - k >= 0:
                grad[m] += h[m - k] * dx
        return grad

    best_h = h.copy()
    best_c5 = 0.0
    start_time = time.time()

    # Gradient ascent steps
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        autocorrs = compute_autocorrs(best_h)
        k_min = np.argmin(autocorrs)
        if k_min == 0:
            grad = 2.0 * dx * best_h
        else:
            grad = compute_gradient(best_h, k_min)
        # Update h in the direction of the gradient
        learning_rate = 0.05
        h_update = best_h + learning_rate * grad
        # Clip and normalize
        h_clipped = np.clip(h_update, 0, 1)
        h_norm = h_clipped * target_sum / np.sum(h_clipped)
        best_h = h_norm
        # Evaluate C5
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5

    # Local optimization with SLSQP
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-7)

    # Final evaluation
    corr = np.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(corr) * dx

    return (final_h, final_c5, n_points)
