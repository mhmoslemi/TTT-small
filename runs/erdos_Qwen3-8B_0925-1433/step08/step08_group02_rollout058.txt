import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    iterations = 0

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        max_lag = np.argmax(corr)

        def obj(x):
            return corr[max_lag] * dx

        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)

        if result.success:
            h = result.x
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            if total > 0:
                h = h * target_sum / total
            else:
                h = np.zeros(n_points)

            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

        if iterations % 100 == 0:
            perturb = np.sin(np.linspace(0, 2 * np.pi, n_points) * max_lag / 5) * 0.05
            new_h = np.clip(h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h
            result = minimize(lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                total = np.sum(h)
                if total > 0:
                    h = h * target_sum / total
                else:
                    h = np.zeros(n_points)
                current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5

        iterations += 1
    return (best_h, best_c5, n_points)
