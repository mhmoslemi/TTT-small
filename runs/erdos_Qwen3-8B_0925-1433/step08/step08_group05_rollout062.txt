import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points) + target_sum / n_points

    def compute_min_lag_grad(h):
        autocorr = np.correlate(h, h, mode='full') * dx
        possible_lags = autocorr[n_points:2*n_points - 1]
        min_idx = np.argmin(possible_lags)
        lag = min_idx + 1
        grad = np.zeros(n_points)
        for m in range(n_points):
            if m + lag < n_points:
                grad[m] += h[m + lag]
            if m - lag >= 0:
                grad[m] += h[m - lag]
        grad *= dx
        return lag, grad

    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        lag, grad = compute_min_lag_grad(h)
        h += 0.05 * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)

    def objective(x):
        autocorr = np.correlate(x, x, mode='full') * dx
        possible_lags = autocorr[n_points:2*n_points - 1]
        return -np.min(possible_lags)

    bounds = [(0, 1)] * n_points
