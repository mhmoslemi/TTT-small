from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        n_points = 45
        x = np.linspace(0, 2, n_points)

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    initial_guesses = [h.copy(), 
                       np.random.permutation(h), 
                       h + 0.1 * np.random.randn(n_points), 
                       h - 0.1 * np.random.randn(n_points),
                       0.5 * np.ones(n_points)]
    initial_guesses = [np.clip(g, 0, 1) * target_sum / np.sum(np.clip(g, 0, 1)) if np.sum(np.clip(g, 0, 1)) > 0 else h for g in initial_guesses]
    
    best_h = h
    best_c5 = np.inf
    start_time = time.time()
    
    for guess in initial_guesses:
        result = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            if result.fun < best_c5:
                best_h, best_c5 = result.x, result.fun
    
    for _ in range(10):
