import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize with structured patterns
    def generate_pattern(base, freq, phase):
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.cos(2 * np.pi * freq * (x - phase)))
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    h_candidates = [initial_h_values.copy()]
    h_candidates.append(generate_pattern(initial_h_values, freq=0.3, phase=0.5))
    h_candidates.append(generate_pattern(initial_h_values, freq=0.2, phase=0.25))

    best_h, best_c5 = None, float('inf')

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    for h in h_candidates:
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
        result = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if result.success:
            curr_c5 = result.fun
            if curr_c5 < best_c5:
                best_h, best_c5 = result.x, curr_c5

    # Perturbation refinement
    if best_h is None:
        best_h = np.zeros(n_points)
        best_c5 = float('inf')

    for _ in range(100):
        if time.time() - time.time() > budget_s:
            break
        # Sinusoidal perturbation
        perturb = 0.05 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        result = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 50})
        if result.success and result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun

    return (best_h, best_c5, n_points)
