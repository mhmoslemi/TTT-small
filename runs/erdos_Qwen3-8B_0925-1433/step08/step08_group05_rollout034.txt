import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def perturb(x, amp, freq):
        pert = amp * np.sin(freq * np.linspace(0, 2 * np.pi, n_points))
        return np.clip(x + pert, 0, 1) * target_sum / np.sum(np.clip(x + pert, 0, 1))

    def generate_base_pattern():
        spaced = np.zeros(n_points)
        idx = np.argsort(np.random.rand(n_points))
        spaced[idx[:int(n_points/2)]] = 1.0
        spaced = spaced * target_sum / np.sum(spaced)
        return spaced

    base_patterns = [generate_base_pattern() for _ in range(5)]
    candidates = [h] + base_patterns

    start_time = time.time()
    best_h, best_c5 = h, objective(h)

    for h_candidate in candidates:
        res = minimize(objective, h_candidate, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        freqs = np.random.choice([1, 2, 3, 4, 5], size=3)
        for freq in freqs:
            new_h = perturb(best_h, 0.05, freq)
            res = minimize(objective, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
