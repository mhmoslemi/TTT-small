import numpy as np
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        k_max = np.argmax(corr)
        lag_offset = n_points - 1
        k_lag = k_max - lag_offset
        shifted_ones_minus_h = np.roll(1 - h, k_lag)
        gradient = np.zeros_like(h)
        gradient[k_lag:] = shifted_ones_minus_h[k_lag:]
        step_size = 0.1
        h = h - step_size * gradient
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        wave = np.sin(2 * np.pi * np.arange(n_points) / n_points)
        perturb = 0.05 * wave * (1 - np.abs(h - np.roll(h, k_lag)))
        h = np.clip(h + perturb, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        current_corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(current_corr)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    return (best_h, best_c5, n_points)
