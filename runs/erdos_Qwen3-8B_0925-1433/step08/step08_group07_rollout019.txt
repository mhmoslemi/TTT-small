        total_grad = grad_max + lambda_reg * grad_reg
        h_new = h - learning_rate * total_grad
        h_new = np.clip(h_new, 0, 1)
        sum_h_new = np.sum(h_new)
        if sum_h_new > 0:
            h_new = h_new * target_sum / sum_h_new
        else:
            h_new = np.zeros(n_points)
        h_new = np.clip(h_new, 0, 1)

        new_c5 = compute_max_overlap(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            no_improvement_counter = 0
        else:
            no_improvement_counter += 1
        if no_improvement_counter >= 50:
            lambda_reg *= 1.1
            no_improvement_counter = 0
        if _ > 100:
            learning_rate *= 0.95

    return (best_h, best_c5, n_points)
