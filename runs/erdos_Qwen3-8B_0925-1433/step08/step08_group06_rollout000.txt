from scipy.optimize import minimize
import time

    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def grad(x, k):
        eps = 1e-4 * np.max(x) + 1e-6
        grad = np.zeros_like(x)

            x_plus[i] += eps
            corr_plus = np.correlate(x_plus, 1 - x_plus, mode='full')
            grad[i] = (np.max(corr_plus) - obj(x)) / eps
        return grad

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    best_h = h

        if max_k >= n_points - 1:

        try:
            res = minimize(obj, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
