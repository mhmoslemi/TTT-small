import numpy as np

def run(seed=42, budget_s=600):

    from scipy.optimize import minimize
    import time

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h from initial_h_values
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(best_h)

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        # Generate perturbed candidate
        if np.random.rand() < 0.5:
            # Random restart
            new_h = np.random.rand(n_points)
            new_h = new_h * target_sum / np.sum(new_h)
            new_h = np.clip(new_h, 0, 1)
        else:
            # Local perturbation
            perturb = np.random.normal(0, 0.05, n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
