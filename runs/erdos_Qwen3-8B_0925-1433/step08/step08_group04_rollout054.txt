from scipy.optimize import minimize, SLSQP
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if n_points == 0:
        h = np.array([])
    else:
        if initial_h_values is not None:
            h = initial_h_values.copy()
        else:
            h = np.zeros(n_points)
            for i in range(n_points):
                if i % 3 == 0:
                    h[i] = 1.0
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        else:
            h = np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(best_h)
    num_iterations = 0
    max_iterations = 100

    while num_iterations < max_iterations and time.time() - start_time < budget_s:
        num_iterations += 1
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag = np.argmin(corr)  # Index of minimum correlation (considering full length)
        max_lag = np.clip(max_lag, 0, len(corr) - 1)
        k = max_lag - n_points + 1  # Adjust lag for 1D convolution

        # Construct perturbation based on lag
        perturb = 0.05 * np.sin(2 * np.pi * k * np.linspace(0, 2, n_points) / 2)
        new_h = best_h + perturb

        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        else:
            new_h = best_h.copy()

        # Optimization using SLSQP
        result = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success and result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
            print(f"Iter {num_iterations}, C5: {best_c5:.6f}")

    # Final evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
