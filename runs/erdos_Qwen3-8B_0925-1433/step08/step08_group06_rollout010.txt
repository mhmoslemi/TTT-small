import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Rescale and clip initial guess
    h = initial_h_values.copy()

    h = np.clip(h, 0, 1)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def jac_obj(x):
        eps = 1e-6
        grad = np.zeros_like(x)
        for i in range(n_points):
            x_plus = x.copy()
            x_plus[i] += eps
            corr_plus = np.correlate(x_plus, 1 - x_plus, mode='full')
            grad[i] = (np.max(corr_plus) - obj(x)) / eps
        return grad

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0.0, 1.0)] * n_points

    # Initial frequency domain design
    fft_h = fft(h)
    cutoff = int(0.1 * n_points)

    best_h = h_filtered
    best_c5 = obj(best_h)
    start_time = time.time()

    for _ in range(150):

            break
        # L-BFGS-B optimization
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, jac=jac_obj, tol=1e-7)
