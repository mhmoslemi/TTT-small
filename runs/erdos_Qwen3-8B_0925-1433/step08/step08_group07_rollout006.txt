import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel, WhiteKernel

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h from initial_h_values
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    X = np.array([np.arange(n_points)]).T
    y = np.array([best_c5])
    kernel = ConstantKernel(1.0, (1e-2, 1e2)) * RBF(1.0, (1e-2, 1e2)) + WhiteKernel(1e-3)
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-5, n_restarts_optimizer=10)

    def eval_c5(h_vec):
        return np.max(np.correlate(h_vec, 1 - h_vec, mode='full')) * dx

    def expected_improvement(x):
        mean, std = gp.predict(x, return_std=True)
        if std.shape[0] == 0:
            return np.inf
        if std < 1e-3:
            return np.inf
        improvement = mean - best_c5
        prob = np.exp(-((improvement - 1e-3) ** 2) / (2 * std ** 2))
        return improvement * prob

    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        if len(X) < 2:
            new_x = np.random.uniform(0, 1, n_points)
            new_x = np.clip(new_x, 0, 1)
            new_x = new_x * target_sum / np.sum(new_x)
            X = np.vstack((X, new_x))
            y = np.append(y, eval_c5(new_x))
            continue
        acq = np.array([expected_improvement(np.array([x])) for x in X])
        idx = np.argmax(acq)
        best_x = X[idx]
        new_x = np.random.normal(best_x, 0.05, n_points)
        new_x = np.clip(new_x, 0, 1)
        new_x = new_x * target_sum / np.sum(new_x)
        new_c5 = eval_c5(new_x)

        y = np.append(y, new_c5)
        gp.fit(X, y)
        if new_c5 < best_c5:
            best_h, best_c5 = new_x.copy(), new_c5

    return (best_h, best_c5, n_points)
