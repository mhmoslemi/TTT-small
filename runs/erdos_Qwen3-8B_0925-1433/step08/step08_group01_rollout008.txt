    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
    best_h, best_c5 = result.x, result.fun

    start_time = time.time()
    while time.time() - start_time < budget_s:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        if k < 0:
            k = -k
            start = (n_points - 1) - k
            end = n_points - 1
        else:
            start = 0
            end = n_points - 1 - k
        seg_len = max(1, int(n_points * 0.15))
        seg_start = np.random.randint(start, max(end - seg_len + 1, 0))
        seg2 = np.random.randint(0, n_points - seg_len)
        new_h = best_h.copy()
        new_h[seg_start:seg_start + seg_len], new_h[seg2:seg2 + seg_len] = new_h[seg2:seg2 + seg_len], new_h[seg_start:seg_start + seg_len]
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        result = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun

    return (best_h, best_c5, n_points)
