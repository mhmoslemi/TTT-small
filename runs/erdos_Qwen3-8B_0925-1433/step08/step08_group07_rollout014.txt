import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

        h = h * target_sum / np.sum(h)

    if np.sum(h) == 0:
        h = np.zeros(n_points)
    else:
        h = h * target_sum / np.sum(h)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(best_h)

    for _ in range(350):
        if time.time() - start_time > budget_s:
            break
        # Generate perturbed candidate
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
