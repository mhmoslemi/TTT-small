import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):

        return np.max(corr) * dx

    def compute_gradient(h_vals, epsilon=1e-6):
        grad = np.zeros_like(h_vals)
        for i in range(n_points):
            h_plus = h_vals.copy()

            h_minus = h_vals.copy()
