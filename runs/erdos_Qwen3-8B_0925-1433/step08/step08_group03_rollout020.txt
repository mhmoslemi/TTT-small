import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Symmetric parent check and mirror
    if n_points % 2 == 0:
        mir = np.flip(initial_h_values)
        if np.allclose(initial_h_values[:n_points//2], 1 - mir[:n_points//2]):
            sym = initial_h_values
        else:
            sym = np.concatenate((initial_h_values[:n_points//2], 1 - initial_h_values[:n_points//2][::-1]))
    else:
        sym = np.concatenate((initial_h_values[:n_points//2], [0.5], 1 - initial_h_values[:n_points//2][::-1]))
    h = np.copy(sym)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def clip_rescale(x):
        if np.sum(x) <= 0:
            return np.zeros_like(x)
        return np.clip(x, 0, 1) * target_sum / np.sum(x)

    bounds = [(0, 1)] * (n_points//2 + 1)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum / 2}]

    best_h = np.copy(h[:n_points//2 + 1])
    best_c5 = obj(h)

    def perturb(x, strength=0.05):
        return np.clip(x + np.random.normal(0, strength, len(x)), 0, 1)

    for _ in range(150):
        if time.time() - time.time() > budget_s:

    h_final = np.concatenate((best_h, 1 - best_h[:-1][::-1]))
