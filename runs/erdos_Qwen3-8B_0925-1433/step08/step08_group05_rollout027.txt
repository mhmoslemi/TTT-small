import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h_values = np.loadtxt('initial_h_values.txt') if 'initial_h_values' in locals() else np.zeros(45)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def clip_rescale(h):
        h = np.clip(h, 0, 1)
        scale = target_sum / np.sum(h) if np.sum(h) > 0 else 1.0
        return h * scale

    h = clip_rescale(initial_h_values)

    structured = [h.copy()]

    # Generate structured patterns
    even = np.zeros(n_points)
    even[::2] = 1.0
    even = clip_rescale(even)

    alt[::2] = 1.0
    alt[1::2] = 0.5
    alt = clip_rescale(alt)
    structured.append(alt)

    rand = np.random.choice([0, 1], size=n_points)
    rand = clip_rescale(rand)
    structured.append(rand)

    best_h = min(structured, key=evaluate)
    best_c5 = evaluate(best_h)

    def perturb(h, amp):
        noise = np.random.normal(0, amp, n_points)

    start = time.time()
