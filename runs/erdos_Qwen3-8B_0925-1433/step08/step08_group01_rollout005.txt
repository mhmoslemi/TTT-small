    bounds = [(0.0, 1.0)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, jac=jac, tol=1e-6, options={'maxiter': 1})
        if res.success:
            best_h = res.x
            best_c5 = res.fun
        # Perturb for exploration
        if np.random.rand() < 0.3:
            seg_len = int(n_points * 0.2)
            seg1 = np.random.randint(0, n_points - seg_len)
            seg2 = np.random.randint(0, n_points - seg_len)
            new_h = best_h.copy()
            new_h[seg1:seg1+seg_len], new_h[seg2:seg2+seg_len] = new_h[seg2:seg2+seg_len], new_h[seg1:seg1+seg_len]
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, jac=jac, tol=1e-6, options={'maxiter': 1})
            if res.success and res.fun < best_c5:
                best_h = res.x
                best_c5 = res.fun

    return (best_h, best_c5, n_points)
