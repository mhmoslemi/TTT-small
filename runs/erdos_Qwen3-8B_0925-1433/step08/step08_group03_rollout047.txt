import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Generate initial candidates
    candidates = []
    h = np.array(initial_h_values)
    candidates.append(h)
    candidates.append(np.concatenate([h[:n_points//2+1], 1 - h[:n_points//2][::-1]]))
    freq = 0.1
    modulated = 0.5 * (1 + np.cos(2 * np.pi * (np.arange(n_points) / n_points) * freq))
    modulated = np.clip(modulated, 0, 1) * target_sum / np.sum(modulated)
    candidates.append(modulated)

    best_h = np.copy(initial_h_values)
    best_c5 = float('inf')
    start_time = time.time()

    for candidate in candidates:
        candidate = np.clip(candidate, 0, 1)
        if np.sum(candidate) > 0:
            candidate = (candidate * target_sum) / np.sum(candidate)
        else:
            candidate = np.zeros(n_points)
        res = minimize(obj, candidate, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Apply structured perturbations
    for _ in range(10):

        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
