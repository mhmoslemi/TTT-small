import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    import numpy as np
    from scipy.optimize import minimize
    import time

    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1).copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(best_h)

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        # Generate multiple perturbed candidates
        perturb = np.random.normal(0, 0.1, n_points)
        new_h1 = np.clip(best_h + perturb, 0, 1)
        new_h1 = new_h1 * target_sum / np.sum(new_h1) if np.sum(new_h1) > 0 else best_h
        new_h2 = np.clip(np.roll(best_h, 1) + perturb, 0, 1)
        new_h2 = new_h2 * target_sum / np.sum(new_h2) if np.sum(new_h2) > 0 else best_h
        new_h3 = np.clip(np.roll(best_h, -1) + perturb, 0, 1)
        new_h3 = new_h3 * target_sum / np.sum(new_h3) if np.sum(new_h3) > 0 else best_h
        
        candidates = [new_h1, new_h2, new_h3]
        for candidate in candidates:
            res = minimize(obj, candidate, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
            if res.success and res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
