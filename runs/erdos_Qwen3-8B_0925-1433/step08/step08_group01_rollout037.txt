        for i in range(n_points):
            j_plus = i + shift
            j_minus = i - shift
            term1 = 0.0
            if 0 <= j_plus < n_points:
                term1 = 1 - x[j_plus]
            term2 = 0.0
            if 0 <= j_minus < n_points:
                term2 = x[j_minus]
            grad[i] = (term1 - term2) * dx
        return grad

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    best_h = h.copy()
    best_c5 = objective(h)
    start_time = time.time()
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, jac=gradient, tol=1e-6)
    if result.success and result.fun < best_c5:
        best_h = result.x
        best_c5 = result.fun
    return (best_h, best_c5, n_points)
