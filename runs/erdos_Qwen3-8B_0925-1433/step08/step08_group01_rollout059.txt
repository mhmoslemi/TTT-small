    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        lag = max_corr_idx - (n_points - 1)
        a = max(0.0, -lag)
        b = min(2.0, 2.0 - lag)
        i_start = max(0, int(a / dx))
        i_end = min(n_points, int(b / dx))
        
        seg_len = int(n_points * 0.3)
        if seg_len <= 0:
            seg_len = 1
        if n_points < seg_len:
            continue
        
        while True:
            start_idx = np.random.randint(0, n_points - seg_len)
            if not (start_idx + seg_len <= i_start or start_idx >= i_end):
                continue
            break
        
        new_h = h.copy()
        new_h[i_start:i_end], new_h[start_idx:start_idx + seg_len] = new_h[start_idx:start_idx + seg_len], new_h[i_start:i_end]
        new_h = np.clip(new_h, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        else:
            new_h = h.copy()
        
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        try:
            res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
            h_opt = res.x
        except:
            h_opt = new_h
        
        corr_opt = np.correlate(h_opt, 1 - h_opt, mode='full')
        current_c5 = np.max(corr_opt) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
        h = h_opt
    
    return (best_h, best_c5, n_points)
