import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h from initial_h_values
    h = np.clip(initial_h_values, 0, 1)
    h_sum = np.sum(h)
    h = h * target_sum / h_sum if h_sum > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)

    def de_objective(x):
        h_candidate = np.clip(x, 0, 1)
        h_candidate = h_candidate * target_sum / np.sum(h_candidate) if np.sum(h_candidate) > 0 else np.zeros_like(h_candidate)
        corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    result_de = differential_evolution(de_objective, bounds, strategy='best1bin', popsize=20, mutation=(0.5, 0.9), recombination=0.8, tol=1e-6, maxiter=200, seed=seed)
    best_h_de = result_de.x
    best_c5_de = result_de.fun

    elapsed = time.time() - time.time()
    if elapsed < budget_s:
        def slsqp_objective(x):
            h_candidate = np.clip(x, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate) if np.sum(h_candidate) > 0 else np.zeros_like(h_candidate)
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            return np.max(corr) * dx

        def slsqp_cons(x):
            return np.sum(x) - target_sum

        cons = [{'type': 'eq', 'fun': slsqp_cons}]
        result_slsqp = minimize(slsqp_objective, best_h_de, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        best_h = result_slsqp.x
        best_c5 = result_slsqp.fun

    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * target_sum / np.sum(h_final) if np.sum(h_final) > 0 else np.zeros(n_points)
    h_final = np.clip(h_final, 0, 1)

    final_c5 = np.max(corr) * dx
