import numpy as np
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def compute_corr(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return corr

    best_h = h.copy()
    best_c5 = np.max(compute_corr(h))

    start_time = time.time()
    step_size = 0.1
    steps = 0
    last_improvement = 0

    while time.time() - start_time < budget_s:
        steps += 1
        corr = compute_corr(h)
        m = np.argmax(corr)
        k = m - (n_points - 1)

        if k >= 0:
            grad = np.zeros(n_points)
            grad[k:] = 1 - h[0 : n_points - k]
        else:
            m = -k
            grad = np.zeros(n_points)
            grad[0 : n_points - m] = 1 - h[m : n_points]

        direction = -grad
        h_new = h + direction * step_size
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * target_sum / current_sum
        else:
            h_new = np.full(n_points, target_sum / n_points)

        new_c5 = np.max(compute_corr(h_new))
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement = steps

        if steps - last_improvement > 200:
            step_size *= 0.9
        else:
            step_size *= 0.995

    return (best_h, best_c5, n_points)
