def run(seed=42, budget_s=600):

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = objective(h)
    start_time = time.time()

    for _ in range(30):
        if time.time() - start_time > budget_s:
            break
        freq = np.random.choice([1, 2, 3, 4, 5])
        amp = np.random.uniform(0.05, 0.2)
        pert = amp * np.sin(2 * np.pi * freq * np.linspace(0, 2, n_points))
        new_h = h + pert
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(objective, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
            h = best_h.copy()
    
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        noise = np.random.normal(0, 0.03, n_points)
        new_h = best_h + noise
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(objective, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7)
