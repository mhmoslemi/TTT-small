import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def perturb(x, freqs, amps):
        x_pert = x.copy()
        for f, a in zip(freqs, amps):
            x_pert += a * np.sin(2 * np.pi * f * np.linspace(0, 2, n_points))
        x_pert = np.clip(x_pert, 0, 1)
        x_pert = x_pert * target_sum / np.sum(x_pert)
        return x_pert

    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(h)

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        freqs = np.random.choice([1, 2, 3, 4, 5], size=3)
        amps = np.random.uniform(0.05, 0.3, size=3)
        new_h = perturb(h, freqs, amps)
        res = minimize(objective, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(30):

        new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(objective, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
