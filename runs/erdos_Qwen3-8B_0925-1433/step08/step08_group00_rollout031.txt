import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    initial_guess = initial_h_values.copy()
    initial_guess = np.clip(initial_guess, 0, 1)
    if np.sum(initial_guess) > 0:
        initial_guess = initial_guess * target_sum / np.sum(initial_guess)
    else:
        initial_guess = np.zeros(n_points)

    def objective(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * target_sum / sum_h
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    # De phase with seed and maxiter
    result_de = differential_evolution(
        objective, bounds, popsize=20, mutation=(0.5, 1.0), recombination=0.9,
        tol=1e-8, polish=True, workers=-1, maxiter=300, seed=seed
    )
    best_h = result_de.x
    best_c5 = result_de.fun

    # L-BFGS-B refinement
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds_lbgf = [(0, 1)] * n_points
    result_lbgf = minimize(
        objective, best_h, method='L-BFGS-B', bounds=bounds_lbgf,
        constraints=cons, tol=1e-8, options={'maxiter': 100}
    )
    best_h = result_lbgf.x
    best_c5 = result_lbgf.fun

    return (best_h, best_c5, n_points)
