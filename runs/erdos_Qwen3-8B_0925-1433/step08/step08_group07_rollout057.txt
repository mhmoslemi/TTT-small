import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h from initial_h_values
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(best_h)

    # Differential Evolution Phase
    def de_fitness(x):
        nonlocal best_c5
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x) if np.sum(x) > 0 else best_h
        curr_c5 = obj(x)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = x.copy()
        return curr_c5

    result = differential_evolution(
        de_fitness,
        bounds,
        constraints=cons,
        popsize=20,
        maxiter=100,
        tol=1e-5,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )

    # Local Refinement Phase
    if time.time() - start_time < budget_s:
        res = minimize(
            obj,
            result.x,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            tol=1e-6,
            options={'maxiter': 100}
        )
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun

    return (np.clip(best_h, 0, 1), best_c5, n_points)
