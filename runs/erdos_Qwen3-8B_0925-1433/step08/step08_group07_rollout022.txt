from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1).copy()
    h_sum = np.sum(h)
    h = h * target_sum / h_sum if h_sum > 0 else np.zeros(n_points)

    def obj(x):
        h_clipped = np.clip(x, 0, 1)
        h_sum_clip = np.sum(h_clipped)
        if h_sum_clip == 0:
            return np.inf
        h_scaled = h_clipped * target_sum / h_sum_clip
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(best_h)
    evals = 0

    for _ in range(200):
        if time.time() - start_time > budget_s:

        # Generate perturbed candidates
