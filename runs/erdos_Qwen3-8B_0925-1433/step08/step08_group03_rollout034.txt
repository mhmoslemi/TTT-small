import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h_values.copy()
    h_sum = np.sum(h)
    if h_sum > 1e-8:
        h = h * target_sum / h_sum
    else:
        h = np.zeros(n_points)
    
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    pop_size = 30
    n_generations = 200
    mutation_scale = 0.1
    
    population = [h * np.random.rand(n_points) for _ in range(pop_size)]
    for i in range(pop_size):
        population[i] = np.clip(population[i], 0, 1)
        population[i] = population[i] * target_sum / np.sum(population[i])
    
    best_c5 = float('inf')
    start_time = time.time()
    
    for gen in range(n_generations):
        if time.time() - start_time > budget_s:
            break
        
        fitness = [obj(ind) for ind in population]
        top_idx = np.argsort(fitness)[:int(pop_size*0.2)]
        parents = [population[i] for i in top_idx]

        for i in range(0, len(parents), 2):
            p1 = parents[i]
            p2 = parents[i+1] if i+1 < len(parents) else p1
            child = 0.5 * (p1 + p2)
            for j in range(n_points):
                if np.random.rand() < 0.3:
                    child[j] += np.random.normal(0, mutation_scale)
            child = np.clip(child, 0, 1)
            child = child * target_sum / np.sum(child)
            new_pop.append(child)
        
        if np.random.rand() < 0.2:
            structured = np.random.choice([0,1], size=n_points, p=[0.5,0.5])
            structured = structured * target_sum / np.sum(structured)
            new_pop.append(structured)
        
        population = new_pop
    
    top_candidates = np.array([obj(ind) for ind in population])
    best_idx = np.argmin(top_candidates)
    best_h = np.clip(population[best_idx], 0, 1)
