        if np.sum(h) == 0:
            h = np.zeros(n_points)
        else:
            h = h * target_sum / np.sum(h)
        wave = np.sin(2 * np.pi * np.arange(n_points) / n_points)
        perturb = 0.05 * wave * (1 - np.abs(h - np.roll(h, lag)))
        h = np.clip(h + perturb, 0, 1)
        if np.sum(h) == 0:
            h = np.zeros(n_points)
        else:
            h = h * target_sum / np.sum(h)
        current_c5 = corr[max_index]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    return (best_h, best_c5, n_points)
