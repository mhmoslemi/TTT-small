import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize with initial_h_values, normalizing to meet the integral constraint
    h = np.array(initial_h_values, dtype=np.float64)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0.0, 1.0)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def gradient(x):
        corr = np.correlate(x, 1 - x, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - (n_points - 1)  # Maps idx to actual shift
        grad = np.zeros(n_points)
        for m in range(n_points):
            t1 = 0.0
            if 0 <= m + shift < n_points:
                t1 = (1.0 - x[m + shift]) * dx
            t2 = 0.0
            if 0 <= m - shift < n_points:
                t2 = -x[m - shift] * dx
            grad[m] = t1 + t2
        return grad

    # Define bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    result = minimize(
        fun=objective,
        x0=h,
        method='SLSQP',
        jac=gradient,
        bounds=bounds,
        constraints=cons,
        tol=1e-6,
        options={'maxiter': 150}
    )
    best_h, best_c5 = result.x, result.fun

    # Refinement with perturbations
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        # Random perturbation with adaptive learning rate
        perturb = np.random.normal(0, 0.03, size=n_points)
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        result_perturb = minimize(
            fun=objective,
            x0=new_h,
            method='L-BFGS-B',
            jac=gradient,
            bounds=bounds,
            constraints=cons,
            tol=1e-6,
            options={'maxiter': 50}
        )
        if result_perturb.fun < best_c5:
            best_h, best_c5 = result_perturb.x, result_perturb.fun

    best_h = best_h * target_sum / np.sum(best_h) if np.sum(best_h) > 0 else best_h
    return (best_h, best_c5, n_points)
