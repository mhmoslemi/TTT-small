import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if n_points == 0:
        h = np.array([])
    else:
        if initial_h_values is not None:
            h = initial_h_values.copy()
        else:
            # Generate structured initial guess
            h = np.zeros(n_points)
            if n_points % 2 == 0:
                h[::2] = 1.0
            else:
                h[::2] = 1.0
                h[0] = 0.5
            # Ensure sum is correct
            if np.sum(h) != target_sum:
                h = h * target_sum / np.sum(h)
    # Now h is the initial guess

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    best_h, best_c5 = result.x, result.fun

    alpha = 0.05  # perturbation strength
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag_index = np.argmax(corr)
        # Convert lag index to actual lag. For example, if the correlation is of length L = 2n - 1.
        L = len(corr)
        max_lag = max_lag_index - (L // 2)
        # Generate perturbation based on max_lag
        # For simplicity, use a sinusoidal perturbation with frequency related to max_lag
        # Perturbation vector: alpha * sin(2 * pi * max_lag * i / n)
        perturbation = alpha * np.sin(2 * np.pi * max_lag * np.linspace(0, 2, n_points) / 2)
        new_h = best_h + perturbation
        new_h = np.clip(new_h, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        else:
            new_h = best_h
        # Perform local optimization
        result = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.fun < best_c5:

import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if n_points == 0:
        h = np.array([])
    else:
        if initial_h_values is not None:
            h = initial_h_values.copy()
        else:
            h = np.zeros(n_points)
            if n_points % 2 == 0:
                h[::2] = 1.0
            else:
                h[::2] = 1.0
                h[0] = 0.5
            h = h * target_sum / np.sum(h)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    best_h, best_c5 = result.x, result.fun

    alpha = 0.05
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag_index = np.argmax(corr)
        L = len(corr)
        max_lag = max_lag_index - (L // 2)
        perturbation = alpha * np.sin(2 * np.pi * max_lag * np.linspace(0, 2, n_points) / 2)
        new_h = best_h + perturbation

        result = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun

    return (best_h, best_c5, n_points)
