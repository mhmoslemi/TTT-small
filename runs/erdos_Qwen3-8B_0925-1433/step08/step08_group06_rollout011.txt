from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    if initial_h_values is None:

        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * (2.0 / n_points)

        eps = 1e-5
        grad = np.zeros_like(x)
        for i in range(n_points):
            x_p = x.copy()
            x_p[i] += eps
            corr_p = np.correlate(x_p, 1 - x_p, mode='full')
            grad[i] = (np.max(corr_p) - obj(x)) / eps
