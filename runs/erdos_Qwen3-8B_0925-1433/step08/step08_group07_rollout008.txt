import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    import numpy as np
    from scipy.optimize import differential_evolution, minimize
    import time

    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    if sum_h == 0:
        h = 0.5 * np.ones_like(h)
    else:
        h *= target_sum / sum_h
    h = np.clip(h, 0, 1)

    def objective(h_candidate):
        clipped = np.clip(h_candidate, 0, 1)
        sum_clipped = np.sum(clipped)
        if sum_clipped == 0:
            scaled = 0.5 * np.ones_like(clipped)
        else:
            scaled = clipped * target_sum / sum_clipped
        corr = np.correlate(scaled, 1 - scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points

    start_time = time.time()
    # Run differential evolution for global exploration
    result_de = differential_evolution(objective, bounds, strategy='best1bin',
                                      popsize=20, maxiter=50, tol=1e-6,
                                      mutation=(0.5, 1), recombination=0.9,
                                      seed=seed)
    best_h = result_de.x
    best_c5 = objective(best_h)

    # Local refinement with SLSQP
    def obj_slsqp(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds_slsqp = [(0.0, 1.0)] * n_points
    result_slsqp = minimize(obj_slsqp, best_h, method='SLSQP',
                            bounds=bounds_slsqp, constraints=cons,
                            tol=1e-8, options={'maxiter': 100})

    final_h = result_slsqp.x
    final_c5 = obj_slsqp(final_h)

    # Check initial solution
    initial_c5 = objective(h)
    if initial_c5 < final_c5:
        final_h = h
        final_c5 = initial_c5

    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        return (final_h, final_c5, n_points)

    return (final_h, final_c5, n_points)
