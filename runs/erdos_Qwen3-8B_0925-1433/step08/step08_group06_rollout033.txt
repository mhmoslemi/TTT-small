import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def grad(x):
        corr = np.correlate(x, 1 - x, mode='full')
        max_k = np.argmax(corr)
        k = max_k - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            i_plus_k = i + k
            if 0 <= i_plus_k < n_points:
                grad[i] += (1 - x[i_plus_k]) * dx
            i_minus_k = i - k
            if 0 <= i_minus_k < n_points:
                grad[i] -= x[i_minus_k] * dx
        return grad

    def constraint(x):
        return np.sum(x) - target_sum

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0, 1)] * n_points

    best_h = h
    best_c5 = objective(best_h)
    start_time = time.time()

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, jac=grad, tol=1e-8)
        if res.success:
            best_h = res.x
            best_c5 = res.fun
        else:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        k = max_k - (n_points - 1)
        freq = 2 * np.pi * abs(k) / n_points
        perturb = 0.02 * np.sin(freq * np.arange(n_points))
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res_perturb = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=cons, jac=grad, tol=1e-8)
        if res_perturb.success and res_perturb.fun < best_c5:
            best_h = res_perturb.x
            best_c5 = res_perturb.fun

    return (best_h, best_c5, n_points)
