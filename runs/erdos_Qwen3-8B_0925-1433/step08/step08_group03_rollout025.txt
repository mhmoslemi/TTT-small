import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    if 'initial_h_values' not in globals():
        initial_h_values = np.array([1.0 if i < 10 else 0 for i in range(45)])
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Generate multiple initial candidates
    initial_candidates = []
    h = np.clip(h_initial, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    initial_candidates.append(h)

    freqs = np.random.uniform(0.1, 0.5, 5)
    for freq in freqs:
        wave = np.sin(2 * np.pi * freq * np.linspace(0, 2, n_points))
        perturbed = h_initial + wave * 0.1
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        initial_candidates.append(perturbed)

    for _ in range(3):
        perm = np.random.permutation(n_points)
        permuted = h_initial[perm]
        permuted = np.clip(permuted, 0, 1)
        permuted = permuted * target_sum / np.sum(permuted)
        initial_candidates.append(permuted)

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for init in initial_candidates:
        result = minimize(obj, init, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
        if time.time() - start_time > budget_s:
