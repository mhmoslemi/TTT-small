import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

    population_size = 20
    population = [initial_h_values.copy() + np.random.normal(0, 0.1, n_points) for _ in range(population_size)]
    for i in range(population_size):
        p = population[i]
        p = np.clip(p, 0, 1)
        if np.sum(p) > 0:
            p = p * target_sum / np.sum(p)
        population[i] = p

    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    max_iter = 200
    F = 0.8
    CR = 0.9

    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 200:
            break
        new_pop = []
        for _ in range(population_size):
            indices = np.random.choice(population_size, 3, replace=False)

            d_idx = np.random.randint(population_size)
