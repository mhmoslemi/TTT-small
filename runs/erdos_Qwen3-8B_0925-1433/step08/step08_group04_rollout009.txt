        if n_points % 2 == 0:
            h = np.zeros(n_points)
            h[:n_points//2] = 1.0
        else:
            h = np.zeros(n_points)
            h[: (n_points-1)//2] = 1.0
            h[-1] = 0.5
    else:
        h = initial_h_values.copy()
        n_points = len(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h_clip = np.clip(h, 0, 1)
    if np.sum(h_clip) > 0:
        h = h_clip * target_sum / np.sum(h_clip)
    else:
        h = np.zeros(n_points)
    
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    def cons_eq(x):
        return np.sum(x) - target_sum
    
    cons = [{'type': 'eq', 'fun': cons_eq}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    best_h = result.x
    best_c5 = result.fun
    
    start_time = time.time()
    iterations = 0
    max_iterations = 20
    
    while time.time() - start_time < budget_s and iterations < max_iterations:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr = np.max(corr)
        max_lag_index = np.argmax(corr)
        
        k = max_lag_index
        alpha = 0.05
        perturbation = alpha * np.sin(2 * np.pi * k * np.arange(n_points) / n_points)
        new_h = best_h + perturbation
        new_h = np.clip(new_h, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        else:
            new_h = best_h
        
        result = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun
            iterations += 1
        else:
            noise_perturbation = np.random.normal(0, 0.02, n_points)
            new_h = best_h + noise_perturbation
            new_h = np.clip(new_h, 0, 1)
            if np.sum(new_h) > 0:
                new_h = new_h * target_sum / np.sum(new_h)
            else:
                new_h = best_h
            result = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if result.fun < best_c5:
                best_h, best_c5 = result.x, result.fun
                iterations += 1
    
    return (best_h, best_c5, n_points)
