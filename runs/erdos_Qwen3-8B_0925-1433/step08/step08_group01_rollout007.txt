import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h_values * target_sum / np.sum(initial_h_values)
    h = np.clip(h, 0, 1)
    
    def evaluate_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def feasibility_check(C, h_guess, n_points, target_sum):
        lags = np.arange(-n_points + 1, n_points)
        corr = np.correlate(h_guess, 1 - h_guess, mode='full')
        k_candidates = np.argsort(corr)[-20:]
        k_candidates = np.concatenate((k_candidates, -k_candidates))
        k_candidates = np.unique(k_candidates)
        k_candidates = k_candidates[np.isin(k_candidates, lags)]
        
        constraints = []
        for k in k_candidates:
            if k == 0:
                continue
            def lag_constraint(h_vec):
                return np.correlate(h_vec, 1 - h_vec, mode='full')[k] - (1 - C)
            constraints.append(lag_constraint)
        
        res = minimize(lambda x: 0, h_guess, method='SLSQP',
                       bounds=[(0, 1)] * n_points,
                       constraints=[{'type': 'ineq', 'fun': c} for c in constraints],
                       tol=1e-6)
        if res.success and np.allclose(np.sum(res.x), target_sum, atol=1e-6):
            return True, res.x
        return False, h_guess
    
    low, high = 0.3, 0.4
    best_c5 = high
    best_h = h
    
    start_time = time.time()
    for _ in range(50):

        mid = (low + high) / 2
        feasible, new_h = feasibility_check(mid, h, n_points, target_sum)
        if feasible:

            best_h = new_h

            low = mid
        h = best_h
