import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    if n_points == 0:
        return (np.array([]), 0.0, 0)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def generate_variations(base):
        variations = []
        variations.append(base)
        h_perm = np.random.permutation(base)
        variations.append(h_perm)
        f = 0.25 * (n_points / 4.0)
        sine_wave = 0.5 + 0.5 * np.sin(2 * np.pi * f * np.arange(n_points) / 2)
        variations.append(np.clip(sine_wave, 0, 1) * target_sum / np.sum(np.clip(sine_wave, 0, 1)))
        noise = np.random.normal(0, 0.05, n_points)
        variations.append(np.clip(base + noise, 0, 1) * target_sum / np.sum(np.clip(base + noise, 0, 1)))
        return variations

    start_time = time.time()
    best_h = initial_h_values.copy()
    best_c5 = np.inf

    if initial_h_values is not None:
        variations = generate_variations(initial_h_values)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        for guess in variations:
            try:
                result = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
                if result.success and result.fun < best_c5:
                    best_h = result.x.copy()
                    best_c5 = result.fun
            except:
                pass
            if time.time() - start_time > budget_s:
                break

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag = np.argmax(corr)
        perturbation = 0.03 * np.sin(2 * np.pi * max_lag * np.arange(n_points) / 2)
        new_h = best_h + perturbation
        new_h = np.clip(new_h, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        else:
            new_h = best_h
        result = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success and result.fun < best_c5:
            best_h = result.x.copy()
            best_c5 = result.fun

    return best_h, best_c5, n_points
