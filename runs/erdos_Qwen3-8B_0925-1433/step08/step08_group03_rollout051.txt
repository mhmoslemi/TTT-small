import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial solution
    initial_sum = np.sum(initial_h_values)
    if abs(initial_sum - target_sum) > 1e-6:
        initial_h_values = initial_h_values * target_sum / initial_sum
    initial_h_values = np.clip(initial_h_values, 0, 1)
    
    # Population settings
    pop_size = 30
    generations = 200
    mut_rate = 0.2
    cross_rate = 0.7
    
    # Initial population
    pop = [initial_h_values.copy() for _ in range(pop_size)]
    for i in range(pop_size):
        h = pop[i].copy()
        noise = np.random.normal(0, 0.1, n_points)
        h = np.clip(h + noise, 0, 1)
        h = h * target_sum / np.sum(h)
        pop[i] = h
    
    best_h = initial_h_values.copy()
    best_c5 = float('inf')
    time_start = time.time()
    
    # GA loop
    for _ in range(generations):
        if time.time() - time_start > budget_s:
            break
        # Evaluate all
        fit = []
        for h in pop:
            corr = np.correlate(h, 1 - h, mode='full')
            fit.append(np.max(corr) * dx)
        
        # Select top 20% for breeding
        top_idx = np.argsort(fit)[:int(len(pop)*0.2)]
        top_pop = [pop[i] for i in top_idx]
        
        # Generate new population
        new_pop = []
        while len(new_pop) < pop_size:
            if np.random.rand() < cross_rate:
                # Crossover
                idx1, idx2 = np.random.choice(len(top_pop), 2, replace=False)
                h1, h2 = top_pop[idx1], top_pop[idx2]
                child = 0.5 * h1 + 0.5 * h2
            else:
                # Mutation
                idx = np.random.choice(len(top_pop))
                child = top_pop[idx].copy()
                noise = np.random.normal(0, 0.05, n_points)
                child = np.clip(child + noise, 0, 1)
                child = child * target_sum / np.sum(child)
            new_pop.append(child)
        pop = new_pop
        
        # Track best
        for h in pop:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Final local refinement
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.success:
        best_h = res.x
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
