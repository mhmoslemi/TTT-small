    start_time = time.time()

    for h in initial_candidates:
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h:
            h_clipped = h_clipped * target_sum / sum_h
        result = minimize(compute_c5, h_clipped, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_h = result.x
            best_c5 = current_c5

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        perturb = np.sin(2 * np.pi * np.linspace(0, 1, n_points) * np.random.uniform(0.1, 0.5))
        perturb = np.clip(perturb, -0.1, 0.1)
        new_h = np.clip(best_h + perturb, 0, 1)
        sum_h = np.sum(new_h)
        if sum_h:
            new_h = new_h * target_sum / sum_h
        result = minimize(compute_c5, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_h = result.x
            best_c5 = current_c5

    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n_points)
