import numpy as np
import time

def run(seed=42, budget_s=600):
    import numpy as np
    import time

    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def rescale(h):
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * target_sum / total

    h_init = np.clip(initial_h_values, 0, 1)
    h_init = rescale(h_init)

    num_particles = 20
    particles = []
    for _ in range(num_particles):
        perturb = np.random.normal(0, 0.1, n_points)
        h_perturb = np.clip(h_init + perturb, 0, 1)
        h_perturb = rescale(h_perturb)
        particles.append(h_perturb)

    velocities = [np.random.uniform(-0.05, 0.05, n_points) for _ in range(num_particles)]

    def compute_fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    pbest = [p.copy() for p in particles]
    gbest = min(pbest, key=compute_fitness)
    best_c5 = compute_fitness(gbest)

    start_time = time.time()
    for iteration in range(500):
        if time.time() - start_time > budget_s:
            break
        fitnesses = [compute_fitness(p) for p in particles]
        for i in range(num_particles):
            if fitnesses[i] < compute_fitness(pbest[i]):
                pbest[i] = particles[i].copy()
                if compute_fitness(pbest[i]) < best_c5:
                    best_c5 = compute_fitness(pbest[i])
                    gbest = pbest[i].copy()
        omega = 0.8 - 0.4 * (iteration / 500)
        for i in range(num_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * velocities[i] + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            velocities[i] = np.clip(velocities[i], -0.2 * dx, 0.2 * dx)
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = rescale(particles[i])
    best_h = gbest
    return (best_h, best_c5, n_points)
