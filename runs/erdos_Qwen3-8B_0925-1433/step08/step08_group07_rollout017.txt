    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run DE
    start_time = time.time()
    result_de = differential_evolution(
        fitness_func,
        bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=20,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    elapsed_time = time.time() - start_time
    time_remaining = budget_s - elapsed_time

    best_h_de = result_de.x
    best_c5_de = result_de.fun

    # Refine with SLSQP
    def obj_slsqp(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    cons_slsqp = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    if time_remaining > 0:
        start_slsqp = time.time()
        result_slsqp = minimize(
            obj_slsqp,
            best_h_de,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=cons_slsqp,
            tol=1e-8,
            options={'maxiter': 100}
        )
        elapsed_slsqp = time.time() - start_slsqp
        total_elapsed = elapsed_time + elapsed_slsqp
        if result_slsqp.success and result_slsqp.fun < best_c5_de:
            best_h = result_slsqp.x
            best_c5 = result_slsqp.fun
        else:
            best_h = best_h_de
            best_c5 = best_c5_de
    else:
        best_h = best_h_de
        best_c5 = best_c5_de

    # Final constraint check
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    if sum_best > 0:
        best_h = best_h * (target_sum / sum_best)
    else:
        best_h = np.zeros(n_points)

    return (best_h, best_c5, n_points)
