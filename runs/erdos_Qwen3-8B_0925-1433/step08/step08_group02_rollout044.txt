    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full') * dx
        return np.max(corr)

    def jac(x):
        corr = np.correlate(x, 1 - x, mode='full') * dx
        max_index = np.argmax(corr)
        m = max_index - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j_plus_m = (i + m) % n_points
            j_minus_m = (i - m) % n_points
            grad[i] = (1 - x[j_plus_m] - x[j_minus_m]) * dx
        return grad

    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        result = minimize(obj, h, method='L-BFGS-B', bounds=[(0,1)]*n_points, 
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], 
                          jac=jac, tol=1e-8)
        h = result.x
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = h.copy()
        if _ % 50 == 0:
            corr = np.correlate(h, 1 - h, mode='full') * dx
            max_index = np.argmax(corr)
            m = max_index - (n_points - 1)
            perturbation = np.sin(np.linspace(0, 2 * np.pi, n_points) * 2) * 0.05 * (1 + np.abs(np.roll(h, m) - h))
            h = np.clip(h + perturbation, 0, 1)
            h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h

    return (best_h, best_c5, n_points)
