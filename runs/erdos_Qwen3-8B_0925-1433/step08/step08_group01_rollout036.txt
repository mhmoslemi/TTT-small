import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Scale initial_h_values to meet the sum constraint
    h = np.array(initial_h_values, dtype=np.float64)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Initial optimization with SLSQP using analytical gradient
    def compute_grad(x, m):
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = m
        delta_k = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for j in range(n_points):
            j_plus = j + delta_k
            if 0 <= j_plus < n_points:
                term1 = 1.0 - x[j_plus]
            else:
                term1 = 0.0
            j_minus = j - delta_k
            if 0 <= j_minus < n_points:
                term2 = x[j_minus]
            else:
                term2 = 0.0
            grad[j] = (term1 - term2) * dx
        return grad

    def grad_func(x):
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        return compute_grad(x, k_max)

    start_time = time.time()
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, jac=grad_func, tol=1e-6)
    best_h = result.x
    best_c5 = result.fun

    # Refinement through structured perturbations
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        # Swap segments to avoid local optima
        if n_points < 10:
            continue
        seg_len = int(n_points * 0.3)
        seg1 = np.random.randint(n_points - seg_len + 1)
        seg2 = np.random.randint(n_points - seg_len + 1)
        new_h = best_h.copy()
        new_h[seg1:seg1+seg_len], new_h[seg2:seg2+seg_len] = new_h[seg2:seg2+seg_len], new_h[seg1:seg1+seg_len]
        new_h = np.clip(new_h, 0.0, 1.0)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        else:
            new_h = best_h
        res = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=cons, jac=grad_func, tol=1e-6)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
