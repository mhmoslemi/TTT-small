import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    result = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    best_h, best_c5 = result.x, result.fun

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]

        def focused_obj(x):
            x_clamped = np.clip(x, 0, 1)
            x_scaled = x_clamped * target_sum / np.sum(x_clamped) if np.sum(x_clamped) > 0 else x_clamped
            corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
            return corr[max_idx] * dx

        perturb = np.sin(np.linspace(0, 2 * np.pi, n_points) * (max_idx + 1)) * 0.05
