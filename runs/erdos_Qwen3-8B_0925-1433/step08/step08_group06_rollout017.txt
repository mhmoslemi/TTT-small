import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h
    best_c5 = obj(h)
    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        # Optimize for current worst lag
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        if max_k >= len(corr) - 1:
            break
        # Use finite difference approximation for gradient
        eps = 1e-4 * max(best_h) + 1e-6
        def grad_func(x):
            grad = np.zeros_like(x)
            for i in range(n_points):
                x_plus = np.copy(x)
                x_plus[i] += eps
                corr_plus = np.correlate(x_plus, 1 - x_plus, mode='full')
                grad[i] = (np.max(corr_plus) - obj(x)) / eps
            return grad
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, jac=grad_func, tol=1e-8)
        if res.success:
            best_h = res.x

        else:

        # Apply sinusoidal perturbation aligned with worst lag
