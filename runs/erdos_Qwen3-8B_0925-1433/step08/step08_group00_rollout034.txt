import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Start with initial_h_values
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
    best_h, best_c5 = result.x, result.fun

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Generate a random perturbation
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Final refinement with L-BFGS-B
    result_lbgf = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
    best_c5 = result_lbgf.fun
