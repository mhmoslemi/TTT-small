import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def create_initializations():
        # Base initialization
        h0 = np.clip(initial_h_values, 0, 1)
        h0 = h0 * target_sum / np.sum(h0) if np.sum(h0) > 0 else np.zeros(n_points)
        # Permutation variation
        h_perm = np.random.permutation(h0)
        # Gaussian noise variation
        h_noise = h0 + np.random.normal(0, 0.1, n_points)
        h_noise = np.clip(h_noise, 0, 1)
        h_noise = h_noise * target_sum / np.sum(h_noise) if np.sum(h_noise) > 0 else h0
        # Binary pattern variation
        h_binary = np.zeros(n_points)
        h_binary[:int(n_points/2)] = 1
        np.random.shuffle(h_binary)
        h_binary = h_binary * target_sum / np.sum(h_binary) if np.sum(h_binary) > 0 else h0
        return [h0, h_perm, h_noise, h_binary]

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h, best_c5 = None, float('inf')
    start_time = time.time()

    for init in create_initializations():
        result = minimize(obj, init, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if result.success and result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun
            if time.time() - start_time > budget_s:
                break

    # Refine with structured perturbations
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
