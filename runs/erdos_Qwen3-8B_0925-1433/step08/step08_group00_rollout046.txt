        h_initial = np.zeros(n_points)

    def obj(x):
        h_clipped = np.clip(x, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h > 0:
            h_norm = h_clipped * target_sum / sum_h
        else:
            h_norm = np.zeros_like(h_clipped)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points
    start_time = time.time()

    # Differential evolution phase
    result_de = differential_evolution(
        obj, bounds, strategy='best1bin', 
        popsize=20, tol=1e-8, mutation=(0.5, 1.0),
        recombination=0.9, seed=seed, maxiter=100
    )

    if time.time() - start_time > budget_s - 100:
        best_h = result_de.x
        best_c5 = result_de.fun
    else:
        best_h = result_de.x
        best_c5 = result_de.fun

    # Local refinement with L-BFGS-B
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result_lbgf = minimize(
        obj, best_h, method='L-BFGS-B', bounds=bounds, 
        constraints=cons, tol=1e-8, options={'maxiter': 100}
    )
    best_h = result_lbgf.x
    best_c5 = result_lbgf.fun

    # Final refinement
    result_final = minimize(
        obj, best_h, method='L-BFGS-B', bounds=bounds,
        constraints=cons, tol=1e-8, options={'maxiter': 50}
    )
    best_h = result_final.x
    best_c5 = result_final.fun

    # Final clipping and normalization
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if sum_h > 0:
        best_h = best_h * target_sum / sum_h
    else:
        best_h = np.zeros(n_points)
    
    return (best_h, best_c5, n_points)
