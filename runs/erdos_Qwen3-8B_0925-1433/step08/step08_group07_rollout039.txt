import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    import numpy as np
    from scipy.optimize import minimize
    import time

    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h /= np.sum(h) if np.sum(h) > 0 else 1.0
    h *= target_sum
    h = np.clip(h, 0, 1)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(best_h)

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h /= np.sum(new_h) if np.sum(new_h) > 0 else 1.0
        new_h *= target_sum
        new_h = np.clip(new_h, 0, 1)

        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
