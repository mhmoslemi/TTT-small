import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    h = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * target_sum / sum_h
    else:
        h = np.zeros(n_points)

    def objective_de(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x == 0:
            return np.inf
        x_scaled = x * (target_sum / sum_x)
        corr = np.correlate(x_scaled, 1 - x_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    result_de = differential_evolution(
        objective_de, bounds, strategy='best1bin', maxiter=50, popsize=20, tol=1e-5,
        mutation=(0.5, 1), recombination=0.5
    )
    elapsed_de = time.time() - start_time
    if elapsed_de >= budget_s:
        best_h = h.copy()
        best_c5 = objective_de(h)
    else:
        best_h = result_de.x
        best_c5 = result_de.fun

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds_slsqp = [(0, 1)] * n_points
    cons_slsqp = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    start_slsqp = time.time()
    res_slsqp = minimize(obj, best_h, method='SLSQP', bounds=bounds_slsqp, constraints=cons_slsqp, tol=1e-5)
    elapsed_slsqp = time.time() - start_slsqp

    if elapsed_de + elapsed_slsqp >= budget_s:
        return (best_h, best_c5, n_points)
    if res_slsqp.success:
        best_h = res_slsqp.x
        best_c5 = res_slsqp.fun
    return (best_h, best_c5, n_points)
