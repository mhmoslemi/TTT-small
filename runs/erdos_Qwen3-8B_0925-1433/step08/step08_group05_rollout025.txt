import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h, best_c5 = h.copy(), objective(h)

    def perturb(x, method='rand', amp=0.1):
        if method == 'rand':
            noise = np.random.normal(0, amp, n_points)
        elif method == 'sine':
            freq = np.random.choice([1, 2, 3, 4, 5])
            amp = np.random.uniform(0.05, 0.2)
            x_vals = np.linspace(0, 2, n_points)
            noise = amp * np.sin(2 * np.pi * freq * x_vals)
        else:
            noise = np.zeros(n_points)
        new_h = np.clip(x + noise, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        return new_h

    def optimize(perturbed_h):
        result = minimize(objective, perturbed_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7)
        return result.x, result.fun

    for _ in range(60):
        if time.time() - start_time > budget_s:
            break
        pert_h = perturb(h, method=np.random.choice(['rand', 'sine']))
        curr_h, curr_c5 = optimize(pert_h)
        if curr_c5 < best_c5:
            best_h, best_c5 = curr_h, curr_c5
            h = curr_h

    for _ in range(30):
        if time.time() - start_time > budget_s:
            break
        pert_h = perturb(best_h, method='rand', amp=0.05)
        curr_h, curr_c5 = optimize(pert_h)
        if curr_c5 < best_c5:
            best_h, best_c5 = curr_h, curr_c5

    return (best_h, best_c5, n_points)
