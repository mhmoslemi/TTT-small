import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Generate diverse initial guesses
    def generate_guesses(base):
        guesses = [base.copy()]
        if len(base) == 0:
            return guesses
        # Permutation
        perm = np.random.permutation(base)
        guesses.append(perm)
        # Sinusoidal variations
        freqs = [0.25, 0.5, 0.75]
        for f in freqs:
            x = np.linspace(0, 2, n_points)
            h = 0.5 + 0.5 * np.sin(2 * np.pi * f * x)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            guesses.append(h)
        # Noisy variations
        noise = np.random.normal(0, 0.05, n_points)
        for _ in range(3):
            h_noise = base + noise
            h_noise = np.clip(h_noise, 0, 1)
            h_noise = h_noise * target_sum / np.sum(h_noise)
            guesses.append(h_noise)
        return guesses

    initial_guesses = generate_guesses(initial_h_values)
    unique_guesses = []
    seen = set()
    for h in initial_guesses:
        h_tuple = tuple(h.round(6))
        if h_tuple not in seen:
            seen.add(h_tuple)
            unique_guesses.append(h)
    initial_guesses = unique_guesses

    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    # Initial optimizations
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        guess = np.clip(guess, 0, 1)
        guess = guess * target_sum / np.sum(guess)
        result = minimize(obj, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    # Refinement steps
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag = np.argmax(corr)
        perturb = 0.03 * np.sin(2 * np.pi * max_lag * np.linspace(0, 2, n_points) / 2)
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        result = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    return (best_h, best_c5, n_points)
