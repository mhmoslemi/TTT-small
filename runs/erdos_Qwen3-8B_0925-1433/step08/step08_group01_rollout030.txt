import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    
    low = 0.3
    high = 0.4
    best_c5 = high
    best_h = h.copy()
    time_start = time.time()
    
    def feasibility(C):
        lags = np.arange(-n_points + 1, n_points)
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_candidates = np.argsort(corr)[-20:]
        k_candidates = np.concatenate((k_candidates, -k_candidates))
        k_candidates = np.unique(k_candidates)
        k_valid = k_candidates[np.isin(k_candidates, lags)]
        cons = []
        for k in k_valid:
            if k == 0:
                continue
            def con(h_vec):
                corr = np.correlate(h_vec, 1 - h_vec, mode='full')
                return corr[k] - (1 - C)
            cons.append({'type': 'ineq', 'fun': con})
        res = minimize(lambda x: 0, h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons)
        return res.success
    
    for _ in range(50):
        if time.time() - time_start > budget_s:
            break
        mid = (low + high) / 2
        if feasibility(mid):
            best_c5 = mid
            high = mid
        else:
            low = mid
        h = best_h.copy()
    
    return (best_h, best_c5, n_points)
