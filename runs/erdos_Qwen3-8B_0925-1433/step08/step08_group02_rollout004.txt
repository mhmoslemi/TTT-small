import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full') * dx
        return np.max(corr)

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_corr = np.max(corr)
        k_max = np.argmax(corr)

        # Gradient update for max lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points and i + k_max >= 0:
                grad[i] += (1 - h[i + k_max]) * dx
            if i - k_max >= 0 and i - k_max < n_points:
                grad[i] -= h[i - k_max] * dx

        # Gradient descent step
        h -= 0.1 * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h

        # Periodic perturbation
        if _ % 50 == 0 and _ > 0:
            perturbation = np.sin(np.linspace(0, 2 * np.pi, n_points) * 2) * 0.05 * (1 + np.abs(np.roll(h, k_max) - h))
            new_h = np.clip(h + perturbation, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h
            res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
            if res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

        # Track best
        if obj(h) < best_c5:
            best_h = h.copy()
            best_c5 = obj(h)

    return (best_h, best_c5, n_points)
