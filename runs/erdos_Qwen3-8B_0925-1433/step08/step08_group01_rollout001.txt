        if k < 0:
            valid_i = np.arange(n_points - (-k))
        else:
            valid_i = np.arange(n_points - k)
        if len(valid_i) < 1:
            continue
        a = np.random.randint(len(valid_i))
        start_idx = valid_i[a]
        end_idx = min(start_idx + 5, n_points)
        start_idx2 = np.random.randint(0, n_points - 5)
        end_idx2 = min(start_idx2 + 5, n_points)
        new_h = best_h.copy()
        new_h[start_idx:end_idx], new_h[start_idx2:end_idx2] = new_h[start_idx2:end_idx2], new_h[start_idx:end_idx]
        result = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun

    return (best_h, best_c5, n_points)
