from scipy.optimize import minimize
import time

    global initial_h_values

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def jac_obj(x, k):
        eps = 1e-5
        grad = np.zeros_like(x)
        for i in range(n_points):

            corr_plus = np.correlate(x_plus, 1 - x_plus, mode='full')
            grad[i] = (np.max(corr_plus) - obj(x)) / eps
        return grad

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    best_h = h
    best_c5 = obj(h)
    start_time = time.time()
