from scipy.optimize import minimize
import time

    np.random.seed(seed)
    n_points = len(initial_h_values)

    dx = 2.0 / n_points

    def obj(x):

        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    best_h = h.copy()

    start_time = time.time()

        new_h = new_h * target_sum / np.sum(new_h)
