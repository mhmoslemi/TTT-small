import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(best_h)

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Optimize using L-BFGS-B
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            best_h = res.x
            best_c5 = res.fun
        # Targeted perturbation based on max lag
        if _ % 50 == 0:
            corr = np.correlate(best_h, 1 - best_h, mode='full')
            max_index = np.argmax(corr)
            lag_idx = max_index - n_points + 1
            if lag_idx > 0:  # positive lag
                for i in range(n_points - lag_idx):
                    perturb = 0.05 * np.sin(2 * np.pi * (i + lag_idx) / n_points)
                    best_h[i] += perturb
                    best_h[i + lag_idx] += perturb
            else:  # negative lag
                lag_abs = -lag_idx
                for i in range(lag_abs, n_points):
                    perturb = 0.05 * np.sin(2 * np.pi * i / n_points)
                    best_h[i] += perturb
                    best_h[i - lag_abs] += perturb
            best_h = np.clip(best_h, 0, 1)
            if np.sum(best_h) > 0:
                best_h = best_h * target_sum / np.sum(best_h)
            else:
                best_h = np.zeros(n_points)
        # Update best if improved
        if obj(best_h) < best_c5:
            best_c5 = obj(best_h)

    return (best_h, best_c5, n_points)
