import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    population = [initial_h_values.copy()]
    for _ in range(29):
        var = np.random.normal(0, 0.1, n_points)
        new_h = np.clip(initial_h_values + var, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        population.append(new_h)

    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    start_time = time.time()
    generation = 0

    while time.time() - start_time < budget_s:
        if generation >= 200:
            break
        fitness = np.array([evaluate(h) for h in population])
        idx = np.argsort(fitness)
        selected = idx[:int(len(population)*0.2)]
        parents = [population[i] for i in selected]

        # Crossover
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                c1 = 0.5 * parents[i] + 0.5 * parents[i+1]
                offspring.append(c1)
            else:
                offspring.append(parents[i])

        # Mutation
        for h in offspring:
            m = np.random.normal(0, 0.08, n_points)
            h_mut = np.clip(h + m, 0, 1)
            h_mut = h_mut * target_sum / np.sum(h_mut)
            offspring.append(h_mut)

        replacement = [population[i] for i in idx[-int(len(population)*0.8):]]
        population = replacement + offspring[:len(population)-len(replacement)]

        # Local refinement
        for h in parents:
            def obj(x):
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * dx
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-7)
            if res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x
        generation += 1

    return (best_h, best_c5, n_points)
