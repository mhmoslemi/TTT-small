import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600):
    import numpy as np
    from scipy.optimize import differential_evolution
    import time

    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h from initial_h_values
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        c5 = np.max(corr) * dx
        penalty = 1e4 * abs(np.sum(x) - target_sum)
        return c5 + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    result = differential_evolution(obj, bounds, strategy='best1bin',
                                   popsize=15, maxiter=100, tol=1e-10,
                                   mutation=(0.5, 1.0), recombination=0.9,
                                   seed=seed, callback=lambda x: None)

    best_h = result.x
    # Recompute C5 without penalty
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    return (best_h, best_c5, n_points)
