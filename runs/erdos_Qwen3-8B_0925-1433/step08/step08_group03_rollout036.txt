        if time.time() - start_time > budget_s:
            break

    # Adaptive perturbations
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.03, len(best_h))
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = create_symmetric(new_h)
        half = n_points // 2
        variables = new_h[:half]
        def obj(x):
            full_h = np.zeros(n_points)
            for i in range(len(x)):
                full_h[i] = x[i]
                if i < n_points - 1 - i:
                    full_h[n_points - 1 - i] = 1 - x[i]
            if n_points % 2 == 1:
                full_h[n_points//2] = 0.5
            corr = np.correlate(full_h, 1 - full_h, mode='full')
            return np.max(corr) * dx
        res = minimize(obj, variables, method='SLSQP', bounds=[(0,1)]*len(variables), tol=1e-8)
        if res.success:
            optimized_vars = res.x
            optimized_h = np.zeros(n_points)
            for i in range(len(optimized_vars)):
                optimized_h[i] = optimized_vars[i]
                if i < n_points - 1 - i:
                    optimized_h[n_points - 1 - i] = 1 - optimized_vars[i]
            if n_points % 2 == 1:
                optimized_h[n_points//2] = 0.5
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h

    return (best_h, best_c5, n_points)
