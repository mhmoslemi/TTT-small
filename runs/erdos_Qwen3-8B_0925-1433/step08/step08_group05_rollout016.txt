import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    base_h = np.array(initial_h_values)
    base_h = np.clip(base_h, 0, 1)
    base_h = base_h * target_sum / np.sum(base_h) if np.sum(base_h) > 0 else np.zeros(n_points)
    best_h = base_h.copy()
    best_c5 = np.max(np.correlate(base_h, 1 - base_h, mode='full')) * dx

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()

    # Generate 8 structured perturbations
    perturbations = []
    for _ in range(8):
        freqs = np.random.choice([1, 2, 3, 4, 5], size=3)
        amps = np.random.uniform(0.05, 0.2, size=3)
        pert = np.zeros_like(base_h)
        for f, a in zip(freqs, amps):
            pert += a * np.sin(2 * np.pi * f * np.linspace(0, 2, n_points))
        perturbed = base_h + pert
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        perturbations.append(perturbed)

    for p in perturbations:
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, p, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success and result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun

    # Local refinement
    for _ in range(30):
        if time.time() - start_time > budget_s:
            break
        noise = np.random.normal(0, 0.05, size=n_points)
        perturbed = best_h + noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        result = minimize(objective, perturbed, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success and result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun

    return (best_h, best_c5, n_points)
