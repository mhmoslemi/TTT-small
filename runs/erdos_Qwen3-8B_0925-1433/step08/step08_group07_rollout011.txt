            x_minus = np.clip(x_minus, 0, 1)
            sum_m = np.sum(x_minus)
            x_minus = x_minus * (n / 2) / sum_m if sum_m != 0 else x_minus
            corr_m = np.correlate(x_minus, 1 - x_minus, mode='full')
            v_m = np.max(corr_m) * dx
            grad[i] = (v_p - v_m) / (2 * eps)
        return grad

    def compute_grad_reg(x):
        n = len(x)
        grad = np.zeros(n)
        for i in range(n):
            if i == 0:
                val = -np.sign(x[1] - x[0])
                grad[i] = val
            elif i == n-1:
                val = np.sign(x[i] - x[i-1])
                grad[i] = val
            else:
                val1 = -np.sign(x[i+1] - x[i])
                val2 = np.sign(x[i] - x[i-1])
                grad[i] = val1 + val2
        return grad

    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()
    lr = 0.1
    lambda_reg = 0.5
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        grad = compute_grad_obj(best_h) + lambda_reg * compute_grad_reg(best_h)
        h = best_h - lr * grad
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        h = h * target_sum / s if s != 0 else np.zeros_like(h)
        current_c5 = obj(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            lr *= 0.99
            lambda_reg *= 1.01

    return (best_h, best_c5, n_points)
