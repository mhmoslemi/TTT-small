        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_corr = np.max(corr)
        if max_corr < best_c5:
            best_c5 = max_corr
            best_h = h.copy()

        k_max = np.argmax(corr)
        perturbation = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            perturbation[i] = -h[j]
        gradient = perturbation * dx

        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(lambda x: np.max(np.correlate(x, 1 - x, mode='full') * dx), 
                       h, 
                       method='L-BFGS-B',
                       bounds=bounds,
                       jac=gradient, 
                       constraints=cons,
                       tol=1e-8)
        h = res.x

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h

        if iteration % 50 == 0:
            perturbation = np.sin(np.linspace(0, 2*np.pi, n_points) * 2) * 0.05 * (1 + np.abs(np.roll(h, k_max) - h))
            h = np.clip(h + perturbation, 0, 1)
            h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h
    return (best_h, best_c5, n_points)
