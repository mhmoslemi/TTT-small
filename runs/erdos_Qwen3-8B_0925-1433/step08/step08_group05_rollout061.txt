        overlaps = np.zeros(n - 1)
        for k in range(1, n):
            total = 0.0
            for i in range(n - k):
                total += h[i] * (1 - h[i + k])
            overlaps[k - 1] = total
        return overlaps

    def objective_k(x, k):
        total = 0.0
        for i in range(len(x) - k):
            total += x[i] * (1 - x[i + k])
        return total * dx

    best_h = h.copy()
    best_c5 = np.max(compute_overlaps(h)) * dx
    start_time = time.time()

    while time.time() - start_time < budget_s:
        overlaps = compute_overlaps(h)
        if np.all(overlaps <= 1e-6):
            break
        max_overlap_val = np.max(overlaps)
        max_shift_idx = np.argmax(overlaps)
        k = max_shift_idx + 1
        perturbation = 0.05 * np.random.randn(n_points)
        perturbed_h = np.clip(h + perturbation, 0, 1)
        perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)

        def objective(x):
            return objective_k(x, k)

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points

        result = minimize(objective, perturbed_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = result.x
            new_c5 = np.max(compute_overlaps(h)) * dx
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5

    return (best_h, best_c5, n_points)
