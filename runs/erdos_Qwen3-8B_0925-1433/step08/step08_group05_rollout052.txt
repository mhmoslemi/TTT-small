import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def find_worst_shift(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.argmax(corr)

    def generate_perturbation(x, worst_k, amp):
        pert = np.zeros_like(x)
        for i in range(n_points - worst_k):
            pert[i] += np.random.uniform(-amp, amp) * (1 - x[i])
            pert[i + worst_k] += np.random.uniform(-amp, amp) * x[i + worst_k]
        return pert

    start_time = time.time()
    best_h = h
    best_c5 = objective(h)

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        worst_k = find_worst_shift(best_h)
        pert = generate_perturbation(best_h, worst_k, 0.1)
        new_h = best_h + pert
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        res = minimize(objective, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
