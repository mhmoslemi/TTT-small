import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def compute_corr(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return corr * dx

    def max_corr(x):
        return np.max(compute_corr(x))

    def constraint(x):
        return np.sum(x) - target_sum

    def objective(x, k):
        corr = np.correlate(x, 1 - x, mode='full')
        return corr[k] * dx

    best_h = h.copy()
    best_c5 = max_corr(best_h)
    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        current_c5 = max_corr(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()

        crt_corr = compute_corr(best_h)
        max_lag_idx = np.argmax(crt_corr)
        if crt_corr[max_lag_idx] < 1e-10:
            break

        cons = [{'type': 'eq', 'fun': constraint}]
        bounds = [(0, 1)] * n_points
        res = minimize(lambda x: objective(x, max_lag_idx), best_h, method='L-BFGS-B', bounds=bounds, constraints=cons)
        best_h = res.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h) if np.sum(best_h) > 0 else best_h

        if _ % 50 == 0:
            k = max_lag_idx
            perturbation = np.sin(2 * np.pi * np.arange(n_points) * (k + 1) / n_points) * 0.05
            new_h = np.clip(best_h + perturbation, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            cons = [{'type': 'eq', 'fun': constraint}]
            bounds = [(0, 1)] * n_points
            res = minimize(lambda x: objective(x, max_lag_idx), new_h, method='L-BFGS-B', bounds=bounds, constraints=cons)
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * target_sum / np.sum(best_h) if np.sum(best_h) > 0 else best_h

    final_corr = compute_corr(best_h)
    final_c5 = np.max(final_corr)
    return (best_h, final_c5, n_points)
