import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h, best_c5 = h.copy(), obj(h)

    for iteration in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = (max_idx - (n_points - 1)) * dx  # Convert index to actual lag

        grad = np.zeros(n_points)
        for i in range(n_points):
            if (i * dx + k) >= 0 and (i * dx + k) <= 2:
                j = int((i * dx + k) / dx)
                if 0 <= j < n_points:
                    grad[i] = -h[j] * dx

        h_new = h - 0.01 * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new) if np.sum(h_new) > 0 else h_new

        res = minimize(obj, h_new, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
            h = res.x

        if iteration % 50 == 0:
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr)
            k = (max_idx - (n_points - 1)) * dx
            perturb = np.sin(np.linspace(0, 2 * np.pi, n_points) * 2) * 0.02 * (1 + np.abs(np.roll(h, int(k / dx)) - h))
            h = np.clip(h + perturb, 0, 1)
            h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h

    return (best_h, best_c5, n_points)
