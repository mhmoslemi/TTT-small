import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values
    h = np.array(h)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)

    def objective(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()
    result = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=15, mutation=(0.5, 0.8), 
        recombination=0.85, tol=1e-8, maxiter=100, seed=seed)
    best_h = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time

    if elapsed < budget_s - 200:
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, 
                          constraints=cons, tol=1e-8)
        best_c5 = result.fun
        best_h = result.x

    return (best_h, best_c5, n_points)
