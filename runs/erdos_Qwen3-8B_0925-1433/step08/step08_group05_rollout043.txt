import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    
    best_h = h
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    def perturb(x, amp=0.05):
        pert = amp * np.random.randn(n_points)
        new_h = np.clip(x + pert, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        return new_h
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        new_h = perturb(best_h)
        result = minimize(objective, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success and result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun
    
    return (best_h, best_c5, n_points)
