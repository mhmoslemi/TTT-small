import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial guess
    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)

    def compute_c5(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0, 1)] * n_points

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        # Trust-constr optimization
        res = minimize(compute_c5, best_h, method='trust-constr',
                      bounds=bounds, constraints=cons, tol=1e-8)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun
        # Structured perturbation
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        if max_k < n_points - 1:
            freq = 2 * np.pi * max_k / n_points
            perturb = 0.01 * np.sin(freq * np.arange(n_points))
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            res_perturb = minimize(compute_c5, new_h, method='trust-constr',
                                bounds=bounds, constraints=cons, tol=1e-8)
            if res_perturb.success and res_perturb.fun < best_c5:
                best_h = res_perturb.x
                best_c5 = res_perturb.fun

    return (best_h, best_c5, n_points)
