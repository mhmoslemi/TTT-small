import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def jac_obj(x):
        eps = 1e-3 * (np.max(x) + 1e-6)
        grads = []
        for i in range(n_points):
            x_p = x.copy()
            x_p[i] += eps
            corr_p = np.correlate(x_p, 1 - x_p, mode='full')
            grad = (np.max(corr_p) - obj(x)) / eps
            grads.append(grad)
        return np.array(grads)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    best_h = h
    best_c5 = obj(h)
    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        # Optimize at worst lag
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        if max_k < n_points - 1:
            res = minimize(obj, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          jac=jac_obj, tol=1e-6)
            if res.success:
                best_h = res.x
                best_c5 = res.fun
        # Perturb
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        if max_k < n_points - 1:
            freq = 2 * np.pi * max_k / n_points
            perturb = 0.005 * np.sin(freq * np.arange(n_points))
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            res_perturb = minimize(obj, new_h, method='SLSQP',
                                bounds=bounds, constraints=cons,
                                jac=jac_obj, tol=1e-6)
            if res_perturb.success and res_perturb.fun < best_c5:
                best_h = res_perturb.x
                best_c5 = res_perturb.fun
    return (best_h, best_c5, n_points)
