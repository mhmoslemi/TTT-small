    def create_symmetric(h):
        first_half = h[:half]
        second_half = 1 - first_half[::-1]
        if n_points % 2 == 1:
            return np.concatenate([first_half, [0.5], second_half])
        else:
            return np.concatenate([first_half, second_half])

    h = create_symmetric(initial_h_values)

    def obj(first_half):
        full = create_full_h(first_half)
        corr = np.correlate(full, 1 - full, mode='full')
        return np.max(corr) * dx

    def create_full_h(first_half):
        if n_points % 2 == 0:
            return np.concatenate([first_half, 1 - first_half[::-1]])
        else:
            return np.concatenate([first_half, [0.5], 1 - first_half[::-1]])

    def evaluate_h(first_half):
        full = create_full_h(first_half)
        corr = np.correlate(full, 1 - full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * half
    initial_first_half = h[:half]

    result = minimize(evaluate_h, initial_first_half, method='SLSQP', bounds=bounds, tol=1e-8)
    best_h = create_full_h(result.x)
    best_c5 = result.fun
    best_time = time.time()

    for _ in range(100):
        if time.time() - best_time > budget_s:
            break
        pert = np.random.normal(0, 0.05, half)
        new_first = np.clip(result.x + pert, 0, 1)
        res = minimize(evaluate_h, new_first, method='SLSQP', bounds=bounds, tol=1e-8)
        if res.fun < best_c5:
            best_c5 = res.fun
            result = res
            best_h = create_full_h(result.x)
            best_time = time.time()

    return (best_h, best_c5, n_points)
