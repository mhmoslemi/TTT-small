import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values) if initial_h_values is not None else 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points // 2] = 1.0
        if n_points % 2 == 1:
            h[n_points // 2] = 0.5
        h = np.clip(h, 0, 1)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    best_h, best_c5 = result.x, result.fun

    start_time = time.time()
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_lag = np.argmax(corr)
        alpha = 0.05
        perturb = alpha * np.sin(2 * np.pi * max_lag * np.linspace(0, 2, n_points) / 2)
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        else:
            new_h = best_h
        res_local = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res_local.fun < best_c5:
            best_h, best_c5 = res_local.x, res_local.fun

    return (best_h, best_c5, n_points)
