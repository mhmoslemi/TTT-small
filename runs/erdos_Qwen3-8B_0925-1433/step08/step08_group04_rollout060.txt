import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    if initial_h_values is not None:
        h = initial_h_values.copy()
    else:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * 0.15 * x)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def generate_variants(h_initial):
        variations = [h_initial]
        perm = np.random.permutation(h_initial)
        variations.append(perm)
        noise = 0.05 * np.random.randn(n_points)
        noisy = h_initial + noise
        noisy = np.clip(noisy, 0, 1)
        noisy = noisy * target_sum / np.sum(noisy)
        variations.append(noisy)
        freq = 0.2
        sin_var = 0.5 + 0.5 * np.sin(2 * np.pi * freq * np.linspace(0, 2, n_points))
        sin_var = np.clip(sin_var, 0, 1)
        sin_var = sin_var * target_sum / np.sum(sin_var)
        variations.append(sin_var)
        shifted = np.roll(h_initial, 1)
        shifted = np.clip(shifted, 0, 1)
        shifted = shifted * target_sum / np.sum(shifted)
        variations.append(shifted)
        return variations

    variants = generate_variants(h)
    best_h, best_c5 = h, float('inf')
    start_time = time.time()

    for guess in variants:
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x
        if time.time() - start_time > budget_s:

    for _ in range(10):

        new_h = best_h + perturb

        if np.sum(new_h) > 0:
            new_h = new_h * target_sum / np.sum(new_h)
        else:
