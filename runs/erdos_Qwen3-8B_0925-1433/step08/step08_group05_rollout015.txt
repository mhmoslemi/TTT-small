import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    eta = 0.05
    max_iterations = 1000
    time_limit = budget_s
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > time_limit:
            break

        corr_full = np.correlate(h, h, mode='full') * dx
        autocorrs = corr_full[n-1:n-1 + n]
        k_min = np.argmin(autocorrs)
        lag = k_min

        grad = np.zeros(n_points)
        for m in range(n_points):
            if m + lag < n_points:
                grad[m] += h[m + lag]
            if m - lag >= 0:
                grad[m] += h[m - lag]
        grad *= dx

        h += eta * grad
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            h = np.full(n_points, 0.5)
        else:
            h = h * target_sum / current_sum

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full') * dx
        return np.max(corr)

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    best_h, best_c5 = result.x, result.fun

    return (best_h, best_c5, n_points)
