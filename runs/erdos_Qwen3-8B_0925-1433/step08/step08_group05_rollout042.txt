import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def random_perturb(x, amp=0.1):
        noise = np.random.normal(0, amp, n_points)
        new_h = np.clip(x + noise, 0, 1)
        return new_h * target_sum / np.sum(np.clip(new_h, 0, 1))

    def sine_perturb(x, freq, amp=0.1):
        pert = amp * np.sin(freq * np.linspace(0, 2 * np.pi, n_points))
        new_h = np.clip(x + pert, 0, 1)
        return new_h * target_sum / np.sum(np.clip(new_h, 0, 1))

    def evaluate_and_optimize(start_h, max_iter=20):
        best_h = start_h.copy()
        best_c5 = objective(best_h)
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            pert_funcs = [random_perturb, sine_perturb]
            for pert_func in pert_funcs:
                new_h = pert_func(best_h, freq=np.random.choice([1, 2, 3, 4, 5]))
                res = minimize(objective, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons)
                if res.success:
                    if res.fun < best_c5:
                        best_h, best_c5 = res.x, res.fun
        return best_h, best_c5

    start_time = time.time()
    best_h, best_c5 = h.copy(), objective(h)
    best_h, best_c5 = evaluate_and_optimize(best_h, max_iter=50)

    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        new_h = random_perturb(best_h)
        res = minimize(objective, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
