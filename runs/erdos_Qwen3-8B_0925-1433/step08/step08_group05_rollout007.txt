            autocorrs.append(corr)
        return np.array(autocorrs)

    def compute_gradient(h, lag, dx, n_points):
        grad = np.zeros_like(h)
        for j in range(n_points):
            if j - lag >= 0:
                grad[j] += h[j - lag]
            if j + lag < n_points:
                grad[j] += h[j + lag]
        grad *= dx
        return grad

    # Gradient ascent steps
    learning_rate = 0.05
    max_iterations = 1000
    start_time = time.time()

    best_h, best_c5 = result.x, result.fun

    # Final evaluation
    return (best_h, best_c5, n_points)

def run(seed=42, budget_s=600):
    global initial_h_values

    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def compute_autocorrs(h, dx, n_points):
        autocorrs = []
        for k in range(1, n_points):
            corr = np.dot(h[:n_points - k], h[k:]) * dx
            autocorrs.append(corr)
        return np.array(autocorrs)

    def compute_gradient(h, lag, dx, n_points):
        grad = np.zeros_like(h)
        for j in range(n_points):
            if j - lag >= 0:
                grad[j] += h[j - lag]
            if j + lag < n_points:
                grad[j] += h[j + lag]
        grad *= dx
        return grad

    learning_rate = 0.05
    max_iterations = 500
    start_time = time.time()
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        min_k = np.argmin(autocorrs)
        lag = min_k + 1
        if lag == 0:
