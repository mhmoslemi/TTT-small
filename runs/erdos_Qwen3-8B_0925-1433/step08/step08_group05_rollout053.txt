import numpy as np
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr[n_points - 1: 2 * n_points - 1]) * dx

    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        shifts = corr[n_points - 1: 2 * n_points - 1]
        top_k = min(5, len(shifts))
        top_shift_indices = np.argsort(shifts)[::-1][:top_k]
        top_shifts = [i + 1 for i in top_shift_indices]

        mask = np.zeros(n_points)
        for k in top_shifts:
            for i in range(n_points - k):
                mask[i] += 1
                mask[i + k] += 1

        amp = 0.1 if _ < 100 else 0.05
        perturbation = np.random.normal(0, amp, n_points) * mask
        new_h = h + perturbation
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)

        new_c5 = evaluate_c5(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()

        h = new_h

    return (best_h, best_c5, n_points)
