        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
        
        def obj(x):
            x = np.clip(x, 0, 1)
            x = x * target_sum / np.sum(x)
            return compute_correlation(x)[current_lag]
        
        grad = compute_gradient(best_h, current_lag)
        
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, jac=grad, tol=1e-6)
        if res.success:
            best_h = res.x
        
        if step % 50 == 0:
            perturbation = np.sin(np.linspace(0, 2 * np.pi, n_points) * 2) * 0.05
            perturbed_h = best_h + perturbation
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * target_sum / np.sum(perturbed_h)
            perturbed_c5, _ = max_correlation(perturbed_h)
            if perturbed_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = perturbed_c5
    
    return (best_h, best_c5, n_points)
