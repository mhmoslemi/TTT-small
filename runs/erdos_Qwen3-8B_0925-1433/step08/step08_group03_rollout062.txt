import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize with a sinusoidal pattern based on initial_h_values
    h = np.array(initial_h_values, dtype=np.float64)
    freq = 0.5
    sinusoidal = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n_points) / n_points * freq))

    sum_h = np.sum(h)
    h = h * target_sum / sum_h
    h = np.clip(h, 0, 1)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    result = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    best_h, best_c5 = result.x, result.fun

    # Apply perturbations and refine
    for _ in range(10):
        if time.time() - start_time > budget_s:
