import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial structure: clip and normalize
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    # Generate structured perturbations
    def perturb_structure(h, freq=0.3, shift=0.0):
        modulated = 0.5 * (1 + np.cos(2 * np.pi * (np.arange(n_points) / n_points * freq) + shift))
        return np.clip(h * modulated, 0, 1)

    # Define objective: max correlation
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    # Constraints
    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Optimization loop with L-BFGS-B and structured perturbations
    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()

    for _ in range(400):
        if time.time() - start_time > budget_s:
            break
        # Apply structured perturbation
        h_perturb = perturb_structure(best_h, freq=0.4, shift=np.random.uniform(0, 1))
        h_perturb = h_perturb * target_sum / np.sum(h_perturb) if np.sum(h_perturb) > 0 else best_h

        # Optimize with L-BFGS-B
        res = minimize(obj, h_perturb, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

        # Apply symmetry and random perturbation
        if np.random.rand() < 0.2:

            h_sym = h_sym * target_sum / np.sum(h_sym) if np.sum(h_sym) > 0 else best_h
            res = minimize(obj, h_sym, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success and res.fun < best_c5:

        # Adaptive noise perturbation
