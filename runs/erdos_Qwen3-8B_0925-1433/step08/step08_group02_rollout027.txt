import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj_and_grad(x):
        c = np.correlate(x, 1 - x, mode='full')
        max_c = np.max(c)
        max_idx = np.argmax(c)
        lag = max_idx - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            if 0 <= (i + lag) < n_points and 0 <= (i - lag) < n_points:
                grad[i] = (1 - x[i + lag]) - x[i - lag]
        return max_c * dx, grad * dx

    bounds = [(0., 1.) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj_and_grad(best_h)[0]

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        try:
            result = minimize(obj_and_grad, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, jac=True, tol=1e-4)
            if result.success:
                best_h = result.x
                best_c5 = result.fun
            else:
                result = minimize(obj_and_grad, best_h, method='COBYLA', bounds=bounds, constraints=cons, jac=True, tol=1e-4)
                best_h = result.x
                best_c5 = result.fun
        except:
            continue

        if _ % 50 == 0:
            c = np.correlate(best_h, 1 - best_h, mode='full')
            max_idx = np.argmax(c)
            lag = max_idx - (n_points - 1)
            perturbation = np.sin(np.linspace(0, np.pi * 2, n_points) * 2) * 0.05
            best_h = np.clip(best_h + perturbation, 0., 1.)
            if np.sum(best_h) > 0:
                best_h = best_h * target_sum / np.sum(best_h)

    return (best_h, best_c5, n_points)
