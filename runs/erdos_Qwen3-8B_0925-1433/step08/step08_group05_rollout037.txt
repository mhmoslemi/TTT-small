import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def perturb(x, amp, freq):
        freq_vec = np.linspace(0, 2 * np.pi, n_points)
        pert = amp * np.sin(freq * freq_vec)
        return np.clip(x + pert, 0, 1) * target_sum / np.sum(np.clip(x + pert, 0, 1))

    start_time = time.time()
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    best_h, best_c5 = result.x, result.fun

    for _ in range(40):
        if time.time() - start_time > budget_s:
            break
        shifts = np.argmax(np.correlate(best_h, 1 - best_h, mode='full'), axis=0)
        top_indices = np.unique(shifts[:10])
        for shift in top_indices:
            new_h = best_h.copy()

            mask[shift:] += 0.1 * np.abs(new_h[:n_points - shift])
            new_h -= mask
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h)
            res = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
