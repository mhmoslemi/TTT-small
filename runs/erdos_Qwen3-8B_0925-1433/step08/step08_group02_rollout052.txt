import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()

    def compute_c5(h_):
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    def get_gradient(h_, lag):
        n = len(h_)
        grad = np.zeros(n)
        for i in range(n):
            j_add = i + lag
            if 0 <= j_add < n:
                grad[i] += (1 - h_[j_add]) * dx
            j_sub = i - lag
            if 0 <= j_sub < n:
                grad[i] -= h_[j_sub] * dx
        return grad

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        grad = get_gradient(h, k)

        # Use L-BFGS-B to minimize the current lag's contribution
        def objective(x):
            return compute_c5(x)
        def jac(x):
            return get_gradient(x, k)

        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(objective, h, method='L-BFGS-B', jac=jac, bounds=bounds, constraints=cons, tol=1e-8)
        h = res.x

        if res.fun < best_c5:
            best_c5 = res.fun
            best_h = h.copy()

        if _ % 50 == 0:
            perturbation = np.sin(np.linspace(0, 2*np.pi, n_points) * 2) * 0.05 * (1 + np.abs(np.roll(h, k) - h))
            h = np.clip(h + perturbation, 0, 1)
            h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h

    return (best_h, best_c5, n_points)
