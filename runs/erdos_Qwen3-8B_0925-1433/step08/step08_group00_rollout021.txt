def run(seed=42, budget_s=600):

    # Initialize h
    h = np.array(initial_h_values)
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * target_sum / h_sum
    else:
        h = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = np.inf

    start_time = time.time()
    learning_rate = 0.1
    decay_rate = 0.999
    for _ in range(10000):  # or until budget is up
        if time.time() - start_time > budget_s:
            break
        # Compute full correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        if max_corr < best_c5:
            best_c5 = max_corr
            best_h = h.copy()
        # Find the index of max_corr
        idx = np.argmax(corr)
        # Compute lag k_max
        k_max = (n_points - 1) - idx
        # Compute gradient for this lag
        gradient = np.zeros(n_points)

            j_minus = (j - k_max) % n_points

        # Clip and normalize

    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
