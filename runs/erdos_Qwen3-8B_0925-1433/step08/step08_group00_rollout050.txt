import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if s > 0:
            x = x * target_sum / s
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points

    start_time = time.time()

    result_de = differential_evolution(
        obj, bounds, strategy='best1bin', 
        maxiter=100, popsize=15, tol=1e-6,
        mutation=(0.5, 1), recombination=0.8
    )
    h_de = result_de.x
    c5_de = result_de.fun

    result_slsqp = minimize(
        obj, h_de, method='SLSQP',
        bounds=bounds, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
        tol=1e-8
    )
    h_refined = result_slsqp.x
    c5_refined = result_slsqp.fun

    if time.time() - start_time < budget_s:
        result_lbgf = minimize(
            obj, h_refined, method='L-BFGS-B',
            bounds=bounds, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            tol=1e-8
        )
        c5_refined = result_lbgf.fun
        h_refined = result_lbgf.x

    return (h_refined, c5_refined, n_points)
