    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break

        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        lag_idx = np.argmax(corr)
        lag_k = lag_idx - (n_points - 1)
        lag_k = max(0, min(lag_k, n_points - 1))

        h_shifted = np.roll(h, -lag_k)
        grad = 1.0 - h_shifted
        h = h - 0.1 * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

        wave = np.sin(2 * np.pi * np.arange(n_points) / n_points)
        perturb = 0.05 * wave * (1 - np.abs(h - np.roll(h, lag_k)))
        h = np.clip(h + perturb, 0, 1)
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h

    return (best_h, best_c5, n_points)
