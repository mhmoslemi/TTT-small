```python
import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def cons(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1)] * n_points
    cons_dict = {'type': 'eq', 'fun': cons}

    best_h = h
    best_c5 = obj(h)
    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        H = np.fft.fft(best_h)
        f_c = int(0.1 * n_points)
        filter_ = np.zeros_like(H)
        filter_[:f_c] = 1.0
        H_filtered = H * filter_
        h_filtered = np.fft.ifft(H_filtered).real
        h_filtered = np.clip(h_filtered, 0, 1)
        h_filtered = h_filtered * target_sum / np.sum(h_filtered) if np.sum(h_filtered) > 0 else h_filtered

        result = minimize(obj, h_filtered, method='SLSQP', bounds=bounds, constraints=cons_dict, tol=1e-8)
        if result.success:
            best_h = result.x
            best_c5 = result.fun
        else:
            break

    return (best_h, best_c5, n_points)
```
