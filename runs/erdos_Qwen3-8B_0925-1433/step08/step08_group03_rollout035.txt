import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values)
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    best_h = h
    best_c5 = obj(h)
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Perturbations
        for _ in range(5):
            perturb = np.random.normal(0, 0.05, n_points)
            new_h = np.clip(h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else h
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
            if res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun

        # Structured perturbations
        for _ in range(3):
            if np.random.rand() < 0.5:
                # Random intervals
                h_rand = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
                h_rand = h_rand * target_sum / np.sum(h_rand)
                res = minimize(obj, h_rand, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
                if res.fun < best_c5:
                    best_h, best_c5 = res.x, res.fun
            else:
                # Symmetric pattern
                sym = np.zeros(n_points)
                sym[:n_points//2] = np.random.rand(n_points//2)
                sym = np.concatenate((sym, 1 - sym[:n_points//2][::-1]))
                sym = sym * target_sum / np.sum(sym)
                res = minimize(obj, sym, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
                if res.fun < best_c5:
                    best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
