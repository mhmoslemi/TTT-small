from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()

    h = np.clip(h, 0, 1)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

        eps = 1e-4 * np.max(x) + 1e-6
        grad = np.zeros_like(x)

            x_plus[i] += eps
            x_minus = x.copy()
            x_minus[i] -= eps
            corr_plus = np.correlate(x_plus, 1 - x_plus, mode='full')
            corr_minus = np.correlate(x_minus, 1 - x_minus, mode='full')
            grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        return grad

    def evaluate(x):

        return np.max(corr) * dx

    best_c5 = obj(h)
