import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 1e-12 else np.zeros(n_points)

    def obj(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 1e-12:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(best_h)

    # Global phase with differential evolution
    if time.time() - start_time < budget_s * 0.3:
        result = differential_evolution(obj, bounds, constraints=cons, tol=1e-8, maxiter=50, popsize=15, mutation=(0.5, 1.0), recombination=0.8, seed=seed)
        best_h, best_c5 = result.x, result.fun

    # Local refinement
    if time.time() - start_time < budget_s * 0.5:
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10, options={'maxiter': 100})
        best_h, best_c5 = res.x, res.fun

    # Multi-start perturbation and refinement
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.03, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 1e-12 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10, options={'maxiter': 50})
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Final refinement with tight tolerance
    result_lbgf = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10, options={'maxiter': 100})
