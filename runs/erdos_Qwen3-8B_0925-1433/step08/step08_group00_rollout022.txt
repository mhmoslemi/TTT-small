import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        initial_h = initial_h_values
    except NameError:
        initial_h = np.random.rand(45)
        initial_h = initial_h * 45 / 2 / np.sum(initial_h)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def eval_c5(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        return eval_c5(x)

    def de_population(n_pop=20):
        pop = [initial_h.copy()]
        for _ in range(n_pop - 1):
            perturb = np.random.normal(0, 0.05, n_points)
            new_h = initial_h + perturb
            new_h = np.clip(new_h, 0, 1)
            new_sum = np.sum(new_h)
            if new_sum > 0:
                new_h = new_h * target_sum / new_sum
            pop.append(new_h)
        return np.array(pop)

    population = de_population()
    best_x = initial_h.copy()
    best_c5 = eval_c5(best_x)

    start_time = time.time()
    F = 0.8
    CR = 0.9
    gen = 0
    while time.time() - start_time < budget_s - 200:
        if gen >= 100:
            break
