import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def generate_pattern(m):
        positions = np.linspace(0, 2, m + 1)[1:-1]
        h = np.zeros(n_points)
        for pos in positions:
            idx = int(pos / dx + 0.5)
            if 0 <= idx < n_points:
                h[idx] = 1
        h = h * target_sum / np.sum(h)
        return h

    structured = [generate_pattern(i) for i in [10, 15, 20]]
    structured.append(np.zeros(n_points))
    structured.append(np.ones(n_points))

    best_c5 = float('inf')
    best_h = h.copy()

    def evaluate(h):
        return minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7).fun

    for p in structured:
        res = evaluate(p)
        if res < best_c5:
            best_h, best_c5 = p, res

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        pert = np.random.normal(0, 0.1, n_points)
        new_h = np.clip(best_h + pert, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        res = evaluate(new_h)
        if res < best_c5:
            best_h, best_c5 = new_h, res

    return (best_h, best_c5, n_points)
