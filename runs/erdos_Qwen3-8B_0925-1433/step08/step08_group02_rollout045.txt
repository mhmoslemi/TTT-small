import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
    
    start_time = time.time()
    
    for _ in range(500):  # Max iterations
        if time.time() - start_time > budget_s:
            break
        
        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_corr = np.max(corr)
        if max_corr < best_c5:
            best_c5 = max_corr
            best_h = h.copy()
        
        # Find index of maximum correlation
        m = np.argmax(corr)
        # Compute lag k from the cross-correlation index
        k = m - (n_points - 1)
        
        # Compute gradient for lag k
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = (1 - h[j]) * dx
            else:
                grad[i] = 0.0
        
        # Perform gradient descent step
        learning_rate = 0.01
        h = h - learning_rate * grad
        
        # Clip and rescale to maintain constraints
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        else:
            h = np.zeros(n_points)
        
        # Periodically apply perturbations
        if _ % 50 == 0:
            # Sinusoidal perturbation
            perturb = np.sin(np.linspace(0, 2 * np.pi, n_points) * 2) * 0.05
            h = np.clip(h + perturb, 0, 1)
            h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h
        
        # Early termination if no improvement
        if np.allclose(h, best_h, atol=1e-6):
            break
    
    return (best_h, best_c5, n_points)
