import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h.copy()
    best_c5 = obj(h)
    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        best_h = result.x
        best_c5 = result.fun

        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (len(corr) // 2)
        overlap_start = max(0, int((k) * n_points / 2))
        overlap_end = min(n_points, int((k + 2) * n_points / 2))

        if n_points < 20:
            continue
        seg_len = max(1, int(n_points * 0.25))
        seg1 = np.random.choice(range(overlap_start, overlap_end - seg_len), 1)[0]
        seg2 = np.random.choice(range(n_points - seg_len), 1)[0]
        new_h = best_h.copy()
        new_h[seg1:seg1+seg_len], new_h[seg2:seg2+seg_len] = new_h[seg2:seg2+seg_len], new_h[seg1:seg1+seg_len]
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    return (best_h, best_c5, n_points)
