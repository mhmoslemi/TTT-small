import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        max_k = np.argmax(corr)

        # Compute gradient for the current max lag
        grad = np.zeros_like(h)
        k = max_k
        for j in range(n_points):
            if (j + k) < n_points:
                grad[j] += (1 - h[j + k]) * dx
            if (j - k) >= 0:
                grad[j] -= h[j - k] * dx

        # Update h with gradient descent step
        step_size = 0.1
        h = h - step_size * grad
        h = np.clip(h, 0, 1)
        # Rescale to target_sum
        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h

        # Check if this improves the best_c5
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_max = np.max(new_corr)
        if new_max < best_c5:
            best_c5 = new_max
            best_h = h.copy()

        # Guided perturbation
        if _ % 50 == 0:
            # Add a sinusoidal perturbation with increasing frequency
            perturbation = np.sin(2 * np.pi * np.linspace(0, 1, n_points)) * 0.05

            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h

    return (best_h, best_c5, n_points)
