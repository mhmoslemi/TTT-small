import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    x = np.linspace(0, 2, n_points)

    # Generate varied initial guesses
    initial_guesses = [
        initial_h_values,
        np.clip(initial_h_values + 0.1 * np.sin(2 * np.pi * x), 0, 1),
        np.clip(initial_h_values + 0.1 * np.sin(4 * np.pi * x), 0, 1),
        np.clip(initial_h_values + 0.1 * np.sin(6 * np.pi * x), 0, 1),
        np.clip(initial_h_values * np.sin(2 * np.pi * x), 0, 1),

    ]

    best_h = initial_h_values.copy()

        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
