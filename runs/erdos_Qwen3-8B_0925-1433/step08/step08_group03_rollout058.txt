import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    generations = 200
    mutation_std = 0.1
    elite_ratio = 0.2

    for gen in range(generations):
