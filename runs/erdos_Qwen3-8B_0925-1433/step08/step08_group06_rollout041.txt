    # Rescale initial guess
    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)

    def obj(x):

        return np.max(corr) * dx

    # Constraints

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        # SLSQP optimization
        res = minimize(obj, best_h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      tol=1e-8)
        if res.success:
            best_h = res.x

            break

        # Apply structured perturbations
        corr = np.correlate(best_h, 1 - best_h, mode='full')
