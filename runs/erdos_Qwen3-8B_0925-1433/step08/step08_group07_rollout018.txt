import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    import numpy as np
    from scipy.optimize import minimize
    import time

    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h from initial_h_values
    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        c5 = np.max(corr) * dx
        tv = np.sum(np.abs(np.diff(x)))
        return c5 + 0.1 * tv

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(best_h)
    λ = 0.1
    no_improvement = 0

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Generate perturbed candidate
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
        res = minimize(obj, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-10)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h, best_c5 = res.x, current_c5
                no_improvement = 0
            else:
                no_improvement += 1
                if no_improvement >= 5:
                    λ *= 1.1
        else:
            no_improvement += 1
            if no_improvement >= 5:
                λ *= 1.1

    return (best_h, best_c5, n_points)
