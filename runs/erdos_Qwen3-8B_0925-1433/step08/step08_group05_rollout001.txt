import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def feasible(x):
        return np.sum(x) - target_sum

    h_base = np.clip(initial_h_values, 0, 1)
    h_base = h_base * target_sum / np.sum(h_base) if np.sum(h_base) > 0 else np.zeros(n_points)

    best_c5 = float('inf')
    best_h = h_base

    start_time = time.time()

    # Initial perturbations
    for _ in range(5):
        freq = np.random.choice([1, 2, 3, 4, 5])
        amp = np.random.uniform(0.05, 0.2)
        x = np.linspace(0, 2 * np.pi, n_points)
        pert = amp * np.sin(freq * x)
        h_pert = h_base + pert
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * target_sum / np.sum(h_pert)
        res = minimize(objective, h_pert, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': feasible}])
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    # Refinement
    for _ in range(15):
        if time.time() - start_time > budget_s:
            break
        h_refine = best_h + 0.05 * np.random.randn(n_points)
        h_refine = np.clip(h_refine, 0, 1)
        h_refine = h_refine * target_sum / np.sum(h_refine)
        res = minimize(objective, h_refine, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': feasible}])
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x

    return (best_h, best_c5, n_points)
