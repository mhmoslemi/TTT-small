import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    if np.sum(h) != target_sum:
        h *= target_sum / np.sum(h)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    best_h, best_c5 = result.x, result.fun
    start_time = time.time()

    for _ in range(80):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        if lag < 0:
            lag += 2 * n_points
        start_idx = 0
        end_idx = n_points - lag - 1
        if end_idx < start_idx:
            end_idx = n_points - 1
        seg_start = max(0, start_idx)
        seg_end = min(n_points - 1, end_idx)
        if seg_end - seg_start > 0:
            seg_len = int((seg_end - seg_start) * 0.2)
            seg1 = np.random.randint(seg_start, seg_end - seg_len + 1)
            seg2 = np.random.randint(seg_start, seg_end - seg_len + 1)
            new_h = best_h.copy()
            new_h[seg1:seg1+seg_len], new_h[seg2:seg2+seg_len] = new_h[seg2:seg2+seg_len], new_h[seg1:seg1+seg_len]
            new_h = np.clip(new_h, 0, 1)
            if np.sum(new_h) != target_sum:
                new_h *= target_sum / np.sum(new_h)
            res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.fun < best_c5:
                best_h, best_c5 = res.x, res.fun
    return (best_h, best_c5, n_points)
