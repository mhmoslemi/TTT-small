import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    result = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    best_h, best_c5 = result.x, result.fun

    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
        max_corr = np.max(corr)
        if max_corr <= best_c5:
            break
        best_c5 = max_corr
        max_idx = np.argmax(corr)
        k = max_idx - n_points + 1
        grad = np.zeros(n_points)

        h = h * target_sum / np.sum(h) if np.sum(h) > 0 else best_h
