import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)

    # Frequency domain filtering
    fft_h = np.fft.fft(h)
    cutoff = int(0.2 * n_points)
    fft_h[cutoff:] = 0
    h_filtered = np.fft.ifft(fft_h).real
    h_filtered = np.clip(h_filtered, 0, 1)
    h_filtered = h_filtered * target_sum / np.sum(h_filtered) if np.sum(h_filtered) > 0 else h_filtered

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def grad(x):
        corr = np.correlate(x, 1 - x, mode='full')
        k = np.argmax(corr)
        grad = np.zeros_like(x)
        for i in range(n_points):
            x_plus = x.copy()
            x_plus[i] += 1e-5
            corr_plus = np.correlate(x_plus, 1 - x_plus, mode='full')
            grad[i] = (np.max(corr_plus) - objective(x)) / 1e-5
        return grad

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    best_h = h_filtered
    best_c5 = objective(best_h)
    start_time = time.time()

    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B',
                       bounds=bounds, constraints=cons,
                       jac=grad, tol=1e-6)
        if res.success:
            best_h = res.x
            best_c5 = res.fun
            corr = np.correlate(best_h, 1 - best_h, mode='full')
            max_k = np.argmax(corr)
            if max_k < n_points - 1:
                freq = 2 * np.pi * max_k / n_points
                perturb = 0.01 * np.sin(freq * np.arange(n_points))
                new_h = np.clip(best_h + perturb, 0, 1)
                new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
                res_perturb = minimize(objective, new_h, method='L-BFGS-B',
                                    bounds=bounds, constraints=cons,
                                    jac=grad, tol=1e-6)
                if res_perturb.success and res_perturb.fun < best_c5:
                    best_h = res_perturb.x
                    best_c5 = res_perturb.fun
        else:
            break

    return (best_h, best_c5, n_points)
