import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    half = (n_points + 1) // 2

    def build_h(x):
        return np.concatenate([x, 1 - x[::-1]])

    def obj(x):
        h = build_h(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * half
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum / 2}]

    def eval_sol(x):
        return obj(x), build_h(x)

    # Initial guess from parent symmetrically constructed
    x0 = initial_h_values[:half]
    x0 = np.clip(x0 * target_sum / np.sum(x0) if np.sum(x0) > 0 else x0, 0, 1)

    start_time = time.time()
    result = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    best_x, best_c5 = result.x, result.fun

    # Refine with perturbations and structured patterns
    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, half)
        new_x = np.clip(best_x + perturb, 0, 1)
        new_x = new_x * target_sum / 2.0 / np.sum(new_x) if np.sum(new_x) > 0 else best_x
        res = minimize(obj, new_x, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
        if res.fun < best_c5:
            best_x, best_c5 = res.x, res.fun

    # Add sinusoidal-based pattern
    sin_pat = np.sin(np.linspace(0, np.pi, half)) * 0.5 + 0.5
    sin_pat = np.clip(sin_pat, 0, 1) * target_sum / 2.0 / np.sum(sin_pat)
    res = minimize(obj, sin_pat, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if res.fun < best_c5:

    h_full = build_h(best_x)
    return (h_full, best_c5, n_points)
