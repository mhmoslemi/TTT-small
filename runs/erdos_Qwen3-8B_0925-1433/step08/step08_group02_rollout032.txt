            best_c5 = current_c5
            best_h = h.copy()

        def objective(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return corr[max_lag] * dx

        def grad(x):
            grad = np.zeros(n_points)
            for i in range(n_points):
                x_plus = x.copy()
                x_plus[i] += 1e-4
                x_plus = np.clip(x_plus, 0, 1)
                x_plus = x_plus * target_sum / np.sum(x_plus) if np.sum(x_plus) > 0 else x_plus
                corr = np.correlate(x_plus, 1 - x_plus, mode='full')
                val_plus = corr[max_lag]
                x_minus = x.copy()
                x_minus[i] -= 1e-4
                x_minus = np.clip(x_minus, 0, 1)
                x_minus = x_minus * target_sum / np.sum(x_minus) if np.sum(x_minus) > 0 else x_minus
                corr = np.correlate(x_minus, 1 - x_minus, mode='full')
                val_minus = corr[max_lag]
                grad[i] = (val_plus - val_minus) / (2e-4)
            return grad

        res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, jac=grad, constraints=cons, tol=1e-8)
        if res.success:
            h = res.x
        else:
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h

        if _ % 50 == 0:
            corr = np.correlate(h, 1 - h, mode='full')
            max_lag = np.argmax(corr)
            freq = 2 * np.pi / n_points * 2
            perturb = np.sin(np.linspace(0, 2 * np.pi, n_points) * freq) * 0.05
            h = np.clip(h + perturb, 0, 1)
            h = h * target_sum / np.sum(h) if np.sum(h) > 0 else h

    return (best_h, best_c5, n_points)
