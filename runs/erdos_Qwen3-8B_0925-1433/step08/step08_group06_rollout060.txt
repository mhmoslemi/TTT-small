import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)
    h = np.clip(h, 0, 1)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    H = np.fft.fft(h)
    cutoff_freq = int(0.1 * n_points)
    filter = np.zeros_like(H)
    filter[:cutoff_freq] = 1.0
    H_filtered = H * filter
    h_filtered = np.fft.ifft(H_filtered).real
    h_filtered = np.clip(h_filtered, 0, 1)
    if np.sum(h_filtered) > 0:
        h_filtered = h_filtered * target_sum / np.sum(h_filtered)
    else:
        h_filtered = h_filtered

    start_time = time.time()
    best_h = h_filtered
    best_c5 = obj(best_h)

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})
            if res.success:
                best_h = res.x
                best_c5 = res.fun
        except:
            pass
        if np.sum(best_h) > 0:
            best_h = best_h * target_sum / np.sum(best_h)
        else:
            best_h = np.zeros(n_points)
        best_h = np.clip(best_h, 0, 1)
        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, n_points)
