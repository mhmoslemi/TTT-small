import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def compute_autocorr(h):
        autocorr = np.correlate(h, h, mode='full') * dx
        return autocorr

    def compute_gradient(h, k, dx, n):
        grad = np.zeros(n)
        for m in range(n):
            val_left = h[m - k] if (m - k) >= 0 else 0
            val_right = h[m + k] if (m + k) < n else 0
            grad[m] = dx * (val_left + val_right)
        return grad

    def objective_for_SLSQP(x):
        autocorr = np.correlate(x, x, mode='full') * dx
        return 1 - np.min(autocorr)

    best_h = h.copy()
    best_c5 = float('inf')

    start_time = time.time()
    steps = 0
    step_size = 0.05
    max_steps = 1000

    while steps < max_steps and time.time() - start_time < budget_s:
        autocorr = compute_autocorr(h)
        min_idx = np.argmin(autocorr)
        lag_k = min_idx - (n_points - 1)
        grad = compute_gradient(h, lag_k, dx, n_points)
        h += step_size * grad
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        steps += 1

    autocorr = compute_autocorr(h)
    best_c5 = 1 - np.min(autocorr)
    best_h = h.copy()

    result = minimize(objective_for_SLSQP, best_h, method='SLSQP', bounds=[(0, 1)] * n_points,
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                      tol=1e-7, options={'maxiter': 100})

    best_h = result.x
    best_c5 = result.fun

    return (best_h, best_c5, n_points)
