import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial guess
    h = initial_h_values * target_sum / np.sum(initial_h_values) if np.sum(initial_h_values) > 0 else np.zeros(n_points)

    h_fft = np.fft.fft(h)

    h_fft[f_c:] = 0.0
    h_filtered = np.fft.ifft(h_fft).real
    h_filtered = np.clip(h_filtered, 0, 1)
    h_filtered = h_filtered * target_sum / np.sum(h_filtered) if np.sum(h_filtered) > 0 else h_filtered

    # Objective function
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    # Constraint
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    best_h = h_filtered
    best_c5 = obj(best_h)
    start_time = time.time()

    # Optimization with finite difference Jacobian
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        def jac(x):
            eps = 1e-6
            grad = np.zeros_like(x)
            for i in range(n_points):
                x_plus = np.copy(x)
                x_plus[i] += eps
                corr_plus = np.correlate(x_plus, 1 - x_plus, mode='full')
                grad[i] = (np.max(corr_plus) - obj(x)) / eps
