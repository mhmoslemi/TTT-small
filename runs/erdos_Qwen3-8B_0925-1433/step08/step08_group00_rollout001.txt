import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_norm(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        return h

    h = clip_norm(initial_h_values)

    def obj(x):
        h_ = clip_norm(x)
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h
    best_c5 = obj(h)

    # Global exploration
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Cauchy perturbation
        noise = np.random.standard_cauchy(n_points) * 0.05
        new_h = np.clip(best_h + noise, 0, 1)
        new_h = clip_norm(new_h)
        res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-8)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    # Local refinement
    for _ in range(80):

    # Final refinement
