import numpy as np
from scipy.optimize import minimize, COBYLA, L_BFGS_B
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_norm(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) > 0:
            x = x * target_sum / np.sum(x)
        return x

    h = clip_norm(initial_h_values.copy())

    def obj(x):
        x = clip_norm(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = obj(h)

    # Global exploration with COBYLA
    for _ in range(30):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = h + perturb
        new_h = clip_norm(new_h)
        res = minimize(obj, new_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
        h = res.x
        if res.success:
            h = res.x

    # Local refinement with L-BFGS-B
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.01, n_points)
        new_h = h + perturb
        new_h = clip_norm(new_h)
        res = minimize(obj, new_h, method='L_BFGS_B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
        h = res.x

    # Final refinement with SLSQP
    for _ in range(20):
        if time.time() - start_time > budget_s:

        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)
        if res.fun < best_c5:
