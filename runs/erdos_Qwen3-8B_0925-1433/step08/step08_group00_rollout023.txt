import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    
    def objective(x):
        h_ = x.copy()
        h_ = np.clip(h_, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0)] * n_points
    
    start_time = time.time()
    
    # Run DE for global optimization
    result_de = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=20, tol=1e-8, mutation=(0.5, 1), 
        recombination=0.9, seed=seed, maxiter=200
    )
    best_h_de = result_de.x
    best_c5_de = result_de.fun
    spent_time = time.time() - start_time
    
    # Refine with L-BFGS-B
    def objective_lbgfs(x):
        h_ = np.clip(x, 0, 1)
        sum_h = np.sum(h_)
        if sum_h > 0:
            h_ = h_ * target_sum / sum_h
        corr = np.correlate(h_, 1 - h_, mode='full')
        return np.max(corr) * dx
    
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds_lbgfs = [(0.0, 1.0)] * n_points
    
    # First refinement
    result_lbgfs = minimize(
        objective_lbgfs, best_h_de, method='L-BFGS-B', bounds=bounds_lbgfs, constraints=cons, 
        tol=1e-8, options={'maxiter': 100}
    )
    best_h = result_lbgfs.x
    best_c5 = result_lbgfs.fun
    spent_time += time.time() - start_time

    final_c5 = np.max(np.correlate(final_h, 1 - final_h, mode='full')) * dx
    
    return (final_h, final_c5, n_points)
