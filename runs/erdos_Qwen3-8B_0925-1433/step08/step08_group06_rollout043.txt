from scipy.optimize import minimize
import time

    n_points = len(initial_h_values)

    h_freq = h_freq * target_sum / np.sum(h_freq) if np.sum(h_freq) > 0 else h

    # Optimization setup
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

        corr = np.correlate(x, 1 - x, mode='full')
