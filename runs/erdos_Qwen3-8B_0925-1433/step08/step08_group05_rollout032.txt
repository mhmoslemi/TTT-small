from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.clip(initial_h_values, 0, 1)
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def perturb(x, freqs, amps):

        for f, a in zip(freqs, amps):
            pert += a * np.sin(2 * np.pi * f * np.linspace(0, 2, n_points))
        return np.clip(x + pert, 0, 1) * target_sum / np.sum(np.clip(x + pert, 0, 1))

    def refine(x):
        res = minimize(objective, x, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 200})
        return res.x

    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    for _ in range(8):
        if time.time() - start_time > budget_s:

        new_h = perturb(best_h, freqs, amps)
        res = minimize(objective, new_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 150})
        if res.success and res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun

    for _ in range(50):
