import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)

    # Frequency domain filtering
    H = np.fft.fft(h)
    f_c = int(0.1 * n_points)
    H_filtered = H.copy()
    H_filtered[f_c:] = 0
    h_filtered = np.fft.ifft(H_filtered).real
    h_filtered = np.clip(h_filtered, 0, 1)
    h_filtered = h_filtered * target_sum / np.sum(h_filtered) if np.sum(h_filtered) > 0 else h_filtered

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0., 1.)] * n_points

    best_h = h_filtered
    best_c5 = obj(best_h)
    start_time = time.time()

    # Optimization
    res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8})
    if res.success:

        best_c5 = res.fun
    else:

        best_c5 = obj(best_h)

    # Final clamping and rescaling
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h) if np.sum(best_h) > 0 else best_h
