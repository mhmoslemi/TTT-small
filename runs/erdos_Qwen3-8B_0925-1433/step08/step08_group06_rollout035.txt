import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial guess
    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)

    # Frequency domain shaping
    def apply_lowpass(h, cutoff_ratio):
        H = fft(h)
        n = len(H)
        freqs = np.fft.rfftfreq(n, d=dx)
        H_filtered = H.copy()
        H_filtered[np.abs(freqs) > cutoff_ratio] = 0
        h_filtered = np.real(ifft(H_filtered))
        h_filtered = np.clip(h_filtered, 0, 1)
        if np.sum(h_filtered) > 0:
            h_filtered = h_filtered * target_sum / np.sum(h_filtered)
        return h_filtered

    # Apply low-pass filter with dynamic cutoff
    h = apply_lowpass(h, 0.3)

    # Objective function
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    # Constraints
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    # Optimization with L-BFGS-B
    max_iter = 150
    best_c5 = obj(h)
    best_h = h.copy()

    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, maxiter=50)
        if res.success:
            if res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x
            h = res.x
        else:
            break

    return (best_h, best_c5, n_points)
