from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    if initial_h_values is None:
        n_points = 45
        h = np.zeros(n_points)
        for i in range(n_points):
            j = n_points - 1 - i
            if i <= j:
                h[i] = 1 if i < 22 else 0
                h[j] = 1 - h[i]
        if n_points % 2:
            h[n_points//2] = 0.5
    else:
        n_points = len(initial_h_values)
        h = np.zeros(n_points)
        for i in range(n_points):
            j = n_points - 1 - i
            if i <= j:
                h[i] = initial_h_values[i]
                h[j] = 1 - initial_h_values[i]
        if n_points % 2:
            h[n_points//2] = 0.5
    dx = 2.0 / n_points
    best_h = h
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        if n_points % 2:
            mid = n_points // 2
            pert = np.random.normal(0, 0.05, mid)
        else:
            pert = np.random.normal(0, 0.05, n_points//2)
        new_h = np.zeros(n_points)
        for i in range(n_points):
            j = n_points - 1 - i
            if i <= j:
                new_h[i] = max(0, min(1, h[i] + pert[i] if i < n_points//2 else h[i]))

        if n_points % 2:
