from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if n_points == 0:
        h = np.array([])
    else:
        h = initial_h_values.copy()
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        else:
            h = np.zeros(n_points)
        h = np.clip(h, 0, 1)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h
    best_c5 = obj(h)
    start_time = time.time()

    def generate_initial_guesses(base):
        guesses = []
        if isinstance(base, np.ndarray):
            guesses.append(base.copy())

            guesses.append(perm)

        noise = np.random.normal(0, 0.1, n_points)
        guesses.append(np.clip(base + noise, 0, 1) * target_sum / np.sum(np.clip(base + noise, 0, 1)))
        return guesses

    initial_guesses = generate_initial_guesses(h)

                best_h, best_c5 = result.x, result.fun
