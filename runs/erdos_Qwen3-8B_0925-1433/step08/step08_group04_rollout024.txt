import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    np.random.seed(seed)
    n_points = len(initial_h_values) if initial_h_values is not None else 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    if n_points == 0:
        return np.array([]), 0.0, 0
    h = initial_h_values.copy() if initial_h_values is not None else np.zeros(n_points)
    h[:n_points//2] = 1.0
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = Bounds(0, 1)
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    best_h = h
    best_c5 = obj(h)
    start_time = time.time()

    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            curr_h = result.x
            curr_c5 = result.fun
            if curr_c5 < best_c5:
                best_h, best_c5 = curr_h, curr_c5
        corr = np.correlate(best_h, 1 - best_h, mode='full')

            perturb = 0.05 * np.sin(2 * np.pi * max_lag * np.arange(n_points) / n_points)
            new_h = best_h + perturb
