import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Rescale initial guess
    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def jac(x, k):
        # Central finite difference with adaptive step
        eps = 1e-4 * np.max(x) + 1e-8
        grad = np.zeros_like(x)
        for i in range(n_points):
            x_plus = np.copy(x)
            x_minus = np.copy(x)
            x_plus[i] += eps
            x_minus[i] -= eps
            corr_plus = np.correlate(x_plus, 1 - x_plus, mode='full')
            corr_minus = np.correlate(x_minus, 1 - x_minus, mode='full')
            grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
        return grad

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    best_h = h
    best_c5 = obj(h)
    start_time = time.time()
    improvement = 1e-5
    iter_count = 0

    while time.time() - start_time < budget_s:
        iter_count += 1
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)

        # Update only for the worst lag to focus optimization
        def sub_obj(x):
            return np.abs(np.correlate(x, 1 - x, mode='full')[max_k]) * dx

        def sub_jac(x):
            return jac(x, max_k)

        res = minimize(sub_obj, best_h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      jac=sub_jac, tol=1e-7)
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun
        else:
            # Try structured sinusoidal modulation
            freq = 2 * np.pi * max_k / n_points
            perturb = 0.005 * np.sin(freq * np.arange(n_points))
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            res_perturb = minimize(obj, new_h, method='SLSQP',
                                bounds=bounds, constraints=cons,

            if res_perturb.success and res_perturb.fun < best_c5:
