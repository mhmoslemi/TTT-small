    bounds = [(0, 1)] * n_points

    best_h = h
    best_c5, _ = conv_max(h)

    start_time = time.time()
    iterations = 0
    while time.time() - start_time < budget_s:
        iterations += 1
        if iterations > 50:
            break
        curr_c5, max_m = conv_max(best_h)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
        else:
            perturb = 0.01 * np.sin(2 * np.pi * max_m / n_points * np.arange(n_points))
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            best_h = new_h
            continue

        cons_fun = lambda x: np.sum(x) - target_sum
        jac_cons = lambda x: np.ones_like(x)

        try:
            res = minimize(obj, best_h, method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': jac_cons}],
                           jac=grad_analytical(best_h, max_m), tol=1e-8)
            if res.success:
                best_h = res.x
            else:
                perturb = 0.01 * np.random.randn(n_points)
                new_h = np.clip(best_h + perturb, 0, 1)
                new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
                best_h = new_h
        except:
            perturb = 0.01 * np.random.randn(n_points)
            new_h = np.clip(best_h + perturb, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h) if np.sum(new_h) > 0 else best_h
            best_h = new_h

    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    return (best_h, best_c5, n_points)
