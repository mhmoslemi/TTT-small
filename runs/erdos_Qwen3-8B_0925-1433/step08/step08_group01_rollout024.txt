            seg2 = np.random.choice(n_points - seg_len, size=1)[0]
        new_h = best_h.copy()
        new_h[seg1:seg1+seg_len], new_h[seg2:seg2+seg_len] = new_h[seg2:seg2+seg_len], new_h[seg1:seg1+seg_len]
        total_new = np.sum(new_h)
        if total_new > 0:
            new_h = new_h * target_sum / total_new
        else:
            new_h = best_h
        res = minimize(obj, new_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, maxiter=50)
        if res.fun < best_c5:
            best_h, best_c5 = res.x, res.fun
    
    return (best_h, best_c5, n_points)
