def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    sum_initial = np.sum(initial_h_values)
    if sum_initial == 0:
        h = np.zeros(n_points)
    else:
        h = initial_h_values * (target_sum / sum_initial)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h) if np.sum(h) > 0 else 0)
    
    low = 0.3
    high = 0.4
    best_c5 = high
    best_h = h.copy()
    start_time = time.time()
    
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        mid = (low + high) / 2
        
        def feasibility_check(C, h_current):
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            lags = np.arange(-n_points + 1, n_points)
            valid_lags = lags[(corr >= 0) & (np.isfinite(corr))]
            top_lags = np.argsort(corr)[::-1][:20]
            top_lags = np.unique(top_lags)
            top_lags = np.concatenate((top_lags, -top_lags))
            valid_lags = np.setdiff1d(top_lags, [0])
            valid_lags = valid_lags[(valid_lags >= -n_points + 1) & (valid_lags < n_points)]
            
            constraints = []
            for k in valid_lags:
                idx = n_points - 1 + k
                if 0 <= idx < len(corr):
                    def constraint(x, k=k, C=C):
                        x = np.array(x)
                        corr_val = np.correlate(x, 1 - x, mode='full')[idx]
                        return corr_val - C
                    constraints.append({'type': 'ineq', 'fun': constraint})
            
            res = minimize(
                lambda x: 0,
                h_current,
                method='SLSQP',
                bounds=[(0, 1)] * n_points,
                constraints=[
                    {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                    {'type': 'ineq', 'fun': constraint} for constraint in constraints
                ],
                options={'maxiter': 100, 'ftol': 1e-6}
            )
            return res.success, res.x
        
        feasible, new_h = feasibility_check(mid, h)
        if feasible:
            h = new_h
            high = mid
            best_c5 = mid
        else:
            low = mid
    
    return (h, best_c5, n_points)
