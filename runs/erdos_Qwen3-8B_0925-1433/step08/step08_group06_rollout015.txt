import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = initial_h_values.copy()
    h = h * target_sum / np.sum(h) if np.sum(h) > 0 else np.zeros(n_points)
    h = np.clip(h, 0, 1)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def jac_obj(x, k):
        eps = 1e-6
        grad = np.zeros_like(x)
        for i in range(n_points):
            x_plus = np.copy(x)
            x_plus[i] += eps
            corr_plus = np.correlate(x_plus, 1 - x_plus, mode='full')
            grad[i] = (np.max(corr_plus) - obj(x)) / eps
        return grad

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0, 1)] * n_points

    m = n_points // 4 + 1
    l = 1.0 / m
    centers = np.linspace(0.5, 2 - 0.5, m)
    h_structured = np.zeros(n_points)
    for i in range(m):
        center = centers[i]
        start = center - l / 2
        end = center + l / 2
        start_idx = int(np.round(start / dx))
        end_idx = int(np.round(end / dx))
        start_idx = max(0, min(start_idx, n_points - 1))
        end_idx = max(0, min(end_idx, n_points - 1))
        if start_idx <= end_idx:
            h_structured[start_idx:end_idx+1] = 1.0

    h_init = h_structured * target_sum / np.sum(h_structured) if np.sum(h_structured) > 0 else h
    h_init = np.clip(h_init, 0, 1)

    best_h = h_init
    best_c5 = obj(h_init)
    start_time = time.time()

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        k = np.argmax(np.correlate(best_h, 1 - best_h, mode='full'))
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, jac=lambda x: jac_obj(x, k), tol=1e-8)
        if res.success:

        max_k = np.argmax(corr)

            perturb = 0.01 * np.sin(freq * np.arange(n_points))
