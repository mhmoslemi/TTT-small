import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_idx = np.argmax(corr)
        k = k_max_idx - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    def generate_candidates(base):
        yield base * (target_sum / np.sum(base))
        yield np.clip(base + np.random.normal(0, 0.2, n_points), 0, 1) * (target_sum / np.sum(np.clip(base + np.random.normal(0, 0.2, n_points), 0, 1)))
        yield base * (1 + 0.5 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)))
        yield base[np.random.permutation(n_points)] * (target_sum / np.sum(base[np.random.permutation(n_points)]))
        yield base * (1 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n_points)))

    initial_guesses = []
    for _ in range(7):

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:

        result = differential_evolution(
            objective, bounds=[(0, 1)] * n_points, strategy='best2bin',

        res_slsqp = minimize(
