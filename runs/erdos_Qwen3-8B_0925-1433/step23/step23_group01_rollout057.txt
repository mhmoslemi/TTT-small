import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guesses = [
        np.array(initial_h_values, dtype=float),
        np.array(initial_h_values, dtype=float),
        0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points)),
        np.random.rand(n_points),
        1.0 - np.array(initial_h_values, dtype=float)
    ]
    for i in range(1, len(initial_guesses)):
        initial_guesses[i] = normalize(initial_guesses[i])
    flip_mask = np.random.rand(n_points) < 0.3
    initial_guesses[1][flip_mask] = 1.0 - initial_guesses[1][flip_mask]
    initial_guesses[1] = normalize(initial_guesses[1])

    best_h = normalize(np.array(initial_h_values, dtype=float))
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    def objective(x):
        h = normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = differential_evolution(objective, bounds, strategy='best1bin',
                                     popsize=15, mutation=(0.5, 1.0),
                                     recombination=0.9, maxiter=100, tol=1e-6,
                                     polish=True, disp=False)
        if res.success:
            h = normalize(res.x)
            c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='trust-constr',
                       bounds=bounds, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       tol=1e-6, options={'maxiter': 100})
        if res.success:
            h = normalize(res.x)
            c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    return (np.clip(best_h, 0, 1), best_c5, n_points)
