import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}

    best_h = np.array(initial_h, dtype=float)
    best_h = normalize(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()

    for _ in range(4):
        if time.time() - start_time > budget_s:
            break
        np.random.seed(seed + _)
        m = n_points // 2
        pos = np.linspace(0, 2, m + 1)[:-1]
        h_struc = np.zeros(n_points)
        for p in pos:
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - p))
            h_struc[idx] = 1
        h_struc = normalize(h_struc)
        res = minimize(obj, h_struc, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
