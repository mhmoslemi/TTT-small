import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values) if 'initial_h_values' in locals() else 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(x):
        h = normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    def create_initial_population():
        pop = []
        # Parent
        pop.append(normalize(np.array(initial_h_values, dtype=float)))
        # Binary
        b = np.zeros(n_points)
        b[n_points//2:] = 1.0
        pop.append(normalize(b))
        # Sinusoidal
        s = 0.5 * (1 + np.sin(2 * np.pi * np.linspace(0, 1, n_points)))
        pop.append(normalize(s))
        # Random
        r = normalize(np.random.rand(n_points))
        pop.append(r)
        # Random cluster
        c = np.random.rand(n_points)
        c = (c < 0.5).astype(float)
        pop.append(normalize(c))
        return pop

    pop = create_initial_population()
    best_h = pop[0].copy()
    best_c5 = evaluate_c5(best_h)

    def cross(parent1, parent2):
        child = np.random.choice([parent1, parent2], size=n_points)
        return normalize(child)

    def mutate(x):
        x = x.copy()
        for _ in range(5):
            idx = np.random.randint(n_points)
            x[idx] += np.random.normal(0, 0.1)
            x = np.clip(x, 0, 1)
        return normalize(x)

    start_time = time.time()
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        # Evaluate
        scores = [evaluate_c5(ind) for ind in pop]
        idxs = np.argsort(scores)
        top = [pop[i] for i in idxs[:10]]
        # Crossover
        new_pop = []
        for i in range(0, len(top), 2):
            new_pop.append(cross(top[i], top[i+1] if i+1 < len(top) else top[i]))
        new_pop += [mutate(ind) for ind in top]
        pop = new_pop[:50]
        # Update best
        for h in pop:
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_h, best_c5 = h.copy(), c5

    # Local optimization
    def obj(x):
        return evaluate_c5(x)

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='trust-constr', bounds=[(0,1)*n_points], constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        if res.success:
            best_h = normalize(res.x)
            best_c5 = evaluate_c5(best_h)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
