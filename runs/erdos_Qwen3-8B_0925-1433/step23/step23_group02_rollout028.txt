import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        return evaluate_c5(x)

    def constraints(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraints, 'jac': np.ones(n_points)}

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = evaluate_c5(best_h)

    def mutate(h, mut_type):
        h = h.copy()
        if mut_type == 0:
            idx = np.random.choice(n_points, 5, replace=False)
            h[idx] += np.random.uniform(-0.2, 0.2, 5)
        elif mut_type == 1:
            h += np.random.normal(0, 0.1, n_points)
        elif mut_type == 2:
            h += 0.2 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))
        h = normalize(h)
        return h

    def crossover(a, b):
        return np.where(np.random.rand(n_points) < 0.5, a, b)

    population = [normalize(np.random.rand(n_points)) for _ in range(100)]
    population += [normalize(initial_guess), normalize(0.5 * (1 + np.sin(2 * np.pi * np.linspace(0, 1, n_points))))]
    for _ in range(20):
        if time.time() - time.time() > budget_s:
            break
        scores = [evaluate_c5(ind) for ind in population]
        elite = [ind for _, ind in sorted(zip(scores, population))[:10]]
        new_pop = elite
        while len(new_pop) < 100:
            a, b = np.random.choice(elite, 2, replace=False)
            child = crossover(a, b)
            child = mutate(child, np.random.randint(3))
            new_pop.append(child)
        population = new_pop

    for ind in population:
        if evaluate_c5(ind) < best_c5:
            best_h, best_c5 = ind, evaluate_c5(ind)

    for _ in range(5):
