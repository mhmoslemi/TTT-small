import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': lambda x: np.ones_like(x)}

    def obj(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    start_time = time.time()
    initial_guesses = [initial_guess]
    x = np.linspace(0, 2, n_points)
    initial_guesses.append(normalize(0.5 + 0.3 * np.sin(2 * np.pi * x)))

    binary_pattern[:n_points//2] = 0.0
    binary_pattern[n_points//2:] = 1.0
    initial_guesses.append(normalize(binary_pattern))
    initial_guesses.append(normalize(np.random.rand(n_points)))
    initial_guesses.append(normalize(0.5 + 0.3 * np.cos(4 * np.pi * x)))

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:

            h = normalize(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    for _ in range(5):
