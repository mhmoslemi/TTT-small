    h_current = best_h.copy()

    best_c5 = np.max([np.dot(h, np.roll(1 - h, k)) * dx for k in range(n_points)])
    best_h = h.copy()
    
    def compute_c5(h):
        overlaps = [np.dot(h, np.roll(1 - h, k)) * dx for k in range(n_points)]
        return np.max(overlaps)
    
    best_c5 = compute_c5(best_h)
    h_current = best_h.copy()
    learning_rate = 0.1
    steps = 0
    time_start = time.time()
    
    while time.time() - time_start < budget_s:
        overlaps = [np.dot(h_current, np.roll(1 - h_current, k)) * dx for k in range(n_points)]
        k_max = np.argmax(overlaps)
        gradient = 1.0 - np.roll(1.0 - h_current, k_max)
        h_new = h_current - learning_rate * gradient
        h_new = np.clip(h_new, 0, 1)
        sum_h = np.sum(h_new)
        if sum_h != 0:
            h_new = h_new * (target_sum / sum_h)
        h_current = h_new
        new_c5 = compute_c5(h_current)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_current.copy()
        if steps % 100 == 0 and learning_rate > 1e-5:
            learning_rate *= 0.95
        if steps % 500 == 0:
            noise = np.random.normal(0, 0.05, n_points)
            h_current += noise
            h_current = np.clip(h_current, 0, 1)
            sum_h = np.sum(h_current)
            if sum_h != 0:
                h_current = h_current * (target_sum / sum_h)
        if steps > 10000:
            break
        steps += 1
    
    return (best_h, best_c5, n_points)
