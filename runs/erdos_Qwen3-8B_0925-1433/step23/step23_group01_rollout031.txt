import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    def smooth_objective(h, k):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        exp_terms = np.exp(-k * corr)
        return (1.0 / k) * np.log(np.sum(exp_terms))

    def smooth_gradient(h, k, eps=1e-5):
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            grad[i] = (smooth_objective(h_plus, k) - smooth_objective(h_minus, k)) / (2 * eps)
        return grad

    def objective(x, k):
        return smooth_objective(x, k)

    def gradient(x, k):
        return smooth_gradient(x, k)

    structured_guesses = []
    x = np.linspace(0, 2, n_points)
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_guess = np.clip(sine_guess, 0, 1)
    sine_guess = normalize(sine_guess)
    structured_guesses.append(sine_guess)
    uniform_guess = np.random.rand(n_points)
    uniform_guess = normalize(uniform_guess)
    structured_guesses.append(uniform_guess)
    modified_guess = initial_guess.copy()
    flip_mask = np.random.rand(n_points) < 0.3
    modified_guess[flip_mask] = 1 - modified_guess[flip_mask]
    modified_guess = normalize(modified_guess)
    structured_guesses.append(modified_guess)

    start_time = time.time()
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        h = guess.copy()
        k = 50
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            res = minimize(objective, h, args=(k,), method='L-BFGS-B',
                           bounds=[(0, 1) for _ in range(n_points)],
                           jac=gradient, tol=1e-6)
            if res.success:
                h = np.clip(res.x, 0, 1)
                h = normalize(h)
                current_c5 = evaluate(h)
                if current_c5 < best_c5:
                    best_h, best_c5 = h.copy(), current_c5
                k *= 1.1
        if time.time() - start_time > budget_s:
            break

    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate, best_h, method='SLSQP',
                       bounds=[(0, 1) for _ in range(n_points)],
                       constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                       tol=1e-6)
        if res.success:
            h = np.clip(res.x, 0, 1)
            h = normalize(h)
            current_c5 = evaluate(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
