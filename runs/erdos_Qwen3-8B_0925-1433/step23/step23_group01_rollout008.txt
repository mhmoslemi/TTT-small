import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        h = np.clip(x, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            return float('inf')
        h = h * target_sum / h_sum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate initial guesses
    initial_guesses = []
    initial_guess = np.array(initial_h_values, dtype=float)
    initial_guesses.append(initial_guess)
    
    # Flip 30% of the elements
    flip_mask = np.random.rand(n_points) < 0.3
    flipped = initial_guess.copy()
    flipped[flip_mask] = 1 - flipped[flip_mask]
    initial_guesses.append(flipped)
    
    # Sine wave
    x = np.linspace(0, 2, n_points)
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_guess = np.clip(sine_guess, 0, 1)
    initial_guesses.append(sine_guess)
    
    # Random noise
    noise_guess = np.random.rand(n_points)
    noise_guess = np.clip(noise_guess, 0, 1)
    noise_guess = noise_guess * target_sum / np.sum(noise_guess)
    initial_guesses.append(noise_guess)
    
    best_h = initial_guess.copy()
    best_c5 = objective(initial_guess)
    
    start_time = time.time()
    
    # Run differential evolution on each initial guess
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1) for _ in range(n_points)]
        result = differential_evolution(objective, bounds, strategy='best1bin', 
                                       popsize=15, mutation=(0.5, 1), recombination=0.9, 
                                       tol=1e-6, maxiter=100, seed=seed)
        current_h = result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    
    # Refine with trust-constr
    if time.time() - start_time < budget_s:
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        bounds = [(0, 1) for _ in range(n_points)]

            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
