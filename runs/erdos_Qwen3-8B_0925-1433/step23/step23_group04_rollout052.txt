import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraints(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraints}

    def perturb_block(x, block_size=5):
        idx = np.random.choice(n_points, block_size, replace=False)
        x[idx] = 1 - x[idx]
        return normalize(x)

    def perturb_shift(x, shift=3):
        shift = np.random.randint(-shift, shift)
        x = np.roll(x, shift)
        return normalize(x)

    start_time = time.time()

    def generate_perturbations(base, count=10):
        perturbations = []
        for _ in range(count):
            p = base.copy()
            if np.random.rand() < 0.5:
                p = perturb_block(p)
            else:
                p = perturb_shift(p)
            perturbations.append(p)
        return perturbations

    initial_guesses = generate_perturbations(initial_guess, 20)

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, guess, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = np.clip(res.x, 0, 1)
            h = normalize(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturbed = best_h.copy()
        if np.random.rand() < 0.5:
            perturbed = perturb_block(perturbed)
        else:
            perturbed = perturb_shift(perturbed)
        res = minimize(obj, perturbed, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = np.clip(res.x, 0, 1)
            h = normalize(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
