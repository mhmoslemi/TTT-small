import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_ref = np.array(initial_h_values, dtype=float)
    h_ref = h_ref * target_sum / np.sum(h_ref)
    best_c5 = np.max(np.correlate(h_ref, 1 - h_ref, mode='full')) * dx
    best_h = h_ref.copy()
    best_c5 = min(best_c5, 0.381)

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def de_bounds():
        return [(0, 1)] * n_points

    def de_fitness(x):
        x = clip_rescale(x)
        return objective(x)

    def slsqp_bounds():
        return [(0, 1)] * n_points

    def slsqp_constraints(x):
        return np.sum(x) - target_sum

    def l_bfgs_bounds():
        return [(0, 1)] * n_points

    def l_bfgs_constraints(x):
        return np.sum(x) - target_sum

    def generate_structured_guesses():
        n = n_points
        guesses = []
        # Block patterns
        for m in [3, 5, 7, 9]:
            block_size = n // m
            guess = np.zeros(n)
            for i in range(m):
                if i % 2 == 0:
                    guess[i*block_size:(i+1)*block_size] = 1.0
            shift = np.random.randint(0, block_size)
            guess = np.roll(guess, shift)
            guesses.append(clip_rescale(guess))
        # Sinusoidal
        guess = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n) / (n - 1)))
        guess = np.clip(guess, 0, 1)
        guess = guess * target_sum / np.sum(guess)
        guesses.append(guess)
        # Checkerboard
        guess = np.zeros(n)
        guess[::2] = 1.0
        guess = np.roll(guess, np.random.randint(0, n//2))
        guesses.append(clip_rescale(guess))
        # Permutation
        guess = np.copy(initial_h_values)
        np.random.shuffle(guess)
        guesses.append(clip_rescale(guess))
        # Mirror
        guess = 1.0 - initial_h_values
        guesses.append(clip_rescale(guess))
        return guesses

    initial_guesses = generate_structured_guesses()
    start_time = time.time()

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:

        result = differential_evolution(de_fitness, de_bounds(), 
                                       strategy='best1bin', 
                                       maxiter=100, 
                                       popsize=30, 
                                       tol=1e-6, 
                                       mutation=(0.8, 1), 
                                       recombination=0.9, 
                                       disp=False)
        if result.success:
            res_h = result.x
            res_c5 = objective(res_h)
            if res_c5 < best_c5:
                best_c5 = res_c5
                best_h = res_h.copy()

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, best_h, method='SLSQP',
                          bounds=slsqp_bounds(),
                          constraints=[{'type': 'eq', 'fun': slsqp_constraints}],
                          tol=1e-6, options={'maxiter': 200})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()

            best_c5 = result.fun

    best_h = clip_rescale(best_h)
    return (best_h, best_c5, n_points)
