import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def objective(x):
        h = normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(normalize(x)) - target_sum

    initial_guess = np.array(initial_h_values, dtype=float)
    best_h = normalize(initial_guess)
    best_c5 = objective(initial_guess)

    initial_guesses = []
    # Generate initial guesses
    initial_guesses.append(initial_guess.copy())
    initial_guesses.append(initial_guess.copy())
    initial_guesses.append(initial_guess.copy())
    # Flip 30%
    flip_mask = np.random.rand(n_points) < 0.3
    initial_guesses[1][flip_mask] = 1 - initial_guesses[1][flip_mask]
    # Sine-based
    x = np.linspace(0, 2, n_points)
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_guess = np.clip(sine_guess, 0, 1)
    initial_guesses[2] = normalize(sine_guess)
    # Gaussian noise
    noise = np.random.normal(0, 0.1, n_points)
    initial_guesses[3] = normalize(initial_guess + noise)

    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_guess.copy()

    for i, guess in enumerate(initial_guesses):
        if time.time() - start_time > budget_s:
            break
        # Differential evolution
        bounds = [(0, 1) for _ in range(n_points)]
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.9,
            maxiter=100,
            tol=1e-6,
            polish=True,
            seed=seed + i
        )
        if result.success:
            h = normalize(result.x)
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        # Trust-constr refinement
        if time.time() - start_time > budget_s:
            break
        result = minimize(
            objective,
            result.x,
            method='trust-constr',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            tol=1e-6
        )
        if result.success:
            h = normalize(result.x)
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

    return (np.clip(best_h, 0, 1), best_c5, n_points)
