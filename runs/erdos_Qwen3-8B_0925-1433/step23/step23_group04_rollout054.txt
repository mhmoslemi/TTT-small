from scipy.optimize import minimize, differential_evolution

def run(seed=42, budget_s=600):

    def normalize(x):

        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]

    def generate_initial_guesses(num, base):
        guesses = []
        for _ in range(num):
            perturbed = base.copy()
            noise = np.random.normal(0, 0.05, n_points)
            perturbed += noise
            perturbed = normalize(perturbed)
            guesses.append(perturbed)
        for _ in range(num // 2):
            flipped = base.copy()
            flip_mask = np.random.rand(n_points) < 0.2
            flipped[flip_mask] = 1 - flipped[flip_mask]
            flipped = normalize(flipped)
            guesses.append(flipped)
        for _ in range(num // 2):
            uniform = np.random.rand(n_points)
            uniform = normalize(uniform)
            guesses.append(uniform)
        return guesses

    start_time = time.time()

        res = minimize(obj, guess, method='L-BFGS-B', bounds=bounds, tol=1e-6)
        if res.success:
            h = normalize(res.x)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
