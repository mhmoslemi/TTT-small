    initial_guesses = [initial_guess]
    sin_wave = 0.5 * (1 + np.sin(2 * np.pi * np.linspace(0, 1, n_points)))
    sin_guess = normalize(sin_wave)
    initial_guesses.append(sin_guess)
    binary_pattern = np.random.rand(n_points)
    binary_pattern[:n_points//2] = 0.0
    binary_pattern[n_points//2:] = 1.0
    binary_guess = normalize(binary_pattern)
    initial_guesses.append(binary_guess)
    segment_guess = np.zeros(n_points)
    seg_len = n_points // 3
    segment_guess[:seg_len] = 1.0
    segment_guess[seg_len:2*seg_len] = 0.0
    segment_guess[2*seg_len:] = 0.5
    segment_guess = normalize(segment_guess)
    initial_guesses.append(segment_guess)
    random_guess = normalize(np.random.rand(n_points))
    initial_guesses.append(random_guess)

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, guess, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
