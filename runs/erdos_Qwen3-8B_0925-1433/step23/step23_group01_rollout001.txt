import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def objective(x):
        h = normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_guesses = []
    initial_guess = np.array(initial_h_values, dtype=float)
    initial_guesses.append(initial_guess)
    flip_mask = np.random.rand(n_points) < 0.3
    initial_guess[flip_mask] = 1 - initial_guess[flip_mask]
    initial_guesses.append(initial_guess)
    x = np.linspace(0, 2, n_points)
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_guess = normalize(sine_guess)
    initial_guesses.append(sine_guess)
    noise = np.random.normal(0, 0.1, n_points)
    noise_guess = initial_guess + noise
    noise_guess = normalize(noise_guess)
    initial_guesses.append(noise_guess)
    inverse_guess = 1 - initial_guess
    inverse_guess = normalize(inverse_guess)
    initial_guesses.append(inverse_guess)

    best_h = normalize(initial_guess)
    best_c5 = objective(initial_guess)

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = differential_evolution(
            objective,
            bounds=[(0, 1) for _ in range(n_points)],
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.9,
            maxiter=50,
            tol=1e-6,
            disp=False,
            seed=seed,
            x0=guess
        )
        if result.success:
            current_h = normalize(result.x)
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_h, best_c5 = current_h, current_c5

    if time.time() - start_time < budget_s:
        result = minimize(
            objective,
            best_h,
            method='trust-constr',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': lambda x: np.sum(normalize(x)) - target_sum},
            tol=1e-6,
            options={'maxiter': 50}
        )
        if result.success:
            best_h = normalize(result.x)
            best_c5 = objective(result.x)

    return (np.clip(best_h, 0, 1), best_c5, n_points)
