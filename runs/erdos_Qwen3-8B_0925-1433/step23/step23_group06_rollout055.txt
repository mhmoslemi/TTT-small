import numpy as np
from scipy.optimize import differential_evolution, minimize
import time
from scipy.stats import qmc

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        m = n_points // 2
        halton = qmc.Halton(d=1, scramble=True)
        points = halton.random(m) * 2
        grid = np.linspace(0, 2, n_points)
        for x in points:
            idx = np.argmin(np.abs(grid - x))
            h[idx] = 1.0
        target_sum = n_points / 2.0
        h = h * target_sum / np.sum(h)
    
    def objective(x):
        h_clipped = np.clip(x, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    def clip_normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_halton_scramble(scramble):
        m = n_points // 2
        halton = qmc.Halton(d=1, scramble=scramble)
        points = halton.random(m) * 2
        grid = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for x in points:
            idx = np.argmin(np.abs(grid - x))
            h[idx] = 1.0
        h = clip_normalize(h)
        return h
    
    initial_guesses = [generate_halton_scramble(False), generate_halton_scramble(True), generate_halton_scramble(False)]
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res_de = differential_evolution(objective, [(0, 1)] * n_points, popsize=30, maxiter=50, tol=1e-5)
        h_candidate = res_de.x.copy()
        h_candidate = clip_normalize(h_candidate)
        current_c5 = objective(h_candidate)
        if current_c5 < best_c5:
            best_h = h_candidate.copy()
            best_c5 = current_c5
    
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res_slsqp = minimize(objective, best_h, method='SLSQP',
                            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                            bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 100})
        if res_slsqp.success:
            h_candidate = res_slsqp.x.copy()
            h_candidate = clip_normalize(h_candidate)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5
    
    best_h = clip_normalize(best_h)
    return (best_h, best_c5, n_points)
