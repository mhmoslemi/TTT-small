import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    def de_bounds():
        return [(0, 1)] * n_points
    def de_fitness(x):
        x = clip_rescale(x)
        return objective(x)
    def slsqp_bounds():
        return [(0, 1)] * n_points
    def slsqp_constraints(x):
        return np.sum(x) - target_sum
    def l_bfgs_bounds():
        return [(0, 1)] * n_points
    def l_bfgs_constraints(x):
        return np.sum(x) - target_sum
    def generate_guesses(base=None):
        guesses = []
        for num_blocks in [3, 5, 7, 9]:
            block_size = n_points // num_blocks
            guess = np.zeros(n_points)
            for i in range(num_blocks):
                guess[i*block_size:(i+1)*block_size] = 1.0
            shift = np.random.randint(0, block_size)
            guess = np.roll(guess, shift)
            guesses.append(clip_rescale(guess))
        x = np.linspace(0, 2, n_points)
        guesses.append(clip_rescale(0.5 + 0.5 * np.sin(2 * np.pi * x)))
        guesses.append(clip_rescale(np.random.binomial(1, 0.5, n_points)))
        if base is not None:
            guesses.append(clip_rescale(1 - base))
        guesses.append(clip_rescale(np.random.rand(n_points)))
        return guesses
    start_time = time.time()
    h_ref = np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)
    best_c5 = objective(h_ref)
    best_h = h_ref.copy()
    guesses = generate_guesses(h_ref)
    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        result = differential_evolution(de_fitness, de_bounds(), strategy='best1bin', maxiter=100, popsize=30, tol=1e-6, mutation=(0.5, 1), recombination=0.9, disp=False)
        if result.success:
            res_h = result.x
            res_c5 = objective(res_h)
            if res_c5 < best_c5:
                best_c5 = res_c5
                best_h = res_h.copy()
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, best_h, method='SLSQP', bounds=slsqp_bounds(), constraints=[{'type': 'eq', 'fun': slsqp_constraints}], tol=1e-6, options={'maxiter': 300})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()
    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, best_h, method='L-BFGS-B', bounds=l_bfgs_bounds(), constraints=[{'type': 'eq', 'fun': l_bfgs_constraints}], tol=1e-6, options={'maxiter': 200})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()
    best_h = clip_rescale(best_h)
    return (best_h, best_c5, n_points)
