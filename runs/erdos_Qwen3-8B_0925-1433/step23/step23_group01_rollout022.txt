import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    structured_guesses = []
    x = np.linspace(0, 2, n_points)
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_guess = np.clip(sine_guess, 0, 1)
    sine_guess = normalize(sine_guess)
    structured_guesses.append(sine_guess)
    uniform_guess = np.random.rand(n_points)
    uniform_guess = normalize(uniform_guess)
    structured_guesses.append(uniform_guess)
    modified_guess = initial_h_values.copy()
    flip_mask = np.random.rand(n_points) < 0.3
    modified_guess[flip_mask] = 1 - modified_guess[flip_mask]
    modified_guess = normalize(modified_guess)
    structured_guesses.append(modified_guess)
    structured_guesses.append(initial_h_values)

    def surrogate_objective(x, κ):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        exp_term = np.exp(-κ * corr)
        return (1.0 / κ) * np.log(np.sum(exp_term)) * dx

    def true_objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = initial_h_values.copy()
    best_c5 = true_objective(best_h)

    start_time = time.time()
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        h = normalize(np.array(guess, dtype=float))
        for κ in [50, 100, 150]:
            if time.time() - start_time > budget_s:
                break
            res = minimize(lambda x: surrogate_objective(x, κ), h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
            if res.success:
                h = normalize(res.x)
                current_c5 = true_objective(h)
                if current_c5 < best_c5:
                    best_h, best_c5 = h, current_c5
                    print(f"New best C5: {best_c5}")

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        res = minimize(true_objective, best_h, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, tol=1e-6)
        if res.success:
            h = normalize(res.x)
            current_c5 = true_objective(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h, current_c5
                print(f"Refined C5: {best_c5}")

    return (np.clip(best_h, 0, 1), best_c5, n_points)
