import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def smooth_objective(x, κ=30):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        exp_sum = np.sum(np.exp(-κ * corr))
        return np.log(exp_sum) / κ

    def true_objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = true_objective(best_h)

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    structured_guesses = []
    # Parent solution
    structured_guesses.append(initial_guess)
    # Sinusoidal
    x = np.linspace(0, 2, n_points)
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_guess = normalize(sine_guess)
    structured_guesses.append(sine_guess)
    # Triangular
    triangular_guess = 0.5 + 0.5 * (2 * (x / 2) - 1)
    triangular_guess = normalize(triangular_guess)
    structured_guesses.append(triangular_guess)
    # Polynomial
    poly_guess = 0.5 + 0.2 * (x / 2) + 0.3 * (x / 2)**2
    poly_guess = normalize(poly_guess)
    structured_guesses.append(poly_guess)

    start_time = time.time()
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(smooth_objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = true_objective(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        result = minimize(true_objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = true_objective(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
