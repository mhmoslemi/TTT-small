import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    def smooth_objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k = 30  # Smoothing parameter
        smooth = (1 / k) * np.log(np.sum(np.exp(-k * corr)))
        return smooth * dx

    def true_objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate structured initial guesses
    structured_guesses = []
    x = np.linspace(0, 2, n_points)
    
    # Parent solution
    parent_guess = normalize(np.array(initial_h_values, dtype=float))
    structured_guesses.append(parent_guess)
    
    # Sinusoidal guess
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_guess = np.clip(sine_guess, 0, 1)
    sine_guess = normalize(sine_guess)
    structured_guesses.append(sine_guess)
    
    # Triangular guess
    triangle_guess = 0.5 + 0.5 * (2 * (x / 2) - 1)
    triangle_guess = np.clip(triangle_guess, 0, 1)
    triangle_guess = normalize(triangle_guess)
    structured_guesses.append(triangle_guess)
    
    # Polynomial guess
    poly_guess = 0.5 + 0.2 * (x / 2) + 0.3 * (x / 2) ** 2
    poly_guess = np.clip(poly_guess, 0, 1)
    poly_guess = normalize(poly_guess)
    structured_guesses.append(poly_guess)
    
    start_time = time.time()
    
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(
            smooth_objective,
            guess,
            method='trust-constr',
            bounds=bounds,
            constraints=cons,
            tol=1e-6
        )
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = true_objective(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5
    
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        result = minimize(
            true_objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            tol=1e-6
        )
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = true_objective(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
