        perturbed = best_h + noise
        perturbed = normalize(perturbed)
        current_c5 = compute_c5(perturbed)
        if current_c5 < best_c5:
            best_h, best_c5 = perturbed.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
