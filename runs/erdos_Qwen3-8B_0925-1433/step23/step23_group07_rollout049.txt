def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        grad = np.zeros_like(h)

            if 0 <= j < n_points:

            j = i - k
            if 0 <= j < n_points:
                grad[i] -= h[j]
        return grad

    start_time = time.time()

        result_de = differential_evolution(objective, bounds=[(0, 1)]*n_points, strategy='best1bin', maxiter=50, popsize=20, tol=1e-6, mutation=(0.5, 1), recombination=0.7, disp=False)
