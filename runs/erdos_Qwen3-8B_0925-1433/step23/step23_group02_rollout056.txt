        return np.max(corr) * dx

    def obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraints(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraints, 'jac': lambda x: np.ones_like(x)}

    start_time = time.time()
    initial_guesses = [initial_guess]

    # Additional structured initial guesses
    sin_wave = 0.5 * (1 + np.sin(2 * np.pi * np.linspace(0, 1, n_points)))
    sin_guess = normalize(sin_wave)
    initial_guesses.append(sin_guess)

    square_wave = np.zeros(n_points)
    square_wave[:n_points//2] = 1.0
    square_guess = normalize(square_wave)
    initial_guesses.append(square_guess)

    tri_wave = np.zeros(n_points)
    for i in range(n_points):
        tri_wave[i] = (i / n_points - 0.5) * 2.0
    tri_wave = np.clip(tri_wave, 0, 1)
    tri_guess = normalize(tri_wave)
    initial_guesses.append(tri_guess)

    cluster_wave = np.zeros(n_points)
    num_ones = int(n_points * 0.4)
    positions = np.random.choice(n_points, num_ones, replace=False)
    cluster_wave[positions] = 1.0
    cluster_guess = normalize(cluster_wave)
    initial_guesses.append(cluster_guess)

    random_guess = normalize(np.random.rand(n_points))
    initial_guesses.append(random_guess)

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, guess, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
