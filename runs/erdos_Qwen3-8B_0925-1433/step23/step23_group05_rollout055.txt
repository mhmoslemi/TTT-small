import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

    early_stop = 50
    current_iter = 0
    start_time = time.time()

    while current_iter < max_iter and time.time() - start_time < budget_s:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        actual_lag = max_idx - (n_points - 1)
        gradient = 1.0 - best_h[(np.arange(n_points) + actual_lag) % n_points]
        new_h = best_h - learning_rate * gradient
        new_h = np.clip(new_h, 0.0, 1.0)
        sum_new = np.sum(new_h)
        if sum_new != 0:
            new_h = new_h * target_sum / sum_new
        else:
            new_h = np.zeros(n_points)
        new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx

            best_c5 = new_c5
            best_h = new_h.copy()

            pert = 0.1 * np.random.randn(n_points)
            new_h_pert = best_h + pert

            if sum_pert != 0:

                new_h_pert = np.zeros(n_points)
