import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guesses = []

    # Original parent
    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    initial_guesses.append(initial_guess)

    # Random flip
    modified_guess = initial_guess.copy()
    flip_mask = np.random.rand(n_points) < 0.3
    modified_guess[flip_mask] = 1 - modified_guess[flip_mask]
    initial_guesses.append(normalize(modified_guess))

    # Sine-based
    x = np.linspace(0, 2, n_points)
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_guess = normalize(sine_guess)
    initial_guesses.append(sine_guess)

    # Uniform random
    uniform_guess = np.random.rand(n_points)
    uniform_guess = normalize(uniform_guess)
    initial_guesses.append(uniform_guess)

    # Block-based
    block_size = n_points // 5
    block_guess = np.zeros(n_points)
    for i in range(5):
        block_guess[i * block_size:(i + 1) * block_size] = 1
    block_guess = normalize(block_guess)
    initial_guesses.append(block_guess)

    # Objective function
    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = initial_guesses[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()

    # Differential evolution parameters
    de_params = {
        'strategy': 'best1bin',
        'popsize': 15,
        'mutation': (0.5, 1),
        'recombination': 0.9,
        'maxiter': 100,
        'tol': 1e-6,
        'disp': False
    }

    # Run differential evolution on all initial guesses
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, guess, method='differential_evolution', bounds=[(0, 1)] * n_points, **de_params)
        if result.success:
            h = normalize(result.x)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    # Trust-constr refinement
    if time.time() - start_time < budget_s:
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(objective, best_h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = normalize(result.x)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
