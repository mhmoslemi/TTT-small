import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def objective(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}

    start_time = time.time()

    # Segment-based coarse optimization
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        # Perturb random segments
        seg_len = n_points // 5
        seg_idx = np.random.randint(0, 5)
        start = seg_idx * seg_len
        end = start + seg_len
        perturb = np.random.uniform(-0.05, 0.05, seg_len)
        best_h[start:end] += perturb
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * target_sum / np.sum(best_h)
        current_c5 = objective(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5

    # Fine-tuning with local optimizer
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            best_h = np.clip(res.x, 0, 1)
            best_h = best_h * target_sum / np.sum(best_h)
            current_c5 = objective(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5

    return (best_h, best_c5, n_points)
