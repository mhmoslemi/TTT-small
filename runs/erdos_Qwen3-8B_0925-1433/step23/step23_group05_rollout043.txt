import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    initial_guess = np.array(initial_h_values, dtype=float)
    initial_guess = clip_rescale(initial_guess)
    best_c5 = objective(initial_guess)
    best_h = initial_guess.copy()

    def generate_structured_guesses():
        guesses = []
        blocks = [3, 5, 7]
        for m in blocks:
            block_size = n_points // m
            pattern = np.zeros(n_points)
            for i in range(n_points):
                if (i // block_size) % 2 == 0:
                    pattern[i] = 1.0
            for s in range(3):
                shifted = np.roll(pattern, s)
                shifted = clip_rescale(shifted)
                guesses.append(shifted)
        for _ in range(3):
            idx = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
            pattern = np.zeros(n_points)
            pattern[idx] = 1.0
            pattern = clip_rescale(pattern)
            for s in range(3):
                guesses.append(np.roll(pattern, s))
        antipodal = 1.0 - initial_guess
        antipodal = clip_rescale(antipodal)
        for s in range(3):
            guesses.append(np.roll(antipodal, s))
        return guesses

    guesses = generate_structured_guesses()
    start_time = time.time()

    def de_bounds():
        return [(0, 1)] * n_points

    def de_fitness(x):
        x = clip_rescale(x)
        return objective(x)

    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        result = differential_evolution(
            de_fitness, de_bounds(), strategy='best1bin',
            maxiter=80, popsize=25, tol=1e-4, mutation=(0.8, 1), recombination=0.9,
            disp=False)
        if result.success:
            res_h = result.x
            res_c5 = objective(res_h)
            if res_c5 < best_c5:
                best_c5 = res_c5
                best_h = res_h.copy()

    def slsqp_bounds():
        return [(0, 1)] * n_points

    def slsqp_constraints(x):
        return np.sum(x) - target_sum

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        result = minimize(
            objective, best_h, method='SLSQP',
            bounds=slsqp_bounds(),
            constraints=[{'type': 'eq', 'fun': slsqp_constraints}],
            tol=1e-5, options={'maxiter': 200})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()

    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        result = minimize(
            objective, best_h, method='L-BFGS-B',
            bounds=slsqp_bounds(),
            constraints=[{'type': 'eq', 'fun': slsqp_constraints}],
            tol=1e-5, options={'maxiter': 100})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()

    best_h = clip_rescale(best_h)
    return (best_h, best_c5, n_points)
