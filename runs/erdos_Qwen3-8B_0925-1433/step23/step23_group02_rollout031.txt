        grad = compute_gradient(best_h)
        best_h -= learning_rate * grad
        best_h = normalize(best_h)
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5

    learning_rate = 0.01
    for _ in range(200):
        if time.time() - time.time() > budget_s:
            break
        grad = compute_gradient(best_h)
        best_h -= learning_rate * grad
        best_h = normalize(best_h)
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5

    other_guesses = [
        0.5 * (1 + np.sin(2 * np.pi * np.linspace(0, 1, n_points))),
        np.random.rand(n_points),
        np.zeros(n_points), np.ones(n_points)
    ]
    for guess in other_guesses:
        guess = normalize(guess)
        for _ in range(100):
            if time.time() - time.time() > budget_s:
                break
            grad = compute_gradient(guess)
            guess -= learning_rate * grad
            guess = normalize(guess)
            current_c5 = compute_c5(guess)
            if current_c5 < best_c5:
                best_h, best_c5 = guess, current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
