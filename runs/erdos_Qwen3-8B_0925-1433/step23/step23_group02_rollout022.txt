from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        x = np.clip(x, 0, 1)

        return x

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraints(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraints, 'jac': lambda x: np.ones_like(x)}

    start_time = time.time()

    sin_wave = 0.5 * (1 + np.sin(2 * np.pi * np.linspace(0, 1, n_points)))

    binary_pattern = np.random.rand(n_points)
    binary_pattern[:n_points//2] = 0.0
    binary_pattern[n_points//2:] = 1.0
    binary_guess = normalize(binary_pattern)

    random_guess = normalize(np.random.rand(n_points))
    initial_guesses.append(random_guess)

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:

        res = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-4)
