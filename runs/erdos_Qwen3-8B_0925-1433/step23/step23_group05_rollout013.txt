import numpy as np

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

            noise = np.random.normal(0, perturb_rate, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
    
    return (best_h, best_c5, n_points)
