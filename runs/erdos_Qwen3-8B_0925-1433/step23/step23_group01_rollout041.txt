import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def objective_de(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def check_feasibility(h):
        return np.all((0 <= h) & (h <= 1)) and np.isclose(np.sum(h), target_sum)

    initial_guesses = []
    initial_guesses.append(normalize(np.array(initial_h_values, dtype=float)))
    flipped = np.array(initial_h_values, dtype=float)
    flipped[np.random.rand(len(flipped)) < 0.3] = 1 - flipped[np.random.rand(len(flipped)) < 0.3]
    initial_guesses.append(normalize(flipped))
    x = np.linspace(0, 2, n_points)
    sine = 0.5 + 0.5 * np.sin(np.pi * x)
    sine = normalize(np.clip(sine, 0, 1))
    initial_guesses.append(sine)
    uniform = normalize(np.random.rand(n_points))
    initial_guesses.append(uniform)

    best_h = initial_guesses[0].copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()
    bounds = [(0, 1) for _ in range(n_points)]

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = differential_evolution(
            objective_de, bounds, strategy='best1bin',
            popsize=15, mutation=(0.5, 1), recombination=0.9,
            tol=1e-6, maxiter=100, polish=False
        )
        if result.success:
            h_candidate = result.x
            h = normalize(h_candidate)
            if check_feasibility(h):
                current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
                if current_c5 < best_c5:
                    best_h, best_c5 = h.copy(), current_c5

    if time.time() - start_time < budget_s:
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        result = minimize(objective, best_h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = normalize(result.x)
            if check_feasibility(h):
                current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
                if current_c5 < best_c5:
                    best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
