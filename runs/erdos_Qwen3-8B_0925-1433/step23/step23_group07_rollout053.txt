from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    best_c5 = min(best_c5, 0.381)
    
    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad
    
    def clip_and_normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x
    
    start_time = time.time()
    n_particles = 50
    inertia = 0.8
    cognitive = 1.5
    social = 1.5
    swarm = np.random.rand(n_particles, n_points)
    swarm = clip_and_normalize(swarm)
    velocities = np.random.uniform(-0.5, 0.5, (n_particles, n_points))
    p_best = swarm.copy()
    p_best_val = np.array([objective(p) for p in p_best])
    g_best = swarm[np.argmin(p_best_val)].copy()
    g_best_val = np.min(p_best_val)
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        for i in range(n_particles):

                            social * r2 * (g_best - swarm[i]))
