from scipy.optimize import differential_evolution, minimize

    best_c5 = np.max(np.correlate(h_ref, 1 - h_ref, mode='full')) * dx

    best_c5 = min(best_c5, 0.381)

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

        return [(0, 1)] * n_points

    def de_fitness(x):

        return objective(x)

        guess2 = np.zeros(n)
        for i in range(n):
            if (i % (n // 4)) == 0:
                guess2[i] = 1.0
        guess2 = np.roll(guess2, np.random.randint(0, n // 2))
        guess2 = clip_rescale(guess2)

        guess4 = np.clip(guess4, 0, 1)
