import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}

    def generate_guesses(base):
        guesses = []
        noise = np.random.normal(0, 0.1, n_points)
        guesses.append(normalize(base + noise))
        flip_mask = np.random.rand(n_points) < 0.3
        guesses.append(normalize(base.copy().astype(float)))
        guesses.append(normalize(np.random.rand(n_points)))
        for _ in range(2):
            mask = np.random.rand(n_points) < 0.1
            gen = base.copy()
            gen[mask] = 1 - gen[mask]
            guesses.append(normalize(gen))
        # Binary pattern
        indices = np.linspace(0, n_points, n_points + 1, dtype=int)
        mid = indices[1:-1][n_points // 2]
        dist = np.abs(np.arange(n_points) - mid)
        mask = dist < n_points // 4
        bin_h = np.zeros(n_points)
        bin_h[mask] = 1
        guesses.append(normalize(bin_h))
        return guesses

    start_time = time.time()
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        for guess in generate_guesses(initial_guess):
            res = minimize(obj, guess, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = np.clip(res.x, 0, 1)
                h = normalize(h)
                current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
                if current_c5 < best_c5:
                    best_h, best_c5 = h.copy(), current_c5
    for _ in range(3):
