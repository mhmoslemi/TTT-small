import numpy as np
from scipy.stats import qmc
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(45)
    n_points = len(h)
    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    def clip_norm(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x
    
    def obj(x):
        x = clip_norm(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    best_h = h.copy()
    best_c5 = obj(best_h)
    
    def generate_halton(n):
        m = n // 2
        points = qmc.Halton(d=1, scramble=False).random(m)
        points *= 2.0
        h = np.zeros(n)
        for i, x in enumerate(points):
            idx = np.argmin(np.abs(x - np.linspace(0, 2, n)))
            h[idx] = 1.0
        return h
    
    def generate_sine(n):
        h = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n) / (n - 1)))
        h = np.clip(h, 0, 1)
        return h
    
    def generate_block(n, size=5):
        h = np.zeros(n)
        for i in range(0, n, 2 * size):
            h[i:i + size] = 1.0
        return h
    
    def generate_even(n, phase=0):
        spacing = n // 2
        h = np.zeros(n)
        for i in range(n):
            if (i - phase) % spacing == 0:
                h[i] = 1.0
        return h
    
    guesses = []
    if 'initial_h_values' in globals():
        guesses.append(h)
    else:
        guesses.append(generate_halton(n_points))
    guesses.append(generate_sine(n_points))
    guesses.append(generate_block(n_points))
    guesses.append(generate_even(n_points))
    
    start_time = time.time()
    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        res = differential_evolution(obj, [(0, 1)] * n_points, popsize=25, maxiter=80, tol=1e-5)
        h_candidate = res.x.copy()
        h_candidate = clip_norm(h_candidate)
        current_c5 = obj(h_candidate)
        if current_c5 < best_c5:
            best_h = h_candidate
            best_c5 = current_c5
    
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 100})
        if res.success:
            h_candidate = res.x.copy()
            h_candidate = clip_norm(h_candidate)
            current_c5 = obj(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate
                best_c5 = current_c5
    
    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B',
                       bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 100})
        if res.success:
            h_candidate = res.x.copy()
            h_candidate = clip_norm(h_candidate)
            current_c5 = obj(h_candidate)
            if current_c5 < best_c5:
                best_h = h_candidate
                best_c5 = current_c5
    
    best_h = clip_norm(best_h)
    return (best_h, best_c5, n_points)
