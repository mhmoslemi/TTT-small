from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    particles = np.clip(h + 0.1 * np.random.randn(n_points), 0, 1)
    particles = particles * target_sum / np.sum(particles)
    velocities = 0.5 * np.random.randn(n_points)
    personal_best = particles.copy()
    personal_best_fitness = np.array([objective(p) for p in particles])
    global_best = particles[np.argmin(personal_best_fitness)]
    global_best_fitness = np.min(personal_best_fitness)

    start_time = time.time()

            particles[i] += velocities[i]

            fitness = objective(particles[i])

        global_best = particles[np.argmin(personal_best_fitness)]

        # Local refinement
