import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    personal_fitness = np.inf * np.ones(n_pop)
    global_best = positions[0].copy()
    global_fitness = np.inf

        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
    
    for i in range(n_pop):
        personal_fitness[i] = objective(positions[i])
        if personal_fitness[i] < global_fitness:
            global_fitness = personal_fitness[i]
            global_best = positions[i].copy()
    
    start_time = time.time()
    w, c1, c2 = 0.8, 1.5, 1.5

            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = w * velocities[i] + c1 * r1 * (personal_best[i] - positions[i]) + c2 * r2 * (global_best - positions[i])

            curr_fit = objective(positions[i])
            if curr_fit < personal_fitness[i]:

                global_best = positions[i].copy()
