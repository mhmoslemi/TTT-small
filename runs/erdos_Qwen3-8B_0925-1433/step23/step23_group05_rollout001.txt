from scipy.optimize import differential_evolution, minimize

    best_h = h_ref.copy()

        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def clip_rescale(x):

        x = x * target_sum / np.sum(x)
        return x

    def generate_structured_guesses():
        n = n_points
        guesses = []
        for shift in range(3):
            block_size = n // 5 + (1 if shift % 2 else 0)

                if (i // block_size) % 2 == 0:
                    guess[i] = 1.0
            guess = np.roll(guess, np.random.randint(n // 2))
            guesses.append(clip_rescale(guess))
        for freq in [3, 4, 5]:

            guess = np.clip(guess, 0, 1)
            guess = guess * target_sum / np.sum(guess)
            guesses.append(guess)
        for _ in range(2):
            guess = np.random.rand(n)
            guess = guess * target_sum / np.sum(guess)
            guesses.append(clip_rescale(guess))
        return guesses
