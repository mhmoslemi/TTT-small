import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=np.float64))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def evaluate_c5(h):
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = Bounds(0, 1)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    start_time = time.time()
    guesses = []

    # Generate structured initial guesses
    x_vals = np.linspace(0, 2, n_points)
    guesses.append(initial_guess)
    guesses.append(normalize(0.5 + 0.5 * np.sin(2 * np.pi * x_vals)))
    guesses.append(normalize(0.5 + 0.5 * np.cos(2 * np.pi * x_vals)))
    guesses.append(normalize(0.5 + 0.25 * (np.sin(2 * np.pi * x_vals) + np.cos(4 * np.pi * x_vals))))
    guesses.append(normalize(0.5 + 0.1 * np.sin(6 * np.pi * x_vals)))
    guesses.append(np.random.rand(n_points))
    guesses.append(np.zeros(n_points) + 0.5)
    guesses.append(np.random.rand(n_points) * 0.5)

    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate_c5, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate_c5, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
