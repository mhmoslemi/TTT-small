import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def evaluate_c5(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        h = normalize(np.clip(x, 0, 1))
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def cons(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': cons, 'jac': lambda x: np.ones_like(x)}

    def calc_gradient(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        corr_values = corr * dx
        sum_e = np.sum(np.exp(0.01 * corr_values))
        grad = np.zeros_like(h)
        for i in range(n_points):
            for k in range(len(corr_values)):
                i_plus_k = i + k
                i_minus_k = i - k
                term1 = 0.01 * np.exp(0.01 * corr_values[k]) / sum_e
                term2 = (1 - h[i_plus_k % n_points]) if i_plus_k % n_points else 0.0
                term3 = -h[i_minus_k % n_points] if i_minus_k >= 0 else 0.0
                grad[i] += term1 * (term2 + term3)
        return grad

    def grad_obj(x):
        return calc_gradient(x)

    start_time = time.time()
    initial_guesses = [initial_guess]

    initial_guesses.append(random_guess)

    for guess in initial_guesses:
