from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}

    def generate_guesses(base):
        guesses = []
        noise = np.random.normal(0, 0.1, n_points)
        guesses.append(normalize(base + noise))
        flips = (np.random.rand(n_points) < 0.2).astype(float)
        guesses.append(normalize(base * (1 - 2 * flips) + flips))
        random_guess = np.random.rand(n_points)
        guesses.append(normalize(random_guess))
        for _ in range(3):
            mask = np.random.rand(n_points) < 0.1
            gen = base.copy()
            gen[mask] = 1 - gen[mask]
            guesses.append(normalize(gen))
        return guesses

    start_time = time.time()
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        for guess in generate_guesses(initial_guess):
            res = minimize(obj, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = np.clip(res.x, 0, 1)
                h = normalize(h)
                current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
                if current_c5 < best_c5:

    for _ in range(2):
