import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if total == 0:
            return np.zeros_like(x)
        factor = target_sum / total
        return x * factor

    initial_guess = np.array(initial_h_values, dtype=float)
    h = normalize(initial_guess)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}

    start_time = time.time()
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        # Regularly perturb by shifting half the positions
        shift = np.random.choice(n_points, size=n_points // 2, replace=False)
        h_perturbed = h.copy()
        h_perturbed[shift] = 1 - h_perturbed[shift]
        h_perturbed = normalize(h_perturbed)
        res = minimize(obj, h_perturbed, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h_candidate = np.clip(res.x, 0, 1)
            h_candidate = normalize(h_candidate)
            current_c5 = np.max(np.correlate(h_candidate, 1 - h_candidate, mode='full')) * dx
            if current_c5 < best_c5:

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
