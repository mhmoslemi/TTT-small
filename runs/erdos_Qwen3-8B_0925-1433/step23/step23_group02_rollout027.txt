import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    h_initial = np.array(initial_h_values, dtype=float)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = normalize(x)
        return compute_c5(h)

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint, 'jac': lambda x: np.ones_like(x)}

    best_h = normalize(h_initial)
    best_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, 0.381)

    start_time = time.time()

    # Generate initial guesses
    def sin_guess():
        x = np.linspace(0, 1, n_points)
        return normalize(0.5 + 0.5 * np.sin(2 * np.pi * x))

    def binary_guess():
        x = np.zeros(n_points)
        x[:n_points//2] = 1.0
        return normalize(x)

    def noise_guess():
        x = np.random.rand(n_points)
        return normalize(x)

    def perturbed_guess():
        x = h_initial + 0.05 * np.random.randn(n_points)
        return normalize(x)

    initial_guesses = [best_h, sin_guess(), binary_guess(), noise_guess(), perturbed_guess()]

    for idx, guess in enumerate(initial_guesses):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            h = normalize(res.x)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
