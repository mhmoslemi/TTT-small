import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
    else:
        n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x
    
    def evaluate_c5(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def obj(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': lambda x: np.ones_like(x)}
    
    start_time = time.time()
    best_h = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else np.random.rand(n_points)
    best_h = normalize(best_h)
    best_c5 = evaluate_c5(best_h)
    
    def generate_sine():
        return 0.5 * (1 + np.sin(2 * np.pi * np.linspace(0, 1, n_points)))
    
    def generate_sparse():
        pos = np.linspace(0, 2, n_points)
        s = np.zeros(n_points)
        s[::3] = 1.0
        s[1::3] = 0.5
        s[2::3] = 0.0
        return s
    
    def generate_turbulent():
        return 0.5 * (1 + np.random.normal(0, 0.2, n_points))
    
    initial_guesses = [
        normalize(initial_h_values) if 'initial_h_values' in globals() else normalize(np.random.rand(n_points)),
        normalize(generate_sine()),
        normalize(generate_sparse()),
        normalize(generate_turbulent()),
        normalize(0.5 * (1 + np.random.rand(n_points) * 0.8))
    ]
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, guess, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5
    
    for _ in range(5):
        if time.time() - start_time > budget_s:
