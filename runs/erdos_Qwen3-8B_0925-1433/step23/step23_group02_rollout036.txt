import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x
    
    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    
    def obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraints(x):
        return np.sum(x) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraints, 'jac': lambda x: np.ones_like(x)}
    
    population = [normalize(initial_guess)]
    for _ in range(4):
        if np.random.rand() < 0.5:
            h = np.zeros(n_points)
            h[np.random.choice(n_points, size=n_points//2, replace=False)] = 1.0
        else:
            h = 0.5 * (1 + np.sin(2 * np.pi * np.linspace(0, 1, n_points)))
        population.append(normalize(h))
    
    start_time = time.time()
    
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        new_population = []
        for h in population:
            if np.random.rand() < 0.3:
                h = h * (1 + 0.2 * np.random.randn(n_points))
            h = normalize(h)
            new_population.append(h)
        population = new_population
    
    for h in population:
        res = minimize(obj, h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = normalize(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
