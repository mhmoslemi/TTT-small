import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    def surrogate(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return (1/50) * np.log(np.sum(np.exp(-50 * corr)))

    def true_obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    structured = [
        initial_guess,
        0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points)),
        0.5 + 0.5 * (2 * (np.linspace(0, 2, n_points)/2) - 1),
        0.5 + 0.2 * (np.linspace(0, 2, n_points)/2) + 0.3 * (np.linspace(0, 2, n_points)/2)**2,
        initial_guess.copy()
    ]
    structured = [normalize(g) for g in structured]

    start_time = time.time()
    for g in structured:
        if time.time() - start_time > budget_s:
            break
        res = minimize(surrogate, g, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = np.clip(res.x, 0, 1)
            h = normalize(h)
            current = true_obj(h)
            if current < best_c5:
                best_h, best_c5 = h, current

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(true_obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = np.clip(res.x, 0, 1)
            h = normalize(h)
            current = true_obj(h)
            if current < best_c5:
                best_h, best_c5 = h, current

    return (np.clip(best_h, 0, 1), best_c5, n_points)
