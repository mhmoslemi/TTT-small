            res = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                current_h = res.x
                current_h = normalize(current_h, target_sum)
                current_c5 = compute_c5(current_h, dx)
                if current_c5 < best_c5:
                    best_h, best_c5 = current_h, current_c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_h = res.x
            current_h = normalize(current_h, target_sum)
            current_c5 = compute_c5(current_h, dx)
            if current_c5 < best_c5:
                best_h, best_c5 = current_h, current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
