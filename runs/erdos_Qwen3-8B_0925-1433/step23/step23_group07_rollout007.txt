import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    def clip_and_normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    # Initialize swarm
    swarm_size = 50
    particles = np.random.uniform(0, 1, (swarm_size, n_points))
    particles = np.array([clip_and_normalize(p) for p in particles])
    velocities = np.random.uniform(-0.5, 0.5, (swarm_size, n_points))
    pbest = particles.copy()
    pbest_fitness = np.array([objective(p) for p in particles])
    gbest = particles[np.argmin(pbest_fitness)]
    gbest_fitness = np.min(pbest_fitness)

    start_time = time.time()
    iterations = 0

    while time.time() - start_time < budget_s:
        iterations += 1
        if iterations > 1000:
            break
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            inertia = 0.8 - 0.001 * iterations
            velocities[i] = inertia * velocities[i] + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = clip_and_normalize(particles[i])
        # Evaluate new fitness
        new_fitness = np.array([objective(p) for p in particles])
        # Update personal best
        improved = new_fitness < pbest_fitness
        pbest[improved] = particles[improved]
        pbest_fitness[improved] = new_fitness[improved]
        # Update global best
        current_best_idx = np.argmin(pbest_fitness)
        if pbest_fitness[current_best_idx] < gbest_fitness:
            gbest = pbest[current_best_idx].copy()
            gbest_fitness = pbest_fitness[current_best_idx]
        # Local refinement
        top_indices = np.argsort(pbest_fitness)[:5]
        for idx in top_indices:
            h_candidate = pbest[idx].copy()
            res = minimize(objective, h_candidate, method='SLSQP', bounds=[(0, 1)]*n_points,
                           constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                           jac=compute_gradient, tol=1e-6, options={'maxiter': 100})

                best_c5 = res.fun
        # Early termination if no improvement
        if iterations > 50 and np.abs(gbest_fitness - best_c5) < 1e-7:
            break

    best_h = clip_and_normalize(best_h)
    return (best_h, best_c5, n_points)
