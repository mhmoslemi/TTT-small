import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def evaluate(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = evaluate(best_h)

    def generate_guesses(base):
        guesses = []
        guesses.append(base)
        flips = np.random.rand(n_points) < 0.3
        guesses.append(normalize(base.copy() * (1 - 2 * flips) + flips))
        sin_x = np.sin(np.linspace(0, np.pi, n_points))
        guesses.append(normalize(base + 0.2 * sin_x))
        noise = np.random.normal(0, 0.1, n_points)
        guesses.append(normalize(base + noise))
        return guesses

    guesses = generate_guesses(initial_guess)

    start_time = time.time()

    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        res = differential_evolution(objective, bounds=[(0, 1)]*n_points, 
                                    strategy='best1bin', popsize=15, 
                                    mutation=(0.5, 1), recombination=0.8, 
                                    tol=1e-6, maxiter=50, 
                                    seed=seed, disp=False)
        if res.success:
            h = normalize(res.x)
            c5 = evaluate(h)
            if c5 < best_c5:
                best_h, best_c5 = h, c5

    for _ in range(3):
        if time.time() - start_time > budget_s:

        res = minimize(objective, best_h, method='L-BFGS-B',

            c5 = evaluate(h)
            if c5 < best_c5:
