        sin_guess = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / (n_points - 1) + phi))
        sin_guess = np.clip(sin_guess, 0, 1)
        sin_guess = sin_guess * target_sum / np.sum(sin_guess)
        guesses.append(sin_guess)
        block_size = n_points // 4
        block_guess = np.zeros(n_points)
        for i in range(n_points):
            if (i // block_size) % 2 == 0:
                block_guess[i] = 1.0
        block_guess = np.roll(block_guess, np.random.randint(0, block_size))
        block_guess = np.clip(block_guess, 0, 1)
        block_guess = block_guess * target_sum / np.sum(block_guess)
        guesses.append(block_guess)
        sparse_guess = np.zeros(n_points)
        rand_indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
        sparse_guess[rand_indices] = 1.0
        sparse_guess = np.clip(sparse_guess, 0, 1)
        sparse_guess = sparse_guess * target_sum / np.sum(sparse_guess)
        guesses.append(sparse_guess)
        guesses.append(h.copy())
        return guesses

    structured_guesses = generate_guesses()

    start_time = time.time()
    for guess in structured_guesses:
        h_current = guess.copy()
        best_h_current = h_current.copy()
        best_c5_current = compute_c5(h_current)
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            m_max = corr.argmax()
            k = m_max - (n_points - 1)
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    grad[i] = 1 - h_current[j]
            noise = np.random.normal(0, 0.05, n_points)
            h_current += 0.01 * (-grad) + noise
            h_current = np.clip(h_current, 0, 1)
            total = np.sum(h_current)
            if total > 0:
                h_current = h_current * target_sum / total
            current_c5 = compute_c5(h_current)
            if current_c5 < best_c5_current:
                best_c5_current = current_c5
                best_h_current = h_current.copy()
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = best_h_current.copy()

    def objective(h_vec):
        return compute_c5(h_vec)

    def constraint(h_vec):
        return np.sum(h_vec) - target_sum

    bounds = [(0, 1)] * n_points
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}])
    if result.success:
        best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h)
    if total > 0:
        best_h = best_h * target_sum / total
    else:
        best_h = np.zeros(n_points)
    return (best_h, best_c5, n_points)
