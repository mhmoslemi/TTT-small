import time

        x = x * target_sum / np.sum(x)
        return x
    
    def obj(x, dx):

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def compute_c5(x):
        return obj(x, dx)

            x = np.linspace(0, 1, n_points)
            return 0.5 * (1 + np.sin(2 * np.pi * x))
        elif pattern_type == 'clusters':
            x = np.linspace(0, 1, n_points)
            return np.zeros_like(x)
        elif pattern_type == 'gradient':
            x = np.linspace(0, 1, n_points)
            return np.sort(x)
        elif pattern_type == 'noise':
            return np.random.rand(n_points)
        else:
            return h
    
    best_h = h
    best_c5 = compute_c5(h)
    
    start_time = time.time()
