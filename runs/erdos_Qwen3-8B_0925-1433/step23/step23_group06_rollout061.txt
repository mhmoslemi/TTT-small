        if time.time() - start_time > budget_s:

        res = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)] * n_points, tol=1e-6, options={'maxiter': 100})
        if res.success:
            h_candidate = res.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = np.max(np.correlate(h_candidate, 1 - h_candidate, mode='full')) * dx
            if current_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
