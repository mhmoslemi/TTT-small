import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def objective(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_guesses():
        guesses = []
        x = np.linspace(0, 2, n_points)
        base = np.array(initial_h_values, dtype=float)
        guesses.append(base)
        guesses.append(np.clip(base + np.random.normal(0, 0.1, n_points), 0, 1))
        guesses.append(np.clip(base * np.random.rand(n_points), 0, 1))
        guesses.append(np.clip(0.5 + 0.5 * np.sin(np.pi * x), 0, 1))
        guesses.append(np.clip(base, 0, 1))
        return [normalize(g) for g in guesses]

    initial_guesses = generate_guesses()
    best_h = initial_guesses[0]
    best_c5 = objective(best_h)

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = differential_evolution(
            objective, bounds, strategy='best1bin', popsize=15,
            mutation=(0.5, 1), recombination=0.9, tol=1e-6, maxiter=100,
            polish=True, disp=False, x0=guess)
        if result.success:

            current_c5 = objective(h)

                best_h = h

    for _ in range(3):

            if current_c5 < best_c5:
