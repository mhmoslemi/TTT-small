        return np.max(corr) * dx

    def generate_guesses(h_base):
        guesses = []
        for phase in [0, np.pi/4, np.pi/2, 3*np.pi/4]:

            h_s = h_s * target_sum / np.sum(h_s)
            guesses.append(h_s)

            h_b = h_b * target_sum / np.sum(h_b)
            guesses.append(h_b)
        for period in [5, 7, 9]:
