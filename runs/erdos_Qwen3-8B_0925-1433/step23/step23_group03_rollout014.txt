import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()

        return np.max(np.correlate(x, 1 - x, mode='full')) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}

    start_time = time.time()

    for _ in range(5):

            break

        for shift in [0, dx/2, -dx/2]:
            m = n_points // 2

            for pos in positions:
                idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - pos))
                h[idx] = 1
            h = normalize(h)
            try:
                res = minimize(obj, h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-5)
                if res.success:

                    h = normalize(h)
                    current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
                    if current_c5 < best_c5:
                        best_h, best_c5 = h.copy(), current_c5

                pass

    for _ in range(3):
