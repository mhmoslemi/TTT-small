import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h):
        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr) * dx
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    def get_segments(num_segs):
        segs = []
        start = 0
        for _ in range(num_segs):
            len_seg = n_points // num_segs
            if n_points % num_segs > 0:
                len_seg += 1
            segs.append((start, start + len_seg))
            start += len_seg
        return segs
    
    segments = get_segments(5)
    start_time = time.time()
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        seg_idx = np.random.randint(len(segments))
        start, end = segments[seg_idx]
        avg = np.mean(best_h[start:end])
        direction = np.random.choice([-1, 1])
        delta = np.random.uniform(0.01, 0.05) * direction
        new_val = avg + delta
        
        h_perturbed = best_h.copy()
        h_perturbed[start:end] = new_val
        h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_c5 = compute_c5(h_perturbed)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_perturbed.copy()
            segments = get_segments(min(15, len(segments)+1))
    
    return (np.clip(best_h, 0, 1), best_c5, n_points)
