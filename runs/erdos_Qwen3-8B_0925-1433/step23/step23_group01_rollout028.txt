import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    structured_guesses = []
    x = np.linspace(0, 2, n_points)
    sine_guess = 0.5 + 0.5 * np.sin(2 * np.pi * x)  # Higher frequency
    sine_guess = np.clip(sine_guess, 0, 1)
    sine_guess = normalize(sine_guess)
    structured_guesses.append(sine_guess)
    square_guess = 0.5 * (1 + np.sign(np.sin(np.pi * x)))
    square_guess = np.clip(square_guess, 0, 1)
    square_guess = normalize(square_guess)
    structured_guesses.append(square_guess)
    triangular_guess = 0.5 * (1 - np.cos(np.pi * x))
    triangular_guess = np.clip(triangular_guess, 0, 1)
    triangular_guess = normalize(triangular_guess)
    structured_guesses.append(triangular_guess)
    uniform_guess = np.random.rand(n_points)
    uniform_guess = normalize(uniform_guess)
    structured_guesses.append(uniform_guess)
    modified_guess = initial_guess.copy()
    flip_mask = np.random.rand(n_points) < 0.3
    modified_guess[flip_mask] = 1 - modified_guess[flip_mask]
    modified_guess = normalize(modified_guess)
    structured_guesses.append(modified_guess)

    start_time = time.time()
    for guess in structured_guesses:

        result = minimize(objective, guess, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-5, options={'maxiter': 100})
        if result.success:

            h = normalize(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
