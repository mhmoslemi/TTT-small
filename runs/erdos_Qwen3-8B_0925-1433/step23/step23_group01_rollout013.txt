import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    structured_guesses = []
    x = np.linspace(0, 2, n_points)
    # Sine-based guess
    sine_guess = 0.5 + 0.5 * np.sin(np.pi * x)
    sine_guess = np.clip(sine_guess, 0, 1)
    sine_guess = normalize(sine_guess)
    structured_guesses.append(sine_guess)
    # Triangle wave guess
    triangle_guess = np.zeros_like(x)
    triangle_guess[:n_points//2] = x[:n_points//2] / (n_points//2)
    triangle_guess[n_points//2:] = 1 - (x[n_points//2:] - n_points//2) / (n_points//2)
    triangle_guess = np.clip(triangle_guess, 0, 1)
    triangle_guess = normalize(triangle_guess)
    structured_guesses.append(triangle_guess)
    # Square wave guess
    square_guess = np.zeros_like(x)
    square_guess[x < 1.0] = 1.0
    square_guess = np.clip(square_guess, 0, 1)
    square_guess = normalize(square_guess)
    structured_guesses.append(square_guess)
    # Polynomial guess
    poly_guess = 0.5 + 0.25 * x**2 - 0.5

    start_time = time.time()
