from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1-h, mode='full')) * dx

    def objective(x):
        h = x
        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1-h, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    def project(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def gen_initial_guesses(base):
        yield base * (1 + 0.3 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)))
        yield project(base * (1 + 0.2 * np.random.normal(0, 1, n_points)))
        yield base[np.random.permutation(n_points)]
        yield base * (1 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points)))

    start_time = time.time()
    
    initial_guesses = []
    for _ in range(4):
        if np.random.rand() < 0.5:
            initial_guesses.append(project(np.array(initial_h_values, dtype=float) * (target_sum / np.sum(np.array(initial_h_values, dtype=float)))))
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result_de = differential_evolution(objective, bounds=[(0, 1)]*n_points, strategy='best1bin', maxiter=100, popsize=20, tol=1e-6, mutation=(0.4, 1), recombination=0.7, disp=False)
        h_de = result_de.x.copy()
        h_de = project(h_de)
        res_slsqp = minimize(objective, h_de, method='SLSQP', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], jac=compute_gradient, tol=1e-6, options={'maxiter': 150})
        if res_slsqp.success and res_slsqp.fun < best_c5:
