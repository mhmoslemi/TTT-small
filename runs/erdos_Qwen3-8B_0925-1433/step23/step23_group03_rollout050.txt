        shifts = np.roll(h, np.random.randint(0, n_points))
        variations.append(shifts)
        perturbed = h.copy()
        idx = np.random.choice(n_points, size=5, replace=False)
        perturbed[idx] = 1.0 - perturbed[idx]
        variations.append(perturbed)

    start_time = time.time()
    for guess in variations:
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_h = np.clip(res.x, 0, 1)
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = current_h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
