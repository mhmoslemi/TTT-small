import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def evaluate_c5(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': lambda x: np.ones_like(x)}

    start_time = time.time()
    initial_guesses = [initial_guess]
    freq = 4.0
    sin_wave = 0.5 * (1 + np.sin(2 * np.pi * freq * np.linspace(0, 1, n_points)))

    initial_guesses.append(sin_guess)
    binary_pattern = np.zeros(n_points)
    binary_pattern[:n_points//3] = 1.0
    binary_guess = normalize(binary_pattern)
    initial_guesses.append(binary_guess)
    binary_pattern = np.zeros(n_points)
    binary_pattern[n_points//4 : 3*n_points//4] = 1.0
    binary_guess = normalize(binary_pattern)
    initial_guesses.append(binary_guess)

    initial_guesses.append(random_guess)

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:

        res = minimize(evaluate_c5, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = normalize(h)
