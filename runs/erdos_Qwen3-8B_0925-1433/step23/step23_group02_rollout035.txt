import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraints(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraints, 'jac': lambda x: np.ones_like(x)}

    start_time = time.time()
    initial_guesses = [initial_guess]
    
    # Generate structured initial guesses
    x = np.linspace(0, 2, n_points)
    cos_freq = 2.0  # Frequency parameter
    cos_guess = 0.5 + 0.5 * np.cos(np.pi * cos_freq * x)
    cos_guess = normalize(cos_guess)
    initial_guesses.append(cos_guess)
    
    tri_guess = 0.5 + 0.5 * np.abs(np.sin(np.pi * x))
    tri_guess = normalize(tri_guess)
    initial_guesses.append(tri_guess)
    
    # Add a binary-like pattern with spacing
    binary_pattern = np.zeros(n_points)
    binary_pattern[:n_points//2] = 0.0
    binary_pattern[n_points//2:] = 1.0
    binary_guess = normalize(binary_pattern)
    initial_guesses.append(binary_guess)
    
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = normalize(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    # Additional refinement with L-BFGS-B
    for _ in range(3):
        if time.time() - start_time > budget_s:

        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
