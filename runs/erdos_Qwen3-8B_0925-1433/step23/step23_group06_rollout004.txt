import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.random.rand(45) * 2
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.clip(h, 0, 1)

        h_clipped = np.clip(x, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    def constraint(x):
        return np.sum(np.clip(x, 0, 1)) - target_sum
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    def generate_block(n, size=5):
        return np.array([(i % (2 * size) < size) for i in range(n)], dtype=float)
    
    def generate_staggered(n, spacing=5):
        return np.array([(i % spacing == 0) for i in range(n)], dtype=float)
    
    def generate_gaussian(n):
        x = np.linspace(0, 2, n)
        return 0.5 * (1 + np.sin(2 * np.pi * x / (n - 1)))
    
    def generate_diagonal(n, size=2):
        h = np.zeros(n)
        for i in range(0, n, 2 * size):
            h[i:i+size] = 1.0
        return h
    
    patterns = [
        generate_block(n_points),
        generate_staggered(n_points),
        generate_gaussian(n_points),
        generate_diagonal(n_points)
    ]
    if 'initial_h_values' in globals():
        patterns.append(h)
    
    for p in patterns:
        if time.time() - start_time > budget_s:
            break
        p_norm = p * target_sum / np.sum(p)
        res = minimize(objective, p_norm, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': constraint}],
                       bounds=[(0, 1)] * n_points, tol=1e-7, options={'maxiter': 200})
        if res.success:
            h_candidate = res.x.copy()
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = objective(h_candidate)
            if current_c5 < best_c5:

        perturb = np.random.normal(0, 0.05, n_points)
        h_perturbed = best_h + perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
