        block_size = np.random.choice([3, 5, 7])
        h_block = np.zeros(n_points)
        for i in range(0, n_points, block_size):
            h_block[i:i+block_size] = 1
        h_block = h_block * target_sum / np.sum(h_block)
        guesses.append(h_block)
        h_rand = h.copy() + np.random.randn(n_points) * 0.1
        h_rand = h_rand * target_sum / np.sum(h_rand)
        guesses.append(h_rand)
        period = np.random.randint(4, 8)
        h_period = np.zeros(n_points)
        for i in range(n_points):
            if i % period == 0:
                h_period[i] = 1
        h_period = h_period * target_sum / np.sum(h_period)
        guesses.append(h_period)
        h_comb = 0.5 * h_sine + 0.5 * h_block
        h_comb = h_comb * target_sum / np.sum(h_comb)
        guesses.append(h_comb)
        unique_guesses = []
        for g in guesses:
            if not any(np.allclose(g, gu, atol=1e-6) for gu in unique_guesses):
                unique_guesses.append(g)
        return unique_guesses

    structured_guesses = generate_structured_guesses(n_points, target_sum)

    start_time = time.time()
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        result = minimize(objective_de, guess, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6, options={'maxiter': 200})
        if result.success:
            current_h = result.x
            current_c5 = objective_de(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1)] * n_points
        result = minimize(objective_de, best_h, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-6, options={'maxiter': 100})
        if result.success and result.fun < best_c5:
            best_h = result.x.copy()
            best_c5 = result.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h, best_c5, n_points)
