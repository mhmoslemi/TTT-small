import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_ref = np.array(initial_h_values, dtype=float) if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    h_ref = h_ref * target_sum / np.sum(h_ref)
    best_c5 = np.max(np.correlate(h_ref, 1 - h_ref, mode='full')) * dx
    best_h = h_ref.copy()

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def de_bounds():
        return [(0, 1)] * n_points

    def de_fitness(x):
        x = clip_rescale(x)
        return objective(x)

    def slsqp_constraints(x):
        return np.sum(x) - target_sum

    def l_bfgs_constraints(x):
        return np.sum(x) - target_sum

    def generate_guesses():
        guesses = []
        block = np.zeros(n_points)
        for i in range(n_points // 5):
            block[i*5:(i+1)*5] = 1.0
        block = np.roll(block, np.random.randint(n_points))
        guesses.append(block)
        sin = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / (n_points - 1)))
        sin = np.roll(sin, np.random.randint(n_points))
        guesses.append(sin)
        sparse = np.zeros(n_points)
        idx = np.random.choice(n_points, size=int(n_points * 0.2), replace=False)
        sparse[idx] = 1.0
        sparse = np.roll(sparse, np.random.randint(n_points))
        sparse = sparse * target_sum / np.sum(sparse)
        guesses.append(sparse)
        anti = 1 - block
        anti = np.roll(anti, np.random.randint(n_points))
        guesses.append(anti)
        return guesses

    guesses = generate_guesses()
    start_time = time.time()

    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        result = differential_evolution(de_fitness, de_bounds(), strategy='best1bin', maxiter=100, popsize=30, tol=1e-6)
        res_h = result.x
        res_c5 = objective(res_h)
        if res_c5 < best_c5:
            best_c5 = res_c5
            best_h = res_h.copy()

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, best_h, method='SLSQP', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': slsqp_constraints}], tol=1e-6, options={'maxiter': 200})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()

    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': l_bfgs_constraints}], tol=1e-6, options={'maxiter': 100})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()

    best_h = clip_rescale(best_h)
    return (best_h, best_c5, n_points)
