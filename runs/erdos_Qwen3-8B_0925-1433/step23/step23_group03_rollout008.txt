from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}

    def generate_guesses(base):
        guesses = [base.copy()]
        if n_points % 2 == 0:
            pattern = np.zeros(n_points)
            pattern[:n_points//2] = 1.0
            guesses.append(normalize(pattern))
        else:
            pattern = np.zeros(n_points)
            pattern[:n_points//2] = 1.0
            pattern[-1] = 0.5
            guesses.append(normalize(pattern))
        for _ in range(3):
            noise = np.random.normal(0, 0.05, n_points)
            guesses.append(normalize(base + noise))
        for _ in range(2):
            flip_mask = np.random.rand(n_points) < 0.2
            temp = base.copy()
            temp[flip_mask] = 1 - temp[flip_mask]
            guesses.append(normalize(temp))
        return guesses

    start_time = time.time()

        if time.time() - start_time > budget_s:

            res = minimize(obj, guess, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-5)
            if res.success:
