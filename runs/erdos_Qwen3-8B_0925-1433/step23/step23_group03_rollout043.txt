def run(seed=42, budget_s=600):

    target_sum = n_points / 2.0

    if initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        m = n_points // 2
        positions = np.linspace(0, 2, m)
        h = np.zeros(n_points)
        for i in range(m):
            x = positions[i]
            idx = np.argmin(np.abs(x - np.linspace(0, 2, n_points)))
            h[idx] = 1.0
        if n_points % 2 == 1:
            zero_indices = np.where(h == 0)[0]
            if zero_indices.size > 0:
                h[zero_indices[0]] = 0.5
        h_init = h

    h_init = np.clip(h_init, 0, 1)
    sum_h = np.sum(h_init)
    if np.abs(sum_h - target_sum) > 1e-6:
        scale = target_sum / sum_h
        h_init = h_init * scale

    def obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

        s = np.sum(h)
        if s < target_sum - 1e-6:

            h[np.where(h == 0)[0][0]] += diff

            h[np.where(h == 1)[0][0]] -= diff

                h = np.clip(h, 0, 1)
                s = np.sum(h)

                    h *= scale
