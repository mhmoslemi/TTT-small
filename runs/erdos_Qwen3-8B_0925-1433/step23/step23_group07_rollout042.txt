    initial_guesses.append(0.5 + 0.5 * np.sin(np.pi * x))
    initial_guesses.append(np.random.permutation(0.5 + 0.5 * np.sin(np.pi * x)))
    initial_guesses.append(np.clip(np.array(initial_h_values) + 0.1 * np.random.randn(n_points), 0, 1))

    best_h = initial_h_values.copy()
    best_c5 = float('inf')

    def objective(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(x)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - x[j]
        return grad

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        guess = normalize_guess(guess)
        res = minimize(objective, guess, method='SLSQP', bounds=[(0, 1)]*n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       jac=compute_gradient, tol=1e-6, options={'maxiter': 200})
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x.copy()

    best_h = normalize_guess(best_h)
    return (best_h, best_c5, n_points)
