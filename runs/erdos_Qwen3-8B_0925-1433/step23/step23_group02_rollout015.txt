    binary_guess = normalize(binary_pattern)
    initial_guesses.append(binary_guess)
    random_guess = normalize(np.random.rand(n_points))
    initial_guesses.append(random_guess)
    tri_wave = 0.5 * (1 + np.abs(np.sin(2 * np.pi * np.linspace(0, 1, n_points))))
    tri_guess = normalize(tri_wave)
    initial_guesses.append(tri_guess)

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate_c5, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        res = minimize(evaluate_c5, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
