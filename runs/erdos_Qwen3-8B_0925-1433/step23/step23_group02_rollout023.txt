def run(seed=42, budget_s=600):

    n_points = len(initial_h_values) if hasattr(initial_h_values, '__len__') else 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x) if np.sum(x) > 0 else x
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()

        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')

        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
