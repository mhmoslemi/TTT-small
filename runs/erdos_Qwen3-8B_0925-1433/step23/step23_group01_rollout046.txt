    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    def objective_surrogate(h):

        return (1 / 50) * np.log(np.sum(np.exp(-50 * corr)))

    def objective_exact(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    def finite_diff_grad(h):

            h_m[i] -= eps

            grad[i] = (val_p - val_m) / (2 * eps)
        return grad

    structured_guesses = []
