import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_structured_guesses():
        guesses = []
        for g in [np.zeros(n_points), np.ones(n_points), np.sin(np.linspace(0, np.pi, n_points))]:
            g = g * target_sum / np.sum(g)
            guesses.append(g)
        return guesses

    structured_guesses = generate_structured_guesses()

    start_time = time.time()
    particles = [np.clip(g, 0, 1) for g in structured_guesses]
    particles += [np.random.rand(n_points) * target_sum / np.sum(np.random.rand(n_points)) for _ in range(50 - len(structured_guesses))]
    velocities = np.random.rand(len(particles), n_points) * 0.1
    best_positions = particles[:]
    best_fitness = [objective(p) for p in particles]
    personal_best = particles[:]
    global_best = np.argmin(best_fitness)
    inertia = 0.8
    cognitive = 1.5
    social = 1.5

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        for i in range(len(particles)):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (inertia * velocities[i] +
                             cognitive * r1 * (personal_best[i] - particles[i]) +
                             social * r2 * (best_positions[global_best] - particles[i]))
            particles[i] = np.clip(particles[i] + velocities[i], 0, 1)
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
            fitness = objective(particles[i])
            if fitness < best_fitness[i]:
                best_fitness[i] = fitness
                personal_best[i] = particles[i].copy()
        global_best = np.argmin(best_fitness)
        best_h = particles[global_best].copy()
        best_c5 = best_fitness[global_best]

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       bounds=[(0, 1)] * n_points,
                       tol=1e-6, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_h = res.x.copy()
            best_c5 = res.fun

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B',
                       bounds=[(0, 1)] * n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       tol=1e-6, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_h = res.x.copy()
            best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
