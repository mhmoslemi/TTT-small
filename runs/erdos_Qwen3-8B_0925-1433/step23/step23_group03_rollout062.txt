import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}

    start_time = time.time()
    for _ in range(4):
        if time.time() - start_time > budget_s:
            break
        # Generate perturbed guess based on structured pattern
        m = n_points // 2
        positions = np.linspace(0, 2, m, endpoint=False)
        h_structured = np.zeros(n_points)
        for pos in positions:
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - pos))
            h_structured[idx] = 1
        guess = h_structured.copy()
        guess = normalize(guess + np.random.normal(0, 0.05, n_points))
        res = minimize(obj, guess, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = np.clip(res.x, 0, 1)
            h = normalize(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = np.clip(res.x, 0, 1)
            h = normalize(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
