import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_ref = np.array(initial_h_values, dtype=float)
    h_ref = h_ref * target_sum / np.sum(h_ref)
    best_c5 = np.max(np.correlate(h_ref, 1 - h_ref, mode='full')) * dx
    best_h = h_ref.copy()
    best_c5 = min(best_c5, 0.381)

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def generate_structured_guesses():
        n = n_points
        guesses = []
        for m in [3, 5, 7, 9, 11]:
            block_size = n // m
            guess = np.zeros(n)
            for i in range(n):
                if (i // block_size) % 2 == 0:
                    guess[i] = 1.0
            guess = np.roll(guess, np.random.randint(0, block_size))
            guesses.append(guess)
        for freq in [2, 4, 6, 8]:
            guess = 0.5 * (1 + np.sin(2 * np.pi * freq * np.arange(n) / (n - 1)))
            guess = np.clip(guess, 0, 1)
            guess = guess * target_sum / np.sum(guess)
            guesses.append(guess)
        guess = np.random.rand(n)
        guess = np.clip(guess, 0, 1)
        guess = guess * target_sum / np.sum(guess)
        guesses.append(guess)
        guess = 1 - h_ref
        guess = np.clip(guess, 0, 1)
        guess = guess * target_sum / np.sum(guess)
        guesses.append(guess)
        return guesses

    initial_guesses = generate_structured_guesses()
    start_time = time.time()

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = differential_evolution(
            lambda x: objective(clip_rescale(x)),
            [(0, 1)] * n_points,
            strategy='best1bin',
            maxiter=100,
            popsize=50,
            tol=1e-6,
            mutation=(0.8, 1),
            recombination=0.9,
            disp=False
        )
        if res.success:
            res_h = res.x
            res_c5 = objective(res_h)
            if res_c5 < best_c5:
                best_c5 = res_c5
                best_h = res_h.copy()

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            tol=1e-6,
            options={'maxiter': 200}
        )
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x.copy()

    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        res = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            tol=1e-6,
            options={'maxiter': 100}
        )
        if res.success and res.fun < best_c5:
            best_c5 = res.fun
            best_h = res.x.copy()

    best_h = clip_rescale(best_h)
    return (best_h, best_c5, n_points)
