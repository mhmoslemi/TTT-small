import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    h = np.array(initial_h_values, dtype=float)
    h = normalize(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    temp = 1.0
    cooling = 0.995
    max_iter = 5000
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        pert = np.zeros(n_points)
        flip_mask = np.random.rand(n_points) < 0.1
        pert[flip_mask] = 1
        noise = np.random.normal(0, 0.05, n_points)
        new_h = h + noise + pert
        new_h = normalize(np.clip(new_h, 0, 1))
        new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
        if new_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - new_c5) / temp):
            best_h = new_h.copy()
            best_c5 = new_c5
        temp *= cooling

    return (np.clip(best_h, 0, 1), best_c5, n_points)
