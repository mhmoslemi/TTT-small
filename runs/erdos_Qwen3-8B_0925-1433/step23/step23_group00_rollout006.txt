        for period in [5, 7, 9]:
            h_period = np.zeros(n_points)
            for i in range(n_points):
                if (i % period) == 0:
                    h_period[i] = 1
            h_period = h_period * target_sum / np.sum(h_period)
            guesses.append(h_period)
        h_rand = base_h + 0.05 * np.random.randn(n_points)
        h_rand = h_rand * target_sum / np.sum(h_rand)
        guesses.append(h_rand)
        for alpha in [0.3, 0.5, 0.7]:
            h_comb = 0.5*(1 + np.sin(2*np.pi*np.arange(n_points)/(n_points-1))) * (1-alpha) + \
                     np.zeros(n_points) * alpha
            h_comb = h_comb * target_sum / np.sum(h_comb)
            guesses.append(h_comb)
        unique_guesses = []
        for g in guesses:
            if not any(np.allclose(g, gu, atol=1e-6) for gu in unique_guesses):
                unique_guesses.append(g)
        return unique_guesses
    
    structured_guesses = generate_guesses(h_initial)
    start_time = time.time()
    def objective(x):
        h = x.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    def constraint(x):
        return np.sum(x) - target_sum
    bounds = [(0, 1)] * n_points
    
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}], tol=1e-6, options={'maxiter': 200})
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_h = res.x.copy()
                best_c5 = current_c5
    
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}], tol=1e-6, options={'maxiter': 100})
        if res.success and res.fun < best_c5:
            best_h = res.x.copy()
            best_c5 = res.fun
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
