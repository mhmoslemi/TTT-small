import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    def perturb(h):
        h = h.copy()
        h += np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    start_time = time.time()
    candidates = [best_h]
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        pert = perturb(best_h)
        candidates.append(pert)
        candidates.append(np.roll(pert, int(np.random.randint(1, n_points))))
        candidates.append(pert * (1 + 0.25 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))))

    for h_candidate in candidates:
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, h_candidate, method='SLSQP', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], jac=compute_gradient, tol=1e-6, options={'maxiter': 200})
        if res.success and res.fun < best_c5:
            best_h = res.x.copy()
            best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
