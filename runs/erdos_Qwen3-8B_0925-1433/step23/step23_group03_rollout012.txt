import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    try:
        n_points = len(initial_h_values)
    except NameError:
        n_points = 45  # Default if no parent construction
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize with evenly spaced 1s
    h = np.zeros(n_points)
    m = n_points // 2
    x_vals = np.linspace(0, 2, n_points)
    pos = np.linspace(0, 2 - (m * dx), m)
    for p in pos:
        idx = np.argmin(np.abs(x_vals - p))
        h[idx] = 1.0
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)

    def obj(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def constraint(h_vec):
        return np.sum(h_vec) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}

    best_h = h.copy()
    best_c5 = obj(best_h)

    start_time = time.time()
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        # Perturb by shifting a subset of positions
        shift_mask = np.random.rand(n_points) < 0.2
        h_perturb = h.copy()
        h_perturb[shift_mask] = 1 - h_perturb[shift_mask]
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        res = minimize(obj, h_perturb, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h_candidate = np.clip(res.x, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = obj(h_candidate)
            if current_c5 < best_c5:
                best_h, best_c5 = h_candidate, current_c5

    # Final refinement
    for _ in range(2):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h_candidate = np.clip(res.x, 0, 1)
            h_candidate = h_candidate * target_sum / np.sum(h_candidate)
            current_c5 = obj(h_candidate)
            if current_c5 < best_c5:
                best_h, best_c5 = h_candidate, current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
