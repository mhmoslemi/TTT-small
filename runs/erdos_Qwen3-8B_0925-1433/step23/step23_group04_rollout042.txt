import numpy as np
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    temp = 1.0

    max_iter = 10000
    start_time = time.time()

        block_size = np.random.randint(5, 20)

        h_perturbed[start:start + block_size] = 1 - h_perturbed[start:start + block_size]

        h_perturbed += noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
