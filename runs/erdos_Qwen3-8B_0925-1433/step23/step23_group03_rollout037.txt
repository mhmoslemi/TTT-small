from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    # Generate structured initial guess with evenly spaced 1s
    m = n_points // 2
    positions = np.linspace(0, 2, m, endpoint=False)
    h_struc = np.zeros(n_points)
    for pos in positions:
        idx = int(np.round(pos / dx))
        if 0 <= idx < n_points:
            h_struc[idx] = 1
    h_initial = normalize(h_struc)

    def obj(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}

    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        # Perturb
        mask = np.random.rand(n_points) < 0.1
        perturb = np.zeros(n_points)
        perturb[mask] = np.random.rand(sum(mask)) * 0.3 - 0.15
        guess = normalize(best_h + perturb)
        res = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = np.clip(res.x, 0, 1)
            h = normalize(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5
        # Flip
        mask = np.random.rand(n_points) < 0.1
        guess = normalize(best_h.copy())
        guess[mask] = 1 - guess[mask]
        res = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)

            if current_c5 < best_c5:
