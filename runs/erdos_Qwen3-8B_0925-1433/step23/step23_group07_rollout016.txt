import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    initial_guesses = []
    for _ in range(3):
        base = np.array(initial_h_values, dtype=float)
        base = base * target_sum / np.sum(base)
        initial_guesses.append(base * (1 + 0.3 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))))
        initial_guesses.append(base.copy() * (target_sum / np.sum(base)))
        initial_guesses.append(np.clip(base + np.random.normal(0, 0.15, n_points), 0, 1) * (target_sum / np.sum(np.clip(base + np.random.normal(0, 0.15, n_points), 0, 1))))
        initial_guesses.append(base[np.random.permutation(n_points)] * (target_sum / np.sum(base[np.random.permutation(n_points)])))

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        h_guess = guess.copy()
        h_guess = np.clip(h_guess, 0, 1) * target_sum / np.sum(h_guess)
        res = minimize(objective, h_guess, method='L-BFGS-B', bounds=[(0, 1)]*n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       jac=compute_gradient, tol=1e-6, options={'maxiter': 200})
        if res.success and res.fun < best_c5:
            best_h = res.x.copy()
            best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
