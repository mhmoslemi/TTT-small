import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = np.clip(h, 0, 1)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1)] * n_points

    structured_guesses = []
    phi = np.array([0, np.pi/4, np.pi/2, 3*np.pi/4])
    for p in phi:
        h_s = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / (n_points - 1) + p))
        h_s = h_s * target_sum / np.sum(h_s)
        structured_guesses.append(h_s)
    for block_size in [4, 6, 8]:
        h_b = np.zeros(n_points)
        for i in range(n_points):
            if i % block_size == 0:
                h_b[i] = 1
        h_b = h_b * target_sum / np.sum(h_b)
        structured_guesses.append(h_b)
    for period in [5, 7, 9]:
        h_p = np.zeros(n_points)
        for i in range(n_points):
            if i % period == 0:
                h_p[i] = 1
        h_p = h_p * target_sum / np.sum(h_p)
        structured_guesses.append(h_p)

    for guess in structured_guesses:
        if time.time() - time.time() > budget_s:
            break
        res = minimize(objective, guess, method='SLSQP', bounds=Bounds(0, 1), constraints=[{'type': 'eq', 'fun': constraint}], tol=1e-6, options={'maxiter': 100})
        if res.success:
            current_h = res.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5

    for _ in range(5):

        res = minimize(objective, best_h, method='L-BFGS-B', bounds=Bounds(0, 1), constraints=[{'type': 'eq', 'fun': constraint}], tol=1e-6, options={'maxiter': 100})
