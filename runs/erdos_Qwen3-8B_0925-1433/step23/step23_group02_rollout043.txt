import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    cons = {'type': 'eq', 'fun': constraint, 'jac': lambda x: np.ones_like(x)}
    bounds = [(0, 1) for _ in range(n_points)]

    best_h = initial_h_values.copy()
    best_c5 = evaluate_c5(best_h)

    start_time = time.time()

    def make_guess(prefix, *args):
        if prefix == 'fft':
            x = np.linspace(0, 2, n_points)
            return 0.5 + 0.5 * np.cos(np.pi * x)
        elif prefix == 'bin':
            h = np.zeros(n_points)
            h[:n_points//2] = 1.0
            return h
        elif prefix == 'rand':
            return np.random.rand(n_points)
        return np.array(initial_h_values)

    guesses = [make_guess('fft'), make_guess('bin'), make_guess('rand')]

    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        for method in ['L-BFGS-B', 'SLSQP']:
            if time.time() - start_time > budget_s:
                break
            res = minimize(objective, guess, method=method, bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = res.x
                h = np.clip(h, 0, 1)
                h = h * (target_sum / np.sum(h))
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
        if time.time() - start_time > budget_s:
            break

    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (target_sum / np.sum(h_final))
    return (h_final, best_c5, n_points)
