from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def objective(x, kappa=10):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        approx = (1.0 / kappa) * np.log(np.sum(np.exp(-kappa * corr)))
        return approx * dx

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = evaluate(best_h)

    structured_guesses = [
        normalize(initial_h_values),
        normalize(1 - initial_h_values),
        normalize(np.roll(initial_h_values, 1)),
        0.5 + 0.5 * np.sin(np.pi * np.linspace(0, 2, n_points)),
        0.5 + 0.5 * np.cos(np.pi * np.linspace(0, 2, n_points)),
        np.clip(np.random.rand(n_points) * 0.5 + 0.25, 0, 1),
        normalize(np.random.rand(n_points)),
        normalize([0.5 if i % 3 == 0 else 0.3 for i in range(n_points)]),
        normalize([0.5 if i % 5 == 0 else 0.1 for i in range(n_points)])
    ]

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    for guess in structured_guesses:
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize(h)
            current_c5 = evaluate(h)
            if current_c5 < best_c5:

    for _ in range(3):
