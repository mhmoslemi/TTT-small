import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess
    best_c5 = compute_c5(best_h)

    def optimize(start_guess):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        res = minimize(lambda x: compute_c5(x), start_guess, method='L-BFGS-B', bounds=bounds, constraints=cons)
        return res

    perturbations = []
    for shift in [0.2, 0.3, 0.4]:
        shifted = initial_guess.copy()
        start = int(shift * n_points)
        end = start + int(0.15 * n_points)
        shifted[start:end] = shifted[start:end][::-1]
        perturbations.append(normalize(shifted))
    for _ in range(4):
        inverted = initial_guess.copy()
        mask = np.random.rand(n_points) < 0.2
        inverted[mask] = 1 - inverted[mask]
        perturbations.append(normalize(inverted))
    for _ in range(3):
        noisy = initial_guess + np.random.normal(0, 0.02, n_points)
        noisy = normalize(noisy)
        perturbations.append(noisy)

    start_time = time.time()
    for guess in perturbations:
        if time.time() - start_time > budget_s:

        if res.success:
            h = normalize(np.clip(res.x, 0, 1))
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:

    for _ in range(5):
