import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    # Generate structured initial guess with evenly spaced 1s
    m = n_points // 2
    positions = np.linspace(0, 2, m, endpoint=False)

        x = i * dx
        idx = np.argmin(np.abs(positions - x))
        if np.abs(positions[idx] - x) <= dx / 2:

    initial_guess = normalize(h_struct)

    def obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}

    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

            h = normalize(h)
