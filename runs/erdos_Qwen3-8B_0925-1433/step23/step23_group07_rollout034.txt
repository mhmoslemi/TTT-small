    best_c5 = min(best_c5, 0.381)

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    initial_guesses = []

    initial_guesses.append(base[np.random.permutation(n_points)] * target_sum / np.sum(base[np.random.permutation(n_points)]))

        result = differential_evolution(
