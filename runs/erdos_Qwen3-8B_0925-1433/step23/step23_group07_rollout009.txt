import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    def generate_initial_guesses(base_h):
        yield base_h * (target_sum / np.sum(base_h))
        yield np.clip(base_h + np.random.normal(0, 0.15, n_points), 0, 1) * (target_sum / np.sum(np.clip(base_h + np.random.normal(0, 0.15, n_points), 0, 1)))
        yield np.clip(base_h * (1 + 0.5 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))), 0, 1) * (target_sum / np.sum(np.clip(base_h * (1 + 0.5 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))), 0, 1)))
        yield np.clip(base_h[np.random.permutation(n_points)], 0, 1) * (target_sum / np.sum(np.clip(base_h[np.random.permutation(n_points)], 0, 1)))
        yield np.random.rand(n_points) * (target_sum / np.sum(np.random.rand(n_points)))

    initial_guesses = []
    for _ in range(3):
        initial_guesses.append(generate_initial_guesses(np.array(initial_h_values, dtype=float)))

    start_time = time.time()
    for guess in np.concatenate(initial_guesses):
        if time.time() - start_time > budget_s:
            break
        result_de = differential_evolution(objective, bounds=[(0, 1)]*n_points, strategy='best1bin', maxiter=150, popsize=25, tol=1e-7, mutation=(0.6, 0.9), recombination=0.8, disp=False)
        h_de = result_de.x.copy()
        h_de = np.clip(h_de, 0, 1) * target_sum / np.sum(h_de)
        res_slsqp = minimize(objective, h_de, method='SLSQP', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], jac=compute_gradient, tol=1e-7, options={'maxiter': 250})
        if res_slsqp.success and res_slsqp.fun < best_c5:
            best_h = res_slsqp.x.copy()
            best_c5 = res_slsqp.fun
        res_l_bfgs = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], jac=compute_gradient, tol=1e-7, options={'maxiter': 150})
        if res_l_bfgs.success and res_l_bfgs.fun < best_c5:

    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
