import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_index = np.argmax(corr)
        k = k_max_index - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    def clip_normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    particles = [h + 0.1 * np.random.randn(n_points) for _ in range(50)]
    particles = [clip_normalize(p) for p in particles]
    best_fitness = [objective(p) for p in particles]
    global_best = particles[np.argmin(best_fitness)]
    global_best_fit = np.min(best_fitness)

    start_time = time.time()
    for t in range(500):
        if time.time() - start_time > budget_s:
            break
        inertia = 0.8 - 0.001 * t
        for i in range(len(particles)):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = particles[i] * inertia + 1.5 * r1 * (particles[i] - global_best) + 1.5 * r2 * (global_best - particles[i])
            new_pos = particles[i] + vel
            new_pos = clip_normalize(new_pos)
            new_fit = objective(new_pos)
            if new_fit < best_fitness[i]:
                particles[i] = new_pos
                best_fitness[i] = new_fit
            if new_fit < global_best_fit:
                global_best = new_pos
                global_best_fit = new_fit
        top_indices = np.argsort(best_fitness)[:5]
        for idx in top_indices:
            res = minimize(objective, particles[idx], method='SLSQP', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], jac=compute_gradient, tol=1e-6, options={'maxiter': 100})
            if res.success and res.fun < global_best_fit:

        if np.abs(global_best_fit - best_c5) < 1e-6:
            break
    best_h = clip_normalize(global_best)
    return (best_h, global_best_fit, n_points)
