from scipy.optimize import differential_evolution, minimize
import time

        return x

        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

            objective, bounds, strategy='best1bin',

            maxiter=100, tol=1e-6, polish=False, seed=seed)
        if result.success:
            return normalize(np.clip(result.x, 0, 1)), result.fun
        return guess, objective(guess)

    start_time = time.time()

        initial_guess,
