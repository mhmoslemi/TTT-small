import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_ref = np.array(initial_h_values, dtype=float)
    h_ref = h_ref * target_sum / np.sum(h_ref)
    best_c5 = np.max(np.correlate(h_ref, 1 - h_ref, mode='full')) * dx
    best_h = h_ref.copy()
    best_c5 = min(best_c5, 0.381)

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def de_bounds():
        return [(0, 1)] * n_points

    def de_fitness(x):
        x = clip_rescale(x)
        return objective(x)

    def slsqp_bounds():
        return [(0, 1)] * n_points

    def slsqp_constraints(x):
        return np.sum(x) - target_sum

    def l_bfgs_bounds():
        return [(0, 1)] * n_points

    def l_bfgs_constraints(x):
        return np.sum(x) - target_sum

    def generate_structured_guesses():
        n = n_points
        guesses = []
        block_counts = [3, 5, 7]
        for bc in block_counts:
            block_size = n // bc
            guess = np.zeros(n)
            for i in range(n):
                if (i // block_size) % 2 == 0:
                    guess[i] = 1.0
            guess = np.roll(guess, np.random.randint(0, n//2))
            guess = clip_rescale(guess)
            guesses.append(guess)
        freqs = [1.5, 2.0, 2.5]
        for f in freqs:
            guess = 0.5 * (1 + np.sin(2 * np.pi * f * np.arange(n) / n))
            guess = np.clip(guess, 0, 1)
            guess = guess * target_sum / np.sum(guess)
            guesses.append(guess)
        probs = [0.4, 0.5, 0.6]
        for p in probs:
            guess = np.random.binomial(1, p, n)
            guess = guess * target_sum / np.sum(guess)
            guesses.append(guess)
        guess = np.random.rand(n)
        guess = clip_rescale(guess)
        guesses.append(guess)
        if 'initial_h_values' in globals():
            mirrored = 1 - initial_h_values
            mirrored = mirrored * target_sum / np.sum(mirrored)
            guesses.append(mirrored)
        return guesses

    initial_guesses = generate_structured_guesses()
    start_time = time.time()

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = differential_evolution(

        )
        if result.success:
            res_h = result.x
            res_c5 = objective(res_h)
            if res_c5 < best_c5:

        if time.time() - start_time > budget_s:
            break
        result = minimize(

            bounds=slsqp_bounds(),

            best_h = result.x.copy()

    return (best_h, best_c5, n_points)
