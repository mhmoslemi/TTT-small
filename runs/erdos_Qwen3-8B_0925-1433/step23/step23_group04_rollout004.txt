import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraints(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraints}

    start_time = time.time()
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        pert = best_h.copy()
        noise = np.random.normal(0, 0.02, n_points)
        pert += noise
        pert = normalize(pert)
        res = minimize(obj, pert, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            curr_h = np.clip(res.x, 0, 1)
            curr_h = normalize(curr_h)
            curr_c5 = np.max(np.correlate(curr_h, 1 - curr_h, mode='full')) * dx
            if curr_c5 < best_c5:
                best_h, best_c5 = curr_h.copy(), curr_c5

    for _ in range(15):
        if time.time() - start_time > budget_s:
            break
        pert = best_h.copy()
        shift = np.random.randint(0, 3)
        idx = np.random.choice(n_points, 3, replace=False)
        pert[idx] = 1 - pert[idx]
        pert = normalize(pert)
        res = minimize(obj, pert, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:

            curr_h = normalize(curr_h)
