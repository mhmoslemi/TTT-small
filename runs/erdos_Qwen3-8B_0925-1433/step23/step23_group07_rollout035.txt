import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h_initial = np.array(initial_h_values, dtype=float) * target_sum / np.sum(initial_h_values)

    def mutate(h):
        h = h.copy()
        noise = np.random.normal(0, 0.1, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    def sinusoidal(h):
        freq = 1.0 + np.random.rand() * 2.0
        amp = 0.2 + np.random.rand() * 0.3
        x = np.linspace(0, 1, n_points)
        h = h * (1 + amp * np.sin(2 * np.pi * x * freq))
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h

    population = [h_initial]
    for _ in range(50):
        population.append(mutate(h_initial))
        population.append(sinusoidal(h_initial))
    population = list(set([tuple(p) for p in population]))[:50]
    population = [np.array(p) for p in population]

    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        evaluations = [(compute_c5(h), h) for h in population]
        evaluations.sort()
        elites = evaluations[:int(len(evaluations)*0.2)]
        new_pop = [e[1] for e in elites]
        while len(new_pop) < 50:
            p1, p2 = np.random.choice(evaluations, 2, replace=False)
            h1, h2 = p1[1], p2[1]
            h_cross = (h1 + h2) / 2
            h_cross = mutate(h_cross)
            new_pop.append(h_cross)
            h_cross = sinusoidal(h_cross)
            new_pop.append(h_cross)
        population = new_pop

        for c, h in evaluations[:5]:
            h_proj = np.clip(h, 0, 1) * target_sum / np.sum(np.clip(h, 0, 1))
            res = minimize(compute_c5, h_proj, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                           constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}])
            if res.success and res.fun < best_c5:
                best_c5 = res.fun
                best_h = res.x.copy()

    best_h = np.clip(best_h, 0, 1) * target_sum / np.sum(np.clip(best_h, 0, 1))
    return (best_h, best_c5, n_points)
