import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        initial_guess = np.array(initial_h_values, dtype=float)
    else:
        n_points = 100
        initial_guess = np.random.rand(n_points)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = normalize(initial_guess.copy())
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()

    def refine_h(h, max_tries=3):
        for _ in range(max_tries):
            if time.time() - start_time > budget_s:
                break
            res = minimize(lambda x: evaluate_c5(x), h, method='trust-constr', bounds=[(0,1)]*n_points,
                           constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
            if res.success:
                h_new = normalize(res.x)
                current_c5 = evaluate_c5(h_new)
                if current_c5 < evaluate_c5(h):
                    h = h_new.copy()
                    best_h, best_c5 = h.copy(), current_c5

    # Generate diverse initial guesses
    guesses = [best_h.copy()]
    if 'initial_h_values' in globals():
        perm = initial_guess.copy()
        np.random.shuffle(perm)
        guesses.append(perm)
        sin_wave = 0.5 * (1 + np.sin(2 * np.pi * np.linspace(0, 1, n_points)))
        guesses.append(normalize(sin_wave))
        rand = normalize(np.random.rand(n_points))
        guesses.append(rand)
        binary = np.zeros(n_points)
        binary[:n_points//2] = 1
        np.random.shuffle(binary)
        guesses.append(normalize(binary))

    # Evaluate and refine all guesses
    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        h = normalize(guess.copy())
        refine_h(h, max_tries=5)
        if evaluate_c5(h) < best_c5:

    # Local refinement
    for _ in range(5):
        if time.time() - start_time > budget_s:
