            if (i // block_size) % 2 == 0:
                guess2[i] = 1.0
        guess2 = np.roll(guess2, np.random.randint(0, block_size))
        guess2 = np.clip(guess2, 0, 1)
        guess2 = guess2 * target_sum / np.sum(guess2)
        guess3 = np.zeros(n)
        indices = np.random.choice(n, size=n//5, replace=False)
        guess3[indices] = 1.0
        guess3 = np.clip(guess3, 0, 1)
        guess3 = guess3 * target_sum / np.sum(guess3)
        return [guess1, guess2, guess3]

    initial_guesses = generate_structured_guesses()
    start_time = time.time()

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result = differential_evolution(objective, [(0, 1)] * n_points,
                                       strategy='best1bin', maxiter=80, popsize=25,
                                       tol=1e-6, mutation=(0.7, 1), recombination=0.8, disp=False)
        if result.success:
            res_h = result.x
            res_c5 = objective(res_h)
            if res_c5 < best_c5:
                best_c5 = res_c5
                best_h = res_h.copy()

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        result = minimize(objective, best_h, method='L-BFGS-B',
                          bounds=[(0, 1)] * n_points,
                          tol=1e-6, options={'maxiter': 100})
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()

    best_h = clip_rescale(best_h)
    return (best_h, best_c5, n_points)
