import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(h):
        h = np.clip(h, 0, 1)
        return h * target_sum / np.sum(h)

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    eps = 0.01
    learning_rate = 0.1
    decay_rate = 0.99
    max_iter = 500
    time_start = time.time()

    def gradient_step(h):
        corr = np.correlate(h, 1 - h, mode='full')
        sum_exp = np.sum(np.exp(eps * corr))
        soft_max = (1 / eps) * np.log(sum_exp)
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_plus = normalize(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            soft_max_plus = (1 / eps) * np.log(np.sum(np.exp(eps * corr_plus)))
            grad[i] = (soft_max_plus - soft_max) / 1e-5
        h -= learning_rate * grad
        h = normalize(h)
        return h

    for _ in range(max_iter):
        if time.time() - time_start > budget_s:
            break
        best_h = gradient_step(best_h)
        current_c5 = evaluate_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
        learning_rate *= decay_rate

    return (np.clip(best_h, 0, 1), best_c5, n_points)
