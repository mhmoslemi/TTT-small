from scipy.optimize import minimize
import time

    def normalize(x):

        return x

    best_c5 = min(best_c5, 0.381)

    def obj(x):

        return np.max(corr) * dx

    def constraints(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraints}

    def generate_initial_guesses(num, base):

            perturbed = base.copy()
            noise = np.random.normal(0, 0.025, n_points)

            flipped = normalize(flipped)
            guesses.append(flipped)
        for _ in range(num // 2):
            uniform = np.random.rand(n_points)
            uniform = normalize(uniform)
            guesses.append(uniform)
        return guesses

        if res.success:
