from scipy.optimize import differential_evolution, minimize

    h = h * target_sum / np.sum(h)
    
    def objective(x):
        h_clipped = np.clip(x, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    def generate_evenly_spaced(n, phase=0):
        spacing = n // 2
        h = np.zeros(n)
        for i in range(n):
            if (i - phase) % spacing == 0:
                h[i] = 1.0
        return h
    
    def generate_sine(n):
        h = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n) / (n - 1)))
        h = np.clip(h, 0, 1)
        return h
    
    def generate_block(n, size=6):
        h = np.zeros(n)
        for i in range(0, n, 2 * size):
            h[i:i + size] = 1.0
        return h

    for phase in [0, 1, 2, 3]:

        guess = guess * target_sum / np.sum(guess)
        initial_guesses.append(guess)
    initial_guesses.append(generate_sine(n_points))
    initial_guesses.append(generate_block(n_points, 6))
    initial_guesses.append(generate_block(n_points, 4))
    initial_guesses.append(generate_block(n_points, 8))
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    best_c5 = min(best_c5, 0.381)
    
    start_time = time.time()

        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
