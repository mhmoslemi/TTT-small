def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    # Generate Sobol-based initial guess
    m = n_points // 2
    samp = qmc.Sobol(d=1, scramble=True)
    sobol_samples = samp.generate(m)
    positions = np.sort(sobol_samples * 2)
    h_sobol = np.zeros(n_points)
    for pos in positions:
        idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - pos))
        h_sobol[idx] = 1
    h_sobol_normalized = normalize(h_sobol)

    # Initial guess from provided values
    h_initial = normalize(np.array(initial_h_values, dtype=float))

    # Objective function
    def obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Constraint
    def cons(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons_eq = {'type': 'eq', 'fun': cons}

    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()

    # Try initial guesses
