import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}

    best_h = normalize(np.array(initial_h_values, dtype=float))
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    initial_guesses = [best_h]

    # Diverse pattern generator
    def generate_pattern(m):
        pattern = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if np.floor(x * m) % 2 == 0:
                pattern[i] = 1.0
        return normalize(pattern)

    for m in range(3, 10):
        initial_guesses.append(generate_pattern(m))

    def perturb(h, noise=0.05):
        h_perturbed = h.copy()
        mask = np.random.rand(n_points) < 0.1
        h_perturbed[mask] += np.random.normal(0, noise, np.sum(mask))
        h_perturbed = normalize(h_perturbed)
        return h_perturbed

    new_guesses = [perturb(g) for g in initial_guesses]
    initial_guesses.extend(new_guesses)

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, guess, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = np.clip(res.x, 0, 1)
            h = normalize(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = np.clip(res.x, 0, 1)
            h = normalize(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
