import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def evaluate_c5(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraints(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraints, 'jac': lambda x: np.ones_like(x)}

    start_time = time.time()
    initial_guesses = [initial_guess]

    # Structured initial patterns
    # 1. Alternating wave
    wave = 0.5 * (1 + np.sin(2 * np.pi * np.linspace(0, 1, n_points)))
    sin_guess = normalize(wave)
    initial_guesses.append(sin_guess)
    # 2. Clustered binary
    cluster = np.zeros(n_points)
    cluster[:n_points//3] = 1.0
    cluster[n_points//3:2*n_points//3] = 0.0
    cluster[2*n_points//3:] = 1.0
    binary_guess = normalize(cluster)
    initial_guesses.append(binary_guess)
    # 3. Evenly spaced
    m = n_points // 2
    pos = np.linspace(0, 2, m + 1)[:-1]
    spaced = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        for p in pos:
            if abs(x - p) < dx * 0.5:
                spaced[i] = 1.0
                break
    spaced_guess = normalize(spaced)
    initial_guesses.append(spaced_guess)
    # 4. Turbulent noise
    noise = np.random.rand(n_points)
    turb_guess = normalize(noise)
    initial_guesses.append(turb_guess)
    # 5. Gradient-based
    def grad_init(x):
        return 0.5 * (1 + np.sin(2 * np.pi * x * dx))
    grad_guess = normalize(grad_init(np.linspace(0, 2, n_points)))
    initial_guesses.append(grad_guess)

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, guess, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:

            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5
