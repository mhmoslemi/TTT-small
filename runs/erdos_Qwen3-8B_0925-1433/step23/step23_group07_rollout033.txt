import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max_idx = np.argmax(corr)
        k = k_max_idx - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    def gen_initial_guesses(base_h):
        yield base_h * (1 + 0.5 * np.sin(2 * np.pi * np.linspace(0, 1, n_points)))
        yield np.clip(base_h + np.random.normal(0, 0.1, n_points), 0, 1) * (target_sum / np.sum(np.clip(base_h + np.random.normal(0, 0.1, n_points), 0, 1)))
        yield base_h[np.random.permutation(n_points)] * (target_sum / np.sum(base_h[np.random.permutation(n_points)]))
        yield np.expand_dims(np.sin(2 * np.pi * np.linspace(0, 1, n_points)) * 0.5 + 0.5, axis=0).repeat(n_points, axis=0).flatten()

    initial_guesses = []
    for _ in range(3):
        if np.random.rand() < 0.5:
            base = np.array(initial_h_values, dtype=float)
            base = base * target_sum / np.sum(base)
            initial_guesses.extend(gen_initial_guesses(base))

    start_time = time.time()
    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        result_de = differential_evolution(objective, bounds=[(0, 1)]*n_points, strategy='best1bin', maxiter=50, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.7, disp=False)
        h_de = np.clip(result_de.x, 0, 1) * target_sum / np.sum(np.clip(result_de.x, 0, 1))
        res_slsqp = minimize(objective, h_de, method='SLSQP', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], jac=compute_gradient, tol=1e-6, options={'maxiter': 150})
        if res_slsqp.success and res_slsqp.fun < best_c5:
            best_h = res_slsqp.x.copy()

        res_l_bfgs = minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], jac=compute_gradient, tol=1e-6, options={'maxiter': 100})
