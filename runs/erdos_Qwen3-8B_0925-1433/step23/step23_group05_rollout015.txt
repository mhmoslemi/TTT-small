import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(n_points)
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    learning_rate = 0.1
    decay_rate = 0.995
    start_time = time.time()
    for step in range(10000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        indices = (np.arange(n_points) + k_max) % n_points
        grad = 1 - h[indices]
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        if np.sum(h) > 0:
            h = h * target_sum / np.sum(h)
        corr_new = np.correlate(h, 1 - h, mode='full')
        new_c5 = np.max(corr_new) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        if np.abs(new_c5 - best_c5) < 1e-4:
            learning_rate *= decay_rate
        if step % 100 == 0:
            noise = np.random.normal(0, 0.05, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            if np.sum(h) > 0:
                h = h * target_sum / np.sum(h)
    return (best_h, best_c5, n_points)
