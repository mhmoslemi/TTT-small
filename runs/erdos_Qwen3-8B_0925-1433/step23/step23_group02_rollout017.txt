        res = minimize(obj, best_h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            h = normalize(h)
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    best_h = np.clip(best_h, 0, 1)
    best_h = normalize(best_h)
    return (best_h, best_c5, n_points)
