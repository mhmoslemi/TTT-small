import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

    max_iter = 500
    inertia = 0.8
    cognitive = 1.5
    social = 1.5

    particles = []
    velocities = []

        perturb = np.random.normal(0, 0.1, n_points)
        candidate = np.clip(h + perturb, 0, 1)
        candidate = candidate * target_sum / np.sum(candidate)
        particles.append(candidate)
        velocities.append(np.random.rand(n_points) * 0.1)
    
    best_particles = particles.copy()
    best_fitness = np.array([np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in particles])
    best_global = best_particles[np.argmin(best_fitness)]
    best_global_val = np.min(best_fitness)
    
    start = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start > budget_s:
            break
