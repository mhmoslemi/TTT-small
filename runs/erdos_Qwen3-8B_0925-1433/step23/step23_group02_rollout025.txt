import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def evaluate_c5(x):
        h = np.clip(x, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraints(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraints, 'jac': lambda x: np.ones_like(x)}

    start_time = time.time()
    initial_guesses = [initial_guess]

    # Generate structured initial guesses
    # 1. Sine wave with higher frequency
    sin_wave = 0.5 * (1 + np.sin(2 * np.pi * 2 * np.linspace(0, 1, n_points)))
    sin_guess = normalize(sin_wave)
    initial_guesses.append(sin_guess)

    # 2. Binary pattern with clusters
    binary_pattern = np.zeros(n_points)
    binary_pattern[::3] = 1.0  # Every third element is 1
    binary_guess = normalize(binary_pattern)
    initial_guesses.append(binary_guess)

    # 3. Gradient-based guess
    grad_wave = np.linspace(0, 1, n_points)
    grad_guess = normalize(grad_wave)
    initial_guesses.append(grad_guess)

    # 4. Turbulent noise pattern (filtered Gaussian noise)
    noise = np.random.normal(0, 0.2, n_points)
    noise = (noise - np.min(noise)) / (np.max(noise) - np.min(noise))
    noise_guess = normalize(noise)
    initial_guesses.append(noise_guess)

    # 5. Checkerboard pattern
    checkerboard = np.zeros(n_points)
    checkerboard[::2] = 1.0
    checkerboard[1::2] = 0.0
    checkerboard_guess = normalize(checkerboard)
    initial_guesses.append(checkerboard_guess)

    for guess in initial_guesses:

        if res.success:
