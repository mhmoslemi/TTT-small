            bounds = [(0, 1) for _ in range(15)]
            res = minimize(lambda x: objective(x, m, len_seg, n_points, dx, target_sum), initial, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': lambda x: constraint(x, m)}, tol=1e-6)
            if res.success:
                x = res.x
                h_new = np.zeros(n_points)
                for i in range(m):
                    start_idx = i * len_seg
                    end_idx = start_idx + len_seg
                    if i < (n_points % m):
                        end_idx += 1
                    h_new[start_idx:end_idx] = x[i]
                h_new = h_new * target_sum / np.sum(h_new)
                curr_c5 = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
                if curr_c5 < best_c5:
                    nonlocal best_h, best_c5
                    best_h, best_c5 = h_new, curr_c5
        remaining = budget_s - (time.time() - start)
        if remaining < 0:
            return (np.clip(best_h, 0, 1), best_c5, n_points)
        return optimize(m + 1, n_points // (m + 1))

    optimize(5, n_points // 5)
    return (np.clip(best_h, 0, 1), best_c5, n_points)
