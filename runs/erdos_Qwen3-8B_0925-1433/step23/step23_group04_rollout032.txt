import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def obj(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    initial_guess = np.array(initial_h_values, dtype=float)

    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    def generate_perturbations(base):
        guesses = []
        noise = np.random.normal(0, 0.03, n_points)
        guesses.append(base + noise)
        flip_mask = np.random.rand(n_points) < 0.2
        guesses.append(np.where(flip_mask, 1 - base, base))
        uniform = np.random.rand(n_points)
        guesses.append(uniform)
        return guesses

    start_time = time.time()
    for i in range(30):
        if time.time() - start_time > budget_s:
            break
        guess = initial_guess + np.random.normal(0, 0.03, n_points)
        guess = np.clip(guess, 0, 1)
        guess = guess * target_sum / np.sum(guess)
        res = minimize(obj, guess, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-8)
        if res.success:
            h = res.x
            c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if c5 < best_c5:
                best_h, best_c5 = h, c5

    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-10)
        if res.success:
            h = res.x
            c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if c5 < best_c5:
                best_h, best_c5 = h, c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
