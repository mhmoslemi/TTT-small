import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_initial = np.array(initial_h_values, dtype=float)
    h_initial = h_initial * target_sum / np.sum(h_initial)
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(h_initial, 1 - h_initial, mode='full')) * dx

    pop_size = 50
    particles = [h_initial.copy() + 0.1 * np.random.randn(n_points) for _ in range(pop_size)]
    particles = [np.clip(p, 0, 1) * target_sum / np.sum(np.clip(p, 0, 1)) for p in particles]
    personal_best = [p.copy() for p in particles]
    personal_fitness = [np.max(np.correlate(p, 1 - p, mode='full')) * dx for p in particles]
    global_best = min(particles, key=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx)
    global_fitness = np.max(np.correlate(global_best, 1 - global_best, mode='full')) * dx

    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        velocities = [0.5 * np.random.randn(n_points) for _ in range(pop_size)]
        for i in range(pop_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = 0.8 * velocities[i] + 1.5 * r1 * (personal_best[i] - particles[i]) + 1.5 * r2 * (global_best - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1) * target_sum / np.sum(np.clip(particles[i], 0, 1))
            curr_c5 = np.max(np.correlate(particles[i], 1 - particles[i], mode='full')) * dx
            if curr_c5 < personal_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_fitness[i] = curr_c5
            if curr_c5 < global_fitness:
                global_best = particles[i].copy()
                global_fitness = curr_c5

        top_idx = np.argsort(personal_fitness)[:int(pop_size * 0.05)]
        for idx in top_idx:
            def obj(x):
                return np.max(np.correlate(x, 1 - x, mode='full')) * dx
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
            res = minimize(obj, personal_best[idx], method='SLSQP', bounds=[(0, 1)]*n_points, constraints=cons, tol=1e-6, options={'maxiter': 100})
            if res.success and res.fun < global_fitness:
                global_best = res.x.copy()
                global_fitness = res.fun

    best_h = np.clip(global_best, 0, 1) * target_sum / np.sum(np.clip(global_best, 0, 1))
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h, best_c5, n_points)
