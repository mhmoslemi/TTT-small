import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        target_sum = len(h) / 2.0
        h = h * target_sum / np.sum(h)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        target_sum = n_points / 2.0
        h = h * target_sum / np.sum(h)
    n_points = int(len(h))
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    phase_shifts = np.linspace(0, 2 * np.pi, 5)
    block_sizes = [5, 7, 9]
    periods = [5, 7, 9]
    guesses = []

    x = np.linspace(0, 2 * np.pi, n_points)
    for phi in phase_shifts:
        guess = 0.5 * (1 + np.sin(x + phi))
        guess = guess * target_sum / np.sum(guess)
        guesses.append(guess)

    for block_size in block_sizes:
        guess = np.zeros(n_points)
        for i in range(n_points):
            guess[i] = 1 if (i % block_size == 0) else 0
        guess = guess * target_sum / np.sum(guess)
        guesses.append(guess)

    for period in periods:
        guess = np.zeros(n_points)
        for i in range(n_points):
            guess[i] = 1 if (i % period == 0) else 0
        guess = guess * target_sum / np.sum(guess)
        guesses.append(guess)

    guesses.append(h)

    start_time = time.time()
    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        res = minimize(lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx, guess,
                       method='L-BFGS-B', bounds=[(0,1)]*n_points,

        if res.success and res.fun < best_c5:
            best_h = res.x.copy()
            best_c5 = res.fun
