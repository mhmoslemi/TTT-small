        return np.max(corr) * dx

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guesses = []
    n = n_points

    # Generate structured guesses
    phi = np.random.uniform(0, 2 * np.pi)
    guess1 = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n) / (n - 1) + phi))
    guess1 = np.clip(guess1, 0, 1)
    guess1 = guess1 * target_sum / np.sum(guess1)
    initial_guesses.append(guess1)

    block_size = n // 4
    guess2 = np.zeros(n)
    for i in range(n):
        if (i // block_size) % 2 == 0:
            guess2[i] = 1.0
    guess2 = np.roll(guess2, np.random.randint(0, block_size))
    guess2 = clip_rescale(guess2)
    initial_guesses.append(guess2)

    guess3 = np.zeros(n)
    for i in range(n):
        if (i % (n // 2)) == 0:
            guess3[i] = 1.0
    guess3 = np.roll(guess3, np.random.randint(0, n // 2))
    guess3 = clip_rescale(guess3)
    initial_guesses.append(guess3)

    best_guess = min(initial_guesses, key=lambda x: objective(x))
    best_h = best_guess.copy()
    best_c5 = objective(best_h)

    start_time = time.time()
    learning_rate = 0.1
    max_iterations = int((budget_s - 0.5) / 0.2)

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        h = best_h.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = compute_gradient_vectorized(h, k_max, n)
        perturbation = 0.05 * np.random.randn(n)
        h_perturbed = h + perturbation
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (target_sum / h_sum) if h_sum > 1e-8 else h
        new_c5 = objective(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()

    best_h = clip_rescale(best_h)
    return (best_h, best_c5, n_points)
