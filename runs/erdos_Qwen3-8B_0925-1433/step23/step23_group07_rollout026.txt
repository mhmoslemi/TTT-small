from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)

    target_sum = n_points / 2.0

        h = np.clip(h, 0, 1)
        return h * target_sum / np.sum(h)

    # Generate diverse initial population
    def generate_pop(base):
        return [
            project_h(base + np.random.normal(0, 0.1, n_points)),
            project_h(base[np.random.permutation(n_points)]),
            project_h(base * (1 + 0.2 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))))
        ]

    population = generate_pop(h)
    population = [np.clip(p, 0, 1) * target_sum / np.sum(np.clip(p, 0, 1)) for p in population]

        fitness = [objective(p) for p in population]

        # Generate new candidates
        new_pop = elites[:]
        for e in elites:

        # Keep top performers

        # Update best
        if np.min(fitness) < best_c5:
