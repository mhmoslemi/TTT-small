import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        h = np.clip(x, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_guesses():
        guesses = []
        # Sine waves with phase shifts
        for phase in [0, np.pi/4, np.pi/2, 3*np.pi/4]:
            x = np.arange(n_points)
            sine = 0.5 * (1 + np.sin(np.pi * x / (n_points-1) + phase))
            sine = sine * target_sum / np.sum(sine)
            guesses.append(sine)
        # Block patterns
        for size in [5, 6, 7]:
            block = np.zeros(n_points)
            for i in range(0, n_points, size):
                block[i:i+size] = 1
            block = block * target_sum / np.sum(block)
            guesses.append(block)
        # Periodic patterns
        for period in [5, 7, 9]:
            periodic = np.zeros(n_points)
            for i in range(n_points):
                if i % period == 0:
                    periodic[i] = 1
            periodic = periodic * target_sum / np.sum(periodic)
            guesses.append(periodic)
        # Combining sine and periodic
        for phase in [0, np.pi/2]:
            sine = 0.5 * (1 + np.sin(np.pi * np.arange(n_points) / (n_points-1) + phase))
            sine = sine * target_sum / np.sum(sine)
            periodic = np.zeros(n_points)
            for i in range(n_points):
                if i % 7 == 0:
                    periodic[i] = 1
            periodic = periodic * target_sum / np.sum(periodic)
            combo = 0.5 * sine + 0.5 * periodic
            guesses.append(combo)
        # Random perturbation
        rand = h.copy()
        noise = 0.1 * np.random.randn(n_points)
        rand += noise
        rand = np.clip(rand, 0, 1)
        rand = rand * target_sum / np.sum(rand)
        guesses.append(rand)
        # Unique guesses
        unique = []
        for g in guesses:
            if not any(np.allclose(g, gu, atol=1e-6) for gu in unique):
                unique.append(g)
        return unique

        if res.success and res.fun < best_c5:
