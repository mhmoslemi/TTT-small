import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def evaluate_c5(x):
        h = normalize(np.clip(x, 0, 1))
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': np.ones(n_points)}

    start_time = time.time()
    population = [initial_guess]
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        sin_wave = 0.5 * (1 + np.sin(2 * np.pi * np.linspace(0, 1, n_points)))
        sin_guess = normalize(sin_wave + np.random.normal(0, 0.05, n_points))
        population.append(sin_guess)
        binary = np.zeros(n_points)
        binary[:n_points//2] = 0.0
        binary[n_points//2:] = 1.0
        binary_guess = normalize(binary + np.random.normal(0, 0.1, n_points))
        population.append(binary_guess)
        random_guess = normalize(np.random.rand(n_points))
        population.append(random_guess)

    for gen in range(20):
        if time.time() - start_time > budget_s:
            break
        scores = [evaluate_c5(ind) for ind in population]
        ordered = [x for _, x in sorted(zip(scores, population))]
        top5 = ordered[:5]
        candidates = top5 + [np.random.rand(n_points) for _ in range(10)]
        new_pop = []
        for ind in candidates:
            if time.time() - start_time > budget_s:
                break
            p1, p2 = np.random.choice(top5, 2, replace=False)
            child = np.copy(p1)
            mask = np.random.rand(n_points) < 0.7
            child[mask] = p2[mask]
            if np.random.rand() < 0.15:
                child += np.random.normal(0, 0.2, n_points)
            if np.random.rand() < 0.1:
                child += 0.1 * np.sin(2 * np.pi * np.linspace(0, 1, n_points))
            child = normalize(np.clip(child, 0, 1))
            new_pop.append(child)
        population = new_pop[:100]

    for _ in range(5):
        if time.time() - start_time > budget_s:

        res = minimize(evaluate_c5, best_h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)

            h = normalize(res.x)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h, current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
