import time

def run(seed=42, budget_s=600):

    h_ref = h_ref * target_sum / np.sum(h_ref)
    best_c5 = np.max(np.correlate(h_ref, 1 - h_ref, mode='full')) * dx
    best_h = h_ref.copy()

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def grad_step(h, lr):
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad[i] = 1 - h[j]
        h_new = h - lr * grad
        return clip_rescale(h_new)

    def perturb(h, mag=0.1):
        idx = np.random.choice(n_points, int(n_points * mag), replace=False)
        h[idx] += np.random.choice([-1, 1], size=len(idx)) * 0.1
        return clip_rescale(h)

    h = best_h.copy()
    best_c5 = objective(h)
    lr = 0.1
    iter_count = 0
    start_time = time.time()
