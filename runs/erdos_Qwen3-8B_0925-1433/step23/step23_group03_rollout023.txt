import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def obj(x):
        return np.max(np.correlate(x, 1 - x, mode='full')) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}

    best_h = np.zeros(n_points)
    best_c5 = 1.0
    start_time = time.time()

    # Start with initial_h_values
    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    # Explore structured initial guesses
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        positions = np.linspace(0, 2, n_points)
        m = n_points // 2
        selected = np.sort(np.random.choice(positions, size=m, replace=False))
        h_initial = np.zeros(n_points)
        for pos in selected:
            idx = np.argmin(np.abs(positions - pos))
            h_initial[idx] = 1
        h_initial = normalize(h_initial)
        res = minimize(obj, h_initial, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-5)
        if res.success:
            h = np.clip(res.x, 0, 1)
            h = normalize(h)
            c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if c5 < best_c5:
                best_h, best_c5 = h.copy(), c5

    # Final refinement
    for _ in range(5):
        if time.time() - start_time > budget_s:
