import time

    h = h * target_sum / np.sum(h)
    
    def obj(x):
        h_clipped = np.clip(x, 0, 1)
        h_scaled = h_clipped * target_sum / np.sum(h_clipped)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = obj(best_h)
    best_c5 = min(best_c5, 0.381)

    def generate_halton(n, scramble=True):
        m = n // 2
        points = Halton(d=1, scramble=scramble).random(m)
        points *= 2
        h = np.zeros(n)
        for i, x in enumerate(points):
            idx = np.argmin(np.abs(x - np.linspace(0, 2, n)))
            h[idx] = 1.0
        return h * target_sum / np.sum(h)

    def clip_normalize(x):
        return np.clip(x, 0, 1) * target_sum / np.sum(np.clip(x, 0, 1))

    initial_guesses = [h]
