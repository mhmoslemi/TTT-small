from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

    initial_h = np.array(initial_h_values, dtype=float)
    h = initial_h.copy()
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)
    best_time = 0

    def calc_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def perturb_shift(h, shift_size, direction):
        shift = np.roll(h, shift_size * direction)
        return np.clip(shift, 0, 1)

    def perturb_invert(h, invert_ratio):
        mask = np.random.rand(len(h)) < invert_ratio
        return np.clip(1 - h * mask + h * (1 - mask), 0, 1)

    def perturb_sine(h, freq, amp):
        h = h + amp * np.sin(2 * np.pi * freq * np.arange(len(h)) / n_points)
        return np.clip(h, 0, 1)

    def eval_func(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        return calc_c5(h)

    start_time = time.time()

            new_h = perturb_sine(best_h, freq, amp)
        new_c5 = eval_func(new_h)
        diff = best_c5 - new_c5
        if np.random.rand() < np.exp(-diff / temp) if diff > 0 else True:
