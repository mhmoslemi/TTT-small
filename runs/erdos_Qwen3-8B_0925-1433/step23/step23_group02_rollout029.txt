import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in locals():
        n_points = len(initial_h_values)
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 100
        h = np.random.rand(n_points)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    def evaluate_c5(x):
        h = normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        return evaluate_c5(x)

    def constraints(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraints, 'jac': lambda x: np.ones_like(x)}

    best_h = h
    best_c5 = evaluate_c5(h)
    start_time = time.time()

    def generate_initial_population():
        pop = []
        pop.append(best_h)
        pop.append(normalize(0.5 * (1 + np.sin(2 * np.pi * np.linspace(0, 1, n_points)))))  # sinusoidal
        pop.append(normalize(np.where(np.random.rand(n_points) < 0.5, 0, 1)))  # binary
        pop.append(normalize(np.random.rand(n_points)))
        return [normalize(x) for x in pop]

    population = generate_initial_population()

    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        scores = np.array([evaluate_c5(x) for x in population])
        elite = population[np.argsort(scores)[:10]]
        new_population = elite.copy()
        for _ in range(len(population) - 10):
            p1, p2 = np.random.choice(elite, 2, replace=False)
            child = np.copy(p1)
            indices = np.random.choice(n_points, size=int(n_points * 0.2), replace=False)
            child[indices] = p2[indices]
            if np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.1, n_points)
                child += noise
                child = np.clip(child, 0, 1)
            child = normalize(child)
            new_population.append(child)
        population = new_population

    top_candidates = sorted(population, key=evaluate_c5)[:5]
    for h_candidate in top_candidates:
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, h_candidate, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h, current_c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            h = normalize(h)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h, current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
