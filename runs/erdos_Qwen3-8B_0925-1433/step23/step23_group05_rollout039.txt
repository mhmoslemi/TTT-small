import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def clip_rescale(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x == 0:
            return np.zeros_like(x)
        scale = target_sum / sum_x
        return x * scale

    h = np.array(initial_h_values, dtype=float)
    h = clip_rescale(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    learning_rate = 0.1
    iteration = 0
    start_time = time.time()

    while time.time() - start_time < budget_s:
        iteration += 1

        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(cross_corr)
        lag = max_index - (n_points - 1)

        grad = np.zeros(n_points)
        for j in range(n_points):
            j_plus_k = (j + lag) % n_points
            j_minus_k = (j - lag) % n_points
            grad[j] = (1 - h[j_plus_k]) - h[j_minus_k]

        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        sum_h_new = np.sum(h_new)
        scale_factor = target_sum / sum_h_new
        h_new = h_new * scale_factor

        h = h_new.copy()
        new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()

        if iteration % 100 == 0:
            learning_rate *= 0.95

        if iteration > 1000 and np.abs(best_c5 - new_c5) < 1e-6:
            break

    return (best_h, best_c5, n_points)
