import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(h):
        return h * target_sum / np.sum(h)

    h = np.array(initial_h_values, dtype=float)
    h = normalize(h)

    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}

    best_h = h.copy()
    best_c5 = obj(best_h)
    start_time = time.time()

    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.1, n_points)
        guess = h + perturb
        guess = np.clip(guess, 0, 1)
        guess = normalize(guess)
        res = minimize(obj, guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_h = res.x
            current_c5 = obj(current_h)
            if current_c5 < best_c5:
                best_h, best_c5 = current_h, current_c5

    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
