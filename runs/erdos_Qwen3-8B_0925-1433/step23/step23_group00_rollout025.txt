from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def objective_de(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * target_sum / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_guesses():
        guesses = []
        block_size = 5
        h_block = np.zeros(n_points)
        for i in range(0, n_points, block_size):
            h_block[i:i+block_size] = 1
        h_block = h_block * target_sum / np.sum(h_block)
        guesses.append(h_block)
        h_sine = 0.5 * (1 + np.sin(np.pi * np.arange(n_points) / (n_points - 1)))
        h_sine = h_sine * target_sum / np.sum(h_sine)
        guesses.append(h_sine)
        period = 4
        h_period = np.zeros(n_points)
        for i in range(n_points):
            h_period[i] = 1 if (i % period == 0) else 0
        h_period = h_period * target_sum / np.sum(h_period)
        guesses.append(h_period)
        h_greedy = np.zeros(n_points)
        for i in range(n_points):
            h_greedy[i] = 1 if (i % 2 == 0) else 0
        h_greedy = h_greedy * target_sum / np.sum(h_greedy)
        guesses.append(h_greedy)
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * target_sum / np.sum(h_rand)
        guesses.append(h_rand)
        return guesses

    guesses = generate_guesses()

    start_time = time.time()
    for guess in guesses:
        if time.time() - start_time > budget_s:
            break
        bounds = [(0, 1)] * n_points
        result = differential_evolution(objective_de, bounds, strategy='best1bin', popsize=20, maxiter=100, tol=1e-6)
        if result.success:
            current_h = result.x
            current_c5 = objective_de(current_h)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective_de, best_h, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       bounds=[(0, 1)] * n_points,
                       tol=1e-6, options={'maxiter': 200})
        if res.success and res.fun < best_c5:
            best_h = res.x.copy()
            best_c5 = res.fun

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective_de, best_h, method='L-BFGS-B',
                       bounds=[(0, 1)] * n_points,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       tol=1e-6, options={'maxiter': 200})
        if res.success and res.fun < best_c5:
            best_h = res.x.copy()
            best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    return (best_h, best_c5, n_points)
