import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values) if 'initial_h_values' in locals() else 50
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        if np.sum(x) == 0:
            return np.full_like(x, 0.5)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float) if 'initial_h_values' in locals() else np.random.rand(n_points))

    def evaluate_c5(x):
        h = normalize(np.clip(x, 0, 1))
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        return evaluate_c5(x)

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': lambda x: np.ones_like(x)}

    best_h = initial_guess.copy()
    best_c5 = evaluate_c5(best_h)

    start_time = time.time()
    strategies = [
        lambda: 0.5 * (1 + np.sin(2 * np.pi * np.linspace(0, 1, n_points))),
        lambda: np.random.choice([0, 1], size=n_points, p=[0.4, 0.6]),
        lambda: np.kron(np.ones(3), np.array([0.5, 0.5, 0.5]))[:n_points],
        lambda: np.random.rand(n_points)
    ]
    for strategy in strategies:
        guess = normalize(strategy())
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, guess, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
