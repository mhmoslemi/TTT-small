import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def evaluate_c5(x):
        h = normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        return evaluate_c5(x)

    def constraints(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraints, 'jac': lambda x: np.ones_like(x)}

    population = [initial_guess]
    for _ in range(20):
        p = np.random.permutation(initial_guess)
        population.append(normalize(p))
    for _ in range(25):
        b = np.random.rand(n_points)
        b[:n_points//2] = 0
        b[n_points//2:] = 1
        population.append(normalize(b))
    for _ in range(20):
        s = 0.5 * (1 + np.sin(2*np.pi*np.linspace(0,1,n_points)))
        n = np.random.normal(0,0.1,n_points)
        population.append(normalize(s + n))
    for _ in range(20):
        r = np.random.rand(n_points)
        population.append(normalize(r))
    population = [normalize(x) for x in population]

    for gen in range(20):
        scores = np.array([evaluate_c5(x) for x in population])
        elites = np.argsort(scores)[:int(len(population)*0.2)]
        elites = [population[i] for i in elites]
        new_pop = []
        for _ in range(len(population) - len(elites)):
            p1, p2 = np.random.choice(elites, 2, replace=False)
            child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
            if np.random.rand() < 0.3:
                idx = np.random.choice(n_points)

                noise = np.random.normal(0,0.05,n_points)
                child = normalize(child + s + noise)
            new_pop.append(child)
        population = elites + new_pop

    for _ in range(5):
        if time.time() - time.time() > budget_s:
            break
        res = minimize(obj, best_h, method='trust-constr', bounds=bounds, constraints=cons, tol=1e-6)

            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h.copy(), current_c5

    return (np.clip(best_h, 0, 1), best_c5, n_points)
