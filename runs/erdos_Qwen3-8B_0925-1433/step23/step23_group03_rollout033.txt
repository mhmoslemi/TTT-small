from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guess = normalize(np.array(initial_h_values, dtype=float))
    best_h = initial_guess.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    best_c5 = min(best_c5, 0.381)

    def obj(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}

    def generate_guesses(base):
        guesses = []
        noise = np.random.normal(0, 0.1, n_points)
        guesses.append(normalize(base + noise))
        flip_mask = np.random.rand(n_points) < 0.3
        guesses.append(normalize(base.copy().astype(float)))
        guesses.append(normalize(np.random.rand(n_points)))
        for _ in range(2):

            gen = base.copy()
            gen[mask] = 1 - gen[mask]
            guesses.append(normalize(gen))

        binary_pattern[::2] = 1

        # Structured spaced pattern
        spaced_pattern = np.zeros(n_points)
        spaced_pattern[::3] = 1
        guesses.append(normalize(spaced_pattern))
        # Tiling pattern
        tile_pattern = np.tile([1, 0, 1, 0], n_points // 4)
        guesses.append(normalize(tile_pattern))
        return guesses

    start_time = time.time()
    for _ in range(5):

                    h = normalize(h)
                    current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
