import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h = np.array(initial_h_values, dtype=float)
    else:

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def normalize(x):
        x = np.clip(x, 0, 1)
        x = x * target_sum / np.sum(x)
        return x

    initial_guesses = []
    if 'initial_h_values' in globals():
        initial_guesses.append(normalize(np.array(initial_h_values, dtype=float)))
    sin_wave = 0.5 * (1 + np.sin(2 * np.pi * np.linspace(0, 1, n_points)))
    initial_guesses.append(normalize(sin_wave))
    binary_pattern = np.zeros(n_points)
    binary_pattern[n_points//2:] = 1.0
    initial_guesses.append(normalize(binary_pattern))
    random_guess = normalize(np.random.rand(n_points))
    initial_guesses.append(random_guess)

    def objective(x):
        h = normalize(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': lambda x: np.ones_like(x)}

    best_h = normalize(h)
    best_c5 = objective(best_h)
    start_time = time.time()

    for guess in initial_guesses:
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = normalize(res.x)
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_h, best_c5 = h, current_c5

        if time.time() - start_time > budget_s:
