import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def objective(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] = 1.0 - h[j]
        return grad

    start_time = time.time()
    n_particles = 50
    particles = np.clip(h + 0.1 * np.random.randn(n_particles, n_points), 0, 1)
    particles = particles * (target_sum / np.sum(particles, axis=1, keepdims=True))
    velocities = 0.1 * np.random.randn(n_particles, n_points)
    pbest = particles.copy()
    pbest_fitness = np.full(n_particles, np.inf)
    gbest = particles[0].copy()
    gbest_fitness = np.inf
    iter_count = 0
    max_iter = 1000

    while time.time() - start_time < budget_s and iter_count < max_iter:
        for i in range(n_particles):
            omega = 0.8 - 0.001 * iter_count
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * velocities[i] + 1.5 * r1 * (pbest[i] - particles[i]) + 1.5 * r2 * (gbest - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
            fitness = objective(particles[i])
            if fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fitness
        current_best = np.min(pbest_fitness)
        if current_best < gbest_fitness:
            gbest = pbest[np.argmin(pbest_fitness)].copy()
            gbest_fitness = current_best
        iter_count += 1

        if iter_count % 20 == 0:
            top_indices = np.argsort(pbest_fitness)[:5]
            for idx in top_indices:
                res = minimize(objective, pbest[idx], method='SLSQP', bounds=[(0, 1)]*n_points,
                               constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                               jac=compute_gradient, tol=1e-6, options={'maxiter': 100})
                if res.success and res.fun < best_c5:
                    best_h = res.x.copy()
                    best_c5 = res.fun

    best_h = np.clip(best_h, 0, 1) * (target_sum / np.sum(best_h))
    return (best_h, best_c5, n_points)
