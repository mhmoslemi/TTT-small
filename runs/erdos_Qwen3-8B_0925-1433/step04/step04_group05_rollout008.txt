import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def scale_h(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / sum_h, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    particles = [initial_h_values.copy()]
    for _ in range(5):
        h_perturb = initial_h_values + np.random.normal(0, 0.1, n_points)
        particles.append(scale_h(h_perturb))
    h_binary = np.random.choice([0, 1], size=n_points)
    particles.append(scale_h(h_binary))
    h_square = np.zeros(n_points)
    for i in range(n_points):
        val = 1 if i < n_points // 2 else 0
        h_square[i] = val
    particles.append(scale_h(h_square))
    h_triangular = np.linspace(0, 1, n_points)
    particles.append(scale_h(h_triangular))

    def obj(x):
        h = scale_h(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_c5 = min(compute_c5(p) for p in particles)
    best_h = particles[np.argmin([compute_c5(p) for p in particles])].copy()

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        c5_values = [compute_c5(p) for p in particles]
        elite_idx = np.argsort(c5_values)[:5]
        elites = [particles[i] for i in elite_idx]
        new_particles = []
        for i in range(len(elites) // 2):
            parent1 = elites[2*i]
            parent2 = elites[2*i + 1]
            child = parent1 * np.random.rand() + parent2 * (1 - np.random.rand())
            new_particles.append(scale_h(child))
        for p in elites:
            noise = np.random.normal(0, 0.05, n_points)
            mut = p + noise
            new_particles.append(scale_h(mut))
        particles = elites + new_particles
        top_idx = np.argsort(c5_values)[:5]
        for idx in top_idx:
            h_candidate = particles[idx].copy()
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(obj, h_candidate, bounds=bounds, constraints=cons, method='L-BFGS-B', tol=1e-5)
            if res.success:
                particles[idx] = res.x
            else:
                particles[idx] = scale_h(h_candidate)
        current_best = min(c5_values)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = particles[np.argmin(c5_values)].copy()
        if best_c5 < 0.3807:
            break
    return (best_h, best_c5, n_points)
