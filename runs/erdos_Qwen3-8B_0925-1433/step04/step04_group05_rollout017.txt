from scipy.optimize import minimize
import time

            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / sum_h, 0, 1)

    def generate_binary():
        h = np.random.choice([0, 1], size=n_points)
        return scale_h(h)

        freq = np.random.uniform(0.3, 0.7)

        return scale_h(h)
    
    def generate_piecewise():
        split_points = np.random.choice(np.arange(1, n_points), size=5, replace=False)
        split_points.sort()
        h = np.zeros(n_points)

            start, end = split_points[i]-1, split_points[i+1]-1
            h[start:end+1] = np.random.choice([0, 1], size=end - start + 1)
        return scale_h(h)

    particles = [initial_h_values.copy()]
    for _ in range(5):
        particles.append(generate_binary())
        particles.append(generate_periodic())
        particles.append(generate_piecewise())

        return np.max(corr) * dx

        return np.max(corr) * dx
