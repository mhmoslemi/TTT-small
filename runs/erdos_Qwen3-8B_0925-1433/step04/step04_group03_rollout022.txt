import numpy as np
import time

def run(seed=42, budget_s=600):
    n_points = len(initial_h_values)
    h = np.round(initial_h_values).astype(int)
    target_sum = n_points / 2
    current_sum = np.sum(h)
    delta = target_sum - current_sum
    if delta > 0:
        zeros = np.where(h == 0)[0]
        if zeros.size > 0:
            h[zeros[:int(delta)]] = 1
    else:
        ones = np.where(h == 1)[0]
        if ones.size > 0:
            h[ones[:int(-delta)]] = 0
    max_iterations = 1000
    start_time = time.time()
    for _ in range(max_iterations):
        overlaps = np.correlate(h, 1 - h, mode='full')
        k_star = np.argmax(overlaps)
        best_diff = 0
        best_i = -1
        best_j = -1
        for i in range(n_points):
            if h[i] == 1:
                for j in range(n_points):
                    if h[j] == 0:
                        i_plus = (i + k_star) % n_points
                        j_plus = (j + k_star) % n_points
                        change = h[i_plus] - h[j_plus]
                        if change < best_diff:
                            best_diff = change
                            best_i = i
                            best_j = j
        if best_diff < 0:
            h[best_i] = 0
            h[best_j] = 1
        else:
            break
        if time.time() - start_time > budget_s:
            break
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * (2.0 / n_points)
