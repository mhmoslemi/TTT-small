        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if res.success:
            new_h = res.x
            new_h = np.clip(new_h, 0, 1)
            sum_new = np.sum(new_h)
            if sum_new != 0:
                scale = (n_points / 2) / sum_new
                new_h = np.clip(new_h * scale, 0, 1)
            else:
                new_h = np.ones(n_points) * 0.5
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
                if best_c5 < 0.3807:
                    return (best_h, best_c5, n_points)
        else:
            pass

    return (best_h, best_c5, n_points)
