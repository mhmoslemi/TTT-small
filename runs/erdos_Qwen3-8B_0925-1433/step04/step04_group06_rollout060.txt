import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)

    start_time = time.time()
    max_iterations = 300
    perturb_count = 0

    while time.time() - start_time < budget_s and perturb_count < max_iterations:
        corr = np.correlate(h, 1 - h, mode='full')
        top_indices = np.argsort(corr)[-3:][::-1]
        top_corr = corr[top_indices]
        weights = np.exp([-0.2, -0.4, -0.6])
        def obj(x):
            h_curr = x
            corr_curr = np.correlate(h_curr, 1 - h_curr, mode='full')
            selected_corr = corr_curr[top_indices]
            return np.sum(weights * selected_corr) * dx
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = res.x
        except:
            pass
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 < 0.3807:
            break
        if perturb_count % 10 == 0:
            noise = np.random.normal(0, 0.1, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        perturb_count += 1
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
