    best_h = np.zeros(n_points)

    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    def objective(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x == 0:
            return 1e6
        scale = (n_points / 2) / sum_x
        x_opt = np.clip(x * scale, 0, 1)
        return compute_c5(x_opt)

    def constraint(x):
        x = np.clip(x, 0, 1)
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    for h in initial_candidates:
        if time.time() - start_time > budget_s:
            break
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if res.success:
            optimized_h = res.x
            optimized_h = np.clip(optimized_h, 0, 1)
            sum_opt = np.sum(optimized_h)
            if sum_opt != 0:
                scale_opt = (n_points / 2) / sum_opt
                optimized_h = np.clip(optimized_h * scale_opt, 0, 1)
            current_c5 = compute_c5(optimized_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h.copy()

    return (best_h, best_c5, n_points)
