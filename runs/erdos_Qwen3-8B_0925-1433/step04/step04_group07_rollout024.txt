        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project(h):
    h_clipped = np.clip(h, 0, 1)
    sum_clipped = np.sum(h_clipped)
    if sum_clipped == 0:
        return np.zeros_like(h_clipped)
    target_sum = n_points / 2.0
    scale = target_sum / sum_clipped
    return np.clip(h_clipped * scale, 0, 1)
