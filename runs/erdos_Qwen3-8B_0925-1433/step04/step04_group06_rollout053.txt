import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    sum_h = np.sum(h)

    start_time = time.time()

    max_iter = 200
    perturb_interval = 100

    while time.time() - start_time < budget_s:
        iteration += 1
        if iteration > max_iter:
            break

        def obj(x):

            return weights[0] * corr_x[top_lags[0]] + weights[1] * corr_x[top_lags[1]] + weights[2] * corr_x[top_lags[2]]

            if res.success:

            h = h * scale
