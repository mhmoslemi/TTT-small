    h_result = np.clip(h_result * scale, 0, 1)
    best_c5 = compute_c5(h_result, dx, n_points)
    return (h_result, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
