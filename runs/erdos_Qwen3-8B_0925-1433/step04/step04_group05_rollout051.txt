import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def scale_h(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / s, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Generate structured candidates
    candidates = []
    # Binary slice candidates
    for shift in range(5):
        p = np.zeros(n_points)
        for i in range(n_points//2):
            p[(shift + i*2) % n_points] = 1
        candidates.append(scale_h(p))
    # Periodic candidates
    freqs = [0.5, 1.0, 2.0]
    for f in freqs:
        x = np.linspace(0, 2, n_points)
        p = 0.5 + 0.5 * np.sin(2 * np.pi * f * x)
        candidates.append(scale_h(p))
    # Interval spacing candidate
    p = np.zeros(n_points)
    ones = n_points // 2
    pos = 0
    while ones > 0:
        p[pos] = 1
        ones -= 1
        pos += 1
        if pos < n_points:
            p[pos] = 0
            pos += 1
    candidates.append(scale_h(p))

    best_c5 = min(compute_c5(c) for c in candidates)
    best_h = candidates[np.argmin([compute_c5(c) for c in candidates])]

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        c5s = [compute_c5(c) for c in candidates]
        top_idx = np.argsort(c5s)[:3]
        for i in top_idx:
            h = candidates[i].copy()
            bounds = [(0, 1)] * n_points
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2})
            res = minimize(lambda x: compute_c5(scale_h(x)), h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-5)
            if res.success:
                h_opt = res.x
                c5 = compute_c5(scale_h(h_opt))
                if c5 < compute_c5(candidates[i]):
                    candidates[i] = scale_h(h_opt)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = scale_h(h_opt)
        if best_c5 < 0.3807:
            break
    return (best_h, best_c5, n_points)
