    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.abs(cors) * dx
        max_c5 = np.max(max_overlap)
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
        
        worst_lag = np.argmax(np.abs(corr))
        best_i = -1
        max_gain = 0.0
        for i in range(n_points):
            shifted = np.roll(h, -worst_lag)
            old = h[i] * (1 - shifted[i])
            new = (1 - h[i]) * (1 - shifted[i])
            gain = old - new
            if gain > max_gain:
                max_gain = gain
                best_i = i
        
        if max_gain <= 0:
            break
        
        h[best_i] = 1 - h[best_i]
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h = h * scale
        h = np.clip(h, 0, 1)
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(np.abs(corr)) * dx
