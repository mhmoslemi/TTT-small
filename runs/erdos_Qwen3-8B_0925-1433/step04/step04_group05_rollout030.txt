            if 0 <= idx_plus < n_points:
                term1 = 1.0 - h[idx_plus]
            term2 = 0.0
            if 0 <= idx_minus < n_points:
                term2 = h[idx_minus]
            grad[i] = term1 - term2
        h = h + step_size * (-grad)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            scale = (n_points / 2) / sum_h
        else:
            scale = 1.0
        h = h * scale
        
        current_c5 = corr[k_max] * dx
        if best_c5 is None or current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if _ > 50 and abs(best_c5 - corr[k_max] * dx) < tol:
            break
    
    return (best_h, best_c5, n_points)
