                h_candidate = np.clip(h_candidate, 0, 1)
                sum_c = np.sum(h_candidate)
                scale_c = (n_points / 2) / sum_c if sum_c != 0 else 1.0
                h_candidate = np.clip(h_candidate * scale_c, 0, 1)
                candidates[i] = h_candidate
        if best_c5 < 0.3807:
            break

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
