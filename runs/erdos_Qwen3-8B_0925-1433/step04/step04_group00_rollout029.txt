        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        bounds = [(0, 1)] * n_points
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h_new = res.x
            h_new = np.clip(h_new, 0, 1)
            sum_h_new = np.sum(h_new)
            if sum_h_new != 0:
                scale = (n_points / 2) / sum_h_new
                h_new = np.clip(h_new * scale, 0, 1)
            else:
                h_new = np.ones(n_points) * 0.5
            current_c5 = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_new.copy()
                if best_c5 < 0.3807:
                    return (best_h, best_c5, n_points)
    return (best_h, best_c5, n_points)
