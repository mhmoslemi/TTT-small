import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values.copy()
    if np.sum(h) != 0:
        scale = target_sum / np.sum(h)
        h = np.clip(h * scale, 0, 1)
    else:
        h = np.full(n_points, 1.0 / n_points)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()
    iterations = 0
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 < 0.3807:
            break
        if iterations % 50 == 0:
            noise = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * target_sum / np.sum(h)
        def obj(x):
            h_opt = x
            corr = np.correlate(h_opt, 1 - h_opt, mode='full')
            return np.max(corr) * dx
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5, options={'maxiter': 100})
            if res.success:
                h = res.x
        except:
            pass
        iterations += 1
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
