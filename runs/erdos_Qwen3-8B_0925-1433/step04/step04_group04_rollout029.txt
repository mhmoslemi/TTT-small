import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def scale_h(h):
        s = (n_points / 2) / np.sum(h)
        return np.clip(h * s, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population = np.tile(initial_h_values, (50, 1))
    np.random.normal(0, 0.1, population.shape).astype(np.float64, copy=False)
    population = np.clip(population, 0, 1)
    population = np.array([scale_h(p) for p in population])

    pbest = population.copy()
    pbest_vals = np.array([compute_c5(p) for p in population])
    gbest = population[np.argmin(pbest_vals)]
    gbest_val = np.min(pbest_vals)

    start_time = time.time()
    iterations = 0

    while time.time() - start_time < budget_s and iterations < 1000:
        iterations += 1
        velocities = np.random.uniform(-0.5, 0.5, population.shape)
        for i in range(population.shape[0]):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = 0.8 * velocities[i] + 1.5 * r1 * (pbest[i] - population[i]) + 1.5 * r2 * (gbest - population[i])
            population[i] += velocities[i]
            population[i] = np.clip(population[i], 0, 1)
            population[i] = scale_h(population[i])
            curr_val = compute_c5(population[i])
            if curr_val < pbest_vals[i]:
                pbest[i] = population[i].copy()
                pbest_vals[i] = curr_val
        gbest_idx = np.argmin(pbest_vals)
        gbest = pbest[gbest_idx].copy()
        gbest_val = pbest_vals[gbest_idx]
        if gbest_val < 0.3807:
            break

    return (gbest, gbest_val, n_points)
