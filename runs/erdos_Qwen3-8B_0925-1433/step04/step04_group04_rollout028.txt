import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Initialize population
    population_size = 50
    particles = []
    velocities = []
    pbest = []
    # Initial population based on initial_h_values
    for _ in range(population_size):
        h = initial_h_values.copy()
        # Add noise
        h += np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        # Scale to meet sum constraint
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h = np.clip(h * scale, 0, 1)
        particles.append(h)
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())

    gbest = particles[0].copy()
    best_c5 = compute_c5(gbest, dx, n_points)

    start_time = time.time()
    omega = 0.8
    c1 = 1.5
    c2 = 1.5
    max_velocity = 0.5

    while time.time() - start_time < budget_s:
        current_scores = []
        for i in range(population_size):
            h = particles[i]
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            current_scores.append(current_c5)

        # Update pbest and gbest
        for i in range(population_size):
            current_c5 = current_scores[i]
            pbest_h = pbest[i]
            pbest_corr = np.correlate(pbest_h, 1 - pbest_h, mode='full')
            pbest_c5 = np.max(pbest_corr) * dx
            if current_c5 < pbest_c5:
                pbest[i] = particles[i].copy()
            if current_c5 < best_c5:
                best_c5 = current_c5
                gbest = particles[i].copy()

        # Update velocities and positions
        for i in range(population_size):
            # Retrieve position and velocity
            pos = particles[i]
            vel = velocities[i]
            # Get pbest and gbest
            # Update velocity
            r1 = np.random.rand()

            pos = np.clip(pos * scale, 0, 1)
            particles[i] = pos

        # Check if target is met
        if best_c5 < 0.3807:
            break

        # Early stopping if no improvement for many iterations
        if np.allclose(best_c5, best_c5, atol=1e-6):  # Not sure about this line
            break

    return (gbest, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    population_size = 50
    particles = []
    velocities = []
    pbest = []
    for _ in range(population_size):
        h = initial_h_values.copy()
        noise = np.random.normal(0, 0.15, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h = np.clip(h * scale, 0, 1)
        particles.append(h)
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())

    gbest = particles[0].copy()
    best_c5 = compute_c5(gbest, dx, n_points)

    start_time = time.time()
    omega = 0.8
    c1 = 1.5
    c2 = 1.5
    max_velocity = 0.5

    while time.time() - start_time < budget_s:
        current_scores = []
        for i in range(population_size):
            h = particles[i]
            corr = np.correlate(h, 1 - h, mode='full')
            current_scores.append(np.max(corr) * dx)

        for i in range(population_size):
            current_c5 = current_scores[i]
            pbest_h = pbest[i]
            pbest_corr = np.correlate(pbest_h, 1 - pbest_h, mode='full')
            pbest_c5 = np.max(pbest_corr) * dx
            if current_c5 < pbest_c5:
                pbest[i] = particles[i].copy()
            if current_c5 < best_c5:
                best_c5 = current_c5
                gbest = particles[i].copy()

        for i in range(population_size):
            pos = particles[i]
            vel = velocities[i]
            r1 = np.random.rand()
            r2 = np.random.rand()
            vel = omega * vel + c1 * r1 * (pbest[i] - pos) + c2 * r2 * (gbest - pos)
            vel = np.clip(vel, -max_velocity, max_velocity)
            velocities[i] = vel
            pos += vel
            pos = np.clip(pos, 0, 1)
            sum_h = np.sum(pos)
            scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
            pos = np.clip(pos * scale, 0, 1)
            particles[i] = pos

        if best_c5 < 0.3807:
            break

    return (gbest, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
