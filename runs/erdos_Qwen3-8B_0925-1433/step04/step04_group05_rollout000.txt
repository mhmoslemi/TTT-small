from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    def scale_h(h):

        if sum_h == 0:
            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / sum_h, 0, 1)

    # Generate structured candidates
    candidates = []
    # Binary sequences with exactly n_points//2 ones
    for _ in range(10):
        b = np.random.choice([0, 1], size=n_points)
        if np.sum(b) != n_points // 2:
            b[np.random.choice(n_points, size=n_points - np.sum(b), replace=False)] = 1
        candidates.append(scale_h(b))
    # Periodic functions
    freqs = np.logspace(0.1, 1.5, 10)
    for f in freqs:
        h = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) * f / n_points)
        candidates.append(scale_h(h))
    # Stepped functions
    for _ in range(10):
        split = np.random.choice(n_points, size=5, replace=False)
        split = np.sort(np.concatenate((split, [0, n_points])))
        val = np.random.choice([0, 1], size=6)
        h = np.zeros(n_points)
        for i in range(6):
            h[split[i]:split[i+1]] = val[i]
        candidates.append(scale_h(h))
    # Mutations of initial_h_values
    for _ in range(5):
        h = initial_h_values + np.random.normal(0, 0.02, n_points)
        candidates.append(scale_h(h))
    # Add initial_h_values
    candidates.append(scale_h(initial_h_values))

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        h = scale_h(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    particles = candidates[:]
    best_c5 = min(compute_c5(p) for p in particles)
    best_h = particles[np.argmin([compute_c5(p) for p in particles])].copy()

    start_time = time.time()
    for _ in range(150):
        if time.time() - start_time > budget_s:

        c5_values = [compute_c5(p) for p in particles]
        idxs = np.argsort(c5_values)
        elites = [particles[i] for i in idxs[:20]]
        new_particles = []
        # Crossover
        for i in range(0, len(elites), 2):
            if i + 1 < len(elites):
                p1 = elites[i]

                child = p1 * np.random.rand() + p2 * (1 - np.random.rand())

        top_idx = idxs[:10]
