            bounds = [(0, 1) for _ in range(n_points)]

                res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
                if res.success:
                    h = res.x
                    sum_h = np.sum(h)
                    h = np.clip(h * (target_sum / sum_h), 0, 1)
                    corr = np.correlate(h, 1 - h, mode='full')
                    c5 = np.max(corr) * dx
                    if c5 < fitness[idx]:
                        population[idx] = h
                        fitness[idx] = c5
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = h
            except:
                pass
    return (best_h, best_c5, n_points)
