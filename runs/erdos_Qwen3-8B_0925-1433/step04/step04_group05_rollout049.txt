from scipy.optimize import minimize
import time

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial population

    binary = np.random.choice([0, 1], size=n_points)
    population.append(scale_h(binary))

        freq = np.random.uniform(0.1, 0.9)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) * freq)
        population.append(scale_h(h))
    for _ in range(3):
        segs = np.random.randint(1, 3, size=n_points // 5)
        h = np.random.choice([0, 1], size=n_points)
        for i in range(1, len(segs)):
            h[sum(segs[:i]):sum(segs[:i+1])] = np.random.choice([0, 1], size=sum(segs[i]))
        population.append(scale_h(h))

    def obj(x):
        h = scale_h(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        evaluations = [compute_c5(p) for p in population]
        elites = np.argsort(evaluations)[:5]
        new_pop = []
        for i in range(5):
            for j in range(i+1, 5):
                child = population[elites[i]] * np.random.rand() + population[elites[j]] * (1 - np.random.rand())
                new_pop.append(scale_h(child))
        for p in population:
            noise = np.random.normal(0, 0.05, n_points)
            new_pop.append(scale_h(p + noise))
        population = [population[i] for i in elites] + new_pop
        for idx in elites:
            h = population[idx].copy()
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(obj, h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
