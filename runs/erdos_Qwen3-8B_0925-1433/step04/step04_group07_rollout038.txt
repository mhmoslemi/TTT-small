import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    h = h * (target_sum / sum_h) if sum_h != 0 else np.full(n_points, 0.5)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Find worst lag
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        
        # Adjust contributing points
        contributions = []
        for i in range(n_points):
            j = (i + lag) % n_points
            contributions.append(h[i] * (1 - h[j]))
        contrib = np.array(contributions)
        top_indices = np.argsort(contrib)[::-1][:5]
        for i in top_indices:
            j = (i + lag) % n_points
            h[i] = np.clip(h[i] - 0.03 * h[i], 0, 1)
            h[j] = np.clip(h[j] - 0.03 * h[j], 0, 1)
        
        # Normalize
        sum_h = np.sum(h)
        scale = target_sum / sum_h if sum_h != 0 else 1.0
        h = np.clip(h * scale, 0, 1)
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
