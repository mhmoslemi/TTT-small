        max_corr = np.max(corr)
        if max_corr * dx <= 0.3807:
            return (best_h, max_corr * dx, n_points)

        # Find the index of maximum in corr
        max_corr_idx = np.argmax(corr)

        # Define optimization problem
        def objective(x):
            x_clipped = np.clip(x, 0, 1)
            sum_x = np.sum(x_clipped)
            if sum_x == 0:
                return 1e6
            scale = (n_points / 2) / sum_x
            x_scaled = np.clip(x_clipped * scale, 0, 1)
            shifted = 1 - x_scaled
            corr_optimized = np.correlate(x_scaled, shifted, mode='full')
            return corr_optimized[max_corr_idx] * dx

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}]
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if result.success:
                h_new = result.x
                h_new = np.clip(h_new, 0, 1)
                sum_h_new = np.sum(h_new)
                if sum_h_new != 0:
                    scale = (n_points / 2) / sum_h_new
                    h_new = np.clip(h_new * scale, 0, 1)
                else:
                    h_new = np.ones(n_points) * 0.5
                h = h_new
                current_c5 = compute_c5(h)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                if best_c5 < 0.3807:
                    return (best_h, best_c5, n_points)
        except:
            pass

    return (best_h, best_c5, n_points)
