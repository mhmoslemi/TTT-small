import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def scale_h(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / sum_h, 0, 1)

    # Generate initial population with structured candidates
    particles = [initial_h_values.copy()]
    for _ in range(3):
        # Binary patterns
        h_binary = np.random.choice([0, 1], size=n_points)
        h_binary = scale_h(h_binary)
        particles.append(h_binary)
        # Periodic functions
        freq = np.random.uniform(1, 2)
        phase = np.random.uniform(0, 2 * np.pi)
        h_period = 0.5 + 0.5 * np.sin(2 * np.pi * freq * np.arange(n_points) / n_points + phase)
        h_period = scale_h(h_period)
        particles.append(h_period)
        # Piecewise constant
        segments = np.random.randint(5, 10)
        indices = np.random.choice(n_points, size=segments, replace=False)
        indices.sort()
        h_pw = np.zeros(n_points)
        for i in range(segments):
            if i % 2 == 0:
                h_pw[indices[i]] = 1
            else:
                h_pw[indices[i]] = 0
        h_pw = scale_h(h_pw)
        particles.append(h_pw)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        h = scale_h(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_c5 = min(compute_c5(p) for p in particles)
    best_h = particles[np.argmin([compute_c5(p) for p in particles])].copy()

    start_time = time.time()
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        c5_values = [compute_c5(p) for p in particles]
        elite_idx = np.argsort(c5_values)[:10]
        elites = [particles[i] for i in elite_idx]
        new_particles = []
        for i in range(len(elites) // 2):
            parent1 = elites[2*i]
            parent2 = elites[2*i + 1]
            child = parent1 * np.random.rand() + parent2 * (1 - np.random.rand())
            new_particles.append(scale_h(child))
        for p in elites:
            noise = np.random.normal(0, 0.1, n_points)
            mut = p + noise
            new_particles.append(scale_h(mut))
        particles = elites + new_particles
        top_idx = np.argsort(c5_values)[:5]
        for idx in top_idx:
            h_candidate = particles[idx].copy()
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(obj, h_candidate, bounds=bounds, constraints=cons, method='L-BFGS-B', tol=1e-6)
            if res.success:
                particles[idx] = res.x
            else:
                particles[idx] = scale_h(h_candidate)
        current_best = min(c5_values)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = particles[np.argmin(c5_values)].copy()
        if best_c5 < 0.3807:
            break
    return (best_h, best_c5, n_points)
