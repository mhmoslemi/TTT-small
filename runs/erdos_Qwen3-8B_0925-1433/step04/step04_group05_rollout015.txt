def run(seed=42, budget_s=600):

        k_max = np.argmax(corr)
        current_c5 = corr[k_max] * dx

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        if best_c5 < 0.3807:
            break

        deltas = 0.001
