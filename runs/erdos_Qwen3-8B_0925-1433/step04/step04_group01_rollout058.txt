    np.random.seed(seed)

    dx = 2.0 / n_points

    start_time = time.time()
    T = 1.0
    cooling_rate = 0.995
    max_iter = int(budget_s / 0.1)  # ~6000 iterations

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:

        perturbation = np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h + perturbation, 0, 1)
        h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
