def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Scale initial_h_values
    initial_h = initial_h_values.copy()
    sum_initial = np.sum(initial_h)
    if sum_initial != 0:
        scale = (n_points / 2) / sum_initial
        initial_h = np.clip(initial_h * scale, 0, 1)
    else:
        initial_h = np.ones(n_points) * 0.5
    h_current = initial_h.copy()
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current)

    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Compute cross-correlation
        shifted = 1 - h_current
        corr = np.correlate(h_current, shifted, mode='full')
        max_corr_val = np.max(corr)
        j_max = np.argmax(corr)

        # Define objective function
        def objective(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            if sum_x == 0:
                return 1e6
            scale = (n_points / 2) / sum_x
            x = np.clip(x * scale, 0, 1)
            shifted = 1 - x
            corr = np.correlate(x, shifted, mode='full')
            val = corr[j_max]
            return val * dx

        # Define constraint
        def constraint(x):
            x = np.clip(x, 0, 1)
            return np.sum(x) - n_points / 2

        cons = [{'type': 'eq', 'fun': constraint}]
        bounds = [(0, 1) for _ in range(n_points)]

        # Perform optimization
        res = minimize(objective, h_current, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)

        if res.success:
            h_candidate = res.x
            h_candidate = np.clip(h_candidate, 0, 1)
            sum_candidate = np.sum(h_candidate)
            if sum_candidate != 0:
                scale = (n_points / 2) / sum_candidate
                h_candidate = np.clip(h_candidate * scale, 0, 1)
            else:
                h_candidate = np.ones(n_points) * 0.5
            h_current = h_candidate.copy()
            current_c5 = compute_c5(h_current)
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
                if best_c5 < 0.3807:
                    return (best_h, best_c5, n_points)
        else:
            h_current = best_h.copy()

    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n_points)

def compute_c5(h):
    shifted = 1 - h
    corr = np.correlate(h, shifted, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
