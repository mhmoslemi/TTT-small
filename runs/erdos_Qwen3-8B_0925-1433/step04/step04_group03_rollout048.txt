from scipy.optimize import minimize

def run(seed=42, budget_s=600):

    h0 = np.clip(h0 * scale, 0, 1)
    
    # Generate diverse initial candidates
    def sine_wave():
        h = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
        return np.clip(h * (n_points / 2) / np.sum(h), 0, 1)
    
    def binary_mut():
        h = np.random.choice([0, 1], size=n_points)
        sum_h = np.sum(h)
        return np.clip(h * (n_points / 2) / sum_h, 0, 1) if sum_h else np.zeros(n_points)
    
    def regular_spacing(m=15):
        h = np.zeros(n_points)
        for i in range(m):
            pos = int(i * n_points / m)
            h[pos] = 1.0
        return np.clip(h * (n_points / 2) / np.sum(h), 0, 1)

        return np.max(corr) * dx
    
    best_h = h0
