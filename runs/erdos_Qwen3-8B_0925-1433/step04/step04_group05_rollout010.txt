import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def scale_h(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / sum_h, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        h = scale_h(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_binary(n_points, shift):
        h = np.zeros(n_points)
        idxs = np.arange(n_points)
        for i in idxs:
            if (i + shift) % 3 == 0:
                h[i] = 1.0
        return h

    def generate_periodic(n_points, freq):
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x)
        return h

    def generate_greedy(n_points):
        h = np.zeros(n_points)
        used = []
        for _ in range(n_points // 2):
            best = -1
            pos = 0
            for i in range(n_points):
                d = 0
                for u in used:
                    d += abs(i - u)
                if d > best:
                    best = d
                    pos = i
            h[pos] = 1.0
            used.append(pos)
        return h

    candidates = [initial_h_values.copy()]
    for shift in range(3):
        candidates.append(generate_binary(n_points, shift))
    for freq in [1.0, 0.5, 2.0]:
        candidates.append(generate_periodic(n_points, freq))
    for _ in range(3):
        candidates.append(generate_greedy(n_points))

    def optimize_candidate(h):
        bounds = [(0, 1)] * n_points
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        res = minimize(obj, h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
        return res.x if res.success else scale_h(h)

    timespent = 0
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)

    for idx in range(len(candidates)):
        if timespent >= budget_s:
            break
        h = candidates[idx].copy()
        h = scale_h(h)
        h = optimize_candidate(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        timespent += 10

    for _ in range(100):
        if timespent >= budget_s:
            break
        h = best_h.copy()
        h += np.random.normal(0, 0.05, n_points)
        h = scale_h(h)
        h = optimize_candidate(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        timespent += 5

    return (best_h, best_c5, n_points)
