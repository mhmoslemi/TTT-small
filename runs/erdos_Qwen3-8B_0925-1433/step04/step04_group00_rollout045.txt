            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            if sum_x == 0:
                return 1e6
            scale = (n_points / 2) / sum_x
            x = np.clip(x * scale, 0, 1)
            shifted = 1 - x
            corr_opt = np.correlate(x, shifted, mode='full')
            return corr_opt[max_index] * dx
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h_new = res.x
            h_new = np.clip(h_new, 0, 1)
            sum_h_new = np.sum(h_new)
            if sum_h_new != 0:
                scale_new = (n_points / 2) / sum_h_new
                h_new = np.clip(h_new * scale_new, 0, 1)
            else:
                h_new = np.ones(n_points) * 0.5
            new_c5 = compute_c5(h_new)
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                h = h_new
        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, n_points)
