import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()
    alpha = 0.05
    no_improvement = 0
    max_iter = int(budget_s / 0.1)  # ~1000 iterations in 600 seconds
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        worst_lag = np.argmax(corr)
        # Central difference gradient estimation
        h_perturbed = h.copy()
        h_perturbed += 1e-6
        corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
        grad = (corr_perturbed[worst_lag] - corr[worst_lag]) / 1e-6
        h -= alpha * grad
        # Projection
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 1e-6:
            h = h * target_sum / current_sum
        new_c5 = compute_c5(h, dx, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > 50:
                alpha *= 0.995
                no_improvement = 0
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
