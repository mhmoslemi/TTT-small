import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    h = h * target_sum / sum_h
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    
    start_time = time.time()
    step_size = 0.05
    max_iter = 1000
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        cross_corr = np.correlate(h, 1 - h, mode='full')
        worst_lag_index = np.argmax(cross_corr)
        lag_k = worst_lag_index - (n_points - 1)
        grad = np.zeros(n_points)
        for j in range(n_points):
            val1 = 0.0
            if 0 <= j + lag_k < n_points:
                val1 = 1 - h[j + lag_k]
            val2 = 0.0
            if 0 <= j - lag_k < n_points:
                val2 = -h[j - lag_k]
            grad[j] = val1 + val2
        h = h - step_size * grad
