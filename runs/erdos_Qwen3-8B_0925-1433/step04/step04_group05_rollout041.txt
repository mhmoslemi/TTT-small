import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def scale_h(h):
        s = np.sum(h)
        if s == 0:
            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / s, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initial population
    pop = [initial_h_values.copy()]
    for _ in range(5):
        h = np.random.choice([0, 1], size=n_points)
        pop.append(scale_h(h))
    freqs = np.random.uniform(1, 3, 10)
    for f in freqs:
        h = 0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points * f)
        pop.append(scale_h(h))
    for _ in range(5):
        splits = np.random.randint(1, n_points-1, 5)
        splits = np.sort(np.unique(np.concatenate([np.zeros(1), splits, np.array([n_points])])))
        h = np.zeros(n_points)
        for i in range(len(splits)-1):
            h[splits[i]:splits[i+1]] = np.random.choice([0, 1], size=splits[i+1]-splits[i])
        pop.append(scale_h(h))

    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        c5_values = [compute_c5(p) for p in pop]
        elite_idx = np.argsort(c5_values)[:10]
        elites = [pop[i] for i in elite_idx]
        
        new_pop = []
        for i in range(len(elites)//2):
            p1, p2 = elites[2*i], elites[2*i+1]
            new_pop.append(scale_h(p1 * np.random.rand() + p2 * (1 - np.random.rand())))
        for p in elites:
            new_pop.append(scale_h(p + np.random.normal(0, 0.1, n_points)))
        pop = elites + new_pop

        top_idx = np.argsort(c5_values)[:5]
        for idx in top_idx:
            h_candidate = pop[idx].copy()
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2})
            bounds = [(0,1)]*n_points
            res = minimize(lambda x: compute_c5(x), h_candidate, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
            if res.success:
                pop[idx] = res.x
            else:
                pop[idx] = scale_h(h_candidate)
        
        current_best = min(c5_values)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = pop[np.argmin(c5_values)].copy()
            if best_c5 < 0.3807:
                break

    return (best_h, best_c5, n_points)
