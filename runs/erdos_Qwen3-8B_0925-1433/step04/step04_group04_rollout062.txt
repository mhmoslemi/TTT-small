    start_time = time.time()
    iterations = 0

    while time.time() - start_time < budget_s:
        iterations += 1
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        if best_c5 < 0.3807:
            break

        k = np.argmax(corr)

        def obj(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            scale = (n_points / 2) / sum_x if sum_x != 0 else 1.0
            x = np.clip(x * scale, 0, 1)
            corr_obj = np.correlate(x, 1 - x, mode='full')
            return corr_obj[k] * dx

        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        res = minimize(obj, h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
        if res.success:
            h = res.x
        else:
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
            h = np.clip(h * scale, 0, 1)

        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h = np.clip(h * scale, 0, 1)

    return (best_h, best_c5, n_points)
