import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.zeros(45)
        h[::2] = 1.0
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_scale = target_sum / np.sum(h)
    h = np.clip(h * h_scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)

    start_time = time.time()
    Perturb = True

    while time.time() - start_time < budget_s:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        top_lags = np.argsort(corr)[-5:]
        current_max = np.max(corr[top_lags]) * dx

        def obj(x):
            h = x
            corr = np.correlate(h, 1 - h, mode='full')
            top_lags = np.argsort(corr)[-5:]
            return np.max(corr[top_lags]) * dx

        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

        try:
            res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5)
            if res.success:
                best_h = res.x
                best_h = np.clip(best_h, 0, 1)
                sum_h = np.sum(best_h)
                if sum_h != 0:
                    best_h *= target_sum / sum_h
                current_c5 = compute_c5(best_h, dx, n_points)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                if best_c5 < 0.3807:
                    break
        except:
            pass

        if Perturb:
            noise = np.random.normal(0, 0.1, n_points)
            best_h = np.clip(best_h + noise, 0, 1)
            sum_h = np.sum(best_h)
            if sum_h != 0:
                best_h *= target_sum / sum_h
            Perturb = False

        if best_c5 < 0.3807:
            break

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
