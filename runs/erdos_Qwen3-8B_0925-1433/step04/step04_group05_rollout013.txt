import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    h_candidate = h_initial if compute_c5(h_initial) < compute_c5(h_binary) else h_binary

        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = compute_c5(h_candidate)
    best_h = h_candidate.copy()

    start_time = time.time()
