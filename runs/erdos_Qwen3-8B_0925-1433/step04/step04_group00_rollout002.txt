    best_c5 = compute_c5(h, dx)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        idx = np.argmax(corr)
        
        def objective(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            if sum_x == 0:
                return 1e6
            scale = (n_points / 2) / sum_x
            x = np.clip(x * scale, 0, 1)
            shifted = 1 - x
            corr_i = np.correlate(x, shifted, mode='full')
            return corr_i[idx] * dx
        
        def constraint(x):
            return np.sum(x) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraint}]
        
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        
        if res.success:
            h_new = res.x
            h_new = np.clip(h_new, 0, 1)
            sum_h_new = np.sum(h_new)
            if sum_h_new != 0:
                scale = (n_points / 2) / sum_h_new
                h_new = np.clip(h_new * scale, 0, 1)
            else:
                h_new = np.ones(n_points) * 0.5
            current_c5 = compute_c5(h_new, dx)
            if current_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = current_c5
        else:
            h = np.random.uniform(0, 1, n_points)
            sum_h = np.sum(h)
            if sum_h != 0:
                scale = (n_points / 2) / sum_h
                h = np.clip(h * scale, 0, 1)
            else:
                h = np.ones(n_points) * 0.5
        
        if best_c5 < 0.3807:
            break
    
    return (best_h, best_c5, n_points)
