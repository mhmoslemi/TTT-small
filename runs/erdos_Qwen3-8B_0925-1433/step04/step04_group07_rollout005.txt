            current_sum = np.sum(h)
            surplus = current_sum - (n_points / 2)
            if surplus != 0:
                idx = np.argsort(np.abs(grad))
                for i in idx:
                    if surplus == 0:
                        break
                    adj = surplus / n_points
                    h[i] += adj
                    surplus -= adj
                h = np.clip(h, 0, 1)
            new_c5 = compute_c5(h, dx, n_points)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    
    return (best_h, best_c5, n_points)
