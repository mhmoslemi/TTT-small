import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

        if sum_h == 0:
            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / sum_h, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def obj(x):
        h = scale_h(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    initial = scale_h(initial_h_values.copy())
    particles = [initial]
    
    for _ in range(10):
        particles.append(scale_h(np.random.choice([0, 1], size=n_points)))
        particles.append(scale_h(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points)))
        particles.append(scale_h(np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])))
        particles.append(scale_h(np.concatenate([np.zeros(n_points//2), np.ones(n_points//2)])))
        particles.append(scale_h(np.random.normal(0.5, 0.2, n_points)))
    
    best_c5 = compute_c5(initial)
    best_h = initial.copy()

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        c5_values = [compute_c5(p) for p in particles]
        elite_idx = np.argsort(c5_values)[:10]
        elites = [particles[i] for i in elite_idx]

            p1 = elites[2*i]

            new_particles.append(scale_h(child))
        for p in elites:
            noise = np.random.normal(0, 0.05, n_points)
            new_particles.append(scale_h(p + noise))
        
        particles = elites + new_particles
        top_idx = np.argsort(c5_values)[:5]
        for idx in top_idx:
            h_candidate = particles[idx].copy()
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
            bounds = [(0, 1) for _ in range(n_points)]
