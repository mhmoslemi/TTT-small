import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Normalize initial_h_values
    sum_h = np.sum(initial_h_values)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h = initial_h_values * scale
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()

    for _ in range(100):  # Number of optimization steps
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        
        def objective(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            scale_x = (n_points / 2) / sum_x if sum_x != 0 else 1.0
            x = np.clip(x * scale_x, 0, 1)
            corr = np.correlate(x, 1 - x, mode='full')
            overlap = corr[max_idx] * dx
            return overlap

        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0, 1)] * n_points
        try:
            res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = res.x
                current_c5 = compute_c5(h, dx, n_points)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
