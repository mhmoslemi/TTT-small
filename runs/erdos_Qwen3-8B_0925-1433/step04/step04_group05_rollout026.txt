import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def scale_h(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / sum_h, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_binary_slices(m):
        h = np.zeros(n_points)
        for i in range(m):
            h[i::m] = 1
        h = scale_h(h)
        return h

    def generate_periodic():
        h = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
        h = scale_h(h)
        return h

    def generate_greedy():
        h = np.zeros(n_points)
        h[::2] = 1
        h = scale_h(h)
        return h

    candidates = []
    for _ in range(5):
        candidates.append(generate_binary_slices(4))
        candidates.append(generate_periodic())
        candidates.append(generate_greedy())

    def obj(x):
        h = scale_h(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_c5 = min(compute_c5(c) for c in candidates)
    best_h = candidates[np.argmin([compute_c5(c) for c in candidates])].copy()

    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        c5_values = [compute_c5(c) for c in candidates]
        top_idx = np.argsort(c5_values)[:5]
        for idx in top_idx:
            h_candidate = candidates[idx].copy()
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(obj, h_candidate, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
            if res.success:
                candidates[idx] = res.x
            else:
                candidates[idx] = scale_h(h_candidate)
        current_best = min(c5_values)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = candidates[np.argmin(c5_values)].copy()
        if best_c5 < 0.3807:
            break
    return (best_h, best_c5, n_points)
