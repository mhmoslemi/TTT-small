import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()

    def objective(x):
        h_new = x
        corr = np.correlate(h_new, 1 - h_new, mode='full')
        top_vals = corr[np.argsort(corr)[::-1][:5]]
        return np.max(top_vals) * dx

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint})
    while time.time() - start_time < budget_s:
        try:
            res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
            if res.success:
                h = np.clip(res.x, 0, 1)
                sum_h = np.sum(h)
                scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
                h = np.clip(h * scale, 0, 1)
        except:
            pass

        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 < 0.3807:
            break

        if (time.time() - start_time) > 300:
            noise = np.random.normal(0, 0.1, n_points)
            h = np.clip(h + noise, 0, 1)
            sum_h = np.sum(h)
            scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
            h = np.clip(h * scale, 0, 1)

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
