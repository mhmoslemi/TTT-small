def run(seed=42, budget_s=600):
    np.random.seed(seed)

    n_points = len(initial_h_values)

    population_size = 50

    for i in range(population_size):
        positions[i] = np.clip(positions[i], 0, 1)
        sum_p = np.sum(positions[i])

        positions[i] = np.clip(positions[i] * scale, 0, 1)
    velocities = [np.zeros(n_points) for _ in range(population_size)]

    gbest = positions[0].copy()
    best_c5 = compute_c5(gbest, dx, n_points)
    for i in range(population_size):
        current_c5 = compute_c5(positions[i], dx, n_points)
        if current_c5 < best_c5:
            gbest = positions[i].copy()
            best_c5 = current_c5

    start_time = time.time()
    iterations = 0
    while time.time() - start_time < budget_s:

            r1, r2 = np.random.random(), np.random.random()

            positions[i] += velocities[i]
            positions[i] = np.clip(positions[i], 0, 1)
            sum_p = np.sum(positions[i])
            scale = (n_points / 2) / sum_p if sum_p != 0 else 1.0
            positions[i] = np.clip(positions[i] * scale, 0, 1)
            current_c5 = compute_c5(positions[i], dx, n_points)
            if current_c5 < compute_c5(pbest[i], dx, n_points):
                pbest[i] = positions[i].copy()
                if current_c5 < best_c5:

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
