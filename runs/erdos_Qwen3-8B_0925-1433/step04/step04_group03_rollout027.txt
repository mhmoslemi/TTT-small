        # Find the worst shift
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        worst_k = np.argmax(corr)
        # Define objective to minimize overlap at worst_k
        def obj(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            scale = (n_points / 2) / sum_x if sum_x != 0 else 1.0
            x = np.clip(x * scale, 0, 1)
            corr = np.correlate(x, 1 - x, mode='full')
            return corr[worst_k] * dx
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2})
        bounds = [(0, 1)] * n_points
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                best_h = res.x
                best_c5 = compute_c5(best_h, dx, n_points)
        except:
            # Fallback: random perturbation
            perturb = np.random.normal(0, 0.05, n_points)
            best_h = np.clip(best_h + perturb, 0, 1)
            sum_h = np.sum(best_h)
            scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
            best_h = np.clip(best_h * scale, 0, 1)
            best_c5 = compute_c5(best_h, dx, n_points)
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
