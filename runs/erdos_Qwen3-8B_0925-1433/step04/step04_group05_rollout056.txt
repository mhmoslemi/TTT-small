import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def scale_h(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / sum_h, 0, 1)

    # Generate structured candidates
    def generate_periodic():
        freqs = np.random.uniform(0.1, 1.5, size=10)
        amps = np.random.uniform(0.1, 0.4, size=10)
        phase = np.random.uniform(0, 2*np.pi, size=10)
        res = np.zeros(n_points)
        for i in range(10):
            res += amps[i] * np.sin(2*np.pi*freqs[i]*np.arange(n_points)/n_points + phase[i])
        res = 0.5 + 0.5 * res / np.max(res)
        return scale_h(res)
    
    def generate_binary():
        while True:
            h = np.random.choice([0, 1], size=n_points)
            if np.abs(np.sum(h) - n_points/2) < 1e-3:
                return scale_h(h)
    
    base_pop = [initial_h_values.copy()]
    for _ in range(5):
        base_pop.append(generate_periodic())
        base_pop.append(generate_binary())
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Evolution loop
    particles = base_pop[:30]
    best_h = particles[0].copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()
    
    for _ in range(60):
        if time.time() - start_time > budget_s:
            break
        c5_values = [compute_c5(p) for p in particles]
        elites = np.argsort(c5_values)[:10]
        
        # Crossover
        new_p = []
        for i in range(0, len(elites), 2):
            p1 = particles[elites[i]]
            p2 = particles[elites[i+1]]
            child = p1 * np.random.rand() + p2 * (1 - np.random.rand())
            new_p.append(scale_h(child))
        
        # Mutation
        for idx in elites:
            p = particles[idx]
            noise = np.random.normal(0, 0.08, n_points)
            mut = p + noise
            new_p.append(scale_h(mut))
        
        # Local optimization
        top_idx = np.argsort(c5_values)[:5]
        for idx in top_idx:
            h_candidate = particles[idx].copy()
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(lambda x: compute_c5(x), h_candidate, bounds=bounds, constraints=cons, method='L-BFGS-B', tol=1e-6)
            if res.success:
                particles[idx] = res.x
            else:
                particles[idx] = scale_h(h_candidate)
        
        # Update best
        for p in particles:
            current_c5 = compute_c5(p)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = p.copy()
            if best_c5 < 0.3807:
                break
        else:
            continue
        break
    
    return (best_h, best_c5, n_points)
