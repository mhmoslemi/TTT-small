import numpy as np
from scipy.optimize import minimize, basinhopping
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        h = h * (n_points / 2) / sum_h
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)

    start_time = time.time()
    iterations = 0
    n_iter = 200  # Increased number of iterations for better exploration
    perturbation_scale = 0.1  # Control magnitude of random perturbations

    def objective(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if sum_x != 0:
            x = x * (n_points / 2) / sum_x
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})

    while time.time() - start_time < budget_s and iterations < n_iter:
        iterations += 1
        # Perturb current h
        perturb = np.random.normal(0, perturbation_scale, n_points)
        h_perturb = h + perturb
        h_perturb = np.clip(h_perturb, 0, 1)
        if np.sum(h_perturb) != 0:
            h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)

        # Local optimization
        res = minimize(objective, h_perturb, bounds=bounds, constraints=cons, method='L-BFGS-B', tol=1e-6)
        if res.success:
            h = res.x
        else:
            h = h_perturb

        # Evaluate and update best solution
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        if best_c5 < 0.3807:
            break

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
