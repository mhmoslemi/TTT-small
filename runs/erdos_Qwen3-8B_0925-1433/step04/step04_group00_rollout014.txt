import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    h_initial = initial_h_values.copy()
    sum_h = np.sum(h_initial)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h_initial = np.clip(h_initial * scale, 0, 1)
    else:
        h_initial = np.ones(n_points) * 0.5
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)

    def objective(h_candidate):
        h = np.clip(h_candidate, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            scale = 1.0
        else:
            scale = (n_points / 2) / sum_h
        h_scaled = np.clip(h * scale, 0, 1)
        shifted = 1 - h_scaled
        corr = np.correlate(h_scaled, shifted, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()

    try:
        result = differential_evolution(
            objective, bounds, popsize=10, strategy='best1bin', tol=1e-6, maxiter=100
        )
        best_h_candidate = result.x
        current_c5 = compute_c5(best_h_candidate)
        if current_c5 < best_c5:
            best_h = best_h_candidate
            best_c5 = current_c5
    except:
        pass

    for _ in range(20):
        if time.time() - start_time >= budget_s:
            break
        shifted = 1 - best_h
        corr = np.correlate(best_h, shifted, mode='full')
        max_corr_idx = np.argmax(corr)
        max_corr_val = corr[max_corr_idx]

        def local_obj(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            if sum_x == 0:
                return 1e6
            scale = (n_points / 2) / sum_x
            x_scaled = np.clip(x * scale, 0, 1)
            shifted = 1 - x_scaled
            corr = np.correlate(x_scaled, shifted, mode='full')
            return corr[max_corr_idx] * dx

        def local_cons(x):

            return np.sum(x) - n_points / 2

        cons = [{'type': 'eq', 'fun': local_cons}]
        res = minimize(local_obj, best_h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons, tol=1e-6)
        if res.success:
            best_h = res.x
            best_c5 = compute_c5(best_h)
        else:
            break

    return (best_h, best_c5, n_points)
