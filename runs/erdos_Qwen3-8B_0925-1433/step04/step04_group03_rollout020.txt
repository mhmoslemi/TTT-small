def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    if not np.isclose(sum_h, n_points / 2):
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)

    start_time = time.time()
    max_iterations = 200
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        ones_idx = np.where(h > 0.5)[0]
        zeros_idx = np.where(h < 0.5)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h, dx, n_points)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
        else:
            h[i], h[j] = h[j], h[i]
    return (best_h, best_c5, n_points)
