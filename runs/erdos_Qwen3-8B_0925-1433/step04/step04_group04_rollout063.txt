from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    global initial_h_values

    h = initial_h_values.copy()

    scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)

    start_time = time.time()
    iterations = 0

    while time.time() - start_time < budget_s and iterations < 200:
        iterations += 1

        current_max = np.max(corr) * dx

        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()

        if best_c5 < 0.3807:
            break

        k_max = np.argmax(corr)
        if k_max >= len(h) - 1:

        else:
            k_max_adj = k_max

        def obj(x):

            x = np.clip(x * scale, 0, 1)

        bounds = [(0, 1) for _ in range(n_points)]
