import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def scale_h(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / sum_h, 0, 1)

    # Generate structured initial candidates
    initial_candidates = [initial_h_values.copy()]
    binary_candidates = []
    for _ in range(5):
        h = np.random.choice([0, 1], size=n_points)
        while np.sum(h) != n_points // 2:
            h = np.random.choice([0, 1], size=n_points)
        binary_candidates.append(scale_h(h))
    initial_candidates.extend(binary_candidates)

    periodic_candidates = []
    for _ in range(5):
        freq = np.random.uniform(0.5, 2.0)
        phase = np.random.uniform(0, 1)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * np.arange(n_points) / n_points + phase * np.pi)
        periodic_candidates.append(scale_h(h))
    initial_candidates.extend(periodic_candidates)

    piecewise_candidates = []
    for _ in range(5):
        split = np.random.randint(1, n_points - 1, size=4)
        split = np.sort(np.unique(np.concatenate((split, [0, n_points]))))
        h = np.zeros(n_points)
        for i in range(len(split) - 1):
            mid = (split[i] + split[i+1]) // 2
            h[mid] = 1
        while np.sum(h) != n_points // 2:
            h = np.random.choice([0, 1], size=n_points)
        piecewise_candidates.append(scale_h(h))
    initial_candidates.extend(piecewise_candidates)

    particles = initial_candidates

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_c5 = min(compute_c5(p) for p in particles)
    best_h = particles[np.argmin([compute_c5(p) for p in particles])].copy()

    start_time = time.time()
    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        c5_values = [compute_c5(p) for p in particles]
        elite_idx = np.argsort(c5_values)[:10]
        elites = [particles[i] for i in elite_idx]

        new_particles = []
        for i in range(len(elites)//2):
            parent1 = elites[2*i]
            parent2 = elites[2*i+1]
            child = parent1 * np.random.rand() + parent2 * (1 - np.random.rand())
            new_particles.append(scale_h(child))
        for p in elites:
            noise = np.random.normal(0, 0.15 if _ < 50 else 0.05, n_points)
            mut = p + noise
            new_particles.append(scale_h(mut))
        particles = elites + new_particles

        top_idx = np.argsort(c5_values)[:5]
        for idx in top_idx:
            h_candidate = particles[idx].copy()
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(lambda x: compute_c5(x), h_candidate, bounds=bounds, constraints=cons, method='L-BFGS-B', tol=1e-5)
            if res.success:
                particles[idx] = res.x
            else:
                particles[idx] = scale_h(h_candidate)

        current_best = min(c5_values)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = particles[np.argmin(c5_values)].copy()
        if best_c5 < 0.3807:
            break

    return (best_h, best_c5, n_points)
