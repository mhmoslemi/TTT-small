import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    h = initial_h_values.copy()
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)

    start_time = time.time()
    iter = 0
    max_iter = 150
    decay = 0.2

    while time.time() - start_time < budget_s and iter < max_iter:
        corr = np.correlate(h, 1 - h, mode='full')
        valid_corr = corr[1:-1]  # Remove edge cases
        sort_idx = np.argsort(valid_corr)[::-1]
        top_lags = sort_idx[:3]
        weights = [np.exp(-decay * i) for i in range(3)]
        weights = [w / np.sum(weights) for w in weights]
        obj = lambda x: np.dot(weights, corr[1 + top_lags]) * dx

        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = np.clip(res.x, 0, 1)
                h_sum = np.sum(h)
                h = np.clip(h * (n_points / 2) / h_sum, 0, 1)
        except:
            pass

        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 < 0.3807:
            break
        iter += 1

        if iter % 10 == 0:
            noise = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + noise, 0, 1)
            h_sum = np.sum(h)
            h = np.clip(h * (n_points / 2) / h_sum, 0, 1)

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
