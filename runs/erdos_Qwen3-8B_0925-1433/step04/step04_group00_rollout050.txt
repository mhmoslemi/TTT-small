import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    if 'initial_h_values' in locals():
        h = initial_h_values.copy()
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5

    n_points = len(h)
    dx = 2.0 / n_points

    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    sum_h = np.sum(h)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    best_h = h.copy()
    best_c5 = compute_c5(h)

    start_time = time.time()
    iterations = 0
    no_improvement = 0
    while time.time() - start_time < budget_s:
        iterations += 1
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        max_corr_index = np.argmax(corr)

        def objective(x):
            x = np.clip(x, 0, 1)
            corr = np.correlate(x, 1 - x, mode='full')
            return corr[max_corr_index] * dx

        def constraint(x):
            return np.sum(x) - n_points / 2

        cons = [{'type': 'eq', 'fun': constraint}]
        bounds = [(0, 1) for _ in range(n_points)]

        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h_new = res.x
            new_c5 = compute_c5(h_new)
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                no_improvement = 0
            else:
                no_improvement += 1
            h = h_new
        else:

            else:
                h = np.ones(n_points) * 0.5
            no_improvement = 0

    return (best_h, best_c5, n_points)
