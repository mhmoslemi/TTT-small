import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    h = initial_h_values.copy()
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)

    start_time = time.time()

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        top_indices = np.argsort(corr)[::-1][:5]
        weights = np.exp(-np.arange(1, 6))
        def obj(x):
            h_candidate = x
            corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            return np.dot(weights, corr_candidate[top_indices])
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = res.x
        except:
            pass
        
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h = np.clip(h * scale, 0, 1)
        
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 < 0.3807:
            break

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
