from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_c5 = 0.3807
    best_h = initial_h_values.copy()

    scale = (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h * scale, 0, 1)

    def fitness(h):
        return compute_c5(h, dx, n_points)

    pop_size = 50
    pop = [np.clip((initial_h_values * (n_points / 2) / np.sum(initial_h_values)), 0, 1)]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points) * (n_points / 2) / np.sum(np.random.rand(n_points))
        pop.append(np.clip(h, 0, 1))

    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitnesses = np.array([fitness(h) for h in pop])
        best_idx = np.argmin(fitnesses)
        curr_c5 = fitnesses[best_idx]

            best_h, best_c5 = pop[best_idx].copy(), curr_c5

                break

        new_pop = []
        for _ in range(pop_size):
            if np.random.rand() < 0.8:
                p1, p2 = np.random.choice(parents, 2, replace=False)

                child = np.concatenate([p1[:idx], p2[idx:]])
