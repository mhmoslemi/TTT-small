from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

        return np.max(corr) * dx

            return np.ones(n_points) * 0.5

            return 1e6

        corr = np.correlate(h, 1 - h, mode='full')
        return np.abs(corr[lag]) * dx
