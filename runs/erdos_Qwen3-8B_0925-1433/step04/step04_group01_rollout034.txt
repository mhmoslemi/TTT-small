import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    initial_sum = np.sum(initial_h_values)
    initial_scaled = initial_h_values * (target_sum / initial_sum)
    pop = [initial_scaled.copy() for _ in range(15)]
    best_h = initial_scaled.copy()
    best_c5 = compute_c5(best_h, dx, n_points)

    start_time = time.time()
    temp = 1.0
    cooling_rate = 0.995
    iterations = 0
    n_best = 5

    while time.time() - start_time < budget_s:
        iterations += 1
        new_pop = []
        for h in pop:
            noise = np.random.normal(0, 0.05, n_points)
            h_perturb = h + noise
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb_sum = np.sum(h_perturb)
            h_perturb = h_perturb * (target_sum / h_perturb_sum)
            new_pop.append(h_perturb)

        pop = new_pop
        fitness = [compute_c5(h, dx, n_points) for h in pop]
        top_idx = np.argsort(fitness)[:n_best]
        top_candidates = [pop[i] for i in top_idx]

        for h in top_candidates:
            bounds = [(0, 1)] * n_points
            def obj(x):
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * dx
            res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}], tol=1e-6, options={'maxiter': 100})
            refined = res.x
            refined_sum = np.sum(refined)
            refined = refined * (target_sum / refined_sum)
            curr = compute_c5(refined, dx, n_points)
            if curr < best_c5:
                best_c5 = curr
                best_h = refined.copy()
                print(f"New best C5: {best_c5}")

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
