import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def normalize(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.ones(n_points) * 0.5
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
        return h
    
    def objective(x):
        x = np.clip(x, 0, 1)
        x = normalize(x)
        shifted = 1 - x
        corr = np.correlate(x, shifted, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx + 1e-6 * (max_corr * dx) ** 2
    
    def constraint(x):
        x = np.clip(x, 0, 1)
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    
    h_initial = normalize(initial_h_values.copy())
    result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    best_h = result.x
    best_c5 = objective(best_h)
    
    def generate_initial_guess(n_points):
        x = np.linspace(0, 2, n_points)
        sine = np.sin(np.pi * x) + 1
        sine = (sine - np.min(sine)) / (np.max(sine) - np.min(sine))
        return sine
    
    for _ in range(5):
        if np.random.rand() < 0.5:
            h = initial_h_values.copy() + np.random.normal(0, 0.05, n_points)
        elif np.random.rand() < 0.5:
            h = generate_initial_guess(n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
        h = normalize(h)
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_c5 = objective(res.x)
            if current_c5 < best_c5:
                best_h = res.x
                best_c5 = current_c5
    
    return (best_h, best_c5, n_points)
