import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def compute_c5(h):
        shifted = 1.0 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx
    
    def optimize_candidate(h):
        def objective(x):
            h_ = x
            shifted = 1.0 - h_
            corr = np.correlate(h_, shifted, mode='full')
            return np.max(corr) * dx
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        return result.x if result.success else h
    
    # Generate initial population
    initial_population = [initial_h_values.copy()]
    x = np.linspace(0, 2, n_points)
    for freq in range(1, 9):
        for phase in [0, np.pi/2, np.pi, 3*np.pi/2]:
            h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x + phase)
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                scale = (n_points / 2) / sum_h
                h = np.clip(h * scale, 0, 1)
            else:
                h = np.ones(n_points) * 0.5
            initial_population.append(h)
    
    # Evaluate initial candidates
    candidate_c5 = [compute_c5(h) for h in initial_population]
    best_idx = np.argmin(candidate_c5)
    best_h = initial_population[best_idx]
    best_c5 = candidate_c5[best_idx]
    
    if best_c5 < 0.3807:
        return (best_h, best_c5, n_points)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        new_h = optimize_candidate(best_h)
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h
            if best_c5 < 0.3807:
                return (best_h, best_c5, n_points)
        # Early exit if optimization isn't improving
        if np.abs(new_c5 - best_c5) < 1e-5:
            break
    
    return (best_h, best_c5, n_points)
