                best_h = h.copy()
        
        if best_c5 < 0.3807:
            break
        
        worst_idx = np.argmax([np.max(np.correlate(c, 1 - c, mode='full')) * dx for c in candidates])
        h_to_optimize = candidates[worst_idx]
        corr = np.correlate(h_to_optimize, 1 - h_to_optimize, mode='full')
        max_k = np.argmax(corr)
        
        def objective(x):
            x = np.clip(x, 0, 1)
            sum_x = np.sum(x)
            scale_x = (n_points / 2) / sum_x if sum_x != 0 else 1.0
            x = np.clip(x * scale_x, 0, 1)
            corr_opt = np.correlate(x, 1 - x, mode='full')
            return corr_opt[max_k] * dx
        
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0, 1)] * n_points
        
        try:
            res = minimize(objective, h_to_optimize, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                optimized_h = res.x
                corr_opt = np.correlate(optimized_h, 1 - optimized_h, mode='full')
                opt_c5 = np.max(corr_opt) * dx
                if opt_c5 < best_c5:
                    best_c5 = opt_c5
                    best_h = optimized_h.copy()
                candidates[worst_idx] = optimized_h
        except:
            perturb_h = h_to_optimize + np.random.normal(0, 0.01, n_points)
            perturb_h = np.clip(perturb_h, 0, 1)
            sum_pert = np.sum(perturb_h)
            scale_pert = (n_points / 2) / sum_pert if sum_pert != 0 else 1.0
            perturb_h = np.clip(perturb_h * scale_pert, 0, 1)
            candidates[worst_idx] = perturb_h
        
        candidates = candidates[:len(candidates)//2] + [np.random.rand(n_points) for _ in range(len(candidates)//2)]
    
    return (best_h, best_c5, n_points)
