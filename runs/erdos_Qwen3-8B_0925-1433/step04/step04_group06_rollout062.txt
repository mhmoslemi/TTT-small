import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

        s = (n_points / 2) / np.sum(h)
        h = np.clip(h * s, 0, 1)
        return h

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population
    pop_size = 50
    pop = []
    h_init = scale_h(initial_h_values.copy())
    for _ in range(pop_size // 2):
        h = h_init + np.random.uniform(-0.05, 0.05, n_points)
        pop.append(scale_h(h))
        h = np.random.uniform(0, 1, n_points)
        pop.append(scale_h(h))
    pop = np.array(pop)

    best_h = h_init
    best_c5 = compute_c5(h_init)
    start_time = time.time()

    for gen in range(200):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = np.array([compute_c5(p) for p in pop])
        idx = np.argsort(fitness)
        pop = pop[idx[:int(pop_size * 0.3)]]
        pop = np.vstack([pop, np.random.uniform(0, 1, (pop_size - len(pop), n_points))])
        pop = np.array([scale_h(p) for p in pop])

        # Crossover
        for _ in range(pop_size // 2):
            p1, p2 = np.random.choice(len(pop), 2, replace=False)
            crossover = np.random.choice(n_points, 1)
            pop[np.random.choice(len(pop))][:crossover] = pop[p1][:crossover]
            pop[np.random.choice(len(pop))][crossover:] = pop[p2][crossover:]

        # Mutation
        for i in range(len(pop)):
            if np.random.rand() < 0.1:
                idxs = np.random.choice(n_points, 10, replace=False)
                pop[i][idxs] += np.random.uniform(-0.1, 0.1, 10)
                pop[i] = scale_h(pop[i])

        # Local refinement
        top_idx = np.argsort(fitness)[:5]
        for i in top_idx:
            h = pop[i].copy()
            corr = np.correlate(h, 1 - h, mode='full')
            max_k = np.argmax(corr)
            if max_k < n_points:
                h[:n_points - max_k] = 1 - h[:n_points - max_k]
