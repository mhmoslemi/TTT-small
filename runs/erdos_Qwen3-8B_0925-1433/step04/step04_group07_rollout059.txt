        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()
    learning_rate = 0.1
    for t in range(10000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_index = np.argmax(corr)
        k = max_corr_index - (n_points - 1)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            gradient[i] = (1 - h[j]) * dx
        h = h - learning_rate * gradient
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.ones(n_points) * (n_points / 2) / n_points
        else:
            scale = (n_points / 2) / sum_h
            h = h * scale
            h = np.clip(h, 0, 1)
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if t % 100 == 0:
            noise = np.random.normal(0, 0.05, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h == 0:
                h = np.ones(n_points) * (n_points / 2) / n_points
            else:
                scale = (n_points / 2) / sum_h
                h = h * scale
                h = np.clip(h, 0, 1)
    return (best_h, best_c5, n_points)
