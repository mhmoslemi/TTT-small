import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Generate even spaced binary initial setup
    m = n_points // 2
    if n_points % 2 == 1:
        m = (n_points + 1) // 2
    positions = [2 * i / m for i in range(m)]
    indices = np.round(np.array(positions) / dx).astype(int)
    indices = np.unique(indices)
    h = np.zeros(n_points)
    for idx in indices:
        if idx < n_points:
            h[idx] = 1
    sum_h = np.sum(h)
    while abs(sum_h - n_points / 2) > 1e-6:
        if sum_h < n_points / 2:
            idx = np.where(h == 0)[0][0]
            h[idx] = 1
        else:
            idx = np.where(h == 1)[0][0]
            h[idx] = 0
        sum_h = np.sum(h)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx, n_points)

    start_time = time.time()
    iterations = 0
    max_iterations = 1000

    while time.time() - start_time < budget_s and iterations < max_iterations:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if not ones.size or not zeros.size:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h, dx, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        iterations += 1

    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
