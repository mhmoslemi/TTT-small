import numpy as np

def run(seed=42, budget_s=600):
    import time
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def generate_random():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros(n_points)
        scale = (n_points / 2) / sum_h
        return np.clip(h * scale, 0, 1)

    def evaluate(h):
        if np.any(h < 0) or np.any(h > 1):
            return np.inf
        if np.abs(np.sum(h) - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode='full')
        return -np.max(corr * dx)

    population = []
    for _ in range(30):
        population.append(initial_h_values.copy())
    for _ in range(20):
        population.append(generate_random())
    population = np.array(population)

    best_h = initial_h_values.copy()
    best_c5 = evaluate(best_h)

    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([evaluate(h) for h in population])
        sorted_indices = np.argsort(fitness)
        sorted_pop = population[sorted_indices]

        new_population = []
        for _ in range(50):
            p1, p2 = np.random.choice(len(sorted_pop), 2, replace=False)
            child = sorted_pop[p1].copy()
            idxs = np.random.choice(n_points, 5, replace=False)
            for i in idxs:
                child[i] = sorted_pop[p2][i]
            idxs_mut = np.random.choice(n_points, 3, replace=False)
            for i in idxs_mut:
                child[i] += np.random.uniform(-0.1, 0.1)
                child[i] = np.clip(child[i], 0, 1)
            sum_h = np.sum(child)
            if sum_h == 0:
                child = np.zeros(n_points)
            else:
                scale = (n_points / 2) / sum_h
                child = np.clip(child * scale, 0, 1)
            new_population.append(child)
