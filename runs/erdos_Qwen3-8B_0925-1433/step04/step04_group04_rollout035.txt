import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def evaluate(h):
        if np.any(h < 0) or np.any(h > 1):
            return np.inf
        if np.abs(np.sum(h) - n_points/2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode='full')
        return -np.max(corr * dx)

    def mutate(h):
        h = h.copy()
        idxs = np.random.choice(n_points, size=3, replace=False)
        for i in idxs:
            h[i] += np.random.uniform(-0.1, 0.1)
            h[i] = np.clip(h[i], 0, 1)
        scale_factor = (n_points/2) / np.sum(h)
        h = np.clip(h * scale_factor, 0, 1)
        return h

    def crossover(parent1, parent2):
        offspring = parent1.copy()
        idxs = np.random.choice(n_points, size=5, replace=False)
        for idx in idxs:
            offspring[idx] = parent2[idx]
        return offspring

    population = []
    for _ in range(50):
        if _ < 30:
            h = initial_h_values.copy()
        else:
            h = np.random.uniform(0, 1, n_points)
            scale = (n_points/2) / np.sum(h)
            h = np.clip(h * scale, 0, 1)
        population.append(h)

    best_h = np.array(population[0])
    best_c5 = evaluate(best_h)

    start_time = time.time()
    for gen in range(300):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([evaluate(h) for h in population])
        sorted_idx = np.argsort(fitness)
        top_idx = sorted_idx[:int(len(population)*0.1)]
        elite = [population[i] for i in top_idx]

        new_population = elite[:]
        while len(new_population) < 50:
            p1, p2 = np.random.choice(top_idx, size=2, replace=False)
            child = crossover(population[p1], population[p2])
            child = mutate(child)
            new_population.append(child)
        population = new_population

        current_best = np.array(population[0])
        if evaluate(current_best) < best_c5:
            best_h = current_best
            best_c5 = evaluate(best_h)
            if best_c5 < 0.3807:
                break

    return (best_h, -best_c5, n_points)
