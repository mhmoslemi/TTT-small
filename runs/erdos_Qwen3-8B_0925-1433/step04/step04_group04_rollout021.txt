from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):

    global initial_h_values

    dx = 2.0 / n_points

    def scale_h(h):
        s = (n_points / 2) / np.sum(h)
        return np.clip(h * s, 0, 1)

    # Initialize population
    population_size = 50
    particles = []
    for _ in range(population_size):
        if initial_h_values is not None:
            h = initial_h_values.copy() + np.random.uniform(-0.1, 0.1, n_points)
        else:
            h = np.zeros(n_points)

            h[idx] = 1
        h = scale_h(h)
        particles.append(h)
    particles = np.array(particles)

    best_h = particles[0]
    best_c5 = compute_c5(best_h, dx, n_points)

    start_time = time.time()

    while time.time() - start_time < budget_s and iterations < 150:
        iterations += 1
        # Evaluate all particles
        c5_vals = np.zeros(population_size)
        for i in range(population_size):
            h = particles[i]
            corr = np.correlate(h, 1 - h, mode='full')
            c5_vals[i] = np.max(corr) * dx
        best_idx = np.argmin(c5_vals)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
