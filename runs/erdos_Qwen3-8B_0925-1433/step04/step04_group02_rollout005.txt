import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx

    h = initial_h_values.copy()
    sum_h = np.sum(h)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    best_h = h.copy()
    best_c5 = compute_c5(h)

    T = 1.0
    alpha = 0.995
    max_iter = 1000
    start_time = time.time()

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        perturb = np.random.normal(0, 0.1, n_points)
        h_perturbed = h + perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = np.sum(h_perturbed)
        if sum_perturbed != 0:
            scale = (n_points / 2) / sum_perturbed
            h_perturbed = h_perturbed * scale
        else:
            h_perturbed = np.ones(n_points) * 0.5
        new_c5 = compute_c5(h_perturbed)
        delta = new_c5 - best_c5
