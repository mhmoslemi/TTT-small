    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = scale_h(h, n_points)

    def compute_max_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_max_overlap(h)
    best_h = h.copy()

    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        original_value = corr[k_max] * dx
        delta = 0.01
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_perturb = h.copy()
            h_perturb[i] += delta
            h_perturb = scale_h(h_perturb, n_points)
            corr_perturb = np.correlate(h_perturb, 1 - h_perturb, mode='full')
            perturbed_value = corr_perturb[k_max] * dx
            grad[i] = (perturbed_value - original_value) / delta
        step_size = 0.05
        h_update = h - step_size * grad
        h_update = np.clip(h_update, 0, 1)
        sum_h = np.sum(h_update)
        if sum_h == 0:
            h_update = np.zeros_like(h_update)
        else:
            h_update = h_update * (n_points / 2) / sum_h
        h = h_update
        current_c5 = compute_max_overlap(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 < 0.3807:
            break
    return (best_h, best_c5, n_points)
