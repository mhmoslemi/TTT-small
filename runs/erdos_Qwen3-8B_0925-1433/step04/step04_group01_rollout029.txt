import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx, target_sum = 2.0 / n_points, n_points / 2
    h = initial_h_values * target_sum / np.sum(initial_h_values)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    
    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        def obj(x):
            new_h = x.copy()
            new_corr = np.correlate(new_h, 1 - new_h, mode='full')
            return new_corr[max_idx] * dx
        bounds = [(0, 1)] * n_points
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons)
        h = res.x
        curr_c5 = compute_c5(h, dx, n_points)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()
        noise = np.random.normal(0, 0.02, n_points)
        h = np.clip(h + noise, 0, 1)
        h = h * target_sum / np.sum(h)
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
