    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        perturb = np.random.normal(0, 0.05, n_points)
        new_h = initial_h_values * target_sum / np.sum(initial_h_values) + perturb
        new_h = np.clip(new_h, 0, 1)
        pop.append(new_h)
    
    best_h = pop[0]
    best_c5 = obj(best_h)
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
