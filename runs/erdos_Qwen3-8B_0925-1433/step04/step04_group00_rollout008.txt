import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        return np.max(corr) * dx
    
    def normalize(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
        else:
            h = np.ones(n_points) * 0.5
        return h
    
    h_best = normalize(initial_h_values.copy())
    best_c5 = compute_c5(h_best)
    
    particles = [normalize(initial_h_values.copy()) for _ in range(50)]
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(50)]
    pbest = particles.copy()
    pbest_c5 = [compute_c5(h) for h in pbest]
    gbest = h_best.copy()
    gbest_c5 = best_c5
    
    start = time.time()
    while time.time() - start < budget_s:
        for i in range(len(particles)):
            h = particles[i].copy()
            h = normalize(h)
            current_c5 = compute_c5(h)
            if current_c5 < pbest_c5[i]:
                pbest[i] = h.copy()
                pbest_c5[i] = current_c5
                if current_c5 < gbest_c5:
                    gbest = h.copy()
                    gbest_c5 = current_c5
                    if gbest_c5 < 0.3807:
                        return (gbest, gbest_c5, n_points)
        
        w, c1, c2 = 0.7, 1.5, 1.5
        for i in range(len(particles)):
            vel = velocities[i]
            r1, r2 = np.random.rand(), np.random.rand()
            vel = w * vel + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            velocities[i] = np.clip(vel, -0.5, 0.5)
            particles[i] += velocities[i]
            particles[i] = normalize(particles[i])
        
        for i in range(5):
            h = particles[i].copy()
            h = normalize(h)
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr)
            max_corr = corr[max_idx] * dx
            
            def obj(x):

                corr = np.correlate(x, 1 - x, mode='full')
                return corr[max_idx] * dx
            
            def cons(x):
                x = normalize(x)
                return np.sum(x) - n_points / 2
            
            cons_func = [{'type': 'eq', 'fun': cons}]
            res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n_points, constraints=cons_func, tol=1e-6)
