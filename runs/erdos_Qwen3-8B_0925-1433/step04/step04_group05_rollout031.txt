import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def scale_h(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / sum_h, 0, 1)

    h = scale_h(initial_h_values.copy())
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()

    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        current_c5 = corr[k_max] * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 < 0.3807:
            break

        grad = np.zeros(n_points)
        for i in range(n_points):
            h_perturb = h.copy()
            h_perturb[i] += 0.01
            h_perturb = scale_h(h_perturb)
            corr_perturb = np.correlate(h_perturb, 1 - h_perturb, mode='full')
            sensitivity = (corr_perturb[k_max] - corr[k_max]) / 0.01
            grad[i] = -sensitivity

        alpha = 0.01
        h = h + alpha * grad
        h = np.clip(h, 0, 1)
        h = scale_h(h)

    return (best_h, best_c5, n_points)
