import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def generate_sine():
        return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points), 0, 1)

    def generate_vdc():
        seq = np.zeros(n_points, dtype=int)
        for i in range(n_points):
            seq[i] = np.sum(np.floor(np.mod(i * np.arange(1, n_points+1), 2**n_points - 1) / (2**(n_points - i - 1))))
        return np.clip(seq / (2**n_points - 1), 0, 1)

    def compute_correlation(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def refine(h):
        h = h.copy()
        for _ in range(200):
            corr = np.correlate(h, 1 - h, mode='full')
            lags = np.argsort(corr)[::-1][:5]
            if not lags.size:
                break
            grad = np.zeros_like(h)
            for k in lags:
                overlap = np.zeros(n_points)
                for i in range(n_points):
                    j = (i + k) % n_points
                    overlap[i] = (1 - h[i]) * h[j]
                grad += 2 * np.sum(overlap)
            grad /= np.sum(np.abs(grad))
            h += -0.1 * grad
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h
        return h

    initial = initial_h_values.copy()
    initial = initial * (target_sum / np.sum(initial)) if np.sum(initial) != 0 else initial
    best_h = initial
    best_c5 = compute_correlation(initial)

    candidates = [initial, generate_sine(), generate_vdc()]
    for _ in range(10):
        candidates.append(np.random.choice([0, 1], size=n_points, p=[0.5, 0.5]))
        candidates.append(generate_sine() * np.random.uniform(0.8, 1.2))

    for cand in candidates:
        cand = cand * (target_sum / np.sum(cand)) if np.sum(cand) != 0 else cand
        c5 = compute_correlation(cand)
        if c5 < best_c5:
            best_h = cand
            best_c5 = c5

    start_time = time.time()
    for _ in range(100):
