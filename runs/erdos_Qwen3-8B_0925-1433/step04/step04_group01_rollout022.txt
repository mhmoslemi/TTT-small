import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    # Assume initial_h_values is available

    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    h = initial_h_values.copy() * target_sum / np.sum(initial_h_values)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    alpha = 0.05
    max_iterations = 500
    start_time = time.time()
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        worst_lag = np.argmax(corr)
        # Compute gradient using finite differences
        delta = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += delta
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            h_minus = h.copy()
            h_minus[i] -= delta
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            grad[i] = (corr_plus[worst_lag] - corr_minus[worst_lag]) / (2 * delta)
        # Update h
        h -= alpha * grad
        # Project back
        current_sum = np.sum(h)

            alpha *= 0.5
        else:
            alpha = max(alpha * 0.99, 0.001)
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
