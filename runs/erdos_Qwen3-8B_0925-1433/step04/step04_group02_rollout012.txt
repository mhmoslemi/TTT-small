import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def compute_c5(h):
        shifted = 1 - h
        corr = np.correlate(h, shifted, mode="full")
        return np.max(corr) * dx

    h = initial_h_values.copy()
    sum_h = np.sum(h)
    if sum_h != 0:
        scale = target_sum / sum_h
        h = np.clip(h * scale, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    T = 1.0
    cooling_rate = 0.995
    max_iterations = int(budget_s * 0.8)

    start_time = time.time()
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        h_perturb = h.copy()
        num_perturb = np.random.randint(1, 4)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for i in indices:
            delta = np.random.uniform(-0.1, 0.1)
            h_perturb[i] += delta
        h_perturb = np.clip(h_perturb, 0, 1)
        sum_perturb = np.sum(h_perturb)
        if sum_perturb == 0:
            h_perturb = np.ones(n_points) * 0.5
        else:
            scale = target_sum / sum_perturb
            h_perturb *= scale
            h_perturb = np.clip(h_perturb, 0, 1)
        candidate_c5 = compute_c5(h_perturb)
        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_perturb.copy()
            current_c5 = candidate_c5
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            if best_c5 < 0.3807:
                return (best_h, best_c5, n_points)
        T *= cooling_rate
    return (best_h, best_c5, n_points)
