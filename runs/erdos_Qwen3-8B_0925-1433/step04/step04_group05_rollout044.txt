import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def scale_h(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / sum_h, 0, 1)

    # Initial candidate
    h = initial_h_values.copy()
    h = scale_h(h)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Optimization loop
    start_time = time.time()
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Evaluate current best
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Perturb and optimize
        perturb = np.random.normal(0, 0.05, n_points)
        h_candidate = h + perturb
        h_candidate = scale_h(h_candidate)
        
        # Minimize objective with constraints
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        res = minimize(obj, h_candidate, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
        if res.success:
            h = res.x
        else:
            h = scale_h(h_candidate)
        
        # Early exit if target is reached
        if best_c5 < 0.3807:
            break
    
    return (best_h, best_c5, n_points)
