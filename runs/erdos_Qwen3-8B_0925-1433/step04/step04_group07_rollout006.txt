For example, if the maximum is at lag=0, then the integral is sum h(x)*(1 - h(x)) which is 1 - sum h(x)^2. This is a fixed value, so adjusting for lag=0 may not be helpful. Thus, the code should prioritize lags that are not zero.

To address this, perhaps after finding the max_index, we can skip the lag=0 case, or ensure that the adjustment is done only for lags that are not zero.

import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    def compute_c5(h, dx, n_points):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()
    
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        if lag == 0:
            continue
        
        h_shifted = np.roll(h, lag)
        products = h * h_shifted
        indices = np.argsort(-products)[:10]
        
        h_temp = h.copy()
        for i in indices:
            h_temp[i] -= 0.05 * h[i]
        
        sum_h = np.sum(h_temp)
        if sum_h != 0:
            factor = (n_points / 2) / sum_h
            h = np.clip(h_temp * factor, 0, 1)
        else:
            h = np.clip(h_temp, 0, 1)
        
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return (best_h, best_c5, n_points)
