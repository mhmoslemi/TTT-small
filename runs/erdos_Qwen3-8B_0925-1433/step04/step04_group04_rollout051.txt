import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    def generate_valid():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros(n_points)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
        return h
    def evaluate(h):
        if np.any(h < 0) or np.any(h > 1):
            return np.inf
        sum_h = np.sum(h)
        if abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode='full')
        return -np.max(corr) * dx
    def crossover(p1, p2):
        child = p1.copy()
        idxs = np.random.choice(n_points, 5, replace=False)
        for i in idxs:
            child[i] = p2[i]
        return child
    def mutate(h):
        h = h.copy()
        for i in np.random.choice(n_points, 3):
            h[i] += np.random.uniform(-0.1, 0.1)
            h[i] = np.clip(h[i], 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return h
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
        return h
    population = []
    for _ in range(20):
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            h = np.zeros(n_points)
        else:
            scale = (n_points / 2) / sum_h
            h = np.clip(h * scale, 0, 1)
        population.append(h)
    for _ in range(30):
        population.append(generate_valid())
