import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def scale_h(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / sum_h, 0, 1)

    h = scale_h(initial_h_values.copy())

    def obj(x):
        h_scaled = scale_h(x)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    for _ in range(30):
        if time.time() - start_time > budget_s:
            break
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(obj, h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6, options={'maxiter': 100})
        if res.success:
            h = res.x
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            if best_c5 < 0.3807:
                break
        else:
            h = scale_h(h + np.random.normal(0, 0.05, n_points))
    return (best_h, best_c5, n_points)
