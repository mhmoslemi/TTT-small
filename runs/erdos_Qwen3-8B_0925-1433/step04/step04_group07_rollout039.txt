import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    h = initial_h_values.copy()

    def compute_max_corr():
        corr = np.correlate(h, h, mode='full')
        return np.max(corr)

    max_corr = compute_max_corr()
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    for _ in range(2000):
        if time.time() > budget_s:
            break
        corr = np.correlate(h, h, mode='full')
        max_corr_val = np.max(corr)
        lag_idx = np.argmax(corr)
        k = lag_idx
        gradient = h[(np.arange(n_points) + k) % n_points]
        step_size = 0.05 * max_corr_val / np.sqrt(np.sum(gradient**2))
        h -= step_size * gradient
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        delta = (target_sum - current_sum) / n_points
        h += delta
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-8:
            continue
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    return (best_h, best_c5, n_points)
