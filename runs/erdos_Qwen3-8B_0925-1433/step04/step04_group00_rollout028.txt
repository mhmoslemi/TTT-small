            if current_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = current_c5
                if best_c5 < 0.3807:
                    return (best_h, best_c5, n_points)
            h = h_new.copy()
        else:
            # Fallback: Perturb and retry
            h += np.random.uniform(-0.05, 0.05, n_points)
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if sum_h != 0:
                scale = (n_points / 2) / sum_h
                h = np.clip(h * scale, 0, 1)
            else:
                h = np.ones(n_points) * 0.5

    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n_points)
