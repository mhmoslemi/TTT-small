import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.round(initial_h_values).astype(int)
    target_sum = n_points // 2
    current_sum = np.sum(h)
    delta = target_sum - current_sum
    if delta > 0:
        idx = np.where(h == 0)[0]
        h[idx[:delta]] = 1
    elif delta < 0:
        idx = np.where(h == 1)[0]
        h[idx[:delta]] = 0
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    
    for _ in range(100):
        overlaps = np.zeros(2*n_points - 1)
        for k in range(-n_points + 1, n_points):
            shifted = np.roll(h, k)
            overlaps[k + n_points - 1] = np.dot(h, 1 - shifted)
        max_k = np.argmax(overlaps)
        
        best_delta = -np.inf
        best_i = -1
        best_j = -1
        for i in np.where(h == 1)[0]:
            for j in np.where(h == 0)[0]:
                term_i = 1 - h[(i + max_k) % n_points]
                term_j = 1 - h[(j + max_k) % n_points]
                delta = -term_i + term_j
                if delta > best_delta:
                    best_delta = delta
                    best_i = i
                    best_j = j
        if best_delta <= 0:
            break
        h[best_i] = 0
        h[best_j] = 1
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
