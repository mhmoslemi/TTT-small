import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def generate_h():
        h = np.random.rand(n_points)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0, 1)
        return h

    population = [initial_h_values.copy() for _ in range(30)]
    population += [generate_h() for _ in range(20)]
    best_h = initial_h_values.copy()
    best_c5 = np.inf

    start_time = time.time()
    while time.time() - start_time < budget_s:
        fitness = []
        for h in population:
            if not (np.all(h >= 0) and np.all(h <= 1)):
                fitness.append(np.inf)
                continue
            s = np.sum(h)
            if np.abs(s - n_points / 2) > 1e-6:
                fitness.append(np.inf)
                continue
            corr = np.correlate(h, 1 - h, mode='full')
            fitness.append(-np.max(corr * dx))

        best_idx = np.argmin(fitness)
        current_c5 = -fitness[best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[best_idx].copy()

        if best_c5 < 0.3807:
            break

        top_indices = np.argsort(fitness)[:10]
        elites = [population[i] for i in top_indices]
        new_pop = elites[:2]

        while len(new_pop) < n_points:
            p1, p2 = np.random.choice(top_indices, 2, replace=False)
            child = population[p1].copy()
            idxs = np.random.choice(n_points, 5, replace=False)
            for i in idxs:
                child[i] = population[p2][i]
            child = np.clip(child, 0, 1)
            s = np.sum(child)
            scale = (n_points / 2) / s if s != 0 else 1.0
            child = np.clip(child * scale, 0, 1)
            new_pop.append(child)

        population = new_pop

    return (best_h, best_c5, n_points)
