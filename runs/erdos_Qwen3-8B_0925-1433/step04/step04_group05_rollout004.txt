            mut = c + noise
            new_candidates.append(scale_h(mut))
        # Optimize top candidates
        top_indices = np.argsort(curr_c5s)[:5]
        for idx in top_indices:
            h_candidate = candidates[idx].copy()
            result = minimize(obj, h_candidate, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-5)
            if result.success:
                candidates[idx] = result.x
            else:
                candidates[idx] = scale_h(h_candidate)
        # Update best
        curr_c5s = [compute_c5(c) for c in candidates]
        if np.min(curr_c5s) < best_c5:
            best_c5 = np.min(curr_c5s)
            best_h = candidates[np.argmin(curr_c5s)].copy()
        if best_c5 < 0.3807:
            break

    return (best_h, best_c5, n_points)
