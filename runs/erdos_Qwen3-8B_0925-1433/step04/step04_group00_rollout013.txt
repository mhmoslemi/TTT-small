        if res.success:
            h_new = res.x
            sum_h_new = np.sum(h_new)
            if sum_h_new != 0:
                scale = (n_points / 2) / sum_h_new
                h_new_scaled = np.clip(h_new * scale, 0, 1)
            else:
                h_new_scaled = np.ones(n_points) * 0.5
            corr_new = np.correlate(h_new_scaled, 1 - h_new_scaled, mode='full')
            current_c5 = np.max(corr_new) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_new_scaled.copy()
                if best_c5 < 0.3807:
                    return (best_h, best_c5, n_points)
        else:
            pass
        
        h = h_new_scaled
    
    return (best_h, best_c5, n_points)
