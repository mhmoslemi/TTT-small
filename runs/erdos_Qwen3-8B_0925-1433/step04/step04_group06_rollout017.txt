import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    global initial_h_values
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h = np.clip(h * scale, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    start_time = time.time()
    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full')
        top_indices = np.argsort(corr)[-5:]
        top_values = corr[top_indices]
        def obj(x):
            h_current = x
            corr_current = np.correlate(h_current, 1 - h_current, mode='full')
            return np.max(corr_current[top_indices]) * dx
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = res.x
        except:
            pass
        current_c5 = compute_c5(h, dx, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if best_c5 < 0.3807:
            break
        if np.random.rand() < 0.1:
            noise = np.random.normal(0, 0.1, n_points)
            h_perturbed = np.clip(h + noise, 0, 1)
            sum_perturbed = np.sum(h_perturbed)
            scale_perturbed = (n_points / 2) / sum_perturbed
            h_perturbed = np.clip(h_perturbed * scale_perturbed, 0, 1)
            h = h_perturbed
    return (best_h, best_c5, n_points)

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
