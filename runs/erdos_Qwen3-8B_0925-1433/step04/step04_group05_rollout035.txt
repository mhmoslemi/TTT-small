    def scale_h(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        return np.clip(h * (n_points / 2) / sum_h, 0, 1)

    candidates = []
    # Periodic candidates
    for freq in [1.0, 0.5, 2.0, 1.5, 0.7]:
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x)
        candidates.append(scale_h(h))
    # Symmetric candidates
    for s in [0.5, 1.0, 1.5, 2.0, 0.7]:
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.5 * np.cos(np.pi * s * x)
        candidates.append(scale_h(h))
    # Binary pattern
    binary = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            binary[i] = 1.0
    candidates.append(scale_h(binary))
    # Initial H
    candidates.append(initial_h_values)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def optimize(h):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        res = minimize(lambda x: compute_c5(x), h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
        return res.x if res.success else h

    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        current_c5 = [compute_c5(c) for c in candidates]
        elite_idx = np.argsort(current_c5)[:5]
        elites = [candidates[i] for i in elite_idx]
        optimized = [optimize(c) for c in elites]
        for i, c in enumerate(candidates):
            if i in elite_idx:
                candidates[i] = optimized.pop(0)
        new_c5 = [compute_c5(c) for c in candidates]
        best_idx = np.argmin(new_c5)
        if new_c5[best_idx] < best_c5:
            best_h = candidates[best_idx].copy()
            best_c5 = new_c5[best_idx]
        if best_c5 < 0.3807:
            break

    return (best_h, best_c5, n_points)
